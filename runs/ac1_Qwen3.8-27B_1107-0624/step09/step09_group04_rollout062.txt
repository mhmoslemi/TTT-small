"""
Optimized step function coefficient search using:
1. L-BFGS with log-sum-exp smoothing for global optimization
2. Active set LP for precise peak equalization (new)
3. Multi-n exploration with progressive refinement
4. KKT-based optimality verification
Target: beat 1.505661 (previous best)

import numpy as np
import time
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)
    def grad(self, z):
        return _smograd(z, self.beta, self.n)

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005) * beta_scale
    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625,
        0.03, 0.015, 0.0075, 0.004,
        0.002, 0.001, 0.0005, 0.0002, 0.0001,
        0.00005, 0.00002, 0.00001
    ])
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        if beta < base_beta * 0.01:
            est_time_per_iter = 0.004 * max(1.0, n / 2048.0)
            max_iter = min(6000, max(50, int(remaining / (est_time_per_iter * 15))))
        else:
            est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
            max_iter = min(4000, max(30, int(remaining / (est_time_per_iter * 12))))
        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-17,
                         'gtol': 1e-14, 'maxcor': 100, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a

def _active_set_lp(a, active_indices, max_step=0.01):
    """
    Solve LP to find direction d that reduces all active peaks.
    minimize -delta
    subject to: conv(d, a)[k] + delta <= 0 for k in active_indices
                sum(d) = 0
                -a[j] <= d[j] <= max_step
    """
    n = len(a)
    m = len(active_indices)
    if m == 0:
        return None
    
    # conv(d, a)[k] = sum_j d[j] * a[k-j] (for valid j)
    # Build constraint matrix A where A[k, j] = a[k-j] if valid
    # We want: A @ d + delta <= 0, i.e., -A @ d - delta >= 0
    # And: sum(d) = 0, -a <= d <= max_step
    # Minimize: -delta (i.e., maximize delta)
    
    # Variables: [d(0), ..., d(n-1), delta]
    # Objective: minimize [-1, ..., -1, -1] (minimize -delta, last variable)
    # Actually: c = [0, 0, ..., 0, -1] (just minimize -delta)
    
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = -1.0  # minimize -delta = maximize delta
    
    # Inequality constraints: A_ieq @ x <= b_ieq
    # For each k in active: conv(d,a)[k] + delta <= 0
    # => sum_j A[k,j]*d[j] + delta <= 0
    # => A[k,:] @ d + [0,...,0,1] @ x <= 0
    
    A_ieq_rows = []
    b_ieq_vals = []
    
    for k in active_indices:
        row = np.zeros(n_var)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                row[j] = a[idx]
        row[-1] = 1.0  # coefficient of delta
        A_ieq_rows.append(row)
        b_ieq_vals.append(0.0)
    
    # d[j] <= max_step => d[j] <= max_step
    # -d[j] <= a[j] => d[j] >= -a[j]
    
    for j in range(n):
        row = np.zeros(n_var)
        row[j] = 1.0
        A_ieq_rows.append(row)
        b_ieq_vals.append(max_step)
        
        row = np.zeros(n_var)
        row[j] = -1.0
        A_ieq_rows.append(row)
        b_ieq_vals.append(a[j])
    
    # delta <= max_step (limit the step)
    row = np.zeros(n_var)
    row[-1] = 1.0
    A_ieq_rows.append(row)
    b_ieq_vals.append(max_step)
    
    # delta >= -max_step (allow negative delta, i.e., no improvement)
    row = np.zeros(n_var)
    row[-1] = -1.0
    A_ieq_rows.append(row)
    b_ieq_vals.append(max_step)
    
    A_ieq = np.array(A_ieq_rows)
    b_ieq = np.array(b_ieq_vals)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: (lo, hi) for each variable
    # d[j]: (-a[j], max_step)
    # delta: (-max_step, max_step)
    bounds = [(-a[j], max_step) for j in range(n)]
    bounds.append((-max_step, max_step))
    
    try:
        result = linprog(c_obj, A_ieq=A_ieq, b_ieq=b_ieq,
                         A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if result.status == 0:
            d = result.x[:n]
            delta = result.x[n]
            if delta > 1e-10:
                return d, delta
    except Exception:
        pass
    return None

ΔC[k] from a[p] change = 2t*a[k-p] + t^2 (if 2p=k) or 2t*a[k-p] (if 2p≠k)

After changing a[q] to a[q]-t:
ΔC[k] from a[q] change = -2t*a[k-q] + t^2 (if 2q=k) or -2t*a[k-q] (if 2q≠k)

Total: ΔC[k] = 2t*(a[k-p] - a[k-q]) + O(t^2)

So the first-order change is 2t*(a[k-p] - a[k-q]). To decrease C[k], we need a[k-p] > a[k-q] (transfer from q to p) or a[k-p] < a[k-q] (transfer from p to q).

For the max to decrease, we need ΔC[k] < 0 for all k at the max. This means:
a[k-p] < a[k-q] for all k in S (if transferring from p to q, i.e., t < 0 means a[p]-t, a[q]+t)

This gives us a simple criterion: for a transfer from p to q (reducing a[p], increasing a[q]), the max decreases iff a[k-p] < a[k-q] for all active k.

This is equivalent to: the "shifted" versions of a (shifted by p and q) satisfy a[·-p] < a[·-q] at all active positions.

This is a very simple check! And it gives us a natural direction for improvement.

**Pairwise transfer algorithm:**
1. Find active set S
2. For each pair (p, q) in support(a):
   - Check if a[k-p] < a[k-q] for all k in S (transfer from p to q reduces all active peaks)
   - If yes, find the optimal t (how much to transfer) via line search
   - Apply the transfer
3. Repeat until no improving transfer is found

The challenge is that checking all pairs is O(n^2 * |S|) which is too slow for n=16384.

But we can be smarter: for a fixed p, the condition a[k-p] < a[k-q] for all k in S means that q should be in a region where a is "lower" than a shifted by p. We can find candidate q's efficiently.

"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with log-sum-exp smoothing, active set LP for precise peak equalization,
multi-n exploration, and iterative refinement cycles.
Key innovations over previous approach:
1. Active set LP (convex) for direction finding that reduces all active peaks simultaneously
2. KKT-based optimality checking to verify local optimality
3. Aggressive iterative cycle between smooth optimization and LP-based peak reduction
4. Progressive beta refinement with more iterations at small beta
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _conv_and_max(a):
    c = _full_convolve(a, a)
    return c, float(c.max())

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)
    def grad(self, z):
        return _smograd(z, self.beta, self.n)

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0, max_single_beta_iters=3000):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005) * beta_scale
    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625,
        0.03, 0.015, 0.0075, 0.004,
        0.002, 0.001, 0.0005, 0.0002, 0.0001,
        0.00005, 0.00002, 0.00001, 0.000005
    ])
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time_per_iter = 0.003 * max(1.0, n / 4096.0)
        if beta < base_beta * 0.01:
            max_iter = min(max_single_beta_iters, max(100, int(remaining / (est_time_per_iter * 20))))
        else:
            max_iter = min(max_single_beta_iters // 2, max(50, int(remaining / (est_time_per_iter * 10))))
        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-15, 'maxcor': 100, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a

def _active_set_lp_step(a, deadline, min_time_remaining=10.0, max_active=20, max_lp_iters=30):
    """
    Active set LP: find direction d that reduces all active peaks.
    Solves: maximize delta subject to conv(d,a)[k] + delta <= 0 for k in S,
            sum(d) = 0, -a[j]*0.1 <= d[j] <= 0.01
    Takes a step if improvement is found.
    """
    n = len(a)
    c, c_max = _conv_and_max(a)
    current_sc = 2.0 * n * c_max
    
    # Find active set: peaks within 0.05% of max
    threshold = c_max * (1.0 - 0.0005)
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    # Limit active set size
    if len(active) > max_active:
        active = active[np.argsort(c[active])[::-1][:max_active]]
    
    m = len(active)
    # For very large n, subsample the active set or use a simpler approach
    if n > 32768 and m > 10:
        active = active[:10]
        m = 10
    
    # Build sparse constraint matrix for LP
    # Variables: d[0..n-1], delta (total n+1 vars)
    # Constraints:
    #   conv(d,a)[k_i] + delta <= 0 for each active k_i  (m constraints)
    #   d[j] <= upper for each j  (n constraints)  
    #   -d[j] <= a[j]*frac for each j  (n constraints)
    #   delta <= max_delta (1 constraint)
    #   -delta <= max_delta (1 constraint)
    # Equality: sum(d) = 0
    
    max_delta = 0.005
    lower_frac = 0.05
    upper_val = 0.005
    
    # Build sparse A for inequality constraints
    rows = []
    cols = []
    vals = []
    
    # Active set constraints: conv(d,a)[k] + delta <= 0
    # conv(d,a)[k] = sum_j d[j] * a[k-j]
    for k in active:
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                rows.append(len(rows) % 10**9)  # placeholder, will build properly
                break
        # Actually, let's build row by row
        pass
    
    # For efficiency with large n, use a banded structure
    # Rebuild properly:
    rows = []
    cols = []
    vals = []
    b_ieq = []
    
    # Active set constraints
    for k in active:
        row_idx = len(b_ieq)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                rows.append(row_idx)
                cols.append(j)
                vals.append(a[idx])
        # delta coefficient
        rows.append(row_idx)
        cols.append(n)  # delta is variable index n
        vals.append(1.0)
        b_ieq.append(0.0)
    
    # Upper bounds on d: d[j] <= upper_val
    for j in range(n):
        row_idx = len(b_ieq)
        rows.append(row_idx)
        cols.append(j)
        vals.append(1.0)
        b_ieq.append(upper_val)
    
    # Lower bounds on d: -d[j] <= a[j] * lower_frac
    for j in range(n):
        row_idx = len(b_ieq)
        rows.append(row_idx)
        cols.append(j)
        vals.append(-1.0)
        b_ieq.append(a[j] * lower_frac)
    
    # Delta bounds
    row_idx = len(b_ieq)
    rows.append(row_idx)
    cols.append(n)
    vals.append(1.0)
    b_ieq.append(max_delta)
    
    row_idx = len(b_ieq)
    rows.append(row_idx)
    cols.append(n)
    vals.append(-1.0)
    b_ieq.append(max_delta)
    
    n_rows = len(b_ieq)
    n_cols = n + 1
    
    A_ieq = sp.csr_matrix((vals, (rows, cols)), shape=(n_rows, n_cols))
    b_ieq = np.array(b_ieq)
    
    # Equality: sum(d) = 0
    A_eq = sp.csr_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))), shape=(1, n_cols))
    b_eq = np.array([0.0])
    
    # Objective: minimize -delta (maximize delta)
    c_obj = np.zeros(n_cols)
    c_obj[n] = -1.0
    
    # Bounds
    bounds = [(None, upper_val) for _ in range(n)]
    bounds.append((-max_delta, max_delta))
    
    try:
        result = linprog(c_obj, A_ieq=A_ieq, b_ieq=b_ieq,
                         A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if result.status == 0:
            d = result.x[:n]
            delta = result.x[n]
            if delta > 1e-8:
                # Take the step with line search
                a_new = a + d
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new / s_new
                    sc_new = _score(a_new)
                    if sc_new < current_sc - 1e-14:
                        return a_new, sc_new
    except Exception:
        pass
    
    return a, current_sc

def _gradient_peak_reduction(a, deadline, min_time_remaining=10.0, max_iters=500):
    """
    Gradient-based peak reduction with adaptive step size.
    Uses the analytical gradient structure for efficiency.
    """
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    lr = 0.01
    no_improve = 0
    K = 5
    
    for it in range(max_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        c, c_max = _conv_and_max(a)
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            K = min(K + 1, 30)
        
        # Top K peaks
        if len(c) > K:
            top_k_idx = np.argsort(c)[::-1][:K]
        else:
            top_k_idx = np.arange(len(c))
        
        # Average subgradient of top K peaks
        w_S = np.zeros(2 * n - 1)
        w_S[top_k_idx] = 1.0 / len(top_k_idx)
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h - 2.0 * np.mean(h)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        alpha = lr / gnorm
        improved = False
        for ls in range(40):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.4
        
        if not improved:
            no_improve += 1
            lr *= 0.8
            if no_improve % 30 == 0:
                K = max(3, K // 2)
            if no_improve >= 300:
                a = best_a.copy()
                lr = 0.01
                K = 5
                no_improve = 0
                if no_improve > 600:
                    break
        else:
            lr = min(0.03, lr * 1.02)
    
    return best_a, best_sc

def _kkt_check(a, tol=0.01):
    """
    Check KKT conditions for the active set problem.
    Returns (is_optimal, violation).
    """
    n = len(a)
    c, c_max = _conv_and_max(a)
    
    # Active set
    threshold = c_max * (1.0 - tol * 0.01)
    S = np.where(c >= threshold)[0]
    
    if len(S) == 0:
        return True, 0.0
    
    # KKT: there exist lambda_k >= 0, sum lambda = 1, and mu such that
    # for all j with a[j] > 0: sum_k lambda_k * 2*a[k-j] = mu
    # (approximately, for the active set)
    
    support = np.where(a > 1e-10)[0]
    if len(support) == 0:
        return True, 0.0
    
    # Uniform lambda as initial guess
    m = len(S)
    # Compute the "KKT residual": for uniform lambda, check if 
    # sum_k a[k-j] is constant over j in support
    vals = np.zeros(len(support))
    for idx, j in enumerate(support):
        vals[idx] = 0.0
        for k in S:
            idx2 = k - j
            if 0 <= idx2 < n:
                vals[idx] += a[idx2]
        vals[idx] /= m
    
    # The KKT violation is how non-constant vals is
    violation = vals.max() - vals.min()
    relative_violation = violation / (vals.mean() + 1e-15)
    
    return relative_violation < 0.05, relative_violation

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.7) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal_asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.3) ** 2) + 0.7 * np.exp(-12.0 * (t - 0.7) ** 2) + 0.05
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 7.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.arange(n)
        a = 1.0 + 0.5 * t / n
        return a / a.sum()
    elif kind == "quadratic_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * (t - 0.3) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _asymmetric_perturb(a, rng, strength=0.03):
    n = len(a)
    t = np.arange(n) / n
    ramp = 0.4 * (t - 0.5)
    noise = 0.6 * rng.standard_normal(n)
    perturbation = ramp + noise
    perturbation = perturbation - perturbation.mean()
    z = np.log(np.maximum(a, 1e-30))
    z = z + strength * perturbation
    z = z - z.mean()
    return _softmax(z)

        break

    # === Phase 4: Multi-n exploration ===
    print(f"\n[P4] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([len(best_seq), 16384, 32768, 12288, 24576]), reverse=True)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")
        
        # L-BFGS on new n
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=60.0, beta_scale=0.5)
        if _try_update(a_opt, f"multi_n_{n_try}"):
            # Follow up with gradient reduction
            a_grad, _ = _gradient_peak_reduction(a_opt, deadline, min_time_remaining=40.0, max_iters=100)
            _try_update(a_grad, f"multi_n_{n_try}_grad")

    # === Phase 5: Final fine L-BFGS ===
    print(f"\n[P5] Final fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 30 - time.monotonic()
    if remaining > 20:
        a_final = _run_lbfgs(best_seq, deadline, min_time_remaining=25.0, beta_scale=0.2)
        _try_update(a_final, "fine_final")
        
        # Gradient polish
        if time.monotonic() < deadline - 15:
            a_polish, _ = _gradient_peak_reduction(best_seq, deadline, min_time_remaining=12.0, max_iters=200)
            _try_update(a_polish, "grad_polish")
        
        # Final LP attempt
        if time.monotonic() < deadline - 5:
            a_lp, _ = _active_set_lp_step(best_seq, deadline, min_time_remaining=5.0)
            _try_update(a_lp, "lp_final")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
