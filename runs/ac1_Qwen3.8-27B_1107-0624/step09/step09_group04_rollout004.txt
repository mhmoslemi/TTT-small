def active_set_solve(S, J, w, a):
    """Given active set S of convolution indices, support J of a, and weights w,
    solve for a on J and lambda."""
    n = len(a)
    # Build the linear system: for j in J, sum_{k in S} w_k * (a[j+k] + a[j-k]) = lambda
    # Unknowns: a[j] for j in J, and lambda
    m = len(J)
    A = np.zeros((m, m + 1))
    J_arr = np.array(J)
    
    for i, j in enumerate(J_arr):
        for k_idx, k in enumerate(S):
            wk = w[k_idx]
            # a[j+k]
            jk = j + k
            if 0 <= jk < n:
                if jk in J:
                    A[i, list(J).index(jk)] += wk
            # a[j-k]
            jk2 = j - k
            if 0 <= jk2 < n:
                if jk2 in J:
                    A[i, list(J).index(jk2)] += wk
        A[i, m] = -1  # coefficient for lambda
    
    # Solve A * [a_J; lambda] = 0 (homogeneous, up to scale)
    # Use SVD to find the null space
    U, sigma, Vt = np.linalg.svd(A)
    # The last row of Vt gives the null space vector
    x = Vt[-1]
    a_solution = x[:m]
    lambda_val = x[m]
    
    # Normalize so sum(a_J) = 1
    if a_solution.sum() < 1e-15:
        return None
    a_solution = a_solution / a_solution.sum()
    if (a_solution < -1e-10).any():
        return None
    
    a_new = np.zeros(n)
    a_new[J_arr] = np.maximum(a_solution, 0)
    a_new = a_new / a_new.sum()
    
    return a_new, lambda_val

The Frank-Wolfe algorithm for minimizing f(a) over the simplex:
1. Compute the subgradient g of f at a
2. Find s = argmin_{x in simplex} g^T x = e_{argmin g} (a vertex of the simplex)
3. Update: a_new = (1-t) * a + t * s, where t is chosen by line search

This is attractive because:
- No projection needed (the update automatically stays in the simplex)
- Simple to implement
- Convergence is O(1/T) which is slow but reliable
- Can be accelerated with strong Wolfe line search

Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses a multi-phase approach: L-BFGS warmup, Frank-Wolfe, active-set equalization,
multi-n exploration, and final polish. Key improvements over previous version:
- Frank-Wolfe steps for better simplex navigation
- Adaptive step sizes with Polyak acceleration
- Dynamic active-set management
- Periodic restarts to escape local minima

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _proj_simplex(v):
    """Euclidean projection onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(v - theta, 0)

def _subgrad_at_peak(a, k):
    """Compute the subgradient of (a*a)[k] with respect to a.
    d/d(a[j]) (a*a)[k] = 2*a[k-j] (with a[m]=0 for invalid m)"""
    n = len(a)
    g = np.zeros(n)
    # a[k-j] is valid when 0 <= k-j < n, i.e., k-(n-1) <= j <= k
    j_start = max(0, k - (n - 1))
    j_end = min(n - 1, k)
    if j_start <= j_end:
        idx = k - np.arange(j_start, j_end + 1)
        g[j_start:j_end+1] = 2.0 * a[idx]
    return g

def _compute_active_subgrad(a, active_set):
    """Compute the average subgradient over the active set."""
    n = len(a)
    g = np.zeros(n)
    for k in active_set:
        g += _subgrad_at_peak(a, k)
    return g / len(active_set)

def _lbfgs_smooth(a_init, deadline, min_time_remaining, beta_scale=1.0):
    """L-BFGS on softmax parameterization with log-sum-exp smoothing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(min(30, len(c_sorted)-1), len(c_sorted)-1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005) * beta_scale
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(3000, max(30, int(remaining / (est_time * 12))))
        
        # Define objective and gradient
        def obj(z_val):
            a_val = _softmax(z_val)
            c_val = _full_convolve(a_val, a_val)
            c_max_val = float(c_val.max())
            shifted = (c_val - c_max_val) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                return 2.0 * n * c_max_val
            return 2.0 * n * (c_max_val + beta * np.log(s))
        
        def grad(z_val):
            a_val = _softmax(z_val)
            c_val = _full_convolve(a_val, a_val)
            c_max_val = float(c_val.max())
            shifted = (c_val - c_max_val) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c_val)
                w[int(np.argmax(c_val))] = 1.0
            else:
                w = e / s
            a_rev = a_val[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n-1:2*n-1]
            W2 = float(np.dot(w, c_val))
            return 4.0 * n * a_val * (h - W2)
        
        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-17, 'gtol': 1e-14, 'maxcor': 50, 'maxls': 30}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

def _frank_wolfe(a_init, deadline, min_time_remaining=30.0):

    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * float(_full_convolve(a, a).max())

        it += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())

        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()

        threshold = c_max * (1 - 0.0005)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        # Subgradient
        g = _compute_active_subgrad(a, active)
        
        # Frank-Wolfe: s = e_{argmin g}
        s_idx = int(np.argmin(g))
        
        # Line search
        best_t = 0.0
        best_f = current_score

                best_t = t_c

            a[s_idx] += best_t
        else:

            active2 = np.where(c >= threshold)[0]
            if len(active2) > len(active):
                g2 = _compute_active_subgrad(a, active2)
                s_idx2 = int(np.argmin(g2))

                    a_try[s_idx2] += 0.01

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
