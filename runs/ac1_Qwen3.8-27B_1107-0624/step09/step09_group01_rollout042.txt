    return e / s

def _A_matvec(v, S, n, K, F_mask=None):
    """Compute A*v where A[i,k] = (1/K)*1_{i+k in S, i in F, k in F}."""
    N = _next_pow2(3 * n)
    v_pad = np.zeros(N)
    v_pad[:n] = v
    if F_mask is not None:
        v_pad[~F_mask[:n]] = 0.0
    w = np.zeros(N)
    for j in S:
        if 0 <= j < N:
            w[j] = 1.0
    F_v = np.fft.rfft(v_pad)
    F_w = np.fft.rfft(w)
    cross = np.fft.irfft(F_v * np.conj(F_w), N)
    result = cross[:n] / K
    if F_mask is not None:
        result[~F_mask[:n]] = 0.0
    return result

def _cg_solve_kkt(S, F_mask, n, K, rhs, max_iter=500, tol=1e-14):
    """Solve A_FF * x = rhs using CG with FFT matvec."""
    x = np.zeros(n)
    # Initial guess: uniform on F
    x[F_mask] = 1.0 / F_mask.sum()
    r = rhs.copy() - _A_matvec(x, S, n, K, F_mask)
    p = r.copy()
    # Mask p to F
    p[~F_mask] = 0.0
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Ap = _A_matvec(p, S, n, K, F_mask)
        pHp = float(p @ Ap)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        if pHp <= 0:
            # Try a different direction
            p = r.copy()
            p[~F_mask] = 0.0
            rs = float(r @ r)
            if rs < 1e-30:
                break
            continue
        alpha = rs / pHp
        x_new = x + alpha * p
        x_new[~F_mask] = 0.0
        r_new = r - alpha * Ap
        r_new[~F_mask] = 0.0
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            alpha *= 0.5
            x_new = x + alpha * p
            x_new[~F_mask] = 0.0
            r_new = r - alpha * Ap
            r_new[~F_mask] = 0.0
            rs_new = float(r_new @ r_new)
            if not np.isfinite(rs_new):
                break
        x = x_new
        r = r_new
        if rs < tol * tol * rs0:
            break
        p_new = (rs_new / rs) * p + r
        p_new[~F_mask] = 0.0
        # Occasional restart if stall
        if i % 50 == 49:
            p = r.copy()
            p[~F_mask] = 0.0
            rs = float(r @ r)
            if rs < 1e-30:
                break
        else:
            p = p_new
            rs = rs_new
    return x

def _kkt_solve_direct(a_init, max_iter=200, max_K=3000, deadline=None):
    """
    Solve the KKT system using growing active set.
    Returns the optimal a for the given n.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    
    F_mask = np.ones(n, dtype=bool)  # Free set (all indices initially)
    
    for outer in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Determine active set: positions where (a*a)_j is close to max
        # Use a tolerance that grows slightly
        tol = max(1e-10, 1e-6 * c_max)
        S = np.where(c >= c_max - tol)[0].tolist()
        
        # Ensure at least the global max is in S
        if len(S) == 0:
            S = [int(np.argmax(c))]
        
        K = len(S)
        K = min(K, max_K)
        # Take top K if too many
        if len(S) > K:
            S_arr = np.array(S)
            S = S_arr[np.argsort(c[S_arr])[-K:]].tolist()
            K = len(S)
        
        # Solve KKT: A_FF * a_F = lambda * 1, sum(a_F) = 1
        # Which gives: a_F = (A_FF^{-1} * 1_F) / (1_F^T A_FF^{-1} 1_F)
        rhs = np.zeros(n)
        rhs[F_mask] = 1.0
        
        x = _cg_solve_kkt(S, F_mask, n, K, rhs, max_iter=300, tol=1e-14)
        
        # Normalize
        x_F_sum = float(x[F_mask].sum())
        if x_F_sum < 1e-15 or not np.isfinite(x_F_sum):
            # Fallback: reduce and retry
            break
        
        a_new = np.zeros(n)
        a_new[F_mask] = x[F_mask] / x_F_sum
        
        # Check non-negativity: if any x[F] < 0, move to zero set
        neg_idx = F_mask & (x < -1e-12)
        if neg_idx.any():
            F_mask[neg_idx] = False
            a_new[neg_idx] = 0
            # Renormalize
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            continue  # Re-solve with updated F
        
        # Check KKT for zero set: (A*a)_i >= lambda for i in Z
        # lambda = 1 / (1_F^T * x) * (1/K) ... actually lambda = 1/x_F_sum / K * ... 
        # Let's compute (A*a_new) and check
        A_a = _A_matvec(a_new, S, n, K)
        # lambda = (A*a)_i for i in F (should be constant)
        lambda_val = float(A_a[F_mask].mean()) if F_mask.sum() > 0 else 0
        Z_mask = ~F_mask
        if Z_mask.any():
            z_viol = Z_mask & (A_a < lambda_val - 1e-8)
            if z_viol.any():
                # Move some back to F (the worst violators)
                z_idx = np.where(z_viol)[0]
                # Move the worst 5%
                worst = min(len(z_idx), max(1, len(z_idx) // 20))
                z_A_vals = A_a[z_idx]
                move_idx = z_idx[np.argsort(z_A_vals)[:worst]]
                F_mask[move_idx] = True
                continue  # Re-solve
        
        # Check convolution feasibility: (a*a)_j <= c_max_new for j not in S
        c_new = _full_convolve(a_new, a_new)
        c_max_new = float(c_new.max())
        
        # Check if active set needs updating
        viol = (c_new > c_max_new - 1e-12) & (np.arange(2*n-1)[:, None] != np.array(S)[None, :]).all(axis=1)
        # Simpler: check if any non-S position exceeds c_max_new
        S_set = set(S)
        viol_positions = []
        for j in range(2*n-1):
            if j not in S_set and c_new[j] > float(np.max(c_new[S_set])) + 1e-10:
                viol_positions.append(j)
        
        if viol_positions:
            # Add to S and continue
            S.extend(viol_positions[:10])  # Add up to 10 new
            K = min(len(S), max_K)
            continue
        
        # Check if solution improved
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
        
        # Check convergence
        a_diff = float(np.linalg.norm(a_new - a))
        a = a_new
        if a_diff < 1e-12:
            break
    
    # Final evaluation
    a = np.maximum(best_a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    sc = _score(a)
    return a, sc

def _kkt_solve_smoothed(a_init, deadline, max_steps=3000, beta_start=0.01):
    """
    Optimize using log-sum-exp smoothing of the max, then refine with KKT.
    Uses L-BFGS in log-space.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    beta = beta_start
    
    def _obj_grad(z_vec, beta_val):
        a = _softmax(z_vec)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / beta_val
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n * c_max, np.zeros(n)
        obj = 2.0 * n * (c_max + beta_val * np.log(s))
        w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        return obj, grad
    
    while beta > 1e-7:
        if time.monotonic() > deadline:
            break
        
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        
        max_iter = min(3000, max(20, int(remaining / 30.0)))
        
        def _obj_fn(z_vec):
            val, _ = _obj_grad(z_vec, beta)
            return val
        
        def _grad_fn(z_vec):
            _, g = _obj_grad(z_vec, beta)
            return g
        
        try:
            result = minimize(
                _obj_fn, z, jac=_grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 80, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
        
        # Adapt beta based on flatness
        c = _full_convolve(current_a, current_a)
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(100, len(c_sorted)-1)]
        if gap < 1e-6 * c_sorted[0]:
            beta *= 0.05
        elif gap < 1e-4 * c_sorted[0]:
            beta *= 0.1
        else:
            beta *= 0.2
        beta = min(beta, 0.0005 * c_sorted[0])
    
    return current_a, current_sc

def _subgrad_refine(a_init, deadline, max_steps=5000):
    """Subgradient method with momentum for refinement."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive tolerance for active set
        for tol_frac in [1e-5, 5e-5, 1e-4, 5e-4, 1e-3, 5e-3, 0.01]:
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            if active.sum() >= 5:
                break
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        W = np.zeros_like(c)
        W[active] = 1.0 / active.sum()
        
        a_rev = a[::-1]
        wc = _full_convolve(W, a_rev)
        h = wc[n-1:2*n-1]
        W_val = float(np.dot(W, c))
        grad = 2.0 * a * (h - W_val)
        grad = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        v = momentum * v - lr * grad
        z = z + v
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        
        if i % 100 == 99:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, 0.03)
            else:
                patience += 1
                lr *= 0.85
                momentum = min(momentum + 0.005, 0.95)
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)
                    patience = 0
                    lr = 0.008
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        # Taper edges
        taper = max(3, n // 50)
        for i in range(taper):
            frac = (taper - i) / taper
            a[i] *= (1 - frac * 0.8)
            a[-1-i] *= (1 - frac * 0.8)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)
