```python
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj_and_grad(z, beta, n):
    """Compute smooth objective and its gradient."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        obj = 2.0 * n * c_max
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    
    # Gradient
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return obj, grad
```

```python
def _adam_optimize(z, beta_schedule, n, deadline, min_time=30.0):
    """Optimize using Adam with a schedule of temperature values."""
    best_sc = _score(_softmax(z))
    best_z = z.copy()
    
    m = np.zeros_like(z)  # First moment
    v = np.zeros_like(z)  # Second moment
    t = 0
    lr_base = 0.01
    
    for beta in beta_schedule:
        if time.monotonic() + min_time > deadline:
            break
        
        # Adam parameters
        lr = lr_base / (1 + 0.1 * t)
        b1, b2 = 0.9, 0.999
        eps = 1e-8
        
        max_iter = 2000
        iters_done = 0
        prev_sc = best_sc
        
        for it in range(max_iter):
            if time.monotonic() + min_time > deadline:
                break
            t += 1
            iters_done += 1
            
            obj, grad = _smoobj_and_grad(z, beta, n)
            
            # Gradient clipping
            gnorm = np.linalg.norm(grad)
            if gnorm > 10.0:
                grad = grad * (10.0 / gnorm)
            
            # Adam update
            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * grad ** 2
            m_hat = m / (1 - b1 ** t)
            v_hat = v / (1 - b2 ** t)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Check score every 50 iterations
            if (it + 1) % 50 == 0:
                sc = _score(_softmax(z))
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_z = z.copy()
                elif sc > prev_sc * 1.01:
                    # Diverging, reduce learning rate
                    lr *= 0.5
                    z = best_z.copy()
                prev_sc = sc
        
        # After each temperature, check
        sc = _score(_softmax(z))
        if sc < best_sc:
            best_sc = sc
            best_z = z.copy()
            print(f"  beta={beta:.2e}: score={best_sc:.10f} (iters={iters_done})")
    
    return best_z, best_sc
```

```python
def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _smoobj_grad(z, beta, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta

    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return obj, grad


class _LbfgsFn:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        obj, _ = _smoobj_grad(z, self.beta, self.n)
        return obj

    def grad(self, z):
        _, g = _smoobj_grad(z, self.beta, self.n)
        return g


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0, max_iters=None):

def _adam_optimize(a_init, deadline, min_time_remaining=30.0):
    """Adam optimizer on log-space with cosine-annealed temperature."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    base_beta = max_c * 0.02

    best_sc = _score(a)
    best_a = a.copy()

    # Cosine-anneal temperature from base_beta to base_beta * 1e-5
    total_budget = max(10.0, deadline - min_time_remaining - time.monotonic())
    n_phases = 8
    phase_time = total_budget / n_phases

    lr = 0.005
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    t_adam = 0
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    stall_count = 0
    prev_best = best_sc

    for phase in range(n_phases):
        if time.monotonic() + min_time_remaining > deadline:
            break

        # Cosine anneal
        progress = phase / max(1, n_phases - 1)
        beta = base_beta * (1e-5 + (1.0 - 1e-5) * 0.5 * (1 + np.cos(np.pi * progress)))
        phase_deadline = time.monotonic() + phase_time

        # Decay learning rate
        lr = 0.005 * (0.85 ** phase)

        while True:
            remaining = min(time.monotonic() + phase_time > phase_deadline and 0 or (phase_deadline - time.monotonic()),
                            deadline - min_time_remaining - time.monotonic())
            if remaining < 2:
                break
            if time.monotonic() > phase_deadline:
                break

            t_adam += 1
            obj, grad = _smoobj_grad(z, beta, n)

            # Gradient clipping
            gnorm = np.linalg.norm(grad)
            if gnorm > 5.0:
                grad = grad * (5.0 / gnorm)

            # Adam update
            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * grad ** 2
            m_hat = m / (1 - b1 ** t_adam)
            v_hat = v / (1 - b2 ** t_adam)
            z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Check score periodically
            if t_adam % 100 == 0:
                a_try = _softmax(z_new)
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    z = z_new
                    stall_count = 0
                else:
                    z = z_new
                    stall_count += 1
                    if stall_count > 20:
                        # Revert and reduce lr
                        z = np.log(np.maximum(best_a, 1e-30))
                        z = z - z.mean()
                        lr *= 0.5
                        m = np.zeros_like(z)
                        v = np.zeros_like(z)
                        stall_count = 0
                        if lr < 1e-6:
                            break

            else:
                z = z_new

    # Final check
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    print(f"  Adam: best_sc={best_sc:.10f}")
    return best_a, best_sc


def _active_set_equalize(a_init, deadline, min_time_remaining=30.0):
    """Direct active-set peak equalization with exact-ish line search."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    K = 5
    lr = 0.01
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            if K < min(60, n):
                K += 1
        elif no_improve > 100:
            a = best_a.copy()
            K = max(3, K // 2)
            lr = 0.01
            no_improve = 0
            if K <= 3:
                break

        # Get top K peaks
        top_k_idx = np.argsort(c)[::-1][:K]

        # Compute average gradient
        w_S = np.zeros(2 * n - 1)
        w_S[top_k_idx] = 1.0 / K
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Backtracking line search
        alpha = lr / (gnorm + 1e-15)
        improved = False
        for ls_iter in range(40):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if improved:
            lr = min(0.03, lr * 1.02)
        else:
            no_improve += 1
            lr *= 0.7
            if no_improve >= 300:
                break

    return best_a, best_sc


def _nesterov_polish(a_init, deadline, min_time_remaining=20.0):
    """Nesterov-accelerated subgradient method for final polish."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    y = a.copy()  # Nesterov momentum variable
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    momentum = 0.9
    base_lr = 0.005
    it_count = 0

    while time.monotonic() + min_time_remaining < deadline:
        it_count += 1
        lr = base_lr / np.sqrt(1 + 0.01 * it_count)

        # Nesterov lookahead
        y_new = a + momentum * (a - y)
        c = _full_convolve(y_new, y_new)
        c_max = float(c.max())

        # Active set: peaks within 0.1% of max
        threshold = c_max * (1 - 0.001)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        if len(active) > 100:
            active = np.argsort(c)[::-1][:100]

        # Subgradient
        w_S = np.zeros(2 * n - 1)
        w_S[active] = 1.0 / len(active)
        y_new_rev = y_new[::-1]
        h = _full_convolve(w_S, y_new_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Update
        a_new = a - lr * grad / (gnorm + 1e-10)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        # Check improvement
        if it_count % 20 == 0:
            sc_new = 2.0 * n * float(_full_convolve(a_new, a_new).max())
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()

        y = a
        a = a_new

    return best_a, best_sc


def _structured_perturb(a, rng, mode=0):
    """Apply a structured perturbation to break symmetry."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    if mode == 0:
        # Linear ramp (asymmetry)
        t = np.arange(n) / n
        perturbation = np.sin(2 * np.pi * (t - 0.5)) * 0.5 + (t - 0.5) * 0.3
        strength = 0.03
    elif mode == 1:
        # High-frequency noise
        perturbation = rng.standard_normal(n)
        perturbation = perturbation - perturbation.mean()
        strength = 0.02
    elif mode == 2:
        # Low-frequency modulation
        t = np.arange(n) / n
        perturbation = np.sin(2 * np.pi * 3 * t) * 0.4 + np.cos(2 * np.pi * 7 * t) * 0.3
        strength = 0.04
    elif mode == 3:
        # Edge boost (concentrate mass at edges)
        t = np.arange(n) / n
        perturbation = -np.sin(np.pi * t) * 0.5  # negative in middle, positive at edges
        strength = 0.05
    else:
        perturbation = rng.standard_normal(n)
        perturbation = perturbation - perturbation.mean()
        strength = 0.02

    z = z + strength * perturbation
    z = z - z.mean()
    return _softmax(z)


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edge_concentrated":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * np.sin(np.pi * t) ** 2) + 0.1
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.arange(n)
        a = 1.0 + 0.4 * np.sin(2.0 * np.pi * 5.0 * t / n) + 0.2 * np.cos(2.0 * np.pi * 11.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

    # === Phase 1: Quick L-BFGS warmup ===

    a_p1 = _run_lbfgs(parent, deadline, min_time_remaining=1000.0, beta_scale=1.0, max_iters=800)
    _update_best(a_p1, "lbfgs_warmup")

    # === Phase 2: Adam optimization with diverse restarts ===
    print(f"\n[P2] Adam optimization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        for trial in range(6):
            if time.monotonic() > deadline - 500:
                break
            if trial == 0:
                a_init = best_seq.copy()
            else:
                a_init = _structured_perturb(best_seq, rng, mode=trial % 4)
            a_opt, sc_opt = _adam_optimize(a_init, deadline, min_time_remaining=500.0)
            _update_best(a_opt, f"adam_{trial}")

    # === Phase 3: L-BFGS refinement ===
    print(f"\n[P3] L-BFGS refinement (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a_p3 = _run_lbfgs(best_seq, deadline, min_time_remaining=400.0, beta_scale=0.3)
        _update_best(a_p3, "lbfgs_refine")

    # === Phase 4: Active-set equalization ===
    print(f"\n[P4] Active-set equalization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        pol_deadline = min(deadline - 300, time.monotonic() + 200)
        a_eq, sc_eq = _active_set_equalize(best_seq, pol_deadline, min_time_remaining=50.0)
        _update_best(a_eq, "equalize")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_current = len(best_seq)
    n_candidates = sorted(set([n_current, 16384, 12288, 24576, 8192, 32768]), reverse=True)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        if n_try == n_current:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=150.0, beta_scale=0.5, max_iters=500)
        _update_best(a_opt, f"multi_n_{n_try}")

    # === Phase 6: Nesterov polish ===
    print(f"\n[P6] Nesterov polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        pol_deadline2 = min(deadline - 80, time.monotonic() + 100)
        a_ns, sc_ns = _nesterov_polish(best_seq, pol_deadline2, min_time_remaining=30.0)
        _update_best(a_ns, "nesterov")

    # === Phase 7: Final L-BFGS with very small beta ===
    print(f"\n[P7] Final ultra-fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 10 - time.monotonic()
    if remaining > 20:
        a_final = _run_lbfgs(best_seq, deadline, min_time_remaining=10.0, beta_scale=0.1)
        _update_best(a_final, "final_ultra")

    # === Phase 8: Tiny active-set final touch ===
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _active_set_equalize(best_seq, deadline - 3, min_time_remaining=3.0)
        _update_best(a_fin, "final_equalize")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
