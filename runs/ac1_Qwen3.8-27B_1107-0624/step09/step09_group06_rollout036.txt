"""
Alternating LP + L-BFGS hybrid with multi-resolution for minimizing
2*n*max(conv(a,a))/sum(a)^2. Key ideas: (1) Alternating LP with active sets
solves a small LP at each step for exact peak reduction, (2) L-BFGS with
power-smoothed max for polishing, (3) multi-resolution refinement, (4) random
restarts to escape local minima.

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy import sparse
import math

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)

def _convolve_with_matrix(a, K_indices):
    """Build the (K x n) matrix where row k has a[k-j] in column j."""
    n = len(a)
    rows = []
    for k in K_indices:
        row = np.zeros(n)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            row[j_lo:j_hi+1] = a[k - np.arange(j_lo, j_hi+1)]
        rows.append(row)
    return np.array(rows)

def _alternating_lp(a_init, deadline, min_time_remaining=30.0, K=30, max_iters=500):
    """Alternating LP: fix a, solve LP for b that minimizes peak of a*b."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        # Find active set (top K peaks)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = float(2.0 * n * c_max / (a.sum()**2))
        
        if current_sc < best_sc - 1e-12:
            best_sc = current_sc
            best_a = a.copy()
        
        # Select top K peaks
        K_actual = min(K, len(c))
        active_idx = np.argsort(c)[::-1][:K_actual]
        active_vals = c[active_idx]
        
        # Solve LP: min M s.t. (a*b)[k] <= M for k in active_idx, sum(b)=1, b>=0
        # Variables: b[0..n-1], M (index n)
        # Objective: minimize M -> c_obj = [0, ..., 0, 1]
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        # Inequality constraints: A_ub @ x <= b_ub
        # For each active peak k: sum_j a[k-j]*b[j] - M <= 0
        A_rows = []
        for k in active_idx:
            row = np.zeros(n + 1)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                row[j_lo:j_hi+1] = a[k - np.arange(j_lo, j_hi+1)]
            row[n] = -1.0  # -M
            A_rows.append(row)
        A_ub = np.array(A_rows)
        b_ub = np.zeros(len(active_idx))
        
        # Equality constraint: sum(b) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        # Bounds: b >= 0, M >= 0
        bounds = [(0, None)] * n + [(0, None)]
        
        try:
            result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                            bounds=bounds, method='highs')
            if result.success:
                b_new = result.x[:n]
                b_new = np.maximum(b_new, 0)
                s_new = b_new.sum()
                if s_new > 1e-10:
                    b_new = b_new / s_new
                    sc_new = _score(b_new)
                    if sc_new < best_sc:
                        best_sc = sc_new
                        best_a = b_new.copy()
                    a = b_new
            else:
                break
        except Exception:
            break
    
    return best_a, best_sc

def _solve_active_lp(a, active_idx):
    """Solve LP: min M s.t. (a*b)[k] <= M for k in active_idx, sum(b)=1, b>=0.
    Returns optimal b or None if infeasible."""
    n = len(a)
    K = len(active_idx)
    if K == 0:
        return None

    # Objective: minimize M (last variable)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    # Inequality constraints: (a*b)[k] - M <= 0 for each active k
    A_ub = np.zeros((K, n + 1))
    for i, k in enumerate(active_idx):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            A_ub[i, j_lo:j_hi + 1] = a[k - np.arange(j_lo, j_hi + 1)]
        A_ub[i, n] = -1.0
    b_ub = np.zeros(K)

    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'dual': False})
        if result.success:
            b_new = np.maximum(result.x[:n], 0.0)
            s_new = b_new.sum()
            if s_new > 1e-10:
                return b_new / s_new
    except Exception:
        pass
    return None


def _alternating_lp(a_init, deadline, min_time_remaining=30.0,
                    K_init=20, K_max=200, max_iters=2000, blend=0.5, seed=0):
    """Alternating LP with adaptive active set size."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    rng = np.random.default_rng(seed)
    K = K_init
    no_improve = 0
    stall_count = 0
    t = 0

    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max  # assuming sum(a)=1

        if current_sc < best_sc - 1e-12:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive K
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
        if gap < c_max * 0.01 and K < K_max:
            K = min(K_max, int(K * 1.4) + 1)
        elif gap > c_max * 0.1 and K > K_init:
            K = max(K_init, int(K * 0.8))

        # Select active peaks
        K_actual = min(K, 2 * n - 1)
        active_idx = np.argsort(c)[::-1][:K_actual]

        # Solve LP
        b_new = _solve_active_lp(a, active_idx)
        if b_new is None:
            # Fallback: simple subgradient step
            grad = np.zeros(n)
            for k in active_idx[:min(K_actual, 20)]:
                j_lo = max(0, k - n + 1)
                j_hi = min(n - 1, k)
                if j_lo <= j_hi:
                    grad[j_lo:j_hi + 1] += a[k - np.arange(j_lo, j_hi + 1)]
                j_lo2 = max(0, k)
                j_hi2 = min(n - 1, k + n - 1)
                if j_lo2 <= j_hi2:
                    grad[j_lo2:j_hi2 + 1] += a[np.arange(j_lo2, j_hi2 + 1) - k]
            grad_t = grad - np.mean(grad)
            lr = 0.01 / (np.linalg.norm(grad_t) + 1e-18)
            a_cand = a - lr * grad_t
            a = _project_simplex(a_cand)
            continue

        # Blend for stability
        a_try = (1.0 - blend) * a + blend * b_new
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            a = b_new
            continue
        a_try = a_try / s_try

        sc_try = _score(a_try)
        if sc_try <= current_sc + 1e-12:
            a = a_try
        else:
            # Try pure b_new
            sc_pure = _score(b_new)
            if sc_pure < current_sc:
                a = b_new
            else:
                a = a_try  # still accept blended

        # Stall detection

            stall_count += 1
            if stall_count > 8:
                break
            # Random perturbation

            a_pert = _project_simplex(a + perturb)
            sc_pert = _score(a_pert)
            if sc_pert < best_sc:
                a = a_pert
                best_a = a.copy()
                best_sc = sc_pert
            no_improve = 0
            K = K_init

        if t % 200 == 0:
            print(f"    AltLP t={t}: best={best_sc:.10f}, K={K}, no_interp={no_improve}")

    return best_a, best_sc


def _dual_guided_subgrad(a_init, deadline, min_time_remaining=30.0,
                         K=50, seed=0):
    """Subgradient method with dual-weighted active set (solves dual LP for weights)."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    momentum = 0.8
    base_lr = 0.02
    lr = base_lr
    no_improve = 0
    stall_count = 0
    t = 0

    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-12:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Select active peaks
        K_actual = min(K, 2 * n - 1)
        active_idx = np.argsort(c)[::-1][:K_actual]

        # Solve dual LP for weights: max lambda s.t. sum(y)=1, A^T y >= lambda, y>=0
        # Where A[k,j] = a[k-j] for valid indices
        # Dual variables: y (K) + lambda (1) = K+1 vars
        # Constraints: (A^T y)[j] - lambda >= 0 for j=0..n-1 (n constraints)
        #              sum(y) = 1 (1 constraint)
        #              y >= 0
        # Objective: maximize lambda -> minimize -lambda
        try:
            c_dual = np.zeros(K + 1)
            c_dual[-1] = -1.0  # minimize -lambda

            # A^T y - lambda >= 0 for all j -> lambda - A^T y <= 0 for all j
            A_dual_ub = np.zeros((n, K + 1))
            A_dual_ub[:, -1] = 1.0  # lambda coefficient
            for i, k in enumerate(active_idx):
                j_lo = max(0, k - n + 1)
                j_hi = min(n - 1, k)
                if j_lo <= j_hi:
                    A_dual_ub[j_lo:j_hi + 1, i] = -a[k - np.arange(j_lo, j_hi + 1)]
            b_dual_ub = np.zeros(n)

            A_dual_eq = np.zeros((1, K + 1))
            A_dual_eq[0, :K] = 1.0  # sum(y) = 1
            b_dual_eq = np.array([1.0])

            bounds_dual = [(0, None)] * K + [(None, None)]

            result_dual = linprog(c_dual, A_ub=A_dual_ub, b_ub=b_dual_ub,
                                  A_eq=A_dual_eq, b_eq=b_dual_eq,
                                  bounds=bounds_dual, method='highs',
                                  options={'presolve': True})
            if result_dual.success:
                y_w = np.maximum(result_dual.x[:K], 0.0)
                y_w /= y_w.sum()
            else:
                y_w = np.ones(K_actual) / K_actual
        except Exception:
            y_w = np.ones(K_actual) / K_actual

        # Compute weighted gradient
        grad = np.zeros(n)
        for i, k in enumerate(active_idx):
            w_i = y_w[i]
            if w_i < 1e-15:
                continue
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                grad[j_lo:j_hi + 1] += w_i * a[k - np.arange(j_lo, j_hi + 1)]
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_lo2 <= j_hi2:
                grad[j_lo2:j_hi2 + 1] += w_i * a[np.arange(j_lo2, j_hi2 + 1) - k]

        # Project to tangent space
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break

        # Nesterov step
        effective_lr = lr / (gnorm + 1e-18) * 0.1
        v = momentum * v - effective_lr * grad_t
        a_cand = a + v
        a_proj = _project_simplex(a_cand)

        sc_new = 2.0 * n * float(_full_convolve(a_proj, a_proj).max())
        if sc_new < current_sc:
            a = a_proj
        else:
            v = 0.5 * v
            lr *= 0.95
            a = a_proj

        if no_improve > 300:
            stall_count += 1
            if stall_count > 10:
                break
            perturb = 0.003 * rng.standard_normal(n)
            a = _project_simplex(a + perturb)
            v = np.zeros(n)
            lr = base_lr
            no_improve = 0
            K = max(20, K // 2)

        if t % 500 == 0:
            print(f"    DualSG t={t}: best={best_sc:.10f}, K={K}")

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, n_betas=12):
    """L-BFGS with adaptive beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03125, 0.015, 0.008, 0.004,
                                  0.002, 0.001, 0.0005, 0.0002, 0.0001])[:n_betas]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 8))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-15, 'maxcor': 80, 'maxls': 50})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.3) ** 2) + np.exp(-12.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    elif kind == "flat_taper":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        taper = max(1, n // 40)
        for i in range(taper):
            a[i] = (i + 1) / taper
            a[n - 1 - i] = (i + 1) / taper
        return a / a.sum()
    else:
        return np.ones(n) / n


def _shrinkage_optimize(a, deadline):
    """Try zeroing out the smallest elements to improve score."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.001, 0.003, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def _random_restart_optimize(a_init, deadline, min_time_remaining=60.0, seed=0):
    """Try multiple random perturbations and optimize the best one."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    best_a = a_init.copy()
    best_sc = _score(a_init)

    n_restarts = 5
    time_per_restart = max(5.0, (deadline - min_time_remaining - time.monotonic()) / n_restarts)

    for r in range(n_restarts):
        if time.monotonic() + min_time_remaining > deadline:
            break

        # Generate perturbation
        scale = 0.05 + 0.03 * r
        perturb = scale * rng.standard_normal(n)
        a_pert = _project_simplex(a_init + perturb)
        sc_pert = _score(a_pert)

        # Quick optimization of perturbed version
        if time.monotonic() + min_time_remaining > deadline:
            break
        a_opt = _run_lbfgs(a_pert, deadline, min_time_remaining=min_time_remaining, n_betas=6)
        sc_opt = _score(a_opt)

        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"    Restart {r}: improved to {best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Main optimization: alternating LP + L-BFGS with multi-resolution."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    target_n = n_parent if n_parent > 0 else 16384

    # === Phase 1: Alternating LP (main workhorse) ===
    print(f"\n[P1] Alternating LP (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 600, t_start + 500)
    a_alt, sc_alt = _alternating_lp(best_seq, p1_deadline,
                                    min_time_remaining=60.0,
                                    K_init=20, K_max=200,

    if sc_alt < best_score:
        best_seq = a_alt.copy()
        best_score = sc_alt
        print(f"  NEW BEST: {best_score:.10f} [alt_lp] t={time.monotonic()-t_start:.1f}s")

        print(f"  AltLP: no improvement (best={best_score:.10f})")

    # === Phase 2: L-BFGS polish ===
    print(f"\n[P2] L-BFGS polish (t={time.monotonic()-t_start:.1f}s)")

        best_score = sc_lb

    # === Phase 3: Dual-guided subgradient ===
    print(f"\n[P3] Dual-guided subgradient (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 400, t_start + 900)
    a_dg, sc_dg = _dual_guided_subgrad(best_seq, p3_deadline,
                                       min_time_remaining=60.0, K=50, seed=seed + 1)

        best_seq = a_dg.copy()
        best_score = sc_dg
        print(f"  NEW BEST: {best_score:.10f} [dual_sg] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Second alternating LP with different params ===
    print(f"\n[P4] AltLP round 2 (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 300, t_start + 1100)
    a_alt2, sc_alt2 = _alternating_lp(best_seq, p4_deadline,
                                      min_time_remaining=60.0,

    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30:

        if sc_fin1 < best_score:

            best_score = sc_fin1

    remaining = deadline - 5 - time.monotonic()
    if remaining > 15:
        a_fin2 = _run_lbfgs(best_seq, deadline - 5, min_time_remaining=5.0, n_betas=8)
        sc_fin2 = _score(a_fin2)
        if sc_fin2 < best_score:
            best_seq = a_fin2.copy()
            best_score = sc_fin2
            print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)

        if s < 0.01:

        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
