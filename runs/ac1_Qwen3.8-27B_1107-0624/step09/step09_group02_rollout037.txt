def compute_gradient(a, w, n):
    """Compute gradient of sum_k w[k]*c[k] where c=conv(a,a).
    w has length 2n-1. Returns array of length n."""
    # g[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # = cross-correlation computed via FFT
    w_rev = w[::-1]  # length 2n-1
    conv_len = n + (2*n-1) - 1  # = 3n-2
    N = 1
    while N < conv_len:
        N *= 2
    A = np.fft.rfft(a, N)
    Wr = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(A * Wr, N)[:conv_len]
    # g[i] = conv[2n-2-i]
    g = conv[n-1:2*n-1][::-1]
    return 2.0 * g
```

def hankel_multiply(p, v, n):
    """Compute M_p * v where M_p[i,j] = p[i+j] for i+j < 2n-1.
    p has length 2n-1, v has length n. Returns array of length n."""

    conv_len = n + (2*n-1) - 1  # 3n-2
    N = 1
    while N < conv_len:
        N *= 2
    V = np.fft.rfft(v, N)
    Pr = np.fft.rfft(p_rev, N)
    conv = np.fft.irfft(V * Pr, N)[:conv_len]
    return conv[n-1:2*n-1][::-1]
```

"""
Optimized step function generator using corrected gradient, Hankel CG solver,
log-sum-exp L-BFGS, active-set refinement, and multi-n exploration.

correction). Uses M_p * v via FFT for O(n log n) iterations.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _hankel_multiply(p, v, n):
    """Compute M_p * v where M_p[i,j] = p[i+j] for i+j < 2n-1.
    p has length 2n-1, v has length n. Returns array of length n.
    Uses FFT: result[i] = sum_j p[i+j]*v[j]."""
    p_rev = p[::-1]
    conv_len = n + (2 * n - 1) - 1
    N = _next_pow2(conv_len)
    V = np.fft.rfft(v, N)
    Pr = np.fft.rfft(p_rev, N)
    conv = np.fft.irfft(V * Pr, N)[:conv_len]
    return conv[n - 1:2 * n - 1][::-1]


def _grad_smoothed(a, w, n):
    """Correct gradient of sum_k w[k]*c[k] where c=conv(a,a).
    w has length 2n-1. Gradient[i] = 2 * sum_j a[j]*w[i+j]."""
    return 2.0 * _hankel_multiply(w, a, n)


def _cg_solve_hankel(p, rhs, n, tol=1e-10, max_iter=2000, reg=0.0):
    """Solve (M_p + reg*I) * x = rhs using conjugate gradient.
    Returns x of length n."""
    x = np.ones(n) / n
    if reg > 0:
        def mv(v):
            return _hankel_multiply(p, v, n) + reg * v
    else:
        def mv(v):
            return _hankel_multiply(p, v, n)

    r = rhs - mv(x)
    s = r.copy()
    rs_old = np.dot(r, r)
    if rs_old < 1e-30:
        return x
    for i in range(max_iter):
        As = mv(s)
        ss = np.dot(s, s)
        if ss < 1e-30:
            break
        alpha = rs_old / ss
        x = x + alpha * s
        r = r - alpha * As
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * np.dot(rhs, rhs) + 1e-20:
            break
        s = r + (rs_new / rs_old) * s
        rs_old = rs_new
    return x


def _active_set_solve(a, p, n, blend=1.0):
    """Given weights p on lags, solve M_p * x = 1, normalize, blend with a."""
    rhs = np.ones(n)
    x = _cg_solve_hankel(p, rhs, n, tol=1e-10, max_iter=1000)
    s = float(x.sum())
    if s < 1e-15:
        return a.copy()
    x = x / s
    x = np.maximum(x, 0)
    s2 = float(x.sum())
    if s2 < 1e-15:
        return a.copy()
    x = x / s2
    if blend >= 1.0:
        return x
    return (1.0 - blend) * a + blend * x


def _make_beta_profile(n, p, q):
    """Create a beta-distribution shaped profile: a[i] ∝ (i+1)^p * (n-i)^q."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p * (n - i + 1.0) ** q
    a = np.maximum(a, 1e-30)
    return a / a.sum()

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _active_set_iteration(a_init, deadline, min_time_remaining=60.0):
    """Iterative active-set method using Hankel CG solver.
    Alternates between updating p (active lag weights) and solving for a."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    k_schedule = [2, 3, 4, 5, 7, 10, 15, 20, 30, 50, 80, 120, 200, 400, 800, 1600]
    k_idx = 0
    it = 0
    max_it = 300
    blend = 1.0

    while time.monotonic() + min_time_remaining < deadline and it < max_it:
        it += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Determine active set size
        k = k_schedule[min(k_idx, len(k_schedule) - 1)]
        k = min(k, 2 * n - 1)

        # Top-k lags
        top_k = np.argsort(c)[::-1][:k]
        p = np.zeros(2 * n - 1)
        p[top_k] = 1.0 / k

        # Solve for a using Hankel CG
        a_new = _active_set_solve(a, p, n, blend=blend)
        a = a_new

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # Adapt step
        if it % 10 == 0:
            k_idx = min(k_idx + 1, len(k_schedule) - 1)
            blend = max(0.3, blend * 0.95)

    return best_a, best_sc

        beta = base_beta * (0.85 ** min(it, 80))
        p = _softmax(c / beta)

        a_new = _active_set_solve(a, p, n, blend=1.0)
        a = a_new

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _proj_grad_run(a_init, deadline, min_time_remaining=60.0):
    """Projected gradient with Nesterov momentum and corrected gradient."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    momentum = np.zeros(n)
    beta_m = 0.9
    patience = 0
    max_patience = 30
    k_active = 3

    for step in range(3000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        a_curr = _softmax(z)
        c = _full_convolve(a_curr, a_curr)
        c_max = float(c.max())

        top_k = np.argsort(c)[::-1][:k_active]
        p = np.zeros(2 * n - 1)
        p[top_k] = 1.0 / k_active

        grad = _grad_smoothed(a_curr, p, n)
        # Project gradient to tangent space of simplex
        at_ga = float(np.dot(a_curr, grad))
        grad_t = grad - a_curr * at_ga
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-15:
            break

        momentum = beta_m * momentum + (1 - beta_m) * grad_t
        lr = 0.1 / (np.linalg.norm(momentum) + 1e-10)
        z = z - lr * momentum
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if step % 5 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros(n)
                    patience = 0
                    beta_m *= 0.9
                    k_active = min(k_active + 2, 50)

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer with corrected gradients, Hankel CG active-set solver,
    L-BFGS with log-sum-exp, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s",
                  flush=True)

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}", flush=True)
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 32768
        parent = _make_beta_profile(n_parent, 2.0, 1.0)
        print(f"No parent, using beta(2,1) n={n_parent}", flush=True)
        update_best(parent, _score(parent), "init")

    # === Phase 1: Parametric sweep (30s) ===
    print(f"\n[Phase 1] Parametric sweep (t={time.monotonic() - t_start:.1f}s)", flush=True)
    phase1_dl = t_start + 30
    n_try = min(n_parent, 4096)  # Use smaller n for quick sweep
    a_param, sc_param = _make_parametric(n_try, None)
    a_param_full = _resample(a_param, n_parent)
    sc_param_full = _score(a_param_full)
    update_best(a_param_full, sc_param_full, "parametric")

    # === Phase 2: L-BFGS from best init (500s) ===
    print(f"\n[Phase 2] L-BFGS (t={time.monotonic() - t_start:.1f}s)", flush=True)
    phase2_dl = min(deadline - 800, t_start + 530)
    init_a = best_seq if best_seq is not None else parent
    a_lb = _lbfgs_run(init_a, phase2_dl, min_time_remaining=30.0)
    update_best(a_lb, _score(a_lb), "lbfgs_1")

    # Try from parametric init
    if time.monotonic() < phase2_dl - 20 and sc_param_full < best_score + 0.01:
        a_lb2 = _lbfgs_run(a_param_full, phase2_dl, min_time_remaining=20.0)
        update_best(a_lb2, _score(a_lb2), "lbfgs_param")

    # === Phase 3: Active-set CG iteration (400s) ===
    print(f"\n[Phase 3] Active-set CG (t={time.monotonic() - t_start:.1f}s)", flush=True)
    phase3_dl = min(deadline - 600, phase2_dl + 400)
    if best_seq is not None and time.monotonic() < phase3_dl:
        a_as, sc_as = _active_set_iteration(best_seq, phase3_dl, min_time_remaining=30.0)
        update_best(a_as, sc_as, "active_set_1")

    # === Phase 4: Softmax dual iteration (300s) ===
    print(f"\n[Phase 4] Softmax Dual (t={time.monotonic() - t_start:.1f}s)", flush=True)
    phase4_dl = min(deadline - 400, phase3_dl + 300)
    if best_seq is not None and time.monotonic() < phase4_dl:
        a_sd, sc_sd = _softmax_dual_iteration(best_seq, phase4_dl, min_time_remaining=30.0)
        update_best(a_sd, sc_sd, "softmax_dual_1")

    # === Phase 5: Projected gradient with momentum (200s) ===
    print(f"\n[Phase 5] Proj Gradient (t={time.monotonic() - t_start:.1f}s)", flush=True)
    phase5_dl = min(deadline - 250, phase4_dl + 200)
    if best_seq is not None and time.monotonic() < phase5_dl:
        a_pg, sc_pg = _proj_grad_run(best_seq, phase5_dl, min_time_remaining=20.0)
        update_best(a_pg, sc_pg, "proj_grad_1")

    # === Phase 6: Second round L-BFGS + active set (250s) ===
    print(f"\n[Phase 6] Round 2 (t={time.monotonic() - t_start:.1f}s)", flush=True)
    phase6_dl = min(deadline - 180, phase5_dl + 250)
    if best_seq is not None and time.monotonic() < phase6_dl:
        a_lb3 = _lbfgs_run(best_seq, phase6_dl, min_time_remaining=30.0)
        update_best(a_lb3, _score(a_lb3), "lbfgs_r2")
        if time.monotonic() < phase6_dl - 20:
            a_as2, sc_as2 = _active_set_iteration(best_seq, phase6_dl,
                                                   min_time_remaining=15.0)
            update_best(a_as2, sc_as2, "active_set_r2")

    # === Phase 7: Multi-n exploration (200s) ===
    print(f"\n[Phase 7] Multi-n (t={time.monotonic() - t_start:.1f}s)", flush=True)
    phase7_dl = min(deadline - 100, phase6_dl + 200)
    n_options = [16384, 65536, 49152, 24576]
    for n_opt in n_options:
        if time.monotonic() > phase7_dl - 30:
            break
        if n_opt == len(best_seq) if best_seq is not None else False:
            continue
        if best_seq is not None:
            a_res = _resample(best_seq, n_opt)
        else:
            a_res = _make_beta_profile(n_opt, 2.0, 1.0)
        a_opt = _lbfgs_run(a_res, phase7_dl, min_time_remaining=20.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_opt}")

        if sc_opt < best_score + 0.002 and time.monotonic() < phase7_dl - 20:
            a_opt2, sc_opt2 = _active_set_iteration(a_opt, phase7_dl,
                                                     min_time_remaining=15.0)
            update_best(a_opt2, sc_opt2, f"n_{n_opt}_as")

    # === Phase 8: Perturbation + re-optimize (150s) ===
    print(f"\n[Phase 8] Perturb+Reopt (t={time.monotonic() - t_start:.1f}s)", flush=True)
    phase8_dl = min(deadline - 50, phase7_dl + 150)
    if best_seq is not None and time.monotonic() < phase8_dl:
        n_pert = [0.01, 0.03, 0.05, 0.1]
        for sig in n_pert:
            if time.monotonic() > phase8_dl - 20:
                break
            a_p = _perturb(best_seq, sig, rng)
            a_opt = _lbfgs_run(a_p, phase8_dl, min_time_remaining=15.0)
            update_best(a_opt, _score(a_opt), f"pert_{sig}")

    # === Phase 9: Final ultra-tight polish (50s) ===
    print(f"\n[Phase 9] Final Polish (t={time.monotonic() - t_start:.1f}s)", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_fin, sc_fin = _active_set_iteration(best_seq, deadline - 15,
                                               min_time_remaining=5.0)
        update_best(a_fin, sc_fin, "final_as")

        if time.monotonic() < deadline - 15:
            a_fin2, sc_fin2 = _softmax_dual_iteration(best_seq, deadline - 10,
                                                       min_time_remaining=5.0)
            update_best(a_fin2, sc_fin2, "final_sd")

    # Final validation
    print(f"\n{'=' * 60}", flush=True)
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_beta_profile(256, 1.0, 1.0)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s", flush=True)
        return [float(x) for x in final]

    fallback = _make_beta_profile(8192, 2.0, 1.0)
    print(f"FALLBACK: score={_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
