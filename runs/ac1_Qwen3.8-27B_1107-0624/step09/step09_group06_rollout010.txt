"""
Hybrid spectral shaping + Nesterov subgradient + L-BFGS continuation optimizer
for minimizing 2*n*max(conv(a,a))/sum(a)^2. Key new idea: Gerchberg-Saxton-style
spectral flattening to reshape the magnitude spectrum, combined with improved
active-set subgradient and continuation-based L-BFGS polishing.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a)=1 (faster)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    """Project x onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _conv_grad_single(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    if j_lo <= j_hi:
        idx = k - np.arange(j_lo, j_hi + 1)
        grad[j_lo:j_hi + 1] += a[idx]
    j_lo2 = max(0, k)
    j_hi2 = min(n - 1, k + n - 1)
    if j_lo2 <= j_hi2:
        idx = np.arange(j_lo2, j_hi2 + 1) - k
        grad[j_lo2:j_hi2 + 1] += a[idx]
    return grad


def _conv_grad_multi(a, active_indices):
    """Gradient of sum of conv(a,a)[k] for k in active_indices."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_indices:
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            idx = k - np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi + 1] += a[idx]
        j_lo2 = max(0, k)
        j_hi2 = min(n - 1, k + n - 1)
        if j_lo2 <= j_hi2:
            idx = np.arange(j_lo2, j_hi2 + 1) - k
            grad[j_lo2:j_hi2 + 1] += a[idx]
    return grad


def _spectral_shaping(a, n_iters=200, gamma_schedule=None, deadline=None,
                      min_time_remaining=30.0, alpha_init=0.95, alpha_min=0.5):
    """
    Gerchberg-Saxton style spectral flattening.
    Iteratively flattens the magnitude spectrum while maintaining a >= 0, sum(a)=1.
    """
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score_normalized(a)

    N = _next_pow2(2 * n - 1)

    if gamma_schedule is None:
        gamma_schedule = np.linspace(0.1, 0.6, n_iters)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() + min_time_remaining > deadline:
            break

        # Ensure on simplex
        a = _project_simplex(a)

        # FFT
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)

        # Target magnitude: geometric mean (log-uniform spectrum)
        log_mag = np.log(mag + 1e-30)
        target_mag = np.exp(np.mean(log_mag))

        gamma = gamma_schedule[min(it, len(gamma_schedule) - 1)]
        alpha = alpha_init + (alpha_min - alpha_init) * (it / max(1, n_iters))

        # Blend current magnitude toward target
        mag_new = mag * (1.0 - gamma) + target_mag * gamma

        # Additional spectral flattening: compress dynamic range
        max_mag = mag_new.max()
        min_mag = mag_new.min()
        if max_mag > 1e-30:
            normalized = (mag_new - min_mag) / (max_mag - min_mag + 1e-30)
            compressed = normalized ** alpha
            mag_new = min_mag + compressed * (max_mag - min_mag)

        # Reconstruct
        A_new = mag_new * phase
        a_new = np.fft.irfft(A_new, N)[:n].real

        # Project to simplex
        a_new = _project_simplex(a_new)

        # Evaluate
        sc = _score_normalized(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
        else:
            a = a_new  # still move forward in spectral domain

    return best_a, best_sc


def _nesterov_subgrad_v2(a_init, deadline, min_time_remaining=60.0,
                         K_init=10, K_max=500, threshold_frac=0.999,
                         seed=0):
    """
    Improved Nesterov subgradient with threshold-based active set and
    Barzilai-Borwein adaptive step size.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    momentum = 0.9
    lr = 0.02
    t = 0
    no_improve = 0
    stall_count = 0

    # Track previous for BB step size
    a_prev = a.copy()
    grad_prev = None

    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Active set: all peaks above threshold
        threshold = c_max * threshold_frac
        active = np.where(c >= threshold)[0]
        if len(active) < 3:
            c_idx = np.argsort(c)[::-1]
            active = c_idx[:max(3, K_init)]
        active = active[:min(len(active), K_max)]

        # Subgradient
        grad = _conv_grad_multi(a, active)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-18:
            break

        # Barzilai-Borwein step size
        if grad_prev is not None:
            s_bb = a - a_prev
            y_bb = grad_t - grad_prev
            sy = np.dot(s_bb, y_bb)
            ss = np.dot(s_bb, s_bb)
            yy = np.dot(y_bb, y_bb)
            if ss > 1e-18 and yy > 1e-18:
                if t % 2 == 0:
                    lr_bb = ss / sy if sy > 1e-15 else lr
                else:
                    lr_bb = sy / yy
                lr = 0.5 * (lr + max(1e-6, min(0.1, lr_bb)))

        a_prev = a.copy()
        grad_prev = grad_t.copy()

        # Nesterov step
        effective_lr = lr / (gnorm + 1e-18) * 0.5
        v = momentum * v - effective_lr * grad_t
        a_cand = a + v
        a_proj = _project_simplex(a_cand)

        sc_new = _score_normalized(a_proj)

        if sc_new < current_sc:
            a = a_proj
        else:
            # Backtrack: try smaller step
            v = 0.5 * v
            a_half = _project_simplex(a + 0.5 * v)
            sc_half = _score_normalized(a_half)
            if sc_half < current_sc:
                a = a_half
            else:
                a = a_proj  # still stay feasible
            lr *= 0.8

        # Periodic restart
        if no_improve % 300 == 0 and no_improve > 0:
            stall_count += 1
            if stall_count > 15:
                break
            perturb = 0.0005 * rng.standard_normal(n)
            a_pert = _project_simplex(a + perturb)
            sc_pert = _score_normalized(a_pert)
            if sc_pert < best_sc:
                a = a_pert
                best_a = a.copy()
                best_sc = sc_pert
            v = np.zeros(n)
            lr = 0.02
            no_improve = 0
            momentum = min(0.95, momentum * 1.01)

        if t % 500 == 0:
            print(f"    NesterovV2 t={t}: best={best_sc:.10f}, K={len(active)}, lr={lr:.6f}")

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs_continuation(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with continuation method (coarse to fine beta)."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.01)

    # Coarse to fine beta schedule (geometric)
    n_betas = 15
    betas = base_beta * (0.6 ** np.arange(n_betas))
    # Ensure beta doesn't get too small
    min_beta = max_c * 1e-5
    betas = np.maximum(betas, min_beta)

    current_a = a_init.copy()
    current_sc = _score_normalized(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        # More iterations for larger beta (easier subproblem)
        est_time_per_iter = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(100, int(remaining / (est_time_per_iter * 15))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score_normalized(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 3 == 0:
                    print(f"    LBFGS cont beta_idx={i}/{n_betas}: score={current_sc:.10f}, beta={beta:.2e}")
        except Exception as e:
            print(f"    LBFGS error at beta_idx={i}: {e}")

    return current_a


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        freq = 5.0 + (7.0 if rng is not None else 7.0)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    elif kind == "flat_taper":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        taper = max(1, n // 30)
        for i in range(taper):
            a[i] = (i + 1) / taper
            a[n - 1 - i] = (i + 1) / taper
        return a / a.sum()
    elif kind == "random_smooth":
        if rng is None:
            rng = np.random.default_rng(0)
        # Generate random smooth profile
        k_modes = 3 + rng.integers(5)
        t = np.arange(n) / n
        a = np.ones(n)
        for _ in range(k_modes):
            freq = rng.uniform(1, 20)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.05, 0.3)
            a += amp * np.cos(2 * np.pi * freq * t + phase)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _asymmetric_init(n, rng):
    """Create asymmetric initializations that can break the symmetry barrier."""
    t = np.linspace(0, 1, n)
    # Shifted Gaussian
    center = rng.uniform(0.2, 0.8)
    width = rng.uniform(0.1, 0.4)
    a = np.exp(-0.5 * ((t - center) / width) ** 2) + 0.01
    # Add asymmetry
    a *= (1.0 + rng.uniform(0.1, 0.5) * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _shrinkage_optimize(a, deadline):
    """Try zeroing out small elements to reduce effective length."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.001, 0.003, 0.005, 0.01, 0.02, 0.03]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc

    if time.monotonic() < deadline - 200:
        p5_deadline = min(deadline - 200, t_start + 1200)
        for round_i in range(3):
            if time.monotonic() + 60 > p5_deadline:
                break
            # Nesterov step
            a_n5, sc_n5 = _nesterov_subgrad_v2(best_seq, p5_deadline,
                                               min_time_remaining=30.0,
                                               K_init=5, K_max=200,
                                               threshold_frac=0.999,
                                               seed=seed + 10 * (round_i + 1))
            sc_n5_full = _score(a_n5)
            if sc_n5_full < best_score:
                best_seq = a_n5.copy()
                best_score = sc_n5_full
                print(f"  P5 round {round_i} nesterov: {best_score:.10f}")

            # Spectral step
            if time.monotonic() + 30 > p5_deadline:
                break
            a_s5, sc_s5 = _spectral_shaping(best_seq, n_iters=100,
                                            deadline=p5_deadline,
                                            min_time_remaining=20.0,
                                            alpha_init=0.8, alpha_min=0.4)
            sc_s5_full = _score(a_s5)
            if sc_s5_full < best_score:
                best_seq = a_s5.copy()
                best_score = sc_s5_full
                print(f"  P5 round {round_i} spectral: {best_score:.10f}")

    # === Phase 6: Multi-n exploration with different inits ===
    print(f"\n[P6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [16384, 32768, 8192, 65536, 24576]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == len(best_seq):
            continue

        # Try multiple initializations for this n
        for kind in ["uniform", "tapered", "asymmetric"]:
            if time.monotonic() > deadline - 60:
                break
            if kind == "asymmetric":
                a_init = _asymmetric_init(n_try, rng)
            else:
                a_init = _make_profile(n_try, kind, rng)

            # Also try resampling from best
            a_res = _resample(best_seq, n_try)
            # Use the better of the two
            sc_init1 = _score(a_init)
            sc_init2 = _score(a_res)
            if sc_init2 < sc_init1:
                a_init = a_res

            # Quick spectral + L-BFGS
            if time.monotonic() < deadline - 50:
                a_opt = _spectral_shaping(a_init, n_iters=50,
                                          deadline=deadline - 30,
                                          min_time_remaining=20.0)
                a_opt = a_opt[0]
                sc_opt = _score(a_opt)

                if time.monotonic() < deadline - 30:
                    a_fin = _run_lbfgs_continuation(a_opt, deadline, min_time_remaining=25.0)
                    sc_fin = _score(a_fin)
                    if sc_fin < sc_opt:
                        a_opt = a_fin
                        sc_opt = sc_fin

                if sc_opt < best_score:
                    best_seq = a_opt.copy()
                    best_score = sc_opt
                    print(f"  NEW BEST: {best_score:.10f} (n={n_try}, {kind}) t={time.monotonic()-t_start:.1f}s")
                else:
                    print(f"  n={n_try}, {kind}: score={sc_opt:.10f} (not better)")

    # === Phase 7: Shrinkage ===
    print(f"\n[P7] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 20)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} [shrinkage] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final L-BFGS polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 10 - time.monotonic()
    if remaining > 20:
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        c_sorted = np.sort(c_best)[::-1]
        gap = mb - c_sorted[min(20, len(c_sorted) - 1)]
        beta_fin = max(gap * 0.05, mb * 1e-4)
        prob = _LbfgsProb(beta_fin, n)
        max_iter = min(3000, max(50, int(remaining / 0.03)))
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxcor': 80})
            a_fin2 = _softmax(result.x)
            sc_fin2 = _score(a_fin2)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
        except Exception:
            pass

    # === Final validation ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
