            max_iter = min(3000, max(30, int(remaining / 0.004)))
            ftol = 1e-17
            gtol = 1e-14

        prob = _SmoProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': ftol, 'gtol': gtol,
                         'maxcor': maxcor, 'maxls': 80}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_equalize(a_init, deadline, min_time_remaining=20.0):
    """Iteratively equalize top-K peaks using analytical gradient structure."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    K = 5
    K_max = min(200, n)
    lr = 0.03
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            if K < K_max:
                K = min(K_max, K + 1)

        top_k_idx = np.argsort(c)[::-1][:K]

        w_S = np.zeros(2 * n - 1)
        w_S[top_k_idx] = 1.0 / K
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        alpha = lr / (gnorm + 1e-15)
        improved = False
        for _ in range(40):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            lr *= 0.9
            if no_improve % 30 == 0:
                K = max(5, K // 2)
            if no_improve >= 300:
                a = best_a.copy()
                lr = 0.03
                K = 5
                no_improve = 0
                if no_improve > 600:
                    break
        else:
            lr = min(0.06, lr * 1.01)

    return best_a, best_sc


def _nesterov_subgrad(a_init, deadline, min_time_remaining=30.0):
    """Nesterov-accelerated subgradient on simplex for global convergence."""
    n = len(a_init)
    x = a_init.copy()
    y = a_init.copy()
    x_old = x.copy()
    best_a = x.copy()
    best_sc = _score(x)

    for t in range(1, 100000):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(y, y)
        c_max = float(c.max())
        K = min(30, n)
        top_k = np.argsort(c)[::-1][:K]

        # Weighted subgradient
        w_vec = np.zeros(2 * n - 1)
        w_vec[top_k] = 1.0 / K
        y_rev = y[::-1]
        h = _full_convolve(w_vec, y_rev)[n - 1:2 * n - 1]
        g = 2.0 * h - 2.0 * np.mean(h)

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-20:
            break

        gamma = 0.5 / (np.sqrt(t) * (gnorm + 1e-15))
        x_new = _proj_simplex(y - gamma * g)

        momentum = (t - 1.0) / (t + 1.0)
        y_new = x_new + momentum * (x_new - x_old)
        y_new = _proj_simplex(y_new)

        x_old = x
        x = x_new
        y = y_new

        if t % 20 == 0:
            sc = _score(x)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_a = x.copy()

    sc_final = _score(x)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = x.copy()

    return best_a, best_sc


def _asymmetric_perturb(a, rng, strength=0.05):
    n = len(a)
    t_arr = np.arange(n) / n
    ramp = 0.3 * (t_arr - 0.5)
    noise = 0.7 * rng.standard_normal(n)
    perturbation = ramp + noise
    perturbation = perturbation - perturbation.mean()
    z = np.log(np.maximum(a, 1e-30))
    z = z + strength * perturbation
    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t_arr = np.linspace(0, 1, n)
        a = np.sin(np.pi * t_arr) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sine_mod":
        t_arr = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 7.0 * t_arr / n) + 0.1 * np.cos(2.0 * np.pi * 3.0 * t_arr / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _direct_peak_equality(a_init, deadline, min_time_remaining=10.0):
    """Directly solve for peak equality using Newton-like iterations on the active set."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    for iteration in range(500):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-16:
            best_sc = current_sc
            best_a = a.copy()

        # Find peaks to equalize: those within 0.1% of max
        threshold = c_max * 0.999
        active = np.where(c >= threshold)[0]
        if len(active) < 2:
            active = np.argsort(c)[::-1][:5]
        K = len(active)
        if K > 50:
            active = np.argsort(c)[::-1][:50]
            K = 50

        # Target: make all active peaks equal to c_max
        # Gradient of (c_k - c_max) for active k
        # We want to move a to reduce the spread of active peaks
        # Use the difference between each active peak and the mean of active peaks
        active_vals = c[active]
        mean_active = active_vals.mean()

        # Compute gradient that equalizes active peaks
        w_vec = np.zeros(2 * n - 1)
        for idx in active:
            w_vec[idx] = (c[idx] - mean_active) / (np.sum((active_vals - mean_active)**2) + 1e-30)

        a_rev = a[::-1]
        h = _full_convolve(w_vec, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-20:
            break

        # Damped Newton step
        scale = 0.1 / (gnorm + 1e-15)
        a_try = a + scale * grad
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            break
        a_try = a_try / s_try
        c_try = _full_convolve(a_try, a_try)
        sc_try = 2.0 * n * float(c_try.max())

        if sc_try < current_sc - 1e-16:
            a = a_try
        else:
            # Try smaller step
            for factor in [0.5, 0.25, 0.1]:
                a_try2 = a + factor * scale * grad
                a_try2 = np.maximum(a_try2, 0.0)
                s_try2 = a_try2.sum()
                if s_try2 < 1e-15:
                    continue
                a_try2 = a_try2 / s_try2
                c_try2 = _full_convolve(a_try2, a_try2)
                sc_try2 = 2.0 * n * float(c_try2.max())
                if sc_try2 < current_sc - 1e-16:
                    a = a_try2
                    break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase optimizer with aggressive small-beta L-BFGS, Nesterov subgradient,
    and direct peak equality polishing."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Initial L-BFGS with moderate beta (warmup) ===
    print(f"\n[P1] L-BFGS warmup (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 1400, t_start + 250)
    a_p1 = _run_lbfgs(best_seq, p1_deadline, min_time_remaining=30.0, beta_scale=1.0)
    sc_p1 = _score(a_p1)
    if sc_p1 < best_score:
        best_seq = a_p1.copy()
        best_score = sc_p1
        print(f"  NEW BEST: {best_score:.10f} [warmup]")

    # === Phase 2: Nesterov subgradient for global convergence ===
    print(f"\n[P2] Nesterov subgradient (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 1000:
        p2_deadline = min(deadline - 1000, t_start + 550)
        a_ns, sc_ns = _nesterov_subgrad(best_seq, p2_deadline, min_time_remaining=30.0)
        if sc_ns < best_score:
            best_seq = a_ns.copy()
            best_score = sc_ns
            print(f"  NEW BEST: {best_score:.10f} [nesterov]")
        else:
            print(f"  Nesterov: no improvement (best={best_score:.10f})")

    # === Phase 3: Fine L-BFGS (main optimization) ===
    print(f"\n[P3] Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        p3_deadline = min(deadline - 500, t_start + 1000)
        a_p3 = _run_lbfgs(best_seq, p3_deadline, min_time_remaining=30.0, beta_scale=0.5)
        sc_p3 = _score(a_p3)
        if sc_p3 < best_score:
            best_seq = a_p3.copy()
            best_score = sc_p3
            print(f"  NEW BEST: {best_score:.10f} [fine_lbfgs]")

    # === Phase 4: Active set equalization ===
    print(f"\n[P4] Active-set equalization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        p4_deadline = min(deadline - 400, t_start + 1200)
        a_eq, sc_eq = _active_set_equalize(best_seq, p4_deadline, min_time_remaining=30.0)
        if sc_eq < best_score:
            best_seq = a_eq.copy()
            best_score = sc_eq
            print(f"  NEW BEST: {best_score:.10f} [equalize]")
        else:
            print(f"  Equalize: no improvement (best={best_score:.10f})")

    # === Phase 5: Asymmetric perturbation + re-optimize ===
    print(f"\n[P5] Asymmetric exploration (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        for trial in range(3):
            if time.monotonic() > deadline - 300:
                break
            strength = 0.02 * (trial + 1)
            a_asym = _asymmetric_perturb(best_seq, rng, strength=strength)
            p5_dl = min(deadline - 300 + trial * 10, t_start + 1400 + trial * 60)
            a_opt = _run_lbfgs(a_asym, p5_dl, min_time_remaining=300 - trial * 60, beta_scale=0.3)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (trial={trial}) [asym]")
            else:
                print(f"  Asym {trial}: {sc_opt:.10f} (no improvement)")

    # === Phase 6: Ultra-fine L-BFGS ===
    print(f"\n[P6] Ultra-fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        p6_deadline = min(deadline - 200, t_start + 1600)
        a_fine = _run_lbfgs(best_seq, p6_deadline, min_time_remaining=30.0, beta_scale=0.2)
        sc_fine = _score(a_fine)
        if sc_fine < best_score:
            best_seq = a_fine.copy()
            best_score = sc_fine
            print(f"  NEW BEST: {best_score:.10f} [ultra_fine]")

    # === Phase 7: Direct peak equality ===
    print(f"\n[P7] Direct peak equality (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        p7_deadline = min(deadline - 100, t_start + 1700)
        a_dp, sc_dp = _direct_peak_equality(best_seq, p7_deadline, min_time_remaining=20.0)
        if sc_dp < best_score:
            best_seq = a_dp.copy()
            best_score = sc_dp
            print(f"  NEW BEST: {best_score:.10f} [peak_eq]")

    # === Phase 8: Multi-n exploration ===
    print(f"\n[P8] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_cands = sorted(set([len(best_seq), 16384, 24576, 12288]), reverse=True)
    for n_try in n_cands:
        if time.monotonic() > deadline - 50:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=0.3)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n]")
        else:
            print(f"  n={n_try}: {sc_opt:.10f} (no improvement)")

    # === Phase 9: Final polished L-BFGS ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 5 - time.monotonic()
    if remaining > 15:
        a_final = _run_lbfgs(best_seq, deadline, min_time_remaining=5.0, beta_scale=0.1)
        sc_final = _score(a_final)
        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final
            print(f"  NEW BEST: {best_score:.10f} [final]")

        # Tiny active set
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _active_set_equalize(best_seq, deadline - 2, min_time_remaining=2.0)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_eq]")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
