import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_w(a, w, n):
    """Compute the gradient contribution: for each i, sum_k w[k] * a[k-i] (valid range)."""
    flip_a = a[::-1]
    out_len = n + len(w) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fw = np.fft.rfft(w, N)
    result = np.fft.irfft(fa * fw, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):

    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _lsobj_factory(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad_factory(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    w_sum = float(w.sum())
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    # d/da [2n * (cmax + beta*log(sum exp((c-cmax)/beta)))]
    # = 2n * W (the weighted average is the derivative of logsumexp w.r.t. cmax)
    # Actually the gradient w.r.t. a is: 2n * (sum_k w_k * dc_k/da) - 2n*W * da_sum/da
    # For the softmax parameterization, we get: 4n * a * (h - W) but need to be careful
    # The standard result: grad_z = 4*n*a*(h - W) where h = corr(a, w) restricted
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec, rhs, max_cg=300, tol=1e-14):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(max(rs_new, 0)) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _newton_step(a, c, c_max, tol_frac):
    n = len(a)
    active = c >= (c_max - tol_frac * c_max)
    if not active.any():
        active[np.argmax(c)] = True
    m = int(active.sum())
    if m < 2 or m > 20 * n:
        return None, None

    A_vec = active.astype(np.float64) / m
    g = 2.0 * _corr_a_w(a, A_vec, n)
    g_t = g - np.mean(g)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-14:
        return None, None

    mu = 1e-6 / m + 1e-10

    A_vec_copy = A_vec
    n_copy = n
    mu_copy = mu

    def _hess_vp(v):
        return 2.0 * _corr_a_w(v, A_vec_copy, n_copy) + mu_copy * v

    u = _cg_solve(_hess_vp, g_t, max_cg=300)
    w = _cg_solve(_hess_vp, np.ones(n), max_cg=300)
    w_sum = float(w.sum())

    if abs(w_sum) < 1e-15:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w

    dx = dx - np.mean(dx)
    return dx, A_vec


def _active_set_newton(a_init, deadline, max_outer=80):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol_list = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        if outer >= max_outer // 2:
            tol_list = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]

        best_improvement = 0
        best_a_new = None

        for tol_frac in tol_list:
            dx, A_vec = _newton_step(a, c, c_max, tol_frac)
            if dx is None:
                continue

            t = 1.0
            for ls in range(50):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < c_max - 1e-14 * t:
                    improvement = c_max - f_try
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_a_new = a_try.copy()
                    break
                t *= 0.5

        if best_a_new is None:
            break

        a = best_a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-15:
            break

    return best_a, best_sc


def _peak_equalization(a, deadline, max_iter=200):
    """
    Iterative peak equalization: compute weighted gradient to flatten peaks.
    Uses adaptive weights based on how far each peak is above the median.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        c_median = np.median(c)

        # Compute weights: higher weight for peaks further above median
        excess = np.maximum(c - c_median, 0)
        if excess.max() < 1e-15:
            break
        w = excess / excess.sum()

        # Compute gradient direction
        grad = 2.0 * _corr_a_w(a, w, n)
        g_t = grad - np.mean(grad)

        # Try multiple step sizes
        applied = False
        for eta in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
            a_try = a - eta * g_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                applied = True
                break

        if not applied:
            # Try even smaller steps
            for eta in [0.0001, 0.0002, 0.0005]:
                a_try = a - eta * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-15:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    applied = True
                    break

        if not applied:
            break

    return best_a, best_sc


def _lbfgs_continuation(a_init, deadline, min_time_remaining=60.0):
    """Aggressive L-BFGS with log-sum-exp continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    idx = min(100, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[idx]
    base_beta = max(gap * 0.2, max_c * 0.001)

    # Very aggressive: go down to max_c * 1e-9
    min_beta = max_c * 1e-9
    n_betas = 30
    log_betas = np.linspace(np.log(base_beta), np.log(min_beta), n_betas)
    betas = np.exp(log_betas)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.002 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 8))))

        try:
            def _obj(z_val):
                return _lsobj_factory(z_val, beta, n)

            def _grad(z_val):
                return _lsgrad_factory(z_val, beta, n)

            result = minimize(
                _obj, z, jac=_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 120, 'maxls': 500}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            if i % 5 == 0:
                print(f"    L-BFGS beta={beta:.2e} (iter {i}/{n_betas}): score={current_sc:.14f}")
        except Exception as ex:
            print(f"    L-BFGS beta={beta:.2e}: exception {ex}")

    return current_a


def _smooth_random_perturbation(n, sigma, rng, n_freq=20):
    """Generate a smooth random perturbation using random Fourier modes."""
    k = np.arange(1, n_freq + 1)
    coeffs = rng.standard_normal(n_freq)
    phases = rng.uniform(0, 2 * np.pi, n_freq)
    t = np.arange(n)
    pert = np.zeros(n)
    for i in range(n_freq):
        pert += coeffs[i] * np.cos(2 * np.pi * k[i] * t / n + phases[i])
    return sigma * pert


def _local_search(a, deadline, n_trials=500):
    """Local search with smooth random perturbations."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        sigma = 0.01 * (0.5 ** (trial / 100))
        pert = _smooth_random_perturbation(n, sigma, rng)
        z = np.log(np.maximum(a, 1e-30)) + pert
        a_try = _softmax(z)
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-15:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _refinement_loop(a, deadline, max_rounds=10):
    """Iterate between Newton, peak equalization, and L-BFGS."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Newton step
        if time.monotonic() < deadline - 30:
            as_dl = min(deadline - 30, time.monotonic() + 40)
            a_newton, sc_newton = _active_set_newton(best_a, as_dl, max_outer=30)
            if sc_newton < best_sc - 1e-15:
                best_sc = sc_newton
                best_a = a_newton.copy()
                print(f"    Refinement round {rnd}: Newton -> {best_sc:.14f}")

        # Peak equalization
        if time.monotonic() < deadline - 20:
            pe_dl = min(deadline - 20, time.monotonic() + 20)
            a_pe, sc_pe = _peak_equalization(best_a, pe_dl, max_iter=50)
            if sc_pe < best_sc - 1e-15:
                best_sc = sc_pe
                best_a = a_pe.copy()
                print(f"    Refinement round {rnd}: PeakEq -> {best_sc:.14f}")

        # Quick L-BFGS polish
        if time.monotonic() < deadline - 15:
            lb_dl = min(deadline - 15, time.monotonic() + 15)
            a_lb = _lbfgs_continuation(best_a, lb_dl, min_time_remaining=15.0)
            sc_lb = _score(a_lb)
            if sc_lb < best_sc - 1e-15:
                best_sc = sc_lb
                best_a = a_lb.copy()
                print(f"    Refinement round {rnd}: L-BFGS -> {best_sc:.14f}")

        if time.monotonic() > deadline - 10:
            break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    - Aggressive log-sum-exp L-BFGS continuation
    - Peak equalization
    - Active-set Newton
    - Multi-scale (n=8192, 16384, 32768)
    - Smooth random local search
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.14f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.14f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Aggressive L-BFGS on parent (n=~16384) ===
    print(f"\n[Phase 1] Aggressive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    lbfgs_dl = min(deadline - 800, time.monotonic() + 400)
    a_lbfgs = _lbfgs_continuation(parent, lbfgs_dl, min_time_remaining=800.0)
    update_best(a_lbfgs, _score(a_lbfgs), "lbfgs_main")

    # === Phase 2: Peak equalization ===
    print(f"\n[Phase 2] Peak equalization (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 600:
        pe_dl = min(deadline - 600, time.monotonic() + 100)
        a_pe, sc_pe = _peak_equalization(best_seq, pe_dl, max_iter=300)
        update_best(a_pe, sc_pe, "peak_eq")

    # === Phase 3: Active-set Newton ===
    print(f"\n[Phase 3] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        as_dl = min(deadline - 500, time.monotonic() + 100)
        a_as, sc_as = _active_set_newton(best_seq, as_dl, max_outer=100)
        update_best(a_as, sc_as, "newton")

    # === Phase 4: Refinement loop on current best ===
    print(f"\n[Phase 4] Refinement loop (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        ref_dl = min(deadline - 400, time.monotonic() + 150)
        a_ref, sc_ref = _refinement_loop(best_seq, ref_dl, max_rounds=10)
        update_best(a_ref, sc_ref, "refinement")

    # === Phase 5: Try n=32768 ===
    print(f"\n[Phase 5] Try n=32768 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        n_large = 32768
        a_large = _interpolate_seq(best_seq, n_large) if best_seq is not None else _make_profile(n_large, "uniform")
        sc_large_init = _score(a_large)
        print(f"  n={n_large} init score: {sc_large_init:.14f}")

        if sc_large_init < best_score + 0.01:
            lbfgs3_dl = min(deadline - 200, time.monotonic() + 100)
            a_large_opt = _lbfgs_continuation(a_large, lbfgs3_dl, min_time_remaining=200.0)
            sc_large = _score(a_large_opt)
            update_best(a_large_opt, sc_large, "n_32768_lbfgs")

            if time.monotonic() < deadline - 150:
                pe3_dl = min(deadline - 150, time.monotonic() + 40)
                a_large_pe, sc_large_pe = _peak_equalization(a_large_opt, pe3_dl, max_iter=50)
                update_best(a_large_pe, sc_large_pe, "n_32768_pe")

    # === Phase 6: Try n=8192 ===
    print(f"\n[Phase 6] Try n=8192 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        n_small = 8192
        a_small = _interpolate_seq(best_seq, n_small) if best_seq is not None else _make_profile(n_small, "uniform")
        sc_small_init = _score(a_small)
        print(f"  n={n_small} init score: {sc_small_init:.14f}")

        if sc_small_init < best_score + 0.01:
            lbfgs4_dl = min(deadline - 100, time.monotonic() + 60)
            a_small_opt = _lbfgs_continuation(a_small, lbfgs4_dl, min_time_remaining=100.0)
            sc_small = _score(a_small_opt)
            update_best(a_small_opt, sc_small, "n_8192_lbfgs")

    # === Phase 7: Local search ===
    print(f"\n[Phase 7] Local search (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        ls_dl = min(deadline - 80, time.monotonic() + 60)
        a_ls, sc_ls = _local_search(best_seq, ls_dl, n_trials=300)
        update_best(a_ls, sc_ls, "local_search")

    # === Phase 8: Final aggressive refinement ===
    print(f"\n[Phase 8] Final refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        # Final L-BFGS with very small beta
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()

        betas_final = [mb * 1e-6, mb * 1e-7, mb * 1e-8, mb * 1e-9, mb * 1e-10]
        for beta_f in betas_final:
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - 10 - time.monotonic()
            if remaining < 3:
                break
            max_iter_fin = min(3000, max(100, int(remaining / 0.05)))
            try:
                def _obj_fin(z_val):
                    return _lsobj_factory(z_val, beta_f, n)

                def _grad_fin(z_val):
                    return _lsgrad_factory(z_val, beta_f, n)

                result = minimize(
                    _obj_fin, z, jac=_grad_fin, method='L-BFGS-B',
                    options={'maxiter': max_iter_fin, 'ftol': 1e-19, 'gtol': 1e-16,
                             'maxcor': 150, 'maxls': 500}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                if sc_fin < best_score - 1e-16:
                    best_score = sc_fin
                    best_seq = a_fin.copy()
                    z = np.log(np.maximum(a_fin, 1e-30))
                    z = z - z.mean()
                    print(f"  Final L-BFGS beta={beta_f:.2e}: {sc_fin:.14f}")
            except Exception:
                pass

        # Final peak equalization
        if time.monotonic() < deadline - 8:
            pe_final_dl = deadline - 8
            a_pf, sc_pf = _peak_equalization(best_seq, pe_final_dl, max_iter=50)
            update_best(a_pf, sc_pf, "final_pe")

        # Final Newton
        if time.monotonic() < deadline - 5:
            n_final_dl = deadline - 5
            a_nf, sc_nf = _active_set_newton(best_seq, n_final_dl, max_outer=20)
            update_best(a_nf, sc_nf, "final_newton")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.14f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.14f}")
    return [float(x) for x in fallback]
