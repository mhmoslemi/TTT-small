`= 2 * conv[p, a_rev][n-1+m] - a[m] * p[2m]`

Now, the Hessian-vector product `H * v` where `H` is the Hessian of `sum_k p_k (a*a)[k]`:
`H[m,p] = sum_k p_k * (2*delta_{m+p,k} - delta_{m,p}*I(2m=k))`

So `(H*v)[m] = sum_p H[m,p] v[p] = 2 * sum_k p_k * v[k-m] - v[m] * p[2m]`
`= 2 * conv[p, v_rev][n-1+m] - v[m] * p[2m]`

Note that this is the same formula as the gradient, but with `a` replaced by `v`! So the Hessian-vector product is:
`H*v = 2 * conv[p, v_rev][n-1:2n-1] - v * p[0:2n-1:2]` (with appropriate padding)

Wait, that's interesting. The Hessian is actually a "convolution" operator (plus a diagonal correction). This means we can compute H*v in O(n log n) using FFT.

But more importantly, this means the Hessian has a specific structure: it's a "block"Toeplitz" or more precisely, it's a sum of anti-diagonal matrices. This is the structure that makes FFT-based computation possible.

3. Hessian-vector product: `H * v` (computed in O(n log n)) - only needed if we use a second-order method

For L-BFGS, we only need the objective and gradient. The Hessian-vector product is useful for Newton-type methods or for verifying convergence.

Multi-scale L-BFGS optimizer for minimum peak linear autocorrelation.
Uses log-sum-exp smoothing with adaptive temperature, FFT-based gradient
computation, and progressive upscaling from small to large sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    out_n = len(a) + len(b) - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _autocorr(a):
    return _fft_convolve(a, a)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _autocorr(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(x):
    x = x - x.max()
    e = np.exp(np.clip(x, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s


def _grad_objective(a, w):
    """Compute gradient of sum_k w[k] * (a*a)[k] w.r.t. a.
    
    a: array of length n (the sequence)
    w: array of length 2n-1 (weights on autocorrelation lags)
    Returns: array of length n
    """
    n = len(a)
    a_rev = a[::-1]
    conv = _fft_convolve(w, a_rev)
    grad = 2.0 * conv[n - 1:2 * n - 1].copy()
    # Diagonal correction
    diag_idx = 2 * np.arange(n)
    diag_idx = diag_idx[diag_idx < len(w)]
    diag_w = np.zeros(n)
    diag_w[diag_idx < n] = w[diag_idx[diag_idx < n]]
    # Wait, this is wrong. Let me fix.

    # Since 2m < 2n-1 for all m < n (because 2(n-1) = 2n-2 < 2n-1), all are valid
    grad -= a * w[2 * np.arange(n)]
    return grad

    """Create objective and gradient functions for L-BFGS in log-space."""
    
    def obj(z_local):
        a_local = _softmax(z_local)
        c = _autocorr(a_local)
        c_max = float(c.max())
        shifted = (c - c_max) / beta

        s_e = e_c.sum()

        return 2.0 * n * (c_max + beta * np.log(s_e))

        c = _autocorr(a_local)
        c_max = float(c.max())

        else:

        a_dot_g = float(np.dot(a_local, grad_a))
        grad_z = a_local * (grad_a - a_dot_g)
        return grad_z

            )

        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power2":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power15":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_log(z, sigma, rng):
    z_new = z + sigma * rng.standard_normal(len(z))
    z_new = z_new - z_new.mean()
    return z_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-scale optimizer: learns optimal shape at small n via fast L-BFGS,
    then progressively upsamples and refines at larger n. Uses adaptive
    log-sum-exp temperature and top-K mean objectives for final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, budget_s - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_a = None
    best_score = np.inf

    def update_best(a, label=""):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if s < 0.01:
            return
        sc = _score(a)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_score = float(sc)
            best_a = a.copy()
            print(f"  BEST: {best_score:.10f} (n={len(a)}) [{label}] "
                  f"t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}")
                update_best(parent, "parent")
    except Exception:
        pass

    # === Phase 1: Small-n shape search (80s) ===
    print(f"\n[Phase 1] Small-n optimization (target: best shape)")
    phase1_end = t_start + 80

    # Try several profiles at n=1024
    n_small = 1024
    best_z_small = None
    best_sc_small = np.inf

    for prof in ["tapered", "power15", "power2", "cos", "hann"]:
        if time.monotonic() > phase1_end - 10:
            break
        a_init = _make_profile(n_small, prof)
        z_init = _to_log(a_init)
        # Quick L-BFGS with moderate beta
        if parent is not None:
            # Also try resampling parent
            a_par = _resample(parent, n_small)
            z_par = _to_log(a_par)
            betas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
            z_out = _run_lbfgs(z_par, n_small, betas, phase1_end, min_time=5.0)
            a_out = _softmax(z_out)
            sc = _score(a_out)
            if sc < best_sc_small:
                best_sc_small = sc
                best_z_small = z_out.copy()
                print(f"    parent@{n_small}: {sc:.8f} [{prof}]")
        else:
            betas = [0.01, 0.005, 0.002, 0.001]
            z_out = _run_lbfgs(z_init, n_small, betas, phase1_end, min_time=5.0)
            a_out = _softmax(z_out)
            sc = _score(a_out)
            if sc < best_sc_small:
                best_sc_small = sc
                best_z_small = z_out.copy()
                print(f"    {prof}@{n_small}: {sc:.8f}")

    if best_z_small is None:
        best_z_small = _to_log(_make_profile(n_small, "tapered"))

    # === Phase 2: Progressive upscaling (900s total for phases 2-5) ===
    # n = 2048 -> 4096 -> 8192 -> 16384 -> 32768 -> 65536

    scales = [2048, 4096, 8192, 16384, 32768]
    if parent is not None and len(parent) >= 32768:
        scales = [4096, 8192, 16384, 32768]

    phase5_end = t_start + 1000  # Leave 770s for phases 5-6

    current_z = best_z_small
    current_n = n_small
    current_a = _softmax(best_z_small)
    update_best(current_a, f"scale_{current_n}")

    for n_target in scales:
        if time.monotonic() > phase5_end - 100:
            break
        print(f"\n[Phase 2] Upscaling to n={n_target}")

        # Upsample
        current_a = _resample(current_a, n_target)
        current_z = _to_log(current_a)
        current_n = n_target

        # Optimize with decreasing beta
        c = _autocorr(current_a)
        c_max = float(c.max())
        c_sorted = np.sort(c)[::-1]
        # Gap-based initial beta
        gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
        beta_start = max(gap * 0.5, c_max * 0.005)

        # Create beta schedule: geometric from beta_start to c_max * 1e-7
        n_betas = 8
        beta_end = c_max * 1e-7
        betas = beta_start * (beta_end / beta_start) ** (np.arange(n_betas) / (n_betas - 1))
        betas = np.maximum(betas, 1e-15)

        phase_end = min(phase5_end - 50, time.monotonic() + 180)
        current_z = _run_lbfgs(current_z, n_target, betas, phase_end, min_time=10.0)
        current_a = _softmax(current_z)
        sc = _score(current_a)
        print(f"    n={n_target}: score={sc:.10f}")
        update_best(current_a, f"scale_{n_target}")

        # Second pass with tighter betas if time allows
        if time.monotonic() < phase5_end - 100:
            c = _autocorr(current_a)
            c_max2 = float(c.max())
            betas2 = np.array([c_max2 * 1e-4, c_max2 * 5e-5, c_max2 * 2e-5,
                               c_max2 * 1e-5, c_max2 * 5e-6, c_max2 * 2e-6,
                               c_max2 * 1e-6, c_max2 * 5e-7])
            phase_end2 = min(phase5_end - 80, time.monotonic() + 80)
            current_z = _run_lbfgs(current_z, n_target, betas2, phase_end2, min_time=5.0)
            current_a = _softmax(current_z)
            sc = _score(current_a)
            print(f"    n={n_target} (refined): score={sc:.10f}")
            update_best(current_a, f"scale_{n_target}_r2")

    # === Phase 3: Try n=65536 (300s) ===
    print(f"\n[Phase 3] Try n=65536")
    n_large = 65536
    phase3_end = min(deadline - 200, time.monotonic() + 300)

    if time.monotonic() < phase3_end - 30:
        a_large = _resample(current_a, n_large)
        z_large = _to_log(a_large)

        c = _autocorr(a_large)
        c_max = float(c.max())
        beta_start = c_max * 0.003
        betas_large = [beta_start, beta_start * 0.3, beta_start * 0.1,
                       beta_start * 0.03, beta_start * 0.01, beta_start * 0.003,
                       c_max * 1e-5, c_max * 1e-6]
        z_large = _run_lbfgs(z_large, n_large, betas_large, phase3_end, min_time=20.0)
        a_large = _softmax(z_large)
        sc = _score(a_large)
        print(f"    n={n_large}: score={sc:.10f}")
        update_best(a_large, f"n_{n_large}")

        if sc < best_score + 0.001 and time.monotonic() < deadline - 150:
            # Refine at 65536
            betas_refine = [c_max * 1e-5, c_max * 5e-6, c_max * 2e-6,
                            c_max * 1e-6, c_max * 5e-7, c_max * 2e-7, c_max * 1e-7]
            z_large = _run_lbfgs(z_large, n_large, betas_refine,
                                  min(deadline - 150, phase3_end + 100), min_time=10.0)
            a_large = _softmax(z_large)
            sc = _score(a_large)
            print(f"    n={n_large} (refined): score={sc:.10f}")
            update_best(a_large, f"n_{n_large}_r2")

    # === Phase 4: Top-K polish at best n (200s) ===
    print(f"\n[Phase 4] Top-K polish")
    if best_a is not None and time.monotonic() < deadline - 50:
        n_best = len(best_a)
        z_best = _to_log(best_a)
        phase4_end = min(deadline - 40, time.monotonic() + 200)

        # Try different K values
        for K in [5, 3, 2, 1]:
            if time.monotonic() > phase4_end - 10:
                break
            z_new = _run_lbfgs_topk(z_best, n_best, K, phase4_end, min_time=5.0,
                                     max_rounds=3)
            a_new = _softmax(z_new)
            sc = _score(a_new)
            print(f"    K={K}: score={sc:.10f}")
            if sc < best_score:
                z_best = z_new.copy()
                update_best(a_new, f"topK_{K}")

    # === Phase 5: Random restarts at best n (80s) ===
    print(f"\n[Phase 5] Random restarts")
    if best_a is not None and time.monotonic() < deadline - 30:
        n_best = len(best_a)
        z_best = _to_log(best_a)
        phase5_end = min(deadline - 25, time.monotonic() + 80)

        for attempt in range(5):
            if time.monotonic() > phase5_end - 5:
                break
            z_pert = _perturb_log(z_best, 0.02, rng)
            c = _autocorr(_softmax(z_pert))
            c_max = float(c.max())
            betas = [c_max * 1e-4, c_max * 1e-5, c_max * 1e-6]
            z_out = _run_lbfgs(z_pert, n_best, betas, phase5_end, min_time=5.0)
            a_out = _softmax(z_out)
            sc = _score(a_out)
            if sc < best_score:
                z_best = z_out.copy()
                update_best(a_out, f"restart_{attempt}")

    # === Phase 6: Final ultra-tight L-BFGS (remaining time) ===
    print(f"\n[Phase 6] Final ultra-tight optimization")
    if best_a is not None and time.monotonic() < deadline - 10:
        n_best = len(best_a)
        z_best = _to_log(best_a)
        c = _autocorr(best_a)
        c_max = float(c.max())

        betas_final = [c_max * 5e-7, c_max * 2e-7, c_max * 1e-7,
                       c_max * 5e-8, c_max * 2e-8, c_max * 1e-8,
                       c_max * 5e-9, c_max * 2e-9]
        z_final = _run_lbfgs(z_best, n_best, betas_final, deadline - 5.0, min_time=3.0)
        a_final = _softmax(z_final)
        sc = _score(a_final)
        print(f"    Final: score={sc:.10f}, n={n_best}")
        update_best(a_final, "final")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_a is not None:
        final = np.clip(best_a, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(4096, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
