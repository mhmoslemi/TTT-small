import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):

    c = _full_convolve(a, a)

        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)

            if len(c_sorted) > 10:
                gap = c_sorted[0] - c_sorted[min(10, len(c_sorted)-1)]

    return best_a, best_sc

def _newton_step(a, c, c_max, tol_frac):
    """Newton step for the KKT conditions."""
    n = len(a)
    active = c >= (c_max - tol_frac * c_max)
    if not active.any():
        active[int(np.argmax(c))] = True
    m = int(active.sum())
    if m < 2:
        return None
    if m > 50 * n:
        return None
    
    A_vec = active.astype(np.float64) / m
    a_rev = a[::-1]
    out_len = n + m - 1  # Wait, A_vec has length 2n-1
    # Actually A_vec has the same length as c, which is 2n-1
    out_len = n + (2*n-1) - 1  # = 3n - 2
    N = _next_pow2(out_len)
    fa = np.fft.rfft(a_rev, N)
    fA = np.fft.rfft(A_vec, N)
    corr = np.fft.irfft(fa * fA, N)[:out_len]
    g = 2.0 * corr[n-1:2*n-1]
    g_t = g - np.mean(g)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-14:
        return None
    
    mu_reg = 1e-6 / m + 1e-10
    A_vec_copy = A_vec
    n_copy = n
    mu_copy = mu_reg
    
    def _hess_vp(v):
        out_len2 = n_copy + len(A_vec_copy) - 1
        N2 = _next_pow2(out_len2)
        fv = np.fft.rfft(v[::-1], N2)
        fA2 = np.fft.rfft(A_vec_copy, N2)
        res = np.fft.irfft(fv * fA2, N2)[:out_len2]
        return 2.0 * res[n_copy-1:2*n_copy-1] + mu_copy * v
    
    # CG solve
    x = np.zeros(n)
    r = g_t.copy()
    p = r.copy()
    rs = np.dot(r, r)
    for _ in range(100):
        Ap = _hess_vp(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < 1e-12:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    
    # Also solve for the constraint sum(dx) = 0
    ones = np.ones(n)
    w = np.zeros(n)
    r2 = ones - _hess_vp(np.zeros(n))
    p2 = r2.copy()
    rs2 = np.dot(r2, r2)
    for _ in range(100):
        Ap2 = _hess_vp(p2)
        pAp2 = np.dot(p2, Ap2)
        if pAp2 < 1e-30:
            break
        alpha2 = rs2 / pAp2
        w = w + alpha2 * p2
        r2 = r2 - alpha2 * Ap2
        rs2_new = np.dot(r2, r2)
        if np.sqrt(rs2_new) < 1e-12:
            break
        p2 = r2 + (rs2_new / rs2) * p2
        rs2 = rs2_new
    
    w_sum = float(w.sum())
    if abs(w_sum) < 1e-15:
        dx = -x
    else:
        lam = float(x.sum()) / w_sum
        dx = -x + lam * w
    
    dx = dx - np.mean(dx)
    return dx

def _newton_polish(a_init, deadline, max_outer=50):
    """Newton-based polishing with active set."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for outer in range(max_outer):
        if time.monotonic() > deadline - 5:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        best_improvement = 0
        best_a_new = None
        
        for tol_frac in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            dx = _newton_step(a, c, c_max, tol_frac)
            if dx is None:
                continue
            
            t = 1.0
            for ls in range(30):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-15:
                    if best_sc - sc_try > best_improvement:
                        best_improvement = best_sc - sc_try
                        best_a_new = a_try.copy()
                    break
                t *= 0.5
        
        if best_a_new is None:
            break
        
        a = best_a_new
        best_sc = _score(a)
        best_a = a.copy()
    
    return best_a, best_sc

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20*(t-0.2)**2) + np.exp(-20*(t-0.8)**2)
        return a / a.sum()
    elif kind == "random_smooth":
        rng = np.random.default_rng(12345)
        a = rng.random(n)
        # Smooth it
        kernel_size = max(3, n // 100)
        if kernel_size % 2 == 0:
            kernel_size += 1
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    return np.ones(n) / n

def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s
