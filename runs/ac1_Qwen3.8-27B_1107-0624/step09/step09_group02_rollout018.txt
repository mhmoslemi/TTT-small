def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive temperature based on convolution flatness."""
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    current_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    best_a = a.copy()
    best_sc = current_sc

    # Adaptive temperature: based on flatness of convolution
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    mean_c = float(c_cur.mean())
    flatness = mean_c / max_c  # close to 1 means flat
    base_beta = max_c * (1.0 - flatness) * 2.0
    base_beta = max(base_beta, max_c * 0.001)
    base_beta = min(base_beta, max_c * 0.1)

    ratios = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005,
              0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005,
              0.00002, 0.00001, 0.000005, 0.000002, 0.000001,
              0.0000005, 0.0000002, 0.0000001, 0.00000005, 0.00000002,
              0.00000001, 0.000000005, 0.000000002, 0.000000001]

    for i, ratio in enumerate(ratios):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        beta = base_beta * ratio
        est_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 5))))

        try:
            def obj_func(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    return 2.0 * n * c_max_local
                return 2.0 * n * (c_max_local + beta * np.log(s_e))

            def grad_func(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    w = np.zeros_like(c_local)
                    w[int(np.argmax(c_local))] = 1.0
                else:
                    w = e / s_e
                # Gradient of smoothed max w.r.t. a
                grad_a = _grad_smoothed_fft(a_local, w, n)
                # Chain rule: d/dz = a * (d/da - mean(d/da))
                # (since softmax: da_i/dz_j = a_i(δ_ij - a_j))
                return a_local * (grad_a - np.dot(a_local, grad_a))

            result = minimize(
                obj_func, z,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                # Update base_beta
                c_new = _full_convolve(a_new, a_new)

                mean_c_new = float(c_new.mean())
                flatness_new = mean_c_new / max_c_new
                base_beta = max_c_new * (1.0 - flatness_new) * 2.0
                base_beta = max(base_beta, max_c_new * 0.001)
                base_beta = min(base_beta, max_c_new * 0.1)
        except Exception:

    return best_a


def _peak_shaving(a_init, deadline, min_time_remaining=20.0):
    """Targeted peak shaving: iteratively reduce the top convolution values."""
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    no_improve = 0
    max_no_improve = 30

    for it in range(2000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        if no_improve > max_no_improve:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find top peaks
        n_peaks = min(20, len(c))
        peaks = np.argsort(c)[::-1][:n_peaks]
        # Only include peaks that are actually at the max level
        peak_threshold = c_max * 0.999
        active_peaks = peaks[c[peaks] >= peak_threshold]
        if len(active_peaks) == 0:
            active_peaks = peaks[:5]
        
        # Build weight vector
        w = np.zeros(2 * n - 1)
        w[active_peaks] = 1.0 / len(active_peaks)
        
        # Compute gradient
        grad = _grad_smoothed_fft(a, w, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        step = 0.1 / (gnorm + 1e-10)
        improved = False
        for ls in range(20):
            a_new = _proj_simplex(a - step * grad)
            c_new = _full_convolve(a_new, a_new)
            if float(c_new.max()) < c_max * (1 - 1e-8):
                a = a_new
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                improved = True
                break
            step *= 0.5
            if step < 1e-14:
                break
        if not improved:
            no_improve += 1

    return best_a, best_sc


def _run_primal_dual(a_init, deadline, min_time_remaining=30.0):

    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()

    a = a / a.sum()
    best_a = a.copy()

    momentum = np.zeros(n)

    for outer in range(300):

            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Adaptive temperature for dual
        sorted_c = np.sort(c)[::-1]
        gap = sorted_c[0] - sorted_c[min(20, len(sorted_c) - 1)]
        beta = max(gap * 0.05, c_max * 0.002)
        
        # Dual: p = softmax(c / beta)
        p = _softmax(c / beta)
        
        # Primal: projected gradient with momentum
        grad = _grad_smoothed_fft(a, p, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        momentum = 0.9 * momentum + 0.1 * grad
        lr = 0.05 / (np.linalg.norm(momentum) + 1e-10)

        if outer % 5 == 0:

def _resample(a, n_new):
    """Resample sequence to new length using interpolation."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * t
        return a / a.sum()
    elif kind == "exp":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t)
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _spectral_flatten(a_init, n_iters=100):
    """Spectral domain flattening: alternate between time and frequency domain projections."""
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    
    N = _next_pow2(2 * n - 1)
    
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(n_iters):
        # FFT
        A = np.fft.rfft(a, N)
        
        # Flatten magnitude: target |A|^2 = constant
        # Choose constant so that sum(a) = 1 is preserved (approximately)
        # sum(a) = A[0]/N (real part), so A[0] should remain ~ N
        # For other frequencies, set |A[k]| = target
        mag = np.abs(A)
        # Target: spread energy evenly
        target_mag = np.full(N // 2 + 1, A[0].real / np.sqrt(N))
        target_mag[0] = A[0].real  # Keep DC component
        
        # Interpolate: blend current and target
        alpha = 0.3
        new_mag = (1 - alpha) * mag + alpha * target_mag
        # Keep DC
        new_mag[0] = mag[0]
        
        # Apply new magnitude, keep phase
        phase = A / np.maximum(mag, 1e-30)
        A_new = new_mag * phase
        
        # Inverse FFT
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project: clip and normalize
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-scale optimizer with adaptive L-BFGS, spectral flattening,
    peak shaving, and primal-dual refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Try to load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 32768
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Multi-scale warm start (200s) ===
    print(f"\n[Phase 1] Multi-scale warm start (t={time.monotonic() - t_start:.1f}s)")
    phase1_deadline = t_start + 200
    scales = [256, 1024, 4096, 16384]
    
    # Start from parent at smallest scale
    a_small = _resample(parent, scales[0])
    for scale in scales:
        if time.monotonic() > phase1_deadline - 20:
            break
        print(f"  Optimizing at n={scale}")
        a_opt = _run_lbfgs_adaptive(a_small, phase1_deadline - 10, min_time_remaining=5.0)
        sc_opt = _score(a_opt)
        print(f"    n={scale} score={sc_opt:.10f}")
        # Upscale to next level
        if scale < scales[-1]:
            next_scale = scales[scales.index(scale) + 1]
            a_small = _resample(a_opt, next_scale)
        # Also update best if this is at the target scale
        if scale == 16384:
            a_16k = a_opt
            # Try to get to 32768
            if time.monotonic() < phase1_deadline - 10:
                a_32k = _resample(a_16k, 32768)
                a_32k_opt = _run_lbfgs_adaptive(a_32k, phase1_deadline - 5, min_time_remaining=5.0)
                update_best(a_32k_opt, _score(a_32k_opt), "multiscale_32k")

    # === Phase 2: Main L-BFGS on best sequence (500s) ===
    print(f"\n[Phase 2] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    phase2_deadline = t_start + 700
    if best_seq is not None:
        a_opt = _run_lbfgs_adaptive(best_seq, phase2_deadline, min_time_remaining=30.0)
        update_best(a_opt, _score(a_opt), "lbfgs_main")

    # === Phase 3: Peak shaving (200s) ===
    print(f"\n[Phase 3] Peak shaving (t={time.monotonic() - t_start:.1f}s)")
    phase3_deadline = min(deadline - 500, phase2_deadline + 200)
    if best_seq is not None and time.monotonic() < phase3_deadline:
        a_ps, sc_ps = _peak_shaving(best_seq, phase3_deadline, min_time_remaining=20.0)
        update_best(a_ps, sc_ps, "peak_shave_1")

    # === Phase 4: Primal-dual (200s) ===
    print(f"\n[Phase 4] Primal-dual (t={time.monotonic() - t_start:.1f}s)")
    phase4_deadline = min(deadline - 300, phase3_deadline + 200)
    if best_seq is not None and time.monotonic() < phase4_deadline:
        a_pd, sc_pd = _run_primal_dual(best_seq, phase4_deadline, min_time_remaining=20.0)
        update_best(a_pd, sc_pd, "primal_dual")

    # === Phase 5: L-BFGS round 2 (200s) ===
    print(f"\n[Phase 5] L-BFGS Round 2 (t={time.monotonic() - t_start:.1f}s)")
    phase5_deadline = min(deadline - 200, phase4_deadline + 200)
    if best_seq is not None and time.monotonic() < phase5_deadline:
        a_opt2 = _run_lbfgs_adaptive(best_seq, phase5_deadline, min_time_remaining=20.0)
        update_best(a_opt2, _score(a_opt2), "lbfgs_r2")

    # === Phase 6: Peak shaving round 2 (150s) ===
    print(f"\n[Phase 6] Peak shaving R2 (t={time.monotonic() - t_start:.1f}s)")
    phase6_deadline = min(deadline - 150, phase5_deadline + 150)
    if best_seq is not None and time.monotonic() < phase6_deadline:
        a_ps2, sc_ps2 = _peak_shaving(best_seq, phase6_deadline, min_time_remaining=15.0)
        update_best(a_ps2, sc_ps2, "peak_shave_2")

    # === Phase 7: Try larger n (150s) ===
    print(f"\n[Phase 7] Try larger n (t={time.monotonic() - t_start:.1f}s)")
    phase7_deadline = min(deadline - 100, phase6_deadline + 150)
    n_current = len(best_seq) if best_seq is not None else 32768
    if n_current < 65536 and time.monotonic() < phase7_deadline - 30:
        a_big = _resample(best_seq, 65536)
        a_big_opt = _run_lbfgs_adaptive(a_big, phase7_deadline, min_time_remaining=20.0)
        update_best(a_big_opt, _score(a_big_opt), "n_65536")
        if best_seq is not None and len(best_seq) == 65536:
            a_ps3, sc_ps3 = _peak_shaving(best_seq, deadline - 30, min_time_remaining=10.0)
            update_best(a_ps3, sc_ps3, "n_65536_ps")

    # === Phase 8: Spectral flattening + refinement (100s) ===
    print(f"\n[Phase 8] Spectral + final (t={time.monotonic() - t_start:.1f}s)")
    phase8_deadline = deadline - 20
    if best_seq is not None and time.monotonic() < phase8_deadline - 10:
        a_spec, sc_spec = _spectral_flatten(best_seq, n_iters=50)
        update_best(a_spec, sc_spec, "spectral")
        if sc_spec < best_score * 1.001:  # if comparable
            a_final = _run_lbfgs_adaptive(a_spec, phase8_deadline, min_time_remaining=10.0)
            update_best(a_final, _score(a_final), "spectral_refined")

    # === Phase 9: Final peak shaving (remaining time) ===
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _peak_shaving(best_seq, deadline - 3, min_time_remaining=3.0)
        update_best(a_fin, sc_fin, "final_ps")

    # === Phase 10: One more L-BFGS with very tight beta ===
    if best_seq is not None and time.monotonic() < deadline - 8:
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        beta_fin = mb * 1e-6
        remaining = deadline - 8 - time.monotonic()
        if remaining > 3:
            max_iter = min(2000, max(50, int(remaining / 0.005)))
            try:
                def final_obj(z_local):
                    a_local = _softmax(z_local)
                    c_local = _full_convolve(a_local, a_local)
                    c_max_local = float(c_local.max())
                    shifted = (c_local - c_max_local) / beta_fin
                    e = np.exp(np.clip(shifted, -700, 0))
                    s_e = e.sum()
                    if s_e < 1e-300:
                        return 2.0 * n * c_max_local
                    return 2.0 * n * (c_max_local + beta_fin * np.log(s_e))

                def final_grad(z_local):
                    a_local = _softmax(z_local)
                    c_local = _full_convolve(a_local, a_local)
                    c_max_local = float(c_local.max())
                    shifted = (c_local - c_max_local) / beta_fin
                    e = np.exp(np.clip(shifted, -700, 0))
                    s_e = e.sum()
                    if s_e < 1e-300:
                        w = np.zeros_like(c_local)
                        w[int(np.argmax(c_local))] = 1.0
                    else:
                        w = e / s_e
                    grad_a = _grad_smoothed_fft(a_local, w, n)
                    return a_local * (grad_a - np.dot(a_local, grad_a))

                result = minimize(
                    final_obj, z,
                    jac=final_grad,
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                             'maxcor': 50}
                )
                a_flb = _softmax(result.x)
                update_best(a_flb, _score(a_flb), "final_tight")
            except Exception:
                pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(32768, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(32768, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
