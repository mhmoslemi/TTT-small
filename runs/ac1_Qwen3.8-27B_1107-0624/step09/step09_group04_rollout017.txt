"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses adaptive smoothed-max L-BFGS, direct active-set subgradient with adaptive
step size, diverse perturbation restarts, and multi-n exploration.
Key improvements: adaptive beta based on peak gap, curvature-aware step sizing,
and exploitation of near-optimal active set structure.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _adaptive_lbfgs(a_init, deadline, min_remaining=30.0, beta_scale=1.0, max_total_time=None):
    """L-BFGS with adaptive beta schedule based on peak structure."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.0005:
        gap = max_c * 0.0005
    base_beta = max(gap, max_c * 0.003) * beta_scale

    # Adaptive beta: start from current gap, decrease geometrically
    num_betas = 12
    ratio = 0.5
    betas = [base_beta * (ratio ** i) for i in range(num_betas)]
    # Ensure minimum beta
    min_beta = max_c * 0.00005
    betas = [max(b, min_beta) for b in betas]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    t_func_start = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break

        # More iterations for smaller beta (finer optimization)
        if beta < base_beta * 0.01:
            max_iter = 8000
        elif beta < base_beta * 0.1:
            max_iter = 5000
        else:
            max_iter = 3000

        est_tpi = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(max_iter, max(30, int(remaining / (est_tpi * 10))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 60,
                    'maxls': 40,
                    'eps': 1e-14,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_optimize(a_init, deadline, min_remaining=20.0):
    """Direct active-set subgradient with curvature-aware step sizing.
    Uses the exact subgradient of the max and adapts step size based on
    observed curvature ( improvement per unit step^2 )."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(best_a)

    K = 5
    max_K = min(200, n)
    lr = 0.1
    no_improve = 0
    progress_count = 0

    # Track curvature for adaptive step size
    prev_sc = best_sc
    prev_lr = lr
    curvature = 1.0  # estimated reduction per lr^2

    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            progress_count += 1

        # Adaptive K: grow when making progress, shrink when stuck
        if progress_count > 0 and no_improve == 0 and K < max_K:
            K = min(max_K, K + 2)
        elif no_improve > 50 and K > 3:
            K = max(3, K // 2)

        # Get active set
        top_k_idx = np.argsort(c)[::-1][:K]

        # Compute subgradient (average of top-K peak gradients)
        w_S = np.zeros(2 * n - 1)
        w_S[top_k_idx] = 1.0 / K
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)
        grad = 2.0 * h[n - 1:2 * n - 1]

        # Project to tangent space of simplex (remove mean)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Adaptive step size based on curvature
        # We want: lr * gnorm ≈ fraction of c_max reduction
        # Step: a_new = a - lr * grad / gnorm (unit step in gradient direction)
        # Expected reduction ≈ lr * (grad · subgrad_level) ≈ lr * gnorm * c_max * some_factor
        eta = lr / (gnorm * c_max + 1e-30)
        eta = min(eta, 0.5)

        # Line search (backtracking)
        alpha = eta
        improved = False
        best_try = a
        best_try_sc = current_sc
        for _ in range(30):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())

            if sc_try < current_sc - 1e-14:
                # Check curvature
                if alpha > 1e-20 and current_sc - sc_try > 1e-16:
                    curv_est = (current_sc - sc_try) / (alpha * gnorm) ** 2
                    curvature = 0.7 * curvature + 0.3 * max(curv_est, 1e-10)
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            lr *= 0.7
            if no_improve > 100:
                # Restart from best with fresh parameters
                a = best_a.copy()
                lr = 0.1
                K = max(3, K // 3)
                no_improve = 0
                if no_improve > 300:
                    break
        else:
            no_improve = 0
            lr = min(0.2, lr * 1.02)

    return best_a, best_sc


def _detailed_active_set(a_init, deadline, min_remaining=10.0):
    """More careful active-set optimization using pairwise peak equalization.
    For pairs of near-equal peaks, compute the direction that reduces both."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(best_a)

    no_improve = 0

            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        threshold = c_max * 0.999
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        if len(active) > 50:
            active = np.argsort(c)[::-1][:50]

        # Compute weighted gradient
        K = len(active)
        w_S = np.zeros(2 * n - 1)
        w_S[active] = 1.0 / K
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)
        grad = 2.0 * h[n - 1:2 * n - 1]
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        eta = 0.01 / (gnorm + 1e-15)

        if s_try > 1e-15:

            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-16:
                a = a_try
            else:
                no_improve += 1
                if no_improve > 200:

        else:

    return best_a, best_sc


def _perturb(a, rng, mode=0, strength=0.05):
    """Apply different types of perturbations to break symmetry."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    if mode == 0:
        # Linear ramp + noise
        t = np.arange(n) / n
        pert = 0.4 * (t - 0.5) + 0.6 * rng.standard_normal(n)
    elif mode == 1:
        # Sine perturbation (low freq)
        t = np.arange(n) / n
        freq = rng.integers(2, 8)
        phase = rng.uniform(0, 2 * np.pi)
        pert = np.sin(2 * np.pi * freq * t + phase) + 0.3 * rng.standard_normal(n)
    elif mode == 2:
        # Sine perturbation (high freq)
        t = np.arange(n) / n
        freq = rng.integers(10, 50)
        phase = rng.uniform(0, 2 * np.pi)
        pert = np.sin(2 * np.pi * freq * t + phase) + 0.3 * rng.standard_normal(n)
    elif mode == 3:
        # Quadratic asymmetry
        t = np.arange(n) / n
        pert = 0.5 * (t - 0.5) ** 2 - 0.5 * (t - 0.5) + 0.5 * rng.standard_normal(n)
    elif mode == 4:
        # Pure random
        pert = rng.standard_normal(n)
    else:
        # Bump perturbation
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.05, 0.3)
        t = np.arange(n) / n
        pert = np.exp(-((t - center) / width) ** 2) + 0.3 * rng.standard_normal(n)

    pert = pert - pert.mean()
    pert = pert / (np.linalg.norm(pert) + 1e-15)
    z = z + strength * pert
    return _softmax(z)


def _make_b_spline_profile(n, order=4):
    """Create a B-spline-like profile of given order."""
    # Start with uniform and convolve (order-1) times
    a = np.ones(n) / n
    for _ in range(min(order, n - 1)):
        a = _full_convolve(a, np.ones(2) / 2)
        # Resample back to length n
        if len(a) != n:
            x_old = np.linspace(0, 1, len(a))
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, a)
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
    return a


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
