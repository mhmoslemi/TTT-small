Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_soft_structured(a, beta, n):
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft_structured(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)


class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)

    """L-BFGS with adaptive beta schedule - fewer levels, more iterations."""

    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < c0_max * 0.002:
        gap = c0_max * 0.002

    # Fewer, more strategically placed beta levels

    betas = gap * multipliers

    t_start = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:

def _active_set_refine_v2(a_init, deadline, min_time_remaining=30.0):
    """Improved KKT-inspired active-set refinement with momentum."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    no_improve = 0
    max_no_improve = 200
    tol_frac = 0.01
    min_tol = 0.0003
    momentum = np.zeros(n)
    beta_mom = 0.7

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.5)
            if tol_frac <= min_tol * 1.1:
                break
            no_improve = 0

        # Identify active set (peaks near max)
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Compute average gradient from active set
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]

        # Identify support of a
        support = a > 1e-10 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        # KKT: grad_avg should be constant on support
        lam = float(np.mean(grad_avg[support]))
        residual = grad_avg - lam

        # Momentum update
        momentum = beta_mom * momentum - residual[support]

        # Step size with backtracking
        base_eta = 0.03
        eta = base_eta
        found_improve = False
        for trial in range(6):
            trial_eta = eta * (0.5 ** trial)
            a_new = a.copy()
            a_new[support] -= trial_eta * momentum
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-13:
                a = a_new
                found_improve = True
                break
            eta = trial_eta

        if not found_improve:
            # Try without momentum
            for trial_eta in [0.02, 0.01, 0.005, 0.002, 0.001]:
                a_try = a.copy()
                a_try[support] -= trial_eta * residual[support]
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-13:
                    a = a_try
                    found_improve = True
                    momentum = np.zeros(n)
                    break

        if not found_improve:
            tol_frac = max(min_tol, tol_frac * 0.9)
            # Decay momentum
            momentum *= 0.5

        # Gradually tighten
        if found_improve:
            tol_frac = max(min_tol, tol_frac * 0.98)
            beta_mom = min(0.9, beta_mom + 0.005)

    return best_a, best_sc


def _nesterov_subgradient(a_init, deadline, min_time_remaining=30.0):
    """Nesterov-accelerated subgradient with Polyak step size."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Lower bound on max(conv(a,a)) given sum(a)=1
    lb_max = 1.0 / (2.0 * n - 1.0)
    lb_score = 2.0 * n * lb_max  # = 2n/(2n-1)

    t = 0
    y = a.copy()  # Nesterov's lookahead point
    a_prev = a.copy()
    step_multiplier = 1.0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(y, y)
        c_max = float(c.max())
        current_max = c_max  # max at lookahead point

        # Compute subgradient at y
        # Use a small tolerance for active set
        tol = 1e-4 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        y_rev = y[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, y_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        # Project gradient onto the tangent space of the simplex
        grad_proj = grad - np.mean(grad)
        g_norm_sq = float(np.dot(grad_proj, grad_proj))
        if g_norm_sq < 1e-30:
            break

        # Polyak step size
        current_score = 2.0 * n * c_max
        gap = current_score - lb_score
        if gap < 1e-12:
            break
        step = step_multiplier * (gap / (4.0 * n * n * g_norm_sq))
        step = min(step, 0.05)

        # Nesterov update
        t_new = t + 1
        theta = (t - 1.0) / (t_new + 2.0) if t >= 1 else 0.0
        y_new = a - step * grad_proj
        y_new = np.maximum(y_new, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-10:
            break
        y_new = y_new / s_new

        # Update a using Nesterov interpolation
        if t >= 1:
            a_new = y_new + theta * (y_new - a)
        else:
            a_new = y_new.copy()
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-10:
            break
        a_new = a_new / s_a

        a_prev = a
        a = a_new
        y = y_new
        t = t_new

        # Track best
        sc_a = _score(a)
        if sc_a < best_sc - 1e-12:
            best_sc = sc_a
            best_a = a.copy()

        # Decay step multiplier slowly
        if t % 200 == 0:
            step_multiplier *= 0.98

    return best_a, best_sc


def _make_profile(n, kind="uniform"):

    elif kind == "raised_cosine":

        a = 1.0 - 0.5 * (2 * t - 1) ** 2

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

    print(f"\n[P1] L-BFGS main (n={n_main}, t={time.monotonic()-t_start_global:.0f}s)")
    p1_end = t_start_global + min(500, (budget_s - 10) * 0.28)
    a1, sc1 = _run_lbfgs_improved(best_seq, p1_end, min_time_remaining=200, verbose=True)
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  P1 result: {best_score:.10f}")

    # === Phase 2: Active-set KKT refinement ===
    print(f"\n[P2] Active-set refine v2 (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p2_end = min(deadline - 400, time.monotonic() + 300)
        a2, sc2 = _active_set_refine_v2(best_seq, p2_end, min_time_remaining=50)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(f"  P2 result: {best_score:.10f}")

    # === Phase 3: L-BFGS round 2 (tighter) ===
    print(f"\n[P3] L-BFGS round 2 (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p3_end = min(deadline - 300, time.monotonic() + 350)
        a3, sc3 = _run_lbfgs_improved(best_seq, p3_end, min_time_remaining=100, verbose=True)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3 result: {best_score:.10f}")

    # === Phase 4: Nesterov subgradient ===
    print(f"\n[P4] Nesterov subgradient (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p4_end = min(deadline - 200, time.monotonic() + 200)
        a4, sc4 = _nesterov_subgradient(best_seq, p4_end, min_time_remaining=50)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4 result: {best_score:.10f}")

    # === Phase 5: L-BFGS after Nesterov ===
    if time.monotonic() + 20 < deadline:
        p5_end = min(deadline - 150, time.monotonic() + 120)
        a5, sc5 = _run_lbfgs_improved(best_seq, p5_end, min_time_remaining=50, verbose=False)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  P5 result: {best_score:.10f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n exploration (t={time.monotonic()-t_start_global:.0f}s)")
    n_options = [12288, 20480, 24576]
    n_options = [x for x in n_options if abs(x - n_main) > 1000]
    for n_try in n_options:
        if time.monotonic() + 100 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if time.monotonic() + 60 > deadline:
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
            continue
        p6_end = min(deadline - 80, time.monotonic() + 120)
        a6, sc6 = _run_lbfgs_improved(a_r, p6_end, min_time_remaining=30, verbose=False)
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            n_main = n_try
            print(f"  n={n_try}: {best_score:.10f}")

    # === Phase 7: Perturbation restarts ===
    print(f"\n[P7] Perturbation restarts (t={time.monotonic()-t_start_global:.0f}s)")
    for pert_scale in [0.005, 0.002, 0.001]:
        if time.monotonic() + 60 > deadline:
            break
        p7_end = min(deadline - 20, time.monotonic() + 80)
        a7, sc7 = _perturb_optimize(best_seq, p7_end, min_time_remaining=20, pert_scale=pert_scale)
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()
            print(f"  Perturb {pert_scale}: {best_score:.10f}")

    # === Phase 8: Final polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 15 < deadline:
        p8_end = min(deadline - 5, time.monotonic() + 100)
        a8, sc8 = _run_lbfgs_improved(best_seq, p8_end, min_time_remaining=5, verbose=False)
        if sc8 < best_score:
            best_score = sc8
            best_seq = a8.copy()

        if time.monotonic() + 10 < deadline:
            p8b_end = min(deadline - 3, time.monotonic() + 50)
            a8b, sc8b = _active_set_refine_v2(best_seq, p8b_end, min_time_remaining=3)
            if sc8b < best_score:
                best_score = sc8b
                best_seq = a8b.copy()

    # Final
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start_global:.1f}s")
    return [float(x) for x in final]
