import numpy as np

from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_normalized(a):
    """Score assuming a is already normalized to sum=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)

def _flatten_spectrum(a, n_iter, relax=0.3, target_power=None):
    """Gerchberg-Saxton style: flatten |FFT(a)|^2 while maintaining a >= 0."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    a_work = a.copy()
    for i in range(n_iter):
        A = np.fft.fft(a_work, N)
        power = np.abs(A) ** 2
        if target_power is None:
            target_power = np.ones(N) * power.sum() / N
        # Blend power spectrum toward flat
        new_power = (1 - relax) * power + relax * target_power
        # Recover A with new amplitude, same phase
        phase = np.angle(A)
        A_new = np.sqrt(new_power) * np.exp(1j * phase)
        # Ensure A is conjugate symmetric (for real a)
        A_new[0] = A_new[0].real
        for k in range(1, N // 2 + 1):
            A_new[N - k] = np.conj(A_new[k])
        a_new = np.fft.ifft(A_new).real[:n]
        # Project to non-negative, normalize
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_work = a_new / s
    return a_work

def _peak_reduction_freq(a, n_iter, n_peaks=5, relax=0.1):
    """Reduce peaks in convolution by modifying frequency domain."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    a_work = a.copy()
    for i in range(n_iter):
        c = _full_convolve(a_work, a_work)
        c_max = c.max()
        # Find top peaks
        peak_indices = np.argsort(c)[::-1][:n_peaks]
        A = np.fft.fft(a_work, N)
        # For each peak, compute the "interference pattern" in freq domain
        # c[k] = (1/N) * sum_m A[m]^2 * exp(2*pi*i*m*k/N)
        # To reduce c[k*], reduce |A[m]| where A[m]^2 * exp(2*pi*i*m*k*/N) 
        # has positive real part
        total_notch = np.zeros(N)
        for k_star in peak_indices:
            phase_factor = np.exp(2j * np.pi * np.arange(N) * k_star / N)
            contrib = np.real(A ** 2 * phase_factor)
            # Only notch where contribution is positive
            total_notch += np.maximum(contrib, 0)
        # Apply notch
        notch = 1.0 - relax * total_notch / (total_notch.max() + 1e-30)
        A_mod = A * np.sqrt(np.maximum(notch, 0.1))
        # Ensure conjugate symmetry
        A_mod[0] = A_mod[0].real
        for k in range(1, N // 2 + 1):
            A_mod[N - k] = np.conj(A_mod[k])
        a_new = np.fft.ifft(A_mod).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_work = a
            break
        a_work = a_new / s
    return a_work

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

class _LbfgsProb:

    def __call__(self, z):

Let me also think about the beta schedule for L-BFGS. The smoothed objective is:
f_beta(a) = 2n * (max(c) + beta * log(sum(exp((c - max(c))/beta))))

As beta → 0, this approaches the true max. For large beta, it's a softer approximation. We want to start with large beta (easy to optimize) and decrease to small beta (more accurate).

The previous code used a geometric decay of beta. I'll use a similar approach but with more steps and smaller final beta.

One more idea: **use the analytical structure of the problem**. The convolution at lag k is:
c[k] = sum_j a[j] * a[k-j]

For the peak to be minimized, we want all c[k] to be approximately equal to 1/(2n-1) (the mean). The "error" e[k] = c[k] - 1/(2n-1) should be small.

In the frequency domain, the mean of c is |A[0]|^2/N = 1/N (for normalized a, the DC component A[0] = sum(a) = 1, so |A[0]|^2/N = 1/N, and the mean of c is 1/(2n-1) ≈ 1/N for N ≈ 2n).

The AC components of c are determined by the AC components of A^2. To minimize the peak of c, we want to minimize the L-infinity norm of the AC part, which is equivalent to making the AC part of A^2 as "flat" as possible in frequency domain (i.e., the amplitude of A^2 should be as flat as possible for k ≠ 0).

This is the key insight: **flatten |A[k]| for k ≠ 0**.

If |A[k]| = constant for all k ≠ 0, then A^2 has |A[k]|^2 = constant for k ≠ 0, and the IFFT of A^2 has a constant value (plus the DC spike). This would give a nearly flat autocorrelation!

The constraint is that a = IFFT(A) must be non-negative. This is the hard constraint.

So the algorithm is:
1. Start with a (non-negative, normalized).
2. Compute A = FFT(a).
3. Modify |A[k]| for k ≠ 0 to be as flat as possible (while keeping A[0] = 1).
4. Compute a' = IFFT(A_modified).
5. Project a' to non-negative, normalize.
6. Repeat.

This is essentially the "flat spectrum" design, well-known in antenna theory and signal processing.

The projection step (step 5) is what makes this hard. If a' has negative values, we clip them to 0, which changes the spectrum and undoes some of the flattening.

To make this work better, we can use a **soft projection**: instead of hard clipping, use a[clip] = max(a', epsilon) for small epsilon > 0, or use a smooth reparameterization like a = softmax(z).

Let me implement this with the softmax reparameterization:
- Work with z (logit space), a = softmax(z).

"""
Hybrid optimizer: frequency-domain spectrum flattening + aggressive L-BFGS
with adaptive beta + multi-n exploration + peak-targeted refinement.
Key novel approach: Gerchberg-Saxton style iterative spectrum flattening
to reduce autocorrelation peaks, combined with smoothed L-BFGS for polish.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score for already-normalized (sum=1) sequence."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _flatten_spectrum(a, n_iter, relax_start=0.15, relax_end=0.02, seed=0):
    """
    Gerchberg-Saxton style: iteratively flatten |FFT(a)| to reduce
    autocorrelation peaks. Uses progressive relaxation.
    """
    n = len(a)
    N = _next_pow2(2 * n - 1)
    rng = np.random.default_rng(seed)
    a_work = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a_work)

    for i in range(n_iter):
        # Progressive relaxation
        t_frac = i / max(1, n_iter)
        relax = relax_start + (relax_end - relax_start) * t_frac

        A = np.fft.fft(a_work, N)
        power = np.abs(A) ** 2

        # Target: flat power spectrum (excluding DC)
        ac_power = power.copy()
        ac_power[0] = 0
        if N > 1:
            ac_power[N // 2] = ac_power[N // 2]  # Nyquist
        target_level = ac_power.sum() / (N - 1)
        target_power = np.full(N, target_level)
        target_power[0] = power[0]  # Keep DC

        # Blend
        alpha = 1.0 - relax
        new_power = alpha * power + relax * target_power
        new_power = np.maximum(new_power, 0)

        # Recover with new amplitude, same phase
        phase = np.angle(A)
        A_new = np.sqrt(new_power) * np.exp(1j * phase)

        # Enforce conjugate symmetry for real signal
        A_new[0] = A_new[0].real
        if N % 2 == 0:
            A_new[N // 2] = A_new[N // 2].real
        for k in range(1, N // 2 + (1 if N % 2 == 0 else 0)):
            if N - k < N:
                A_new[N - k] = np.conj(A_new[k])

        a_new = np.fft.ifft(A_new).real[:n]

        # Soft projection: clip negatives, add tiny floor, normalize
        a_new = np.maximum(a_new, 1e-12)
        s = a_new.sum()
        if s < 1e-15:
            a_work = best_a.copy()
            break
        a_work = a_new / s

        sc = _score_norm(a_work)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_work.copy()

    return best_a, best_sc


def _peak_notch_freq(a, n_iter, n_peaks=10, strength=0.05, seed=0):
    """
    Reduce specific peaks in the convolution by applying
    frequency-domain notches at constructive interference points.
    """
    n = len(a)
    N = _next_pow2(2 * n - 1)
    rng = np.random.default_rng(seed)
    a_work = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a_work)
    m = np.arange(N)

    for i in range(n_iter):
        c = _full_convolve(a_work, a_work)
        c_max = c.max()
        peak_idx = np.argsort(c)[::-1][:n_peaks]

        A = np.fft.fft(a_work, N)
        A_sq = A ** 2

        # Compute notch: where does each peak get constructive contribution?
        notch = np.zeros(N)
        for k_star in peak_idx:
            phase_k = np.exp(2j * np.pi * m * k_star / N)
            contrib = np.real(A_sq * phase_k)
            notch += np.maximum(contrib, 0)

        # Apply notch (reduce amplitude where interference is constructive)
        max_notch = notch.max()
        if max_notch < 1e-30:
            break
        scale = 1.0 - strength * notch / max_notch
        scale = np.maximum(scale, 0.3)

        A_mod = A * np.sqrt(scale)
        A_mod[0] = A_mod[0].real
        if N % 2 == 0:
            A_mod[N // 2] = A_mod[N // 2].real
        for k in range(1, N // 2 + (1 if N % 2 == 0 else 0)):
            if N - k < N:
                A_mod[N - k] = np.conj(A_mod[k])

        a_new = np.fft.ifft(A_mod).real[:n]
        a_new = np.maximum(a_new, 1e-12)
        s = a_new.sum()
        if s < 1e-15:
            a_work = best_a.copy()
            break
        a_work = a_new / s

        sc = _score_norm(a_work)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_work.copy()

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_remaining=30, n_betas=20, seed=0):
    """L-BFGS with adaptive beta schedule, multiple passes."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    # More betas for finer convergence
    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.0156, 0.0078, 0.0039, 0.0019, 0.00095,
        0.00048, 0.00024, 0.00012, 0.00006, 0.00003,
        0.000015, 0.000007, 0.000003, 0.000001
    ])[:n_betas]

    current_a = a_init.copy()
    current_sc = _score_norm(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break

        # Estimate time per iteration based on n
        est_time = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(8000, max(20, int(remaining / (est_time * 15))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score_norm(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 5 == 0:
                    print(f"      L-BFGS beta={beta:.2e}: score={current_sc:.12f}")
        except Exception:
            pass

    return current_a


def _nesterov_subgrad(a_init, deadline, min_time_remaining=30.0,
                      K_init=5, K_max=150, seed=0):
    """Nesterov-accelerated subgradient with adaptive active set."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    momentum = 0.8
    base_lr = 0.003
    lr = base_lr
    K = K_init
    t = 0
    no_improve = 0
    stall_count = 0

    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
        if gap < c_max * 0.01 and K < K_max:
            K = min(K_max, int(K * 1.4) + 1)

        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]

        # Compute subgradient (sum of gradients of active peaks)
        grad = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                idx = k - np.arange(j_lo, j_hi + 1)
                grad[j_lo:j_hi + 1] += a[idx]
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_lo2 <= j_hi2:
                idx = np.arange(j_lo2, j_hi2 + 1) - k
                grad[j_lo2:j_hi2 + 1] += a[idx]

        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-18:
            break

        effective_lr = min(base_lr, lr) / (gnorm + 1e-18) * 0.05
        v = momentum * v - effective_lr * grad_t
        a_cand = a + v
        a_proj = _project_simplex(a_cand)

        c_new = _full_convolve(a_proj, a_proj)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < current_sc:
            a = a_proj
        else:
            v = 0.5 * v
            a = a_proj

        if no_improve % 100 == 0 and no_improve > 0:
            stall_count += 1
            if stall_count > 8:
                break
            # Random restart
            perturb = 0.002 * rng.standard_normal(n)
            a_pert = _project_simplex(a + perturb)
            c_pert = _full_convolve(a_pert, a_pert)
            sc_pert = 2.0 * n * float(c_pert.max())
            if sc_pert < best_sc:
                a = a_pert
                best_a = a.copy()
                best_sc = sc_pert
            v = np.zeros(n)
            no_improve = 0

        if t % 300 == 0:
            print(f"      Nesterov t={t}: best={best_sc:.10f}, K={K}")

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _spectral_random_restart(a, n_restarts=20, relax=0.1, seed=0):
    """Random phase perturbations in frequency domain."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    rng = np.random.default_rng(seed)
    best_a = a.copy()
    best_sc = _score_norm(best_a)

    for _ in range(n_restarts):
        A = np.fft.fft(best_a, N)
        # Random phase perturbation
        phase_pert = rng.uniform(-relax, relax, N)
        phase_pert[0] = 0  # Keep DC
        if N % 2 == 0:
            phase_pert[N // 2] = 0
            phase_pert[N // 2 + 1:] = -phase_pert[1:N // 2][::-1]
        else:
            phase_pert[(N // 2) + 1:] = -phase_pert[1:(N // 2) + 1][::-1]
        A_pert = A * np.exp(1j * phase_pert)
        a_pert = np.fft.ifft(A_pert).real[:n]
        a_pert = np.maximum(a_pert, 1e-12)
        a_pert = a_pert / a_pert.sum()
        sc = _score_norm(a_pert)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_pert.copy()

    return best_a, best_sc

        max_iter = min(3000, max(20, int(remaining / 0.03)))
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxcor': 80})
            a_fin = _softmax(result.x)
            sc_fin = _score_norm(a_fin)
            if sc_fin < best_score:
                best_seq = a_fin.copy()
                best_score = sc_fin
                print(f"  NEW BEST: {best_score:.12f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
        except Exception:
            pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
