"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Improved active-set Newton with adaptive active set detection
2. Coordinate descent with exact quadratic line search
3. Subspace optimization (block coordinate descent)
4. Spectral initialization and multi-n exploration
5. KKT residual monitoring for convergence detection

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _kkt_residual(a, c, c_max, tol_frac=1e-4):
    """Compute the KKT residual: how far from optimal is the current point?"""
    n = len(a)
    active = c >= (c_max - tol_frac * c_max)
    if not active.any():
        return 1.0
    w = active.astype(np.float64)
    w = w / w.sum()
    g = 2.0 * _corr_a_A(a, w, n)
    g_t = g - np.mean(g)
    # KKT residual: for active coordinates (a_i > 0), gradient should be constant
    pos_mask = a > 1e-10
    if pos_mask.sum() < 2:
        return np.linalg.norm(g_t)
    return float(np.std(g_t[pos_mask]))


def _newton_step_direct(a, c, c_max, tol_frac, mu):
    """Newton step with direct solve for the active set KKT system."""
    n = len(a)
    active = c >= (c_max - tol_frac * c_max)
    if not active.any():
        active[np.argmax(c)] = True
    m = int(active.sum())
    if m < 2 or m > 2000:
        return None
    
    A_vec = active.astype(np.float64) / m
    g = 2.0 * _corr_a_A(a, A_vec, n)
    g_t = g - np.mean(g)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-15:
        return None
    
    # Build the Hessian as a matvec: H*v = 2*corr(v, A_vec) + mu*v
    A_vec_copy = A_vec
    n_copy = n
    mu_copy = mu
    
    def _hess_vp(v):
        return 2.0 * _corr_a_A(v, A_vec_copy, n_copy) + mu_copy * v
    
    # Use CG for the linear system
    x = np.zeros(n)
    r = g_t - _hess_vp(x)
    p = r.copy()
    rs = np.dot(r, r)
    for _ in range(300):
        Ap = _hess_vp(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < 1e-12 * (1 + np.linalg.norm(g_t)):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new

def _coordinate_descent(a_init, deadline, max_iters=5000, min_time=30.0):
    """Coordinate descent with exact quadratic line search."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()

        return a, np.inf

    # update the convolution when one coordinate changes

    # If we change a[j] by delta:
    # c[k] changes by delta * (a[j-k] + a[j+k]) for appropriate k
    
    # For efficiency, we'll use a blocking approach
    # Group coordinates and optimize in blocks
    
    block_size = max(1, n // 100)
    n_blocks = (n + block_size - 1) // block_size
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    for outer in range(max_iters):
        if time.monotonic() + min_time > deadline:
            break
        
        improved = False
        for block_idx in range(n_blocks):
            start = block_idx * block_size
            end = min(start + block_size, n)
            
            # Get current sub-vector
            a_sub = a[start:end].copy()
            
            # Compute gradient for this block
            # d/d a[j] of c[k] = a[j-k] + a[j+k] (with bounds)
            # We want to reduce max(c), so minimize the active lags
            
            # Simple approach: compute full gradient, take step for this block
            w = np.zeros(2 * n - 1)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            active = c >= c_max * 0.99
            if not active.any():
                active[np.argmax(c)] = True
            w[active] = 1.0 / active.sum()
            
            g = 2.0 * _corr_a_A(a, w, n)
            g_sub = g[start:end]
            
            # Exact line search along -g_sub direction
            # f(t) = max_k (c_k + t * (g_k contribution))
            # Approximate: minimize quadratic
            
            a_try = a.copy()
            best_t = 0.0
            best_f = c_max
            
            for t in [0.0001, 0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                a_try[start:end] = a_sub - t * g_sub
                a_try[start:end] = np.maximum(a_try[start:end], 0)
                a_try = a_try / a_try.sum()
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < best_f:
                    best_f = f_try
                    best_t = t
            
            if best_t > 0:
                a[start:end] = a_sub - best_t * g_sub
                a[start:end] = np.maximum(a[start:end], 0)
                a = a / a.sum()
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            # Try smaller steps
            for scale in [0.01, 0.005, 0.001]:
                rng = np.random.default_rng(outer)
                perturb = scale * rng.standard_normal(n)
                a_try = np.maximum(a + perturb, 0)
                a_try = a_try / a_try.sum()
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            if not improved:
                break
    
    return best_a, best_sc


def _subspace_optimize(a_init, deadline, n_dims=50, min_time=30.0):
    """Optimize in a low-dimensional subspace defined by top eigenvectors."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    # Build subspace from gradient structure
    c = _full_convolve(a, a)
    c_max = float(c.max())
    active = c >= c_max * 0.95
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / active.sum()
    
    g = 2.0 * _corr_a_A(a, w, n)
    g_t = g - np.mean(g)
    
    # Use top components of the gradient as basis
    # Or use sinusoidal basis
    t = np.arange(n, dtype=np.float64)
    basis = []
    for freq in range(1, min(n_dims, n // 2) + 1):
        basis.append(np.cos(2 * np.pi * freq * t / n))
        if len(basis) >= n_dims:
            break
    
    # Also add the gradient direction
    g_norm = np.linalg.norm(g_t)
    if g_norm > 1e-15:
        basis.append(g_t / g_norm)
    
    # Orthonormalize using QR
    B = np.column_stack(basis[:n_dims])
    try:
        Q, R = np.linalg.qr(B)
        # Keep only valid basis vectors
        valid = np.abs(np.diag(R)) > 1e-10
        Q = Q[:, valid]
    except:
        Q = g_t.reshape(-1, 1) / g_norm
    
    # Optimize in the subspace: a + sum_k alpha_k * Q[:,k]
    n_sub = Q.shape[1]
    
    def _subspace_obj(alpha):
        a_new = a + Q @ alpha
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            return 1e10
        a_new = a_new / sn
        c_new = _full_convolve(a_new, a_new)
        return float(c_new.max())
    
    # Use L-BFGS in the subspace
    alpha0 = np.zeros(n_sub)
    
    def _sub_obj(alpha):
        return _subspace_obj(alpha)
    
    def _sub_grad(alpha):
        # Numerical gradient
        eps = 1e-7
        grad = np.zeros(n_sub)
        f0 = _subspace_obj(alpha)
        for i in range(n_sub):
            alpha_p = alpha.copy()
            alpha_p[i] += eps
            alpha_m = alpha.copy()
            alpha_m[i] -= eps
            grad[i] = (_subspace_obj(alpha_p) - _subspace_obj(alpha_m)) / (2 * eps)
        return grad
    
    remaining = deadline - min_time - time.monotonic()
    if remaining < 5:
        return best_a, best_sc
    
    max_iter = min(500, max(50, int(remaining / (0.1 * n_sub))))
    
    try:
        result = minimize(
            _sub_obj, alpha0,
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12}
        )
        a_new = a + Q @ result.x
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            print(f"    Subspace: score={sc_new:.12f}")
    except Exception as e:
        print(f"    Subspace error: {e}")
    
    return best_a, best_sc


def _make_tapered(n, taper_power=1.0):
    """Create a tapered sequence (smooth at boundaries)."""
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** taper_power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_gaussian(n, width=0.3):
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - 0.5) / width) ** 2)
    return a / a.sum()


def _deflation_step(a, deadline, n_attempts=100, min_time=20.0):
    """Multi-peak deflation: reduce all significant peaks simultaneously."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for attempt in range(n_attempts):
        if time.monotonic() + min_time > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find peaks at various thresholds
        for thresh_frac in [0.999, 0.99, 0.98, 0.95]:
            threshold = c_max * thresh_frac
            peak_mask = c >= threshold
            n_peaks = int(peak_mask.sum())
            if n_peaks < 1:
                continue
            
            w = np.zeros(2 * n - 1)
            w[peak_mask] = 1.0 / n_peaks
            
            grad = 2.0 * _corr_a_A(a, w, n)
            g_t = grad - np.mean(grad)
            
            applied = False
            for strength in [0.0001, 0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
                a_try = a - strength * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    applied = True
                    break
            
            if applied:
                break
    
    return best_a, best_sc


def _lbfgs_logsumexp(a_init, deadline, min_time=30.0):
    """L-BFGS with log-sum-exp smoothing of the max."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    # Determine initial beta from the gap
    c_sorted = np.sort(c_cur)[::-1]
    gap_idx = min(20, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[gap_idx]
    base_beta = max(gap * 0.2, max_c * 0.001)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    def _lsobj(z_val, beta):
        a_val = _softmax(z_val)
        c_val = _full_convolve(a_val, a_val)
        c_max_val = float(c_val.max())
        shifted = (c_val - c_max_val) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300:
            return 2.0 * n * c_max_val
        return 2.0 * n * (c_max_val + beta * np.log(s))
    
    def _lsgrad(z_val, beta):
        a_val = _softmax(z_val)
        c_val = _full_convolve(a_val, a_val)
        c_max_val = float(c_val.max())
        shifted = (c_val - c_max_val) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300:
            w_val = np.zeros(2 * n - 1)
            w_val[int(np.argmax(c_val))] = 1.0
        else:
            w_val = e / s
        wc = _full_convolve(w_val, a_val)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w_val, c_val))
        return 4.0 * n * a_val * (h - W)
    
    # Beta schedule: geometric from base_beta down to very small
    n_betas = 15
    min_beta = max_c * 1e-8
    betas = np.geomspace(base_beta, min_beta, n_betas)
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 5:
            break
        
        est_time_per_iter = 0.005 * max(1.0, n / 8192.0)
        max_iter = min(3000, max(20, int(remaining / (est_time_per_iter * 5))))
        
        try:
            result = minimize(
                lambda zv: _lsobj(zv, beta),
                z,
                jac=lambda zv: _lsgrad(zv, beta),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxcor': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception as e:
            pass
    
    return current_a


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining Newton refinement, coordinate descent,
    subspace optimization, and spectral methods to minimize
    2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 40.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.12f} (n={len(p_arr)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass
    
    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={len(parent)}, score={sc_parent:.12f}")
        update_best(parent, sc_parent, "parent")
        n_main = len(parent)
    else:
        n_main = 16384
        parent = _make_tapered(n_main, 1.0)
        update_best(parent, _score(parent), "init_tapered")
    
    n = n_main
    
    # === Phase 1: Newton refinement from parent ===
    print(f"\n[Phase 1] Newton refinement (n={n}, t={time.monotonic() - t_start:.1f}s)")
    newton_dl = min(deadline - 600, time.monotonic() + 300)
    a_newton, sc_newton = _newton_refine(best_seq, newton_dl, max_outer=200, min_time=600)
    update_best(a_newton, sc_newton, "newton")
    
    # === Phase 2: Deflation ===
    print(f"\n[Phase 2] Multi-peak deflation (t={time.monotonic() - t_start:.1f}s)")
    def_dl = min(deadline - 500, time.monotonic() + 100)
    a_def, sc_def = _deflation_step(best_seq, def_dl, n_attempts=200, min_time=500)
    update_best(a_def, sc_def, "deflation")
    
    # === Phase 3: L-BFGS log-sum-exp ===
    print(f"\n[Phase 3] L-BFGS log-sum-exp (t={time.monotonic() - t_start:.1f}s)")
    lbfgs_dl = min(deadline - 400, time.monotonic() + 150)
    a_lbfgs = _lbfgs_logsumexp(best_seq, lbfgs_dl, min_time=400)
    update_best(a_lbfgs, _score(a_lbfgs), "lbfgs_lse")
    
    # === Phase 4: Coordinate descent ===
    print(f"\n[Phase 4] Coordinate descent (t={time.monotonic() - t_start:.1f}s)")
    cd_dl = min(deadline - 300, time.monotonic() + 100)
    a_cd, sc_cd = _coordinate_descent(best_seq, cd_dl, max_iters=3000, min_time=300)
    update_best(a_cd, sc_cd, "coord_descent")
    
    # === Phase 5: Subspace optimization ===
    print(f"\n[Phase 5] Subspace optimization (t={time.monotonic() - t_start:.1f}s)")
    sub_dl = min(deadline - 250, time.monotonic() + 80)
    a_sub, sc_sub = _subspace_optimize(best_seq, sub_dl, n_dims=30, min_time=250)
    update_best(a_sub, sc_sub, "subspace")
    
    # === Phase 6: Iterative refinement loop ===
    print(f"\n[Phase 6] Iterative refinement (t={time.monotonic() - t_start:.1f}s)")
    for rnd in range(6):
        if time.monotonic() + 200 > deadline:
            break
        
        # Newton
        nd = min(deadline - 150, time.monotonic() + 60)
        a_n, sc_n = _newton_refine(best_seq, nd, max_outer=50, min_time=150)
        update_best(a_n, sc_n, f"refine_n{rnd}")
        
        # Deflation
        dd = min(deadline - 100, time.monotonic() + 40)
        a_d, sc_d = _deflation_step(best_seq, dd, n_attempts=50, min_time=100)
        update_best(a_d, sc_d, f"refine_d{rnd}")
        
        # Check convergence
        if _kkt_residual(best_seq, _full_convolve(best_seq, best_seq), 
                         float(_full_convolve(best_seq, best_seq).max()), 1e-5) < 1e-6:
            print("    KKT residual very small, converging")
            break
    
    # === Phase 7: Try different n ===
    print(f"\n[Phase 7] Try different n (t={time.monotonic() - t_start:.1f}s)")
    for n_try in [8192, 32768, 16384]:
        if time.monotonic() + 100 > deadline:
            break
        if n_try == n_main:
            continue
        a_try = _interpolate_seq(best_seq, n_try)
        sc_try = _score(a_try)
        print(f"  n={n_try} interpolated score: {sc_try:.12f}")
        
        # Quick Newton on this
        nd = min(deadline - 80, time.monotonic() + 50)
        a_opt, sc_opt = _newton_refine(a_try, nd, max_outer=30, min_time=80)
        update_best(a_opt, sc_opt, f"n_{n_try}")
    
    # === Phase 8: Randomized restarts ===
    print(f"\n[Phase 8] Randomized restarts (t={time.monotonic() - t_start:.1f}s)")
    for sigma in [0.001, 0.003, 0.01, 0.03]:
        if time.monotonic() + 60 > deadline:
            break
        p_start = _perturb(best_seq, sigma, rng)
        # Quick Newton
        nd = min(deadline - 40, time.monotonic() + 30)
        a_r, sc_r = _newton_refine(p_start, nd, max_outer=30, min_time=40)
        update_best(a_r, sc_r, f"restart_s{sigma}")
    
    # === Phase 9: Final aggressive polish ===
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    # Very tight Newton
    nd = min(deadline - 10, time.monotonic() + 60)
    a_fin, sc_fin = _newton_refine(best_seq, nd, max_outer=100, min_time=10)
    update_best(a_fin, sc_fin, "final_newton")
    
    # Final deflation
    dd = min(deadline - 5, time.monotonic() + 30)
    a_fin2, sc_fin2 = _deflation_step(best_seq, dd, n_attempts=50, min_time=5)
    update_best(a_fin2, sc_fin2, "final_deflation")
    
    # Final L-BFGS with tiny beta
    if time.monotonic() < deadline - 8:
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 1e-9
        remaining = deadline - 8 - time.monotonic()
        if remaining > 3:
            max_iter_fin = min(2000, max(50, int(remaining / 0.05)))
            try:
                result = minimize(
                    lambda zv: _lsobj(zv, beta_final),
                    z,
                    jac=lambda zv: _lsgrad(zv, beta_final),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter_fin, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 100}
                )
                a_fin3 = _softmax(result.x)
                update_best(a_fin3, _score(a_fin3), "final_tiny_beta")
            except:
                pass
    
    # Final output
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
