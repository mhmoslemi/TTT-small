"""
Optimized step function coefficient minimizer.
Uses adaptive KKT-Newton with gap-based active set selection,
smooth max L-BFGS with adaptive temperature, and multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute H@v where H is the Hankel matrix from weights W."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve_hankel(W, n, rhs, max_iter=300, tol=1e-12):
    """Solve Hx = rhs where H is Hankel matrix from W."""
    x = np.zeros(n)
    r = rhs - _hankel_matvec(x, W, n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.05:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _compute_gradient(a, W, n):
    """Compute gradient of the objective using weights W on convolution."""
    a_rev = a[::-1]
    wc = _full_convolve(W, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(W, _full_convolve(a, a)))
    return 2.0 * a * (h - W_val)


def _adaptive_kkt_newton(a_init, deadline, max_rounds=500, min_K=1, max_K=5000):
    """
    KKT-based Newton with adaptive K selection based on gap structure.
    The key improvement: K is chosen based on where the "gap" is in the sorted convolution,
    rather than a fixed schedule. This allows the method to automatically respond to
    the structure of the problem.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0
    current_K = min_K

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        c_sorted = np.sort(c)[::-1]
        
        # Adaptive K: find the natural gap in the sorted convolution
        # We want K such that c_sorted[K-1] is close to c_max
        # but c_sorted[K] has a noticeable drop
        adaptive_K = current_K
        if len(c_sorted) > 10:
            # Look for the first significant gap
            for k in range(min(10, len(c_sorted) - 1), min(len(c_sorted) - 1, max_K + 10)):
                gap = c_sorted[k-1] - c_sorted[k]
                if gap > 0.01 * c_max:
                    adaptive_K = max(min_K, k)
                    break
            else:
                adaptive_K = min(max_K, current_K * 2)
        
        # Also consider the fraction of energy in the top values
        # If top K values contain most of the "excess", use that K
        for test_K in [max(1, int(0.001 * n)), max(1, int(0.005 * n)), 
                       max(1, int(0.01 * n)), max(1, int(0.05 * n)),
                       max(1, int(0.1 * n))]:
            if c_sorted[test_K - 1] > 0.9 * c_max:
                adaptive_K = max(adaptive_K, test_K)
                break
        
        K = min(adaptive_K, 2 * n - 1, max_K)
        K = max(K, min_K)
        
        # Select active set
        idx = np.argpartition(c, -K)[-K:]
        W = np.zeros(2 * n - 1)
        W[idx] = 1.0 / K
        
        # Compute gradient
        g = _compute_gradient(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break
        
        # Newton step via CG
        Hg = _cg_solve_hankel(W, n, g, max_iter=200)
        H1 = _cg_solve_hankel(W, n, np.ones(n), max_iter=200)
        H1_denom = float(np.ones(n) @ H1)

            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / H1_denom
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)
        
        # Line search with stronger sufficient decrease
        alpha = 1.0
        step_done = False
        for ls in range(80):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            # Stronger sufficient decrease: require at least 1% relative decrease per unit step
            if c_max_try < c_max * (1.0 - 0.02 * alpha) - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break
        
        # Fallback: subgradient step
        if not step_done:
            for trial in [0.1, 0.05, 0.02, 0.01, 0.005, 0.001]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break

            best_sc = sc
            best_a = a.copy()
            no_improve = 0
            # Increase K when improving
            current_K = min(max_K, max(current_K, K))
        else:
            no_improve += 1
            # Decrease K when stuck (try smaller active set)
            current_K = max(min_K, current_K // 2)
        
        if not step_done and no_improve > 10:
            break

    return best_a, best_sc


def _smooth_max_optimize(a_init, deadline, min_remaining=60.0):

    L-BFGS optimization of the smooth max approximation.
    Uses adaptive temperature with gap-based adjustment.
    """

    current_a = a_init.copy()

    c_sorted = np.sort(c_cur)[::-1]

    beta = max(gap * 0.5, max_c * 0.001)

    z = z - z.mean()
    
    outer_iter = 0
    while beta > max_c * 5e-5:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break
        
        outer_iter += 1
        scale = max(1.0, n / 8192.0)
        est_iter = 0.003 * scale
        max_iter = min(3000, max(50, int(remaining / (est_iter * 40))))

            a_vals = _softmax(z_vals)
            c_vals = _full_convolve(a_vals, a_vals)
            c_max_val = float(c_vals.max())
            shifted = (c_vals - c_max_val) / beta
            e_vals = np.exp(np.clip(shifted, -700, 0))
            s_vals = e_vals.sum()
            if s_vals < 1e-300 or not np.isfinite(s_vals):
                return 2.0 * n * c_max_val
            return 2.0 * n * (c_max_val + beta * np.log(s_vals))
        
        def grad_fn(z_vals):
            a_vals = _softmax(z_vals)
            c_vals = _full_convolve(a_vals, a_vals)
            c_max_val = float(c_vals.max())
            shifted = (c_vals - c_max_val) / beta
            e_vals = np.exp(np.clip(shifted, -700, 0))
            s_vals = e_vals.sum()
            if s_vals < 1e-300 or not np.isfinite(s_vals):
                w_vals = np.zeros_like(c_vals)
                w_vals[int(np.argmax(c_vals))] = 1.0
            else:
                w_vals = e_vals / s_vals
            a_rev = a_vals[::-1]
            wc = _full_convolve(w_vals, a_rev)
            h = wc[n - 1:2 * n - 1]
            W_val = float(np.dot(w_vals, c_vals))
            return 4.0 * n * a_vals * (h - W_val)
        
        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 80, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
        
        # Update beta based on new gap structure
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]

        new_gap = c_sorted_new[0] - c_sorted_new[new_gap_idx]

            beta *= 0.08
        elif new_gap < 2e-4 * c_max_new:
            beta *= 0.15
        elif new_gap < 1e-3 * c_max_new:
            beta *= 0.25
        else:
            beta *= 0.4
        
        beta = min(beta, c_max_new * 0.0005)
        # Don't decrease beta too fast
        beta = max(beta, c_max_new * 5e-6)

def _subgradient_adaptive(a_init, deadline, max_steps=10000):
    """
    Subgradient method with adaptive learning rate and Nesterov-like momentum.
    Uses soft active set (weighted by proximity to max) instead of hard threshold.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.03
    momentum = 0.9
    v = np.zeros(n)
    
    # Nesterov momentum state
    z_nesterov = z.copy()
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        # Nesterov lookahead
        z_check = z_nesterov + momentum * (z_nesterov - z)
        a = _softmax(z_check)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Soft active set: weight by exponential proximity to max
        # This is like a tempered distribution over the convolution
        temp = c_max * 0.001
        weights = np.exp(-(c_max - c) / temp)
        # Only keep weights that are significant
        threshold = weights.max() * 0.01
        active_mask = weights > threshold
        if not active_mask.any():
            active_mask[int(np.argmax(c))] = True
        W = np.zeros_like(c)
        W[active_mask] = weights[active_mask] / weights[active_mask].sum()
        
        # Compute subgradient
        a_rev = a[::-1]
        wc = _full_convolve(W, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(W, c))
        grad = 2.0 * a * (h - W_val)
        grad = grad - np.mean(grad)
        grad_norm = float(np.linalg.norm(grad))
        
        if grad_norm < 1e-16:
            break
        
        # Update with momentum
        v = momentum * v - lr * grad / (grad_norm / np.sqrt(n))
        z_nesterov = z + v
        z = z_nesterov
        
        # Clip and recenter
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()
        
        # Periodic evaluation
        if i % 100 == 99:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, 0.06)
            else:
                patience += 1
                if patience % 5 == 0:
                    lr *= 0.9
                    momentum = min(momentum + 0.005, 0.95)
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    z_nesterov = z.copy()
                    v = np.zeros(n)
                    patience = 0
                    lr = 0.02
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    elif kind == "optimal_shape":
        # Try a shape that might minimize peak convolution
        # Use a "raised cosine" with optimized taper
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * t / 2) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        taper = 0.15
        a = np.ones(n)
        mask_left = t < taper
        mask_right = t > 1 - taper
        a[mask_left] = 0.5 * (1 + np.cos(np.pi * t[mask_left] / taper))
        a[mask_right] = 0.5 * (1 + np.cos(np.pi * (1 - t[mask_right]) / taper))
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)

Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive KKT-Newton with gap-based active set, Nesterov subgradient
with soft active set, and smooth max L-BFGS with adaptive temperature.

# Global state for smooth max optimization (avoids closures)
_SM_STATE = {'beta': 0.01, 'n': 1000}


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve_hankel(W, n, rhs, max_iter=300, tol=1e-12):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.05:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

    wc = _full_convolve(W, a_rev)
    h = wc[n - 1:2 * n - 1]

    return 2.0 * a * (h - W_val)

        a = _softmax(z_check)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Soft active set
        temp = max(c_max * 0.0005, 1e-15)
        weights = np.exp(-np.maximum(0, c_max - c) / temp)
        threshold = weights.max() * 0.005
        active_mask = weights > threshold
        if not active_mask.any():
            active_mask[int(np.argmax(c))] = True
        W = np.zeros_like(c)
        W[active_mask] = weights[active_mask]
        W = W / W.sum()
        
        # Subgradient
        a_rev = a[::-1]
        wc = _full_convolve(W, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(W, c))
        grad = 2.0 * a * (h - W_val)
        grad = grad - np.mean(grad)
        grad_norm = float(np.linalg.norm(grad))
        
        if grad_norm < 1e-16:
            break
        
        # Nesterov update
        v = momentum * v - lr * grad / max(grad_norm / np.sqrt(n), 1e-30)
        z_nv = z + v
        z = z_nv
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()
        
        # Periodic evaluation
        if i % 100 == 99:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, 0.06)
            else:
                patience += 1
                if patience % 5 == 0:
                    lr *= 0.92
                    momentum = min(momentum + 0.003, 0.95)
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    z_nv = z.copy()
                    v = np.zeros(n)
                    patience = 0
                    lr = 0.02
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        taper = 0.12
        a = np.ones(n)
        mask_left = t < taper
        mask_right = t > 1 - taper
        a[mask_left] = 0.5 * (1 + np.cos(np.pi * t[mask_left] / taper))
        a[mask_right] = 0.5 * (1 + np.cos(np.pi * (1 - t[mask_right]) / taper))
        return a / a.sum()
    elif kind == "raised":
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * t / 2) ** 1.2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)
