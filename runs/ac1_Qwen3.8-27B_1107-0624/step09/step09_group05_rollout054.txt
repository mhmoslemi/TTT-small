def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_conv_blended(a, w, n):
    """Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    = 2 * conv(w, a_rev)[n-1 : 2n-1]
    """
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _smoobj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    grad_a = _grad_conv_blended(a, w, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)

def _optimal_lambda_active_set(a, c, active_mask, n, n_iter=50):
    """Find optimal lambda on active set to minimize gradient variance on support.
    
    Uses projected gradient with FFT-based matvec for Toeplitz structure.
    """
    S_indices = np.where(active_mask)[0]
    n_S = len(S_indices)
    if n_S == 0:
        return None
    if n_S == 1:
        lam = np.array([1.0])
        return lam

    # Support of a
    support = a > 1e-10 * a.max()
    n_T = int(support.sum())
    if n_T < 2:
        return np.ones(n_S) / n_S

    # Build B matrix implicitly via FFT
    # B[j, k] = a[S_indices[k] - j] for j in support (valid range)
    # M = B^T B is Toeplitz-like: M[k, k'] = sum_{j in support} a[S[k]-j] * a[S[k']-j]
    # Approximate as: M[k,k'] = conv(a,a_rev)[n-1 + S[k'] - S[k]] (for dense support)
    
    a_rev = a[::-1]
    cc = _full_convolve(a, a_rev)  # length 2n-1
    # cc[n-1+d] = sum_u a[u] a[u-d] (cross-correlation at lag d, approx)
    
    # For the QP: min lambda^T M lambda s.t. lambda >= 0, sum = 1
    # Use projected gradient with matvec via FFT
    
    lam = np.ones(n_S) / n_S
    scale = 1.0 / n_T
    
    for it in range(n_iter):
        # Compute g = 2*M@lam - 2*(1/n_T)*(1^T B lam) * (B^T 1)
        # Approximate: (B^T lam)[j] = sum_k lam[k] * a[S[k]-j] = conv(lam_extended, a_rev)[n-1+j]
        
        # Compute B^T lam
        lam_ext = np.zeros(2 * n - 1)
        lam_ext[S_indices] = lam
        bt_lam = _full_convolve(lam_ext, a_rev)
        bt_lam = bt_lam[n - 1:2 * n - 1]  # length n
        
        # Compute 1^T B lam = sum_j (B^T lam)[j] over support
        one_t_b_lam = float(np.sum(bt_lam[support]))
        
        # Compute B^T 1 (vector): [B^T 1][j] = sum_k a[S[k]-j] = conv(ones_ext, a_rev)[n-1+j]
        # ones_ext has 1 at positions S_indices
        b_t_one = _full_convolve(lam_ext * 0 + np.zeros(2*n-1), a_rev)  # dummy
        # Actually: B^T 1 at position j = sum_k 1 * a[S[k]-j]
        # = conv(indicator_of_S, a_rev)[n-1+j] where indicator has 1s at S
        ind_S = np.zeros(2 * n - 1)
        ind_S[S_indices] = 1.0
        b_t_one = _full_convolve(ind_S, a_rev)
        b_t_one = b_t_one[n - 1:2 * n - 1]
        
        # Gradient of objective: g = 2*(B^T B lam)/n_T - 2*(1^T B lam)/n_T^2 * B^T 1
        # Need B^T B lam = B^T (B lam)
        # B lam at position j = sum_k lam[k] a[S[k]-j] = bt_lam[j] (already computed)
        # B^T (B lam) at position k = sum_{j in support} bt_lam[j] * a[S[k]-j]
        # = conv(bt_lam_restricted, a_rev_shifted)... 
        
        # Simpler: use the Toeplitz approximation
        # (B^T B)[k, k'] ≈ cc[n-1 + S[k'] - S[k]]
        # So (B^T B lam)[k] = sum_{k'} cc[n-1 + S[k'] - S[k]] * lam[k']
        # This is a conv

        # Compute B^T B lam using FFT
        # (B^T B lam)[k] = sum_{k'} cc[n-1 + S[k'] - S[k]] * lam[k']
        # Let d = S[k'] - S[k], then this is like a cross-correlation
        
        # Alternative: just compute g = 2 * bt_lam on the support, then "lift" back
        # The gradient w.r.t. lambda[k] is: 2 * sum_{j in support} a[S[k]-j] * (bt_lam[j] - mean)
        # = 2 * (B^T (bt_lam - mean*1)) [at S[k]]
        
        mean_bt = float(np.mean(bt_lam[support]))
        centered = bt_lam - mean_bt
        # B^T centered at S_indices:
        ind_c = np.zeros(2 * n - 1)
        # We need sum_j centered[j] * a[S[k]-j] = conv(centered_ext, a_rev)[n-1 + S[k]]
        centered_ext = np.zeros(2 * n - 1)
        centered_ext[support] = centered[support]
        tmp = _full_convolve(centered_ext, a_rev)
        g = 2.0 * scale * tmp[n - 1 + S_indices]
        
        # Projected gradient step
        step = 0.1 / (it + 1)
        lam = lam - step * g
        # Project onto simplex

        ls = lam.sum()
        if ls > 1e-30:

        else:

            no_improve += 2
            if no_improve > max_no_improve:
                tol_frac = max(min_tol, tol_frac * 0.7)
                if tol_frac <= min_tol * 1.2:
                    break

        # Tighten tolerance when improving
        if improved:
            tol_frac = max(min_tol, tol_frac * 0.95)

        iter_count += 1
        if iter_count % 200 == 0 and verbose:
            print(f"    {label} iter={iter_count}: current={current_sc:.12f}, best={best_sc:.12f}")

    return best_a, best_sc


def _subgradient_polyak(a_init, deadline, min_time_remaining=30.0, verbose=True, label=""):
    """Subgradient method with Polyak step size and active set."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lb = 2.0 * n / (2 * n - 1)
    t_phase = time.monotonic()
    step_scale = 1.0
    t_iter = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            if verbose and t_iter % 100 == 0:
                print(f"    {label} SG iter={t_iter}: score={best_sc:.12f}, t={time.monotonic()-t_phase:.0f}s")

        # Active set
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        grad = _grad_conv_blended(a, w, n)

        # Center gradient (for simplex constraint)
        support = a > 1e-12
        if int(support.sum()) > 0:
            mean_g = float(np.mean(grad[support]))
        else:
            mean_g = float(np.mean(grad))
        grad_c = grad - mean_g

        # Polyak step
        gap = current_sc - lb
        if gap < 1e-12:
            break
        g_norm_sq = float(np.dot(grad_c[support], grad_c[support])) if int(support.sum()) > 0 else float(np.dot(grad_c, grad_c))
        if g_norm_sq < 1e-30:
            break
        step = step_scale * gap / (g_norm_sq * 2.0 * n)
        step = min(step, 0.05)

        a_new = a - step * grad_c
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new
        t_iter += 1

        if t_iter % 200 == 0:
            step_scale *= 0.97

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.6 * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * (t - 0.5)))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flattop":
        t = np.linspace(0, 1, n)
        a = (1.0 + 0.5 * np.cos(2 * np.pi * (t - 0.5))
             + 0.2 * np.cos(4 * np.pi * (t - 0.5))
             + 0.1 * np.cos(6 * np.pi * (t - 0.5)))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb_and_optimize(a_init, deadline, min_time_remaining=60.0, rng=None):
    """Add small random perturbation and re-optimize quickly."""
    if rng is None:
        rng = np.random.default_rng(123)
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    # Small random perturbation (maintaining non-negativity)
    noise_scale = 0.01
    noise = rng.normal(0, noise_scale, size=n)
    a_pert = a * np.exp(noise)
    a_pert = np.maximum(a_pert, 0.0)
    s_pert = a_pert.sum()
    if s_pert < 1e-10:
        return a_init, _score(a_init)
    a_pert = a_pert / s_pert

    sc_pert = _score(a_pert)
    # Quick L-BFGS on perturbed
    end = time.monotonic() + 80
    a_opt, sc_opt = _run_lbfgs(a_pert, end, min_time_remaining=10, verbose=False, label="pert")
    return a_opt, sc_opt


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining:
    P1: L-BFGS with soft-max (moderate to small beta)
    P2: Active-set refinement with optimal lambda
    P3: L-BFGS with very small beta (fine-tuning)
    P4: More active-set refinement
    P5: Random perturbation + re-optimization
    P6: Final polish (L-BFGS + active set + subgradient)
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.12f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(16384, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n=16384: score={best_score:.12f}")

    n_main = len(best_seq)

    # === Phase 1: L-BFGS main ===
    print(f"\n[P1] L-BFGS main (n={n_main}, t={time.monotonic()-t_start:.0f}s)")
    p1_end = t_start + min(400, (budget_s - 5) * 0.25)
    a1, sc1 = _run_lbfgs(best_seq, p1_end, min_time_remaining=200, verbose=True, label="P1")
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  P1 result: {best_score:.12f}")

    # === Phase 2: Active-set refinement with optimal lambda ===
    print(f"\n[P2] Active-set refine v2 (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p2_end = min(deadline - 400, time.monotonic() + 250)
        a2, sc2 = _active_set_refine_v2(best_seq, p2_end, min_time_remaining=50, verbose=True, label="P2")
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(f"  P2 result: {best_score:.12f}")

    # === Phase 3: L-BFGS with very small beta ===
    print(f"\n[P3] L-BFGS fine (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p3_end = min(deadline - 300, time.monotonic() + 350)
        a3, sc3 = _run_lbfgs(best_seq, p3_end, min_time_remaining=100, verbose=True, label="P3")
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3 result: {best_score:.12f}")

    # === Phase 4: More active-set refinement ===
    print(f"\n[P4] Active-set refine round 2 (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 40 < deadline:
        p4_end = min(deadline - 200, time.monotonic() + 180)
        a4, sc4 = _active_set_refine_v2(best_seq, p4_end, min_time_remaining=50, verbose=True, label="P4")
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4 result: {best_score:.12f}")

    # === Phase 5: Random perturbation + re-optimization ===
    print(f"\n[P5] Perturbation + re-opt (t={time.monotonic()-t_start:.0f}s)")
    n_perturb = 5
    for pi in range(n_perturb):
        if time.monotonic() + 100 > deadline:
            break
        print(f"  Perturbation {pi+1}/{n_perturb}")
        a_p, sc_p = _perturb_and_optimize(best_seq, deadline, min_time_remaining=100, rng=rng)
        if sc_p < best_score:
            best_score = sc_p
            best_seq = a_p.copy()
            print(f"  Perturb {pi+1} improved: {best_score:.12f}")
        else:
            print(f"  Perturb {pi+1}: {sc_p:.12f} (no improvement, best={best_score:.12f})")

    # === Phase 6: Subgradient polish ===
    print(f"\n[P6] Subgradient (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 40 < deadline:
        p6_end = min(deadline - 80, time.monotonic() + 100)
        a6, sc6 = _subgradient_polyak(best_seq, p6_end, min_time_remaining=30, verbose=True, label="P6")
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  P6 result: {best_score:.12f}")

    # === Phase 7: Final L-BFGS + active set ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 20 < deadline:
        p7_end = min(deadline - 20, time.monotonic() + 120)
        a7, sc7 = _run_lbfgs(best_seq, p7_end, min_time_remaining=10, verbose=False, label="P7")
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()
            print(f"  P7a: {best_score:.12f}")

        if time.monotonic() + 15 < deadline:
            p7b_end = min(deadline - 5, time.monotonic() + 50)
            a7b, sc7b = _active_set_refine_v2(best_seq, p7b_end, min_time_remaining=5, verbose=False, label="P7b")
            if sc7b < best_score:
                best_score = sc7b
                best_seq = a7b.copy()
                print(f"  P7b: {best_score:.12f}")

    # === Phase 8: Multi-n quick check ===
    print(f"\n[P8] Multi-n check (t={time.monotonic()-t_start:.0f}s)")
    n_opts = [12288, 20480, 24576]
    for n_try in n_opts:
        if time.monotonic() + 60 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if time.monotonic() + 40 > deadline:
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  n={n_try}: {best_score:.12f}")
            continue
        p8_end = min(deadline - 20, time.monotonic() + 60)
        a8, sc8 = _run_lbfgs(a_r, p8_end, min_time_remaining=10, verbose=False, label=f"n{n_try}")
        if sc8 < best_score:
            best_score = sc8
            best_seq = a8.copy()
            print(f"  n={n_try}: {best_score:.12f}")

    # Final
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
