from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

        return np.inf
    c = _full_convolve(a, a)

    """Build the row of the constraint matrix for lag k.
    The constraint is: sum_i f[k-i] * b_i <= M
    where f[k-i] is 0 if k-i is out of [0, n-1].
    """

    i_min = max(0, k - n + 1)
    i_max = min(k, n - 1)

    rows = []
    cols = []

"""
LP-based iterative optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses cutting-plane LP to find descent directions, combined with momentum
gradient polishing, multi-n exploration, and final fine-tuning.
Key innovation: Direct use of the LP from the problem statement with
cutting-plane constraint selection, which provides exact (unsmoothed)
descent directions unlike log-sum-exp approaches.
"""
import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_lp_matrix(f, lags, n):
    """Build sparse constraint matrix for LP. Row i corresponds to lag lags[i].
    A[i, j] = f[lags[i] - j] if 0 <= lags[i] - j < n, else 0."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(lags):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        if i_min <= i_max:
            indices = np.arange(i_min, i_max + 1)
            values = f[k - indices]
            mask = values > 0
            if mask.any():
                rows.extend([idx] * int(mask.sum()))
                cols.extend(indices[mask].tolist())
                vals.extend(values[mask].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(len(lags), n))
    return A


def _solve_lp(f, lags, M, n):
    """Solve: max sum(b) s.t. b >= 0, (f*b)_k <= M for k in lags."""
    if len(lags) == 0:
        return None
    A = _build_lp_matrix(f, lags, n)
    b_rhs = np.full(len(lags), M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_rhs, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.status == 0:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def _lp_direction(f, c, c_max, n, max_rounds=15, max_constraints=800):
    """Find LP direction using cutting plane. Returns normalized direction or None."""
    sorted_idx = np.argsort(c)[::-1]
    K_init = min(80, 2 * n - 1)
    active_lags = sorted_idx[:K_init].copy()

    for _ in range(max_rounds):
        b_sol = _solve_lp(f, active_lags, c_max, n)
        if b_sol is None:
            # Try fewer constraints
            active_lags = sorted_idx[:max(10, len(active_lags) // 2)]
            b_sol = _solve_lp(f, active_lags, c_max, n)
            if b_sol is None:
                return None

        sum_b = float(b_sol.sum())
        if sum_b < 1e-10:
            return None

        # Check all constraints
        f_b = _full_convolve(f, b_sol)
        tol = 1e-10 * c_max
        violations = np.where(f_b > c_max + tol)[0]

        if len(violations) == 0:
            return b_sol / sum_b

        # Add most violated constraints
        viol_amount = f_b[violations] - c_max
        n_add = min(20, len(violations))
        worst = violations[np.argsort(viol_amount)[::-1][:n_add]]
        new_lags = np.union1d(active_lags, worst)
        if len(new_lags) > max_constraints:
            # Keep only the most important (highest c values)
            c_vals = c[new_lags]
            top = new_lags[np.argsort(c_vals)[::-1][:max_constraints]]
            new_lags = top
        active_lags = new_lags

    return b_sol / sum_b


def _active_set_gradient(a, c, tol_frac):
    """Compute active-set gradient for max autocorrelation minimization."""
    n = len(a)
    c_max = float(c.max())
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if not active.any():
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True

    eps = 1e-12 * max(c_max, 1e-30)
    w = np.zeros_like(c)
    w[active] = 1.0 / (c_max - c[active] + eps)
    w = w / w.sum()

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad = 2.0 * h
    grad = grad - np.mean(grad)
    return grad


def _gradient_polish(a_init, deadline, tol_start=0.005, tol_end=0.0002,
                     lr_init=0.03, momentum=0.85, max_stall=80):
    """Momentum gradient descent with adaptive step and tolerance."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    v = np.zeros(n)
    tol_frac = tol_start
    lr = lr_init
    no_improve = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        grad = _active_set_gradient(a, c, tol_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        v = momentum * v - lr * grad
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            a_try = _normalize(a + alpha * v)
            sc_try = 2.0 * n * float(_full_convolve(a_try, a_try).max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                current_sc = sc_try
                improved = True
                break

        if not improved:
            v = np.zeros(n)
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_end, tol_frac * 0.7)
                lr *= 0.8
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(tol_end, tol_frac * 0.5)
                lr = lr_init * 0.4
                no_improve = 0
        else:
            if tol_frac > tol_end and no_improve < 10:
                tol_frac = max(tol_end, tol_frac * 0.99)

    return best_a, best_sc


def _line_search_step(a, direction, current_sc, n, step_sizes=None):
    """Find best step size towards direction."""
    if step_sizes is None:
        step_sizes = [1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
    best_t = 0.0
    best_sc = current_sc
    for t in step_sizes:
        a_try = _normalize((1.0 - t) * a + t * direction)
        sc_try = 2.0 * n * float(_full_convolve(a_try, a_try).max())
        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_t = t
    return best_t, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return _normalize(a_new)

                a = _normalize(a)
                consecutive_fail = 0
                print(f"  [perturb] iter={iteration}: score={_score(a):.10f} t={time.monotonic()-t_start:.0f}s")
            iteration += 1
            continue

        consecutive_fail = 0

        # Line search
        best_t, best_sc_step = _line_search_step(a, direction, current_sc, n)

        if best_t > 0 and best_sc_step < best_score - 1e-14:
            a = _normalize((1.0 - best_t) * a + best_t * direction)
            best_seq = a.copy()
            best_score = best_sc_step
            lp_success_count += 1
            iteration += 1

            if iteration % 10 == 0:
                print(f"  LP iter {iteration}: score={best_score:.10f} (t={best_t:.3f}, "
                      f"successes={lp_success_count}) t={time.monotonic()-t_start:.0f}s")

            # Adaptively adjust K
            if lp_success_count % 50 == 0:
                K_adaptive = min(200, K_adaptive + 10)
        else:
            # LP direction didn't help, try partial step or gradient
            grad = _active_set_gradient(a, c, 0.01)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-15:
                grad_dir = -grad / gnorm
                for alpha in [0.01, 0.005, 0.002]:
                    a_try = _normalize(a + alpha * grad_dir)
                    sc_try = _score(a_try)
                    if sc_try < best_score - 1e-14:
                        a = a_try
                        best_seq = a_try.copy()
                        best_score = sc_try
                        break
            iteration += 1
            if iteration % 20 == 0:
                print(f"  [no LP improve] iter={iteration}: score={best_score:.10f} t={time.monotonic()-t_start:.0f}s")

    print(f"  Phase 1 complete: {iteration} iterations, {lp_success_count} LP successes, "
          f"score={best_score:.10f}")

    # ================================================================
    # Phase 2: Intensive gradient polishing (~300s)
    # ================================================================
    print(f"\n[Phase 2] Intensive gradient polishing")
    phase2_deadline = min(deadline - 300, time.monotonic() + 300)

    a_pol, sc_pol = _gradient_polish(best_seq, phase2_deadline,
                                     tol_start=0.003, tol_end=0.00005,
                                     lr_init=0.02, momentum=0.85, max_stall=100)
    if sc_pol < best_score:
        best_seq = a_pol.copy()
        best_score = sc_pol
        print(f"  NEW BEST: {best_score:.10f} [phase2_grad]")
    else:
        print(f"  No improvement (stayed at {best_score:.10f})")

    # ================================================================
    # Phase 3: Second round of LP iterations on polished sequence (~250s)
    # ================================================================
    print(f"\n[Phase 3] LP round 2")
    phase3_deadline = min(deadline - 150, time.monotonic() + 250)

    a = _normalize(best_seq.copy())
    iter2 = 0

    while time.monotonic() < phase3_deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if iter2 > 0 and iter2 % 10 == 0:
            polish_dl = min(phase3_deadline - 10, time.monotonic() + 5)
            a_p, sc_p = _gradient_polish(a, polish_dl,
                                         tol_start=0.002, tol_end=0.0005,
                                         lr_init=0.008, momentum=0.8, max_stall=20)
            if sc_p < current_sc - 1e-12:
                a = a_p
                best_seq = a_p.copy()
                best_score = sc_p
                c = _full_convolve(a, a)
                c_max = float(c.max())
                current_sc = 2.0 * n * c_max
            iter2 += 1
            continue

        direction = _lp_direction(a, c, c_max, n, max_rounds=8, max_constraints=500)

        if direction is not None:
            best_t, best_sc_step = _line_search_step(a, direction, current_sc, n)
            if best_t > 0 and best_sc_step < best_score - 1e-14:
                a = _normalize((1.0 - best_t) * a + best_t * direction)
                best_seq = a.copy()
                best_score = best_sc_step
                if iter2 % 20 == 0:
                    print(f"  LP2 iter {iter2}: {best_score:.10f} t={time.monotonic()-t_start:.0f}s")
        iter2 += 1

    print(f"  Phase 3 complete: {iter2} iterations, score={best_score:.10f}")

    # ================================================================
    # Phase 4: Multi-n exploration (~150s)
    # ================================================================
    print(f"\n[Phase 4] Multi-n exploration")
    phase4_deadline = min(deadline - 80, time.monotonic() + 150)

    n_candidates = []
    for factor in [1.0, 1.25, 0.8, 1.5, 0.6, 2.0]:
        n_try = int(n * factor)
        if 2048 <= n_try <= 100000 and n_try != n:
            n_candidates.append(n_try)

    seen = set()
    n_unique = []
    for nn in n_candidates:
        if nn not in seen:
            seen.add(nn)
            n_unique.append(nn)

    for n_try in n_unique:
        if time.monotonic() > phase4_deadline - 15:
            break
        a_try = _resample(best_seq, n_try)
        # Quick gradient polish
        polish_dl = min(phase4_deadline - 5, time.monotonic() + 12)
        a_opt, sc_opt = _gradient_polish(a_try, polish_dl,
                                         tol_start=0.01, tol_end=0.002,
                                         lr_init=0.02, momentum=0.8, max_stall=40)
        sc_final = _score(a_opt)
        if sc_final < best_score - 1e-10:
            best_seq = a_opt.copy()
            best_score = sc_final
            n = n_try
            a = _normalize(best_seq.copy())
            print(f"  n={n_try}: NEW BEST {best_score:.10f} t={time.monotonic()-t_start:.0f}s")
        else:
            print(f"  n={n_try}: score={sc_final:.10f}")

    # ================================================================
    # Phase 5: Final ultra-fine polish (~80s)
    # ================================================================
    print(f"\n[Phase 5] Final polish")
    final_deadline = deadline - 3

    # Gradient with very small tolerance
    a_fin, sc_fin = _gradient_polish(best_seq, final_deadline,
                                     tol_start=0.001, tol_end=0.00001,
                                     lr_init=0.005, momentum=0.8, max_stall=60)
    if sc_fin < best_score:
        best_seq = a_fin.copy()
        best_score = sc_fin
        print(f"  NEW BEST: {best_score:.10f} [final_grad]")

    # One more LP round if time permits
    if time.monotonic() < deadline - 20:
        a = _normalize(best_seq.copy())
        c = _full_convolve(a, a)
        c_max = float(c.max())
        for _ in range(5):
            if time.monotonic() > deadline - 10:
                break
            direction = _lp_direction(a, c, c_max, n, max_rounds=5, max_constraints=300)
            if direction is not None:
                best_t, best_sc_step = _line_search_step(a, direction, 2.0 * n * c_max, n)
                if best_t > 0 and best_sc_step < best_score - 1e-14:
                    a = _normalize((1.0 - best_t) * a + best_t * direction)
                    best_seq = a.copy()
                    best_score = best_sc_step
                    c = _full_convolve(a, a)
                    c_max = float(c.max())
                    print(f"  [final LP] {best_score:.10f} t={time.monotonic()-t_start:.0f}s")

    # Final output
    print(f"\n{'=' * 60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) / 256
    final_score = _score(final)
    print(f"FINAL: score={final_score:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    return [float(x) for x in final]
