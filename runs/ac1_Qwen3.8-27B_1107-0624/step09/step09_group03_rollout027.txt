"""
Optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Multiplicative weights (mirror descent) with self-consistent active set weights
2. Active-set Newton with adaptive tolerance and KKT verification
3. Continuation on active set tightness and softmax temperature
4. Structured diversification (swap moves, block moves, anti-symmetric perturbations)
5. Multi-phase time allocation with final high-precision Newton polish
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _gradient_at_active(a, K_weights, n):
    """Compute gradient signal from active lags with given weights.
    K_weights is a vector of length 2n-1 with weights for active lags.
    Returns g where g[i] = sum_k K_weights[k] * 2 * a[k-i] (valid).
    """
    return 2.0 * _corr_a_A(a, K_weights, n)


def _mirror_descent_phase(a_init, deadline, beta_range=None, max_iter=None):

        c_init = _full_convolve(a, a)
        c_max_init = float(c_init.max())

def _newton_kkt(a_init, deadline, max_outer=200):
    """
    Active-set Newton method with KKT verification.
    At each step:
    1. Identify active set (lags near max)
    2. Solve the KKT linear system
    3. Take a Newton step with careful line search
    4. Verify KKT conditions; if not satisfied, update active set
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_sc = float(2.0 * n * _full_convolve(a, a).max())

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive tolerance: start loose, tighten as we converge
        progress = outer / max_outer
        if progress < 0.3:
            tol_frac = 0.01
        elif progress < 0.6:
            tol_frac = 0.003
        elif progress < 0.8:
            tol_frac = 0.001
        else:
            tol_frac = 0.0003

        threshold = c_max * (1.0 - tol_frac)
        active = c >= threshold
        if not active.any():
            active[np.argmax(c)] = True
        m = int(active.sum())

        if m < 2:
            active[np.argmax(c)] = True
            m = max(m, 2)
        if m > 5 * n:
            # Too many active; use a harder threshold
            active = c >= threshold * 0.99
            m = int(active.sum())

        # Build the active set weight vector
        A_vec = np.zeros(2 * n - 1)
        A_vec[active] = 1.0 / m

        # Compute gradient (uniform weights on active set)
        g = _gradient_at_active(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)

        if gnorm < 1e-15:
            break

        # Newton direction: solve (H + mu*I) dx = -g_t, then adjust for sum constraint
        # H is the Hessian of the weighted sum of active b[k]'s
        # H[i,j] = 2 * sum_k A_vec[k] * delta(i+j, k) (approximately)
        # Matvec: (H v)[i] = 2 * sum_k A_vec[k] * v[k-i] (for valid k-i)
        mu = 1e-8 / m + 1e-12

        A_vec_ref = A_vec
        n_ref = n
        mu_ref = mu

        def _hess_matvec(v):
            return 2.0 * _corr_a_A(v, A_vec_ref, n_ref) + mu_ref * v

        # CG solve for (H + mu*I) x = -g_t
        x0 = np.zeros(n)
        r = -g_t - _hess_matvec(x0)
        p = r.copy()
        rs = np.dot(r, r)
        rhs_norm = np.linalg.norm(-g_t)
        dx_u = np.zeros(n)
        for cg_iter in range(300):
            Ap = _hess_matvec(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            dx_u = dx_u + alpha * p
            r = r - alpha * Ap
            rs_new = np.dot(r, r)
            if np.sqrt(rs_new) < 1e-13 * (1.0 + rhs_norm):
                break
            p = r + (rs_new / rs) * p
            rs = rs_new

        # Also solve (H + mu*I) w = 1 for the sum constraint adjustment
        r_w = np.ones(n) - _hess_matvec(np.zeros(n))
        p_w = r_w.copy()
        rs_w = np.dot(r_w, r_w)
        w = np.zeros(n)
        for cg_iter in range(300):
            Ap_w = _hess_matvec(p_w)
            pAp_w = np.dot(p_w, Ap_w)
            if pAp_w < 1e-30:
                break
            alpha_w = rs_w / pAp_w
            w = w + alpha_w * p_w
            r_w = r_w - alpha_w * Ap_w
            rs_w_new = np.dot(r_w, r_w)
            if np.sqrt(rs_w_new) < 1e-13:
                break
            p_w = r_w + (rs_w_new / rs_w) * p_w
            rs_w = rs_w_new

        w_sum = float(w.sum())
        if abs(w_sum) > 1e-15:
            lam = float(dx_u.sum()) / w_sum
            dx = dx_u - lam * w
        else:
            dx = dx_u
        dx = dx - np.mean(dx)

        # Line search
        t = 1.0
        step_taken = False
        for ls in range(60):
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < c_max - 1e-14 * t:
                a = a_try
                sc_try = 2.0 * n * f_try
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                step_taken = True
                break
            t *= 0.5

        if not step_taken:
            # Gradient fallback with more aggressive line search
            t = 0.1
            for ls in range(40):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    f_try = float(c_try.max())
                    if f_try < c_max - 1e-15:
                        a = a_try
                        sc_try = 2.0 * n * f_try
                        if sc_try < best_sc:
                            best_sc = sc_try
                            best_a = a_try.copy()
                        step_taken = True
                        break
                t *= 0.5

        if not step_taken:
            break

        # Convergence check
        c_check = _full_convolve(a, a)
        if abs(float(c_check.max()) - c_max) < 1e-15:
            break

    return best_a, best_sc


def _self_consistent_weights(a, c, c_max, max_iter=20):
    """
    Find weights on active lags that make the 'pressure' as flat as possible
    on the support of a. This is a self-consistent iteration.
    """
    n = len(a)
    threshold = 0.95 * c_max
    active = c >= threshold
    if not active.any():
        active[np.argmax(c)] = True
    
    w = active.astype(np.float64)
    w = w / w.sum()
    
    for _ in range(max_iter):
        # Compute pressure
        p = _corr_a_A(a, w, n)
        
        # On the support (where a > 0), we want p to be constant
        support = a > 1e-10
        if support.sum() < 2:
            break
        
        p_support = p[support]
        p_mean = p_support.mean()
        p_std = p_support.std()
        if p_std < 1e-15:
            break
        
        # Update weights: increase weight on lags where pressure gradient is large
        # This pushes the pressure toward uniformity
        # The gradient of p[i] w.r.t. w[k] is a[k-i] (for valid k-i)
        # We want to minimize Var(p on support) w.r.t. w
        # d(Var)/dw[k] = 2 * sum_{i in support} (p[i] - p_mean) * a[k-i]
        # (approximately, ignoring the derivative of p_mean)
        
        # Simpler: use the fact that at optimum, p should be constant on support
        # so we want to "flatten" p by adjusting w
        # Use a gradient step: w_new = w - eta * grad, projected to simplex
        
        # Compute the "deficit" at each active lag
        # If increasing w[k] makes p more uniform, we want to increase w[k]
        # The change in p[i] from increasing w[k] is a[k-i]
        # We want sum_{i in support} (p[i] - p_mean) * a[k-i] to be negative
        # (so that increasing w[k] reduces the variance)
        
        active_idx = np.where(active)[0]
        if len(active_idx) < 2:
            break
        
        # Compute the "benefit" of each active lag
        benefits = np.zeros(len(active_idx))
        for idx, k in enumerate(active_idx):
            # p change at i from increasing w[k]: a[k-i] (for valid i)
            valid_low = max(0, k - (n - 1))
            valid_high = min(n - 1, k)
            if valid_high < valid_low:
                continue
            a_slice = a[valid_low:valid_high + 1]
            p_slice = p[valid_low:valid_high + 1]
            p_dev = p_slice - p_mean
            # Only count support positions
            supp_mask = support[valid_low:valid_high + 1]
            if supp_mask.any():
                benefits[idx] = np.dot(a_slice[supp_mask], p_dev[supp_mask])
        
        # We want to decrease high-variance: move weight toward lags with negative benefit
        # (negative benefit means increasing that weight reduces variance)
        benefits -= benefits.mean()
        w_new = w[active_idx] * np.exp(0.1 * (-benefits))
        w_new = np.maximum(w_new, 0)
        w_new_sum = w_new.sum()
        if w_new_sum < 1e-15:
            break
        w_new = w_new / w_new_sum
        
        w = np.zeros(2 * n - 1)
        w[active_idx] = w_new
    
    return w


def _structured_perturb(a, rng, mode="swap"):
    """
    Apply a structured perturbation to a.
    Modes: 'swap' (swap pairs), 'block' (move a block), 'antisym' (anti-symmetric bias)
    """
    n = len(a)
    a_new = a.copy()
    
    if mode == "swap":
        # Swap small random blocks
        n_swaps = rng.integers(3, 10)
        for _ in range(n_swaps):
            i = rng.integers(0, n - 50)
            j = rng.integers(0, n - 50)
            size = rng.integers(5, 50)
            size = min(size, n - 1 - max(i, j))
            if size > 0 and i != j and abs(i - j) > size:
                a_new[i:i+size], a_new[j:j+size] = a_new[j:j+size].copy(), a_new[i:i+size].copy()
    
    elif mode == "block":
        # Move a block from one side to the other
        size = rng.integers(n // 50, n // 10)
        src = rng.integers(0, n - size)
        dst = rng.integers(0, n - size)
        if abs(src - dst) > size:
            A = a_new[src:src+size].copy()
            a_new[src:src+size] = a_new[dst:dst+size]
            a_new[dst:dst+size] = A
    
    elif mode == "antisym":
        # Add anti-symmetric bias (large on left, small on right or vice versa)
        tilt = rng.uniform(0.02, 0.08)
        t = np.linspace(0, 1, n)
        bias = 1.0 + tilt * (2 * t - 1) * rng.uniform(1, 3)
        a_new = a_new * bias
    
    else:
        # Gaussian in log space
        z = np.log(np.maximum(a, 1e-30))
        z = z + 0.05 * rng.standard_normal(n)
        a_new = _softmax(z)
    
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s


def _lbfgs_logsumexp(a_init, deadline, min_time=30.0):
    """
    L-BFGS with log-sum-exp smoothing (compact version).
    """
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[min(50, len(c_sorted)-1)] if len(c_sorted) > 50 else c_sorted[-1]
    base_beta = max(max_c - gap, max_c * 0.001)

    n_betas = 12
    log_betas = np.linspace(np.log(base_beta), max(np.log(max_c * 1e-6), -30), n_betas)
    betas = np.exp(log_betas)

    current_a = a_init.copy()
    current_sc = float(2.0 * n * max_c)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 3:
            break
        est_time = 0.004 * max(1.0, n / 8192.0)
        max_iter = min(3000, max(30, int(remaining / (est_time * 8))))

        try:
            def _obj(z_val):
                a = _softmax(z_val)
                c = _full_convolve(a, a)
                c_max = float(c.max())
                shifted = (c - c_max) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s = e.sum()
                if s < 1e-300 or not np.isfinite(s):
                    return 2.0 * n * c_max
                return 2.0 * n * (c_max + beta * np.log(s))

            def _grad(z_val):
                a = _softmax(z_val)
                c = _full_convolve(a, a)
                c_max = float(c.max())
                shifted = (c - c_max) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s = e.sum()
                if s < 1e-300 or not np.isfinite(s):
                    w = np.zeros_like(c)
                    w[int(np.argmax(c))] = 1.0
                else:
                    w = e / s
                h = _corr_a_A(a, w, n)
                W = float(np.dot(w, c))
                return 4.0 * n * a * (h - W)

            result = minimize(
                _obj, z, jac=_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'gtol': 1e-14, 'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            c_new = _full_convolve(a_new, a_new)
            sc_new = float(2.0 * n * c_new.max())
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _refinement_alternation(a, deadline, max_rounds=15):
    """
    Alternate between mirror descent (with self-consistent weights) and Newton.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = float(2.0 * n * _full_convolve(a, a).max())

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Newton step
        ntl_dl = min(deadline - 3, time.monotonic() + 20)
        a_n, sc_n = _newton_kkt(a, ntl_dl, max_outer=30)
        if sc_n < best_sc:
            best_sc = sc_n
            best_a = a_n.copy()
            a = a_n

        # Mirror descent step with self-consistent weights
        md_dl = min(deadline - 3, time.monotonic() + 15)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        w = _self_consistent_weights(a, c, c_max, max_iter=10)
        
        g = _gradient_at_active(a, w, n)
        g = g - np.mean(g)
        g_norm = np.linalg.norm(g) / np.sqrt(n)
        if g_norm > 1e-10:
            for factor in [0.01, 0.005, 0.002, 0.001]:
                h = factor / (1.0 + g_norm)
                a_try = a * np.exp(-h * g)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < c_max * (1.0 - 1e-10):
                    a = a_try
                    sc_try = 2.0 * n * f_try
                    if sc_try < best_sc:
                        best_sc = sc_try
                        best_a = a_try.copy()
                    break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: Quick L-BFGS start (60s)
    Phase 2: Mirror descent with continuation (500s)
    Phase 3: Newton + self-consistent refinement (400s)
    Phase 4: Diversified structured restarts (400s)
    Phase 5: High-precision Newton polish (300s)
    Phase 6: Final L-BFGS with tiny beta (140s)
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.12f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={len(parent)}, score={sc_parent:.12f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_init = 16384
        parent = np.ones(n_init) / n_init
        update_best(parent, _score(parent), "uniform_init")

    n = len(parent)

    # === Phase 1: Quick L-BFGS (60s) ===
    print(f"\n[Phase 1] Quick L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    p1_dl = min(deadline, time.monotonic() + 60)
    a_p1 = _lbfgs_logsumexp(parent, p1_dl, min_time=10.0)
    update_best(a_p1, None, "lbfgs_quick")

    # === Phase 2: Mirror descent with continuation (500s) ===
    print(f"\n[Phase 2] Mirror descent continuation (t={time.monotonic() - t_start:.1f}s)")
    p2_dl = min(deadline, time.monotonic() + 500)
    a_p2, sc_p2 = _mirror_descent_phase(best_seq, p2_dl)
    update_best(a_p2, sc_p2, "mirror_descent")

    # === Phase 3: Newton + self-consistent refinement (400s) ===
    print(f"\n[Phase 3] Newton + self-consistent (t={time.monotonic() - t_start:.1f}s)")
    p3_dl = min(deadline, time.monotonic() + 400)
    a_p3, sc_p3 = _refinement_alternation(best_seq, p3_dl, max_rounds=20)
    update_best(a_p3, sc_p3, "newton_sc")

    # === Phase 4: Diversified structured restarts (400s) ===
    print(f"\n[Phase 4] Diversified restarts (t={time.monotonic() - t_start:.1f}s)")
    p4_dl = min(deadline, time.monotonic() + 400)
    modes = ["swap", "block", "antisym", "gaussian", "swap", "antisym"]
    for i in range(min(12, len(modes) * 2)):
        if time.monotonic() > p4_dl:
            break
        mode = modes[i % len(modes)]
        perturbed = _structured_perturb(best_seq, rng, mode=mode)
        
        # Quick optimize the perturbed point
        sub_dl = min(p4_dl - (time.monotonic() - time.monotonic()), time.monotonic() + 30)
        a_sub = _lbfgs_logsumexp(perturbed, sub_dl, min_time=5.0)
        sc_sub = _score(a_sub)
        update_best(a_sub, sc_sub, f"restart_{mode}_{i}")
        
        # Also try Newton on it
        if time.monotonic() < p4_dl - 10:
            ntl_dl = min(p4_dl - 5, time.monotonic() + 15)
            a_ntl, sc_ntl = _newton_kkt(a_sub, ntl_dl, max_outer=20)
            update_best(a_ntl, sc_ntl, f"newton_{mode}_{i}")

    # === Phase 5: High-precision Newton polish (300s) ===
    print(f"\n[Phase 5] High-precision Newton (t={time.monotonic() - t_start:.1f}s)")
    p5_dl = min(deadline, time.monotonic() + 300)
    a_p5, sc_p5 = _newton_kkt(best_seq, p5_dl, max_outer=200)
    update_best(a_p5, sc_p5, "newton_precise")

    # === Phase 6: Final L-BFGS with tiny beta (remaining time) ===
    print(f"\n[Phase 6] Final L-BFGS polish (t={time.monotonic() - t_start:.1f}s)")
    p6_dl = deadline - 5
    if time.monotonic() < p6_dl:
        a_p6 = _lbfgs_logsumexp(best_seq, p6_dl, min_time=5.0)
        update_best(a_p6, None, "lbfgs_final")

    # Final: one more Newton pass
    if time.monotonic() < deadline - 3:
        final_dl = deadline - 1
        a_fin, sc_fin = _newton_kkt(best_seq, final_dl, max_outer=50)
        update_best(a_fin, sc_fin, "final_newton")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
