import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_raw(a):
    """Score assuming sum=1: just 2*n*max(conv)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _normalize(a):

    if s < 1e-15:

    return a / s


def _grad_maxconv(a, w):

    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _smoothed_obj_grad(z, beta, K, n):
    """Top-K log-sum-exp smoothed objective and gradient in log-domain."""
    z_max = z.max()
    a = np.exp(z - z_max)

    c = _full_convolve(a, a)

    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap * 0.8, max_c * 0.005)
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # Extended schedule with more gradual transitions
    schedule = [
        (1.5, 0.15), (1.0, 0.12), (0.8, 0.10), (0.6, 0.08),
        (0.4, 0.06), (0.3, 0.05), (0.2, 0.04), (0.15, 0.03),
        (0.1, 0.025), (0.07, 0.02), (0.05, 0.015), (0.03, 0.012),
        (0.02, 0.01), (0.01, 0.008), (0.005, 0.006), (0.003, 0.004),
        (0.001, 0.003), (0.0005, 0.002), (0.0002, 0.001),
    ]
    
    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 15:
            break
        
        beta = base_beta * beta_mult
        K = max(3, int((2 * n - 1) * K_frac))
        
        # Estimate time per function evaluation
        est_time = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(10000, max(100, int(remaining / (est_time * 30))))
        max_fun = min(50000, int(remaining / est_time))
        
        obj_fn = _make_smoothed_fn(beta, K, n)
        grad_fn = _make_smoothed_grad(beta, K, n)
        
        try:
            result = minimize(
                obj_fn, z, jac=grad_fn, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxcor': 100, 'maxls': 50, 'maxfun': max_fun}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if label:
                    print(f"    {label} beta={beta:.6f} K={K}: {sc_new:.12f}")
        except Exception as e:
            pass
    
    return current_a, current_sc


def _coordinate_descent(a_init, deadline, min_remaining):
    """Coordinate descent: optimize one element at a time via line search."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    current_sc = 2.0 * n * c_max
    
    no_improve = 0
    max_no_improve = 50
    
    # Process coordinates in order of their contribution to the max peak
    peak_k = int(np.argmax(c))
    contrib = np.zeros(n)
    i_lo = max(0, peak_k - (n - 1))
    i_hi = min(n - 1, peak_k)
    for i in range(i_lo, i_hi + 1):
        j = peak_k - i
        contrib[i] = a[j]
        if i != j:
            contrib[j] += a[i]
    
    order = np.argsort(contrib)[::-1]
    
    while time.monotonic() + min_remaining < deadline and no_improve < max_no_improve:
        improved_this_pass = False
        
        for idx in order:
            if time.monotonic() + min_remaining > deadline:
                break
            
            # Optimize a[idx] via parabolic interpolation / line search
            a_orig = a[idx]
            if a_orig < 1e-15:
                continue
            
            # Try several candidate values
            candidates = [a_orig * 0.5, a_orig * 0.7, a_orig * 0.85,
                         a_orig * 1.15, a_orig * 1.3, a_orig * 1.5]
            # Also try small absolute shifts
            step = a_orig * 0.1
            candidates.extend([max(0, a_orig - step), a_orig + step])
            
            best_val = a_orig
            best_local_sc = current_sc
            
            for cv in candidates:
                if cv < 0:
                    cv = 0.0
                a[idx] = cv
                s = a.sum()
                if s < 1e-15:
                    a[idx] = a_orig
                    continue
                a_norm = a / s
                c_try = _full_convolve(a_norm, a_norm)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < best_local_sc - 1e-15:
                    best_local_sc = sc_try
                    best_val = cv
            
            if best_val != a_orig:
                a[idx] = best_val
                a = _normalize(a)
                current_sc = best_local_sc
                improved_this_pass = True
                
                if current_sc < best_sc - 1e-15:
                    best_sc = current_sc
                    best_a = a.copy()
                
                # Update contributions
                c = _full_convolve(a, a)
                c_max = float(c.max())
                peak_k = int(np.argmax(c))
                contrib = np.zeros(n)
                i_lo = max(0, peak_k - (n - 1))
                i_hi = min(n - 1, peak_k)
                for i in range(i_lo, i_hi + 1):
                    j = peak_k - i
                    contrib[i] = a[j]
                    if i != j:
                        contrib[j] += a[i]
                order = np.argsort(contrib)[::-1]
        
        if improved_this_pass:
            no_improve = 0
        else:
            no_improve += 1
    
    return best_a, best_sc


def _peak_equalize(a_init, deadline, min_remaining):
    """ADMM-style peak equalization: try to make top peaks equal."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    # Target: make the top peaks as equal as possible
    # Strategy: iteratively identify the set of "active" peaks and
    # use gradient descent to flatten them
    
    tol = 0.02
    lr = 0.01
    momentum = 0.5
    velocity = np.zeros(n)
    stall = 0
    
    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                break
        
        # Active peaks: within tol of max
        active = c >= c_max * (1.0 - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx_max = int(np.argmax(c))
            active[idx_max] = True
            n_active = 1
        
        # Weight: emphasize peaks that are higher
        w = np.zeros(len(c))
        active_vals = c[active]
        w[active] = 1.0 + 10.0 * (active_vals - active_vals.min()) / max(active_vals.max() - active_vals.min(), 1e-15)
        w = w / w.sum()
        
        # Gradient step
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break
        
        # Momentum + line search
        velocity = momentum * velocity - lr * g_centered
        alpha = 1.0
        improved = False
        for _ in range(20):
            a_try = a + alpha * velocity
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                current_sc = sc_try
                improved = True
                break
            alpha *= 0.5
        
        if improved:
            lr = min(0.05, lr * 1.02)
            tol = max(0.001, tol * 0.97)
        else:
            velocity = np.zeros(n)
            lr = max(0.001, lr * 0.7)
            tol = min(0.05, tol * 1.2)
    
    return best_a, best_sc


def _fine_gradient_descent(a_init, deadline, min_remaining):
    """Very fine active-set gradient descent with adaptive step."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.002
    tol_frac = 0.002
    no_improve = 0
    prev_sc = best_sc
    
    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-16:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                a = best_a.copy()
                lr = max(1e-5, lr * 0.5)
                tol_frac = max(1e-5, tol_frac * 0.5)
                no_improve = 0
                if tol_frac < 1e-7 or lr < 1e-7:
                    break
        
        # Active set: peaks within tol_frac*c_max of the max
        threshold = c_max * (1.0 - tol_frac)
        active = c >= threshold
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break
        
        # Backtracking line search
        alpha = min(lr, 1.0 / (gnorm * gnorm + 1e-15))
        improved = False
        for _ in range(50):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-16:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if improved:
            if tol_frac > 1e-6:
                tol_frac = max(1e-6, tol_frac * 0.95)
            lr = min(0.01, lr * 1.01)
        else:
            no_improve += 5
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * t) + 0.15
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym2":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.3) ** 2 * 10) + 0.1
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.35) ** 2) + np.exp(-30.0 * (t - 0.65) ** 2) + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss_center":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "skew_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.4) ** 2) + 0.05 * np.exp(-4.0 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS coarse-to-fine, coordinate descent,
    peak equalization, fine gradient descent, multi-restart, and final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent if available
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass
    
    if parent is not None:
        n_main = len(parent)
        parent_sc = _score(parent)
        print(f"Parent: n={n_main}, score={parent_sc:.12f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_main = 16384
        best_seq = _make_profile(n_main, "tapered")
        best_score = _score(best_seq)
        print(f"No parent, tapered n={n_main}, score={best_score:.12f}")
    
    # === Phase 1: Main L-BFGS optimization ===
    print(f"\n[P1] L-BFGS coarse-to-fine (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = deadline - 700
    a1, sc1 = _run_lbfgs_phase(best_seq, p1_deadline, 100, "P1")
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.12f} [lbfgs1] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.12f}) t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 2: Peak equalization ===
    print(f"\n[P2] Peak equalization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        p2_deadline = min(deadline - 450, time.monotonic() + 300)
        a2, sc2 = _peak_equalize(best_seq, p2_deadline, 30)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.12f} [peak_eq] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.12f}) t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 3: Coordinate descent ===
    print(f"\n[P3] Coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        p3_deadline = min(deadline - 350, time.monotonic() + 200)
        a3, sc3 = _coordinate_descent(best_seq, p3_deadline, 30)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.12f} [coord] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.12f}) t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 4: Second L-BFGS from improved point ===
    print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        p4_deadline = min(deadline - 250, time.monotonic() + 200)
        a4, sc4 = _run_lbfgs_phase(best_seq, p4_deadline, 50, "P4")
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.12f} [lbfgs2] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 5: Fine gradient descent ===
    print(f"\n[P5] Fine gradient descent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        p5_deadline = min(deadline - 150, time.monotonic() + 150)
        a5, sc5 = _fine_gradient_descent(best_seq, p5_deadline, 20)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.12f} [fine_gd] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 6: Multi-restart ===
    print(f"\n[P6] Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 120:
        # Perturbed restarts
        sigmas = [0.02, 0.04, 0.06, 0.03, 0.05, 0.08, 0.01, 0.1]
        n_restarts = min(8, max(2, int((deadline - 120 - time.monotonic()) / 30)))
        for i in range(n_restarts):
            if time.monotonic() > deadline - 60:
                break
            sigma = sigmas[i % len(sigmas)]
            a_pert = _perturb(best_seq, sigma, rng)
            restart_deadline = min(deadline - 50, time.monotonic() + 30)
            a_r, sc_r = _run_lbfgs_phase(a_pert, restart_deadline, 5, f"P6r{i}")
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.12f} [restart_{i}] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 7: Try different profiles ===
    print(f"\n[P7] Profile exploration (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        n = len(best_seq)
        profiles = ["asym", "skew_gauss", "asym2", "cos_taper", "bimodal"]
        for prof_kind in profiles:
            if time.monotonic() > deadline - 50:
                break
            a_prof = _make_profile(n, prof_kind)
            prof_deadline = min(deadline - 40, time.monotonic() + 25)
            a_p, sc_p = _run_lbfgs_phase(a_prof, prof_deadline, 5, f"P7_{prof_kind}")
            if sc_p < best_score:
                best_seq = a_p.copy()
                best_score = sc_p
                print(f"  NEW BEST: {best_score:.12f} [profile_{prof_kind}] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 8: Multi-n exploration ===
    print(f"\n[P8] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        n_main = len(best_seq)
        n_options = [16384, 24576, 12288, 32768]
        n_options = [x for x in n_options if x != n_main]
        for n_try in n_options[:2]:
            if time.monotonic() > deadline - 30:
                break
            a_res = _resample(best_seq, n_try)
            n_deadline = min(deadline - 25, time.monotonic() + 35)
            a_opt, sc_opt = _run_lbfgs_phase(a_res, n_deadline, 5, f"P8_n{n_try}")
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.12f} [n={n_try}] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 9: Final fine polish ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        # Fine gradient descent
        fin_deadline = deadline - 3
        a_fin, sc_fin = _fine_gradient_descent(best_seq, fin_deadline, 2)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.12f} [final_gd] t={time.monotonic()-t_start:.1f}s")
        
        # Tiny L-BFGS
        remaining = deadline - 2 - time.monotonic()
        if remaining > 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = max(mb * 0.0003, 1e-10)
            K_fin = max(3, int(n * 0.001))
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            obj_fn = _make_smoothed_fn(beta_fin, K_fin, n)
            grad_fn = _make_smoothed_grad(beta_fin, K_fin, n)
            max_iter = min(5000, max(50, int(remaining / 0.02)))
            try:
                result = minimize(obj_fn, z, jac=grad_fn, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-20,
                                           'gtol': 1e-16, 'maxcor': 100})
                a_fin2 = np.exp(result.x - result.x.max())
                a_fin2 = a_fin2 / a_fin2.sum()
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.12f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass
    
    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
