def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _cg_solve_hankel(y_rev_fft, N, rhs, n, mu, max_cg=300):
    """Solve (Y + mu*I) x = rhs using CG, where (Yv)[j] = sum_m y[j+m]*v[m]."""
    x = np.zeros(n)

    p = rhs.copy()

    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-15:
        return x
    for it in range(max_cg):
        fp = np.fft.rfft(p, N)
        rtmp = np.fft.irfft(y_rev_fft * fp, N)
        Ar = rtmp[n - 1:2 * n - 1][::-1] + mu * p
        pAr = np.dot(p, Ar)
        if pAr < 1e-30:

        alpha = rs / pAr
        x = x + alpha * p
        r = r - alpha * Ar
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < 1e-12 * rhs_norm:

        p = r + (rs_new / rs) * p
        rs = rs_new

    y is uniform on set_indices.
    Returns the solution a_new or None on failure.

    y[set_indices] = 1.0 / len(set_indices)
    y_rev = y[::-1]
    # FFT size for convolution of y_rev (len 2n-1) with v (len n): output len 3n-1

    if w_sum < 1e-10 or not np.isfinite(w_sum):

    a_new = w / w_sum
    # Clip small negatives

        s = a_new.sum()
        if s > 1e-10:

            c = _full_convolve(a, a)
            c_max = float(c.max())
            if c_max < 1e-15:
                break
            # Active set
            threshold = c_max * (1.0 - tol_frac)
            active = np.where(c >= threshold)[0]
            m = len(active)
            if m < 3:
                active = np.argsort(c)[::-1][:10]
                m = 10
            if m > 50 * n:
                active = np.argsort(c)[::-1][:min(m, 50 * n)]
                m = len(active)

            # Solve QP
            a_qp = _solve_qp_active(active, n, mu=mu, max_cg=300)
            if a_qp is None:
                # Try with more regularization
                a_qp = _solve_qp_active(active, n, mu=mu * 100, max_cg=300)
            if a_qp is None:
                break

            # Line search: a_try = (1-t)*a + t*a_qp
            found = False
            for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
                a_try = (1.0 - t) * a + t * a_qp
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-15:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    found = True
                    break
            if not found:
                break

            # Check convergence
            c_check = _full_convolve(a, a)
            if abs(float(c_check.max()) - c_max) < 1e-14:
                break

    return best_a, best_sc


def _lsobj(z, beta, n, c_max_ref):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # Gradient of (w . c) w.r.t. a[i] = sum_k w[k] * 2*a[k-i]
    # = 2 * (w_rev * a)[...] using the Hankel structure
    w_rev = w[::-1]
    N = _next_pow2(len(w_rev) + n - 1)
    fw = np.fft.rfft(w_rev, N)
    fa = np.fft.rfft(a, N)
    conv = np.fft.irfft(fw * fa, N)[:len(w_rev) + n - 1]
    # (w_rev * a)[2n-2-i] gives sum_k w[k]*a[k-i]
    # conv has length (2n-1)+(n)-1 = 3n-1, we want indices n-1 to 2n-2 reversed
    h = conv[n - 1:2 * n - 1][::-1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_polish(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with log-sum-exp continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 30:
        gap = c_sorted[0] - c_sorted[30]
    else:
        gap = c_sorted[0] - c_sorted[-1]
    base_beta = max(gap * 0.5, max_c * 0.005)

    n_betas = 20
    log_betas = np.linspace(np.log(base_beta), np.log(max(max_c * 1e-7, 1e-30)), n_betas)
    betas = np.exp(log_betas)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.004 * max(1.0, n / 8192.0)
        max_iter = min(3000, max(50, int(remaining / (est_time * 8))))
        try:
            result = minimize(
                _lsobj, z, args=(beta, n, max_c),
                jac=_lsgrad, args=(beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.12f}")
        except Exception as ex:
            print(f"    L-BFGS beta={beta:.2e}: exception {ex}")
    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _deflation_step(a, deadline, n_attempts=30):
    """Multi-peak deflation: reduce all significant peaks simultaneously."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        threshold = c_max * 0.995
        peak_mask = c >= threshold
        if not peak_mask.any():
            peak_mask[np.argmax(c)] = True
        w = np.zeros_like(c)
        w[peak_mask] = 1.0 / peak_mask.sum()
        # Gradient of sum_k w[k]*c[k] w.r.t. a[i] = 2 * sum_k w[k]*a[k-i]
        w_rev = w[::-1]
        N = _next_pow2(len(w_rev) + n - 1)
        fw = np.fft.rfft(w_rev, N)
        fa = np.fft.rfft(a, N)
        conv = np.fft.irfft(fw * fa, N)[:len(w_rev) + n - 1]
        grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
        g_t = grad - np.mean(grad)

        applied = False
        for strength in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
            a_try = a - strength * g_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                applied = True
                break
        if not applied:
            break
    return best_a, best_sc
