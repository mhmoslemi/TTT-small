    k = 1
    while k < n:

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) structured initialization, (2) aggressive L-BFGS with log-sum-exp
smoothing and very fine temperature schedule, (3) active-set peak equalization
via fixed-point iteration, (4) multi-n exploration, (5) subgradient polishing.
Key innovations: geometric temperature sweep to 1e-9, active-set Newton
iteration, and diversified random restarts.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _convolve_w_a(w, a):
    """Compute g[i] = sum_j w[i+j]*a[j] for i=0..n-1 using FFT."""
    n = len(a)
    a_rev = a[::-1]
    N = _next_pow2(3 * n)
    wf = np.fft.rfft(w, N)
    arf = np.fft.rfft(a_rev, N)
    conv_result = np.fft.irfft(wf * arf, N)
    return conv_result[n - 1:2 * n - 1]


def _smooth_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smooth_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    h = _convolve_w_a(w, a)
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smooth_obj(z, self.beta, self.n)

    def grad(self, z):
        return _smooth_grad(z, self.beta, self.n)


def _run_lbfgs_aggressive(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with aggressive geometric temperature schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

    # Very fine geometric schedule - go much lower than before
    ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
                       0.015625, 0.0078125, 0.00390625, 0.001953125,
                       0.0009765625, 0.00048828125, 0.000244140625,
                       0.0001220703125, 0.00006103515625,
                       0.000030517578125, 0.0000152587890625,
                       0.00000762939453125, 0.000003814697265625])
    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # More iterations for smaller beta (finer optimization)
        if i < len(betas) // 3:
            max_iter = min(3000, max(100, int(remaining / 0.02)))
        elif i < 2 * len(betas) // 3:
            max_iter = min(5000, max(200, int(remaining / 0.03)))
        else:
            max_iter = min(8000, max(500, int(remaining / 0.04)))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 100,
                    'maxls': 80,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS beta={beta:.2e} (idx={i}): score={current_sc:.12f}")
        except Exception:
            pass

    return current_a


def _active_set_equalize(a_init, deadline, min_time_remaining=30.0):
    """Iteratively equalize top K peaks using fixed-point / Newton ideas."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    K = max(5, min(50, int(n * 0.003)))
    lr = 0.01
    no_improve = 0
    stall_count = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            stall_count += 1
            if stall_count > 15:
                break
            lr *= 0.5
            if lr < 1e-6:
                break
            no_improve = 0

        # Select active peaks (top K)
        c_idx = np.argsort(c)[::-1]
        active_k = c_idx[:K]

        # Compute weighted gradient (soft subgradient)
        active_vals = c[active_k]
        spread = active_vals.max() - active_vals.min()
        if spread < 1e-15:
            # Peaks are already equalized, try a different K
            K = min(K * 2, 200)
            lr *= 0.7
            continue

        # Temperature for weighting
        temp = max(spread * 0.1, c_max * 1e-6)
        w_active = np.exp((active_vals - active_vals.max()) / temp)
        w_active = w_active / w_active.sum()

        # Compute the soft gradient direction
        w_full = np.zeros(len(c))
        w_full[active_k] = w_active
        h = _convolve_w_a(w_full, a)
        W = float(np.dot(w_full, c))
        grad = 4.0 * n * a * (h - W)

        # Project to simplex tangent (sum = 0)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break

        # Step with backtracking
        step_size = lr / (gnorm + 1e-18)
        a_cand = a - step_size * grad_t
        a_proj = _project_simplex(a_cand)

        c_new = _full_convolve(a_proj, a_proj)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < current_sc:
            a = a_proj
        else:
            # Try smaller step
            a_cand = a - 0.1 * step_size * grad_t
            a_proj = _project_simplex(a_cand)
            c_new = _full_convolve(a_proj, a_proj)
            sc_new = 2.0 * n * float(c_new.max())
            if sc_new < current_sc:
                a = a_proj
            else:
                lr *= 0.7

        if no_improve % 1000 == 0 and no_improve > 0:
            print(f"    Active-set: it={no_improve}, best={best_sc:.12f}, K={K}, lr={lr:.6f}")

    return best_a, best_sc


def _subgradient_polish(a_init, deadline, min_time_remaining=20.0):
    """Subgradient method with Nesterov momentum for final polishing."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    v = np.zeros(n)
    momentum = 0.9
    t = 0
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5000:
                break

        # Use top-5 peaks for a softer subgradient
        K = 5
        c_idx = np.argsort(c)[::-1]
        active_k = c_idx[:K]
        active_vals = c[active_k]
        spread = active_vals.max() - active_vals.min()
        temp = max(spread * 0.05, c_max * 1e-7)
        w_act = np.exp((active_vals - active_vals.max()) / temp)
        w_act = w_act / w_act.sum()
        w_full = np.zeros(len(c))
        w_full[active_k] = w_act

        h = _convolve_w_a(w_full, a)
        W = float(np.dot(w_full, c))
        grad = 4.0 * n * a * (h - W)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break

        # Nesterov step with adaptive lr
        base_lr = 0.002 / (1.0 + 0.01 * t)
        lr = base_lr / (gnorm + 1e-18) * 0.5
        v = momentum * v - lr * grad_t
        a_cand = a + v
        a = _project_simplex(a_cand)

        if t % 10000 == 0:
            print(f"    Subgrad t={t}: best={best_sc:.12f}, v_norm={np.linalg.norm(v):.6f}")

    return best_a, best_sc


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * t / 2) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian_center":
        t = np.linspace(-1, 1, n)
        a = np.exp(-2.0 * t ** 2)
        return a / a.sum()
    elif kind == "random":
        if rng is None:
            rng = np.random.default_rng(0)
        a = rng.random(n) + 0.1
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    elif kind == "flat_taper":
        a = np.ones(n)
        taper = max(1, n // 30)
        for i in range(taper):
            a[i] = (i + 1) / taper
            a[n - 1 - i] = (i + 1) / taper
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, np.maximum(a, 0))
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

    sc_main2 = _score(a_main2)
    if sc_main2 < best_score:
        best_score = sc_main2
        best_seq = a_main2.copy()
        print(f"  NEW BEST [lbfgs2]: {best_score:.12f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [8192, 16384, 32768]
    if n_base not in n_candidates:
        n_candidates.insert(0, n_base)

    p5_deadline = p4_deadline
    for n_try in n_candidates:
        if time.monotonic() > p5_deadline - 30:
            break
        if n_try == len(best_seq):
            continue
        a_resamp = _resample(best_seq, n_try)
        sc_resamp = _score(a_resamp)
        print(f"  n={n_try}: resampled score={sc_resamp:.12f}")

        if sc_resamp < best_score + 0.05:
            # Worth optimizing
            a_opt_n = _run_lbfgs_aggressive(a_resamp, p5_deadline, min_time_remaining=30.0)
            sc_opt_n = _score(a_opt_n)
            if sc_opt_n < best_score:
                best_score = sc_opt_n
                best_seq = a_opt_n.copy()
                print(f"  NEW BEST [n={n_try}]: {best_score:.12f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Subgradient polishing ===
    print(f"\n[P6] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    p6_deadline = min(deadline - 40, t_start + 1550)
    a_sg, sc_sg = _subgradient_polish(best_seq, p6_deadline, min_time_remaining=30.0)
    if sc_sg < best_score:
        best_score = sc_sg
        best_seq = a_sg.copy()
        print(f"  NEW BEST [subgrad]: {best_score:.12f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final ultra-fine L-BFGS ===
    print(f"\n[P7] Final L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 10 - time.monotonic()
    if remaining > 30:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        c_sorted = np.sort(c_best)[::-1]
        gap = mb - c_sorted[min(20, len(c_sorted) - 1)]
        beta_fin = max(gap * 0.005, mb * 1e-6)

        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        prob = _LbfgsProb(beta_fin, n)
        max_iter = min(5000, max(100, int(remaining / 0.05)))
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxcor': 80, 'maxls': 60})
            a_fin = _softmax(result.x)
            sc_fin = _score(a_fin)
            if sc_fin < best_score:
                best_score = sc_fin
                best_seq = a_fin.copy()
                print(f"  NEW BEST [final_lbfgs]: {best_score:.12f} t={time.monotonic()-t_start:.1f}s")
        except Exception:
            pass

    # Random perturbations for final attempt
    remaining = deadline - 5 - time.monotonic()
    if remaining > 20:
        for attempt in range(3):
            if remaining < 10:
                break
            perturb = 0.0005 * rng.standard_normal(len(best_seq))
            a_pert = _project_simplex(best_seq + perturb)
            z_pert = np.log(np.maximum(a_pert, 1e-30))
            z_pert = z_pert - z_pert.mean()

            c_pert = _full_convolve(a_pert, a_pert)
            mb_pert = float(c_pert.max())
            c_s = np.sort(c_pert)[::-1]
            gap_p = mb_pert - c_s[min(10, len(c_s) - 1)]
            beta_p = max(gap_p * 0.01, mb_pert * 1e-5)

            prob = _LbfgsProb(beta_p, len(best_seq))
            max_it = min(2000, max(50, int(remaining / 0.1)))
            try:
                result = minimize(prob, z_pert, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_it, 'ftol': 1e-20,
                                           'gtol': 1e-15, 'maxcor': 50})
                a_r = _softmax(result.x)
                sc_r = _score(a_r)
                if sc_r < best_score:
                    best_score = sc_r
                    best_seq = a_r.copy()
                    print(f"  NEW BEST [perturb_{attempt}]: {best_score:.12f}")
            except Exception:
                pass
            remaining = deadline - 5 - time.monotonic()

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
