"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
- Mirror descent (exponentiated gradient) with adaptive active set on the simplex
- Frank-Wolfe (conditional gradient) for robustness
- L-BFGS on smoothed max via log-sum-exp for polishing
- Multi-n exploration with resampling
- Binomial and tapered initializations to escape local minima
- Dual-guided step direction for faster convergence
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming a is already normalized to sum=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _conv_subgrad(a, active_k):
    """Compute subgradient sum_{k in active} dh_k/da, where h_k = conv(a,a)[k]."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_k:
        # dh_k/da[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
        # First term: a[k-i] for i in [max(0,k-n+1), min(n-1,k)]
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += a[k - np.arange(i_lo, i_hi + 1)]
        # Second term: a[i-k] for i in [max(0,k), min(n-1, k+n-1)]
        i_lo2 = max(0, k)
        i_hi2 = min(n - 1, k + n - 1)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2 + 1] += a[np.arange(i_lo2, i_hi2 + 1) - k]
    return grad


def _make_binomial_init(n):
    """Initialize with central binomial coefficient shape: C(2j,j)/4^j."""
    j = np.arange(n)
    # Use log to avoid overflow: log(C(2j,j)) - j*log(4)
    log_vals = np.zeros(n)
    for i in range(n):
        # log(C(2i,i)) = sum_{k=1}^{i} log(2i+1-k) - log(k) approx using gammaln
        log_vals[i] = _log_binom(2 * i, i) - i * np.log(4.0)
    a = np.exp(log_vals - log_vals.max())
    a = a / a.sum()
    return a


def _log_binom(n, k):
    """Compute log of binomial coefficient n choose k."""
    if k < 0 or k > n:
        return -np.inf
    k = min(k, n - k)
    if k == 0:
        return 0.0
    result = 0.0
    for i in range(k):
        result += np.log(n - i) - np.log(i + 1)
    return result


def _make_tapered_init(n, alpha=1.0):
    """Initialize with a tapered shape: sin(pi*x)^alpha."""
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** alpha
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_uniform_init(n):
    return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _mirror_descent(a_init, deadline, min_time_remaining=30.0, K_init=5,
                    K_max=200, eta_init=0.01, seed=0):
    """Mirror descent (exponentiated gradient) on the simplex."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    a = np.maximum(a, 1e-30)
    a = a / a.sum()

    best_a = a.copy()
    c = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c.max())

    rng = np.random.default_rng(seed)
    w = np.log(a)
    w = w - w.mean()  # center for numerical stability
    eta = eta_init
    K = K_init
    no_improve = 0
    t = 0

    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            # Slightly increase eta on improvement
            eta = min(eta * 1.02, 0.1)
        else:
            no_improve += 1

        # Adaptive K
        if t % 100 == 0:
            c_sorted = np.sort(c)[::-1]
            gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
            if gap < c_max * 0.003 and K < K_max:
                K = min(K_max, int(K * 1.2) + 1)

        # Select active peaks
        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]

        # Compute subgradient
        grad = _conv_subgrad(a, active)

        # Mirror descent update: w_j -= eta * a_j * (g_j - <a,g>)
        a_dot_g = float(np.dot(a, grad))
        mirror_grad = a * (grad - a_dot_g)
        w = w - eta * mirror_grad

        # Renormalize w for stability
        if t % 100 == 0:
            w = w - w.mean()

        a_new = _softmax(w)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < current_sc + 1e-12:
            a = a_new
        else:
            # Backtrack: halve eta and try again
            eta *= 0.5
            w = np.log(a)
            w = w - w.mean()
            if eta < 1e-8:
                break

        # Periodic restart
        if no_improve % 100 == 0 and no_improve > 0:
            if no_improve > 500:
                break
            # Small perturbation in w-space
            w = w + 0.001 * rng.standard_normal(n)
            w = w - w.mean()
            a = _softmax(w)
            a = np.maximum(a, 1e-30)
            a = a / a.sum()
            eta = eta_init * 0.5
            no_improve = 0

        if t % 2000 == 0:
            print(f"      MD t={t}: best={best_sc:.10f}, cur={sc_new:.10f}, K={K}, eta={eta:.6f}")

    return best_a, best_sc


def _frank_wolfe(a_init, deadline, min_time_remaining=30.0, K_init=5, K_max=200, seed=0):
    """Frank-Wolfe (conditional gradient) method for the simplex."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s

    best_a = a.copy()
    c = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c.max())

    rng = np.random.default_rng(seed)
    K = K_init
    no_improve = 0
    t = 0
    gamma = 1.0

    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            break

        # Adaptive K
        if t % 100 == 0:
            c_sorted = np.sort(c)[::-1]
            gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
            if gap < c_max * 0.003 and K < K_max:
                K = min(K_max, int(K * 1.2) + 1)

        # Active set
        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]

        # Subgradient
        grad = _conv_subgrad(a, active)

        # Frank-Wolfe: s = e_{argmin grad_j}
        j_min = int(np.argmin(grad))
        s_vec = np.zeros(n)
        s_vec[j_min] = 1.0

        # Update with diminishing step size
        gamma = 2.0 / (t + 2.0)
        a_new = (1.0 - gamma) * a + gamma * s_vec
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        a = a_new

        if t % 5000 == 0:
            sc_now = 2.0 * n * float(_full_convolve(a, a).max())
            print(f"      FW t={t}: best={best_sc:.10f}, cur={sc_now:.10f}, K={K}")

    return best_a, best_sc


def _smoothed_obj_grad(z, beta, n):
    """Compute smoothed objective and gradient for L-BFGS."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max, None
    obj = 2.0 * n * (c_max + beta * np.log(s))

    # Gradient
    w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 4.0 * n * a * (h - W)
    # Mirror: grad_z = a * (grad_a - <a, grad_a>)
    a_dot_g = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - a_dot_g)
    return obj, grad_z


def _lbfgs_polish(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS on smoothed max objective with adaptive beta."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = max_c - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.008, 0.004,
                                  0.002, 0.001, 0.0005])

    current_a = a_init.copy()
    current_sc = 2.0 * n * max_c
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # Estimate iterations
        est_time = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 10))))

        def fun(z_inner):
            val, _ = _smoothed_obj_grad(z_inner, beta, n)
            return val

        def jac(z_inner):
            _, g = _smoothed_obj_grad(z_inner, beta, n)
            return g

        try:
            result = minimize(fun, z, jac=jac, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-16,
                                       'gtol': 1e-14, 'maxcor': 60, 'maxls': 40})
            a_new = _softmax(result.x)
            sc_new = 2.0 * n * float(_full_convolve(a_new, a_new).max())
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 3 == 0:
                    print(f"      LBFGS beta_idx={i}: score={current_sc:.10f}")
        except Exception:
            pass

    return current_a


def _dual_guide_descent(a_init, deadline, min_time_remaining=30.0, K_init=10,
                        K_max=300, eta_init=0.005, seed=0):
    """
    Descent method guided by dual weights. At each step, find lambda_k for active
    peaks that makes the gradient most uniform, then take a projected gradient step.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s

    best_a = a.copy()
    c = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c.max())

    rng = np.random.default_rng(seed)
    eta = eta_init
    K = K_init
    no_improve = 0
    t = 0
    v = np.zeros(n)  # momentum
    momentum = 0.7

    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            eta = min(eta * 1.01, 0.05)
        else:
            no_improve += 1

        if no_improve > 400:
            break

        # Adaptive K
        if t % 50 == 0:
            c_sorted = np.sort(c)[::-1]
            gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
            if gap < c_max * 0.002 and K < K_max:
                K = min(K_max, int(K * 1.15) + 1)

        # Active set: top K peaks
        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]

        # Compute subgradient with uniform weights
        grad = _conv_subgrad(a, active)

        # Project gradient to tangent space of simplex
        grad_t = grad - np.mean(grad)

        # Momentum step
        v = momentum * v - eta * grad_t
        a_cand = a + v
        a_proj = _project_simplex(a_cand)

        c_new = _full_convolve(a_proj, a_proj)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < current_sc + 1e-12:
            a = a_proj
        else:
            eta *= 0.7
            v = 0.5 * v
            # Still move to a_proj but with smaller effective step
            a = (1 - 0.5) * a + 0.5 * a_proj
            a = _project_simplex(a)

        # Restart
        if no_improve % 150 == 0 and no_improve > 0:
            if no_improve > 600:
                break
            perturb = 0.002 * rng.standard_normal(n)
            a_pert = _project_simplex(a + perturb)
            c_pert = _full_convolve(a_pert, a_pert)
            sc_pert = 2.0 * n * float(c_pert.max())
            if sc_pert < best_sc:
                a = a_pert
                best_a = a.copy()
                best_sc = sc_pert
            v = np.zeros(n)
            eta = eta_init
            no_improve = 0

        if t % 2000 == 0:
            print(f"      DG t={t}: best={best_sc:.10f}, cur={sc_new:.10f}, K={K}, eta={eta:.6f}")

    return best_a, best_sc


def _gmd_combination(a1, a2, deadline, min_time_remaining=30.0):
    """Combine two solutions by geometric mean and re-optimize."""
    n1, n2 = len(a1), len(a2)
    if n1 != n2:
        a2_r = _resample(a2, n1)
        n2 = n1
    # Geometric mean
    a_combined = np.sqrt(np.maximum(a1, 1e-30) * np.maximum(a2_r, 1e-30))
    a_combined = a_combined / a_combined.sum()
    sc = _score_normalized(a_combined)
    return a_combined, sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function: hybrid mirror descent + Frank-Wolfe + L-BFGS
    with multi-n exploration and diverse initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                parent_sc = _score_normalized(parent)
                best_seq = parent.copy()
                best_score = parent_sc
                print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_uniform_init(n_parent)
        best_seq = parent.copy()
        best_score = _score_normalized(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # ================================================================
    # Phase 1: Dual-guided descent on parent
    # ================================================================
    print(f"\n[P1] Dual-guided descent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        p1_dead = min(deadline - 600, t_start + 300)
        a1, sc1 = _dual_guide_descent(best_seq, p1_dead, min_time_remaining=30,
                                      K_init=10, K_max=200, eta_init=0.005, seed=seed)
        if sc1 < best_score:
            best_seq = a1.copy()
            best_score = sc1
            print(f"  NEW BEST: {best_score:.10f} [dual_guide] t={time.monotonic()-t_start:.1f}s")

    # ================================================================
    # Phase 2: Mirror descent
    # ================================================================
    print(f"\n[P2] Mirror descent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        p2_dead = min(deadline - 500, t_start + 400)
        a2, sc2 = _mirror_descent(best_seq, p2_dead, min_time_remaining=30,
                                  K_init=5, K_max=200, eta_init=0.01, seed=seed + 1)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [mirror] t={time.monotonic()-t_start:.1f}s")

    # ================================================================
    # Phase 3: L-BFGS polish
    # ================================================================
    print(f"\n[P3] L-BFGS polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        p3_dead = min(deadline - 400, t_start + 500)
        a3 = _lbfgs_polish(best_seq, p3_dead, min_time_remaining=30)
        sc3 = _score_normalized(a3)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [lbfgs] t={time.monotonic()-t_start:.1f}s")

    # ================================================================
    # Phase 4: Try binomial initialization
    # ================================================================
    print(f"\n[P4] Binomial init exploration (t={time.monotonic()-t_start:.1f}s)")
    n_bin = n_parent
    a_bin = _make_binomial_init(n_bin)
    sc_bin = _score_normalized(a_bin)
    print(f"  Binomial init n={n_bin}: score={sc_bin:.10f}")

    if sc_bin < best_score:
        best_seq = a_bin.copy()
        best_score = sc_bin
        print(f"  NEW BEST: {best_score:.10f} [binomial_init] t={time.monotonic()-t_start:.1f}s")

    if time.monotonic() < deadline - 300:
        p4_dead = min(deadline - 300, t_start + 600)
        a4, sc4 = _dual_guide_descent(a_bin, p4_dead, min_time_remaining=30,
                                      K_init=10, K_max=200, eta_init=0.005, seed=seed + 2)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [binomial_opt] t={time.monotonic()-t_start:.1f}s")

        if time.monotonic() < deadline - 250:
            a4b = _lbfgs_polish(best_seq, deadline, min_time_remaining=30)
            sc4b = _score_normalized(a4b)
            if sc4b < best_score:
                best_seq = a4b.copy()
                best_score = sc4b
                print(f"  NEW BEST: {best_score:.10f} [binomial_lbfgs] t={time.monotonic()-t_start:.1f}s")

    # ================================================================
    # Phase 5: Tapered init exploration
    # ================================================================
    print(f"\n[P5] Tapered init (t={time.monotonic()-t_start:.1f}s)")
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        if time.monotonic() > deadline - 200:
            break
        a_tap = _make_tapered_init(n_parent, alpha)
        sc_tap = _score_normalized(a_tap)
        print(f"  Tapered alpha={alpha}: score={sc_tap:.10f}")
        if sc_tap < best_score:
            best_seq = a_tap.copy()
            best_score = sc_tap

        if time.monotonic() < deadline - 200:
            p5_dead = min(deadline - 200, t_start + 200 + 50 * alpha)
            a5, sc5 = _mirror_descent(a_tap, p5_dead, min_time_remaining=20,
                                      K_init=5, K_max=150, eta_init=0.01, seed=seed + int(alpha * 10))
            if sc5 < best_score:
                best_seq = a5.copy()
                best_score = sc5
                print(f"  NEW BEST: {best_score:.10f} [tapered_{alpha}] t={time.monotonic()-t_start:.1f}s")

    # ================================================================
    # Phase 6: Multi-n exploration
    # ================================================================
    print(f"\n[P6] Multi-n exploration (t={time.mont-onic()-t_start:.1f}s)" if False else
          f"\n[P6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = []
    for n_try in [8192, 12288, 16384, 20480, 24576, 32768]:
        if n_try != n_parent:
            n_candidates.append(n_try)
    n_candidates = n_candidates[:4]  # Limit to 4

    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        a_res = _resample(best_seq, n_try)
        sc_res = _score_normalized(a_res)
        print(f"  n={n_try}: resampled score={sc_res:.10f}")

        if time.monotonic() < deadline - 100:
            p6_dead = min(deadline - 100, t_start + 100 + (n_try // 4096) * 50)
            a6, sc6 = _dual_guide_descent(a_res, p6_dead, min_time_remaining=20,
                                          K_init=5, K_max=100, eta_init=0.003, seed=seed + n_try)
            if sc6 < best_score:
                best_seq = a6.copy()
                best_score = sc6
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")

            if time.monotonic() < deadline - 60:
                a6b = _lbfgs_polish(best_seq if len(best_seq) == n_try else a6,
                                    deadline, min_time_remaining=20)
                sc6b = _score_normalized(a6b)
                if sc6b < best_score:
                    best_seq = a6b.copy()
                    best_score = sc6b
                    print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_lbfgs] t={time.monotonic()-t_start:.1f}s")

    # ================================================================
    # Phase 7: Final L-BFGS polish on best
    # ================================================================
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        a7 = _lbfgs_polish(best_seq, deadline, min_time_remaining=10)
        sc7 = _score_normalized(a7)
        if sc7 < best_score:
            best_seq = a7.copy()
            best_score = sc7
            print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")

    # ================================================================
    # Phase 8: Final mirror descent squeeze
    # ================================================================
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a8, sc8 = _mirror_descent(best_seq, deadline - 5, min_time_remaining=5,
                                  K_init=3, K_max=50, eta_init=0.003, seed=seed + 999)
        if sc8 < best_score:
            best_seq = a8.copy()
            best_score = sc8
            print(f"  NEW BEST: {best_score:.10f} [final_mirror] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_uniform_init(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_uniform_init(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
