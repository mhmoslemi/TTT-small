    if s < 1e-15:

    c = _full_convolve(a, a)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:

    return a / s

    c_pos = np.maximum(c, 0.0)
    if p == np.inf:
        return float(c_pos.max())
    return float(np.sum(c_pos ** p))

    """
    Gradient of sum(max(c,0)^p) / (sum(a))^2 w.r.t. a,
    where c = conv(a,a).

    gradient of sum(c^p) in the simplex tangent space.
    """

    c_pos = np.maximum(c, 0.0)

    elif p == np.inf:

        w /= w.sum()
    else:

def _make_profile(n, kind, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    t = np.linspace(0, 1, n)
    
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'power':
        alpha = rng.uniform(0.5, 3.0)
        a = (1 - t + 1e-10) ** alpha
        return a / a.sum()
    elif kind == 'exp':
        beta = rng.uniform(0.5, 5.0)
        a = np.exp(-beta * t)
        return a / a.sum()
    elif kind == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == 'hann_shift':
        shift = rng.uniform(0, 1)
        a = 0.5 * (1 + np.cos(2 * np.pi * (t - shift)))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == 'gauss':
        mu = rng.uniform(0.2, 0.8)
        sigma = rng.uniform(0.1, 0.3)
        a = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
        return a / a.sum()
    elif kind == 'ramp':
        a = 1.0 - t
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == 'ramp2':
        a = (1 - t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == 'random':
        a = np.maximum(rng.random(n), 1e-10)
        return a / a.sum()
    elif kind == 'kaiser':
        beta = rng.uniform(2, 10)
        a = _kaiser_window(n, beta)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _kaiser_window(n, beta):
    """Kaiser window."""
    from scipy.special import i0
    m = np.arange(n)
    alpha = (n - 1) / 2.0
    r = (m - alpha) / alpha
    r = np.clip(r, -1, 1)
    return i0(beta * np.sqrt(1 - r**2)) / i0(beta)

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _targeted_perturb(a, c, rng, strength=0.05):
    """
    Perturb a in a way that specifically targets reducing the peak
    of the convolution.
    """
    n = len(a)
    c_max = c.max()
    peak_idx = np.where(c >= c_max - 0.05 * c_max)[0]
    
    a_new = a.copy()
    
    # For each peak position k, the contribution to c[k] from a[i] is a[k-i]
    # To reduce c[k], we should decrease a[i] where a[k-i] is large
    # This is a soft heuristic: decrease a more where it contributes to peaks
    
    # Compute "pressure" on each a[i]
    pressure = np.zeros(n)
    for k in peak_idx:
        # c[k] = sum_j a[j] * a[k-j]
        # d c[k] / d a[i] = a[k-i] (for valid k-i)
        # We want to decrease a[i] proportional to a[k-i]
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                pressure[i] += a[j]
    
    pressure /= max(1, len(peak_idx))
    pnorm = np.linalg.norm(pressure)
    if pnorm > 1e-10:
        pressure /= pnorm
    
    # Apply perturbation: decrease in pressure direction + random
    noise = rng.standard_normal(n)
    a_new = a - strength * pressure + 0.01 * noise
    a_new = _normalize(a_new)
    return a_new

def _momentum_descent(a_init, deadline, min_time_remaining=30, 
                       use_lp=True, p_start=4, p_end=64):
    """
    Momentum-based gradient descent on the smoothed objective.
    Uses Lp norm with increasing p, then switches to active-set.
    """
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    v = np.zeros(n)
    momentum = 0.9
    lr = 0.05
    no_improve = 0
    max_stall = 200
    
    p = p_start
    tol_frac = 0.05
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _conv(a)
        c_max = float(c.max())
        sc_current = 2.0 * n * c_max  # with sum(a)=1
        
        if sc_current < best_sc - 1e-14:
            best_sc = sc_current
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute gradient
        if use_lp and p < p_end:
            grad = _lp_gradient(a, p)
        else:
            # Active-set gradient
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            w = np.zeros_like(c)
            eps = 1e-12
            w[active] = 1.0 / (c_max - c[active] + eps)
            w = w / w.sum()
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1 : 2 * n - 1]
            grad = 2.0 * h
            grad = grad - np.mean(grad)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Momentum update
        v = momentum * v - lr * grad
        
        # Backtracking line search
        alpha = 1.0
        improved = False
        for _ in range(40):
            a_try = _normalize(a + alpha * v)
            sc_try = _score(a_try)
            if sc_try < best_sc * (1 - 1e-10) or sc_try < sc_current * (1 - 1e-12):
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            v = np.zeros(n)
            lr *= 0.7
            momentum = min(0.95, momentum + 0.01)
            if no_improve % 50 == 0:
                tol_frac = max(0.001, tol_frac * 0.7)
                p = min(p_end, p * 1.5)
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                lr = 0.02
                momentum = 0.85
                tol_frac = 0.02
                p = p_start
                no_improve = 0
        else:
            if p < p_end:
                p = min(p_end, p * 1.02)
            tol_frac = max(0.001, tol_frac * 0.99)
            lr = min(0.1, lr * 1.005)
    
    return best_a, best_sc

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Lp-annealing, Nesterov momentum, informed ILS, and multi-n search.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _conv(a):
    return _full_convolve(a, a)

def _grad_max_weighted(a, c, w):
    """Compute gradient: 2 * conv(w, a_rev)[n-1:2n-1], projected to tangent."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1 : 2 * n - 1]
    grad = 2.0 * h
    grad = grad - np.mean(grad)
    return grad

def _lp_grad(a, p):
    """Gradient of sum(c^p) where c = conv(a,a)."""
    c = _conv(a)
    c_pos = np.maximum(c, 0.0)
    if p <= 1:
        w = np.ones_like(c) / len(c)
    else:
        w = c_pos ** (p - 1)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones_like(c) / len(c)
        else:
            w = w / ws
    return _grad_max_weighted(a, c, w), c

def _active_grad(a, c, tol_frac):
    """Active-set gradient with inverse-gap weighting."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a), c
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    n_active = int(active.sum())
    if n_active == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        n_active = 1
    eps = 1e-12 * c_max
    w = np.zeros_like(c)
    w[active] = 1.0 / (c_max - c[active] + eps)
    w = w / w.sum()
    return _grad_max_weighted(a, c, w), c

def _nesterov_optimize(a_init, deadline, min_time_remaining=30,
                        mode='lp_start', p_start=4, p_end=128,
                        tol_start=0.05, tol_end=0.0005,
                        lr_init=0.05, momentum_init=0.9):
    """
    Nesterov momentum optimization with adaptive mode switching.
    mode: 'lp_start' = start with Lp, switch to active-set
          'active' = always active-set
          'lp' = always Lp
    """
    global t_start
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    v = np.zeros(n)
    momentum = momentum_init
    lr = lr_init
    no_improve = 0
    max_stall = 300
    
    c = _conv(a)
    c_max = float(c.max())
    p = p_start
    tol_frac = tol_start
    
    step_count = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        # Compute gradient
        if mode == 'lp' or (mode == 'lp_start' and p < p_end):
            grad, c = _lp_grad(a, p)
        else:
            grad, c = _active_grad(a, c, tol_frac)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Nesterov momentum
        v = momentum * v - lr * grad
        
        # Line search (backtracking)
        alpha = 1.0
        c_cur = _conv(a)
        sc_cur = 2.0 * n * float(c_cur.max())
        
        improved = False
        for _ in range(40):
            a_try = _normalize(a + alpha * v)
            c_try = _conv(a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < sc_cur - 1e-13 * sc_cur:
                a = a_try
                c = c_try
                improved = True
                step_count += 1
                
                if sc_try * 0.5 / n < best_sc * 0.5 / n:  # equivalent to sc_try < best_sc
                    best_sc = sc_try
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                break
            alpha *= 0.5
        
        if not improved:
            v *= 0.3
            lr *= 0.7
            no_improve += 1
            
            if no_improve % 100 == 0:
                tol_frac = max(tol_end, tol_frac * 0.5)
                p = min(p_end, p * 2.0)
                momentum = min(0.95, momentum + 0.02)
            
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                lr = lr_init * 0.3
                momentum = momentum_init
                tol_frac = tol_start * 0.5
                p = p_start
                no_improve = 0
        
        else:
            if mode == 'lp_start' and p < p_end:
                p = min(p_end, p * 1.05)
            tol_frac = max(tol_end, tol_frac * 0.995)
            lr = min(lr_init * 2, lr * 1.003)
            momentum = min(0.95, momentum + 0.0001)
        
        # Periodic check
        if step_count % 5000 == 0 and step_count > 0:
            print(f"    step={step_count}, p={p:.1f}, tol={tol_frac:.5f}, lr={lr:.5f}, sc={best_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
    
    return best_a, best_sc

def _make_profile(n, kind, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    t = np.linspace(0, 1, n)
    
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind.startswith('pow'):
        alpha = float(kind[3:]) if kind[3:] != '' else 1.0
        a = (1 - t + 1e-10) ** alpha
        return a / a.sum()
    elif kind.startswith('exp'):
        beta = float(kind[3:]) if kind[3:] != '' else 2.0
        a = np.exp(-beta * t)
        return a / a.sum()
    elif kind == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == 'hann_shift':
        shift = rng.uniform(0, 1)
