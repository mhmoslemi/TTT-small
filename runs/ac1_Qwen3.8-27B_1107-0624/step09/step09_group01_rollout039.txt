def _cg_solve_lm(matvec, b, max_iter=80):

        return x, True
    rs0 = rs_old

        Hp = matvec(p)
        pHp = float(p @ Hp)
        if not np.isfinite(pHp) or pHp < 1e-30:
            return x, False

        x = x + alpha * p
        r = r - alpha * Hp

            r = r + alpha * Hp

        p = r + beta_cg * p
        rs_old = rs_new

        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.02
        return a / a.sum()
    elif kind == "flat_top":
        # Flat top with smooth edges (like a window function)
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        edge = 0.1
        for i in range(n):
            if t[i] < edge:
                a[i] = 0.5 + 0.5 * np.cos(np.pi * t[i] / edge)
            elif t[i] > 1 - edge:
                a[i] = 0.5 + 0.5 * np.cos(np.pi * (t[i] - (1 - edge)) / edge)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return a / s
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    a_p = a + sigma * rng.standard_normal(len(a))
    return _project_simplex(a_p)


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate():
    t_start = time.monotonic()
    deadline = t_start + 1770.0
    rng = np.random.default_rng(42)

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s", flush=True)

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            if float(parent_raw.sum()) > 0.01:
                parent = parent_raw / parent_raw.sum()
                par_sc = _score(parent)
                print(f"Parent loaded: n={len(parent)}, score={par_sc:.10f}", flush=True)
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        parent = _make_profile(32768, "tapered")
        update_best(parent, _score(parent), "init_tapered")

    n0 = len(parent)
    print(f"Starting with n={n0}", flush=True)

    # Phase 1: L-BFGS with adaptive smoothing
    print(f"\n[Phase 1] Smooth L-BFGS (t={time.monotonic() - t_start:.1f}s)", flush=True)
    t1_end = min(deadline - 900, t_start + 400)
    a1 = _run_smooth_lbfgs(best_seq, t1_end, time_budget=380, beta_init=0.02)
    update_best(a1, _score(a1), "lbfgs_1")

    # Phase 2: Subgradient with momentum
    print(f"\n[Phase 2] Subgrad+momentum (t={time.monotonic() - t_start:.1f}s)", flush=True)
    t2_end = min(deadline - 600, t_start + 700)
    a2 = _run_subgrad(best_seq, t2_end, time_budget=280)
    update_best(a2, _score(a2), "subgrad")

    # Phase 3: LM-Newton
    print(f"\n[Phase 3] LM-Newton (t={time.monotonic() - t_start:.1f}s)", flush=True)
    t3_end = min(deadline - 300, t_start + 900)
    a3 = _run_lm_newton(best_seq, t3_end, time_budget=180)
    update_best(a3, _score(a3), "lm_newton")

    # Phase 4: Cycle with perturbation and different methods
    print(f"\n[Phase 4] Cycling (t={time.monotonic() - t_start:.1f}s)", flush=True)
    cycle_end = min(deadline - 200, t_start + 1400)
    for cyc in range(12):
        if time.monotonic() > cycle_end:
            break

        # Alternate between methods
        if cyc % 3 == 0:
            sigma = 0.005 + 0.002 * cyc
            a_pert = _perturb(best_seq, sigma, rng)
            t_sub = min(cycle_end, time.monotonic() + 40)
            a_c = _run_subgrad(a_pert, t_sub, time_budget=35)
        elif cyc % 3 == 1:
            a_pert = _perturb(best_seq, 0.003, rng)
            t_sub = min(cycle_end, time.monotonic() + 60)
            a_c = _run_lm_newton(a_pert, t_sub, time_budget=55)
        else:
            a_pert = _perturb(best_seq, 0.002, rng)
            t_sub = min(cycle_end, time.monotonic() + 50)
            a_c = _run_smooth_lbfgs(a_pert, t_sub, time_budget=45, beta_init=0.005)

        update_best(a_c, _score(a_c), f"cyc_{cyc}")

    # Phase 5: Multi-n exploration
    print(f"\n[Phase 5] Multi-n (t={time.monotonic() - t_start:.1f}s)", flush=True)
    n_options = sorted(set([n0 // 2, n0, n0 * 2, 16384, 32768, 65536]))
    n_options = [x for x in n_options if 4096 <= x <= 65536]

    for n_try in n_options:

            break
        a_res = _resample(best_seq, n_try)
        sc_res = _score(a_res)
        if sc_res < best_score:
            update_best(a_res, sc_res, f"resample_n{n_try}")

        if time.monotonic() < deadline - 60:
            t_end = min(deadline - 50, time.monotonic() + 40)
            a_opt = _run_smooth_lbfgs(a_res, t_end, time_budget=35, beta_init=0.01)
            update_best(a_opt, _score(a_opt), f"lbfgs_n{n_try}")

    # Phase 6: Final polish - run best method on current best
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 5:

        a_fin1 = _run_subgrad(best_seq, t_end, time_budget=55)
        update_best(a_fin1, _score(a_fin1), "final_sg")

        # Try LM-Newton
        if time.monotonic() < deadline - 3:

            a_fin2 = _run_lm_newton(best_seq, t_end2, time_budget=35)
            update_best(a_fin2, _score(a_fin2), "final_nm")

    # Final output
    print(f"\n{'=' * 60}", flush=True)
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s", flush=True)
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
