"""
Convex optimization approach for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key improvements: adaptive log-sum-exp temperature, Nesterov-accelerated
subgradient, frequency-domain peak analysis, multi-n exploration (up to 32768),
and aggressive multi-restart with diverse profiles.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _topk_smooth_obj_grad(z, beta, K, n):
    """Top-K log-sum-exp smoothed objective in log-domain with softmax parameterization."""
    # Compute a from z (softmax)
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a = a / a.sum()
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    # Get top K values
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        c_top = c
        top_idx = np.arange(len(c))
    
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    
    # Gradient through softmax: d/da of (2n * smooth_max) dotted with da/dz
    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


class _SmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _topk_smooth_obj_grad(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _topk_smooth_obj_grad(z, self.beta, self.K, self.n)[1]

    base_beta = max(gap, max_c * 0.01)
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

        (0.001, 0.002), (0.0005, 0.001),

def _nesterov_subgrad(a_init, deadline, min_time_remaining=60.0):

    best_a = a.copy()
    best_sc = _score(a)
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    y_z = z.copy()  # Nesterov's lookahead point
    
    n_2nm1 = 2 * n - 1
    active_frac = 0.02
    momentum = 0.5  # Nesterov momentum (will increase)
    lr = 0.01
    
    velocity = np.zeros(n)
    prev_g = None
    stall = 0
    max_stall = 100
    iteration = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        
        # Nesterov lookahead: evaluate at y = z + momentum * (z - z_prev)
        a_y = np.exp(y_z - y_z.max())
        a_y = a_y / a_y.sum()
        
        c = _full_convolve(a_y, a_y)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_active]
        
        active_vals = c[active_idx]
        w = np.zeros(n_2nm1)
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w[active_idx] = (active_vals - active_vals.min()) / spread + 0.1
        else:
            w[active_idx] = 1.0 / n_active
        w = w / w.sum()
        
        # Gradient at y
        g = _grad_maxconv(a_y, w)
        # Convert to z-space gradient
        a_y_arr = a_y
        W = float(np.dot(a_y_arr, g))
        g_z = a_y_arr * (g - W)
        g_z_centered = g_z - g_z.mean()

            best_a = a_try.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive learning rate

            if stall > max_stall:
                lr = max(1e-6, lr * 0.7)
                active_frac = min(0.1, active_frac * 1.5)
                stall = 0
            else:
                lr = min(0.05, lr * 1.1)
        
        # Periodic: check if we should restart with L-BFGS
        if iteration % 300 == 0 and time.monotonic() + min_time_remaining + 120 < deadline:
            lbfgs_deadline = time.monotonic() + 100
            a_tmp = _run_lbfgs_adaptive(a_try, lbfgs_deadline, min_time_remaining=10)
            sc_tmp = _score(a_tmp)

def _make_profile(n, kind="tapered"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.6
    elif kind == "cos_taper":
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
    elif kind == "asym":
        a = np.exp(-2.0 * t) + 0.1
    elif kind == "bimodal":
        a = np.exp(-25.0 * (t - 0.3) ** 2) + np.exp(-25.0 * (t - 0.7) ** 2) + 0.05
    elif kind == "flat_taper":
        # Mostly flat with gentle taper at edges
        taper = n // 10
        a = np.ones(n)
        for i in range(taper):
            f = (i + 1) / (taper + 1)
            a[i] = f ** 0.5
            a[n - 1 - i] = f ** 0.5
    elif kind == "super_taper":
        a = np.sin(np.pi * t) ** 0.3
    elif kind == "gauss":
        a = np.exp(-3.0 * (t - 0.5) ** 2 / 0.25) + 0.01
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    """Fine active-set polish."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.003
    tol = 0.003
    no_improve = 0
    n_2nm1 = 2 * n - 1

            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        # Active set: peaks within tol of max
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())

            active[int(np.argmax(c))] = True

def _multiplicative_update(a_init, deadline, min_time_remaining=30.0):
    """Multiplicative weight update (mirror descent with KL)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = _normalize(a_init.copy())
    best_sc = _score(best_a)
    
    eta = 0.02
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        
        # Compute pressure on each a_i
        # a_i contributes to c_k for k in [i, i+n-1] with weight a_{k-i}
        # Pressure = sum over peaks k of w_k * a_{k-i}
        n_2nm1 = 2 * n - 1
        n_active = max(10, int(n_2nm1 * 0.01))
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_active]
        
        w = np.zeros(n_2nm1)
        active_vals = c[active_idx]
        w[active_idx] = (active_vals - active_vals.min() + 1e-10)
        w = w / w.sum()
        
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        
        # Multiplicative update in log-space (equivalent to mirror descent with entropy)
        z = z - eta * g_centered / (np.abs(g_centered).mean() + 1e-15)
        
        if no_improve > 100:
            eta = max(0.001, eta * 0.8)
    
    return best_a, best_sc

Convex optimization approach for minimizing 2n*max(conv(a,a))/sum(a)^2.

def _next_pow2(n):
    k = 1

    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _topk_smooth_obj_grad(z, beta, K, n):
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        c_top = c
        top_idx = np.arange(len(c))
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


class _SmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _topk_smooth_obj_grad(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _topk_smooth_obj_grad(z, self.beta, self.K, self.n)[1]


def _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a
    c_sorted = np.sort(c_cur)[::-1]
    n_2nm1 = 2 * n - 1
    gap = c_sorted[0] - c_sorted[min(200, n_2nm1 - 1)]
    if gap < max_c * 0.0005:
        gap = max_c * 0.0005
    base_beta = max(gap, max_c * 0.005)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    schedule = [
        (3.0, 0.2), (1.5, 0.15), (0.8, 0.1), (0.5, 0.07),
        (0.3, 0.05), (0.15, 0.035), (0.08, 0.025), (0.04, 0.015),
        (0.02, 0.01), (0.01, 0.007), (0.005, 0.005), (0.002, 0.003),
        (0.001, 0.002), (0.0005, 0.001), (0.0002, 0.0008),
    ]
    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 15:
            break
        beta = base_beta * beta_mult
        K = max(8, int(n_2nm1 * K_frac))
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(10000, max(100, int(remaining / (est_time * 12))))
        prob = _SmoProb(beta, K, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 120, 'maxls': 50}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a


def _nesterov_subgrad(a_init, deadline, min_time_remaining=60.0):
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    y_z = z.copy()
    n_2nm1 = 2 * n - 1
    active_frac = 0.015
    t_nest = 1.0
    lr = 0.008
    stall = 0
    max_stall = 120
    iteration = 0
    last_lbfgs = time.monotonic()
    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        a_y = np.exp(y_z - y_z.max())
        a_y = a_y / a_y.sum()
        c = _full_convolve(a_y, a_y)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        n_active = max(8, min(n_2nm1, int(n_2nm1 * active_frac)))
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_active]
        active_vals = c[active_idx]
        w = np.zeros(n_2nm1)
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w[active_idx] = (active_vals - active_vals.min()) / spread + 0.05
        else:
            w[active_idx] = 1.0 / n_active
        w = w / w.sum()
        g = _grad_maxconv(a_y, w)
        W = float(np.dot(a_y, g))
        g_z = a_y * (g - W)
        g_z_centered = g_z - g_z.mean()
        gnorm = np.linalg.norm(g_z_centered)
        if gnorm < 1e-18:
            break
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        coeff = (t_nest - 1.0) / t_new
        z_new = y_z - lr * g_z_centered
        y_z = z_new + coeff * (z_new - z)
        z = z_new
        t_nest = t_new
        a_try = np.exp(z - z.max())
        a_try = a_try / a_try.sum()
        sc_try = 2.0 * n * float(_full_convolve(a_try, a_try).max())
        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try.copy()
            stall = 0
        else:
            stall += 1
        if stall > max_stall:
            lr = max(1e-7, lr * 0.6)
            active_frac = min(0.08, active_frac * 2.0)
            t_nest = 1.0
            stall = 0
        elif stall % 50 == 0:
            lr = min(0.04, lr * 1.05)
        if iteration % 400 == 0 and time.monotonic() - last_lbfgs > 150:
            if time.monotonic() + min_time_remaining + 100 < deadline:
                lbfgs_deadline = time.monotonic() + 80
                a_tmp = _run_lbfgs_adaptive(a_try, lbfgs_deadline, min_time_remaining=10)
                sc_tmp = _score(a_tmp)
                if sc_tmp < best_sc:
                    best_a = a_tmp.copy()
                    best_sc = sc_tmp
                    z = np.log(np.maximum(a_tmp, 1e-30))
                    z = z - z.mean()
                    y_z = z.copy()
                    t_nest = 1.0
                    lr = 0.006
                    active_frac = 0.015
                last_lbfgs = time.monotonic()
    return best_a, best_sc


def _mirror_descent(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = _normalize(a_init.copy())
    best_sc = _score(best_a)
    eta = 0.015
    no_improve = 0
    n_2nm1 = 2 * n - 1
    while time.monotonic() + min_time_remaining < deadline:
        a = np.exp(z - z.max())
        a = a / a.sum()
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 800:
                break
        n_active = max(10, int(n_2nm1 * 0.01))
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_active]
        w = np.zeros(n_2nm1)
        active_vals = c[active_idx]
        w[active_idx] = active_vals - active_vals.min() + 1e-10
        w = w / w.sum()
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        g_scale = np.abs(g_centered).mean() + 1e-15
        z = z - eta * g_centered / g_scale
        if no_improve > 200:
            eta = max(0.001, eta * 0.85)
        elif no_improve % 100 == 0:
            eta = max(0.001, eta * 0.9)
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.6
    elif kind == "cos_taper":
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
    elif kind == "asym":
        a = np.exp(-2.0 * t) + 0.1
    elif kind == "bimodal":
        a = np.exp(-25.0 * (t - 0.3) ** 2) + np.exp(-25.0 * (t - 0.7) ** 2) + 0.05
    elif kind == "flat_taper":
        a = np.ones(n)
        taper = max(1, n // 8)
        for i in range(taper):
            f = (i + 1) / (taper + 1)
            a[i] = f ** 0.4
            a[n - 1 - i] = f ** 0.4
    elif kind == "super_taper":
        a = np.sin(np.pi * t) ** 0.3
    elif kind == "gauss":
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.02
    elif kind == "pow_taper":
        a = np.sin(np.pi * t) ** 0.8
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.002
    tol = 0.002
    no_improve = 0
    n_2nm1 = 2 * n - 1
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > int(n_2nm1 * 0.03):
            c_sorted_idx = np.argsort(c)[::-1]
            active = np.zeros(n_2nm1, dtype=bool)
            active[c_sorted_idx[:int(n_2nm1 * 0.03)]] = True
            n_active = int(n_2nm1 * 0.03)
        w = np.zeros(n_2nm1)
        w[active] = 1.0 / n_active
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-17:
            break
        alpha = min(lr, 0.3 / (gnorm + 1e-15))
        improved = False
        for ls in range(50):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-16:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            no_improve += 1
            if no_improve % 40 == 0:
                tol = max(5e-5, tol * 0.6)
                lr *= 0.65
            if no_improve > 500:
                a = best_a.copy()
                tol = max(2e-5, tol * 0.4)
                lr = 0.001
                no_improve = 0
        else:
            if tol > 5e-5:
                tol = max(5e-5, tol * 0.95)
            lr = min(0.006, lr * 1.03)
    return best_a, best_sc


def _parametric_optimize(n, deadline, min_time_remaining=20.0):
    """Optimize over a parametric family: a_i = sin(pi*t)^p * (1 + amp*cos(2*pi*f*t)) + base."""
    best_a = None
    best_sc = np.inf
    p_vals = [0.4, 0.6, 0.8, 1.0, 1.5, 2.0]
    t = np.linspace(0, 1, n)
    for p in p_vals:
        if time.monotonic() + min_time_remaining > deadline:
            break
        a_base = np.sin(np.pi * t) ** p
        a_base = np.maximum(a_base, 1e-10)
        a_base = a_base / a_base.sum()
        a_opt = _run_lbfgs_adaptive(a_base, min(deadline - min_time_remaining, time.monotonic() + 60),
                                     min_time_remaining=min_time_remaining)
        sc = _score(a_opt)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
    if best_a is not None:
        return best_a, best_sc
    return _make_profile(n, "tapered"), _score(_make_profile(n, "tapered"))


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase convex optimizer: adaptive L-BFGS, Nesterov subgradient,
    mirror descent, parametric optimization, multi-n (16K/32K) exploration,
    and multi-restart with diverse perturbations.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, tapered n={n_parent}, score={best_score:.10f}")

    # Phase 1: L-BFGS adaptive on parent
    print(f"\n[P1] L-BFGS adaptive (t={time.monotonic()-t_start:.1f}s)")
    a1 = _run_lbfgs_adaptive(best_seq, deadline, min_time_remaining=800)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs1] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # Phase 2: Nesterov subgradient
    print(f"\n[P2] Nesterov subgradient (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        a2, sc2 = _nesterov_subgrad(best_seq, min(deadline - 400, time.monotonic() + 500),
                                    min_time_remaining=60)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [nesterov] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # Phase 3: Mirror descent
    print(f"\n[P3] Mirror descent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        a3, sc3 = _mirror_descent(best_seq, min(deadline - 300, time.monotonic() + 200),
                                  min_time_remaining=30)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [mirror] t={time.monotonic()-t_start:.1f}s")

    # Phase 4: L-BFGS round 2
    print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a4 = _run_lbfgs_adaptive(best_seq, min(deadline - 250, time.monotonic() + 300),
                                  min_time_remaining=50)
        sc4 = _score(a4)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [lbfgs2] t={time.monotonic()-t_start:.1f}s")

    # Phase 5: Try n=32768
    print(f"\n[P5] Try n=32768 (t={time.monotonic()-t_start:.1f}s)")
    n_main = len(best_seq)
    if time.monotonic() < deadline - 200 and n_main < 32768:
        a_res = _resample(best_seq, 32768)
        a_opt = _run_lbfgs_adaptive(a_res, min(deadline - 150, time.monotonic() + 200),
                                     min_time_remaining=30)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            n_main = 32768
            print(f"  NEW BEST: {best_score:.10f} [n=32768] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  n=32768 no improvement ({sc_opt:.10f} vs {best_score:.10f})")

    # Phase 6: Fine polish
    print(f"\n[P6] Fine polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 120:
        a5, sc5 = _fine_polish(best_seq, min(deadline - 100, time.monotonic() + 150),
                                min_time_remaining=20)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [polish] t={time.monotonic()-t_start:.1f}s")

    # Phase 7: Multi-restart
    print(f"\n[P7] Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        restart_budget = min(120, (deadline - 100 - time.monotonic()) * 0.7)
        n_restart = max(3, int(restart_budget / 25))
        sigmas = [0.015, 0.03, 0.05, 0.08, 0.02, 0.04, 0.1, 0.01, 0.06, 0.12]
        for i in range(n_restart):
            if time.monotonic() > deadline - 80:
                break
            sigma = sigmas[i % len(sigmas)]
            a_pert = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs_adaptive(a_pert, min(deadline - 70, time.monotonic() + 22),
                                       min_time_remaining=5)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.10f} [restart_{i}] t={time.monotonic()-t_start:.1f}s")

    # Phase 8: Parametric optimization
    print(f"\n[P8] Parametric (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        n_p = len(best_seq)
        a_p, sc_p = _parametric_optimize(n_p, deadline, min_time_remaining=30)
        if sc_p < best_score:
            best_seq = a_p.copy()
            best_score = sc_p
            print(f"  NEW BEST: {best_score:.10f} [parametric] t={time.monotonic()-t_start:.1f}s")

    # Phase 9: Final L-BFGS + polish
    print(f"\n[P9] Final (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 10:
        a_fin = _run_lbfgs_adaptive(best_seq, deadline - 5, min_time_remaining=5)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
        a_fin2, sc_fin2 = _fine_polish(best_seq, deadline - 2, min_time_remaining=2)
        if sc_fin2 < best_score:
            best_seq = a_fin2.copy()
            best_score = sc_fin2
            print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")

    # Final
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
