    k = 1
    while k < n:

    return max(2, k)

    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _topk_grad_contiguous(a, lo, hi):
    """Compute (2/K) * sum_{k=lo}^{hi} a[k-i] for all i in [0, n-1].
    Uses prefix sums for O(n) computation."""
    n = len(a)
    K = hi - lo + 1
    # prefix[j] = sum_{j'=0}^{j} a[j']
    prefix = np.zeros(n + 1)
    prefix[1:] = np.cumsum(a)
    # For each i: sum_{k=lo}^{hi} a[k-i] where a[j]=0 for j outside [0,n-1]
    # = sum_{j=max(0,lo-i)}^{min(n-1,hi-i)} a[j]
    # = prefix[min(n-1,hi-i)+1] - prefix[max(0,lo-i)]
    i = np.arange(n)
    j_lo = np.maximum(0, lo - i)
    j_hi = np.minimum(n - 1, hi - i)
    valid = j_lo <= j_hi
    grad = np.zeros(n)
    grad[valid] = prefix[j_hi[valid] + 1] - prefix[j_lo[valid]]
    return (2.0 / K) * grad


def _topk_grad_general(a, topk_idx, K):
    """Compute (2/K) * sum_{k in topk_idx} a[k-i] for all i. O(n*K) or O(n log n)."""
    n = len(a)
    # Check if contiguous
    sorted_idx = np.sort(topk_idx)
    if len(topk_idx) > 1 and np.all(np.diff(sorted_idx) == 1):
        return _topk_grad_contiguous(a, sorted_idx[0], sorted_idx[-1])
    # General case: O(n*K)
    grad = np.zeros(n)
    for k in topk_idx:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            j = k - np.arange(i_lo, i_hi + 1)
            grad[i_lo:i_hi + 1] += a[j]
    return (2.0 / K) * grad


def _topk_obj(a, topk_idx, K):
    """Compute (1/K) * sum_{k in topk} conv(a,a)[k]."""
    c = _full_convolve(a, a)
    return float(np.mean(c[topk_idx]))


def _topk_projected_grad(a_init, deadline, K=200, max_inner_iters=200, 
                         outer_iters=50, min_time_remaining=30.0, seed=0):
    """Top-K mean projected gradient method."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_score = _score(a)
    
    rng = np.random.default_rng(seed)
    lr = 0.4  # initial step size (safe: 1/Lipschitz ~ 0.5)
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        # Compute current autocorrelation
        c = _full_convolve(a, a)
        current_score = 2.0 * n * float(c.max())
        
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            break
        
        # Find top-K lags
        topk_idx = np.argsort(c)[::-1][:K]
        topk_sorted = np.sort(topk_idx)
        
        # Check if contiguous
        is_contiguous = (len(topk_sorted) > 1 and 
                        np.all(np.diff(topk_sorted) == 1))
        
        if is_contiguous:
            lo, hi = topk_sorted[0], topk_sorted[-1]
        else:
            lo, hi = topk_sorted.min(), topk_sorted.max()
        
        # Inner loop: optimize with fixed top-K
        for inner_it in range(max_inner_iters):
            if time.monotonic() + min_time_remaining > deadline:
                break
            
            # Compute gradient
            if is_contiguous:
                grad = _topk_grad_contiguous(a, lo, hi)
            else:
                grad = _topk_grad_general(a, topk_idx, K)
            
            # Project onto tangent space (sum = 0) for better conditioning
            grad_t = grad - grad.mean()
            
            # Projected gradient step
            a_new = _project_simplex(a - lr * grad_t)
            
            # Evaluate
            c_new = _full_convolve(a_new, a_new)
            obj_new = float(np.mean(c_new[topk_idx]))
            c_old_obj = float(np.mean(c[topk_idx]))
            
            if obj_new < c_old_obj - 1e-16:
                a = a_new
                c = c_new
            else:
                lr *= 0.5
                if lr < 1e-8:
                    break
        
        # Periodic restart
        if no_improve % 30 == 0 and no_improve > 0:
            pert = 0.001 * rng.standard_normal(n)
            a_pert = _project_simplex(a + pert)
            c_pert = _full_convolve(a_pert, a_pert)
            sc_pert = 2.0 * n * float(c_pert.max())
            if sc_pert < best_score:
                a = a_pert
                best_a = a.copy()
                best_score = sc_pert
            lr = 0.4
            no_improve = 0
    
    return best_a, best_score


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, beta_start=None, min_time_remaining=30.0,
               n_betas=12):
    """L-BFGS with log-sum-exp smoothing and adaptive beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = max_c - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    if beta_start is None:
        beta_start = max(gap, max_c * 0.005)
    
    betas = beta_start * np.array([1.0] + [0.5 ** i for i in range(1, n_betas)])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.sum() / n  # center for numerical stability
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        est_time = 0.005 * max(1.0, n / 16384.0)
        max_iter = min(5000, max(100, int(remaining / (est_time * 5))))
        
        def fun(z_in):
            return _smoobj(z_in, beta, n)
        
        def grad(z_in):
            return _smograd(z_in, beta, n)
        
        try:
            result = minimize(fun, z, jac=grad, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'ftol': 1e-18,
                                      'gtol': 1e-14, 'maxcor': 100, 'maxls': 30})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.sum() / n
        except Exception:
            pass
    
    return current_a


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n
