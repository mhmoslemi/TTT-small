import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _convolve_and_max(a):
    c = _full_convolve(a, a)
    return c, float(c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _conv_grad_single(a, k):
    n = len(a)
    grad = np.zeros(n)
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    if j_lo <= j_hi:
        idx = k - np.arange(j_lo, j_hi + 1)
        grad[j_lo:j_hi + 1] += a[idx]
    j_lo2 = max(0, k)
    j_hi2 = min(n - 1, k + n - 1)
    if j_lo2 <= j_hi2:
        idx = np.arange(j_lo2, j_hi2 + 1) - k
        grad[j_lo2:j_hi2 + 1] += a[idx]
    return grad


def _conv_grad_multi(a, active_indices):

        beta = base_gap * beta_mult
        if beta < c_max_init * 1e-6:
            beta = c_max_init * 1e-6

        K_eff = min(K, len(_full_convolve(current_a, current_a)))

        # Estimate time per L-BFGS iteration
        est_time = 0.003 * max(1.0, n / 8000.0) * (1 + K_eff / 100.0)
        max_iter = min(8000, max(30, int(remaining / (est_time * 8))))

        def func(zv):
            return _topk_smooth_obj(zv, beta, K_eff, n)

        def grad(zv):
            return _topk_smooth_grad(zv, beta, K_eff, n)

        try:
            result = minimize(
                func, z, jac=grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 3 == 0:
                    print(f"    P2 step={i}: score={current_sc:.10f}, K={K_eff}, beta={beta:.2e}")
            # Update base_gap based on current state
            if i % 4 == 3 and i < len(schedule) - 1:
                c_cur = _full_convolve(current_a, current_a)
                c_s = np.sort(c_cur)[::-1]
                base_gap = max(c_s[0] - c_s[min(50, len(c_s)-1)], c_s[0] * 0.0005)

        except Exception:
            pass

    return current_a, current_sc


def _phase3_equalize(a_init, deadline, min_remaining=30.0):
    """Direct peak equalization: iteratively reduce the top peaks by
    moving mass from peak contributors to non-contributors."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    no_improve = 0
    tol = 1e-9

    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                break

        # Find peaks within tolerance of max
        tol_abs = max(c_max * 1e-6, 1e-12)
        active_mask = c >= (c_max - tol_abs)
        active_indices = np.where(active_mask)[0]

        if len(active_indices) == 0:
            active_indices = np.array([int(np.argmax(c))])

        # Compute gradient of the max (equal-weighted over active)
        K = len(active_indices)
        if K > 100:
            # Too many active, take top 100
            c_idx = np.argsort(c)[::-1]
            active_indices = c_idx[:100]
            K = 100

        grad = _conv_grad_multi(a, active_indices) / K
        grad_t = grad - np.mean(grad)

        # ADMM-like step: move in negative gradient direction
        # with a step size that aims to equalize the peaks
        peak_vals = c[active_indices]
        peak_mean = peak_vals.mean()
        peak_var = peak_vals.var()

        if peak_var < tol_abs * tol_abs:
            # Peaks are already equalized, try to reduce the level
            # Move mass from high-gradient to low-gradient positions
            hg = np.argsort(grad_t)[::-1][:max(10, n // 200)]
            lg = np.argsort(grad_t)[:max(10, n // 200)]
            
            move = 1e-6 * a.max()
            for j in hg:
                a[j] -= move / len(hg)
            for j in lg:
                a[j] += move * len(hg) / (len(lg) * (a[j] + 1e-15))
            
            a = np.maximum(a, 0.0)
            ss = a.sum()
            if ss < 1e-10:
                break
            a = a / ss
            continue

        # Step size based on gradient magnitude
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-20:
            break

        # Try to reduce the max by moving opposite to gradient
        step_size = 0.01 * c_max / (gnorm + 1e-20)
        step_size = min(step_size, 0.05 * a.max())

        a_new = a - step_size * grad_t
        a_new = _project_simplex(a_new)

        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < current_sc:
            a = a_new
        else:
            # Halve step
            a_new2 = a - 0.3 * step_size * grad_t
            a_new2 = _project_simplex(a_new2)
            c_new2 = _full_convolve(a_new2, a_new2)
            sc_new2 = 2.0 * n * float(c_new2.max())
            if sc_new2 < current_sc:
                a = a_new2
            else:
                # Very small step
                a_new3 = a - 0.01 * step_size * grad_t
                a_new3 = _project_simplex(a_new3)
                a = a_new3

        if no_improve > 500 and no_improve % 500 == 0:
            # Try a random perturbation
            rng = np.random.default_rng(hash((t,)) % 2**31)
            perturb = 1e-5 * rng.standard_normal(n)
            a_pert = _project_simplex(a + perturb)
            c_pert = _full_convolve(a_pert, a_pert)
            sc_pert = 2.0 * n * float(c_pert.max())
            if sc_pert < best_sc:
                a = a_pert
                best_a = a.copy()
                best_sc = sc_pert
                no_improve = 0

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_initial(n, kind="structured", seed=42):
    """Create various structured initializations."""
    rng = np.random.default_rng(seed)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "gaussian":
        a = np.abs(rng.standard_normal(n))
        return a / a.sum()
    elif kind == "exponential":
        a = rng.exponential(1.0, n)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "structured":
        # Use a structured sequence based on number theory
        t = np.arange(n)
        # Use a low-discrepancy-like structure
        a = 1.0 + 0.3 * np.sin(2 * np.pi * t * 7 / n) + 0.2 * np.cos(2 * np.pi * t * 13 / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "fibonacci":
        # Golden ratio based structure
        phi = (1 + np.sqrt(5)) / 2
        t = np.arange(n)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * t / phi)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _shrink_and_reoptimize(a, deadline, min_remaining=30.0):
    """Try zeroing out the smallest elements and re-optimizing."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    fractions = [0.001, 0.003, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12]
    a_sorted_idx = np.argsort(a)

    for frac in fractions:
        if time.monotonic() + min_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    1. Load parent, try multiple initializations
    2. Fast subgradient for coarse optimization
    3. Top-K L-BFGS for medium-precision optimization
    4. Direct peak equalization for fine optimization
    5. Shrinkage exploration
    6. Multi-n exploration
    7. Final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_initial(n_parent, "structured", seed)
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, structured n={n_parent}, score={best_score:.10f}")

    # ============ Phase 1: Subgradient (coarse) ============
    print(f"\n[Phase 1] Subgradient (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 600, t_start + 250)
    if time.monotonic() < p1_deadline:
        a1, sc1 = _phase1_subgradient(best_seq, p1_deadline, min_remaining=30, seed=seed)
        if sc1 < best_score:
            best_seq = a1.copy()
            best_score = sc1
            print(f"  Phase1: NEW BEST={best_score:.10f}")
        else:
            print(f"  Phase1: no improvement (best={best_score:.10f})")

    # ============ Phase 2: Top-K L-BFGS (medium) ============
    print(f"\n[Phase 2] Top-K L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 400, t_start + 700)
    if time.monotonic() < p2_deadline:
        a2, sc2 = _phase2_topk_lbfgs(best_seq, p2_deadline, min_remaining=30)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  Phase2: NEW BEST={best_score:.10f}")
        else:
            print(f"  Phase2: no improvement (best={best_score:.10f})")

    # ============ Phase 3: Peak equalization (fine) ============
    print(f"\n[Phase 3] Peak equalization (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 250, t_start + 1000)
    if time.monotonic() < p3_deadline:
        a3, sc3 = _phase3_equalize(best_seq, p3_deadline, min_remaining=20)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  Phase3: NEW BEST={best_score:.10f}")
        else:
            print(f"  Phase3: no improvement (best={best_score:.10f})")

    # ============ Phase 4: Second L-BFGS pass (finer) ============
    print(f"\n[Phase 4] L-BFGS pass 2 (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 200, t_start + 1200)
    if time.monotonic() < p4_deadline:
        a4, sc4 = _phase2_topk_lbfgs(best_seq, p4_deadline, min_remaining=20)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  Phase4: NEW BEST={best_score:.10f}")
        else:
            print(f"  Phase4: no improvement (best={best_score:.10f})")

    # ============ Phase 5: More equalization ============
    print(f"\n[Phase 5] Equalization pass 2 (t={time.monotonic()-t_start:.1f}s)")
    p5_deadline = min(deadline - 150, t_start + 1350)
    if time.monotonic() < p5_deadline:
        a5, sc5 = _phase3_equalize(best_seq, p5_deadline, min_remaining=15)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  Phase5: NEW BEST={best_score:.10f}")

    # ============ Phase 6: Shrinkage ============
    print(f"\n[Phase 6] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    p6_deadline = min(deadline - 100, t_start + 1450)
    if time.monotonic() < p6_deadline:
        a6, sc6 = _shrink_and_reoptimize(best_seq, p6_deadline, min_remaining=15)
        if sc6 < best_score:
            best_seq = a6.copy()
            best_score = sc6
            print(f"  Phase6: NEW BEST={best_score:.10f}")
            # Re-optimize after shrinkage
            if time.monotonic() < deadline - 80:
                a6b, sc6b = _phase2_topk_lbfgs(best_seq, deadline - 40, min_remaining=10)
                if sc6b < best_score:
                    best_seq = a6b.copy()
                    best_score = sc6b
                    print(f"  Phase6b: NEW BEST={best_score:.10f}")

    # ============ Phase 7: Multi-n exploration ============
    print(f"\n[Phase 7] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_options = [16384, 20480, 12288, 24576]
    if n_parent not in n_options:
        n_options.insert(0, n_parent)

    for n_try in n_options:
        if time.monotonic() > deadline - 60:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init={sc_init:.10f}")
        if sc_init < best_score + 0.01:  # Only continue if competitive
            if time.monotonic() < deadline - 40:
                a_opt, sc_opt = _phase2_topk_lbfgs(a_init, deadline - 20, min_remaining=10)
                if sc_opt < best_score:
                    best_seq = a_opt.copy()
                    best_score = sc_opt
                    print(f"  Phase7: NEW BEST={best_score:.10f} (n={n_try})")

    # ============ Phase 8: Final equalization polish ============
    print(f"\n[Phase 8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a8, sc8 = _phase3_equalize(best_seq, deadline - 2, min_remaining=2)
        if sc8 < best_score:
            best_seq = a8.copy()
            best_score = sc8
            print(f"  Phase8: NEW BEST={best_score:.10f}")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
