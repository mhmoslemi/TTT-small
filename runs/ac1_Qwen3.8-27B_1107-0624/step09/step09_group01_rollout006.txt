import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _ls_smoothed_obj_and_grad(z, beta, n):
    """Compute smoothed log-sum-exp objective and gradient."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        w = e / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    
    # Gradient
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W_val = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W_val)
    
    return obj, grad

Optimized sequence search minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses aggressive L-BFGS with adaptive log-sum-exp smoothing, Adam with
importance weighting, exact subgradient polish, and multi-scale n exploration.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _ls_obj_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        w = e / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W_val)
    return obj, grad

    iteration = 0
    while beta > min_beta and time.monotonic() < deadline:
        iteration += 1

        def obj_fn(z):
            o, _ = _ls_obj_grad(z, beta, n)
            return o

        def grad_fn(z):
            _, g = _ls_obj_grad(z, beta, n)
            return g

        z = np.log(np.maximum(best_a, 1e-30)) - np.log(np.maximum(best_a, 1e-30)).mean()

    # Phase 1: Main L-BFGS optimization (most critical)
    print(f"\n[Phase 1] Primary L-BFGS (n={len(parent)})")
    t_p1 = t_start + 650
    a1, sc1 = _lbfgs_aggressive(parent, t_p1, verbose=True)
    update_best(a1, sc1)

    # Phase 2: Adam for different local minima
    print(f"\n[Phase 2] Adam importance (t={time.monotonic()-t_start:.1f}s)")
    t_p2 = time.monotonic() + 250
    a2, sc2 = _adam_global(best_seq, min(t_p2, deadline - 300))
    update_best(a2, sc2)

    # Phase 3: Second L-BFGS pass from improved point
    print(f"\n[Phase 3] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    t_p3 = time.monotonic() + 350
    a3, sc3 = _lbfgs_aggressive(best_seq, min(t_p3, deadline - 200), verbose=True)
    update_best(a3, sc3)

    # Phase 4: Multi-n exploration
    print(f"\n[Phase 4] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n0 = len(best_seq)
    n_set = sorted(set([n0 // 2, n0, n0 * 2, 16384, 32768, 65536]))
    for n_try in n_set:
        if time.monotonic() > deadline - 150:
            break
        if n_try < 1024 or n_try > 98304:
            continue
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if sc_r < best_score:
            update_best(a_r, sc_r)
        if time.monotonic() < deadline - 130:
            t_sub = min(deadline - 120, time.monotonic() + 50)
            a_opt, sc_opt = _lbfgs_aggressive(a_r, t_sub)
            update_best(a_opt, sc_opt)

    # Phase 5: Subgradient fine polish
    print(f"\n[Phase 5] Subgrad polish (t={time.monotonic()-t_start:.1f}s)")
    t_p5 = min(deadline - 60, time.monotonic() + 100)
    a5, sc5 = _subgrad_fine(best_seq, t_p5)
    update_best(a5, sc5)

    # Phase 6: Final L-BFGS with tightest beta
    print(f"\n[Phase 6] Final L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    a6, sc6 = _lbfgs_aggressive(best_seq, deadline - 5, verbose=True)
    update_best(a6, sc6)

    # Final
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
