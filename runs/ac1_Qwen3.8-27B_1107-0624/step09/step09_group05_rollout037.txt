The Fejér kernel is the square of the Dirichlet kernel, and it has the property that its autocorrelation is a triangle (which is as flat as possible for a "main lobe" shape). But we want the autocorrelation to be flat, not the signal.

Actually, the Fejér kernel F_n(x) = (1/n) * (sin(n*x/2)/sin(x/2))^2 has the property that its autocorrelation is the triangle function. This isn't what we want.

What we want is: a signal a whose autocorrelation c = a*a is as flat as possible. This is the dual of the Fejér kernel problem.

In signal processing, a signal with constant (or nearly constant) autocorrelation is called a "ZAC" (Zero Autocorrelation) sequence or a "flat autocorrelation" sequence. These exist for complex sequences (like m-sequences), but for real non-negative sequences, it's much harder.

For real non-negative sequences, the best we can do is minimize the peak sidelobe level (PSL) of the autocorrelation. This is a well-studied problem.

The best known PSL for non-negative sequences of length n is O(1/n) (same order as the main lobe), achieved by certain window functions. The optimal window for minimizing PSL is related to the Chebyshev (Dolph-Chebyshev) window!

The Dolph-Chebyshev window has the property that all sidelobes are equal in magnitude. If we could use a real non-negative version of this, we'd get a nearly flat autocorrelation.

But the Dolph-Chebyshev window can have negative values. We need a non-negative version.

For example: a[k] = w(k/n) * (1 + eps * cos(2*pi*f*k))

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _spectral_waterfill(a, n_iter=20, alpha=0.5, n_peg=100):
    """Water-filling in the frequency domain to flatten the power spectrum.
    
    The idea: |FFT(a_padded)|^2 is the circular autocorrelation.
    We iteratively reduce peaks and raise the rest to make it flatter.
    """
    n = len(a)
    a = a.copy()
    N = _next_pow2(2 * n - 1)
    
    for it in range(n_iter):
        a_padded = np.zeros(N)
        a_padded[:n] = a
        A = np.fft.rfft(a_padded)
        power = np.abs(A) ** 2
        
        # Water-filling: target a flatter spectrum
        # Reduce peaks, raise valleys
        p_max = power.max()
        p_mean = power.mean()
        
        # Peg factor: how much to reduce peaks
        # For bins above threshold, scale down
        threshold = p_mean * 1.0  # all bins above mean get reduced
        mask_high = power > threshold
        
        if mask_high.sum() == 0:
            break
            
        # Reduce high-power bins
        # New power for high bins: interpolate between p_mean and p_max
        reduction = power[mask_high] - p_mean
        power[mask_high] = p_mean + reduction * (1 - alpha)
        
        # Reconstruct
        new_amp = np.sqrt(power)
        new_phase = np.angle(A)
        A_new = new_amp * np.exp(1j * new_phase)
        
        a_new_padded = np.fft.irfft(A_new, N)
        a_new = np.real(a_new_padded[:n])
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s
    
    return a

def _spectral_waterfill_v2(a, n_iter=50, peg_fraction=0.3):
    """Improved water-filling: peg the top fraction of bins."""
    n = len(a)
    a = a.copy()
    N = _next_pow2(2 * n - 1)
    
    for it in range(n_iter):
        a_padded = np.zeros(N)
        a_padded[:n] = a
        A = np.fft.fft(a_padded)  # Full FFT for symmetry
        power = np.abs(A) ** 2
        
        # Find the threshold: peg top fraction
        sorted_power = np.sort(power)[::-1]
        n_peg = max(1, int(peg_fraction * len(power)))
        threshold = sorted_power[n_peg - 1] if n_peg < len(power) else 0
        
        # Water fill: set all pegged bins to threshold
        mask_peg = power > threshold
        n_peg_actual = mask_peg.sum()
        if n_peg_actual == 0:
            break
        
        # The total power is conserved
        total_power = power.sum()
        peg_power = threshold * n_peg_actual
        
        # Remaining power goes to non-pegged bins proportionally
        mask_free = ~mask_peg
        n_free = mask_free.sum()
        if n_free == 0:
            break
        remaining = total_power - peg_power
        if remaining < 0:
            # Can't peg this high, lower threshold
            threshold = total_power / n_peg_actual
            power[mask_peg] = threshold
        else:
            # Scale free bins
            current_free_power = power[mask_free]
            if current_free_power.sum() < 1e-30:
                power[mask_free] = remaining / n_free
            else:
                scale = remaining / current_free_power.sum()
                power[mask_free] = current_free_power * scale
            power[mask_peg] = threshold
        
        # Reconstruct
        new_amp = np.sqrt(power)
        new_phase = np.angle(A)
        A_new = new_amp * np.exp(1j * new_phase)
        
        a_new_padded = np.fft.ifft(A_new)
        a_new = np.real(a_new_padded[:n])
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Check convergence
        if np.allclose(a_new, a, atol=1e-12):
            a = a_new
            break
        a = a_new
    
    return a

        amp_norm = np.maximum(amp_norm, 1e-10)
        amp_compressed = amp_norm ** beta
        A_new = amp_compressed * amp_max * np.exp(1j * phase)
        
        a_new_padded = np.fft.ifft(A_new)
        a_new = np.real(a_new_padded[:n])
        a_new = np.maximum(a_new, 0.0)
        
        if use_symmetry:
            a_new = _symmetrize(a_new)
        
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Check convergence
        if np.max(np.abs(a_new - a)) < 1e-13:
            a = a_new
            break
        a = a_new
    
    return a


def _spectral_soft_blend(a, beta=0.9, n_iter=5, t_blend=0.5, use_symmetry=True):
    """Spectral compression with soft blending to avoid aggressive projection."""
    n = len(a)
    a = a.copy()
    N = _next_pow2(2 * n - 1)
    
    for it in range(n_iter):
        a_padded = np.zeros(N, dtype=np.float64)
        a_padded[:n] = a
        A = np.fft.fft(a_padded)
        amp = np.abs(A)
        phase = np.angle(A)
        
        amp_max = amp.max()
        if amp_max < 1e-15:
            break
        
        amp_norm = np.maximum(amp / amp_max, 1e-10)
        amp_compressed = amp_norm ** beta
        A_new = amp_compressed * amp_max * np.exp(1j * phase)
        
        a_new_padded = np.fft.ifft(A_new)
        a_raw = np.real(a_new_padded[:n])
        a_proj = np.maximum(a_raw, 0.0)
        
        # Blend: keep some of the original to avoid too aggressive change
        a_new = (1.0 - t_blend) * a + t_blend * _normalize(a_proj)
        
        if use_symmetry:
            a_new = _symmetrize(a_new)
        else:
            a_new = _normalize(a_new)
        
        if np.max(np.abs(a_new - a)) < 1e-13:
            a = a_new
            break
        a = a_new
    
    return a


def _peg_waterfill(a, peg_frac=0.2, n_iter=20, use_symmetry=True):
    """Hard water-filling in frequency domain: peg top fraction of bins."""
    n = len(a)
    a = a.copy()
    N = _next_pow2(2 * n - 1)
    
    for it in range(n_iter):
        a_padded = np.zeros(N, dtype=np.float64)
        a_padded[:n] = a
        A = np.fft.fft(a_padded)
        power = np.abs(A) ** 2
        phase = np.angle(A)
        
        p_max = power.max()
        if p_max < 1e-15:
            break
        
        # Peg threshold
        sorted_p = np.sort(power)[::-1]
        n_peg = max(1, int(peg_frac * N))
        threshold = sorted_p[min(n_peg - 1, N - 1)]
        
        mask_peg = power > threshold
        n_pegged = mask_peg.sum()
        if n_pegged == 0:
            break
        n_free = N - n_pegged
        if n_free == 0:
            break
        
        total = power.sum()
        # Set pegged bins to threshold, distribute remaining to free bins
        remaining = total - threshold * n_pegged
        if remaining < 0:
            threshold = total / n_pegged
            remaining = 0
        
        power_new = power.copy()
        power_new[mask_peg] = threshold
        if n_free > 0:
            mask_free = ~mask_peg
            current_free = power[mask_free].sum()
            if current_free > 1e-30:
                power_new[mask_free] = power[mask_free] * (remaining / current_free)
            else:
                power_new[mask_free] = remaining / n_free
        
        amp_new = np.sqrt(power_new)
        A_new = amp_new * np.exp(1j * phase)
        a_new_padded = np.fft.ifft(A_new)
        a_new = np.real(a_new_padded[:n])
        a_new = np.maximum(a_new, 0.0)
        
        if use_symmetry:
            a_new = _symmetrize(a_new)
        else:
            a_new = _normalize(a_new)
        
        s = a_new.sum()
        if s < 1e-15:
            break
        
        if np.max(np.abs(a_new - a)) < 1e-13:
            a = a_new
            break
        a = a_new
    
    return a


def _kkt_refine(a_init, deadline, min_time_remaining=30.0, use_symmetry=True):
    """Improved KKT active-set refinement with backtracking line search."""
    n = len(a_init)
    a = a_init.copy()
    a = _normalize(a)
    if use_symmetry:
        a = _symmetrize(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    tol_frac = 0.01
    min_tol = 0.00005
    no_improve = 0
    max_no_improve = 200
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.7)
            if tol_frac <= min_tol * 1.2:
                break
            no_improve = 0
        
        # Identify active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros(len(c), dtype=bool)
            active[idx] = True
            n_active = 1
        
        # Compute gradient: grad[j] = (2/n_active) * sum_{k in active} a[k-j] * [valid]
        # This equals (2/n_active) * conv(w, a_rev)[n-1+j] where w is active weights
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]
        
        # Support of a
        support = a > 1e-10 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break
        
        # KKT: grad should be constant on support
        lam = float(np.mean(grad[support]))
        residual = np.zeros(n)
        residual[support] = grad[support] - lam
        
        # Backtracking line search
        best_local_a = a.copy()
        best_local_sc = current_sc
        eta = 0.1
        for _ in range(10):
            a_try = a.copy()
            a_try[support] -= eta * residual[support]
            a_try = np.maximum(a_try, 0.0)
            if use_symmetry:
                a_try = _symmetrize(a_try)
            else:
                a_try = _normalize(a_try)
            s_try = a_try.sum()
            if s_try < 1e-10:
                eta *= 0.5
                continue
            a_try = _normalize(a_try)
            if use_symmetry:
                a_try = _symmetrize(a_try)
            sc_try = _score(a_try)
            if sc_try < best_local_sc - 1e-14:
                best_local_a = a_try.copy()
                best_local_sc = sc_try
                if sc_try < current_sc - 1e-10:
                    break
            eta *= 0.5
            if eta < 1e-6:
                break
        
        a = best_local_a
        
        # Tighten tolerance on improvement
        if no_improve == 0:
            tol_frac = max(min_tol, tol_frac * 0.95)
    
    return best_a, best_sc


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_soft(a, beta, n):
    """Gradient of soft_max(conv(a,a)) using structured computation."""
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _smoobj(z, beta, n):
    a = _softmax_safe(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax_safe(z)
    grad_a = _grad_soft(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)


def _softmax_safe(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, verbose=True, use_symmetry=True):
    """L-BFGS with adaptive beta schedule on log-parameterization."""
    n = len(a_init)
    a = _normalize(a_init)
    if use_symmetry:
        a = _symmetrize(a)
    
    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    # Compute gap for beta scheduling
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(200, len(c_sorted) - 1)]
    if gap < c0_max * 0.002:
        gap = c0_max * 0.002
    
    # Fine-grain beta levels (more aggressive than before)
    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
        0.00003125, 0.000015625, 0.0000078, 0.000004, 0.000002, 0.000001
    ])
    betas = gap * multipliers
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        size_factor = max(1.0, n / 8192.0)
        est_iter_time = 0.003 * size_factor
        max_iter = min(6000, max(50, int(remaining / (est_iter_time * 15))))
        
        if beta < gap * 0.001:
            max_iter = min(max_iter, 1500)
        
        def obj_func(z_in):
            return _smoobj(z_in, beta, n)
        
        def grad_func(z_in):
            return _smograd(z_in, beta, n)
        
        try:
            result = minimize(
                obj_func, z, jac=grad_func, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax_safe(result.x)
            if use_symmetry:
                a_new = _symmetrize(a_new)
            else:
                a_new = _normalize(a_new)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"      beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}")
            elif verbose and i < 8:
                print(f"      beta={beta:.2e}: no improvement ({best_sc:.10f})")
        except Exception as e:
            if verbose:
                print(f"      beta={beta:.2e}: error {e}")
    
    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    elif kind == "quartic":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2.0
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "periodic_taper":
        t = np.linspace(0, 1, n)
        window = np.sin(np.pi * t) ** 0.7
        mod = 1.0 + 0.1 * np.cos(2 * np.pi * 13.0 * t)
        a = np.maximum(window * mod, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer with spectral compression as a new ingredient.
    
    Phases:
    P1: Spectral compression exploration (multiple betas)
    P2: KKT active-set refinement
    P3: Soft-max L-BFGS
    P4: Spectral + KKT again on L-BFGS result
    P5: Multi-n exploration (32768)
    P6: Final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass
    
    if parent is not None:
        print(f"Parent loaded: n={len(parent)}")
        best_seq = _normalize(parent)
        best_score = _score(best_seq)
        print(f"Parent score: {best_score:.10f}")
    else:
        best_seq = _make_profile(16384, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n=16384 score: {best_score:.10f}")
    
    n_main = len(best_seq)
    
    # ============ Phase 1: Spectral compression exploration ============
    print(f"\n[P1] Spectral compression exploration (n={n_main}, t={time.monotonic()-t_start:.0f}s)")
    p1_deadline = min(deadline - 600, t_start + 300)
    
    betas_to_try = [0.95, 0.92, 0.88, 0.85, 0.80, 0.75]
    for beta in betas_to_try:
        if time.monotonic() > p1_deadline:
            break
        a_sp = _spectral_compress(best_seq, beta=beta, n_iter=15, use_symmetry=True)
        sc_sp = _score(a_sp)
        if sc_sp < best_score:
            best_score = sc_sp
            best_seq = a_sp.copy()
            print(f"  beta={beta}: score={best_score:.10f} (improved)")
        else:
            # Try soft blend
            a_sb = _spectral_soft_blend(best_seq, beta=beta, n_iter=8, t_blend=0.3, use_symmetry=True)
            sc_sb = _score(a_sb)
            if sc_sb < best_score:
                best_score = sc_sb
                best_seq = a_sb.copy()
                print(f"  soft_blend beta={beta}: score={best_score:.10f} (improved)")
    
    # Also try peg waterfill
    if time.monotonic() + 30 < p1_deadline:
        for peg_f in [0.3, 0.5, 0.7]:
            if time.monotonic() + 20 > p1_deadline:
                break
            a_peg = _peg_waterfill(best_seq, peg_frac=peg_f, n_iter=15, use_symmetry=True)
            sc_peg = _score(a_peg)
            if sc_peg < best_score:
                best_score = sc_peg
                best_seq = a_peg.copy()
                print(f"  peg={peg_f}: score={best_score:.10f} (improved)")
    
    # ============ Phase 2: KKT active-set refinement ============
    print(f"\n[P2] KKT refinement (t={time.monotonic()-t_start:.0f}s)")
    p2_deadline = min(deadline - 400, time.monotonic() + 250)
    a_kkt, sc_kkt = _kkt_refine(best_seq, p2_deadline, min_time_remaining=60, use_symmetry=True)
    if sc_kkt < best_score:
        best_score = sc_kkt
        best_seq = a_kkt.copy()
        print(f"  KKT result: {best_score:.10f}")
    
    # Without symmetry too
    if time.monotonic() + 60 < p2_deadline:
        a_kkt2, sc_kkt2 = _kkt_refine(best_seq, min(deadline - 400, time.monotonic() + 120),
                                       min_time_remaining=30, use_symmetry=False)
        if sc_kkt2 < best_score:
            best_score = sc_kkt2
            best_seq = a_kkt2.copy()
            print(f"  KKT (no sym) result: {best_score:.10f}")
    
    # ============ Phase 3: Soft-max L-BFGS ============
    print(f"\n[P3] L-BFGS (t={time.monotonic()-t_start:.0f}s)")
    p3_deadline = min(deadline - 200, time.monotonic() + 400)
    a_lb, sc_lb = _run_lbfgs(best_seq, p3_deadline, min_time_remaining=100, verbose=True,
                             use_symmetry=True)
    if sc_lb < best_score:
        best_score = sc_lb
        best_seq = a_lb.copy()
        print(f"  L-BFGS result: {best_score:.10f}")
    
    # ============ Phase 4: Spectral + KKT on L-BFGS result ============
    print(f"\n[P4] Spectral + KKT round 2 (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 60 < deadline:
        # Try a few spectral compressions
        for beta in [0.97, 0.94, 0.90]:
            if time.monotonic() + 40 > deadline:
                break
            a_sp4 = _spectral_soft_blend(best_seq, beta=beta, n_iter=5, t_blend=0.2,
                                         use_symmetry=True)
            sc_sp4 = _score(a_sp4)
            if sc_sp4 < best_score:
                best_score = sc_sp4
                best_seq = a_sp4.copy()
                print(f"  spectral beta={beta}: score={best_score:.10f}")
        
        # KKT polish
        if time.monotonic() + 60 < deadline:
            p4_deadline = min(deadline - 100, time.monotonic() + 150)
            a_k4, sc_k4 = _kkt_refine(best_seq, p4_deadline, min_time_remaining=30,
                                       use_symmetry=True)
            if sc_k4 < best_score:
                best_score = sc_k4
                best_seq = a_k4.copy()
                print(f"  KKT round2: {best_score:.10f}")
    
    # ============ Phase 5: Try larger n ============
    print(f"\n[P5] Multi-n exploration (t={time.monotonic()-t_start:.0f}s)")
    n_large = n_main * 2 if n_main <= 16384 else n_main + 8192
    n_large = min(n_large, 65536)  # Cap at 65536 for time reasons
    
    if time.monotonic() + 120 < deadline and n_large != n_main:
        print(f"  Trying n={n_large}")
        a_large = _resample(best_seq, n_large)
        a_large = _symmetrize(a_large)
        sc_large = _score(a_large)
        print(f"  Resampled score (unoptimized): {sc_large:.10f}")
        
        if time.monotonic() + 100 < deadline:
            # Quick KKT on the larger sequence
            p5_deadline = min(deadline - 60, time.monotonic() + 150)
            a_l_kkt, sc_l_kkt = _kkt_refine(a_large, p5_deadline, min_time_remaining=20,
                                             use_symmetry=True)
            if sc_l_kkt < best_score:
                best_score = sc_l_kkt
                best_seq = a_l_kkt.copy()
                print(f"  n={n_large} KKT: {best_score:.10f}")
            
            if time.monotonic() + 60 < deadline and best_seq is a_l_kkt:
                # Quick L-BFGS on larger n
                p5b_deadline = min(deadline - 30, time.monotonic() + 100)
                a_l_lb, sc_l_lb = _run_lbfgs(a_large, p5b_deadline, min_time_remaining=10,
                                              verbose=False, use_symmetry=True)
                if sc_l_lb < best_score:
                    best_score = sc_l_lb
                    best_seq = a_l_lb.copy()
                    print(f"  n={n_large} L-BFGS: {best_score:.10f}")
    
    # Also try a different n (smaller might sometimes help due to discrete effects)
    n_smaller = n_main // 2 if n_main > 8192 else n_main
    if n_smaller != n_main and time.monotonic() + 60 < deadline:
        a_small = _resample(best_seq, n_smaller)
        a_small = _symmetrize(a_small)
        sc_small = _score(a_small)
        if time.monotonic() + 40 < deadline:
            p5c_deadline = min(deadline - 30, time.monotonic() + 60)
            a_s_kkt, sc_s_kkt = _kkt_refine(a_small, p5c_deadline, min_time_remaining=10,
                                             use_symmetry=True)
            if sc_s_kkt < best_score:
                best_score = sc_s_kkt
                best_seq = a_s_kkt.copy()
                print(f"  n={n_smaller} KKT: {best_score:.10f}")
    
    # ============ Phase 6: Final polish ============
    print(f"\n[P6] Final polish (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 20 < deadline:
        p6_deadline = deadline - 5
        # One more L-BFGS with tight beta
        a_final, sc_final = _run_lbfgs(best_seq, p6_deadline, min_time_remaining=5,
                                        verbose=False, use_symmetry=True)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
        
        # One more KKT
        if time.monotonic() + 15 > deadline:
            p6b_deadline = deadline - 3
            a_f2, sc_f2 = _kkt_refine(best_seq, p6b_deadline, min_time_remaining=3,
                                       use_symmetry=True)
            if sc_f2 < best_score:
                best_score = sc_f2
                best_seq = a_f2.copy()
    
    # ============ Final ============
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
        s = final.sum()
        if s < 0.01:
            final = np.ones(256)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
          f"time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
