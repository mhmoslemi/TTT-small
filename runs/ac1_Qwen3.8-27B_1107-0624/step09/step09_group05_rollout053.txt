Multi-phase optimizer using Adam + KKT refinement + perturbation restarts.
Key innovations over previous L-BFGS approach:
- Adam optimizer for faster convergence and better handling of ill-conditioning
- Huber-smoothed max for better conditioning at small smoothing
- Active-set KKT projection for final polish
- Perturbation-based escape from local minima
- Multi-length exploration

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_normalized(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _logsumexp_softmax(c, beta):
    """Compute soft max value and weights."""
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return c_max, w
    return c_max + beta * np.log(s), e / s

def _grad_of_softmax_c(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n-1:2*n-1]

def _adam_step(z, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam update step. Returns updated m, v, z."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z_new, m, v

def _adam_optimize(a_init, deadline, min_time_remaining=30.0, verbose=True):
    """Adam optimization on log-parameters with annealed beta."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    
    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    # Initialize Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    # Beta schedule: start large, end very small
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted)-1)]
    if gap < c0_max * 0.01:
        gap = c0_max * 0.01
    
    # More aggressive schedule
    n_betas = 40
    betas = gap * (0.0001 ** (np.linspace(0, 1, n_betas)))
    # Ensure monotone decreasing
    for i in range(1, n_betas):
        betas[i] = min(betas[i], betas[i-1])
    
    t_start = time.monotonic()
    patience = 0
    max_patience = 20000
    
    # Main loop: iterate through betas
    beta_idx = 0
    while time.monotonic() + min_time_remaining < deadline:
        if beta_idx >= n_betas:
            break
        beta = betas[beta_idx]
        
        # Determine number of iterations for this beta based on remaining time
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Adaptive learning rate: start larger, decay
        base_lr = 0.01 * (n / 16384.0) ** 0.5
        lr = base_lr * (1.0 / (1.0 + 0.01 * t))
        
        # For small beta, use smaller learning rate
        if beta < gap * 0.01:
            lr *= 0.1
        if beta < gap * 0.001:
            lr *= 0.1
        
        max_iter = min(50000, max(500, int(remaining / (0.001 * n / 16384.0))))
        
        for it in range(max_iter):
            if time.monotonic() + min_time_remaining > deadline:
                break
            
            a = _softmax_stable(z)
            c = _full_convolve(a, a)
            soft_val, w = _logsumexp_softmax(c, beta)
            
            # Check actual score
            actual_sc = 2.0 * n * float(c.max())
            if actual_sc < best_sc - 1e-14:
                best_sc = actual_sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            
            if patience > max_patience:
                break
            
            # Gradient
            grad_a = _grad_of_softmax_c(a, w)
            W = float(np.dot(grad_a, a))
            grad_z = a * (grad_a - W)
            
            # Adam step
            t += 1
            z, m, v = _adam_step(z, grad_z, m, v, t, lr)
        
        # Check if we should advance beta
        if patience > max_patience * 0.5:
            beta_idx += 1
        
        if verbose and beta_idx % 5 == 0:
            elapsed = time.monotonic() - t_start
            print(f"      beta={beta:.2e} ({beta_idx+1}/{n_betas}): "
                  f"score={best_sc:.10f}, t={elapsed:.0f}s, patience={patience}")
    
    return best_a, best_sc

def _kkt_refine(a_init, deadline, min_time_remaining=20.0):
    """KKT-based refinement using active set projection."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score_normalized(a)
    
    tol_frac = 0.003
    min_tol = 0.00005
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            tol_frac = max(min_tol, tol_frac * 0.5)
            if tol_frac <= min_tol * 1.1:
                break
            no_improve = 0
        
        # Active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        
        # Average gradient from active set
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n-1:2*n-1]
        
        # Support of a
        support = a > 1e-10 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break
        
        # KKT: grad_avg should be constant on support
        lam = float(np.mean(grad_avg[support]))
        residual = grad_avg[support] - lam
        
        # Update
        found = False
        for eta in [0.1, 0.05, 0.02, 0.01, 0.005]:
            a_new = a.copy()
            a_new[support] -= eta * residual
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            sc_new = _score_normalized(a_new)
            if sc_new < current_sc - 1e-14:
                a = a_new
                found = True
                break
        
        if not found:
            tol_frac = max(min_tol, tol_frac * 0.9)
            continue
        
        if no_improve == 0:
            tol_frac = max(min_tol, tol_frac * 0.98)
    
    return best_a, best_sc

def _perturb_and_optimize(a_init, n_perturb, deadline, min_time_remaining=30.0, verbose=False):
    """Try random perturbations and optimize the best one."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score_normalized(a_init)
    
    rng = np.random.default_rng(12345)
    
    for i in range(n_perturb):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        # Different perturbation strategies
        strat = i % 4
        if strat == 0:
            # Small Gaussian noise
            noise = rng.normal(0, 0.01, n)
            a_pert = a_init + noise
        elif strat == 1:
            # Reduce peaks in convolution
            c = _full_convolve(a_init, a_init)
            c_max = float(c.max())
            active = c >= 0.99 * c_max
            # Find which a[j] contribute most to active peaks
            a_rev = a_init[::-1]
            w = np.zeros(len(c))
            w[active] = 1.0
            wc = _full_convolve(w, a_rev)
            contrib = wc[n-1:2*n-1]
            a_pert = a_init - 0.05 * contrib / max(contrib.max(), 1e-15)
        elif strat == 2:
            # Redistribution: move mass from high-grad to low-grad
            c = _full_convolve(a_init, a_init)
            c_max = float(c.max())
            active = c >= (c_max - 0.01 * c_max)
            a_rev = a_init[::-1]
            w = np.zeros(len(c))
            w[active] = 1.0 / max(1, int(active.sum()))
            wc = _full_convolve(w, a_rev)
            grad = wc[n-1:2*n-1]
            grad_mean = grad.mean()
            a_pert = a_init - 0.03 * (grad - grad_mean)
        else:
            # Random coordinate swap
            a_pert = a_init.copy()
            i1 = rng.integers(0, n)
            i2 = rng.integers(0, n)
            delta = 0.01 * a_pert.max()
            a_pert[i1] -= delta
            a_pert[i2] += delta
        
        a_pert = np.maximum(a_pert, 0.0)
        s_pert = a_pert.sum()
        if s_pert > 1e-10:
            a_pert = a_pert / s_pert
            sc_pert = _score_normalized(a_pert)
            if sc_pert < best_sc:
                best_sc = sc_pert
                best_a = a_pert.copy()
                if verbose:
                    print(f"    Perturb {i}: score={best_sc:.10f}")
    
    return best_a, best_sc

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2*np.pi*11*t/n)
             + 0.15 * np.cos(2*np.pi*5*t/n)
             + 0.1 * np.sin(2*np.pi*23*t/n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n
