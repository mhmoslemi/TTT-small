    top_idx = np.argsort(c)[::-1][:top_k]

    constraints = [b >= 0]
    for k in top_idx:

    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.GUROBI)
    
    if b.value is not None:
        return b.value
    return None

"""
LP-Direction + Spectral Shaping + Fine Polish optimizer for minimizing
2*n*max(convolve(a,a))/sum(a)^2.

Key innovations:
- LP-based direction finding (solving the dual/primal LP for active constraints)
- Convex 1-D line search (score is convex in the mixing parameter t)
- Spectral (FFT-domain) shaping as an alternative direction
- Adaptive active set management
- Multi-n exploration

import numpy as np
import time
from scipy.optimize import minimize, linprog, minimize_scalar
from scipy.sparse import coo_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _score_normalized(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())


def _solve_lp_direction(f, M, top_k=50):
    """
    Solve the LP:
    maximize sum(b)
    subject to (f*b)_k <= M for k in top_k peaks
               b >= 0
    Returns the optimal b vector, or None if infeasible.
    """
    n = len(f)
    c = _full_convolve(f, f)
    top_idx = np.argsort(c)[::-1][:top_k]
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(top_idx):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        js = np.arange(j_lo, j_hi + 1)
        ks = k - js
        rows.extend([idx] * len(js))
        cols.extend(js.tolist())
        vals.extend(f[ks].tolist())
    
    A_ub = coo_matrix((vals, (rows, cols)), shape=(top_k, n))
    b_ub = np.full(top_k, M)
    
    result = linprog(
        c=-np.ones(n),
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method='highs',
        options={'presolve': True, 'time_limit': 30}
    )
    
    if result.success:
        b = result.x
        if b.sum() > 1e-15:
            return b, top_idx
    return None, top_idx


def _line_search_lp(f, b, n):
    """
    Find optimal t in [0,1] for f_new = (1-t)*f + t*b.
    The score as a function of t is convex (max of quadratics with non-negative
    leading coefficients), so we use bounded scalar optimization.
    """
    def score_t(t):
        f_new = (1 - t) * f + t * b
        c = _full_convolve(f_new, f_new)
        return 2.0 * n * c.max()
    
    result = minimize_scalar(score_t, bounds=(0, 1), method='bounded',
                             options={'xatol': 1e-6})
    return result.x, result.fun


def _lp_phase(f_init, deadline, min_time_remaining=100):
    """LP-based iterative refinement."""
    n = len(f_init)
    f = _normalize(f_init.copy())
    best_f = f.copy()
    best_sc = _score_normalized(f)
    
    top_k = 30
    max_top_k = 200
    min_top_k = 10
    stall = 0
    max_stall = 30
    total_iter = 0
    
    print(f"  [LP Phase] Starting: n={n}, score={best_sc:.10f}")
    
    while time.monotonic() + min_time_remaining < deadline:
        total_iter += 1
        c = _full_convolve(f, f)
        M = float(c.max())
        current_sc = 2.0 * n * M
        
        if current_sc < best_sc - 1e-12:
            best_sc = current_sc
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
        
        # Adapt top_k
        if stall > max_stall:
            top_k = min(max_top_k, top_k * 2)
            stall = 0
        elif stall == 0 and top_k > min_top_k:
            top_k = max(min_top_k, int(top_k * 0.8))
        
        # Solve LP
        b, top_idx = _solve_lp_direction(f, M, top_k)
        if b is None:
            # LP failed, try smaller top_k
            top_k = max(min_top_k, top_k // 2)
            continue
        
        b = _normalize(b)
        
        # Line search
        t_opt, sc_opt = _line_search_lp(f, b, n)
        
        if t_opt > 1e-6 and sc_opt < current_sc - 1e-12:
            f = (1 - t_opt) * f + t_opt * b
            f = np.maximum(f, 0)
            f = _normalize(f)
            sc_new = _score_normalized(f)
            if sc_new < best_sc:
                best_sc = sc_new
                best_f = f.copy()
                if total_iter % 50 == 1:
                    print(f"    iter {total_iter}: score={best_sc:.10f}, t={t_opt:.4f}, top_k={top_k}")
        else:
            # No improvement, slightly perturb
            f = _normalize(f + 0.001 * np.random.standard_normal(n))
    
    return best_f, best_sc


def _spectral_phase(f_init, deadline, min_time_remaining=60):
    """FFT-domain shaping."""
    n = len(f_init)
    N = _next_pow2(2 * n - 1)
    f = _normalize(f_init.copy())
    best_f = f.copy()
    best_sc = _score_normalized(f)
    
    print(f"  [Spectral Phase] Starting: n={n}, N={N}, score={best_sc:.10f}")
    
    # Get initial spectrum
    A = np.fft.rfft(f, N)
    phase = np.angle(A)
    C = np.abs(A) ** 2
    
    lr = 0.1
    for it in range(500):
        if time.monotonic() + min_time_remaining >= deadline:
            break
        
        # Current autoconvolution
        c = np.fft.irfft(C, N)[:2 * n - 1]
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-12:
            best_sc = current_sc
            best_f = f.copy()
        
        # Find peaks
        n_peaks = max(5, int(2 * n * 0.005))
        peak_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        
        # Compute gradient of max(c) w.r.t. C
        # c = IFFT(C), so dc/dC[omega] = exp(2πi*omega*k/N) for peak k
        # We want to reduce C where it contributes to peaks
        grad_C = np.zeros_like(C)
        for k in peak_idx:
            # Contribution of frequency omega to c[k] is Re(C[omega] * exp(2πi*omega*k/N))
            omega = np.arange(len(C))
            contrib = np.cos(2 * np.pi * omega * k / N)
            grad_C += contrib
        grad_C = grad_C / n_peaks
        
        # Update C: reduce where gradient is positive (contributes to peaks)
        C_new = C - lr * grad_C
        C_new = np.maximum(C_new, 0)
        
        # Ensure C is consistent with a valid spectrum (real, non-negative)
        # Reconstruct signal
        A_new = np.sqrt(C_new) * np.exp(1j * phase)
        A_new[0] = np.sqrt(C_new[0])  # DC: keep real
        f_new = np.fft.irfft(A_new, N)[:n]
        
        # Project
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-10:
            f = best_f.copy()
            C = np.abs(np.fft.rfft(f, N)) ** 2
            lr *= 0.5
            continue
        f_new = f_new / s
        
        # Update
        f = f_new
        A = np.fft.rfft(f, N)
        phase = np.angle(A)
        C = np.abs(A) ** 2
        
        # Adaptive learning rate
        c_new_check = np.fft.irfft(C, N)[:2 * n - 1]
        if c_new_check.max() > c_max * 1.1:
            lr *= 0.7
        else:
            lr = min(0.5, lr * 1.02)
    
    return best_f, best_sc


def _topk_smoobj_grad(z, beta, K, n):
    """Top-K log-sum-exp smoothed objective and gradient in log-domain."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c
    
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    
    # Gradient
    n_2nm1 = 2 * n - 1
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    ga = 2.0 * wc[n - 1:2 * n - 1]
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z

    """Fine L-BFGS with top-K smoothing."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score_normalized(a)
    current_a = a.copy()

        return current_a

    if gap < max_c * 0.001:
        gap = max_c * 0.001

    z = z - z.mean()

For the linear convolution c[k] = sum_i a[i] * a[k-i] where the sum is over all i such that 0 <= i < n and 0 <= k-i < n.

So for c_f[k]:
c_f[k] = sum_{i: max(0,k-n+1) <= i <= min(n-1,k)} f[i] * f[k-i]

Let's define the valid range: i from i_lo to i_hi where i_lo = max(0, k-n+1) and i_hi = min(n-1, k).

Then:
c_f[k] = sum_{i=i_lo}^{i_hi} f[i] * f[k-i]
h[k] = sum_{i=i_lo}^{i_hi} f[i] * b[k-i]
c_b[k] = sum_{i=i_lo}^{i_hi} b[i] * b[k-i]

Now, c_f[k] + c_b[k] - 2*h[k] = sum_{i=i_lo}^{i_hi} (f[i]*f[k-i] + b[i]*b[k-i] - 2*f[i]*b[k-i])

This is NOT the same as sum (f[i] - b[k-i])^2 because the terms don't match up:
(f[i] - b[k-i])^2 = f[i]^2 - 2f[i]*b[k-i] + b[k-i]^2

But we have f[i]*f[k-i] + b[i]*b[k-i] - 2f[i]*b[k-i].

These are different! So the t^2 coefficient is:
sum (f[i]*f[k-i] + b[i]*b[k-i] - 2*f[i]*b[k-i])

This is not obviously non-negative. Let me check with a simple example:
n=2, k=1:
c_f[1] = f[0]*f[1] + f[1]*f[0] = 2*f[0]*f[1]
h[1] = f[0]*b[1] + f[1]*b[0]
c_b[1] = 2*b[0]*b[1]

t^2 coeff = 2f[0]f[1] + 2b[0]b[1] - 2(f[0]b[1] + f[1]b[0])
= 2(f[0]f[1] + b[0]b[1] - f[0]b[1] - f[1]b[0])
= 2(f[0](f[1]-b[1]) + b[0](b[1]-f[1]))
= 2(f[0]-b[0])(f[1]-b[1])

This can be negative! For example, if f[0] > b[0] and f[1] < b[1], then (f[0]-b[0])(f[1]-b[1]) < 0.

So the score is NOT necessarily convex in t! This means I can't use a simple convex 1-D optimizer.

"""
LP-Direction + Spectral Shaping + Fine Polish optimizer.
Minimizes 2*n*max(convolve(a,a))/sum(a)^2 using:
- LP-based direction finding with adaptive active sets
- FFT-domain spectral peak reduction
- Top-K smoothed L-BFGS polishing
- Multi-restart with diverse initializations

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import coo_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a.copy()
    return a / s


def _solve_lp_direction(f, M, top_k):
    n = len(f)
    c = _full_convolve(f, f)
    top_idx = np.argsort(c)[::-1][:top_k]

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(top_idx):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        js = np.arange(j_lo, j_hi + 1)
        ks = k - js
        rows.extend([idx] * len(js))
        cols.extend(js.tolist())
        vals.extend(f[ks].tolist())

    A_ub = coo_matrix((vals, (rows, cols)), shape=(top_k, n))
    b_ub = np.full(top_k, M)

    result = linprog(
        c=-np.ones(n),
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method='highs',
        options={'presolve': True, 'time_limit': 20}
    )

    if result.success:
        b = result.x
        if b.sum() > 1e-15:
            return b, top_idx
    return None, top_idx


def _line_search_mix(f, b, n):
    best_t = 0.0
    best_sc = _score_norm(f)
    c_f = _full_convolve(f, f)
    c_b = _full_convolve(b, b)
    c_fb = _full_convolve(f, b)

    for t in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
        c_mix = (1-t)**2 * c_f + 2*t*(1-t) * c_fb + t**2 * c_b
        sc = 2.0 * n * float(c_mix.max())
        if sc < best_sc:
            best_sc = sc
            best_t = t
    return best_t, best_sc


def _lp_phase(f_init, deadline, min_time_remaining):
    n = len(f_init)
    f = _normalize(f_init.copy())
    best_f = f.copy()
    best_sc = _score_norm(f)

    top_k = 25
    max_top_k = 150
    min_top_k = 10
    stall = 0
    max_stall = 25
    total_iter = 0
    t_decay = 1.0

    print(f"  [LP] Start: n={n}, score={best_sc:.10f}")

    while time.monotonic() + min_time_remaining < deadline:
        total_iter += 1
        c = _full_convolve(f, f)
        M = float(c.max())
        current_sc = 2.0 * n * M

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_f = f.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            top_k = min(max_top_k, top_k * 2)
            stall = 0
        elif stall == 0 and top_k > min_top_k:
            top_k = max(min_top_k, int(top_k * 0.85))

        b, top_idx = _solve_lp_direction(f, M, top_k)
        if b is None:
            top_k = max(min_top_k, top_k // 2)
            f = _normalize(f + 0.002 * np.random.standard_normal(n))
            continue

        b = _normalize(b)
        t_opt, sc_opt = _line_search_mix(f, b, n)

        if t_opt > 1e-5 and sc_opt < current_sc - 1e-13:
            f = (1 - t_opt) * f + t_opt * b
            f = np.maximum(f, 0)
            f = _normalize(f)
            sc_new = _score_norm(f)
            if sc_new < best_sc:
                best_sc = sc_new
                best_f = f.copy()
                if total_iter % 100 == 1:
                    print(f"    iter {total_iter}: {best_sc:.10f} t={t_opt:.4f} k={top_k}")
        else:
            f = _normalize(f + 0.001 * np.random.standard_normal(n))

        if total_iter % 200 == 0:
            print(f"    iter {total_iter}: best={best_sc:.10f} cur={current_sc:.10f} k={top_k} stall={stall}")

    return best_f, best_sc


def _spectral_phase(f_init, deadline, min_time_remaining):
    n = len(f_init)
    N = _next_pow2(2 * n - 1)
    f = _normalize(f_init.copy())
    best_f = f.copy()
    best_sc = _score_norm(f)
    print(f"  [Spec] Start: n={n}, score={best_sc:.10f}")

    A = np.fft.rfft(f, N)
    phase = np.angle(A)
    C = np.abs(A) ** 2
    lr = 0.05
    it = 0

    while time.monotonic() + min_time_remaining < deadline:
        it += 1
        c = np.fft.irfft(C, N)[:2 * n - 1]
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_f = f.copy()

        n_peaks = max(5, int(2 * n * 0.003))
        peak_idx = np.argpartition(c, -n_peaks)[-n_peaks:]

        omega = np.arange(len(C))
        grad_C = np.zeros(len(C))
        for k in peak_idx:
            grad_C += np.cos(2.0 * np.pi * omega * k / N)
        grad_C /= n_peaks

        C_new = C - lr * grad_C
        C_new = np.maximum(C_new, 0)

        sqrt_C = np.sqrt(C_new)
        A_new = sqrt_C * np.exp(1j * phase)
        A_new[0] = sqrt_C[0]
        f_new = np.fft.irfft(A_new, N)[:n]

        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-10:
            lr *= 0.5
            C = np.abs(np.fft.rfft(f, N)) ** 2
            continue
        f_new = f_new / s

        f = f_new
        A = np.fft.rfft(f, N)
        phase = np.angle(A)
        C = np.abs(A) ** 2

        c_check = np.fft.irfft(C, N)[:2 * n - 1]
        if c_check.max() > c_max * 1.05:
            lr *= 0.7
        else:
            lr = min(0.3, lr * 1.03)

        if it % 100 == 0:
            print(f"    spec iter {it}: best={best_sc:.10f} lr={lr:.4f}")

    return best_f, best_sc


class _TopkSmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def obj_val(self, z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if self.K < len(c):
            top_idx = np.argpartition(c, -self.K)[-self.K:]
            c_top = c[top_idx]
        else:
            top_idx = np.arange(len(c))
            c_top = c
        shifted = (c_top - c_max) / self.beta
        shifted = np.clip(shifted, -700, 0)
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * self.n * c_max
        return 2.0 * self.n * (c_max + self.beta * np.log(s))

    def grad_val(self, z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if self.K < len(c):
            top_idx = np.argpartition(c, -self.K)[-self.K:]
            c_top = c[top_idx]
        else:
            top_idx = np.arange(len(c))
            c_top = c
        shifted = (c_top - c_max) / self.beta
        shifted = np.clip(shifted, -700, 0)
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(2 * self.n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            e_full = np.zeros(2 * self.n - 1)
            e_full[top_idx] = e
            w = e_full / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        ga = 2.0 * wc[self.n - 1:2 * self.n - 1]
        W = float(np.dot(a, ga))
        return a * (ga - W)

    def __call__(self, z):
        return self.obj_val(z)

    def grad(self, z):
        return self.grad_val(z)


def _run_lbfgs(a_init, deadline, min_time_remaining):
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score_norm(a)
    current_a = a.copy()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    schedule = [
        (1.0, 0.06), (0.6, 0.04), (0.3, 0.03), (0.15, 0.02),
        (0.08, 0.015), (0.04, 0.01), (0.02, 0.008), (0.01, 0.005),
        (0.005, 0.003), (0.002, 0.002),
    ]

    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        beta = base_beta * beta_mult
        K = max(5, int((2 * n - 1) * K_frac))
        max_iter = min(5000, max(30, int(remaining / 0.04)))

        prob = _TopkSmoProb(beta, K, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-17, 'gtol': 1e-13,
                         'maxcor': 100, 'maxls': 50}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score_norm(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_profile(n, kind):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-25.0 * (t - 0.3) ** 2) + np.exp(-25.0 * (t - 0.7) ** 2) + 0.03
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        fade = int(n * 0.1)
        for i in range(fade):
            a[i] = (i + 1) / fade
            a[n - 1 - i] = (i + 1) / fade
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

                best_seq = a_p.copy()
                best_score = sc_p
                print(f"  NEW BEST: {best_score:.10f} [profile_{prof_kind}]")

    # Phase 7: Multi-n exploration
    print(f"\n[P7] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        n_main = len(best_seq)
        n_options = [16384, 8192, 24576, 12288]
        n_options = [x for x in n_options if x != n_main]
        for n_try in n_options[:2]:
            if time.monotonic() > deadline - 20:
                break
            a_res = _resample(best_seq, n_try)
            a_opt = _run_lbfgs(a_res, min(deadline - 10, time.monotonic() + 25),
                               min_time_remaining=3)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [n={n_try}]")

    # Phase 8: Final L-BFGS polish
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - time.monotonic()
    if remaining > 5:
        a_fin = _run_lbfgs(best_seq, deadline - 1, min_time_remaining=1)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final]")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
