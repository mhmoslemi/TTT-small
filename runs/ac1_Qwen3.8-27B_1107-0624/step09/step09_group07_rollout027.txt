- Multi-scale L-BFGS with top-K smoothing for coarse optimization
- Adaptive momentum and periodic restarts to escape local minima
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _grad_single_peak(a, k, n):
    """Gradient of c_k = sum_i a_i * a_{k-i} w.r.t. a."""
    g = np.zeros(n)
    j_lo = max(0, k - (n - 1))
    j_hi = min(n - 1, k)
    for j in range(j_lo, j_hi + 1):
        idx = k - j
        if 0 <= idx < n:
            g[j] += 2.0 * a[idx]
    return g


def _grad_single_peak_vec(a, k, n):
    """Vectorized gradient of c_k w.r.t. a."""
    g = np.zeros(n)
    j_lo = max(0, k - (n - 1))
    j_hi = min(n - 1, k)
    idx_lo = k - j_hi
    idx_hi = k - j_lo
    g[j_lo:j_hi + 1] = 2.0 * a[idx_lo:idx_hi + 1][::-1]
    return g


def _freq_filter(direction, n, cutoff_frac=0.3):
    """Apply a smooth low-pass filter to the direction in frequency domain."""
    N = _next_pow2(n)
    D = np.fft.rfft(direction, N)
    freqs = np.fft.rfftfreq(N) * N
    # Smooth tanh filter
    omega_c = cutoff_frac * N / 2.0
    if omega_c < 1:
        omega_c = 1.0
    mask = np.tanh(freqs / (omega_c + 1e-10))
    D_filtered = D * mask
    result = np.fft.irfft(D_filtered, N)[:n]
    return result


def _topk_smoobj_grad(z, beta, K, n):
    """Top-K log-sum-exp smoothed objective and gradient in log-domain."""
    z_max = z.max()
    a = np.exp(z - z_max)
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())

    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c

    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()

    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))

    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


class _TopkSmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[1]


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with top-K smoothing and adaptive K/beta annealing."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.008)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    schedule = [
        (1.0, 0.1), (0.7, 0.08), (0.5, 0.06), (0.3, 0.04),
        (0.2, 0.03), (0.1, 0.02), (0.05, 0.015), (0.03, 0.01),
        (0.01, 0.008), (0.005, 0.005), (0.002, 0.003), (0.001, 0.002),
    ]

    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        beta = base_beta * beta_mult
        K = max(5, int((2 * n - 1) * K_frac))

        est_time = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(6000, max(50, int(remaining / (est_time * 20))))

        prob = _TopkSmoProb(beta, K, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 50}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgrad_with_filter(a_init, deadline, min_time_remaining=60.0):
    """Exact subgradient with frequency-domain filtering and BB steps."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    n_2nm1 = 2 * n - 1
    active_frac = 0.02
    momentum = 0.5
    lr = 0.01
    lr_min = 1e-5
    lr_max = 0.05
    stall = 0
    max_stall = 100
    velocity = np.zeros(n)
    freq_cutoff = 0.2
    prev_a = None
    prev_g = None
    iteration = 0
    last_lbfgs_time = time.monotonic()

    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive active set: peaks within margin of max
        margin = max(c_max * 0.01, c_max * np.exp(-iteration / 500.0))
        active_thresh = c_max - margin
        active_mask = c >= active_thresh
        n_active = int(active_mask.sum())
        if n_active == 0:
            active_mask[int(np.argmax(c))] = True
            n_active = 1
        if n_active > n_2nm1 * active_frac:
            top_idx = np.argpartition(c, -int(n_2nm1 * active_frac))[-int(n_2nm1 * active_frac):]
            active_mask = np.zeros(n_2nm1, dtype=bool)
            active_mask[top_idx] = True
            n_active = int(active_mask.sum())

        # Compute weighted gradient
        active_vals = c[active_mask]
        w = np.zeros(n_2nm1)
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w[active_mask] = (active_vals - active_vals.min()) / spread + 0.1
        else:
            w[active_mask] = 1.0 / n_active
        w = w / w.sum()

        g = _grad_maxconv(a, w)

        # Frequency filter
        g = _freq_filter(g, n, freq_cutoff)

        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break

        # BB step size
        if prev_a is not None and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g_centered - prev_g
            s_norm = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm > 1e-18 and abs(sy) > 1e-18:
                lr_bb = s_norm / abs(sy)
                lr = np.clip(lr_bb, lr_min, lr_max)

        # Momentum step with line search
        velocity = momentum * velocity - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()

        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            for ls in range(20):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-15:
                    improved = True
                    break
                velocity = momentum * velocity * 0.5 - lr * 0.5 * g_centered
                a_try = a + velocity
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    velocity = np.zeros(n)
                    a_try = a
                    break
                a_try = a_try / s_try
                lr *= 0.7

        if improved:
            prev_a = a.copy()
            prev_g = g_centered.copy()
            a = a_try
            lr = min(lr_max, lr * 1.02)
            freq_cutoff = min(0.5, freq_cutoff * 1.005)
        else:
            velocity = np.zeros(n)
            lr = max(lr_min, lr * 0.6)
            momentum = min(0.8, momentum * 1.02)

        # Adapt on stall
        if stall > max_stall:
            active_frac = min(0.08, active_frac * 1.5)
            lr = max(lr_min, lr * 0.5)
            momentum = min(0.9, momentum + 0.05)
            freq_cutoff = max(0.05, freq_cutoff * 0.7)
            stall = 0
            # Reset to best
            if stall > 0:
                a = best_a.copy()
                velocity = np.zeros(n)
                prev_a = None
                prev_g = None

    return best_a, best_sc


def _lp_frank_wolfe(a_init, deadline, min_time_remaining=60.0):
    """LP-guided Frank-Wolfe for the linearized peak minimization problem."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n_2nm1 = 2 * n - 1

    gamma = 0.1
    momentum = 0.3
    velocity = np.zeros(n)
    stall = 0
    max_stall = 200
    active_count = 20
    freq_cutoff = 0.15
    lp_count = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
            gamma = min(0.3, gamma * 1.1)
        else:
            stall += 1

        # Select active peaks
        n_active = min(n_2nm1, active_count + stall // 50)
        top_idx = np.argsort(c)[::-1][:n_active]

        # Build LP
        # Variables: a[0..n-1], t  (total n+1 variables)
        # Minimize t
        # Constraints: grad_k . a - t <= c_k(a_current) for k in active set
        #               sum(a) = 1
        #               a >= 0

        c_active = c[top_idx]
        # Build constraint matrix (n_active rows, n+1 columns)
        A_ineq = np.zeros((n_active, n + 1))
        b_ineq = np.zeros(n_active)

        for ii, k in enumerate(top_idx):
            # Gradient of c_k at current a
            gk = np.zeros(n)
            j_lo = max(0, k - (n - 1))
            j_hi = min(n - 1, k)
            idx = k - np.arange(j_lo, j_hi + 1)
            idx = idx[::-1]  # reverse because k-j decreases as j increases
            # Actually: for j in [j_lo, j_hi], the contribution is 2*a[k-j]
            idx = k - np.arange(j_lo, j_hi + 1)
            gk[j_lo:j_hi + 1] = 2.0 * a[idx]
            A_ineq[ii, :n] = gk
            A_ineq[ii, n] = -1.0
            b_ineq[ii] = c[ii]  # c_k(a_current)

        # Equality: sum(a) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        # Objective: minimize t
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        # Bounds
        bounds = [(0, None)] * n + [(None, None)]

        t_lp_start = time.monotonic()
        try:
            res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 10})
        except Exception:
            res = None

        if res is not None and res.success and res.x is not None:
            a_lp = res.x[:n]
            a_lp = np.maximum(a_lp, 0.0)
            s_lp = a_lp.sum()
            if s_lp > 1e-10:
                a_lp = a_lp / s_lp

                # Frank-Wolfe step with momentum
                direction = a_lp - a
                velocity = momentum * velocity + (1 - momentum) * direction
                velocity = _freq_filter(velocity, n, freq_cutoff)
                velocity -= velocity.mean()  # Preserve sum

                # Line search
                improved = False
                for ls_gamma in [gamma, gamma * 0.5, gamma * 0.25, gamma * 0.1,
                                 gamma * 0.5 * 1.2, gamma * 0.3]:
                    a_try = a + ls_gamma * velocity
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    sc_try = 2.0 * n * float(c_try.max())
                    if sc_try < current_sc - 1e-15:
                        a = a_try
                        improved = True
                        break

                if improved:
                    lp_count += 1
                    # Slowly increase active set and freq cutoff
                    if lp_count % 50 == 0:
                        active_count = min(50, active_count + 1)
                        freq_cutoff = min(0.4, freq_cutoff * 1.02)
                else:
                    velocity *= 0.5
        else:
            # LP failed, fall back to subgradient
            w = np.zeros(n_2nm1)
            w[top_idx] = 1.0 / n_active
            g = _grad_maxconv(a, w)
            g = _freq_filter(g, n, freq_cutoff)
            g_centered = g - g.mean()
            a_try = a - 0.01 * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-10:
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-15:
                    a = a_try
                    improved = True

        # Stall handling
        if stall > max_stall:
            a = best_a.copy()
            velocity = np.zeros(n)
            gamma = 0.05
            momentum = 0.5
            active_count = 10
            freq_cutoff = 0.1
            stall = 0
            max_stall = max(50, max_stall // 2)

        if stall > 50:
            gamma = max(0.02, gamma * 0.8)

    return best_a, best_sc


def _direct_peak_reduction(a_init, deadline, min_time_remaining=30.0):
    """Directly reduce the top few peaks by adjusting contributing elements."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    patience = 0
    max_patience = 300

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break

        # Find top peaks and compute combined gradient
        n_peaks = max(3, min(n, int(n * 0.003)))
        peak_idx = np.argsort(c)[::-1][:n_peaks]

        # Combined gradient direction
        g = np.zeros(n)
        for k in peak_idx:
            j_lo = max(0, k - (n - 1))
            j_hi = min(n - 1, k)
            idx = k - np.arange(j_lo, j_hi + 1)
            g[j_lo:j_hi + 1] += 2.0 * a[idx]

        g = _freq_filter(g, n, 0.2)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break

        # Line search
        alpha = 0.02 / (gnorm + 1e-15)
        improved = False
        for ls in range(30):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            patience += 5

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.4 + 0.6 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2) + 0.03
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "super_tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.clip(1.0 - np.abs(2.0 * t - 1.0) * 1.5, 0.1, 1.0)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    """Very fine active-set polish with small steps."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.003
    tol = 0.003
    no_improve = 0
    freq_cutoff = 0.1

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Active set: peaks within tol of max
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active

        g = _grad_maxconv(a, w)
        g = _freq_filter(g, n, freq_cutoff)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-16:
            break

        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        improved = False
        for ls in range(40):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                tol = max(0.0001, tol * 0.6)
                lr *= 0.6
                freq_cutoff = max(0.02, freq_cutoff * 0.8)
            if no_improve > 400:
                a = best_a.copy()
                tol = max(0.0001, tol * 0.5)
                lr = 0.002
                no_improve = 0
        else:
            if tol > 0.0001:
                tol = max(0.0001, tol * 0.97)
            lr = min(0.005, lr * 1.01)

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid optimizer: L-BFGS smoothing, exact subgradient with frequency filtering,
    LP-guided Frank-Wolfe, direct peak reduction, multi-restart, and fine polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, tapered n={n_parent}, score={best_score:.10f}")

    n = len(best_seq)

    # === Phase 1: L-BFGS on parent ===
    print(f"\n[P1] L-BFGS top-K (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        a1 = _run_lbfgs(best_seq, min(deadline - 550, time.monotonic() + 300),
                        min_time_remaining=550)
        sc1 = _score(a1)
        if sc1 < best_score:
            best_seq = a1.copy()
            best_score = sc1
            print(f"  NEW BEST: {best_score:.10f} [lbfgs1] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Subgradient with frequency filtering ===
    print(f"\n[P2] Subgrad + freq filter (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        a2, sc2 = _subgrad_with_filter(best_seq, min(deadline - 450, time.monotonic() + 250),
                                       min_time_remaining=450)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [subgrad] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: LP-guided Frank-Wolfe ===
    print(f"\n[P3] LP Frank-Wolfe (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a3, sc3 = _lp_frank_wolfe(best_seq, min(deadline - 350, time.monotonic() + 300),
                                  min_time_remaining=350)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [lp_fw] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Second L-BFGS from improved point ===
    print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a4 = _run_lbfgs(best_seq, min(deadline - 250, time.monotonic() + 200),
                        min_time_remaining=250)
        sc4 = _score(a4)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [lbfgs2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: LP-guided Frank-Wolfe round 2 ===
    print(f"\n[P5] LP FW round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        a5, sc5 = _lp_frank_wolfe(best_seq, min(deadline - 200, time.monotonic() + 200),
                                  min_time_remaining=200)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [lp_fw2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Direct peak reduction ===
    print(f"\n[P6] Direct peak reduction (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a6, sc6 = _direct_peak_reduction(best_seq, min(deadline - 150, time.monotonic() + 150),
                                         min_time_remaining=150)
        if sc6 < best_score:
            best_seq = a6.copy()
            best_score = sc6
            print(f"  NEW BEST: {best_score:.10f} [peakred] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Multi-restart exploration ===
    print(f"\n[P7] Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        restart_budget = min(200, (deadline - 150 - time.monotonic()) * 0.6)
        n_restart = max(2, int(restart_budget / 35))
        sigmas = [0.03, 0.05, 0.08, 0.12, 0.06, 0.04, 0.1, 0.15, 0.02, 0.07]
        for i in range(n_restart):
            if time.monotonic() > deadline - 100:
                break
            sigma = sigmas[i % len(sigmas)]
            a_pert = _perturb(best_seq, sigma, rng)
            # Quick LP-FW from perturbed start
            a_r, sc_r = _lp_frank_wolfe(a_pert, min(deadline - 80, time.monotonic() + 30),
                                        min_time_remaining=5)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.10f} [restart_{i}] t={time.monotonic()-t_start:.1f}s")

        # Try different profiles
        profiles = ["cos_taper", "gauss", "bimodal", "super_tapered", "plateau"]
        for prof_kind in profiles:
            if time.monotonic() > deadline - 60:
                break
            a_prof = _make_profile(n, prof_kind)
            a_p, sc_p = _lp_frank_wolfe(a_prof, min(deadline - 40, time.monotonic() + 25),
                                        min_time_remaining=5)
            if sc_p < best_score:
                best_seq = a_p.copy()
                best_score = sc_p
                print(f"  NEW BEST: {best_score:.10f} [profile_{prof_kind}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final fine polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 10:
        a_fin, sc_fin = _fine_polish(best_seq, deadline - 5, min_time_remaining=5)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [polish] t={time.monotonic()-t_start:.1f}s")

    # === Final tiny L-BFGS ===
    remaining = deadline - 3 - time.monotonic()
    if remaining > 5:
        n_fin = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        beta_fin = max(mb * 0.0001, 1e-10)
        K_fin = max(5, int(n_fin * 0.002))
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        prob = _TopkSmoProb(beta_fin, K_fin, n_fin)
        max_iter = min(2000, max(10, int(remaining / 0.05)))
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxcor': 50})
            a_fin2 = np.exp(result.x - result.x.max())
            a_fin2 = a_fin2 / a_fin2.sum()
            sc_fin2 = _score(a_fin2)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
        except Exception:
            pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
