        # CG solve for H u = g_tangent
        g_t = g - np.mean(g)
        x = np.zeros(n)
        r = g_t - _hess_vp(x)
        p = r.copy()
        rs = np.dot(r, r)
        gnorm = np.linalg.norm(g_t)
        converged = False
        for cg_it in range(300):
            Ap = _hess_vp(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha_cg = rs / pAp
            x = x + alpha_cg * p
            r = r - alpha_cg * Ap
            rs_new = np.dot(r, r)
            if np.sqrt(rs_new) < 1e-12 * (1 + gnorm):
                converged = True
                break
            p = r + (rs_new / rs) * p
            rs = rs_new

        # Solve for the equality constraint multiplier
        w = np.zeros(n)
        ones = np.ones(n)
        r = ones - _hess_vp(w)
        p = r.copy()
        rs = np.dot(r, r)
        for cg_it in range(200):
            Ap = _hess_vp(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha_cg = rs / pAp
            w = w + alpha_cg * p
            r = r - alpha_cg * Ap
            rs_new = np.dot(r, r)
            if np.sqrt(rs_new) < 1e-12:
                break
            p = r + (rs_new / rs) * p
            rs = rs_new

        w_sum = float(w.sum())
        if abs(w_sum) < 1e-15:
            dx = -x
        else:
            lam = float(x.sum()) / w_sum
            dx = -x + lam * w

        dx = dx - np.mean(dx)

        # Armijo line search
        f_cur = c_max
        improved = False
        t = 1.0
        for ls in range(40):
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_cur - 1e-14 * t * np.linalg.norm(dx):
                a = a_try
                improved = True
                break
            t *= 0.5

        if improved:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        else:
            # Try subgradient fallback
            k_star = int(np.argmax(c))
            g_sg = _subgrad_direction(a, k_star, n)
            g_sg = 2.0 * n * g_sg
            g_sg_t = g_sg - np.mean(g_sg)
            gn = float(np.dot(g_sg_t, g_sg_t))
            if gn > 1e-30:
                alpha_sg = min(1.0, (f_cur - 1.0 / (2 * n)) / (gn + 1e-20) * 0.1)
                a_try = a - alpha_sg * g_sg_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_sc:
                        best_sc = sc_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True

        if not improved:
            break

        # Check convergence
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-16:
            break

    return best_a, best_sc


def _lbfgs_continuation(a_init, deadline, min_time_remaining=60.0):
    """
    L-BFGS with extended beta continuation schedule.
    40 stages going from moderate to very small beta.
    """
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 100:
        gap = c_sorted[0] - c_sorted[50]
    elif len(c_sorted) > 10:
        gap = c_sorted[0] - c_sorted[len(c_sorted) // 2]
    else:
        gap = c_sorted[0] - c_sorted[-1]
    base_beta = max(gap * 0.5, max_c * 0.01)

    # Extended schedule: 40 stages
    n_betas = 40
    log_start = np.log(base_beta)
    log_end = np.log(max(max_c * 1e-9, 1e-30))
    log_betas = np.linspace(log_start, log_end, n_betas)
    betas = np.exp(log_betas)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.005 * max(1.0, n / 8192.0)
        # More iterations for smaller beta (harder problem)
        if i < n_betas // 3:
            max_iter = min(2000, max(50, int(remaining / (est_time * 20))))
        elif i < 2 * n_betas // 3:
            max_iter = min(3000, max(100, int(remaining / (est_time * 10))))
        else:
            max_iter = min(5000, max(200, int(remaining / (est_time * 5))))

        try:
            def _obj_fn(z_val):
                return _lsobj(z_val, beta, n)

            def _grad_fn(z_val):
                return _lsgrad(z_val, beta, n)

            result = minimize(
                _obj_fn,
                z,
                jac=_grad_fn,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-15,
                    'maxcor': 200,
                    'maxls': 500,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            if i % 10 == 0:
                print(f"    L-BFGS stage {i}/{n_betas} beta={beta:.2e}: score={sc_new:.12f}")
        except Exception as ex:
            print(f"    L-BFGS stage {i}/{n_betas}: exception {ex}")

    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "bilateral":
        t = np.linspace(0, 1, n)
        a = np.exp(-2 * np.abs(t - 0.5))
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _optimize_sequence(a_init, deadline, min_time=30.0, label=""):
    """
    Run a full optimization pipeline on a sequence.
    """
    best_a = a_init.copy()
    best_sc = _score(a_init)

    # Step 1: L-BFGS continuation
    if time.monotonic() < deadline - min_time:
        lb_dl = min(deadline - min_time, time.monotonic() + (deadline - time.monotonic()) * 0.4)
        a1 = _lbfgs_continuation(best_a, lb_dl, min_time_remaining=min_time)
        sc1 = _score(a1)
        if sc1 < best_sc:
            best_sc = sc1
            best_a = a1.copy()
            print(f"  [{label}] After L-BFGS: {best_sc:.12f}")

    # Step 2: Active-set Newton
    if time.monotonic() < deadline - min_time:
        an_dl = min(deadline - min_time, time.monotonic() + (deadline - time.monotonic()) * 0.25)
        a2, sc2 = _active_set_newton(best_a, an_dl, max_outer=100)
        if sc2 < best_sc:
            best_sc = sc2
            best_a = a2.copy()
            print(f"  [{label}] After Newton: {best_sc:.12f}")

    # Step 3: Subgradient polish
    if time.monotonic() < deadline - min_time:
        sg_dl = min(deadline - min_time, time.monotonic() + (deadline - time.monotonic()) * 0.2)
        a3, sc3 = _subgradient_polish(best_a, sg_dl, max_iter=20000, label=label)
        if sc3 < best_sc:
            best_sc = sc3
            best_a = a3.copy()
            print(f"  [{label}] After Subgrad: {best_sc:.12f}")

    # Step 4: Final L-BFGS polish
    if time.monotonic() < deadline - min_time + 10:
        fin_dl = min(deadline - 5, time.monotonic() + 30)
        a4 = _lbfgs_continuation(best_a, fin_dl, min_time_remaining=10)
        sc4 = _score(a4)
        if sc4 < best_sc:
            best_sc = sc4
            best_a = a4.copy()
            print(f"  [{label}] Final polish: {best_sc:.12f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    1. L-BFGS log-sum-exp continuation (extended 40-stage schedule)
    2. Active-set Newton with adaptive tolerance
    3. Projected subgradient with Polyak step
    4. Multi-scale (16384 -> 32768)
    5. Diversified restarts
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 40.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: {best_score:.12f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent if available
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={len(parent)}, score={sc_parent:.12f}")
        update_best(parent, sc_parent, "parent")
        n_work = len(parent)
    else:
        n_work = 16384
        parent = _make_profile(n_work, "uniform")
        update_best(parent, _score(parent), "init_uniform")

    # === Phase 1: Main optimization at original n ===
    print(f"\n[Phase 1] Main optimization at n={n_work} (t={time.monotonic() - t_start:.1f}s)")
    p1_dl = min(deadline - 600, time.monotonic() + 500)
    a_main, sc_main = _optimize_sequence(best_seq, p1_dl, min_time=600, label="main")
    update_best(a_main, sc_main, "phase1")

    # === Phase 2: Subgradient intensification ===
    print(f"\n[Phase 2] Subgradient intensification (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        sg_dl = min(deadline - 400, time.monotonic() + 200)
        # Run subgradient with more iterations and tighter step control
        a_sg, sc_sg = _subgradient_polish(best_seq, sg_dl, max_iter=100000, label="intensify")
        update_best(a_sg, sc_sg, "subgrad_intensify")

    # === Phase 3: Newton refinement on improved sequence ===
    print(f"\n[Phase 3] Newton refinement (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        nw_dl = min(deadline - 300, time.monotonic() + 150)
        a_nw, sc_nw = _active_set_newton(best_seq, nw_dl, max_outer=300)
        update_best(a_nw, sc_nw, "newton_refine")

    # === Phase 4: L-BFGS with ultra-small beta ===
    print(f"\n[Phase 4] Ultra-small beta L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        ul_dl = min(deadline - 250, time.monotonic() + 150)
        a_ul = _lbfgs_continuation(best_seq, ul_dl, min_time_remaining=250)
        update_best(a_ul, _score(a_ul), "ultra_beta")

    # === Phase 5: Multi-scale - try n=32768 ===
    print(f"\n[Phase 5] Multi-scale n=32768 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        n_large = 32768
        a_large = _interpolate_seq(best_seq, n_large)
        sc_large_init = _score(a_large)
        print(f"  n={n_large} interpolated init score: {sc_large_init:.12f}")
        update_best(a_large, sc_large_init, "interp_32768")

        if sc_large_init < best_score or True:  # Always try to optimize
            p5_dl = min(deadline - 100, time.monotonic() + 250)
            a_large_opt, sc_large_opt = _optimize_sequence(a_large, p5_dl, min_time=100, label="n32768")
            update_best(a_large_opt, sc_large_opt, "n32768_optimized")

    # === Phase 6: Diversified restarts ===
    print(f"\n[Phase 6] Diversified restarts (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        for sigma_val in [0.01, 0.03, 0.08]:
            if time.monotonic() > deadline - 60:
                break
            p_start = _perturb(best_seq, sigma_val, rng)
            r_dl = min(deadline - 50, time.monotonic() + 40)
            a_r = _lbfgs_continuation(p_start, r_dl, min_time_remaining=50)
            update_best(a_r, _score(a_r), f"restart_s{sigma_val}")

    # === Phase 7: Final tight cycle ===
    print(f"\n[Phase 7] Final tight cycle (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 15:
        # Quick subgradient burst
        sg_fin_dl = min(deadline - 10, time.monotonic() + 15)
        a_fin, sc_fin = _subgradient_polish(best_seq, sg_fin_dl, max_iter=50000, label="final")
        update_best(a_fin, sc_fin, "final_subgrad")

        # Quick L-BFGS
        if time.monotonic() < deadline - 8:
            lf_dl = min(deadline - 5, time.monotonic() + 10)
            a_lf = _lbfgs_continuation(best_seq, lf_dl, min_time_remaining=5)
            update_best(a_lf, _score(a_lf), "final_lbfgs")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
