Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
New approach: combines spectral shaping (Gerchberg-Saxton), stochastic active-set
subgradient with Nesterov momentum, L-BFGS with adaptive top-K smoothing,
basin hopping, and multi-n exploration.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _topk_lse_obj_grad(z, beta, K, n):
    """Top-K log-sum-exp smoothed objective and gradient in log-domain."""
    a = np.exp(z - z.max())
    s_a = a.sum()
    if s_a < 1e-300:
        a = np.ones(n) / n
    else:
        a = a / s_a
    c = _full_convolve(a, a)
    c_max = float(c.max())

    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c

    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -500, 0)
    e = np.exp(shifted)
    s_e = e.sum()

    if s_e < 1e-300 or not np.isfinite(s_e):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s_e
        obj = 2.0 * n * (c_max + beta * np.log(s_e))

    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z

        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):

    def grad(self, z):
        return _topk_lse_obj_grad(z, self.beta, self.K, self.n)[1]


def _gs_iterations(a_init, deadline, min_time_remaining=30.0):
    """Gerchberg-Saxton style spectral shaping iterations."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    N = _next_pow2(2 * n - 1)

    alpha = 0.3
    alpha_min = 0.01
    alpha_max = 2.0
    no_improve = 0
    max_no_improve = 50

    t_start = time.monotonic()
    while time.monotonic() + 5 < deadline:
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        phase = np.angle(A)

        c_perm = np.fft.irfft(P, N)
        c_shifted = np.roll(c_perm, n - 1)
        c_approx = c_shifted[max(0, n - 1):max(0, n - 1) + (2 * n - 1)]

        c_mean = c_approx.mean()
        c_excess = np.clip(c_approx - c_mean * 1.1, 0, None)

        if c_excess.max() < 1e-15:
            break

        E = np.fft.rfft(c_excess, N)
        E_abs = np.abs(E)
        E_norm = E_abs / max(E_abs.max(), 1e-15)

        P_new = P * np.exp(-alpha * E_norm)
        P_new = np.maximum(P_new, 1e-30)

        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            alpha *= 0.5

        a_new = a_new / s_new

            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
            alpha = min(alpha_max, alpha * 1.02)
        else:
            no_improve += 1
            alpha = max(alpha_min, alpha * 0.85)
            if no_improve > max_no_improve:
                break

    return best_a, best_sc


def _subgradient_nesterov(a_init, deadline, min_time_remaining=30.0):
    """Stochastic active-set subgradient with Nesterov momentum."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n_2nm1 = 2 * n - 1

    momentum = 0.75
    lr = 0.03
    lr_min = 1e-5
    lr_max = 0.15
    stall = 0
    max_stall = 100
    velocity = np.zeros(n)
    y_lookahead = a.copy()

    iteration = 0
    t_start = time.monotonic()

    while time.monotonic() + 5 < deadline:
        iteration += 1
        a_eval = np.maximum(y_lookahead, 0.0)
        s_eval = a_eval.sum()
        if s_eval < 1e-10:
            a_eval = a
            s_eval = a.sum()
        a_eval = a_eval / s_eval

        c = _full_convolve(a_eval, a_eval)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a_eval.copy()
            stall = 0
        else:
            stall += 1

        # Stochastic active set: sample peaks with probability proportional to value
        n_active = max(5, min(n_2nm1, int(n_2nm1 * 0.005) + 5))
        c_sorted_idx = np.argsort(c)[::-1]
        # Mix of top peaks and random peaks for diversity
        n_top = max(2, n_active // 2)
        top_peaks = c_sorted_idx[:n_top]
        n_rand = n_active - n_top
        rand_peaks = c_sorted_idx[np.random.randint(n_top, min(n_2nm1, n_2nm1 - 1), size=n_rand)]
        active_idx = np.concatenate([top_peaks, rand_peaks])

        active_vals = c[active_idx]
        w = np.zeros(n_2nm1)
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w[active_idx] = (active_vals - active_vals.min()) / spread + 0.05
        else:
            w[active_idx] = 1.0 / n_active
        w = w / w.sum()

        g = _grad_maxconv(a_eval, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break

        # Nesterov step
        velocity = momentum * velocity - lr * g_centered
        y_lookahead = a + velocity
        a_try = np.maximum(y_lookahead, 0.0)
        s_try = a_try.sum()

        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            for ls in range(20):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-15:
                    improved = True
                    break
                velocity = momentum * velocity * 0.5 - lr * 0.5 * g_centered
                y_lookahead = a + velocity
                a_try = np.maximum(y_lookahead, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    velocity = np.zeros(n)
                    y_lookahead = a
                    break
                a_try = a_try / s_try
                lr *= 0.7

        if improved:
            a = a_try
            lr = min(lr_max, lr * 1.02)
        else:
            velocity = np.zeros(n)
            y_lookahead = a.copy()
            lr = max(lr_min, lr * 0.6)

        if stall > max_stall:
            lr = max(lr_min, lr * 0.3)
            momentum = min(0.9, momentum + 0.03)
            velocity = np.zeros(n)
            y_lookahead = a.copy()
            stall = 0
            # Occasionally restart from best
            if stall % (2 * max_stall) == 0:
                a = best_a.copy()

    return best_a, best_sc


def _run_lbfgs_multi(a_init, deadline, min_time_remaining=30.0):
    """Multi-stage L-BFGS with aggressive annealing."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    schedule = [
        (1.0, 0.15), (0.7, 0.10), (0.5, 0.07), (0.3, 0.05),
        (0.2, 0.04), (0.12, 0.03), (0.07, 0.025), (0.04, 0.015),
        (0.02, 0.01), (0.01, 0.008), (0.005, 0.005), (0.002, 0.003),
        (0.001, 0.002), (0.0005, 0.001),
    ]

    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        beta = base_beta * beta_mult
        K = max(3, int((2 * n - 1) * K_frac))

        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(30, int(remaining / (est_time * 25))))

        prob = _SmoProb(beta, K, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxcor': 100, 'maxls': 50}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return a


def _fine_polish(a_init, deadline, min_time_remaining=5.0):
    """Iterative refinement targeting peaks within tolerance of max."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.003
    tol = 0.003
    no_improve = 0
    max_no_improve = 500

    while time.monotonic() + 2 < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active

        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-17:
            break

        alpha = min(lr, 0.3 / (gnorm + 1e-15))
        improved = False
        for ls in range(35):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-16:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                tol = max(0.0003, tol * 0.7)
                lr = max(0.0005, lr * 0.7)
            if no_improve > max_no_improve:
                a = best_a.copy()
                tol = max(0.0003, tol * 0.5)
                lr = 0.002
                no_improve = 0
        else:
            if tol > 0.0003:
                tol = max(0.0003, tol * 0.96)
            lr = min(0.008, lr * 1.01)

    return best_a, best_sc


def _make_profile(n, kind="tapered", **kwargs):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        sigma = kwargs.get('sigma', 0.25)
        center = kwargs.get('center', 0.5)
        a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * t) + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym2":
        t = np.linspace(0, 1, n)
        a = 0.1 + 0.9 * np.exp(-2.0 * (1 - t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2) + 0.02
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "platform":
        t = np.linspace(0, 1, n)
        a = np.where(np.abs(t - 0.5) < 0.3, 1.0, 0.3)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "octave":
        t = np.linspace(0, 1, n)
        a = 0.2 + 0.8 * np.cos(2 * np.pi * t) ** 2 * np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "optimal_cont":
        # Approximation of optimal continuous profile (wider than Gaussian)
        t = np.linspace(0, 1, n)
        sigma = kwargs.get('sigma', 0.38)
        center = kwargs.get('center', 0.5)
        a = np.exp(-0.5 * ((t - center

) / sigma) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample_to_n(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _perturb_shift(a, shift, rng):
    """Circular shift + noise perturbation."""
    n = len(a)
    s = int(rng.integers(-shift, shift + 1))
    a_shifted = np.roll(a, s)
    z = np.log(np.maximum(a_shifted, 1e-30))
    z = z + 0.02 * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining GS spectral shaping, Nesterov subgradient,
    multi-stage L-BFGS, basin hopping, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        best_seq = _make_profile(n_parent, "tapered")
        best_score = _score(best_seq)
        print(f"No parent, tapered n={n_parent}, score={best_score:.10f}")

    n_main = len(best_seq)
    improved = False

    # === Phase 1: GS Spectral Shaping ===
    print(f"\n[P1] GS Spectral (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        gs_deadline = min(deadline - 550, time.monotonic() + 120)
        a_gs, sc_gs = _gs_iterations(best_seq, gs_deadline, min_time_remaining=10)
        if sc_gs < best_score:
            best_score = sc_gs
            best_seq = a_gs.copy()
            improved = True
        print(f"  GS: score={sc_gs:.10f}, best={best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # === Phase 2: Multi-stage L-BFGS ===
    print(f"\n[P2] L-BFGS multi-stage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        lbfgs_deadline = min(deadline - 450, time.monotonic() + 300)
        a_lb = _run_lbfgs_multi(best_seq, lbfgs_deadline, min_time_remaining=50)
        sc_lb = _score(a_lb)
        if sc_lb < best_score:
            best_score = sc_lb
            best_seq = a_lb.copy()
            improved = True
        print(f"  L-BFGS: score={sc_lb:.10f}, best={best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # === Phase 3: Nesterov Subgradient ===
    print(f"\n[P3] Nesterov subgradient (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        sg_deadline = min(deadline - 300, time.monotonic() + 200)
        a_sg, sc_sg = _subgradient_nesterov(best_seq, sg_deadline, min_time_remaining=30)
        if sc_sg < best_score:
            best_score = sc_sg
            best_seq = a_sg.copy()
            improved = True
        print(f"  Subgrad: score={sc_sg:.10f}, best={best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # === Phase 4: Basin Hopping ===
    print(f"\n[P4] Basin hopping (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        hop_budget = min(200, (deadline - 200 - time.monotonic()))
        n_hops = max(3, int(hop_budget / 45))

        for i in range(n_hops):
            if time.monotonic() > deadline - 60:
                break

            sigma = [0.05, 0.1, 0.08, 0.15, 0.06, 0.12, 0.2, 0.04][i % 8]
            a_pert = _perturb_log(best_seq, sigma, rng)

            # Short L-BFGS from perturbed point
            hop_deadline = min(deadline - 50, time.monotonic() + 35)
            a_hop = _run_lbfgs_multi(a_pert, hop_deadline, min_time_remaining=8)
            sc_hop = _score(a_hop)

            if sc_hop < best_score:
                best_score = sc_hop
                best_seq = a_hop.copy()
                improved = True
                print(f"  Hop {i}: NEW BEST {best_score:.10f} (sigma={sigma}) t={time.monotonic()-t_start:.1f}s")
            else:
                # Try perturbing the best directly with a shift
                if i % 2 == 1:
                    a_shift = _perturb_shift(best_seq, n_main // 8, rng)
                    a_hop2 = _run_lbfgs_multi(a_shift, min(deadline - 50, time.monotonic() + 25), min_time_remaining=8)
                    sc_hop2 = _score(a_hop2)
                    if sc_hop2 < best_score:
                        best_score = sc_hop2
                        best_seq = a_hop2.copy()
                        improved = True
                        print(f"  Shift {i}: NEW BEST {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: L-BFGS again from best ===
    print(f"\n[P5] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        lbfgs2_deadline = min(deadline - 150, time.monotonic() + 180)
        a_lb2 = _run_lbfgs_multi(best_seq, lbfgs2_deadline, min_time_remaining=40)
        sc_lb2 = _score(a_lb2)
        if sc_lb2 < best_score:
            best_score = sc_lb2
            best_seq = a_lb2.copy()
            improved = True
        print(f"  L-BFGS2: score={sc_lb2:.10f}, best={best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # === Phase 6: Fine Polish ===
    print(f"\n[P6] Fine polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        polish_deadline = min(deadline - 80, time.monotonic() + 100)
        a_pol, sc_pol = _fine_polish(best_seq, polish_deadline, min_time_remaining=10)
        if sc_pol < best_score:
            best_score = sc_pol
            best_seq = a_pol.copy()
            improved = True
        print(f"  Polish: score={sc_pol:.10f}, best={best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        n_main = len(best_seq)
        # Try nearby n values that are powers of 2 or close
        n_options = [n_main // 2, n_main, n_main * 2]
        # Filter: must be >= 1024 and <= 65536, and different from each other
        n_options = [x for x in n_options if 1024 <= x <= 65536]
        # Remove duplicates
        seen = set()
        n_unique = []
        for x in n_options:
            if x not in seen:
                seen.add(x)
                n_unique.append(x)
        n_options = n_unique

        for n_try in n_options:
            if time.monotonic() > deadline - 30:
                break
            if n_try == len(best_seq):
                continue
            a_res = _resample_to_n(best_seq, n_try)
            sc_res = _score(a_res)
            # Quick L-BFGS
            if time.monotonic() < deadline - 25:
                a_opt = _run_lbfgs_multi(a_res, min(deadline - 20, time.monotonic() + 30), min_time_remaining=5)
                sc_opt = _score(a_opt)
            else:
                sc_opt = sc_res
                a_opt = a_res

            best_for_n = min(sc_res, sc_opt)
            if best_for_n < best_score:
                best_score = best_for_n
                if sc_opt <= sc_res:
                    best_seq = a_opt.copy()
                else:
                    best_seq = a_res.copy()
                improved = True
                print(f"  n={n_try}: NEW BEST {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Diverse profile restarts ===
    print(f"\n[P8] Profile restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        profiles = [
            ("optimal_cont", {'sigma': 0.38, 'center': 0.5}),
            ("optimal_cont", {'sigma': 0.42, 'center': 0.5}),
            ("optimal_cont", {'sigma': 0.35, 'center': 0.45}),
            ("gauss", {'sigma': 0.3, 'center': 0.5}),
            ("asym", {}),
            ("bimodal", {}),
            ("octave", {}),
        ]

        for prof_kind, prof_kwargs in profiles:
            if time.monotonic() > deadline - 30:
                break
            a_prof = _make_profile(n_best, prof_kind
, **prof_kwargs)
            a_p = _run_lbfgs_multi(a_prof, min(deadline - 25, time.monotonic() + 25), min_time_remaining=5)
            sc_p = _score(a_p)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_p.copy()
                improved = True
                print(f"  Profile {prof_kind}: NEW BEST {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 9: Final fine polish ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _fine_polish(best_seq, deadline - 2, min_time_remaining=2)
        if sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.copy()
            improved = True
        print(f"  Final: score={sc_fin:.10f}, best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

        # Tiny L-BFGS with very small beta
        remaining = deadline - 2 - time.monotonic()
        if remaining > 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = max(mb * 0.0003, 1e-10)
            K_fin = max(3, int(n * 0.002))
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _SmoProb(beta_fin, K_fin, n)
            max_iter = min(5000, max(20, int(remaining / 0.02)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-20,
                                           'gtol': 1e-15, 'maxcor': 100})
                a_fin2 = np.exp(result.x - result.x.max())
                a_fin2 = np.maximum(a_fin2, 1e-30)
                a_fin2 = a_fin2 / a_fin2.sum()
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_score = sc_fin2
                    best_seq = a_fin2.copy()
                    improved = True
                    print(f"  Final L-BFGS: NEW BEST {best_score:.10f} t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    total_time = time.monotonic() - t_start
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, improved={improved}, time={total_time:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
