"""
Optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Multi-start with structured perturbations (frequency-domain, block, random)
2. L-BFGS log-sum-exp with staircase beta schedule (hold-and-decrease)
3. High-precision active-set Newton with adaptive active set and cubic line search
4. Fine local search with adaptive perturbations
5. Iterative refinement cycles
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    """Compute correlation: result[i] = sum_j a[j] * A_vec[j+i-n+1] (valid range)"""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _lsobj_factory(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad_factory(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec, rhs, max_cg=300, tol=1e-14):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _newton_step_precise(a, c, c_max, tol_frac):
    """Compute one Newton step with precise active set."""
    n = len(a)
    active = c >= (c_max - tol_frac * c_max)
    if not active.any():
        active[np.argmax(c)] = True
    m = int(active.sum())
    if m < 2 or m > 15 * n:
        return None, None

    A_vec = active.astype(np.float64) / m
    g = 2.0 * _corr_a_A(a, A_vec, n)
    g_t = g - np.mean(g)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-15:
        return None, None

    mu = 5e-8 / m + 1e-11
    A_vec_copy = A_vec
    n_copy = n
    mu_copy = mu

    def _hess_vp(v):
        return 2.0 * _corr_a_A(v, A_vec_copy, n_copy) + mu_copy * v

    u = _cg_solve(_hess_vp, g_t, max_cg=300)
    w = _cg_solve(_hess_vp, np.ones(n), max_cg=300)
    w_sum = float(w.sum())

    if abs(w_sum) < 1e-15:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w

    dx = dx - np.mean(dx)
    return dx, A_vec


def _cubic_line_search(a, dx, c_max, best_sc, max_ls=80):
    """Cubic interpolation line search."""
    n = len(a)
    t = 1.0
    best_t = 0.0
    best_val = c_max
    t_prev = 0.0
    val_prev = c_max

    for ls in range(max_ls):
        a_try = a + t * dx
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            t *= 0.5
            continue
        a_try = a_try / s_try
        c_try = _full_convolve(a_try, a_try)
        f_try = float(c_try.max())

        if f_try < best_val:
            best_val = f_try
            best_t = t

        # Cubic interpolation for next step
        if ls >= 2:
            # Simple backtracking
            t = t * 0.7
        else:
            t = t * 0.5

        if best_t > 0 and t < best_t * 0.01:
            break

    if best_t > 0:
        a_best = a + best_t * dx
        a_best = np.maximum(a_best, 0)
        s_best = a_best.sum()
        if s_best < 1e-15:
            return a, c_max
        a_best = a_best / s_best
        return a_best, best_val
    return a, c_max


def _active_set_newton_precise(a_init, deadline, max_outer=50):
    """High-precision active-set Newton with adaptive active set."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()

        return a, _score(a)

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Try multiple tolerance levels
        tol_list = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]

        best_a_new = None
        best_f_new = c_max

        for tol_frac in tol_list:
            dx, A_vec = _newton_step_precise(a, c, c_max, tol_frac)
            if dx is None:
                continue

            a_try, f_try = _cubic_line_search(a, dx, c_max, _score(a))
            if f_try < best_f_new - 1e-15:
                best_f_new = f_try
                best_a_new = a_try

        if best_a_new is None:
            # Gradient fallback
            g_all = 2.0 * _corr_a_A(a, np.ones(2 * n - 1) / (2 * n - 1), n)
            g_t = g_all - np.mean(g_all)

                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)

                    if f_try < c_max - 1e-14:
                        best_a_new = a_try
                        best_f_new = f_try

            break

        a = best_a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Check convergence
        c_new = _full_convolve(a, a)

def _lbfgs_staircase(a_init, deadline, min_time_remaining=30.0):
    """
    L-BFGS with staircase beta schedule:
    Hold beta at each level for multiple optimization calls, then decrease.
    This allows better convergence at each smoothness level.
    """
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Staircase schedule: fewer, longer holds
    n_levels = 12
    # Start at moderate beta, end at very small beta
    beta_start = max_c * 0.005
    beta_end = max_c * 2e-8
    log_betas = np.linspace(np.log(beta_start), np.log(beta_end), n_levels)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(log_betas):
        beta = np.exp(beta)
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Hold at this beta for up to 3 L-BFGS calls
        n_hold = 3
        for hold in range(n_hold):
            remaining = deadline - min_time_remaining - time.monotonic()
            if remaining < 3:
                break
            est_time = 0.002 * max(1.0, n / 8192.0)
            max_iter = min(3000, max(50, int(remaining / (est_time * 15))))

            try:
                def _obj(z_val):
                    return _lsobj_factory(z_val, beta, n)

                def _grad(z_val):
                    return _lsgrad_factory(z_val, beta, n)

                result = minimize(
                    _obj,
                    z,
                    jac=_grad,
                    method='L-BFGS-B',
                    options={
                        'maxiter': max_iter,
                        'ftol': 1e-18,
                        'gtol': 1e-14,
                        'maxcor': 120,
                        'maxls': 300,
                    }
                )
                a_new = _softmax(result.x)
                sc_new = _score(a_new)
                if sc_new < current_sc - 1e-17:
                    current_a = a_new
                    current_sc = sc_new
                    z = np.log(np.maximum(a_new, 1e-30))
                    z = z - z.mean()
                else:
                    # No improvement, break early for this hold level
                    break
            except Exception:
                break

        if time.monotonic() + min_time_remaining > deadline:
            break

    return current_a


def _subgradient_momentum(a_init, deadline, n_iters=50000):
    """
    Nesterov momentum subgradient method with multi-peak weighting.
    Used to escape local minima.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    velocity = np.zeros(n)
    momentum = 0.9
    alpha_0 = 0.1

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        y = a + momentum * velocity

        # Compute peaks
        c = _full_convolve(y, y)
        c_max = float(c.max())
        c_mean = float(c.mean())

        # Multi-peak weighting: weight peaks proportional to excess over mean
        threshold = c_mean + 0.3 * (c_max - c_mean)
        peak_mask = c > threshold
        if not peak_mask.any():
            peak_mask[np.argmax(c)] = True

        weights = np.zeros_like(c)
        excess = np.maximum(c - threshold, 0)
        if excess.sum() > 0:
            weights[peak_mask] = excess[peak_mask]
            weights = weights / weights.sum()
        else:
            peak_mask2 = c >= c_max * 0.99
            if not peak_mask2.any():
                peak_mask2[np.argmax(c)] = True
            weights[peak_mask2] = 1.0 / peak_mask2.sum()

        # Compute combined gradient
        # g[i] = sum_k w[k] * (a[k-i] + a[i-k]) (where valid)
        # Use FFT-based correlation
        w_vec = weights
        a_rev = a[::-1]
        wc = _full_convolve(w_vec, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]
        grad = grad - np.mean(grad)

        # Step size
        alpha = alpha_0 / (1.0 + 0.01 * it)

        # Update
        velocity = momentum * velocity - alpha * grad
        a_new = a + velocity
        a_new = _project_simplex(a_new)

        # Check improvement
        if it % 500 == 0:
            sc_new = _score(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()

        a = a_new
        # Gradually increase momentum
        momentum = min(0.99, 0.9 + 0.0001 * it)

    # Final score
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()

    return best_a


def _freq_domain_perturb(a, eta, freq, rng):
    """Perturb a in frequency domain: modify magnitude spectrum."""
    n = len(a)
    F = np.fft.rfft(a)
    mag = np.abs(F)
    phase = np.angle(F)
    target = np.exp(1j * freq * np.pi / n)
    # Reduce peaks in spectrum
    mag_mean = mag.mean()
    smoothed_mag = mag * (1.0 - eta) + mag_mean * eta
    # Add small random phase perturbation
    phase_noise = eta * 0.1 * rng.standard_normal(len(phase))
    phase = phase + phase_noise
    F_new = smoothed_mag * np.exp(1j * phase)
    a_new = np.fft.irfft(F_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _block_perturb(a, n_blocks, rng, eta):
    """Perturb block means while preserving within-block structure."""
    n = len(a)
    block_size = n // n_blocks
    a_new = a.copy()
    for b in range(n_blocks):
        start = b * block_size
        end = start + block_size if b < n_blocks - 1 else n
        block_sum = a[start:end].sum()
        new_mean = block_sum * (1.0 + eta * rng.standard_normal())
        if new_mean > 0:
            scale = new_mean / max(block_sum, 1e-15)
            a_new[start:end] *= scale
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _random_perturb(a, sigma, rng):
    """Log-domain random perturbation."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

        update_best(a_fin, sc_fin, "final_newton")

        # One more L-BFGS with very small beta
        if time.monotonic() < deadline - 25:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 1e-8
            remaining = deadline - 25 - time.monotonic()
            if remaining > 5:
                max_iter_fin = min(1500, max(50, int(remaining / 0.05)))
                try:
                    def _obj_fin(z_val):
                        return _lsobj_factory(z_val, beta_final, n)

                    def _grad_fin(z_val):
                        return _lsgrad_factory(z_val, beta_final, n)

                    result = minimize(
                        _obj_fin,
                        z,
                        jac=_grad_fin,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter_fin, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 100}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_tiny_beta")
                except Exception:
                    pass

    # === Phase 8: Local search with micro-perturbations (50s) ===
    print(f"\n[Phase 8] Local search (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        p8_deadline = min(deadline - 10, t_start + 1160)
        n = len(best_seq)
        for attempt in range(200):
            if time.monotonic() > p8_deadline:
                break
            # Very small perturbation
            sigma = 0.001 * (1.0 + attempt / 50.0)
            a_try = _random_perturb(best_seq, sigma, rng)
            # Quick Newton polish
            a_try2, sc_try2 = _active_set_newton_precise(a_try, min(p8_deadline, time.monotonic() + 3), max_outer=5)
            update_best(a_try2, sc_try2, f"local_{attempt}")
            if time.monotonic() > p8_deadline:
                break

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
