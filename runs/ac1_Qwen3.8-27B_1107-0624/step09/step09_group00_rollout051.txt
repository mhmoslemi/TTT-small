"""
Improved optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:
- Variance-aware objective to promote flat convolution
- Two-stage optimization (variance then peak)
- Better active-set handling with exact identification
- Targeted ILS perturbations based on gradient structure
- More aggressive multi-n exploration

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos2":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos3":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) ** 3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(12345)
        a = rng.random(n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _variance_objective(a):
    """Compute variance of the self-convolution (proxy for flatness)."""
    c = _full_convolve(a, a)
    return float(np.var(c))

def _variance_gradient(a):
    """Gradient of sum(C[k]^2) with respect to a."""
    n = len(a)
    c = _full_convolve(a, a)
    a_rev = a[::-1]
    # grad[j] = 4 * sum_{m=0}^{n-1} C[j+m] * a[m]
    # = 4 * (a_rev * C)[n-1 + j] for j = 0, ..., n-1
    ac = _full_convolve(a_rev, c)
    grad = 4.0 * ac[n - 1:2 * n - 1]
    grad = grad - np.mean(grad)
    return grad

def _variance_optimize(a_init, deadline, min_time_remaining=30.0):
    """Gradient descent on variance of self-convolution."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_var = _variance_objective(a)
    best_sc = _score(a)
    lr = 0.01
    momentum = np.zeros(n)
    mu = 0.9
    
    while time.monotonic() + min_time_remaining < deadline:
        grad = _variance_gradient(a)
        gn = np.linalg.norm(grad)
        if gn < 1e-16:
            break
        # Normalize gradient for consistent step sizes
        grad = grad / gn
        momentum = mu * momentum + grad
        a_new = _normalize(a - lr * momentum)
        var_new = _variance_objective(a_new)
        if var_new < best_var - 1e-15:
            best_var = var_new
            best_a = a_new.copy()
        if var_new < _variance_objective(a) * (1 - 1e-10):
            a = a_new
            if best_sc > _score(a):
                best_sc = _score(a)
                best_a = a.copy()
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
    
    return best_a, _score(best_a)

def _active_set_gradient(a, c, n_peaks=10):
    """Compute gradient based on top n_peaks of the convolution."""
    n = len(a)
    c_max = float(c.max())
    # Get indices of top n_peaks
    k = min(n_peaks, len(c))
    top_idx = np.argsort(c)[::-1][:k]
    # Weight by how close to max
    eps = 1e-12
    w = c[top_idx] / (c_max + eps)
    w = w / w.sum()
    
    # grad[j] = 2 * sum_k w[k] * a[k-j]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad = 2.0 * h
    grad = grad - np.mean(grad)
    return grad

def _active_set_polish(a_init, deadline, min_time_remaining=20.0,
                       tol_frac_start=0.01, tol_frac_end=0.0001,
                       lr_init=0.03, momentum=0.85, n_peaks=20):
    """Active-set gradient with momentum and adaptive tolerance."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    v = np.zeros(n)
    tol_frac = tol_frac_start
    lr = lr_init
    no_improve = 0
    max_stall = 300
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        # Identify active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            idx = int(np.argmax(c))
            active[idx] = True
        
        # Weighted gradient
        eps = 1e-12 * c_max
        w = np.zeros_like(c)
        w[active] = 1.0 / (c_max - c[active] + eps)
        w = w / w.sum()
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        v = momentum * v - lr * grad
        alpha = 1.0
        improved = False
        for _ in range(50):
            a_try = _normalize(a + alpha * v)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            v = np.zeros(n)
            no_improve += 1
            if no_improve % 30 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.6)
                lr *= 0.7
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(tol_frac_end, tol_frac * 0.3)
                lr = lr_init * 0.3
                no_improve = 0
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.98)
            lr = min(lr_init, lr * 1.01)
    
    return best_a, _score(best_a)

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
