"""
Refined optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
1. Exact peak equalization via linear algebra (minimum-norm solution)
2. Very large active sets for fine-grained optimization
3. Multi-path independent optimization
4. Adaptive active set management with hysteresis
5. Improved L-BFGS with better beta scheduling

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm1(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / self.beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * self.n * c_max
        return 2.0 * self.n * (c_max + self.beta * np.log(s))

    def grad(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / self.beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[self.n - 1:2 * self.n - 1]
        W = float(np.dot(w, c))
        return 4.0 * self.n * a * (h - W)

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.01) * beta_scale

    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625,
        0.03, 0.015, 0.0075, 0.004,
        0.002, 0.001, 0.0005, 0.0002, 0.0001,
        0.00005, 0.00002, 0.00001, 0.000005, 0.000001
    ])

    current_a = a_init.copy()
    current_sc = _score_norm1(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        if beta < base_beta * 0.01:
            est_time = 0.003 * max(1.0, n / 2048.0)
            max_iter = min(8000, max(100, int(remaining / (est_time * 10))))
        else:
            est_time = 0.002 * max(1.0, n / 2048.0)
            max_iter = min(5000, max(50, int(remaining / (est_time * 8))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score_norm1(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a

def _exact_equalize(a, active_idx, target_c):

    best_a = a.copy()

    no_improve = 0
    K_target = 10
    lr = 0.1

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

Key innovations over previous approach:

5. Multi-path optimization with independent perturbations

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[self.n - 1:2 * self.n - 1]
        W = float(np.dot(w, c))
        return 4.0 * self.n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.01) * beta_scale

    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625,
        0.03, 0.015, 0.0075, 0.004,
        0.002, 0.001, 0.0005, 0.0002, 0.0001,
        0.00005, 0.00002, 0.00001, 0.000005, 0.000001
    ])

    current_a = a_init.copy()
    current_sc = _score_norm1(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        if beta < base_beta * 0.01:
            est_time = 0.002 * max(1.0, n / 2048.0)
            max_iter = min(8000, max(100, int(remaining / (est_time * 8))))
        else:
            est_time = 0.002 * max(1.0, n / 2048.0)
            max_iter = min(5000, max(50, int(remaining / (est_time * 6))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score_norm1(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _exact_equalize_step(a, c, active_idx, target_c):
    """Compute minimum-norm delta that equalizes active peaks to target_c (first order).
    
    Returns delta array or None if solving fails.
    """
    n = len(a)
    K = len(active_idx)
    if K == 0 or K > 500:
        return None

    # Compute autocorrelation for building MMt
    ac = _autocorr_linear(a)  # ac[d] = sum_j a[j]*a[j+d], d=0..n-1

    # Build MMt: (K+1) x (K+1)
    # (MMt)[i,j] = 4 * autocorr[|m_i - m_j|] for i,j < K
    # (MMt)[i,K] = 2 for i < K (since sum a = 1)
    # (MMt)[K,K] = n
    MMt = np.zeros((K + 1, K + 1))
    for i in range(K):
        for j in range(K):
            d = abs(int(active_idx[i]) - int(active_idx[j]))
            MMt[i, j] = 4.0 * ac[d]
        MMt[i, K] = 2.0
    MMt[K, K] = float(n)

    # b vector
    b = np.zeros(K + 1)
    for i in range(K):
        b[i] = target_c - c[active_idx[i]]

    # Solve
    try:
        alpha = np.linalg.solve(MMt, b)
    except np.linalg.LinAlgError:
        try:
            alpha = np.linalg.lstsq(MMt, b, rcond=None)[0]
        except Exception:
            return None

    if not np.all(np.isfinite(alpha)):
        return None

    # Compute delta = M^T @ alpha
    # delta[j] = 2 * sum_i alpha[i] * a[m_i - j] + alpha[K]
    delta = np.full(n, alpha[K], dtype=np.float64)
    for i in range(K):
        ai = alpha[i]
        if abs(ai) < 1e-30:
            continue
        m = int(active_idx[i])
        lo = max(0, m - n + 1)
        hi = min(m, n - 1)
        if lo <= hi:
            delta[lo:hi + 1] += 2.0 * ai * a[m - hi:m - lo + 1][::-1]

    return delta


def _active_set_equalize(a_init, deadline, min_time_remaining=30.0, max_K=300):
    """Active-set equalization using exact linear algebra solution."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score_norm1(a_init)
    a = a / s

    best_a = a.copy()
    best_sc = _score_norm1(a)

    no_improve = 0
    K_target = 10
    lr = 0.1
    rng_local = np.random.default_rng(12345)

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            K_target = min(max_K, K_target + 2)
        else:
            no_improve += 1

        if no_improve > 800:
            # Perturb and reset
            a = best_a + 0.005 * rng_local.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0
            K_target = 5
            lr = 0.05
            continue

        # Find active set
        tol = c_max * 0.002
        active = np.where(c >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        if len(active) > K_target:
            top_vals = c[active]
            order = np.argsort(top_vals)[::-1][:K_target]
            active = active[order]

        K = len(active)

        # Target: slightly below the min of active set to push all down
        active_vals = c[active]
        target_c = float(np.min(active_vals)) * 0.999
        target_c = max(target_c, c_max * 0.5)  # don't go too aggressive

        # Try exact equalization
        delta = _exact_equalize_step(a, c, active, target_c)

        if delta is None:
            # Fallback: average gradient
            w_S = np.zeros(2 * n - 1)
            w_S[active] = 1.0 / K
            a_rev = a[::-1]
            h = _full_convolve(w_S, a_rev)[n - 1:2 * n - 1]
            grad = 2.0 * h
            grad = grad - np.mean(grad)
            delta = -grad

        # Find max step for non-negativity
        mask = delta < -1e-30
        if mask.any():
            max_step = float(np.min(a[mask] / (-delta[mask]))) * 0.95
        else:
            max_step = 1.0

        step = min(lr, max_step) if max_step > 0 else 0.0
        improved = False

        for _ in range(25):
            if step <= 1e-16:
                break
            a_try = a + step * delta
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                step *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            step *= 0.5

        if not improved:
            lr *= 0.92
            if K_target > 5:
                K_target = max(5, K_target - 1)
            if no_improve >= 500:
                a = best_a.copy()
                lr = 0.1
                K_target = 10
                no_improve = 0
        else:
            lr = min(0.4, lr * 1.03)

    return best_a, best_sc


def _interpolate_equalize(a_init, deadline, min_time_remaining=30.0, max_K=100):
    """Alternative: use interpolation-style equalization with parabolic line search."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm1(a)

    no_improve = 0
    K = 5
    rng_local = np.random.default_rng(99999)

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            K = min(max_K, K + 1)
        else:
            no_improve += 1

        if no_improve > 600:
            a = best_a + 0.003 * rng_local.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0
            K = 3
            continue

        # Active set: top K peaks
        active = np.argsort(c)[::-1][:K]
        active_vals = c[active]

        # Weight: emphasize the highest peaks
        w = 1.0 / (active_vals + 1e-30)
        w = w / w.sum()

        # Compute weighted gradient
        grad = np.zeros(n)
        for i, m in enumerate(active):
            wi = w[i]
            if wi < 1e-15:
                continue
            mm = int(m)
            lo = max(0, mm - n + 1)
            hi = min(mm, n - 1)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * wi * a[mm - hi:mm - lo + 1][::-1]

        # Project to tangent space of simplex
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Line search with parabolic interpolation
        alpha = 0.1 / (gnorm + 1e-15)
        best_alpha = 0
        best_sc_try = current_sc

        for _ in range(30):
            if alpha < 1e-16:
                break
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_alpha = alpha
            alpha *= 0.7

        if best_alpha > 0 and best_sc_try < current_sc - 1e-15:
            a_try = a - best_alpha * grad
            a_try = np.maximum(a_try, 0.0)
            a_try = a_try / a_try.sum()
            a = a_try
        else:
            no_improve += 5

    return best_a, best_sc


def _asymmetric_perturb(a, rng, strength=0.03):
    n = len(a)
    t = np.arange(n) / n
    ramp = 0.3 * (t - 0.5)
    noise = 0.7 * rng.standard_normal(n)
    perturbation = ramp + noise
    perturbation = perturbation - perturbation.mean()
    z = np.log(np.maximum(a, 1e-30))
    z = z + strength * perturbation
    z = z - z.mean()
    return _softmax(z)


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
