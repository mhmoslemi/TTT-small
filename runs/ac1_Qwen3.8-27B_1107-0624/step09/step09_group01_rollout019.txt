Optimized step function sequence generator using adaptive log-sum-exp smoothing,
KKT Newton with dynamic active sets, spectral shaping, and multi-n exploration.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp_smoothed_obj(z, beta, n):
    """Compute the log-sum-exp smoothed objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _hankel_matvec(v, W, n):
    """Compute H(v) where H is the Hessian of w·(a⊛a) for the active set W."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve_hankel(W, n, rhs, max_iter=300, tol=1e-12):
    """Conjugate gradient solver for the Hankel system."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)

            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.05:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:

    return x

        c_new = _full_convolve(current_a, current_a)

        c_sorted = np.sort(c_new)[::-1]
        gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]

        if gap < 1e-8 * c_max_new:
            beta *= 0.05
        elif gap < 1e-5 * c_max_new:
            beta *= 0.1
        elif gap < 1e-3 * c_max_new:
            beta *= 0.2
        else:
            beta *= 0.4

        beta = min(beta, c_max_new * 0.0001)
        if beta < c_max_new * 1e-12:
            break

    return current_a, current_sc


def _spectral_flatten(a, n_iters=100, deadline=None):
    """Flatten the spectrum of a to reduce peak autocorrelation.
    Uses iterative: FFT -> flatten magnitude -> IFFT -> project to non-negative."""
    n = len(a)
    a_work = a.copy()
    s = a_work.sum()
    if s < 1e-15:
        return a_work
    a_work = a_work / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        # Compute FFT
        N = _next_pow2(2 * n - 1)
        A = np.fft.rfft(a_work, N)
        # Flatten: reduce the peak of |A|² by spreading energy
        mag = np.abs(A)
        mag_sq = mag ** 2
        # Target: make mag_sq more uniform
        target = np.ones_like(mag_sq) * mag_sq.sum() / len(mag_sq)
        # Blend
        blend = 0.1
        mag_new = np.sqrt(mag_sq * (1 - blend) + target * blend)
        # Preserve phase
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a_work = a_new

    return a_work


def _subgrad_momentum_v2(a_init, deadline, max_steps=5000):
    """Improved subgradient method with momentum and adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)

    lr = 0.03
    momentum = 0.9
    v = np.zeros(n)

            active = c >= (c_max - tol)

        if active.sum() == 0:

        W = np.zeros_like(c)

        # Gradient
        a_rev = a[::-1]
        wc = _full_convolve(W, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(W, c))
        grad = 2.0 * a * (h - W_val)
        grad = grad - np.mean(grad)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-16:
            break

        # Adaptive learning rate
        lr = lr * (0.999 ** patience)
        v = momentum * v - lr * grad
        z = z + v
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 30 == 29:

                best_sc = sc

                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered2":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.where(t < 0.2, t / 0.2, np.where(t > 0.8, (1 - t) / 0.2, 1.0))
        a = np.maximum(a, 0)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

"""

n exploration, and diverse restarts. Target: score < 1.5027.
"""
import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

        base_K = k_schedule[min(rnd, len(k_schedule) - 1)]
        c_sorted = np.sort(c)[::-1]
        gap_threshold = c_max - 0.005 * c_max
        n_near = int(np.sum(c >= gap_threshold))
        K = max(min(base_K, 2 * n - 1), min(n_near, 2 * n - 1))
        K = min(K, 2 * n - 1)

        idx = np.argpartition(c, -K)[-K:]
        W = np.zeros(2 * n - 1)
        W[idx] = 1.0 / K

        g = _kkt_gradient(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        Hg = _cg_solve_hankel(W, n, g, max_iter=200)
        H1 = _cg_solve_hankel(W, n, np.ones(n), max_iter=200)
        H1_denom = float(np.ones(n) @ H1)
        if abs(H1_denom) < 1e-25 or not np.isfinite(H1_denom):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / H1_denom
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        alpha = 1.0
        step_done = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_max_try = float(_full_convolve(a_try, a_try).max())
            if c_max_try < c_max * (1.0 - 0.005 * alpha) - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not step_done:
            for trial in [0.1, 0.05, 0.02, 0.01, 0.005, 0.001, 0.0005]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                if float(_full_convolve(a_try, a_try).max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 10:
            break

    return best_a, best_sc


def _smoothed_lbfgs(a_init, deadline, beta_init=0.002, min_remaining=60.0):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    c_max_cur = float(c_cur.max())
    beta = min(beta_init * c_max_cur, 0.005)

    while beta > c_max_cur * 1e-13:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 16384.0)
        est_time_per_iter = 0.002 * scale
        max_iter = min(4000, max(20, int(remaining / (est_time_per_iter * 60))))

        def obj_fn(z_val):
            return _lse_obj(z_val, beta, n)

        def grad_fn(z_val):
            return _lse_grad(z_val, beta, n)

        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 400}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted = np.sort(c_new)[::-1]
        gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]

        if gap < 1e-9 * c_max_new:
            beta *= 0.03
        elif gap < 1e-6 * c_max_new:
            beta *= 0.08
        elif gap < 1e-4 * c_max_new:
            beta *= 0.15
        elif gap < 1e-2 * c_max_new:
            beta *= 0.3
        else:
            beta *= 0.5

        beta = min(beta, c_max_new * 0.00005)
        if beta < c_max_new * 1e-13:
            break

    return current_a, current_sc


def _spectral_flatten(a, n_iters=80, deadline=None):
    n = len(a)
    a_work = a.copy()
    s = a_work.sum()
    if s < 1e-15:
        return a_work
    a_work = a_work / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        N = _next_pow2(2 * n - 1)
        A = np.fft.rfft(a_work, N)
        mag_sq = np.abs(A) ** 2
        target = np.ones_like(mag_sq) * mag_sq.sum() / len(mag_sq)
        blend = 0.05 + 0.05 * (i / max(n_iters, 1))
        mag_new = np.sqrt(mag_sq * (1 - blend) + target * blend)
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a_work = a_new

    return a_work


def _subgrad_momentum(a_init, deadline, max_steps=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.025
    momentum = 0.9
    v = np.zeros(n)

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        for tol_frac in [0.00005, 0.0002, 0.0005, 0.001, 0.003, 0.01, 0.05]:
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            if active.sum() >= 5:
                break
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        W = np.zeros_like(c)
        W[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(W, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(W, c))
        grad = 2.0 * a * (h - W_val)
        grad = grad - np.mean(grad)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-16:
            break

        lr = lr * (0.998 ** patience)
        v = momentum * v - lr * grad
        z = z + v
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 30 == 29:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.08, 0.06)
            else:
                patience += 1
                momentum = min(momentum + 0.005, 0.96)
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)
                    patience = 0
                    lr = 0.015

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered2":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.9
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.02
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.where(t < 0.15, t / 0.15, np.where(t > 0.85, (1 - t) / 0.15, 1.0))
        a = np.maximum(a, 0)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "fejer_like":
        t = np.linspace(0, 1, n)
        a = (np.sin(np.pi * t) / (np.pi * t + 1e-10)) ** 2
        a = np.maximum(a, 0)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _two_scale_optimize(a_init, n_small, deadline, min_remaining=100.0):
    """Optimize at smaller n, then upsample and continue at original n."""
    n_orig = len(a_init)
    a_small = _resample(a_init, n_small)
    sc_small = _score(a_small)
    print(f"    Two-scale: n={n_small}, initial score={sc_small:.10f}")

    t_mid = min(deadline - min_remaining - 10, time.monotonic() + (deadline - min_remaining - time.monotonic()) * 0.4)
    a_opt_small, sc_opt_small = _smoothed_lbfgs(a_small, t_mid, beta_init=0.003, min_remaining=10.0)
    print(f"    Two-scale small n result: score={_score(a_opt_small):.10f}")

    a_up = _resample(a_opt_small, n_orig)
    sc_up = _score(a_up)
    print(f"    Two-scale upscaled score: {sc_up:.10f}")

    t_end = min(deadline - min_remaining, time.monotonic() + (deadline - min_remaining - time.monotonic()) * 0.6)
    a_final, sc_final = _smoothed_lbfgs(a_up, t_end, beta_init=0.001, min_remaining=min_remaining)
    return a_final, sc_final


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"imp={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                par_sc = _score(parent)
                print(f"Parent: n={len(parent)}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        parent = _make_profile(32768, "tapered")
        print(f"No parent, using tapered n=32768")
        update_best(parent, _score(parent), "init")

    n0 = len(parent)

    # Phase 1: Quick KKT-Newton to establish good starting point
    print(f"\n[Phase 1] KKT-Newton quick (t={time.monotonic() - t_start:.1f}s)")
    t_p1_end = min(deadline - 800, t_start + 150)
    k_sch = [20, 50, 100, 200, 500, 1000, 2000, 5000]
    a_kkt, sc_kkt = _newton_kkt(parent, t_p1_end, max_rounds=200, k_schedule=k_sch)
    update_best(a_kkt, sc_kkt, "kkt_1")

    # Phase 2: Main L-BFGS optimization (long, adaptive)
    print(f"\n[Phase 2] L-BFGS main (t={time.monotonic() - t_start:.1f}s)")
    t_p2_end = min(deadline - 500, t_start + 700)
    a_lb, sc_lb = _smoothed_lbfgs(best_seq, t_p2_end, beta_init=0.003, min_remaining=450.0)
    update_best(a_lb, sc_lb, "lbfgs_1")

    # Phase 3: Spectral flattening kick
    print(f"\n[Phase 3] Spectral flatten (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a_spec = _spectral_flatten(best_seq, n_iters=60,
                                   deadline=min(deadline - 190, time.monotonic() + 40))
        update_best(a_spec, _score(a_spec), "spectral_1")
        # Follow up with Newton
        t_p3b = min(deadline - 180, time.monotonic() + 30)
        a_s3, sc_s3 = _newton_kkt(best_seq, t_p3b, max_rounds=80,
                                   k_schedule=[10, 50, 200, 1000, 5000])
        update_best(a_s3, sc_s3, "post_spec_kkt")

    # Phase 4: Two-scale optimization
    print(f"\n[Phase 4] Two-scale (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        n_small = max(4096, n0 // 4)
        t_two_end = min(deadline - 250, t_start + 1100)
        a_ts, sc_ts = _two_scale_optimize(best_seq, n_small, t_two_end, min_remaining=200.0)
        update_best(a_ts, sc_ts, "two_scale")

    # Phase 5: Cycling subgradient + Newton + L-BFGS
    print(f"\n[Phase 5] Cycle (t={time.monotonic() - t_start:.1f}s)")
    cycle_end = min(deadline - 200, t_start + 1500)
    for cyc in range(12):
        if time.monotonic() > cycle_end:
            break
        # Subgradient
        t_sub = min(cycle_end, time.monotonic() + 25)
        a_c1, sc_c1 = _subgrad_momentum(best_seq, t_sub, max_steps=3000)
        update_best(a_c1, sc_c1, f"cyc_sg_{cyc}")

        # Newton with varying K
        t_new = min(cycle_end, time.monotonic() + 35)
        if cyc % 2 == 0:
            k_sch2 = [5, 10, 20, 50, 100, 300, 1000]
        else:
            k_sch2 = [50, 100, 500, 1000, 5000]
        a_c2, sc_c2 = _newton_kkt(best_seq, t_new, max_rounds=80, k_schedule=k_sch2)
        update_best(a_c2, sc_c2, f"cyc_np_{cyc}")

        # Periodic L-BFGS burst
        if cyc % 3 == 2 and time.monotonic() < cycle_end - 30:
            t_burst = min(cycle_end - 25, time.monotonic() + 20)
            a_b, sc_b = _smoothed_lbfgs(best_seq, t_burst, beta_init=0.001, min_remaining=25.0)
            update_best(a_b, sc_b, f"cyc_lbfgs_{cyc}")

        # Periodic spectral
        if cyc % 4 == 3 and time.monotonic() < cycle_end - 15:
            a_sf = _spectral_flatten(best_seq, n_iters=40,
                                     deadline=min(cycle_end - 10, time.monotonic() + 15))
            update_best(a_sf, _score(a_sf), f"cyc_spec_{cyc}")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_options = sorted(set([
        n0, n0 * 2, 16384, 32768, 49152, 40960, 36864
    ]))
    n_options = [n for n in n_options if 4096 <= n <= 49152]

    for n_try in n_options:
        if time.monotonic() > deadline - 80:
            break
        a_res = _resample(best_seq, n_try)
        sc_res = _score(a_res)
        if sc_res < best_score:
            update_best(a_res, sc_res, f"resample_n{n_try}")

        if time.monotonic() < deadline - 40 and n_try <= 32768:
            t_end = min(deadline - 30, time.monotonic() + 15)
            a_opt, sc_opt = _newton_kkt(a_res, t_end, max_rounds=50,
                                         k_schedule=[10, 100, 500, 2000])
            update_best(a_opt, sc_opt, f"opt_n{n_try}")

    # Phase 7: Diverse restarts with different profiles
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    profiles = ["tapered2", "fejer_like", "bell", "flat_top", "cos", "raised_cos"]
    for i, pk in enumerate(profiles):
        if time.monotonic() > deadline - 15:
            break
        a_init = _make_profile(n0, pk)
        a_pert = _perturb(a_init, 0.005, rng)
        t_end = min(deadline - 10, time.monotonic() + 8)
        a_r, sc_r = _newton_kkt(a_pert, t_end, max_rounds=40,
                                 k_schedule=[10, 100, 1000])
        update_best(a_r, sc_r, f"restart_{pk}")

    # Phase 8: Final ultra-fine polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 2:
        k_final = [1, 2, 3, 5, 10, 20, 50, 100, 200, 500, 1000]
        a_fin, sc_fin = _newton_kkt(best_seq, deadline - 1, max_rounds=500, k_schedule=k_final)
        update_best(a_fin, sc_fin, "final_kkt")

    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(32768, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
