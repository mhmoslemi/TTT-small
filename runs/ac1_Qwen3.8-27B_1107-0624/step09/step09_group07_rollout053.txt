"""
Adam-based optimizer with LSE smoothing, geometric temperature annealing,
and periodic L-BFGS refinement for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations: Adam in log-space as primary optimizer, full LSE smoothing,
geometric temperature schedule, symmetry-aware initialization, and peak-chasing polish.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _conv_gradient(w, a):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a). w has length 2n-1."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _lse_obj_grad_z(z, beta):
    """Full LSE smoothed objective and gradient in log-space (softmax param)."""
    n = z.size
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    a = e / e.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = c / beta
    shifted = np.clip(shifted - shifted.max(), -700, 0)
    exp_shifted = np.exp(shifted)
    s = exp_shifted.sum()
    if s < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        w = exp_shifted / s
        obj = 2.0 * n * (c_max + beta * np.log(s) + beta * (shifted.max() - c.max() / beta))
        # Simpler: obj = 2.0 * n * beta * (shifted.max() + np.log(s))
        obj = 2.0 * n * beta * (c.max() / beta + np.log(s))
    g_a = _conv_gradient(w, a)
    W = float(np.dot(a, g_a))
    g_z = a * (g_a - W)
    return obj, g_z


def _topk_obj_grad_z(z, beta, K):
    """Top-K LSE smoothed objective and gradient in log-space."""
    n = z.size
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    a = e / e.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c
    shifted = c_top / beta
    shifted = np.clip(shifted - shifted.max(), -700, 0)
    exp_s = np.exp(shifted)
    s = exp_s.sum()
    if s < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = exp_s
        w = e_full / s
        obj = 2.0 * n * beta * (c.max() / beta + np.log(s))
    g_a = _conv_gradient(w, a)
    W = float(np.dot(a, g_a))
    g_z = a * (g_a - W)
    return obj, g_z


def _run_adam(z_init, beta_start, beta_end, n_iters, n, deadline, min_remaining=120.0,
              lr_base=0.005, print_interval=2000):
    """Adam optimizer in log-space with geometric temperature annealing."""
    z = z_init.copy().astype(np.float64)
    z = z - z.mean()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_z = z.copy()
    best_score = np.inf

    a_init = np.exp(z - z.max())
    a_init = a_init / a_init.sum()
    best_score = 2.0 * n * float(_full_convolve(a_init, a_init).max())
    best_z = z.copy()
    best_a = a_init.copy()

    decay = (beta_end / beta_start) ** (1.0 / max(1, n_iters))
    warmup_iters = min(200, n_iters // 10)

    t_start = time.monotonic()
    for t in range(1, n_iters + 1):
        if time.monotonic() + min_remaining > deadline:
            break
        if t % 500 == 0 and (time.monotonic() - t_start) > 120:
            break

        beta = beta_start * (decay ** t)

        # Learning rate schedule: warmup then cosine decay
        if t <= warmup_iters:
            lr = lr_base * (t / warmup_iters)
        else:
            progress = (t - warmup_iters) / max(1, n_iters - warmup_iters)
            lr = lr_base * 0.5 * (1.0 + np.cos(np.pi * progress))

        # Compute objective and gradient
        obj, g_z = _lse_obj_grad_z(z, beta)

        # Update moments
        m = beta1 * m + (1.0 - beta1) * g_z
        v = beta2 * v + (1.0 - beta2) * g_z * g_z
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Adam update
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - z.max()  # numerical stability

        # Periodic score check
        if t % print_interval == 0 or t == n_iters:
            a = np.exp(z - z.max())
            a = a / a.sum()
            sc = 2.0 * n * float(_full_convolve(a, a).max())
            if sc < best_score:
                best_score = sc
                best_z = z.copy()
                best_a = a.copy()
            if t % (print_interval * 5) == 0:
                print(f"    Adam iter {t}/{n_iters}, beta={beta:.6f}, score={sc:.10f}, best={best_score:.10f}")

    return best_a, best_score, best_z


def _run_lbfgs_refine(a_init, deadline, min_remaining=60.0, beta_scale=0.01, K_frac=0.02):
    """L-BFGS refinement with top-K smoothing."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.0005:
        gap = max_c * 0.0005

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    stages = [
        (beta_scale, K_frac),
        (beta_scale * 0.5, K_frac * 0.7),
        (beta_scale * 0.2, K_frac * 0.4),
        (beta_scale * 0.1, K_frac * 0.2),
        (beta_scale * 0.05, K_frac * 0.1),
    ]

    for b_scale, k_frac in stages:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 15:
            break

        beta = max_c * b_scale
        K = max(3, int((2 * n - 1) * k_frac))

        max_iter = min(4000, max(30, int(remaining / 0.02)))

        def obj_func(z):
            return _topk_obj_grad_z(z, beta, K)[0]

        def grad_func(z):
            return _topk_obj_grad_z(z, beta, K)[1]

        try:
            result = minimize(
                obj_func, z, jac=grad_func, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 50}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _peak_chase(a_init, deadline, min_remaining=30.0):
    """Directly chase the top peak with projected gradient steps."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.01
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        # Use peaks within 1% of max
        threshold = c_max * 0.99
        active = c >= threshold
        n_active = max(1, int(active.sum()))
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active
        # Weight more heavily the top peak
        peak_idx = int(np.argmax(c))
        w[peak_idx] += 0.5
        w = w / w.sum()

        g = _conv_gradient(w, a)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break

        # Backtracking line search
        alpha = lr / (gnorm + 1e-15)
        improved = False
        for ls in range(30):
            a_try = np.maximum(a - alpha * g_centered, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2. * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if improved:
            lr = min(0.05, lr * 1.02)
        else:
            lr *= 0.6
            if lr < 1e-6:
                break

    return best_a, best_sc


def _make_symmetric_profile(n, kind="raised_cos"):
    """Create a symmetric initial profile."""
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sin_power":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        # Slightly taper the edges
        taper = int(n * 0.1)
        for i in range(taper):
            f = 1.0 - (taper - i) / taper
            a[i] = 0.3 + 0.7 * f
            a[n - 1 - i] = 0.3 + 0.7 * f
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "optimal_shape":
        # Based on analysis: optimal shape has slight edge tapering
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.3 * (2 * t - 1) ** 4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_log(a, sigma, rng):
    """Perturb in log-space."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

    Multi-phase optimizer using Adam in log-space with LSE smoothing,
    geometric temperature annealing, periodic L-BFGS refinement,
    and peak-chasing polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_symmetric_profile(n_parent, "sin_power")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, sin_power n={n_parent}, score={best_score:.10f}")

    n = len(best_seq)

    # === Phase 1: Adam with geometric annealing (long phase) ===
    print(f"\n[P1] Adam LSE annealing (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        z_init = np.log(np.maximum(best_seq, 1e-30))
        z_init = z_init - z_init.max()

        c_cur = _full_convolve(best_seq, best_seq)
        max_c = float(c_cur.max())
        beta_start = max_c * 0.1
        beta_end = max_c * 0.001

        adam_deadline = min(deadline - 350, time.monotonic() + 500)
        a1, sc1, z1 = _run_adam(z_init, beta_start, beta_end, 15000, n,
                                 adam_deadline, min_remaining=300, lr_base=0.008)
        if sc1 < best_score:
            best_seq = a1.copy()
            best_score = sc1
            print(f"  NEW BEST: {best_score:.10f} [adam1] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement from Adam (best={best_score:.10f})")

    # === Phase 2: L-BFGS refinement ===
    print(f"\n[P2] L-BFGS refine (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a2, sc2 = _run_lbfgs_refine(best_seq, min(deadline - 250, time.monotonic() + 300),
                                     min_remaining=100, beta_scale=0.005, K_frac=0.015)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [lbfgs1] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Second Adam round from improved point ===
    print(f"\n[P3] Adam round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        z_init2 = np.log(np.maximum(best_seq, 1e-30))
        z_init2 = z_init2 - z_init2.max()
        c_cur = _full_convolve(best_seq, best_seq)
        max_c = float(c_cur.max())
        beta_start2 = max_c * 0.01
        beta_end2 = max_c * 0.0001

        adam_deadline2 = min(deadline - 200, time.monotonic() + 250)
        a3, sc3, z3 = _run_adam(z_init2, beta_start2, beta_end2, 15000, n,
                                 adam_deadline2, min_remaining=150, lr_base=0.003)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [adam2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Peak chasing ===
    print(f"\n[P4] Peak chase (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        a4, sc4 = _peak_chase(best_seq, min(deadline - 120, time.monotonic() + 150),
                               min_remaining=20)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [peakchase] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: L-BFGS fine polish ===
    print(f"\n[P5] L-BFGS fine polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        a5, sc5 = _run_lbfgs_refine(best_seq, min(deadline - 50, time.monotonic() + 80),
                                     min_remaining=10, beta_scale=0.001, K_frac=0.005)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [lbfgs2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Restart exploration ===
    print(f"\n[P6] Restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        sigmas = [0.02, 0.04, 0.06, 0.03, 0.05, 0.08]
        for i, sigma in enumerate(sigmas):
            if time.monotonic() > deadline - 40:
                break
            a_pert = _perturb_log(best_seq, sigma, rng)
            # Quick Adam run
            z_p = np.log(np.maximum(a_pert, 1e-30))
            z_p = z_p - z_p.max()
            c_p = _full_convolve(a_pert, a_pert)
            mc = float(c_p.max())
            adam_dl = min(deadline - 30, time.monotonic() + 25)
            a_r, sc_r, _ = _run_adam(z_p, mc * 0.005, mc * 0.0001, 3000, n,
                                      adam_dl, min_remaining=5, lr_base=0.003)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.10f} [restart_{i}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final peak chase ===
    print(f"\n[P7] Final peak chase (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a7, sc7 = _peak_chase(best_seq, deadline - 2, min_remaining=2)
        if sc7 < best_score:
            best_seq = a7.copy()
            best_score = sc7
            print(f"  NEW BEST: {best_score:.10f} [final] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_symmetric_profile(256, "sin_power")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_symmetric_profile(256, "sin_power")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
