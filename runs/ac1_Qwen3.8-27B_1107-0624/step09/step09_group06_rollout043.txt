def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _conv(a):
    return _full_convolve(a, a)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _grad_smoothed_max(a, beta, active_idx=None):

    a = _softmax(z)
    c = _conv(a)
    K_actual = min(K, len(c))
    top_k = np.argpartition(c, -K_actual)[-K_actual:]

    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    momentum = 0.8
    base_lr = 0.003
    t = 0
    no_improve = 0
    
    while time.monotonic() + min_remaining < deadline:
        t += 1
        c = _conv(a)
        c_max = float(c.max())
        sc_cur = 2.0 * n * c_max
        
        if sc_cur < best_sc - 1e-14:
            best_sc = sc_cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
        
        # Active set: top K peaks
        K_actual = min(K, len(c))
        active_idx = np.argpartition(c, -K_actual)[-K_actual:]
        
        # Corrected gradient (average over active peaks)
        grad = np.zeros(n)
        for k in active_idx:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                grad[j_lo:j_hi + 1] += 2.0 * a[k - np.arange(j_hi - j_lo + 1)][::-1]
        grad /= K_actual
        
        # Project to tangent space
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break
        
        lr = base_lr / (gnorm + 1e-18) * 0.05
        v = momentum * v - lr * grad_t
        a_cand = a + v
        a_proj = _project_simplex(a_cand)
        
        sc_new = _score(a_proj)
        if sc_new < sc_cur:
            a = a_proj
        else:
            a = a_proj  # still stay on simplex
            
            if no_improve % 100 == 0:
                perturb = 0.002 * rng.standard_normal(n)
                a_pert = _project_simplex(a + perturb)
                sc_pert = _score(a_pert)
                if sc_pert < best_sc:
                    a = a_pert
                    best_a = a.copy()
                    best_sc = sc_pert
                v = np.zeros(n)
                no_improve = 0
    
    return best_a, best_sc


def _perturb_and_reoptimize(a_init, deadline, min_remaining=60.0, seed=0):
    """Apply random perturbations and re-optimize."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    n_pert = 5
    pert_time = (deadline - min_remaining - time.monotonic()) / n_pert
    pert_time = max(10.0, min(120.0, pert_time))
    
    for p in range(n_pert):
        if time.monotonic() + min_remaining > deadline:
            break
        
        # Random perturbation: perturb a random subset of coordinates
        n_perturb = max(10, n // 100)
        idx = rng.choice(n, size=n_perturb, replace=False)
        a_pert = a_init.copy()
        scale = 0.01 + 0.02 * p / max(1, n_pert - 1)
        a_pert[idx] += scale * rng.standard_normal(n_perturb)
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert /= s
        
        # Quick L-BFGS from perturbed point
        sub_deadline = time.monotonic() + pert_time
        a_opt, sc_opt = _run_active_lbfgs(a_pert, sub_deadline,
                                           min_remaining=10.0,
                                           K_start=15, K_end=60,
                                           beta_scale=0.005,
                                           max_betas=4, seed=seed + p)
        if sc_opt < best_sc:
            best_a = a_opt
            best_sc = sc_opt
            a_init = a_opt  # Continue from improved point
            print(f"    [Perturb {p+1}] New best: {best_sc:.10f}")
    
    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _direct_coord_optimize(a_init, deadline, min_remaining=30.0):
    """Direct coordinate optimization: for each coordinate, find optimal value."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    rng = np.random.default_rng(42)
    n_sweeps = 3
    
    for sweep in range(n_sweeps):
        if time.monotonic() + min_remaining > deadline:
            break
        
        # Compute current convolution
        c = _conv(a)
        c_max = float(c.max())
        S = a.sum()
        
        # Process coordinates in random order
        order = rng.permutation(n)
        n_updated = 0
        
        for j in order:
            if time.monotonic() + min_remaining > deadline:
                break
            
            # c[k] as function of a[j]:
            # c[k] = 2*a[j]*a[k-j] + c_rest[k] for k != 2j
            # c[2j] = a[j]^2 + c_rest[2j]
            # We want to find a[j] >= 0 that minimizes max_k c_new[k]
            # where c_new[k] = c[k] + 2*(a_new - a[j])*a[k-j] (for k where both valid)
            #                   c[k] + (a_new^2 - a[j]^2) (for k=2j)
            
            # For simplicity, use a small step: try a few values around current
            # and pick the best.
            old_val = a[j]
            candidates = [old_val]
            # Try reducing (since we want to reduce peaks)
            if old_val > 1e-10:
                candidates += [old_val * 0.9, old_val * 0.8, old_val * 0.7]
            # Try increasing slightly
            candidates += [old_val * 1.1, old_val * 1.2]
            candidates += [0.0]
            
            best_j_val = old_val
            best_j_sc = 2.0 * n * c_max / (S * S)
            
            for new_val in candidates:
                if new_val < 0:
                    continue
                a_try = a.copy()
                a_try[j] = new_val
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                c_try = _conv(a_try)
                sc_try = 2.0 * n * float(c_try.max()) / (s_try * s_try)
                if sc_try < best_j_sc:
                    best_j_sc = sc_try
                    best_j_val = new_val
            
            if best_j_val != old_val:
                a[j] = best_j_val
                n_updated += 1
        
        # Renormalize
        s = a.sum()
        if s > 1e-15:
            a = a / s
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Main search function combining multiple optimization strategies."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        # Good initialization: raised cosine
        n_init = n_parent
        t = np.linspace(0, 1, n_init)
        parent = 0.5 * (1 + np.cos(2 * np.pi * t))
        parent = np.maximum(parent, 1e-10)
        parent = parent / parent.sum()
        parent_sc = _score(parent)
        print(f"Initialize: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    
    # === Phase 1: Active-set L-BFGS (main workhorse) ===
    print(f"\n[P1] Active-set L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 900, t_start + 400)
    if time.monotonic() < p1_deadline:
        a1, sc1 = _run_active_lbfgs(best_seq, p1_deadline,
                                     min_remaining=60.0,
                                     K_start=20, K_end=80,
                                     beta_scale=0.01, max_betas=6,
                                     seed=seed)
        if sc1 < best_score:
            best_seq = a1.copy()
            best_score = sc1
            print(f"  [P1] New best: {best_score:.10f}")
    
    # === Phase 2: Nesterov with corrected gradient ===
    print(f"\n[P2] Nesterov corrected (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 700, t_start + 500)
    if time.monotonic() < p2_deadline:
        a2, sc2 = _nesterov_corrected(best_seq, p2_deadline,
                                       min_remaining=60.0, K=30, seed=seed+1)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  [P2] New best: {best_score:.10f}")
    
    # === Phase 3: Second L-BFGS pass (finer) ===
    print(f"\n[P3] Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 500, t_start + 700)
    if time.monotonic() < p3_deadline:
        a3, sc3 = _run_active_lbfgs(best_seq, p3_deadline,
                                     min_remaining=60.0,
                                     K_start=30, K_end=150,
                                     beta_scale=0.005, max_betas=6,
                                     seed=seed+2)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  [P3] New best: {best_score:.10f}")
    
    # === Phase 4: Perturbation and re-optimization ===
    print(f"\n[P4] Perturbations (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 300, t_start + 1100)
    if time.monotonic() < p4_deadline:
        a4, sc4 = _perturb_and_reoptimize(best_seq, p4_deadline,
                                           min_remaining=60.0, seed=seed+10)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  [P4] New best: {best_score:.10f}")
    
    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    p5_deadline = min(deadline - 150, t_start + 1400)
    n_candidates = []
    # Try sizes near the parent
    base_n = n_parent if n_parent > 0 else 16384
    for delta in [1, -1, 2, -2, 3, -3, 4, -4]:
        n_try = base_n + delta * 256
        if 4096 <= n_try <= 32768 and n_try not in n_candidates:
            n_candidates.append(n_try)
    
    for n_try in n_candidates:
        if time.monotonic() > p5_deadline - 120:
            break
        if n_try == len(best_seq):
            continue
        a_init_n = _resample(best_seq, n_try)
        sc_init = _score(a_init_n)
        print(f"  n={n_try}: init score={sc_init:.10f}")
        
        sub_dl = min(p5_deadline, time.monotonic() + 150)
        a_n, sc_n = _run_active_lbfgs(a_init_n, sub_dl,
                                       min_remaining=30.0,
                                       K_start=20, K_end=80,
                                       beta_scale=0.008, max_betas=4,
                                       seed=seed + n_try)
        if sc_n < best_score:
            best_seq = a_n.copy()
            best_score = sc_n
            print(f"  [P5] New best (n={n_try}): {best_score:.10f}")
    
    # === Phase 6: Final L-BFGS polish ===
    print(f"\n[P6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 30 - time.monotonic()
    if remaining > 60:
        n = len(best_seq)
        c_best = _conv(best_seq)
        mb = float(c_best.max())
        
        # Very fine beta
        c_sorted = np.sort(c_best)[::-1]
        gap = mb - c_sorted[min(20, len(c_sorted) - 1)]
        if gap < 1e-15:
            gap = mb * 0.001
        
        beta_final = gap * 0.5
        beta_final = max(beta_final, mb * 0.0001)
        
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        
        prob = _ActiveSetLbfgs(beta_final, n, min(50, 2 * n))
        max_iter = min(2000, max(50, int(remaining / 0.05)))
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxcor': 60, 'maxls': 50})
            a_fin = _softmax(result.x)
            sc_fin = _score(a_fin)
            if sc_fin < best_score:
                best_seq = a_fin.copy()
                best_score = sc_fin
                print(f"  [P6] New best: {best_score:.10f}")
        except Exception as e:
            print(f"  [P6] Error: {e}")
    
    # === Phase 7: Ultra-final with even smaller beta ===
    remaining = deadline - 10 - time.monotonic()
    if remaining > 20:
        n = len(best_seq)
        c_best = _conv(best_seq)
        mb = float(c_best.max())
        
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        
        beta_ult = mb * 0.00005
        prob = _ActiveSetLbfgs(beta_ult, n, min(30, 2 * n))
        max_iter = min(1000, max(20, int(remaining / 0.05)))
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxcor': 40})
            a_ult = _softmax(result.x)
            sc_ult = _score(a_ult)
            if sc_ult < best_score:
                best_seq = a_ult.copy()
                best_score = sc_ult
                print(f"  [P7] New best: {best_score:.10f}")
        except Exception:
            pass
    
    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
