def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _cross_corr(a, w, n):
    """
    Compute (a ⋆ w)_i = sum_j a[j] * w[i+j] for i=0..n-1,
    where w is zero-padded beyond its length.
    """
    flip_a = a[::-1].copy()
    out_len = n + len(w) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fw = np.fft.rfft(w, N)
    result = np.fft.irfft(fa * fw, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _cg_solve(matvec_fn, w_ref, rhs, max_iter, tol):
    """Conjugate gradient solver using a pre-allocated workspace."""
    n = len(rhs)
    x = np.zeros(n)
    r = rhs - matvec_fn(np.zeros_like(rhs))  # matvec(0) should be 0 for our use
    # Actually, for (H + mu*I)v, matvec(0) = 0, so r = rhs
    r = rhs.copy()
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_iter):
        Ap = matvec_fn(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x += alpha * p
        r -= alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

    if s < 1e-15:
        return a, np.inf

        c_max = float(c.max())

        if outer < 20:
            active_frac = 0.95
        elif outer < 50:
            active_frac = 0.99
        elif outer < 100:
            active_frac = 0.999
        else:
            active_frac = 0.9999

                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try /= s_try
            c_try = _full_convolve(a_try, a_try)

            if f_try < c_max * (1 - 1e-13) - t * 1e-15:
                a = a_try
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                improved = True
                break
            t *= 0.5
        
        if not improved:
            # Try gradient descent as fallback
            g = 2.0 * _cross_corr(a, np.ones(2*n-1)/(2*n-1), n)
            g_t = g - np.mean(g)

                a_try = np.maximum(a_try, 0)

            if not improved:

        if outer > 20:
            c_check = _full_convolve(a, a)
            if abs(float(c_check.max()) - c_max) < 1e-14:
                break
