def _next_pow2(n):
    k = 1

def _full_convolve(a, b):

    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)

        a = _softmax(z)
        c = _full_convolve(a, a)
        return 2.0 * self.n * _power_mean_value(c, self.p)

    def grad(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        w = _power_mean_weights(c, self.p)
        grad_a = _grad_structured(a, w)
        W = float(np.dot(grad_a, a))
        return a * (grad_a - W)


def _run_lbfgs(z, prob, max_iter, ftol=1e-20):
    result = minimize(
        prob, z, jac=prob.grad, method='L-BFGS-B',
        options={
            'maxiter': max_iter,
            'ftol': ftol,
            'gtol': 1e-16,
            'maxcor': 200,
            'maxls': 100,
        }
    )
    return result


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "one_sided":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t) + 0.005
        return a / a.sum()
    elif kind == "gaussian_asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-5.0 * (t - 0.35) ** 2) + 0.003
        return a / a.sum()
    elif kind == "skewed":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t * (1 - t) * 4) * (0.5 + 0.5 * np.cos(np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n, dtype=np.float64)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "cubic":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample_to_n(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _z_from_a(a):
    a = np.maximum(a, 1e-30)
    z = np.log(a)
    return z - z.mean()


def _subgradient_polish(a, deadline, min_time=10.0):
    """Subgradient method with Polyak step and active set tracking."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    lb = 2.0 * n / (2 * n - 1)

    step_scale = 0.3
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() + min_time < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                step_scale *= 0.7
                if step_scale < 0.01:
                    break
                no_improve = 0

        # Active set: top lags within tolerance
        tol = max(0.001 * c_max, 1e-12)
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
            n_active = 1

        # Average subgradient
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        grad = _grad_structured(a, w)

        # Project gradient: remove mean component (since we're on simplex)
        grad_c = grad - np.mean(grad)

        # Polyak step
        gap = current_sc - lb
        if gap < 1e-10:
            break
        g_norm_sq = float(np.dot(grad_c, grad_c))
        if g_norm_sq < 1e-20:
            break
        step = step_scale * gap / g_norm_sq
        step = min(step, 0.05)

        a_new = a - step * grad_c
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Adaptive multi-phase optimizer:
    P1: Coarse L-BFGS (find good basin)
    P2: Adaptive beta refinement (main optimization, 55% of budget)
    P3: Perturbation-based local search (escape local minima)
    P4: Subgradient polish
    P5: Multi-n exploration (brief)
    P6: Final L-BFGS polish
    """
    t0 = time.monotonic()
    deadline = t0 + budget_s - 20  # 20s buffer for safety

    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = _score(best_seq)
        n_main = len(best_seq)
        print(f"Parent: n={n_main}, score={best_score:.10f}")
    else:
        best_seq = _make_profile(16384, "tapered")
        best_score = _score(best_seq)
        n_main = len(best_seq)
        print(f"No parent. Tapered n={n_main}: score={best_score:.10f}")

    z = _z_from_a(best_seq)
    gap = _compute_gap(best_seq)
    print(f"Initial gap: {gap:.6e}")

    # === Phase 1: Coarse L-BFGS ===
    p1_budget = 0.10 * (budget_s - 20)
    p1_end = t0 + p1_budget
    if time.monotonic() < p1_end:
        print(f"\n[P1] Coarse L-BFGS (budget={p1_budget:.0f}s)")
        for beta_mult in [0.01, 0.003]:
            if time.monotonic() > p1_end:
                break
            beta = gap * beta_mult
            prob = _LSEProb(beta, n_main)
            remaining = p1_end - time.monotonic()
            max_iter = min(400, max(100, int(remaining / 0.05)))
            result = _run_lbfgs(z, prob, max_iter)
            z = result.x
            a = _softmax(z)
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
            print(f"  P1 beta={beta:.2e}: score={sc:.10f}, nit={result.nit}")

    # === Phase 2: Adaptive beta refinement (main) ===
    p2_budget = 0.50 * (budget_s - 20)
    p2_end = t0 + p2_budget
    print(f"\n[P2] Adaptive beta refinement (budget={p2_budget:.0f}s)")
    beta = gap * 0.001
    min_beta = gap * 1e-9
    rounds = 0
    consecutive_no_improve = 0

    while beta > min_beta and time.monotonic() < p2_end:
        rounds += 1
        remaining = p2_end - time.monotonic()
        if remaining < 10:
            break

        n = len(z)
        size_factor = max(1.0, n / 8192.0)
        est_iter_time = 0.004 * size_factor

        # More iterations when beta is small (harder problem)
        if beta < gap * 1e-4:
            max_iter = min(5000, max(500, int(remaining / (est_iter_time * 40))))
        else:
            max_iter = min(3000, max(300, int(remaining / (est_iter_time * 25))))

        prob = _LSEProb(beta, n)
        result = _run_lbfgs(z, prob, max_iter)
        z = result.x
        a = _softmax(z)
        sc = _score(a)

        improved = sc < best_score - 1e-13
        if improved:
            best_score = sc
            best_seq = a.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        # Also try power mean as alternative
        if rounds % 3 == 0 and time.monotonic() < p2_end - 30:
            for p_try in [8, 16]:
                pm_prob = _PMProb(p_try, n)
                remaining_pm = p2_end - time.monotonic()
                if remaining_pm < 20:
                    break
                max_iter_pm = min(500, max(100, int(remaining_pm / 0.1)))
                result_pm = _run_lbfgs(z, pm_prob, max_iter_pm)
                z_pm = result_pm.x
                a_pm = _softmax(z_pm)
                sc_pm = _score(a_pm)
                if sc_pm < best_score - 1e-13:
                    best_score = sc_pm
                    best_seq = a_pm.copy()
                    z = z_pm.copy()
                    print(f"    PM p={p_try} improved: {best_score:.10f}")
                    break

        if rounds % 2 == 0 or improved:
            print(f"  P2 r{rounds}: beta={beta:.2e}, score={sc:.10f}, "
                  f"best={best_score:.10f}, nit={result.nit}, "
                  f"t={time.monotonic()-t0:.0f}s")

        # Adaptive beta reduction
        if improved:
            beta *= 0.4
        elif consecutive_no_improve <= 2:
            beta *= 0.6
        else:
            beta *= 0.3  # Aggressive reduction when stuck

        # If L-BFGS converged very quickly, go to next beta faster
        if result.nit < max_iter * 0.2:
            beta *= 0.5

    print(f"  P2 done: {rounds} rounds, final beta={beta:.2e}, best={best_score:.10f}")

    # === Phase 3: Perturbation-based local search ===
    p3_budget = 0.15 * (budget_s - 20)
    p3_end = t0 + p3_budget
    if time.monotonic() < p3_end:
        print(f"\n[P3] Perturbation local search (budget={p3_budget:.0f}s)")
        z = _z_from_a(best_seq)
        pert_beta = max(gap * 1e-6, 1e-12)
        n_pert_rounds = 0
        max_pert = 20

        while n_pert < max_pert and time.monotonic() < p3_end - 10:
            n_pert += 1
            # Scale noise to decrease over time
            noise_scale = 0.15 / (1.0 + 0.15 * n_pert)
            z_pert = z + noise_scale * rng.standard_normal(len(z))

            remaining = p3_end - 10 - time.monotonic()
            if remaining < 5:
                break
            max_iter = min(400, max(50, int(remaining / 0.15)))
            prob = _LSEProb(pert_beta, len(z))
            result = _run_lbfgs(z_pert, prob, max_iter)
            z_trial = result.x
            a_trial = _softmax(z_trial)
            sc_trial = _score(a_trial)

            if sc_trial < best_score - 1e-13:
                best_score = sc_trial
                best_seq = a_trial.copy()
                z = z_trial.copy()
                print(f"  P3 r{n_pert}: improved to {best_score:.10f}")
            elif n_pert % 5 == 0:
                print(f"  P3 r{n_pert}: best={best_score:.10f}")

    # === Phase 4: Subgradient polish ===
    p4_budget = 0.08 * (budget_s - 20)
    p4_end = t0 + p4_budget
    if time.monotonic() < p4_end:
        print(f"\n[P4] Subgradient polish (budget={p4_budget:.0f}s)")
        a4, sc4 = _subgradient_polish(best_seq, p4_end, min_time=5.0)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4: {best_score:.10f}")
        # Quick L-BFGS after subgradient
        if time.monotonic() < deadline - 60:
            z = _z_from_a(best_seq)
            gap_now = _compute_gap(best_seq)
            prob = _LSEProb(gap_now * 1e-5, len(z))
            remaining = deadline - 60 - time.monotonic()
            max_iter = min(500, max(100, int(remaining / 0.1)))
            result = _run_lbfgs(z, prob, max_iter)
            a = _softmax(result.x)
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"  P4b L-BFGS: {best_score:.10f}")

    # === Phase 5: Multi-n exploration (brief) ===
    p5_budget = 0.10 * (budget_s - 20)
    p5_end = t0 + p5_budget
    if time.monotonic() < p5_end:
        print(f"\n[P5] Multi-n exploration (budget={p5_budget:.0f}s)")
        n_main = len(best_seq)
        n_options = [8192, 12288, 16384, 20480, 24576, 32768]
        n_options = [x for x in n_options if x != n_main and x >= 8192]

        for n_try in n_options:
            if time.monotonic() > p5_end - 10:
                break
            a_r = _resample_to_n(best_seq, n_try)
            sc_r = _score(a_r)

            remaining = p5_end - 10 - time.monotonic()
            if remaining < 15:
                if sc_r < best_score:
                    best_score = sc_r
                    best_seq = a_r.copy()
                continue

            z_r = _z_from_a(a_r)
            gap_r = _compute_gap(a_r)
            beta_r = gap_r * 0.001
            prob = _LSEProb(beta_r, n_try)
            max_iter = min(600, max(100, int(remaining / 0.1)))
            result = _run_lbfgs(z_r, prob, max_iter)
            a_opt = _softmax(result.x)
            sc_opt = _score(a_opt)

            if sc_opt < best_score - 1e-13:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"  n={n_try}: improved to {best_score:.10f}")
            elif sc_r < best_score - 1e-13:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  n={n_try}: resampled {sc_r:.10f}")

    # === Phase 6: Final polish ===
    print(f"\n[P6] Final polish (t={time.monotonic()-t0:.0f}s)")
    if time.monotonic() < deadline:
        z = _z_from_a(best_seq)
        gap_final = _compute_gap(best_seq)

        for beta_mult in [2e-5, 5e-6, 1e-6, 5e-7]:
            if time.monotonic() > deadline - 3:
                break
            beta = gap_final * beta_mult
            remaining = deadline - 3 - time.monotonic()
            if remaining < 5:
                break
            max_iter = min(3000, max(200, int(remaining / 0.05)))
            prob = _LSEProb(beta, len(z))
            result = _run_lbfgs(z, prob, max_iter)
            z = result.x
            a = _softmax(z)
            sc = _score(a)
            if sc < best_score - 1e-14:
                best_score = sc
                best_seq = a.copy()
            print(f"  P6 beta={beta:.2e}: score={sc:.10f}, nit={result.nit}, "
                  f"t={time.monotonic()-t0:.0f}s")

    # Final validation
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    elapsed = time.monotonic() - t0
    print(f"\n{'='*60}")
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={elapsed:.1f}s")
    return [float(x) for x in final]
