```python
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def solve_lp(a, M):
    n = len(a)
    # Build sparse matrix A where A[k,m] = a[k-m] if 0 <= k-m < n
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        for m in range(m_lo, m_hi + 1):
            rows.append(k)
            cols.append(m)
            vals.append(a[k - m])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Minimize -sum(b) subject to A*b <= M, b >= 0
    c_vec = -np.ones(n)
    res = linprog(c_vec, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x, -res.fun
    return None, None
```

"""
LP-based direction finding + log-sum-exp L-BFGS hybrid optimizer.
Key innovation: Uses a linear program to find globally optimal descent
directions (maximizing sum(b) subject to cross-correlation bounds),
combined with adaptive line search and fine L-BFGS polishing.
Also explores reduced-constraint LPs for larger sequences.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _build_sparse_A(a):
    """Build sparse matrix A where A[k,m] = a[k-m] if 0 <= k-m < n."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            ms = np.arange(m_lo, m_hi + 1)
            rows.extend([k] * len(ms))
            cols.extend(ms.tolist())
            vals.extend(a[k - ms].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _solve_lp(a, M, top_k=None):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M, b >= 0.
    If top_k is specified, only use the top_k tightest constraints."""
    n = len(a)
    conv = _full_convolve(a, a)

    if top_k is not None and top_k < len(conv):
        # Use only top_k constraints
        k_idx = np.argsort(conv)[::-1][:top_k]
        A = _build_sparse_A(a)
        A_sub = A[k_idx, :]
        b_ub = np.full(len(k_idx), M)
    else:
        A = _build_sparse_A(a)
        b_ub = np.full(2 * n - 1, M)

    c_vec = -np.ones(n)
    try:
        res = linprog(c_vec, A_ub=A_sub if top_k is not None and top_k < len(conv) else A,
                      b_ub=b_ub, bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 30})
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, None


def _lp_optimize(a_init, deadline, min_time_remaining=60.0, use_full_lp=True, seed=0):
    """LP-based direction finding with adaptive line search."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(seed)

    t = 0
    no_improve = 0
    step = 0.3

    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        conv = _full_convolve(a, a)
        M = float(conv.max())
        current_sc = 2.0 * n * M

        if current_sc < best_sc - 1e-12:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            break

        # Choose top_k for reduced LP
        if n > 5000:
            top_k = min(500, int(0.1 * n))
        else:
            top_k = None

        b_star, sum_b = _solve_lp(a, M, top_k=top_k)
        if b_star is None or sum_b <= 1.0 + 1e-10:
            # LP is stuck; try perturbation
            pert = 0.005 * rng.standard_normal(n)
            a_pert = _project_simplex(a + pert)
            sc_pert = _score(a_pert)
            if sc_pert < best_sc:
                a = a_pert
                best_a = a.copy()
                best_sc = sc_pert
            step *= 0.7
            continue

        g = b_star / sum_b

        # Line search: try several step sizes
        best_t = 0.0
        best_sc_try = current_sc
        for t_try in [step, step * 0.5, step * 0.25, step * 0.1, 0.5, 0.3, 0.15, 0.05]:
            a_try = (1 - t_try) * a + t_try * g
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_t = t_try

        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
            step = min(1.0, step * 1.1)
        else:
            step *= 0.5
            if step < 0.001:
                break

        if t % 20 == 0:
            print(f"    LP t={t}: best={best_sc:.10f}, cur={current_sc:.10f}, step={step:.4f}")

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with log-sum-exp smoothing and adaptive beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.002)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03125, 0.015, 0.0075, 0.00375,
                                  0.0019, 0.00095, 0.0005, 0.00025,
                                  0.00012, 0.00006, 0.00003, 0.000015,
                                  0.000008, 0.000004])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 12))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 3 == 0:
                    print(f"    L-BFGS beta_idx={i}: score={current_sc:.10f}")
        except Exception:
            pass

    return current_a


def _power_sum_obj(z, p, n):
    """Power-sum smoothed objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_pos = np.maximum(c, 1e-30)
    val = np.sum(c_pos ** p) ** (1.0 / p)
    return 2.0 * n * val


def _power_sum_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_pos = np.maximum(c, 1e-30)
    S = np.sum(c_pos ** p)
    S_inv_p = S ** (-1.0 / p)
    # d/dz sum(c^p)^{1/p} = (1/p) * S^{1/p - 1} * sum p * c^{p-1} * dc/dz
    # = S^{1/p - 1} * sum c^{p-1} * dc/dz
    w = c_pos ** (p - 1) / S
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W) * S_inv_p * S ** (1.0 / p - 1.0) * p / p


def _run_power_sum_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with power-sum smoothing (alternative to log-sum-exp)."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Increase p from 4 to 256
    ps = [4, 8, 16, 32, 64, 128, 256]

    for p in ps:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(3000, max(20, int(remaining / (est_time_per_iter * 15))))

        try:
            result = minimize(
                lambda zz: _power_sum_obj(zz, p, n),
                z,
                jac=lambda zz: _power_sum_grad(zz, p, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 40}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    PowerSum p={p}: score={current_sc:.10f}")
        except Exception:
            pass

    return current_a


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.arange(n)
        a = np.exp(-4.0 * ((t - n / 2) / n) ** 2)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "centripetal":
        # Emphasize center
        t = np.arange(n)
        a = 1.0 + 0.5 * np.cos(np.pi * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _nesterov_subgrad(a_init, deadline, min_time_remaining=30.0, K_init=5, K_max=200, seed=0):
    """Nesterov-accelerated subgradient with active set."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    momentum = 0.85
    base_lr = 0.003
    lr = base_lr
    K = K_init
    t = 0
    no_improve = 0
    stall_count = 0

    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive K
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
        if gap < c_max * 0.005 and K < K_max:
            K = min(K_max, int(K * 1.3) + 1)

        # Select active peaks
        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]

        # Compute subgradient (vectorized)
        grad = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                idx = k - np.arange(j_lo, j_hi + 1)
                grad[j_lo:j_hi + 1] += a[idx]
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_lo2 <= j_hi2:
                idx = np.arange(j_lo2, j_hi2 + 1) - k
                grad[j_lo2:j_hi2 + 1] += a[idx]

        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break

        effective_lr = lr / (gnorm + 1e-18) * 0.1
        v = momentum * v - effective_lr * grad_t
        a_cand = a + v
        a_proj = _project_simplex(a_cand)

        c_new = _full_convolve(a_proj, a_proj)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < current_sc:
            a = a_proj
        else:
            v = 0.5 * v
            lr *= 0.9
            a = a_proj

        if no_improve % 100 == 0 and no_improve > 0:
            stall_count += 1
            if stall_count > 8:
                break
            perturb = 0.002 * rng.standard_normal(n)
            a_pert = _project_simplex(a + perturb)
            c_pert = _full_convolve(a_pert, a_pert)
            sc_pert = 2.0 * n * float(c_pert.max())
            if sc_pert < best_sc:
                a = a_pert
                best_a = a.copy()
                best_sc = sc_pert
            v = np.zeros(n)
            lr = base_lr
            no_improve = 0

    return best_a, best_sc


def _shrinkage_optimize(a, deadline):
    """Try zeroing out small elements to reduce effective n."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.001, 0.003, 0.005, 0.01, 0.02, 0.03, 0.05]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid LP + L-BFGS + Nesterov optimizer.
    Uses LP-based direction finding for small/medium n,
    reduced-LP + log-sum-exp L-BFGS for large n,
    with multi-n exploration and shrinkage.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    elapsed = lambda: time.monotonic() - t_start

    # === Phase 1: LP-based optimization for medium n ===
    print(f"\n[P1] LP optimization (t={elapsed():.1f}s)")
    lp_n_values = [2048, 3072, 4096]
    if n_parent not in lp_n_values and n_parent <= 5000:
        lp_n_values.insert(0, n_parent)

    for n_try in lp_n_values:
        if elapsed() > deadline - 400:
            break
        if n_try == len(best_seq):
            a_init = best_seq.copy()
        else:
            a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")

        p1_deadline = min(deadline - 350, t_start + 250)
        a_lp, sc_lp = _lp_optimize(a_init, p1_deadline, min_time_remaining=30, seed=seed + n_try)
        sc_lp = _score(a_lp)
        if sc_lp < best_score:
            best_seq = a_lp.copy()
            best_score = sc_lp
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [LP] t={elapsed():.1f}s")
        else:
            print(f"  LP n={n_try}: no improvement (score={sc_lp:.10f})")

    # === Phase 2: L-BFGS polish on best ===
    print(f"\n[P2] L-BFGS polish (t={elapsed():.1f}s)")
    if elapsed() < deadline - 300:
        p2_deadline = min(deadline - 250, t_start + 450)
        a_lb = _run_lbfgs(best_seq, p2_deadline, min_time_remaining=30.0)
        sc_lb = _score(a_lb)
        if sc_lb < best_score:
            best_seq = a_lb.copy()
            best_score = sc_lb
            print(f"  NEW BEST: {best_score:.10f} [L-BFGS] t={elapsed():.1f}s")

    # === Phase 3: Nesterov subgradient ===
    print(f"\n[P3] Nesterov (t={elapsed():.1f}s)")
    if elapsed() < deadline - 250:
        p3_deadline = min(deadline - 200, t_start + 600)
        a_nes, sc_nes = _nesterov_subgrad(best_seq, p3_deadline,
                                           min_time_remaining=30.0,
                                           K_init=5, K_max=300, seed=seed + 1)
        if sc_nes < best_score:
            best_seq = a_nes.copy()
            best_score = sc_nes
            print(f"  NEW BEST: {best_score:.10f} [Nesterov] t={elapsed():.1f}s")

    # === Phase 4: Power-sum L-BFGS (alternative smoothing) ===
    print(f"\n[P4] Power-sum L-BFGS (t={elapsed():.1f}s)")

        a_ps = _run_power_sum_lbfgs(best_seq, p4_deadline, min_time_remaining=30.0)
        sc_ps = _score(a_ps)

            best_score = sc_ps
