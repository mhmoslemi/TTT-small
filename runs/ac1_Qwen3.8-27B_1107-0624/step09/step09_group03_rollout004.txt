import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    """Compute the part of convolve(flip(a), A_vec) that gives gradient w.r.t. a."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec, rhs, max_cg=300, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _newton_step_active(a, c, c_max, tol_frac, mu):
    """Compute Newton step for active set defined by tol_frac."""
    n = len(a)
    active = c >= (c_max - tol_frac * c_max)
    if not active.any():
        active[np.argmax(c)] = True
    m = int(active.sum())
    if m < 2 or m > 20 * n:
        return None
    
    A_vec = active.astype(np.float64) / m
    g = 2.0 * _corr_a_A(a, A_vec, n)
    g_t = g - np.mean(g)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-14:
        return None
    
    A_vec_c = A_vec.copy()
    n_c = n
    mu_c = mu
    
    def _hess_vp(v):
        return 2.0 * _corr_a_A(v, A_vec_c, n_c) + mu_c * v
    
    u = _cg_solve(_hess_vp, g_t, max_cg=300)
    w = _cg_solve(_hess_vp, np.ones(n), max_cg=300)
    w_sum = float(w.sum())
    
    if abs(w_sum) < 1e-15:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w
    
    dx = dx - np.mean(dx)
    return dx


def _kkt_check(a, c, c_max, tol_frac):
    """Check KKT conditions for the active set."""
    n = len(a)
    active = c >= (c_max - tol_frac * c_max)
    if not active.any():
        return False
    m = int(active.sum())
    A_vec = active.astype(np.float64) / m
    g = 2.0 * _corr_a_A(a, A_vec, n)
    g_mean = np.mean(g)
    # On active set: gradient should be ~ constant (the multiplier)
    active_grad = g[active]
    if np.std(active_grad) > 0.1 * (np.abs(g_mean) + 1e-10):
        return False
    # On inactive set: gradient should be <= multiplier (for minimization)
    inactive_grad = g[~active]
    if inactive_grad.size > 0 and np.max(inactive_grad) > g_mean + 1e-8:
        return False
    return True


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "sqrt_taper":
        t = np.linspace(0, 1, n)
        a = np.sqrt(np.sin(np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        a = np.ones(n)
        fade = n // 10
        for i in range(fade):
            a[i] = 0.5 * (1 - np.cos(np.pi * i / fade))
            a[n - 1 - i] = a[i]
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_fourier(a, rng, n_modes=20, sigma=0.1):
    """Perturb in frequency domain - modify random Fourier modes."""
    n = len(a)
    A = np.fft.rfft(a)
    # Pick random frequency modes to perturb
    n_freq = len(A)
    idx = rng.choice(n_freq, size=min(n_modes, n_freq), replace=False)
    for i in idx:
        if i == 0 or i == n_freq - 1:
            continue
        phase_shift = rng.uniform(-sigma, sigma)
        mag_shift = 1.0 + rng.normal(0, sigma * 0.5)
        A[i] *= mag_shift * np.exp(1j * phase_shift)
        if i < n_freq - 1:
            A[n_freq - 1 - i] = np.conj(A[i])
    a_new = np.fft.irfft(A, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _perturb_additive(a, rng, sigma):
    """Additive perturbation with simplex projection."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _symmetrize(a):
    """Make sequence symmetric: a[i] = a[n-1-i]."""
    n = len(a)
    half = (n + 1) // 2
    a_sym = np.zeros(n)
    a_sym[:half] = a[:half]
    a_sym[half:] = a[:half][::-1][:n - half]
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s > 0:
        a_sym = a_sym / s
    return a_sym


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _lbfgs_run(a_init, deadline, min_remaining=30.0, n_betas=20):
    """L-BFGS with log-sum-exp continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 50:
        gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    else:
        gap = c_sorted[0] - c_sorted[-1]
    base_beta = max(gap * 0.2, max_c * 0.003)
    min_beta = max(max_c * 5e-8, 1e-15)
    
    log_betas = np.linspace(np.log(base_beta), np.log(min_beta), n_betas)
    betas = np.exp(log_betas)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        
        est_time = 0.004 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 8))))
        
        beta_c = beta
        n_c = n
        
        def _obj(z_val):
            return _lsobj(z_val, beta_c, n_c)
        
        def _grad(z_val):
            return _lsgrad(z_val, beta_c, n_c)
        
        try:
            result = minimize(
                _obj, z, jac=_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 150, 'maxls': 500}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _newton_active_set(a_init, deadline, max_outer=200):
    """Active-set Newton with adaptive tolerance and KKT check."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive tolerance: start loose, tighten
        if outer < 30:
            tol_list = [0.05, 0.02, 0.01, 0.005]
        elif outer < 80:
            tol_list = [0.005, 0.002, 0.001, 0.0005]
        else:
            tol_list = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
        
        best_improvement = 0
        best_a_new = None
        
        for tol_frac in tol_list:
            mu = 1e-6 / max(1, int((c >= c_max - tol_frac * c_max).sum())) + 1e-9
            dx = _newton_step_active(a, c, c_max, tol_frac, mu)
            if dx is None:
                continue
            
            # Backtracking line search
            t = 1.0
            for ls in range(60):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < c_max - 1e-10 * t:
                    improvement = c_max - f_try
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_a_new = a_try.copy()
                    break
                t *= 0.5
        
        if best_a_new is None:
            # Try gradient descent fallback
            g_all = 2.0 * _corr_a_A(a, np.ones(2 * n - 1) / (2 * n - 1), n)
            g_t = g_all - np.mean(g_all)
            t = 0.05
            improved = False
            for ls in range(40):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_sc - 1e-14:
                        best_a_new = a_try.copy()
                        best_improvement = 1
                        improved = True
                        break
                t *= 0.5
            if not improved:
                break
        
        a = best_a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # KKT convergence check
        if _kkt_check(a, _full_convolve(a, a), float(_full_convolve(a, a).max()), 0.001):
            break
    
    return best_a, best_sc


def _deflation_refine(a, deadline, n_attempts=50):
    """Multi-peak deflation refinement."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find peaks
        for thresh_frac in [0.999, 0.995, 0.99, 0.98]:
            threshold = c_max * thresh_frac
            peak_mask = c >= threshold
            n_peaks = int(peak_mask.sum())
            if n_peaks == 0:
                peak_mask[np.argmax(c)] = True
                n_peaks = 1
            
            w = np.zeros_like(c)
            w[peak_mask] = 1.0 / n_peaks
            
            grad = 2.0 * _corr_a_A(a, w, n)
            g_t = grad - np.mean(grad)
            
            applied = False
            for strength in [0.0002, 0.0005, 0.001, 0.002, 0.005, 0.01]:
                a_try = a - strength * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    applied = True
                    break
            
            if applied:
                break
    
    return best_a, best_sc


def _refinement_loop(a, deadline, max_rounds=8):
    """Iterate between Newton and deflation."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        # Newton
        nt_dl = min(deadline - 10, time.monotonic() + min(60, max(10, (deadline - time.monotonic()) / max(1, max_rounds - rnd))))
        a_nt, sc_nt = _newton_active_set(a, nt_dl, max_outer=40)
        if sc_nt < best_sc:
            best_sc = sc_nt
            best_a = a_nt.copy()
            a = a_nt
        
        # Deflation
        df_dl = min(deadline - 10, time.monotonic() + min(30, max(5, (deadline - time.monotonic()) / max(1, max_rounds - rnd))))
        a_df, sc_df = _deflation_refine(a, df_dl, n_attempts=15)
        if sc_df < best_sc:
            best_sc = sc_df
            best_a = a_df.copy()
            a = a_df
        
        if abs(_score(a) - best_sc) < 1e-15:
            break
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining L-BFGS continuation, active-set Newton,
    Fourier perturbations, and multi-scale exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.12f} (n={len(p_arr)}) [{label}] "
                  f"dt={time.monotonic() - t_start:.1f}s")
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.12f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "raised_cosine")
        update_best(parent, _score(parent), "init")
    
    a = best_seq.copy()
    
    # === Phase 1: Aggressive L-BFGS continuation ===
    print(f"\n[Phase 1] L-BFGS continuation (t={time.monotonic() - t_start:.1f}s)")
    p1_dl = min(deadline - 600, time.monotonic() + 250)
    a_p1 = _lbfgs_run(a, p1_dl, min_remaining=600, n_betas=25)
    update_best(a_p1, _score(a_p1), "lbfgs_p1")
    if best_seq is not None:
        a = best_seq.copy()
    
    # === Phase 2: Active-set Newton ===
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    p2_dl = min(deadline - 450, time.monotonic() + 150)
    a_p2, sc_p2 = _newton_active_set(a, p2_dl, max_outer=100)
    update_best(a_p2, sc_p2, "newton_p2")
    if best_seq is not None:
        a = best_seq.copy()
    
    # === Phase 3: Deflation ===
    print(f"\n[Phase 3] Deflation (t={time.monotonic() - t_start:.1f}s)")
    p3_dl = min(deadline - 350, time.monotonic() + 60)
    a_p3, sc_p3 = _deflation_refine(a, p3_dl, n_attempts=40)
    update_best(a_p3, sc_p3, "deflation_p3")
    if best_seq is not None:
        a = best_seq.copy()
    
    # === Phase 4: Diverse Fourier restarts ===
    print(f"\n[Phase 4] Fourier restarts (t={time.monotonic() - t_start:.1f}s)")
    n_restarts = 8
    for i in range(n_restarts):
        if time.monotonic() > deadline - 250:
            break
        remaining_for_restart = (deadline - 250 - time.monotonic()) / max(1, n_restarts - i)
        
        # Generate diverse perturbation
        restart_type = i % 4
        if restart_type == 0:
            p_start = _perturb_fourier(a, rng, n_modes=30 + 10 * i, sigma=0.05 + 0.02 * i)
        elif restart_type == 1:
            p_start = _perturb_additive(a, rng, sigma=0.02 + 0.03 * i)
        elif restart_type == 2:
            p_start = _symmetrize(_perturb_additive(a, rng, sigma=0.03 + 0.02 * i))
        else:
            # Fourier with larger perturbation
            p_start = _perturb_fourier(a, rng, n_modes=50 + 20 * i, sigma=0.1 + 0.05 * i)
        
        sc_start = _score(p_start)
        if sc_start < best_score:
            update_best(p_start, sc_start, f"restart_init_{i}")
        
        # Quick L-BFGS on the restart
        r_dl = min(deadline - 200, time.monotonic() + remaining_for_restart * 0.6)
        a_r = _lbfgs_run(p_start, r_dl, min_remaining=200, n_betas=10)
        update_best(a_r, _score(a_r), f"restart_lbfgs_{i}")
        
        # Quick Newton
        if time.monotonic() < deadline - 100:
            nt_dl = min(deadline - 100, time.monotonic() + remaining_for_restart * 0.3)
            a_nt, sc_nt = _newton_active_set(a_r, nt_dl, max_outer=30)
            update_best(a_nt, sc_nt, f"restart_nt_{i}")
    
    if best_seq is not None:
        a = best_seq.copy()
    
    # === Phase 5: Refinement loop on best ===
    print(f"\n[Phase 5] Refinement loop (t={time.monotonic() - t_start:.1f}s)")
    p5_dl = min(deadline - 150, time.monotonic() + 100)
    a_p5, sc_p5 = _refinement_loop(a, p5_dl, max_rounds=6)
    update_best(a_p5, sc_p5, "refinement_p5")
    if best_seq is not None:
        a = best_seq.copy()
    
    # === Phase 6: Try longer sequence ===
    print(f"\n[Phase 6] Longer sequence (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 120:
        n_large = 32768
        a_large = _interpolate_seq(a, n_large)
        sc_large = _score(a_large)
        print(f"  n={n_large} init score: {sc_large:.12f}")
        
        lb_dl = min(deadline - 80, time.monotonic() + 60)
        a_large_opt = _lbfgs_run(a_large, lb_dl, min_remaining=80, n_betas=12)
        update_best(a_large_opt, _score(a_large_opt), "n32768_lbfgs")
        
        if time.monotonic() < deadline - 60:
            nt_dl = min(deadline - 60, time.monotonic() + 30)
            a_large_nt, sc_large_nt = _newton_active_set(a_large_opt, nt_dl, max_outer=30)
            update_best(a_large_nt, sc_large_nt, "n32768_nt")
    
    # === Phase 7: Final polish ===
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        a = best_seq.copy()
        
        # Very small beta L-BFGS
        n = len(a)
        c_best = _full_convolve(a, a)
        mb = float(c_best.max())
        z = np.log(np.maximum(a, 1e-30))
        z = z - z.mean()
        beta_final = mb * 2e-7
        remaining = deadline - 10 - time.monotonic()
        if remaining > 5:
            beta_c = beta_final
            n_c = n
            max_iter_fin = min(3000, max(100, int(remaining / 0.05)))
            
            def _obj_fin(z_val):
                return _lsobj(z_val, beta_c, n_c)
            
            def _grad_fin(z_val):
                return _lsgrad(z_val, beta_c, n_c)
            
            try:
                result = minimize(
                    _obj_fin, z, jac=_grad_fin,
                    method='L-BFGS-B',
                    options={'maxiter': max_iter_fin, 'ftol': 1e-18, 'gtol': 1e-15,
                             'maxcor': 200, 'maxls': 500}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_tiny_beta")
            except Exception:
                pass
        
        # Final Newton
        if time.monotonic() < deadline - 5:
            a = best_seq.copy()
            fin_dl = deadline - 3
            a_fin2, sc_fin2 = _newton_active_set(a, fin_dl, max_outer=50)
            update_best(a_fin2, sc_fin2, "final_newton")
    
    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
