"""
Optimized step function generator minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses multi-stage L-BFGS annealing, peak-equalization refinement,
targeted stochastic search, and ultra-fine active-set polish.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _lbe_val(z, beta, n):
    """Log-sum-exp smoothed objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lbe_grad(z, beta, n):
    """Gradient of log-sum-exp smoothed objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

class _LBE:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def f(self, z):
        return _lbe_val(z, self.beta, self.n)
    def g(self, z):
        return _lbe_grad(z, self.beta, self.n)

def _run_lbfgs_anneal(a_init, deadline, min_remaining=30.0):
    """L-BFGS with adaptive beta annealing - the main workhorse."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    # Estimate the gap between top peaks
    top_k = min(50, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[top_k]
    if gap < max_c * 0.0005:
        gap = max_c * 0.0005
    base_beta = max(gap, max_c * 0.003)

    # More aggressive and longer annealing schedule
    n_stages = 25
    betas = base_beta * np.exp(np.linspace(0, np.log(max(1e-7, 1.0 / (10 * base_beta))), n_stages))
    betas = np.maximum(betas, max_c * 5e-7)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break

        # Time allocation per stage
        if i < n_stages // 2:
            # Early stages: less time
            stage_time = remaining * 0.3 / (n_stages // 2)
        else:
            # Late stages: more time (critical for convergence)
            stage_time = remaining * 0.7 / (n_stages - n_stages // 2)

        est_per_iter = 0.003 * max(1.0, n / 16384.0)
        max_iter = min(8000, max(50, int(stage_time / (est_per_iter * 12))))

        prob = _LBE(beta, n)
        try:
            result = minimize(
                prob.f, z, jac=prob.g, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        if (i + 1) % 5 == 0:
            print(f"    LBE stage {i+1}/{n_stages}, beta={beta:.2e}, score={current_sc:.10f}")

    return current_a

def _active_set_polish(a_init, deadline, tol_frac=0.001, max_iter=50000):
    """Active-set gradient descent with adaptive step size and momentum."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    v = np.zeros(n)
    momentum = 0.9
    lr = 0.01
    no_improve = 0
    stall_limit = 100

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        # Weighted gradient (inverse-gap weighting)
        eps = 1e-12 * c_max
        w = np.zeros_like(c)
        w[active] = 1.0 / (c_max - c[active] + eps)
        w = w / w.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        grad = 4.0 * n * h  # gradient of 2n*sum(w*c) w.r.t. a
        
        # Project to tangent space
        grad = grad - np.mean(grad)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Heavy-ball momentum with step halving
        v = momentum * v - lr * grad
        alpha = 1.0
        improved = False
        for _ in range(30):
            a_try = _normalize(a + alpha * v)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            v = np.zeros(n)
            no_improve += 1
            if no_improve % 50 == 0:
                tol_frac = max(1e-6, tol_frac * 0.6)
                lr *= 0.5
                momentum = min(0.95, momentum * 1.01)
            if no_improve >= stall_limit:
                # Restart from best
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(1e-7, tol_frac * 0.3)
                lr = 0.005
                no_improve = 0
        else:
            if tol_frac > 1e-7:
                tol_frac = max(1e-7, tol_frac * 0.99)
            lr = min(0.02, lr * 1.01)

    return best_a, best_sc

def _peak_equalize(a_init, deadline):
    """Iteratively equalize the top peaks using Newton-like steps."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    K = 20  # Number of top peaks to equalize
    
    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
        
        # Get top K peaks
        top_idx = np.argsort(c)[::-1][:K]
        top_vals = c[top_idx]
        mean_top = float(top_vals.mean())
        std_top = float(top_vals.std())
        
        if std_top < 1e-15 * c_max:
            break  # Already nearly flat
        
        # Compute gradient to equalize: push peaks down, valleys up
        # Direction: sum_{k in top} (c[k] - mean_top) * a[k-i]
        w = np.zeros_like(c)
        w[top_idx] = (top_vals - mean_top) / (std_top * K)
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        grad = h - np.mean(h)
        
        # Take a small step
        lr = std_top / (np.linalg.norm(grad) + 1e-15) * 0.1
        a_new = _normalize(a - lr * grad)
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())
        
        if sc_new < current_sc - 1e-15:
            a = a_new
        else:
            # Try smaller step
            lr *= 0.1
            a_new = _normalize(a - lr * grad)
            c_new = _full_convolve(a_new, a_new)
            sc_new = 2.0 * n * float(c_new.max())
            if sc_new < current_sc - 1e-15:
                a = a_new
            else:
                break  # Can't improve
    
    return best_a, best_sc

def _stochastic_search(a_init, deadline, rng, min_remaining=60.0):
    """Targeted stochastic local search with diverse perturbations."""
    n = len(a_init)
    current_a = _normalize(a_init.copy())
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc
    
    attempt = 0
    while time.monotonic() + min_remaining < deadline:
        if attempt % 50 == 0:
            print(f"    ILS attempt {attempt}, best={best_sc:.10f}")
        
        strategy = attempt % 6
        
        if strategy == 0:
            # Small global Gaussian perturbation
            sigma = 0.003 + 0.005 * (attempt % 5) / 5.0
            z = np.log(np.maximum(current_a, 1e-30))
            z += sigma * rng.standard_normal(n)
            z -= z.mean()
            a_pert = _softmax(z)
            
        elif strategy == 1:
            # Targeted perturbation near peak-contributing elements
            c = _full_convolve(current_a, current_a)
            peak_lag = int(np.argmax(c))
            # Elements that most affect this peak: a[0..peak_lag] and a[peak_lag..n-1]
            n_target = max(20, n // 10)
            center = peak_lag // 2 if peak_lag < n else n // 2
            start = max(0, center - n_target // 2)
            end = min(n, start + n_target)
            a_pert = current_a.copy()
            seg = a_pert[start:end]
            z_seg = np.log(np.maximum(seg, 1e-30))
            z_seg += 0.01 * rng.standard_normal(end - start)
            a_pert[start:end] = np.exp(z_seg - np.max(z_seg))
            a_pert = _normalize(a_pert)
            
        elif strategy == 2:
            # Geometric taper perturbation
            seg_len = max(50, n // 10)
            start = int(rng.integers(0, n - seg_len))
            a_pert = current_a.copy()
            t = np.linspace(0, 1, seg_len)
            rate = rng.uniform(0.5, 2.0)
            taper = np.exp(-rate * t)
            a_pert[start:start+seg_len] *= taper
            a_pert = _normalize(a_pert)
            
        elif strategy == 3:
            # Sparse modification
            n_mod = max(10, n // 50)
            idx = rng.choice(n, size=n_mod, replace=False)
            a_pert = current_a.copy()
            a_pert[idx] *= np.exp(0.02 * rng.standard_normal(n_mod))
            a_pert = _normalize(a_pert)
            
        elif strategy == 4:
            # Mirror swap
            seg_len = max(20, n // 20)
            start = int(rng.integers(0, n - seg_len))
            a_pert = current_a.copy()
            seg = a_pert[start:start+seg_len].copy()
            a_pert[start:start+seg_len] = seg[::-1]
            
        else:
            # Phase-based perturbation (multiply by cosine)
            a_pert = current_a.copy()
            freq = rng.uniform(1, 5)
            phase = rng.uniform(0, 2*np.pi)
            t = np.arange(n) / n
            mod = 1.0 + 0.01 * np.cos(2*np.pi*freq*t + phase)
            a_pert *= mod
            a_pert = _normalize(a_pert)
        
        # Quick optimization of perturbed version
        opt_deadline = min(deadline - min_remaining, time.monotonic() + 5)
        z = np.log(np.maximum(a_pert, 1e-30))
        z = z - z.mean()
        
        # Short L-BFGS
        try:
            c_tmp = _full_convolve(a_pert, a_pert)
            mc = float(c_tmp.max())
            beta = mc * 0.01
            prob = _LBE(beta, n)
            result = minimize(prob.f, z, jac=prob.g, method='L-BFGS-B',
                             options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 50})
            a_opt = _softmax(result.x)
        except Exception:
            a_opt = a_pert
        
        sc_opt = _score(a_opt)
        
        if sc_opt < best_sc - 1e-13:
            best_sc = sc_opt
            best_a = a_opt.copy()
            current_a = a_opt.copy()
            print(f"    ILS NEW BEST: {best_sc:.10f} (strategy={strategy})")
        elif sc_opt < current_sc - 1e-13:
            current_a = a_opt.copy()
        
        attempt += 1
    
    return best_a, best_sc

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _quick_optimize(a_init, deadline, min_remaining=10.0):
    """Quick combined optimization for perturbed sequences."""
    n = len(a_init)
    # Short L-BFGS
    c_cur = _full_convolve(a_init, a_init)
    mc = float(c_cur.max())
    if mc > 1e-15:
        z = np.log(np.maximum(a_init, 1e-30))
        z = z - z.mean()
        betas = [mc * 0.05, mc * 0.01, mc * 0.003, mc * 0.001, mc * 0.0003]
        current_a = a_init.copy()
        for beta in betas:
            remaining = deadline - min_remaining - time.monotonic()
            if remaining < 3:
                break
            try:
                prob = _LBE(beta, n)
                max_iter = min(1000, max(30, int(remaining / (0.005 * max(1.0, n/16384.0) * 10))))
                result = minimize(prob.f, z, jac=prob.g, method='L-BFGS-B',
                                 options={'maxiter': max_iter, 'ftol': 1e-19,
                                          'gtol': 1e-16, 'maxcor': 50})
                a_new = _softmax(result.x)
                if _score(a_new) < _score(current_a):
                    current_a = a_new
                    z = np.log(np.maximum(a_new, 1e-30))
                    z = z - z.mean()
            except Exception:
                pass
    else:
        current_a = a_init.copy()
    
    # Quick polish
    if time.monotonic() < deadline - min_remaining - 2:
        pol_deadline = min(deadline - min_remaining, time.monotonic() + 3)
        a_pol, sc_pol = _active_set_polish(current_a, pol_deadline, tol_frac=0.003)
        if sc_pol < _score(current_a):
            current_a = a_pol
    
    return current_a
