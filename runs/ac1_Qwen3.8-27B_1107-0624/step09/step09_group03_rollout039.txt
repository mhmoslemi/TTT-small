"""
KKT-based active set optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: FFT-based CG solver for the KKT system, active set management,
multiple diverse initializations, and iterative refinement.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _hess_a(A_vec, a, n, mu_reg):
    """Compute H(a) = corr(a, A_vec) + mu_reg * a, where corr uses the Hankel structure."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    corr = result[n - 1:2 * n - 1]
    return corr + mu_reg * a


def _cg_solve_hess(A_vec, n, mu_reg, rhs, max_iter=300, tol=1e-12):
    """Solve H(x) = rhs using conjugate gradients, where H is the Hankel-type operator."""
    x = np.zeros(n)
    r = rhs - _hess_a(A_vec, x, n, mu_reg)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Ap = _hess_a(A_vec, p, n, mu_reg)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r_new = r - alpha * Ap
        rs_new = np.dot(r_new, r_new)
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            r = r_new
            break
        p = r_new + (rs_new / rs) * p
        r = r_new
        rs = rs_new
    return x


def _kkt_solve(A_vec, n, mu_reg=1e-10):

    Solve the KKT system: H(a) = nu * 1, sum(a) = 1.
    Returns a, or None if solution has negative components.

    ones = np.ones(n)
    u = _cg_solve_hess(A_vec, n, mu_reg, ones, max_iter=500, tol=1e-13)
    u_sum = float(u.sum())
    if u_sum < 1e-15 or not np.isfinite(u_sum):
        return None
    a = u / u_sum
    return a

def _gradient_descent_refine(a_init, deadline, max_iter=500):
    """
    Gradient-based refinement using the subgradient of the max.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find active peaks (within 99.5% of max)
        threshold = c_max * 0.995
        active = c >= threshold
        n_act = max(1, int(active.sum()))
        
        # Subgradient direction
        A_vec = np.zeros(2 * n - 1)
        A_vec[active] = 1.0 / n_act
        grad = 2.0 * _hess_a(A_vec, a, n, 0.0)
        grad_t = grad - np.mean(grad)
        
        # Backtracking line search
        improved = False
        for step in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]:
            a_try = a - step * grad_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-13:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        
        if not improved:
            break
    
    return best_a, best_sc


def _perturb_refine(a_best, deadline, n_perturb=20, sigma_range=(0.001, 0.05)):
    """
    Try random perturbations around the best solution to escape local optima.
    """
    n = len(a_best)
    best_a = a_best.copy()
    best_sc = _score(a_best)
    rng = np.random.default_rng(12345)
    
    for i in range(n_perturb):
        if time.monotonic() > deadline:
            break
        
        sigma = np.exp(np.linspace(np.log(sigma_range[0]), np.log(sigma_range[1]), n_perturb)[i])
        z = np.log(np.maximum(a_best, 1e-30))
        z = z + sigma * rng.standard_normal(n)
        z = z - z.mean()
        e = np.exp(np.clip(z, -50, 50))
        a_pert = e / e.sum()
        
        sc_pert = _score(a_pert)
        if sc_pert < best_sc:
            best_a = a_pert.copy()
            best_sc = sc_pert
            print(f"    Perturb {i}: sigma={sigma:.4f}, score={best_sc:.12f}")
    
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp_up":
        t = np.linspace(0, 1, n)
        a = t + 0.1
        return a / a.sum()
    elif kind == "ramp_down":
        t = np.linspace(0, 1, n)
        a = (1 - t) + 0.1
        return a / a.sum()
    elif kind == "two_block":
        a = np.zeros(n)
        a[:n // 2] = 1.0
        a[n // 2:] = 0.3
        return a / a.sum()
    elif kind == "two_block_alt":
        a = np.zeros(n)
        a[n // 3:] = 1.0
        a[:n // 3] = 0.4
        return a / a.sum()
    elif kind == "exponential_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-2 * t)
        return a / a.sum()
    elif kind == "exponential_rise":
        t = np.linspace(0, 1, n)
        a = np.exp(2 * t)
        return a / a.sum()
    elif kind == "gaussian_center":
        t = np.linspace(0, 1, n)
        a = np.exp(-4 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "gaussian_offcenter":
        t = np.linspace(0, 1, n)
        a = np.exp(-4 * (t - 0.35) ** 2)
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * t)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _lbfgs_smooth(a_init, deadline, min_remaining=10.0):
    """
    L-BFGS with log-sum-exp smoothing for final polishing.
    """
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    # Start with moderate beta and decrease
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 100:
        gap = c_sorted[0] - c_sorted[100]
    else:
        gap = c_sorted[0] - c_sorted[-1]
    base_beta = max(gap * 0.5, max_c * 0.01)
    
    # Fewer beta steps, focus on getting close
    n_betas = 10
    min_beta = max_c * 1e-6
    log_betas = np.linspace(np.log(base_beta), np.log(max(min_beta, max_c * 1e-8)), n_betas)
    betas = np.exp(log_betas)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Estimate iterations
        est_time_per_iter = 0.005 * max(1.0, n / 8192.0)
        max_iter = min(2000, max(20, int(remaining / (est_time_per_iter * 5))))
        
        try:
            result = minimize(
                _lsobj(z, beta, n),
                z,
                jac=_lsgrad(z, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxcor': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            continue
    
    return current_a


def _lsobj(params, beta, n):
    a = _softmax(params)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(params, beta, n):
    a = _softmax(params)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy KKT-based optimizer with diverse initializations and
    aggressive active set refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.12f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass
    
    n_primary = 16384
    
    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={len(parent)}, score={sc_parent:.12f}")
        update_best(parent, sc_parent, "parent")
        n_primary = len(parent)
    else:
        parent = _make_profile(n_primary, "uniform")
        update_best(parent, _score(parent), "uniform_init")
    
    # === Phase 1: Diverse initializations + quick KKT refinement ===
    print(f"\n[Phase 1] Diverse initializations (t={time.monotonic() - t_start:.1f}s)")
    profiles = ["ramp_up", "ramp_down", "two_block", "two_block_alt",
                "exponential_decay", "exponential_rise", "gaussian_offcenter",
                "sine_mod", "gaussian_center"]
    
    phase1_deadline = min(deadline - 600, time.monotonic() + 300)
    time_per_profile = (phase1_deadline - time.monotonic()) / len(profiles) if len(profiles) > 0 else 60
    
    for prof_name in profiles:
        if time.monotonic() > phase1_deadline:
            break
        a_init = _make_profile(n_primary, prof_name)
        # Quick gradient refinement
        gd_deadline = min(phase1_deadline, time.monotonic() + max(2, time_per_profile * 0.6))
        a_gd, sc_gd = _gradient_descent_refine(a_init, gd_deadline, max_iter=100)
        if sc_gd < best_score:
            # Quick KKT refinement
            kkt_deadline = min(phase1_deadline, time.monotonic() + max(2, time_per_profile * 0.4))
            a_kkt, sc_kkt = _active_set_refine(a_gd, kkt_deadline, max_iter=50)
            update_best(a_kkt, sc_kkt, f"init_{prof_name}")
        else:
            update_best(a_gd, sc_gd, f"init_{prof_name}_gd")
    
    # Also try random exponential initialization
    if time.monotonic() < phase1_deadline:
        rng = np.random.default_rng(seed)
        a_rand = rng.exponential(1.0, n_primary)
        a_rand = a_rand / a_rand.sum()
        gd_d = min(phase1_deadline, time.monotonic() + 30)
        a_gd, sc_gd = _gradient_descent_refine(a_rand, gd_d, max_iter=50)
        kkt_d = min(phase1_deadline - 2, time.monotonic() + 20)
        a_kkt, sc_kkt = _active_set_refine(a_gd, kkt_d, max_iter=30)
        update_best(a_kkt, sc_kkt, "init_random_exp")
    
    # === Phase 2: Deep KKT refinement on best ===
    print(f"\n[Phase 2] Deep KKT refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        phase2_deadline = min(deadline - 400, time.monotonic() + 300)
        a_kkt, sc_kkt = _active_set_refine(best_seq, phase2_deadline, max_iter=500)
        update_best(a_kkt, sc_kkt, "deep_kkt")
    
    # === Phase 3: Gradient refinement ===
    print(f"\n[Phase 3] Gradient refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        phase3_deadline = min(deadline - 300, time.monotonic() + 200)
        a_gd, sc_gd = _gradient_descent_refine(best_seq, phase3_deadline, max_iter=1000)
        update_best(a_gd, sc_gd, "grad_refine")
    
    # === Phase 4: More KKT refinement ===
    print(f"\n[Phase 4] Second KKT pass (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        phase4_deadline = min(deadline - 250, time.monotonic() + 200)
        a_kkt, sc_kkt = _active_set_refine(best_seq, phase4_deadline, max_iter=500)
        update_best(a_kkt, sc_kkt, "kkt_pass2")
    
    # === Phase 5: Perturbation + re-refine ===
    print(f"\n[Phase 5] Perturbation search (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        phase5_deadline = min(deadline - 200, time.monotonic() + 150)
        a_pert, sc_pert = _perturb_refine(best_seq, phase5_deadline, n_perturb=15)
        update_best(a_pert, sc_pert, "perturb")
        
        if sc_pert < best_score + 1e-10:
            # Re-refine the perturbed solution
            kkt_d = min(deadline - 150, time.monotonic() + 100)
            a_kkt, sc_kkt = _active_set_refine(a_pert, kkt_d, max_iter=200)
            update_best(a_kkt, sc_kkt, "perturb_kkt")
    
    # === Phase 6: L-BFGS smooth polishing ===
    print(f"\n[Phase 6] L-BFGS polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        phase6_deadline = min(deadline - 100, time.monotonic() + 150)
        a_smooth = _lbfgs_smooth(best_seq, phase6_deadline, min_remaining=100.0)
        update_best(a_smooth, _score(a_smooth), "lbfgs_smooth")
    
    # === Phase 7: Final KKT + gradient alternation ===
    print(f"\n[Phase 7] Final alternation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        for rnd in range(5):
            if time.monotonic() > deadline - 30:
                break
            # KKT step
            kkt_d = min(deadline - 20, time.monotonic() + 40)
            a_k, sc_k = _active_set_refine(best_seq, kkt_d, max_iter=50)
            if sc_k < best_score:
                best_seq = a_k
                best_score = sc_k
                print(f"    Alt KKT: {sc_k:.12f}")
            # Gradient step
            gd_d = min(deadline - 15, time.monotonic() + 30)
            a_g, sc_g = _gradient_descent_refine(best_seq, gd_d, max_iter=50)
            if sc_g < best_score:
                best_seq = a_g
                best_score = sc_g
                print(f"    Alt Grad: {sc_g:.12f}")
    
    # === Phase 8: Try larger n ===
    print(f"\n[Phase 8] Larger n (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_large = 32768
        a_large = _interpolate(best_seq, n_large)
        sc_large = _score(a_large)
        print(f"  n={n_large} init: {sc_large:.12f}")
        
        if sc_large < best_score + 0.01:
            # Quick refinement on larger n
            phase8_deadline = min(deadline - 30, time.monotonic() + 60)
            a_large_kkt, sc_large_kkt = _active_set_refine(a_large, phase8_deadline, max_iter=50)
            update_best(a_large_kkt, sc_large_kkt, "n_32768")
    
    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
