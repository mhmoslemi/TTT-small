"""
Optimized peak-convolution minimizer using adaptive log-sum-exp smoothing,
L-BFGS in log-space, peak-shaving gradient with momentum, and multi-n search.

import numpy as np
import time
from scipy.optimize import minimize

# Helper functions (top-level, no closures)

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))

def softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def obj_and_grad(z, beta, n, top_k=1):
    """Compute objective and gradient of smoothed max using top-k peaks."""
    a = softmax(z)
    c = full_conv(a, a)
    
    if top_k == 1:
        # Standard log-sum-exp
        c_max = float(c.max())
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s_e = e.sum()
        if s_e < 1e-300:
            obj = 2.0 * n * c_max
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            obj = 2.0 * n * (c_max + beta * np.log(s_e))
            w = e / s_e
    else:
        # Sum of top-k peaks (smoothed)
        sorted_c = np.sort(c)[::-1]
        k = min(top_k, len(c))
        c_top = sorted_c[:k]
        c_kth = c_top[-1]
        # Smooth approximation of sum of top-k: use softmax over top portion
        shifted = (c - c_kth) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        # Only keep top k
        top_idx = np.argsort(c)[::-1][:k]
        e_other = np.zeros_like(e)
        e_other[top_idx] = e[top_idx]
        s_e = e_other.sum()
        if s_e < 1e-300:
            obj = 2.0 * n * float(c_top.sum())
            w = np.zeros_like(c)
            w[top_idx] = 1.0 / k
        else:
            # Approximate: obj ≈ 2n * (c_kth + beta * log(sum of top-k exp((c-c_kth)/beta)))
            obj = 2.0 * n * float(c_top.sum())  # hard top-k for now
            w = np.zeros_like(c)
            w[top_idx] = 1.0 / k
    
    # Gradient
    a_rev = a[::-1]
    wc = full_conv(w, a_rev)
    h = wc[n-1:2*n-1]
    W_val = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W_val)
    
    return obj, grad

Hmm, but `cp.quad_form` for the quadratic terms might be slow for large n. The Hessian of `sum_k p_k (a*a)[k]` is a banded matrix (bandwidth n), so it's sparse.

Actually, `(a*a)[k] = sum_i a[i]*a[k-i]`, so the second derivative with respect to a[j] and a[m] is:
- If j+m = k: 1 (or 2 if j ≠ m)
- Otherwise: 0

So the Hessian is a "Hankel-like" matrix. For large n, this is expensive to form explicitly.

Let me skip the exact QP solve and stick with gradient methods, but make them more efficient.

1. Efficient log-sum-exp + L-BFGS with adaptive beta
2. Peak-shaving with momentum

Peak-convolution minimizer: minimizes 2*n*max(a*a)/(sum a)^2 for non-negative a.
Uses adaptive log-sum-exp smoothing with L-BFGS in log-space, peak-shaving
gradient with momentum, and multi-length exploration.

import numpy as np

from scipy.optimize import minimize


def _npow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _npow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _conv(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700.0, 700.0))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


class _ObjLS:
    """Log-sum-exp smoothed objective."""
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def __call__(self, z):
        a = _softmax(z)
        c = _conv(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / self.beta
        e = np.exp(np.clip(shifted, -700.0, 0.0))
        s_e = e.sum()
        if s_e < 1e-300:
            return 2.0 * self.n * c_max
        return 2.0 * self.n * (c_max + self.beta * np.log(s_e))


class _GradLS:
    """Gradient of log-sum-exp smoothed objective."""
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def __call__(self, z):
        a = _softmax(z)
        c = _conv(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / self.beta
        e = np.exp(np.clip(shifted, -700.0, 0.0))
        s_e = e.sum()
        if s_e < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s_e
        a_rev = a[::-1]
        wc = _conv(w, a_rev)
        h = wc[self.n - 1:2 * self.n - 1]
        W_val = float(np.dot(w, c))
        return 4.0 * self.n * a * (h - W_val)


def _run_lbfgs(z, n, beta, max_iter, deadline):
    """Run L-BFGS from z with given beta, respecting deadline."""
    obj = _ObjLS(beta, n)
    grad = _GradLS(beta, n)
    try:
        res = minimize(obj, z, jac=grad, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxcor': 50, 'maxls': 50})
        return res.x
    except Exception:
        return z

    momentum = np.zeros(n)

        top_idx = np.argsort(c)[::-1][:k]
        w = np.zeros(len(c))
        w[top_idx] = 1.0 / k
        
        # Gradient
        a_rev = a[::-1]
        wc = _conv(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)

            sc = _score(a_check)

                best_sc = sc
                best_z = z.copy()

def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s
