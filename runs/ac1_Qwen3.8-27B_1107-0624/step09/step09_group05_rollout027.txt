```python
"""
Multi-phase optimizer combining L-BFGS soft-max, KKT active-set linear solver,
and perturbation-based local search.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)

def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

def _grad_soft_structured(a, beta, n):
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)

def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft_structured(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)

class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    
    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)
    
    def grad(self, z):
        return _smograd(z, self.beta, self.n)
```

```python
    """Compute M @ v where M[j,i] = w[support_idx[i] + support_idx[j]] for j,i in support."""
    v_full = np.zeros(n)
    v_full[support_idx] = v
    
    L = _next_pow2(3 * n)  # enough for linear correlation
    G = np.fft.rfft(w, L)
    V = np.fft.rfft(v_full, L)
    corr = np.fft.irfft(G * np.conjugate(V), L)

```

This is O(n log n) per matvec. CG typically needs O(sqrt(kappa)) iterations, so total should be reasonable.

Now, the full KKT solver using CG:

We want to solve: `M @ a_T = C * ones(|T|)` with `sum(a_T) = 1`, `a_T >= 0`.

Approach: 
1. Form the augmented system and solve with least squares (for small |T|)
2. Or use CG on the normal equations (for large |T|)

For simplicity and robustness, let me use the following iterative approach:
1. Initialize `a_T` (e.g., uniform)
2. Compute `r = M @ a_T - mean(M @ a_T)` (residual: deviation from constant)
3. Solve the CG system: find `d` such that `M @ d = -r + alpha * ones` (to also adjust the sum)
4. Actually, this is getting complicated. Let me use a simpler approach.

**Simpler iterative KKT approach:**

At each iteration:
1. Compute `g = M @ a_T` (using FFT matvec)
2. The ideal `a_T` should make all components of `g` equal
3. Update: `a_T_new[i] = a_T[i] * exp(-eta * (g[i] - mean(g)))` (multiplicative weights / mirror descent)
4. Normalize: `a_T_new = a_T_new / sum(a_T_new)`
5. Check convergence

This is the **exponential weights** / **multiplicative weights** update. It's simple, preserves non-negativity, and should converge to the KKT point.

The step size eta can be adapted (e.g., based on the Lipschitz constant of the gradient, or simply halved if the objective increases).

This is essentially a form of **mirror descent** with the negative entropy as the mirror map, which is well-suited for simplex-constrained problems.

```python
            else:
```

"""
Multi-phase optimizer combining L-BFGS soft-max, KKT multiplicative-weights
mirror descent, and perturbation-based local search.

Key innovations over previous:
- KKT-based multiplicative weights (mirror descent) refiner using FFT matvec
- Perturbation-based local search to escape local optima
- Adaptive active-set management
- Improved time management with more iterations at small beta
- Support for non-power-of-2 sequence lengths
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_soft_structured(a, beta, n):
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft_structured(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)


class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _kkt_matvec(v, w, support_idx, n):
    """Compute M @ v where M[j,i] = w[support_idx[i] + support_idx[j]] for j,i in support.
    Uses FFT-based cross-correlation for O(n log n) per call.
    """
    v_full = np.zeros(n)
    v_full[support_idx] = v

    L = _next_pow2(3 * n)
    G = np.fft.rfft(w, L)
    V = np.fft.rfft(v_full, L)
    corr = np.fft.irfft(G * np.conjugate(V), L)

    return corr[support_idx]


def _kkt_mw_refine(a_init, deadline, min_time_remaining=30.0, verbose=False):
    """KKT refinement using multiplicative weights (mirror descent with entropy).
    
    At KKT point with active set S and support T:
      sum_{k in S} w_k * a[k-j] = const for all j in T
      
    We use multiplicative weights: a[j] *= exp(-eta * (grad[j] - mean(grad)))
    which converges to the KKT point while preserving non-negativity.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    eta = 0.3
    eta_min = 1e-7
    tol_active = 0.008
    no_improve = 0
    max_no_improve = 80
    iter_count = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            if verbose and iter_count % 100 == 0:
                print(f"    KKT-MW iter {iter_count}: score={best_sc:.10f}, eta={eta:.4f}")
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            eta = min(eta * 1.05, 3.0)
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                tol_active = max(tol_active * 0.7, 0.001)
                no_improve = 0
                if tol_active <= 0.0011:
                    break
            eta = max(eta * 0.9, eta_min)

        # Identify active set
        active_thresh = c_max * (1.0 - tol_active)
        active = np.zeros(2 * n - 1, dtype=bool)
        active[c >= active_thresh] = True
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        # Cap active set for efficiency (keep top peaks)
        max_active = min(1000, max(20, n // 10))
        if n_active > max_active:
            top_idx = np.argsort(c)[::-1][:max_active]
            active = np.zeros(2 * n - 1, dtype=bool)
            active[top_idx] = True
            n_active = max_active

        # Identify support
        support_thresh = a.max() * 1e-7
        support = a > support_thresh
        support_idx = np.where(support)[0]
        T = len(support_idx)
        if T < 2:
            break

        # Build uniform weights over active set
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active

        # Compute gradient via FFT matvec
        v = a[support_idx]
        g = 2.0 * _kkt_matvec(v, w, support_idx, n)

        # Multiplicative weights update
        g_centered = g - np.mean(g)
        # Clip to avoid overflow
        g_centered = np.clip(g_centered, -50.0 / max(eta, 1e-10), 50.0 / max(eta, 1e-10))
        scale = np.exp(-eta * g_centered)
        a_sup_new = v * scale
        a_sup_new = np.maximum(a_sup_new, 1e-30)
        s_new = float(a_sup_new.sum())
        if s_new < 1e-15:
            break
        a_sup_new = a_sup_new / s_new

        # Construct full a with decay of off-support elements
        a_new = np.zeros(n)
        a_new[support_idx] = a_sup_new
        off_support = ~support
        if np.any(off_support):
            a_new[off_support] = a[off_support] * 0.3

        s_total = float(a_new.sum())
        if s_total < 1e-15:
            break
        a_new = a_new / s_total

        sc_new = _score(a_new)
        if sc_new < current_sc - 1e-14:
            a = a_new
        else:
            # Backtracking
            found = False
            for factor in [0.5, 0.25, 0.1, 0.05]:
                eta_try = eta * factor
                scale_try = np.exp(-eta_try * g_centered)
                a_sup_try = v * scale_try
                a_sup_try = np.maximum(a_sup_try, 1e-30)
                a_sup_try = a_sup_try / a_sup_try.sum()
                a_try = np.zeros(n)
                a_try[support_idx] = a_sup_try
                a_try[off_support] = a[off_support] * 0.3
                s_try = float(a_try.sum())
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-14:
                    a = a_try
                    eta = eta_try
                    found = True
                    break
            if not found and eta <= eta_min * 2:
                break

        iter_count += 1

    return best_a, best_sc

                T = 2000

                break

                sc_new = _score(a_blend)

            a_new[~support] = a[~support] * 0.5
            s_total = float(a_new.sum())
            if s_total > 1e-15:

                sc_new = _score(a_new)

def _perturbation_search(a_init, deadline, min_time_remaining=60.0, verbose=False):
    """Perturbation-based local search: add noise, re-optimize, keep best."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    rng = np.random.default_rng(12345)

    perturb_schedules = [
        (0.02, 0.95), (0.01, 0.97), (0.005, 0.98),
        (0.002, 0.99), (0.05, 0.93),
    ]

    for sigma, keep_frac in perturb_schedules:
        if time.monotonic() + min_time_remaining >= deadline:
            break
        # Create perturbed version
        noise = rng.normal(0, sigma, n)
        a_pert = best_a * (1.0 + noise)
        # Keep only top fraction of mass
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert = a_pert / s

        # Quick L-BFGS on perturbed
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            continue
        p_end = min(deadline - min_time_remaining, time.monotonic() + max(5, remaining * 0.3))
        a_opt, sc_opt = _run_lbfgs(a_pert, p_end, min_time_remaining=5, verbose=False)
        if sc_opt < best_sc:
            if verbose:
                print(f"    Perturb sigma={sigma}: {best_sc:.10f} -> {sc_opt:.10f}")
            best_sc = sc_opt
            best_a = a_opt.copy()

    # Also try structured perturbations
    # Flip a random contiguous block to zero and redistribute
    if time.monotonic() + min_time_remaining < deadline:
        for _ in range(5):
            if time.monotonic() + min_time_remaining >= deadline:
                break
            block_len = n // (10 + rng.randint(0, 10))
            start = rng.randint(0, max(1, n - block_len))
            a_mod = best_a.copy()
            a_mod[start:start + block_len] = 0.0
            mass = a_mod.sum()
            if mass < 1e-10:
                continue
            # Redistribute mass uniformly
            a_mod += (1.0 - mass) / n
            a_mod = np.maximum(a_mod, 0.0)
            a_mod = a_mod / a_mod.sum()

            remaining = deadline - min_time_remaining - time.monotonic()
            if remaining < 10:
                break
            p_end = min(deadline - min_time_remaining, time.monotonic() + max(5, remaining * 0.3))
            a_opt, sc_opt = _run_lbfgs(a_mod, p_end, min_time_remaining=5, verbose=False)
            if sc_opt < best_sc:
                if verbose:
                    print(f"    Block perturb: {sc_opt:.10f}")
                best_sc = sc_opt
                best_a = a_opt.copy()

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0, verbose=True):
    """L-BFGS with adaptive beta schedule."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    # Compute initial gap for beta scheduling
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c0_max * 0.003:
        gap = c0_max * 0.003

    # Fine-grain beta levels
    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
        0.00003125, 0.000015625, 0.0000078, 0.0000039,
        0.000002, 0.000001, 0.0000005, 0.0000002
    ])
    betas = gap * multipliers

    t_start = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        size_factor = max(1.0, n / 8192.0)
        est_iter_time = 0.003 * size_factor
        max_iter = min(10000, max(200, int(remaining / (est_iter_time * 15))))

        if beta < gap * 0.001:
            max_iter = min(max_iter, 3000)

        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-18,
                    'maxcor': 120,
                    'maxls': 80,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    elapsed = time.monotonic() - t_start
                    print(f"    LBFGS beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}, t={elapsed:.0f}s")
        except Exception:
            if verbose:
                print(f"    LBFGS beta={beta:.2e}: error")

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    elif kind == "random_smooth":
        rng = np.random.default_rng(777)
        # Low-pass filtered random
        freq = np.arange(n)
        mask = freq < n // 8
        a = np.zeros(n)
        a[mask] = rng.standard_normal(mask.sum())
        a = np.abs(a) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
