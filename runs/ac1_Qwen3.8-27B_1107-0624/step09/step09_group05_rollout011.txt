Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _convolve_and_max(a):
    c = _full_convolve(a, a)
    return c, float(c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())

    if gap < c0_max * 0.003:
        gap = c0_max * 0.003
    gap *= beta_scale

    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
        0.00003125, 0.000015625, 0.0000078, 0.0000039,
        0.000002, 0.000001, 0.0000005
    ])
    betas = gap * multipliers

    t0 = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        size_factor = max(1.0, n / 8192.0)
        est_iter_time = 0.0015 * size_factor
        max_iter = min(6000, max(200, int(remaining / (est_iter_time * 15))))
        if beta < gap * 0.002:
            max_iter = min(max_iter, 1500)

        prob = _LBFGSProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxcor': 150, 'maxls': 40}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"      L-BFGS beta={beta:.2e}: score={best_sc:.10f}, nit={result.nit}")
        except Exception:
            pass

    return best_a, best_sc


def _bb_projected_gradient(a_init, deadline, min_time_remaining=30.0, verbose=False):
    """Projected subgradient method with Barzilai-Borwein adaptive step size."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init.copy(), _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    a_prev = a.copy()
    grad_prev = None
    best_no_improve = 0
    max_no_improve = 200
    step_size = 0.01
    rng = np.random.default_rng(12345)

    t0 = time.monotonic()
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            best_no_improve = 0
        else:
            best_no_improve += 1

        if best_no_improve > max_no_improve:
            # Restart with perturbation
            noise = rng.normal(0, 0.001, n)
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp > 1e-10:
                a_pert = a_pert / sp
                sc_pert = _score(a_pert)
                if sc_pert < best_sc:
                    a = a_pert
                    best_sc = sc_pert
                    best_a = a.copy()
                best_no_improve = 0
            step_size *= 0.5
            continue

        # Compute subgradient (average over active peaks)
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        # Limit active set size for stability
        if n_active > 50:
            top_idx = np.argsort(c)[::-1][:50]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 50

        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        # BB step size
        if grad_prev is not None:
            s_vec = a - a_prev
            y_vec = grad - grad_prev
            s_dot_s = float(np.dot(s_vec, s_vec))
            s_dot_y = float(np.dot(s_vec, y_vec))
            y_dot_y = float(np.dot(y_vec, y_vec))
            if s_dot_y > 1e-12 and y_dot_y > 1e-12:
                bb1 = s_dot_s / s_dot_y
                bb2 = s_dot_y / y_dot_y
                step_size = 0.5 * (bb1 + bb2)
                step_size = min(step_size, 0.05)
                step_size = max(step_size, 1e-6)

        # Projected gradient step
        grad_centered = grad - np.mean(grad)
        a_new = a - step_size * grad_centered
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Check improvement, backtrack if needed
        sc_new = _score(a_new)
        tries = 0
        while sc_new >= current_sc - 1e-14 and tries < 8:
            step_size *= 0.5
            a_new = a - step_size * grad_centered
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            sc_new = _score(a_new)
            tries += 1

        a_prev = a.copy()
        grad_prev = grad.copy()
        a = a_new

    return best_a, best_sc


def _active_set_refine(a_init, deadline, min_time_remaining=30.0, verbose=False):
    """KKT-inspired active-set refinement with adaptive tolerance."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init.copy(), _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    no_improve = 0
    max_no_improve = 500
    tol_frac = 0.003
    min_tol = 0.00005
    eta = 0.08
    rng = np.random.default_rng(777)

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.7)
            eta = max(0.01, eta * 0.8)
            if tol_frac <= min_tol * 1.05 and eta < 0.015:
                # Try random perturbation
                noise = rng.normal(0, 0.0005, n)
                a_try = a + noise
                a_try = np.maximum(a_try, 0)
                st = a_try.sum()
                if st > 1e-10:
                    a_try = a_try / st
                    sc_try = _score(a_try)
                    if sc_try < best_sc:
                        a = a_try
                        best_sc = sc_try
                        best_a = a.copy()
                        no_improve = 0
                        continue
                break
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]

        support = a > 1e-10 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        lam = float(np.mean(grad_avg[support]))
        residual = grad_avg[support] - lam

        a_new = a.copy()
        a_new[support] -= eta * residual
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new >= current_sc:
            found = False
            for trial_eta in [eta * 0.5, eta * 0.25, eta * 0.1, 0.005]:
                a_try = a.copy()
                a_try[support] -= trial_eta * residual
                a_try = np.maximum(a_try, 0.0)
                st = a_try.sum()
                if st < 1e-10:
                    continue
                a_try = a_try / st
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-13:
                    a_new = a_try
                    sc_new = sc_try
                    found = True
                    break
            if not found:
                tol_frac = max(min_tol, tol_frac * 0.9)
                continue

        a = a_new
        if no_improve == 0:
            tol_frac = max(min_tol, tol_frac * 0.98)
            eta = min(0.15, eta * 1.01)

    return best_a, best_sc


def _hankel_qp(a_init, n_qp, deadline, min_time_remaining=30.0):
    """Solve the Hankel QP: min a^T W a s.t. sum(a)=1, a>=0.
    W is Hankel with W[i,j] = w[i+j], where w = current convolution weights.
    Returns the solution resampled to n_qp.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    c = _full_convolve(a, a)
    c_max = float(c.max())

    # Use weights proportional to c (or c^2 for more aggressive peaking)
    w = c.copy()
    w = w / w.sum()
    w = w + 1e-10  # ensure positive for conditioning

    # Size the QP
    n_qp = min(n_qp, n)
    # Resample w to length 2*n_qp - 1
    w_old = c  # length 2n-1
    if len(w_old) != 2 * n_qp - 1:
        x_old = np.linspace(0, 1, len(w_old))
        x_new = np.linspace(0, 1, 2 * n_qp - 1)
        w_sampled = np.interp(x_new, x_old, w_old)
        w_sampled = np.maximum(w_sampled, 1e-10)
    else:
        w_sampled = np.maximum(w_old, 1e-10)

    # Build Hankel matrix W_qp[i,j] = w_sampled[i+j]
    W = np.zeros((n_qp, n_qp))
    for i in range(n_qp):
        W[i, :] = w_sampled[i:i + n_qp]

    # Solve W @ a_qp = 1 (unconstrained except sum)
    try:
        a_qp = np.linalg.solve(W + 1e-8 * np.eye(n_qp), np.ones(n_qp))
    except np.linalg.LinAlgError:
        a_qp = np.linalg.lstsq(W + 1e-8 * np.eye(n_qp), np.ones(n_qp), rcond=None)[0]

    # Active set: zero out negative components and re-solve
    for iteration in range(50):
        neg_mask = a_qp < 0
        n_neg = int(neg_mask.sum())
        if n_neg == 0:
            break
        if n_neg >= n_qp:
            break
        active_mask = ~neg_mask
        n_act = int(active_mask.sum())
        if n_act < 2:
            break
        W_sub = W[np.ix_(active_mask, active_mask)]
        try:
            a_sub = np.linalg.solve(W_sub + 1e-8 * np.eye(n_act), np.ones(n_act))
        except np.linalg.LinAlgError:
            a_sub = np.linalg.lstsq(W_sub + 1e-8 * np.eye(n_act), np.ones(n_act), rcond=None)[0]
        a_qp = np.zeros(n_qp)
        a_qp[active_mask] = a_sub
        # Check if any active became negative
        if np.all(a_qp[active_mask] > 0):
            break

    a_qp = np.maximum(a_qp, 0)
    s_qp = a_qp.sum()
    if s_qp < 1e-10:
        a_qp = np.ones(n_qp) / n_qp
    else:
        a_qp = a_qp / s_qp

    # Resample back to n
    if n_qp != n:
        x_old = np.linspace(0, 1, n_qp)
        x_new = np.linspace(0, 1, n)
        a_resampled = np.interp(x_new, x_old, a_qp)
        a_resampled = np.maximum(a_resampled, 0)
        sr = a_resampled.sum()
        if sr > 1e-10:
            a_resampled = a_resampled / sr
        else:
            a_resampled = a.copy()
    else:
        a_resampled = a_qp

    return a_resampled


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edge_boost":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _random_perturb(a, scale, rng):
    """Apply a structured random perturbation."""
    n = len(a)
    kind = rng.integers(4)
    if kind == 0:
        # Add sinusoid
        freq = rng.uniform(1, 20)
        phase = rng.uniform(0, 2 * np.pi)
        t = np.arange(n)
        pert = scale * np.sin(2 * np.pi * freq * t / n + phase)
    elif kind == 1:
        # Add low-frequency cosine
        freq = rng.uniform(0.5, 5)
        phase = rng.uniform(0, 2 * np.pi)
        t = np.arange(n)
        pert = scale * np.cos(2 * np.pi * freq * t / n + phase)
    elif kind == 2:
        # Random smooth perturbation (low-pass filtered noise)
        noise = rng.normal(0, scale, n)
        k = max(2, n // 64)
        kernel = np.ones(k) / k
        pert = np.convolve(noise, kernel, mode='same')
    else:
        # Modulated perturbation
        freq = rng.uniform(5, 30)
        mod_freq = rng.uniform(1, 3)
        t = np.arange(n)
        envelope = 0.5 + 0.5 * np.sin(2 * np.pi * mod_freq * t / n)
        pert = scale * envelope * np.sin(2 * np.pi * freq * t / n)

    a_new = a + pert
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining L-BFGS, BB projected gradient,
    active-set refinement, Hankel QP, and multi-start with perturbations.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(16384, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n=16384: score={best_score:.10f}")

    n_main = len(best_seq)

    # === Phase 1: L-BFGS main optimization ===
    print(f"\n[P1] L-BFGS (n={n_main}, t={time.monotonic()-t_start:.0f}s)")
    p1_end = t_start + min(500, (budget_s - 10) * 0.30)
    a1, sc1 = _run_lbfgs(best_seq, p1_end, min_time_remaining=200,
                          verbose=True, beta_scale=1.0)
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  P1: {best_score:.10f}")

    # === Phase 2: BB Projected Gradient ===
    print(f"\n[P2] BB Projected Gradient (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p2_end = min(deadline - 300, time.monotonic() + 250)
        a2, sc2 = _bb_projected_gradient(best_seq, p2_end, min_time_remaining=50,
                                         verbose=True)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(f"  P2: {best_score:.10f}")

    # === Phase 3: Active-set refinement ===
    print(f"\n[P3] Active-set refine (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 40 < deadline:
        p3_end = min(deadline - 250, time.monotonic() + 250)
        a3, sc3 = _active_set_refine(best_seq, p3_end, min_time_remaining=50,
                                     verbose=True)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3: {best_score:.10f}")

    # === Phase 4: L-BFGS round 2 ===
    print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p4_end = min(deadline - 200, time.monotonic() + 300)
        a4, sc4 = _run_lbfgs(best_seq, p4_end, min_time_remaining=100,
                              verbose=True, beta_scale=0.7)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4: {best_score:.10f}")

    # === Phase 5: Multi-start with perturbations ===
    print(f"\n[P5] Multi-start (t={time.monotonic()-t_start:.0f}s}")
    if time.monotonic() + 60 < deadline:
        n_starts = 5
        for k in range(n_starts):
            if time.monotonic() + 80 > deadline:
                break
            scale = 0.003 * (1.0 - 0.15 * k)
            a_pert = _random_perturb(best_seq, scale, rng)
            sc_pert = _score(a_pert)
            p5_end = min(deadline - 100, time.monotonic() + 100)
            a5, sc5 = _run_lbfgs(a_pert, p5_end, min_time_remaining=40,
                                  verbose=False, beta_scale=0.8)
            if sc5 < best_score:
                best_score = sc5
                best_seq = a5.copy()
                print(f"  start {k+1}: {best_score:.10f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n (t={time.monotonic()-t_start:.0f}s)")
    n_options = [12288, 20480, 24576]
    n_options = [x for x in n_options if x != n_main and x <= 32768]
    for n_try in n_options:
        if time.monotonic() + 80 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if sc_r < best_score:
            best_score = sc_r
            best_seq = a_r.copy()
            print(f"  n={n_try} resample: {best_score:.10f}")
        p6_end = min(deadline - 60, time.monotonic() + 90)
        a6, sc6 = _run_lbfgs(a_r, p6_end, min_time_remaining=30,
                              verbose=False, beta_scale=0.9)
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  n={n_try}: {best_score:.10f}")

    # === Phase 7: Hankel QP warm-start ===
    print(f"\n[P7] Hankel QP (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 60 < deadline:
        n_hankel = min(256, len(best_seq))
        t_h = time.monotonic()
        a_h = _hankel_qp(best_seq, n_hankel, deadline, min_time_remaining=30)
        a_h_full = _resample(a_h, len(best_seq))
        sc_h = _score(a_h_full)
        print(f"  Hankel n_qp={n_hankel} raw score: {sc_h:.10f} (t={time.monotonic()-t_h:.1f}s)")
        if sc_h < best_score + 0.1:  # only refine if competitive
            p7_end = min(deadline - 40, time.monotonic() + 80)
            a7, sc7 = _run_lbfgs(a_h_full, p7_end, min_time_remaining=20,
                                  verbose=False, beta_scale=1.0)
            if sc7 < best_score:
                best_score = sc7
                best_seq = a7.copy()
                print(f"  Hankel refined: {best_score:.10f}")

    # === Phase 8: Final combined polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 20 < deadline:
        # Active set
        p8a_end = min(deadline - 20, time.monotonic() + 60)
        a8a, sc8a = _active_set_refine(best_seq, p8a_end, min_time_remaining=10)
        if sc8a < best_score:
            best_score = sc8a
            best_seq = a8a.copy()
            print(f"  P8a: {best_score:.10f}")

        if time.monotonic() + 15 < deadline:
            # L-BFGS
            p8b_end = min(deadline - 5, time.monotonic() + 60)
            a8b, sc8b = _run_lbfgs(best_seq, p8b_end, min_time_remaining=5,
                                    verbose=False, beta_scale=0.5)
            if sc8b < best_score:
                best_score = sc8b
                best_seq = a8b.copy()
                print(f"  P8b: {best_score:.10f}")

        if time.monotonic() + 10 < deadline:
            # BB gradient
            p8c_end = min(deadline - 3, time.monotonic() + 30)
            a8c, sc8c = _bb_projected_gradient(best_seq, p8c_end, min_time_remaining=3)
            if sc8c < best_score:
                best_score = sc8c
                best_seq = a8c.copy()
                print(f"  P8c: {best_score:.10f}")

    # Final
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
