            delta = np.zeros(2*n-1)
            if 2*i < 2*n-1:
                delta[2*i] = 1.0
            
            S_b = float(b.sum())  # = s - a_i_old since a[i]=0 now
            
            # Minimize f(x) = 2*n*max_k(c_b[k] + 2*x*d[k] + x^2*delta[k]) / (S_b + x)^2
            # over x >= 0
            def f(x):
                c_new = c_b + 2*x*d + x*x*delta
                return 2.0*n*float(np.max(c_new)) / ((S_b + x)**2)
            
            # Search in [0, 10*a_i_old + 0.01*max(a)]
            x_max = max(10.0 * a_i_old, 0.01 * float(np.max(a))) + 1e-10
            res = minimize_scalar(f, bounds=(0.0, x_max), method='bounded',
                                  options={'xatol': 1e-12})
            a_new_val = res.x
            a[i] = a_new_val
            
            # Check improvement
            sc_new = _score(a)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a.copy()
                improved = True
        
        if not improved and iteration > 5:
            break
    
    return best_a

def _coord_descent_fast(a, deadline=None, min_time=30.0):
    """Faster coordinate descent using FFT-based updates."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    N = _next_pow2(2*n-1)
    N2 = _next_pow2(4*n-3)  # for convolution of d with d if needed
    
    for iteration in range(200):
        if deadline is not None and time.monotonic() + min_time > deadline:
            break
        
        s_cur = float(a.sum())
        improved = False
        
        for i in range(n):
            a_i_old = a[i]
            a[i] = 0.0
            
            # FFT of b (= a with a[i]=0)
            Fb = np.fft.rfft(a, N)
            c_b = np.fft.irfft(Fb * Fb, N)[:2*n-1]
            
            # d[k] = b[k-i] for valid k
            # This is a shifted version of b
            d = np.zeros(2*n-1)
            # d[k] = b[k-i] where 0 <= k-i < n
            lo = max(0, i)
            hi = min(2*n-1, i+n)
            d[lo:hi] = b_range = a[max(0, lo-i):min(n, hi-i)]
            
            delta = 0.0
            if 2*i < 2*n-1:
                delta = 1.0
            
            S_b = float(a.sum())
            
            # Minimize f(x) = 2n * max(c_b + 2x*d + x^2*delta) / (S_b + x)^2
            def f(x):
                pk = float(np.max(c_b + 2.0*x*d + x*x*delta))
                return 2.0*n*pk / ((S_b+x)**2)
            
            x_max = max(5.0*a_i_old, 0.01*float(np.max(a))) + 1e-10
            res = minimize_scalar(f, bounds=(0.0, x_max), method='bounded',
                                  options={'xatol': 1e-10})
            a[i] = res.x
            a[i] = max(0.0, a[i])
            
            if iteration % 20 == 0 or i == n-1:
                sc = _score(a)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
        
        if not improved and iteration > 3:
            break
    
    return best_a

def _lbfgs_optimize(a_init, deadline, min_time=30.0, betas=None):
    """L-BFGS optimization with log-sum-exp smoothing and adaptive beta."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = max(0.0, c_sorted[0] - c_sorted[min(20, len(c_sorted)-1)])
    base_beta = max(gap * 0.05, max_c * 0.0005)
    
    if betas is None:
        ratios = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
                  0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001,
                  0.000005, 0.000002, 0.000001, 0.0000005, 0.0000002, 0.0000001,
                  0.00000005, 0.00000002, 0.00000001]
    else:
        ratios = betas
    
    for i, ratio in enumerate(ratios):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 5:
            break
        beta = base_beta * ratio
        
        # Estimate time per iteration
        est_time = 0.002 * max(1.0, n / 2048.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 3))))
        
        try:
            def obj(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    return 2.0 * n * c_max_local
                return 2.0 * n * (c_max_local + beta * np.log(s_e))
            
            def grad(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    w = np.zeros_like(c_local)
                    w[int(np.argmax(c_local))] = 1.0
                else:
                    w = e / s_e
                a_rev = a_local[::-1]
                wc = _full_convolve(w, a_rev)
                h = wc[n-1:2*n-1]
                W_val = float(np.dot(w, c_local))
                return 4.0 * n * a_local * (h - W_val)
            
            result = minimize(
                obj, z,
                jac=grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-16:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS beta={beta:.2e}: score={best_sc:.10f}")
        except Exception as e:
            pass
    
    return best_a

def _active_set_optimize(a_init, deadline, min_time=30.0):
    """Active-set optimization: focus on the top few peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    c = _full_convolve(a, a)
    max_c = float(c.max())
    base_beta = max_c * 0.001
    
    n_outer = 0
    max_outer = 200
    
    while time.monotonic() + min_time < deadline and n_outer < max_outer:
        n_outer += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive active set: lags within a fraction of the peak
        for frac in [0.001, 0.005, 0.01, 0.05, 0.1]:
            if time.monotonic() + min_time > deadline:
                break
            threshold = c_max * (1.0 - frac)
            active = c >= threshold
            if active.sum() < 2:
                # Take top-k
                k = min(10, len(c))
                top_k = np.argsort(c)[::-1][:k]
                p = np.zeros(2*n-1)
                p[top_k] = 1.0/k
            else:
                p = np.zeros(2*n-1)
                p[active] = 1.0/active.sum()
            
            # Minimize sum_k p_k * c_k using L-BFGS in log-space
            beta = base_beta * (0.7 ** min(n_outer, 30))
            
            try:
                def obj(z_local):
                    a_local = _softmax(z_local)
                    c_local = _full_convolve(a_local, a_local)
                    return float(np.dot(p, c_local))
                
                def grad(z_local):
                    a_local = _softmax(z_local)
                    c_local = _full_convolve(a_local, a_local)
                    a_rev = a_local[::-1]
                    wc = _full_convolve(p, a_rev)
                    h = wc[n-1:2*n-1]
                    P_val = float(np.dot(p, c_local))
                    return 2.0 * a_local * (h - P_val)
                
                result = minimize(
                    obj, z,
                    jac=grad,
                    method='L-BFGS-B',
                    options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-14,
                             'maxcor': 40}
                )
                a_new = _softmax(result.x)
                sc_new = _score(a_new)
                if sc_new < best_sc - 1e-15:
                    best_sc = sc_new
                    best_a = a_new.copy()
                    z = np.log(np.maximum(a_new, 1e-30))
                    z = z - z.mean()
            except Exception:
                pass
    
    return best_a

def _gradient_descent_smooth(a_init, deadline, min_time=30.0):
    """Direct gradient descent with momentum on the smooth approximation."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    c = _full_convolve(a, a)
    max_c = float(c.max())
    c_sorted = np.sort(c)[::-1]
    gap = max(0.0, c_sorted[0] - c_sorted[min(20, len(c_sorted)-1)])
    beta = max(gap * 0.1, max_c * 0.01)
    
    momentum = np.zeros(n)
    beta_m = 0.9
    lr = 0.01
    patience = 0
    max_patience = 100
    
    for step in range(100000):
        if time.monotonic() + min_time > deadline:
            break
        
        a_curr = _softmax(z)
        c_curr = _full_convolve(a_curr, a_curr)
        c_max = float(c_curr.max())
        shifted = (c_curr - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s_e = e.sum()
        
        if s_e < 1e-300:
            w = np.zeros_like(c_curr)
            w[int(np.argmax(c_curr))] = 1.0
        else:
            w = e / s_e
        
        a_rev = a_curr[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        W_val = float(np.dot(w, c_curr))
        grad = 4.0 * n * a_curr * (h - W_val)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Adaptive learning rate
        lr = 0.05 / (gnorm / (n * 0.01) + 1.0)
        
        momentum = beta_m * momentum + (1.0 - beta_m) * grad
        z_new = z - lr * momentum
        z_new = np.clip(z_new, -100.0, 100.0)
        z = z_new - z_new.mean()
        
        # Cool beta
        if step % 100 == 0:
            beta *= 0.95
        
        # Check score periodically
        if step % 50 == 0:
            sc = _score(a_curr)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_curr.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros(n)
                    patience = 0
                    beta_m = max(0.5, beta_m * 0.9)
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "two_lobe":
        t = np.linspace(0, 1, n)
        a = np.exp(-20*(t-0.25)**2) + np.exp(-20*(t-0.75)**2) + 0.01
        return a / a.sum()
    elif kind == "three_lobe":
        t = np.linspace(0, 1, n)
        a = np.exp(-15*(t-0.2)**2) + np.exp(-15*(t-0.5)**2) + np.exp(-15*(t-0.8)**2) + 0.01
        return a / a.sum()
    elif kind == "four_lobe":
        t = np.linspace(0, 1, n)
        a = np.exp(-12*(t-0.15)**2) + np.exp(-12*(t-0.4)**2) + np.exp(-12*(t-0.6)**2) + np.exp(-12*(t-0.85)**2) + 0.01
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * t
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS with log-sum-exp, coordinate descent,
    active-set optimization, and multi-length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev-best_score:.2e} t={time.monotonic()-t_start:.1f}s")
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass
    
    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")
    
    # === Phase 1: L-BFGS from parent (500s) ===
    print(f"\n[Phase 1] L-BFGS from parent (t={time.monotonic()-t_start:.1f}s)")
    phase1_dl = t_start + 500
    a1 = _lbfgs_optimize(parent, phase1_dl, min_time=30.0)
    update_best(a1, _score(a1), "lbfgs_parent")
    
    # === Phase 2: Active-set optimization (300s) ===
    print(f"\n[Phase 2] Active-set (t={time.monotonic()-t_start:.1f}s)")
    phase2_dl = min(deadline - 700, phase1_dl + 300)
    if best_seq is not None and time.monotonic() < phase2_dl:
        a2 = _active_set_optimize(best_seq, phase2_dl, min_time=30.0)
        update_best(a2, _score(a2), "active_set")
    
    # === Phase 3: Coordinate descent (300s) ===
    print(f"\n[Phase 3] Coordinate Descent (t={time.monotonic()-t_start:.1f}s)")
    phase3_dl = min(deadline - 500, phase2_dl + 300)
    if best_seq is not None and time.monotonic() < phase3_dl:
        n = len(best_seq)
        # For large n, coordinate descent on all coordinates is too slow
        # Use a subset or the fast version
        if n <= 4096:
            a3 = _coord_descent_fast(best_seq, phase3_dl, min_time=30.0)
        else:
            # Subsampled coordinate descent
            a3 = best_seq.copy()
            a3 = a3 / a3.sum()
            # Only optimize a subset of coordinates per pass
            indices = rng.permutation(n)
            coords_per_pass = min(n, 512)
            for it in range(200):
                if time.monotonic() + 30 > phase3_dl:
                    break
                subset = indices[(it * coords_per_pass) % n: ((it + 1) * coords_per_pass) % n]
                if len(subset) == 0:
                    subset = indices[:coords_per_pass]
                for i in subset:
                    a_i_old = a3[i]
                    a3[i] = 0.0
                    b = a3
                    c_b = _full_convolve(b, b)
                    d = np.zeros(2*n-1)
                    lo = max(0, i)
                    hi = min(2*n, i+n)
                    d[lo:hi] = b[max(0, lo-i):min(n, hi-i)]
                    delta = 1.0 if 2*i < 2*n-1 else 0.0
                    S_b = float(b.sum())
                    
                    def f_cd(x):
                        pk = float(np.max(c_b + 2.0*x*d + x*x*delta))
                        return 2.0*n*pk / ((S_b+x)**2)
                    
                    x_max = max(5.0*a_i_old, 0.01*float(np.max(a3))) + 1e-10
                    try:
                        res = minimize_scalar(f_cd, bounds=(0.0, x_max), method='bounded',
                                              options={'xatol': 1e-10})
                        a3[i] = max(0.0, res.x)
                    except Exception:
                        a3[i] = a_i_old
                sc = _score(a3)
                if sc < best_score:
                    update_best(a3, sc, "coord_descent")
            update_best(a3, _score(a3), "coord_descent_final")
    
    # === Phase 4: Second L-BFGS round (300s) ===
    print(f"\n[Phase 4] L-BFGS Round 2 (t={time.monotonic()-t_start:.1f}s)")
    phase4_dl = min(deadline - 300, phase3_dl + 300)
    if best_seq is not None and time.monotonic() < phase4_dl:
        a4 = _lbfgs_optimize(best_seq, phase4_dl, min_time=30.0)
        update_best(a4, _score(a4), "lbfgs_r2")
    
    # === Phase 5: Multi-length exploration (200s) ===
    print(f"\n[Phase 5] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    phase5_dl = min(deadline - 100, phase4_dl + 200)
    n_tries = [32768, 24576, 8192, 16384]
    for n_try in n_tries:
        if time.monotonic() > phase5_dl - 30:
            break
        if n_try == n_parent and best_seq is not None and len(best_seq) == n_try:
            continue
        a_n = _resample(best_seq, n_try) if best_seq is not None else _make_profile(n_try, "tapered")
        a_n = _lbfgs_optimize(a_n, phase5_dl, min_time=20.0)
        sc_n = _score(a_n)
        update_best(a_n, sc_n, f"n_{n_try}")
    
    # === Phase 6: Final coordinate descent polish (100s) ===
    print(f"\n[Phase 6] Final CD (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        n = len(best_seq)
        if n <= 4096:
            a_fin = _coord_descent_fast(best_seq, deadline - 20, min_time=10.0)
            update_best(a_fin, _score(a_fin), "final_cd")
        else:
            # Quick coordinate descent on random subset
            a_fin = best_seq.copy()
            a_fin = a_fin / a_fin.sum()
            indices = rng.permutation(n)
            for it in range(50):
                if time.monotonic() > deadline - 25:
                    break
                subset = indices[(it * 256) % n: ((it + 1) * 256) % n]
                if len(subset) == 0:
                    subset = indices[:256]
                for i in subset:
                    a_i_old = a_fin[i]
                    a_fin[i] = 0.0
                    b = a_fin
                    c_b = _full_convolve(b, b)
                    d = np.zeros(2*n-1)
                    lo = max(0, i)
                    hi = min(2*n, i+n)
                    d[lo:hi] = b[max(0, lo-i):min(n, hi-i)]
                    delta = 1.0 if 2*i < 2*n-1 else 0.0
                    S_b = float(b.sum())
                    def f_cd2(x):
                        pk = float(np.max(c_b + 2.0*x*d + x*x*delta))
                        return 2.0*n*pk / ((S_b+x)**2)
                    x_max = max(5.0*a_i_old, 0.01*float(np.max(a_fin))) + 1e-10
                    try:
                        res = minimize_scalar(f_cd2, bounds=(0.0, x_max), method='bounded',
                                              options={'xatol': 1e-10})
                        a_fin[i] = max(0.0, res.x)
                    except Exception:
                        a_fin[i] = a_i_old
                sc = _score(a_fin)
                if sc < best_score:
                    update_best(a_fin, sc, "final_cd_sub")
            update_best(a_fin, _score(a_fin), "final_cd_subset")
    
    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
