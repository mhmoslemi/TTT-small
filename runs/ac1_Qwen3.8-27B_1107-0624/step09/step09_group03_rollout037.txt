"""
Optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Aggressive L-BFGS log-sum-exp continuation with extended beta range
2. Subgradient Nesterov acceleration for final refinement
3. Active-set Newton with better convergence
4. Coordinate descent polish
5. Multi-length exploration (n=16384, 32768)
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _corr_a_A(a, A_vec, n):
    """Compute correlation of a with A_vec, restricted to valid range."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]

def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)

def _nesterov_subgrad(a_init, deadline, max_iter=50000):
    """
    Nesterov accelerated subgradient method for min max(b_i) over simplex.
    b_i = (a*a)_i, we minimize max_i b_i subject to a >= 0, sum(a) = 1.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    
    # Compute Lipschitz constant estimate
    # ||∇(a*a)_i||_inf = 2 * max(a) for each i
    # So L for max is 2 * max(a)
    
    a_prev = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_sc = _score(a)
    
    for k in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Nesterov interpolation
        t_prev = t_k
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev)) / 2.0
        theta = (t_prev - 1.0) / t_k
        y = a + theta * (a - a_prev)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a
        else:
            y = y / s_y
        
        # Compute convolution and subgradient
        c = _full_convolve(y, y)
        c_max = float(c.max())
        
        # Subgradient: weighted average of active gradients
        # Use log-sum-exp with small temperature for smoothing
        beta_sub = c_max * 1e-6
        shifted = (c - c_max) / beta_sub
        e = np.exp(np.clip(shifted, -700, 0))
        w = e / e.sum()
        
        # Gradient of sum w_i * b_i w.r.t. a
        # b_i = sum_j a_j * a_{i-j}
        # d(b_i)/d(a_m) = a_{i-m} + a_{m-i} (for valid indices)
        # So grad = 2 * corr(a, w) (using the appropriate correlation)
        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        g = 2.0 * wc[n - 1:2 * n - 1]
        
        # Projected subgradient step
        L = 2.0 * float(y.max()) + 1e-10
        step = 1.0 / (L * (k + 1))
        a_new = y - step * g
        a_new = _project_simplex(a_new)
        
        a_prev = a
        a = a_new
        
        # Track best (using actual score, not the smoothed one)
        # Do this less frequently to save time
        if k % 100 == 0 or k < 10:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    # Final score
    sc = _score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    
    return best_a, best_sc

"""

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _ls_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _ls_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _nesterov_subgrad(a_init, deadline, max_iter=30000):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()

        return a, _score(a)
    a = a / s

    a_prev = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_sc = _score(a)
    f_best = float(c_max) if False else _score(a)

    for k in range(max_iter):
        if time.monotonic() > deadline:

        t_prev = t_k

        theta = (t_prev - 1.0) / t_k
        y = a + theta * (a - a_prev)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a / a.sum()
        else:

        beta_sub = max(c_max * 1e-5, 1e-30)
        shifted = (c - c_max) / beta_sub
        e = np.exp(np.clip(shifted, -700, 0))
        w = e / e.sum()

        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        g = 2.0 * wc[n - 1:2 * n - 1]

        L = 2.0 * float(y.max()) + 1e-12

def _coord_descent(a_init, deadline, max_rounds=20):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))

        # Find coordinates that most affect the peak
        a_rev = a[::-1]
        e_peak = np.zeros(2 * n - 1)
        e_peak[peak_idx] = 1.0
        grad_peak = 2.0 * _full_convolve(e_peak, a_rev)[n - 1:2 * n - 1]

        # Sort by gradient magnitude, try top coordinates
        grad_abs = np.abs(grad_peak)
        top_k = min(100, n)
        top_coords = np.argsort(grad_abs)[::-1][:top_k]

        improved = False
        for i in top_coords:
            if time.monotonic() > deadline:
                break
            if a[i] < 1e-15:
                continue

            # 1D minimization: vary a[i] while keeping others fixed
            # Normalize: a[i] new = a[i] * t, then rescale
            best_t = 1.0
            best_local = best_sc
            for t_try in [0.5, 0.7, 0.85, 0.95, 1.05, 1.1, 1.3]:
                a_try = a.copy()
                a_try[i] = a[i] * t_try
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_local:
                    best_local = sc_try
                    best_t = t_try

            if best_t != 1.0 and best_local < best_sc - 1e-14:
                a[i] = a[i] * best_t
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                best_sc = best_local
                best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_sc


def _lbfgs_continuation(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 100:
        gap = c_sorted[0] - c_sorted[100]
    else:
        gap = c_sorted[0] - c_sorted[-1]
    base_beta = max(gap * 0.5, max_c * 0.001)

    # Extended schedule: 35 levels down to max_c * 1e-12
    n_betas = 35
    log_base = np.log(base_beta)
    log_min = max(np.log(max_c * 1e-12), -50.0)
    if log_min >= log_base:
        log_min = log_base - 5.0
    log_betas = np.linspace(log_base, log_min, n_betas)
    betas = np.exp(log_betas)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.004 * max(1.0, n / 8192.0)
        # More iterations at smaller beta
        if beta < max_c * 1e-6:
            max_iter = min(8000, max(100, int(remaining / (est_time * 8))))
        else:
            max_iter = min(5000, max(50, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _ls_obj,
                z,
                args=(beta, n),
                jac=_ls_grad,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-15,
                    'maxcor': 200,
                    'maxls': 500,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-18:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            if i % 5 == 0:
                print(f"    L-BFGS beta={beta:.2e} (level {i}/{n_betas}): score={current_sc:.12f}")
        except Exception as ex:
            print(f"    L-BFGS beta={beta:.2e}: exception {ex}")

    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _active_set_newton(a_init, deadline, max_outer=50):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Try multiple tolerance levels
        tol_list = [0.01, 0.005, 0.002, 0.001, 0.0005]
        best_improvement = 0
        best_a_new = None

        for tol_frac in tol_list:
            active = c >= (c_max - tol_frac * c_max)
            if not active.any():
                active[np.argmax(c)] = True
            m = int(active.sum())
            if m < 2 or m > 5 * n:
                continue

            A_vec = active.astype(np.float64) / m
            g = 2.0 * _corr_a_A(a, A_vec, n)
            g_t = g - np.mean(g)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                continue

            mu = 1e-6 / m + 1e-10
            A_vec_copy = A_vec
            n_copy = n
            mu_copy = mu

            # Conjugate gradient solve
            def hess_vp(v):
                return 2.0 * _corr_a_A(v, A_vec_copy, n_copy) + mu_copy * v

            # CG solve for H^{-1} g_t
            x = np.zeros(n)
            r = g_t.copy()
            p = r.copy()
            rs = np.dot(r, r)
            for _ in range(200):
                Ap = hess_vp(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x = x + alpha * p
                r = r - alpha * Ap
                rs_new = np.dot(r, r)
                if np.sqrt(rs_new) < 1e-12:
                    break
                p = r + (rs_new / rs) * p
                rs = rs_new

            # Solve H^{-1} 1 for normalization
            x2 = np.zeros(n)
            r2 = np.ones(n)
            p2 = r2.copy()
            rs2 = np.dot(r2, r2)

                Ap2 = hess_vp(p2)
                pAp2 = np.dot(p2, Ap2)
                if pAp2 < 1e-30:
                    break
                alpha2 = rs2 / pAp2
                x2 = x2 + alpha2 * p2
                r2 = r2 - alpha2 * Ap2
                rs2_new = np.dot(r2, r2)
                if np.sqrt(rs2_new) < 1e-12:
                    break
                p2 = r2 + (rs2_new / rs2) * p2
                rs2 = rs2_new

            w_sum = float(x2.sum())
            if abs(w_sum) < 1e-15:
                dx = -x
            else:
                lam = float(x.sum()) / w_sum
                dx = -x + lam * x2

            dx = dx - np.mean(dx)

            # Line search
            t = 1.0
            for _ in range(40):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < c_max - 1e-14:
                    imp = c_max - f_try
                    if imp > best_improvement:
                        best_improvement = imp
                        best_a_new = a_try.copy()
                    break
                t *= 0.5

        if best_a_new is None:
            break

        a = best_a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc

    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        as_dl = min(deadline - 500, time.monotonic() + 200)
        a_as, sc_as = _active_set_newton(best_seq, as_dl, max_outer=40)
        update_best(a_as, sc_as, "as_newton")

    # === Phase 3: Nesterov subgradient ===
    print(f"\n[Phase 3] Nesterov subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        nest_dl = min(deadline - 400, time.monotonic() + 300)
        a_nest, sc_nest = _nesterov_subgrad(best_seq, nest_dl, max_iter=25000)
        update_best(a_nest, sc_nest, "nesterov")

    # === Phase 4: Coordinate descent polish ===
    print(f"\n[Phase 4] Coordinate descent (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        cd_dl = min(deadline - 300, time.monotonic() + 150)
        a_cd, sc_cd = _coord_descent(best_seq, cd_dl, max_rounds=30)
        update_best(a_cd, sc_cd, "coord_desc")

    # === Phase 5: L-BFGS polish with very small beta ===
    print(f"\n[Phase 5] L-BFGS final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        lbfgs2_dl = min(deadline - 200, time.monotonic() + 150)
        a_pol = _lbfgs_continuation(best_seq, lbfgs2_dl, min_time_remaining=200.0)
        update_best(a_pol, _score(a_pol), "lbfgs_polish")

    # === Phase 6: Try n=32768 ===
    print(f"\n[Phase 6] Try n=32768 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 150 and best_seq is not None:
        n_large = 32768
        a_large = _interpolate_seq(best_seq, n_large)
        sc_large_init = _score(a_large)
        print(f"  n={n_large} init score: {sc_large_init:.12f}")
        update_best(a_large, sc_large_init, "n32768_init")

        lbfgs3_dl = min(deadline - 100, time.monotonic() + 100)
        a_large_opt = _lbfgs_continuation(a_large, lbfgs3_dl, min_time_remaining=100.0)
        update_best(a_large_opt, _score(a_large_opt), "n32768_opt")

    # === Phase 7: Diversified restarts ===
    print(f"\n[Phase 7] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        for sigma, label in [(0.003, "tiny"), (0.01, "small"), (0.03, "med")]:
            if time.monotonic() > deadline - 40:
                break
            p_start = _perturb(best_seq, sigma, rng)
            rest_dl = min(deadline - 30, time.monotonic() + 60)
            a_r = _lbfgs_continuation(p_start, rest_dl, min_time_remaining=30.0)
            update_best(a_r, _score(a_r), f"restart_{label}")

    # === Phase 8: Final refinement cycle ===
    print(f"\n[Phase 8] Final refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        # Quick Newton
        if time.monotonic() < deadline - 25:
            as_final = min(deadline - 15, time.monotonic() + 30)
            a_fn, sc_fn = _active_set_newton(best_seq, as_final, max_outer=20)
            update_best(a_fn, sc_fn, "final_newton")

        # Quick L-BFGS with tiny beta
        if time.monotonic() < deadline - 12:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 1e-9
            remaining = deadline - 12 - time.monotonic()
            if remaining > 3:
                max_iter_fin = min(3000, max(100, int(remaining / 0.05)))
                try:
                    result = minimize(
                        _ls_obj, z, args=(beta_final, n),
                        jac=_ls_grad, method='L-BFGS-B',
                        options={'maxiter': max_iter_fin, 'ftol': 1e-20,
                                 'gtol': 1e-16, 'maxcor': 200, 'maxls': 500}
                    )
                    a_fin = _softmax(result.x)
                    update_best(a_fin, _score(a_fin), "final_tiny_beta")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
