def _kkt_solve(a, T, S, n):
    """Solve the KKT linear system given active set S and support T."""
    w = np.zeros(2*n-1)
    w[S] = 1.0 / len(S)
    
    T_idx = np.where(T)[0]
    n_T = len(T_idx)
    
    # Build W_T: W_T[i, j] = w[T_idx[j] + T_idx[i]]
    W_T = np.zeros((n_T, n_T))
    for i, j in enumerate(T_idx):
        for m, tm in enumerate(T_idx):
            idx = tm + j
            if 0 <= idx < 2*n-1:
                W_T[i, m] = w[idx]
    
    # System: [2*W_T, 1; 1^T, 0] [a_T; lambda] = [lambda*1; 1]
    # i.e., 2*W_T * a_T - 1*lambda = 0
    #        1^T * a_T = 1
    
    A = np.zeros((n_T + 1, n_T + 1))
    A[:n_T, :n_T] = 2 * W_T
    A[:n_T, n_T] = -1.0
    A[n_T, :n_T] = 1.0
    
    b = np.zeros(n_T + 1)
    b[n_T] = 1.0
    
    sol = np.linalg.solve(A, b)
    a_T = sol[:n_T]
    lambda_val = sol[n_T]
    
    # Check non-negativity
    if np.min(a_T) < -1e-10:
        return None, None
    
    # Construct full a
    a_new = np.zeros(n)
    a_new[T_idx] = np.maximum(a_T, 0)
    a_new /= a_new.sum()
    
    return a_new, lambda_val

def _kkt_iteration(a, n, S, tol=0.01):
    """One KKT iteration: given active set S, update a to satisfy KKT."""
    w = np.zeros(2*n-1)
    S_idx = np.where(S)[0]
    w[S_idx] = 1.0 / len(S_idx)
    
    # Solve W * u = 1 using CG with regularization
    # W[i,j] = w[i+j]
    mu = 0.01  # regularization
    u = _cg_solve_hankel(w, np.ones(n), n, mu, max_iter=500)
    
    # a = u / (1^T u)
    s = u.sum()
    if s < 1e-10 or np.min(u) < -1e-6:
        return None  # Regularization failed, fall back
    
    a_new = u / s
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    return a_new

def _cg_solve_hankel(w, b, n, mu, max_iter=500, tol=1e-10):
    """Solve (W + mu*I) x = b where W[i,j] = w[i+j] using CG."""
    def matvec(v):
        # W * v
        v_flip = v[::-1]
        c = _full_convolve(w, v_flip)
        Wv = c[n-1 : 2*n-1]
        return Wv + mu * v
    
    x = np.zeros(n)
    r = b.copy()  # b - A*x = b - 0 = b
    p = r.copy()
    rs = r @ r
    for _ in range(max_iter):
        Ap = matvec(p)
        pAp = p @ Ap
        if abs(pAp) < 1e-30:
            break
        alpha = rs / pAp
        x += alpha * p
        r -= alpha * Ap
        rs_new = r @ r
        if np.sqrt(rs_new) < tol * np.sqrt(b @ b):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

    """Optimize by reducing KKT violations."""
    best_a = a.copy()
    best_sc = _score(a)
    
    eta = 0.1  # step size
    tol_frac = 0.005  # active set tolerance
    
    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        if sc < best_sc - 1e-12:
            best_sc, best_a = sc, a.copy()
        
        # Active set
        active = c >= c_max - tol_frac * c_max
        n_act = max(1, int(active.sum()))
        
        # Gradient (subgradient)
        w = np.zeros(2*n-1)
        w[active] = 1.0 / n_act
        g = 2.0 * _hankel_product(w, a, n)  # g[j] = 2 * sum_m a[m] * w[m+j]
        
        # Violation on support
        support = a > 1e-10
        if not np.any(support):
            break
        lam = float(np.mean(g[support]))
        r = g - lam
        r[~support] = 0.0
        
        # Gradient of violation
        # grad_V[m] = 4 * sum_{j in support} r[j] * w[m+j] = 4 * (W * r)[m]
        grad_V = 4.0 * _hankel_product(w, r, n)
        
        # Center grad_V (to maintain sum(a) = 1)
        grad_V -= np.mean(grad_V[support]) * support
        
        # Step
        step = eta * grad_V
        a_new = a - step
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        # Check improvement
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())
        
        if sc_new >= sc:
            # Try smaller step
            improved = False
            for factor in [0.5, 0.25, 0.1]:
                a_try = a - factor * eta * grad_V
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < sc - 1e-12:
                    a_new = a_try
                    improved = True
                    break
            if not improved:

                eta *= 0.5
                if eta < 1e-6:
                    # Try adjusting active set
                    tol_frac = max(0.001, tol_frac * 0.8)
                break
        else:
            a = a_new
            grad_V = grad_V.copy()  # ensure we have the right variable
            # Decay step size slowly
            eta = max(0.01, eta * 0.999)
            # Tighten active set tolerance

"""
Multi-phase optimizer combining L-BFGS, KKT violation reduction,
and active-set refinement for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations: FFT-based gradient for KKT violations, adaptive
active set tightening, momentum-accelerated iterations.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _hankel_mv(w, v, n):
    """Compute W*v where W[i,j] = w[i+j] for i,j in [0,n-1].
    w has length 2n-1. Uses FFT."""
    v_flip = v[::-1]
    c = _full_convolve(w, v_flip)
    return c[n-1:2*n-1]

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)

def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    grad_a = 2.0 * _hankel_mv(w, a, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)

        self.n = n
    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)
    def grad(self, z):
        return _smograd(z, self.beta, self.n)

def _make_profile(n, kind="tapered"):

def _resample(a, new_n):

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
