"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses spectral flattening initialization, adaptive log-sum-exp L-BFGS,
subgradient with BB steps, and basin hopping.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _lsobj_grad(z, beta, K, n):
    """Log-sum-exp smoothed objective and gradient. Only top K peaks."""
    z_max = z.max()
    a = np.exp(z - z_max)
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c
    
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    
    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z

    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _lsobj_grad(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _lsobj_grad(z, self.beta, self.K, self.n)[1]

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n].real
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new
    
    return a


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive log-sum-exp smoothing."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a
    
    # Adaptive base beta
    c_sorted = np.sort(c_cur)[::-1]
    n_ranks = min(len(c_sorted), 50)
    gap = c_sorted[0] - c_sorted[n_ranks - 1]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # Schedule: (beta_multiplier, K_fraction)
    # Start with full smoothing, end with sharp
    schedule = [
        (2.0, 1.0), (1.5, 0.8), (1.0, 0.5), (0.7, 0.3),
        (0.5, 0.2), (0.3, 0.1), (0.2, 0.06), (0.1, 0.04),
        (0.05, 0.03), (0.03, 0.02), (0.01, 0.01), (0.005, 0.006),
        (0.002, 0.004), (0.001, 0.003), (0.0005, 0.002),
    ]
    
    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        beta = base_beta * beta_mult
        K = max(10, int((2 * n - 1) * K_frac))
        
        est_time = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(10000, max(100, int(remaining / (est_time * 30))))
        
        prob = _LsProb(beta, K, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 150, 'maxls': 50}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                # Update base_beta based on new gap
                c_new = _full_convolve(a_new, a_new)
                mc_new = float(c_new.max())
                cs_new = np.sort(c_new)[::-1]
                gap_new = cs_new[0] - cs_new[min(50, len(cs_new)-1)-1]
                if gap_new < mc_new * 0.001:
                    gap_new = mc_new * 0.001
                base_beta = max(gap_new, mc_new * 0.005)
        except Exception:
            pass
    
    return current_a


def _subgradient_bb(a_init, deadline, min_time_remaining=60.0):
    """Active-set subgradient with BB steps and momentum."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    c = _full_convolve(a, a)
    n_2nm1 = 2 * n - 1
    
    active_frac = 0.02
    momentum = 0.7
    lr = 0.03
    lr_min = 1e-5
    lr_max = 0.15
    stall = 0
    max_stall = 100
    velocity = np.zeros(n)
    
    prev_a = None
    prev_g = None
    
    iteration = 0
    last_lbfgs_time = time.monotonic()
    
    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Select active peaks
        n_active = max(5, min(n_2nm1, int(n_2nm1 * active_frac)))
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_active]
        
        active_vals = c[active_idx]
        w = np.zeros(n_2nm1)
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w[active_idx] = (active_vals - active_vals.min()) / spread + 0.1
        else:
            w[active_idx] = 1.0 / n_active
        w = w / w.sum()
        
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break
        
        # BB step
        if prev_a is not None and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g_centered - prev_g
            s_norm = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm > 1e-18 and abs(sy) > 1e-18:
                lr_bb = s_norm / abs(sy)
                lr = np.clip(lr_bb, lr_min, lr_max)
        
        velocity = momentum * velocity - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        
        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            for ls in range(30):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-15:
                    improved = True
                    break
                velocity = momentum * velocity * 0.5 - lr * 0.5 * g_centered
                a_try = a + velocity
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    velocity = np.zeros(n)
                    a_try = a
                    break
                a_try = a_try / s_try
                lr *= 0.7
        
        if improved:
            prev_a = a.copy()
            prev_g = g_centered.copy()
            a = a_try
            lr = min(lr_max, lr * 1.02)
            active_frac = max(0.003, active_frac * 0.97)
        else:
            velocity = np.zeros(n)
            lr = max(lr_min, lr * 0.65)
            active_frac = min(0.1, active_frac * 1.3)
        
        if stall > max_stall:
            active_frac = min(0.1, active_frac * 2.0)
            lr = max(lr_min, lr * 0.3)
            momentum = min(0.9, momentum + 0.05)
            stall = 0
        
        # Periodic L-BFGS
        if iteration % 300 == 0 and time.monotonic() - last_lbfgs_time > 90:
            if time.monotonic() + min_time_remaining + 60 < deadline:
                a_tmp = _run_lbfgs(a, time.monotonic() + 50, min_time_remaining=5)
                sc_tmp = _score(a_tmp)
                if sc_tmp < best_sc:
                    best_a = a_tmp.copy()
                    best_sc = sc_tmp
                    a = a_tmp.copy()
                    velocity = np.zeros(n)
                    prev_a = None
                    prev_g = None
                last_lbfgs_time = time.monotonic()
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "optimal_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2) + 0.03
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _perturb_freq(a, sigma, rng):
    """Perturb in frequency domain for different types of noise."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    A += sigma * (rng.standard_normal(len(A)) + 1j * rng.standard_normal(len(A)))
    a_new = np.fft.irfft(A, N)[:n].real
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    """Very fine active-set polish."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.003
    tol = 0.003
    no_improve = 0
    n_2nm1 = 2 * n - 1
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        # Active set: peaks within tol of max
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        elif n_active > 200:
            # Limit active set size
            top_idx = np.argsort(c)[::-1][:200]
            active = np.zeros(n_2nm1, dtype=bool)
            active[top_idx] = True
            n_active = 200
        
        w = np.zeros(n_2nm1)
        w[active] = 1.0 / n_active
        
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-16:
            break
        
        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        improved = False
        for ls in range(50):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                tol = max(0.0001, tol * 0.7)
                lr *= 0.7
            if no_improve > 500:
                a = best_a.copy()
                tol = max(0.0001, tol * 0.5)
                lr = 0.002
                no_improve = 0
        else:
            if tol > 0.0001:
                tol = max(0.0001, tol * 0.98)
            lr = min(0.008, lr * 1.005)
    
    return best_a, best_sc


def _coordinate_descent(a_init, deadline, min_time_remaining=10.0):
    """Coordinate descent: adjust pairs of elements to reduce top peaks."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n_2nm1 = 2 * n - 1
    
    patience = 0
    max_patience = 200
    rng = np.random.default_rng(123)
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        
        # Find top peak
        k_peak = int(np.argmax(c))
        
        # Contributions to this peak: c[k_peak] = sum_i a[i]*a[k_peak-i]
        i_lo = max(0, k_peak - (n - 1))
        i_hi = min(n - 1, k_peak)
        
        # Find the pair (i, j) with i+j = k_peak that has the largest product
        # Reduce both slightly, increase others
        best_pair_product = 0
        best_i, best_j = -1, -1
        for i in range(i_lo, i_hi + 1):
            j = k_peak - i
            prod = a[i] * a[j]
            if prod > best_pair_product:
                best_pair_product = prod
                best_i = i
                best_j = j
        
        if best_i < 0:
            break
        
        # Reduce the top contributing pair
        # c[k_peak] decreases by approximately 2*a[i]*delta if we reduce a[i] by delta
        # (when i != j) or a[i]^2 - (a[i]-delta)^2 = 2*a[i]*delta - delta^2 if i == j
        delta = a[best_i] * 0.05
        a_new = a.copy()
        a_new[best_i] -= delta
        if best_j != best_i:
            a_new[best_j] -= delta
        else:
            a_new[best_i] = a[best_i] - delta  # already done above, just correct
        
        # Redistribute: increase elements that don't contribute to this peak
        # (or contribute less)
        # Find elements far from the peak
        mask = np.ones(n, dtype=bool)
        mask[best_i] = False
        if best_j != best_i:
            mask[best_j] = False
        # Also reduce contribution to nearby peaks
        total_reduce = delta * (1 if best_j == best_i else 2)
        
        # Add to random elements
        n_add = max(1, n // 1000)
        candidates = np.where(mask)[0]
        add_idx = rng.choice(candidates, size=min(n_add, len(candidates)), replace=False)
        a_new[add_idx] += total_reduce / len(add_idx)
        
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new
            c_try = _full_convolve(a_new, a_new)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_new
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining spectral flattening, adaptive L-BFGS,
    subgradient with BB steps, coordinate descent, and basin hopping.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Try to load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, tapered n={n_parent}, score={best_score:.10f}")
    
    n = len(best_seq)
    
    # === Phase 1: Spectral flattening from parent ===
    print(f"\n[P1] Spectral flattening (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 1200:
        a_sp = _spectral_flatten(best_seq, n_iter=30, alpha_start=0.4, alpha_end=0.6)
        sc_sp = _score(a_sp)
        print(f"  Spectral flatten score: {sc_sp:.10f}")
        if sc_sp < best_score:
            best_seq = a_sp.copy()
            best_score = sc_sp
            print(f"  NEW BEST: {best_score:.10f} [spectral] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 2: L-BFGS round 1 ===
    print(f"\n[P2] L-BFGS round 1 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 800:
        a1 = _run_lbfgs(best_seq, min(deadline - 600, time.monotonic() + 400),
                        min_time_remaining=600)
        sc1 = _score(a1)
        if sc1 < best_score:
            best_seq = a1.copy()
            best_score = sc1
            print(f"  NEW BEST: {best_score:.10f} [lbfgs1] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 3: Subgradient BB ===
    print(f"\n[P3] Subgradient BB (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a2, sc2 = _subgradient_bb(best_seq, min(deadline - 350, time.monotonic() + 300),
                                  min_time_remaining=60)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [subgrad] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 4: L-BFGS round 2 ===
    print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        a4 = _run_lbfgs(best_seq, min(deadline - 200, time.monotonic() + 200),
                        min_time_remaining=40)
        sc4 = _score(a4)
        if sc4 < best_score:

            best_score = sc4

        a5, sc5 = _coordinate_descent(best_seq, min(deadline - 150, time.monotonic() + 150),
                                      min_time_remaining=20)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [coord] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 6: Fine polish ===
    print(f"\n[P6] Fine polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        a6, sc6 = _fine_polish(best_seq,
