def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = scipy.fft.rfft(a, N)
    fb = scipy.fft.rfft(b, N)
    return scipy.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
