"""
Multi-phase optimizer using power-mean smoothing, LP-subgradient, and L-BFGS.
Key innovations:
- Power-mean (p-norm) objective for cleaner gradients than log-sum-exp
- LP-linearized subgradient with Nesterov acceleration
- Multi-scale and multi-n exploration
- Adaptive p-schedule in L-BFGS
- Hierarchical active-set refinement
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 2
    while k < n:
        k <<= 1
    return k


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _pm_score_grad(a, p):
    """Power-mean objective and gradient.
    Objective: (sum_k c[k]^p)^(1/p) where c = conv(a,a)
    We actually minimize (sum c^p) for computational simplicity (monotone transform).
    Returns (obj, grad) where obj = sum_k c[k]^p, grad = d(obj)/da
    """
    n = len(a)
    c = _full_convolve(a, a)
    # Clip to avoid numerical issues
    c = np.maximum(c, 1e-30)
    obj = float(np.sum(c ** p))
    # Gradient: d/d(a[i]) sum_k c[k]^p = p * sum_k c[k]^(p-1) * dc[k]/da[i]
    # dc[k]/da[i] = a[k-i] + a[i-k] (with boundary)
    # So grad[i] = p * (conv(c^(p-1), a_rev))[i] (with appropriate slicing)
    cp1 = c ** (p - 1)
    a_rev = a[::-1]
    wc = _full_convolve(cp1, a_rev)
    grad = p * wc[n - 1:2 * n - 1]
    return obj, grad


class _PMProblem:
    def __init__(self, p, n):
        self.p = p
        self.n = n
    
    def __call__(self, z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        obj, _ = _pm_score_grad(a, self.p)
        return obj
    
    def grad(self, z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        _, ga = _pm_score_grad(a, self.p)
        # Chain rule: d/dz = da/dz * da/dz
        # a = exp(z - z.max()) / sum(exp(z - z.max()))
        # In log-space, we parameterize as z, and a = softmax(z) (approximately, if z is centered)
        # Actually, let's use the proper chain rule.
        # Let u = z - z.max(), a = exp(u)/sum(exp(u))
        # d(a[i])/d(z[j]) = a[i]*(delta_ij - a[j])
        # So grad_z[i] = sum_j d(a[i])/d(z[j]) * grad_a[j] ... no
        # grad_z[i] = sum_j d(obj)/d(a[j]) * d(a[j])/d(z[i])
        #           = sum_j ga[j] * a[j] * (delta_ji - a[i])
        #           = a[i] * ga[i] - a[i] * sum_j ga[j] * a[j]
        #           = a[i] * (ga[i] - sum_j a[j]*ga[j])
        W = float(np.dot(a, ga))
        grad_z = a * (ga - W)
        return grad_z


def _pm_lbfgs(a_init, deadline, p_start=10, p_end=200, min_time_remaining=60):
    """L-BFGS with power-mean, annealing p from p_start to p_end."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()
    
    # Determine number of stages
    n_stages = 8
    p_schedule = np.geomspace(p_start, p_end, n_stages)
    
    for p in p_schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Initial step size estimate
        est_eval_time = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(30, int(remaining / (est_eval_time * 15))))
        
        z = np.log(np.maximum(current_a, 1e-30))
        z = z - z.mean()
        
        prob = _PMProblem(int(p), n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 50}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
        except Exception:
            pass
    
    return current_a


def _accelerated_subgrad(a_init, deadline, min_time_remaining=60):
    """Accelerated (Nesterov) subgradient on the linearized problem."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    # Nesterov momentum
    y = a.copy()
    velocity = np.zeros(n)
    t_k = 1.0
    lr = 0.01
    lr_decay = 0.9999
    n_iter = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        n_iter += 1
        # Compute gradient at y
        c = _full_convolve(y, y)
        c_max = float(c.max())
        k_max = int(np.argmax(c))
        
        # Subgradient of max(conv(a,a)) at a=y:
        # d/dc[k_max]: dc[k_max]/da[i] = y[k_max-i] + y[i-k_max] (with bounds)
        # This is conv(e_{k_max}, y_rev) sliced appropriately
        # Actually: dc[k]/da[i] = y[k-i] if 0<=k-i<n, else 0
        # So grad[i] = y[k_max - i] if 0 <= k_max - i < n, plus y[i - k_max] if 0 <= i - k_max < n
        # Wait, c[k] = sum_j a[j] * a[k-j]
        # dc[k]/da[i] = a[k-i] (from first factor) + a[i-k] (from second factor, if i != k-j for some j)
        # Actually: c[k] = sum_j a[j]*a[k-j]
        # d/dc[k]/da[i]: if i appears as j, contributes a[k-i]; if i appears as k-j (i.e., j=k-i), contributes a[i]
        # So dc[k]/da[i] = a[k-i] + a[i-k] = 2*a[k-i] (if k-i is in range, and i != k-i)
        # Wait, more carefully:
        # c[k] = sum_{j=0}^{n-1} a[j] * a[k-j] (where 0 <= k-j <= n-1)
        # d(c[k])/d(a[i]) = sum_j [delta_{j,i} * a[k-j] + a[j] * delta_{k-j,i}]
        #                  = a[k-i] + a[i-k] (if both indices are valid)
        # If k-i == i (i.e., i = k/2), then it's 2*a[i]... no wait
        # If k-i is valid (0 <= k-i <= n-1), first term gives a[k-i]
        # If i-k is valid (0 <= i-k <= n-1), second term gives a[i-k]
        # Note k-i and i-k can't both be valid unless... k-i >= 0 means i <= k, and i-k >= 0 means i >= k
        # So both are valid only when i = k (giving a[0] + a[0] = 2*a[0])
        # Actually no: k-i valid means 0 <= k-i <= n-1, i.e., k-(n-1) <= i <= k
        # i-k valid means 0 <= i-k <= n-1, i.e., k <= i <= k+(n-1)
        # Both valid: max(0, k-(n-1)) <= i <= min(k, k+(n-1)) AND k <= i... 
        # Hmm, this is getting complicated. Let me just use the FFT approach.
        
        # Use the FFT-based gradient computation
        w = np.zeros(2 * n - 1)
        w[k_max] = 1.0
        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        g = 2.0 * wc[n - 1:2 * n - 1]
        
        # Center the gradient (remove mean to maintain sum=1 constraint)
        g_centered = g - np.mean(g)
        
        # Step size (decreasing)
        step = lr / (1.0 + t_k)
        
        # Nesterov step
        velocity = 0.9 * velocity - step * g_centered
        a_new = y + velocity
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = best_a.copy()
            y = a_new.copy()
        else:
            a_new = a_new / s_new
        
        # Update momentum parameter
        t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        
        # Check improvement
        sc_new = 2.0 * n * float(_full_convolve(a_new, a_new).max())
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
        
        # Update y (Nesterov lookahead)
        beta = (t_k - 1.0) / (t_k + 1.0)
        y = a_new + beta * (a_new - y)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-10:
            y = y / s_y
        
        # Decay learning rate slowly
        lr *= lr_decay
    
    return best_a, best_sc


def _lp_subgradient(a_init, deadline, min_time_remaining=60):
    """LP-linearized subgradient: at each step, solve min t s.t. conv(a_old, a) <= t, a>=0, sum(a)=1
    Using projected subgradient on the equivalent problem."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    prev_a = None
    prev_g = None
    lr = 0.05
    momentum = 0.5
    velocity = np.zeros(n)
    stall = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        # Current "old" a for linearization
        a_old = a.copy()
        
        # Compute c = conv(a_old, a) - this is the current max we're trying to reduce
        c = _full_convolve(a_old, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Find active peaks (top few)
        n_active = max(3, int((2*n-1) * 0.005))
        top_idx = np.argpartition(c, -n_active)[-n_active:]
        c_top = c[top_idx]
        
        # Weighted subgradient
        w = np.zeros(2*n-1)
        spread = c_top.max() - c_top.min()
        if spread > 1e-18:
            w[top_idx] = (c_top - c_top.min()) / spread + 0.1
        else:
            w[top_idx] = 1.0 / n_active
        w = w / w.sum()
        
        # Gradient of sum_k w[k] * conv(a_old, a)[k] w.r.t. a
        # = conv(a_old_rev, w) sliced
        a_old_rev = a_old[::-1]
        wc = _full_convolve(a_old_rev, w)
        g = 2.0 * wc[n-1:2*n-1]
        # Wait, this isn't quite right. Let me reconsider.
        # conv(a_old, a)[k] = sum_i a_old[k-i]*a[i]
        # d/d(a[j]) conv(a_old,a)[k] = a_old[k-j] (if valid)
        # So d/d(a[j]) sum_k w[k]*conv(a_old,a)[k] = sum_k w[k]*a_old[k-j]
        # This is (conv_rev(w, a_old))[j] = sum_k w[k]*a_old[k-j]
        # In terms of standard convolution: sum_k w[k]*a_old[k-j] = (w * a_old)[j] reversed...
        # Actually: (w * a_old)[m] = sum_k w[k]*a_old[m-k]
        # We want sum_k w[k]*a_old[k-j] = sum_k w[k]*a_old[-(j-k)] 
        # Let m = -j: sum_k w[k]*a_old[m-k] = (w*a_old)[m]... hmm
        # Let me just use: grad[j] = sum_k w[k] * a_old[k-j] (where 0<=k-j<n)
        # This is the cross-correlation of w and a_old, which is:
        # (w * a_old_rev)[j + n - 1]... 
        # Actually, let's just compute it directly with FFT:
        # grad = ifft(fft(w) * conj(fft(a_old, N)))... no
        # 
        # Simplest: use the same _full_convolve trick
        # sum_k w[k] * a_old[k-j] for each j
        # = for j=0: sum_k w[k]*a_old[k] = (w * a_old) at some offset
        # This is actually: flip(w) * a_old, sliced
        # flip(w) * a_old = conv(w[::-1], a_old)
        # conv(w[::-1], a_old)[j] = sum_k w[::-1][k]*a_old[j-k] = sum_k w[len(w)-1-k]*a_old[j-k]
        # Let k' = len(w)-1-k: = sum_k' w[k']*a_old[j-len(w)+1+k']
        # Hmm, this is getting confusing. Let me just use the direct formula.
        #
        # Actually, the simplest way: 
        # d/d(a[j]) sum_k w[k] * conv(a_old, a)[k]
        # conv(a_old, a)[k] = sum_i a_old[k-i] * a[i]
        # d/d(a[j]) = sum_k w[k] * a_old[k-j] (where 0 <= k-j <= n-1)
        # Let me define h[j] = sum_k w[k] * a_old[k-j]
        # This is the cross-correlation: h = correlate(w, a_old)
        # In numpy: np.correlate(w, a_old, mode='full') gives the full cross-correlation
        # correlate(w, a_old)[m] = sum_n w[n+m] * a_old[n]
        # We want h[j] = sum_k w[k] * a_old[k-j]
        # Let k = n+j: h[j] = sum_n w[n+j] * a_old[n] = correlate(w, a_old) at offset j
        # So h = np.correlate(w, a_old, mode='full')[n-1:n-1+n]... 
        # Actually, np.correlate(w, a_old, 'full')[m] = sum_n w[n+m-(len(a_old)-1)] * a_old[n]
        # This is getting too complicated. Let me just use the FFT approach that I know works.
        
        # Use the gradient formula I already have:
        # For c = conv(a, b), dc/d(a[i]) = 2*conv(c_w, a_rev)[i] where c_w is the weight
        # Wait no, that formula is for c = conv(a, a). For c = conv(a_old, a):
        # dc[k]/da[j] = a_old[k-j]
        # sum_k w[k] * dc[k]/da[j] = sum_k w[k] * a_old[k-j]
        # This is exactly: (w * flip(a_old)) shifted... 
        #
        # OK let me just compute it with a direct loop for small arrays or use the trick:
        # sum_k w[k] * a_old[k-j] = conv(w, a_old_shifted)[j] where a_old_shifted[i] = a_old[-i]... 
        #
        # Actually the cleanest way: 
        # Define v[j] = a_old[j] (length n)
        # Define w[k] (length 2n-1)
        # We want h[j] = sum_k w[k]*v[k-j] for j=0,...,n-1
        # This is (v * w) at position... let's see
        # (v*w)[m] = sum_k v[k]*w[m-k]
        # h[j] = sum_k w[k]*v[k-j] 
        # Let l = k-j: h[j] = sum_l w[l+j]*v[l] = (w_shifted * v)[0]... no
        # h[j] = sum_l v[l]*w[l+j]
        # (w*v)[m] = sum_l w[l]*v[m-l]
        # Let m-l = l' => l = m-l': (w*v)[m] = sum_l' w[m-l']*v[l']
        # h[j] = sum_l v[l]*w[l+j] = sum_l w[l+j]*v[l]
        # Compare with (w*v)[m]: sum_l' w[m-l']*v[l']
        # If we set m-l' = l+j, i.e., l' = m-l-j... this doesn't simplify nicely.
        #
        # Let me just use: h[j] = sum_k w[k]*a_old[k-j]
        # Extend a_old to length 2n-1 (zero-padded)
        # Then h = correlate(w, a_old_extended) at appropriate positions
        # 
        # You know what, let me just use the same _grad_maxconv function with a trick:
        # _grad_maxconv(a, w) computes 2*conv(w, a_rev)[n-1:2n-1]
        # This gives: 2 * sum_k w[k] * a_rev[k-(n-1)] for each position
        # a_rev[k-(n-1)] = a[n-1-k+(n-1)] = a[2n-2-k]
        # So _grad_maxconv(a, w)[j] = 2 * sum_k w[k] * a[2n-2-k] ... that's not what I want.
        #
        # Let me just compute it directly using FFT:
        # h[j] = sum_k w[k] * a_old[k-j] for j = 0,...,n-1
        # This is the cross-correlation of w and a_old.
        # Using FFT: h = ifft(FFT(w) * conj(FFT(a_old))) ... but with appropriate zero-padding.
        
        # Actually, I'll just use a simpler approach:
        # h[j] = sum_k w[k] * a_old[k-j]
        # = sum_k w[k] * a_old_reversed[j-k+n-1] ... no
        #
        # Let me define: a_ext = a_old padded to length 2n-1 (zeros on the right)
        # Then h[j] = sum_{k=j}^{j+n-1} w[k] * a_old[k-j]
        # This is (w * a_old)[j] in the full convolution if we pad properly.
        #
        # FULL convolution: (w * a_old)[m] = sum_k w[k] * a_old[m-k]
        # We want h[j] = sum_k w[k] * a_old[k-j]
        # = sum_k w[k] * a_old[-(j-k)]
        # If we define a_neg[i] = a_old[-i] (reversed and time-indexed from -n+1 to 0):
        # a_neg[i] = a_old[-i] for -n+1 <= i <= 0
        # h[j] = sum_k w[k] * a_neg[j-k]
        # = (w * a_neg)[j] in the full convolution
        #
        # This is getting too complicated. Let me just use the approach that works:
        # The gradient of f(a) = max_k conv(a, a)[k] at point a is:
        # g[i] = 2 * sum_{k in active} w_k * a[k-i] (where 0 <= k-i < n)
        # For the LINEARIZED version (conv(a_old, a)):
        # g[i] = sum_k w_k * a_old[k-i] (where 0 <= k-i < n)
        #
        # I'll compute this using the same trick as _grad_maxconv but adapted:
        # _grad_maxconv computes 2*conv(w, a_rev)[n-1:2n-1]
        # conv(w, a_rev)[m] = sum_k w[k]*a_rev[m-k] = sum_k w[k]*a[n-1-(m-k)] = sum_k w[k]*a[n-1-m+k]
        # For m in [n-1, 2n-2]: let j = m-(n-1), so m = n-1+j
        # conv(w, a_rev)[n-1+j] = sum_k w[k]*a[n-1-(n-1+j)+k] = sum_k w[k]*a[k-j]
        # 
        # So conv(w, a_rev)[n-1+j] = sum_k w[k]*a[k-j] = h[j] (exactly what I want!)
        # 
        # So for the linearized version:
        # g[i] = conv(w, a_old_rev)[n-1+i] where a_old_rev = a_old[::-1]
        # And we DON'T multiply by 2 (since it's not a quadratic, just linear in a)
        
        a_old_rev = a_old[::-1]
        wc = _full_convolve(w, a_old_rev)
        g = wc[n-1:2*n-1]  # No factor of 2!
        
        g_centered = g - np.mean(g)
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break
        
        # BB step size
        if prev_a is not None and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g_centered - prev_g
            s_norm = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm > 1e-18 and abs(sy) > 1e-18:
                lr_bb = s_norm / abs(sy)
                lr = np.clip(0.5 * lr + 0.5 * lr_bb, 1e-5, 0.1)
        
        # Momentum step
        velocity = momentum * velocity - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        
        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            # Check if this reduces the linearized objective
            c_try = _full_convolve(a_old, a_try)
            sc_try = float(c_try.max())
            if sc_try < c_max - 1e-15:
                a = a_try
                improved = True
            else:
                # Halve
                lr *= 0.5
                velocity = momentum * velocity * 0.5 - lr * g_centered
                a_try = a + velocity
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_old, a_try)
                    if float(c_try.max()) < c_max - 1e-15:
                        a = a_try
                        improved = True
        
        if improved:
            prev_a = a.copy()
            prev_g = g_centered.copy()
            lr = min(0.1, lr * 1.05)
        else:
            velocity = np.zeros(n)
            lr = max(1e-5, lr * 0.5)
            if stall > 100:
                # Reset to best and try different active set
                a = best_a.copy()
                n_active = min(2*n-1, n_active * 2)
                stall = 0
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2) + 0.03
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "optimal_taper":
        t = np.linspace(0, 1, n)
        # Optimal taper: minimize the edge contribution to max convolution
        # c[0] = a[0]^2, so we want small edges
        # But also c[n-1] = sum a[i]^2, which we want small
        # Trade-off: not too concentrated, not too spread
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _fine_active_set(a_init, deadline, min_time_remaining=5):
    """Very fine active-set subgradient with adaptive tolerance."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    tol = 0.01
    lr = 0.01
    no_improve = 0
    prev_a = None
    prev_g = None
    velocity = np.zeros(n)
    momentum = 0.5
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        # Active set
        active = c >= (c_max - tol * c_max)
        n_active = max(1, int(active.sum()))
        
        w = np.zeros(2 * n - 1)
        active_vals = c[active]
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w[active] = (active_vals - active_vals.min()) / spread + 0.05
        else:
            w[active] = 1.0 / n_active
        w = w / w.sum()
        
        # Gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        g = 2.0 * wc[n-1:2*n-1]
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break
        
        # BB step
        if prev_a is not None and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g_centered - prev_g
            sn = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if sn > 1e-18 and abs(sy) > 1e-18:
                lr_bb = sn / sy if sy > 0 else lr
                lr = np.clip(lr_bb, 1e-6, 0.05)
        
        velocity = momentum * velocity - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        
        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                prev_a = prev_a  # will update below
            else:
                # Backtrack
                for _ in range(10):
                    lr *= 0.5
                    velocity = momentum * velocity * 0.5 - lr * g_centered
                    a_try = a + velocity
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-10:
                        break
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    sc_try = 2.0 * n * float(c_try.max())
                    if sc_try < current_sc - 1e-15:
                        a = a_try
                        improved = True
                        break
        
        if improved:
            prev_a = a.copy()
            prev_g = g_centered.copy()
            tol = max(1e-5, tol * 0.98)
        else:
            velocity = np.zeros(n)
            no_improve += 1
            if no_improve % 30 == 0:
                tol = max(1e-5, tol * 0.8)
                lr = max(1e-6, lr * 0.5)
            if no_improve > 200:
                a = best_a.copy()
                velocity = np.zeros(n)
                prev_a = None
                prev_g = None
                no_improve = 0
                tol = max(1e-5, tol * 0.5)
    
    return best_a, best_sc


def _entropy_lbfgs(a_init, deadline, min_time_remaining=30):
    """L-BFGS using negative entropy of convolution as smooth objective."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    def ent_obj(z):
        a_local = np.exp(z - z.max())
        a_local = a_local / a_local.sum()
        c = _full_convolve(a_local, a_local)
        c = np.maximum(c, 1e-30)
        return float(-np.sum(c * np.log(c)))
    
    def ent_grad(z):
        a_local = np.exp(z - z.max())
        a_local = a_local / a_local.sum()
        c = _full_convolve(a_local, a_local)
        c = np.maximum(c, 1e-30)
        # d/d(a[i]) (-sum c log c) = -sum_k (log(c[k]) + 1) * dc[k]/da[i]
        # dc[k]/da[i] = a[k-i] + a[i-k]
        # So grad_a[i] = -sum_k (log(c[k])+1) * (a[k-i] + a[i-k])
        # = -2 * conv(log(c)+1, a_rev)[i] (sliced)
        lc = np.log(c) + 1.0
        a_rev = a_local[::-1]
        wc = _full_convolve(lc, a_rev)
        ga = -2.0 * wc[n-1:2*n-1]
        W = float(np.dot(a_local, ga))
        return a_local * (ga - W)
    
    # Run L-BFGS
    est_time = 0.003 * max(1.0, n / 8192.0)
    max_iter = min(3000, max(50, int((deadline - min_time_remaining - time.monotonic()) / (est_time * 15))))
    
    try:
        result = minimize(ent_obj, z, jac=ent_grad, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                                   'maxcor': 100})
        a_new = np.exp(result.x - result.x.max())
        a_new = a_new / a_new.sum()
        sc_new = _score(a_new)
        if sc_new < current_sc:
            current_a = a_new
            current_sc = sc_new
    except Exception:
        pass
    
    return current_a
