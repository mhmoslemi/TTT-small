"""
Adaptive simultaneous iteration with L-BFGS polish for minimizing peak linear
autocorrelation of non-negative sequences. Uses multi-scale initialization,
Nesterov momentum, adaptive active set, and periodic L-BFGS refinement.
Key improvements: correct gradient computation, adaptive beta, multi-n search,
and perturbation restarts to escape local minima.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_autocorr(a, p, n):
    """Compute gradient of sum_k p[k]*R(k) w.r.t. a where R(k)=conv(a,a)[k].
    grad[i] = 2 * sum_j a[j] * p[i+j] for j=0..n-1.
    Computed via FFT cross-correlation.
    """
    N = _next_pow2(3 * n + 2)
    Pa = np.fft.rfft(p, N)
    Aa = np.fft.rfft(a, N)
    corr = np.fft.irfft(Pa * np.conj(Aa), N)
    g = corr[:n]
    return 2.0 * g


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * t
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.3) ** 2) + 0.1
        return a / a.sum()
    elif kind == "bump2":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.7) ** 2) + 0.1
        return a / a.sum()
    elif kind == "double_bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.25) ** 2) + np.exp(-12.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _run_momentum_descent(a_init, deadline, min_time_remaining=60.0,
                          max_steps=5000, top_k_range=(3, 30)):
    """Nesterov momentum gradient descent in log-space with adaptive active set."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-30)
    a = a / a.sum()
    z = np.log(a)
    z = z - z.mean()
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_sc = _score(a)
    beta_m = 0.9
    patience = 0
    max_patience = 80
    step_count = 0

    for step in range(max_steps):
        if time.monotonic() + min_time_remaining > deadline:
            break

        a_curr = _softmax(z)
        c = _full_convolve(a_curr, a_curr)
        c_max = float(c.max())

        # Adaptive active set: use peaks within a fraction of max
        frac = 0.001 + 0.01 * min(1.0, step / 500.0)
        threshold = c_max * (1.0 - frac)
        active = c >= threshold
        n_active = int(active.sum())
        if n_active < 1:
            active[int(np.argmax(c))] = True
            n_active = 1
        # Cap the active set size
        if n_active > top_k_range[1]:
            top_idx = np.argsort(c)[::-1][:top_k_range[1]]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = top_k_range[1]

        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / n_active

        # Compute gradient
        grad = _grad_autocorr(a_curr, p, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Nesterov momentum
        v = beta_m * v + (1.0 - beta_m) * grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break

        # Adaptive step size with backtracking
        lr = 0.1 / (v_norm + 1e-10)
        z_trial = z - lr * v
        z_trial = np.clip(z_trial, -40.0, 40.0)
        z_trial = z_trial - z_trial.mean()
        a_trial = _softmax(z_trial)
        c_trial = _full_convolve(a_trial, a_trial)

        # Backtracking line search
        if float(c_trial.max()) > c_max * 1.001:
            for _ in range(10):
                lr *= 0.5
                if lr < 1e-12:
                    break
                z_trial = z - lr * v
                z_trial = np.clip(z_trial, -40.0, 40.0)
                z_trial = z_trial - z_trial.mean()
                a_trial = _softmax(z_trial)
                c_trial = _full_convolve(a_trial, a_trial)
                if float(c_trial.max()) <= c_max:
                    break
            else:
                z = z_trial  # accept anyway with tiny step
                step_count += 1
                continue

        z = z_trial
        a_curr = a_trial
        step_count += 1

        # Check improvement periodically
        if step % 20 == 0 or c_trial.max() < best_sc * best_a.sum() ** 2 / (2.0 * n) - 1e-10:
            sc = _score(a_curr)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_curr.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    # Restart from best with reduced momentum
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)
                    beta_m = max(0.5, beta_m * 0.85)
                    patience = 0

    return best_a, best_sc


def _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with adaptive beta based on peak gap."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-30)
    a = a / a.sum()
    current_sc = _score(a)
    z = np.log(a)
    z = z - z.mean()
    best_a = a.copy()
    best_sc = current_sc

    # Adaptive beta schedule
    ratios = [0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002,
              0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]

    for i, ratio in enumerate(ratios):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # Compute adaptive base_beta from current peak structure
        c_cur = _full_convolve(a, a)
        c_max = float(c_cur.max())
        c_sorted = np.sort(c_cur)[::-1]
        gap = c_sorted[0] - c_sorted[min(5, len(c_sorted) - 1)]
        base_beta = max(gap * 0.1, c_max * 0.0005)
        beta = base_beta * ratio

        est_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(2000, max(20, int(remaining / (est_time * 8))))

        try:
            def obj_func(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    return 2.0 * n * c_max_local
                return 2.0 * n * (c_max_local + beta * np.log(s_e))

            def grad_func(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    w = np.zeros_like(c_local)
                    w[int(np.argmax(c_local))] = 1.0
                else:
                    w = e / s_e
                grad = _grad_autocorr(a_local, w, n)
                return n * grad

            result = minimize(
                obj_func, z,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 40, 'maxls': 80}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                a = a_new.copy()
        except Exception:
            pass

    return best_a, best_sc


def _run_simultaneous_iter(a_init, deadline, min_time_remaining=60.0, max_rounds=200):

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find peaks (within tolerance of max)
        tol = max(1e-8, c_max * 0.0001)
        peaks = np.where(c >= c_max - tol)[0]
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(c))])

        # For each peak, compute the "anti-peak" direction
        # We want to reduce c[k] for k in peaks
        # The gradient of c[k] w.r.t. a is 2*a[k-i] (shifted)
        # Sum of gradients over peaks gives the direction to decrease
        grad = np.zeros(n)
        for k in peaks:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j]

        # Normalize and take step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Line search
        best_step = 0.0
        best_cmax = c_max
        for step_frac in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            lr = step_frac / (gnorm * n**0.5 + 1e-10)
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            c_new = _full_convolve(a_new, a_new)
            if float(c_new.max()) < best_cmax:
                best_cmax = float(c_new.max())
                best_step = step_frac
                a_best = a_new.copy()

        if best_step > 0:
            a = a_best
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining momentum gradient descent, adaptive L-BFGS,
    and simultaneous iteration. Explores multiple sequence lengths and uses
    perturbation restarts to escape local minima.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Momentum gradient descent on parent (400s) ===
    print(f"\n[Phase 1] Momentum Descent (t={time.monotonic() - t_start:.1f}s)")
    phase1_deadline = t_start + 400
    a1, sc1 = _run_momentum_descent(parent, phase1_deadline, min_time_remaining=30.0,
                                    max_steps=10000)
    update_best(a1, sc1, "momentum_1")

    # === Phase 2: L-BFGS polish (300s) ===
    print(f"\n[Phase 2] L-BFGS Adaptive (t={time.monotonic() - t_start:.1f}s)")
    phase2_deadline = min(deadline - 800, phase1_deadline + 300)
    if best_seq is not None and time.monotonic() < phase2_deadline:
        a2, sc2 = _run_lbfgs_adaptive(best_seq, phase2_deadline, min_time_remaining=30.0)
        update_best(a2, sc2, "lbfgs_1")

    # === Phase 3: More momentum descent (300s) ===
    print(f"\n[Phase 3] Momentum Descent R2 (t={time.monotonic() - t_start:.1f}s)")
    phase3_deadline = min(deadline - 500, phase2_deadline + 300)
    if best_seq is not None and time.monotonic() < phase3_deadline:
        a3, sc3 = _run_momentum_descent(best_seq, phase3_deadline, min_time_remaining=30.0,
                                        max_steps=10000, top_k_range=(5, 50))
        update_best(a3, sc3, "momentum_2")

    # === Phase 4: L-BFGS again (200s) ===
    print(f"\n[Phase 4] L-BFGS R2 (t={time.monotonic() - t_start:.1f}s)")
    phase4_deadline = min(deadline - 300, phase3_deadline + 200)
    if best_seq is not None and time.monotonic() < phase4_deadline:
        a4, sc4 = _run_lbfgs_adaptive(best_seq, phase4_deadline, min_time_remaining=20.0)
        update_best(a4, sc4, "lbfgs_2")

    # === Phase 5: Perturbation restarts (200s) ===
    print(f"\n[Phase 5] Perturbation Restarts (t={time.monotonic() - t_start:.1f}s)")
    phase5_deadline = min(deadline - 150, phase4_deadline + 200)
    if best_seq is not None and time.monotonic() < phase5_deadline:
        n_perturb = 5
        perturb_stops = np.linspace(0.01, 0.05, n_perturb)
        for pi in range(n_perturb):
            if time.monotonic() > phase5_deadline - 20:
                break
            sigma = perturb_stops[pi]
            a_pert = _perturb(best_seq, sigma, rng)
            per_deadline = min(phase5_deadline - 10, time.monotonic() + 30)
            a_p, sc_p = _run_momentum_descent(a_pert, per_deadline,
                                              min_time_remaining=5.0, max_steps=1000)
            update_best(a_p, sc_p, f"perturb_{pi}")

    # === Phase 6: Simultaneous iteration (100s) ===
    print(f"\n[Phase 6] Simultaneous Iteration (t={time.monotonic() - t_start:.1f}s)")
    phase6_deadline = min(deadline - 60, phase5_deadline + 100)
    if best_seq is not None and time.monotonic() < phase6_deadline:
        a6, sc6 = _run_simultaneous_iter(best_seq, phase6_deadline,
                                         min_time_remaining=10.0, max_rounds=100)
        update_best(a6, sc6, "simultaneous")

    # === Phase 7: Multi-n exploration (60s) ===
    print(f"\n[Phase 7] Multi-n Exploration (t={time.monotonic() - t_start:.1f}s)")
    phase7_deadline = min(deadline - 30, phase6_deadline + 60)
    if best_seq is not None and time.monotonic() < phase7_deadline:
        n_tries = [16384, 65536]
        for n_try in n_tries:
            if time.monotonic() > phase7_deadline - 15:
                break
            if n_try == len(best_seq):
                continue
            a_init_n = _resample(best_seq, n_try)
            a_n, sc_n = _run_momentum_descent(a_init_n, min(phase7_deadline - 5,
                                                           time.monotonic() + 25),
                                              min_time_remaining=5.0, max_steps=2000)
            update_best(a_n, sc_n, f"n_{n_try}")

    # === Phase 8: Final ultra-tight L-BFGS (30s) ===
    print(f"\n[Phase 8] Final Polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _run_lbfgs_adaptive(best_seq, deadline - 3, min_time_remaining=3.0)
        update_best(a_fin, sc_fin, "final_lbfgs")

        # One more momentum pass
        if time.monotonic() < deadline - 5:
            a_fm, sc_fm = _run_momentum_descent(best_seq, deadline - 3,
                                                min_time_remaining=3.0, max_steps=3000,
                                                top_k_range=(1, 10))
            update_best(a_fm, sc_fm, "final_momentum")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
