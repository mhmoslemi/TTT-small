"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Multi-scale log-sum-exp L-BFGS for coarse optimization
2. Nesterov-accelerated gradient with adaptive soft active set for fine optimization  
3. Direct peak-pair reduction for final polish
4. Diversification via structured perturbations and multi-n search
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _proj_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _soft_max_weights(c, beta):
    """Compute soft active-set weights: w_k = exp(c_k/beta) / sum exp(c_k/beta)."""
    c_shifted = (c - c.max()) / beta
    e = np.exp(np.clip(c_shifted, -500, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

def _grad_soft_max(c, a, w):
    """Gradient of soft-max of convolution with respect to a.
    Returns gradient vector of shape (n,)."""
    n = len(a)
    a_rev = a[::-1]
    h = _full_convolve(w, a_rev)
    # The gradient of (1/beta)*log(sum exp(c_k/beta)) w.r.t. a[m]
    # = sum_k w_k * 2 * a[k-m] * 1_{0<=k-m<n}
    # = 2 * (w * a_rev) convolved, extracted at the right positions
    grad = 2.0 * h[n - 1:2 * n - 1]
    return grad

def _smoothed_obj_score(a, beta):
    """Compute the smoothed objective: 2n * (max_c + beta * log(sum exp((c-max_c)/beta))) / sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -500, 0))
    s_e = e.sum()
    if s_e < 1e-300:
        smooth_max = c_max
    else:
        smooth_max = c_max + beta * np.log(s_e)
    return 2.0 * n * smooth_max / (s * s)

def _smoothed_grad(a, beta):
    """Gradient of the smoothed objective (with sum(a) normalized to 1)."""
    n = len(a)
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    # Objective (with sum a = 1): 2n * soft_max(c)
    # Gradient: 2n * grad_soft_max(c, a, w)
    grad = 2.0 * n * _grad_soft_max(c, a, w)
    return grad

def _nesterov_optimize(a_init, deadline, min_remaining=60.0, beta_start=0.01, beta_end=0.001, n_inner_iters=200):
    """Nesterov-accelerated gradient descent on the smoothed objective.
    Works directly on the simplex (a >= 0, sum a = 1)."""
    n = len(a_init)
    a = _proj_simplex(a_init.copy())
    best_a = a.copy()
    best_score = _score(a)
    
    t_mom = 1.0  # Nesterov momentum parameter
    y = a.copy()  # Nesterov extrapolation point
    lr = 0.1 / (2.0 * n)  # Initial learning rate (conservative)
    
    # Beta schedule: geometric from beta_start to beta_end
    total_budget = deadline - time.monotonic() - min_remaining
    if total_budget <= 0:
        return best_a, best_score
    
    n_steps = max(10, int(total_budget / 0.05))  # Estimate number of steps we can do
    if n_steps > 5000:
        n_steps = 5000
    
    betas = np.geomspace(beta_start, beta_end, min(n_steps, 500))
    
    for step in range(n_steps):
        if time.monotonic() + min_remaining > deadline:
            break
        
        # Current beta
        beta = betas[min(step, len(betas) - 1)]
        
        # Evaluate at y (Nesterov point)
        y = _proj_simplex(y)
        c_y = _full_convolve(y, y)
        w = _soft_max_weights(c_y, beta)
        grad_y = 2.0 * n * _grad_soft_max(c_y, y, w)
        
        # Project gradient to tangent space (sum d = 0)
        grad_y = grad_y - np.mean(grad_y)
        
        gnorm = np.linalg.norm(grad_y)
        if gnorm < 1e-18:
            break
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        alpha_nest = (t_mom - 1.0) / t_new
        
        # Step
        y_new = y - lr * grad_y
        y_new = _proj_simplex(y_new)
        
        # Nesterov extrapolation
        x_new = y_new + alpha_nest * (y_new - y)
        x_new = _proj_simplex(x_new)
        
        # Check improvement (every 5 steps to save time)
        if step % 5 == 0:
            sc = _score(x_new)
            if sc < best_score:
                best_score = sc
                best_a = x_new.copy()
            
            # Adaptive learning rate
            sc_y = _score(y)
            if sc_y > best_score * 1.001:
                lr *= 0.7  # Reduce if stuck
            elif step % 50 == 0 and lr < 0.5 / (2.0 * n * gnorm + 1e-15):
                lr *= 1.05  # Slowly increase if making progress
        
        y = x_new
        t_mom = t_new
    
    return best_a, best_score
