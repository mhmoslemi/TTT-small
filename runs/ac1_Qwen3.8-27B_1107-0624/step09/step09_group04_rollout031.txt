```python
def _variance_grad_z(z, n):
    """Gradient of variance objective w.r.t. z (log-space)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    # Gradient w.r.t. a: g_a[j] = 4 * conv(c, a^R)[j + n - 1]
    a_rev = a[::-1]
    h = _full_convolve(c, a_rev)
    g_a = 4.0 * h[n-1 : 2*n-1]
    # Chain rule: dz -> da
    # a[i] = exp(z[i]) / sum(exp(z))
    # da[i]/dz[j] = a[i] * (delta_ij - a[j])
    # g_z[j] = sum_i g_a[i] * da[i]/dz[j] = sum_i g_a[i] * a[i] * (delta_ij - a[j])
    #        = g_a[j] * a[j] - a[j] * sum_i g_a[i] * a[i]
    sum_ga_a = float(np.dot(g_a, a))
    g_z = a * (g_a - sum_ga_a)
    return g_z
```

```python
"""
Variance-flattening + active-set peak equalization optimizer.
Key innovation: uses sum(c[k]^2) as a smooth surrogate to flatten the
autocorrelation before applying peak-equalization, achieving better
convergence than direct log-sum-exp optimization.

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _variance_objective_z(z, n):
    """sum(c[k]^2) where c = conv(softmax(z), softmax(z))."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    return float(np.sum(c * c))

def _variance_grad_z(z, n):
    """Gradient of sum(c[k]^2) w.r.t. log-space z."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    a_rev = a[::-1]
    h = _full_convolve(c, a_rev)
    g_a = 4.0 * h[n-1 : 2*n-1]
    sum_ga_a = float(np.dot(g_a, a))
    return a * (g_a - sum_ga_a)

def _optimize_variance(a_init, deadline, min_remaining=100.0):
    """Minimize sum(c^2) using Adam in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.01
    best_a = a_init.copy()
    best_obj = _variance_objective_z(z, n)
    best_score = _score(a_init)
    
    t_start = time.monotonic()
    pat = 0  # patience
    
    for t in range(500000):
        if time.monotonic() + min_remaining > deadline:
            break
        
        g = _variance_grad_z(z, n)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** (t + 1))
        v_hat = v / (1 - beta2 ** (t + 1))
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic evaluation
        if (t + 1) % 200 == 0:
            a = _softmax(z)
            obj = _variance_objective_z(z, n)
            sc = _score(a)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                best_score = sc

            else:

    a_final = _softmax(z)

def _peak_equalize(a_init, deadline, min_remaining=50.0):

    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    K = 5
    lr = 0.01
    no_improve = 0
```

```python
"""
Variance-flattening + active-set peak equalization optimizer.
Key innovation: uses sum(conv(a,a)[k]^2) as a smooth surrogate to flatten
the autocorrelation before applying peak-equalization, with Adam optimizer
in log-space for fast convergence. Multi-start with diverse initializations
and adaptive time allocation across sequence lengths.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _variance_obj_z(z, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return float(np.sum(c * c))


def _variance_grad_z(z, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    a_rev = a[::-1]
    h = _full_convolve(c, a_rev)
    g_a = 4.0 * h[n - 1:2 * n - 1]
    sum_ga_a = float(np.dot(g_a, a))
    return a * (g_a - sum_ga_a)


def _adam_variance_minimize(a_init, deadline, min_remaining=60.0):

def _lbfgs_lse(a_init, deadline, min_remaining=30.0):
    """L-BFGS on log-sum-exp surrogate for max(c) minimization."""
    n = len(a_init)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        c_curr = _full_convolve(a_curr, a_curr)

            a_t = _softmax(z)
            c_t = _full_convolve(a_t, a_t)

            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:

            return 2.0 * n * (cm + beta * np.log(s))

            c_t = _full_convolve(a_t, a_t)

        max_iter = min(3000, max(50, int(remaining / 0.01)))

                current_sc = sc_new

        except Exception:

def _peak_equalize(a_init, deadline, min_remaining=30.0):
    """Active-set peak equalization with adaptive K and line search."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    K = 3
    lr = 0.02
    no_improve = 0

    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        cur_sc = 2.0 * n * c_max

        if cur_sc < best_sc - 1e-14:
            best_sc = cur_sc
            best_a = a.copy()
            no_improve = 0
            K = min(K + 1, 80)

        top_idx = np.argsort(c)[::-1][:K]
        w = np.zeros(len(c))
        w[top_idx] = 1.0 / K
        a_rev = a[::-1]
        h = _full_convolve(w, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        alpha = lr / (gnorm + 1e-15)
        improved = False
        for _ in range(40):
            a_try = np.maximum(a - alpha * grad, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try /= s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < cur_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

            lr = min(0.05, lr * 1.01)
        else:
            no_improve += 1
            lr *= 0.85
            if no_improve % 30 == 0:
                K = max(3, K // 2)
            if no_improve >= 300:
                a = best_a.copy()
                lr = 0.02
                K = 3
                no_improve = 0
                break

    return best_a, best_sc

        return a / a.sum()
    elif kind == "rand_rect":
        if rng is None:

    else:

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new

        print(f"evaluate error: {e}")

    return [float(x) for x in final]
```
