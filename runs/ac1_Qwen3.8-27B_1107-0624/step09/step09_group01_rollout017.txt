Uses aggressive K=1 Newton, momentum subgradient with Nesterov,

"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _hankel_matvec(v, W, n):
    """Compute Hessian-vector product for the quadratic form involving W."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]

def _cg_solve_hankel(W, n, rhs, max_iter=300, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.05:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

    best_a = a.copy()
    best_sc = _score(a)

        c = _full_convolve(a)
        c_max = float(c.max())

        K = min(K, 2 * n - 1)
        idx = np.argpartition(c, -K)[-K:]
        W = np.zeros(2 * n - 1)
        W[idx] = 1.0 / K

        wc = _full_convolve(W, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(W, c))
        g = 2.0 * a * (h - W_val)

        gn = float(np.linalg.norm(g_t))

                dx = -g_t / (gn + 1e-30)

        step_done = False

            c_try = _full_convolve(a_try)

            if c_max_try < c_max * (1.0 - 0.005 * alpha) - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-18:
                break

        if not step_done:
            # Fallback: subgradient step
            for trial in [0.05, 0.02, 0.01, 0.005, 0.001, 0.0005, 0.0001]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try)
                if float(c_try.max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 10:
            break

    return best_a, best_sc

def _subgrad_momentum(a_init, deadline, max_steps=5000, lr0=0.03, mom0=0.9):

    momentum = mom0

        z_look = z + momentum * v
        a = _softmax(z_look)

        # Adaptive active set
        for tol_frac in [0.00005, 0.0001, 0.0005, 0.001, 0.003, 0.01, 0.05]:
            tol = tol_frac * c_max

                break

        W[active] = 1.0 / active.sum()

        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(W, c))

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Actually let me just use v = momentum * v - lr * grad

def _run_lbfgs_burst(a_init, deadline, min_remaining=30.0, beta_frac=0.001):
    n = len(a_init)
    c_cur = _full_convolve(a_init)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    beta = max(gap * 0.3, max_c * beta_frac)

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    current_a = a_init.copy()
    current_sc = _score(current_a)

    while beta > max_c * 1e-10:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 8192.0)
        est_iter = 0.004 * scale
        max_iter = min(5000, max(20, int(remaining / (est_iter * 30))))

        def obj_fn(z, beta=beta, n=n):
            a = _softmax(z)
            c = _full_convolve(a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                return 2.0 * n * c_max
            return 2.0 * n * (c_max + beta * np.log(s))

        def grad_fn(z, beta=beta, n=n):
            a = _softmax(z)
            c = _full_convolve(a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W_val = float(np.dot(w, c))
            return 4.0 * n * a * (h - W_val)

        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        c_new = _full_convolve(current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

        if new_gap < 1e-7 * c_max_new:
            beta *= 0.1
        elif new_gap < 1e-4 * c_max_new:
            beta *= 0.2
        else:
            beta *= 0.3

        beta = min(beta, c_max_new * 0.0003)

    return current_a

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.5) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _peak_shave(a, deadline, max_iters=200):
    """Targeted peak shaving: iteratively reduce the maximum peak."""
    n = len(a)
    a = np.maximum(a.copy(), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a)
        k_star = int(np.argmax(c))
        c_max = float(c[k_star])
        
        # Direction: reduce a[i] where a[k_star - i] is large
        # This reduces c[k_star] = sum a[i] * a[k_star - i]
        idx_valid = np.arange(n)
        partner = np.clip(k_star - idx_valid, 0, n - 1)
        # Only use valid partners (where 0 <= k_star - i < n)
        valid = (k_star - idx_valid >= 0) & (k_star - idx_valid < n)
        
        direction = np.zeros(n)
        direction[valid] = a[partner[valid]]  # proportional to partner values
        
        # Normalize direction
        d_norm = np.linalg.norm(direction)
        if d_norm < 1e-15:
            break
        direction = direction / d_norm
        
        # Try decreasing a in the direction
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]:
            a_try = a - step * direction
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max - 1e-16:
                a = a_try
                break
        
        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc

c[k_star] = sum_{i: 0 <= k_star - i < n} a[i] * a[k_star - i]

To reduce c[k_star], I want to decrease the terms a[i] * a[k_star - i]. The gradient of c[k_star] with respect to a[i] is a[k_star - i] (if 0 <= k_star - i < n, else 0).

So the steepest descent direction for c[k_star] is:
direction[i] = a[k_star - i] (if valid)

But we also need to maintain sum(a) = constant (or at least non-zero). So we should subtract the mean:
direction[i] = a[k_star - i] - mean(a[k_star - i] for valid i)

Actually, since the objective is homogeneous (invariant to scaling), we don't need to maintain sum(a) = 1. We just need sum(a) > 0. So we can just decrease in the direction of the gradient without worrying about the constraint.

But wait, if we decrease all a[i] proportionally, the sum decreases but the objective stays the same (since it's homogeneous). So we need to decrease in a direction that's not a scalar multiple of a itself.

**Alternative: Primal-Dual method**

min max_k c_k(a) s.t. a >= 0, sum(a) = 1

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):

def _hankel_matvec(v, W, n):

    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":

        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.5) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.5) ** 2) + 0.02

    else:

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s
