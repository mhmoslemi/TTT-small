import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _conv_max(a):
    c = _full_convolve(a, a)
    return float(c.max()), c


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)

def _normalize(a):

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos2":
        t = np.linspace(0, 1, n)

    elif kind == "flat_top":

    elif kind == "sqrt_hann":
        t = np.linspace(0, 1, n)
        a = np.sqrt(0.5 * (1.0 + np.cos(2.0 * np.pi * t)))

def _active_set_gradient(a, c, tol_frac):
    n = len(a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros(n)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    n_active = int(active.sum())
    if n_active == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        n_active = 1
    eps = 1e-12 * c_max
    w = np.zeros_like(c)
    w[active] = 1.0 / (c_max - c[active] + eps)
    w = w / w.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad = 2.0 * h
    grad = grad - np.mean(grad)
    return grad


def _nesterov_polish(a_init, deadline, tol_start=0.01, tol_end=0.00002,
                     lr_init=0.04, momentum=0.85, max_stall=200):

    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    v = np.zeros(n)
    tol_frac = tol_start
    lr = lr_init
    no_improve = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        grad = _active_set_gradient(a, c, tol_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        v = momentum * v - lr * grad
        alpha = 1.0
        improved = False
        for _ in range(40):
            a_try = _normalize(a + alpha * v)
            sc_try = _conv_max(a_try)[0] * 2.0 * n
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            v = np.zeros(n)
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_end, tol_frac * 0.6)
                lr *= 0.7
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(tol_end, tol_frac * 0.3)
                lr = lr_init * 0.4
                no_improve = 0
        else:
            if tol_frac > tol_end:
                tol_frac = max(tol_end, tol_frac * 0.99)
            lr = min(lr_init, lr * 1.005)

    return best_a, _score(best_a)


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def f(self, z):
        return _smoobj(z, self.beta, self.n)

    def g(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs_spring(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with spring (reheat) annealing schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Spring schedule: multiple passes with decreasing base beta
    # Each pass: cool down, then slightly reheat
    passes = [
        (1.0, 0.01, 8),     # (start_mult, end_mult, n_segments)
        (0.5, 0.005, 8),
        (0.25, 0.002, 8),
        (0.1, 0.0005, 8),
        (0.05, 0.0001, 6),
        (0.02, 0.00003, 6),
    ]

    for pass_idx, (start_mult, end_mult, n_seg) in enumerate(passes):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        # Generate beta schedule for this pass
        betas = base_beta * np.geomspace(start_mult, end_mult, n_seg)

        for beta in betas:
            if time.monotonic() + min_time_remaining > deadline:
                break
            remaining = deadline - min_time_remaining - time.monotonic()
            if remaining < 5:
                break

            # Adaptive max iterations based on remaining time and n
            est_time = 0.003 * max(1.0, n / 4096.0)
            max_iter = min(4000, max(30, int(remaining / (est_time * 20))))

            prob = _LbfgsProblem(beta, n)
            try:
                result = minimize(
                    prob.f, z, jac=prob.g, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                             'maxcor': 80, 'maxls': 40}
                )
                a_new = _softmax(result.x)
                sc_new = _score(a_new)
                if sc_new < current_sc - 1e-16:
                    current_a = a_new
                    current_sc = sc_new
                    z = np.log(np.maximum(a_new, 1e-30))
                    z = z - z.mean()
            except Exception:
                pass

        # Reheat: slightly increase z noise to escape potential local min
        if pass_idx < len(passes) - 1 and time.monotonic() < deadline - min_time_remaining - 20:
            z = np.log(np.maximum(current_a, 1e-30))

            z = z + 0.002 * rng_reheat.standard_normal(n)
            z = z - z.mean()
            current_a = _softmax(z)

    return current_a


def _targeted_peak_correction(a, deadline, n_corr=50):

            a_perturbed += sigma * bump * a_perturbed.max()
            a_perturbed = _normalize(a_perturbed)
        elif kind == 'block_large':
            block_size = max(50, n // 3)
            start = int(rng.integers(0, n - block_size))
            a_perturbed = current_a.copy()
            z = np.log(np.maximum(a_perturbed[start:start + block_size], 1e-30))
            z = z + sigma * rng.standard_normal(block_size)
            a_perturbed[start:start + block_size] = np.exp(z - np.max(z))
            a_perturbed = _normalize(a_perturbed)
        elif kind == 'sparse_large':
            n_perturb = max(50, n // 5)
            idx = rng.choice(n, size=n_perturb, replace=False)
            a_perturbed = current_a.copy()
            factors = np.exp(sigma * rng.standard_normal(n_perturb))
            a_perturbed[idx] *= factors
            a_perturbed = _normalize(a_perturbed)
        else:
            a_perturbed = current_a.copy()

        opt_deadline = min(deadline - min_time_remaining, time.monotonic() + opt_time)
        # Quick optimize: L-BFGS then targeted correction
        a_opt = _quick_optimize(a_perturbed, opt_deadline, min_time_remaining=2)
        sc_opt = _score(a_opt)

        if sc_opt < best_sc - 1e-14:
            best_sc = sc_opt
            best_a = a_opt.copy()
            current_a = a_opt.copy()
            print(f"    ILS NEW BEST: {best_sc:.10f} ({kind}, sigma={sigma}) t={time.monotonic()-t_start:.0f}s")
        elif sc_opt < current_sc - 1e-14:
            current_a = a_opt.copy()

        attempt += 1

    return best_a, best_sc


def _quick_optimize(a_init, deadline, min_time_remaining=10.0):
    """Quick combined optimization: L-BFGS + targeted correction + polish."""
    n = len(a_init)
    remaining = deadline - time.monotonic()
    if remaining < 3:
        return a_init

    # L-BFGS with moderate beta
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.002:
        gap = max_c * 0.002
    beta = max(gap, max_c * 0.003)

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    prob = _LbfgsProblem(beta, n)

    est_time = 0.003 * max(1.0, n / 4096.0)
    max_iter = min(2000, max(20, int((remaining - min_time_remaining) / (est_time * 25))))

    try:
        result = minimize(
            prob.f, z, jac=prob.g, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-17, 'gtol': 1e-14,
                     'maxcor': 60, 'maxls': 30}
        )
        a_new = _softmax(result.x)
    except Exception:
        a_new = a_init.copy()

    # Targeted correction
    remaining2 = deadline - min_time_remaining - time.monotonic()
    if remaining2 > 2:
        a_new, _ = _targeted_peak_correction(a_new, time.monotonic() + remaining2 * 0.5)

    # Quick polish
    remaining3 = deadline - min_time_remaining - time.monotonic()
    if remaining3 > 1:
        a_new, _ = _nesterov_polish(a_new, time.monotonic() + remaining3,
                                     tol_start=0.005, tol_end=0.0005,
                                     lr_init=0.02, momentum=0.8, max_stall=50)

    return a_new


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _final_ultra_polish(a_init, deadline):
    """Ultra-fine final polish: alternating very small beta L-BFGS and active-set gradient."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    # Multiple rounds of alternating methods
    rounds = [
        ('lbfgs', 0.0001),
        ('targeted', None),
        ('ghost', 0.00005),
        ('lbfgs', 0.00003),
        ('targeted', None),
        ('ghost', 0.00002),
        ('lbfgs', 0.00001),
        ('targeted', None),
        ('ghost', 0.00001),
    ]

    for method, beta in rounds:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        if method == 'lbfgs':
            z = np.log(np.maximum(a, 1e-30))
            z = z - z.mean()
            prob = _LbfgsProblem(float(beta) * float(_full_convolve(a, a).max()), n)
            max_iter = min(2000, max(20, int(remaining / 0.05)))
            try:
                result = minimize(prob.f, z, jac=prob.g, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-19,
                                           'gtol': 1e-16, 'maxcor': 60, 'maxls': 30})
                a_new = _softmax(result.x)
            except Exception:
                a_new = a
            sc = _score(a_new)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a_new.copy()
                a = a_new
        elif method == 'targeted':
            a, sc = _targeted_peak_correction(a, time.monotonic() + min(10, remaining * 0.5), n_corr=100)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a.copy()
        elif method == 'ghost':
            a, sc = _nesterov_polish(a, time.monotonic() + min(10, remaining * 0.5),
                                      tol_start=0.001 * float(beta) * 100000 if beta else 0.001,
                                      tol_end=0.00001, lr_init=0.005, momentum=0.8, max_stall=100)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a.copy()

    return best_a, _score(best_a)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer with spring annealing, targeted peak correction,
    enhanced ILS, and ultra-fine final polish.
    """
    global t_start
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 24576
        parent = _make_profile(n_parent, "hann")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, hann n={n_parent}, score={best_score:.10f}")

    n_main = n_parent

    # === Phase 1: Multi-n initialization (60s) ===
    print(f"\n[P1] Multi-n init (t={time.monotonic() - t_start:.1f}s)")
    p1_deadline = t_start + 60

    n_tries = [n_main]
    for nn in [n_main * 2, n_main // 2, int(n_main * 1.25), int(n_main * 1.5)]:
        if nn >= 1024 and nn <= 65536 and nn not in n_tries:
            n_tries.append(nn)

    for n_try in n_tries:
        if time.monotonic() > p1_deadline - 10:
            break
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        a_opt = _quick_optimize(a_init, min(p1_deadline, time.monotonic() + 8), min_time_remaining=2)
        sc_opt = _score(a_opt)
        print(f"  n={n_try}: init={sc_init:.10f}, opt={sc_opt:.10f}")
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            n_main = n_try
            print(f"  NEW BEST: {best_score:.10f} [p1/n={n_try}]")

    # === Phase 2: Spring L-BFGS main optimization (400s) ===
    print(f"\n[P2] Spring L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    p2_deadline = t_start + 460
    a_main = _run_lbfgs_spring(best_seq, p2_deadline, min_time_remaining=40)
    sc_main = _score(a_main)
    if sc_main < best_score:
        best_seq = a_main.copy()
        best_score = sc_main
        print(f"  NEW BEST: {best_score:.10f} [p2/spring_lbfgs]")
    else:
        print(f"  No improvement: {sc_main:.10f} vs {best_score:.10f}")

    # === Phase 3: Targeted correction + polish (120s) ===
    print(f"\n[P3] Targeted + polish (t={time.monotonic() - t_start:.1f}s)")
    p3_deadline = t_start + 580
    a_corr, sc_corr = _targeted_peak_correction(best_seq, p3_deadline - 40, n_corr=200)
    if sc_corr < best_score:
        best_seq = a_corr.copy()
        best_score = sc_corr
        print(f"  NEW BEST: {best_score:.10f} [p3/targeted]")
    a_pol, sc_pol = _nesterov_polish(best_seq, p3_deadline,
                                      tol_start=0.002, tol_end=0.00001,
                                      lr_init=0.015, momentum=0.85, max_stall=150)
    if sc_pol < best_score:
        best_seq = a_pol.copy()
        best_score = sc_pol
        print(f"  NEW BEST: {best_score:.10f} [p3/polish]")

    # === Phase 4: ILS round 1 (350s) ===
    print(f"\n[P4] ILS round 1 (t={time.monotonic() - t_start:.1f}s)")
    p4_deadline = t_start + 930
    a_ils, sc_ils = _stochastic_ils(best_seq, p4_deadline, rng, min_time_remaining=60)
    if sc_ils < best_score:
        best_seq = a_ils.copy()
        best_score = sc_ils
        print(f"  NEW BEST: {best_score:.10f} [p4/ils1]")

    # === Phase 5: L-BFGS refinement on ILS result (200s) ===
    print(f"\n[P5] L-BFGS refinement (t={time.monotonic() - t_start:.1f}s)")
    p5_deadline = t_start + 1130
    if time.monotonic() < p5_deadline - 30:
        a_ref = _run_lbfgs_spring(best_seq, p5_deadline, min_time_remaining=40)
        sc_ref = _score(a_ref)
        if sc_ref < best_score:
            best_seq = a_ref.copy()
            best_score = sc_ref
            print(f"  NEW BEST: {best_score:.10f} [p5/refine]")

    # === Phase 6: ILS round 2 (200s) ===
    print(f"\n[P6] ILS round 2 (t={time.monotonic() - t_start:.1f}s)")
    p6_deadline = t_start + 1330
    if time.monotonic() < p6_deadline - 30:
        rng2 = np.random.default_rng(int(seed) + 7777)
        a_ils2, sc_ils2 = _stochastic_ils(best_seq, p6_deadline, rng2, min_time_remaining=40)
        if sc_ils2 < best_score:
            best_seq = a_ils2.copy()
            best_score = sc_ils2
            print(f"  NEW BEST: {best_score:.10f} [p6/ils2]")

    # === Phase 7: Multi-n exploration (150s) ===
    print(f"\n[P7] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    p7_deadline = t_start + 1480
    n_current = len(best_seq)
    n_cands = [n_current]
    for mult in [0.75, 1.0, 1.25, 1.5, 2.0, 0.5]:
        nn = int(n_current * mult)
        if nn >= 2048 and nn <= 65536 and nn not in n_cands:
            n_cands.append(nn)

    for n_try in n_cands:
        if time.monotonic() > p7_deadline - 15:
            break
        if n_try == n_current:
            continue
        a_init = _resample(best_seq, n_try)
        a_opt = _quick_optimize(a_init, min(p7_deadline, time.monotonic() + 20), min_time_remaining=5)
        sc_opt = _score(a_opt)
        print(f"  n={n_try}: score={sc_opt:.10f}")
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [p7/n={n_try}]")

    # === Phase 8: Final ultra-fine polish (150s) ===
    print(f"\n[P8] Final ultra-polish (t={time.monotonic() - t_start:.1f}s)")
    p8_deadline = deadline
    a_fin, sc_fin = _final_ultra_polish(best_seq, p8_deadline)
    if sc_fin < best_score:
        best_seq = a_fin.copy()
        best_score = sc_fin
        print(f"  NEW BEST: {best_score:.10f} [p8/final]")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
