def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    out_n = len(a) + len(b) - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_fast(a):
    """Compute evaluation score efficiently."""
    n = len(a)
    s = float(a.sum())

        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def _run_active_set(a_init, deadline, min_remaining=60.0):

    n = len(a_init)
    a = np.maximum(a_init, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_fast(a)

            break

        c_max = float(c.max())

        # Progressive tolerance
        tol = c_max * max(0.001, 0.01 * (0.95 ** rnd))
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / active.sum()

        # Projected gradient inner loop
        for inner in range(30):
            a_rev = a[::-1]
            pc = _fft_convolve(p, a_rev)
            h = pc[n - 1:2 * n - 1]
            W_val = float(np.dot(p, c))
            grad = 4.0 * n * a * (h - W_val)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            lr = 0.1 / (gnorm + 1e-10)
            a = _proj_simplex(a - lr * grad)

        sc = _score_fast(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc

        k_active = min(20, len(c))
        top_k = np.argsort(c)[::-1][:k_active]
        p = np.zeros(2 * n - 1)
        p[top_k] = 1.0 / k_active

        a_rev = a_curr[::-1]
        pc = _fft_convolve(p, a_rev)
        h = pc[n - 1:2 * n - 1]
        W_val = float(np.dot(p, c))
        grad = 4.0 * n * a_curr * (h - W_val)

        # Mirror descent step in log-space
        lr = 0.1 / (1.0 + step * 0.01)
        z = z - lr * grad
        z = z - z.mean()
        z = np.clip(z, -100, 100)

        if step % 5 == 0:
            sc = _score_fast(_softmax(z))
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = _softmax(z).copy()

    return best_a, best_sc


def _resample(a, n_new):
    """Resample sequence to new length."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    """Perturb in log-space."""
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS, projected gradient with momentum,
    active-set QP, mirror descent, and multi-scale exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score_fast(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is None:
        n_init = 32768
        parent = _make_profile(n_init, "tapered")
        print(f"No parent found, using tapered n={n_init}")
        update_best(parent, "init")
    else:
        print(f"Parent: n={len(parent)}, score={_score_fast(parent):.10f}")
        update_best(parent, "parent")

    n_parent = len(best_seq) if best_seq is not None else 32768

    # === Phase 1: L-BFGS from parent (400s) ===
    print(f"\n[Phase 1] L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    p1_deadline = t_start + 400
    a1 = _run_lbfgs_smoothed(best_seq, p1_deadline, min_remaining=60)
    update_best(a1, "lbfgs_1")

    # === Phase 2: Projected gradient with momentum (300s) ===
    print(f"\n[Phase 2] Proj Grad Momentum (t={time.monotonic() - t_start:.1f}s)")
    p2_deadline = t_start + 700
    a2, sc2 = _run_proj_grad_momentum(best_seq, p2_deadline, min_remaining=60)
    update_best(a2, "proj_grad_1")

    # === Phase 3: Active set (300s) ===
    print(f"\n[Phase 3] Active Set (t={time.monotonic() - t_start:.1f}s)")
    p3_deadline = t_start + 1000
    a3, sc3 = _run_active_set(best_seq, p3_deadline, min_remaining=60)
    update_best(a3, "active_1")

    # === Phase 4: L-BFGS round 2 with perturbation (200s) ===
    print(f"\n[Phase 4] L-BFGS perturbed (t={time.monotonic() - t_start:.1f}s)")
    p4_deadline = t_start + 1200
    a_pert = _perturb(best_seq, 0.02, rng)
    a4 = _run_lbfgs_smoothed(a_pert, p4_deadline, min_remaining=60)
    update_best(a4, "lbfgs_pert")

    # === Phase 5: Mirror descent (150s) ===
    print(f"\n[Phase 5] Mirror Descent (t={time.monotonic() - t_start:.1f}s)")
    p5_deadline = t_start + 1350
    a5, sc5 = _run_mirror_descent(best_seq, p5_deadline, min_remaining=60)
    update_best(a5, "mirror_1")

    # === Phase 6: Multi-length exploration (200s) ===
    print(f"\n[Phase 6] Multi-length (t={time.monotonic() - t_start:.1f}s)")
    p6_deadline = min(deadline - 50, t_start + 1550)
    for n_try in [16384, 65536, 49152]:
        if time.monotonic() > p6_deadline - 60:
            break
        if n_try == n_parent:
            continue
        a_res = _resample(best_seq, n_try)
        a_n = _run_lbfgs_smoothed(a_res, min(p6_deadline, time.monotonic() + 100), min_remaining=15)
        update_best(a_n, f"n_{n_try}")
        if _score_fast(a_n) < best_score + 0.01 and time.monotonic() < p6_deadline - 30:
            a_nq, sc_nq = _run_active_set(a_n, min(p6_deadline, time.monotonic() + 60), min_remaining=10)
            update_best(a_nq, f"n_{n_try}_active")

    # === Phase 7: Final polish (remaining time) ===
    print(f"\n[Phase 7] Final Polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 20:
        a_fin = _run_lbfgs_smoothed(best_seq, deadline - 10, min_remaining=5)
        update_best(a_fin, "final_lbfgs")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "uniform")
        sc_final = _score_fast(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(8192, "uniform")
    print(f"FALLBACK: score={_score_fast(fallback):.10f}")
    return [float(x) for x in fallback]
