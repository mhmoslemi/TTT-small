Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: Gauss-Newton active-set peak equalization exploiting the
quadratic structure of autocorrelation, combined with L-BFGS on log-domain
parameterization for general optimization. Multi-phase strategy with adaptive
active set growth.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = _fft_convolve(a, a)
    return 2.0 * n * float(conv.max()) / (s * s)


def _gn_equalize(a_init, deadline, min_remaining=30.0):
    """Gauss-Newton active-set peak equalization.
    Exploits that conv[k] is quadratic in a, so Gauss-Newton = Newton (exact).
    Iteratively identifies the active set (near-max peaks) and takes exact
    Newton-like steps to equalize them."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * float(_fft_convolve(a, a).max())
    
    K = 2
    K_max = min(100, n // 2)
    no_improve = 0
    total_no_improve = 0
    damping = 1.0
    i_arr = np.arange(n)
    
    while time.monotonic() + min_remaining < deadline:
        conv = _fft_convolve(a, a)
        max_conv = float(conv.max())
        current_score = 2.0 * n * max_conv
        
        if current_score < best_score - 1e-13:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
            total_no_improve = 0
            # Grow active set
            K = min(K_max, K + 1)
            damping = min(1.0, damping * 1.5)
        
        if no_improve > 50:
            # Reset
            a = best_a.copy()
            K = max(2, K // 3)
            damping = 0.1
            no_improve = 0
            total_no_improve += 1
            if total_no_improve > 20:
                break
        
        # Identify active set: top K peaks
        top_idx = np.argsort(conv)[::-1][:K]
        top_vals = conv[top_idx]
        
        # Skip if already well-equalized
        spread = top_vals.max() - top_vals.min()
        if spread < max_conv * 1e-6:
            no_improve += 1
            continue
        
        # Target: minimum of active peaks (we want to bring the top down to the bottom)
        # Actually, target the average for balanced reduction
        t_target = top_vals.mean()

        idx_mat = top_idx[:, None].astype(np.int64) - i_arr[None, :]
        valid_mat = (idx_mat >= 0) & (idx_mat < n)
        clamped = np.clip(idx_mat, 0, n - 1)
        G = np.where(valid_mat, 2.0 * a[clamped], 0.0)
        
        # RHS: how much each active peak needs to change
        b = t_target - top_vals
        
        # Levenberg-Marquardt for robustness
        lam = 1e-4 * max(1.0, max_conv)
        GtG = G.T @ G + lam * np.eye(n)
        Gtb = G.T @ b
        d = np.linalg.solve(GtG, Gtb)
        
        # Project to sum-zero
        d = d - d.mean()
        
        # Limit step for non-negativity
        neg_mask = d < 0
        if neg_mask.any():
            max_alpha = float(np.min(a[neg_mask] / (-d[neg_mask])))
        else:
            max_alpha = 1.0
        max_alpha = max(max_alpha, 1e-15)

        improved = False

            if (a_try < -1e-14).any():
                alpha *= 0.5
                continue
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-14:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            conv_try = _fft_convolve(a_try, a_try)
            score_try = 2.0 * n * float(conv_try.max())
            if score_try < current_score - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

            if alpha < damping * 0.1:
                damping = max(0.01, alpha * 1.5)
        else:
            no_improve += 1
            damping = max(0.01, damping * 0.5)
            K = max(2, K - 1)
    
    return best_a, best_score


def _softmax(z):
    z = z - z.max()
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoothed_obj(z, beta, n):
    a = _softmax(z)
    conv = _fft_convolve(a, a)
    c_max = float(conv.max())
    shifted = (conv - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smoothed_grad(z, beta, n):
    a = _softmax(z)
    conv = _fft_convolve(a, a)
    c_max = float(conv.max())
    shifted = (conv - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _fft_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, conv))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_remaining=30.0, beta_scale=1.0):
    """L-BFGS optimization on log-domain parameterization."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _compute_score(a)
    
    z = np.log(a)
    z = z - z.mean()
    
    # Adaptive beta sequence
    conv = _fft_convolve(a, a)
    sorted_c = np.sort(conv)[::-1]
    gap = sorted_c[0] - sorted_c[min(30, len(sorted_c) - 1)]
    if gap < sorted_c[0] * 0.001:
        gap = sorted_c[0] * 0.001
    base_beta = max(gap, sorted_c[0] * 0.005) * beta_scale
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001])
    
    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break
        
        def obj(z):
            return _smoothed_obj(z, beta, n)
        
        def grad(z):
            return _smoothed_grad(z, beta, n)
        
        max_iter = min(3000, max(20, int(remaining / 0.05)))
        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 60}
            )

Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using Gauss-Newton

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = _fft_convolve(a, a)
    return 2.0 * n * float(conv.max()) / (s * s)

        if step_improved:
            if alpha < damping * 0.05:
                damping = max(0.01, alpha * 2.0)
        else:
            no_improve += 1
            damping = max(0.01, damping * 0.6)
            K = max(2, K - 1)

    return best_a, best_score


def _softmax(z):
    z = z - z.max()
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoothed_obj(z, beta, n):
    a = _softmax(z)
    conv = _fft_convolve(a, a)
    c_max = float(conv.max())
    shifted = (conv - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smoothed_grad(z, beta, n):
    a = _softmax(z)
    conv = _fft_convolve(a, a)
    c_max = float(conv.max())
    shifted = (conv - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _fft_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, conv))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_remaining=30.0, beta_scale=1.0):
    """L-BFGS on log-domain with adaptive beta annealing."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _compute_score(a)

    z = np.log(a)
    z = z - z.mean()

    conv = _fft_convolve(a, a)
    sorted_c = np.sort(conv)[::-1]
    gap = float(sorted_c[0] - sorted_c[min(50, len(sorted_c) - 1)])
    if gap < sorted_c[0] * 0.001:
        gap = float(sorted_c[0]) * 0.001
    base_beta = max(gap, float(sorted_c[0]) * 0.005) * beta_scale

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005])

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break

        def obj_func(z):
            return _smoothed_obj(z, beta, n)

        def grad_func(z):
            return _smoothed_grad(z, beta, n)

        max_iter = min(4000, max(30, int(remaining / 0.04)))
        try:
            result = minimize(
                obj_func, z, jac=grad_func, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 80}
            )
            a_new = _softmax(result.x)
            sc_new = _compute_score(a_new)
            if sc_new < best_score - 1e-14:
                best_a = a_new
                best_score = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return best_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 7.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new
    new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _subgradient_step(a, deadline, min_remaining=10.0):
    """Lightweight subgradient polish for final touches."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * float(_fft_convolve(a, a).max())
    i_arr = np.arange(n)

    for it in range(500):
        if time.monotonic() + min_remaining > deadline:
            break
        conv = _fft_convolve(a, a)
        c_max = float(conv.max())
        current_score = 2.0 * n * c_max

        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()

        # Active set: peaks within 0.05% of max
        threshold = c_max * (1 - 0.0005)
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Average gradient of active peaks
        idx_mat = active[:, None].astype(np.int64) - i_arr[None, :]
        valid_mat = (idx_mat >= 0) & (idx_mat < n)
        clamped = np.clip(idx_mat, 0, n - 1)
        G = np.where(valid_mat, 2.0 * a[clamped], 0.0)
        grad = G.mean(axis=0)
        grad = grad - grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        eta = 0.02 / (1.0 + it * 0.01)
        eta = min(eta, 0.4 / (np.max(np.abs(grad)) + 1e-15))

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        # Check improvement every 3 iterations
        if (it + 1) % 3 == 0:
            conv_new = _fft_convolve(a_new, a_new)
            if float(conv_new.max()) < c_max:
                a = a_new
            else:
                pass  # keep old a, eta will shrink

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase optimizer: L-BFGS warmup, Gauss-Newton equalization,
    fine L-BFGS, repeated GN with larger active sets, multi-n, final polish."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        n_parent = len(parent)
        parent_sc = _compute_score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _compute_score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS warmup (explore the landscape) ===
    print(f"\n[P1] L-BFGS warmup (t={time.monotonic()-t_start:.1f}s)")
    a_p1 = _run_lbfgs(parent, deadline, min_remaining=1000.0, beta_scale=1.0)
    sc_p1 = _compute_score(a_p1)
    if sc_p1 < best_score:
        best_seq = a_p1.copy()
        best_score = sc_p1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_warmup]")
    else:
        print(f"  No improvement: {sc_p1:.10f} vs {best_score:.10f}")

    # === Phase 2: Gauss-Newton equalization (main improvement phase) ===
    print(f"\n[P2] Gauss-Newton equalization (t={time.monotonic()-t_start:.1f}s)")
    gn_dead1 = min(deadline - 700, time.monotonic() + 350)
    a_gn1, sc_gn1 = _gn_equalize(best_seq, gn_dead1, min_remaining=30.0)
    if sc_gn1 < best_score:
        best_seq = a_gn1.copy()
        best_score = sc_gn1
        print(f"  NEW BEST: {best_score:.10f} [gn1]")
    else:
        print(f"  GN1: no improvement (best={best_score:.10f})")

    # === Phase 3: Fine L-BFGS (refine after GN) ===
    print(f"\n[P3] Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        a_fine = _run_lbfgs(best_seq, deadline, min_remaining=500.0, beta_scale=0.3)
        sc_fine = _compute_score(a_fine)
        if sc_fine < best_score:
            best_seq = a_fine.copy()
            best_score = sc_fine
            print(f"  NEW BEST: {best_score:.10f} [fine_lbfgs]")
        else:
            print(f"  Fine L-BFGS: no improvement ({sc_fine:.10f})")

    # === Phase 4: Second GN round (larger active sets) ===
    print(f"\n[P4] GN round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        gn_dead2 = min(deadline - 350, time.monotonic() + 250)
        a_gn2, sc_gn2 = _gn_equalize(best_seq, gn_dead2, min_remaining=30.0)
        if sc_gn2 < best_score:
            best_seq = a_gn2.copy()
            best_score = sc_gn2
            print(f"  NEW BEST: {best_score:.10f} [gn2]")
        else:
            print(f"  GN2: no improvement (best={best_score:.10f})")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_cur = len(best_seq)
    n_candidates = sorted(set([n_cur, int(n_cur * 1.25), int(n_cur * 0.8)]), reverse=True)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 120:
            break
        if n_try == n_cur:
            continue
        if n_try < 4096 or n_try > 49152:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _compute_score(a_init)
        print(f"  n={n_try}: init={sc_init:.10f}")
        a_opt = _run_lbfgs(a_init, deadline, min_remaining=120.0, beta_scale=0.5)
        sc_opt = _compute_score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n]")
        # Quick GN on the new n
        if time.monotonic() < deadline - 60:
            a_gn_m, sc_gn_m = _gn_equalize(a_opt, deadline - 30, min_remaining=15.0)
            if sc_gn_m < best_score:
                best_seq = a_gn_m.copy()
                best_score = sc_gn_m
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_gn]")

    # === Phase 6: Final GN polish ===
    print(f"\n[P6] Final GN polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 10 - time.monotonic()
    if remaining > 20:
        a_fin, sc_fin = _gn_equalize(best_seq, deadline - 5, min_remaining=5.0)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_gn]")

    # === Phase 7: Subgradient micro-polish ===
    print(f"\n[P7] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 5 - time.monotonic()
    if remaining > 15:
        a_sg, sc_sg = _subgradient_step(best_seq, deadline - 3, min_remaining=3.0)
        if sc_sg < best_score:
            best_seq = a_sg.copy()
            best_score = sc_sg
            print(f"  NEW BEST: {best_score:.10f} [subgrad]")

    # Final validation
    print(f"\n{'=' * 60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _compute_score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        ev_score = evaluate_sequence([float(x) for x in final])
        print(f"evaluate_sequence: {ev_score:.10f}")
    except Exception as e:
        print(f"evaluate_sequence error: {e}")
    return [float(x) for x in final]
