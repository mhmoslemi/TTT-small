    lr = 0.01
    tol_expand = 0.005  # initial tolerance for "near-peak"
    tol_shrink = 0.0001
    momentum = 0.5
    velocity = np.zeros(n)
    stall = 0
    max_stall = 100

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive tolerance: start wide, shrink as we stall
        tol = max(tol_shrink, tol_expand * (0.95 ** stall) if stall < 200 else tol_shrink)

        # Find active peaks
        active_mask = c >= (c_max - tol * c_max)
        n_active = int(active_mask.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active_mask = np.zeros(n_corr, dtype=bool)
            active_mask[idx] = True
            n_active = 1

        # Weights: emphasize peaks that are higher than mean of active
        active_vals = c[active_mask]
        mean_active = active_vals.mean()
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w = np.zeros(n_corr)
            w[active_mask] = (active_vals - mean_active + spread) / (n_active * spread + 1e-30)
            w = np.maximum(w, 0.05 / n_active)
            w = w / w.sum()
        else:
            w = np.zeros(n_corr)
            w[active_mask] = 1.0 / n_active

        # Gradient
        g = _grad_weighted(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break

        # Momentum step with backtracking
        velocity = momentum * velocity - lr * g_centered
        improved = False
        alpha = 1.0
        for ls in range(30):
            a_try = a + alpha * velocity
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-16:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if improved:
            lr = min(0.05, lr * 1.02)
            momentum = min(0.7, momentum + 0.01)
        else:
            velocity = np.zeros(n)
            lr = max(1e-5, lr * 0.5)
            momentum = max(0.3, momentum - 0.05)

        if stall > max_stall:
            # Reset and try wider active set
            a = best_a.copy()
            velocity = np.zeros(n)
            lr = max(1e-4, lr * 0.5)
            tol_expand = max(tol_shrink, tol_expand * 0.8)
            stall = 0
            if tol_expand < tol_shrink * 2:
                break

    return best_a, best_sc


def _iterative_target(a_init, deadline, min_time_remaining=30.0):
    """Iteratively reduce a target level for the autocorrelation peaks."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    target = float(c.max())
    gamma = 2.0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()

        # Objective: minimize sum of (c[k] - target)^2 for c[k] > target
        # plus a small penalty for max(c)
        def obj(z):
            aa = np.exp(z - z.max())
            aa = aa / aa.sum()
            cc = _full_convolve(aa, aa)
            excess = np.maximum(cc - target, 0)
            return float(np.sum(excess * excess)) + 1e-4 * float(cc.max()) ** 2

        def obj_grad(z):
            aa = np.exp(z - z.max())
            aa = aa / aa.sum()
            cc = _full_convolve(aa, aa)
            excess = np.maximum(cc - target, 0)
            # d/dck of sum(excess^2) = 2*excess
            # d/dck of max(c)^2 * 1e-4 = 2e-4 * max_c if k == argmax
            w = 2.0 * excess
            max_idx = int(np.argmax(cc))
            w[max_idx] += 2e-4 * cc[max_idx]
            ga = _grad_weighted(aa, w)
            W = float(np.dot(aa, ga))
            return aa * (ga - W)

        z = np.log(np.maximum(a, 1e-30))
        z = z - z.mean()

        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        max_iter = min(3000, max(50, int(remaining / 0.1)))

        try:
            result = minimize(obj, z, jac=obj_grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12})
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_a = a_new.copy()
                best_sc = sc_new
                a = a_new
            # Tighten target
            c_new = _full_convolve(a_new, a_new)
            target = max(float(c_new.max()) * 0.995, best_sc / (2.0 * n) * 0.99)
        except Exception:
            target = c_max * 0.998

        # If target hasn't decreased in a while, break
        if target >= c_max * 0.999:
            target = c_max * 0.998

    return best_a, best_sc


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    """Ultra-fine active-set polish with very small steps."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    n_corr = 2 * n - 1
    lr = 0.003
    no_improve = 0
    max_no_improve = 500

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-16:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Active set: peaks within very tight tolerance
        active = c >= (c_max - 0.002 * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros(n_corr, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros(n_corr)
        w[active] = 1.0 / n_active

        g = _grad_weighted(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break

        # Very careful line search
        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        improved = False
        for ls in range(40):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-16:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.7
            if no_improve > max_no_improve:
                break
        else:
            lr = min(0.005, lr * 1.01)

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        # Taper edges
        edge = int(0.1 * n)
        for i in range(edge):
            frac = (i + 1) / (edge + 1)
            taper = 0.5 * (1 + np.cos(np.pi * frac))
            a[i] = taper
            a[n - 1 - i] = taper
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "sin_p":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.8
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb_log(a, sigma, rng):
    """Perturb in log-space for diversity."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

    return (mix * a1 + (1 - mix) * a2)

def _coordinate_descent(a_init, deadline, min_time_remaining=10.0):
    """Coordinate descent on a small set of shape parameters."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    # Define shape as base * (1 + small perturbations at control points)
    # Use cosine basis for smooth perturbations
    n_ctrl = min(8, n // 100)
    t = np.linspace(0, 1, n)
    basis = np.zeros((n_ctrl, n))
    for k in range(n_ctrl):
        basis[k] = np.cos(2 * np.pi * (k + 1) * t)

    params = np.zeros(n_ctrl)
    step = 0.01

    for _ in range(200):
        if time.monotonic() + min_time_remaining >= deadline:
            break
        for k in range(n_ctrl):
            if time.monotonic() + min_time_remaining >= deadline:
                break
            # Try positive and negative steps
            for direction in [1, -1]:
                params[k] += direction * step * basis[k].max()
                a_try = a + np.dot(params, basis)
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    params[k] -= direction * step * basis[k].max()
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-16:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    break
                else:
                    params[k] -= direction * step * basis[k].max()
            step *= 0.9
            if step < 1e-5:
                break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer with peak equalization, iterative target reduction,
    adaptive L-BFGS, and fine polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        n_main = len(parent)
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"Parent: n={n_main}, score={best_score:.10f}")
    else:
        n_main = 16384
        best_seq = _make_profile(n_main, "tapered")
        best_score = _score(best_seq)
        print(f"No parent, tapered n={n_main}, score={best_score:.10f}")

    # === Phase 1: L-BFGS with adaptive K schedule ===
    print(f"\n[P1] L-BFGS schedule (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 700, t_start + 350)
    a1 = _run_lbfgs_schedule(best_seq, p1_deadline, min_time_remaining=60)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  Updated: {best_score:.10f} [P1] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Peak Equalization ===
    print(f"\n[P2] Peak equalization (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 450, time.monotonic() + 250)
    a2, sc2 = _peak_equalization(best_seq, p2_deadline, min_time_remaining=30)
    if sc2 < best_score:
        best_seq = a2.copy()
        best_score = sc2
        print(f"  Updated: {best_score:.10f} [P2] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Second L-BFGS from improved point ===
    print(f"\n[P3] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 300, time.monotonic() + 200)
    k_sched_fine = [
        (0.01, 0.003), (0.005, 0.002), (0.003, 0.001),
        (0.001, 0.0005), (0.0005, 0.0003),
    ]
    a3 = _run_lbfgs_schedule(best_seq, p3_deadline, min_time_remaining=30,
                             k_schedule=k_sched_fine)
    sc3 = _score(a3)
    if sc3 < best_score:
        best_seq = a3.copy()
        best_score = sc3
        print(f"  Updated: {best_score:.10f} [P3] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Iterative Target Reduction ===
    print(f"\n[P4] Iterative target (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 200, time.monotonic() + 150)
    a4, sc4 = _iterative_target(best_seq, p4_deadline, min_time_remaining=20)
    if sc4 < best_score:
        best_seq = a4.copy()
        best_score = sc4
        print(f"  Updated: {best_score:.10f} [P4] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Fine Polish ===
    print(f"\n[P5] Fine polish (t={time.monotonic()-t_start:.1f}s)")
    p5_deadline = min(deadline - 100, time.monotonic() + 100)
    a5, sc5 = _fine_polish(best_seq, p5_deadline, min_time_remaining=10)
    if sc5 < best_score:
        best_seq = a5.copy()
        best_score = sc5
        print(f"  Updated: {best_score:.10f} [P5] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Diversification ===
    print(f"\n[P6] Diversification (t={time.monotonic()-t_start:.1f}s)")
    n_main = len(best_seq)

    # 6a: Perturb + quick L-BFGS
    if time.monotonic() < deadline - 80:
        sigmas = [0.02, 0.04, 0.06, 0.03, 0.05, 0.01, 0.08, 0.015]
        for i, sig in enumerate(sigmas):
            if time.monotonic() > deadline - 40:
                break
            a_pert = _perturb_log(best_seq, sig, rng)
            a_r = _run_lbfgs_schedule(a_pert, min(deadline - 30, time.monotonic() + 25),

                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  Updated: {best_score:.10f} [perturb_{i}, sig={sig}] "
                      f"t={time.monotonic()-t_start:.1f}s")

    # 6b: Try different profiles
    if time.monotonic() < deadline - 60:
        profiles = ["hann", "flat_top", "sin_p", "gauss", "cos_taper"]
        for prof_kind in profiles:
            if time.monotonic() > deadline - 30:
                break
            a_prof = _make_profile(n_main, prof_kind)
            # Mix with best
            a_mixed = _mix_profiles(best_seq, a_prof, 0.5)
            a_mixed = _normalize(a_mixed)
            a_p = _run_lbfgs_schedule(a_mixed, min(deadline - 25, time.monotonic() + 20),
                                      min_time_remaining=5)
            sc_p = _score(a_p)
            if sc_p < best_score:
                best_seq = a_p.copy()
                best_score = sc_p
                print(f"  Updated: {best_score:.10f} [profile_{prof_kind}] "
                      f"t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 50:
        n_options = [12288, 24576, 8192]
        n_options = [x for x in n_options if x != n_main]
        for n_try in n_options:
            if time.monotonic() > deadline - 25:
                break
            a_res = _resample(best_seq, n_try)
            a_opt = _run_lbfgs_schedule(a_res, min(deadline - 20, time.monotonic() + 30),
                                        min_time_remaining=5)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  Updated: {best_score:.10f} [n={n_try}] "
                      f"t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    # Peak equalization one more time
    if time.monotonic() < deadline - 15:
        a_fe, sc_fe = _peak_equalization(best_seq, min(deadline - 10, time.monotonic() + 40),
                                         min_time_remaining=5)
        if sc_fe < best_score:
            best_seq = a_fe.copy()
            best_score = sc_fe
            print(f"  Updated: {best_score:.10f} [final_eq] t={time.monotonic()-t_start:.1f}s")

    # Fine polish
    if time.monotonic() < deadline - 5:
        a_fp, sc_fp = _fine_polish(best_seq, deadline - 2, min_time_remaining=2)
        if sc_fp < best_score:
            best_seq = a_fp.copy()
            best_score = sc_fp
            print(f"  Updated: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")

    # Tiny L-BFGS at the end
    remaining = deadline - 3 - time.monotonic()
    if remaining > 5:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        beta_fin = max(mb * 0.0001, 1e-10)
        K_fin = max(3, int(n * 0.0005))
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        prob = _TopkSmoProb(beta_fin, K_fin, n)
        max_iter = min(5000, max(50, int(remaining / 0.02)))
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',

            a_fin2 = a_fin2 / a_fin2.sum()
