        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _hess_apply(w, v):
    """Apply Hessian of sum_k w[k]*(a*a)[k] to v. H[i,j] = 2*w[i+j]."""
    n = len(v)
    v_rev = v[::-1]
    wc = _full_convolve(w, v_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _topk_smoobj_grad(z, beta, K, n):
    z_max = z.max()
    a = np.exp(z - z_max)
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        c_top = c
    
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    
    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


class _TopkSmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def fval(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[0]

    def fgrad(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[1]


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.008)
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    schedule = [
        (1.0, 0.12), (0.7, 0.09), (0.5, 0.07), (0.3, 0.05),
        (0.2, 0.04), (0.1, 0.03), (0.05, 0.02), (0.03, 0.015),
        (0.015, 0.01), (0.008, 0.007), (0.004, 0.005), (0.002, 0.003),
        (0.001, 0.002), (0.0005, 0.0015),
    ]
    
    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        beta = base_beta * beta_mult
        K = max(5, int((2 * n - 1) * K_frac))
        
        prob = _TopkSmoProb(beta, K, n)
        est_time = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(50, int(remaining / (est_time * 25))))
        
        try:
            result = minimize(
                prob.fval, z, jac=prob.fgrad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 128, 'maxls': 50}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _optimal_subgradient_direction(a, active_idx):
    """Find optimal averaging of subgradients for active set."""
    n = len(a)
    n_act = len(active_idx)
    
    if n_act == 1:
        w = np.zeros(2*n-1)
        w[active_idx[0]] = 1.0
        return _grad_maxconv(a, w)
    
    # Compute gradient matrix G (n × n_act)
    G = np.zeros((n, n_act))
    for gi, k in enumerate(active_idx):
        e_k = np.zeros(2*n-1)
        e_k[k] = 1.0
        G[:, gi] = _grad_maxconv(a, e_k)
    
    # Center (remove mean to respect sum constraint)
    G_mean = G.mean(axis=0)
    Gc = G - G_mean[np.newaxis, :]
    
    # Solve min_p ||Gc^T p||^2 s.t. p >= 0, sum p = 1
    # For small n_act, use direct QP
    if n_act <= 200:
        # Gram matrix
        Hg = Gc.T @ Gc  # n_act × n_act
        # KKT: 2*Hg*p + lambda*1 + mu = 0
        # Try all subsets? No, too expensive. Use projected gradient.
        p = np.ones(n_act) / n_act
        for _ in range(50):
            g = 2.0 * (Hg @ p)
            p_new = p - 0.01 * g
            p_new = np.maximum(p_new, 0)
            s = p_new.sum()
            if s < 1e-15:
                break
            p_new /= s
            if np.linalg.norm(p_new - p) < 1e-12:
                break
            p = p_new
    else:
        p = np.ones(n_act) / n_act
    
    direction = -(Gc.T @ p)
    return direction


def _linearized_peak_equalization(a, active_idx, max_iter=100):
    """Solve linearized system to equalize active peaks."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    # Target: reduce all active peaks to (1-epsilon) * c_max
    target = c_max * 0.999
    
    # Build system: 2*(a*d)[k] = target - c[k] for k in active_idx
    # plus sum(d) = 0
    n_act = len(active_idx)
    
    # M is (n_act+1) × n
    # Row k: 2*a reversed and shifted
    # Actually M[k,j] = 2*a[j-k] (with a out of range = 0)
    # Wait: (a*d)[k] = sum_j a[j] d[k-j]... no.
    # (a*d)[k] = sum_i a[i] d[k-i]
    # d/d a_t of [(a+d')(a+d')[k]] linearized = d'(k-t) + d'(t) ... 
    # Let me just use: the gradient of (a*a)[k] at point a in direction d is 2*(conv of a and d) at position k.
    # Actually: d/da_j [(a*a)[k]] = 2*a[k-j] (as derived).
    # So the linearized change in (a*a)[k] when we add t*d to a is: 2*t*sum_j a[k-j]*d[j] = 2*t*(a_rev * d_rev)... 
    
    # Let me just compute the matrix directly for small n_act
    # M[k, j] = 2 * a[k-j] if 0 <= k-j < n, else 0
    
    # For efficiency, compute MM^T directly
    # (MM^T)[k1, k2] = sum_j M[k1,j] * M[k2,j] = 4 * sum_j a[k1-j] * a[k2-j]
    
    n_sys = n_act + 1
    MMt = np.zeros((n_sys, n_sys))
    b = np.zeros(n_sys)
    
    for i1 in range(n_act):
        k1 = active_idx[i1]
        b[i1] = target - c[k1]
        for i2 in range(n_act):
            k2 = active_idx[i2]
            # sum_j a[k1-j] * a[k2-j] where j ranges over [0, n-1] and k1-j, k2-j in [0,n-1]
            # = sum_j a[k1-j] a[k2-j]
            # Let l = k1-j: j = k1-l, range: 0<=k1-l<n and 0<=l<n and 0<=k2-(k1-l)<n
            # = sum_l a[l] a[l+(k2-k1)] (with appropriate bounds)
            # This is the autocorrelation of a at lag (k2-k1)
            d = k2 - k1
            if abs(d) < n:
                if d >= 0:
                    # sum_{l=0}^{n-1-d} a[l] * a[l+d]
                    MMt[i1, i2] = 4.0 * np.dot(a[:n-d], a[d:n])
                else:
                    d = -d
                    MMt[i1, i2] = 4.0 * np.dot(a[d:n], a[:n-d])
            else:
                MMt[i1, i2] = 0.0
        # Cross term with sum constraint row
        # sum_j 2*a[k1-j] * 1 = 2 * sum_{j: 0<=k1-j<n} a[k1-j] = 2 * sum of a over valid range
        lo = max(0, k1 - (n-1))
        hi = min(n-1, k1)
        MMt[i1, n_act] = 2.0 * np.sum(a[lo:hi+1])
        MMt[n_act, i1] = MMt[i1, n_act]
    
    MMt[n_act, n_act] = float(n)
    b[n_act] = 0.0
    
    # Solve MMt * y = b, then d = M^T * y
    try:

        MMt_reg = MMt + 1e-10 * np.eye(n_sys)
