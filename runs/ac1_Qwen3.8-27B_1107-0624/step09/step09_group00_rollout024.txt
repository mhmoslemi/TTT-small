def _next_pow2(n):

    r = np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):

def _active_set_gradient(a, c, tol_frac):
    """Compute weighted active-set gradient for L-infinity convolution."""
    n = len(a)
    c_max = float(c.max())
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    n_active = int(active.sum())
    if n_active == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        n_active = 1

    eps = 1e-12 * c_max
    w = np.zeros_like(c)
    w[active] = 1.0 / (c_max - c[active] + eps)
    w = w / w.sum()

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad = 2.0 * h
    grad = grad - np.mean(grad)
    return grad

def _saxton_init(n, rng, n_iter=50):
    """Gerchberg-Saxton spectral equalization for flat-spectrum initialization."""
    N = _next_pow2(2 * n - 1)
    # Start with random phases, fixed magnitudes
    phases = rng.random(N // 2 + 1) * 2 * np.pi
    A = np.ones(N // 2 + 1, dtype=complex)
    A[0] = 1.0  # DC component
    A[1:] = np.exp(1j * phases[1:])

    for _ in range(n_iter):
        a_full = np.fft.irfft(A, N)
        a = np.maximum(a_full[:n], 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

        A_new = np.fft.rfft(a, N)
        # Equalize magnitudes (keep DC)
        mag_target = float(np.mean(np.abs(A_new[1:] ** 2))) ** 0.5
        if mag_target < 1e-15:
            mag_target = 1.0 / np.sqrt(n)
        A_new[0] = A_new[0]  # keep DC
        A_new[1:] = mag_target * np.exp(1j * np.angle(A_new[1:]))
        A = A_new

    return _normalize(a)


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "two_bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        # Cosine-modulated flat top
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        if rng is None:
            rng = np.random.default_rng(42)
        a = rng.random(n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def f(self, z):
        return _smoobj(z, self.beta, self.n)

    def g(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=20.0):
    """L-BFGS on log-softmax with adaptive beta annealing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015,
                                  0.008, 0.004, 0.002, 0.001, 0.0005,
                                  0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 20))))

        prob = _LbfgsProblem(beta, n)
        try:
            result = minimize(
                prob.f, z, jac=prob.g, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 80, 'maxls': 40}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _hybrid_optimize(a_init, deadline, min_time_remaining=15.0):
    """Two-phase: L2 flatten then L-infinity polish."""
    n = len(a_init)
    # Phase A: L2 spectral flattening
    l2_deadline = min(deadline - min_time_remaining, time.monotonic() + 10)
    if time.monotonic() < l2_deadline:
        a_l2, sc_l2 = _l2_flatten(a_init, l2_deadline, lr_init=0.03,
                                   min_time_remaining=2.0)
    else:
        a_l2, sc_l2 = a_init, _score(a_init)

    # Phase B: L-infinity polish
    if time.monotonic() < deadline - 3:
        a_pol, sc_pol = _active_set_polish(
            a_l2, deadline - 2, tol_start=0.005, tol_end=0.0005,
            lr_init=0.03, momentum=0.82)
        if sc_pol <= sc_l2:
            return a_pol, sc_pol
        else:
            return a_l2, sc_l2
    return a_l2, sc_l2


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _ils(a_init, deadline, rng, min_time_remaining=40.0):
    """Iterated Local Search with diverse perturbations and hybrid sub-optimization."""
    n = len(a_init)
    current_a = _normalize(a_init.copy())
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc

    strategies = [
        ('global_small', 0.003),
        ('global_med', 0.015),
        ('block', 0.05),
        ('sparse', 0.1),
        ('mirror_seg', 0.0),
        ('global_tiny', 0.001),
        ('block_small', 0.03),
        ('gaussian_mix', 0.0),
    ]

    attempt = 0
    while time.monotonic() + min_time_remaining < deadline:
        kind, sigma = strategies[attempt % len(strategies)]

        if kind == 'global_small' or kind == 'global_med' or kind == 'global_tiny':
            z = np.log(np.maximum(current_a, 1e-30))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif kind == 'block' or kind == 'block_small':
            block_size = max(30, n // (10 if kind == 'block' else 6))
            start = int(rng.integers(0, n - block_size))
            a_pert = current_a.copy()
            z = np.log(np.maximum(a_pert[start:start + block_size], 1e-30))
            z = z + sigma * rng.standard_normal(block_size)
            a_pert[start:start + block_size] = np.exp(z - np.max(z))
            a_pert = _normalize(a_pert)
        elif kind == 'sparse':
            n_pert = max(15, n // 20)
            idx = rng.choice(n, size=n_pert, replace=False)
            a_pert = current_a.copy()
            factors = np.exp(sigma * rng.standard_normal(n_pert))
            a_pert[idx] *= factors
            a_pert = _normalize(a_pert)
        elif kind == 'mirror_seg':
            seg_len = max(20, n // 15)
            start = int(rng.integers(0, n - seg_len))
            a_pert = current_a.copy()
            seg = a_pert[start:start + seg_len].copy()
            a_pert[start:start + seg_len] = seg[::-1]
            # Small noise
            z = np.log(np.maximum(a_pert, 1e-30))
            z = z + 0.002 * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif kind == 'gaussian_mix':
            # Mix current with a Gaussian profile
            t = np.linspace(0, 1, n)
            gauss = np.exp(-3.0 * (t - 0.5) ** 2)
            gauss = gauss / gauss.sum()
            mix = rng.random() * 0.3
            a_pert = (1 - mix) * current_a + mix * gauss
            a_pert = _normalize(a_pert)
        else:
            a_pert = current_a.copy()

        opt_deadline = min(deadline - min_time_remaining, time.monotonic() + 6)
        a_opt, sc_opt = _hybrid_optimize(a_pert, opt_deadline, min_time_remaining=2)

        if sc_opt < best_sc - 1e-13:
            best_sc = sc_opt
            best_a = a_opt.copy()
            current_a = a_opt.copy()
            print(f"    ILS NEW BEST: {best_sc:.10f} ({kind}) t={time.monotonic() - t_start:.0f}s")
        elif sc_opt < current_sc - 1e-13:
            current_a = a_opt.copy()

        attempt += 1

    return best_a, best_sc

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L2 spectral flattening + active-set L-infinity polish
    + L-BFGS + ILS + multi-n exploration for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    """
    global t_start
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    n_main = n_parent if n_parent > 0 else 16384

    # === Phase 1: L2 spectral flattening from parent + diverse starts (250s) ===
    print(f"\n[P1] L2 spectral flatten (t={time.monotonic() - t_start:.1f}s)")
    ms_deadline = t_start + min(250, (deadline - t_start) // 5)

    starts = []
    starts.append(('parent', parent.copy()))
    starts.append(('saxton', _saxton_init(n_main, rng, n_iter=30)))
    starts.append(('cos_mod', _make_profile(n_main, "cos_mod")))
    starts.append(('two_bump', _make_profile(n_main, "two_bump")))
    starts.append(('gaussian', _make_profile(n_main, "gaussian")))

    for name, a_init in starts:
        if time.monotonic() > ms_deadline - 30:
            break
        sc_init = _score(a_init)
        a_opt, sc_opt = _l2_flatten(a_init, ms_deadline, lr_init=0.04,
                                     min_time_remaining=40)
        print(f"  {name}: init={sc_init:.10f}, l2_opt={sc_opt:.10f}")
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [l2/{name}] t={time.monotonic() - t_start:.1f}s")

    # === Phase 2: Active-set L-infinity polish on best (200s) ===
    print(f"\n[P2] Active-set polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        pol_deadline = min(deadline - 400, time.monotonic() + 200)
        a_pol, sc_pol = _active_set_polish(
            best_seq, pol_deadline, tol_start=0.008, tol_end=0.00003,
            lr_init=0.03, momentum=0.84)
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  NEW BEST: {best_score:.10f} [active_set] t={time.monotonic() - t_start:.1f}s")

    # === Phase 3: L-BFGS with log-sum-exp annealing (200s) ===
    print(f"\n[P3] L-BFGS annealing (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        lbfgs_deadline = min(deadline - 400, time.monotonic() + 200)
        a_lbfgs = _run_lbfgs(best_seq, lbfgs_deadline, min_time_remaining=50)
        sc = _score(a_lbfgs)
        if sc < best_score:
            best_seq = a_lbfgs.copy()
            best_score = sc
            print(f"  NEW BEST: {best_score:.10f} [lbfgs] t={time.monotonic() - t_start:.1f}s")

    # === Phase 4: ILS with hybrid sub-optimizer (350s) ===
    print(f"\n[P4] ILS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        ils_deadline = min(deadline - 250, time.monotonic() + 350)
        a_ils, sc_ils = _ils(best_seq, ils_deadline, rng, min_time_remaining=60)
        if sc_ils < best_score:
            best_seq = a_ils.copy()
            best_score = sc_ils
            print(f"  NEW BEST: {best_score:.10f} [ils] t={time.monotonic() - t_start:.1f}s")

    # === Phase 5: Multi-n exploration (250s) ===
    print(f"\n[P5] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [n_main]
    # Add nearby sizes and some alternatives
    for offset in [-2048, -1024, 1024, 2048, 4096, -4096, 8192]:
        nn = n_main + offset
        if nn >= 4096 and nn <= 65536:
            n_candidates.append(nn)
    # Add absolute sizes
    for nn in [12288, 20480, 24576, 32768, 40960, 16384, 8192]:
        if nn not in n_candidates:
            n_candidates.append(nn)

    seen = set()
    n_unique = []
    for nn in n_candidates:
        if nn not in seen:
            seen.add(nn)
            n_unique.append(nn)

    # Sort by proximity to current best
    n_unique.sort(key=lambda x: abs(x - len(best_seq)))

    for n_try in n_unique:
        if time.monotonic() > deadline - 150:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)

        a_opt, sc_opt = _hybrid_optimize(a_init, deadline, min_time_remaining=60)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic() - t_start:.1f}s")
        else:
            print(f"  n={n_try}: init={sc_init:.10f}, opt={sc_opt:.10f} (no improvement)")

    # === Phase 6: Second ILS round (150s) ===
    print(f"\n[P6] ILS round 2 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        ils2_deadline = min(deadline - 80, time.monotonic() + 150)
        rng2 = np.random.default_rng(int(seed) + 777)
        a_ils2, sc_ils2 = _ils(best_seq, ils2_deadline, rng2, min_time_remaining=40)
        if sc_ils2 < best_score:
            best_seq = a_ils2.copy()
            best_score = sc_ils2
            print(f"  NEW BEST: {best_score:.10f} [ils2] t={time.monotonic() - t_start:.1f}s")

    # === Phase 7: Final ultra-fine polish (80s) ===
    print(f"\n[P7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _active_set_polish(
            best_seq, deadline - 3, tol_start=0.002, tol_end=0.000002,
            lr_init=0.008, momentum=0.78)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic() - t_start:.1f}s")

        # Final L-BFGS with very small beta
        remaining = deadline - 3 - time.monotonic()
        if remaining > 15:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.000005
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _LbfgsProblem(beta_fin, n)
            max_iter = min(3000, max(50, int(remaining / 0.04)))
            try:
                result = minimize(prob.f, z, jac=prob.g, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-20,
                                           'gtol': 1e-16, 'maxcor': 80})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic() - t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
