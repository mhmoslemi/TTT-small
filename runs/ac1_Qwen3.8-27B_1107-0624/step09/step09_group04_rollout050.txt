softmax-smoothed L-BFGS, Nesterov-accelerated active-set gradient descent,
and multi-scale exploration to minimize 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _quartic_obj(z, n):
    """Smooth surrogate: sum of squares of autocorrelation, parameterized by log-softmax."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    return float(np.sum(c * c))

def _quartic_grad(z, n):
    """Gradient of sum(c^2) w.r.t. z (softmax parameterization)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    # d(sum c^2)/da = 2 * (c * a_rev) shifted to [n-1:2n-1]
    # where c * a_rev is the convolution of c with reversed a
    # Actually: d/dc_k (sum_j c_j^2) = 2 c_k
    # da_i/da = ... 
    # d(sum c^2)/da_i = 2 * sum_k c_k * dc_k/da_i = 2 * sum_k c_k * 2*a_{k-i}
    # = 4 * (c * a_rev)[n-1+i] ... let me think more carefully.
    # 
    # dc_k/da_i = 2*a_{k-i} (as derived earlier)
    # d(sum_k c_k^2)/da_i = 2*sum_k c_k * dc_k/da_i = 4*sum_k c_k * a_{k-i}
    # = 4 * (c * a_rev)[n-1+i]  (where a_rev = a[::-1])
    #
    # Wait: sum_k c_k * a_{k-i} = (c * a_rev)[(k-i) + (n-1)] ... 
    # Let me use the convolution identity: (c * a_rev)[m] = sum_k c_k * a_rev[m-k] = sum_k c_k * a[n-1-(m-k)]
    # We want sum_k c_k * a_{k-i}. Let j = k-i, so = sum_j c_{j+i} * a_j = (c_shifted * a)[i]
    # Hmm, let me just compute it directly using FFT.
    
    # d(sum c^2)/da[i] = 4 * sum_k c[k] * a[k-i] (where a[k-i]=0 if k-i outside [0,n-1])
    # = 4 * (flip(c) * a)[n-1+i] ... 
    
    # Actually, let's use: sum_k c[k] * a[k-i] = (c_flipped * a)[n-1-i + n-1] ... 
    # I'll just compute it via FFT: correlate c with a
    # corr[i] = sum_k c[k] * a[k-i] = (c * flip(a))[i + n - 1]
    a_rev = a[::-1]
    h = _full_convolve(c, a_rev)  # h[m] = sum_k c[k] * a_rev[m-k]
    # h[m] = sum_k c[k] * a[n-1-(m-k)] = sum_k c[k] * a[n-1-m+k]
    # We want sum_k c[k] * a[k-i]. Let n-1-m+k = k-i => k-i = n-1-m+k => i = m - (n-1) => m = i + n - 1
    # So d(sum c^2)/da[i] = 4 * h[i + n - 1]
    
    grad_a = 4.0 * h[n-1:2*n-1]
    
    # Chain rule: da/dz = diag(a) - outer(a, a) (for softmax)
    # dz: grad_z[i] = grad_a[i] * a[i] - a[i] * sum_j grad_a[j] * a[j]
    W = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - W)
    return grad_z

def _run_quartic_lbfgs(a_init, deadline, min_time_remaining=60.0):
    """Optimize the quartic surrogate using L-BFGS."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    class QuarticProb:
        def __call__(self, z):
            return _quartic_obj(z, n)
        def grad(self, z):
            return _quartic_grad(z, n)
    
    prob = QuarticProb()
    try:
        result = minimize(
            prob, z, jac=prob.grad, method='L-BFGS-B',
            options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-14, 'maxcor': 100, 'maxls': 50}
        )
        a_new = _softmax(result.x)
        return a_new
    except Exception:
        return a_init

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _run_smo_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0):
    """L-BFGS on the smoothed max objective."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003) * beta_scale
    
    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005,
        0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005
    ])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        if beta < base_beta * 0.01:
            max_iter = min(8000, max(100, int(remaining / 0.05)))
        else:
            max_iter = min(5000, max(50, int(remaining / 0.04)))
        
        class Prob:
            def __init__(self, beta, n):
                self.beta = beta
                self.n = n
            def __call__(self, z):
                return _smoobj(z, self.beta, self.n)
            def grad(self, z):
                return _smograd(z, self.beta, self.n)
        
        prob = Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-17, 'gtol': 1e-15, 'maxcor': 100, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

def _nesterov_active_set(a_init, deadline, min_time_remaining=30.0):
    """Nesterov-accelerated projected gradient on the active set."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    v = np.zeros(n)  # momentum
    beta_mom = 0.9  # momentum coefficient
    lr = 0.01
    K = 10  # active set size
    no_improve = 0
    max_no_improve = 100
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            lr = min(0.05, lr * 1.02)
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
            lr *= 0.95
            if no_improve % 30 == 0:
                K = max(3, K // 2)
        
        # Active set: top K peaks
        top_k = np.argsort(c)[::-1][:K]
        
        # Compute gradient: average of active peak gradients
        # grad[i] = (2/K) * sum_{k in top_K} a[k-i] (with boundary)
        w_S = np.zeros(2 * n - 1)
        w_S[top_k] = 1.0 / K
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        
        # Project gradient to sum-zero subspace
        grad = grad - np.mean(grad)
        
        # Nesterov step
        v = beta_mom * v + grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Check improvement (every 10 iterations to save time)
        if no_improve % 10 == 0 or no_improve < 5:
            c_new = _full_convolve(a_new, a_new)
            sc_new = 2.0 * n * float(c_new.max())
            if sc_new < current_sc:
                a = a_new
            else:
                v = beta_mom * v  # decay momentum when rejected
        else:
            a = a_new  # accept unconditionally for momentum to build
    
    return best_a, best_sc

def _active_set_equalize(a_init, deadline, min_time_remaining=30.0):
    """Iterative peak equalization with backtracking line search."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    K = 5
    lr = 0.03
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            if K < min(50, n):
                K += 1
        else:
            no_improve += 1
            lr *= 0.9
            if no_improve > 50:
                K = max(3, K // 2)
            if no_improve > 200:
                break
        
        top_k = np.argsort(c)[::-1][:K]
        w_S = np.zeros(2 * n - 1)
        w_S[top_k] = 1.0 / K
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h - 2.0 * np.mean(2.0 * h)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Backtracking line search
        alpha = lr / (gnorm + 1e-15)
        improved = False
        for _ in range(40):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if improved:
            lr = min(0.08, lr * 1.01)
    
    return best_a, best_sc

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal_asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.3) ** 2) + 0.6 * np.exp(-15.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    else:
        return np.ones(n) / n

Wait, I realize I have a potential issue with the quartic gradient. Let me double-check.

We want to minimize $f(a) = \sum_k (a*a)_k^2$ where $a$ is parameterized as $a = \text{softmax}(z)$.

$\frac{\partial f}{\partial a_i} = 2 \sum_k (a*a)_k \cdot \frac{\partial (a*a)_k}{\partial a_i} = 2 \sum_k c_k \cdot 2a_{k-i} = 4 \sum_k c_k a_{k-i}$

where $a_j = 0$ for $j < 0$ or $j \geq n$.

Now, $\sum_k c_k a_{k-i}$: Let's compute this.

Define $h[m] = \sum_k c_k \cdot a_{rev}[m-k]$ where $a_{rev}[j] = a[n-1-j]$.

$h[m] = \sum_k c_k \cdot a[n-1-(m-k)] = \sum_k c_k \cdot a[n-1-m+k]$

We want $\sum_k c_k \cdot a[k-i]$. Setting $n-1-m+k = k-i$ gives $n-1-m = -i$, so $m = i + n - 1$.

Therefore: $\sum_k c_k \cdot a[k-i] = h[i + n - 1]$.

And $h = c * a_{rev}$ (full convolution).

So $\frac{\partial f}{\partial a_i} = 4 \cdot (c * a_{rev})[i + n - 1]$ for $i = 0, \ldots, n-1$.

Then the chain rule for softmax:
$\frac{\partial f}{\partial z_i} = \frac{\partial f}{\partial a_i} a_i - a_i \sum_j \frac{\partial f}{\partial a_j} a_j = a_i (g_i - \sum_j g_j a_j)$

where $g_i = 4 (c * a_{rev})[i + n - 1]$.

"""
Hybrid optimizer combining quartic surrogate pre-optimization,
softmax-smoothed L-BFGS with adaptive temperatures, Nesterov-accelerated
active-set gradient descent with adaptive active set, and multi-scale
exploration to minimize 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _quartic_obj(z, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return float(np.sum(c * c))


def _quartic_grad_fn(z, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    a_rev = a[::-1]
    h = _full_convolve(c, a_rev)
    grad_a = 4.0 * h[n - 1:2 * n - 1]
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)


def _run_quartic_lbfgs(a_init, deadline, min_time_remaining=30.0):

def _smoobj_binary(z, beta, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())

    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd_binary(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0

    h = wc[n - 1:2 * n - 1]
