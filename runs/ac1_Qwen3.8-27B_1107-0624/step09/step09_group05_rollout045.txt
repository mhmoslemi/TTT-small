Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) L-BFGS with adaptive soft-max beta, (2) Active-set QP with 
FFT-based mat-vec and FISTA acceleration, (3) Multi-n exploration.

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)

def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

def _qp_matvec(S, a, n):
    """Compute Q_S @ a where Q_S[j,m] = S[j+m]/|S|.
    Result: (S * a_rev)[n-1 : 2n-1] / |S|
    """
    a_rev = a[::-1]
    S_a = _full_convolve(S, a_rev)
    S_sum = S.sum()
    if S_sum < 1e-30:
        return np.zeros(n)
    return S_a[n-1:2*n-1] / S_sum

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-30:
        return np.ones(n) / n
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / w.sum()

def _fista_qp(a_init, S, deadline, min_time=30):
    """Solve min a^T Q_S a s.t. a >= 0, sum(a) = 1 using FISTA.
    
    Q_S mat-vec: Q_S @ a = conv(S, a_rev)[n-1:2n-1] / |S|
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    
    S_sum = S.sum()
    if S_sum < 1:
        S = S.astype(float)
        S = S / S.sum()
        S_sum = 1.0
    
    # Lipschitz constant estimate: 2 * max eigenvalue of Q_S
    # Upper bound: 2 * max_k S[k] ... actually for a Hankel matrix from S,
    # the max eigenvalue is bounded by max_row_sum
    # Row sum of Q_S at row j: sum_m S[j+m]/S_sum
    # = (sum of S from j to j+n-1) / S_sum
    # Upper bound: 1 (since all elements of S are non-negative and sum to S_sum)
    # Actually the row sum is at most 1 (it's a fraction of S_sum)
    # So L <= 2 (for the objective a^T Q_S a, gradient is 2*Q_S*a, 
    # so Lipschitz of gradient is 2*lambda_max(Q_S) <= 2*max_row_sum <= 2)
    
    L = 2.0  # Conservative upper bound on Lipschitz constant of gradient
    step = 1.0 / L
    
    # FISTA parameters
    t_k = 1.0
    y = a.copy()
    
    best_a = a.copy()
    best_obj = None
    
    max_iter = 5000
    
    for it in range(max_iter):
        if time.monotonic() + min_time > deadline:
            break
        
        # Compute gradient at y
        grad_y = 2.0 * _qp_matvec(S, y, n)
        
        # Proximal gradient step
        z = y - step * grad_y
        # Project onto simplex
        x = _project_simplex(z)
        
        # Evaluate objective at x
        c = _full_convolve(x, x)
        obj = float(c @ S / S_sum)  # = a^T Q_S a
        if best_obj is None or obj < best_obj:
            best_obj = obj
            best_a = x.copy()
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x + ((t_k - 1.0) / t_new) * (x - a)

Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) L-BFGS with adaptive soft-max beta, (2) Active-set QP with
FFT-based mat-vec and FISTA acceleration, (3) Multi-n exploration.

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())

    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if float(v.sum()) < 1e-30:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = float(w.sum())
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = float(a_new.sum())
    if s > 1e-15:
        a_new = a_new / s
    return a_new

        if parent_raw.size > 0:

            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(16384, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n=16384: score={best_score:.10f}")

    n_main = len(best_seq)

    # === Phase 1: L-BFGS ===
    print(f"\n[P1] L-BFGS main (n={n_main}, t={time.monotonic()-t0:.0f}s)")
    p1_end = t0 + min(550, (budget_s - 10) * 0.35)
    a1, sc1 = _lbfgs_softmax(best_seq, p1_end, min_time=200, verbose=True, label="P1")
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  P1 result: {best_score:.10f}")

    # === Phase 2: Active-set QP ===
    print(f"\n[P2] Active-set QP (t={time.monotonic()-t0:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p2_end = min(deadline - 400, time.monotonic() + 300)
        a2, sc2 = _active_set_loop(best_seq, p2_end, min_time=100)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(f"  P2 result: {best_score:.10f}")

    # === Phase 3: L-BFGS round 2 ===
    print(f"\n[P3] L-BFGS round 2 (t={time.monotonic()-t0:.0f}s)")
    if time.monotonic() + 40 < deadline:
        p3_end = min(deadline - 250, time.monotonic() + 350)
        a3, sc3 = _lbfgs_softmax(best_seq, p3_end, min_time=100, verbose=True, label="P3")
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3 result: {best_score:.10f}")

    # === Phase 4: Subgradient + L-BFGS ===
    print(f"\n[P4] Subgradient (t={time.monotonic()-t0:.0f}s)")
    if time.monotonic() + 40 < deadline:
        p4_end = min(deadline - 150, time.monotonic() + 120)
        a4, sc4 = _subgradient_refine(best_seq, p4_end, min_time=50)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4a result: {best_score:.10f}")
        if time.monotonic() + 30 < deadline:
            p4b_end = min(deadline - 80, time.monotonic() + 100)
            a4b, sc4b = _lbfgs_softmax(best_seq, p4b_end, min_time=40, verbose=False, label="P4b")
            if sc4b < best_score:
                best_score = sc4b
                best_seq = a4b.copy()
                print(f"  P4b result: {best_score:.10f}")

    # === Phase 5: Multi-n ===
    print(f"\n[P5] Multi-n (t={time.monotonic()-t0:.0f}s)")
    n_options = [12288, 20480, 24576, 32768]
    n_options = [x for x in n_options if x != n_main]
    for n_try in n_options:
        if time.monotonic() + 80 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if time.monotonic() + 60 > deadline:
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
            continue
        p5_end = min(deadline - 60, time.monotonic() + 80)
        a5, sc5 = _lbfgs_softmax(a_r, p5_end, min_time=30, verbose=False, label=f"n={n_try}")
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  n={n_try}: {best_score:.10f}")

    # === Phase 6: Final polish ===
    print(f"\n[P6] Final polish (t={time.monotonic()-t0:.0f}s)")
    if time.monotonic() + 20 < deadline:
        p6_end = min(deadline - 5, time.monotonic() + 150)
        a6, sc6 = _lbfgs_softmax(best_seq, p6_end, min_time=5, verbose=False, label="P6")
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()

        if time.monotonic() + 15 < deadline:
            p6b_end = min(deadline - 3, time.monotonic() + 50)
            a6b, sc6b = _active_set_loop(best_seq, p6b_end, min_time=3)
            if sc6b < best_score:
                best_score = sc6b
                best_seq = a6b.copy()
                print(f"  P6b active-set: {best_score:.10f}")

    # Final
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t0:.1f}s")
    return [float(x) for x in final]
