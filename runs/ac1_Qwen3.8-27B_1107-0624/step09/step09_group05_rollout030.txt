def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)

    e = np.exp(np.clip(shifted, -700, 0))

        w[int(np.argmax(c))] = 1.0

    n = len(a)
    N = _next_pow2(2 * n - 1)
    a = a.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

        A = np.fft.rfft(a, N)
        P = np.abs(A)**2
        
        # Compute autocorrelation
        c = np.fft.irfft(P, N)[:2*n-1]
        c_max = float(c.max())
        sc = 2.0 * n * c_max

            best_a = a.copy()

            peaks[k_peak] = True
            n_peaks = 1

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS with adaptive soft-max, frequency-domain spectral shaping,
KKT active-set refinement, and multi-n exploration.
Key new techniques: FFT-based spectral peak shaving, non-uniform active-set weights,
and adaptive restart with perturbation to escape local minima.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_soft_structured(a, beta, n):
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft_structured(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)


class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)

def _spectral_shaping(a_init, deadline, min_time_remaining=30.0, verbose=False):
    """Frequency-domain peak reduction via iterative amplitude modification."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    N = _next_pow2(2 * n - 1)
    freqs = np.arange(N // 2 + 1)

    # Adaptive parameters
    alpha = 0.3  # compression strength
    n_iter_max = 500

    for it in range(n_iter_max):
        if time.monotonic() + min_time_remaining > deadline:
            break

        # FFT
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        phase = np.angle(A)

        # Autocorrelation
        c = np.fft.irfft(P, N)[:2 * n - 1]
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # Find peaks
        threshold = c_max * 0.999
        peak_idx = np.where(c >= threshold)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(c)])

        # Compute influence of each frequency on the peaks
        # c[k] = (1/N) * sum_m P[m] * cos(2*pi*m*k/N) (for real symmetric case)
        # Use FFT of the peak indicator
        peak_indicator = np.zeros(2 * n - 1)
        peak_indicator[peak_idx] = 1.0

        # Influence via correlation: influence[m] = Re(FFT(peak_indicator)[m])
        # But we need to be careful about the relationship
        # c[k] = IFFT(|A|^2)[k] = (1/N) sum_m |A[m]|^2 * exp(2*pi*i*m*k/N)
        # To reduce c at peak positions, reduce |A[m]|^2 where exp(2*pi*i*m*k/N) 
        # has positive real part (constructive) at those k positions.
        
        pi_fft = np.fft.rfft(peak_indicator, N)
        influence = np.real(pi_fft)  # [m] -> how much frequency m contributes to peaks
        
        # Normalize
        inf_max = np.abs(influence).max()
        if inf_max < 1e-15:
            break
        influence = influence / inf_max

        # Apply spectral compression: reduce P where influence is positive
        # Use a soft threshold to avoid discontinuities
        P_new = P.copy()
        constructive = influence > 0
        if constructive.any():
            # Reduce proportionally to influence
            reduction = alpha * influence[constructive]
            # Cap the reduction to not go below zero
            P_new[constructive] -= reduction * P[constructive]
            P_new = np.maximum(P_new, 0)

        # Reconstruct
        mag = np.sqrt(P_new)
        A_new = mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]

        # Project to non-negative, normalize
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        # Decay alpha slowly
        alpha *= 0.995

        if verbose and it % 100 == 0:
            print(f"    spectral iter {it}: score={best_sc:.10f}, c_max={c_max:.2e}")

    return best_a, best_sc

            a_try[support] -= eta * residual
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-12:
                a = a_try
                improvements += 1
                break

        if improvements == 0:
            tol_frac = max(min_tol, tol_frac * 0.85)
            if tol_frac <= min_tol * 1.05:
                break
        else:
            tol_frac = max(min_tol, tol_frac * 0.97)

    return best_a, best_sc


def _subgradient_polish(a_init, deadline, min_time_remaining=30.0):
    """Polyak subgradient method for final polish."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lb = 2.0 * n / (2 * n - 1)
    step_scale = 0.3
    t = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-12:
            best_sc = current_sc
            best_a = a.copy()

        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        gap = current_sc - lb
        if gap < 1e-10:
            break
        grad_c = grad - np.mean(grad)
        g_norm_sq = float(np.dot(grad_c, grad_c))
        if g_norm_sq < 1e-20:
            break
        step = step_scale * gap / g_norm_sq
        step = min(step, 0.05)

        a_new = a - step * grad_c
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a = a_new / s_new
        t += 1

        if t % 50 == 0:
            step_scale *= 0.95

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.8) ** 0.9
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.4) ** 2) + 0.003
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, eps, seed=0):
    """Add small random perturbation and reproject."""
    rng = np.random.default_rng(seed)
    n = len(a)
    noise = rng.standard_normal(n) * eps
    a_new = a + noise
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _iterative_active_set_optimal_weights(a, c, active_mask, n):
    """Solve for optimal lambda weights using conjugate gradient on the Gram matrix."""
    active_indices = np.where(active_mask)[0]
    m = len(active_indices)
    if m < 2:
        return active_mask.astype(float) / max(1, int(active_mask.sum()))

    # G[k,k'] = 4 * sum_j a[k-j] a[k'-j] for k, k' in active_indices
    # Compute G @ v efficiently using convolution
    # (G @ v)[i] = 4 * sum_j a[active[i]-j] * sum_{i'} v[i'] * a[active[i']-j]
    # = 4 * sum_j a[active[i]-j] * (conv_indicator)[j]
    # where conv_indicator[j] = sum_{i'} v[i'] * a[active[i']-j]
    
    # Alternative: (G @ v)[i] = 4 * sum_{i'} v[i'] * G[i,i']
    # G[i,i'] = 4 * sum_j a[k_i - j] * a[k_i' - j]
    # = 4 * <a_shifted_by_k_i, a_shifted_by_k_i'>
    
    # This is equivalent to: (G @ v)[i] = 4 * inner(a_win[k_i], (sum v[i'] * a_win[k_i']))
    # where a_win[k] is the vector [a[k-0], a[k-1], ..., a[k-(n-1)]] (padded)
    
    # Let me use the convolution trick:
    # Define b = 0, then for each active index k_i with weight v[i]:
    #   b += v[i] * a_shifted(k_i)
    # where a_shifted(k)[j] = a[k-j] (for valid j)
    # Then (G @ v)[i] = 4 * inner(a_shifted(k_i), b)
    # = 4 * sum_j a[k_i - j] * b[j]
    # = 4 * (a correlated with b) at position k_i
    
    # a correlated with b at position k = sum_j a[k-j] b[j]
    # = conv(a_rev, b)[k + n - 1] (approximately, with padding)
    
    # Let me just compute it directly for small active sets
    if m <= 200:
        # Direct computation
        Gv = np.zeros(m)
        for i in range(m):
            ki = active_indices[i]
            # G[i,:] = 4 * [sum_j a[ki-j] a[k_i'-j] for i' in range(m)]
            # = 4 * cross_correlation of a window at ki with a windows at k_i'
            # Simplify: G[i,i'] = 4 * sum_j a[ki-j] a[ki'-j]
            # Let's compute via the overlap
            lo = max(0, ki - (n - 1), active_indices[i'] - (n - 1)) if m > 1 else 0
            # Actually just iterate:
            for i_prime in range(m):
                ki_prime = active_indices[i_prime]
                # sum_j a[ki-j] * a[ki_prime-j] for valid j
                j_lo = max(0, ki - (n - 1), ki_prime - (n - 1))
                j_hi = min(n - 1, ki, ki_prime)
                if j_lo > j_hi:
                    Gv[i] += v[i_prime] * 0  # zero
                else:
                    Gv[i] += 4.0 * v[i_prime] * np.dot(a[ki - j_hi:ki - j_lo + 1], 
                                                         a[ki_prime - j_hi:ki_prime - j_lo + 1])
        return Gv
    else:
        # For large active sets, use uniform weights as approximation
        w = np.zeros(len(c))
        w[active_mask] = 1.0 / m
        return w


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    P1: Quick L-BFGS warm-start from parent
    P2: Spectral shaping (frequency domain peak reduction)
    P3: Refined L-BFGS with fine beta
    P4: Active-set KKT refinement
    P5: Multi-n exploration
    P6: Perturbation + re-optimization
    P7: Final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(16384, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n=16384: score={best_score:.10f}")

    n_main = len(best_seq)
    print(f"Starting: n={n_main}, score={best_score:.10f}")

    # === Phase 1: Quick L-BFGS ===
    t_elapsed = time.monotonic() - t_start
    print(f"\n[P1] L-BFGS warm-start (t={t_elapsed:.0f}s)")
    if time.monotonic() + 50 < deadline:
        p1_end = min(deadline - 400, time.monotonic() + 200)
        a1, sc1 = _run_lbfgs(best_seq, p1_end, min_time_remaining=300, verbose=True)
        if sc1 < best_score:
            best_score = sc1
            best_seq = a1.copy()
            print(f"  P1: {best_score:.10f}")

    # === Phase 2: Spectral Shaping ===
    t_elapsed = time.monotonic() - t_start
    print(f"\n[P2] Spectral shaping (t={t_elapsed:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p2_end = min(deadline - 300, time.monotonic() + 200)
        a2, sc2 = _spectral_shaping(best_seq, p2_end, min_time_remaining=250, verbose=True)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(f"  P2: {best_score:.10f}")
        # Quick L-BFGS after spectral
        if time.monotonic() + 30 < deadline:
            p2b_end = min(deadline - 250, time.monotonic() + 120)
            a2b, sc2b = _run_lbfgs(best_seq, p2b_end, min_time_remaining=200, verbose=False)
            if sc2b < best_score:
                best_score = sc2b
                best_seq = a2b.copy()
                print(f"  P2b: {best_score:.10f}")

    # === Phase 3: Refined L-BFGS ===
    t_elapsed = time.monotonic() - t_start
    print(f"\n[P3] Refined L-BFGS (t={t_elapsed:.0f}s)")
    if time.monotonic() + 40 < deadline:
        p3_end = min(deadline - 200, time.monotonic() + 250)
        a3, sc3 = _run_lbfgs(best_seq, p3_end, min_time_remaining=150, verbose=True)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3: {best_score:.10f}")

    # === Phase 4: Active-set refinement ===
    t_elapsed = time.monotonic() - t_start
    print(f"\n[P4] Active-set (t={t_elapsed:.0f}s)")
    if time.monotonic() + 50 < deadline:
        p4_end = min(deadline - 150, time.monotonic() + 150)
        a4, sc4 = _active_set_refine(best_seq, p4_end, min_time_remaining=100)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4: {best_score:.10f}")
        # Spectral after active set
        if time.monotonic() + 30 < deadline:
            p4b_end = min(deadline - 100, time.monotonic() + 100)
            a4b, sc4b = _spectral_shaping(best_seq, p4b_end, min_time_remaining=80)
            if sc4b < best_score:
                best_score = sc4b
                best_seq = a4b.copy()
                print(f"  P4b: {best_size:.10f}" if False else f"  P4b: {best_score:.10f}")

    # === Phase 5: Multi-n exploration ===
    t_elapsed = time.monotonic() - t_start
    print(f"\n[P5] Multi-n (t={t_elapsed:.0f}s)")
    n_options = sorted(set([n_main // 2, n_main, n_main + 4096, n_main + 8192, 24576]))
    n_options = [x for x in n_options if 4096 <= x <= 49152 and x != n_main]
    for n_try in n_options:
        if time.monotonic() + 60 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if sc_r < best_score:
            best_score = sc_r
            best_seq = a_r.copy()
            print(f"  n={n_try} (resample): {best_score:.10f}")
        if time.monotonic() + 80 > deadline:
            continue
        p5_end = min(deadline - 60, time.monotonic() + 100)
        a5, sc5 = _run_lbfgs(a_r, p5_end, min_time_remaining=40, verbose=False)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  n={n_try}: {best_score:.10f}")

    # === Phase 6: Perturbation + re-optimize ===
    t_elapsed = time.monotonic() - t_start
    print(f"\n[P6] Perturbation (t={t_elapsed:.0f}s)")
    n_current = len(best_seq)
    for pert_seed in range(5):
        if time.monotonic() + 50 > deadline:
            break
        a_pert = _perturb(best_seq, eps=0.02 / n_current ** 0.5, seed=100 + pert_seed)
        p6_end = min(deadline - 30, time.monotonic() + 60)
        a_p, sc_p = _run_lbfgs(a_pert, p6_end, min_time_remaining=20, verbose=False)
        if sc_p < best_score:
            best_score = sc_p
            best_seq = a_p.copy()
            print(f"  pert {pert_seed}: {best_score:.10f}")

    # === Phase 7: Final polish ===
    t_elapsed = time.monotonic() - t_start
    print(f"\n[P7] Final polish (t={t_elapsed:.0f}s)")
    if time.monotonic() + 20 < deadline:
        p7_end = min(deadline - 5, time.monotonic() + 100)
        a7, sc7 = _run_lbfgs(best_seq, p7_end, min_time_remaining=5, verbose=False)
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()

    if time.monotonic() + 15 < deadline:
        p7b_end = min(deadline - 3, time.monotonic() + 50)
        a7b, sc7b = _active_set_refine(best_seq, p7b_end, min_time_remaining=3)
        if sc7b < best_score:
            best_score = sc7b
            best_seq = a7b.copy()

    if time.monotonic() + 10 < deadline:
        p7c_end = min(deadline - 2, time.monotonic() + 30)
        a7c, sc7c = _subgradient_polish(best_seq, p7c_end, min_time_remaining=2)
        if sc7c < best_score:
            best_score = sc7c
            best_seq = a7c.copy()

    # Final
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={elapsed:.1f}s")
    return [float(x) for x in final]
