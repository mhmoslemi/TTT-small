import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    max_c = float(np.max(c))
    return 2.0 * n * max_c / (s ** 2)

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _soft_max_value(c, beta):
    n = len(c)
    sum_c = float(c.sum())
    mean_c = float(sum_c) / n
    if beta < 1e-30:
        return np.max(c)
    shifted = c - np.max(c) 
# Note: clipping below max avoids overflow in exponential, 
# and normalization ensures gap condition
    if np.min(shifted) < 0:
        shifted = shifted / (beta * mean_c or 1)
    else:
        shifted = shifted / beta
    e = np.exp(np.clip(shifted, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.max(c)
    return np.max(c) + beta * np.log(s)

def _soft_max_weights(c, beta):
    n = len(c)
    if beta < 1e-30:
        return np.zeros_like(c) + 1/n
    shifted = (c - np.max(c)) / beta
    shifted = np.clip(shifted, -700, None)  # Don't clip upper
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return e / s

def _run_lbfgs(a_init, deadline, min_time_remaining=60.0, verbose=True):
    n = len(a_init)

    s = a.sum() if a.sum() > 0.01 else 0.1

    best_score = _score(a)
    
    # Map to Fourier space more efficiently
    log_z = np.log(np.maximum(a, 1e-30))

    remaining = time.time() + min_time_remaining

            res = minimize(lambda x: _score(_softmax(x.reshape(n))), 
                           x0=log_z.reshape(n),
                           method='L-BFGS-B',
                           options={'maxiter':2000, 'ftol':1e-15, 'gtol':1e-12, 
                                 'maxcor':10, 'maxls':200, 'disp':False})
            new_params = res.x
            a_new = _softmax(new_params.reshape(n))
        else:
            # Stay in parameter space if score doesn't warrant big improvement
            continue_message = time.time() + remaining > timeout
            if continue_message:

            else:

        pass

    if np.min(a) < 0 or np.max(a) > 1000:
