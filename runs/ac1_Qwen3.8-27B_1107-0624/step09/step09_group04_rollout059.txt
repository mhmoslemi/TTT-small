"""
Optimized peak-shaving algorithm for minimizing 2*n*max(conv(a,a))/sum(a)^2.

Uses momentum-based active-set descent with exact quadratic line search,
periodic random restarts in log-space, multi-scale n exploration, and
parameterized family search (exponential, quadratic-exponential profiles).
Key improvement over previous: Nesterov momentum on direction, adaptive
step sizes with backtracking, and structured exploration to escape local minima.

import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _conv_grad(c, a, n):
    """Compute gradient of max_k c[k] w.r.t. a, using top-K peaks.
    Returns the subgradient direction (before simplex projection)."""
    top_k = np.argsort(c)[::-1][:K_global]
    w = np.zeros(2 * n - 1)
    w[top_k] = 1.0 / len(top_k)
    a_rev = a[::-1]
    h = _full_convolve(w, a_rev)
    return 2.0 * h[n - 1:2 * n - 1]


K_global = 50


def _proj_simplex_dir(a, d):
    """Project direction d so that a + alpha*d stays on simplex for some alpha > 0.
    Returns modified direction and max feasible alpha."""
    n = len(a)

        alpha_max = max(alpha_max, 1e-10)

    return d, min(alpha_max, 1.0)

    best_val = float(c_top.max())
    for alpha in alphas:
        vals = c_top + alpha * lin_top + alpha * alpha * quad_top
        v = float(vals.max())
        if v < best_val:
            best_val = v
            best_alpha = alpha
    
    # Refine with a few more points around best
    if best_alpha > 0:
        delta = best_alpha * 0.2
        for frac in [0.5, 0.75, 1.0, 1.25, 1.5]:
            a_try = best_alpha * frac
            if a_try > max_alpha or a_try < 0:
                continue
            vals = c_top + a_try * lin_top + a_try * a_try * quad_top
            v = float(vals.max())
            if v < best_val:
                best_val = v
                best_alpha = a_try
    
    return best_alpha, best_val


def _peak_shave(a_init, deadline, min_time_remaining=60.0, K_start=10, K_max=200):
    """Momentum-based active-set peak shaving with adaptive steps."""
    global K_global
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    best_a = a.copy()
    best_sc = _score(best_a)
    
    # Nesterov momentum
    y = a.copy()
    velocity = np.zeros(n)
    beta_mom = 0.9
    
    lr = 0.05
    no_improve = 0
    K = K_start
    total_iters = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max / (a.sum() ** 2)
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.1)
            if K < K_max and no_improve == 0:
                K = min(K_max, K + 2)
        
        K_global = K
        # Compute subgradient of max over top-K peaks
        grad = _conv_grad(c, a, n)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Nesterov: update y = a + beta * (a - a_prev)
        # Simplified: use velocity
        velocity = beta_mom * velocity - lr * grad / (gnorm + 1e-15)
        d = velocity
        d = d - np.mean(d)
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-18:
            break
        
        d = d / (dnorm + 1e-15)
        
        # Project to feasible direction
        d, alpha_feas = _proj_simplex_dir(a, d)
        if alpha_feas < 1e-12:
            # Can't move in this direction; try opposite
            d = -d
            d = d - np.mean(d)
            d, alpha_feas = _proj_simplex_dir(a, d)
            if alpha_feas < 1e-12:
                # Reset momentum
                velocity = np.zeros(n)
                no_improve += 1
                if no_improve > 200:
                    break
                continue
        
        alpha_feas = min(alpha_feas, 1.0)
        
        # Line search
        best_alpha, best_val = _line_search(a, d, c_max, n, max_alpha=alpha_feas)
        
        if best_alpha > 1e-10:
            a_new = a + best_alpha * d
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a = best_a.copy()
                velocity = np.zeros(n)
                no_improve += 5
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            
            if sc_new < current_sc - 1e-15:
                # Accept (Nesterov update)
                y_prev = y.copy()
                y = a_new
                a = a_new
                beta_mom = min(0.98, beta_mom + 0.001)
            else:
                no_improve += 1
                beta_mom = max(0.5, beta_mom * 0.95)
                lr *= 0.7
                K = max(K_start, K // 2)
        else:
            no_improve += 1
            lr *= 0.5
            K = max(K_start, K // 2)
        
        # Periodic restart
        if no_improve > 100:
            a = best_a.copy()
            velocity = np.zeros(n)
            beta_mom = 0.9
            lr = 0.05
            K = K_start
            # Add small perturbation
            z = np.log(np.maximum(a, 1e-30))
            z += 0.01 * np.random.standard_normal(n)
            z = z - z.mean()
            a = _softmax(z)
            no_improve = 0
        
        if no_improve > 400:
            break
        
        total_iters += 1
        if total_iters % 500 == 0:
            print(f"    peak_shave: iter={total_iters}, best={best_sc:.10f}, K={K}, lr={lr:.4f}, no_improve={no_improve}")
    
    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0):
    from scipy.optimize import minimize
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005) * beta_scale
    
    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625,
        0.03, 0.015, 0.0075, 0.004,
        0.002, 0.001, 0.0005, 0.0002, 0.0001
    ])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(3000, max(30, int(remaining / (est_time_per_iter * 15))))
        
        def obj(z_):
            return _smoobj(z_, beta, n)
        
        def grad(z_):
            return _smograd(z_, beta, n)
        
        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-17,
                         'gtol': 1e-14, 'maxcor': 80, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            a_new = np.clip(a_new, 0, 1000)
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "exp_ramp":
        # a[j] = exp(c * j / n) normalized, c around 1-3
        c = 1.5 if rng is None else rng.uniform(0.5, 3.0)
        t = np.arange(n) / n
        a = np.exp(c * (t - 0.5))
        return a / a.sum()
    elif kind == "quad_exp":
        # a[j] = exp(a + b*t + c*t^2) 
        if rng is None:
            a, b, c = 0, 1.0, -0.5
        else:
            a = rng.normal(0, 1)
            b = rng.normal(0, 2)
            c = rng.normal(0, 2)
        t = np.arange(n) / n
        x = a + b * t + c * t * t
        x = x - x.max() + 3  # shift to make exp safe
        arr = np.exp(x)
        return arr / arr.sum()
    elif kind == "sine_mod":
        if rng is None:
            f1, f2, eps = 3, 7, 0.3
        else:
            f1 = rng.integers(1, 10)
            f2 = rng.integers(5, 20)
            eps = rng.uniform(0.1, 0.5)
        t = np.arange(n)
        a = 1.0 + eps * np.sin(2 * np.pi * f1 * t / n) + eps * 0.5 * np.cos(2 * np.pi * f2 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        if rng is None:
            mu1, mu2, sig = 0.3, 0.7, 0.15
        else:
            mu1 = rng.uniform(0.2, 0.4)
            mu2 = rng.uniform(0.6, 0.8)
            sig = rng.uniform(0.05, 0.25)
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - mu1) / sig) ** 2) + 0.7 * np.exp(-((t - mu2) / sig) ** 2) + 0.01
        return a / a.sum()
    elif kind == "tapered_asym":
        t = np.linspace(0, 1, n)
        p = 1.5 if rng is None else rng.uniform(0.5, 4.0)
        a = np.sin(np.pi * t ** p) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _fast_refine(a_init, deadline, min_time_remaining=10.0):
    """Quick subgradient polish for the final few seconds."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(best_a)
    
    it = 0
    while time.monotonic() + min_time_remaining < deadline and it < 2000:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
        
        # Active set: peaks within 0.1% of max
        threshold = c_max * 0.999
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        K_use = min(len(active), 100)
        active = active[np.argsort(c[active])[::-1][:K_use]]
        
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        a_rev = a[::-1]
        h = _full_convolve(w, a_rev)
        grad = 2.0 * h[n - 1:2 * n - 1]
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        eta = 0.005 / (1 + it * 0.001)
        a_new = a - eta * grad / (gnorm + 1e-15)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        it += 1
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Multi-strategy optimizer combining momentum peak-shaving, L-BFGS smoothing,
    profile family exploration, and multi-n resampling."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "exp_ramp", rng)
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, exp_ramp n={n_parent}, score={best_score:.10f}")
    
    # === Phase 1: Quick L-BFGS warmup on parent ===
    print(f"\n[P1] L-BFGS warmup (t={time.monotonic()-t_start:.1f}s)")
    a_p1 = _run_lbfgs(parent, deadline, min_time_remaining=1200.0, beta_scale=1.0)
    sc_p1 = _score(a_p1)
    if sc_p1 < best_score:
        best_seq = a_p1.copy()
        best_score = sc_p1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_warmup]")
    
    # === Phase 2: Momentum peak-shaving (main optimization) ===
    print(f"\n[P2] Momentum peak-shaving (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        pol_deadline = min(deadline - 600, time.monotonic() + 400)
        a_ps, sc_ps = _peak_shave(best_seq, pol_deadline, min_time_remaining=100.0,
                                  K_start=20, K_max=150)
        if sc_ps < best_score:
            best_seq = a_ps.copy()
            best_score = sc_ps
            print(f"  NEW BEST: {best_score:.10f} [peak_shave_1]")
        else:
            print(f"  Peak shave 1: no improvement (best={best_score:.10f})")
    
    # === Phase 3: Try different profile families ===
    print(f"\n[P3] Profile family exploration (t={time.monotonic()-t_start:.1f}s)")
    n_main = len(best_seq)
    profiles = ["exp_ramp", "quad_exp", "sine_mod", "bimodal", "tapered_asym"]
    for prof in profiles:
        if time.monotonic() > deadline - 400:
            break
        a_prof = _make_profile(n_main, prof, rng)
        sc_prof = _score(a_prof)
        # Quick refine
        a_ref = _run_lbfgs(a_prof, deadline, min_time_remaining=400.0, beta_scale=0.5)
        sc_ref = _score(a_ref)
        if sc_ref < best_score:
            best_seq = a_ref.copy()
            best_score = sc_ref
            print(f"  NEW BEST: {best_score:.10f} [profile_{prof}]")
    
    # === Phase 4: Second peak-shaving round ===
    print(f"\n[P4] Peak-shaving round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        pol_deadline2 = min(deadline - 300, time.monotonic() + 200)
        a_ps2, sc_ps2 = _peak_shave(best_seq, pol_deadline2, min_time_remaining=50.0,
                                     K_start=30, K_max=200)
        if sc_ps2 < best_score:
            best_seq = a_ps2.copy()
            best_score = sc_ps2
            print(f"  NEW BEST: {best_score:.10f} [peak_shave_2]")
    
    # === Phase 5: Fine L-BFGS ===
    print(f"\n[P5] Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a_fine = _run_lbfgs(best_seq, deadline, min_time_remaining=200.0, beta_scale=0.3)
        sc_fine = _score(a_fine)
        if sc_fine < best_score:
            best_seq = a_fine.copy()
            best_score = sc_fine
            print(f"  NEW BEST: {best_score:.10f} [fine_lbfgs]")
    
    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([len(best_seq), 16384, 12288, 24576, 8192, 32768]), reverse=True)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=80.0, beta_scale=0.5)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n]")
    
    # === Phase 7: Final peak-shave ===
    print(f"\n[P7] Final peak-shave (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 50:
        pol_deadline3 = min(deadline - 40, time.monotonic() + 80)
        a_ps3, sc_ps3 = _peak_shave(best_seq, pol_deadline3, min_time_remaining=20.0,
                                     K_start=50, K_max=300)
        if sc_ps3 < best_score:
            best_seq = a_ps3.copy()
            best_score = sc_ps3
            print(f"  NEW BEST: {best_score:.10f} [peak_shave_final]")
    
    # === Phase 8: Subgradient polish ===
    print(f"\n[P8] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 10:
        sg_deadline = min(deadline - 5, time.monotonic() + 40)
        a_sg, sc_sg = _fast_refine(best_seq, sg_deadline, min_time_remaining=3.0)
        if sc_sg < best_score:
            best_seq = a_sg.copy()
            best_score = sc_sg
            print(f"  NEW BEST: {best_score:.10f} [subgrad]")
    
    # === Final: one more L-BFGS with ultra-small beta ===
    print(f"\n[P9] Ultra-fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 3 - time.monotonic()
    if remaining > 15:
        a_ult = _run_lbfgs(best_seq, deadline, min_time_remaining=3.0, beta_scale=0.1)
        sc_ult = _score(a_ult)
        if sc_ult < best_score:
            best_seq = a_ult.copy()
            best_score = sc_ult
            print(f"  NEW BEST: {best_score:.10f} [ultra_fine]")
    
    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
