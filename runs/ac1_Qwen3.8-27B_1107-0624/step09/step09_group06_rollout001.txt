"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS smooth-max, LP-based Frank-Wolfe direction, 
active-set subgradient, and multi-n exploration.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize
from scipy.sparse import coo_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Assumes sum(a) = 1, returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    """Project x onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.n, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_factor=1.0):
    """L-BFGS with adaptive beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Adaptive beta based on gap
    c_sorted = np.sort(c_cur)[::-1]
    K_check = min(50, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[K_check]
    if gap < 1e-15:
        gap = max_c * 0.001
    base_beta = max(gap * 0.1, max_c * 0.001) * beta_factor

    # Beta schedule: start moderate, go very fine
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03125, 0.015, 0.0075, 0.00375,
                                  0.0019, 0.00095, 0.0005, 0.00025,
                                  0.00012])

    current_a = a_init.copy()
    current_sc = _score_normalized(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.004 * max(1.0, n / 16384.0)
        max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 10))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score_normalized(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                # Recompute gap for next beta
                c_new = _full_convolve(a_new, a_new)
                mc = float(c_new.max())
                cs = np.sort(c_new)[::-1]
                g = mc - cs[min(50, len(cs)-1)]
                if g > 1e-15:
                    # Adjust remaining betas
                    remaining_betas = betas[i+1:]
                    if len(remaining_betas) > 0:
                        scale = g / (c_sorted[0] - c_sorted[K_check] + 1e-30)
                        # Don't rescale, just print
                        pass
                if i % 3 == 0:
                    print(f"    L-BFGS beta_idx={i}: score={current_sc:.10f}")
        except Exception as ex:
            print(f"    L-BFGS error at beta_idx={i}: {ex}")
            break

    return current_a


def _lp_frank_wolfe_direction(a, M, K=200, solver=None):
    """Solve the LP to find the Frank-Wolfe direction.
    
    max sum(b) s.t. b >= 0, (a*b)[k] <= M for k in top K peaks.
    
    Returns the optimal b (not normalized).
    """
    n = len(a)
    c = _full_convolve(a, a)
    # Get top K peak indices
    top_k_idx = np.argsort(c)[::-1][:K]
    
    # Build constraint matrix C (K x n)
    # C[k_idx, j] = a[k_idx - j] if 0 <= k_idx - j < n
    C = np.zeros((K, n))
    for i, k in enumerate(top_k_idx):
        # (a*b)[k] = sum_j a[k-j] * b[j]
        # Valid j: max(0, k-n+1) <= j <= min(n-1, k)
        j_lo = max(0, int(k) - n + 1)
        j_hi = min(n - 1, int(k))
        if j_lo <= j_hi:
            C[i, j_lo:j_hi+1] = a[k - np.arange(j_lo, j_hi + 1)]
    
    # Set up LP
    b_var = cp.Variable(n)
    constraints = [b_var >= 0]
    constraints.append(C @ b_var <= M)
    
    problem = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    # Try different solvers
    tried = []
    solvers_to_try = [solver] if solver else []
    solvers_to_try += ['GLPK', 'ECOS', 'SCS']  # SCS might not be available
    # Actually available: CBC,CVXOPT,GLOP,GLPK,GUROBI,MOSEK,PDLP,SCIP,XPRESS,ECOS
    solvers_to_try = [s for s in solvers_to_try if s in ['GLPK', 'ECOS', 'PDLP', 'GUROBI', 'MOSEK']]
    
    for sol in solvers_to_try:
        if sol in tried:
            continue
        tried.append(sol)
        try:
            problem.solve(solver=sol)
            if problem.status == 'optimal' and b_var.value is not None:
                return np.maximum(b_var.value, 0)
        except Exception:
            continue
    
    # Fallback: uniform direction
    return np.ones(n)


def _alternating_lp(a_init, deadline, min_time_remaining=60.0, K=200):
    """Alternating method: fix a, solve LP for b, blend."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    
    best_a = a.copy()
    best_sc = _score_normalized(a)
    
    lr = 0.5
    no_improve = 0
    stall_count = 0
    
    t = 0
    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        M = float(c.max())
        current_sc = 2.0 * n * M
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50:
            stall_count += 1
            if stall_count > 8:
                break
            lr = max(0.1, lr * 0.7)
            no_improve = 0
        
        # Solve LP for direction
        t_lp_start = time.monotonic()
        b_dir = _lp_frank_wolfe_direction(a, M, K=K)
        t_lp = time.monotonic() - t_lp_start
        
        if b_dir is None or not np.any(b_dir > 0):
            break
        
        S = b_dir.sum()
        if S < 1e-15:
            break
        b_norm = b_dir / S
        
        # Blend
        a_new = (1 - lr) * a + lr * b_norm
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-10:
            break
        a_new = a_new / sn
        
        sc_new = _score_normalized(a_new)
        
        if sc_new < current_sc:
            a = a_new
        else:
            # Try smaller step
            for trial_lr in [lr * 0.5, lr * 0.25]:
                a_try = (1 - trial_lr) * a + trial_lr * b_norm
                a_try = np.maximum(a_try, 0)
                st = a_try.sum()
                if st < 1e-10:
                    continue
                a_try = a_try / st
                sc_try = _score_normalized(a_try)
                if sc_try < current_sc:
                    a = a_try
                    lr = trial_lr
                    break
            else:
                a = a_new  # accept anyway to avoid stalling
        
        if t % 5 == 0:
            print(f"    LP-Alt t={t}: best={best_sc:.10f}, cur={sc_new:.10f}, lr={lr:.3f}, LP_time={t_lp:.1f}s")
    
    return best_a, best_sc


def _subgradient_active(a_init, deadline, min_time_remaining=60.0):
    """Subgradient method with exact active set (all peaks within tol of max)."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    
    best_a = a.copy()
    best_sc = _score_normalized(a)
    
    base_lr = 0.003
    lr = base_lr
    momentum_v = np.zeros(n)
    momentum = 0.9
    tol = 1e-6  # relative tolerance for active set
    
    no_improve = 0
    stall_count = 0
    t = 0
    
    rng = np.random.default_rng(12345)
    
    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            stall_count += 1
            if stall_count > 10:
                break
            # Perturb
            perturb = 0.0005 * rng.standard_normal(n)
            a_pert = _project_simplex(a + perturb)
            sc_pert = _score_normalized(a_pert)
            if sc_pert < best_sc:
                a = a_pert
                best_a = a.copy()
                best_sc = sc_pert
            momentum_v = np.zeros(n)
            lr = base_lr
            no_improve = 0
            stall_count = 0
        
        # Active set: all peaks within tol of max
        active_mask = c >= c_max * (1 - tol)
        active_idx = np.where(active_mask)[0]
        
        if len(active_idx) == 0:
            active_idx = [int(np.argmax(c))]
        
        # Compute gradient of sum of active peaks
        grad = np.zeros(n)
        for k in active_idx:
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                grad[j_lo:j_hi+1] += a[k - np.arange(j_lo, j_hi + 1)]
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_lo2 <= j_hi2:
                grad[j_lo2:j_hi2+1] += a[np.arange(j_lo2, j_hi2 + 1) - k]
        
        # Project gradient to tangent space of simplex
        grad_t = grad - np.mean(grad)
        
        # Nesterov step
        momentum_v = momentum * momentum_v - lr * grad_t
        a_cand = a + momentum_v
        a_proj = _project_simplex(a_cand)
        
        sc_new = _score_normalized(a_proj)
        
        if sc_new < current_sc:
            a = a_proj
            lr = min(lr * 1.01, 0.01)
        else:
            momentum_v = 0.5 * momentum_v
            lr *= 0.95
            # Still project to stay feasible
            a = a_proj
        
        if t % 500 == 0:
            print(f"    SubGrad t={t}: best={best_sc:.10f}, cur={sc_new:.10f}, |active|={len(active_idx)}, lr={lr:.6f}")
    
    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * (t - 0.5)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":
        a = np.ones(n)
        taper = max(1, n // 100)
        for i in range(taper):
            a[i] = (i + 1) / taper
            a[n - 1 - i] = (i + 1) / taper
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _shrinkage_optimize(a, deadline):
    """Try zeroing out small elements."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score_normalized(a)
    
    a_sorted_idx = np.argsort(a)
    fractions = [0.001, 0.002, 0.005, 0.01, 0.02, 0.03, 0.05]
    
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score_normalized(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)
            print(f"  Shrinkage: frac={frac}, score={best_sc:.10f}")
    
    return best_a, best_sc
