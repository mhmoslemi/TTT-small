def _next_pow2(n):
    """Calculate the next power of two >= n for FFT."""
    k = 1
    while k < n:

        a = np.exp(-15.0 * (t - 0.25) ** 2) + np.exp(-15.0 * (t - 0.75) ** 2) + 0.05
        a = a / a.sum()
    elif kind == "flat_taper":
        a = np.ones(n)
        taper = max(1, n // 50)
        for i in range(taper):
            a[i] = (i + 1) / taper
            a[n-1-i] = (i + 1) / taper
        a = a / a.sum()
    else:
        a = np.ones(n) / n

    return a
