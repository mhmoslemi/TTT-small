import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _grad_smoothed_fast(a, p, n):
    b = a[::-1]
    conv_len = len(b) + len(p) - 1
    N = _next_pow2(conv_len)
    Fb = np.fft.rfft(b, N)
    Fp = np.fft.rfft(p, N)
    conv = np.fft.irfft(Fb * Fp, N)[:conv_len]
    result = 2.0 * conv[n - 1:2 * n - 1].copy()
    for k in range(0, 2 * n - 1, 2):
        m = k // 2
        if m < n and k < len(p):
            result[m] -= a[m] * p[k]
    return result

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _run_active_set_cg(a_init, deadline, min_time_remaining=60.0, max_rounds=200):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    tol_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                    0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
                    0.000001, 0.0000005, 0.0000002, 0.0000001]
    tol_idx = 0
    rounds_per_tol = 8

    for rnd in range(max_rounds):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / active.sum()

        grad = _grad_smoothed_fast(a, p, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            tol_idx += 1
            continue

        # Conjugate gradient-like step with backtracking
        d = -grad
        alpha = 0.1 / (gnorm + 1e-10)
        for ls in range(30):
            a_new = _proj_simplex(a + alpha * d)
            c_new = _full_convolve(a_new, a_new)
            if float(c_new.max()) <= c_max + 1e-12:
                a = a_new
                break
            alpha *= 0.5

        # Also try a steeper step
        alpha2 = 0.5 / (gnorm + 1e-10)
        for ls in range(20):
            a_new = _proj_simplex(a - alpha2 * grad)
            c_new = _full_convolve(a_new, a_new)
            if float(c_new.max()) < c_max:
                a = a_new
                c_max = float(c_new.max())
                break
            alpha2 *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
        elif rnd > 10:
            a = best_a.copy()

        if (rnd + 1) % rounds_per_tol == 0:
            tol_idx += 1

    return best_a, best_sc


def _run_momentum_grad(a_init, deadline, min_time_remaining=60.0):

            break

        lr = lr_base * (1.0 + 0.05 * min(step, 100)) / (mnorm / grad_norm + 1e-10)
        lr = min(lr, 1.0 / (mnorm + 1e-10))

        z_new = z - lr * momentum
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()
        z = z_new

        if step % 5 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros(n)
                    patience = 0
                    beta_m = max(0.5, beta_m * 0.9)
                    lr_base *= 0.7

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _run_dual_primal(a_init, deadline, min_time_remaining=60.0, max_outer=300):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    base_beta = max_c * 0.02

    for outer in range(max_outer):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        beta = base_beta * (0.85 ** min(outer, 80))
        p = _softmax(c / beta)

        # Primal: projected gradient with a few iterations
        lr = 0.02
        for inner in range(30):
            if time.monotonic() + min_time_remaining > deadline:
                break
            grad = _grad_smoothed_fast(a, p, n)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            lr = 0.1 / (gnorm * (1.0 + 0.02 * inner))
            a = _proj_simplex(a - lr * grad)

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _multi_start_small_n(deadline, min_time_remaining=30.0, n_target=256):
    """Try multiple profiles at small n and pick the best."""
    profiles = ["ramp", "ramp_bend", "ramp_bend2", "ramp_bend3",
                "super_ramp", "ramp05", "ramp_quadratic", "cos_ramp",
                "exp_ramp", "ramp2"]
    best = None
    best_sc = np.inf
    per_profile_time = (deadline - min_time_remaining - time.monotonic()) / len(profiles)

    for prof in profiles:
        if time.monotonic() + min_time_remaining > deadline:
            break
        a_init = _make_profile(n_target, prof)
        sc_init = _score(a_init)
        dl = time.monotonic() + max(5.0, per_profile_time)
        a_opt = _run_lbfgs(a_init, dl, min_time_remaining=5.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best = a_opt.copy()
            print(f"    small_n({prof}): score={sc_opt:.10f} (init={sc_init:.10f})")

    return best, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    # === Phase 1: Small-n optimization (120s) ===
    print(f"\n[Phase 1] Small-n multi-start (t={time.monotonic()-t_start:.1f}s)")
    phase1_deadline = t_start + 120
    if parent is not None and n_parent <= 512:
        a_small = parent
    else:
        a_small, sc_small = _multi_start_small_n(phase1_deadline, min_time_remaining=30.0, n_target=256)
        if a_small is not None:
            update_best(a_small, sc_small, "small_n")
            if parent is None:
                parent = a_small
                n_parent = len(a_small)
        else:
            a_small = _make_profile(256, "ramp_bend")
            update_best(a_small, _score(a_small), "small_n_init")
            parent = a_small
            n_parent = 256

    # Thorough L-BFGS at small n
    if time.monotonic() < phase1_deadline - 20:
        a_s1 = _run_lbfgs(a_small, phase1_deadline, min_time_remaining=15.0)
        update_best(a_s1, _score(a_s1), "small_n_lbfgs")

    # === Phase 2: Upscale to medium n (150s) ===
    print(f"\n[Phase 2] Medium-n (n=2048) (t={time.monotonic()-t_start:.1f}s)")
    phase2_deadline = min(t_start + 270, deadline - 900)
    if best_seq is not None and time.monotonic() < phase2_deadline:
        a_med = _resample_log(best_seq, 2048)
        a_m1 = _run_lbfgs(a_med, phase2_deadline, min_time_remaining=30.0)
        update_best(a_m1, _score(a_m1), "medium_n")
        if time.monotonic() < phase2_deadline - 15:
            a_m2, sc_m2 = _run_momentum_grad(a_m1, phase2_deadline, min_time_remaining=10.0)
            update_best(a_m2, sc_m2, "medium_momentum")
        if time.monotonic() < phase2_deadline - 10:
            a_m3, sc_m3 = _run_active_set_cg(a_m1, phase2_deadline, min_time_remaining=5.0, max_rounds=50)
            update_best(a_m3, sc_m3, "medium_as")

    # === Phase 3: Upscale to n=8192 (180s) ===
    print(f"\n[Phase 3] n=8192 (t={time.monotonic()-t_start:.1f}s)")
    phase3_deadline = min(t_start + 450, deadline - 700)
    if best_seq is not None and time.monotonic() < phase3_deadline:
        a_8k = _resample_log(best_seq, 8192)
        a_8k1 = _run_lbfgs(a_8k, phase3_deadline, min_time_remaining=30.0)
        update_best(a_8k1, _score(a_8k1), "n8192")
        if time.monotonic() < phase3_deadline - 20:
            a_8k2, sc_8k2 = _run_active_set_cg(a_8k1, phase3_deadline, min_time_remaining=15.0, max_rounds=80)
            update_best(a_8k2, sc_8k2, "n8192_as")
        if time.monotonic() < phase3_deadline - 10:
            a_8k3, sc_8k3 = _run_momentum_grad(a_8k1, phase3_deadline, min_time_remaining=5.0)
            update_best(a_8k3, sc_8k3, "n8192_mom")

    # === Phase 4: Upscale to n=32768 (300s) ===
    print(f"\n[Phase 4] n=32768 (t={time.monotonic()-t_start:.1f}s)")
    phase4_deadline = min(t_start + 750, deadline - 500)
    if best_seq is not None and time.monotonic() < phase4_deadline:
        a_32k = _resample_log(best_seq, 32768)
        a_32k1 = _run_lbfgs(a_32k, phase4_deadline, min_time_remaining=60.0)
        update_best(a_32k1, _score(a_32k1), "n32768")

        if time.monotonic() < phase4_deadline - 30:
            a_32k2, sc_32k2 = _run_dual_primal(a_32k1, phase4_deadline, min_time_remaining=20.0, max_outer=200)
            update_best(a_32k2, sc_32k2, "n32768_dp")

        if time.monotonic() < phase4_deadline - 20:
            a_32k3, sc_32k3 = _run_momentum_grad(a_32k1, phase4_deadline, min_time_remaining=10.0)
            update_best(a_32k3, sc_32k3, "n32768_mom")

    # === Phase 5: Active-set QP at n=32768 (300s) ===
    print(f"\n[Phase 5] Active-set QP (t={time.monotonic()-t_start:.1f}s)")
    phase5_deadline = min(t_start + 1050, deadline - 350)
    if best_seq is not None and time.monotonic() < phase5_deadline:
        a_qp = _run_active_set_cg(best_seq, phase5_deadline, min_time_remaining=30.0, max_rounds=300)
        update_best(a_qp, _score(a_qp), "qp_1")

        if time.monotonic() < phase5_deadline - 20:
            a_qp2 = _run_lbfgs(a_qp, phase5_deadline, min_time_remaining=15.0)
            update_best(a_qp2, _score(a_qp2), "qp_lbfgs")

    # === Phase 6: Second round L-BFGS (250s) ===
    print(f"\n[Phase 6] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    phase6_deadline = min(t_start + 1300, deadline - 250)
    if best_seq is not None and time.monotonic() < phase6_deadline:
        a_r2 = _run_lbfgs(best_seq, phase6_deadline, min_time_remaining=30.0)
        update_best(a_r2, _score(a_r2), "lbfgs_r2")

        if time.monotonic() < phase6_deadline - 30:
            a_r2m, sc_r2m = _run_momentum_grad(a_r2, phase6_deadline, min_time_remaining=15.0)
            update_best(a_r2m, sc_r2m, "mom_r2")

    # === Phase 7: Multi-n exploration (200s) ===
    print(f"\n[Phase 7] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    phase7_deadline = min(t_start + 1500, deadline - 200)
    n_tries = [16384, 24576, 65536, 4096]
    n_current = len(best_seq) if best_seq is not None else 0
    for n_try in n_tries:
        if time.monotonic() > phase7_deadline - 30:
            break
        if n_try == n_current:
            continue
        if best_seq is not None:
            a_init_n = _resample_log(best_seq, n_try)
        else:
            a_init_n = _make_profile(n_try, "ramp_bend")
        a_n = _run_lbfgs(a_init_n, phase7_deadline, min_time_remaining=15.0)
        sc_n = _score(a_n)
        update_best(a_n, sc_n, f"n_{n_try}")

        if sc_n < best_score + 0.003 and time.monotonic() < phase7_deadline - 20:
            a_nq, sc_nq = _run_active_set_cg(a_n, phase7_deadline,
                                             min_time_remaining=10.0, max_rounds=50)
            update_best(a_nq, sc_nq, f"n_{n_try}_as")

    # === Phase 8: Perturbation and restart (150s) ===
    print(f"\n[Phase 8] Perturbations (t={time.monotonic()-t_start:.1f}s)")
    phase8_deadline = min(t_start + 1650, deadline - 120)
    if best_seq is not None and time.monotonic() < phase8_deadline:
        for sigma in [0.02, 0.01, 0.005]:
            if time.monotonic() > phase8_deadline - 20:
                break
            a_pert = _perturb(best_seq, sigma, rng)
            a_p = _run_lbfgs(a_pert, phase8_deadline, min_time_remaining=10.0)
            update_best(a_p, _score(a_p), f"perturb_{sigma}")

    # === Phase 9: Final ultra-tight polish (100s) ===
    print(f"\n[Phase 9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_fin, sc_fin = _run_active_set_cg(best_seq, deadline - 10,
                                           min_time_remaining=5.0, max_rounds=100)
        update_best(a_fin, sc_fin, "final_as")

        if time.monotonic() < deadline - 12:
            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta_fin = mb * 1e-9
            remaining = deadline - 12 - time.monotonic()
            if remaining > 3:
                max_iter = min(2000, max(50, int(remaining / 0.005)))

                def final_obj(z_local):
                    a_local = _softmax(z_local)
                    c_local = _full_convolve(a_local, a_local)
                    c_max_local = float(c_local.max())
                    shifted = (c_local - c_max_local) / beta_fin
                    e = np.exp(np.clip(shifted, -700, 0))
                    s_e = e.sum()
                    if s_e < 1e-300:
                        return 2.0 * n * c_max_local
                    return 2.0 * n * (c_max_local + beta_fin * np.log(s_e))

                def final_grad(z_local):
                    a_local = _softmax(z_local)
                    c_local = _full_convolve(a_local, a_local)
                    c_max_local = float(c_local.max())
                    shifted = (c_local - c_max_local) / beta_fin
                    e = np.exp(np.clip(shifted, -700, 0))
                    s_e = e.sum()
                    if s_e < 1e-300:
                        w = np.zeros_like(c_local)
                        w[int(np.argmax(c_local))] = 1.0
                    else:
                        w = e / s_e
                    a_rev = a_local[::-1]
                    wc = _full_convolve(w, a_rev)
                    h = wc[n - 1:2 * n - 1]
                    W_val = float(np.dot(w, c_local))
                    return 4.0 * n * a_local * (h - W_val)

                try:
                    result = minimize(
                        final_obj, z,
                        jac=final_grad,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-22, 'gtol': 1e-18,
                                 'maxcor': 50}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "ramp_bend")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "ramp_bend")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
