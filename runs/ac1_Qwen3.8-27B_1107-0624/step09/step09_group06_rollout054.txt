For the Newton-based peak equalization, here's the idea:
- Identify the top K peaks: `k_1, ..., k_K`
- We want to find `a` (near the current `a`) such that `(a*a)[k_i] = (a*a)[k_j]` for all `i, j`
- This gives `K-1` equations
- We also want `sum(a) = 1` (1 equation)
- Total: `K` equations in `n` unknowns
- Use the "minimum norm" solution: `a_new = a + V * alpha` where `V` is an `n x K` matrix of "tangent directions" and `alpha` is chosen to satisfy the equations.

The Jacobian of the system is `K x n`. We can solve for `alpha` using the pseudoinverse.

Specifically, let `F(a) = [(a*a)[k_1] - (a*a)[k_2], ..., (a*a)[k_{K-1}] - (a*a)[k_K], sum(a) - 1]`. We want `F(a) = 0`.

`J = dF/da` is a `K x n` matrix. The Newton step is `da = -J^+ * F(a)` where `J^+` is the pseudoinverse.

Then `a_new = a + t * da` for some step size `t` (determined by line search).

This is a clean approach! Let me implement it.

Actually, the Jacobian entries are:
- `d(a*a)[k_i]/da[j] = a[k_i - j] + a[j - k_i]` (for valid indices)
- `d(sum(a))/da[j] = 1`

So the Jacobian row for peak `k` is: `[a[k-0] + a[0-k], a[k-1] + a[1-k], ..., a[k-(n-1)] + a[(n-1)-k]]` (with invalid indices contributing 0).

This is essentially the "correlation" of `a` with a shifted version of itself, at shift `k`.

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)

    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)

def _jacobian_row(a, k):
    """Compute d(a*a)[k]/da as a vector of length n."""
    n = len(a)
    grad = np.zeros(n)
    # (a*a)[k] = sum_j a[j] * a[k-j]
    # d/d a[j] = a[k-j] (if 0 <= k-j < n) + a[j-k] (if 0 <= j-k < n)
    for j in range(n):
        val = 0.0
        if 0 <= k - j < n:
            val += a[k - j]
        if 0 <= j - k < n:
            val += a[j - k]
        grad[j] = val
    return grad

def _jacobian_row_fast(a, k):
    """Vectorized version."""
    n = len(a)
    grad = np.zeros(n)
    # First term: a[k-j] for j where 0 <= k-j < n, i.e., max(0, k-n+1) <= j <= min(n-1, k)
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    if j_lo <= j_hi:

    if j_lo2 <= j_hi2:
        idx = np.arange(j_lo2, j_hi2 + 1) - k
        grad[j_lo2:j_hi2 + 1] += a[idx]
    return grad

    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):

            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
        
        # Identify top K peaks
        c_idx = np.argsort(c)[::-1]
        active_k = c_idx[:K]
        
        # Check if peaks are already equalized
        peak_vals = c[active_k]
        peak_range = peak_vals.max() - peak_vals.min()
        if peak_range < 1e-12 * c_max:
            break
        
        # Compute Jacobian (K x n matrix)
        # Row i: d(a*a)[active_k[i]]/da
        # We want to minimize the spread of the top K peaks
        # Use the difference: (a*a)[k_i] - (a*a)[k_{K-1}] for i = 0, ..., K-2
        # Plus the constraint sum(a) = 1
        
        # Actually, let's use a different formulation:

        # This is equivalent to minimizing the variance of the active peaks

        weights = 2.0 * (peak_vals - mean_pv)

        for i, k in enumerate(active_k):
            grad += weights[i] * _jacobian_row_fast(a, k)
        
        # Project to direction that maintains sum(a) = 1
        grad = grad - np.mean(grad)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Line search

        for ls in range(20):

            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:

            if sc_try < current_sc:
                a = a_try
                break
            alpha *= 0.5
        
        if alpha < 1e-10:
            break
    
    return best_a, best_sc

"""
Optimized sequence finder using corrected gradient L-BFGS with adaptive
beta schedule, Newton peak equalization, and multi-scale exploration.
Key improvement: correct gradient computation (includes both forward and 
backward convolution derivative terms) for better convergence.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd_correct(z, beta, n):
    """Corrected gradient that includes both forward and backward terms."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    
    # h[j] = sum_k w[k] * a[k-j] for j = 0,...,n-1
    # Computed as elements [n-1, 2n-2] of conv(w, a_rev)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    
    # h'[j] = sum_k w[k] * a[j-k] for j = 0,...,n-1
    # This is elements [0, n-1] of conv(w, a)
    wa = _full_convolve(w, a)
    h_prime = wa[:n]
    
    W = float(np.dot(w, c))
    W_prime = float(np.dot(a, h_prime))
    
    return 2.0 * n * a * (h + h_prime - W - W_prime)


class _LbfgsProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    
    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)
    
    def grad(self, z):
        return _smograd_correct(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, n_betas=12):
    """L-BFGS with corrected gradient and adaptive beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)
    
    # Geometric beta schedule
    ratio = 0.5
    betas = base_beta * (ratio ** np.arange(n_betas))
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        # Estimate time per iteration
        est_time = 0.005 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 5))))
        
        prob = _LbfgsProblem(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS beta={beta:.2e}: score={current_sc:.12f} (iter={result.nit})")
            else:
                print(f"    L-BFGS beta={beta:.2e}: no improvement (score={sc_new:.12f} vs {current_sc:.12f})")
        except Exception as ex:
            print(f"    L-BFGS beta={beta:.2e}: error: {ex}")
    
    return current_a, current_sc

    s = a.sum()

        a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    while time.monotonic() + min_time_remaining < deadline:

        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Identify active peaks

        active_k = c_idx[:K]
        peak_vals = c[active_k]
        peak_mean = peak_vals.mean()
        peak_spread = peak_vals.max() - peak_vals.min()
        
        if peak_spread < 1e-10 * c_max:
            break
        
        # Compute gradient to reduce peak spread
        # Minimize sum_i ((c[k_i] - peak_mean)^2) via steepest descent
        weights = 2.0 * (peak_vals - peak_mean)
        
        # Full gradient: sum_i weights[i] * dc[k_i]/da
        grad = np.zeros(n)
        for i, k in enumerate(active_k):
            if abs(weights[i]) < 1e-15:
                continue
            # dc[k]/da[j] = a[k-j] + a[j-k]
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                idx = k - np.arange(j_lo, j_hi + 1)
                grad[j_lo:j_hi + 1] += weights[i] * a[idx]
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_lo2 <= j_hi2:
                idx = np.arange(j_lo2, j_hi2 + 1) - k
                grad[j_lo2:j_hi2 + 1] += weights[i] * a[idx]
        
        # Project to maintain sum(a) = 1
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Line search
        alpha = 1.0
        improved = False
        for _ in range(30):
            a_try = a - alpha * grad / (gnorm * gnorm)
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved and alpha < 1e-10:
            break
    
    return best_a, best_sc


def _subgradient_optimize(a_init, deadline, min_time_remaining=60.0,
                           K_init=10, K_max=200, seed=0):
    """Subgradient method with momentum for coarse optimization."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())
    
    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    momentum = 0.8
    base_lr = 0.01
    lr = base_lr
    K = K_init
    no_improve = 0
    stall_count = 0
    t = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive K
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
        if gap < c_max * 0.003 and K < K_max:
            K = min(K_max, int(K * 1.3) + 1)
        
        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                idx = k - np.arange(j_lo, j_hi + 1)
                grad[j_lo:j_hi + 1] += a[idx]
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_lo2 <= j_hi2:
                idx = np.arange(j_lo2, j_hi2 + 1) - k
                grad[j_lo2:j_hi2 + 1] += a[idx]
        
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break
        
        effective_lr = lr / (gnorm + 1e-18) * 0.1
        v = momentum * v - effective_lr * grad_t
        a_cand = a + v
        a_proj = _project_simplex(a_cand)
        
        c_new = _full_convolve(a_proj, a_proj)
        sc_new = 2.0 * n * float(c_new.max())
        
        if sc_new < current_sc:
            a = a_proj
        else:
            v = 0.5 * v
            lr *= 0.95
            a = a_proj
        
        if no_improve % 300 == 0 and no_improve > 0:
            stall_count += 1
            if stall_count > 8:
                break
            perturb = 0.002 * rng.standard_normal(n)
            a_pert = _project_simplex(a + perturb)
            c_pert = _full_convolve(a_pert, a_pert)
            sc_pert = 2.0 * n * float(c_pert.max())
            if sc_pert < best_sc:
                a = a_pert
                best_a = a.copy()
                best_sc = sc_pert
            v = np.zeros(n)
            lr = base_lr
            no_improve = 0
        
        if t % 500 == 0:
            print(f"    Subgrad t={t}: best={best_sc:.10f}, K={K}")
    
    return best_a, best_sc
