def _next_pow2(n):

    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

        return np.inf
    c = _full_convolve(a, a)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))

    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0

    h = wc[n - 1:2 * n - 1]

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

    n = len(a)
    c = _full_convolve(a, a)
    M = float(c.max())
    
    # LP: maximize sum(b) subject to b>=0, conv(a,b)[k] <= M for all k
    # conv(a,b)[k] = sum_j a[j]*b[k-j] (linear in b)
    # Variables: b[0], ..., b[n-1]
    # Constraints: for each k in 0..2n-2, sum_{j valid} a[j]*b[k-j] <= M
    # Objective: maximize sum(b)
    
    # This is too large for most LP solvers with n=24576 (49151 constraints, 24576 vars)
    # Instead, use a simplified version: only constrain the top few peaks
    n_top = 50
    c_top = np.sort(c)[::-1][:n_top]
    top_idx = np.argsort(c)[::-1][:n_top]
    
    # Actually, for large n this LP is intractable. Skip it and use gradient-based approach.
    # Return None to indicate LP wasn't used
    return None


def _stochastic_ils(a_init, deadline, rng, min_time_remaining=40.0):
    """Stochastic Iterated Local Search."""
    n = len(a_init)
    current_a = _normalize(a_init.copy())
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc

    strategies = [
        ('global_small', 0.003),
        ('global_med', 0.01),
        ('global_large', 0.03),
        ('block', 0.03),
        ('sparse', 0.05),
        ('global_tiny', 0.001),
    ]

    attempt = 0
    global t_start
    while time.monotonic() + min_time_remaining < deadline:
        kind, sigma = strategies[attempt % len(strategies)]

        if kind == 'global_small' or kind == 'global_med' or kind == 'global_large' or kind == 'global_tiny':
            z = np.log(np.maximum(current_a, 1e-30))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_perturbed = _softmax(z)
        elif kind == 'block':
            block_size = max(20, n // 6)
            start = int(rng.integers(0, n - block_size))
            a_perturbed = current_a.copy()
            z = np.log(np.maximum(a_perturbed[start:start + block_size], 1e-30))
            z = z + sigma * rng.standard_normal(block_size)
            a_perturbed[start:start + block_size] = np.exp(z - np.max(z))
            a_perturbed = _normalize(a_perturbed)
        elif kind == 'sparse':
            n_perturb = max(10, n // 20)
            idx = rng.choice(n, size=n_perturb, replace=False)
            a_perturbed = current_a.copy()
            factors = np.exp(sigma * rng.standard_normal(n_perturb))
            a_perturbed[idx] *= factors
            a_perturbed = _normalize(a_perturbed)
        else:
            a_perturbed = current_a.copy()

        opt_deadline = min(deadline - min_time_remaining, time.monotonic() + 6)
        a_opt = _quick_optimize(a_perturbed, opt_deadline, min_time_remaining=3)
        sc_opt = _score(a_opt)

        if sc_opt < best_sc - 1e-13:
            best_sc = sc_opt
            best_a = a_opt.copy()
            current_a = a_opt.copy()
            print(f"    ILS NEW BEST: {best_sc:.12f} ({kind}) t={time.monotonic()-t_start:.0f}s")
        elif sc_opt < current_sc - 1e-13:
            current_a = a_opt.copy()

        attempt += 1

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: aggressive L-BFGS, peak-targeted subgradient,
    stochastic ILS, multi-n exploration, and final ultra-fine polish.
    """
    global t_start
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.12f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 24576
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.12f}")

    n_main = n_parent

    # === Phase 1: Aggressive L-BFGS on parent (400s) ===
    print(f"\n[P1] Aggressive L-BFGS on parent (t={time.monotonic() - t_start:.1f}s)")
    p1_deadline = t_start + 400
    a_opt = _run_lbfgs_aggressive(best_seq, p1_deadline, min_time_remaining=50)
    sc_opt = _score(a_opt)
    if sc_opt < best_score:
        best_seq = a_opt.copy()
        best_score = sc_opt
        print(f"  NEW BEST: {best_score:.12f} [lbfgs_parent] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement: {sc_opt:.12f}")

    # === Phase 2: Peak-targeted subgradient (150s) ===
    print(f"\n[P2] Peak-targeted subgradient (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 900:
        p2_deadline = min(deadline - 900, time.monotonic() + 150)
        a2, sc2 = _peak_target_subgrad(best_seq, p2_deadline, n_iter=5000)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.12f} [peak_subgrad] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement: {sc2:.12f}")

    # === Phase 3: Multi-start with diverse profiles (250s) ===
    print(f"\n[P3] Multi-start (t={time.monotonic() - t_start:.1f}s)")
    p3_deadline = min(deadline - 800, time.monotonic() + 250)
    starts = []
    for kind in ['hann', 'gaussian', 'raised_cos2', 'triangle']:
        starts.append((kind, _make_profile(n_main, kind)))

    for name, a_init in starts:
        if time.monotonic() > p3_deadline - 30:
            break
        a_opt = _quick_optimize(a_init, p3_deadline, min_time_remaining=30)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.12f} [ms/{name}] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  {name}: {sc_opt:.12f} (no improvement)")

    # === Phase 4: ILS on best (300s) ===
    print(f"\n[P4] ILS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        ils_deadline = min(deadline - 350, time.monotonic() + 300)
        a_ils, sc_ils = _stochastic_ils(best_seq, ils_deadline, rng, min_time_remaining=50)
        if sc_ils < best_score:
            best_seq = a_ils.copy()
            best_score = sc_ils
            print(f"  NEW BEST: {best_score:.12f} [ils] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No ILS improvement: {sc_ils:.12f}")

    # === Phase 5: Multi-n exploration (250s) ===
    print(f"\n[P5] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [n_main - 1024, n_main + 1024, n_main + 2048, 20480, 28672, 16384, 32768]
    seen = set()
    n_unique = []
    for nn in n_candidates:
        if nn not in seen and nn != len(best_seq) and nn > 4096:
            seen.add(nn)
            n_unique.append(nn)

    for n_try in n_unique:
        if time.monotonic() > deadline - 200:
            break
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init={sc_init:.12f}")

        a_opt = _quick_optimize(a_init, deadline, min_time_remaining=30)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.12f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  n={n_try}: {sc_opt:.12f} (no improvement)")

    # === Phase 6: Final intensive L-BFGS + subgradient (200s) ===
    print(f"\n[P6] Final intensive polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        # First: aggressive L-BFGS
        lbfgs_dl = min(deadline - 30, time.monotonic() + 120)
        a_fin = _run_lbfgs_aggressive(best_seq, lbfgs_dl, min_time_remaining=15)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.12f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")

        # Then: peak-targeted subgradient
        if time.monotonic() < deadline - 5:
            subg_dl = deadline - 5
            a_sub, sc_sub = _peak_target_subgrad(best_seq, subg_dl, n_iter=10000)
            if sc_sub < best_score:
                best_seq = a_sub.copy()
                best_score = sc_sub
                print(f"  NEW BEST: {best_score:.12f} [final_subgrad] t={time.monotonic()-t_start:.1f}s")

        # Ultra-fine L-BFGS with tiny beta
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_tiny = mb * 0.000001
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _LbfgsProblem(beta_tiny, n)
            remaining = deadline - 3 - time.monotonic()
            max_iter = min(3000, max(50, int(remaining / 0.02)))
            try:
                result = minimize(prob.f, z, jac=prob.g, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-20,
                                           'gtol': 1e-16, 'maxcor': 200})
                a_ultra = _softmax(result.x)
                sc_ultra = _score(a_ultra)
                if sc_ultra < best_score:
                    best_seq = a_ultra.copy()
                    best_score = sc_ultra
                    print(f"  NEW BEST: {best_score:.12f} [ultra_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
