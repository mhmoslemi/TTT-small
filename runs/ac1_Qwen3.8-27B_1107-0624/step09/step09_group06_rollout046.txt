def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 13.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod2":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 7.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 31.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.25) ** 2) + np.exp(-10.0 * (t - 0.75) ** 2) + 0.1
        return a / a.sum()
    elif kind == "flat_taper":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        taper = max(1, n // 40)
        for i in range(taper):
            a[i] = (i + 1) / taper
            a[n - 1 - i] = (i + 1) / taper
        return a / a.sum()
    else:
        return np.ones(n) / n


def _nesterov_v2(a_init, deadline, min_remaining=50.0, K_init=20, K_max=1000, seed=0):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    rng = np.random.default_rng(seed)
    a_prev = a.copy()
    momentum = 0.9
    step_lr = 0.02
    K = K_init
    t = 0
    no_improve = 0
    stall = 0
    best_step_lr = step_lr

    while time.monotonic() + min_remaining < deadline:
        t += 1
        y = a + momentum * (a - a_prev)
        a_prev = a.copy()
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy

        c = _full_convolve(y, y)
        c_max = float(c.max())
        cur_sc = 2.0 * n * c_max

        if cur_sc < best_sc - 1e-14:
            best_sc = cur_sc
            best_a = y.copy()
            no_improve = 0
            best_step_lr = step_lr
        else:
            no_improve += 1

        # Adaptive K
        c_sorted_desc = np.sort(c)[::-1]
        if K < K_max:
            gap = c_sorted_desc[0] - c_sorted_desc[min(K, len(c_sorted_desc) - 1)]
            if gap < c_max * 0.005:
                K = min(K_max, int(K * 1.4) + 1)
        if K > 100 and no_improve > 200:
            gap_top50 = c_sorted_desc[0] - c_sorted_desc[49]
            if gap_top50 > c_max * 0.05:
                K = max(K_init, K // 2)

        c_idx = np.argsort(c)[::-1]
        active = c_idx[:min(K, len(c_idx))]

        grad = _conv_grad_multi(y, active)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-20:
            break

        step = step_lr / (gnorm + 1e-18)
        a_proj = y
        for _ in range(15):
            a_cand = y - step * grad_t
            a_proj = _project_simplex(a_cand)
            c_new = _full_convolve(a_proj, a_proj)
            sc_new = 2.0 * n * float(c_new.max())
            if sc_new <= cur_sc + 1e-12:
                break
            step *= 0.5

        a = a_proj

        # Schedule
        if no_improve > 100:
            step_lr = max(0.001, step_lr * 0.98)
        if no_improve > 300:
            stall += 1
            if stall > 10:
                break
            pert_mag = 0.003 * (1 + stall * 0.5)
            pert = pert_mag * rng.standard_normal(n)
            a_pert = _project_simplex(a + pert)
            c_pert = _full_convolve(a_pert, a_pert)
            sc_pert = 2.0 * n * float(c_pert.max())
            if sc_pert < best_sc:
                best_a = a_pert.copy()
                best_sc = sc_pert
            a = a_pert
            a_prev = a.copy()
            step_lr = best_step_lr
            no_improve = 0
            momentum = 0.9

        if t % 3000 == 0:
            print(f"    NesV2 t={t}: best={best_sc:.10f}, K={K}, lr={step_lr:.5f}")

    return best_a, best_sc


def _adam_optimize(a_init, deadline, min_remaining=50.0, seed=0):
    """ADAM optimizer on the log-sum-exp smoothed objective in z-space."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_cur = _full_convolve(a, a)
    c_max_cur = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < 1e-15:
        gap = c_max_cur * 0.01
    beta_init = max(gap, c_max_cur * 0.005)

    # ADAM hyperparameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    rng = np.random.default_rng(seed)

    # Adaptive beta schedule
    beta = beta_init
    beta_decay = 0.7
    stall = 0
    no_improve = 0

    while time.monotonic() + min_remaining < deadline:
        t += 1

        # Periodic beta reduction
        if t % 5000 == 0 and beta > c_max_cur * 0.0001:
            beta *= beta_decay
            m = np.zeros(n)
            v = np.zeros(n)
            lr = 0.01

        g = _smograd(z, beta, n)
        gnorm = np.linalg.norm(g)

        if not np.isfinite(gnorm) or gnorm < 1e-20:
            break

        # ADAM update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Convert back to simplex
        a_new = _softmax(z)
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            z = np.log(np.maximum(a, 1e-30))
            z = z - z.mean()
        else:
            no_improve += 1
            # Revert z if score got too bad
            if sc_new > best_sc * 1.05:
                a = best_a.copy()
                z = np.log(np.maximum(a, 1e-30))
                z = z - z.mean()
                m = np.zeros(n)
                v = np.zeros(n)

        if no_improve > 5000:
            stall += 1
            if stall > 5:
                break
            # Reduce lr and beta
            lr *= 0.5
            beta *= 0.5
            no_improve = 0
            # Small perturbation
            pert = 0.001 * rng.standard_normal(n)
            z = z + pert
            m = np.zeros(n)
            v = np.zeros(n)

        if t % 5000 == 0:
            print(f"    ADAM t={t}: best={best_sc:.10f}, beta={beta:.6f}, lr={lr:.6f}")

    return best_a, best_sc


def _mirror_descent(a_init, deadline, min_remaining=50.0, seed=0):
    """Mirror descent (exponentiated gradient) with Nesterov acceleration."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    a_prev = a.copy()
    momentum = 0.7
    eta = 0.05
    K = 30
    t = 0
    no_improve = 0
    stall = 0

        # Nesterov lookahead
        y = a + momentum * (a - a_prev)

        y = y / y.sum()

        c_max = float(c.max())
        cur_sc = 2.0 * n * c_max

            best_sc = cur_sc
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        c_sorted = np.sort(c)[::-1]
        if K < 500:
            gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
            if gap < c_max * 0.005:
