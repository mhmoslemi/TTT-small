        # Compute gradient
        grad = np.zeros(n)

                    grad[i] += a[j]

Optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Multiplicative weight descent (fast O(n log n) per iteration)
2. L-BFGS with log-sum-exp continuation (aggressive beta schedule)
3. Active-set Newton with FFT-based Hessian
4. Multi-peak deflation
5. Multiple diverse initializations and sequence lengths
6. Symmetrization exploitation (reversal invariance)
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    """Compute the gradient contribution: for each i, sum_k A_vec[k] * a[k-i] (valid)."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj_factory(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec, rhs, max_cg=200, tol=1e-13):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _newton_step(a, c, c_max, tol_frac):
    n = len(a)
    active = c >= (c_max - tol_frac * c_max)
    if not active.any():
        active[np.argmax(c)] = True
    m = int(active.sum())
    if m < 2 or m > 20 * n:
        return None, None

    A_vec = active.astype(np.float64) / m
    g = 2.0 * _corr_a_A(a, A_vec, n)
    g_t = g - np.mean(g)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-14:
        return None, None

    mu = 1e-8 / m + 1e-12
    A_vec_copy = A_vec
    n_copy = n
    mu_copy = mu

    def _hess_vp(v):
        return 2.0 * _corr_a_A(v, A_vec_copy, n_copy) + mu_copy * v

    u = _cg_solve(_hess_vp, g_t, max_cg=300)
    w = _cg_solve(_hess_vp, np.ones(n), max_cg=300)
    w_sum = float(w.sum())

    if abs(w_sum) < 1e-15:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w

    dx = dx - np.mean(dx)
    return dx, A_vec


def _multiplicative_descent(a, deadline, max_iters=5000, threshold=0.98):
    """Fast multiplicative weight descent using FFT for O(n log n) per iteration."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    best_sc = _score(a)
    best_a = a.copy()
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        S = (c > c_max * threshold).astype(np.float64)

        # Compute grad[i] = sum_k S[k] * a[k-i] (valid indices only)
        # grad[i] = sum_m a[m] * S[i+m] for valid m, i+m
        # = conv(a, S_rev)[n-1-i] where S_rev = S[::-1]
        S_rev = S[::-1]
        conv_as = _full_convolve(a, S_rev)
        grad = conv_as[n - 1::-1]  # grad[i] = conv_as[n-1-i]

        # Adaptive step size
        g_max = grad.max()
        if g_max < 1e-15:
            break

        # Try a few eta values
        best_eta = None
        best_a_try = None
        best_sc_try = best_sc
        for eta in [0.001, 0.003, 0.01, 0.03, 0.1]:
            a_try = a * np.exp(-eta * grad / g_max)
            a_try = np.maximum(a_try, 1e-15)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_a_try = a_try
                best_eta = eta

        if best_a_try is not None:
            a = best_a_try
            if best_sc_try < best_sc - 1e-15:
                best_sc = best_sc_try
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1
        else:
            stall_count += 1

        if stall_count > 30:
            break

        # Occasionally try symmetrization
        if it % 100 == 50:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = np.maximum(a_sym, 1e-15)
            a_sym = a_sym / a_sym.sum()
            sc_sym = _score(a_sym)
            if sc_sym < best_sc - 1e-14:
                best_sc = sc_sym
                best_a = a_sym.copy()
                a = a_sym

    return best_a, best_sc


def _lbfgs_continuation(a_init, deadline, min_time_remaining=30.0, max_betas=30):
    """L-BFGS with aggressive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap_idx = min(20, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[gap_idx]
    base_beta = max(gap * 0.2, max_c * 0.003)

    # Aggressive schedule down to very small beta
    min_beta = max_c * 1e-8
    if min_beta > 1e-15:
        n_betas = min(max_betas, 30)
        log_betas = np.linspace(np.log(base_beta), np.log(min_beta), n_betas)
        betas = np.exp(log_betas)
    else:
        betas = [base_beta]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        est_time = 0.002 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 12))))
        # Budget per beta stage
        stage_time = min(remaining * 0.8, 120.0)
        max_iter = min(max_iter, max(50, int(stage_time / est_time)))

        try:
            def _obj(z_val):
                return _lsobj_factory(z_val, beta, n)

            def _grad(z_val):
                return _lsgrad_factory(z_val, beta, n)

            result = minimize(
                _obj, z, jac=_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 100, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.12f}")
        except Exception as ex:
            print(f"    L-BFGS beta={beta:.2e}: exception {ex}")

    return current_a


def _active_set_newton(a_init, deadline, max_outer=200):
    """Active-set Newton method for the minmax problem."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Try multiple tolerances
        if outer < max_outer // 3:
            tol_list = [0.03, 0.015, 0.008, 0.004]
        elif outer < 2 * max_outer // 3:
            tol_list = [0.004, 0.002, 0.001, 0.0005]
        else:
            tol_list = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]

        best_improvement = 0
        best_a_new = None

        for tol_frac in tol_list:
            dx, A_vec = _newton_step(a, c, c_max, tol_frac)
            if dx is None:
                continue

            # Line search
            t = 1.0
            found = False
            for ls in range(60):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 1e-15)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < c_max - 1e-14 * t:
                    improvement = c_max - f_try
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_a_new = a_try.copy()
                    found = True
                    break
                t *= 0.5
            if found:
                break

        if best_a_new is not None:
            a = best_a_new
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

            c_new = _full_convolve(a, a)
            if abs(float(c_new.max()) - c_max) < 1e-15:
                break
        else:
            # Try deflation as fallback
            peak_mask = c >= c_max * 0.995
            w = np.zeros_like(c)
            w[peak_mask] = 1.0 / max(peak_mask.sum(), 1)
            grad = 2.0 * _corr_a_A(a, w, n)
            g_t = grad - np.mean(grad)

            for strength in [0.0001, 0.0003, 0.001, 0.003, 0.01]:
                a_try = a - strength * g_t
                a_try = np.maximum(a_try, 1e-15)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    a = a_try
                    best_sc = sc_try
                    best_a = a_try.copy()
                    break
            else:
                break

    return best_a, best_sc
