"""
Optimized step function coefficient search using multi-phase gradient-based
methods with active-set refinement, Nesterov acceleration, and multi-restart
strategies to minimize 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_soft_structured(a, beta, n):
    """Gradient of soft_max(conv(a,a)) w.r.t. a using FFT structure."""
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


class _SoftObj:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        return 2.0 * self.n * _soft_max_value(c, self.beta)

    def grad(self, z):
        a = _softmax(z)
        grad_a = _grad_soft_structured(a, self.beta, self.n)
        W = float(np.dot(grad_a, a))
        return a * (grad_a - W)


def _active_set_grad(a, tol_frac):
    """Compute gradient at active set (peaks of conv)."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    active = c >= (c_max - tol_frac * c_max)
    n_active = int(active.sum())
    if n_active == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        n_active = 1
    a_rev = a[::-1]
    w = np.zeros(len(c))
    w[active] = 1.0 / n_active
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad, c, c_max


def _lbfgs_optimize(a_init, deadline, min_remaining=30.0, verbose=True, tag=""):
    """L-BFGS with adaptive beta schedule."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    
    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < c0_max * 0.002:
        gap = c0_max * 0.002
    
    multipliers = np.array([
        1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002,
        0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6
    ])
    betas = gap * multipliers
    
    t_start = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:

        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break
        
        size_factor = max(1.0, n / 8192.0)

                prob, z, jac=prob.grad, method='L-BFGS-B',

            a_new = _softmax(result.x)

                z = np.log(np.maximum(a_new, 1e-30))

def _active_set_equalize(a_init, deadline, min_remaining=30.0, verbose=True, tag=""):
    """Aggressive active-set equalization with momentum."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    tol_frac = 0.003
    min_tol = 0.00005
    momentum = np.zeros(n)
    beta_mom = 0.7
    no_improve = 0
    max_no_improve = 200
    
    t_start = time.monotonic()
    it = 0
    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-12:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.5)
            if tol_frac <= min_tol * 1.2:
                break
            no_improve = 0
            momentum = np.zeros(n)
        
        grad, c, c_max = _active_set_grad(a, tol_frac)
        
        # Project gradient to tangent space of simplex
        grad_proj = grad - np.mean(grad)
        
        # Momentum
        momentum = beta_mom * momentum + (1 - beta_mom) * grad_proj
        g_norm_sq = float(np.dot(momentum, momentum))
        if g_norm_sq < 1e-20:
            break
        
        # Step size: try to reduce c_max to the mean of active set
        active = c >= (c_max - tol_frac * c_max)
        active_mean = float(c[active].mean())
        target_reduction = (c_max - active_mean) * 0.5
        step = target_reduction / np.sqrt(g_norm_sq)
        step = min(step, 0.05)
        
        a_new = a - step * momentum
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        # Line search
        sc_new = _score(a_new)
        if sc_new >= current_sc:
            for factor in [0.5, 0.25, 0.1, 0.05]:
                a_try = a - step * factor * momentum
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc:
                    a_new = a_try
                    break
            else:
                # No improvement
                tol_frac = max(min_tol, tol_frac * 0.9)
                continue
        
        a = a_new
        it += 1
        if verbose and it % 500 == 0:
            print(f"    {tag} active-set it={it}: score={best_sc:.10f}, tol={tol_frac:.2e}")
    
    return best_a, best_sc


def _perturb(a, magnitude=0.01, rng=None):
    """Add small random perturbation and renormalize."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    # Add perturbation in log domain
    z = np.log(np.maximum(a, 1e-30))
    noise = rng.normal(0, magnitude, n)
    z += noise
    a_new = _softmax(z)
    return a_new


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with adaptive smoothing, momentum-based active-set equalization,
perturbation-based restarts, and multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_soft_structured(a, beta, n):
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


class _SoftObj:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        return 2.0 * self.n * _soft_max_value(c, self.beta)

    def grad(self, z):
        a = _softmax(z)
        grad_a = _grad_soft_structured(a, self.beta, self.n)
        W = float(np.dot(grad_a, a))
        return a * (grad_a - W)


def _active_set_grad(a, tol_frac):
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    thr = c_max - tol_frac * c_max
    active = c >= thr
    n_active = int(active.sum())
    if n_active == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        n_active = 1
    a_rev = a[::-1]
    w = np.zeros(len(c))
    w[active] = 1.0 / n_active
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad, c, c_max


def _lbfgs_optimize(a_init, deadline, min_remaining=0.0, verbose=True, tag=""):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c0_max * 0.002:
        gap = c0_max * 0.002

    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.008, 0.004,
        0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6
    ])
    betas = gap * multipliers

    t_start = time.monotonic()

        if time.monotonic() + min_remaining > deadline:

            break

        est_iter = 0.003 * size_factor

        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',

                         'maxcor': 200, 'maxls': 100}

def _active_set_equalize(a_init, deadline, min_remaining=0.0, verbose=True, tag=""):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_frac = 0.004
    min_tol = 0.00003
    momentum = np.zeros(n)
    beta_mom = 0.6
    no_improve = 0
    max_no_improve = 300
    step_decay = 1.0

    it = 0
    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.5)
            if tol_frac <= min_tol * 1.3:
                break
            no_improve = 0
            momentum *= 0.5
            step_decay = max(0.3, step_decay * 0.7)

        grad, c, c_max = _active_set_grad(a, tol_frac)
        grad_proj = grad - np.mean(grad)

        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad_proj
        g_norm_sq = float(np.dot(momentum, momentum))
        if g_norm_sq < 1e-22:
            break

        # Adaptive step: target reducing c_max toward active mean
        active = c >= (c_max - tol_frac * c_max)
        n_active = int(active.sum())
        if n_active > 1:
            active_mean = float(c[active].mean())
            target_red = (c_max - active_mean) * 0.4
        else:
            target_red = c_max * 0.001

        step = step_decay * target_red / np.sqrt(g_norm_sq)
        step = min(step, 0.08)
        step = max(step, 1e-6)

        a_new = a - step * momentum
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new >= current_sc:
            found = False
            for factor in [0.5, 0.2, 0.1, 0.05]:
                a_try = a - step * factor * momentum
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-14:
                    a_new = a_try
                    found = True
                    break
            if not found:
                tol_frac = max(min_tol, tol_frac * 0.92)
                step_decay = max(0.1, step_decay * 0.8)
                continue

        a = a_new
        it += 1
        if no_improve == 0:
            step_decay = min(1.0, step_decay * 1.02)
            tol_frac = max(min_tol, tol_frac * 0.985)

        if verbose and it % 2000 == 0:
            print(f"    {tag} active-set it={it}: score={best_sc:.10f}, "
                  f"tol={tol_frac:.2e}, n_act={n_active}")

    return best_a, best_sc


def _perturb(a, magnitude, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    noise = rng.normal(0, magnitude, n)
    z += noise
    a_new = _softmax(z)
    return a_new


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":
        a = np.ones(n)
        fade = max(1, n // 80)
        for i in range(fade):
            w = (i + 1) / (fade + 1)
            a[i] = w
            a[n - 1 - i] = w
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _nesterov_gradient(a_init, deadline, min_remaining=0.0, verbose=True, tag=""):

    if gap < c0_max * 0.002:
        gap = c0_max * 0.002

    # Start with moderate beta and decrease
    beta_start = gap * 0.01
    beta_end = gap * 1e-6
    n_inner = 200
    betas = np.geomspace(beta_start, beta_end, n_inner)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    v = np.zeros(n)  # velocity for Nesterov

    momentum_coeff = 0.9
    it = 0
    for bi, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break

        # Multiple steps per beta level
        n_steps = max(5, int(50 * n / 16384))
        for _ in range(n_steps):
            if time.monotonic() + min_remaining > deadline:
                break

            z_look = z + momentum_coeff * v
            a_look = _softmax(z_look)
            grad = np.zeros(n)
            c = _full_convolve(a_look, a_look)

            # Step

            lr = min(lr, 0.5)
            v = momentum_coeff * v - lr * grad
            z_new = z + v

            # Check improvement
            a_new = _softmax(z_new)
            sc_new = float(2.0 * n * _full_convolve(a_new, a_new).max())
            if sc_new < best_sc - 1e-13:

            z = z_new
            it += 1

        if verbose and (bi % 40 == 0 or bi == n_inner - 1):
            print(f"    {tag} Nesterov beta={beta:.1e} ({bi+1}/{n_inner}): "

    p1_time = 0.30 * total_budget
    p1_deadline = t0 + p1_time
    print(f"\n[P1] L-BFGS main (budget={p1_time:.0f}s)")
    a1, sc1 = _lbfgs_optimize(best_seq, p1_deadline, min_remaining=0, verbose=True, tag="P1")
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  P1 improved to: {best_score:.10f}")
    else:
        print(f"  P1 no improvement (best={best_score:.10f})")

    # === Phase 2: Active-set equalization (15%) ===
    p2_time = 0.15 * total_budget
    p2_deadline = t0 + p1_time + p2_time
    print(f"\n[P2] Active-set equalize (budget={p2_time:.0f}s)")
    a2, sc2 = _active_set_equalize(best_seq, p2_deadline, min_remaining=0, verbose=True, tag="P2")
    if sc2 < best_score:
        best_score = sc2
        best_seq = a2.copy()
        print(f"  P2 improved to: {best_score:.10f}")
    else:
        print(f"  P2 no improvement (best={best_score:.10f})")

    # === Phase 3: L-BFGS refinement (20%) ===
    p3_time = 0.20 * total_budget
    p3_deadline = t0 + p1_time + p2_time + p3_time
    print(f"\n[P3] L-BFGS refine (budget={p3_time:.0f}s)")
    a3, sc3 = _lbfgs_optimize(best_seq, p3_deadline, min_remaining=0, verbose=True, tag="P3")
    if sc3 < best_score:
        best_score = sc3
        best_seq = a3.copy()
        print(f"  P3 improved to: {best_score:.10f}")

    # === Phase 4: Nesterov gradient (10%) ===
    p4_time = 0.10 * total_budget
    p4_deadline = t0 + p1_time + p2_time + p3_time + p4_time
    print(f"\n[P4] Nesterov (budget={p4_time:.0f}s)")
    a4, sc4 = _nesterov_gradient(best_seq, p4_deadline, min_remaining=0, verbose=True, tag="P4")
    if sc4 < best_score:
        best_score = sc4
        best_seq = a4.copy()
        print(f"  P4 improved to: {best_score:.10f}")

    # === Phase 5: Perturbation restarts (10%) ===
    p5_time = 0.10 * total_budget
    p5_deadline = t0 + p1_time + p2_time + p3_time + p4_time + p5_time
    print(f"\n[P5] Perturb restarts (budget={p5_time:.0f}s)")
    perturb_time_each = p5_time / 3.0

    for attempt in range(3):
        if time.monotonic() > p5_deadline:
            break
        mags = [0.003, 0.008, 0.015]
        a_pert = _perturb(best_seq, mags[attempt], rng)
        a_opt, sc_opt = _lbfgs_optimize(a_pert, min(p5_deadline, time.monotonic() + perturb_time_each),
                                         min_remaining=0, verbose=False, tag=f"P5.{attempt}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"  P5.{attempt} (mag={mags[attempt]}): improved to {best_score:.10f}")

    # === Phase 6: Multi-n exploration (8%) ===
    p6_time = 0.08 * total_budget
    p6_deadline = min(deadline - 0.07 * total_budget, t0 + p1_time + p2_time + p3_time + p4_time + p5_time + p6_time)
    print(f"\n[P6] Multi-n (budget={p6_time:.0f}s)")

    n_options = sorted(set([
        max(4096, int(n_main * 0.75)),
        max(4096, int(n_main * 1.25)),
        max(4096, int(n_main * 1.5)),
        max(4096, int(n_main * 0.5)),
    ]))
    n_options = [x for x in n_options if x != n_main and x <= 65536]

    time_per_n = max(20.0, p6_time / max(1, len(n_options)))
    for n_try in n_options:
        if time.monotonic() > p6_deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        n_deadline = min(p6_deadline, time.monotonic() + time_per_n)
        a6, sc6 = _lbfgs_optimize(a_r, n_deadline, min_remaining=0, verbose=False, tag=f"P6.n{n_try}")
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  n={n_try}: improved to {best_score:.10f}")

    # === Phase 7: Final polish (7%) ===
    p7_time = 0.07 * total_budget
    p7_deadline = deadline - 1.0
    print(f"\n[P7] Final polish (budget={p7_time:.0f}s)")

    a7, sc7 = _active_set_equalize(best_seq, p7_deadline - p7_time * 0.4,
                                    min_remaining=0, verbose=False, tag="P7a")
    if sc7 < best_score:
        best_score = sc7
        best_seq = a7.copy()
        print(f"  P7a: {best_score:.10f}")

    a8, sc8 = _lbfgs_optimize(best_seq, p7_deadline, min_remaining=0, verbose=False, tag="P7b")
    if sc8 < best_score:
        best_score = sc8
        best_seq = a8.copy()
        print(f"  P7b: {best_score:.10f}")

    # Final
    print(f"\n{'=' * 60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t0:.1f}s")
    return [float(x) for x in final]
