```python
def _project_simplex(v):
    """Project v onto the probability simplex {x : sum(x)=1, x>=0}."""
    n = len(v)
    # Algorithm from Wang & Carreira-Perpinan (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

**Key insight I haven't tried: The problem is equivalent to a max-flow / optimal transport problem in disguise.**

The autocorrelation conv(a,a)[k] represents the "mass" of pairs (i, k-i) that sum to k. We want to distribute this mass as evenly as possible across all k values.

This is like: given a distribution a on {0,...,n-1}, the "sum distribution" of two independent draws from a is conv(a,a)/sum(a)^2. We want this sum distribution to be as flat as possible.

The most flat distribution on {0,...,2n-2} is the uniform distribution with each value = 1/(2n-1). The question is: what non-negative a minimizes the L-infinity distance between conv(a,a)/sum(a)^2 and the uniform distribution?

This is a moment problem! The moments of the sum distribution are determined by the moments of a. Specifically:
- E[X+Y] = 2*E[X] (where X,Y are iid from a)
- E[(X+Y)^2] = 2*E[X^2] + 2*(E[X])^2
- etc.

The uniform distribution on {0,...,2n-2} has mean n-1 and variance (2n-1)^2-1)/12.

For a to produce a sum distribution close to uniform, we need:
- E[X] = (n-1)/2 (the mean of the sum is n-1, so the mean of X is (n-1)/2)
- E[X^2] should be appropriate

This suggests that a good starting point is a distribution with mean (n-1)/2 and appropriate variance. A symmetric distribution centered at (n-1)/2 would have E[X] = (n-1)/2 automatically.

```python
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _cmax(a):
    """Return just max of autocorrelation (for normalized sum=1 sequences)."""
    c = _full_convolve(a, a)
    return float(c.max()), c


def _project_simplex(v):
    """Project v onto probability simplex {x: sum(x)=1, x>=0} in O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _active_gradient(a, c, tol_frac):
    """Compute active-set gradient using FFT-based cross-correlation.
    
    Returns grad[j] = sum_{k active} w[k]*(a[k-j] + a[j]) * [0<=k-j<n] / sum(w)
    """
    n = len(a)
    c_max = float(c.max())
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    n_active = int(active.sum())
    if n_active == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
        n_active = 1

    # Inverse-gap weighting
    eps = 1e-14 * c_max + 1e-30
    w = np.zeros_like(c)
    w[active] = 1.0 / (c_max - c[active] + eps)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = active.astype(float)
        w_sum = w.sum()
    w = w / w_sum

    # h[j] = sum_{m=0}^{n-1} a[m] * w[m+j] via FFT cross-correlation
    nw = len(w)  # 2n-1
    N = _next_pow2(n + nw - 1)
    A = np.fft.rfft(a, N)
    W_rev = np.fft.rfft(w[::-1], N)
    conv_result = np.fft.irfft(A * W_rev, N)
    # h[j] = conv_result[nw-1-j] for j=0,...,n-1
    # conv_result has length n+nw-1 = 3n-2
    h = conv_result[nw - 1 : nw - 1 - n : -1]  # h[0]..h[n-1]

    # W_j = sum_{m=0}^{n-1} w[m+j] (sliding window sum of w)
    # w has length 2n-1, we need windows of size n starting at j=0..n-1
    # W_j = prefix[j+n] - prefix[j] where prefix is cumulative sum of w (padded)
    w_padded = np.zeros(3 * n)
    w_padded[:nw] = w
    prefix = np.concatenate([[0.0], np.cumsum(w_padded)])
    W_j = prefix[n : 2 * n] - prefix[0 : n - 1 + 1]  # W_j = sum w[j:j+n]
    # More carefully: W_j = sum_{m=0}^{n-1} w[m+j] = prefix[j+n]-prefix[j]
    W_j = prefix[n:2 * n + 1] - prefix[0:n + 1]

    grad = h + a * W_j
    # Project onto tangent space of sum(a)=1
    grad = grad - np.mean(grad)
    return grad


def _active_gradient_simple(a, c, tol_frac):
    """Simpler active-set gradient using direct convolution approach."""
    n = len(a)
    c_max = float(c.max())
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    n_active = int(active.sum())
    if n_active == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
        n_active = 1

    eps = 1e-14 * c_max + 1e-30
    w = np.zeros_like(c)
    w[active] = 1.0 / (c_max - c[active] + eps)
    w = w / w.sum()

    # grad[j] = sum_k w[k] * (a[k-j] + a[j]) * [0<=k-j<n]
    # = sum_k w[k]*a[k-j]*[0<=k-j<n] + a[j]*sum_k w[k]*[0<=k-j<n]
    # First term: cross-correlation of a and w
    # Second term: a[j] * (number of active k in [j, j+n-1] weighted by w)
    
    # Use the convolution approach:
    # sum_k w[k]*a[k-j] = (w_rev * a)[something]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    # wc[m] = sum_k w[k]*a_rev[m-k] = sum_k w[k]*a[n-1-(m-k)] = sum_k w[k]*a[n-1-m+k]
    # We want sum_k w[k]*a[k-j]. Let n-1-m+k = k-j => n-1-m = -j => m = n-1+j
    # So first term = wc[n-1+j] for j=0,...,n-1
    first_term = wc[n - 1 : 2 * n - 1]
    
    # Second term: a[j] * sum_{k: 0<=k-j<n} w[k] = a[j] * sum_{k=j}^{j+n-1} w[k]
    # w has length 2n-1. For j in [0,n-1], k ranges from j to j+n-1, which is within [0, 2n-2].
    # W_j = sum_{k=j}^{j+n-1} w[k]
    w_cumsum = np.concatenate([[0.0], np.cumsum(w)])
    W_j = w_cumsum[n:n + n] - w_cumsum[0:n]
    
    grad = first_term + a * W_j
    grad = grad - np.mean(grad)
    return grad


def _protein_descent(a_init, deadline, tol_start=0.1, tol_end=0.001,
                     lr_init=0.02, momentum=0.8, min_time=5.0):
    """Projected gradient with momentum and adaptive active set."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    _, best_cm = _cmax(a)
    best_sc = float(best_cm.max()) * 2.0 * n
    v = np.zeros(n)
    tol_frac = tol_start
    lr = lr_init
    no_improve = 0
    max_stall = 300
    t_start = time.monotonic()

    while time.monotonic() < deadline - min_time:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        grad = _active_gradient_simple(a, c, tol_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Momentum update
        v = momentum * v - lr * grad / (gnorm + 1e-30)
        v_norm = np.linalg.norm(v)
        if v_norm > 0:
            v = v / v_norm * min(v_norm, 0.1)

        # Line search
        alpha = 1.0
        improved = False
        for _ in range(60):
            a_step = a + alpha * v
            a_try = _project_simplex(a_step)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            v *= 0.5
            no_improve += 1
            if no_improve % 30 == 0:
                tol_frac = max(tol_end, tol_frac * 0.7)
                lr *= 0.7
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(tol_end, tol_frac * 0.5)
                lr = lr_init * 0.5
                no_improve = 0
        else:
            if tol_frac > tol_end:
                tol_frac = max(tol_end, tol_frac * 0.99)
            lr = min(lr_init * 2, lr * 1.005)

    return best_a, _score(best_a)


def _peak_shaving(a_init, deadline, min_time=5.0):
    """Focus exclusively on reducing the single maximum peak."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.01
    no_improve = 0

    while time.monotonic() < deadline - min_time:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Find the argmax
        k_max = int(np.argmax(c))

        # Gradient of conv(a,a)[k_max] w.r.t. a[j] = a[k_max-j] + a[j] (when valid)
        k = k_max
        # d/d(a[j]) conv(a,a)[k] = a[k-j] + a[j] for j in [max(0,k-n+1), min(n-1,k)]
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)

        grad = np.zeros(n)
        for j in range(j_lo, j_hi + 1):
            ki = k - j
            if 0 <= ki < n:
                if ki == j:
                    grad[j] = 2.0 * a[j]
                else:
                    grad[j] = a[ki] + a[j]

        # Project onto tangent space
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        grad = grad / gnorm

        # Line search
        alpha = lr
        improved = False
        for _ in range(40):
            a_try = _project_simplex(a - alpha * grad)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            lr *= 0.8
            if no_improve > 200:
                a = best_a.copy()
                lr = 0.01
                no_improve = 0
        else:
            lr = min(0.05, lr * 1.02)

    return best_a, _score(best_a)


def _logsumexp_objective(z, beta, n):
    a = np.exp(z - np.max(z))
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _logsumexp_gradient(z, beta, n):
    a = np.exp(z - np.max(z))
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs_smoothed(a_init, deadline, min_time=30.0):
    """L-BFGS with log-sum-exp smoothing and adaptive beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap * 0.5, max_c * 0.003)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.008, 0.004, 0.002,
                                  0.001, 0.0005, 0.0002, 0.0001,
                                  0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        remaining = deadline - min_time - time.monotonic()
        if remaining < 10:
            break

        est_time = 0.003 * max(1.0, n / 16384.0)
        max_iter = min(8000, max(50, int(remaining / (est_time * 15))))

        def func(z_val):
            return _logsumexp_objective(z_val, beta, n)

        def grad(z_val):

        try:

                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bipolar":
        # Two peaks at 1/4 and 3/4
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.75) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos4":
        t = np.linspace(0, 1, n)
        a = (0.5 * (1.0 + np.cos(2.0 * np.pi * t))) ** 4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _quick_optimize(a_init, deadline, min_time=5.0):
    """Combined optimization: projected gradient + peak shaving."""
    a = _protein_descent(a_init, min(deadline - min_time, time.monotonic() + 30),
                         tol_start=0.05, tol_end=0.001,
                         lr_init=0.02, momentum=0.8, min_time=3)
    if time.monotonic() < deadline - min_time:
        a2, sc2 = _peak_shaving(a, deadline, min_time=min_time)
        if sc2 < _score(a):
            a = a2
    return a, _score(a)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    1. Careful projected gradient with adaptive active set
    2. Peak shaving
    3. L-BFGS with log-sum-exp smoothing
    4. Simulated annealing ILS
    5. Multi-n exploration
    6. Final ultra-fine polish
    """
    global t_start
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_default = 16384
        parent = _make_profile(n_default, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_default}, score={best_score:.10f}")

    n_main = len(best_seq)

    # === Phase 1: Projected gradient with adaptive active set (250s) ===
    print(f"\n[P1] Projected gradient (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = t_start + 250
    a1, sc1 = _protein_descent(best_seq, p1_deadline,
                               tol_start=0.1, tol_end=0.0005,
                               lr_init=0.03, momentum=0.85, min_time=10)
    if sc1 < best_score:
        best_seq = a1
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [P1-projgrad] t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Peak shaving (150s) ===
    print(f"\n[P2] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(t_start + 400, deadline - 900)
    if time.monotonic() < p2_deadline:
        a2, sc2 = _peak_shaving(best_seq, p2_deadline, min_time=10)
        if sc2 < best_score:
            best_seq = a2
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [P2-peakshave] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: L-BFGS with smoothing (300s) ===
    print(f"\n[P3] L-BFGS smoothed (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(t_start + 700, deadline - 600)
    if time.monotonic() < p3_deadline:
        a3, sc3 = _run_lbfgs_smoothed(best_seq, p3_deadline, min_time=30)
        if sc3 < best_score:
            best_seq = a3
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [P3-lbfgs] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Simulated annealing ILS (400s) ===
    print(f"\n[P4] ILS with SA (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(t_start + 1100, deadline - 400)
    if time.monotonic() < p4_deadline:
        a4, sc4 = _ils_annealing(best_seq, p4_deadline, rng, min_time=50)
        if sc4 < best_score:
            best_seq = a4
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [P4-ils-sa] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Multi-n exploration (300s) ===
    print(f"\n[P5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    p5_deadline = min(t_start + 1400, deadline - 200)
    n_candidates = sorted(set([n_main, n_main * 2, n_main // 2,
                                32768, 49152, 24576, 16384, 65536]),
                          key=lambda x: abs(x - n_main))
    n_tried = set()
    for n_try in n_candidates:
        if time.monotonic() > p5_deadline - 30:
            break
        if n_try in n_tried or n_try < 4096 or n_try > 65536:
            n_tried.add(n_try)
            continue
        n_tried.add(n_try)
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        if sc_init < best_score - 0.01:
            # Skip if resampling alone is much worse (will likely not recover)
            # Still try but with less time
            pass
        a_opt, sc_opt = _quick_optimize(a_init, min(p5_deadline, time.monotonic() + 45),
                                         min_time=5)
        if sc_opt < best_score:
            best_seq = a_opt
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [P5-multin] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Second round of gradient + peak shave (150s) ===
    print(f"\n[P6] Second round (t={time.monotonic()-t_start:.1f}s)")
    p6_deadline = min(deadline - 80, time.monotonic() + 150)
    if time.monotonic() < p6_deadline:
        a6, sc6 = _protein_descent(best_seq, min(p6_deadline - 50, time.monotonic() + 80),
                                    tol_start=0.01, tol_end=0.0001,
                                    lr_init=0.01, momentum=0.85, min_time=5)
        if sc6 < best_score:
            best_seq = a6
            best_score = sc6
        if time.monotonic() < p6_deadline - 10:
            a6b, sc6b = _peak_shaving(best_seq, p6_deadline, min_time=5)
            if sc6b < best_score:
                best_seq = a6b
                best_score = sc6b
                print(f"  NEW BEST: {best_score:.10f} [P6-round2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final ultra-fine L-BFGS (80s) ===
    print(f"\n[P7] Final L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 15:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_fin = mb * 0.00005
        remaining = deadline - 15 - time.monotonic()
        max_iter = min(3000, max(30, int(remaining / 0.02)))

        def func_fin(z_val):
            return _logsumexp_objective(z_val, beta_fin, n)

        def grad_fin(z_val):
            return _logsumexp_gradient(z_val, beta_fin, n)

        try:
            result = minimize(func_fin, z, jac=grad_fin, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-15, 'maxcor': 100})
            a_fin = np.exp(result.x - np.max(result.x))
            a_fin = a_fin / a_fin.sum()
            sc_fin = _score(a_fin)
            if sc_fin < best_score:
                best_seq = a_fin
                best_score = sc_fin
                print(f"  NEW BEST: {best_score:.10f} [P7-final] t={time.monotonic()-t_start:.1f}s")
        except Exception:
            pass

    # === Final tiny peak shave with whatever time remains ===
    remaining = deadline - 5 - time.monotonic()
    if remaining > 3:
        a_final, sc_final = _peak_shaving(best_seq, deadline - 3, min_time=1)
        if sc_final < best_score:
            best_seq = a_final
            best_score = sc_final

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
