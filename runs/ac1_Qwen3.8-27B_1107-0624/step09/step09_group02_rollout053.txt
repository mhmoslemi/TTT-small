This is a key structural insight. The dual problem involves inverting a Hankel matrix, which can be done in O(n^2) using the Levinson-Durbin algorithm (though it's for Toeplitz, not Hankel... but Hankel = Toeplitz * J where J is the exchange matrix).

def _resample(a, n_new):
    n_old = len(a)

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:

    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Progressive k: start small, increase
    k_schedule = [5, 10, 20, 50, 100, 200, 500, 1000]
    rounds_per_k = max(5, max_rounds // len(k_schedule))

    for ki, k_act in enumerate(k_schedule):
        if time.monotonic() + min_remaining > deadline:
            break

        for rnd in range(rounds_per_k):
            if time.monotonic() + min_remaining > deadline:
                break

            c = _full_convolve(a, a)
            c_max = float(c.max())

            # Get top-k lags
            k_eff = min(k_act, 2 * n - 1)
            top_k = np.argsort(c)[::-1][:k_eff]
            p = np.zeros(2 * n - 1)
            p[top_k] = 1.0 / k_eff

            # Projected gradient with backtracking
            grad = _grad_acf(a, p, n)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break

            # Lipschitz estimate via power iteration
            # L ≈ 2 * max eigenvalue of Q = sum p_k H_k
            # Upper bound: L <= 2 * sum p_k * max_row_sum(H_k) = 2 * sum p_k * n = 2
            # Better: L <= 2 * max_k p_k * (something)... use 4 as safe bound
            L_est = 4.0
            lr = 1.0 / L_est

            for backtrack in range(30):
                a_new = _proj_simplex(a - lr * grad)
                c_new = _full_convolve(a_new, a_new)
                obj_new = float(np.dot(p, c_new))
                obj_old = float(np.dot(p, c))
                # Quadratic model decrease
                model_dec = 0.5 * lr * float(np.dot(grad, a - a_new))
                if obj_new <= obj_old - model_dec:
                    a = a_new
                    break
                lr *= 0.5
            else:
                if lr < 1e-12:
                    break
                a = _proj_simplex(a - lr * grad)

            sc = _score(a)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _run_momentum_pg(a_init, deadline, min_remaining=30.0):
    """Projected gradient with Nesterov momentum in log-space."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    momentum = np.zeros(n)
    beta_m = 0.85
    patience = 0
    max_patience = 30

    for step in range(3000):
        if time.monotonic() + min_remaining > deadline:
            break

        a_curr = _softmax(z)
        c = _full_convolve(a_curr, a_curr)
        c_max = float(c.max())

        # Top-k active set
        k_act = min(20, 2 * n - 1)
        top_k = np.argsort(c)[::-1][:k_act]
        p = np.zeros(2 * n - 1)
        p[top_k] = 1.0 / k_act

        grad_a = _grad_acf(a_curr, p, n)
        # Convert to z-gradient
        z_grad = a_curr * (grad_a - float(np.dot(a_curr, grad_a)))
        gnorm = np.linalg.norm(z_grad)
        if gnorm < 1e-14:
            break

        momentum = beta_m * momentum + (1 - beta_m) * z_grad
        lr = 0.05 / (np.linalg.norm(momentum) / (gnorm + 1e-15))
        z_new = z - lr * momentum
        z_new = np.clip(z_new, -40.0, 40.0)
        z_new = z_new - z_new.mean()
        z = z_new

        if step % 5 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros(n)
                    patience = 0
                    beta_m *= 0.95

    return best_a, best_sc


def _run_fine_lbfgs(a_init, deadline, min_remaining=20.0):
    """Very fine L-BFGS with a small number of betas for final polish."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(5, len(c_sorted) - 1)]
    base_beta = max(gap * 0.05, max_c * 1e-6)

    # Very tight betas
    ratios = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
              0.0005, 0.0002, 0.0001]

    for i, ratio in enumerate(ratios):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        beta = base_beta * ratio

        def obj_func(z_local):
            a_local = _softmax(z_local)
            c_local = _full_convolve(a_local, a_local)
            c_max_local = float(c_local.max())
            shifted = (c_local - c_max_local) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s_e = e.sum()
            if s_e < 1e-300:
                return 2.0 * n * c_max_local
            return 2.0 * n * (c_max_local + beta * np.log(s_e))

        def grad_func(z_local):
            a_local = _softmax(z_local)
            c_local = _full_convolve(a_local, a_local)
            c_max_local = float(c_local.max())
            shifted = (c_local - c_max_local) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s_e = e.sum()
            if s_e < 1e-300:
                w = np.zeros_like(c_local)
                w[int(np.argmax(c_local))] = 1.0
            else:
                w = e / s_e
            h = _grad_acf(a_local, w, n) / 2.0
            W_val = float(np.dot(w, c_local))
            return 4.0 * n * a_local * (h - W_val)

        est_time = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(1500, max(30, int(remaining / (est_time * 8))))

        try:
            result = minimize(
                obj_func, z,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 30, 'maxls': 40}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid multi-phase optimizer:
    Phase 1: L-BFGS continuation with geometric beta schedule
    Phase 2: Active-set QP with projected gradient
    Phase 3: Momentum projected gradient in log-space
    Phase 4: Fine L-BFGS polish
    Phase 5: Multi-length exploration
    Phase 6: Final ultra-tight active-set refinement
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  ** NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 32768
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: L-BFGS continuation (700s) ===
    p1_deadline = t_start + 700
    print(f"\n[Phase 1] L-BFGS Continuation (budget to {p1_deadline - t_start:.0f}s)")
    a1, sc1 = _run_lbfgs_phase(parent, p1_deadline, min_remaining=30.0)
    update_best(a1, sc1, "lbfgs_cont")

    # === Phase 2: Active-set QP refinement (350s) ===
    p2_deadline = min(t_start + 1050, deadline - 400)
    print(f"\n[Phase 2] Active-Set QP (budget to {p2_deadline - t_start:.0f}s)")
    if best_seq is not None:
        a2, sc2 = _run_active_set_qp(best_seq, p2_deadline, min_remaining=30.0, max_rounds=150)
        update_best(a2, sc2, "active_qp")

    # === Phase 3: Momentum PG (200s) ===
    p3_deadline = min(t_start + 1250, deadline - 300)
    print(f"\n[Phase 3] Momentum PG (budget to {p3_deadline - t_start:.0f}s)")
    if best_seq is not None:
        a3, sc3 = _run_momentum_pg(best_seq, p3_deadline, min_remaining=20.0)
        update_best(a3, sc3, "momentum_pg")

    # === Phase 4: Fine L-BFGS (250s) ===
    p4_deadline = min(t_start + 1500, deadline - 200)
    print(f"\n[Phase 4] Fine L-BFGS (budget to {p4_deadline - t_start:.0f}s)")
    if best_seq is not None:
        a4, sc4 = _run_fine_lbfgs(best_seq, p4_deadline, min_remaining=20.0)
        update_best(a4, sc4, "fine_lbfgs")

    # === Phase 5: Multi-length exploration (150s) ===
    p5_deadline = min(t_start + 1650, deadline - 120)
    print(f"\n[Phase 5] Multi-Length (budget to {p5_deadline - t_start:.0f}s)")
    n_tries = [49152, 16384, 65536]
    for n_try in n_tries:
        if time.monotonic() > p5_deadline - 40:
            break
        if n_try == n_parent:
            continue
        print(f"  Trying n={n_try}")
        if best_seq is not None:
            a_init_n = _resample(best_seq, n_try)
        else:
            a_init_n = _make_profile(n_try, "tapered")
        a_n, sc_n = _run_lbfgs_phase(a_init_n, p5_deadline, min_remaining=20.0)
        update_best(a_n, sc_n, f"n_{n_try}")
        # Quick QP refinement
        if sc_n < best_score + 0.01 and time.monotonic() < p5_deadline - 30:
            a_nq, sc_nq = _run_active_set_qp(a_n, p5_deadline, min_remaining=15.0, max_rounds=40)
            update_best(a_nq, sc_nq, f"n_{n_try}_qp")

    # === Phase 6: Final ultra-tight refinement (100s) ===
    p6_deadline = min(deadline - 10, t_start + 1780)
    print(f"\n[Phase 6] Final Polish (budget to {p6_deadline - t_start:.0f}s)")
    if best_seq is not None and time.monotonic() < p6_deadline - 5:
        # Tight active-set QP
        a6, sc6 = _run_active_set_qp(best_seq, p6_deadline, min_remaining=5.0, max_rounds=80)
        update_best(a6, sc6, "final_qp")

        # Fine L-BFGS
        if time.monotonic() < p6_deadline - 5:
            a6b, sc6b = _run_fine_lbfgs(best_seq, p6_deadline, min_remaining=5.0)
            update_best(a6b, sc6b, "final_lbfgs")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
