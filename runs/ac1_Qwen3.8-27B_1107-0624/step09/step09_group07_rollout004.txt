import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n-1:2*n-1]

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using adaptive
top-K log-sum-exp smoothing, subgradient methods with Nesterov
acceleration, and multi-scale exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_conv(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _smooth_obj_grad(z, beta, K, n):
    """Top-K log-sum-exp smoothed objective and gradient in log-domain."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c
    
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    
    ga = _grad_conv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


class _SmoothProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _smooth_obj_grad(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _smooth_obj_grad(z, self.beta, self.K, self.n)[1]


class _HingeProb:
    """Penalty-based smooth objective using sum of squared hinge losses."""
    def __init__(self, t_target, beta, n):
        self.t_target = t_target
        self.beta = beta
        self.n = n

    def __call__(self, z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        c = _full_convolve(a, a)
        # Hinge penalty: sum of max(0, c[k] - t)^beta
        hinge = np.maximum(c - self.t_target, 0.0)
        if self.beta == 2:
            penalty = np.sum(hinge * hinge)
        else:
            penalty = np.sum(hinge ** self.beta) / self.beta
        return 2.0 * self.n * (self.t_target + penalty)

    def grad(self, z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        c = _full_convolve(a, a)
        hinge = np.maximum(c - self.t_target, 0.0)
        if self.beta == 2:
            w = hinge  # d/dt of (t-target)^2 = 2(t-target), but we use (t-target)
        else:
            w = hinge ** (self.beta - 1)
        w = w / (w.sum() + 1e-30)
        ga = _grad_conv(a, w)
        W = float(np.dot(a, ga))
        grad_z = a * (ga - W)
        return grad_z


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive top-K smoothing."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a
    
    # Adaptive base beta based on the gap
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # Adaptive schedule
    schedule = [
        (1.5, 0.15), (1.0, 0.12), (0.7, 0.10), (0.5, 0.08),
        (0.3, 0.06), (0.2, 0.05), (0.1, 0.04), (0.05, 0.03),
        (0.03, 0.02), (0.01, 0.015), (0.005, 0.01), (0.002, 0.008),
        (0.001, 0.005), (0.0005, 0.003),
    ]
    
    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        beta = base_beta * beta_mult
        K = max(3, int((2 * n - 1) * K_frac))
        
        est_time = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(50, int(remaining / (est_time * 15))))
        
        prob = _SmoothProb(beta, K, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxcor': 120, 'maxls': 80}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            # Re-estimate base_beta after improvement
            c_new = _full_convolve(a_new, a_new)
            max_c_new = float(c_new.max())
            c_sorted_new = np.sort(c_new)[::-1]
            gap_new = c_sorted_new[0] - c_sorted_new[min(50, len(c_sorted_new) - 1)]
            if gap_new < max_c_new * 0.001:
                gap_new = max_c_new * 0.001
            base_beta = max(gap_new, max_c_new * 0.005)
        except Exception:
            pass
    
    return current_a


def _subgradient_nesterov(a_init, deadline, min_time_remaining=60.0):
    """Subgradient with Nesterov momentum and adaptive active set."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n_2nm1 = 2 * n - 1
    
    momentum = 0.85
    lr = 0.01
    lr_min = 1e-6
    lr_max = 0.05
    stall = 0
    max_stall = 100
    velocity = np.zeros(n)
    prev_a = None
    prev_g = None
    active_frac = 0.02
    
    iteration = 0
    t_start = time.monotonic()
    
    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive active set
        n_active = max(5, min(n_2nm1, int(n_2nm1 * active_frac)))
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_active]
        
        active_vals = c[active_idx]
        w = np.zeros(n_2nm1)
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w[active_idx] = (active_vals - active_vals.min()) / spread + 0.05
        else:
            w[active_idx] = 1.0 / n_active
        w = w / w.sum()
        
        g = _grad_conv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break
        
        # BB step size
        if prev_a is not None and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g_centered - prev_g
            s_norm = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm > 1e-18 and abs(sy) > 1e-18:
                lr_bb = s_norm / abs(sy)
                lr = np.clip(lr_bb, lr_min, lr_max)
        
        # Nesterov step
        velocity = momentum * velocity - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        
        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            for ls in range(30):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-16:
                    improved = True
                    break
                velocity = momentum * velocity * 0.5 - lr * 0.5 * g_centered
                a_try = a + velocity
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    velocity = np.zeros(n)
                    a_try = a
                    break
                a_try = a_try / s_try
                lr *= 0.7
        
        if improved:
            prev_a = a.copy()
            prev_g = g_centered.copy()
            a = a_try
            lr = min(lr_max, lr * 1.02)
            active_frac = max(0.003, active_frac * 0.98)
        else:
            velocity = np.zeros(n)
            lr = max(lr_min, lr * 0.5)
            active_frac = min(0.10, active_frac * 1.5)
        
        if stall > max_stall:
            active_frac = min(0.10, active_frac * 2.0)
            lr = max(lr_min, lr * 0.2)
            momentum = min(0.95, momentum + 0.03)
            stall = 0
        
        # Periodic L-BFGS restart
        if iteration % 300 == 0 and time.monotonic() - t_start > 60:
            if time.monotonic() + min_time_remaining + 60 < deadline:
                lbfgs_deadline = time.monotonic() + 50
                a_tmp = _run_lbfgs(a, lbfgs_deadline, min_time_remaining=5)
                sc_tmp = _score(a_tmp)
                if sc_tmp < best_sc:
                    best_a = a_tmp.copy()
                    best_sc = sc_tmp
                    a = a_tmp.copy()
                    velocity = np.zeros(n)
                    prev_a = None
                    prev_g = None
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == " hann_sq":
        t = np.linspace(0, 1, n)
        a = (0.5 + 0.5 * np.cos(2 * np.pi * t)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "fejer":
        # Fejér kernel-like: (sin(x)/x)^2 shape
        t = np.linspace(-1, 1, n)
        a = np.sinc(t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t) + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2) + 0.03
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        # Flat top window
        a = 1.0 + 0.5 * np.cos(np.pi * t / (0.3)) * (t < 0.3) + 0.5 * np.cos(np.pi * (1-t) / (0.3)) * (t > 0.7)
        # Simpler: just use a raised flat
        edge = 0.2
        a = np.ones(n)
        m = int(n * edge)
        for i in range(m):
            a[i] = 0.5 * (1 + np.cos(np.pi * i / m))
            a[n-1-i] = 0.5 * (1 + np.cos(np.pi * i / m))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    """Very fine active-set polish with adaptive tolerance."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.003
    tol = 0.003
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-16:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        # Active set: peaks within tol of max
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active
        
        g = _grad_conv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-16:
            break
        
        alpha = min(lr, 0.3 / (gnorm + 1e-15))
        improved = False
        for ls in range(50):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-16:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                tol = max(0.00005, tol * 0.6)
                lr *= 0.6
            if no_improve > 400:
                a = best_a.copy()
                tol = max(0.00005, tol * 0.4)
                lr = 0.002
                no_improve = 0
        else:
            tol = max(0.00005, tol * 0.96)
            lr = min(0.008, lr * 1.005)
    
    return best_a, best_sc
