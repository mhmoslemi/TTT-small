def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp_max_approx(z, beta, n):
    """Smooth approximation of max using log-sum-exp."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max, c
    return 2.0 * n * (c_max + beta * np.log(s)), c


def _grad_smooth_max(z, beta, n):
    """Gradient of the smooth max approximation."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)

def _active_set_optimize(a_init, deadline, min_time_remaining=30.0):
    """Active-set peak equalization with backtracking line search."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    K = min(10, n)
    lr = 0.01
    no_improve = 0
    max_no_improve = 300
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            K = min(K + 1, min(100, n))
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.8
            if no_improve >= max_no_improve:
                a = best_a.copy()
                lr = 0.01
                K = min(10, n)
                no_improve = 0
                break
        
        # Get active set
        top_k_idx = np.argsort(c)[::-1][:K]
        
        # Compute subgradient: average gradient of top-K peaks
        w_S = np.zeros(2 * n - 1)
        w_S[top_k_idx] = 2.0 / K
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Backtracking line search
        alpha = lr / (gnorm + 1e-15)
        improved = False
        for _ in range(40):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc * (1 - 1e-12):
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            lr *= 0.9
            if lr < 1e-10:
                break
    
    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, rng, strength=0.03):

    pert = strength * np.sin(2 * np.pi * freq * t + phase)
    z = z + pert
    z = z - z.mean()
    return _softmax(z)
