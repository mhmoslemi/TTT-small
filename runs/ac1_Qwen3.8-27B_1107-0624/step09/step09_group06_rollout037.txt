Key innovations over previous approach:

- Anti-symmetric perturbation initialization (theoretically motivated)

import numpy as np
import time
from scipy.optimize import minimize


def _fft_convolve(a, b):
    """FFT-based linear convolution."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    if out_n == 0:
        return np.array([], dtype=np.float64)
    N = 1 << (out_n - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    """Compute the objective: 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(x):
    """Project x onto probability simplex {a >= 0, sum(a) = 1}."""
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _grad_single(a, k):
    """Exact gradient of conv(a,a)[k] w.r.t. a.
    Uses identity: d conv(a,a)[k] / d a[m] = 2*a[k-m] (if 0<=k-m<n)."""
    n = len(a)
    grad = np.zeros(n)
    m_lo = max(0, k - n + 1)
    m_hi = min(n - 1, k)
    if m_lo <= m_hi:
        j_lo = k - m_hi
        j_hi = k - m_lo
        grad[m_lo:m_hi + 1] = 2.0 * a[j_lo:j_hi + 1][::-1]
    return grad


def _grad_active_set(a, active_k):
    """Gradient of sum of conv(a,a)[k] for k in active_k."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_k:
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            j_lo = k - m_hi
            j_hi = k - m_lo
            grad[m_lo:m_hi + 1] += 2.0 * a[j_lo:j_hi + 1][::-1]
    return grad


def _grad_smooth_max(a, c, beta):
    """Exact gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a, using FFT.
    grad[m] = 2 * (a_rev * w)[m + n - 1] where w = softmax(beta*c)."""
    n = len(a)
    bc = beta * c
    bc_shifted = bc - bc.max()
    e = np.exp(np.clip(bc_shifted, -700, 0.0))
    s = e.sum()
    if s < 1e-300:
        k_max = int(np.argmax(c))
        return _grad_single(a, k_max)
    w = e / s
    # grad[m] = 2 * sum_j a[j] * w[m+j] = 2 * (a_rev * w)[m + n - 1]
    a_rev = a[::-1]
    conv_len = n + len(w) - 1
    N = 1 << (conv_len - 1).bit_length()
    A_rev = np.fft.rfft(a_rev, N)
    W = np.fft.rfft(w, N)
    conv = np.fft.irfft(A_rev * W, N)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad


def _smooth_max_value(c, beta):
    """Value of (1/beta)*log(sum exp(beta*c))."""
    bc = beta * c
    bc_shifted = bc - bc.max()
    e = np.exp(np.clip(bc_shifted, -700, 0.0))
    s = e.sum()
    if s < 1e-300:
        return float(c.max())
    return float(c.max() + np.log(s) / beta)


def _spectral_equalize(a, n_iter=100, width=0.3):
    """Iteratively equalize the power spectrum to be more peaked (DC-concentrated).
    This makes the autocorrelation flatter (lower peak)."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    N = 1 << (2 * n - 1).bit_length()
    
    for _ in range(n_iter):
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        P += 1e-30
        # We want P to be more peaked at DC
        # Compute a "target" that's a raised cosine centered at DC
        freqs = np.arange(len(P))
        # Half-bandwidth of the target peak
        hw = max(1, int(width * len(P)))
        target = np.ones(len(P))
        # Boost DC relative to the rest
        P_norm = P / P.sum()
        # Apply a spectral "boost" to low frequencies
        boost = 1.0 + 2.0 * np.exp(-freqs.astype(float) / hw)
        # New spectrum: normalize P * boost
        P_new = P * boost
        phase = A / (np.abs(A) + 1e-30)
        A_new = phase * np.sqrt(P_new)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new
    return a


def _nesterov_subgrad(a_init, deadline, min_remaining=30.0,
                       K_init=10, K_max=100, seed=0, base_lr=0.01):
    """Nesterov-accelerated subgradient with adaptive active set."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    c = _fft_convolve(a, a)
    best_sc = 2.0 * n * float(c.max())
    
    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    momentum = 0.9
    lr = base_lr
    K = K_init
    t = 0
    no_improve = 0
    stall_count = 0
    stall_threshold = 500
    
    while time.monotonic() + min_remaining < deadline:
        t += 1
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        cur_sc = 2.0 * n * c_max
        
        if cur_sc < best_sc - 1e-15:
            best_sc = cur_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive K
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
        if gap < c_max * 0.01 and K < K_max:
            K = min(K_max, int(K * 1.3) + 1)
        elif gap > c_max * 0.05 and K > K_init:
            K = max(K_init, int(K * 0.8))
        
        # Select active peaks
        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]
        
        # Compute subgradient (exact)
        grad = _grad_active_set(a, active)
        # Project to tangent space of simplex
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        
        if gnorm < 1e-18:
            break
        
        # Adaptive learning rate
        effective_lr = lr / (np.sqrt(t) + 1.0)
        
        # Nesterov step
        v = momentum * v - effective_lr * grad_t
        a_cand = a + v
        a_new = _project_simplex(a_cand)
        
        # Check improvement
        c_new = _fft_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())
        
        if sc_new < cur_sc * (1 + 1e-6):
            a = a_new
        else:
            v *= 0.5
            a = a_new  # Stay on simplex
        
        # Periodic perturbation to escape local minima
        if no_improve > stall_threshold:
            stall_count += 1
            if stall_count > 8:
                break
            # Anti-symmetric perturbation
            half = n // 2
            perturb = rng.standard_normal(half)
            full_pert = np.zeros(n)
            full_pert[:half] = perturb
            full_pert[half:] = -perturb[::-1]
            # Ensure sum is preserved
            full_pert -= full_pert.mean()
            scale = 0.01 * a.max()
            a_pert = _project_simplex(a + scale * full_pert)
            c_pert = _fft_convolve(a_pert, a_pert)
            sc_pert = 2.0 * n * float(c_pert.max())
            if sc_pert < best_sc:
                a = a_pert
                best_a = a.copy()
                best_sc = sc_pert
            v = np.zeros(n)
            no_improve = 0
            stall_threshold = max(100, stall_threshold * 0.7)
        
        if t % 1000 == 0:
            elapsed = time.monotonic()
            remaining = deadline - min_remaining - elapsed
            if remaining > 0:
                print(f"    [Nesterov] t={t}: best={best_sc:.10f}, K={K}, "
                      f"lr={lr:.6f}, remaining={remaining:.0f}s")
    
    return best_a, best_sc


class _SmoothMaxProb:
    """Objective and gradient for L-BFGS on smooth max."""
    
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    
    def __call__(self, z):
        a = _softmax(z)
        c = _fft_convolve(a, a)
        val = _smooth_max_value(c, self.beta)
        return 2.0 * self.n * val
    
    def gradient(self, z):
        a = _softmax(z)
        c = _fft_convolve(a, a)
        grad_a = _grad_smooth_max(a, c, self.beta)
        # Chain rule: d/dz = a * (I - 1/n * 11^T) @ grad_a
        grad_z = a * (grad_a - np.mean(grad_a))
        return 2.0 * self.n * grad_z


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _run_lbfgs(a_init, deadline, min_remaining=30.0):
    """L-BFGS on smooth max with adaptive beta schedule."""
    n = len(a_init)
    current_a = a_init.copy()
    s = current_a.sum()
    if s > 1e-15:
        current_a = current_a / s
    current_sc = _score(current_a)
    
    c_cur = _fft_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a
    
    # Determine base beta from gap
    c_sorted = np.sort(c_cur)[::-1]
    top_K = min(20, len(c_sorted))
    gap = c_sorted[0] - c_sorted[top_K - 1]
    base_beta = max(gap * 0.5, max_c * 0.05)
    
    # Beta schedule: start large (smooth), end small (sharp)
    n_betas = 12
    betas = base_beta * (0.5 ** np.arange(n_betas))
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        
        # Estimate time per iteration
        est_time = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(3000, max(20, int(remaining / (est_time * 5))))
        
        prob = _SmoothMaxProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16,
                         'gtol': 1e-14, 'maxcor': 50, 'maxls': 40}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(current_a, 1e-30))
                z = z - z.mean()
                if i % 3 == 0:
                    print(f"    [L-BFGS] beta_idx={i}: score={current_sc:.10f}")
        except Exception:
            pass
    
    return current_a


def _anti_sym_perturb(a, rng, scale=0.05):
    """Add anti-symmetric perturbation to break symmetry."""
    n = len(a)
    half = n // 2
    perturb = rng.standard_normal(half)
    full = np.zeros(n)
    full[:half] = perturb
    full[half:] = -perturb[::-1]
    if n % 2 == 1:
        full[half] = 0.0
    full -= full.mean()
    a_new = a + scale * full * (a.max() + 1e-10)
    return _project_simplex(a_new)


def _make_init(n, kind="uniform", seed=0):
    """Create various initializations."""
    rng = np.random.default_rng(seed)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "anti_sym":
        a = np.ones(n) / n
        half = n // 2
        pert = rng.standard_normal(half) * 0.3
        full = np.zeros(n)
        full[:half] = pert
        full[half:] = -pert[::-1]
        if n % 2 == 1:
            full[half] = 0.0
        a = np.maximum(a + full, 0.0)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered_anti":
        t = np.linspace(0, 1, n)
        base = np.sin(np.pi * t) ** 0.5
        # Add anti-symmetric component
        asym = np.sin(2 * np.pi * 3 * t) * 0.2
        a = np.maximum(base + asym, 0.01)
        return a / a.sum()
    elif kind == "random_clipped":
        a = np.abs(rng.standard_normal(n))
        return a / a.sum()
    elif kind == "bimodal_anti":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2) + 0.1
        # Add anti-symmetric perturbation
        rng2 = np.random.default_rng(seed + 1)
        half = n // 2
        pert = rng2.standard_normal(half) * 0.1
        full = np.zeros(n)
        full[:half] = pert
        full[half:] = -pert[::-1]
        a = np.maximum(a + full, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _spectral_optimize(a, deadline, min_remaining=20.0, n_iter_max=500):
    """Iterative spectral equalization to reduce autocorrelation peak.
    Alternates between:
    1. Time domain: project to non-negative simplex
    2. Frequency domain: boost DC power relative to high frequencies
    """
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    N = 1 << (2 * n - 1).bit_length()
    
    # Heat parameter: controls how aggressive the spectral boost is
    heat = 0.1
    min_heat = 0.01
    
    for it in range(n_iter_max):
        if time.monotonic() + min_remaining > deadline:
            break
        
        # Compute current power spectrum
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2 + 1e-30
        
        # Create spectral boost: emphasize low frequencies
        freqs = np.arange(len(P))
        # Gaussian boost centered at DC
        sigma = max(1.0, heat * len(P))
        boost = np.exp(-freqs.astype(float) ** 2 / (2.0 * sigma ** 2))
        
        # New target spectrum
        P_target = P * (1.0 + boost)
        
        # Reconstruct with target amplitude, keep phase
        phase = A / (np.abs(A) + 1e-30)
        A_new = phase * np.sqrt(P_target)
        
        # Back to time domain
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Anneal: reduce heat over time
        heat = max(min_heat, heat * 0.98)
        
        if it % 100 == 0:
            print(f"    [Spectral] it={it}: score={best_sc:.10f}, heat={heat:.4f}")
    
    return best_a, best_sc
