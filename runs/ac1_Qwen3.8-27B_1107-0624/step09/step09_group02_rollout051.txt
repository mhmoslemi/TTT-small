"""

Key improvements over previous approach:
1. FISTA (accelerated projected gradient) on the QP subproblem with FFT-based Hessian-vector products.
2. Progressive active set growth (Fibonacci-like: 1,2,3,5,8,13,...) for natural curriculum learning.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _hvp_qp(v, active_lags, n):
    """Compute Q*v where Q = sum_{k in active_lags} H_k.
    (Q*v)[i] = 2 * sum_{k in active_lags} v[k-i] (with bounds 0<=k-i<n)
    Uses FFT for O(n log n) computation.
    """
    if len(active_lags) == 0:
        return np.zeros(n)
    p = np.zeros(2 * n - 1)
    p[active_lags] = 1.0
    v_rev = v[::-1]
    conv_len = len(p) + len(v_rev) - 1
    N = _next_pow2(conv_len)
    Fp = np.fft.rfft(p, N)
    Fv = np.fft.rfft(v_rev, N)
    conv = np.fft.irfft(Fp * Fv, N)[:conv_len]
    # (Q*v)[i] = 2 * conv[i + n - 1]
    result = 2.0 * conv[n - 1:2 * n - 1]
    return result


def _hvp_qp_direct(v, active_lags, n):
    """Direct computation of Q*v for small active sets."""
    result = np.zeros(n)
    for k in active_lags:
        # (H_k * v)[i] = 2 * v[k-i] for 0 <= k-i < n
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        if i_min <= i_max:
            idx = np.arange(i_min, i_max + 1)
            result[idx] += 2.0 * v[k - idx]
    return result


def _grad_qp(a, active_lags, n):
    """Gradient of sum_{k in active_lags} c[k] w.r.t. a."""
    return _hvp_qp(a, active_lags, n)


def _fista_qp(a_init, active_lags, n, max_iter, L_override=None, tol=1e-12):
    """FISTA on minimize sum_{k in active} c[k] s.t. a in simplex.
    Returns the optimized a.
    """
    a = a_init.copy()
    a = _proj_simplex(a)
    y = a.copy()

    L = L_override if L_override else 2.0 * len(active_lags)

    L = max(L, 2.0)

    for it in range(max_iter):
        grad = _hvp_qp(y, active_lags, n)
        
        # Backtracking line search (every 10 iterations)
        if it % 10 == 0 and it > 0:
            # Estimate L from recent progress
            step_test = 1.0 / L
            a_test = _proj_simplex(y - step_test * grad)
            f_test = float(_hvp_qp(a_test, active_lags, n) @ a_test) / 2.0
            f_y = float(grad @ y) / 2.0
            # If f(a_test) > f(y) - <grad, a_test> + L/2 * ||a_test - y||^2, increase L
            # Simplified: just use the known bound
            pass
        
        step = 1.0 / L
        a_new = _proj_simplex(y - step * grad)
        
        # Objective check
        obj = float(_hvp_qp(a_new, active_lags, n) @ a_new) / 2.0
        if abs(prev_obj - obj) < tol * max(1.0, abs(obj)):
            break
        prev_obj = obj
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0)
        # Renormalize y to stay close to simplex (FISTA requires this for the projection)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a_new.copy()
        else:
            y = y / s_y * s_y  # keep as is, the projection handles it
        a = a_new
        t = t_new
    
    return a


def _cg_solve_kkt(n, active_lags, max_iter=500, tol=1e-10):
    """Solve the KKT system for the QP on full support:
    minimize a^T Q a / 2 s.t. sum(a) = 1, a >= 0
    KKT: Q a = lambda * 1 (on support where a > 0)
    Try to find a >= 0, sum(a) = 1, Q a = lambda * 1.
    """
    # Solve Q x = 1 using CG, then a = x / (sum(x))
    # Check non-negativity
    b = np.ones(n)
    x = np.ones(n) / n  # initial guess
    r = b - _hvp_qp(x, active_lags, n)
    p_vec = r.copy()
    rs_old = float(r @ r)
    if rs_old < tol:
        x = x / x.sum()
        if np.min(x) >= 0:
            return x
    
    for it in range(max_iter):
        Ap = _hvp_qp(p_vec, active_lags, n)
        pAp = float(p_vec @ Ap)
        if pAp < 1e-15:
            break
        alpha = rs_old / pAp
        x_new = x + alpha * p_vec
        r_new = r - alpha * Ap
        rs_new = float(r_new @ r_new)
        if rs_new < tol:
            x = x_new
            break
        beta = rs_new / rs_old
        p_vec = r_new + beta * p_vec
        r = r_new
        rs_old = rs_new
    
    s = x.sum()
    if s > 0:
        x = x / s
    return x


def _active_set_qp_direct(a_init, active_lags, n, max_iter=200):
    """Solve QP on the active set using iterative refinement.
    Combines CG for the KKT system with non-negativity enforcement.
    """
    a = a_init.copy()
    support = np.where(a > 1e-15)[0]
    if len(support) == 0:
        support = np.arange(n)
    
    for outer in range(max_iter):
        # Solve Q_S x = lambda * 1 on the support
        # Using CG with the full HVP (but restricted to support)
        x = np.zeros(n)
        x[support] = 1.0 / len(support)
        
        # CG on the subproblem
        r = _hvp_qp(x, active_lags, n)
        r[~support] = 0
        r[support] -= np.mean(r[support])  # enforce sum constraint via Lagrange multipler
        # Actually, let's just do: find a on support s.t. (Q a)_support = lambda * 1
        # Initialize
        x = np.zeros(n)
        x[support] = 1.0 / len(support)
        p_vec = np.zeros(n)
        # Compute residual: Q x - mean(Q x on support) * 1 (projected to support)
        qx = _hvp_qp(x, active_lags, n)
        lambda_est = np.mean(qx[support])
        res = np.zeros(n)
        res[support] = qx[support] - lambda_est
        p_vec[support] = res[support].copy()
        rs = float(res[support] @ res[support])
        
        for cg_it in range(100):
            if rs < 1e-20:
                break
            Ap = _hvp_qp(p_vec, active_lags, n)
            Ap[~support] = 0
            # Project to ensure the constraint
            Ap[support] -= np.mean(Ap[support])
            pAp = float(p_vec[support] @ Ap[support])
            if abs(pAp) < 1e-15:
                break
            alpha = rs / pAp
            x[support] += alpha * p_vec[support]
            new_qx = _hvp_qp(x, active_lags, n)
            new_lambda = np.mean(new_qx[support])
            new_res = np.zeros(n)
            new_res[support] = new_qx[support] - new_lambda
            rs_new = float(new_res[support] @ new_res[support])
            beta = rs_new / rs
            p_vec[support] = new_res[support] + beta * p_vec[support]
            r = new_res
            rs = rs_new
        
        # Enforce non-negativity
        x[~support] = 0
        if np.min(x[support]) < -1e-10:
            # Remove the most negative element from support
            worst = support[np.argmin(x[support])]
            x[worst] = 0
            support = np.delete(support, worst)

                a = np.zeros(n)
                a[support] = 1.0 / len(support)

        else:
            x = np.maximum(x, 0)
            s = x.sum()
            if s < 1e-15:
                a = np.ones(n) / n
                return a
            a = x / s
            return a
    
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def _run_fista_progressive(a_init, deadline, min_time_remaining=30.0):

def _run_lbfgs_quick(a_init, deadline, min_time_remaining=10.0):
    """Quick L-BFGS with log-sum-exp smoothing."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    a = a / s
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    best_a = a.copy()
    best_sc = _score(a)
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(5, len(c_sorted) - 1)]
    base_beta = max(gap * 0.05, max_c * 0.005)
    
    ratios = [1.0, 0.1, 0.01, 0.001, 0.0001]
    
    for ratio in ratios:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        beta = base_beta * ratio
        est_time = 0.005 * max(1.0, n / 16384.0)
        max_iter = min(2000, max(20, int(remaining / (est_time * 3))))
        
        try:
            def obj_func(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    return 2.0 * n * c_max_local
                return 2.0 * n * (c_max_local + beta * np.log(s_e))
            
            def grad_func(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    w = np.zeros_like(c_local)
                    w[int(np.argmax(c_local))] = 1.0
                else:
                    w = e / s_e
                a_rev = a_local[::-1]
                wc = _full_convolve(w, a_rev)
                h = wc[n - 1:2 * n - 1]
                W_val = float(np.dot(w, c_local))
                return 4.0 * n * a_local * (h - W_val)
            
            result = minimize(
                obj_func, z, jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxcor': 30}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return best_a

    best_sc = _score(a)
    
    # Use top-1 (exact subgradient) with FISTA-like acceleration
    y = a.copy()
    t = 1.0
    L = 2.0  # Lipschitz constant for single lag
    
    c = _full_convolve(y, y)
    k_max = int(np.argmax(c))
    
    for it in range(10000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        # Update active set dynamically
        if it % 50 == 0:
            c = _full_convolve(y, y)
            c_max = float(c.max())
            # Use lags within 1% of max
            tol = 0.01 * c_max
            active = np.where(c >= c_max - tol)[0]
            if len(active) == 0:
                active = np.array([k_max])
            L = 2.0 * len(active)
        
        grad = _hvp_qp(y, active, n)
        step = 1.0 / L
        a_new = _proj_simplex(y - step * grad)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - y)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y > 0:
            y = y / s_y
        a = a_new
        t = t_new
        
        if it % 100 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * t
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.5) ** 2) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _run_fista_multi_k(a_init, deadline, min_time_remaining=15.0):
    """Run FISTA with multiple fixed K values, using the best result at each."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    K_values = [3, 5, 8, 13, 21, 34, 55, 89]
    
    for K in K_values:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        c = _full_convolve(a, a)
        top_K = np.argsort(c)[::-1][:K]
        active_lags = list(top_K)
        
        est_time_per_iter = 0.003 * max(1.0, n / 16384.0)
        max_iter = min(3000, max(50, int(remaining / (est_time_per_iter * 4))))
        
        a_new = _fista_qp(a, active_lags, n, max_iter)
        sc_new = _score(a_new)
        
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_sc

            return

            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")
    
    # Load parent

            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
