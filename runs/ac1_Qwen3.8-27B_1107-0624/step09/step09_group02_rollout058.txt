"""
Optimized flat auto-convolution sequence generator.
Uses continuation L-BFGS with FFT-based gradients, active-set refinement,
and multi-profile initialization to minimize 2n*max(convolve(a,a))/(sum(a))^2.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _autoconv(a):
    """Compute convolve(a, a) using FFT."""
    return _full_convolve(a, a)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _autoconv(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_fft(a, w):
    """Compute gradient of sum_k w[k]*b[k] w.r.t. a, where b=convolve(a,a).
    
    grad[m] = 2 * sum_k w[k] * a[k-m] (a[j]=0 outside [0,n-1])
    = 2 * (w cross-corr a)[m]
    
    Using FFT: h = ifft(conj(FFT(a,N)) * FFT(w,N))[:n]
    """
    n = len(a)
    N = _next_pow2(3 * n - 1)
    Fa = np.fft.rfft(a, N)
    Fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(Fa) * Fw, N)[:n]
    return 2.0 * h


def _smoothed_obj_params(a, beta, n):
    """Compute the smoothed objective value and softmax weights."""
    c = _autoconv(a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s_e = e.sum()
    if s_e < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        w = e / s_e
        obj = 2.0 * n * (c_max + beta * np.log(s_e))
    return obj, w, c_max


def _obj_and_grad(z, beta, n):
    """Compute objective and gradient for L-BFGS in log-space (softmax parameterization)."""
    a = _softmax(z)
    obj, w, c_max = _smoothed_obj_params(a, beta, n)
    
    # Gradient w.r.t. a
    grad_a = _grad_fft(a, w)
    
    # Gradient w.r.t. z: grad_z[j] = a[j] * (grad_a[j] - dot(a, grad_a))
    dot_a = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - dot_a)
    
    return obj, grad_z


def _run_continuation_lbfgs(a_init, deadline, min_remaining=60.0):
    """Run L-BFGS with continuation on beta."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    
    c_cur = _autoconv(a)
    c_max = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    # Gap between top values
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted)-1)]
    
    # Beta schedule: start wide, narrow down
    base_beta = max(gap * 0.5, c_max * 0.01)
    ratios = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
              0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001,
              0.000005, 0.000001, 0.0000005, 0.0000001]
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    for i, ratio in enumerate(ratios):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break
        
        beta = base_beta * ratio
        est_time_per_iter = 0.005 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 8))))
        
        try:
            def obj_func(z_local):
                a_local = _softmax(z_local)
                o, _, _ = _smoothed_obj_params(a_local, beta, n)
                return o
            
            def grad_func(z_local):
                _, g = _obj_and_grad(z_local, beta, n)
                return g
            
            result = minimize(
                obj_func, z,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)

                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))

            z = result.x - result.x.mean()
        except Exception as e:

def _run_active_set(a_init, deadline, min_remaining=30.0, max_rounds=200):
    """Active-set method: focus on the top-k autocorrelation values."""
    n = len(a_init)

    a = a / a.sum()
    best_a = a.copy()

                    0.00005, 0.00002, 0.00001, 0.000005, 0.000002, 0.000001,
                    0.0000005, 0.0000001]
    tol_idx = 0
    
    for rnd in range(max_rounds):
        if time.monotonic() + min_remaining > deadline:
            break
        
        c = _autoconv(a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule)-1)]
        
        # Active set: lags within tol of the max
        active_mask = c >= (c_max * (1.0 - tol))
        if active_mask.sum() == 0:
            active_mask[int(np.argmax(c))] = True
        
        p = np.zeros(len(c))
        p[active_mask] = 1.0 / active_mask.sum()
        
        # Gradient descent on the smoothed objective (smoothed over active set)
        for inner in range(30):
            if time.monotonic() + min_remaining > deadline:
                break
            grad_a = _grad_fft(a, p)
            gnorm = np.linalg.norm(grad_a)
            if gnorm < 1e-18:
                break
            # Adaptive step size
            step = 0.5 / (gnorm * n * 0.01 + 1e-10)
            # Projected gradient step
            a_try = a - step * grad_a
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 0:
                a_try = a_try / s_try
            # Check if this improves
            c_try = _autoconv(a_try)
            if float(c_try.max()) < c_max * (1 + 1e-10):
                a = a_try
            else:
                # Halve the step
                step *= 0.5
                if step < 1e-15:
                    break
        
        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
        elif rnd > 10:
            a = best_a.copy()

def _run_smoothed_descent(a_init, deadline, min_remaining=30.0):

        if time.monotonic() + min_remaining > deadline:
            break
        
        decay = 0.97 ** min(step, 100)
        beta = base_beta * decay

        c_max = float(c.max())
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s_e = e.sum()
        if s_e < 1e-300:
            w = np.zeros_like(c)

        else:
            w = e / s_e
        
        grad_a = _grad_fft(a, w)
        gnorm = np.linalg.norm(grad_a)
        if gnorm < 1e-15:
            break
        
        # Adam-like update in log-space
        z = np.log(np.maximum(a, 1e-30))
        z = z - z.mean()
        # Simple gradient step in z-space
        dot_a = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot_a)
        znorm = np.linalg.norm(grad_z)
        if znorm < 1e-15:

        lr = 0.1 / (znorm + 1e-10)
        z_new = z - lr * grad_z

            best_a = a_new.copy()
            a = a_new.copy()
            patience = 0
        else:

def _make_profile(n, kind="sqrt"):
    t = np.arange(n, dtype=np.float64)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "tapered":
        a = np.sin(np.pi * (t + 0.5) / n) ** 0.5
    elif kind == "sqrt":
        # 1/sqrt(k+1) profile - theoretical approximation
        a = 1.0 / np.sqrt(t + 1.0)
    elif kind == "sqrt_sym":
        # Symmetric version: 1/sqrt(min(k+1, n-k))
        a = 1.0 / np.sqrt(np.minimum(t + 1.0, n - t).astype(np.float64))
    elif kind == "cos":
        a = 0.5 + 0.5 * np.cos(np.pi * t / (n - 1))
    elif kind == "ramp":
        a = 0.3 + 0.7 * t / max(n-1, 1)
    elif kind == "bump":
        a = np.exp(-4.0 * ((t / max(n-1, 1)) - 0.3) ** 2) + 0.05
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s
