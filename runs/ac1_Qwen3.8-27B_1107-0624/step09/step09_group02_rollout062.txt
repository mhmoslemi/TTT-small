FFT-accelerated peak autocorrelation minimizer using L-BFGS continuation,
active-set QP with CG inner solver, and multi-resolution refinement.
Key innovations: Hankel-structured FFT matrix products, adaptive active set,
and aggressive beta continuation for tight convergence.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _npo2(n):
    return 1 << max(0, n - 1).bit_length()


def _conv(a, b):
    n, m = len(a), len(b)
    N = _npo2(n + m - 1)
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:n + m - 1]


def _autocorr(a):
    return _conv(a, a)


def _grad_autocorr(a, p):
    n = len(a)
    a_rev = a[::-1]
    conv = _conv(p, a_rev)
    return 2.0 * conv[n - 1:2 * n - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _autocorr(a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _softmax_lse(z, beta=1.0):
    z = z / beta
    z = z - z.max()
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rh = rho[-1]
    theta = (cssv[rh] - 1) / (rh + 1)
    return np.maximum(v - theta, 0)


def _hankel_mv(h, v):
    n = len(v)
    w = v[::-1]
    conv = _conv(h, w)
    return conv[n - 1:2 * n - 1]


def _cg_hankel(h, b, max_iter=200, tol=1e-12, mask=None):
    """Solve Hx = b where H is Hankel with H[i,j]=h[i+j], using CG.
    If mask is given, solve on the subset: (H_MM x)_m = b_m for m in mask,
    with x_{not in mask} = 0.
    """
    n = len(b)
    if mask is not None:
        x = np.zeros(n)
        x[mask] = b
        b_full = np.zeros(n)
        b_full[mask] = b
        b_use = b_full
    else:
        x = np.zeros(n)
        b_use = b.copy()

    r = b_use - _apply_hankel_masked(h, x, mask)
    p_vec = r.copy()
    rs_old = float(r @ r)
    if rs_old < tol * tol:
        return x

    for _ in range(max_iter):
        Hp = _apply_hankel_masked(h, p_vec, mask)
        alpha = rs_old / float(p_vec @ Hp)
        if not np.isfinite(alpha) or alpha <= 0:
            break
        x += alpha * p_vec
        if mask is not None:
            x[~mask] = 0
        r -= alpha * Hp
        rs_new = float(r @ r)
        if rs_new < tol * tol:
            break
        p_vec = r + (rs_new / rs_old) * p_vec
        if mask is not None:
            p_vec[~mask] = 0
        rs_old = rs_new

    return x


def _apply_hankel_masked(h, v, mask):
    result = _hankel_mv(h, v)
    if mask is not None:
        result[~mask] = 0
    return result


def _solve_qp_hankel(h, mask=None):
    """Solve min (1/2)x^T H x s.t. sum(x)=1, x>=0 (on support).
    H is Hankel with row/col structure h[i+j].
    Returns the optimal x (length n) or None.
    """
    n = len(h) // 2 + 1  # n = (len(h)-1)//2
    n = (len(h) - 1) // 2

    if mask is None:
        mask = np.ones(n, dtype=bool)

    m_count = int(mask.sum())
    if m_count <= 0:
        return None
    if m_count == 1:
        x = np.zeros(n)
        x[mask] = 1.0
        return x

    # Solve H_SS x_S = 1 via CG, then normalize
    # h_full has length 2n-1
    # For the subproblem on mask, we do CG with masked matrix-vector product
    b = np.ones(n)
    b[~mask] = 0

    x = _cg_hankel(h, b, max_iter=300, tol=1e-14, mask=mask)

    sx = float(x.sum())
    if sx < 1e-15 or not np.isfinite(sx):
        return None
    x = x / sx

        # Active set violation: fix the most negative to 0 and retry
        worst = int(np.argmin(x[mask]))
        mask[worst] = False
        return _solve_qp_hankel(h, mask=mask)

    x = np.maximum(x, 0)
    x = x / x.sum()

        # Projected gradient fallback if QP solve fails or doesn't improve
        if rnd % 5 == 0:
            grad = _grad_autocorr(a, p_w)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-15:
                lr = 0.1 / (gnorm + 1e-10)
                for ls in range(20):
                    a_try = _proj_simplex(a - lr * grad)
                    sc_try = _score(a_try)
                    if sc_try < best_sc:
                        best_sc = sc_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        break
                    lr *= 0.5
                    if lr < 1e-15:
                        break

        a = best_a.copy()

        if (rnd + 1) % 8 == 0:
            tol_idx += 1

    return best_a, best_sc


def _run_proj_grad_bt(a_init, deadline, min_remaining=15.0, max_iter=5000):
    """Projected gradient with backtracking line search and momentum."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    momentum = np.zeros(n)
    beta_m = 0.85
    patience = 0

    for step in range(max_iter):
        if time.monotonic() > deadline - min_remaining:
            break
        a_curr = _softmax_lse(z)
        c = _autocorr(a_curr)
        c_max = float(c.max())

        # Top-k active set
        k_act = min(20, 2 * n - 1)
        top_k = np.argsort(c)[::-1][:k_act]
        beta_w = max(c_max * 0.01, 1e-18)
        p = np.zeros(2 * n - 1)
        c_top = c[top_k]
        w_top = np.exp((c_top - c_top.max()) / beta_w)
        p[top_k] = w_top / w_top.sum()

        grad = _grad_autocorr(a_curr, p)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        momentum = beta_m * momentum + (1 - beta_m) * grad
        lr = 0.05 / (np.linalg.norm(momentum) / (gnorm + 1e-10) + 1e-10)

        z_new = z - lr * momentum
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()
        z = z_new

        if step % 5 == 0:
            a_check = _softmax_lse(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 80:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros(n)
                    beta_m = max(0.5, beta_m * 0.9)
                    patience = 0

    a_final = _softmax_lse(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.2 + 0.8 * t
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_lse(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS continuation, active-set QP with CG/FFT,
    projected gradient with backtracking, multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s", flush=True)

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}", flush=True)
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}", flush=True)
        update_best(parent, _score(parent), "init")

    # === Phase 1: L-BFGS continuation (250s) ===
    print(f"\n[Phase 1] L-BFGS continuation (t={time.monotonic() - t_start:.1f}s)", flush=True)
    p1_deadline = t_start + 250
    a1 = _run_lbfgs_smoothed(parent, p1_deadline, min_remaining=20.0)
    update_best(a1, _score(a1), "lbfgs_main")

    # Try different profiles at same n
    for prof in ["cos", "gauss", "ramp"]:
        if time.monotonic() > p1_deadline - 20:
            break
        a_init = _make_profile(n_parent, prof)
        a_p = _run_lbfgs_smoothed(a_init, p1_deadline, min_remaining=10.0)
        update_best(a_p, _score(a_p), f"lbfgs_{prof}")

    # Perturbation restart
    if time.monotonic() < p1_deadline - 15:
        a_pert = _perturb(best_seq, 0.02, rng)
        a_p2 = _run_lbfgs_smoothed(a_pert, p1_deadline, min_remaining=5.0)
        update_best(a_p2, _score(a_p2), "lbfgs_perturb")

    # === Phase 2: Active-set QP (350s) ===
    print(f"\n[Phase 2] Active-set QP (t={time.monotonic() - t_start:.1f}s)", flush=True)
    p2_deadline = min(deadline - 600, t_start + 600)
    a_qp, sc_qp = _run_active_set_qp(best_seq, p2_deadline, min_remaining=20.0, max_rounds=200)
    update_best(a_qp, sc_qp, "qp_1")

    # === Phase 3: Projected gradient with backtracking (250s) ===
    print(f"\n[Phase 3] ProjGrad BT (t={time.monotonic() - t_start:.1f}s)", flush=True)
    p3_deadline = min(deadline - 400, p2_deadline + 250)
    a_pg, sc_pg = _run_proj_grad_bt(best_seq, p3_deadline, min_remaining=15.0, max_iter=3000)
    update_best(a_pg, sc_pg, "pg_1")

    # === Phase 4: Second L-BFGS round (200s) ===
    print(f"\n[Phase 4] L-BFGS round 2 (t={time.monotonic() - t_start:.1f}s)", flush=True)
    p4_deadline = min(deadline - 250, p3_deadline + 200)
    a_l2 = _run_lbfgs_smoothed(best_seq, p4_deadline, min_remaining=15.0)
    update_best(a_l2, _score(a_l2), "lbfgs_r2")

    # === Phase 5: Second QP round (200s) ===
    print(f"\n[Phase 5] QP round 2 (t={time.monotonic() - t_start:.1f}s)", flush=True)
    p5_deadline = min(deadline - 150, p4_deadline + 200)
    a_qp2, sc_qp2 = _run_active_set_qp(best_seq, p5_deadline, min_remaining=15.0, max_rounds=150)
    update_best(a_qp2, sc_qp2, "qp_2")

    # === Phase 6: Multi-n exploration (150s) ===
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)", flush=True)
    p6_deadline = min(deadline - 80, p5_deadline + 150)
    n_tries = [n_parent * 2, n_parent // 2, n_parent * 3 // 2]
    seen_ns = set([n_parent])
    for n_try in n_tries:
        if time.monotonic() > p6_deadline - 15:
            break
        if n_try in seen_ns or n_try < 512:
            continue
        seen_ns.add(n_try)
        a_init_n = _resample(best_seq, n_try)
        a_n = _run_lbfgs_smoothed(a_init_n, p6_deadline, min_remaining=5.0)
        update_best(a_n, _score(a_n), f"n_{n_try}")

    # === Phase 7: Final combined polish (80s) ===
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)", flush=True)
    p7_deadline = deadline - 5

    if time.monotonic() < p7_deadline - 20:
        a_fin, sc_fin = _run_active_set_qp(best_seq, p7_deadline - 10, min_remaining=5.0, max_rounds=100)
        update_best(a_fin, sc_fin, "final_qp")

    if time.monotonic() < p7_deadline - 5:
        a_pg2, sc_pg2 = _run_proj_grad_bt(best_seq, p7_deadline, min_remaining=2.0, max_iter=2000)
        update_best(a_pg2, sc_pg2, "final_pg")

    # Final L-BFGS with very tight beta
    if time.monotonic() < p7_deadline - 3:
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        c_b = _autocorr(best_seq)
        cmax_b = float(c_b.max())
        beta_fin = cmax_b * 2e-6
        remaining = max(1.0, p7_deadline - 3 - time.monotonic())
        max_iter = int(min(500, max(20, remaining / 0.02)))

        def final_obj(z_l):
            a_l = _softmax_lse(z_l)
            c_l = _autocorr(a_l)
            shifted = c_l / beta_fin
            shifted = shifted - shifted.max()
            se = np.exp(shifted).sum()
            if se < 1e-300:
                return 2.0 * n * float(c_l.max())
            return 2.0 * n * beta_fin * (shifted.max() + np.log(se))

        def final_grad(z_l):
            a_l = _softmax_lse(z_l)
            c_l = _autocorr(a_l)
            shifted = c_l / beta_fin
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            se = e.sum()
            if se < 1e-300:
                w = np.zeros(2 * n - 1)
                w[int(np.argmax(c_l))] = 1.0
            else:
                w = e / se
            g = _grad_autocorr(a_l, w)
            return 2.0 * n * a_l * (g - float(np.dot(a_l, g)))

        try:
            result = minimize(
                final_obj, z,
                jac=final_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 40, 'maxls': 30}
            )
            a_flb = _softmax_lse(result.x)
            update_best(a_flb, _score(a_flb), "final_lbfgs")
        except Exception:
            pass

    # Final validation
    print(f"\n{'=' * 60}", flush=True)
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s", flush=True)
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
