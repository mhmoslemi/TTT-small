"""

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(np.arange(1, n + 1) * u > cssv - 1)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _grad_active(a, p, n):
    """Gradient of sum_k p[k]*conv(a,a)[k] w.r.t. a using FFT cross-correlation."""
    N = _next_pow2(2 * n + len(p))
    Fx = np.fft.rfft(a, N)
    Fy = np.fft.rfft(p, N)
    r = np.fft.irfft(np.conj(Fx) * Fy, N)
    return 2.0 * r[:n]


def _make_profile(n, kind="uniform"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        a = 0.3 + 0.7 * t
        return a / a.sum()
    elif kind == "ramp2":
        a = 0.1 + 0.9 * t
        return a / a.sum()
    elif kind == "bump":
        a = np.exp(-8.0 * (t - 0.3) ** 2) + 0.1
        return a / a.sum()
    elif kind == "bump2":
        a = np.exp(-8.0 * (t - 0.7) ** 2) + 0.1
        return a / a.sum()
    elif kind == "two_bump":
        a = np.exp(-12.0 * (t - 0.25) ** 2) + np.exp(-12.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    elif kind == "asym":
        a = 0.2 + 0.8 * t ** 2
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS on smoothed max-convobjective in log-space."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    best_a = a.copy()
    best_sc = _score(a)

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)]
    base_beta = max(gap * 0.1, max_c * 0.001)

    ratios = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
              0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001,
              0.000005, 0.000002, 0.000001]

    for ratio in ratios:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        beta = max(base_beta * ratio, 1e-12)
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(30, int(remaining / (est_time * 5))))

        try:
            def obj_func(zl):
                al = _softmax(zl)
                cl = _full_convolve(al, al)
                cm = float(cl.max())
                shifted = (cl - cm) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                se = e.sum()
                if se < 1e-300:
                    return 2.0 * n * cm
                return 2.0 * n * (cm + beta * np.log(se))

            def grad_func(zl):
                al = _softmax(zl)
                cl = _full_convolve(al, al)
                cm = float(cl.max())
                shifted = (cl - cm) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                se = e.sum()
                if se < 1e-300:
                    w = np.zeros_like(cl)
                    w[int(np.argmax(cl))] = 1.0
                else:
                    w = e / se
                a_rev = al[::-1]
                wc = _full_convolve(w, a_rev)
                h = wc[n - 1:2 * n - 1]
                W_val = float(np.dot(w, cl))
                return 4.0 * n * al * (h - W_val)

            result = minimize(obj_func, z, jac=grad_func,
                              method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxcor': 50, 'maxls': 100})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return best_a


def _run_grad_active(a_init, deadline, min_time_remaining=30.0, K=10):
    """Multiplicative gradient ascent targeting top-K active lags."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    patience = 0
    max_pat = 80

    for it in range(10000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = _full_convolve(a, a)
        c_max = c.max()
        top_idx = np.argsort(c)[::-1][:K]
        p = np.zeros(2 * n - 1)
        p[top_idx] = 1.0 / K

        grad = _grad_active(a, p, n)
        g_max = np.max(np.abs(grad))
        if g_max < 1e-15:
            break
        step_rate = 0.03 * (1.0 - 0.005 * min(it, 50)) / g_max
        a_new = a * np.exp(-step_rate * grad)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_a = a_new.copy()
            patience = 0
            a = a_new
        else:
            patience += 1
            a = a_new
            if patience > max_pat:
                a = best_a.copy()
                patience = 0

    return best_a, best_sc


def _run_freq_shave(a_init, deadline, min_time_remaining=30.0):
    """Frequency-domain peak shaving: iteratively reduce peaks in conv via magnitude attenuation."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    N = _next_pow2(2 * n - 1)

    for it in range(2000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        Ac = A * A
        c = np.fft.irfft(Ac, N)[:2 * n - 1]
        c_max = float(c.max())
        k_peak = int(np.argmax(c))

        # Attenuate frequencies contributing constructively to the peak
        freqs = np.arange(len(A))
        theta = 2.0 * np.pi * freqs * k_peak / N
        T = np.cos(theta)
        alpha = 0.003 * (1.0 + 0.5 * np.cos(np.pi * min(it, 100) / 100.0))
        atten = 1.0 - alpha * np.maximum(T, 0)
        atten = np.maximum(atten, 0.01)
        A_new = A * np.sqrt(atten)

        a_pad_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_pad_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_sc


def _run_active_set_qp(a_init, deadline, min_time_remaining=30.0, max_rounds=150):
    """Active-set QP with progressive tightening."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    tol_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                    0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
                    0.000001, 0.0000005, 0.0000002, 0.0000001]
    tol_idx = 0

    for rnd in range(max_rounds):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        K = int(active.sum())
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / K

        # Projected gradient with backtracking for the QP
        step = 0.01
        for inner in range(80):
            if time.monotonic() + min_time_remaining > deadline:
                break
            grad = _grad_active(a, p, n)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 0.05 / (gnorm + 1e-10)
            a_try = _proj_simplex(a - step * grad)
            # Check KKT: no inactive lag should exceed max active
            c_try = _full_convolve(a_try, a_try)
            if c_try.max() <= c_max * (1.0 + 1e-6):
                a = a_try
            else:
                step *= 0.5
                a = a_try
                if step < 1e-15:
                    break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        if (rnd + 1) % 5 == 0:
            tol_idx += 1

    return best_a, best_sc


def _run_proj_grad_momentum(a_init, deadline, min_time_remaining=30.0):
    """Projected gradient with Nesterov momentum on top-K active set."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    momentum = np.zeros(n)
    beta_m = 0.9
    patience = 0
    max_pat = 100

    for step in range(8000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        a_curr = _softmax(z)
        c = _full_convolve(a_curr, a_curr)
        c_max = float(c.max())
        k_active = min(8, len(c))
        top_k = np.argsort(c)[::-1][:k_active]
        p = np.zeros(2 * n - 1)
        p[top_k] = 1.0 / k_active

        grad = _grad_active(a_curr, p, n)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        momentum = beta_m * momentum + (1 - beta_m) * grad
        lr = 0.015 * (1.0 + 0.05 * step) / (np.linalg.norm(momentum) / (grad_norm + 1e-10) + 1e-10)
        z_new = z - lr * momentum
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new = z_new - z_new.mean()
        z = z_new

        if step % 15 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_pat:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros(n)
                    patience = 0
                    beta_m *= 0.9

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _run_subspace(a_init, deadline, min_time_remaining=30.0, top_m=200):
    """Subspace optimization: only update top-m most important coordinates."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    # Identify important coordinates (where a is large)
    c = _full_convolve(a, a)
    # Coordinates that contribute most to the peaks
    peak_lags = np.argsort(c)[::-1][:10]
    contrib = np.zeros(n)
    for k in peak_lags:
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                contrib[m] += a[j]
    # Top m coordinates by contribution
    if top_m < n:
        top_idx = np.argsort(contrib)[::-1][:top_m]
    else:
        top_idx = np.arange(n)

    # Optimize only these coordinates using small L-BFGS
    a_work = a.copy()
    z_sub = np.log(np.maximum(a_work[top_idx], 1e-30))
    z_sub = z_sub - z_sub.mean()

    c_cur = _full_convolve(a_work, a_work)
    c_max_cur = float(c_cur.max())
    beta = c_max_cur * 0.005

    for inner_iter in range(200):
        if time.monotonic() + min_time_remaining > deadline:
            break
        a_work[top_idx] = _softmax(z_sub) * a_work[top_idx].sum()
        # Renormalize
        a_work = np.maximum(a_work, 0)
        s = a_work.sum()
        if s < 1e-15:
            break
        a_work = a_work / s
        c_w = _full_convolve(a_work, a_work)
        c_max_w = float(c_w.max())
        if c_max_w < c_max_cur - 1e-12:
            c_max_cur = c_max_w
            sc_w = _score(a_work)
            if sc_w < best_sc:
                best_sc = sc_w
                best_a = a_work.copy()

        # Simple gradient step in the subspace
        top_idx_set = set(top_idx.tolist())
        p_tmp = np.zeros(2 * n - 1)
        active = c_w >= c_max_w * 0.99
        p_tmp[active] = 1.0 / max(active.sum(), 1)
        grad_full = _grad_active(a_work, p_tmp, n)
        grad_sub = grad_full[top_idx]
        gn = np.linalg.norm(grad_sub)
        if gn < 1e-14:
            break
        lr = 0.02 / (gn + 1e-10)
        z_sub = z_sub - lr * grad_sub
        z_sub = np.clip(z_sub, -30, 30)
        z_sub = z_sub - z_sub.mean()

    return best_a, best_sc
