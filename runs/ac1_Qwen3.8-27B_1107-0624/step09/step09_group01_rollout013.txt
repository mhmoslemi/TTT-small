Let me also think about the "active set" approach more carefully. Once we have a good approximate solution, we can:
1. Identify the active set $S = \{k : c_k = \max_m c_m\}$ (or approximately)
2. Solve the KKT conditions on this active set:
   - $\sum_{k \in S} w_k (a*a)_k = \max$ (automatically satisfied)
   - $\sum_{k \in S} w_k \frac{\partial c_k}{\partial a_j} = \lambda$ for all $j$ (with $w_k \geq 0$, $\sum w_k = 1$)
   - $a_j > 0 \forall j$ (assuming interior solution)

This gives us a system of linear equations in $w$ and $\lambda$:
$\sum_{k \in S} w_k \cdot 2a_{k-j} \cdot \mathbb{1}_{0 \leq k-j < n} = \lambda$ for $j = 0, \ldots, n-1$
$\sum_{k \in S} w_k = 1$
$w_k \geq 0$

This is a linear feasibility problem. If we can find a feasible solution, then our current $a$ is a KKT point.

Actually, a simpler approach: just use the top-K elements as the "active set" and do a few iterations of the projected gradient or Newton method on the smoothed objective.

"""
Optimized step function height minimizer using log-sum-exp smoothing,
L-BFGS in log-space, active-set KKT refinement, and multi-scale n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    """Next power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a, b):
    """FFT-based convolution."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(np.clip(z, -100, 100))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _log_softmax_grad(a):
    """Gradient of softmax w.r.t. log-input: a_k * (delta_kj - a_j)."""
    return a  # We'll compute this in the gradient function

def _smoothed_obj_grad(z, beta, n):
    """
    Compute objective and gradient of the log-sum-exp smoothed objective
    in log-space (z = log(a_normalized)).
    
    F_beta = 2n * beta * log(sum_k exp(c_k / beta))
    where c = conv(a, a), a = softmax(z), sum(a) = 1.
    """
    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    
    if beta < c_max * 1e-10:
        # Effectively the hard max
        obj = 2.0 * n * c_max
        # Gradient uses one-hot at argmax
        k_max = int(np.argmax(c))
        p = np.zeros_like(c)
        p[k_max] = 1.0
    else:
        shifted = c / beta - c_max / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300:
            obj = 2.0 * n * c_max
            p = np.zeros_like(c)
            p[int(np.argmax(c))] = 1.0
        else:
            obj = 2.0 * n * (c_max + beta * np.log(s))
            p = e / s
    
    # Gradient of F_beta w.r.t. a: 2n * sum_k p_k * (partial c_k / partial a_j)
    # partial c_k / partial a_j = 2 * a_{k-j} if 0 <= k-j < n
    # So grad_a_j = 4n * sum_m a_m * p_{j+m}
    # = 4n * (cross-corr of a and p at lag j)
    
    a_rev = a[::-1]
    N = _next_pow2(2 * len(a) + len(p))
    F_a = np.fft.rfft(a_rev, N)
    F_p = np.fft.rfft(p, N)
    corr = np.fft.irfft(F_a * F_p, N)
    h = corr[n - 1:2 * n - 1]  # cross-correlation at lags 0..n-1
    
    grad_a = 4.0 * n * h
    
    # Convert to z-space: grad_z_j = a_j * (grad_a_j - sum_k a_k * grad_a_k)
    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)
    
    return obj, grad_z

def _optimize_smoothed(a_init, deadline, beta_start=0.1, beta_min=1e-6, min_time=30.0):
    """
    Optimize the smoothed objective using L-BFGS in log-space.
    Adaptively decreases beta.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-300))
    z = z - z.mean()
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    beta = beta_start
    
    while time.monotonic() + min_time < deadline:
        if time.monotonic() > deadline:
            break
        
        # Estimate time per iteration
        t0 = time.monotonic()
        
        def obj_fn(z_val):
            obj, _ = _smoothed_obj_grad(z_val, beta, n)
            return obj
        
        def grad_fn(z_val):
            _, g = _smoothed_obj_grad(z_val, beta, n)
            return g
        
        # Run L-BFGS
        max_iter = 200
        est_time_per_iter = 0.05  # rough estimate
        remaining = deadline - min_time - time.monotonic()
        max_iter = min(max_iter, max(10, int(remaining / max(est_time_per_iter, 0.01))))
        
        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12,
                         'maxcor': 40, 'maxls': 50}
            )
            new_a = _softmax(result.x)
            new_sc = _score(new_a)
            
            if new_sc < current_sc - 1e-12:
                current_a = new_a
                current_sc = new_sc
                z = np.log(np.maximum(new_a, 1e-300))
                z = z - z.mean()
        except Exception as e:
            pass
        
        # Check if we should decrease beta
        c = _fft_convolve(current_a, current_a)
        c_max = float(c.max())
        c_sorted = np.sort(c)[::-1]
        # If the top elements are very close, we're in a "flat" regime
        top_50 = c_sorted[:min(50, len(c_sorted))]
        spread = top_50[0] - top_50[-1]
        
        if spread < c_max * 0.001:
            beta *= 0.3
        elif spread < c_max * 0.01:
            beta *= 0.5
        else:
            beta *= 0.7
        
        beta = max(beta, c_max * 1e-8)
        
        if beta < c_max * 1e-6:
            break
    
    return current_a, current_sc

def _kkt_refine(a_init, deadline, max_rounds=100):
    """
    Active-set KKT refinement. Identify the active set (top-K convolution values)
    and solve the KKT conditions.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    k_schedule = [1, 3, 5, 10, 20, 50, 100, 200, 500, 1000]
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        K = k_schedule[rnd % len(k_schedule)]
        K = min(K, 2 * n - 1)
        idx = np.argpartition(c, -K)[-K:]
        W = np.zeros(2 * n - 1)
        W[idx] = 1.0 / K
        
        # Compute gradient of the "W-average" of convolution values
        # d/d a_j [sum_k W_k * c_k] = 4n * (W-cross-corr with a) at lag j
        a_rev = a[::-1]
        N = _next_pow2(3 * n)
        F_a = np.fft.rfft(a_rev, N)
        F_W = np.fft.rfft(W, N)
        corr = np.fft.irfft(F_a * F_W, N)
        h = corr[n - 1:2 * n - 1]
        
        grad = 4.0 * n * h
        W_val = float(np.dot(W, c))
        
        # Normalize gradient
        grad_centered = grad - np.mean(grad)
        gn = float(np.linalg.norm(grad_centered))
        if gn < 1e-14 * n:
            break
        
        # Newton step using CG on the "Hankel" operator
        # H[v] = 4n * (W-cross-corr of v with a-reversed) ... 
        # Actually the Hessian in this formulation is complex.
        # Let's use a steepest descent step with line search.
        
        dx = -grad_centered / gn
        
        # Line search
        alpha = 1.0
        step_done = False
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _fft_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1.0 - 0.005 * alpha) - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break
        
        if not step_done:
            # Try smaller fixed steps
            for trial in [0.01, 0.005, 0.001, 0.0005]:
                a_try = a - trial * grad_centered / gn
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _fft_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break
        
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc

def _resample(a, n_new):
    """Resample sequence to new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.where(np.abs(t - 0.5) < 0.3, 1.0, 0.1)
        return a / a.sum()
    else:
        return np.ones(n) / n

import numpy as np

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -200, 200))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoothed_obj_grad(z, beta, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = float(c.max())

    if beta <= c_max * 1e-10:
        obj = 2.0 * n * c_max
        k_max = int(np.argmax(c))
        p = np.zeros_like(c)
        p[k_max] = 1.0
    else:
        shifted = c / beta - c_max / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300:
            obj = 2.0 * n * c_max
            p = np.zeros_like(c)
            p[int(np.argmax(c))] = 1.0
        else:
            obj = 2.0 * n * (c_max + beta * np.log(s))
            p = e / s

    a_rev = a[::-1]
    size = len(a_rev) + len(p) - 1
    N = _next_pow2(size)
    F_a = np.fft.rfft(a_rev, N)
    F_p = np.fft.rfft(p, N)
    corr = np.fft.irfft(F_a * F_p, N)
    h = corr[n - 1:2 * n - 1]

    grad_a = 4.0 * n * h
    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)

    return obj, grad_z

def _kkt_refine(a_init, deadline, max_rounds=150):

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n
