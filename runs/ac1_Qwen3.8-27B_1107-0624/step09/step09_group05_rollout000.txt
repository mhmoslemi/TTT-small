def _next_pow2(n):

def _fft_convolve(a, b):

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):

        return np.inf
    c = _fft_conv(a, a)

def _conv(a):

def _grad_for_active(a, active):
    """Gradient of mean c[k] over active set, w.r.t. a.
    grad[j] = 2/|S| * sum_{k in S} a[k-j]
    """
    n = len(a)
    a_rev = a[::-1]
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / max(1, active.sum())

    return 2.0 * wc[n - 1:2 * n - 1]


def _project_onto_simplex(v):
    """Project v onto {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
