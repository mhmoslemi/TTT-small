"""
Multi-restart L-BFGS optimizer with log-sum-exp smoothing for minimum
peak autocorrelation sequences. Uses FFT-based gradient computation,
adaptive beta continuation, and diverse initializations to escape local minima.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 2
    while k < n:
        k <<= 1
    return k


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _grad_smoothed_fft(a, w, n):
    """Compute dF/da where F = 2n * beta * log(sum exp(c/beta)).
    Returns the gradient in a-space (before softmax chain rule).
    grad[i] = 4n * (w * a_rev)[n-1+i]
    """
    a_rev = a[::-1]
    conv = _full_convolve(w, a_rev)
    return 4.0 * n * conv[n-1:2*n-1]


def _lbfgs_step(a_init, beta, max_iter, deadline, min_time_remaining=5.0):
    """One L-BFGS step at a given beta. Returns (a_new, success)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    def obj_func(z_local):
        a_local = _softmax(z_local)
        c_local = _full_convolve(a_local, a_local)
        c_max_local = float(c_local.max())
        shifted = (c_local - c_max_local) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s_e = e.sum()
        if s_e < 1e-300:
            return 2.0 * n * c_max_local
        return 2.0 * n * (c_max_local + beta * np.log(s_e))
    
    def grad_func(z_local):
        a_local = _softmax(z_local)
        c_local = _full_convolve(a_local, a_local)
        c_max_local = float(c_local.max())
        shifted = (c_local - c_max_local) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s_e = e.sum()
        if s_e < 1e-300:
            w = np.zeros_like(c_local)
            w[int(np.argmax(c_local))] = 1.0
        else:
            w = e / s_e
        g = _grad_smoothed_fft(a_local, w, n)
        W = float(np.dot(g, a_local))
        return a_local * (g - W)
    
    est_time = 0.005 * max(1.0, n / 8192.0)
    actual_max_iter = min(max_iter, max(30, int((deadline - min_time_remaining - time.monotonic()) / (est_time * 5))))
    if actual_max_iter < 10:
        return a_init, False
    
    try:
        result = minimize(
            obj_func, z,
            jac=grad_func,
            method='L-BFGS-B',
            options={'maxiter': actual_max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxcor': 50, 'maxls': 100}
        )
        a_new = _softmax(result.x)
        return a_new, True
    except Exception:
        return a_init, False

def _direct_subgrad(a_init, deadline, min_time_remaining=10.0, num_steps=2000):
    """Direct subgradient method with top-K active set."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.05
    for step in range(num_steps):
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Top-K active
        K = min(10, len(c))
        top_k = np.argsort(c)[::-1][:K]
        # Weight by proximity to max
        thresh = c_max * 0.99
        active = c >= thresh
        if active.sum() == 0:
            active[top_k[0]] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        g = _grad_smoothed_fft(a, w, n)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size
        lr = 2.0 / (gnorm * (1.0 + 0.01 * step))
        a_new = _proj_simplex(a - lr * g)
        
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.5
    
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "triangular":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s
