"""
Peak-equalization L-BFGS optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
- Newton-style peak equalization exploiting convolution Toeplitz structure
- Multi-phase adaptive top-K smoothing with aggressive annealing
- Proximal gradient with exact subgradient and simplex projection
- Multi-scale warm-start for restarts
- Hyperparameter sweeps over beta/K schedules
- Diverse initialization with spectral profiles
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _topk_smoobj_grad(z, beta, K, n):
    z_max = z.max()
    a = np.exp(z - z_max)
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    L = len(c)
    if K < L:
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(L)
        c_top = c
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s_val = e.sum()
    if s_val < 1e-300 or not np.isfinite(s_val):
        w = np.zeros(L)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(L)
        e_full[top_idx] = e
        w = e_full / s_val
        obj = 2.0 * n * (c_max + beta * np.log(s_val))
    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _peak_equalize(a, M=80, max_iters=50, deadline=None, min_time_remaining=5.0):
    """Newton-style peak equalization using convolution structure."""
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    current_sc = 2.0 * n * c_max

    for it in range(max_iters):
        if deadline is not None and time.monotonic() + min_time_remaining > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()

        L = 2 * n - 1
        M_eff = min(M, L)
        top_idx = np.argpartition(c, -M_eff)[-M_eff:]
        c_top = c[top_idx]
        spread = c_top.max() - c_top.min()
        if spread < c_max * 1e-10:
            break

        # Build G = H_M @ H_M^T using convolution values
        # G[j,l] = c[top_idx[l] - top_idx[j]] (with bounds check)
        G = np.zeros((M_eff, M_eff))
        for j in range(M_eff):
            for l in range(j, M_eff):
                diff = top_idx[l] - top_idx[j]
                if 0 <= diff < L:
                    G[j, l] = c[diff]
                elif -L < diff < 0:
                    # By symmetry of autocorrelation
                    G[j, l] = c[-diff]
                else:
                    G[j, l] = 0.0
                G[l, j] = G[j, l]

        # Target: reduce all top peaks toward their mean minus a small amount
        target = c_top.mean() - spread * 0.3
        rhs = c_top - target

        # Regularize and solve
        lam = 1e-6 * G.trace() / M_eff
        if lam < 1e-30:
            lam = 1e-30
        try:
            sol = np.linalg.solve(G + lam * np.eye(M_eff), rhs)
        except np.linalg.LinAlgError:
            sol = np.linalg.lstsq(G + lam * np.eye(M_eff), rhs, rcond=None)[0]

        # Compute d = -H^T @ sol
        # d[i] = -sum_j sol[j] * a[top_idx[j] - i] (valid indices)
        d = np.zeros(n)
        for j in range(M_eff):
            k_j = int(top_idx[j])
            i_lo = max(0, k_j - (n - 1))
            i_hi = min(n - 1, k_j)
            for i in range(i_lo, i_hi + 1):
                d[i] -= sol[j] * a[k_j - i]

        # Enforce sum(d) = 0
        d = d - d.mean()

        # Line search
        alpha = 0.01
        improved = False
        for ls in range(30):
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            break

    return best_a, best_sc


def _proximal_gradient(a_init, deadline, min_time_remaining=30.0, max_iters=5000):
    """Proximal gradient with exact subgradient and simplex projection."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    best_c_max = c_max
    current_sc = 2.0 * n * c_max

    eta = 1.0 / (4.0 * c_max + 1e-30)
    eta = min(eta, 0.5)
    tol = c_max * 0.005
    stall = 0
    max_stall = 100
    L = 2 * n - 1

    for it in range(max_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            best_c_max = c_max
            stall = 0
            eta = min(eta * 1.1, 1.0)
        else:
            stall += 1
            if stall > max_stall:
                eta *= 0.5
                tol = max(tol * 0.5, c_max * 0.0005)
                stall = 0
                if eta < 1e-6:
                    break

        # Active set
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > L:
            top_idx = np.argpartition(c, -L)[-L:]
            active = np.zeros(L, dtype=bool)
            active[top_idx] = True
            n_active = L

        # Subgradient: average gradient of active peaks
        w = np.zeros(L)
        w[active] = 1.0 / n_active
        g = _grad_maxconv(a, w)

        # Proximal step with projection onto simplex
        a_try = _project_simplex(a - eta * g)
        c_try = _full_convolve(a_try, a_try)
        sc_try = 2.0 * n * float(c_try.max())
        if sc_try < current_sc - 1e-15:
            a = a_try
        else:
            # Backtracking
            a = a_try  # Still move (helps escape)
            eta *= 0.5

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, schedule=None):
    """L-BFGS with top-K smoothing and adaptive schedule."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.008)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    if schedule is None:
        schedule = [
            (1.0, 0.15), (0.7, 0.10), (0.5, 0.08), (0.35, 0.06),
            (0.2, 0.04), (0.1, 0.03), (0.05, 0.02), (0.03, 0.015),
            (0.015, 0.01), (0.007, 0.007), (0.003, 0.005), (0.001, 0.003),
            (0.0005, 0.002), (0.0002, 0.001),
        ]

    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 15:
            break

        beta = base_beta * beta_mult
        K = max(5, int((2 * n - 1) * K_frac))

        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(30, int(remaining / (est_time * 15))))

        class _Prob:
            def __init__(self, b, K, nn):
                self.b = b
                self.K = K
                self.nn = nn
            def __call__(self, z):
                return _topk_smoobj_grad(z, self.b, self.K, self.nn)[0]
            def grad(self, z):
                return _topk_smoobj_grad(z, self.b, self.K, self.nn)[1]

        prob = _Prob(beta, K, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxcor': 128, 'maxls': 50}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgradient_bb(a_init, deadline, min_time_remaining=60.0):
    """Active-set subgradient with Barzilai-Borwein steps and momentum."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    L = 2 * n - 1
    active_frac = 0.02
    momentum = 0.6
    lr = 0.01
    lr_min = 1e-5
    lr_max = 0.05
    stall = 0
    max_stall = 100
    velocity = np.zeros(n)
    prev_a = None
    prev_g = None
    iteration = 0
    last_lbfgs_time = time.monotonic()

    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        n_active = max(5, min(L, int(L * active_frac)))
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_active]

        active_vals = c[active_idx]
        w = np.zeros(L)
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w[active_idx] = (active_vals - active_vals.min()) / spread + 0.05
        else:
            w[active_idx] = 1.0 / n_active
        w = w / w.sum()

        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break

        if prev_a is not None and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g_centered - prev_g
            s_norm = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm > 1e-18 and abs(sy) > 1e-18:
                lr_bb = s_norm / abs(sy)
                lr = np.clip(lr_bb, lr_min, lr_max)

        velocity = momentum * velocity - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            for ls in range(20):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-15:
                    improved = True
                    break
                velocity = momentum * velocity * 0.5 - lr * 0.5 * g_centered
                a_try = a + velocity
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    velocity = np.zeros(n)
                    a_try = a
                    break
                a_try = a_try / s_try
                lr *= 0.7

        if improved:
            prev_a = a.copy()
            prev_g = g_centered.copy()
            a = a_try
            lr = min(lr_max, lr * 1.02)
            active_frac = max(0.003, active_frac * 0.97)
        else:
            velocity = np.zeros(n)
            lr = max(lr_min, lr * 0.6)
            active_frac = min(0.1, active_frac * 1.5)

        if stall > max_stall:
            active_frac = min(0.1, active_frac * 2.0)
            lr = max(lr_min, lr * 0.3)
            momentum = min(0.85, momentum + 0.05)
            stall = 0

        if iteration % 400 == 0 and time.monotonic() - last_lbfgs_time > 100:
            if time.monotonic() + min_time_remaining + 60 < deadline:
                lbfgs_deadline = time.monotonic() + 50
                a_tmp = _run_lbfgs(a, lbfgs_deadline, min_time_remaining=5)
                sc_tmp = _score(a_tmp)
                if sc_tmp < best_sc:
                    best_a = a_tmp.copy()
                    best_sc = sc_tmp
                    a = a_tmp.copy()
                    velocity = np.zeros(n)
                    prev_a = None
                    prev_g = None
                    last_lbfgs_time = time.monotonic()

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":
        t = np.linspace(0, 1, n)
        a = 0.8 + 0.2 * np.cos(np.pi * t) ** 4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t) + 0.1
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2) + 0.03
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "super_flat":
        t = np.linspace(0, 1, n)
        a = 0.95 + 0.05 * np.sin(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _perturb_structured(a, rng, n_blocks=16):
    """Perturb in blocks to preserve local structure."""
    n = len(a)
    block_size = max(1, n // n_blocks)
    z = np.log(np.maximum(a, 1e-30))
    for start in range(0, n, block_size):
        end = min(start + block_size, n)
        shift = rng.standard_normal() * 0.02
        z[start:end] += shift
    z = z - z.mean()
    e = np.exp(z)
    return e / e.sum()


def _multi_scale_optimize(profile_fn, n_final, deadline, min_time_remaining=60, rng=None):
    """Multi-scale optimization: coarse to fine."""
    if rng is None:
        rng = np.random.default_rng(0)
    scales = [256, 512, 1024, 2048, 4096, 8192, n_final]
    scales = [s for s in scales if s <= n_final]
    if scales[-1] != n_final:
        scales.append(n_final)

    a = profile_fn(scales[0])
    for i, n_s in enumerate(scales):
        if i > 0:
            a = _resample(a, n_s)
        if time.monotonic() + min_time_remaining > deadline:
            break
        alloc = (deadline - min_time_remaining - time.monotonic()) * (0.3 ** (len(scales) - 1 - i))
        alloc = min(alloc, 300.0)
        if alloc < 10:
            continue
        a = _run_lbfgs(a, time.monotonic() + alloc, min_time_remaining=5)
        sc = _score(a)
        print(f"    multi-scale n={n_s}: score={sc:.10f}")
    return a


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    """Very fine active-set polish with small steps and adaptive tolerance."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.003
    tol = 0.003
    no_improve = 0
    L = 2 * n - 1

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 500:
            top_idx = np.argpartition(c, -500)[-500:]
            active = np.zeros(L, dtype=bool)
            active[top_idx] = True
            n_active = 500

        w = np.zeros(L)
        w[active] = 1.0 / n_active
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-16:
            break

        alpha = min(lr, 0.3 / (gnorm + 1e-15))
        improved = False
        for ls in range(35):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 15 == 0:
                tol = max(0.0001, tol * 0.7)
                lr *= 0.7
            if no_improve > 200:
                a = best_a.copy()
                tol = max(0.00005, tol * 0.5)
                lr = 0.002
                no_improve = 0
        else:
            if tol > 0.00005:
                tol = max(0.00005, tol * 0.96)
            lr = min(0.008, lr * 1.01)

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS with top-K smoothing,
    Newton-style peak equalization, proximal gradient, and multi-restart.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, tapered n={n_parent}, score={best_score:.10f}")

    n_main = len(best_seq)

    # === Phase 1: Coarse L-BFGS ===
    print(f"\n[P1] Coarse L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline, t_start + 250.0)
    a1 = _run_lbfgs(best_seq, p1_deadline, min_time_remaining=300)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [P1] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Peak equalization ===
    print(f"\n[P2] Peak equalization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        p2_deadline = min(deadline, time.monotonic() + 200.0)
        a2, sc2 = _peak_equalize(best_seq, M=100, max_iters=100,
                                 deadline=p2_deadline, min_time_remaining=300)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [P2] t={time.monotonic()-t_start:.1f}s")
        # Follow up with L-BFGS on the equalized point
        if time.monotonic() < deadline - 400:
            a2b = _run_lbfgs(best_seq, min(deadline, time.monotonic() + 100.0),
                             min_time_remaining=350)
            sc2b = _score(a2b)
            if sc2b < best_score:
                best_seq = a2b.copy()
                best_score = sc2b
                print(f"  NEW BEST: {best_score:.10f} [P2b] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Fine L-BFGS ===
    print(f"\n[P3] Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        p3_deadline = min(deadline, time.monotonic() + 250.0)
        a3 = _run_lbfgs(best_seq, p3_deadline, min_time_remaining=200)
        sc3 = _score(a3)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [P3] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Proximal gradient ===
    print(f"\n[P4] Proximal gradient (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        p4_deadline = min(deadline, time.monotonic() + 150.0)
        a4, sc4 = _proximal_gradient(best_seq, p4_deadline, min_time_remaining=200)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [P4] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Subgradient BB ===
    print(f"\n[P5] Subgradient BB (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        p5_deadline = min(deadline, time.monotonic() + 150.0)
        a5, sc5 = _subgradient_bb(best_seq, p5_deadline, min_time_remaining=200)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [P5] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Second peak equalization + L-BFGS ===
    print(f"\n[P6] Peak eq + L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        pe_deadline = min(deadline, time.monotonic() + 80.0)
        a6, sc6 = _peak_equalize(best_seq, M=150, max_iters=80,
                                 deadline=pe_deadline, min_time_remaining=150)
        if sc6 < best_score:
            best_seq = a6.copy()
            best_score = sc6
            print(f"  NEW BEST: {best_score:.10f} [P6a] t={time.monotonic()-t_start:.1f}s")
        if time.monotonic() < deadline - 180:
            a6b = _run_lbfgs(best_seq, min(deadline - 100, time.monotonic() + 80.0),
                             min_time_remaining=100)
            sc6b = _score(a6b)
            if sc6b < best_score:
                best_seq = a6b.copy()
                best_score = sc6b
                print(f"  NEW BEST: {best_score:.10f} [P6b] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Multi-restart exploration ===
    print(f"\n[P7] Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        restart_budget = deadline - 120 - time.monotonic()
        n_restart = max(2, int(restart_budget / 35))
        sigmas = [0.02, 0.04, 0.06, 0.10, 0.03, 0.05, 0.08, 0.12, 0.015, 0.025]
        for i in range(n_restart):
            if time.monotonic() > deadline - 80:
                break
            sigma = sigmas[i % len(sigmas)]
            a_pert = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs(a_pert, min(deadline - 60, time.monotonic() + 25.0),
                             min_time_remaining=5)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.10f} [restart_{i} sig={sigma}] "
                      f"t={time.monotonic()-t_start:.1f}s")

        # Structured perturbations
        for i in range(3):
            if time.monotonic() > deadline - 70:
                break
            n_blocks = [8, 16, 32][i]
            a_pert = _perturb_structured(best_seq, rng, n_blocks=n_blocks)
            a_r = _run_lbfgs(a_pert, min(deadline - 50, time.monotonic() + 20.0),
                             min_time_remaining=5)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.10f} [struct_{n_blocks}] "
                      f"t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Different profiles ===
    print(f"\n[P8] Different profiles (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        profiles = ["super_flat", "flat_taper", "cos_taper", "gauss"]
        for prof_kind in profiles:
            if time.monotonic() > deadline - 60:
                break
            a_prof = _make_profile(len(best_seq), prof_kind)
            # Perturb toward best
            a_prof = 0.5 * a_prof + 0.5 * best_seq
            a_prof = _normalize(a_prof)
            a_p = _run_lbfgs(a_prof, min(deadline - 40, time.monotonic() + 20.0),
                             min_time_remaining=5)
            sc_p = _score(a_p)
            if sc_p < best_score:
                best_seq = a_p.copy()
                best_score = sc_p
                print(f"  NEW BEST: {best_score:.10f} [profile_{prof_kind}] "
                      f"t={time.monotonic()-t_start:.1f}s")

    # === Phase 9: Multi-n exploration ===
    print(f"\n[P9] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        n_main = len(best_seq)
        n_options = [n_main // 2, n_main * 3 // 2, n_main * 2]
        n_options = [x for x in n_options if x != n_main and x >= 1024 and x <= 65536]
        for n_try in n_options[:2]:
            if time.monotonic() > deadline - 40:
                break
            a_res = _resample(best_seq, n_try)
            a_opt = _run_lbfgs(a_res, min(deadline - 30, time.monotonic() + 30.0),
                               min_time_remaining=5)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [n={n_try}] "
                      f"t={time.monotonic()-t_start:.1f}s")

    # === Phase 10: Final peak equalization + polish ===
    print(f"\n[P10] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        pe_deadline = min(deadline, time.monotonic() + 50.0)
        a_fin, sc_fin = _peak_equalize(best_seq, M=200, max_iters=50,
                                       deadline=pe_deadline, min_time_remaining=20)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_pe] t={time.monotonic()-t_start:.1f}s")

    if time.monotonic() < deadline - 15:
        a_fp, sc_fp = _fine_polish(best_seq, min(deadline - 5, time.monotonic() + 40.0),
                                   min_time_remaining=5)
        if sc_fp < best_score:
            best_seq = a_fp.copy()
            best_score = sc_fp
            print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 11: Tiny L-BFGS ===
    if time.monotonic() < deadline - 5:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        beta_fin = max(mb * 0.0001, 1e-10)
        K_fin = max(3, int(n * 0.001))
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()

        class _FinalProb:
            def __init__(self, b, K, nn):
                self.b = b
                self.K = K
                self.nn = nn
            def __call__(self, z):
                return _topk_smoobj_grad(z, self.b, self.K, self.nn)[0]
            def grad(self, z):
                return _topk_smoobj_grad(z, self.b, self.K, self.nn)[1]

        prob = _FinalProb(beta_fin, K_fin, n)
        remaining = deadline - 5 - time.monotonic()
        if remaining > 3:
            max_iter = min(2000, max(10, int(remaining / 0.02)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-20,
                                           'gtol': 1e-16, 'maxcor': 64})
                a_fin2 = np.exp(result.x - result.x.max())
                a_fin2 = a_fin2 / a_fin2.sum()
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] "
                          f"t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
