def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_vec(v, A_vec, n):
    """Compute corr(v, A)[j] = sum_m v[m]*A[m+j] for j=0..n-1."""
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fv = np.fft.rfft(v, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fv * fA, N)[:out_len]
    return result[:n]


def _hess_matvec(v, active_vec, n, scale):
    """Apply Hessian of (scale * mean_active_autocorrelation) to v."""
    return scale * _corr_vec(v, active_vec, n)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec_fn, rhs, max_cg=300, tol=1e-14):
    x = np.zeros_like(rhs)
    r = rhs - matvec_fn(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_cg):
        Ap = matvec_fn(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            break
        if rs < 1e-30:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _newton_step(a, c, c_max, active, mu):
    """Compute Newton step for active-set subproblem.
    active: boolean array of same length as c
    Returns (dx, active_vec) or (None, None) on failure.
    """
    n = len(a)
    active_bool = active
    m = int(active_bool.sum())
    if m < 1:
        return None, None
    if m > 2 * n:
        return None, None

    A_vec = active_bool.astype(np.float64) / m
    scale = 2.0

    # Gradient: g = 2 * corr(a, A_vec) restricted to [0, n-1]
    g = scale * _corr_vec(a, A_vec, n)
    g_t = g - np.mean(g)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-15:
        return None, None

    A_vec_ref = A_vec
    n_ref = n
    mu_ref = mu
    scale_ref = scale

    def _matvec(v):
        return scale_ref * _corr_vec(v, A_vec_ref, n_ref) + mu_ref * v

    u = _cg_solve(_matvec, g_t, max_cg=300)
    w = _cg_solve(_matvec, np.ones(n), max_cg=300)
    w_sum = float(w.sum())

    if abs(w_sum) < 1e-14:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w

    dx = dx - np.mean(dx)
    return dx, A_vec

def _make_profile(n, kind="uniform"):

        return np.ones(n) / n
    elif kind == "raised_cosine":

        a = 0.5 * (1 + np.cos(2 * np.pi * t))

    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.12f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={len(parent)}, score={sc_parent:.12f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        update_best(parent, _score(parent), "init")

    # === Phase 1: L-BFGS continuation (aggressive) ===
    print(f"\n[Phase 1] L-BFGS continuation (t={time.monotonic() - t_start:.1f}s)")
    p1_dl = min(deadline - 600, time.monotonic() + 300)
    a_p1 = _lbfgs_continuation(parent, p1_dl, min_time_remaining=600.0)
    update_best(a_p1, _score(a_p1), "lbfgs_phase1")

    # === Phase 2: Active-set Newton ===
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        p2_dl = min(deadline - 400, time.monotonic() + 200)
        a_p2, sc_p2 = _active_set_newton(best_seq, p2_dl, max_outer=100)
        update_best(a_p2, sc_p2, "as_newton_phase2")

    # === Phase 3: Multi-peak deflation ===
    print(f"\n[Phase 3] Multi-peak deflation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        p3_dl = min(deadline - 300, time.monotonic() + 60)
        a_p3, sc_p3 = _multi_peak_deflation(best_seq, p3_dl, n_attempts=80)
        update_best(a_p3, sc_p3, "deflation_phase3")

    # === Phase 4: Iterative refinement ===
    print(f"\n[Phase 4] Iterative refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        p4_dl = min(deadline - 200, time.monotonic() + 150)
        a_p4, sc_p4 = _refinement_loop(best_seq, p4_dl, max_rounds=15)
        update_best(a_p4, sc_p4, "refinement_phase4")

    # === Phase 5: L-BFGS polish ===
    print(f"\n[Phase 5] L-BFGS polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        p5_dl = min(deadline - 150, time.monotonic() + 100)
        a_p5 = _lbfgs_continuation(best_seq, p5_dl, min_time_remaining=150.0)
        update_best(a_p5, _score(a_p5), "lbfgs_polish")

    # === Phase 6: Try n=32768 ===
    print(f"\n[Phase 6] Try n=32768 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 120 and best_seq is not None:
        n_large = 32768
        a_large = _interpolate_seq(best_seq, n_large)
        sc_large_init = _score(a_large)
        print(f"  n={n_large} init score: {sc_large_init:.12f}")

        p6_dl = min(deadline - 80, time.monotonic() + 60)
        a_large_opt = _lbfgs_continuation(a_large, p6_dl, min_time_remaining=80.0)
        sc_large = _score(a_large_opt)
        update_best(a_large_opt, sc_large, "n_32768")

        # Quick Newton on large
        if time.monotonic() < deadline - 60:
            p6b_dl = min(deadline - 60, time.monotonic() + 40)
            a_large_n, sc_large_n = _active_set_newton(a_large_opt, p6b_dl, max_outer=30)
            update_best(a_large_n, sc_large_n, "n_32768_newton")

    # === Phase 7: Diversified restarts ===
    print(f"\n[Phase 7] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        for sigma, label in [(0.003, "tiny"), (0.01, "small"), (0.03, "med")]:
            if time.monotonic() > deadline - 40:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _lbfgs_continuation(p_start, deadline, min_time_remaining=40.0)
            update_best(a_r, _score(a_r), f"restart_{label}")

    # === Phase 8: Final super-polish ===
    print(f"\n[Phase 8] Final super-polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        # Refinement
        ref_dl = min(deadline - 5, time.monotonic() + 40)
        a_fin, sc_fin = _refinement_loop(best_seq, ref_dl, max_rounds=10)
        update_best(a_fin, sc_fin, "final_ref")

        # Final tight Newton
        if time.monotonic() < deadline - 3:
            fin_dl = deadline - 2
            a_fin2, sc_fin2 = _final_polish(best_seq, fin_dl)
            update_best(a_fin2, sc_fin2, "final_tight_newton")

        # Very tiny beta L-BFGS
        if time.monotonic() < deadline - 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 1e-8
            remaining = deadline - 4 - time.monotonic()
            if remaining > 3:
                max_iter_fin = min(3000, max(100, int(remaining / 0.04)))
                try:
                    result = minimize(
                        _lsobj, z, args=(beta_final, n),
                        jac=_lsgrad, method='L-BFGS-B',
                        options={'maxiter': max_iter_fin, 'ftol': 1e-20,
                                 'gtol': 1e-16, 'maxcor': 200, 'maxls': 500}
                    )
                    a_fin3 = _softmax(result.x)
                    a_fin3 = np.maximum(a_fin3, 0)
                    s_fin3 = a_fin3.sum()
                    if s_fin3 > 1e-15:
                        a_fin3 = a_fin3 / s_fin3
                    update_best(a_fin3, _score(a_fin3), "final_tiny_beta")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
