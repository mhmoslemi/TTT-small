    """Frequency-domain subgradient descent.
    
    Operates on P = |FFT(a)|^2. The gradient of conv[m] w.r.t. P[k] is
    (1/N)*exp(2*pi*i*k*m/N). We take steps in P-space to reduce the peak.
    """
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(seed)
    
    # Adaptive step size
    base_lr = 0.5
    lr = base_lr
    K = min(5, 2 * n - 1)
    no_improve = 0
    stall_count = 0
    iter_count = 0
    
    while time.monotonic() < deadline:
        iter_count += 1
        
        # Compute FFT and power spectrum
        A = np.fft.fft(a, N)
        P = np.abs(A) ** 2
        phase = np.angle(A)
        
        # Compute circular autocorrelation = IDFT(P)
        r_circ = np.fft.ifft(P).real
        
        # Extract linear convolution from circular autocorrelation
        # For zero-padded real signal: conv[m] = N * r_circ[m] for m = 0..n-1
        # and conv[2n-1-m] = conv[m] by symmetry
        # Actually, let's just compute conv directly for correctness
        conv_lin = np.zeros(2 * n - 1)
        conv_lin[:n] = r_circ[:n] * N
        conv_lin[n:] = r_circ[N - n + 1:N][::-1] * N
        
        c_max = float(conv_lin.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-12:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and iter_count % 200 == 0:
            print(f"  FreqSG iter={iter_count}: best={best_sc:.10f}, K={K}, lr={lr:.6f}")
        
        # Adaptive K: grow active set
        c_sorted = np.sort(conv_lin)[::-1]
        if K < min(K_max, 2 * n - 1):
            gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
            if gap < c_max * 0.002:
                K = min(K_max, int(K * 1.3) + 1)
        
        # Find active peaks (top K lags)
        top_idx = np.argsort(conv_lin)[::-1][:K]
        
        # Compute subgradient in P-space
        # d conv[m] / d P[k] = (1/N) * exp(2*pi*i*k*m/N)
        # For real P, the effective gradient (real part) is (1/N)*cos(2*pi*k*m/N)
        # We want to DECREASE P where it contributes to the peak
        k_vals = np.arange(N)
        # Build gradient matrix: grad_P[k, m] = (1/N)*cos(2*pi*k*m/N)
        # For efficiency, compute as outer product
        angles = 2.0 * np.pi * np.outer(k_vals, top_idx) / N
        grad_P = np.cos(angles).mean(axis=1)  # average over active peaks
        
        # Step: P_new = P - lr * grad_P (clipped at 0)
        P_new = P - lr * grad_P
        P_new = np.maximum(P_new, 0.0)
        
        # Reconstruct A and a
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real[:n]
        
        # Project: clip negatives, normalize
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            # Failed, try smaller step
            lr *= 0.5
            continue
        a_new = a_new / s_new
        
        # Check if this step improved things
        c_new = np.zeros(2 * n - 1)
        A_check = np.fft.fft(a_new, N)
        r_check = np.fft.ifft(np.abs(A_check)**2).real
        c_new[:n] = r_check[:n] * N
        c_new[n:] = r_check[N - n + 1:N][::-1] * N
        sc_new = 2.0 * n * float(c_new.max())
        
        if sc_new < current_sc * 1.001:
            a = a_new
        else:
            # Step was bad, reduce lr
            lr *= 0.7
        
        # Periodic restart
        if no_improve > 500 and no_improve % 500 == 0:
            stall_count += 1
            if stall_count > 8:
                break
            # Random perturbation
            perturb = 0.002 * rng.standard_normal(n)
            a_pert = _clip_normalize(a + perturb)
            sc_pert = _score(a_pert)
            if sc_pert < best_sc:
                a = a_pert
                best_a = a_pert.copy()
                best_sc = sc_pert
            lr = base_lr
            no_improve = 0
        
        # Gradually decrease lr
        if iter_count % 100 == 0:
            lr *= 0.98
    
    return best_a, best_sc


def _spatial_subgrad_iter(a_init, deadline, seed=0, K_init=10, K_max=200, verbose=True):
    """Spatial-domain projected subgradient on the simplex."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())
    
    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    momentum = 0.85
    base_lr = 0.003
    lr = base_lr
    K = K_init
    t = 0
    no_improve = 0
    stall_count = 0
    
    while time.monotonic() < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive K
        c_sorted = np.sort(c)[::-1]
        if K < K_max:
            gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
            if gap < c_max * 0.003:
                K = min(K_max, int(K * 1.4) + 1)
        
        # Active peaks
        c_idx = np.argsort(c)[::-1][:K]
        
        # Subgradient
        grad = _conv_grad_multi(a, c_idx)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        
        if gnorm < 1e-18:
            break
        
        # Nesterov step
        v = momentum * v - (lr / (gnorm + 1e-18)) * 0.1 * grad_t
        a_cand = a + v
        a_proj = _project_simplex(a_cand)
        
        c_new = _full_convolve(a_proj, a_proj)
        sc_new = 2.0 * n * float(c_new.max())
        
        if sc_new < current_sc:
            a = a_proj
            lr = min(base_lr * 2, lr * 1.002)
        else:
            v = 0.5 * v
            lr *= 0.95
            a = a_proj
        
        # Periodic restart
        if no_improve > 300 and no_improve % 300 == 0:
            stall_count += 1
            if stall_count > 10:
                break
            perturb = 0.001 * rng.standard_normal(n)
            a_pert = _project_simplex(a + perturb)
            sc_pert = _score(a_pert)
            if sc_pert < best_sc:
                a = a_pert
                best_a = a_pert.copy()
                best_sc = sc_pert
            v = np.zeros(n)
            lr = base_lr
            no_improve = 0
        
        if verbose and t % 500 == 0:
            print(f"  SpatialSG t={t}: best={best_sc:.10f}, K={K}, lr={lr:.6f}")
    
    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time=30.0, verbose=True):
    """L-BFGS with adaptive beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03125, 0.016, 0.008, 0.004,
                                  0.002, 0.001, 0.0005, 0.00025,
                                  0.0001, 0.00005, 0.00002, 0.00001])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 3:
            break
        
        est_time = 0.003 * max(1.0, n / 16384.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 15))))
        
        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 100, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose and i % 3 == 0:
                    print(f"    L-BFGS beta_idx={i}, beta={beta:.6f}: score={current_sc:.10f}")
        except Exception:
            pass
    
    return current_a


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":
        a = np.ones(n)
        taper = max(1, n // 100)
        for i in range(taper):
            a[i] = (i + 1) / taper
            a[n - 1 - i] = (i + 1) / taper
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """Hybrid spectral + spatial + L-BFGS optimizer with multiple phases."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")
    
    n = len(best_seq)
    
    # === Phase 1: Frequency-domain subgradient (NEW) ===
    print(f"\n[P1] Frequency-domain subgradient (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 600, t_start + 400)
    if time.monotonic() < p1_deadline:
        a_freq, sc_freq = _freq_subgrad_iter(best_seq, p1_deadline, seed=seed,
                                             K_max=50, verbose=True)
        if sc_freq < best_score:
            best_seq = a_freq.copy()
            best_score = sc_freq
            print(f"  NEW BEST: {best_score:.10f} [freq_subgrad] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  FreqSG: no improvement (best={best_score:.10f})")
    
    # === Phase 2: Spatial subgradient ===
    print(f"\n[P2] Spatial subgradient (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 500, t_start + 550)
    if time.monotonic() < p2_deadline:
        a_spat, sc_spat = _spatial_subgrad_iter(best_seq, p2_deadline, seed=seed + 1,
                                                K_init=10, K_max=150, verbose=True)
        if sc_spat < best_score:
            best_seq = a_spat.copy()
            best_score = sc_spat
            print(f"  NEW BEST: {best_score:.10f} [spatial_subgrad] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 3: L-BFGS polish ===
    print(f"\n[P3] L-BFGS polish (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 400, t_start + 700)
    if time.monotonic() < p3_deadline:
        a_lb = _run_lbfgs(best_seq, p3_deadline, min_time=60.0, verbose=True)
        sc_lb = _score(a_lb)
        if sc_lb < best_score:
            best_seq = a_lb.copy()
            best_score = sc_lb
            print(f"  NEW BEST: {best_score:.10f} [lbfgs] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 4: Second frequency pass ===
    print(f"\n[P4] Frequency subgradient round 2 (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 300, t_start + 900)
    if time.monotonic() < p4_deadline:
        a_freq2, sc_freq2 = _freq_subgrad_iter(best_seq, p4_deadline, seed=seed + 2,
                                               K_max=80, verbose=True)
        if sc_freq2 < best_score:
            best_seq = a_freq2.copy()
            best_score = sc_freq2
            print(f"  NEW BEST: {best_score:.10f} [freq2] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 5: Spatial round 2 ===
    print(f"\n[P5] Spatial subgradient round 2 (t={time.monotonic()-t_start:.1f}s)")
    p5_deadline = min(deadline - 250, t_start + 1050)
    if time.monotonic() < p5_deadline:
        a_spat2, sc_spat2 = _spatial_subgrad_iter(best_seq, p5_deadline, seed=seed + 3,
                                                  K_init=5, K_max=100, verbose=True)
        if sc_spat2 < best_score:
            best_seq = a_spat2.copy()
            best_score = sc_spat2
            print(f"  NEW BEST: {best_score:.10f} [spatial2] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 6: L-BFGS round 2 (finer) ===
    print(f"\n[P6] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    p6_deadline = min(deadline - 200, t_start + 1200)
    if time.monotonic() < p6_deadline:
        a_lb2 = _run_lbfgs(best_seq, p6_deadline, min_time=60.0, verbose=True)
        sc_lb2 = _score(a_lb2)
        if sc_lb2 < best_score:
            best_seq = a_lb2.copy()
            best_score = sc_lb2
            print(f"  NEW BEST: {best_score:.10f} [lbfgs2] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [n, 8192, 16384, 24576, 32768]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)
    
    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == len(best_seq):
            continue
        if n_try > 40000:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")
        
        if time.monotonic() < deadline - 60:
            # Quick L-BFGS
            p7d = min(deadline - 60, time.monotonic() + 80)
            a_opt = _run_lbfgs(a_init, p7d, min_time=20.0, verbose=False)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_lbfgs]")
        
        if time.monotonic() < deadline - 50:
            # Quick freq subgrad
            p7d2 = min(deadline - 50, time.monotonic() + 60)
            a_freq_n, sc_freq_n = _freq_subgrad_iter(
                a_opt if 'a_opt' in dir() else a_init, p7d2,
                seed=seed + n_try, K_max=30, verbose=False)
            if sc_freq_n < best_score:
                best_seq = a_freq_n.copy()
                best_score = sc_freq_n
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_freq]")
    
    # === Phase 8: Try different initializations at best n ===
    print(f"\n[P8] Alternative initializations (t={time.monotonic()-t_start:.1f}s)")
    best_n = len(best_seq)
    inits = ["cos_mod", "flat_taper", "tapered"]
    for kind in inits:
        if time.monotonic() > deadline - 40:
            break
        a_alt = _make_profile(best_n, kind)
        # Quick optimization
        if time.monotonic() < deadline - 30:
            pd = min(deadline - 30, time.monotonic() + 50)
            a_alt_opt, sc_alt = _freq_subgrad_iter(a_alt, pd, seed=seed + 99,
                                                   K_max=30, verbose=False)
            if sc_alt < best_score:
                best_seq = a_alt_opt.copy()
                best_score = sc_alt
                print(f"  NEW BEST: {best_score:.10f} [init={kind}]")
    
    # === Phase 9: Final polish ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 10 - time.monotonic()
    if remaining > 30:
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        c_sorted = np.sort(c_best)[::-1]
        gap = mb - c_sorted[min(10, len(c_sorted) - 1)]
        beta_fin = max(gap * 0.01, mb * 0.00002)
        prob = _LbfgsProb(beta_fin, n)
        max_iter = min(3000, max(50, int(remaining / 0.05)))
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-15, 'maxcor': 80, 'maxls': 50})
            a_fin = _softmax(result.x)
            sc_fin = _score(a_fin)
            if sc_fin < best_score:
                best_seq = a_fin.copy()
                best_score = sc_fin
                print(f"  NEW BEST: {best_score:.10f} [final_lbfgs]")
        except Exception:
            pass
    
    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
