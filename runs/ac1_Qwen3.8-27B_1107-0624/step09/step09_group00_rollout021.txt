"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines parametric shape initialization, subgradient with backtracking,
log-sum-exp L-BFGS polish, spectral shaping, and iterated local search.
Key innovations: beta-distribution parametric starts, adaptive active-set
weighting with top-K lags, frequency-domain spectral flattening, and
curriculum-based beta annealing with plateaus.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _normalize(a):
    a = np.maximum(a, 0.0)

    if s < 1e-30:

def _subgradient(a, c, top_k=100, mode="inverse_gap"):
    """Compute weighted subgradient using top-K active lags."""
    n = len(a)
    c_max = float(c.max())
    
    # Get top-K lags
    k_eff = min(top_k, len(c))
    top_indices = np.argpartition(c, -k_eff)[-k_eff:]
    
    # Weighting scheme
    if mode == "inverse_gap":
        eps = 1e-12 * c_max
        w = np.zeros_like(c)
        gaps = c_max - c[top_indices]
        w[top_indices] = 1.0 / (gaps + eps)
    elif mode == "Gumbel":
        w = np.zeros_like(c)
        vals = c[top_indices] / max(c_max * 0.01, 1e-15)
        w[top_indices] = np.exp(np.clip(vals - vals.max(), -50, 0))
    elif mode == "uniform_active":
        tol = 0.1 * c_max
        w = np.zeros_like(c)
        w[c >= c_max - tol] = 1.0
    else:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        w_sum = 1.0
    w = w / w_sum
    
    # Gradient: grad[m] = 2 * sum_k w[k] * a[k-m] (valid indices)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad = 2.0 * h
    
    # Project onto tangent space of sum(a) = const
    grad = grad - np.mean(grad)
    return grad

        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        # Adapt top_k: start wide, narrow down
        progress = min(1.0, no_improve / max_stall)
        k_current = max(top_k_end, int(top_k_start * (1.0 - 0.8 * progress)))
        
        grad = _subgradient(a, c, top_k=k_current, mode=mode)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Backtracking line search
        alpha = lr
        improved = False
        for _ in range(40):
            a_try = _normalize(a - alpha * grad)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            lr = lr * 0.9
            if no_improve >= max_stall:
                a = best_a.copy()
                lr = lr_init * 0.5
                no_improve = 0
        else:
            lr = min(lr_init, lr * 1.01)
    
    return best_a, _score(best_a)


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def f(self, z):
        return _smoobj(z, self.beta, self.n)

    def g(self, z):
        return _smograd(z, self.beta, self.n)


def _lbfgs_optimize(a_init, deadline, min_time=20.0):
    """L-BFGS with curriculum beta annealing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

    # Curriculum: start with larger beta (smoother), anneal down
    betas = base_beta * np.array([
        1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.035, 0.025, 0.018, 0.012, 0.008, 0.005,
        0.003, 0.002, 0.001, 0.0005, 0.0002
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 10:
            break

        # Scale iterations with remaining time and problem size
        est_per_iter = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(30, int(remaining / (est_per_iter * 20))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob.f, z, jac=prob.g, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 120, 'maxls': 60}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _spectral_shaping(a, amount=0.3):
    """Reduce high-frequency components that cause convolution 'ringing'."""
    n = len(a)
    N = _next_pow2(2 * n)
    A = np.fft.rfft(a, N)
    # Smooth the spectrum: blend with a low-pass version
    freqs = np.fft.rfftfreq(N)
    # Gaussian low-pass
    bandwidth = 0.15
    lowpass = np.exp(-((freqs - 0.5) ** 2) / (2 * bandwidth ** 2))
    A_smooth = A * (1 - amount + amount * lowpass)
    a_new = np.fft.irfft(A_smooth, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _quick_optimize(a_init, deadline, min_time=15.0):
    """Combined optimization: subgradient then L-BFGS."""
    # Phase 1: Subgradient
    if time.monotonic() + min_time + 5 < deadline:
        sg_deadline = min(deadline - min_time - 5, time.monotonic() + 8)
        a1, sc1 = _subgradient_descent(a_init, sg_deadline, min_time=2,
                                        top_k_start=200, top_k_end=10,
                                        lr_init=0.015, mode="inverse_gap")
        if sc1 < _score(a_init):
            a_init = a1

    # Phase 2: L-BFGS
    if time.monotonic() + min_time < deadline:
        a2 = _lbfgs_optimize(a_init, deadline, min_time=min_time)
        if _score(a2) < _score(a_init):
            a_init = a2

    return a_init


def _ils_optimize(a_init, deadline, rng, min_time=45.0):
    """Iterated Local Search with diverse perturbations."""
    n = len(a_init)
    current_a = _normalize(a_init.copy())
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc
    attempt = 0

    while time.monotonic() + min_time < deadline:
        # Choose perturbation strategy
        strat = attempt % 6
        if strat == 0:
            # Global small noise
            z = np.log(np.maximum(current_a, 1e-30)) + 0.008 * rng.standard_normal(n)
            a_p = _softmax(z - z.mean())
        elif strat == 1:
            # Block perturbation
            bs = max(30, n // 6)
            st = int(rng.integers(0, n - bs))
            a_p = current_a.copy()
            seg = np.log(np.maximum(a_p[st:st + bs], 1e-30))
            seg += 0.04 * rng.standard_normal(bs)
            a_p[st:st + bs] = np.exp(seg - seg.max())
            a_p = _normalize(a_p)
        elif strat == 2:
            # Sparse perturbation
            npert = max(20, n // 12)
            idx = rng.choice(n, size=npert, replace=False)
            a_p = current_a.copy()
            a_p[idx] *= np.exp(0.06 * rng.standard_normal(npert))
            a_p = _normalize(a_p)
        elif strat == 3:
            # Spectral perturbation
            a_p = _spectral_shaping(current_a, amount=rng.uniform(0.1, 0.5))
        elif strat == 4:
            # Mirror + noise
            a_p = current_a.copy()
            slen = max(20, n // 15)
            st = int(rng.integers(0, n - slen))
            a_p[st:st + slen] = current_a[st:st + slen][::-1]
            z = np.log(np.maximum(a_p, 1e-30)) + 0.005 * rng.standard_normal(n)
            a_p = _softmax(z - z.mean())
        else:
            # Shape morphing: blend with a different parametric shape
            p = rng.uniform(0.5, 4.0)
            q = rng.uniform(0.5, 4.0)
            shape = _beta_profile(n, p, q)
            t = rng.uniform(0.1, 0.4)
            a_p = _normalize((1 - t) * current_a + t * shape)

        # Optimize the perturbation
        opt_dl = min(deadline - min_time, time.monotonic() + 6)
        a_opt = _quick_optimize(a_p, opt_dl, min_time=2)
        sc_opt = _score(a_opt)

        if sc_opt < best_sc - 1e-13:
            best_sc = sc_opt
            best_a = a_opt.copy()
            current_a = a_opt.copy()
            print(f"    ILS BEST: {best_sc:.10f} (strat={strat}) t={time.monotonic()-_t_start:.0f}s")
        elif sc_opt < current_sc - 1e-13:
            current_a = a_opt.copy()

        attempt += 1

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _final_polish(a, deadline):
    """Ultra-fine final optimization with very small beta."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    for beta_mult in [0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
        if time.monotonic() > deadline - 3:
            break
        beta = c_max * beta_mult
        remaining = deadline - 3 - time.monotonic()
        if remaining < 5:
            break
        prob = _LbfgsProb(beta, n)
        max_iter = min(3000, max(20, int(remaining / 0.02)))
        try:
            result = minimize(prob.f, z, jac=prob.g, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxcor': 100})
            a_new = _softmax(result.x)
            if _score(a_new) < _score(a):
                a = a_new
                z = np.log(np.maximum(a, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return a

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    P0: Load parent, evaluate
    P1: Parametric shape search (find best family)
    P2: Multi-start optimization (subgradient + L-BFGS)
    P3: Intensive L-BFGS on best
    P4: Subgradient polish
    P5: Iterated Local Search
    P6: Multi-n exploration
    P7: Second ILS round
    P8: Final ultra-fine polish
    """
    global _t_start
    _t_start = time.monotonic()
    deadline = _t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _general_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    n_main = n_parent

    # === P1: Parametric shape search (120s) ===
    print(f"\n[P1] Parametric shape search (t={time.monotonic()-_t_start:.1f}s)")
    p1_deadline = _t_start + 120

    shape_configs = [
        ("beta_1_2", "asym_beta", 1.0, 2.0),
        ("beta_2_1", "asym_beta", 2.0, 1.0),
        ("beta_1_3", "asym_beta", 1.0, 3.0),
        ("beta_3_1", "asym_beta", 3.0, 1.0),
        ("beta_0_5_2", "asym_beta", 0.5, 2.0),
        ("beta_2_0_5", "asym_beta", 2.0, 0.5),
        ("beta_1_5_1", "beta", 1.5, 1.0),
        ("beta_1_1_5", "beta", 1.0, 1.5),
        ("skew_gauss", "skew_gauss", 0, 0),
        ("power_law", "power_law", 0, 0),
        ("smoothstep", "smoothstep", 0, 0),
    ]

    for name, kind, p, q in shape_configs:
        if time.monotonic() > p1_deadline - 20:
            break
        a_init = _general_profile(n_main, kind, p, q)
        sc_init = _score(a_init)
        if sc_init < best_score:
            a_opt = _quick_optimize(a_init, p1_deadline, min_time=30)
            sc_opt = _score(a_opt)
            print(f"  {name}: init={sc_init:.8f}, opt={sc_opt:.8f}")
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [{name}]")
        else:
            print(f"  {name}: init={sc_init:.8f} (skip)")

    # Also optimize the parent
    if time.monotonic() < p1_deadline - 20:
        a_opt = _quick_optimize(parent, p1_deadline, min_time=20)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  parent_opt: {sc_opt:.10f} NEW BEST")

    # === P2: Intensive L-BFGS on best (200s) ===
    print(f"\n[P2] Intensive L-BFGS (t={time.monotonic()-_t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        lbfgs_dl = min(deadline - 600, time.monotonic() + 200)
        a_main = _lbfgs_optimize(best_seq, lbfgs_dl, min_time=60)
        sc = _score(a_main)
        if sc < best_score:
            best_seq = a_main.copy()
            best_score = sc
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_int]")

    # === P3: Subgradient polish (150s) ===
    print(f"\n[P3] Subgradient polish (t={time.monotonic()-_t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        sg_dl = min(deadline - 400, time.monotonic() + 150)
        a_sg, sc_sg = _subgradient_descent(best_seq, sg_dl, min_time=30,
                                           top_k_start=300, top_k_end=5,
                                           lr_init=0.008, mode="inverse_gap")
        if sc_sg < best_score:
            best_seq = a_sg.copy()
            best_score = sc_sg
            print(f"  NEW BEST: {best_score:.10f} [subgrad]")

    # === P4: Combined quick optimize on best ===
    print(f"\n[P4] Combined optimize (t={time.monotonic()-_t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        comb_dl = min(deadline - 350, time.monotonic() + 60)
        a_comb = _quick_optimize(best_seq, comb_dl, min_time=20)
        sc_comb = _score(a_comb)
        if sc_comb < best_score:
            best_seq = a_comb.copy()
            best_score = sc_comb
            print(f"  NEW BEST: {best_score:.10f} [combined]")

    # === P5: Iterated Local Search (200s) ===
    print(f"\n[P5] ILS Round 1 (t={time.monotonic()-_t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        ils_dl = min(deadline - 250, time.monotonic() + 200)
        a_ils, sc_ils = _ils_optimize(best_seq, ils_dl, rng, min_time=60)
        if sc_ils < best_score:
            best_seq = a_ils.copy()
            best_score = sc_ils
            print(f"  NEW BEST: {best_score:.10f} [ils1]")

    # === P6: Multi-n exploration (150s) ===
    print(f"\n[P6] Multi-n exploration (t={time.monotonic()-_t_start:.1f}s)")
    n_candidates = [n_main, 20480, 28672, 32768, 16384, 40960, 12288]
    seen = set()
    n_unique = []
    for nn in n_candidates:
        if nn not in seen and nn != len(best_seq) and 4096 <= nn <= 65536:
            seen.add(nn)
            n_unique.append(nn)

    for n_try in n_unique:
        if time.monotonic() > deadline - 120:
            break
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        opt_dl = min(deadline - 120, time.monotonic() + 25)
        a_opt = _quick_optimize(a_init, opt_dl, min_time=5)
        sc_opt = _score(a_opt)
        if sc_opt < best_score - 1e-10:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  n={n_try}: NEW BEST {best_score:.10f}")
        else:
            print(f"  n={n_try}: {sc_opt:.8f} (no improvement)")

    # === P7: Second ILS (120s) ===
    print(f"\n[P7] ILS Round 2 (t={time.monotonic()-_t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        rng2 = np.random.default_rng(int(seed) + 7777)
        ils2_dl = min(deadline - 80, time.monotonic() + 120)
        a_ils2, sc_ils2 = _ils_optimize(best_seq, ils2_dl, rng2, min_time=40)
        if sc_ils2 < best_score:
            best_seq = a_ils2.copy()
            best_score = sc_ils2
            print(f"  NEW BEST: {best_score:.10f} [ils2]")

    # === P8: Final ultra-fine polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-_t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin = _final_polish(best_seq, deadline - 2)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish]")

        # One more subgradient pass with very small top_k
        if time.monotonic() < deadline - 3:
            a_sg2, sc_sg2 = _subgradient_descent(best_seq, deadline - 2, min_time=2,
                                                 top_k_start=10, top_k_end=3,
                                                 lr_init=0.003, mode="inverse_gap")
            if sc_sg2 < best_score:
                best_seq = a_sg2.copy()
                best_score = sc_sg2
                print(f"  NEW BEST: {best_score:.10f} [final_subgrad]")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _general_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-_t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _general_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
