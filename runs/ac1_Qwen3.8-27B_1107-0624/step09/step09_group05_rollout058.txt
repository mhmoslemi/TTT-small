    res_padded[:n] = residual
    
    # Kernel for the linearization
    h = np.zeros(N)

    h[:len(w)] = 2.0 * w
    
    # FFT solve
    H = np.fft.rfft(h)
    R = np.fft.rfft(res_padded)
    
    # Avoid division by zero
    mask = np.abs(H) > 1e-15
    da_fft = np.zeros_like(H)
    da_fft[mask] = R[mask] / H[mask]
    
    da = np.fft.irfft(da_fft, N)[:n]
    
    return da

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _soft_max(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_soft(a, beta):

    n = len(a)

    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad, c


def _project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(a)
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(a - theta, 0)

def _smoothed_gradient_descent(a_init, deadline, min_remaining=60.0, verbose=False):

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.25) ** 2) + np.exp(-40.0 * (t - 0.75) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n, dtype=np.float64)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.6 * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    elif kind == "flat_middles":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        taper = 0.15
        for i in range(n):
            if t[i] < taper:
                a[i] = 0.5 + 0.5 * t[i] / taper
            elif t[i] > 1 - taper:
                a[i] = 0.5 + 0.5 * (1 - t[i]) / taper
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _diversify(a, rng):
    """Create a perturbed version of a for multi-start."""
    n = len(a)
    # Add small structured noise
    noise = rng.normal(0, 0.02, n)
    a_new = a + noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

            print(f"  P4 result: {best_sc:.10f}")
        else:
            print(f"  P4: no improvement (still {best_sc:.10f})")

    # === Phase 5: Multi-n exploration and diversification ===
    print(f"\n[P5] Multi-n and diversification (t={time.monotonic() - t0:.0f}s)")
    if time.monotonic() + 120 < deadline:
        # Try nearby n values
        n_options = [n_main - 1024, n_main + 1024, n_main * 3 // 2, n_main * 5 // 4]
        n_options = [x for x in n_options if x >= 2048 and x <= 32768 and x != n_main]
        for n_try in sorted(set(n_options)):
            if time.monotonic() + 60 >= deadline:
                break
            a_r = _resample(best_a, n_try)
            sc_r = _score(a_r)
            if sc_r < best_sc:
                best_sc = sc_r
                best_a = a_r.copy()
                print(f"  n={n_try}: resample improved to {best_sc:.10f}")
            # Quick KKT on resampled
            if time.monotonic() + 40 < deadline:
                a_r2, sc_r2 = _kkt_refine(a_r, min(deadline - 30, time.monotonic() + 40),
                                          min_remaining=5)
                if sc_r2 < best_sc:
                    best_sc = sc_r2
                    best_a = a_r2.copy()
                    print(f"  n={n_try}: KKT improved to {best_sc:.10f}")

        # Diversification: try perturbed versions
        if time.monotonic() + 80 < deadline:
            for div_i in range(3):
                if time.monotonic() + 30 >= deadline:
                    break
                a_div = _diversify(best_a, rng)
                sc_div = _score(a_div)
                if sc_div < best_sc + 0.05:  # only proceed if promising
                    a_div2, sc_div2 = _kkt_refine(a_div, min(deadline - 20, time.monotonic() + 30),
                                                  min_remaining=5)
                    if sc_div2 < best_sc:
                        best_sc = sc_div2
                        best_a = a_div2.copy()
                        print(f"  Diversify {div_i}: improved to {best_sc:.10f}")

    # === Phase 6: Final polish ===
    print(f"\n[P6] Final polish (t={time.monotonic() - t0:.0f}s)")
    if time.monotonic() + 20 < deadline:
        a6, sc6 = _kkt_refine(best_a, deadline - 5, min_remaining=3)
        if sc6 < best_sc:
            best_sc = sc6
            best_a = a6.copy()

    # Final output
    final = np.clip(best_a, 0.0, 1000.0)
    s_final = float(final.sum())
    if s_final < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"\n{'=' * 60}")
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t0:.1f}s")
    print(f"Target: 1.5027, Achieved: {sc_final:.10f}")
    return [float(x) for x in final]
