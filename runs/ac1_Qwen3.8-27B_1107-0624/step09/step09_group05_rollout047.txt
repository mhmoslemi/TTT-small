    Let me think again.
    
    dc[k]/da[j] = 2*a[k-j] if 0<=k-j<=n-1 and k != 2j
    dc[k]/da[j] = a[j] if k = 2j (i.e., k-j = j)
    
    So sum_k w[k]*dc[k]/da[j] = 2*sum_k w[k]*a[k-j]*1_{0<=k-j<=n-1, k!=2j} + w[2j]*a[j]
    
    = 2*conv(w, a_rev)[n-1+j] - 2*w[2j]*a[j] + w[2j]*a[j]  (if 2j < 2n-1)
    = 2*conv(w, a_rev)[n-1+j] - w[2j]*a[j]
    
    Hmm, this diagonal correction is annoying. For n=16384, the diagonal has n elements out of 2n-1, so it's not negligible.
    
    Actually wait. Let me reconsider.
    
    conv(w, a_rev)[m] = sum_k w[k] * a_rev[m-k] = sum_k w[k] * a[n-1-(m-k)] = sum_k w[k] * a[n-1-m+k]
    
    I want: sum_k w[k] * a[k-j] for k-j in [0, n-1]
    = sum_k w[k] * a[k-j]
    
    Let's use: conv(w, a)[j] = sum_k w[k] * a[j-k] ... no, that's cross-correlation.
    
    Let me just use the direct formula:
    sum_k w[k] * a[k-j] where 0 <= k-j <= n-1, i.e., j <= k <= j+n-1
    
    This is a "sliding dot product". I can compute it efficiently with FFT.
    
    Define b[m] = a[m] for m in [0, n-1], 0 otherwise.
    Define w_ext[k] = w[k] for k in [0, 2n-2], 0 otherwise.
    
    sum_k w_ext[k] * b[k-j] = (w_ext * b)[j] ... no, that's convolution.
    
    (w_ext * b)[m] = sum_k w_ext[k] * b[m-k]
    
    I want sum_k w_ext[k] * b[k-j] = sum_k w_ext[k] * b[k-j]
    
    Let l = k-j, so k = l+j:
    = sum_l w_ext[l+j] * b[l]
    
    This is the cross-correlation of w_ext and b, evaluated at j.
    = (b * w_ext)[j] ... no.
    
    Cross-correlation: (b ⋆ w_ext)[j] = sum_l b[l] * w_ext[l+j]
    
    Yes! So sum_k w_ext[k] * b[k-j] = (b ⋆ w_ext)[j]
    
    And cross-correlation can be computed via FFT: 
    (b ⋆ w_ext)[j] = ifft(conj(FFT(b)) * FFT(w_ext))[j]
    
    Or equivalently: conv(b, w_ext_reversed)
    
    Let me just use: np.convolve or FFT-based correlation.
    
    Actually, for efficiency, let me use the formula from the previous program:
    
    grad[j] = 2 * conv(w, a_rev)[n-1+j]
    
    where a_rev = a[::-1].
    
    conv(w, a_rev)[m] = sum_k w[k] * a_rev[m-k] = sum_k w[k] * a[n-1-m+k]
    
    For m = n-1+j: conv(w, a_rev)[n-1+j] = sum_k w[k] * a[n-1-(n-1+j)+k] = sum_k w[k] * a[k-j]
    
    Yes! This is exactly what we want. So:
    
    sum_k w[k] * a[k-j] (for valid k-j) = conv(w, a_rev)[n-1+j]
    
    (The convolution automatically handles the bounds since w is zero outside [0, 2n-2] and a is zero outside [0, n-1].)
    
    So the gradient (without the diagonal correction) is:
    g_no_diag[j] = 2 * conv(w, a_rev)[n-1+j]
    
    And the diagonal correction: when k=2j, the term is w[2j]*a[j] instead of 2*w[2j]*a[j].
    So the correction is -w[2j]*a[j] (subtract the extra a[j] that was double-counted).
    
    But w[2j] is only defined for 2j in [0, 2n-2], i.e., j in [0, n-1]. So for all j in [0, n-1], the correction is -w[2j]*a[j].
    
    Therefore: g[j] = 2*conv(w, a_rev)[n-1+j] - w[2j]*a[j]
    """
    n = len(a)
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    # Diagonal correction
    diag_w = w[0::2]  # w[0], w[2], w[4], ...
    # w[2j] for j in range(n): indices 0, 2, 4, ..., 2n-2
    # That's n elements
    if len(diag_w) == n:
        grad -= diag_w * a
    else:
        # Shouldn't happen for full convolution of length 2n-1
        m = min(len(diag_w), n)
        grad[:m] -= diag_w[:m] * a[:m]
    return grad

def _smoobj_sq(x, beta, n):
    """Objective using squared parameterization."""
    a = _sqnorm(x)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)

def _smograd_sq(x, beta, n):
    """Gradient using squared parameterization."""
    a = _sqnorm(x)
    S = float(np.dot(x, x))
    if S < 1e-300:
        return np.zeros_like(x)
    ga = _grad_conv_structured(a, beta)
    # g_x[m] = 2*x[m] * (ga[m] - sum_j ga[j]*a[j]) / S
    dot = float(np.dot(ga, a))
    return 2.0 * x * (ga - dot) / S

class _ProbSq:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, x):
        return _smoobj_sq(x, self.beta, self.n)

    def grad(self, x):
        return _smograd_sq(x, self.beta, self.n)

def _smoobj_sm(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)

def _smograd_sm(z, beta, n):
    a = _softmax(z)
    ga = _grad_conv_structured(a, beta)
    W = float(np.dot(ga, a))
    return a * (ga - W)

class _ProbSm:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj_sm(z, self.beta, self.n)

    def grad(self, z):
        return _smograd_sm(z, self.beta, self.n)

def _run_lbfgs_sq(x_init, deadline, min_time_remaining=60.0, verbose=True):
    """L-BFGS with squared parameterization."""
    n = len(x_init)
    best_a = _sqnorm(x_init)
    best_sc = _score(best_a)
    
    c0 = _full_convolve(best_a, best_a)
    c0_max = float(c0.max())
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < c0_max * 0.003:
        gap = c0_max * 0.003

    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
        0.00003125, 0.000015625, 0.0000078, 0.0000039,
        0.000002, 0.000001, 0.0000005
    ])
    betas = gap * multipliers

    t_start = time.monotonic()
    x = x_init.copy()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        size_factor = max(1.0, n / 8192.0)
        est_iter_time = 0.003 * size_factor
        max_iter = min(6000, max(50, int(remaining / (est_iter_time * 30))))

        prob = _ProbSq(beta, n)
        try:
            result = minimize(
                prob, x, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 50,
                    'maxls': 40,
                }
            )
            a_new = _sqnorm(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                x = np.sqrt(best_a)
                x[np.random.default_rng(len(betas)+i).integers(0, n)] += 1e-3
                # Add tiny perturbation to avoid stuck
                if verbose:
                    elapsed = time.monotonic() - t_start
                    print(f"    [sq] beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}, t={elapsed:.0f}s")
            else:
                if verbose and i < 8:
                    print(f"    [sq] beta={beta:.2e}: no improvement (best={best_sc:.10f})")
        except Exception as e:
            if verbose:
                print(f"    [sq] beta={beta:.2e}: error {e}")

    return best_a, best_sc

def _peak_equalize(a_init, deadline, min_time_remaining=30.0):
    """Peak equalization: iteratively make the top peaks of conv(a,a) equal.
    
    At the optimum, the active peaks should all have the same value.
    We find the top K peaks and solve for adjustments to equalize them.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    max_iter = 500
    for it in range(max_iter):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
        
        # Find top peaks
        # Use a tolerance to define the "active" peaks
        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros(len(c), dtype=bool)
            active[idx] = True
            n_active = 1
        
        # Compute: for each active peak k, c[k] = sum_j a[j]*a[k-j]
        # We want to make all c[k] for k in active equal to the maximum
        # (i.e., reduce the peaks that are above the mean of active peaks)
        
        # Alternative: compute the gradient direction that reduces the variance of active peaks
        active_indices = np.where(active)[0]
        c_active = c[active_indices]
        c_mean = float(c_active.mean())
        c_spread = float(c_active.max() - c_active.min())
        
        if c_spread < 1e-10 * c_max:
            # Peaks are already well-equalized
            tol *= 0.9
            if tol < 1e-6 * c_max:
                break
            continue
        
        # For each active peak k, the gradient w.r.t. a is:
        # dc[k]/da[j] = 2*a[k-j] (for valid k-j, k != 2j) or a[j] (for k=2j)
        # We want to move a in a direction that reduces the variance of c[active]
        
        # Subgradient of variance: for peaks above mean, we want to decrease c[k]
        # For peaks below mean, we want to increase c[k]
        weights = c_active - c_mean  # positive for peaks above mean
        
        # Gradient direction: sum_k weights[k] * dc[k]/da
        # = 2 * sum_k weights[k] * a[k-j] (approximately, ignoring diagonal)
        
        w_vec = np.zeros(len(c))
        w_vec[active_indices] = weights
        a_rev = a[::-1]
        wc = _full_convolve(w_vec, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]
        
        # Project to zero mean (to preserve sum)
        grad -= np.mean(grad)
        
        # Normalize
        g_norm = float(np.sqrt(np.dot(grad, grad)))
        if g_norm < 1e-15:
            break
        
        # Step size: try to reduce the spread
        # The spread should decrease by about step * g_norm * spread
        for eta in [0.1, 0.05, 0.02, 0.01, 0.005]:
            a_new = a - eta * grad / g_norm * c_spread
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-12:
                a = a_new
                break
    
    return best_a, best_sc

def _active_set_kkt(a_init, deadline, min_time_remaining=30.0):
    """KKT-based active set refinement.
    
    At the KKT optimum with active set S and support T:
    - For j in T: sum_{k in S} a[k-j] / |S| = lambda (constant)
    - a[j] >= 0 for all j, sum(a) = 1
    
    We iterate:
    1. Find S (active peaks) and T (support)
    2. Compute the "target" gradient from S
    3. Update a to equalize the gradient on T
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    no_improve = 0
    max_no_improve = 200
    tol_frac = 0.005
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            tol_frac = max(0.0005, tol_frac * 0.7)
            if tol_frac <= 0.0005:
                break
            no_improve = 0
        
        # Identify active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        
        # Compute average gradient from active set
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]
        
        # KKT: grad_avg should be constant on support T
        support = a > 1e-10 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break
        
        lam = float(np.mean(grad_avg[support]))
        residual = grad_avg[support] - lam
        
        # Update: reduce residual on support
        for eta in [0.1, 0.05, 0.02, 0.01]:
            a_new = a.copy()
            a_new[support] -= eta * residual
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-13:
                a = a_new
                break
        else:
            # No improvement with any step size
            tol_frac = max(0.0005, tol_frac * 0.9)
            continue
        
        # Tighten tolerance when making progress
        if no_improve == 0:
            tol_frac = max(0.0005, tol_frac * 0.98)
    
    return best_a, best_sc

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp_up":
        t = np.linspace(0, 1, n)
        a = t + 0.01
        return a / a.sum()
    elif kind == "ramp_down":
        t = np.linspace(0, 1, n)
        a = (1 - t) + 0.01
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t) + 0.005
        return a / a.sum()
    elif kind == "exp_growth":
        t = np.linspace(0, 1, n)
        a = np.exp(3.0 * t) + 0.005
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "one_sided_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t * 0.5) ** 1.5 + 0.005
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quadratic_ramp":
        t = np.linspace(0, 1, n)
        a = t**0.5 + 0.01
        return a / a.sum()
    elif kind == "inv_quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 / (t + 0.1)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _try_n_augment(a, extra_n, deadline):
    """Try extending the sequence by adding zeros or small values at the end."""
    n = len(a)
    new_n = n + extra_n
    a_new = np.zeros(new_n)
    a_new[:n] = a
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    return a_new

def _perturb_and_refine(a, n_perturb, deadline, min_time_remaining=30.0):
    """Perturb random coordinates and refine."""
    n = len(a)
    a_pert = a.copy()
    rng = np.random.default_rng(12345 + n_perturb)
    idx = rng.choice(n, size=max(1, n // 10), replace=False)
    a_pert[idx] *= (1.0 + 0.1 * rng.standard_normal(len(idx)))
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s > 0:
        a_pert = a_pert / s
    return _active_set_kkt(a_pert, deadline, min_time_remaining)

_t_global = 0.0

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining:
    1. Squared parameterization L-BFGS (allows exact zeros)
    2. Active-set KKT refinement
    3. Peak equalization
    4. Multi-profile exploration (asymmetric profiles)
    5. Multi-n search
    6. Final polish with both parameterizations
    """
    global _t_global
    _t_global = time.monotonic()
    deadline = _t_global + max(1.0, float(budget_s) - 5.0)
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(16384, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n=16384: score={best_score:.10f}")
    
    n_main = len(best_seq)
    
    # === Phase 1: Squared param L-BFGS on parent ===
    print(f"\n[P1] Squared L-BFGS (n={n_main}, t={time.monotonic()-_t_global:.0f}s)")
    p1_end = _t_global + min(400, (budget_s - 5) * 0.25)
    x_init = np.sqrt(best_seq)
    x_init[np.random.default_rng(42).integers(0, n_main)] += 0.1  # break symmetry
    a1, sc1 = _run_lbfgs_sq(x_init, p1_end, min_time_remaining=200, verbose=True)
    if sc1 < best_score:
