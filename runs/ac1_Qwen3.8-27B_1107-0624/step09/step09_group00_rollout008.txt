"""
Optimized active-set gradient descent with Barzilai-Borwein steps,
spectral initialization, multi-resolution refinement, and targeted ILS
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Key improvements over previous approach:
- Barzilai-Borwein adaptive step sizes for faster convergence
- Spectral (random-phase) initialization for diverse starts
- Multi-resolution coarse-to-fine optimization
- KKT-residual-driven active set management
- Targeted pairwise mass-transfer perturbations in ILS
- Power-mean smoothing as alternative to log-sum-exp

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _active_set_weights(c, tol_frac):
    """Compute weights for active set with inverse-gap weighting."""
    c_max = float(c.max())
    if c_max < 1e-30:
        w = np.ones_like(c) / len(c)
        return w, c_max
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if not active.any():
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
    eps = 1e-14 * c_max
    w = np.zeros_like(c)
    w[active] = 1.0 / (c_max - c[active] + eps)
    w = w / w.sum()
    return w, c_max


def _convolve_with_weights(w, a):
    """Compute conv(w, a) and extract the gradient-relevant portion."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return h


def _gradient_smoothed(a, beta, n):
    """Gradient of 2n * LSE_beta(conv(a,a)) with respect to a."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    h = _convolve_with_weights(w, a)
    g = 2.0 * n * h
    g = g - np.mean(g)
    return g


def _objective_smoothed(a, beta, n):
    """Smoothed objective: 2n * LSE_beta(conv(a,a))."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _power_mean_obj(a, p, n):
    """Objective using p-th power mean of convolution."""
    c = _full_convolve(a, a)
    c = np.maximum(c, 1e-30)
    pm = (np.mean(c ** p)) ** (1.0 / p)
    return 2.0 * n * pm


def _bb_step(y, s, mu_prev):
    """Barzilai-Borwein step size computation."""
    sy = float(np.dot(s, y))
    ss = float(np.dot(s, s))
    if ss < 1e-30:
        return mu_prev
    if sy < 1e-15:
        return mu_prev * 0.5
    gamma = ss / sy
    if gamma < 1e-12:
        return 1e-12
    if gamma > 10.0:
        return 10.0
    return gamma


def _nesterov_bb_polish(a_init, deadline, min_remaining=10.0,
                        tol_start=0.01, tol_end=0.00001,
                        lr_init=0.03, momentum_init=0.7):
    """
    Active-set gradient descent with Nesterov momentum and BB step sizes.
    """
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score_normalized(a)
    v = np.zeros(n)
    momentum = momentum_init
    lr = lr_init
    tol_frac = tol_start
    no_improve = 0
    max_stall = 200
    prev_grad = None
    prev_a = None
    prev_lr = lr

    while time.monotonic() < deadline - min_remaining:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        w, _ = _active_set_weights(c, tol_frac)
        h = _convolve_with_weights(w, a)
        grad = 2.0 * n * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # BB step size update
        if prev_grad is not None and prev_a is not None:
            y = grad - prev_grad
            s = a - prev_a
            lr = _bb_step(y, s, lr)
            lr = np.clip(lr, 1e-8, 1.0)

        prev_grad = grad.copy()
        prev_a = a.copy()

        # Nesterov look-ahead
        a_look = a + momentum * v
        c_look = _full_convolve(a_look, a_look)
        w_look, _ = _active_set_weights(c_look, tol_frac)
        h_look = _convolve_with_weights(w_look, a_look)
        grad_look = 2.0 * n * h_look
        grad_look = grad_look - np.mean(grad_look)

        # Update velocity
        v = momentum * v - lr * grad_look
        a_new = _project_simplex(a + v)

        # Line search
        alpha = 1.0
        improved = False
        c_cur = _full_convolve(a, a)
        sc_cur = 2.0 * n * float(c_cur.max())

        for _ in range(30):
            a_try = _project_simplex(a + alpha * v)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < sc_cur - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            v = np.zeros(n)
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_end, tol_frac * 0.6)
                lr *= 0.5
                momentum = max(0.3, momentum * 0.9)
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                momentum = momentum_init
                tol_frac = max(tol_end, tol_frac * 0.3)
                lr = lr_init
                no_improve = 0
                max_stall = int(max_stall * 1.3)
        else:
            if tol_frac > tol_end:
                tol_frac = max(tol_end, tol_frac * 0.97)
            momentum = min(0.95, momentum * 1.002)

    return best_a, _score_normalized(best_a)


def _lbfgs_smoothed(a_init, deadline, min_remaining=30.0):
    """L-BFGS on log-softmax parameterization with adaptive beta."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < 1e-18:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015,
                                  0.007, 0.003, 0.0015, 0.0007, 0.0003,
                                  0.0001, 0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score_normalized(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(50, int(remaining / (est_time * 20))))

        def func(z_inner):
            a_inner = _softmax(z_inner)
            return _objective_smoothed(a_inner, beta, n)

        def grad(z_inner):
            a_inner = _softmax(z_inner)
            g_a = _gradient_smoothed(a_inner, beta, n)
            return a_inner * g_a

        try:
            result = minimize(func, z, jac=grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-15, 'maxcor': 200, 'maxls': 50})
            a_new = _softmax(result.x)
            sc_new = _score_normalized(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _spectral_init(n, rng):
    """Generate a non-negative sequence with random phase spectrum."""
    freqs = np.fft.rfftfreq(n)
    phases = rng.uniform(0, 2 * np.pi, len(freqs))
    spectrum = np.exp(1j * phases)
    # Add some amplitude variation
    amps = 1.0 + 0.3 * rng.standard_normal(len(freqs))
    amps = np.maximum(amps, 0.1)
    spectrum = amps * np.exp(1j * phases)
    a = np.fft.irfft(spectrum, n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _structured_init(n, kind="chirp", rng=None):
    """Generate structured initial sequences."""
    if kind == "chirp":
        t = np.arange(n)
        phase = 2 * np.pi * t * (t + 1) / (2 * n)
        a = np.abs(np.exp(1j * phase)) + 0.1
        return a / a.sum()
    elif kind == "barker_like":
        if rng is None:
            rng = np.random.default_rng(0)
        a = rng.random(n)
        # Apply element-wise tanh for some structure
        a = np.tanh(2 * a - 1)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "two_hump":
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.3) ** 2) + np.exp(-20 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "comb":
        a = np.ones(n)
        a[::2] = 0.5
        return a / a.sum()
    else:
        return np.ones(n) / n


def _downsample(a, factor):
    """Downsample by averaging blocks."""
    n = len(a)
    new_n = n // factor
    a_padded = a[:new_n * factor]
    a_blocks = a_padded.reshape(new_n, factor)
    a_coarse = a_blocks.mean(axis=1)
    return a_coarse / a_coarse.sum()


def _upsample(a, factor):
    """Upsample by repeating/interpolating."""
    n = len(a)
    new_n = n * factor
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _targeted_ils(a_init, deadline, rng, min_remaining=30.0):
    """
    Targeted ILS with mass-transfer perturbations focused on
    high-convolution-contribution coordinates.
    """
    n = len(a_init)
    current_a = _normalize(a_init.copy())
    current_sc = _score_normalized(current_a)
    best_a = current_a.copy()
    best_sc = current_sc

    attempt = 0
    while time.monotonic() + min_remaining < deadline:
        # Identify "important" coordinates: those with high a_j or high contribution to peak
        c = _full_convolve(current_a, current_a)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))

        # Contribution of a_j to the peak: a_j * a[peak_idx - j] (if valid)
        contrib = np.zeros(n)
        for j in range(n):
            k = peak_idx - j
            if 0 <= k < n:
                contrib[j] = current_a[j] * current_a[k]
        contrib_norm = contrib / (contrib.sum() + 1e-30)

        # Choose perturbation type
        r = rng.random()
        if r < 0.3:
            # Mass transfer between high-contribution coordinates
            n_transfer = max(5, n // 500)
            # Weight by contribution
            weights = contrib_norm + 1e-10
            weights = weights / weights.sum()
            idx_src = rng.choice(n, size=n_transfer, replace=False, p=weights)
            idx_dst = rng.choice(n, size=n_transfer, replace=False, p=weights)
            a_perturbed = current_a.copy()
            for i in range(n_transfer):
                amt = a_perturbed[idx_src[i]] * rng.uniform(0.01, 0.1)
                a_perturbed[idx_src[i]] -= amt
                a_perturbed[idx_dst[i]] += amt
            a_perturbed = np.maximum(a_perturbed, 0.0)
            a_perturbed = _normalize(a_perturbed)
        elif r < 0.5:
            # Global small noise
            sigma = 0.003
            z = np.log(np.maximum(current_a, 1e-30))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_perturbed = _softmax(z)
        elif r < 0.7:
            # Block swap
            block_size = max(32, n // 64)
            start1 = int(rng.integers(0, n - block_size))
            start2 = int(rng.integers(0, n - block_size))
            if start1 == start2:
                start2 = (start1 + block_size) % (n - block_size)
            a_perturbed = current_a.copy()
            a_perturbed[start1:start1 + block_size], a_perturbed[start2:start2 + block_size] = \
                a_perturbed[start2:start2 + block_size].copy(), a_perturbed[start1:start1 + block_size].copy()
            a_perturbed = _normalize(a_perturbed)
        elif r < 0.85:
            # Sparse large perturbation
            n_perturb = max(20, n // 100)
            idx = rng.choice(n, size=n_perturb, replace=False)
            a_perturbed = current_a.copy()
            factors = np.exp(rng.standard_normal(n_perturb) * 0.1)
            a_perturbed[idx] *= factors
            a_perturbed = _normalize(a_perturbed)
        else:
            # Mirror a segment
            seg_len = max(16, n // 32)
            start = int(rng.integers(0, n - seg_len))
            a_perturbed = current_a.copy()
            a_perturbed[start:start + seg_len] = a_perturbed[start:start + seg_len][::-1]
            z = np.log(np.maximum(a_perturbed, 1e-30))
            z = z + 0.002 * rng.standard_normal(n)
            z = z - z.mean()
            a_perturbed = _softmax(z)

        # Quick local optimization
        opt_deadline = min(deadline - min_remaining, time.monotonic() + 5)
        a_opt, sc_opt = _nesterov_bb_polish(
            a_perturbed, opt_deadline, min_remaining=1,
            tol_start=0.005, tol_end=0.0005,
            lr_init=0.02, momentum_init=0.7)

        if sc_opt < best_sc - 1e-14:
            best_sc = sc_opt
            best_a = a_opt.copy()
            current_a = a_opt.copy()
        elif sc_opt < current_sc - 1e-14:
            current_a = a_opt.copy()

        attempt += 1
        if attempt % 50 == 0:
            print(f"    ILS iter {attempt}: best={best_sc:.10f}, t={time.monotonic()-t_start:.0f}s")

    return best_a, best_sc

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining spectral initialization, L-BFGS with
    log-sum-exp annealing, Barzilai-Borwein Nesterov active-set gradient,
    multi-resolution refinement, and targeted ILS.
    """
    global t_start
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = np.ones(n_parent) / n_parent
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    n_main = n_parent if n_parent > 0 else 16384

    # === Phase 1: Diverse initialization + quick optimize (180s) ===
    print(f"\n[P1] Diverse init + quick opt (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = t_start + 180

    starts = [('parent', parent.copy())]
    starts.append(('spectral1', _spectral_init(n_main, rng)))
    starts.append(('spectral2', _spectral_init(n_main, rng)))
    starts.append(('chirp', _structured_init(n_main, 'chirp')))
    starts.append(('two_hump', _structured_init(n_main, 'two_hump')))
    starts.append(('comb', _structured_init(n_main, 'comb')))

    for name, a_init in starts:
        if time.monotonic() > p1_deadline - 20:
            break
        a_opt = _quick_optimize(a_init, min(p1_deadline, time.monotonic() + 25),
                                min_remaining=5)
        sc_opt = _score(a_opt)
        print(f"  {name}: score={sc_opt:.10f}")
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [{name}]")

    # === Phase 2: Intensive L-BFGS on best (350s) ===
    print(f"\n[P2] Intensive L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        lbfgs_deadline = min(deadline - 600, time.monotonic() + 350)
        a_main = _lbfgs_smoothed(best_seq, lbfgs_deadline, min_remaining=30)
        sc = _score(a_main)
        if sc < best_score:
            best_seq = a_main.copy()
            best_score = sc
            print(f"  NEW BEST: {best_score:.10f} [lbfgs]")

    # === Phase 3: Nesterov BB polish (250s) ===
    print(f"\n[P3] Nesterov BB polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        pol_deadline = min(deadline - 400, time.monotonic() + 250)
        a_pol, sc_pol = _nesterov_bb_polish(
            best_seq, pol_deadline, min_remaining=10,
            tol_start=0.002, tol_end=0.00001,
            lr_init=0.015, momentum_init=0.75)
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  NEW BEST: {best_score:.10f} [nesterov_bb]")

    # === Phase 4: Targeted ILS (300s) ===
    print(f"\n[P4] Targeted ILS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        ils_deadline = min(deadline - 300, time.monotonic() + 300)
        a_ils, sc_ils = _targeted_ils(best_seq, ils_deadline, rng, min_remaining=30)
        if sc_ils < best_score:
            best_seq = a_ils.copy()
            best_score = sc_ils
            print(f"  NEW BEST: {best_score:.10f} [ils]")

    # === Phase 5: Multi-resolution refinement (200s) ===
    print(f"\n[P5] Multi-resolution (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        mr_deadline = min(deadline - 200, time.monotonic() + 200)
        # Coarse optimization
        for factor in [4, 2]:
            if time.monotonic() > mr_deadline - 30:
                break
            a_coarse = _downsample(best_seq, factor)
            n_coarse = len(a_coarse)
            print(f"  Coarse n={n_coarse} (factor={factor}), score={_score(a_coarse):.10f}")
            a_coarse_opt, _ = _nesterov_bb_polish(
                a_coarse, min(mr_deadline, time.monotonic() + 40),
                min_remaining=5, tol_start=0.005, tol_end=0.0005,
                lr_init=0.03, momentum_init=0.7)
            a_up = _upsample(a_coarse_opt, factor)
            sc_up = _score(a_up)
            if sc_up < best_score:
                # Refine at full resolution
                a_refined = a_up.copy()
                # Blend with best
                a_blend = 0.5 * a_refined + 0.5 * best_seq
                a_blend = _normalize(a_blend)
                a_refined2, sc_ref = _nesterov_bb_polish(
                    a_blend, min(mr_deadline, time.monotonic() + 30),
                    min_remaining=5, tol_start=0.003, tol_end=0.0003,
                    lr_init=0.015, momentum_init=0.7)
                if sc_ref < best_score:
                    best_seq = a_refined2.copy()
                    best_score = sc_ref
                    print(f"  NEW BEST: {best_score:.10f} [multi_res factor={factor}]")

    # === Phase 6: Second round of intensive optimization (200s) ===
    print(f"\n[P6] Second round (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        r2_deadline = min(deadline - 150, time.monotonic() + 200)
        # L-BFGS
        a_r2 = _lbfgs_smoothed(best_seq, r2_deadline, min_remaining=30)
        sc_r2 = _score(a_r2)
        if sc_r2 < best_score:
            best_seq = a_r2.copy()
            best_score = sc_r2
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_r2]")

        # Nesterov polish
        if time.monotonic() < r2_deadline - 5:
            a_r2b, sc_r2b = _nesterov_bb_polish(
                best_seq, r2_deadline, min_remaining=5,
                tol_start=0.001, tol_end=0.000005,
                lr_init=0.01, momentum_init=0.8)
            if sc_r2b < best_score:
                best_seq = a_r2b.copy()
                best_score = sc_r2b
                print(f"  NEW BEST: {best_score:.10f} [nesterov_r2]")

    # === Phase 7: Final ILS round (100s) ===
    print(f"\n[P7] Final ILS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        rng2 = np.random.default_rng(int(seed) + 7777)
        ils2_deadline = min(deadline - 60, time.monotonic() + 100)
        a_ils2, sc_ils2 = _targeted_ils(best_seq, ils2_deadline, rng2, min_remaining=20)
        if sc_ils2 < best_score:
            best_seq = a_ils2.copy()
            best_score = sc_ils2
            print(f"  NEW BEST: {best_score:.10f} [ils2]")

    # === Phase 8: Ultra-fine final polish (50s) ===
    print(f"\n[P8] Ultra-fine polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _nesterov_bb_polish(
            best_seq, deadline - 3, min_remaining=2,
            tol_start=0.0005, tol_end=0.000001,
            lr_init=0.005, momentum_init=0.85)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_ultra]")

    # Final
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
