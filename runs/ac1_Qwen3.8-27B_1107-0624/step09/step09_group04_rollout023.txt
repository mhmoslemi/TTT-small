`α = (M_max - M_target) / (2 * B_max @ grad)`

This is getting complicated. Let me just use a robust line search.

2. Improve the active set equalization:

3. Add a "mass redistribution" step: identify positions that contribute a lot to the top peaks but little to other peaks, and move mass from them to positions that contribute to non-peak areas.

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Log-softmax L-BFGS for smooth optimization
2. Active-set peak equalization with adaptive step size
3. Direct simplex-projected gradient steps
4. Local search with random perturbations
5. Multi-n exploration
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003) * beta_scale
    
    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625,
        0.03, 0.015, 0.0075, 0.004,
        0.002, 0.001, 0.0005, 0.0002, 0.0001,
        0.00005, 0.00002, 0.00001, 0.000005
    ])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # For very small beta, use more iterations
        if beta < base_beta * 0.005:
            max_iter = min(8000, max(100, int(remaining / 0.008)))
        elif beta < base_beta * 0.05:
            max_iter = min(5000, max(50, int(remaining / 0.006)))
        else:
            max_iter = min(3000, max(30, int(remaining / 0.005)))
        
        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

def _active_set_equalize(a_init, deadline, min_time_remaining=10.0):
    """Active-set peak equalization with adaptive step size and direct simplex projection."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    K = 5  # Start with 5 active peaks
    K_max = min(100, n)
    no_improve = 0
    lr = 0.03
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            # Grow K slowly
            if K < K_max and no_improve == 0:
                K = min(K_max, K + 2)
        
        # Get top K indices with weights (higher weight for higher peaks)
        sorted_idx = np.argsort(c)[::-1]
        top_k_idx = sorted_idx[:K]
        top_k_vals = c[top_k_idx]
        
        # Weight by peak value (more weight on higher peaks)
        w_S = np.zeros(2 * n - 1)
        # Use softmax weighting on the top-K peaks
        w_top = top_k_vals - top_k_vals.max()
        w_top = np.exp(w_top / (max_c * 0.01 + 1e-15))
        w_top = w_top / w_top.sum()
        w_S[top_k_idx] = w_top
        
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Line search with backtracking
        alpha = lr / (gnorm * np.linalg.norm(a) + 1e-15)
        improved = False
        best_alpha = 0
        for _ in range(60):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc * (1 - 1e-10):
                a = a_try
                best_alpha = alpha
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            lr *= 0.8
            if no_improve % 30 == 0:
                K = max(3, K // 2)
            if no_improve >= 300:
                a = best_a.copy()
                lr = 0.03
                K = 5
                no_improve = 0
                break
        else:
            lr = min(0.08, lr * 1.05)
    
    return best_a, best_sc

def _direct_simplex_descent(a_init, deadline, min_time_remaining=10.0):
    """Direct projected gradient descent on the simplex."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    lr = 0.01
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        # Subgradient of max(c)
        # Use the argmax peak
        k_max = int(np.argmax(c))
        # dc[k_max]/da[i] = 2*a[k_max - i]
        grad = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        # Center for simplex
        grad = grad - np.mean(grad)
        
        # Projected gradient step
        a_try = a - lr * grad
        a_try = _proj_simplex(a_try)
        
        c_try = _full_convolve(a_try, a_try)
        sc_try = 2.0 * n * float(c_try.max())
        
        if sc_try < current_sc:
            a = a_try
            lr = min(0.05, lr * 1.02)
        else:
            no_improve += 1
            lr *= 0.9
            if no_improve > 200:
                break
    
    return best_a, best_sc

def _local_search(a_init, deadline, min_time_remaining=10.0, rng=None):
    """Local search with random perturbations."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    no_improve = 0
    step_size = 0.02
    
    while time.monotonic() + min_time_remaining < deadline:
        if no_improve > 500:
            a = best_a.copy()
            no_improve = 0
            step_size = 0.02
        
        # Random perturbation
        pert = rng.standard_normal(n) * step_size
        a_try = a + pert
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            no_improve += 1
            continue
        a_try = a_try / s_try
        
        c_try = _full_convolve(a_try, a_try)
        sc_try = 2.0 * n * float(c_try.max())
        
        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            no_improve = 0
            step_size = min(0.05, step_size * 1.01)
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                step_size *= 0.95
    
    return best_a, best_sc

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.arange(n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * 7.0 * t / n) + 0.2 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "fejer":
        # Fejér kernel-like: (sin(x)/x)^2
        t = np.linspace(-1, 1, n)
        x = np.pi * t
        a = np.sinc(t) ** 2  # sin(pi*t)/(pi*t) squared
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

Hmm, wait. Let me reconsider the approach. The gap between 1.505661 and 1.5027 is about 0.003. This is a small gap, so I need to be very precise.

Let me think about what's limiting the previous approach:

1. **The log-softmax parameterization**: For large n (16384), the log-softmax can be ill-conditioned. The variable z can have a large range, and the mapping from z to a can be very sensitive. This might cause L-BFGS to struggle.

2. **The active set size**: Using too few or too many active peaks can be suboptimal. The optimal active set size depends on the structure of the solution.

3. **The step size in active set equalization**: A fixed or slowly-decaying step size might not be optimal.

Let me think about a different approach: **What if I directly optimize the active set equalization condition?**

At the optimum, if the top-K peaks are all equal to M, then:
- `c[k_m] = M` for all `m = 1, ..., K`
- The gradient of `c[k_m]` with respect to `a` should be "aligned" with the constraint (sum a = 1, a >= 0)

The gradient of `c[k_m]` is `2 * a[k_m - :]` (shifted version of a).

For the optimum, we need:
`a[k_m - :] - λ * 1 = μ_m * a` (for some Lagrange multipliers)

Wait, that's not quite right. The KKT conditions for:
minimize M subject to c[k_m] <= M for all m, sum a = 1, a >= 0

At the optimum (assuming interior solution, a > 0):
- There exist multipliers λ_m >= 0 for each peak constraint
- There exists a multiplier ν for the sum constraint
- For each i: 2 * sum_m λ_m * a[k_m - i] - ν = 0 (where terms with k_m - i out of range are 0)

This means: `sum_m λ_m * a[k_m - i] = ν/2` for all i where a[i] > 0.

In other words, the weighted sum of shifted copies of `a` should be constant (on the support of `a`).

This is a very specific condition! It means that the "autocorrelation" of `a` with the "dual" sequence `λ` should be flat.

This gives me an idea for a new algorithm:
1. Initialize `a` and `λ`
2. Update `a` to minimize the maximum peak (given `λ`)
3. Update `λ` to satisfy the flatness condition
4. Repeat

For step 2: Given `λ`, the objective is `sum_m λ_m * c[k_m]` (a weighted sum of peaks). This is a quadratic function of `a`:
`sum_m λ_m * sum_j a[j] * a[k_m - j] = a^T (sum_m λ_m B_{k_m}) a`

where `B_k` is the matrix with 1s at positions (j, k-j).

Minimizing this quadratic subject to `sum a = 1`, `a >= 0` is a QP! We can solve it efficiently.

For step 3: Given `a`, we want to find `λ` such that `sum_m λ_m * a[k_m - i]` is as flat as possible. This is a least-squares problem.

This alternating approach might converge faster than the gradient-based methods!

However, solving the QP in step 2 for n=16384 might be expensive. The QP has n variables and n constraints (non-negativity) plus 1 equality constraint. With n=16384, this is a large QP.

But the QP has a special structure: the objective is `a^T Q a` where `Q = sum_m λ_m B_{k_m}`. The matrix `Q` is sparse (each `B_k` has at most n non-zero entries, so `Q` has at most K*n non-zero entries). For K=50 and n=16384, that's about 819K non-zeros, which is manageable.

Actually, I realize that solving this QP exactly at each iteration might be too expensive. Let me use a simpler approach:

**Power iteration / coordinate descent on the QP:**

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Log-softmax L-BFGS with adaptive beta for smooth optimization
2. Softmax-weighted active-set peak equalization with adaptive step size

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.002) * beta_scale

    # More fine-grained beta schedule for better convergence
    betas = base_beta * np.array([
        1.0, 0.5, 0.3, 0.15, 0.08,
        0.04, 0.02, 0.01, 0.005,
        0.0025, 0.00125, 0.000625,
        0.0003, 0.00015, 0.00008,
        0.00004, 0.00002, 0.00001, 0.000005
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Scale iterations based on beta and remaining time
        if beta < base_beta * 0.003:
            max_iter = min(10000, max(200, int(remaining / 0.006)))
        elif beta < base_beta * 0.03:
            max_iter = min(6000, max(100, int(remaining / 0.005)))
        else:
            max_iter = min(4000, max(50, int(remaining / 0.004)))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-16,
                    'maxcor': 120,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_equalize(a_init, deadline, min_time_remaining=10.0):
    """Active-set peak equalization with softmax weighting and adaptive step."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    K = 8  # Start with larger K
    K_max = min(200, n)
    no_improve = 0
    lr = 0.05

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            if K < K_max:
                K = min(K_max, K + 3)

        # Get top K indices with softmax weighting
        sorted_idx = np.argsort(c)[::-1]
        top_k_idx = sorted_idx[:K]
        top_k_vals = c[top_k_idx]

        # Softmax weighting: higher peaks get more weight
        # Use a temperature that adapts to the spread of top-k values
        spread = top_k_vals[0] - top_k_vals[-1]
        if spread < c_max * 1e-6:
            w_top = np.ones(K) / K
        else:
            temp = spread * 0.5
            w_top = np.exp((top_k_vals - top_k_vals[0]) / temp)
            w_top = w_top / w_top.sum()

        w_S = np.zeros(2 * n - 1)
        w_S[top_k_idx] = w_top

        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Line search: start with a reasonably large step
        alpha = min(0.5 / (gnorm + 1e-15), lr * 3.0)
        improved = False
        for _ in range(80):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc * (1 - 1e-12):
                a = a_try
                improved = True
                break
            alpha *= 0.7  # Less aggressive halving

        if not improved:
            no_improve += 1
            lr *= 0.85
            if no_improve % 40 == 0:
                K = max(4, K // 2)
            if no_improve >= 400:
                a = best_a.copy()
                lr = 0.05
                K = 8
                no_improve = 0
                break
        else:
            lr = min(0.12, lr * 1.08)

    return best_a, best_sc


def _flatten_peaks(a_init, deadline, min_time_remaining=10.0):
    """Targeted flattening: reduce the dominant peak by moving mass
    from positions that strongly contribute to it."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    max_iters = 500
    for it in range(max_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()

        # Find the top peak and second peak
        sorted_idx = np.argsort(c)[::-1]
        k1 = sorted_idx[0]
        k2 = sorted_idx[1]
        val1 = c[k1]
        val2 = c[k2]

        if val1 - val2 < val1 * 1e-5:
            # Peaks are already flat, stop
            break

        # The dominant peak k1: reduce contribution by reducing a at
        # positions that feed into k1
        # Contribution of a[i] to c[k1] is a[k1-i]
        contrib1 = np.zeros(n)
        for i in range(n):
            j = k1 - i
            if 0 <= j < n:
                contrib1[i] = a[j]

        # The second peak k2: we want to not increase it
        contrib2 = np.zeros(n)
        for i in range(n):
            j = k2 - i
            if 0 <= j < n:
                contrib2[i] = a[j]

        # Move mass from positions with high contrib1 and low contrib2
        # to positions with low contrib1 and high contrib2
        surplus = contrib1 - contrib2  # positive means position feeds peak1 more
        # Reduce a where surplus > 0, increase where surplus < 0
        # Scale by a to only move mass from positions that have mass
        direction = surplus * a
        direction = direction - direction.mean()  # zero-sum

        if np.abs(direction).max() < 1e-20:
            break

        # Try a small step
        for scale in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            a_try = a - scale * direction
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                break

    return best_a, best_sc


def _local_search(a_init, deadline, min_time_remaining=10.0, rng=None):
    """Local search with structured perturbations (low-freq + high-freq)."""
    if rng is None:
        rng = np.random.default_rng(42)

    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    no_improve = 0
    step_size = 0.015

    while time.monotonic() + min_time_remaining < deadline:
        if no_improve > 1000:
            a = best_a.copy()
            no_improve = 0
            step_size = 0.015

        # Structured perturbation: mix of low-freq and high-freq
        k_low = rng.integers(2, max(3, n // 100))
        k_high = rng.integers(max(n // 4, 4), n)
        t = np.arange(n)
        phase1 = rng.uniform(0, 2 * np.pi)
        phase2 = rng.uniform(0, 2 * np.pi)
        pert = (0.5 * np.sin(2 * np.pi * k_low * t / n + phase1) +
                0.3 * np.sin(2 * np.pi * k_high * t / n + phase2) +
                0.2 * rng.standard_normal(n))
        pert = pert - pert.mean()

        a_try = a + step_size * pert
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            no_improve += 1
            continue
        a_try = a_try / s_try

        c_try = _full_convolve(a_try, a_try)
        sc_try = 2.0 * n * float(c_try.max())

        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            no_improve = 0
            step_size = min(0.04, step_size * 1.02)
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                step_size *= 0.9

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "fejer":
        t = np.linspace(-1, 1, n)
        a = np.sinc(t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.arange(n)
        a = 1.0 + 0.4 * np.sin(2.0 * np.pi * 5.0 * t / n) + 0.2 * np.cos(2.0 * np.pi * 11.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _asymmetric_perturb(a, rng, strength=0.03):

    ramp = 0.4 * (t - 0.5)
    noise = 0.6 * rng.standard_normal(n)
    perturbation = ramp + noise
    perturbation = perturbation - perturbation.mean()
    z = np.log(np.maximum(a, 1e-30))
    z = z + strength * perturbation
    z = z - z.mean()
    return _softmax(z)

        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS warmup ===
    print(f"\n[P1] L-BFGS warmup (t={time.monotonic()-t_start:.1f}s)")
    a_p1 = _run_lbfgs(parent, deadline, min_time_remaining=500.0, beta_scale=1.0)
    sc_p1 = _score(a_p1)
    if sc_p1 < best_score:
        best_seq = a_p1.copy()
        best_score = sc_p1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_warmup]")

    # === Phase 2: Active-set equalization (long) ===
    print(f"\n[P2] Active-set equalization (t={time.monotonic()-t_start:.1f}s)")
    pol_deadline = min(deadline - 450, time.monotonic() + 350)
    a_eq, sc_eq = _active_set_equalize(best_seq, pol_deadline, min_time_remaining=50.0)
    if sc_eq < best_score:
        best_seq = a_eq.copy()
        best_score = sc_eq
        print(f"  NEW BEST: {best_score:.10f} [active_set_1]")

    # === Phase 3: Fine L-BFGS ===
    print(f"\n[P3] Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    a_fine = _run_lbfgs(best_seq, deadline, min_time_remaining=300.0, beta_scale=0.3)
    sc_fine = _score(a_fine)
    if sc_fine < best_score:
        best_seq = a_fine.copy()
        best_score = sc_fine
        print(f"  NEW BEST: {best_score:.10f} [fine_lbfgs]")

    # === Phase 4: Flatten peaks ===
    print(f"\n[P4] Flatten peaks (t={time.monotonic()-t_start:.1f}s)")
    fl_deadline = min(deadline - 250, time.monotonic() + 100)
    a_fl, sc_fl = _flatten_peaks(best_seq, fl_deadline, min_time_remaining=30.0)
    if sc_fl < best_score:
        best_seq = a_fl.copy()
        best_score = sc_fl
        print(f"  NEW BEST: {best_score:.10f} [flatten]")

    # === Phase 5: Active-set round 2 ===
    print(f"\n[P5] Active-set round 2 (t={time.monotonic()-t_start:.1f}s)")
    pol_deadline2 = min(deadline - 200, time.monotonic() + 150)
    a_eq2, sc_eq2 = _active_set_equalize(best_seq, pol_deadline2, min_time_remaining=30.0)
    if sc_eq2 < best_score:
        best_seq = a_eq2.copy()
        best_score = sc_eq2
        print(f"  NEW BEST: {best_score:.10f} [active_set_2]")

    # === Phase 6: Asymmetric perturbation + re-optimize ===
    print(f"\n[P6] Asymmetric exploration (t={time.monotonic()-t_start:.1f}s)")
    for trial in range(3):
        if time.monotonic() > deadline - 150:
            break
        strength = 0.02 * (trial + 1)
        a_asym = _asymmetric_perturb(best_seq, rng, strength=strength)
        a_opt = _run_lbfgs(a_asym, deadline, min_time_remaining=150.0, beta_scale=0.5)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (trial={trial}) [asym]")
        else:
            print(f"  Asym trial {trial}: score={sc_opt:.10f} (no improvement)")

    # === Phase 7: Local search ===
    print(f"\n[P7] Local search (t={time.monotonic()-t_start:.1f}s)")
    ls_deadline = min(deadline - 80, time.monotonic() + 60)
    a_ls, sc_ls = _local_search(best_seq, ls_deadline, min_time_remaining=30.0, rng=rng)
    if sc_ls < best_score:
        best_seq = a_ls.copy()
        best_score = sc_ls
        print(f"  NEW BEST: {best_score:.10f} [local_search]")

    # === Phase 8: Multi-n exploration ===
    print(f"\n[P8] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    current_n = len(best_seq)
    n_candidates = sorted(set([current_n, 24576, 32768]), reverse=True)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 50:
            break
        if n_try == current_n:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")
        if sc_init < best_score:
            a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=50.0, beta_scale=0.5)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n]")

    # === Phase 9: Final active-set polish ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _active_set_equalize(best_seq, deadline - 3, min_time_remaining=3.0)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish]")

    # === Phase 10: Ultra-fine L-BFGS ===
    print(f"\n[P10] Ultra-fine (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 3 - time.monotonic()
    if remaining > 5:
        a_ult = _run_lbfgs(best_seq, deadline, min_time_remaining=3.0, beta_scale=0.1)
        sc_ult = _score(a_ult)
        if sc_ult < best_score:
            best_seq = a_ult.copy()
            best_score = sc_ult
            print(f"  NEW BEST: {best_score:.10f} [ultra_fine]")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    return [float(x) for x in fallback]
