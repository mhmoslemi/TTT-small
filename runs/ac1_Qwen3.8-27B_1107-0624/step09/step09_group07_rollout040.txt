    c = _full_convolve(a, a)

def _grad_conv(a, w):
    """Gradient of sum_k w[k]*c_k w.r.t. a, where c = conv(a,a). Length n."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _lse_compute(a, beta, n):
    """Compute log-sum-exp smoothed value, weights, and gradient w.r.t. a."""
    c = _full_convolve(a, a)

def _hess_vec_lse(a, z, beta, v):
    """Approximate Hessian-vector product for LSE smoothed objective in a-space.
    Used for Newton step computation."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # First term: Hessian of weighted sum = grad_conv(v, w) (by our derivation)
    term1 = _grad_conv(v, w)
    # Second term: (1/beta) * sum_k (w_k - w_k^2) * (grad_c_k)(grad_c_k)^T
    # Only compute for significant weights
    sig_thresh = 0.01 / max(1, (2 * n - 1))
    sig_idx = np.where(w > sig_thresh)[0]
    term2 = np.zeros(n)
    for k in sig_idx:
        coeff = (w[k] - w[k] ** 2) / beta
        gk = _grad_conv(a, np.zeros(2 * n - 1) + np.eye(1, 1)[0, 0] * (lambda x: x)(np.zeros(2*n-1)))
        # Actually need grad of c_k w.r.t. a which is 2*S_k*a
        # (S_k a)_i = a[k-i] for valid i
        gk = np.zeros(n)
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            gk[i] = 2.0 * a[k - i]
        term2 += coeff * np.dot(gk, v) * gk
    return term1 + term2

        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        n_active = max(5, min(n_2nm1, int(n_2nm1 * active_frac)))
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_active]

        active_vals = c[active_idx]
        w = np.zeros(n_2nm1)
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w[active_idx] = (active_vals - active_vals.min()) / spread + 0.05
        else:
            w[active_idx] = 1.0 / n_active
        w = w / w.sum()

        g = _grad_conv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break

        if prev_a is not None and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g_centered - prev_g
            s_norm = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm > 1e-18 and abs(sy) > 1e-18:
                lr_bb = s_norm / abs(sy)
                lr = np.clip(lr_bb, lr_min, lr_max)

        velocity = momentum * velocity - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()

        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            for ls in range(30):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-15:
                    improved = True
                    break
                velocity = momentum * velocity * 0.5 - lr * 0.5 * g_centered
                a_try = a + velocity
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    velocity = np.zeros(n)
                    a_try = a
                    break
                a_try = a_try / s_try
                lr *= 0.7

        if improved:
            prev_a = a.copy()
            prev_g = g_centered.copy()
            a = a_try
            lr = min(lr_max, lr * 1.02)
            active_frac = max(0.003, active_frac * 0.98)
        else:
            velocity = np.zeros(n)
            lr = max(lr_min, lr * 0.6)
            active_frac = min(0.1, active_frac * 1.2)

        if stall > max_stall:
            active_frac = min(0.1, active_frac * 2.0)
            lr = max(lr_min, lr * 0.3)
            momentum = min(0.9, momentum + 0.05)
            stall = 0

        if iteration % 300 == 0 and time.monotonic() - last_lbfgs_time > 180:
            if time.monotonic() + min_remaining + 120 < deadline:
                lbfgs_deadline = time.monotonic() + 100
                a_tmp = _run_lse_lbfgs(a, [max(c_max * 0.005, 1e-6)], lbfgs_deadline, 5)
                sc_tmp = _score(a_tmp)
                if sc_tmp < best_sc:
                    best_a = a_tmp.copy()
                    best_sc = sc_tmp
                    a = a_tmp.copy()
                    velocity = np.zeros(n)
                    prev_a = None
                    prev_g = None
                    last_lbfgs_time = time.monotonic()

    return best_a, best_sc
