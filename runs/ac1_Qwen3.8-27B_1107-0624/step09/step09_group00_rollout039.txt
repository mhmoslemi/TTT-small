def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _top_m_sum(c, m):
    """Sum of the top m values in c."""
    if m >= len(c):
        return c.sum()
    # Use partial sort for efficiency
    top = np.argpartition(c, -m)[-m:]
    return c[top].sum()

def _top_m_gradient(c, a, m):
    """Gradient of sum of top m values of c = a*a with respect to a."""
    if m >= len(c):
        w = np.ones_like(c) / len(c)
    else:
        threshold_idx = len(c) - m
        threshold = np.partition(c, threshold_idx)[threshold_idx]
        w = (c >= threshold).astype(float)
        n_active = w.sum()
        if n_active == 0:
            w[int(np.argmax(c))] = 1.0
            n_active = 1
        else:
            w = w / n_active

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    n = len(a)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h

Let me also add a "frequency domain" perspective: the DFT of a non-negative sequence has a specific structure. The magnitude of the DFT is related to the square root of the autocorrelation's DFT (by the Wiener-Khinchin theorem). So minimizing the peak of the autocorrelation is equivalent to making the DFT magnitude as flat as possible.

For a non-negative sequence of length n, the DFT at frequency k is:
$\hat{a}(k) = \sum_{j=0}^{n-1} a_j e^{-2\pi i jk/n}$

The magnitude is $|\hat{a}(k)| = \sqrt{(\sum a_j \cos(2\pi jk/n))^2 + (\sum a_j \sin(2\pi jk/n))^2}$

For the autocorrelation to be flat, we need $|\hat{a}(k)|$ to be constant for all k. This means:

- Hysteresis-based active set to avoid oscillation
"""
import numpy as np
import time
from scipy.optimize import minimize

_t_start = 0.0


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_from_conv(a, c):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _top_m_sum(c, m):
    if m >= len(c):
        return float(c.sum())
    idx = np.argpartition(c, -m)[-m:]
    return float(c[idx].sum())


def _conv_gradient(w, a):
    """Compute grad_j = 2 * sum_k w[k] * a[k-j] (for valid k-j)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _top_m_weighted_grad(c, a, m, power=1.0):
    """Gradient of sum of top-m c[k]^power / power."""
    if m >= len(c):
        w = np.ones_like(c)
    else:
        threshold_idx = len(c) - m
        threshold = np.partition(c, threshold_idx)[threshold_idx]
        w = (c >= threshold).astype(float)
        n_active = w.sum()
        if n_active == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            if power > 1.0:
                w = w * (c ** (power - 1.0))
            w = w / w.sum()
    return _conv_gradient(w, a)


def _make_profile(n, kind="hann"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos4":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) ** 3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _top_m_optimize(a_init, deadline, m_start=8, m_end=2, min_time=20.0,
                    momentum=0.85, lr_init=0.03, power=1.0):
    """Optimize using sum-of-top-m objective with momentum and BB step size."""
    global _t_start
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    v = np.zeros(n)
    lr = lr_init
    m = m_start
    prev_c_max = None
    prev_gnorm = None
    no_improve = 0
    max_stall = 80

    while time.monotonic() < deadline - min_time:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt m: oscillate between m_start and m_end
        if no_improve > max_stall:
            m = m_end if m > m_end else m_start
            no_improve = 0

        # Compute gradient
        grad = _top_m_weighted_grad(c, a, m, power=power)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # BB step size update
        if prev_c_max is not None and prev_gnorm is not None:
            delta_f = prev_c_max - c_max
            delta_g = gnorm - prev_gnorm
            if abs(delta_g) > 1e-12:
                bb_lr = delta_f / max(abs(delta_g), 1e-12)
                bb_lr = np.clip(bb_lr, 0.001, 5.0)
                lr = 0.5 * lr + 0.5 * bb_lr
        prev_c_max = c_max
        prev_gnorm = gnorm

        # Momentum
        v = momentum * v - lr * (grad / (gnorm + 1e-15))
        vmax = np.linalg.norm(v)
        if vmax > 0.5:
            v = v * 0.5 / vmax

        # Line search
        alpha = 1.0
        improved = False
        for _ in range(25):
            a_try = a + alpha * v
            a_try = _normalize(a_try)
            c_try = _full_convolve(a_try, a_try)
            cm = float(c_try.max())
            if cm < c_max - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            v = np.zeros(n)
            lr = max(0.001, lr * 0.7)
            if no_improve > max_stall * 2:
                a = best_a.copy()
                v = np.zeros(n)
                m = max(2, m - 2)
                lr = lr_init
                no_improve = 0

    return best_a, _score(best_a)


def _water_fill_optimize(a_init, deadline, min_time=15.0):
    """Water-filling: equalize top peaks by targeted gradient steps."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    while time.monotonic() < deadline - min_time:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()

        # Find top peaks
        sorted_idx = np.argsort(c)[::-1]
        top_10 = sorted_idx[:10]
        top_vals = c[top_10]
        spread = top_vals[0] - top_vals[min(4, len(top_vals) - 1)]

        if spread < c_max * 1e-7:
            # Well equalized, use very small gradient step
            break

        # Gradient of the single worst peak
        w = np.zeros_like(c)
        w[sorted_idx[0]] = 1.0
        grad = _conv_gradient(w, a)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: try to reduce the max by spread
        alpha = spread / (gnorm * gnorm + 1e-15)
        alpha = min(alpha, 0.05)

        improved = False
        for trial in range(5):
            a_try = _normalize(a - alpha * grad)
            c_try = _full_convolve(a_try, a_try)
            if float(c_try.max()) < c_max - 1e-16:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # Try a multi-peak gradient
            w_multi = np.zeros_like(c)
            n_active = min(5, len(c))
            for idx in sorted_idx[:n_active]:
                w_multi[idx] = 1.0 / n_active
            grad_multi = _conv_gradient(w_multi, a)
            grad_multi = grad_multi - np.mean(grad_multi)
            gnorm_m = np.linalg.norm(grad_multi)
            if gnorm_m < 1e-15:
                break
            alpha_m = spread / (gnorm_m * gnorm_m + 1e-15)
            alpha_m = min(alpha_m, 0.03)
            a_try = _normalize(a - alpha_m * grad_multi)
            c_try = _full_convolve(a_try, a_try)
            if float(c_try.max()) < c_max - 1e-16:
                a = a_try
                improved = True

        if not improved:
            break

    return best_a, _score(best_a)


def _coord_descent_peaks(a_init, deadline, min_time=20.0, max_rounds=500):
    """Coordinate descent: optimize elements that contribute to worst peaks."""
    global _t_start
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(777)

    round_num = 0
    while time.monotonic() < deadline - min_time and round_num < max_rounds:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()

        # Find worst peak position
        k_star = int(np.argmax(c))

        # Find top contributing elements to this peak
        valid_i = np.array([i for i in range(n) if 0 <= k_star - i < n])
        if len(valid_i) == 0:
            break
        contribs = a[valid_i] * a[k_star - valid_i]
        order = np.argsort(contribs)[::-1]
        top_count = min(15, len(valid_i))
        top_elems = valid_i[order[:top_count]]

        improved_this_round = False
        for idx in top_elems:
            if time.monotonic() > deadline - min_time:
                break
            # Try a few multiplicative factors
            best_factor = 1.0
            best_cm = c_max
            factors = [0.85, 0.9, 0.95, 1.05, 1.1]
            for f in factors:
                a_try = a.copy()
                a_try[idx] *= f
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try /= s
                c_try = _full_convolve(a_try, a_try)
                cm = float(c_try.max())
                if cm < best_cm - 1e-16:
                    best_cm = cm
                    best_factor = f
            if best_factor != 1.0:
                a[idx] *= best_factor
                a = _normalize(a)
                improved_this_round = True

        # Also try some random coordinates
        for _ in range(20):
            if time.monotonic() > deadline - min_time:
                break
            idx = int(rng.integers(0, n))
            best_factor = 1.0
            best_cm = float(_full_convolve(a, a).max())
            for f in [0.9, 0.95, 1.05, 1.1]:
                a_try = a.copy()
                a_try[idx] *= f
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try /= s
                c_try = _full_convolve(a_try, a_try)
                cm = float(c_try.max())
                if cm < best_cm - 1e-16:
                    best_cm = cm
                    best_factor = f
            if best_factor != 1.0:
                a[idx] *= best_factor
                a = _normalize(a)
                improved_this_round = True

        if not improved_this_round:
            # Try a second peak
            k_star2 = int(np.argmax(np.where(np.arange(len(c)) == k_star, 0, c)))
            if k_star2 == k_star:
                break
            valid_i2 = np.array([i for i in range(n) if 0 <= k_star2 - i < n])
            if len(valid_i2) > 0:
                contribs2 = a[valid_i2] * a[k_star2 - valid_i2]
                order2 = np.argsort(contribs2)[::-1]
                for idx in valid_i2[order2[:5]]:
                    best_factor = 1.0
                    best_cm = c_max
                    for f in [0.9, 0.95, 1.05]:
                        a_try = a.copy()
                        a_try[idx] *= f
                        s = a_try.sum()
                        if s < 1e-10:
                            continue
                        a_try /= s
                        c_try = _full_convolve(a_try, a_try)
                        cm = float(c_try.max())
                        if cm < best_cm - 1e-16:
                            best_cm = cm
                            best_factor = f
                    if best_factor != 1.0:
                        a[idx] *= best_factor
                        a = _normalize(a)
                        improved_this_round = True

        if not improved_this_round:
            break

        round_num += 1

    return best_a, _score(best_a)


def _freq_domain_perturb(a, rng, strength=0.02):
    """Perturb in the frequency domain."""
    n = len(a)
    N = _next_pow2(2 * n)
    A = np.fft.rfft(a, N)
    # Add random phase perturbation (preserves magnitude approximately)
    phase_pert = strength * rng.standard_normal(len(A))
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _freq_magnitude_perturb(a, rng, strength=0.01):
    """Perturb the magnitude of the DFT."""
    n = len(a)
    N = _next_pow2(2 * n)
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_pert = mag * (1.0 + strength * rng.standard_normal(len(mag)))
    mag_pert = np.maximum(mag_pert, 0.0)
    A_new = mag_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _stochastic_ils(a_init, deadline, rng, min_time=50.0):
    """Stochastic ILS with diverse perturbations including frequency-domain."""
    n = len(a_init)
    current_a = _normalize(a_init.copy())
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc

    attempt = 0
    while time.monotonic() < deadline - min_time:
        # Choose perturbation strategy
        strat = attempt % 5
        if strat == 0:
            # Global small noise in log-space
            z = np.log(np.maximum(current_a, 1e-30))
            z += 0.008 * rng.standard_normal(n)
            z -= z.mean()
            a_pert = np.exp(z - np.max(z))
            a_pert = _normalize(a_pert)
        elif strat == 1:
            # Frequency domain phase perturbation
            a_pert = _freq_domain_perturb(current_a, rng, strength=0.03)
        elif strat == 2:
            # Frequency domain magnitude perturbation
            a_pert = _freq_magnitude_perturb(current_a, rng, strength=0.015)
        elif strat == 3:
            # Block perturbation
            a_pert = current_a.copy()
            block = max(30, n // 12)
            start = int(rng.integers(0, n - block))
            a_pert[start:start + block] *= np.exp(0.04 * rng.standard_normal(block))
            a_pert = _normalize(a_pert)
        else:
            # Sparse large perturbation
            a_pert = current_a.copy()
            n_perturb = max(20, n // 20)
            idx = rng.choice(n, size=n_perturb, replace=False)
            a_pert[idx] *= np.exp(0.08 * rng.standard_normal(n_perturb))
            a_pert = _normalize(a_pert)

        # Quick optimization
        opt_deadline = min(deadline - min_time, time.monotonic() + 12)
        a_opt, sc_opt = _top_m_optimize(a_pert, opt_deadline, m_start=3, m_end=1,
                                         min_time=2, lr_init=0.015)
        sc_opt = _score(a_opt)

        if sc_opt < best_sc - 1e-13:
            best_sc = sc_opt
            best_a = a_opt.copy()
            current_a = a_opt.copy()
            print(f"    ILS NEW BEST: {best_sc:.10f} (attempt {attempt}, strat {strat}) "
                  f"t={time.monotonic()-_t_start:.0f}s")
        elif sc_opt < current_sc - 1e-13:
            current_a = a_opt.copy()
            current_sc = sc_opt

        attempt += 1

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _lbfgs_smooth(a_init, deadline, min_time=30.0):
    """L-BFGS with log-sum-exp smoothing as backup optimizer."""
    n = len(a_init)
    a = a_init.copy()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-15:
        return a

    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < c_max * 0.001:
        gap = c_max * 0.001
    base_beta = max(gap, c_max * 0.003)

    def obj(z):
        a_s = np.exp(z - np.max(z))
        a_s = _normalize(a_s)
        c_s = _full_convolve(a_s, a_s)
        cm = float(c_s.max())
        shifted = (c_s - cm) / base_beta
        e = np.exp(np.clip(shifted, -500, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n * cm
        return 2.0 * n * (cm + base_beta * np.log(s))

    def grad(z):
        a_s = np.exp(z - np.max(z))
        a_s = _normalize(a_s)
        c_s = _full_convolve(a_s, a_s)
        cm = float(c_s.max())
        shifted = (c_s - cm) / base_beta
        e = np.exp(np.clip(shifted, -500, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c_s)
            w[int(np.argmax(c_s))] = 1.0
        else:
            w = e / s
        grad_c = _conv_gradient(w, a_s)
        W = float(np.dot(w, c_s))
        # Gradient in z-space: chain rule through softmax
        # a_s = softmax(z), so da_s/dz_j = a_s_j (delta_jk - a_s_k)
        # d(c_k)/da_s = 2 * (shifted a)
        # d(obj)/dz_j = sum_k w_k * d(c_k)/da_s_j * da_s_j/dz_j ... 
        # Actually simpler: obj depends on a_s which depends on z
        # d obj / d z_j = sum_k w_k * (2 a_{k-j} - 2 * W * something)
        # Let me just use: grad_a = 2n * (conv_gradient(w, a) - mean)
        # then grad_z = diag(a_s) @ (grad_a - mean(grad_a)) ... 
        # Actually for softmax: if f(a) and a = softmax(z), then
        # df/dz_j = a_j * (df/da_j - sum_k a_k * df/da_k)
        grad_a = 2.0 * n * _conv_gradient(w, a_s)
        grad_a = grad_a - np.mean(grad_a)
        return a_s * (grad_a - float(np.dot(a_s, grad_a)))

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    remaining = deadline - min_time - time.monotonic()
    if remaining < 10:
        return a

    max_iter = min(5000, max(50, int(remaining / 0.02)))
    try:
        result = minimize(obj, z, jac=grad, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'gtol': 1e-12,
                                   'maxcor': 80})
        a_new = np.exp(result.x - np.max(result.x))
        a_new = _normalize(a_new)
        sc_new = _score(a_new)
        sc_old = _score(a)
        if sc_new < sc_old:
            return a_new
    except Exception:
        pass
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid Peak-Focused Optimizer combining:
    1. Sum-of-top-m gradient optimization with BB step size
    2. Water-filling peak equalization
    3. Coordinate descent on peak contributors
    4. Stochastic ILS with frequency-domain perturbations
    5. L-BFGS with log-sum-exp smoothing (backup)
    6. Multi-n exploration
    """
    global _t_start
    _t_start = time.monotonic()
    deadline = _t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent = parent / parent.sum()
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 24576
        best_seq = _make_profile(n_parent, "hann")
        best_score = _score(best_seq)
        print(f"No parent, Hann n={n_parent}, score={best_score:.10f}")

    n_main = len(best_seq)

    # === Phase 1: Top-m optimization (250s) ===
    print(f"\n[P1] Top-m optimize (t={time.monotonic()-_t_start:.1f}s)")
    p1_deadline = _t_start + 250
    a1, sc1 = _top_m_optimize(best_seq, p1_deadline, m_start=10, m_end=3,
                               min_time=20, lr_init=0.02, momentum=0.82)
    if sc1 < best_score:
        best_seq = a1
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [top_m] t={time.monotonic()-_t_start:.1f}s")

    # === Phase 2: Water filling (150s) ===
    print(f"\n[P2] Water fill (t={time.monotonic()-_t_start:.1f}s)")
    p2_deadline = min(_t_start + 400, deadline - 500)
    if time.monotonic() < p2_deadline:
        a2, sc2 = _water_fill_optimize(best_seq, p2_deadline, min_time=15)
        if sc2 < best_score:
            best_seq = a2
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [water] t={time.monotonic()-_t_start:.1f}s")

    # === Phase 3: Coordinate descent (250s) ===
    print(f"\n[P3] Coord descent (t={time.monotonic()-_t_start:.1f}s)")
    p3_deadline = min(_t_start + 650, deadline - 400)
    if time.monotonic() < p3_deadline:
        a3, sc3 = _coord_descent_peaks(best_seq, p3_deadline, min_time=20, max_rounds=300)
        if sc3 < best_score:
            best_seq = a3
            best_score = sc3
