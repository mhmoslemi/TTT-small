"""
Focused optimizer for minimum peak linear autocorrelation sequences.
Key insight: asymmetric (power-law) profiles beat symmetric ones.
Uses L-BFGS in log-space with aggressive beta continuation,
periodic perturbations, and active-set refinement.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _init_power_law(n, p=1.0):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    a = a / a.sum()
    return a


def _init_mixed(n, p=1.5, alpha=0.3):
    i = np.arange(1, n + 1, dtype=np.float64)
    a1 = i ** p
    t = np.linspace(0, 1, n)
    a2 = np.sin(np.pi * t) ** 0.5
    a = alpha * a1 / a1.sum() + (1 - alpha) * a2 / a2.sum()
    return a / a.sum()


def _run_lbfgs_continuation(a_init, deadline, min_time_remaining=60.0,
                            n_perturb=5, perturb_every=200):
    """L-BFGS with aggressive beta continuation and periodic perturbations."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    current_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    best_a = a.copy()
    best_sc = current_sc

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    mean_c = float(c_cur.mean())
    gap = max_c - mean_c
    base_beta = max(gap * 0.05, max_c * 0.001)

    ratios = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
              0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001,
              0.000005, 0.000002, 0.000001, 0.0000005, 0.0000002, 0.0000001,
              0.0000005, 0.0000002, 0.0000001, 0.00000005, 0.00000002, 0.00000001]

    rng = np.random.default_rng(12345)
    total_iter = 0

    for i, ratio in enumerate(ratios):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        beta = base_beta * ratio
        est_time = 0.004 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 4))))

        def obj_func(z_local):
            a_local = _softmax(z_local)
            c_local = _full_convolve(a_local, a_local)
            c_max_local = float(c_local.max())
            shifted = (c_local - c_max_local) / beta
            e_loc = np.exp(np.clip(shifted, -700, 0))
            s_e = e_loc.sum()
            if s_e < 1e-300:
                return 2.0 * n * c_max_local
            return 2.0 * n * (c_max_local + beta * np.log(s_e))

        def grad_func(z_local):
            a_local = _softmax(z_local)
            c_local = _full_convolve(a_local, a_local)
            c_max_local = float(c_local.max())
            shifted = (c_local - c_max_local) / beta
            e_loc = np.exp(np.clip(shifted, -700, 0))
            s_e = e_loc.sum()
            if s_e < 1e-300:
                w = np.zeros_like(c_local)
                w[int(np.argmax(c_local))] = 1.0
            else:
                w = e_loc / s_e
            a_rev = a_local[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W_val = float(np.dot(w, c_local))
            return 4.0 * n * a_local * (h - W_val)

        try:
            result = minimize(
                obj_func, z,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS beta_ratio={ratio:.6f}: score={sc_new:.10f}"
                      f" (iter={result.nit})")
            total_iter += result.nit

            # Periodic perturbation to escape local minima
            if total_iter > perturb_every * (i + 1) and beta > base_beta * 0.001:
                sigma = 0.02 * (1.0 + i * 0.1)
                z = z + sigma * rng.standard_normal(n)
                z = z - z.mean()
        except Exception as e:
            print(f"    L-BFGS error at ratio={ratio}: {e}")
            pass

    return best_a


def _run_active_set_refine(a_init, deadline, min_time_remaining=60.0):
    """Refine using active-set QP with projected gradient inner solve."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    max_c = float(c.max())

    for rnd in range(200):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max_c * (0.01 * (0.9 ** rnd))
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / active.sum()

        # Projected gradient with backtracking
        for inner in range(30):
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_rev = a[::-1]
            wp = _full_convolve(p, a_rev)
            h = wp[n - 1:2 * n - 1]
            W_val = float(np.dot(p, c))
            grad = 2.0 * a * (h - W_val)

            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-16:
                break
            step = 1.0
            for _ in range(20):
                a_new = np.maximum(a - step * grad / gnorm, 0)
                sn = a_new.sum()
                if sn < 1e-15:
                    break
                a_new = a_new / sn
                c_new = _full_convolve(a_new, a_new)
                if float(c_new.max()) < c_max * 1.001:
                    a = a_new
                    break
                step *= 0.5
            c_max = float(_full_convolve(a, a).max())

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            if rnd % 20 == 0:
                print(f"    Active-set rnd={rnd}: score={sc:.10f}")

    return best_a


def _run_block_refine(a_init, deadline, min_time_remaining=30.0):
    """Block coordinate descent: optimize subsets of coordinates at a time."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(999)

    block_size = min(512, n // 4)

    for it in range(500):
        if time.monotonic() + min_time_remaining > deadline:
            break

        # Random block
        start = rng.integers(0, max(1, n - block_size))
        idx = np.arange(start, min(n, start + block_size))

        # SPSA-like step: evaluate score with small perturbation
        sigma = 0.1 / (1.0 + it * 0.1)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Gradient of score w.r.t. a at active lags
        active = c >= c_max * 0.99
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / max(1, active.sum())

        # Compute gradient via FFT
        a_rev = a[::-1]
        wp = _full_convolve(p, a_rev)
        h = wp[n - 1:2 * n - 1]
        W_val = float(np.dot(p, c))
        grad = 2.0 * a * (h - W_val)

        # Update the block
        g_block = grad[idx]
        gnorm = np.linalg.norm(g_block)
        if gnorm < 1e-16:
            continue
        lr = 0.5 / (gnorm * gnorm + 1e-10)
        a_new = a.copy()
        a_new[idx] = np.maximum(a_new[idx] - lr * g_block, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            continue
        a_new = a_new / sn

        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new

    return best_a
