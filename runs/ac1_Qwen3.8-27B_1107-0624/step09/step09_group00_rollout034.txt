LP-assisted subgradient optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses exact subgradient with Nesterov acceleration, LP-based direction finding,
adaptive active set, multi-n exploration, and L-BFGS with log-sum-exp smoothing.

import numpy as np
import time
from scipy.optimize import minimize, linprog

# Global timer
_t_start = 0.0

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _simplex_project(x):
    """Project x onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(x)
    if n == 0:
        return x
    # Sort in decreasing order
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _active_set(c, frac=0.01, max_count=200):
    """Find active set: indices where c[k] >= max(c) * (1 - frac)."""
    c_max = c.max()
    threshold = c_max * (1.0 - frac)
    active = np.where(c >= threshold)[0]
    if len(active) > max_count:
        # Keep top max_count
        vals = c[active]
        idx = np.argsort(vals)[::-1][:max_count]
        active = active[idx]
    return active, c_max

def _subgradient(a, c, active, c_max):
    """Compute subgradient of max_k (a*a)_k at a.
    
    Returns the gradient vector (length n).
    """
    n = len(a)
    active_c = c[active]
    # Weights: proportional to exp((c[k] - c_max) / (c_max * 0.001))
    # This gives more weight to higher peaks
    beta = c_max * 0.001
    w = np.exp((active_c - c_max) / beta)
    w = w / w.sum()
    
    # grad = 2 * sum_k w[k] * shift(a, k)

"""
LP-assisted subgradient optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses exact subgradient with Nesterov acceleration, LP-based direction finding,
adaptive active set management, L-BFGS with log-sum-exp smoothing,
multi-n exploration, and spectral pre-conditioning.

import numpy as np
import time
import scipy.optimize
from scipy.optimize import minimize, linprog
import scipy.sparse
from scipy import linalg

_t_start = 0.0
_t_deadline = 0.0


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _simplex_project(x):
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _convolve_max_and_active(a, frac=0.005, max_active=300):
    """Compute convolution, peak, and active set in one call."""
    n = len(a)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    threshold = c_max * (1.0 - frac)
    active = np.where(c >= threshold)[0]
    if len(active) > max_active:
        vals = c[active]
        top_idx = np.argsort(vals)[::-1][:max_active]
        active = active[top_idx]
    return c, c_max, active


def _subgradient_fft(a, c, active, c_max, rel_tol=0.002):
    """Compute weighted subgradient of max_k (a*a)_k using FFT."""
    n = len(a)
    active_c = c[active]
    beta = c_max * rel_tol
    w = np.exp((active_c - c_max) / beta)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(active)) / len(active)
    else:
        w = w / w_sum

    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(active):
        w_full[k] = w[i]

    a_rev = a[::-1]
    wc = _fft_convolve(w_full, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    grad = grad - np.mean(grad)
    return grad


def _lp_direction(a, c, active, c_max, max_constraints=40):
    """Solve LP to find direction d that reduces active autocorrelation values.
    
    min t  s.t.  (a*d)_k <= t  for k in active
                 sum(d) = 0
                 d >= -a
    """
    n = len(a)
    # Limit active set size for LP
    if len(active) > max_constraints:
        active_c = c[active]
        top_idx = np.argsort(active_c)[::-1][:max_constraints]
        active = active[top_idx]
    m = len(active)
    if m == 0:
        return None

    # Build constraint matrix (sparse)
    # Row i: sum_j a[active[i] - j] * d[j] + (-1)*t <= 0
    rows = []
    cols = []
    data = []
    for i, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            val = a[k - j]
            if val > 1e-15:
                rows.append(i)
                cols.append(j)
                data.append(val)
        # -t term
        rows.append(i)
        cols.append(n)  # t is variable index n
        data.append(-1.0)

    A_ineq = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(m, n + 1))
    b_ineq = np.zeros(m)

    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    # Bounds: d_j >= -a_j, t free
    bounds = [(-a[j], None) for j in range(n)]
    bounds.append((None, None))

    try:
        res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq,
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.fun < -1e-12:
            d = res.x[:n]
            d = d - np.mean(d)  # ensure sum = 0 numerically
            return d
        elif res.success and res.x is not None:
            d = res.x[:n]
            d = d - np.mean(d)
            if np.linalg.norm(d) > 1e-10:
                return d
    except Exception:
        pass
    return None


def _lbfgs_smoothed(a_init, deadline, min_beta_frac=0.0001, max_beta_frac=0.05):
    """L-BFGS on log-softmax parameterization with adaptive beta annealing."""
    n = len(a_init)
    c = _fft_convolve(a_init, a_init)
    c_max = float(c.max())
    if c_max < 1e-15:
        return a_init

    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < c_max * 0.001:
        gap = c_max * 0.001
    base_beta = max(gap, c_max * max_beta_frac)

    # Beta schedule: from large to small
    n_beta = 20
    betas = base_beta * (max_beta_frac / (max_beta_frac * base_beta)) ** np.linspace(0, 1, n_beta)
    betas = base_beta * np.geomspace(1.0, max(min_beta_frac, 1e-6), n_beta)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - 10:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break

        def f(z_val):
            a = _softmax(z_val)
            c_val = _fft_convolve(a, a)
            cm = float(c_val.max())
            shifted = (c_val - cm) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                return 2.0 * n * cm
            return 2.0 * n * (cm + beta * np.log(s))

        def g(z_val):
            a = _softmax(z_val)
            c_val = _fft_convolve(a, a)
            cm = float(c_val.max())
            shifted = (c_val - cm) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                w = np.zeros_like(c_val)
                w[int(np.argmax(c_val))] = 1.0
            else:
                w = e / s
            a_rev = a[::-1]
            wc = _fft_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c_val))
            return 4.0 * n * a * (h - W)

        est_time = 0.003 * max(1.0, n / 16384.0)
        max_iter = min(8000, max(50, int(remaining / (est_time * 20))))

        try:
            result = minimize(
                f, z, jac=g, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _nesterov_subgradient(a_init, deadline, lr_init=0.01, momentum=0.9,
                          rel_tol_start=0.005, rel_tol_end=0.0005,
                          lp_interval=80):
    """Nesterov-accelerated subgradient with periodic LP direction finding."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    v = np.zeros(n)
    rel_tol = rel_tol_start
    lr = lr_init
    no_improve = 0
    max_stall = 200
    iter_count = 0

    while time.monotonic() < deadline - 5:
        c, c_max, active = _convolve_max_and_active(a, frac=rel_tol)
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Get direction: LP every lp_interval steps, else subgradient
        if iter_count % lp_interval == lp_interval - 1:
            d = _lp_direction(a, c, active, c_max, max_constraints=30)
            if d is not None:
                d_norm = np.linalg.norm(d)
                if d_norm > 1e-12:
                    d = d / d_norm
                    # Line search along LP direction
                    improved_ls = False
                    for alpha in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
                        a_try = _simplex_project(a + alpha * d)
                        sc_try = _score(a_try)
                        if sc_try < current_sc - 1e-13:
                            a = a_try
                            improved_ls = True
                            break
                    if improved_ls:
                        iter_count += 1
                        continue
                    else:
                        # LP didn't help, fall through to subgradient
                        d = None

        # Subgradient step
        grad = _subgradient_fft(a, c, active, c_max, rel_tol=rel_tol)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            # Try LP
            d = _lp_direction(a, c, active, c_max, max_constraints=40)
            if d is None:
                break
            grad = d
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            grad = grad / gnorm

        # Nesterov momentum
        v = momentum * v - lr * grad
        step = v

        # Line search
        improved = False
        alpha = 1.0
        for _ in range(40):
            a_try = _simplex_project(a + alpha * step)
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            v = np.zeros(n)
            no_improve += 1
            if no_improve % 20 == 0:
                rel_tol = max(rel_tol_end, rel_tol * 0.7)
                lr *= 0.8
            if no_improve >= max_stall:
                # Try LP to kick out
                d = _lp_direction(a, c, active, c_max, max_constraints=50)
                if d is not None:
                    for alpha in [0.05, 0.02, 0.01]:
                        a_try = _simplex_project(a + alpha * d)
                        if _score(a_try) < best_sc:
                            a = a_try
                            break
                    a = best_a.copy()
                v = np.zeros(n)
                rel_tol = max(rel_tol_end, rel_tol * 0.5)
                lr = lr_init * 0.6
                no_improve = 0
        else:
            if rel_tol > rel_tol_end:
                rel_tol = max(rel_tol_end, rel_tol * 0.99)
            lr = min(lr_init * 2, lr * 1.005)

        iter_count += 1

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos2":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "super_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 4)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.where(np.abs(t - 0.5) < 0.3, 1.0, 0.5 * (1 + np.cos(np.pi * (np.abs(t - 0.5) - 0.3) / 0.2)))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bickley":
        # Bickley-Naylor (psi) function approximation
        t = np.linspace(0.01, 0.99, n)
        a = np.exp(-1.0 / (t * (1 - t) * 4))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _spectral_damp(a, strength=0.01):
    """Dampen low-frequency components that contribute to the autocorrelation peak."""
    n = len(a)
    A = np.fft.rfft(a)
    # The DC and low-frequency components dominate the peak
    # Dampen frequencies 1..K (keep DC)
    K = max(5, n // 100)
    A[1:K+1] *= (1.0 - strength)
    a_new = np.fft.irfft(A, n)
    return _normalize(a_new)


def _quick_optimize(a_init, deadline):
    """Combined L-BFGS + subgradient optimization."""
    remaining = deadline - time.monotonic()
    if remaining < 5:
        return a_init

    # L-BFGS phase
    lbfgs_deadline = time.monotonic() + remaining * 0.5
    a = _lbfgs_smoothed(a_init, lbfgs_deadline)

    # Subgradient polish
    sg_deadline = min(deadline, time.monotonic() + remaining * 0.4)
    a2, sc2 = _nesterov_subgradient(a, sg_deadline,
                                     lr_init=0.01, momentum=0.9,
                                     rel_tol_start=0.003, rel_tol_end=0.0003,
                                     lp_interval=50)
    sc1 = _score(a)
    if sc2 < sc1:
        a = a2

    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining:
    1. Diverse initialization with asymmetric profiles
    2. L-BFGS with log-sum-exp annealing
    3. Nesterov-accelerated exact subgradient
    4. LP-based direction finding for plateau escape
    5. Multi-n exploration with resampling
    6. Spectral pre-conditioning
    """
    global _t_start, _t_deadline
    _t_start = time.monotonic()
    _t_deadline = _t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 24576
        parent = _make_profile(n_parent, "hann")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, hann n={n_parent}, score={best_score:.10f}")

    n_main = n_parent if n_parent > 0 else 24576

    # === Phase 1: Quick multi-start (350s) ===
    print(f"\n[P1] Multi-start (t={time.monotonic()-_t_start:.1f}s)")
    ms_deadline = _t_start + 350

    starts = [('parent', parent.copy())]
    for kind in ['hann', 'gaussian', 'raised_cos2', 'super_gauss', 'flat_top', 'bickley', 'triangle']:
        starts.append((kind, _make_profile(n_main, kind)))

    for name, a_init in starts:
        if time.monotonic() > ms_deadline - 30:
            break
        a_opt = _quick_optimize(a_init, ms_deadline)
        sc_opt = _score(a_opt)
        print(f"  {name}: opt={sc_opt:.10f}")
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [ms/{name}] t={time.monotonic()-_t_start:.1f}s")

    # === Phase 2: Intensive L-BFGS (400s) ===
    print(f"\n[P2] Intensive L-BFGS (t={time.monotonic()-_t_start:.1f}s)")
    lbfgs_deadline = min(_t_deadline - 600, time.monotonic() + 400)
    a_main = _lbfgs_smoothed(best_seq, lbfgs_deadline, min_beta_frac=0.00005, max_beta_frac=0.03)
    sc = _score(a_main)
    if sc < best_score:
        best_seq = a_main.copy()
        best_score = sc
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_int] t={time.monotonic()-_t_start:.1f}s")

    # === Phase 3: Nesterov subgradient with LP (300s) ===
    print(f"\n[P3] Nesterov+LP (t={time.monotonic()-_t_start:.1f}s)")
    sg_deadline = min(_t_deadline - 400, time.monotonic() + 300)
    a_sg, sc_sg = _nesterov_subgradient(best_seq, sg_deadline,
                                         lr_init=0.008, momentum=0.92,
                                         rel_tol_start=0.003, rel_tol_end=0.0002,
                                         lp_interval=40)
    if sc_sg < best_score:
        best_seq = a_sg.copy()
        best_score = sc_sg
        print(f"  NEW BEST: {best_score:.10f} [nesterov_lp] t={time.monotonic()-_t_start:.1f}s")

    # === Phase 4: Multi-n exploration (250s) ===
    print(f"\n[P4] Multi-n (t={time.monotonic()-_t_start:.1f}s)")
    n_base = len(best_seq)
    n_candidates = []
    for dn in [-4096, -2048, -1024, 1024, 2048, 4096, 6144, 8192]:
        nn = n_base + dn
        if nn >= 4096 and nn <= 65536 and nn != n_base:
            n_candidates.append(nn)
    n_candidates = list(set(n_candidates))
    # Sort by proximity to n_base
    n_candidates.sort(key=lambda x: abs(x - n_base))
    n_candidates = n_candidates[:6]  # limit to 6

    for n_try in n_candidates:
        if time.monotonic() > _t_deadline - 200:
            break
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        a_opt = _quick_optimize(a_init, _t_deadline)
        sc_opt = _score(a_opt)
        print(f"  n={n_try}: init={sc_init:.10f}, opt={sc_opt:.10f}")
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-_t_start:.1f}s")

    # === Phase 5: LP-focused polish (200s) ===
    print(f"\n[P5] LP polish (t={time.monotonic()-_t_start:.1f}s)")
    lp_deadline = min(_t_deadline - 100, time.monotonic() + 200)
    a_lp, sc_lp = _nesterov_subgradient(best_seq, lp_deadline,
                                         lr_init=0.005, momentum=0.95,
                                         rel_tol_start=0.002, rel_tol_end=0.0001,
                                         lp_interval=15)
    if sc_lp < best_score:
        best_seq = a_lp.copy()
        best_score = sc_lp
        print(f"  NEW BEST: {best_score:.10f} [lp_polish] t={time.monotonic()-_t_start:.1f}s")

    # === Phase 6: Asymmetric perturbation + optimize (150s) ===
    print(f"\n[P6] Asymmetric (t={time.monotonic()-_t_start:.1f}s)")
    if time.monotonic() < _t_deadline - 80:
        n = len(best_seq)
        # Try slight asymmetry
        for asym in [0.01, 0.02, 0.03]:
            if time.monotonic() > _t_deadline - 60:
                break
            t = np.linspace(0, 1, n)
            asym_factor = 1.0 + asym * np.cos(np.pi * t)
            a_asym = best_seq * asym_factor
            a_asym = _normalize(a_asym)
            sc_asym = _score(a_asym)
            if sc_asym < best_score + 0.01:  # only optimize if close
                a_opt = _quick_optimize(a_asym, min(_t_deadline, time.monotonic() + 30))
                sc_opt = _score(a_opt)
                if sc_opt < best_score:
                    best_seq = a_opt.copy()
                    best_score = sc_opt
                    print(f"  NEW BEST: {best_score:.10f} [asym={asym}] t={time.monotonic()-_t_start:.1f}s")

    # === Phase 7: Final ultra-fine polish (80s) ===
    print(f"\n[P7] Final polish (t={time.monotonic()-_t_start:.1f}s)")
    if time.monotonic() < _t_deadline - 5:
        # L-BFGS with very small beta
        a_fin = _lbfgs_smoothed(best_seq, _t_deadline - 3,
                                min_beta_frac=0.00001, max_beta_frac=0.01)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-_t_start:.1f}s")

        # Final subgradient
        if time.monotonic() < _t_deadline - 5:
            a_fin2, sc_fin2 = _nesterov_subgradient(best_seq, _t_deadline - 2,
                                                     lr_init=0.003, momentum=0.97,
                                                     rel_tol_start=0.001,
                                                     rel_tol_end=0.00005,
                                                     lp_interval=20)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_sg] t={time.monotonic()-_t_start:.1f}s")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-_t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    return [float(x) for x in fallback]
