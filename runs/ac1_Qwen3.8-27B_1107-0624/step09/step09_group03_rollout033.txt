Optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 using:

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n-1:2*n-1]

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}"""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)

def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _cg_solve(matvec, rhs, max_cg=300, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    rhs_norm = max(rhs_norm, 1e-30)
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

def _kkt_active_set(a, c, c_max, gradient_tol=1e-6):
    """
    Determine active set using KKT conditions.
    Active positions are those where c[k] is close to c_max AND
    the gradient condition is nearly satisfied.
    """
    n = len(a)
    # Primary criterion: convolution value close to max
    tol_frac = 0.01
    primary = c >= (c_max - tol_frac * c_max)
    
    if not primary.any():
        primary[np.argmax(c)] = True
    
    return primary

def _newton_step_kkt(a, c, c_max, tol_frac, mu=1e-8):
    """
    Compute Newton step with KKT verification.
    """
    n = len(a)
    active = c >= (c_max - tol_frac * c_max)
    if not active.any():
        active[np.argmax(c)] = True
    m = int(active.sum())
    if m < 2 or m > 15 * n:
        return None, None
    
    A_vec = active.astype(np.float64) / m
    g = 2.0 * _corr_a_A(a, A_vec, n)
    g_t = g - np.mean(g)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-14:
        return None, None
    
    # Regularization
    mu_val = mu / m + 1e-10
    
    A_vec_copy = A_vec
    n_copy = n
    
    def _hess_vp(v):
        return 2.0 * _corr_a_A(v, A_vec_copy, n_copy) + mu_val * v
    
    u = _cg_solve(_hess_vp, g_t, max_cg=300)
    w = _cg_solve(_hess_vp, np.ones(n), max_cg=300)
    w_sum = float(w.sum())
    
    if abs(w_sum) < 1e-15:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w
    
    dx = dx - np.mean(dx)
    return dx, A_vec

def _nesterov_step(a, deadline, max_iter=5000):
    """
    Nesterov-accelerated projected gradient method.
    Minimizes the smoothed objective (using subgradient of max).
    """
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    
    # Nesterov momentum
    y = a.copy()
    t_val = 1.0
    gamma = 1.0  # step size (will adapt)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute subgradient at y
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        # Subgradient: uniform over argmax positions
        peak_mask = c_y >= c_max * (1 - 1e-10)
        w = np.zeros_like(c_y)
        w[peak_mask] = 1.0 / peak_mask.sum()
        
        # Gradient of max(conv(y,y)) w.r.t. y
        g = 2.0 * _corr_a_A(y, w, n)
        
        # Normalize gradient (scale-invariant objective)
        s_y = y.sum()
        grad_obj = 2.0 * n * g / (s_y * s_y) - 4.0 * n * c_max / (s_y * s_y * s_y) * np.ones(n)
        # Actually, let's use the normalized gradient
        g_t = g - np.mean(g)
        
        # Gradient descent step
        gamma = 1.0 / (np.linalg.norm(g_t) ** 2 + 1e-20) * 0.5
        a_try = y - gamma * g_t
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            a_try = a
        else:
            a_try = a_try / s_try
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        beta_mom = (t_val - 1.0) / t_new
        y_new = a_try + beta_mom * (a_try - y)
        y_new = np.maximum(y_new, 0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = a_try
        else:
            y_new = y_new / s_new
        
        sc = _score(a_try)
        if sc < best_sc:
            best_sc = sc
            best_a = a_try.copy()
        
        a = a_try
        y = y_new
        t_val = t_new
    
    return best_a, best_sc

def _lbfgs_continuation(a_init, deadline, min_time_remaining=30.0, n_betas=20):
    """
    L-BFGS with adaptive beta continuation.
    """
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    # Adaptive initial beta based on the spread of top convolution values
    c_sorted = np.sort(c_cur)[::-1]
    # Use the gap between top 1 and top min(100, 2n-1)
    k_ref = min(100, len(c_sorted))
    gap = c_sorted[0] - c_sorted[k_ref - 1]
    base_beta = max(gap * 0.5, max_c * 0.001)
    
    # Schedule: geometric from base_beta to very small
    min_beta = max_c * 1e-8
    log_betas = np.linspace(np.log(base_beta), np.log(min_beta), n_betas)
    betas = np.exp(log_betas)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Estimate iterations based on time
        est_time_per_iter = 0.002 * max(1.0, n / 8192.0)
        max_iter = min(3000, max(20, int(remaining / (est_time_per_iter * 8))))
        
        try:
            result = minimize(
                _lsobj,
                z,
                args=(beta, n),
                jac=_lsgrad,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.12f}")
        except Exception as ex:
            print(f"    L-BFGS beta={beta:.2e}: exception {ex}")
    
    return current_a

def _active_set_newton(a_init, deadline, max_outer=50):
    """
    Active-set Newton with adaptive tolerances.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Try multiple tolerances, from loose to tight
        tol_list = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
        
        best_improvement = 0
        best_a_new = None
        
        for tol_frac in tol_list:
            dx, A_vec = _newton_step_kkt(a, c, c_max, tol_frac)
            if dx is None:
                continue
            
            # Line search
            t = 1.0
            for ls in range(40):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < c_max * (1 - 1e-12):
                    improvement = c_max - f_try
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_a_new = a_try.copy()
                    break
                t *= 0.5
        
        if best_a_new is None:
            # Try gradient step as fallback
            g_all = 2.0 * _corr_a_A(a, np.ones(2*n-1) / (2*n-1), n)
            g_t = g_all - np.mean(g_all)
            t = 0.01
            improved = False
            for ls in range(20):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    sc_try = _score(a_try)

                        best_improvement = 1
                        improved = True

                break
        
        a = best_a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Check convergence

            break

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.25)**2) + np.exp(-20 * (t - 0.75)**2)
        return a / a.sum()
    elif kind == "random_nonneg":
        rng = np.random.default_rng(12345)
        a = np.abs(rng.standard_normal(n))
        return a / a.sum()
    else:
        return np.ones(n) / n

def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _refinement_loop(a, deadline, max_rounds=10):

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:

        newton_dl = min(deadline - 5, time.monotonic() + 40)

            best_a = a_nest.copy()
            a = a_nest
        
        if abs(_score(a) - best_sc) < 1e-15:
            break
    
    return best_a, best_sc

Wait, I realize I need to double-check my `_lsobj` and `_lsgrad` functions. They take `z` as the first argument and `beta`, `n` as additional arguments. When used with `scipy.optimize.minimize`, the `args` parameter will pass `beta` and `n` correctly.

Let me also think about whether the Nesterov step is correct. The subgradient of `max_k c[k]` with respect to `a` is:
`g_i = 2 * sum_{k in argmax} a[k-i]` (where valid indices are used)

But since the objective is scale-invariant, we should normalize by the sum. Let me reconsider.

The objective is `f(a) = 2n * max(c) / (sum(a))^2`.

The gradient with respect to `a[i]` is:
`df/da[i] = 2n * (dc_max/da[i]) / (sum(a))^2 - 2 * 2n * max(c) / (sum(a))^3 * 1`
`= 2n * (dc_max/da[i]) / (sum(a))^2 - 4n * max(c) / (sum(a))^3`

where `dc_max/da[i] = 2 * sum_{k in argmax} a[k-i]` (for the subgradient in the argmax case).

Actually, for the smoothed version (log-sum-exp), the gradient is:
`df/da[i] = 2n * sum_k w[k] * (dc[k]/da[i]) / (sum(a))^2 - 4n * f_smooth / (sum(a))^3`

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)

    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):

    a_rev = a[::-1]

    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))

def _cg_solve(matvec, rhs, max_cg=300, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = max(np.linalg.norm(rhs), 1e-30)
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

    active = c >= (c_max - tol_frac * c_max)
    if not active.any():

    m = int(active.sum())
    if m < 2 or m > 20 * n:

    A_vec = active.astype(np.float64) / m
    g = 2.0 * _corr_a_A(a, A_vec, n)
    g_t = g - np.mean(g)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-14:
        return None, None

    mu = mu_base / m + 1e-10

                _lsobj, z, args=(beta, n),
                jac=_lsgrad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxcor': 100, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            if i % 5 == 0 or i == n_betas - 1:
                print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.12f}")
        except Exception as ex:
            print(f"    L-BFGS beta={beta:.2e}: exc {ex}")

    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15 * (t - 0.25) ** 2) + np.exp(-15 * (t - 0.75) ** 2)
        return a / a.sum()
    elif kind == "triangular":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2 * t - 1)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _refinement_loop(a, deadline, max_rounds=8):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        newton_dl = min(deadline - 10, time.monotonic() + 30)
        a_newton, sc_newton = _active_set_newton(a, newton_dl, max_outer=15)
        if sc_newton < best_sc:
            best_sc = sc_newton
            best_a = a_newton.copy()
            a = a_newton

        nested_dl = min(deadline - 10, time.monotonic() + 20)
        a_nest, sc_nest = _nesterov_projected(a, nested_dl, max_iter=3000)
        if sc_nest < best_sc:
            best_sc = sc_nest
            best_a = a_nest.copy()
            a = a_nest

        if abs(_score(a) - best_sc) < 1e-15:
            break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: adaptive L-BFGS continuation, KKT-verified
    active-set Newton, Nesterov projected gradient, multi-scale, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.12f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.12f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        update_best(parent, _score(parent), "init")

    # === Phase 1: L-BFGS continuation ===
    print(f"\n[Phase 1] L-BFGS continuation (t={time.monotonic() - t_start:.1f}s)")
    lbfgs_dl = min(deadline - 600, time.monotonic() + 350)
    a_lbfgs = _lbfgs_continuation(parent, lbfgs_dl, min_time_remaining=600.0, n_betas=25)
    update_best(a_lbfgs, _score(a_lbfgs), "lbfgs")

    # === Phase 2: Active-set Newton ===
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        as_dl = min(deadline - 500, time.monotonic() + 300)
        a_as, sc_as = _active_set_newton(best_seq, as_dl, max_outer=80)
        update_best(a_as, sc_as, "newton")

    # === Phase 3: Nesterov ===
    print(f"\n[Phase 3] Nesterov (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        nest_dl = min(deadline - 400, time.monotonic() + 120)
        a_nest, sc_nest = _nesterov_projected(best_seq, nest_dl, max_iter=8000)
        update_best(a_nest, sc_nest, "nesterov")

    # === Phase 4: Refinement loop ===
    print(f"\n[Phase 4] Refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        ref_dl = min(deadline - 350, time.monotonic() + 120)
        a_ref, sc_ref = _refinement_loop(best_seq, ref_dl, max_rounds=8)
        update_best(a_ref, sc_ref, "refinement")

    # === Phase 5: Try n=32768 ===
    print(f"\n[Phase 5] Try n=32768 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        n_large = 32768
        if len(best_seq) < n_large:
            a_large = _interpolate_seq(best_seq, n_large)
            sc_large_init = _score(a_large)
            print(f"  n={n_large} init: {sc_large_init:.12f}")
            lbfgs3_dl = min(deadline - 150, time.monotonic() + 60)
            a_large_opt = _lbfgs_continuation(a_large, lbfgs3_dl, min_time_remaining=150.0, n_betas=8)
            update_best(a_large_opt, _score(a_large_opt), "n_large")
            if time.monotonic() < deadline - 100:
                nt3_dl = min(deadline - 100, time.monotonic() + 50)
                a_ln, sc_ln = _active_set_newton(a_large_opt, nt3_dl, max_outer=25)
                update_best(a_ln, sc_ln, "n_large_newton")

    # === Phase 6: Try n=8192 with different init ===
    print(f"\n[Phase 6] Try n=8192 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        n_med = 8192
        if len(best_seq) > n_med:
            a_med = _interpolate_seq(best_seq, n_med)
        else:
            a_med = _interpolate_seq(best_seq, n_med)
        sc_med_init = _score(a_med)
        lbfgs4_dl = min(deadline - 100, time.monotonic() + 50)
        a_med_opt = _lbfgs_continuation(a_med, lbfgs4_dl, min_time_remaining=100.0, n_betas=10)
        sc_med = _score(a_med_opt)
        update_best(a_med_opt, sc_med, "n_8192")
        if sc_med < best_score and time.monotonic() < deadline - 80:
            nt4_dl = min(deadline - 80, time.monotonic() + 40)
            a_med_nt, sc_mn = _active_set_newton(a_med_opt, nt4_dl, max_outer=30)
            update_best(a_med_nt, sc_mn, "n_8192_newton")

    # === Phase 7: Diversified restarts ===
    print(f"\n[Phase 7] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        n = len(best_seq)
        for sigma, label in [(0.005, "tiny"), (0.02, "small"), (0.05, "med"), (0.1, "large")]:
            if time.monotonic() > deadline - 60:
                break
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            p_start = _softmax(z)
            r_dl = min(deadline - 50, time.monotonic() + 60)
            a_r = _lbfgs_continuation(p_start, r_dl, min_time_remaining=50.0, n_betas=8)
            update_best(a_r, _score(a_r), f"restart_{label}")

    # === Phase 8: Final polish ===
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        if time.monotonic() < deadline - 30:
            fin_dl = min(deadline - 20, time.monotonic() + 50)
            a_fin, sc_fin = _active_set_newton(best_seq, fin_dl, max_outer=30)
            update_best(a_fin, sc_fin, "final_newton")

        if time.monotonic() < deadline - 15:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 3e-9
            remaining = deadline - 15 - time.monotonic()
            if remaining > 3:
                max_iter_fin = min(1500, max(30, int(remaining / 0.04)))
                try:
                    result = minimize(
                        _lsobj, z, args=(beta_final, n),
                        jac=_lsgrad, method='L-BFGS-B',
                        options={'maxiter': max_iter_fin, 'ftol': 1e-20,
                                 'gtol': 1e-15, 'maxcor': 100, 'maxls': 200}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_tiny_beta")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
