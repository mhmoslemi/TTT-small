Uses direct active-set Newton steps, adaptive L-BFGS, and multi-n exploration.

import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)
    def grad(self, z):
        return _smograd(z, self.beta, self.n)

The idea: given the current sequence $a$ and the active set of peaks $S = \{k_1, ..., k_K\}$, we want to find a direction $d$ such that:
1. Moving in direction $d$ decreases all active peaks
2. $d$ has zero sum (to preserve the normalization)
3. $d$ is the "minimum effort" direction (minimum norm)

The linearized change in peak $k$ when moving by $d$ is:
$\Delta c_k \approx 2 \sum_i a_i d_{k-i}$ (for valid indices)

We want $\Delta c_{k_l} = \Delta$ (same decrease for all active peaks) for $l = 1, ..., K$.

This gives K equations:
$\sum_i a_i d_{k_l - i} = \Delta / 2$ for $l = 1, ..., K$

Plus the constraint $\sum d_i = 0$.

This is a system of K+1 equations in n + 1 unknowns ($d_0, ..., d_{n-1}, \Delta$). We want the minimum-norm solution.

The matrix form is:
$M \cdot [d; \Delta] = b$

where M is (K+1) × (n+1) and b = [0, 0, ..., 0, 0]^T (we want all peaks to decrease by the same amount, and the sum to be preserved).

def _active_set_newton_step(a, K=10):
    """Compute a minimum-norm direction that equalizes the top K peaks."""
    n = len(a)
    c = _full_convolve(a, a)
    
    # Top K peaks
    top_k = np.argsort(c)[::-1][:K]
    
    # Build gradient matrix G (K x n)
    # G[l, j] = a[k_l - j] if 0 <= k_l - j < n, else 0
    G = np.zeros((K, n))
    for l in range(K):
        k = top_k[l]
        # For each j, G[l, j] = a[k - j] if 0 <= k - j < n
        valid = (k - np.arange(n) >= 0) & (k - np.arange(n) < n)
        G[l, valid] = a[k - np.arange(n)[valid]]
    
    # Add sum constraint row
    A = np.vstack([G, np.ones(n)])
    b = np.concatenate([np.full(K, -1.0), [0.0]])
    
    # Minimum norm solution: d = A^T (A A^T)^{-1} b
    AT = A.T  # n x (K+1)
    AAT = A @ AT  # (K+1) x (K+1)
    try:
        AAT_inv_b = np.linalg.solve(AAT + 1e-12 * np.eye(K+1), b)
        d = AT @ AAT_inv_b
    except np.linalg.LinAlgError:
        return None
    
    return d

Minimizes 2*n*max(full_convolve(a,a))/sum(a)^2 using:
- Multi-scale L-BFGS on smooth (log-sum-exp) approximation
- Direct active-set Newton steps with full-step line search
- Adaptive active set management
- Multi-n exploration with resampling
- Final subgradient polish

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _build_G(a, peak_indices):
    """Build K x n matrix where G[l,j] = a[k_l - j] if 0 <= k_l - j < n."""
    n = len(a)
    K = len(peak_indices)
    G = np.zeros((K, n))
    for l in range(K):
        k = int(peak_indices[l])
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            G[l, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return G


def _newton_direction(a, active_peaks):
    """Compute minimum-norm direction to equalize active peaks.
    Returns direction d (unit norm) or None if failed."""
    n = len(a)
    K = len(active_peaks)
    
    G = _build_G(a, active_peaks)
    ones = np.ones(n)
    A = np.vstack([G, ones.reshape(1, -1)])
    b = np.concatenate([np.full(K, -1.0), [0.0]])
    
    AAT = A @ A.T
    try:
        AAT_inv_b = np.linalg.solve(AAT + 1e-14 * np.eye(K + 1), b)
    except np.linalg.LinAlgError:
        return None
    d = A.T @ AAT_inv_b
    
    dnorm = np.linalg.norm(d)
    if dnorm < 1e-18:
        return None
    return d / dnorm
