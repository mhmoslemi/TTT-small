"""
Multi-strategy optimizer for the step-function peak-minimization problem.
Minimizes 2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences a.

Strategy:

3. Simulated-annealing iterated local search for global exploration
4. Multi-resolution (multi-n) exploration
5. Peak-targeted active-set gradient with heavy-ball momentum for final polish
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(c)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    a_norm = a / s
    return _score_normalized(a_norm)


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _cross_corr(w, a):
    """Compute cross-correlation: result[i] = sum_k w[k] * a[k-i] (linear)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return wc[n - 1:2 * n - 1]


def _lse_objective(z, beta, n):
    """Log-sum-exp smoothed objective in log-parameterization."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    """Gradient of LSE objective w.r.t. z."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    h = _cross_corr(w, a)
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-30:
        a_new = a_new / s
    return a_new


def _make_window(n, kind):
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'hann':
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        return np.maximum(a, 1e-10) / np.maximum(a, 1e-10).sum()
    elif kind == 'gaussian':
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == 'triangle':
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        return np.maximum(a, 1e-10) / np.maximum(a, 1e-10).sum()
    elif kind == 'super_gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.5) ** 4)

    elif kind == 'cos4':
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) ** 2
        return np.maximum(a, 1e-10) / np.maximum(a, 1e-10).sum()
    else:
        return np.ones(n) / n

"""
Multi-strategy optimizer for minimizing 2*n*max(convolve(a,a))/sum(a)^2.

Combines: adaptive log-sum-exp L-BFGS, multiplicative weight (max-projection)
polish, momentum active-set gradient, segment-removal global search, and
spectrum-flatness regularization to escape symmetric local minima.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    """Score for sum(a)=1: returns 2*n*max(c)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _cross_corr(w, a):
    """result[i] = sum_k w[k] * a[k-i] (linear cross-correlation)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return wc[n - 1:2 * n - 1]


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    h = _cross_corr(w, a)
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adaptive_lse_lbfgs(a_init, deadline, min_remaining=20.0):
    """L-BFGS on log-softmax with adaptive beta scheduling."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score_norm(a)

    c0 = _full_convolve(a, a)
    c0s = np.sort(c0)[::-1]
    gap0 = c0s[0] - c0s[min(50, len(c0s) - 1)]
    if gap0 < c0s[0] * 0.001:
        gap0 = c0s[0] * 0.001
    beta = max(gap0, c0s[0] * 0.004)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    patience = 0

    while time.monotonic() < deadline - min_remaining:
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 8:
            break
        est_t = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(6000, max(50, int(remaining / (est_t * 25))))

        try:
            res = minimize(
                _lse_obj, z, args=(beta, n),
                jac=lambda zz, bb, nn: _lse_grad(zz, bb, nn),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 200, 'maxls': 100}
            )
            a_new = _softmax(res.x)
            sc_new = _score_norm(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                patience = 0
            else:
                patience += 1

            c_new = _full_convolve(best_a, best_a)
            cn = np.sort(c_new)[::-1]
            new_gap = cn[0] - cn[min(50, len(cn) - 1)]
            if new_gap < cn[0] * 0.0005:
                new_gap = cn[0] * 0.0005

            if patience < 4:
                beta *= 0.65
            else:
                beta = max(new_gap, cn[0] * 0.001)
                patience = 0
                z = np.log(np.maximum(best_a, 1e-30))
                z = z - z.mean()
        except Exception:
            beta *= 0.7

    return best_a, best_sc


def _projmax_polish(a_init, deadline, min_remaining=5.0):
    """
    Max-projection multiplicative polish (Du & Osher method).
    Finds a, with sum=1, z>=0, such that Q(a) <= z (element-wise) for the
    family of quadratic forms defining the convolution.
    """
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score_norm(a)
    lr = 0.5
    no_improve = 0

    while time.monotonic() < deadline - min_remaining:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        cur_sc = 2.0 * n * c_max

        if cur_sc < best_sc - 1e-15:
            best_sc = cur_sc
            best_a = a.copy()
            no_improve = 0

        # Active set (top peaks)
        tol = 0.02 * c_max
        active = c >= (c_max - tol)
        if not active.any():
            active[int(np.argmax(c))] = True
        eps = 1e-15 * c_max
        w = np.zeros_like(c)
        w[active] = 1.0 / (c_max - c[active] + eps)
        w = w / w.sum()

        # Gradient direction
        h = _cross_corr(w, a)
        grad = 2.0 * h
        grad = grad - np.mean(grad)

        # Max-projection step
        a_new = a + lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        sc_new = _score_norm(a_new)
        if sc_new < cur_sc - 1e-15:
            a = a_new
        else:
            lr *= 0.5
            no_improve += 1
            if no_improve % 40 == 0:
                tol *= 0.6
            if no_improve > 400:
                a = best_a.copy()
                lr = 0.3
                no_improve = 0

    return best_a, best_sc


def _momentum_active(a_init, deadline, min_remaining=5.0):
    """Active-set gradient with heavy-ball momentum and backtracking."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score_norm(a)
    v = np.zeros(n)
    tol_frac = 0.008
    lr = 0.04
    momentum = 0.82
    no_improve = 0

    while time.monotonic() < deadline - min_remaining:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        cur_sc = 2.0 * n * c_max

        if cur_sc < best_sc - 1e-15:
            best_sc = cur_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if not active.any():
            active[int(np.argmax(c))] = True
        eps = 1e-15 * c_max
        w = np.zeros_like(c)
        w[active] = 1.0 / (c_max - c[active] + eps)
        w = w / w.sum()

        h = _cross_corr(w, a)
        grad = 2.0 * h - np.mean(2.0 * h)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        v = momentum * v - lr * grad
        alpha = 1.0
        improved = False
        for _ in range(35):
            a_try = _normalize(a + alpha * v)
            sc_try = _score_norm(a_try)
            if sc_try < cur_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            v = np.zeros(n)
            no_improve += 1
            if no_improve % 30 == 0:
                tol_frac = max(0.0003, tol_frac * 0.6)
                lr *= 0.7
            if no_improve > 300:
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(0.002, tol_frac * 0.5)
                lr = 0.02
                no_improve = 0
        else:
            if tol_frac > 0.001:
                tol_frac *= 0.99
            lr = min(0.05, lr * 1.005)

    return best_a, best_sc


def _quick_opt(a_init, deadline, min_remaining=3.0):

            z = np.log(np.maximum(a_p, 1e-30)) + 0.004 * rng.standard_normal(n)
            a_p = _softmax(z - z.mean())
        elif kind == 'peakt':
            c = _full_convolve(cur_a, cur_a)
            cm = c.max()
            active = c >= cm * 0.9
            w = np.where(active, 3.0, 1.0)
            w = w / w.sum()
            h = _cross_corr(w, cur_a)
            a_p = _normalize(cur_a * np.exp(-0.025 * h))
        elif kind == 'asym':
            # Asymmetric tilt to break symmetry
            t = np.linspace(0, 1, n)
            tilt = np.exp(sigma * (2.0 * t - 1.0) * 3.0)
            a_p = _normalize(cur_a * tilt)
        else:
            a_p = cur_a.copy()

        dl = min(deadline - min_remaining, time.monotonic() + 15)
        a_opt, sc_opt = _quick_opt(a_p, dl, min_remaining=2)

        delta = sc_opt - cur_sc
        if delta < 0:
            cur_a, cur_sc = a_opt, sc_opt
            if sc_opt < best_sc - 1e-12:
                best_a, best_sc = a_opt.copy(), sc_opt
                print(f"    ILS BEST: {best_sc:.10f} ({kind}) t={time.monotonic()-t0:.0f}s")
        else:
            if rng.random() < np.exp(-delta / max(T, 1e-12)):
                cur_a, cur_sc = a_opt, sc_opt

    return best_a, best_sc


def _seg_removal_search(a_init, deadline, rng, min_remaining=60.0):
    """Remove random segments to escape symmetric local minima."""
    n = len(a_init)
    cur_a = _normalize(a_init.copy())
    cur_sc = _score_norm(cur_a)
    best_a = cur_a.copy()
    best_sc = cur_sc

    for _ in range(30):
        if time.monotonic() > deadline - min_remaining:
            break
        # Remove a random segment
        sl = int(rng.integers(n // 15, n // 5))
        st = int(rng.integers(0, n - sl))
        a_try = cur_a.copy()
        a_try[st:st+sl] = 0.0
        a_try = _normalize(a_try)

        dl = min(deadline - min_remaining, time.monotonic() + 12)
        a_opt, sc_opt = _quick_opt(a_try, dl, min_remaining=2)

        if sc_opt < best_sc - 1e-12:
            best_a, best_sc = a_opt.copy(), sc_opt
            cur_a = a_opt.copy()
            cur_sc = sc_opt
            print(f"    SEGRM BEST: {best_sc:.10f} t={time.monotonic()-deadline+deadline:.0f}s")
        elif sc_opt < cur_sc - 1e-12:
            cur_a, cur_sc = a_opt, sc_opt

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-30:
        a_new = a_new / s
    return a_new


def _make_window(n, kind):
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'hann':
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == 'gaussian':
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == 'super_gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.5) ** 4)
        return a / a.sum()
    elif kind == 'cos4':
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) ** 2
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == 'asym_hann':
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-12)
        a = a * np.exp(0.3 * (t - 0.5))
        return a / a.sum()
    else:
        return np.ones(n) / n


def _reg_asymmetry(a, lam):
    """Add asymmetric Gaussian perturbation and re-optimize."""
    n = len(a)
    rng = np.random.default_rng(777)
    center = rng.uniform(0.3, 0.7)
    width = rng.uniform(0.05, 0.2)
    t = np.linspace(0, 1, n)
    bump = np.exp(-0.5 * ((t - center) / width) ** 2)
    a_new = _normalize(a + lam * (bump - bump.sum() / n))
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: adaptive LSE L-BFGS + max-projection polish +
    momentum active-set + SA-ILS + segment removal + asymmetry breaking +
    multi-n exploration + spectrum-flatness regularization.
    """
    global t_start
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 18.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = _score_norm(parent)
        print(f"Parent: n={n_parent}, score={best_score:.10f}")
    else:
        n_parent = 16384
        best_seq = _make_window(n_parent, 'hann')
        best_score = _score_norm(best_seq)
        print(f"Default Hann n={n_parent}: score={best_score:.10f}")

    n_main = n_parent

    # Phase 1: Multi-start (200s)
    print(f"\n[P1] Multi-start (t={time.monotonic()-t_start:.0f}s)")
    p1_dl = t_start + 200
    starts = [('parent', best_seq.copy())]
    for k in ['hann', 'gaussian', 'super_gauss', 'asym_hann']:
        starts.append((k, _make_window(n_main, k)))
    for name, a0 in starts:
        if time.monotonic() > p1_dl - 20:
            break
        sc0 = _score_norm(a0)
        a_opt, sc_opt = _quick_opt(a0, min(p1_dl, time.monotonic() + 35), min_remaining=3)
        print(f"  {name}: {sc0:.10f} -> {sc_opt:.10f}")
        if sc_opt < best_score:
            best_seq, best_score = a_opt.copy(), sc_opt
            print(f"  NEW BEST: {best_score:.10f}")

    # Phase 2: Deep LSE (350s)
    print(f"\n[P2] Deep LSE (t={time.monotonic()-t_start:.0f}s)")
    p2_dl = min(t_start + 550, deadline - 450)
    a2, sc2 = _adaptive_lse_lbfgs(best_seq, p2_dl, min_remaining=20)
    if sc2 < best_score:
        best_seq, best_score = a2.copy(), sc2
        print(f"  NEW BEST: {best_score:.10f} [deep_lse]")

    # Phase 3: Projmax polish (120s)
    print(f"\n[P3] Projmax (t={time.monotonic()-t_start:.0f}s)")
    p3_dl = min(t_start + 670, deadline - 330)
    a3, sc3 = _projmax_polish(best_seq, p3_dl, min_remaining=10)
    if sc3 < best_score:
        best_seq, best_score = a3.copy(), sc3
        print(f"  NEW BEST: {best_score:.10f} [projmax]")

    # Phase 4: Momentum (80s)
    print(f"\n[P4] Momentum (t={time.monotonic()-t_start:.0f}s)")
    p4_dl = min(t_start + 750, deadline - 250)
    a4, sc4 = _momentum_active(best_seq, p4_dl, min_remaining=10)
    if sc4 < best_score:
        best_seq, best_score = a4.copy(), sc4
        print(f"  NEW BEST: {best_score:.10f} [momentum]")

    # Phase 5: SA-ILS (350s)
    print(f"\n[P5] SA-ILS (t={time.monotonic()-t_start:.0f}s)")
    p5_dl = min(t_start + 1100, deadline - 350)
    a5, sc5 = _diverse_ils(best_seq, p5_dl, rng, min_remaining=60)
    if sc5 < best_score:
        best_seq, best_score = a5.copy(), sc5
        print(f"  NEW BEST: {best_score:.10f} [ils]")

    # Phase 6: Segment removal (150s)
    print(f"\n[P6] SegRemoval (t={time.monotonic()-t_start:.0f}s)")
    p6_dl = min(t_start + 1250, deadline - 200)
    a6, sc6 = _seg_removal_search(best_seq, p6_dl, rng, min_remaining=30)
    if sc6 < best_score:
        best_seq, best_score = a6.copy(), sc6
        print(f"  NEW BEST: {best_score:.10f} [segrm]")

    # Phase 7: Asymmetry breaking + re-optimize (100s)
    print(f"\n[P7] AsymBreak (t={time.monotonic()-t_start:.0f}s)")
    p7_dl = min(t_start + 1350, deadline - 150)
    if time.monotonic() < p7_dl - 30:
        for lam in [0.01, 0.03, 0.05]:
            if time.monotonic() > p7_dl - 20:
                break
            a_asym = _reg_asymmetry(best_seq, lam)
            a_opt, sc_opt = _quick_opt(a_asym, min(p7_dl, time.monotonic() + 25), min_remaining=3)
            if sc_opt < best_score:
                best_seq, best_score = a_opt.copy(), sc_opt
                print(f"  NEW BEST: {best_score:.10f} [asym lam={lam}]")

    # Phase 8: Multi-n (150s)
    print(f"\n[P8] Multi-n (t={time.monotonic()-t_start:.0f}s)")
    p8_dl = min(t_start + 1500, deadline - 100)
    n_tries = []
    for fac in [0.5, 0.75, 1.25, 1.5, 2.0]:
        nt = int(n_main * fac)
        if 4096 <= nt <= 65536:
            n_tries.append(nt)
    for nt in [12288, 20000, 32768, 40000]:
        if nt not in n_tries and 4096 <= nt <= 65536:
            n_tries.append(nt)
    n_tries = sorted(set(n_tries))

    for nt in n_tries:
        if time.monotonic() > p8_dl - 15:
            break
        if nt == n_main:
            continue
        a_init = _resample(best_seq, nt)
        sc_init = _score_norm(a_init)
        a_opt, sc_opt = _quick_opt(a_init, min(p8_dl, time.monotonic() + 18), min_remaining=3)
        if sc_opt < best_score:
            best_seq, best_score = a_opt.copy(), sc_opt
            print(f"  NEW BEST: {best_score:.10f} [n={nt}]
")
            n_main = nt

    # Phase 9: Final polish (80s)
    print(f"\n[P9] Final (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() < deadline - 10:
        a9a, sc9a = _adaptive_lse_lbfgs(best_seq, deadline - 12, min_remaining=8)
        if sc9a < best_score:
            best_seq, best_score = a9a.copy(), sc9a
            print(f"  NEW BEST: {best_score:.10f} [final_lse]")
        if time.monotonic() < deadline - 5:
            a9b, sc9b = _projmax_polish(best_seq, deadline - 3, min_remaining=3)
            if sc9b < best_score:
                best_seq, best_score = a9b.copy(), sc9b
                print(f"  NEW BEST: {best_score:.10f} [final_projmax]")

    # Final
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_window(256, 'uniform')
        sc_f = _score_norm(final)
        print(f"FINAL: score={sc_f:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fb = _make_window(256, 'uniform')
    print(f"FALLBACK: score={_score_norm(fb):.10f}")
    return [float(x) for x in fb]
