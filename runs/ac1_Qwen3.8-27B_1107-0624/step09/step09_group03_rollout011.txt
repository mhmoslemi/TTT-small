    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.005)
    log_lo = np.log(max_c * 1e-9)
    log_hi = np.log(base_beta)
    log_betas = np.linspace(log_hi, log_lo, n_betas)
    betas = np.exp(log_betas)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        _g_beta = beta
        est_time = 0.004 * max(1.0, n / 16384.0)
        max_iter = min(10000, max(100, int(remaining / (est_time * 2))))
        try:
            result = minimize(
                _lsobj_scipy, z, jac=_lsgrad_scipy,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxcor': 200, 'maxls': 500}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            if i % 10 == 0 or i == n_betas - 1:
                print(f"      L-BFGS beta={beta:.2e} score={sc_new:.12f}")
        except Exception as ex:
            print(f"      L-BFGS beta={beta:.2e} err={ex}")
    return current_a
