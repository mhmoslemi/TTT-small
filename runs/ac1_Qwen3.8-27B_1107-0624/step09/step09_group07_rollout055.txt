import numpy as np

from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _topk_avg_obj_grad(z, K, n):
    """Top-K average smoothed objective and gradient in log-domain."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
    else:
        top_idx = np.arange(len(c))
    
    c_top = c[top_idx]
    obj = 2.0 * n * c_top.mean()
    
    w = np.zeros(2 * n - 1)
    w[top_idx] = 1.0 / len(top_idx)
    
    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z

    def grad(self, z):
        return _topk_avg_obj_grad(z, self.K, self.n)[1]

                z = z - z.mean()
                if verbose:
                    print(f"    L-BFGS K={K} ({method}): score={current_sc:.10f}")
        except Exception as e:
            if verbose:
                print(f"    L-BFGS error: {e}")
    
    return current_a


def _peak_shave_lp(a_init, deadline, min_time_remaining=30.0, P=15):
    """LP-based peak shaving: find descent direction reducing top P peaks."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    c = _full_convolve(a, a)
    n_c = 2 * n - 1
    
    iteration = 0
    stall = 0
    max_stall = 50
    
    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                break
        
        # Get top P peak indices
        top_idx = np.argpartition(c, -P)[-P:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        
        # Build LP: find direction d such that:
        #   d . grad_k < 0 for each top peak k (reduces peak)
        #   sum d = 0 (preserves sum)
        #   ||d||_inf <= 1 (bounded step)
        #   Maximize -sum d . grad_k (maximal reduction)
        
        # Gradient of b[k] w.r.t. a: grad_k[j] = a[k-j] + a[j+k-...] 
        # Use _grad_maxconv with w = e_k
        grads = np.zeros((P, n))
        for ii, kk in enumerate(top_idx):
            w = np.zeros(n_c)
            w[kk] = 1.0
            grads[ii] = _grad_maxconv(a, w)
        
        # We want to maximize sum_i alpha_i * (grads[i] . d) where alpha_i > 0
        # Subject to: grad_i . d <= -epsilon (reduce each peak)
        #              sum d = 0
        #              -1 <= d_j <= 1
        
        # Simplified: minimize (sum of top P peaks) in direction d
        # = minimize sum_i (b[k_i] + t * grads[i].d)
        # So we want d that minimizes sum_i grads[i].d
        # = minimize c_vec . d where c_vec = sum_i grads[i]
        
        c_vec = grads.sum(axis=0)
        c_vec_centered = c_vec - c_vec.mean()
        
        # LP: minimize c_vec_centered . d
        #     subject to sum d = 0
        #                -1 <= d <= 1
        #     (ignore non-negativity of a+td for now, check later)
        
        # Actually, let's just use the centered gradient direction with a
        # step that respects non-negativity
        d = -c_vec_centered
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm
        
        # Line search with non-negativity
        max_step = min(a) / (np.abs(d).max() + 1e-15)
        t_try = min(0.01, max_step * 0.9)
        
        improved = False
        for ls in range(30):
            a_try = a + t_try * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                t_try *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score_normalized(a_try)
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            t_try *= 0.5
        
        if not improved:
            # Try random direction in the null space of sum
            rng_local = np.random.default_rng(iteration)
            d_rand = rng_local.standard_normal(n)
            d_rand -= d_rand.mean()
            d_rand /= (np.linalg.norm(d_rand) + 1e-15)
            
            for ls in range(15):
                t_try = 0.005
                a_try = a + t_try * d_rand
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    sc_try = _score_normalized(a_try)
                    if sc_try < current_sc - 1e-15:
                        a = a_try
                        improved = True
                        break
                t_try *= 0.5
            
            if not improved:
                break
    
    return best_a, best_sc


def _spectral_perturb(a, frequency, amplitude, rng):
    """Add a sinusoidal perturbation in the frequency domain."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    # Add a sinusoidal component
    t = np.arange(n)
    perturbation = amplitude * np.sin(2.0 * np.pi * frequency * t / n + rng.uniform(0, 2 * np.pi))
    z_pert = z + perturbation
    z_pert = z_pert - z_pert.max()
    a_pert = np.exp(z_pert)
    return a_pert / a_pert.sum()


def _subgradient_nesterov(a_init, deadline, min_time_remaining=60.0):
    """Subgradient method with Nesterov acceleration and adaptive step sizes."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    y = a.copy()  # Nesterov look-ahead point
    best_a = a.copy()
    best_sc = _score(a)
    
    n_c = 2 * n - 1
    active_frac = 0.02
    lr = 0.03
    lr_decay = 0.999
    momentum = 0.9
    stall = 0
    max_stall = 100
    
    k = 0
    while time.monotonic() + min_time_remaining < deadline:
        k += 1
        # Nesterov update: y = a + momentum * (a - a_prev)
        # Simplified: just use momentum on the step
        c = _full_convolve(y, y)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Select active peaks
        n_active = max(5, min(n_c, int(n_c * active_frac)))
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_active]
        
        # Weighted gradient
        active_vals = c[active_idx]
        w = np.zeros(n_c)
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w[active_idx] = ((active_vals - active_vals.min()) / spread) ** 2 + 0.01
        else:
            w[active_idx] = 1.0 / n_active
        w = w / w.sum()
        
        g = _grad_maxconv(y, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-16:
            break
        
        # Adaptive step size (Nesterov-style)
        tk = 1.0 / (k ** 0.5)
        step = lr * tk * 10.0
        step = min(step, 0.5)
        
        a_new = y - step * g_centered / gnorm
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        # Nesterov: a = a_new, y = a + momentum * (a - a_prev)
        step_vec = a_new - a
        a = a_new
        y = a + momentum * step_vec
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-10:
            y = y / sy
        
        # Adapt parameters
        if stall > max_stall:
            active_frac = min(0.1, active_frac * 1.5)
            lr *= 0.5
            momentum = min(0.95, momentum + 0.02)
            stall = 0
            if lr < 1e-4:
                break
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "optimal_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.25) ** 2) + np.exp(-30.0 * (t - 0.75) ** 2) + 0.02
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":
        t = np.linspace(0, 1, n)
        # Flat in the middle, smooth taper at edges
        edge = 0.15
        a = np.ones(n)
        for i in range(n):
            if t[i] < edge:
                a[i] = 0.5 + 0.5 * np.cos(np.pi * t[i] / edge)
            elif t[i] > 1 - edge:
                a[i] = 0.5 + 0.5 * np.cos(np.pi * (t[i] - (1 - edge)) / edge)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _coordinate_descent(a_init, deadline, min_time_remaining=20.0):
    """Cyclic coordinate descent on log-space with sum constraint."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    no_improve = 0
    max_no_improve = 5 * n
    
    order = np.arange(n)
    
    while time.monotonic() + min_time_remaining < deadline:
        np.random.shuffle(order)
        improved_this_pass = False
        
        for idx in order:
            if time.monotonic() + min_time_remaining > deadline:
                break
            
            # Optimize z[idx] with all others fixed
            # a = exp(z) / sum(exp(z))
            # We want to minimize max(conv(a,a))
            
            z_base = z.copy()
            z_base[idx] = 0  # baseline
            
            # Find optimal z[idx] by bisection/line search
            # a[idx] = exp(z[idx]) / (exp(z[idx]) + rest_sum)
            # where rest_sum = sum of exp(z[j]) for j != idx
            
            exp_others = np.exp(z - z.max())
            exp_others[idx] = 0
            rest = exp_others.sum()
            
            # a[idx] = exp(z[idx]-z.max()) / (exp(z[idx]-z.max()) + rest)
            # Let u = exp(z[idx]-z.max()), then a[idx] = u/(u+rest)
            # So a[idx] ranges from 0 (u=0) to 1 (u→∞)
            # We want to find a[idx] in [0, 1] that minimizes the objective.
            
            # Sample a few candidate values
            current_a_idx = a[idx]
            candidates = [
                current_a_idx * 0.5, current_a_idx * 0.8, current_a_idx,
                current_a_idx * 1.2, current_a_idx * 1.5,
                current_a_idx * 0.3, current_a_idx * 2.0
            ]
            candidates = [max(1e-12, min(1.0 - 1e-12, c)) for c in candidates]
            
            best_new_sc = _score_normalized(a)
            best_new_val = current_a_idx
            
            for val in candidates:
                if val <= 0 or val >= 1:
                    continue
                a_try = a.copy()
                a_try[idx] = val
                a_try = a_try / a_try.sum()
                sc_try = _score_normalized(a_try)
                if sc_try < best_new_sc:
                    best_new_sc = sc_try
                    best_new_val = val
            
            if best_new_val != current_a_idx:
                a[idx] = best_new_val
                a = a / a.sum()
                z = np.log(np.maximum(a, 1e-30))
                z = z - z.mean()
                improved_this_pass = True
        
        current_sc = _score(a)
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += n
            if no_improve > max_no_improve:
                break
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS with top-K smoothing,
    subgradient with Nesterov acceleration, peak shaving,
    spectral perturbations, and coordinate descent.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "flat_taper")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, flat_taper n={n_parent}, score={best_score:.10f}")
    
    n_main = len(best_seq)
    
    # === Phase 1: Coarse L-BFGS (large K, log-sum-exp) ===
    print(f"\n[P1] Coarse L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        a1 = _run_lbfgs_topk_avg(best_seq, min(deadline - 550, time.monotonic() + 300),
                                  min_time_remaining=550)
        sc1 = _score(a1)
        if sc1 < best_score:
            best_seq = a1.copy()
            best_score = sc1
            print(f"  NEW BEST: {best_score:.10f} [P1] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 2: Subgradient with Nesterov ===
    print(f"\n[P2] Nesterov subgradient (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a2, sc2 = _subgradient_nesterov(best_seq, min(deadline - 350, time.monotonic() + 250),
                                         min_time_remaining=60)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [P2] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 3: Fine L-BFGS (small K, top-K average) ===
    print(f"\n[P3] Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a3 = _run_lbfgs_topk_avg(best_seq, min(deadline - 250, time.monotonic() + 200),
                                  min_time_remaining=40)
        sc3 = _score(a3)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [P3] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 4: Peak shaving ===
    print(f"\n[P4] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a4, sc4 = _peak_shave_lp(best_seq, min(deadline - 150, time.monotonic() + 150),
                                   min_time_remaining=30, P=20)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [P4] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 5: Multi-restart with spectral perturbations ===
    print(f"\n[P5] Spectral restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        n_restarts = 6
        restart_time = min(25, (deadline - 150 - time.monotonic()) / max(1, n_restarts))
        
        for i in range(n_restarts):
            if time.monotonic() > deadline - 100:
                break
            
            # Different perturbation strategies
            if i < 3:
                # Frequency perturbation
                freq = rng.uniform(0.5, 5.0)
                amp = 0.05 + 0.03 * i
                a_pert = _spectral_perturb(best_seq, freq, amp, rng)
            else:
                # Gaussian perturbation
                sigma = 0.03 + 0.02 * (i - 3)
                a_pert = _perturb(best_seq, sigma, rng)
            
            a_r = _run_lbfgs_topk_avg(a_pert, min(deadline - 80, time.monotonic() + restart_time),
                                       min_time_remaining=5)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.10f} [restart_{i}] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 6: Coordinate descent polish ===
    print(f"\n[P6] Coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        a6, sc6 = _coordinate_descent(best_seq, min(deadline - 60, time.monotonic() + 80),
                                       min_time_remaining=10)
        if sc6 < best_score:
            best_seq = a6.copy()
            best_score = sc6
            print(f"  NEW BEST: {best_score:.10f} [P6] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        n_options = [16384, 8192, 32768, 24576]
        n_options = [x for x in n_options if x != n_main]
        
        for n_try in n_options[:3]:
            if time.monotonic() > deadline - 40:
                break
            a_res = _resample(best_seq, n_try)
            a_opt = _run_lbfgs_topk_avg(a_res, min(deadline - 20, time.monotonic() + 35),
                                         min_time_remaining=5)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [n={n_try}] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 8: Final fine L-BFGS ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a8 = _run_lbfgs_topk_avg(best_seq, deadline - 2, min_time_remaining=2)
        sc8 = _score(a8)
        if sc8 < best_score:
            best_seq = a8.copy()
            best_score = sc8
            print(f"  NEW BEST: {best_score:.10f} [P8] t={time.monotonic()-t_start:.1f}s")
    
    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "flat_taper")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "flat_taper")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
