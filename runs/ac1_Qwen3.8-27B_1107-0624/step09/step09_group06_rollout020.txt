"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) n-exploration, (2) aggressive smooth-max L-BFGS, 
(3) iterative peak-shaving, (4) parametric restarts, (5) final polish.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:

import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)

        j_lo2 = max(0, k)
        j_hi2 = min(n - 1, k + n - 1)
        if j_lo2 <= j_hi2:
            indices = np.arange(j_lo2, j_hi2 + 1) - k
            grad[j_lo2:j_hi2 + 1] += a[indices]
    return grad


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _SmoothMaxProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_adaptive_lbfgs(a_init, deadline, min_remaining=30.0, label=""):
    """L-BFGS with adaptive beta based on actual convolution flatness."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)

    c_sorted = np.sort(c_cur)[::-1]
    n_peaks = min(50, len(c_sorted) - 1)
    gap = max_c - c_sorted[n_peaks]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    n_beta_steps = 20
    scale_factors = np.array([1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.06, 0.04,
                              0.025, 0.015, 0.009, 0.005, 0.003, 0.0018,
                              0.001, 0.0006, 0.00035, 0.0002, 0.0001, 0.00005])

    for i, sf in enumerate(scale_factors):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        beta = base_beta * sf
        est_time_per_iter = 0.003 * max(1.0, n / 8192.0)
        if i < n_beta_steps // 3:
            max_iter = min(3000, max(50, int(remaining / (est_time_per_iter * 15))))
        elif i < 2 * n_beta_steps // 3:
            max_iter = min(4000, max(30, int(remaining / (est_time_per_iter * 10))))
        else:
            max_iter = min(5000, max(20, int(remaining / (est_time_per_iter * 8))))

        prob = _SmoothMaxProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxcor': 100, 'maxls': 60}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 5 == 0:
                    print(f"    {label} LBFGS step {i}: score={current_sc:.10f}, beta={beta:.3e}")
            else:
                # Try with the old z (don't update if no improvement)
                pass
        except Exception:
            pass

    return current_a, current_sc


def _topk_subgradient(a_init, deadline, min_remaining=30.0, K_init=20, K_max=500,
                      label="", seed=0):
    """Top-K subgradient method with adaptive K and Nesterov momentum."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    momentum = 0.8
    lr = 0.003
    K = K_init
    t = 0
    no_improve = 0

    while time.monotonic() + min_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive K
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
        if gap < c_max * 0.003 and K < K_max:
            K = min(K_max, int(K * 1.25) + 1)
        elif gap > c_max * 0.05 and K > K_init:
            K = max(K_init, int(K * 0.7))

        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]
        grad = _conv_grad_for_indices(a, active)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-18:
            break

        effective_lr = lr / (gnorm + 1e-18) * 0.05
        v = momentum * v - effective_lr * grad_t
        a_cand = a + v
        a_proj = _project_simplex(a_cand)

        c_new = _full_convolve(a_proj, a_proj)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new <= current_sc:
            a = a_proj
        else:
            v = 0.3 * v
            lr *= 0.95
            # Half step
            a_half = a + 0.5 * v
            a_proj = _project_simplex(a_half)
            a = a_proj

        if no_improve % 300 == 0 and no_improve > 0:
            if no_improve > 3000:
                break
            # Perturb
            scale = 0.002 * a.max()
            perturb = scale * rng.standard_normal(n)
            a_pert = _project_simplex(a + perturb)
            c_pert = _full_convolve(a_pert, a_pert)
            sc_pert = 2.0 * n * float(c_pert.max())
            if sc_pert < best_sc:
                a = a_pert
                best_a = a.copy()
                best_sc = sc_pert
            v = np.zeros(n)
            no_improve = 0

        if t % 1000 == 0:
            print(f"    {label} TopK t={t}: best={best_sc:.10f}, K={K}, lr={lr:.6f}")

    return best_a, best_sc


def _resample_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_parametric_profile(n, kind, **kwargs):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t - np.pi))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t - np.pi))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "flat_taper":
        taper_frac = kwargs.get('taper_frac', 0.1)
        taper = max(1, int(n * taper_frac))
        a = np.ones(n)
        for i in range(taper):
            frac = (i + 1) / taper
            a[i] = frac
            a[n - 1 - i] = frac
        return a / a.sum()
    elif kind == "polynomial_taper":
        p = kwargs.get('p', 2)
        t = np.linspace(-1, 1, n)
        a = (1 - t**2) ** p
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "gaussian":
        sigma = kwargs.get('sigma', 0.3)
        t = np.linspace(-1, 1, n)
        a = np.exp(-t**2 / (2 * sigma**2))
        return a / a.sum()
    elif kind == "fourier":
        n_harmonics = kwargs.get('n_harm', 3)
        amps = kwargs.get('amps', None)
        if amps is None:
            amps = np.array([0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.003, 0.001, 0.0005])[:n_harmonics]
        else:
            amps = np.array(amps)
        t = np.arange(n)
        a = np.ones(n)
        for i, amp in enumerate(amps):
            a += amp * np.cos(2 * np.pi * (i + 1) * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _peak_shave(a_init, deadline, min_remaining=20.0, label=""):
    """Directly shave the top peaks by reducing contributors."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    lr = 0.01
    no_improve = 0

    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
            if no_improve % 100 == 0:
                lr *= 0.7

        # Find top peaks
        n_peaks = max(5, int(n * 0.002))
        c_idx = np.argsort(c)[::-1]
        active_k = c_idx[:n_peaks]

        # Compute gradient (which a[j] contribute to peaks)
        grad = _conv_grad_for_indices(a, active_k)
        # Normalize
        gmax = grad.max()
        if gmax < 1e-15:
            break

        # Reduce high-gradient elements, increase low-gradient elements
        # using a softmax-like redistribution
        g_norm = grad / gmax
        # Elements to reduce: top 5% by gradient
        n_reduce = max(10, int(n * 0.05))
        n_increase = max(10, int(n * 0.05))
        top_idx = np.argsort(g_norm)[::-1][:n_reduce]
        bot_idx = np.argsort(g_norm)[:n_increase]

        # Compute move amount
        total_reduce = a[top_idx].sum()
        total_increase_room = (a.max() - a[bot_idx]).clip(min=0).sum()
        if total_reduce < 1e-15:
            break
        move_frac = min(lr, 0.1)
        move_amount = move_frac * total_reduce / len(top_idx)

        # Apply
        for j in top_idx:
            a[j] = max(0, a[j] - move_amount)
        excess = move_amount * len(top_idx)
        # Distribute to bottom indices proportionally to their "room"
        room = (a.max() * 0.5 - a[bot_idx]).clip(min=1e-10)
        room_frac = room / room.sum()
        a[bot_idx] += excess * room_frac

        # Renormalize
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

    return best_a, best_sc


def _quick_optimize(a_init, time_budget):
    """Quick optimization: short L-BFGS + short topK."""
    t0 = time.monotonic()
    deadline = t0 + time_budget

    # Quick L-BFGS
    if time.monotonic() < deadline - 10:
        a_lb, sc_lb = _run_adaptive_lbfgs(a_init, deadline, min_remaining=10, label="quick")
        if sc_lb < _score(a_init):
            a_init = a_lb

    # Quick topK
    if time.monotonic() < deadline - 5:
        a_k, sc_k = _topk_subgradient(a_init, deadline, min_remaining=5,
                                       K_init=10, K_max=50, label="quick_k",
                                       seed=123)
        if sc_k < _score(a_init):
            a_init = a_k

    return a_init


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining adaptive L-BFGS, top-K subgradient,
    peak shaving, n-exploration, and parametric restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_parametric_profile(n_parent, "flat_taper", taper_frac=0.05)
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, flat_taper n={n_parent}, score={best_score:.10f}")

    # === Phase 1: n-exploration (quick) ===
    print(f"\n[P1] n-exploration (t={time.monotonic()-t_start:.1f}s)")
    n_options = [8192, 16384, 24576, 32768]
    if n_parent not in n_options:
        n_options.insert(0, n_parent)

    # Remove the current best n (we'll optimize it in phase 2)
    n_to_try = [n for n in n_options if n != len(best_seq)]
    n_to_try = n_to_try[:3]  # Limit to 3 extra n values

    for n_try in n_to_try:
        if time.monotonic() > deadline - 600:
            break
        a_try = _resample_seq(best_seq, n_try)
        sc_try = _score(a_try)
        print(f"  n={n_try}: resampled score={sc_try:.10f}")
        # Quick optimize
        time_left = deadline - time.monotonic()
        alloc = min(60.0, time_left * 0.05)
        if alloc > 20:
            a_opt = _quick_optimize(a_try, alloc)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                n_parent = n_try
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [exploration]")

    # === Phase 2: Main aggressive L-BFGS ===
    print(f"\n[P2] Main L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p2_alloc = min(500.0, (deadline - t_start) * 0.35)
    p2_deadline = t_start + p2_alloc
    a_main, sc_main = _run_adaptive_lbfgs(best_seq, p2_deadline,
                                           min_remaining=30.0, label="main")
    if sc_main < best_score:
        best_seq = a_main.copy()
        best_score = sc_main
        print(f"  NEW BEST: {best_score:.10f} [main_lbfgs]")
    else:
        print(f"  Main L-BFGS: no improvement (best={best_score:.10f})")

    # === Phase 3: Top-K subgradient ===
    print(f"\n[P3] Top-K subgradient (t={time.monotonic()-t_start:.1f}s)")
    p3_alloc = min(300.0, (deadline - t_start) * 0.2)
    p3_deadline = t_start + p3_alloc
    a_k, sc_k = _topk_subgradient(best_seq, p3_deadline, min_remaining=20.0,
                                   K_init=15, K_max=300, label="topk", seed=seed)
    if sc_k < best_score:
        best_seq = a_k.copy()
        best_score = sc_k
        print(f"  NEW BEST: {best_score:.10f} [topk]")
    else:
        print(f"  Top-K: no improvement (best={best_score:.10f})")

    # === Phase 4: Second L-BFGS (finer) ===
    print(f"\n[P4] Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        p4_deadline = min(deadline - 200, t_start + (deadline - t_start) * 0.6)
        a_fine, sc_fine = _run_adaptive_lbfgs(best_seq, p4_deadline,
                                               min_remaining=30.0, label="fine")
        if sc_fine < best_score:
            best_seq = a_fine.copy()
            best_score = sc_fine
            print(f"  NEW BEST: {best_score:.10f} [fine_lbfgs]")

    # === Phase 5: Peak shaving ===
    print(f"\n[P5] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        p5_deadline = min(deadline - 150, t_start + (deadline - t_start) * 0.7)
        a_shave, sc_shave = _peak_shave(best_seq, p5_deadline, min_remaining=20.0,
                                         label="shave")
        if sc_shave < best_score:
            best_seq = a_shave.copy()
            best_score = sc_shave
            print(f"  NEW BEST: {best_score:.10f} [shave]")

    # === Phase 6: Parametric restarts ===
    print(f"\n[P6] Parametric restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        n_best = len(best_seq)
        profiles = [
            ("flat_taper", {'taper_frac': 0.03}),
            ("polynomial_taper", {'p': 3}),
            ("polynomial_taper", {'p': 4}),
            ("fourier", {'n_harm': 5}),
            ("gaussian", {'sigma': 0.25}),
        ]
        for kind, kwargs in profiles:
            if time.monotonic() > deadline - 80:
                break
            a_param = _make_parametric_profile(n_best, kind, **kwargs)
            # Blend with current best (50/50)
            a_blend = 0.5 * best_seq + 0.5 * a_param
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()
            sc_blend = _score(a_blend)
            if sc_blend < best_score:
                print(f"  Blend {kind}: {sc_blend:.10f}")
                best_seq = a_blend.copy()
                best_score = sc_blend
            # Quick optimize the blend
            if time.monotonic() < deadline - 60:
                a_opt = _quick_optimize(a_blend, min(30.0, deadline - time.monotonic() - 40))
                sc_opt = _score(a_opt)
                if sc_opt < best_score:
                    best_seq = a_opt.copy()
                    best_score = sc_opt
                    print(f"  NEW BEST: {best_score:.10f} [param_restart {kind}]")

    # === Phase 7: Final L-BFGS polish ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_fin, sc_fin = _run_adaptive_lbfgs(best_seq, deadline, min_remaining=10.0,
                                             label="final")
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_lbfgs]")

    # === Phase 8: Final top-K polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_fin_k, sc_fin_k = _topk_subgradient(best_seq, deadline, min_remaining=5.0,
                                               K_init=5, K_max=50, label="final_k",
                                               seed=seed + 999)
        if sc_fin_k < best_score:
            best_seq = a_fin_k.copy()
            best_score = sc_fin_k
            print(f"  NEW BEST: {best_score:.10f} [final_topk]")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_parametric_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_parametric_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
