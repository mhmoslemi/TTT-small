    if s < 1e-300:
        return np.ones(n) / n

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]

        a = a_new

    return a


def _run_active_set_qp(a_init, deadline, min_time_remaining=60.0, verbose=True):
    global t_start_global
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    N = _next_pow2(3 * n)

    if verbose:
        print(f"  Active-set QP: n={n}, initial score={best_sc:.10f}")

    tol_active = 0.005
    min_tol_active = 0.00005
    no_improve = 0
    max_no_improve = 30
    momentum = 0.7

    for outer in range(500):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        s = a.sum()
        current_sc = 2.0 * n * c_max / (s * s)

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and (no_improve == 0 or no_improve % 15 == 0):
            elapsed = time.monotonic() - t_start_global
            print(f"    AS-QP: score={current_sc:.10f}, best={best_sc:.10f}, "
                  f"n_active_tol={tol_active:.6f}, imp={no_improve}, t={elapsed:.0f}s")

        if no_improve > max_no_improve:
            if tol_active <= min_tol_active * 1.05:
                break
            tol_active = max(min_tol_active, tol_active * 0.6)
            no_improve = 0
            momentum = min(momentum + 0.05, 0.95)

        threshold = c_max * (1.0 - tol_active)
        active = c > threshold
        n_active = int(active.sum())

        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros(len(c), dtype=bool)
            active[idx] = True
            n_active = 1

        temp = max(tol_active * c_max * 0.5, 1e-12)
        lam = np.zeros(len(c))
        c_active = c[active]
        lam_active = np.exp((c_active - c_max) / temp)
        lam_active /= lam_active.sum()
        lam[active] = lam_active

        a_qp = _fista_qp(a, lam, n, N, max_iter=200, tol=1e-9)
        a_new = momentum * a + (1.0 - momentum) * a_qp
        a_new = _project_simplex(a_new)
        a = a_new

        tol_active = max(min_tol_active, tol_active * 0.99)

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, verbose=True):
    global t_start_global
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < c0_max * 0.002:
        gap = c0_max * 0.002

    multipliers = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = gap * np.array(multipliers)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        size_factor = max(1.0, n / 8192.0)
        max_iter = min(5000, max(200, int(remaining / (0.003 * size_factor * 15))))

        def obj(z_in):
            a_s = _softmax_fn(z_in)
            c = _full_convolve(a_s, a_s)
            return 2.0 * n * _soft_max_val(c, beta)

        def grad(z_in):
            a_s = _softmax_fn(z_in)
            c = _full_convolve(a_s, a_s)
            w = _soft_max_wt(c, beta)
            a_rev = a_s[::-1]
            wc = _full_convolve(w, a_rev)
            grad_a = 2.0 * wc[n - 1:2 * n - 1]
            W = float(np.dot(grad_a, a_s))
            return a_s * (grad_a - W)

        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14, 'maxcor': 80, 'maxls': 50}
            )
            a_new = _softmax_fn(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    elapsed = time.monotonic() - t_start_global
                    print(f"    LBFGS beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}, t={elapsed:.0f}s")
        except Exception:
            pass

    return best_a, best_sc


def _softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_val(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_wt(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _subgradient_refine(a_init, deadline, min_time_remaining=30.0):
    global t_start_global
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lb = 2.0 * n / (2.0 * n - 1.0)
    step_scale = 0.3
    t_iter = 0

    while time.monotonic() + min_time_remaining < deadline and t_iter < 5000:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()

        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        gap = current_sc - lb
        if gap < 1e-12:
            break
        grad_c = grad - np.mean(grad)
        g_norm_sq = float(np.dot(grad_c, grad_c))
        if g_norm_sq < 1e-25:
            break
        step = min(step_scale * gap / g_norm_sq, 0.05)

        a_new = a - step * grad_c
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a = a_new / s_new
        t_iter += 1

        if t_iter % 200 == 0:
            step_scale *= 0.9

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    global t_start_global
    t_start_global = time.monotonic()
    deadline = t_start_global + max(1.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = np.inf

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(16384, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n=16384: score={best_score:.10f}")

    n_main = len(best_seq)

    # Phase 1: Active-set QP (main workhorse)
    print(f"\n[P1] Active-set QP (n={n_main}, t={time.monotonic()-t_start_global:.0f}s)")
    p1_end = t_start_global + min(500, (budget_s - 5) * 0.30)
    a1, sc1 = _run_active_set_qp(best_seq, p1_end, min_time_remaining=300, verbose=True)
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  P1 result: {best_score:.10f}")

    # Phase 2: L-BFGS with adaptive beta
    print(f"\n[P2] L-BFGS (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p2_end = min(deadline - 250, time.monotonic() + 350)
        a2, sc2 = _run_lbfgs(best_seq, p2_end, min_time_remaining=150, verbose=True)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(f"  P2 result: {best_score:.10f}")

    # Phase 3: Active-set QP again (tighter)
    print(f"\n[P3] Active-set QP round 2 (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p3_end = min(deadline - 200, time.monotonic() + 300)
        a3, sc3 = _run_active_set_qp(best_seq, p3_end, min_time_remaining=100, verbose=True)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3 result: {best_score:.10f}")

    # Phase 4: Subgradient refinement
    print(f"\n[P4] Subgradient (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p4_end = min(deadline - 150, time.monotonic() + 120)
        a4, sc4 = _subgradient_refine(best_seq, p4_end, min_time_remaining=50)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4 result: {best_score:.10f}")

        if time.monotonic() + 20 < deadline:
            p4b_end = min(deadline - 80, time.monotonic() + 80)
            a4b, sc4b = _run_lbfgs(best_seq, p4b_end, min_time_remaining=30, verbose=False)
            if sc4b < best_score:
                best_score = sc4b
                best_seq = a4b.copy()
                print(f"  P4b result: {best_score:.10f}")

    # Phase 5: Multi-n exploration
    print(f"\n[P5] Multi-n exploration (t={time.monotonic()-t_start_global:.0f}s)")
    n_options = [8192, 12288, 20480, 24576]
    n_options = [x for x in n_options if x != n_main]
    for n_try in n_options:
        if time.monotonic() + 80 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if verbose_flag(sc_r, best_score):
            print(f"  Resampled n={n_try}: raw score={sc_r:.10f}")
        if time.monotonic() + 60 > deadline:
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
            continue
        p5_end = min(deadline - 60, time.monotonic() + 90)
        a5, sc5 = _run_active_set_qp(a_r, p5_end, min_time_remaining=30, verbose=False)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  n={n_try}: improved to {best_score:.10f}")

    # Phase 6: Diverse profile starts
    print(f"\n[P6] Diverse profiles (t={time.monotonic()-t_start_global:.0f}s)")
    profiles = ["periodic", "bimodal", "cubic_taper", "gaussian"]
    n_for_profiles = len(best_seq)
    for prof in profiles:
        if time.monotonic() + 40 > deadline:
            break
        a_p = _make_profile(n_for_profiles, prof)
        sc_p = _score(a_p)
        if sc_p < best_score + 0.3:
            p6_end = min(deadline - 40, time.monotonic() + 60)
            a_p_opt, sc_p_opt = _run_active_set_qp(a_p, p6_end, min_time_remaining=20, verbose=False)
            if sc_p_opt < best_score:
                best_score = sc_p_opt
                best_seq = a_p_opt.copy()
                print(f"  Profile {prof}: {best_score:.10f}")

    # Phase 7: Final polish
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 15 < deadline:
        p7_end = min(deadline - 3, time.monotonic() + 80)
        a7, sc7 = _run_lbfgs(best_seq, p7_end, min_time_remaining=3, verbose=False)
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()

        if time.monotonic() + 10 < deadline:
            p7b_end = min(deadline - 2, time.monotonic() + 30)
            a7b, sc7b = _run_active_set_qp(best_seq, p7b_end, min_time_remaining=2, verbose=False)
            if sc7b < best_score:
                best_score = sc7b
                best_seq = a7b.copy()

    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"\n{'='*60}")
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start_global:.1f}s")
    return [float(x) for x in final]


def verbose_flag(sc, best_sc):
    return sc < best_sc
