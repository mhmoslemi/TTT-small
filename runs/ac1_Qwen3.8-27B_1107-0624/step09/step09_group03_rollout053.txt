For the spectral approach, here's a concrete algorithm:
1. Given `a`, compute `c = conv(a, a)` (zero-padded FFT)
2. Compute the "desired" autocorrelation: `c_desired = c - alpha * (c - c_mean)` where `c_mean = sum(c)/(2n-1) = (sum(a))^2/(2n-1)` and `alpha` controls the strength of flattening

   - Better: `c_new_target = c - alpha * max(0, c - c_threshold)` where threshold is some percentile
3. Compute `delta_c = c_new_target - c` (the desired change in autocorrelation)
4. Compute `delta_S = fft(delta_c)` (the corresponding change in power spectrum)
5. Update the power spectrum: `S_new = max(0, S + eta * delta_S)` where eta is a step size
6. Recover `a` from `S_new`: we need `a = ifft(A)` where `|A|^2 = S_new`. But we also need the phases!
   - We can keep the phases from the old `a`: `A_new = sqrt(S_new) * exp(i * angle(A_old))`
   - Then `a_new = ifft(A_new)`
   - Project to non-negative: `a_new = max(0, a_new)`
   - Renormalize

This is the "Gerchberg-Saxton" type algorithm (alternating projections between amplitude and phase constraints). It's used in phase recovery and might work here.

Actually, this is very similar to the "FACH" (Flattening Algorithm for Contrast) or similar methods used in optics.

def spectral_flatten(a, n_iters=100, alpha=0.1):
    n = len(a)
    N = _next_pow2(2*n-1)
    a = a.copy()
    a = a / a.sum()
    
    for it in range(n_iters):
        A = np.fft.rfft(a, N)
        S = np.abs(A)**2  # power spectrum
        c = np.fft.irfft(S, N)[:2*n-1]  # autocorrelation
        
        c_max = c.max()
        c_mean = c.sum() / (2*n-1)
        
        # Desired change: reduce peaks
        # c_target = c - alpha * (c - c_mean) = (1-alpha)*c + alpha*c_mean
        # This flattens toward the mean
        c_target = (1 - alpha) * c + alpha * c_mean
        # But don't go below 0
        c_target = np.maximum(c_target, 0)
        
        # delta_c = c_target - c
        delta_c = c_target - c
        # delta_S = fft(delta_c)
        delta_S = np.fft.rfft(delta_c, N)
        
        # Update spectrum
        S_new = S + eta * delta_S
        S_new = np.maximum(S_new, 0)
        
        # Recover a with old phases
        phase = np.angle(A)
        A_new = np.sqrt(S_new) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new
    
    return a

    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _lsobj(z_val, beta, n):
    a = _softmax(z_val)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z_val, beta, n):
    a = _softmax(z_val)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    w_norm = w / w.sum()
    grad = 2.0 * _corr_a_A(a, w_norm, n)
    W = float(np.dot(w_norm, c))
    return 4.0 * n * a * (grad - W)


def _hessvp(context, v):
    A_vec, n, mu = context
    return 2.0 * _corr_a_A(v, A_vec, n) + mu * v


def _cg_solve(context, rhs, max_cg=300, tol=1e-14):
    x = np.zeros_like(rhs)
    r = rhs - _hessvp(context, x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = _hessvp(context, p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _newton_dx(a, c, c_max, tol_frac):
    n = len(a)
    active = c >= (c_max - tol_frac * c_max)
    if not active.any():
        active[np.argmax(c)] = True
    m = int(active.sum())
    if m < 2 or m > 12 * n:
        return None, None

    A_vec = active.astype(np.float64) / m
    g = 2.0 * _corr_a_A(a, A_vec, n)
    g_t = g - np.mean(g)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-14:
        return None, None

    mu = 1e-8 / m + 1e-10
    context = (A_vec, n, mu)
    u = _cg_solve(context, g_t, max_cg=300)
    w = _cg_solve(context, np.ones(n), max_cg=300)
    w_sum = float(w.sum())

    if abs(w_sum) < 1e-15:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w

    dx = dx - np.mean(dx)
    return dx, A_vec


def _active_set_newton(a_init, deadline, max_outer=80):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        if outer < max_outer // 3:
            tol_list = [0.03, 0.015, 0.008, 0.004, 0.002]
        elif outer < 2 * max_outer // 3:
            tol_list = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
        else:
            tol_list = [0.0001, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6]

        best_improvement = 0
        best_a_new = None

        for tol_frac in tol_list:
            dx, A_vec = _newton_dx(a, c, c_max, tol_frac)
            if dx is None:
                continue

            t = 1.0
            for ls in range(60):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < c_max - 1e-14 * t:
                    improvement = c_max - f_try
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_a_new = a_try.copy()
                    break
                t *= 0.5

        if best_a_new is None:
            g_all = 2.0 * _corr_a_A(a, np.ones(2 * n - 1) / (2 * n - 1), n)
            g_t = g_all - np.mean(g_all)
            t = 0.1
            for ls in range(40):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    f_try = float(c_try.max())
                    if f_try < c_max:
                        best_a_new = a_try.copy()
                        best_improvement = c_max - f_try
                        break
                t *= 0.5

        if best_a_new is None:
            break

        a = best_a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if best_improvement < 1e-16 * c_max:
            break

    return best_a, best_sc


def _spectral_flatten(a_init, deadline, max_iters=200):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    N = _next_pow2(2 * n - 1)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, N)
        S = np.abs(A) ** 2
        c = np.fft.irfft(S, N)[:2 * n - 1]
        c_max = float(c.max())
        c_mean = float(c.sum()) / (2 * n - 1)

        # Compute flattening direction in frequency domain
        # We want to reduce peaks: target c is lower at peaks, higher at valleys
        c_target = c.copy()
        threshold = c_mean + 0.3 * (c_max - c_mean)
        excess = np.maximum(c - threshold, 0)
        deficit = np.maximum(threshold - c, 0)
        # Scale to have zero sum
        total_excess = excess.sum()
        total_deficit = deficit.sum()
        if total_excess < 1e-15 or total_deficit < 1e-15:
            break
        scale = min(1.0, total_deficit / total_excess)
        delta_c = -scale * excess + deficit * (scale * total_excess / (total_deficit + 1e-30))
        delta_c = delta_c - delta_c.sum() / (2 * n - 1)

        delta_S = np.fft.rfft(delta_c, N)
        # Only take real part for the power spectrum update
        delta_S_real = np.real(delta_S)

        # Adaptive step size
        eta = 0.01
        applied = False
        for _ in range(20):
            S_new = S + eta * delta_S_real
            if (S_new < -1e-10).any():
                eta *= 0.5
                continue
            S_new = np.maximum(S_new, 0)
            phase = np.angle(A)
            A_new = np.sqrt(S_new) * np.exp(1j * phase)
            a_new = np.fft.irfft(A_new, N)[:n]
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                eta *= 0.5
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new
                applied = True
                break
            eta *= 0.5

        if not applied:
            # Try direct gradient step fallback
            w = np.zeros(2 * n - 1)
            peak_mask = c >= 0.95 * c_max
            n_peaks = int(peak_mask.sum())
            if n_peaks == 0:
                peak_mask[np.argmax(c)] = True
                n_peaks = 1
            w[peak_mask] = 1.0 / n_peaks
            grad = 2.0 * _corr_a_A(a, w, n)
            g_t = grad - np.mean(grad)
            for lr in [0.0001, 0.0005, 0.001, 0.005]:
                a_try = a - lr * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_sc - 1e-14:
                        best_sc = sc_try
                        best_a = a_try.copy()
                        a = a_try
                        applied = True
                        break
            if not applied:
                break

    return best_a, best_sc


def _lbfgs_continuation(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    idx = min(100, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[idx]
    base_beta = max(gap * 0.2, max_c * 0.005)

    # Aggressive schedule: 30 stages from base_beta down to max_c * 1e-8
    n_betas = 30
    log_lo = np.log(base_beta)
    log_hi = max(np.log(max_c * 1e-8), -35)
    log_betas = np.linspace(log_lo, log_hi, n_betas)
    betas = np.exp(log_betas)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.002 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 20))))

        try:
            result = minimize(
                _lsobj,
                z,
                args=(beta, n),
                jac=_lsgrad,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 500,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-18:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.12f} (improve={current_sc - sc_new:.2e}) t={time.monotonic()-t_start_global:.0f}s")
        except Exception as ex:
            print(f"    L-BFGS beta={beta:.2e}: exception {ex}")

    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "wigner":
        t = np.linspace(0, 1, n)
        a = np.abs(t - 0.5) ** 0.3 + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _refinement_loop(a, deadline, max_rounds=10):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Newton step
        as_dl = min(deadline - 3, time.monotonic() + 25)
        a_newton, sc_newton = _active_set_newton(a, as_dl, max_outer=20)
        if sc_newton < best_sc:
            best_sc = sc_newton
            best_a = a_newton.copy()
            a = a_newton

        # Spectral step
        sp_dl = min(deadline - 3, time.monotonic() + 15)
        a_sp, sc_sp = _spectral_flatten(a, sp_dl, max_iters=30)
        if sc_sp < best_sc:
            best_sc = sc_sp
            best_a = a_sp.copy()
            a = a_sp

        if best_improvement_check(best_sc, a):
            break

    return best_a, best_sc


def best_improvement_check(current_sc, a):
    return False


def _kkt_check(a, tol=1e-10):
    """Check how close a is to satisfying KKT conditions."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    active = c >= (c_max - tol * c_max)
    m = int(active.sum())
    if m < 1:
        return np.inf, c_max
    A_vec = active.astype(np.float64) / m
    g = 2.0 * _corr_a_A(a, A_vec, n)
    g_t = g - np.mean(g)
    kkt_viol = np.linalg.norm(g_t) / (np.linalg.norm(g) + 1e-30)
    return kkt_viol, c_max

                    jac=_lsgrad,
                    method='L-BFGS-B',
                    options={'maxiter': max_iter_fin, 'ftol': 1e-20, 'gtol': 1e-16, 'maxcor': 100, 'maxls': 500}
                )
                a_fin3 = _softmax(result.x)
                update_best(a_fin3, _score(a_fin3), "final_tiny_beta")
            except Exception:
                pass

        # One more Newton pass
        if time.monotonic() < deadline - 3:
            a_last, sc_last = _active_set_newton(best_seq, deadline - 1, max_outer=20)
            update_best(a_last, sc_last, "last_newton")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start_global:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
