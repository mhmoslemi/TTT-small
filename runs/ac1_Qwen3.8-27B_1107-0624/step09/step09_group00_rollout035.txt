**Peak Equalization Algorithm:**

Given current `a` and convolution `c`:
1. Find the top K peaks: positions `k_1, ..., k_K` with values `c[k_1] >= ... >= c[k_K]`.
2. We want to find a perturbation `d` (with `sum(d) = 0`) that reduces all K peaks.
3. The change in peak `k_i` is approximately `d(conv(a,e_{k_i}) + conv(e_{k_i},a), d) = d^T g_i` where `g_i = ∂c[k_i]/∂a`.
4. We want `d^T g_i < 0` for all `i`.
5. This is a linear feasibility problem: find `d` such that `d^T g_i <= -ε` for all `i` and `sum(d) = 0`.
6. If feasible, take a step in direction `d`. If not, we're at a local minimum of the max.

In practice, we can solve this as:
- Minimize `max_i d^T g_i` subject to `||d|| = 1, sum(d) = 0, a + d >= 0`.
- This gives the "steepest descent" direction for the max.

A simpler approximation: use `d = -sum_i w_i * g_i` (weighted average of negative gradients) with `w_i` proportional to the "importance" of each peak.

The weights `w_i` can be:
- Uniform: `w_i = 1/K`
- Proportional to peak height: `w_i = c[k_i] / sum_j c[k_j]`
- Inverse gap: `w_i = 1/(c[k_i] - c[K+1] + ε)` (emphasize peaks that are close to the (K+1)-th peak)

One more thought: the **direct max-subgradient** method. At each step:
1. Compute `c = conv(a, a)`.
2. Find all `k` where `c[k] >= c_max - δ` (active set with tolerance δ).
3. Compute the subgradient: `g = sum_{k in active} w_k * grad_k` where `w_k` are chosen weights.
4. The "steepest descent" direction for the max is the solution to:

5. Take a step: `a_new = normalize(max(a + alpha * d, 0))`.

"""
Peak-equalization hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines direct active-set subgradient descent, L-BFGS with adaptive log-sum-exp
smoothing, peak-targeting iterated local search, and multi-scale n exploration.
Key innovations: asymmetric initialization, exact gradient computation via FFT,
Barzilai-Borwein step sizes, and peak-equalization heuristics.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, w):
    """Compute grad[m] = 2 * sum_k w[k]*a[k-m] for m=0,...,n-1.
    w has length 2n-1 (same as convolution output)."""
    n = len(a)
    result = _full_convolve(w, a[::-1])
    grad = 2.0 * result[n - 1:2 * n - 1]
    return grad


def _active_set_grad(a, c, top_k=20, tol_frac=0.01):
    """Compute weighted active-set gradient targeting top peaks."""
    n = len(a)
    c_max = float(c.max())
    
    # Find active peaks: top_k peaks within tolerance of max
    sorted_idx = np.argsort(c)[::-1]
    active_count = min(top_k, len(c))
    # Also include all peaks within tol_frac of max
    tol = tol_frac * c_max
    tol_mask = c >= (c_max - tol)
    
    active_set = set(sorted_idx[:active_count].tolist())
    active_set.update(np.where(tol_mask)[0].tolist())
    active_idx = np.array(sorted(active_set))
    
    if len(active_idx) == 0:
        active_idx = np.array([int(np.argmax(c))])
    
    # Weights: emphasize higher peaks
    c_active = c[active_idx]
    eps = 1e-15 * c_max
    w_active = 1.0 / (c_max - c_active + eps)
    w_active = w_active / w_active.sum()
    
    # Build full weight vector
    w = np.zeros(len(c))
    w[active_idx] = w_active
    
    # Compute gradient
    grad = _compute_grad(a, w)
    
    # Project onto zero-sum subspace
    grad = grad - np.mean(grad)
    
    return grad, c_max


def _peak_subgradient_descent(a_init, deadline, min_time=10.0):
    """Direct active-set subgradient descent with Barzilai-Borwein step size."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.1
    momentum = 0.8
    v = np.zeros(n)
    tol_frac_schedule = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    phase = 0
    no_improve = 0
    prev_sc = best_sc
    prev_a = a.copy()
    prev_grad = None

            break

            s_vec = a - prev_a
            y_vec = grad - prev_grad
            ys = np.dot(s_vec, y_vec)
            ss = np.dot(s_vec, s_vec)
            yy = np.dot(y_vec, y_vec)

                if abs(ys) > 1e-30:
                    lr_bb = ss / abs(ys) if np.random.random() < 0.5 else yy / ss
                    lr = 0.5 * lr + 0.5 * np.clip(lr_bb, 1e-6, 1.0)
                else:
                    lr *= 0.9
        prev_grad = grad.copy()
        prev_a = a.copy()

        alpha = 1.0
        improved = False
        for _ in range(40):

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # Gradient of smoothed objective w.r.t. z
    # d/dz [2n * (c_max + beta * log(sum exp((c-c_max)/beta)))]
    # = 2n * beta * sum_k w[k] * dc[k]/da * da/dz
    # da/dz = a * (I - a) (for softmax)
    # dc[k]/da = 2 * (conv(a, e_k) + conv(e_k, a)) -> use _compute_grad
    grad_a = _compute_grad(a, w)
    # Smoothed objective gradient w.r.t. a: 2n * (w-weighted c) -> actually:
    # The smoothed objective is 2n*(c_max + beta*log(sum exp((c-c_max)/beta)))
    # d/d(a) = 2n * sum_k w[k] * dc[k]/da = 2n * grad_a (where grad_a = _compute_grad(a,w))
    # Wait, _compute_grad already includes the factor of 2.
    # The smoothed obj gradient w.r.t. a is: 2n * sum_k w[k] * (dc/da)[k]
    # dc/da is a matrix. The column for peak k is the gradient of c[k].
    # sum_k w[k] * grad(c[k]) = _compute_grad(a, w) / 2 * 2n? 
    # Let me recheck: _compute_grad(a, w) = 2 * sum_k w[k] * a[k-m]
    # And dc[k]/da[m] = a[k-m] + a[m-k]
    # So sum_k w[k] * dc[k]/da[m] = sum_k w[k]*(a[k-m]+a[m-k])
    # _compute_grad gives 2*sum_k w[k]*a[k-m]
    # By symmetry, sum_k w[k]*a[m-k] = sum_k w[k]*a[k-m] (same sum, just variable rename)
    # So sum_k w[k]*dc[k]/da[m] = _compute_grad(a,w)[m]
    # Therefore: d(smoothed_obj)/da = 2n * _compute_grad(a, w)
    # And d(smoothed_obj)/dz = d(smoothed_obj)/da * da/dz
    # da/dz[j] = a[j]*(delta_jk - a[k]) for each k
    # So d(smoothed_obj)/dz[k] = 2n * _compute_grad(a,w)[k] * a[k] * (1 - a[k])
    # Wait, let me be more careful.
    # a = softmax(z), so da/dz = a*(I - a^T) (outer product form: da_i/dz_j = a_i*(delta_ij - a_j))
    # d(obj)/dz_j = sum_i d(obj)/da_i * da_i/dz_j = sum_i g_i * a_i * (delta_ij - a_j)
    # = g_j * a_j - a_j * sum_i g_i * a_i
    # = a_j * (g_j - sum_i g_i * a_i)
    # where g = d(obj)/da = 2n * _compute_grad(a, w)
    
    g_a = 2.0 * n * _compute_grad(a, w)
    W = float(np.dot(g_a, a))
    grad_z = a * (g_a - W)
    return grad_z


class _LbfgsProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def f(self, z):
        return _smoobj(z, self.beta, self.n)

    def g(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS on log-softmax parameterization with adaptive beta annealing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007,
                                  0.003, 0.0015, 0.0007, 0.0003, 0.0001, 0.00005])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_per_iter = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(50, int(remaining / (est_per_iter * 20))))

        prob = _LbfgsProblem(beta, n)
        try:
            result = minimize(
                prob.f, z, jac=prob.g, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 120, 'maxls': 60}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _quick_optimize(a_init, deadline, min_time_remaining=10.0):
    """Combined L-BFGS + peak subgradient optimization."""
    a = _run_lbfgs(a_init, deadline, min_time_remaining=min_time_remaining + 5)
    if time.monotonic() < deadline - min_time_remaining:
        a2, sc2 = _peak_subgradient_descent(
            a, min(deadline - min_time_remaining, time.monotonic() + 20))
        if sc2 < _score(a):
            a = a2
    return a


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "exp_left":
        # Exponential decay from left (asymmetric)
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "exp_right":
        # Exponential decay from right (asymmetric)
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (1.0 - t))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "skewed_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.35) ** 2)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "logistic_skew":
        t = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + np.exp(4.0 * (t - 0.4)))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "random":
        if rng is None:
            rng = np.random.default_rng(12345)
        a = rng.random(n)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.25) ** 2) + 0.5 * np.exp(-30.0 * (t - 0.75) ** 2)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _stochastic_ils(a_init, deadline, rng, min_time_remaining=30.0):
    """ILS with peak-targeting perturbations."""
    n = len(a_init)
    current_a = _normalize(a_init.copy())
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc
    t_start_ils = time.monotonic()

    while time.monotonic() + min_time_remaining < deadline:
        # Choose perturbation strategy
        r = rng.random()
        if r < 0.25:
            # Global small perturbation
            sigma = 0.003 + 0.005 * rng.random()
            z = np.log(np.maximum(current_a, 1e-30))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_perturbed = _softmax(z)
        elif r < 0.45:
            # Peak-targeting: increase elements that reduce the worst peak
            c = _full_convolve(current_a, current_a)
            k_max = int(np.argmax(c))
            # Direction to reduce peak at k_max: negative gradient
            w = np.zeros(len(c))
            w[k_max] = 1.0
            grad = _compute_grad(current_a, w)
            grad = grad - np.mean(grad)
            # Perturb in negative gradient direction
            step = 0.01 + 0.02 * rng.random()
            a_perturbed = _normalize(current_a - step * grad)
        elif r < 0.60:
            # Block perturbation
            block_size = max(32, n // 10)
            start = int(rng.integers(0, n - block_size))
            a_perturbed = current_a.copy()
            z = np.log(np.maximum(a_perturbed[start:start + block_size], 1e-30))
            z = z + 0.03 * rng.standard_normal(block_size)
            a_perturbed[start:start + block_size] = np.exp(z - np.max(z))
            a_perturbed = _normalize(a_perturbed)
        elif r < 0.75:
            # Sparse perturbation
            n_perturb = max(20, n // 20)
            idx = rng.choice(n, size=n_perturb, replace=False)
            a_perturbed = current_a.copy()
            factors = np.exp(0.05 * rng.standard_normal(n_perturb))
            a_perturbed[idx] *= factors
            a_perturbed = _normalize(a_perturbed)
        elif r < 0.85:
            # Asymmetric shift: bias mass toward one side
            t = np.linspace(0, 1, n)
            shift = 0.1 * rng.standard_normal()
            z = np.log(np.maximum(current_a, 1e-30))
            z = z + shift * t
            z = z - z.mean()
            a_perturbed = _softmax(z)
        else:
            # Mirror + noise
            seg_len = max(20, n // 25)
            start = int(rng.integers(0, n - seg_len))
            a_perturbed = current_a.copy()
            a_perturbed[start:start + seg_len] = a_perturbed[start:start + seg_len][::-1]
            z = np.log(np.maximum(a_perturbed, 1e-30))
            z = z + 0.005 * rng.standard_normal(n)
            z = z - z.mean()
            a_perturbed = _softmax(z)

        # Quick optimize
        opt_deadline = min(deadline - min_time_remaining, time.monotonic() + 6)
        a_opt = _quick_optimize(a_perturbed, opt_deadline, min_time_remaining=2)
        sc_opt = _score(a_opt)

        if sc_opt < best_sc - 1e-12:
            best_sc = sc_opt
            best_a = a_opt.copy()
            current_a = a_opt.copy()
            print(f"    ILS NEW BEST: {best_sc:.10f} t={time.monotonic()-t_start_ils:.0f}s")
        elif sc_opt < current_sc - 1e-12:
            current_a = a_opt.copy()

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _final_ultra_polish(a_init, deadline):
    """Ultra-fine polish: alternate between tiny-beta L-BFGS and direct subgradient."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    # Ultra-small beta L-BFGS
    c = _full_convolve(a, a)
    c_max = float(c.max())
    beta = c_max * 0.00001
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    remaining = deadline - time.monotonic()
    max_iter = min(5000, max(100, int(remaining * 0.5 / 0.02)))
    
    prob = _LbfgsProblem(beta, n)
    try:
        result = minimize(prob.f, z, jac=prob.g, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-18,
                                   'maxcor': 150, 'maxls': 80})
        a_new = _softmax(result.x)
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            a = a_new
            best_a = a.copy()
            best_sc = sc_new
    except Exception:
        pass
    
    # Direct subgradient fine-tune
    if time.monotonic() < deadline - 5:
        a2, sc2 = _peak_subgradient_descent(a, deadline - 3, min_time=3.0)
        if sc2 < best_sc:
            best_a = a2
            best_sc = sc2
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid peak-equalization optimizer. Uses asymmetric initialization,
    L-BFGS with adaptive smoothing, direct active-set subgradient,
    and peak-targeting ILS to minimize 2*n*max(conv(a,a))/sum(a)^2.
    """
    global t_start
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    n_main = n_parent if n_parent > 0 else 16384

    # === Phase 1: Multi-start with diverse asymmetric profiles (180s) ===
    print(f"\n[P1] Multi-start (t={time.monotonic() - t_start:.1f}s)")
    p1_deadline = t_start + 180

    starts = [('parent', parent.copy())]
    for kind in ['exp_left', 'exp_right', 'skewed_gauss', 'logistic_skew', 'hann', 'bimodal']:
        starts.append((kind, _make_profile(n_main, kind, rng)))

    for name, a_init in starts:
        if time.monotonic() > p1_deadline - 30:
            break
        sc_init = _score(a_init)
        a_opt = _quick_optimize(a_init, p1_deadline, min_time_remaining=20)
        sc_opt = _score(a_opt)
        print(f"  {name}: init={sc_init:.10f}, opt={sc_opt:.10f}")
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [ms/{name}]")

    # === Phase 2: Intensive L-BFGS on best (250s) ===
    print(f"\n[P2] Intensive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    p2_deadline = min(deadline - 600, time.monotonic() + 250)
    if time.monotonic() < p2_deadline:
        a_lb = _run_lbfgs(best_seq, p2_deadline, min_time_remaining=40)
        sc = _score(a_lb)
        if sc < best_score:
            best_seq = a_lb.copy()
            best_score = sc
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_int]")

    # === Phase 3: Peak-directed subgradient (200s) ===
    print(f"\n[P3] Peak subgradient (t={time.monotonic() - t_start:.1f}s)")
    p3_deadline = min(deadline - 400, time.monotonic() + 200)
    if time.monotonic() < p3_deadline:
        a_pg, sc_pg = _peak_subgradient_descent(best_seq, p3_deadline, min_time=10)
        if sc_pg < best_score:
            best_seq = a_pg.copy()
            best_score = sc_pg
            print(f"  NEW BEST: {best_score:.10f} [peak_subgrad]")

    # === Phase 4: ILS with peak-targeting (300s) ===
    print(f"\n[P4] ILS (t={time.monotonic() - t_start:.1f}s)")
    p4_deadline = min(deadline - 300, time.monotonic() + 300)
    if time.monotonic() < p4_deadline:
        a_ils, sc_ils = _stochastic_ils(best_seq, p4_deadline, rng, min_time_remaining=30)
        if sc_ils < best_score:
            best_seq = a_ils.copy()
            best_score = sc_ils
            print(f"  NEW BEST: {best_score:.10f} [ils]")

    # === Phase 5: Multi-n exploration (200s) ===
    print(f"\n[P5] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [n_main, int(n_main * 1.5), int(n_main * 0.75), 32768, 20480, 49152]
    seen = set()
    n_unique = []
    for nn in n_candidates:
        if nn not in seen and nn != len(best_seq) and nn >= 1024:
            seen.add(nn)
            n_unique.append(nn)

    for n_try in n_unique[:4]:
        if time.monotonic() > deadline - 150:
            break
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        a_opt = _quick_optimize(a_init, deadline, min_time_remaining=50)
        sc_opt = _score(a_opt)
        print(f"  n={n_try}: init={sc_init:.10f}, opt={sc_opt:.10f}")
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try})")

    # === Phase 6: Second optimization round (250s) ===
    print(f"\n[P6] Second round (t={time.monotonic() - t_start:.1f}s)")
    p6_deadline = min(deadline - 100, time.monotonic() + 250)
    if time.monotonic() < p6_deadline:
        # L-BFGS
        a_lb2 = _run_lbfgs(best_seq, min(p6_deadline, time.monotonic() + 120), min_time_remaining=60)
        sc2 = _score(a_lb2)
        if sc2 < best_score:
            best_seq = a_lb2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [round2_lbfgs]")
        # Peak subgradient
        if time.monotonic() < p6_deadline - 20:
            a_pg2, sc_pg2 = _peak_subgradient_descent(best_seq, p6_deadline - 10, min_time=5)
            if sc_pg2 < best_score:
                best_seq = a_pg2.copy()
                best_score = sc_pg2
                print(f"  NEW BEST: {best_score:.10f} [round2_subgrad]")
        # Short ILS
        if time.monotonic() < p6_deadline - 10:
            rng2 = np.random.default_rng(int(seed) + 7777)
            a_ils2, sc_ils2 = _stochastic_ils(best_seq, p6_deadline, rng2, min_time_remaining=5)
            if sc_ils2 < best_score:
                best_seq = a_ils2.copy()
                best_score = sc_ils2
                print(f"  NEW BEST: {best_score:.10f} [round2_ils]")

    # === Phase 7: Ultra-fine final polish (remaining time) ===
    print(f"\n[P7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _final_ultra_polish(best_seq, deadline - 3)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_ultra]")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
