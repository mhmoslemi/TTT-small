Primal-dual optimizer with CG inner solver for minimum peak autocorrelation step functions.
Uses the strong duality: min_{a in simplex} max_k c[k] = max_{x in simplex} min_{a in simplex} sum_k x_k c[k].
Inner QP solved via conjugate gradient on the Hankel matrix H_x (matvec via FFT).

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _hankel_matvec(x, v, n):
    """Compute (H_x @ v) where H_x[i,j] = x[i+j] for 0<=i,j<n and i+j<=2n-2."""
    v_rev = v[::-1]
    out_len = n + len(x) - 1
    N = _next_pow2(out_len)
    Fv = np.fft.rfft(v_rev, N)
    Fx = np.fft.rfft(x, N)
    conv = np.fft.irfft(Fv * Fx, N)[:out_len]
    return conv[n - 1:2 * n - 1]

def _proj_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0)

        g = _grad_lag(a, k_star, n)

        # Also add small gradients from nearby peaks for stability
        sorted_lags = np.argsort(c)[::-1]
        top_k = min(5, 2 * n - 1)
        active = sorted_lags[:top_k]
        weights = c[active] - c_max + 1.0
        weights = weights / weights.sum()
        g_total = np.zeros(n)
        for w, k in zip(weights, active):
            g_total += w * _grad_lag(a, k, n)

        # Step with backtracking
        gnorm = np.linalg.norm(g_total)
        if gnorm < 1e-15:
            break

        eta = 0.1 / (gnorm + 1e-10)
        for ls in range(20):
            a_new = a - eta * g_total
            a_new = _proj_simplex(a_new)
            c_new = _full_convolve(a_new, a_new)
            if float(c_new.max()) < c_max * (1 + 1e-12):
                a = a_new
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                    if it % 50 == 0:
                        print(f"    subgrad iter {it}: score={best_sc:.10f}")
                break
            eta *= 0.5

    return best_a, best_sc


def _run_lbfgs_polish(a_init, deadline, min_time_remaining=10.0):
    """L-BFGS with log-sum-exp smoothing."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = max_c - c_sorted[min(10, len(c_sorted) - 1)]
    base_beta = max(gap * 0.05, max_c * 0.001)

    _g_n = n
    ratios = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001, 0.0005, 0.0001]

    for i, ratio in enumerate(ratios):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        beta = base_beta * ratio
        _g_beta = beta

        z = np.log(np.maximum(best_a, 1e-30))
        z = z - z.mean()

        est_time = 0.005 * max(1.0, n / 32768.0)
        max_iter = min(3000, max(20, int(remaining / (est_time * 5))))

        try:
            result = minimize(
                _lbfgs_obj, z,
                jac=_lbfgs_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 50, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                print(f"    L-BFGS ratio={ratio}: score={best_sc:.10f}")
        except Exception:
            pass

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: primal-dual with CG inner solver, subgradient method,
    and L-BFGS polish. Explores multiple sequence lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is None:
        parent = _make_profile(32768, "tapered")

    best_score = _score(parent)
    best_a = parent.copy()
    print(f"Parent: n={len(parent)}, score={best_score:.10f}")

    # Determine n values to try
    n_parent = len(parent)
    n_candidates = [n_parent]
    for n_try in [16384, 32768, 65536]:
        if n_try != n_parent:
            n_candidates.append(n_try)
    n_candidates = sorted(set(n_candidates), reverse=True)

    # Phase 1: Primal-dual on parent n (majority of time)
    print(f"\n[Phase 1] Primal-dual n={n_parent}")
    p1_deadline = t_start + 700
    a_pd, sc_pd = _run_primal_dual(parent, n_parent, p1_deadline, num_iters=10000)
    if sc_pd < best_score:
        best_score = sc_pd
        best_a = a_pd.copy()
        print(f"  Primal-dual result: {best_score:.10f}")

    # Phase 2: Subgradient refinement
    print(f"\n[Phase 2] Subgradient")
    p2_deadline = t_start + 1000
    a_sg, sc_sg = _run_subgradient(best_a, len(best_a), p2_deadline, num_iters=5000)
    if sc_sg < best_score:
        best_score = sc_sg
        best_a = a_sg.copy()
        print(f"  Subgradient result: {best_score:.10f}")

    # Phase 3: L-BFGS polish
    print(f"\n[Phase 3] L-BFGS polish")
    p3_deadline = t_start + 1350
    a_lb, sc_lb = _run_lbfgs_polish(best_a, p3_deadline, min_time_remaining=30.0)
    if sc_lb < best_score:
        best_score = sc_lb
        best_a = a_lb.copy()
        print(f"  L-BFGS result: {best_score:.10f}")

    # Phase 4: Try other n values
    print(f"\n[Phase 4] Multi-n exploration")
    for n_try in n_candidates:
        if time.monotonic() > deadline - 30:
            break
        if n_try == n_parent:
            continue

        # Resample
        x_old = np.linspace(0, 1, len(best_a))
        x_new = np.linspace(0, 1, n_try)
        a_res = np.interp(x_new, x_old, best_a)
        a_res = np.maximum(a_res, 0)
        a_res = a_res / a_res.sum()

        sc_res = _score(a_res)
        print(f"  n={n_try}: resampled score={sc_res:.10f}")

        # Quick primal-dual
        p4_deadline = min(deadline - 20, time.monotonic() + 150)
        a_n, sc_n = _run_primal_dual(a_res, n_try, p4_deadline, num_iters=3000)
        if sc_n < best_score:
            best_score = sc_n
            best_a = a_n.copy()
            print(f"  n={n_try} primal-dual: {best_score:.10f}")

    # Phase 5: Second primal-dual round on best
    print(f"\n[Phase 5] Second primal-dual round")
    if time.monotonic() < deadline - 60:
        n_best = len(best_a)
        c_best = _full_convolve(best_a, best_a)
        tau_start = float(c_best.max()) * 0.05
        a2, sc2 = _run_primal_dual(best_a, n_best, deadline - 20, num_iters=10000)
        if sc2 < best_score:
            best_score = sc2
            best_a = a2.copy()
            print(f"  Round 2 result: {best_score:.10f}")

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"\n{'=' * 60}")
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
    return [float(x) for x in final]
