SCP (Successive Convex Approximation) with active-set LP for minimizing
2*n*max(conv(a,a))/sum(a)^2.

Key innovation: At each iteration, linearize the autocorrelation around
the current point and solve a linear program to find the best step.
This directly targets the active (peak) lags and is more effective than
gradient-based methods for the non-smooth L-infinity objective.

Combined with:
- Active-set gradient with momentum for fine-tuning
- Stochastic perturbations to escape local minima
- Multi-n exploration

import time

# ... (helper functions as before) ...

def _scp_lp_step(a_k, C_k, tol_frac=0.05, max_active=300):
    """Solve one SCP linearization subproblem via LP."""
    n = len(a_k)
    c_max = float(C_k.max())
    tol = tol_frac * c_max
    
    active = C_k >= (c_max - tol)
    active_idx = np.where(active)[0]
    
    if len(active_idx) == 0:
        return a_k.copy()
    
    if len(active_idx) > max_active:
        c_vals = C_k[active_idx]
        top = np.argsort(c_vals)[-max_active:]
        active_idx = active_idx[top]
    
    n_active = len(active_idx)
    
    # Build A_ub matrix (n_active x (n+1))
    # Constraint for lag j: sum_p a_k[j-p] * a_p * 2 - t <= C_k[j]
    # i.e., [2*a_k[j-0], 2*a_k[j-1], ..., 2*a_k[j-(n-1)], -1] @ [a; t] <= C_k[j]
    # But a_k[j-p] is 0 when j-p < 0 or j-p >= n
    
    A = np.zeros((n_active, n + 1))
    b = np.zeros(n_active)
    
    for i, j in enumerate(active_idx):
        # For p = 0, ..., n-1: coefficient of a_p is 2 * a_k[j-p] if 0 <= j-p < n
        p_min = max(0, j - n + 1)

    b_eq = np.array([1.0])
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,

```python
"""
SCP (Successive Convex Approximation) with active-set LP for minimizing
2*n*max(conv(a,a))/sum(a)^2.

Key innovation: At each iteration, linearize the autocorrelation around
the current point and solve a linear program to find the best step direction.
This directly targets the peak lags and is more effective than gradient-based
methods for the non-smooth L-infinity objective. Combined with:
- Benders-style constraint generation for the active set
- Robust line search with backtracking
- Active-set gradient with momentum for fine-tuning
- Stochastic iterated local search for global exploration
- Multi-n exploration for optimal sequence length
"""
import numpy as np
import time
from scipy.optimize import linprog, minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos2":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exponential_taper":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2 / (0.1 + 0.4 * (t - 0.5) ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        edges = max(10, n // 20)
        edge_t = np.linspace(0, 1, edges)
        taper = 0.5 * (1 + np.cos(edge_t * np.pi))
        a[:edges] = taper
        a[-edges:] = taper[::-1]
        return a / a.sum()
    else:
        return np.ones(n) / n


def _solve_scp_lp(a_k, C_k, active_idx, n):
    """Solve one SCP LP subproblem. Returns a_new or None."""
    n_active = len(active_idx)
    if n_active == 0:
        return a_k.copy()

    # Build A_ub (n_active x (n+1)) and b_ub
    A = np.zeros((n_active, n + 1))
    b = np.zeros(n_active)

    for i, j in enumerate(active_idx):
        p_min = max(0, j - n + 1)
        p_max = min(n - 1, j)
        if p_max >= p_min:
            # Coefficient of a_p is 2*a_k[j-p] for p in [p_min, p_max]
            # j-p ranges from j-p_min down to j-p_max
            A[i, p_min:p_max + 1] = 2.0 * a_k[j - p_max:j - p_min + 1][::-1]
        A[i, -1] = -1.0
        b[i] = C_k[j]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    bounds = [(0, None)] * n + [(-1e6, 1e6)]

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success:
            a_new = np.maximum(res.x[:n], 0.0)
            s = a_new.sum()
            if s > 1e-10:
                a_new /= s
            return a_new
    except Exception:
        pass
    return None


def _scp_optimize(a_init, deadline, min_active_frac=0.7, max_active_frac=0.95,
                  initial_active_frac=0.5, max_ahead=10.0):
    """SCP-LP optimization with Benders-style active set growth."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    C = _full_convolve(a, a)
    c_max = float(C.max())
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0
    max_no_improve = 80

    active_frac = initial_active_frac

    while time.monotonic() < deadline - max_ahead:
        # Determine active set
        threshold = (1.0 - (1.0 - active_frac)) * c_max
        active = np.where(C >= c_max - (1.0 - active_frac) * c_max)[0]
        # Simpler: active where C >= c_max * active_frac
        active = np.where(C >= c_max * active_frac)[0]

        if len(active) == 0:
            active = np.array([int(np.argmax(C))])

        # Limit active set for speed
        max_active = min(len(active), 500)
        if len(active) > max_active:
            c_vals = C[active]
            top = np.argsort(c_vals)[-max_active:]
            active = active[top]

        # Solve LP
        a_lp = _solve_scp_lp(a, C, active, n)
        if a_lp is None:
            # Fallback: reduce active set
            active = active[:max(5, len(active) // 2)]
            a_lp = _solve_scp_lp(a, C, active, n)
            if a_lp is None:
                break

        # Line search
        current_sc = _score(a)
        improved = False
        for lam in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05]:
            a_try = _normalize((1.0 - lam) * a + lam * a_lp)
            sc_try = _score(a_try)
            if sc_try < current_sc * (1.0 - 1e-9):
                a = a_try
                C = _full_convolve(a, a)
                c_max = float(C.max())
                improved = True
                if sc_try < best_sc * (1.0 - 1e-9):
                    best_sc = sc_try
                    best_a = a.copy()
                    no_improve = 0
                break

        if not improved:
            no_improve += 1
            # Tighten or loosen active set
            if no_improve % 10 == 0:
                if active_frac < max_active_frac:
                    active_frac = min(max_active_frac, active_frac + 0.03)
                else:
                    active_frac = max(min_active_frac, active_frac - 0.05)
            if no_improve >= max_no_improve:
                break

    return best_a, best_sc


def _active_set_gradient(a, c, tol_frac):
    """Compute weighted active-set gradient."""
    n = len(a)
    c_max = float(c.max())
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True

    eps = 1e-12 * c_max
    w = np.zeros_like(c)
    w[active] = 1.0 / (c_max - c[active] + eps)
    w = w / w.sum()

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad = 2.0 * h
    grad = grad - np.mean(grad)
    return grad
