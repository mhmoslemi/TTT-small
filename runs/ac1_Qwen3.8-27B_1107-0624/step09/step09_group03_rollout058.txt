"""

2. Iterative peeling/deflation with adaptive active set
3. High-precision Newton refinement with direct sparse solve

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()

        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec_func, rhs, max_cg=300, tol=1e-14):
    x = np.zeros_like(rhs)
    r = rhs - matvec_func(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec_func(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x_new = x + alpha * p
        r_new = r - alpha * Ap
        rs_new = np.dot(r_new, r_new)
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            x = x_new
            break
        p = r_new + (rs_new / rs) * p
        x = x_new
        r = r_new
        rs = rs_new
    return x


def _newton_direction(a, c, c_max, tol_frac, mu_reg=1e-8):
    n = len(a)
    active = c >= (c_max - tol_frac * c_max)
    if not active.any():
        active[int(np.argmax(c))] = True
    m = int(active.sum())
    if m < 2 or m > 20 * n:
        return None

    A_vec = active.astype(np.float64) / m
    g = 2.0 * _corr_a_A(a, A_vec, n)
    g_t = g - np.mean(g)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-15:
        return None

    mu = mu_reg / m + 1e-12

    A_vec_local = A_vec
    n_local = n
    mu_local = mu

    def _hess_vp(v):
        return 2.0 * _corr_a_A(v, A_vec_local, n_local) + mu_local * v

    u = _cg_solve(_hess_vp, g_t, max_cg=300)
    w = _cg_solve(_hess_vp, np.ones(n), max_cg=300)
    w_sum = float(w.sum())

    if abs(w_sum) < 1e-15:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w

    dx = dx - np.mean(dx)
    return dx


def _deflation_step(a, c, c_max, n_peaks=10):
    n = len(a)
    sorted_idx = np.argsort(c)[::-1]
    peak_indices = sorted_idx[:min(n_peaks, len(c))]
    peak_vals = c[peak_indices]
    weights = peak_vals - c_max + float(peak_vals.max())
    weights = np.maximum(weights, 1e-10)
    weights = weights / weights.sum()

    w = np.zeros_like(c)
    w[peak_indices] = weights
    grad = 2.0 * _corr_a_A(a, w, n)
    g_t = grad - np.mean(grad)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-15:
        return None
    return -g_t / gnorm


def _line_search_deflation(a, direction, c_max, max_steps=40):
    n = len(a)
    best_a = None
    best_cmax = c_max
    t = 1.0
    for _ in range(max_steps):
        a_try = a + t * direction
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            t *= 0.5
            continue
        a_try = a_try / s_try
        c_try = _full_convolve(a_try, a_try)
        cmax_try = float(c_try.max())
        if cmax_try < best_cmax - 1e-14:
            best_cmax = cmax_try
            best_a = a_try.copy()
        t *= 0.7
        if t < 1e-10:
            break
    return best_a, best_cmax

                z = z - z.mean()
            if i % 5 == 0 or i == n_betas - 1:
                print(f"    L-BFGS beta={beta:.2e} (frac={beta/max_c:.2e}): "
                      f"score={current_sc:.12f} t={time.monotonic()-t_start_ref:.1f}s")
        except Exception as ex:
            print(f"    L-BFGS beta={beta:.2e}: exception {ex}")

    return current_a, current_sc


def _peeling_phase(a_init, deadline, max_rounds=200):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        n_peaks = max(3, min(50, int(5 * (c < c_max * 0.99).sum() * 0.1 + 3)))
        direction = _deflation_step(a, c, c_max, n_peaks=max(n_peaks, 5))
        if direction is None:
            direction = _newton_direction(a, c, c_max, tol_frac=0.05)
            if direction is None:
                break

        a_new, cmax_new = _line_search_deflation(a, direction, c_max)
        if a_new is None:
            # Try smaller deflation
            direction = _deflation_step(a, c, c_max, n_peaks=3)
            if direction is not None:
                a_new, cmax_new = _line_search_deflation(a, 0.1 * direction, c_max)
            if a_new is None:
                break

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Check convergence
        if c_max - cmax_new < 1e-15:
            break

    return best_a, best_sc


def _newton_refinement(a_init, deadline, max_outer=200):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break

        # Adaptive tolerance: start loose, tighten as we converge
        progress = outer / max_outer
        if outer < max_outer * 0.3:
            tol_frac = 0.01
        elif outer < max_outer * 0.6:
            tol_frac = 0.003
        elif outer < max_outer * 0.85:
            tol_frac = 0.001
        else:
            tol_frac = 0.0003

        dx = _newton_direction(a, c, c_max, tol_frac=tol_frac)
        if dx is None:
            # Try deflation as fallback
            direction = _deflation_step(a, c, c_max, n_peaks=5)
            if direction is None:
                break
            a_new, cmax_new = _line_search_deflation(a, direction, c_max)
            if a_new is None:
                break
            a = a_new
            c = _full_convolve(a, a)
            c_max = float(c.max())
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
            continue

        # Line search
        a_new, cmax_new = _line_search_deflation(a, dx, c_max, max_steps=50)
        if a_new is None:
            # Try reduced step
            a_try = a + 0.1 * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a_new = a_try
                    cmax_new = float(c_try.max())
                else:
                    break
            else:
                break

        a = a_new
        c = _full_convolve(a, a)
        new_c_max = float(c.max())
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if abs(c_max - new_c_max) < 1e-16:
            break
        c_max = new_c_max

    return best_a, best_sc


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 - (2 * t - 1) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def propose_candidate(seed=42, budget_s=1800):
    global t_start_ref
    t_start = time.monotonic()
    t_start_ref = t_start
    deadline = t_start + max(1.0, float(budget_s) - 40.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.12f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={len(parent)}, score={sc_parent:.12f}")
        update_best(parent, sc_parent, "parent")
    else:
        parent = _make_profile(16384, "uniform")
        update_best(parent, _score(parent), "init")

    n_work = len(parent)
    print(f"\n[Phase 1] Peeling refinement (n={n_work}, t={time.monotonic()-t_start:.1f}s)")
    peel_dl = min(deadline - 800, time.monotonic() + 120)
    a_peel, sc_peel = _peeling_phase(parent, peel_dl, max_rounds=100)
    update_best(a_peel, sc_peel, "peeling")

    print(f"\n[Phase 2] L-BFGS continuation (n={n_work}, t={time.monotonic()-t_start:.1f}s)")
    lbfgs_dl = min(deadline - 500, time.monotonic() + 300)
    a_lbfgs, sc_lbfgs = _lbfgs_phase(
        a_peel, lbfgs_dl,
        min_time_remaining=500.0,
        start_beta_frac=0.005,
        end_beta_frac=5e-9,
        n_betas=22
    )
    update_best(a_lbfgs, sc_lbfgs, "lbfgs_main")

    print(f"\n[Phase 3] Newton refinement (n={n_work}, t={time.monotonic()-t_start:.1f}s)")
    newton_dl = min(deadline - 300, time.monotonic() + 250)
    a_newton, sc_newton = _newton_refinement(a_lbfgs, newton_dl, max_outer=300)
    update_best(a_newton, sc_newton, "newton_ref")

    print(f"\n[Phase 4] Second L-BFGS polish (n={n_work}, t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        lbfgs2_dl = min(deadline - 150, time.monotonic() + 100)
        a_pol, sc_pol = _lbfgs_phase(
            best_seq, lbfgs2_dl,
            min_time_remaining=150.0,
            start_beta_frac=1e-5,
            end_beta_frac=1e-10,
            n_betas=15
        )
        update_best(a_pol, sc_pol, "lbfgs_polish")

    print(f"\n[Phase 5] Final Newton polish (n={n_work}, t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        newton2_dl = min(deadline - 80, time.monotonic() + 80)
        a_fin, sc_fin = _newton_refinement(best_seq, newton2_dl, max_outer=100)
        update_best(a_fin, sc_fin, "final_newton")

    # Try n=32768

    if time.monotonic() < deadline - 120 and best_seq is not None:
        n_large = 32768
        a_large = _interpolate_seq(best_seq, n_large)
        sc_large_init = _score(a_large)
        print(f"  n={n_large} init score: {sc_large_init:.12f}")
        update_best(a_large, sc_large_init, "n32768_init")

        if sc_large_init < best_score + 0.01:

            lbfgs3_dl = min(deadline - 60, time.monotonic() + 60)
            a_large_opt, sc_large_opt = _lbfgs_phase(
                a_large, lbfgs3_dl,
                min_time_remaining=60.0,
                start_beta_frac=0.003,

            )

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)

        if s < 0.01:

        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
