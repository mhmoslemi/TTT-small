"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
- Aggressive multi-scale L-BFGS with log-sum-exp smoothing
- Weighted active-set QP refinement with temperature annealing
- Projected gradient with backtracking line search on the active set
- Multi-start with diverse asymmetric initializations
- Adaptive active set sizing based on peak flatness
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.008) * beta_scale

    # More aggressive beta schedule with more small values for fine tuning
    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.12, 0.06,
        0.03, 0.015, 0.008, 0.004,
        0.002, 0.001, 0.0005, 0.0002, 0.0001,
        0.00005, 0.00002, 0.00001, 0.000005
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # Scale iterations based on beta size
        if beta < base_beta * 0.005:
            est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
            max_iter = min(8000, max(100, int(remaining / (est_time_per_iter * 20))))
        elif beta < base_beta * 0.05:
            est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
            max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 15))))
        else:
            est_time_per_iter = 0.002 * max(1.0, n / 2048.0)
            max_iter = min(3000, max(30, int(remaining / (est_time_per_iter * 10))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _weighted_active_set_refine(a_init, deadline, min_time_remaining=30.0):
    """
    Weighted active-set refinement using temperature-annealed weights.
    At each step, compute weighted gradient of active peaks and take
    a projected gradient step with backtracking line search.
    The temperature controls how many peaks are effectively active.
    """
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    # Temperature schedule: start wide, narrow down
    temps = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    lr = 0.05
    no_improve = 0

    for temp_factor in temps:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        # Estimate max iterations for this temperature
        max_iters = int(min(300, remaining * 5))

        for it in range(max_iters):
            if time.monotonic() + min_time_remaining > deadline:
                break

            c = _full_convolve(a, a)
            c_max = float(c.max())
            current_sc = 2.0 * n * c_max

            if current_sc < best_sc - 1e-15:
                best_sc = current_sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve >= 50:
                    break

            # Compute temperature-based weights
            temp = c_max * temp_factor
            shifted = (c - c_max) / temp
            e = np.exp(np.clip(shifted, -100, 0))
            s = e.sum()
            w = e / s  # Soft weights over all peaks

            # Compute gradient of weighted sum of active peaks
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * a * (h - W) / n  # Gradient of 2n*weighted_sum/sum^2 wrt a on simplex

            # Project gradient to be sum-zero (stay on simplex)
            grad = grad - np.mean(grad)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-18:
                break

            # Backtracking line search
            alpha = lr / (gnorm + 1e-15)
            improved = False
            for _ in range(40):
                a_try = a - alpha * grad
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-14:
                    a = a_try
                    improved = True
                    break
                alpha *= 0.5

            if not improved:
                lr *= 0.9
                if no_improve >= 30:
                    a = best_a.copy()
                    lr = 0.05
                    no_improve = 0
                    break

    return best_a, best_sc

def _peak_equalization_qp(a_init, deadline, min_time_remaining=30.0):
    """
    Iterative peak equalization: alternate between identifying the active set
    and solving a projected gradient QP on that set.
    Uses adaptive active set size based on peak flatness.
    """
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    max_rounds = 50
    for rnd in range(max_rounds):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()

        # Adaptive active set size: include peaks within relative tolerance of max
        # Start with tight tolerance, loosen if not improving
        tolerance = 0.001 * (1 + rnd * 0.5)
        active = np.where(c >= c_max * (1 - tolerance))[0]
        if len(active) < 3:
            active = np.argsort(c)[::-1][:10]
        if len(active) > 500:
            active = np.argsort(c)[::-1][:500]

        K = len(active)
        w_S = np.zeros(2 * n - 1)
        w_S[active] = 1.0 / K
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)[n - 1:2 * n - 1]
        W = float(np.dot(w_S, c))
        grad = 4.0 / n * (h - W)
        # Scale by a for log-space gradient, then project
        grad_proj = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_proj)
        if gnorm < 1e-18:
            break

        # Multiple steps per round
        steps_per_round = 20
        lr = 0.1 / (gnorm * K + 1e-15)
        for _ in range(steps_per_round):
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_try = a - lr * grad_proj
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                lr *= 0.5
                continue
            a_try = a_try / s_try
            # Quick check
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc:
                a = a_try
                current_sc = sc_try
            else:
                lr *= 0.7

    return best_a, best_sc


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "left_heavy":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t) + 0.01
        return a / a.sum()
    elif kind == "right_heavy":
        t = np.linspace(0, 1, n)
        a = np.exp(2.0 * (t - 1)) + 0.01
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.arange(n)
        a = 1.0 + 0.4 * np.sin(2.0 * np.pi * 5.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 11.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.2) ** 2) + 0.5 * np.exp(-30.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "asym_bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.15) ** 2) + 0.3 * np.exp(-40.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "noise":
        if rng is None:
            rng = np.random.default_rng(0)
        a = 0.5 + 0.5 * rng.random(n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sine_asym":
        t = np.arange(n)
        a = np.exp(-1.5 * t / n) * (1.0 + 0.3 * np.sin(2.0 * np.pi * 7.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power_law":
        t = np.arange(1, n + 1)
        a = t ** (-0.3)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _tiny_lbfgs(a_init, deadline, min_time_remaining=5.0):
    """Quick L-BFGS with very small beta for final polish."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap * 0.1, max_c * 0.001)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    betas = [base_beta, base_beta * 0.1, base_beta * 0.01, base_beta * 0.001]
    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        prob = _LbfgsProb(beta, n)
        try:
            remaining = deadline - min_time_remaining - time.monotonic()
            max_iter = min(2000, max(50, int(remaining / 0.05)))
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 60}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining:
    1. L-BFGS with aggressive beta annealing
    2. Weighted active-set refinement with temperature annealing
    3. Top-K active-set QP refinement with adaptive K
    4. Peak equalization with adaptive tolerance
    5. Multi-start from diverse initializations
    6. Multi-n exploration
    7. Final ultra-fine polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                parent_sc = _score(parent)
                best_seq = parent.copy()
                best_score = parent_sc
                print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent. Uniform n={n_parent}, score={best_score:.10f}")

    n = len(best_seq)

    # === Phase 1: L-BFGS warmup on parent ===
    print(f"\n[P1] L-BFGS warmup (t={time.monotonic()-t_start:.1f}s)")
    a_p1 = _run_lbfgs(parent, deadline, min_time_remaining=600.0, beta_scale=1.0)
    sc_p1 = _score(a_p1)
    if sc_p1 < best_score:
        best_seq = a_p1.copy()
        best_score = sc_p1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_warmup]")
    else:
        print(f"  No improvement (score={sc_p1:.10f})")

    # === Phase 2: Weighted active set refinement ===
    print(f"\n[P2] Weighted active set (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a_w, sc_w = _weighted_active_set_refine(best_seq, deadline, min_time_remaining=200.0)
        if sc_w < best_score:
            best_seq = a_w.copy()
            best_score = sc_w
            print(f"  NEW BEST: {best_score:.10f} [weighted_active]")
        else:
            print(f"  No improvement (score={sc_w:.10f}, best={best_score:.10f})")

    # === Phase 3: Top-K active set with varying K ===
    print(f"\n[P3] Top-K active set (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a_k, sc_k = _top_k_active_set_refine(best_seq, deadline, min_time_remaining=150.0,
                                             K_range=(5, 200))
        if sc_k < best_score:
            best_seq = a_k.copy()
            best_score = sc_k
            print(f"  NEW BEST: {best_score:.10f} [topk_active]")
        else:
            print(f"  No improvement (score={sc_k:.10f}, best={best_score:.10f})")

    # === Phase 4: L-BFGS again (refined starting point) ===
    print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        a_p4 = _run_lbfgs(best_seq, deadline, min_time_remaining=150.0, beta_scale=0.5)
        sc_p4 = _score(a_p4)
        if sc_p4 < best_score:
            best_seq = a_p4.copy()
            best_score = sc_p4
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_r2]")
        else:
            print(f"  No improvement (score={sc_p4:.10f})")

    # === Phase 5: Peak equalization with adaptive tolerance ===
    print(f"\n[P5] Peak equalization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a_eq, sc_eq = _peak_equalization_qp(best_seq, deadline, min_time_remaining=100.0)
        if sc_eq < best_score:
            best_seq = a_eq.copy()
            best_score = sc_eq
            print(f"  NEW BEST: {best_score:.10f} [peak_eq]")
        else:
            print(f"  No improvement (score={sc_eq:.10f}, best={best_score:.10f})")

    # === Phase 6: Multi-start from diverse initializations ===
    print(f"\n[P6] Multi-start (t={time.monotonic()-t_start:.1f}s)")
    profiles = ["left_heavy", "sine_asym", "asym_bimodal", "power_law", "noise"]
    for prof in profiles:
        if time.monotonic() > deadline - 80:
            break
        a_init = _make_profile(n, prof, rng)
        # Quick optimization
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=80.0, beta_scale=0.8)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [multi_start:{prof}]")
        else:
            print(f"  {prof}: score={sc_opt:.10f} (no improvement)")

    # === Phase 7: Weighted active set on best ===
    print(f"\n[P7] Weighted active set round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        a_w2, sc_w2 = _weighted_active_set_refine(best_seq, deadline, min_time_remaining=50.0)
        if sc_w2 < best_score:
            best_seq = a_w2.copy()
            best_score = sc_w2
            print(f"  NEW BEST: {best_score:.10f} [weighted_active_2]")
        else:
            print(f"  No improvement (score={sc_w2:.10f})")

    # === Phase 8: Multi-n exploration ===
    print(f"\n[P8] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([n, 16384, 12288, 24576, 8192, 32768]), reverse=True)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 40:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=40.0, beta_scale=0.5)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n]")
        else:
            print(f"  n={n_try}: score={sc_opt:.10f} (init={sc_init:.10f})")

    # === Phase 9: Final ultra-fine L-BFGS ===
    print(f"\n[P9] Final ultra-fine (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 5 - time.monotonic()
    if remaining > 10:
        a_final = _tiny_lbfgs(best_seq, deadline, min_time_remaining=5.0)
        sc_final = _score(a_final)
        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final
            print(f"  NEW BEST: {best_score:.10f} [final_ultra]")

        # Final active set
        if time.monotonic() < deadline - 3:
            a_fin, sc_fin = _weighted_active_set_refine(best_seq, deadline - 2, min_time_remaining=2.0)
            if sc_fin < best_score:
                best_seq = a_fin.copy()
                best_score = sc_fin
                print(f"  NEW BEST: {best_score:.10f} [final_active]")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
