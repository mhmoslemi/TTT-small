import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    """Compute the correlation vector: gradient of sum_k A_vec[k]*c_k w.r.t. a"""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]

    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _hessian_matvec(v, A_vec, n, mu_reg=1e-10):
    """Hessian-vector product for the active-constrained quadratic.
    H = 2 * K^T K where K maps a -> active convolution values.
    H*v = 2 * K^T (K*v)
    K*v gives the convolution values at active indices.
    """
    conv_v = _full_convolve(v, v)  # This is wrong for Hessian
    # Actually H[v]_j = 2 * sum_k A_vec[k] * (v[k-j] + v[j-k])
    # = 4 * corr(v, A_vec) (when symmetrized)
    # More precisely: H*v = 2 * _corr_a_A(v, A_vec, n)
    return 2.0 * _corr_a_A(v, A_vec, n) + mu_reg * v


def _cg_solve(matvec_fn, rhs, max_cg=300, tol=1e-12):
    """Conjugate gradient solver."""
    n = len(rhs)
    x = np.zeros(n)
    r = rhs - matvec_fn(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_cg):
        Ap = matvec_fn(p)
        pAp = np.dot(p, Ap)
        if abs(pAp) < 1e-30:
            break
        alpha = rs / pAp
        if alpha < 0 or not np.isfinite(alpha):
            break
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(max(rs_new, 0)) < tol * (1 + rhs_norm):
            break
        if not np.isfinite(rs_new):
            break
        beta_cg = rs_new / rs
        p = r + beta_cg * p
        rs = rs_new
    return x


def _compute_kkt_weights(a, c, c_max, active_indices, n):
    """
    Compute KKT multipliers for active constraints.
    Solve: sum_k lam_k * grad_k = mu/2 on the support
    subject to sum lam_k = 1, lam_k >= 0.
    
    Returns (weights, mu, residual_norm)
    """
    m = len(active_indices)
    if m == 0:
        return None, 0, np.inf
    if m == 1:
        # Single active constraint: uniform weight
        w = np.ones(1)
        return w, 0, 0.0
    
    # Build the system using the gradient vectors
    # grad_k[j] for j=0..n-1: the j-th component of gradient of c_k w.r.t. a
    # Using FFT: grad = 2 * corr(a, A_vec) where A_vec has 1 at active positions
    # But we need individual grad_k vectors for the linear system.
    
    # For the system: sum_k lam_k * grad_k = (mu/2) * ones (on support)
    # This is equivalent to: the gradient of the weighted sum = constant
    # grad_weighted = 2 * corr(a, w_full) where w_full[k] = lam_k at active positions
    
    # We want: 2 * corr(a, w_full)[j] = mu for all j in support
    # And: sum lam_k = 1
    
    # Build the matrix B where B[j, k] = grad_k[j] for active k
    # Then solve: B * lam = (mu/2) * ones_support, sum(lam) = 1
    
    # For efficiency, compute B * lam using the correlation:
    # B * lam = 2 * corr(a, w_full) where w_full has lam values at active positions
    
    # Set up the least-squares problem:
    # Minimize ||B * lam - (mu/2) * 1||^2
    # subject to: sum(lam) = 1
    
    # Use the augmented system:
    # [B; 1^T] * [lam; 1] ... this doesn't work directly.
    
    # Instead: minimize ||B * lam - alpha * 1||^2 + penalty * (sum(lam) - 1)^2
    # Taking derivative: B^T B * lam - alpha * B^T * 1 + 2*penalty*(sum(lam)-1)*1 = 0
    # And: sum(lam) = 1
    
    # Simpler approach: parameterize mu and solve for lam
    # From: B * lam = (mu/2) * 1 (least squares)
    #   lam = (B^T B)^{-1} B^T (mu/2) * 1
    # Constraint: sum(lam) = 1
    #   => mu/2 * 1^T (B^T B)^{-1} B^T 1 = 1
    #   => mu = 2 / (1^T (B^T B)^{-1} B^T 1)
    #   => lam = mu/2 * (B^T B)^{-1} B^T 1
    
    # This requires inverting B^T B (m x m matrix), which is feasible for small m.
    
    # Compute B^T B and B^T 1 using the correlation structure:
    # (B^T B)[k1, k2] = sum_j grad_{k1}[j] * grad_{k2}[j]
    #                  = sum_j (a[k1-j]+a[j-k1]) * (a[k2-j]+a[j-k2])
    
    # For efficiency, use the fact that:
    # sum_j grad_k1[j] * grad_k2[j] = 2 * sum_j a[k1-j]*a[k2-j] + 2 * sum_j a[k1-j]*a[j-k2]
    #                                  + 2 * sum_j a[j-k1]*a[k2-j] + 2 * sum_j a[j-k1]*a[j-k2]
    #                                  = 2 * (a conv a)[k2-k1] + 2 * (a conv a_rev)[k1+k2-2n+2]
    #                                  + 2 * (a_rev conv a)[k2-k1] + 2 * (a_rev conv a_rev)[k1+k2-2n+2]
    
    # This is getting complex. Let me just compute B directly for small m.
    
    if m <= 200:
        # Direct computation
        B = np.zeros((n, m))
        for idx, k in enumerate(active_indices):
            k = int(k)
            # grad_k[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
            for j in range(n):
                val = 0.0
                if 0 <= k - j < n:
                    val += a[k - j]
                if 0 <= j - k < n:
                    val += a[j - k]
                B[j, idx] = val
        
        # Solve: B * lam = (mu/2) * 1, sum(lam) = 1
        # Using the formula: lam = mu/2 * (B^T B)^{-1} B^T 1
        BtB = B.T @ B
        Bt1 = B.T @ np.ones(n)
        
        # Add regularization for numerical stability
        BtB_reg = BtB + 1e-12 * np.eye(m)
        
        try:
            tmp = np.linalg.solve(BtB_reg, Bt1)
        except np.linalg.LinAlgError:
            tmp = np.linalg.lstsq(BtB_reg, Bt1, rcond=None)[0]
        
        denom = float(np.dot(np.ones(m), tmp))
        if abs(denom) < 1e-20:
            # Fallback to uniform
            return np.ones(m) / m, 0.0, 1.0
        
        lam = tmp / denom  # Ensures sum(lam) = 1
        mu_val = 2.0 * denom  # mu = 2 * (1^T B^T (B^T B)^{-1} B^T 1) / (1^T tmp) ... 
        # Actually mu/2 = 1/denom * ... let me just compute residual
        
        # Residual: ||B * lam - (mu/2) * 1||
        # mu/2 = (1^T B^T (B^T B)^{-1} B^T 1) * mu / (2 * 1^T tmp)
        # This is circular. Let me just compute the gradient and check.
        grad_check = B @ lam  # = sum lam_k * grad_k
        mean_grad = float(np.mean(grad_check))
        residual = float(np.std(grad_check))
        
        # Clamp negative multipliers
        lam_clipped = np.maximum(lam, 0)
        lam_sum = lam_clipped.sum()
        if lam_sum > 1e-15:
            lam_clipped = lam_clipped / lam_sum
        else:
            lam_clipped = np.ones(m) / m
        
        return lam_clipped, mean_grad, residual
    else:
        # For large active set, use uniform as approximation
        return np.ones(m) / m, 0.0, 1.0


def _kkt_newton_step(a, c, c_max, active_indices, n, mu_reg=1e-10):
    """
    Compute a Newton step using KKT multipliers.
    """
    m = len(active_indices)
    if m == 0:
        return None
    
    # Get KKT weights
    weights, mu_val, resid = _compute_kkt_weights(a, c, c_max, active_indices, n)
    if weights is None:
        return None
    
    # Build the weight vector for the full convolution (length 2n-1)
    w_full = np.zeros(2 * n - 1)
    for idx, k in enumerate(active_indices):
        w_full[int(k)] = weights[idx]
    
    # Gradient direction: d = -2 * corr(a, w_full) (this decreases the weighted sum of peaks)
    grad = 2.0 * _corr_a_A(a, w_full, n)
    grad_t = grad - np.mean(grad)  # Project to maintain sum = 1
    
    # Hessian: H*v = 2 * corr(v, w_full) + mu_reg * v
    # (The Hessian of the weighted sum of active constraints)
    
    def _hess_vp(v):
        return 2.0 * _corr_a_A(v, w_full, n) + mu_reg * v
    
    # Newton step: solve H * dx = -grad_t
    dx = _cg_solve(_hess_vp, -grad_t, max_cg=300, tol=1e-12)
    dx = dx - np.mean(dx)  # Maintain sum = 1
    
    # Check if the step is reasonable
    dx_norm = np.linalg.norm(dx)
    a_norm = np.linalg.norm(a)
    if dx_norm > 10 * a_norm or not np.isfinite(dx_norm):
        return None
    
    return dx, weights, resid


def _spectral_shaping(a, n_iterations=50, alpha_schedule="linear"):
    """
    Iterative spectral shaping to flatten the autocorrelation.
    Modifies the power spectrum to reduce peaks.
    """
    n = len(a)
    N = _next_pow2(2 * n - 1)
    a_cur = a.copy().astype(np.float64)
    
    for it in range(n_iterations):
        # Compute current spectrum
        A = np.fft.rfft(a_cur, N)
        P = np.abs(A) ** 2
        phase = np.angle(A)
        
        # Current autocorrelation
        c = np.fft.irfft(P, N)[:2 * n - 1]
        c_max = float(c.max())
        
        if c_max < 1e-15:
            break
        
        # Identify peaks
        threshold = c_max * 0.95
        peak_lags = np.where(c >= threshold)[0]
        
        # Spectral correction: reduce power at frequencies that contribute to peaks
        # For a peak at lag k, the spectral contribution is |A|^2 * exp(ikw)
        # We reduce |A|^2 at frequencies where the peak is constructive
        
        # Simple approach: create a "notch" in the spectrum
        # The peak at lag k corresponds to correlation at that lag
        # In frequency domain, this is the |A(w)|^2 evaluated with a phase shift
        
        # Practical approach: create a "flat target" and blend
        P_mean = P.mean()
        P_dc = P[0]  # DC component = (sum a)^2
        
        # Target: as flat as possible, but maintaining DC
        # P_target = P_mean for all non-DC, P_dc for DC
        # Blend: P_new = (1-alpha)*P + alpha*P_target
        if alpha_schedule == "linear":
            alpha = 0.3 * (1 - it / max(n_iterations, 1)) + 0.05
        else:
            alpha = 0.1
        
        P_target = np.full_like(P, P_mean)
        P_target[0] = P_dc  # Preserve DC
        
        P_new = (1 - alpha) * P + alpha * P_target
        P_new = np.maximum(P_new, 0)
        
        # New amplitude and phase
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        
        # Inverse FFT to get new sequence
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Check improvement
        sc_new = _score(a_new)
        sc_old = _score(a_cur)
        if sc_new > sc_old * 1.01:
            # Revert if it got too much worse
            a_new = a_cur
            break
        a_cur = a_new
    
    return a_cur


def _spectral_shaping_v2(a, n_iterations=100, beta=0.5):
    """
    Improved spectral shaping using peak-specific corrections.
    """
    n = len(a)
    N = _next_pow2(2 * n - 1)
    a_cur = a.copy().astype(np.float64)
    
    for it in range(n_iterations):
        A = np.fft.rfft(a_cur, N)
        P = np.abs(A) ** 2
        phase = np.angle(A)
        
        c = np.fft.irfft(P, N)[:2 * n - 1]
        c_max = float(c.max())
        if c_max < 1e-15:
            break
        
        # For each significant peak, compute a "deconstructive" spectral correction
        # Peak at lag k: reduce spectral components that are constructive at lag k
        # The contribution of frequency w to lag k is |A(w)|^2 * exp(i*w*k)
        # To reduce the peak at lag k, we want to reduce |A(w)|^2 where exp(i*w*k) aligns
        
        # Practical: compute the "peak spectrum" and subtract it
        peak_mask = c >= c_max * 0.9
        n_peaks = max(int(peak_mask.sum()), 1)
        
        # Weighted autocorrelation of peaks
        w = np.zeros(2 * n - 1)
        w[peak_mask] = 1.0 / n_peaks
        
        # The "peak contribution" to the spectrum
        # c_peak = sum_k w[k] * c[k] = IDFT(P) weighted
        # We want to reduce P where it contributes to c_peak
        
        # Gradient of c_peak w.r.t. P: it's the IDFT basis vectors
        # d(c_peak)/d(P[w]) = (1/N) * sum_k w[k] * exp(2*pi*i*w*k/N)
        # = (1/N) * FFT(w)[w]
        
        w_padded = np.zeros(N)
        w_padded[:2 * n - 1] = w
        w_ft = np.fft.rfft(w_padded)
        
        # Reduce P where the gradient is positive (contributes to peak)
        reduction = beta * np.abs(w_ft) ** 2 / max(np.max(np.abs(w_ft) ** 2), 1e-30)
        reduction = np.minimum(reduction, 0.5)  # Don't reduce by more than 50%
        
        P_new = P * (1 - reduction)
        P_new[0] = P[0]  # Preserve DC
        
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        a_cur = a_new
    
    return a_cur


def _active_set_optimize(a_init, deadline, max_outer=200, tol_list=None):
    """
    Active-set Newton optimization with KKT multiplier computation.
    """
    if tol_list is None:
        tol_list = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]
    
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Try different active set tolerances
        found_step = False
        for tol_frac in tol_list:
            if time.monotonic() > deadline:
                break
            
            threshold = c_max * (1 - tol_frac)
            active_mask = c >= threshold
            n_active = int(active_mask.sum())
            
            if n_active < 1:
                continue
            if n_active > min(500, n):
                # Too many active - use only the top ones
                top_indices = np.argsort(c)[::-1][:min(200, n_active)]
                active_mask = np.zeros_like(c, dtype=bool)
                active_mask[top_indices] = True
                n_active = len(top_indices)
            
            active_indices = np.where(active_mask)[0]
            
            # Compute Newton step with KKT weights
            result = _kkt_newton_step(a, c, c_max, active_indices, n)
            if result is None:
                continue
            dx, weights, resid = result
            
            # Line search
            t = 1.0
            for ls in range(60):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < c_max - 1e-15:
                    improvement = c_max - f_try
                    a = a_try
                    found_step = True
                    break
                t *= 0.5
            
            if found_step:
                break
        
        if not found_step:
            # Gradient fallback
            c = _full_convolve(a, a)
            c_max = float(c.max())
            peak_idx = int(np.argmax(c))
            w_single = np.zeros(2 * n - 1)
            w_single[peak_idx] = 1.0
            grad = 2.0 * _corr_a_A(a, w_single, n)
            grad_t = grad - np.mean(grad)
            
            t = 0.01
            applied = False
            for ls in range(40):
                a_try = a - t * grad_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_sc:
                        a = a_try
                        applied = True
                        break
                t *= 0.5
            if not applied:
                break
        
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Convergence check
        if outer > 5:
            c_new = _full_convolve(a, a)
            if abs(float(c_new.max()) - c_max) < 1e-15:
                break
    
    return best_a, best_sc


def _lsobj_factory(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad_factory(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_run(a_init, deadline, min_time_remaining=30.0, max_betas=20):
    """L-BFGS with continuation on beta."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - c_sorted[20]
    else:
        gap = c_sorted[0] - c_sorted[-1]
    base_beta = max(gap * 0.5, max_c * 0.01)
    
    n_betas = max_betas
    log_min = max(np.log(max_c * 1e-8), -40)
    log_betas = np.linspace(np.log(base_beta), log_min, n_betas)
    betas = np.exp(log_betas)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break
        
        est_time = 0.002 * max(1.0, n / 8192.0)
        max_iter = min(3000, max(20, int(remaining / (est_time * 8))))
        
        try:
            def _obj(z_val):
                return _lsobj_factory(z_val, beta, n)
            
            def _grad(z_val):
                return _lsgrad_factory(z_val, beta, n)
            
            result = minimize(
                _obj, z, jac=_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.10f} t={time.monotonic()-t_start_global:.0f}s")
        except Exception as ex:
            print(f"    L-BFGS beta={beta:.2e}: exception {ex}")
    
    return current_a


t_start_global = 0.0


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bipolar":
        # Two Gaussian bumps
        t = np.linspace(0, 1, n)
        a = np.exp(-30 * (t - 0.25) ** 2) + np.exp(-30 * (t - 0.75) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _refinement_loop(a, deadline, max_rounds=15):
    """Iterate between active-set Newton and spectral shaping."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        # Active-set Newton
        as_dl = min(deadline - 10, time.monotonic() + 60)
        a_newton, sc_newton = _active_set_optimize(a, as_dl, max_outer=50,
                                                    tol_list=[0.005, 0.002, 0.001, 0.0005, 0.0001])
        if sc_newton < best_sc:
            best_sc = sc_newton
            best_a = a_newton.copy()
            a = a_newton
        
        # Spectral shaping (gentle)
        if time.monotonic() < deadline - 20:
            a_spec = _spectral_shaping_v2(a, n_iterations=20, beta=0.1)
            sc_spec = _score(a_spec)
            if sc_spec < best_sc:
                best_sc = sc_spec
                best_a = a_spec.copy()
                a = a_spec
        
        # Check convergence
        if rnd > 3:
            sc_cur = _score(a)
            if abs(sc_cur - best_sc) < 1e-14:
                break
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining spectral shaping, KKT active-set Newton,
    and L-BFGS continuation.
    """
    global t_start_global
    t_start_global = time.monotonic()
    deadline = t_start_global + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start_global:.1f}s")
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass
    
    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={len(parent)}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        parent = _make_profile(16384, "uniform")
        update_best(parent, _score(parent), "init_uniform")
    
    n_work = len(best_seq) if best_seq is not None else 16384
    
    # === Phase 1: Spectral pre-shaping ===
    print(f"\n[Phase 1] Spectral pre-shaping (t={time.monotonic() - t_start_global:.1f}s)")
    spec_dl = min(deadline - 800, time.monotonic() + 300)
    a_spec1 = _spectral_shaping_v2(best_seq, n_iterations=80, beta=0.3)
    update_best(a_spec1, _score(a_spec1), "spectral_v2_0.3")
    
    a_spec2 = _spectral_shaping(best_seq, n_iterations=40)
    update_best(a_spec2, _score(a_spec2), "spectral_v1")
    
    # === Phase 2: Active-set Newton (main refinement) ===
    print(f"\n[Phase 2] KKT Active-set Newton (t={time.monotonic() - t_start_global:.1f}s)")
    as_dl = min(deadline - 500, time.monotonic() + 250)
    a_as, sc_as = _active_set_optimize(best_seq, as_dl, max_outer=200,
                                        tol_list=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    update_best(a_as, sc_as, "kkt_newton")
    
    # === Phase 3: L-BFGS continuation ===
    print(f"\n[Phase 3] L-BFGS continuation (t={time.monotonic() - t_start_global:.1f}s)")
    lbfgs_dl = min(deadline - 300, time.monotonic() + 200)
    a_lbfgs = _lbfgs_run(best_seq, lbfgs_dl, min_time_remaining=300.0, max_betas=20)
    update_best(a_lbfgs, _score(a_lbfgs), "lbfgs")
    
    # === Phase 4: Combined refinement loop ===
    print(f"\n[Phase 4] Refinement loop (t={time.monotonic() - t_start_global:.1f}s)")
    ref_dl = min(deadline - 200, time.monotonic() + 150)
    a_ref, sc_ref = _refinement_loop(best_seq, ref_dl, max_rounds=10)
    update_best(a_ref, sc_ref, "refinement_loop")
    
    # === Phase 5: Try different n ===
    print(f"\n[Phase 5] Multi-scale (t={time.monotonic() - t_start_global:.1f}s)")
    for n_try in [8192, 32768]:
        if time.monotonic() > deadline - 100:
            break
        a_try = _interpolate_seq(best_seq, n_try)
        sc_try = _score(a_try)
        print(f"  n={n_try}: initial score={sc_try:.10f}")
        
        # Quick optimization
        opt_dl = min(deadline - 80, time.monotonic() + 60)
        a_opt, sc_opt = _active_set_optimize(a_try, opt_dl, max_outer=50,
                                              tol_list=[0.01, 0.005, 0.001])
        update_best(a_opt, sc_opt, f"n_{n_try}")
    
    # === Phase 6: Diversified restarts ===
    print(f"\n[Phase 6] Diversified restarts (t={time.monotonic() - t_start_global:.1f}s)")
    restart_configs = [
        (0.003, "tiny"), (0.01, "small"), (0.03, "med"), (0.08, "large"),
    ]
    for sigma, label in restart_configs:
        if time.monotonic() > deadline - 60:
            break
        p_start = _perturb(best_seq, sigma, rng)
        # Quick active-set on perturbed
        rst_dl = min(deadline - 50, time.monotonic() + 40)
        a_r, sc_r = _active_set_optimize(p_start, rst_dl, max_outer=30,
                                          tol_list=[0.01, 0.005, 0.001, 0.0005])
        update_best(a_r, sc_r, f"restart_{label}")
    
    # === Phase 7: Final polish ===
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start_global:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 15:
        # Very fine active-set
        fin_dl = min(deadline - 5, time.monotonic() + 40)
        a_fin, sc_fin = _active_set_optimize(best_seq, fin_dl, max_outer=100,
                                              tol_list=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        update_best(a_fin, sc_fin, "final_finewell")
        
        # Tiny beta L-BFGS
        if time.monotonic() < deadline - 10:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 1e-7
            remaining = deadline - 12 - time.monotonic()
            if remaining > 5:
                max_iter_fin = min(2000, max(50, int(remaining / 0.05)))
                try:
                    def _obj_fin(z_val):
                        return _lsobj_factory(z_val, beta_final, n)
                    def _grad_fin(z_val):
                        return _lsgrad_factory(z_val, beta_final, n)
                    result = minimize(_obj_fin, z, jac=_grad_fin, method='L-BFGS-B',
                                      options={'maxiter': max_iter_fin, 'ftol': 1e-18,
                                               'gtol': 1e-15, 'maxcor': 100})
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_tiny_beta")
                except Exception:
                    pass
    
    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start_global:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
