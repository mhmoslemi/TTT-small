def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _binomial_init(n):
    """Initialize with truncated binomial square-root coefficients."""
    # a_k proportional to binom(2k, k) / 4^k
    ks = np.arange(n, dtype=np.float64)
    # Use the recurrence: a_0 = 1, a_k = a_{k-1} * (2k-1)/(2k)
    a = np.ones(n)
    for k in range(1, n):
        a[k] = a[k-1] * (2*k - 1) / (2*k)
    s = a.sum()
    return a / s


def _power_law_init(n, gamma):
    """Initialize with power law a_k ∝ (k+1)^(-gamma)."""
    ks = np.arange(1, n + 1, dtype=np.float64)
    a = ks ** (-gamma)
    return a / a.sum()

    ks = np.arange(n, dtype=np.float64)

    # Hmm, let me use the same approach as _variance_gradient:
    N = _next_pow2(2 * n)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    cross = np.fft.irfft(fw * np.conj(fa), N)[:n]  # T1[i] = sum_k w[k]*a[k-i]
    conv_aw = np.fft.irfft(fa * fw, N)[:n]  # T2[i] = sum_k w[k]*a[i-k]
    
    return 2.0 * (cross + conv_aw)


def _flattening_optimize(a_init, deadline, min_time_remaining=60.0):
    """
    Phase 1: Gradient descent on the variance of conv(a,a).
    This reduces the spread of peaks, which helps before targeting the max specifically.
    Uses Nesterov momentum.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.01
    momentum = 0.9
    n_iter = 0
    
    while time.monotonic() < deadline - min_time_remaining:
        c = _full_convolve(a, a)
        grad = _variance_gradient(a, c, n)
        
        # Nesterov: look-ahead
        a_look = a - momentum * v
        a_look = np.maximum(a_look, 0)
        s_look = a_look.sum()
        if s_look > 1e-15:
            a_look = a_look / s_look
            c_look = _full_convolve(a_look, a_look)
            grad_look = _variance_gradient(a_look, c_look, n)
            # Project gradient onto tangent space of simplex
            grad_look = grad_look - grad_look.mean()
            v = momentum * v + lr * grad_look
        else:
            v = momentum * v + lr * grad
        
        # Step
        a_new = a - v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            v *= 0.5
            continue
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        
        a = a_new
        n_iter += 1
        
        if n_iter % 100 == 0:
            print(f"    Flattening iter {n_iter}: score={sc_new:.12f}, best={best_sc:.12f}")
        
        # Adaptive learning rate
        if n_iter % 50 == 0:
            lr *= 0.95
    
    return best_a


def _lsexp_optimize(a_init, deadline, min_time_remaining=30.0):
    """
    Phase 2: Log-sum-exp smoothing with aggressive beta continuation.
    Uses direct gradient descent (not L-BFGS) for more control.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    c_cur = _full_convolve(a, a)
    c_max_init = c_cur.max()
    
    # Beta schedule: from relatively large to very small
    betas = np.concatenate([
        np.linspace(c_max_init * 0.05, c_max_init * 0.01, 10),
        np.linspace(c_max_init * 0.01, c_max_init * 0.001, 10),
        np.linspace(c_max_init * 0.001, c_max_init * 0.0001, 10),
        np.linspace(c_max_init * 0.0001, c_max_init * 0.00001, 10),
    ])
    
    for beta in betas:
        if time.monotonic() > deadline - min_time_remaining:
            break
        
        # For this beta, do several gradient steps
        n_steps = max(10, min(100, int((deadline - min_time_remaining - time.monotonic()) / (len(betas) * 0.05))))
        
        for step in range(n_steps):
            if time.monotonic() > deadline - min_time_remaining:
                break
            
            c = _full_convolve(a, a)
            c_max = c.max()
            
            # Check if beta is too large relative to current max
            if beta > c_max * 0.5:
                break
            
            grad = _max_gradient(a, c, n, beta=beta)
            # Project onto tangent space
            grad = grad - grad.mean()
            
            # Adaptive step size
            lr = 0.01 * beta / max(c_max, 1e-10)
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            
            sc_new = _score(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
            
            a = a_new
    
    return best_a


def _newton_polish(a_init, deadline, max_iterations=50):
    """
    Phase 3: Active-set Newton method for final polish.
    Uses the structural insight that the Hessian is Hankel-like.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iterations):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Active set: positions within a small fraction of max
        for tol in [0.001, 0.0005, 0.0001]:
            active = c >= c_max * (1 - tol)
            if active.sum() < 2:
                continue
            if active.sum() > n:
                continue
            
            m = int(active.sum())
            A_vec = active.astype(np.float64) / m
            
            # Gradient of sum_{k in active} c[k] / m
            # = (1/m) * sum_k active[k] * grad c[k]
            N = _next_pow2(2 * n)
            fA = np.fft.rfft(A_vec, N)
            fa = np.fft.rfft(a, N)
            cross = np.fft.irfft(fA * np.conj(fa), N)[:n]
            conv_ = np.fft.irfft(fa * fA, N)[:n]
            g = 2.0 * (cross + conv_)
            g_t = g - g.mean()
            
            # Hessian: H[i,j] = (2/m) * |{k in S : i+j=k}| (with diagonal correction)
            # Build the Hessian-vector product using FFT
            A_vec_ = A_vec  # active / m
            
            def hess_vp(v):
                # (H @ v)[i] = sum_j H[i,j] * v[j] = (2/m) * sum_j |{k in S: i+j=k}| * v[j]
                # = (2/m) * sum_k active_count(i ... ) 
                # = (2/m) * sum_k A_vec[k] * (sum_j: i+j=k v[j])
                # = (2/m) * sum_k A_vec[k] * v[k-i] (where v is zero-padded)
                # This is a convolution-like operation
                N2 = _next_pow2(2 * n)
                fv = np.fft.rfft(v, N2)
                fAv = np.fft.rfft(A_vec_, N2)
                # We need: (2/m) * sum_k A_vec_[k] * v[k-i]
                # = (2/m) * conv-related... 
                # Actually: sum_k A_vec[k] * v[k-i] for fixed i
                # Let j = k-i: sum_j A_vec[j+i] * v[j]
                # This is ifft(FFT(A_vec) * conj(FFT(v)))[i] (cross-correlation)
                cross_h = np.fft.irfft(fAv * np.conj(fv), N2)[:n]
                return 2.0 * cross_h
            
            # Actually wait, I need to be more careful.
            # H[i,j] = (2/m) * N(i+j) where N(s) = |{k in S: k = s}|
            # For the active set indicator A_vec (which is 1/m on active, 0 elsewhere):
            # H[i,j] = 2 * A_vec[i+j] (for i+j in [0, 2n-2])
            # (with the diagonal correction: if i=j and 2i is in S, the factor is 1 not 2, but this is a minor effect)
            
            # So (H @ v)[i] = 2 * sum_j A_vec[i+j] * v[j]
            # = 2 * sum_j A_vec[i+j] * v[j]
            # Let p = i+j: = 2 * sum_p A_vec[p] * v[p-i]
            # = 2 * (A_vec ⋆ v)[i] (cross-correlation of A_vec and v at lag i)
            # = 2 * ifft(FFT(A_vec) * conj(FFT(v)))[i]
            
            # Good, so hess_vp is correct.
            
            # Add regularization for the sum constraint
            mu = 1e-8
            def hess_vp_reg(v):
                return 2.0 * np.fft.irfft(np.fft.rfft(A_vec_, _next_pow2(2*n)) * np.conj(np.fft.rfft(v, _next_pow2(2*n))), _next_pow2(2*n))[:n] + mu * v
            
            # Solve H @ dx = -g_t using CG
            # (with the constraint that sum(dx) = 0 handled by projecting g_t)
            x = np.zeros(n)
            r = -g_t - hess_vp_reg(x)
            p_vec = r.copy()
            rs = np.dot(r, r)
            for cg_iter in range(100):
                Ap = hess_vp_reg(p_vec)
                pAp = np.dot(p_vec, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x = x + alpha * p_vec
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-12:
                    break
                p_vec = r_new + (rs_new / rs) * p_vec
                rs = rs_new
                r = r_new
            
            dx = x - np.mean(x)
            
            # Line search
            t = 1.0
            for ls in range(40):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = c_try.max()
                if f_try < c_max - 1e-14:
                    sc_try = _score(a_try)
                    if sc_try < best_sc:
                        best_sc = sc_try
                        best_a = a_try.copy()
                    a = a_try
                    break
                t *= 0.5
            else:
                break
        
        # Check convergence
        c_final = _full_convolve(a, a)
        if abs(c_final.max() - c_max) < 1e-15:
            break
    
    return best_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "binomial":
        return _binomial_init(n)
    elif kind == "power_law":
        return _power_law_init(n, 0.5)
    elif kind == "exp":
        return _exp_init(n, 2.0)
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-3 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _lbfgs_polar(a_init, deadline, min_time_remaining=30.0):
    """
    L-BFGS optimization in log-space with log-sum-exp objective.
    """
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    # Beta schedule
    base_beta = max_c * 0.01
    betas = np.geomspace(base_beta, max_c * 1e-6, 20)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        def _obj(z_val):
            a_val = _softmax(z_val)
            c_val = _full_convolve(a_val, a_val)
            c_max_val = float(c_val.max())
            shifted = (c_val - c_max_val) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                return 2.0 * n * c_max_val
            return 2.0 * n * (c_max_val + beta * np.log(s))
        
        def _grad(z_val):
            a_val = _softmax(z_val)
            c_val = _full_convolve(a_val, a_val)
            c_max_val = float(c_val.max())
            shifted = (c_val - c_max_val) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c_val)
                w[int(np.argmax(c_val))] = 1.0
            else:
                w = e / s
            # Gradient w.r.t. z: d/dz_i [2*n*beta*log(sum exp((c-cmax)/beta))]
            # = 2*n * sum_k w[k] * dc_k/dz_i
            # dc_k/dz_i = dc_k/da_i * da_i/dz_i = (a[k-i]+a[i-k]) * a_i * (delta_{ii} - a_i)
            # So: grad_i = 2*n*a_i*(sum_k w[k]*(a[k-i]+a[i-k]) - sum_k w[k]*c_k)
            
            N = _next_pow2(2 * n)
            fw = np.fft.rfft(w, N)
            fa = np.fft.rfft(a_val, N)
            cross = np.fft.irfft(fw * np.conj(fa), N)[:n]
            conv_w = np.fft.irfft(fa * fw, N)[:n]
            h = 2.0 * (cross + conv_w)  # This is sum_k w[k]*(a[k-i]+a[i-k]) * 2 = grad of sum w*c w.r.t. a
            W = float(np.dot(w, c_val))
            # h[i] = 2*sum_k w[k]*(a[k-i]+a[i-k]), so sum_k w[k]*(a[k-i]+a[i-k]) = h[i]/2
            # Actually h[i] IS 2*(T1+T2) = 2*sum_k w[k]*(a[k-i]+a[i-k])... 
            # Wait: T1[i] = sum_k w[k]*a[k-i], T2[i] = sum_k w[k]*a[i-k]
            # h[i] = 2*(T1[i]+T2[i]) = 2*sum_k w[k]*(a[k-i]+a[i-k])
            # But I want sum_k w[k]*(a[k-i]+a[i-k]) = h[i]/2
            # 
            # Actually, for the gradient of sum_k w[k]*c[k] w.r.t. a[i]:
            # = sum_k w[k] * dc_k/da_i = sum_k w[k] * (a[k-i] + a[i-k])
            # = T1[i] + T2[i] = h[i] / 2
            # 
            # And dc_k/dz_i = (dc_k/da_i) * (da_i/dz_i) = (a[k-i]+a[i-k]) * a_i * (1 - a_i) (for i=i)
            # Wait, softmax: a_i = exp(z_i) / sum_j exp(z_j)
            # da_i/dz_j = a_i * (delta_{ij} - a_j)
            # 
            # So: d/dz_i [sum_k w[k]*c[k]] = sum_k w[k] * sum_j dc_k/da_j * da_j/dz_i
            # = sum_k w[k] * [dc_k/da_i * a_i*(1-a_i) + sum_{j≠i} dc_k/da_j * (-a_i*a_j)]
            # = a_i * [sum_k w[k]*dc_k/da_i - a_i * sum_k w[k]*sum_j dc_k/da_j]
            # = a_i * [g_i - a_i * sum_j g_j] where g_i = sum_k w[k]*dc_k/da_i
            # = a_i * (g_i - a_i * G) where G = sum_i g_i = sum_k w[k]*c[k] (since sum_i dc_k/da_i = c[k])
            # 
            # So grad_z[i] = 2*n*beta*a_i*(g_i - a_i*W) where g_i = h[i]/2
            # = 2*n*beta*a_i*(h[i]/2 - a_i*W)
            # = n*beta*a_i*(h[i] - 2*a_i*W)
            
            return 2.0 * n * beta * a_val * (h / 2.0 - a_val * W)
        
        try:
            max_iter = min(2000, max(30, int(remaining / (0.01 * max(1, n/8192)))))
            result = minimize(
                _obj,
                z,
                jac=_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxcor': 50, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.12f}")
        except Exception as ex:
            print(f"    L-BFGS beta={beta:.2e}: exception {ex}")
    
    return current_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    1. Binomial square-root + parametric initializations
    2. Variance-flattening gradient descent with momentum
    3. Log-sum-exp L-BFGS with aggressive continuation
    4. Active-set Newton polish
    5. Multi-scale and diversification
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            t_now = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.12f} (n={len(p_arr)}) [{label}] improve={prev - best_score:.2e} t={t_now:.1f}s")
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    # === Phase 0: Try multiple initializations ===
    print(f"[Phase 0] Trying initializations (t={time.monotonic()-t_start:.1f}s)")
    
    init_candidates = {}
    
    if parent is not None:
        init_candidates['parent'] = parent
        sc_p = _score(parent)
        update_best(parent, sc_p, "parent")
        print(f"  Parent: n={len(parent)}, score={sc_p:.12f}")
    
    # Try n=16384 with various inits
    n_base = 16384
    inits = [
        ('binomial', _binomial_init(n_base)),
        ('power05', _power_law_init(n_base, 0.5)),
        ('power03', _power_law_init(n_base, 0.3)),
        ('exp2', _exp_init(n_base, 2.0)),
        ('exp4', _exp_init(n_base, 4.0)),
        ('uniform', np.ones(n_base) / n_base),
    ]
    
    if parent is not None and len(parent) != n_base:
        inits.append(('parent_interp', _interpolate_seq(parent, n_base)))
    
    for name, a_init in inits:
        sc_init = _score(a_init)
        print(f"  Init '{name}': n={n_base}, score={sc_init:.12f}")
        init_candidates[name] = a_init
        update_best(a_init, sc_init, f"init_{name}")
    
    # Pick the best initialization
    best_init_name = min(init_candidates.keys(), key=lambda k: _score(init_candidates[k]))
    best_init = init_candidates[best_init_name].copy()
    print(f"  Best init: {best_init_name}")
    
    # === Phase 1: Variance flattening ===
    print(f"\n[Phase 1] Variance flattening (t={time.monotonic()-t_start:.1f}s)")
    flat_dl = min(deadline - 800, time.monotonic() + 200)
    a_flat = _flattening_optimize(best_init, flat_dl, min_time_remaining=800)
    update_best(a_flat, _score(a_flat), "flattening")
    
    # Also try flattening from parent if available
    if parent is not None:
        flat_dl2 = min(deadline - 700, time.monotonic() + 100)
        a_flat2 = _flattening_optimize(parent, flat_dl2, min_time_remaining=700)
        update_best(a_flat2, _score(a_flat2), "flattening_parent")
    
    # === Phase 2: L-BFGS with log-sum-exp ===
    print(f"\n[Phase 2] L-BFGS log-sum-exp (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        lbfgs_dl = min(deadline - 500, time.monotonic() + 300)
        a_lbfgs = _lbfgs_polar(best_seq, lbfgs_dl, min_time_remaining=500)
        update_best(a_lbfgs, _score(a_lbfgs), "lbfgs")
    
    # === Phase 3: Newton polish ===
    print(f"\n[Phase 3] Newton polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
