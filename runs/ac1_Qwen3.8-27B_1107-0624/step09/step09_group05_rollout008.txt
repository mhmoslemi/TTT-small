"""
Multi-algorithm optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines:
1. Spectral flattening (POCS-inspired) for global shape optimization
2. L-BFGS with adaptive beta on soft-max parameterization
3. KKT active-set refinement
4. Multi-n and multi-profile exploration
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)

def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta

Combines spectral flattening (POCS-inspired), adaptive L-BFGS on soft-max,

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft_structured(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)

def _spectral_flatten(a, n_iters=100, alpha_start=0.95, alpha_end=0.5, deadline=None, min_time=10.0):
    """Spectral flattening via iterative magnitude compression."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    N = _next_pow2(2 * n - 1)

    for i in range(n_iters):
        if deadline is not None and time.monotonic() + min_time > deadline:
            break
        # Interpolate alpha
        frac = i / max(1, n_iters - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        # Compress magnitude
        A_new = np.power(mag, alpha) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n].real
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        sc = _score(a_new)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
    return best_a

    min_tol_frac = 0.0001

        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

            no_improve = 0
            step_scale = max(0.01, step_scale * 0.7)

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        gap = current_sc - lb
        if gap < 1e-10:
            break

        grad_c = grad - np.mean(grad)
        g_norm_sq = float(np.dot(grad_c, grad_c))
        if g_norm_sq < 1e-20:
            break

        step = step_scale * gap / g_norm_sq
        step = min(step, 0.05)

        a_new = a - step * grad_c
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            step_scale *= 0.5
            continue
        a_new = a_new / s_new
        a = a_new
        t += 1

        if t % 200 == 0:
            step_scale = max(0.01, step_scale * 0.9)

    return best_a, best_sc


def _active_set_refine(a_init, deadline, min_time_remaining=20.0):
    """KKT-inspired active-set refinement with line search."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_frac = 0.008
    min_tol_frac = 0.0002
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-12:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(min_tol_frac, tol_frac * 0.6)
            if tol_frac <= min_tol_frac * 1.2:
                break
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]

        support = a > 1e-10 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        lam = float(np.mean(grad_avg[support]))
        residual = grad_avg[support] - lam

        a_new = a.copy()
        eta = 0.08
        a_new[support] -= eta * residual
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new >= current_sc - 1e-12:
            found = False
            for trial_eta in [0.04, 0.02, 0.01, 0.005]:
                a_try = a.copy()
                a_try[support] -= trial_eta * residual
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-12:
                    a_new = a_try
                    sc_new = sc_try
                    found = True
                    break
            if not found:
                tol_frac = max(min_tol_frac, tol_frac * 0.8)
                continue

        a = a_new
        if no_improve == 0:
            tol_frac = max(min_tol_frac, tol_frac * 0.96)

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, verbose=False):
    """L-BFGS with adaptive beta schedule and early termination."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(min(50, len(c_sorted) - 1), len(c_sorted) - 1)]
    if gap < c0_max * 0.005:
        gap = c0_max * 0.005

    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.00006,
        0.00003, 0.000015, 0.000008, 0.000004, 0.000002, 0.000001
    ])
    betas = gap * multipliers

    # Adaptive: stop if no improvement for 3 consecutive betas
    no_improve_count = 0
    max_no_improve = 5

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        size_factor = max(1.0, n / 4096.0)
        max_iter = min(6000, max(200, int(remaining / (0.003 * size_factor * 30))))
        if beta < gap * 0.001:
            max_iter = min(max_iter, 3000)

        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-13:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                no_improve_count = 0
                if verbose:
                    print(f"    beta={beta:.2e} ({i+1}/{len(betas)}): score={best_sc:.10f}, nit={result.nit}")
            else:
                no_improve_count += 1
                if no_improve_count > max_no_improve:
                    if verbose:
                        print(f"    Stopping L-BFGS after {no_improve_count} non-improving betas")
                    break
        except Exception:
            no_improve_count += 1
            if no_improve_count > max_no_improve:
                break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-60.0 * (t - 0.25) ** 2) + np.exp(-60.0 * (t - 0.75) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 7.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "one_sided":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * t) + 0.005
        return a / a.sum()
    elif kind == "one_sided_strong":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * t) + 0.002
        return a / a.sum()
    elif kind == "quadratic_taper":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.7 * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-5.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    elif kind == "skewed":
        t = np.linspace(0, 1, n)
        a = (t ** 0.5) * np.exp(-3.0 * t) + 0.005
        return a / a.sum()
    elif kind == "chebyshev":
        t = np.linspace(0, 1, n)
        a = np.abs(np.cos(np.pi * t)) + 0.01
        return a / a.sum()
    elif kind == "random_smooth":
        rng = np.random.default_rng(12345)
        n_coarse = max(10, n // 64)
        coarse = rng.random(n_coarse) + 0.1
        a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_coarse), coarse)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _optimize_candidate(a_init, deadline, min_time_remaining=30.0, verbose=False):
    """Full optimization pipeline on a single candidate."""
    if time.monotonic() + min_time_remaining > deadline:
        return a_init, _score(a_init)

    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Step 1: Spectral flattening (quick global improvement)
    if time.monotonic() + min_time_remaining > deadline:
        return best_a, best_sc
    spec_end = min(deadline - min_time_remaining - 60, time.monotonic() + 30)
    a_spec = _spectral_flatten(a, n_iters=50, alpha_start=0.9, alpha_end=0.6,
                               deadline=spec_end, min_time=5.0)
    sc_spec = _score(a_spec)
    if sc_spec < best_sc:
        best_sc = sc_spec
        best_a = a_spec.copy()
        if verbose:
            print(f"      Spectral: {best_sc:.10f}")

    # Step 2: L-BFGS refinement
    if time.monotonic() + min_time_remaining > deadline:
        return best_a, best_sc
    lbfgs_end = min(deadline - min_time_remaining - 30, time.monotonic() + 120)
    a_lbfgs, sc_lbfgs = _run_lbfgs(best_a, lbfgs_end, min_time_remaining=10, verbose=verbose)
    if sc_lbfgs < best_sc:
        best_sc = sc_lbfgs
        best_a = a_lbfgs.copy()
        if verbose:
            print(f"      L-BFGS: {best_sc:.10f}")

    # Step 3: Active set refinement
    if time.monotonic() + min_time_remaining > deadline:
        return best_a, best_sc
    as_end = min(deadline - min_time_remaining, time.monotonic() + 60)
    a_as, sc_as = _active_set_refine(best_a, as_end, min_time_remaining=5)
    if sc_as < best_sc:
        best_sc = sc_as
        best_a = a_as.copy()
        if verbose:
            print(f"      ActiveSet: {best_sc:.10f}")

    # Step 4: Subgradient
    if time.monotonic() + min_time_remaining > deadline:
        return best_a, best_sc
    sg_end = min(deadline - min_time_remaining, time.monotonic() + 40)
    a_sg, sc_sg = _subgradient_refine(best_a, sg_end, min_time_remaining=5)
    if sc_sg < best_sc:
        best_sc = sc_sg
        best_a = a_sg.copy()
        if verbose:
            print(f"      Subgrad: {best_sc:.10f}")

    # Step 5: Final L-BFGS polish
    if time.monotonic() + min_time_remaining > deadline:
        return best_a, best_sc
    final_end = min(deadline - 2, time.monotonic() + 30)
    a_final, sc_final = _run_lbfgs(best_a, final_end, min_time_remaining=2, verbose=False)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining spectral flattening, L-BFGS,
    active-set KKT refinement, and diverse profile/n exploration.
    """
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(8192, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n=8192: score={best_score:.10f}")

    n_main = len(best_seq)

    # === Phase 1: Best-optimize the parent ===
    if time.monotonic() + 60 < deadline:
        print(f"\n[P1] Full optimization of parent (n={n_main}, t={time.monotonic()-t0:.0f}s)")
        p1_end = min(deadline - 300, time.monotonic() + 400)
        a1, sc1 = _optimize_candidate(best_seq, p1_end, min_time_remaining=200, verbose=True)
        if sc1 < best_score:
            best_score = sc1
            best_seq = a1.copy()
            print(f"  P1 result: {best_score:.10f}")

    # === Phase 2: Spectral + L-BFGS on diverse profiles at same n ===
    print(f"\n[P2] Diverse profiles (t={time.monotonic()-t0:.0f}s)")
    profiles = ["one_sided_strong", "skewed", "chebyshev", "random_smooth",
                "bimodal", "periodic", "quadratic_taper"]
    for prof in profiles:
        if time.monotonic() + 50 > deadline:
            break
        a_p = _make_profile(n_main, prof)
        sc_p = _score(a_p)
        if verbose_sc := (sc_p < best_score + 0.3):
            p2_end = min(deadline - 100, time.monotonic() + 120)
            a_p_opt, sc_p_opt = _optimize_candidate(a_p, p2_end, min_time_remaining=20, verbose=False)
            if sc_p_opt < best_score:
                best_score = sc_p_opt
                best_seq = a_p_opt.copy()
                print(f"  {prof}: {best_score:.10f}")

    # === Phase 3: Multi-n exploration ===
    print(f"\n[P3] Multi-n exploration (t={time.monotonic()-t0:.0f}s)")
    n_main_now = len(best_seq)
    n_options = [4096, 8192, 12288, 16384, 20480, 24576]
    n_options = [x for x in n_options if x != n_main_now]
    # Sort by proximity to current best n first, then by size
    n_options.sort(key=lambda x: abs(x - n_main_now))
    for n_try in n_options[:4]:
        if time.monotonic() + 80 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if time.monotonic() + 60 > deadline:
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  n={n_try} (resample only): {best_score:.10f}")
            continue
        p3_end = min(deadline - 60, time.monotonic() + 150)
        a_r_opt, sc_r_opt = _optimize_candidate(a_r, p3_end, min_time_remaining=30, verbose=False)
        if sc_r_opt < best_score:
            best_score = sc_r_opt
            best_seq = a_r_opt.copy()
            print(f"  n={n_try}: {best_score:.10f}")

    # === Phase 4: Spectral exploration with different alphas ===
    print(f"\n[P4] Spectral exploration (t={time.monotonic()-t0:.0f}s)")
    if time.monotonic() + 40 < deadline:
        a_spec = _spectral_flatten(best_seq, n_iters=200, alpha_start=0.99, alpha_end=0.3,
                                   deadline=min(deadline - 20, time.monotonic() + 120), min_time=10.0)
        sc_spec = _score(a_spec)
        if sc_spec < best_score:
            best_score = sc_spec
            best_seq = a_spec.copy()
            print(f"  Spectral deep: {best_score:.10f}")
        # L-BFGS after spectral
        if time.monotonic() + 30 < deadline:
            p4_end = min(deadline - 20, time.monotonic() + 100)
            a4, sc4 = _run_lbfgs(best_seq, p4_end, min_time_remaining=10, verbose=False)
            if sc4 < best_score:
                best_score = sc4
                best_seq = a4.copy()
                print(f"  After spectral+L-BFGS: {best_score:.10f}")

    # === Phase 5: Final intensive refinement ===
    print(f"\n[P5] Final refinement (t={time.monotonic()-t0:.0f}s)")
    if time.monotonic() + 20 < deadline:
        p5_end = min(deadline - 5, time.monotonic() + 180)
        a5, sc5 = _optimize_candidate(best_seq, p5_end, min_time_remaining=5, verbose=True)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  P5 result: {best_score:.10f}")

    # === Phase 6: One more active-set pass ===
    if time.monotonic() + 15 < deadline:
        p6_end = min(deadline - 2, time.monotonic() + 60)
        a6, sc6 = _active_set_refine(best_seq, p6_end, min_time_remaining=2)
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  P6 active-set: {best_score:.10f}")

    # === Final subgradient polish ===
    if time.monotonic() + 10 < deadline:
        p7_end = min(deadline - 1, time.monotonic() + 40)
        a7, sc7 = _subgradient_refine(best_seq, p7_end, min_time_remaining=1)
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()
            print(f"  P7 subgrad: {best_score:.10f}")

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t0:.1f}s")
    return [float(x) for x in final]
