def project_simplex(v):
    """Project v onto the probability simplex {a: a_i >= 0, sum(a) = 1}."""
    n = len(v)
    if v.sum() <= 1:
        # Already in the simplex if all non-negative
        v = np.maximum(v, 0)
        s = v.sum()
        if s > 0:
            return v / s
        return np.ones(n) / n
    
    # Use the standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
