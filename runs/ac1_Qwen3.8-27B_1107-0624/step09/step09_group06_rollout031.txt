def _gradient_of_conv_at_k(a, k):
    """Return gradient of conv(a,a)[k] w.r.t. a. Shape (n,)."""
    n = len(a)
    i = np.arange(n)

    """Subgradient of max_k conv(a,a)[k]."""
    c = _full_convolve(a, a)
    idx = np.argsort(c)[::-1][:K]

def _solve_direction_lp(a, M, K=50, time_limit=30.0):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for top-K peaks, g >= 0."""
    n = len(a)
    c = _full_convolve(a, a)
    top_k = np.argsort(c)[::-1][:K]

    rows = []
    cols = []
    vals = []
    for ii, k in enumerate(top_k):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            g_idx = k - j
            rows.append(ii)
            cols.append(g_idx)
            vals.append(a[j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(top_k), n))
    c_obj = -np.ones(n)

                  bounds=[(0, np.inf)] * n, method='highs',
                  options={'time_limit': time_limit, 'presolve': True})
