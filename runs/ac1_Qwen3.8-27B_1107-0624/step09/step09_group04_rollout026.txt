"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. L-BFGS warmup on softmax-smoothed objective
2. QP-based peak equalization using analytical gradient structure
3. Multi-n exploration and asymmetric perturbations
4. Final fine-tuning
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)
    def grad(self, z):
        return _smograd(z, self.beta, self.n)

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005) * beta_scale
    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625,
        0.03, 0.015, 0.0075, 0.004,
        0.002, 0.001, 0.0005, 0.0002, 0.0001
    ])
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        if beta < base_beta * 0.01:
            est_time_per_iter = 0.004 * max(1.0, n / 2048.0)
            max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 10))))
        else:
            est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
            max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 8))))
        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-17, 'gtol': 1e-14, 'maxcor': 80, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new

2. QP-based peak equalization exploiting the analytical gradient structure

5. Adaptive active set sizing and regularization

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    if s < 1e-300 or not np.isfinite(s):

        w[int(np.argmax(c))] = 1.0

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

        except np.linalg.LinAlgError:
            d_coeff = np.ones(K) / K
        
        # Compute direction: direction = S*1 - sum_i d_coeff[i] * g_i
        # where S = sum_i d_coeff[i] * means[i], and g_i is the unprojected gradient
        S = float(np.dot(d_coeff, means))
        direction = np.full(n, S)
        for i in range(K):
            p = int(top_k_idx[i])
            L = max(0, p - n + 1)
            R = min(p, n - 1)
            # g_i[j] = 2*a[p-j] for j in [L, R]
            # For j=L: p-j = p-L. For j=R: p-j = p-R.
            # So a[p-j] for j=L..R is a[p-L], a[p-L-1], ..., a[p-R]
            # = a[p-R+1 : p-L+1][::-1] or a[p-L : p-R-1 : -1]
            direction[L:R + 1] -= d_coeff[i] * 2.0 * a[p - R : p - L + 1][::-1]
        
        # Verify sum is approximately zero
        # (it should be by construction, but numerical errors may creep in)
        direction -= direction.mean()
        
        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-18:
            break
        direction = direction / dnorm
        
        # Find maximum step for non-negativity
        neg_mask = direction < -1e-15
        if neg_mask.any():
            max_alpha = float((a[neg_mask] / (-direction[neg_mask])).min())
        else:
            max_alpha = 1.0
        max_alpha = min(max_alpha * 0.9, 0.5)
        
        # Backtracking line search
        alpha = min(0.3 * max_alpha * lr_scale, 0.1)
        improved = False
        for _ in range(40):
            if alpha < 1e-12:
                break
            a_try = a + alpha * direction
            if a_try.min() < -1e-12:
                alpha *= 0.5
                continue
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.6
        
        if not improved:
            lr_scale *= 0.8
    
    return best_a, best_sc

def _subgradient_polish(a_init, deadline, min_time_remaining=15.0):
    """Subgradient method with adaptive step size for final polish."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    avg_norm = 1e-6
    for it in range(20000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
        
        # Active set: peaks within 0.1% of max
        threshold = c_max * 0.999
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        if len(active) > 80:
            active = np.argsort(c)[::-1][:80]
        
        # Compute average gradient
        grads = np.zeros(n)
        for p in active:
            p = int(p)
            L = max(0, p - n + 1)
            R = min(p, n - 1)
            grads[L:R+1] += 2.0 * a[p - R : p - L + 1][::-1]
        grads /= len(active)
        grads -= grads.mean()
        
        gnorm = np.linalg.norm(grads)
        if gnorm < 1e-18:
            break
        avg_norm = 0.95 * avg_norm + 0.05 * gnorm
        
        lb = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound for sum(a)=1
        gap = current_sc - lb
        if gap < 1e-10:
            break
        
        eta = min(0.5 * gap / (avg_norm * avg_norm + 1e-18), 0.1 / (np.sqrt(it + 1) * avg_norm))
        eta = min(eta, 0.5)
        
        a_try = a - eta * grads
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            break
        a_try = a_try / s_try
        
        # Check improvement every 10 iterations
        if (it + 1) % 10 == 0:
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc:
                a = a_try
    
    return best_a, best_sc

def _asymmetric_perturb(a, rng, strength=0.05):
    n = len(a)
    t = np.arange(n) / n
    ramp = 0.3 * (t - 0.5)
    noise = 0.7 * rng.standard_normal(n)
    perturbation = ramp + noise
    perturbation = perturbation - perturbation.mean()
    z = np.log(np.maximum(a, 1e-30))
    z = z + strength * perturbation
    z = z - z.mean()
    return _softmax(z)

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 7.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.3) ** 2) + 0.8 * np.exp(-15.0 * (t - 0.7) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase optimizer combining L-BFGS, QP peak equalization,
    multi-n exploration, and subgradient polish."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "sine_mod")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, sine_mod n={n_parent}, score={best_score:.10f}")
    
    # === Phase 1: L-BFGS warmup (300s) ===
    t_p1_start = time.monotonic()
    print(f"\n[P1] L-BFGS warmup (t={t_p1_start - t_start:.1f}s)")
    p1_deadline = min(deadline - 1100.0, t_p1_start + 300.0)
    a_p1 = _run_lbfgs(parent, p1_deadline, min_time_remaining=5.0, beta_scale=1.0)
    sc_p1 = _score(a_p1)
    if sc_p1 < best_score:
        best_seq = a_p1.copy()
        best_score = sc_p1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_warmup]")
    else:
        print(f"  No improvement: {sc_p1:.10f} vs {best_score:.10f}")
    
    # === Phase 2: QP peak equalization (600s) ===
    t_p2_start = time.monotonic()
    print(f"\n[P2] QP peak equalization (t={t_p2_start - t_start:.1f}s)")
    p2_deadline = min(deadline - 700.0, t_p2_start + 600.0)
    a_qp, sc_qp = _qp_peak_equalize(best_seq, p2_deadline, min_time_remaining=10.0)
    if sc_qp < best_score:
        best_seq = a_qp.copy()
        best_score = sc_qp
        print(f"  NEW BEST: {best_score:.10f} [qp_equalize]")
    else:
        print(f"  No improvement: {sc_qp:.10f} vs {best_score:.10f}")
    
    # === Phase 3: Fine L-BFGS (300s) ===
    t_p3_start = time.monotonic()
    print(f"\n[P3] Fine L-BFGS (t={t_p3_start - t_start:.1f}s)")
    p3_deadline = min(deadline - 400.0, t_p3_start + 300.0)
    a_fine = _run_lbfgs(best_seq, p3_deadline, min_time_remaining=5.0, beta_scale=0.3)
    sc_fine = _score(a_fine)
    if sc_fine < best_score:
        best_seq = a_fine.copy()
        best_score = sc_fine
        print(f"  NEW BEST: {best_score:.10f} [fine_lbfgs]")
    else:
        print(f"  No improvement: {sc_fine:.10f} vs {best_score:.10f}")
    
    # === Phase 4: Multi-n exploration (300s) ===
    t_p4_start = time.monotonic()
    print(f"\n[P4] Multi-n exploration (t={t_p4_start - t_start:.1f}s)")
    n_candidates = sorted(set([len(best_seq), 16384, 12288, 24576, 8192]), reverse=True)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 500:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")
        p4_sub_deadline = min(deadline - 450.0, time.monotonic() + 60.0)
        a_opt = _run_lbfgs(a_init, p4_sub_deadline, min_time_remaining=3.0, beta_scale=0.5)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_lbfgs]")
        # Quick QP on resampled
        if time.monotonic() < deadline - 420:
            p4_qp_deadline = min(deadline - 400.0, time.monotonic() + 30.0)
            a_qp2, sc_qp2 = _qp_peak_equalize(best_seq if sc_opt < sc_init else a_opt,
                                              p4_qp_deadline, min_time_remaining=2.0)
            if sc_qp2 < best_score:
                best_seq = a_qp2.copy()
                best_score = sc_qp2
                print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [multi_n_qp]")
    
    # === Phase 5: Asymmetric perturbation + re-optimize (200s) ===
    t_p5_start = time.monotonic()
    print(f"\n[P5] Asymmetric exploration (t={t_p5_start - t_start:.1f}s)")
    for trial in range(3):
        if time.monotonic() > deadline - 250:
            break
        strength = 0.04 * (trial + 1)
        a_asym = _asymmetric_perturb(best_seq, rng, strength=strength)
        p5_sub = min(deadline - 230.0, time.monotonic() + 40.0)
        a_opt = _run_lbfgs(a_asym, p5_sub, min_time_remaining=2.0, beta_scale=0.3)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (asym trial={trial}) [asym]")
        else:
            print(f"  Asym trial {trial}: {sc_opt:.10f} (no improvement)")
    
    # === Phase 6: Final QP + subgradient (200s) ===
    t_p6_start = time.monotonic()
    print(f"\n[P6] Final polish (t={t_p6_start - t_start:.1f}s)")
    p6_qp = min(deadline - 60.0, t_p6_start + 120.0)
    a_final_qp, sc_final = _qp_peak_equalize(best_seq, p6_qp, min_time_remaining=5.0)
    if sc_final < best_score:
        best_seq = a_final_qp.copy()
        best_score = sc_final
        print(f"  NEW BEST: {best_score:.10f} [final_qp]")
    
    if time.monotonic() < deadline - 30:
        p6_lbg = min(deadline - 25.0, time.monotonic() + 30.0)
        a_final_lb = _run_lbfgs(best_seq, p6_lbg, min_time_remaining=3.0, beta_scale=0.1)
        sc_final_lb = _score(a_final_lb)
        if sc_final_lb < best_score:
            best_seq = a_final_lb.copy()
            best_score = sc_final_lb
            print(f"  NEW BEST: {best_score:.10f} [final_lbfgs]")
    
    if time.monotonic() < deadline - 10:
        p6_sg = min(deadline - 5.0, time.monotonic() + 20.0)
        a_sg, sc_sg = _subgradient_polish(best_seq, p6_sg, min_time_remaining=3.0)
        if sc_sg < best_score:
            best_seq = a_sg.copy()
            best_score = sc_sg
            print(f"  NEW BEST: {best_score:.10f} [subgrad]")
    
    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
