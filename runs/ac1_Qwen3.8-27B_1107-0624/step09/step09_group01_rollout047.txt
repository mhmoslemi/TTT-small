"""

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _hankel_diagonal(W, n):
    """Compute the diagonal of the Hankel matrix H = sum W[k] * H_k."""
    e0 = np.zeros(n)
    e0[0] = 1.0
    return _hankel_matvec(e0, W, n)


def _pcg_solve_hankel(W, n, rhs, max_iter=200, tol=1e-14):
    """Preconditioned CG for solving H x = b where H is Hankel."""
    x = np.zeros(n)
    r = rhs.copy()
    # Jacobi preconditioner
    diag = _hankel_diagonal(W, n)
    diag = np.maximum(np.abs(diag), 1e-30)
    z = r / diag
    p = z.copy()
    rz = float(r @ z)
    rz0 = rz

    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rz / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rz_new = float(r @ (r / diag))
        if not np.isfinite(rz_new):
            break
        if abs(rz_new) < tol * tol * abs(rz0):
            break
        beta = rz_new / rz if abs(rz) > 1e-30 else 0.0
        z = r / diag
        p = z + beta * p
        rz = rz_new

    return x


def _compute_gradient(a, W, n):
    """Compute gradient of sum_k W[k] * c_k with respect to a."""
    a_rev = a[::-1]
    wc = _full_convolve(W, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(W, _full_convolve(a, a)))
    return 2.0 * a * (h - W_val)

def _lbfgs_smooth(a_init, deadline, min_remaining=30.0):
    """L-BFGS on smooth approximation with temperature annealing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    beta = max(gap * 0.5, max_c * 0.003)

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    current_a = a_init.copy()
    current_sc = _score(current_a)

    def _make_obj(z_arr, beta_val, n_val):
        a_arr = _softmax(z_arr)
        c_arr = _full_convolve(a_arr, a_arr)
        c_max_val = float(c_arr.max())
        shifted = (c_arr - c_max_val) / beta_val
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n_val * c_max_val
        return 2.0 * n_val * (c_max_val + beta_val * np.log(s))

    def _make_grad(z_arr, beta_val, n_val):
        a_arr = _softmax(z_arr)
        c_arr = _full_convolve(a_arr, a_arr)
        c_max_val = float(c_arr.max())
        shifted = (c_arr - c_max_val) / beta_val
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c_arr)
            w[int(np.argmax(c_arr))] = 1.0
        else:
            w = e / s
        a_rev = a_arr[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n_val - 1:2 * n_val - 1]
        W_val = float(np.dot(w, c_arr))
        # grad of sum W_k c_k w.r.t. a
        grad_a = 2.0 * a_arr * (h - W_val)
        # Chain rule: a = softmax(z), so grad_z_i = sum_j grad_a_j * (a_i*delta_ij - a_i*a_j)
        # = a_i * (grad_a_i - sum_j a_j * grad_a_j)
        inner = float(np.dot(a_arr, grad_a))
        return 4.0 * n_val * a_arr * (grad_a - inner)

    while beta > max_c * 1e-12:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 8192.0)
        est_iter_time = 0.003 * scale
        max_iter = min(8000, max(30, int(remaining / (est_iter_time * 40))))

        def obj_fn(z_val):
            return _make_obj(z_val, beta, n)

        def grad_fn(z_val):
            return _make_grad(z_val, beta, n)

                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 80, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        # Update beta based on flatness
        c_new = _full_convolve(current_a, current_a)

        else:

def _make_profile(n, kind="tapered"):

    elif kind == "cos":

    elif kind == "bell":

    elif kind == "asym":

        a = np.sin(np.pi * t ** 0.7) ** 0.8 + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

        t_kkt = min(t_p4_end - 10, time.monotonic() + 50)
        k_sch2 = [1, 2, 3, 5, 10, 20, 50, 100, 200, 500]
        a_c1, sc_c1 = _newton_kkt(best_seq, t_kkt, max_rounds=80, k_schedule=k_sch2)
        update_best(a_c1, sc_c1, f"cyc_kkt_{cyc}")

        # L-BFGS polish
        t_lb = min(t_p4_end, time.monotonic() + 40)
        a_c2 = _lbfgs_smooth(best_seq, t_lb, min_remaining=60.0)
        update_best(a_c2, _score(a_c2), f"cyc_lb_{cyc}")

    # Phase 5: Multi-n exploration (100s)
    print(f"\n[Phase 5] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    t_p5_end = min(deadline - 50, t_start + 1400)
    n_options = [n0 // 2, n0 * 3 // 4, n0, n0 * 5 // 4, n0 * 2]
    n_options = sorted(set(n for n in n_options if 8192 <= n <= 65536))

    for n_try in n_options:
        if time.monotonic() > t_p5_end:
            break
        a_res = _resample(best_seq, n_try)
        sc_res = _score(a_res)
        if sc_res < best_score:
            update_best(a_res, sc_res, f"resample_n{n_try}")
        # Quick KKT on resampled
        if time.monotonic() < t_p5_end - 10 and n_try <= 40000:
            t_quick = min(t_p5_end, time.monotonic() + 15)
            a_q, sc_q = _newton_kkt(a_res, t_quick, max_rounds=30,
                                     k_schedule=[1, 3, 5, 10, 20, 50])
            update_best(a_q, sc_q, f"quick_n{n_try}")

    # Phase 6: Final ultra-fine polish (remaining time)
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        k_final = [1, 1, 2, 2, 3, 3, 5, 5, 8, 13, 21, 34, 55]
        a_fin, sc_fin = _newton_kkt(best_seq, deadline - 2, max_rounds=100, k_schedule=k_final)
        update_best(a_fin, sc_fin, "final_kkt")

        # One more L-BFGS pass with very small beta
        if time.monotonic() < deadline - 3:
            a_fin2 = _lbfgs_smooth(best_seq, deadline - 1, min_remaining=2.0)
            update_best(a_fin2, _score(a_fin2), "final_lbfgs")

    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
