The diagonal of H is: `H[j,j] = sum_k lambda_k * (delta_{j,j+k} + delta_{j,j-k})`

For k > 0: `delta_{j,j+k} = 0` and `delta_{j,j-k} = 0` (since j ≠ j±k for k > 0)
For k = 0: `delta_{j,j} + delta_{j,j} = 2`

So the diagonal of H is `2 * lambda_0` if k=0 is in the active set, and 0 otherwise!

This means for k > 0 active constraints, the diagonal is 0, which makes diagonal preconditioning useless.

"""
Hybrid optimizer using GS pre-initialization, preconditioned KKT-Newton,
and multi-scale exploration to minimize 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    elif kind == "broad":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3 + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _gs_iterations(a, alpha=0.3, n_iter=200, deadline=None):
    """Gerchberg-Saxton iterations to flatten autocorrelation spectrum."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    a = a.copy()
    for i in range(n_iter):
        if deadline and time.monotonic() > deadline:
            break
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        P = np.abs(A) ** 2
        # Flatten the spectrum
        P_flat = np.power(P + 1e-30, alpha)
        # Renormalize to preserve total power
        P_flat = P_flat * (P.sum() / P_flat.sum())
        # New spectrum with same phase
        A_new = np.sqrt(P_flat) * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, N)[:n]
        a = np.maximum(a_new, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
    return a


def _hankel_matvec_prencond(p, W, n, cond_eigs):
    """Preconditioned matvec for the KKT system."""
    # First apply H * p (the Hankel-like operator)
    p_rev = p[::-1]
    out_len = 2 * n - 1
    N = _next_pow2(out_len)
    F_p = np.fft.rfft(p_rev, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_p * F_W, N)[:out_len]
    Hp = 2.0 * conv[n - 1:2 * n - 1]
    
    # Apply preconditioner: divide by eigenvalues in frequency domain
    if cond_eigs is not None:
        F_Hp = np.fft.rfft(Hp, N)
        # cond_eigs should be the eigenvalues of H in frequency domain
        # Clip to avoid division by zero
        safe_eigs = np.maximum(cond_eigs, 1e-10)
        F_Hp_precond = F_Hp / safe_eigs
        Hp = np.fft.irfft(F_Hp_precond, N)[:n]
    
    return Hp


def _compute_h_eigs(W, n):
    """Compute eigenvalues of the circulant approximation of H."""
    out_len = 2 * n - 1
    N = _next_pow2(out_len)
    # W has length 2n-1, the eigenvalues of sum_k W[k] * (S_k + S_k^T)
    # in the circulant approximation are 2 * sum_k W[k] * cos(2*pi*m*(k-n+1)/N)
    # Actually, let me think more carefully...
    # H = sum_k W[k] * (S_k + S_k^T) where S_k shifts by k
    # In circulant approximation, eig_m(H) = sum_k W[k] * (exp(i*k*omega_m) + exp(-i*k*omega_m))
    # = 2 * sum_k W[k] * cos(k * omega_m)
    # where omega_m = 2*pi*m/N
    
    # But we need to be careful about the indexing of W.
    # W[k] corresponds to the constraint at lag (k - (n-1)) in the centered sense.
    # Actually, for the KKT system, W is a vector of length 2n-1 where W[k] is the
    # weight for the constraint c[k] <= max.
    # The Hessian of c[k] has the structure S_{k-(n-1)} + S_{k-(n-1)}^T
    # where the shift is centered.
    
    # Let me use a simpler approach: just compute the DFT of W (appropriately centered)
    # and the eigenvalues are 2 * |DFT(W_centered)|
    
    # Center W: shift so that the center lag is at position 0
    center = n - 1
    W_centered = np.zeros(N)
    for k in range(len(W)):
        shift = k - center
        if 0 <= shift < N:
            W_centered[shift] += W[k]
        elif shift < 0:
            W_centered[N + shift] += W[k]
    
    eigs = 2.0 * np.abs(np.fft.rfft(W_centered))
    return eigs


def _cg_solve(W, n, rhs, max_iter=100, tol=1e-10):
    """Solve H x = rhs using CG with frequency-domain preconditioner."""
    x = np.zeros(n)
    r = rhs.copy()
    # Preconditioner
    eigs = _compute_h_eigs(W, n)
    N = _next_pow2(2 * n - 1)
    
    def precond(v):
        Fv = np.fft.rfft(v, N)
        Fv_precond = Fv / np.maximum(eigs, 1e-8)
        return np.fft.irfft(Fv_precond, N)[:n]
    
    z = precond(r)
    p = z.copy()
    rz = float(r @ z)
    rz0 = rz
    
    for i in range(max_iter):
        # matvec: H * p
        p_rev = p[::-1]
        out_len = 2 * n - 1
        F_p = np.fft.rfft(p_rev, N)
        F_W = np.fft.rfft(W, N)
        conv = np.fft.irfft(F_p * F_W, N)[:out_len]
        Hp = 2.0 * conv[n - 1:2 * n - 1]
        
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rz / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        z_new = precond(r_new)
        rz_new = float(r_new @ z_new)
        if not np.isfinite(rz_new) or abs(rz_new) < 1e-30:
            break
        beta_gmres = rz_new / rz
        p_new = z_new + beta_gmres * p
        x = x_new
        r = r_new
        z = z_new
        p = p_new
        rz = rz_new
        if abs(rz) < tol * tol * max(rz0, 1e-30):
            break
    
    return x


def _compute_gradient(a, W, n):
    """Compute gradient of the surrogate objective."""
    a_rev = a[::-1]
    wc = _full_convolve(W, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(W, _full_convolve(a, a)))
    return 2.0 * a * (h - W_val)


def _kkt_newton(a_init, deadline, max_rounds=2000, k_schedule=None):

    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0
    
    if k_schedule is None:
        k_schedule = [1, 2, 3, 5, 10, 20, 50, 100, 200, 500, 1000, 2000]
    
    k_idx = 0
    increase_k = 0
    N = _next_pow2(2 * n - 1)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        K = min(k_schedule[min(k_idx, len(k_schedule) - 1)], 2 * n - 1)
        idx = np.argpartition(c, -K)[-K:]
        W = np.zeros(2 * n - 1)
        W[idx] = 1.0 / K
        
        # Gradient
        g = _compute_gradient(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break
        
        # Newton step via CG
        Hg = _cg_solve(W, n, g, max_iter=80)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=80)
        H1_denom = float(np.ones(n) @ H1)
        
        if abs(H1_denom) < 1e-25 or not np.isfinite(H1_denom):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / H1_denom
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)
        
        # Line search
        alpha = 1.0
        step_done = False
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break
        
        if not step_done:
            # Fallback: subgradient step
            for trial in [0.01, 0.005, 0.001, 0.0005, 0.0001]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break
        
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive K: increase K if stuck
        if step_done:
            increase_k += 1
            if increase_k >= 20 and k_idx < len(k_schedule) - 1:
                k_idx += 1
                increase_k = 0
        else:
            no_improve += 1
            if no_improve > 5:
                # Try increasing K to escape
                if k_idx < len(k_schedule) - 1:

def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

    z = z + sigma * rng.standard_normal(len(a))

"""
Hybrid optimizer: GS pre-initialization + preconditioned KKT-Newton with

Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    elif kind == "broad":

def _gs_iterations(a, alpha, n_iter, deadline):
    """Gerchberg-Saxton iterations to flatten autocorrelation."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    a = a.copy()
    for i in range(n_iter):
        if time.monotonic() > deadline:
            break
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        P = np.abs(A) ** 2
        P_flat = np.power(P + 1e-30, alpha)
        ps = P.sum()
        pfs = P_flat.sum()
        if pfs < 1e-30:
            break
        P_flat = P_flat * (ps / pfs)
        A_new = np.sqrt(P_flat) * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, N)[:n]
        a = np.maximum(a_new, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
    return a


def _hankel_matvec(p, F_W, N, n):
    """Matrix-vector product for the KKT Hessian system."""
    p_rev = p[::-1]
    F_p = np.fft.rfft(p_rev, N)
    conv = np.fft.irfft(F_p * F_W, N)[:2 * n - 1]
    return 2.0 * conv[n - 1:2 * n - 1]


def _compute_h_eigs(W, n, N):
    """Eigenvalues of circulant H approximation."""
    center = n - 1
    W_c = np.zeros(N)
    for k in range(len(W)):
        shift = k - center
        if 0 <= shift < N:
            W_c[shift] += W[k]
        elif shift < 0:
            W_c[N + shift] += W[k]
    eigs = 2.0 * np.abs(np.fft.rfft(W_c))
    return eigs


def _precond(v, eigs, N, n):
    """Apply frequency-domain preconditioner."""
    Fv = np.fft.rfft(v, N)
    safe = np.maximum(eigs, 1e-10)
    Fv_p = Fv / safe
    return np.fft.irfft(Fv_p, N)[:n]


def _cg_solve_p(W, n, rhs, max_iter, tol, N):
    """PCG solver for H x = rhs with frequency preconditioner."""
    F_W = np.fft.rfft(W, N)
    eigs = _compute_h_eigs(W, n, N)
    x = np.zeros(n)
    r = rhs.copy()
    z = _precond(r, eigs, N, n)
    p = z.copy()
    rz = float(r @ z)
    rz0 = abs(rz)
    if rz0 < 1e-30:
        return np.zeros(n)
    for i in range(max_iter):
        Hp = _hankel_matvec(p, F_W, N, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rz / pHp
        if not np.isfinite(alpha):
            break
        x = x + alpha * p
        r = r - alpha * Hp
        z = _precond(r, eigs, N, n)
        rz_new = float(r @ z)
        if not np.isfinite(rz_new) or abs(rz_new) < 1e-35:
            break
        beta_cg = rz_new / rz
        p = z + beta_cg * p
        rz = rz_new
        if abs(rz) < tol * tol * max(rz0, 1e-30):
            break
    return x


def _compute_gradient(a, W, n):
    """Gradient of the KKT surrogate."""
    a_rev = a[::-1]
    wc = _full_convolve(W, a_rev)
    h = wc[n - 1:2 * n - 1]
    c = _full_convolve(a, a)
    W_val = float(np.dot(W, c))
    return 2.0 * a * (h - W_val)


def _kkt_newton(a_init, deadline, max_rounds, k_schedule):
    """KKT-Newton with preconditioned CG and adaptive K."""
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0
    k_idx = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        K = min(k_schedule[min(k_idx, len(k_schedule) - 1)], 2 * n - 1)
        idx = np.argpartition(c, -K)[-K:]
        W = np.zeros(2 * n - 1)
        W[idx] = 1.0 / K
        F_W = np.fft.rfft(W, N)

def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

        print(f"  NEW BEST: {sc:.10f} (n={len(p)}) imp={imp_str} t={elapsed:.1f}s")


def _try_gs_start(b_state, a_init, t_start, deadline):
    """Try GS with different alphas and return best."""
    for alpha in [0.15, 0.25, 0.4, 0.6]:
        if time.monotonic() > deadline:
            break
        a_gs = _gs_iterations(a_init, alpha, 200, deadline)
        sc = _score(a_gs)
        _update_best(b_state, a_gs, sc, t_start)


def _try_profile_start(b_state, n, t_start, deadline):
    """Try different profile types as starting points."""
    for kind in ["tapered", "cos", "bell", "broad", "uniform"]:
        if time.monotonic() > deadline:
            break
        a = _make_profile(n, kind)
        sc = _score(a)
        _update_best(b_state, a, sc, t_start)


def propose_candidate():
    """
    Main search: GS pre-init -> KKT-Newton -> subgrad -> cycle -> multi-n -> polish.
    """
    t_start = time.monotonic()

    rng = np.random.default_rng(123)

        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())

                parent = parent_raw / s
                par_sc = _score(parent)
                print(f"Parent: n={len(parent)}, score={par_sc:.10f}")
                _update_best(b_state, parent, par_sc, t_start)
    except Exception:
        pass

    if parent is None:
        parent = _make_profile(16384, "tapered")
        print(f"No parent, using tapered n=16384")
        _update_best(b_state, parent, _score(parent), t_start)

    n0 = len(parent)

    # Phase 1: GS pre-initialization
    print(f"\n[Phase 1] GS pre-init (t={time.monotonic() - t_start:.1f}s)")
    gs_dl = t_start + 90.0
    _try_gs_start(b_state, parent, t_start, gs_dl)

    # Also try a few profiles at different sizes for diversity
    if time.monotonic() < t_start + 110:
        _try_profile_start(b_state, 16384, t_start, t_start + 130)

    # Phase 2: Main KKT-Newton
    print(f"\n[Phase 2] KKT-Newton main (t={time.monotonic() - t_start:.1f}s)")

    sg_dl = t_start + 950.0
    a_sg, sc_sg = _subgrad_max(b_state['best_seq'], sg_dl, 8000)
    _update_best(b_state, a_sg, sc_sg, t_start)

    # Phase 4: Cycled refinement
    print(f"\n[Phase 4] Cycle (t={time.monotonic() - t_start:.1f}s)")
    cyc_dl = t_start + 1350.0
    for cyc in range(15):
        if time.monotonic() > cyc_dl:
            break
        t_sub = min(cyc_dl, time.monotonic() + 25.0)
        a_c1, sc_c1 = _subgrad_max(b_state['best_seq'], t_sub, 800)
        _update_best(b_state, a_c1, sc_c1, t_start)

        t_new = min(cyc_dl, time.monotonic() + 35.0)
        k_sch2 = [1, 1, 2, 3, 5, 10, 30, 100]
        a_c2, sc_c2 = _kkt_newton(b_state['best_seq'], t_new, 150, k_sch2)
        _update_best(b_state, a_c2, sc_c2, t_start)

    # Phase 5: Multi-n exploration
    print(f"\n[Phase 5] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_options = [8192, 16384, 24576, 32768, 49152, 65536]
    for n_try in n_options:
        if time.monotonic() > deadline - 80:
            break
        a_res = _resample(b_state['best_seq'], n_try)
        sc_res = _score(a_res)
        _update_best(b_state, a_res, sc_res, t_start)

        if time.monotonic() < deadline - 120 and n_try <= 49152:

            k_sch3 = [1, 1, 2, 3, 5, 10, 20]

            _update_best(b_state, a_opt, sc_opt, t_start)

        sigma = 0.008 * (restart + 1)
        a_pert = _perturb(b_state['best_seq'], sigma, rng)
        t_end = min(deadline - 15, time.monotonic() + 15.0)
        a_r, sc_r = _kkt_newton(a_pert, t_end, 80, [1, 2, 3, 5, 10])
        _update_best(b_state, a_r, sc_r, t_start)

    print(f"\n[Phase 7] Final K=1 polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 3:

                                     [1, 1, 1, 1, 2, 2, 3])

    print(f"\n{'=' * 60}")
    if b_state['best_seq'] is not None:
        final = np.clip(b_state['best_seq'], 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    sc_fb = _score(fallback)
    print(f"FALLBACK: score={sc_fb:.10f}")
    return [float(x) for x in fallback]
