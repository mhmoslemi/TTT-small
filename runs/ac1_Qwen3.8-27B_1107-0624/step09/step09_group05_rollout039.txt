"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses POCS-based initialization, L-BFGS with adaptive smooth-max, 
active-set QP refinement, and subgradient polishing.
Explores multiple sequence lengths and initializations.
"""
import numpy as np
import time
from scipy.optimize import minimize

# Global timing
_t0 = 0.0

def _fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    N = max(2, N)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    """Compute the objective value."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    s = float(a.sum())
    if s < 1e-15:
        return a
    return a / s

def _softmax(z):
    z = z - z.max()
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _soft_max(c, beta):
    """Compute soft max of c."""
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)

def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

def _grad_soft(a, beta, n):
    """Gradient of soft_max(conv(a,a)) w.r.t. a."""
    c = _fft_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _fft_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]

def _logobj(z, beta, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    return 2.0 * n * _soft_max(c, beta)

def _loggrad(z, beta, n):
    a = _softmax(z)
    ga = _grad_soft(a, beta, n)
    W = float(np.dot(ga, a))
    return a * (ga - W)

def _pocs_init(n, n_iter=50, seed=42):
    """POCS-based initialization: alternate between flat |DFT| and non-negativity."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = _normalize(np.maximum(a, 1e-10))
    for _ in range(n_iter):
        F = np.fft.fft(a)
        mag = np.abs(F)
        mean_mag = mag.mean()
        # Set magnitude to constant, keep phase
        phase = np.angle(F)
        F_new = mean_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(F_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
        else:
            break
        a = a_new
    return a

def _parametric(n, kind, **kw):
    """Generate parametric sequence."""
    if kind == "power":
        p = kw.get('p', 1.0)
        q = kw.get('q', 0.0)
        j = np.arange(n, dtype=np.float64)
        a = (j + 1.0)**p * (n - j)**q
        a = np.maximum(a, 1e-15)
        return _normalize(a)
    elif kind == "sigmoid":
        center = kw.get('center', 0.7)
        width = kw.get('width', 0.2)
        j = np.arange(n, dtype=np.float64) / n
        a = 1.0 / (1.0 + np.exp(-(j - center) / width))
        a = np.maximum(a, 1e-15)
        return _normalize(a)
    elif kind == "tapered_power":
        p = kw.get('p', 0.5)
        j = np.arange(n, dtype=np.float64) / (n - 1)
        a = (j + 1e-6)**p * (1 - j + 1e-6)**p
        return _normalize(a)
    elif kind == "linear_taper":
        a = np.arange(1, n+1, dtype=np.float64)
        return _normalize(a)
    elif kind == "sqrt_taper":
        a = np.sqrt(np.arange(1, n+1, dtype=np.float64))
        return _normalize(a)
    else:
        return np.ones(n) / n

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, verbose=True, tag=""):
    """L-BFGS optimization with adaptive beta schedule."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    c0 = _fft_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max / (a.sum()**2)
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    # Beta schedule: start large, decrease geometrically
    c_sorted = np.sort(c0)[::-1]
    gap = c0_max - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c0_max * 0.005:
        gap = c0_max * 0.005
    
    multipliers = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
                   0.015625, 0.0078125, 0.00390625, 0.002,
                   0.001, 0.0005, 0.00025, 0.0001, 0.00005,
                   0.00002, 0.00001, 0.000005, 0.000002]
    betas = np.array(multipliers) * gap
    
    t_start = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        size_factor = max(1.0, n / 16384.0)
        est_time_per_iter = 0.001 * size_factor
        max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 30))))
        
        def fun(z_):
            return _logobj(z_, beta, n)
        
        def jac(z_):
            return _loggrad(z_, beta, n)
        
        try:
            result = minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-14,
                    'maxcor': 50,
                    'maxls': 40,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    elapsed = time.monotonic() - t0
                    print(f"    {tag} beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}, t={elapsed:.0f}s")
        except Exception as e:
            if verbose:
                print(f"    {tag} beta={beta:.2e}: error {e}")
    
    return best_a, best_sc

def _active_set_qp(a_init, deadline, min_time_remaining=30.0, verbose=False):
    """Active-set refinement using iterative QP-like steps.
    
    At each iteration:
    1. Identify active set S (conv values near max)
    2. Compute the quadratic form for the active set
    3. Use conjugate gradient to minimize the quadratic
    4. Project onto simplex
    """
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    no_improve = 0
    max_no_improve = 500
    tol_frac = 0.01
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            tol_frac = max(0.001, tol_frac * 0.7)
            if tol_frac <= 0.0012:
                break
            no_improve = 0
        
        # Active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros(len(c), dtype=bool)
            active[idx] = True
            n_active = 1
        
        # Build the gradient: g[j] = (2/n_active) * sum_{k in active} a[k-j]
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _fft_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]
        
        # Center the gradient (tangent to simplex)
        grad = grad - np.mean(grad)
        
        # CG-like iteration: minimize a^T * M * a where M[j,i] = 1 if j+i in active
        # Instead of forming M explicitly, use the gradient as a direction
        # Take a step in the negative gradient direction
        g_norm = float(np.sqrt(np.dot(grad, grad)))
        if g_norm < 1e-15:
            break
        
        # Line search
        best_alpha = 0.01
        best_a_try = None
        best_sc_try = current_sc
        for alpha in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_a_try = a_try
                best_alpha = alpha
        
        if best_a_try is None:
            # No improvement, tighten tolerance
            tol_frac = max(0.001, tol_frac * 0.8)
            continue
        
        a = best_a_try
        
        if no_improve == 0:
            tol_frac = max(0.001, tol_frac * 0.95)
    
    return best_a, best_sc

def _subgradient(a_init, deadline, min_time_remaining=30.0):
    """Subgradient method with adaptive step size."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    lb = 2.0 * n / (2 * n - 1)
    t = 0
    step_scale = 0.3
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
        
        # Active set with tight tolerance
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros(len(c), dtype=bool)
            active[idx] = True
            n_active = 1
        
        # Gradient
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _fft_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]
        
        # Center
        grad = grad - np.mean(grad)
        g_norm_sq = float(np.dot(grad, grad))
        if g_norm_sq < 1e-20:
            break
        
        # Step size
        gap = current_sc - lb
        if gap < 1e-10:
            break
        step = min(step_scale * gap / g_norm_sq, 0.05)
        
        a = a - step * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

def _resample(a, new_n):

"""
Multi-strategy optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses POCS-based initialization, L-BFGS with adaptive smooth-max,
active-set refinement with momentum, subgradient polishing, and multi-n exploration.
Key improvements: larger default n, POCS init for flat circular autocorrelation,

import numpy as np

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    N = max(2, N)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = float(a.sum())
    if s < 1e-15:
        return a
    return a / s


def _softmax(z):
    z = z - z.max()
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_val(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_w(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_soft(a, beta, n):
    c = _fft_convolve(a, a)
    w = _soft_max_w(c, beta)
    a_rev = a[::-1]
    wc = _fft_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _pocs_init(n, n_iter=80, seed=42):
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = _normalize(np.maximum(a, 1e-10))
    for _ in range(n_iter):
        F = np.fft.fft(a)
        mag = np.abs(F)
        mean_mag = float(mag.mean())
        if mean_mag < 1e-15:
            break
        phase = np.angle(F)
        F_new = mean_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(F_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
        else:
            break
        a = a_new
    return a


def _parametric(n, kind, p=1.0, q=0.0, center=0.7, width=0.2):
    j = np.arange(n, dtype=np.float64)
    if kind == "power":
        a = (j + 1.0) ** p * (n - j) ** q
    elif kind == "sigmoid":
        t = j / n
        a = 1.0 / (1.0 + np.exp(-(t - center) / width))
    elif kind == "sym_power":
        t = j / (n - 1)
        a = (t + 1e-8) ** p * (1.0 - t + 1e-8) ** q
    elif kind == "sqrt":
        a = np.sqrt(j + 1.0)
    elif kind == "log":
        a = np.log(j + 1.0)
    elif kind == "exp_ramp":
        a = np.exp(0.5 * j / n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-15)
    return _normalize(a)

def _resample(a, new_n):

        return a.copy()

    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    return _normalize(a_new)


def _coordinate_shave(a, n_shaves=20):
    """Targeted shaving: reduce elements that contribute most to the peak."""
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_sc = _score(a)

    for _ in range(n_shaves):
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        k_peak = int(np.argmax(c))

        # Find which a[j] contribute most to c[k_peak]
        # c[k_peak] = sum_j a[j]*a[k_peak-j]
        contrib = np.zeros(n)
        for j in range(n):
            k2 = k_peak - j
            if 0 <= k2 < n:
                contrib[j] = a[j] * a[k2]

        # Shave the top contributors
        top_idx = np.argsort(contrib)[::-1][:5]
        for j in top_idx:
            for factor in [0.95, 0.9, 0.8, 0.5]:
                a_try = a.copy()
                a_try[j] *= factor
                a_try = _normalize(a_try)
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-12:
                    a = a_try
                    best_sc = sc_try
                    best_a = a_try.copy()
                    break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Phases: POCS init -> L-BFGS -> Active-set momentum -> L-BFGS ->
    Subgradient -> Multi-n -> Final polish.
    """
    global _t0
    _t0 = time.monotonic()
    deadline = _t0 + max(1.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                parent_sc = _score(parent)
                print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
                best_seq = parent.copy()
                best_score = parent_sc
    except Exception:
        pass

    # Choose n
    if parent is not None and 16384 <= len(parent) <= 65536:
        n_main = len(parent)
    else:
        n_main = 32768

    print(f"Main n = {n_main}, budget = {budget_s}s")

    # === Phase 1: Initialization (100s) ===
    print(f"\n[P1] Init (t={time.monotonic()-_t0:.0f}s)")
    p1_end = _t0 + min(100, (budget_s - 5) * 0.07)
    candidates = []

    # POCS
    a_pocs = _pocs_init(n_main, n_iter=60, seed=seed)
    sc_pocs = _score(a_pocs)
    candidates.append((sc_pocs, a_pocs))
    print(f"  POCS: {sc_pocs:.10f}")

    # POCS with different seed
    a_pocs2 = _pocs_init(n_main, n_iter=60, seed=seed + 1)
    sc_pocs2 = _score(a_pocs2)
    candidates.append((sc_pocs2, a_pocs2))
    print(f"  POCS2: {sc_pocs2:.10f}")

    # Parametric families
    param_list = [
        ("power", 1.0, 0.0), ("power", 1.5, 0.0), ("power", 2.0, 0.0),
        ("power", 0.5, 0.0), ("power", 1.0, 0.5), ("power", 1.0, 1.0),
        ("sigmoid", 0, 0), ("sym_power", 1.0, 0.0), ("sqrt", 0, 0),
        ("exp_ramp", 0, 0),
    ]
    for kind, p, q in param_list:
        if time.monotonic() > p1_end - 5:
            break
        if kind == "sigmoid":
            a_p = _parametric(n_main, kind, center=0.7, width=0.2)
        else:
            a_p = _parametric(n_main, kind, p=p, q=q)
        sc_p = _score(a_p)
        candidates.append((sc_p, a_p))

    candidates.sort(key=lambda x: x[0])
    print(f"  Best init: {candidates[0][0]:.10f}")

    for sc, a in candidates[:5]:
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()

    # Also try coordinate shaving on best candidate
    if time.monotonic() < p1_end - 10:
        a_sh, sc_sh = _coordinate_shave(best_seq, n_shaves=10)
        if sc_sh < best_score:
            best_score = sc_sh
            best_seq = a_sh.copy()
            print(f"  After shaving: {best_score:.10f}")

    # === Phase 2: Main L-BFGS (650s) ===
    print(f"\n[P2] L-BFGS main (n={n_main}, t={time.monotonic()-_t0:.0f}s)")
    p2_end = _t0 + min(650, (budget_s - 5) * 0.42)
    a2, sc2 = _run_lbfgs(best_seq, p2_end, min_time_remaining=200, verbose=True, tag="P2")
    if sc2 < best_score:
        best_score = sc2
        best_seq = a2.copy()
        print(f"  P2: {best_score:.10f}")

    # === Phase 3: Active-set momentum (300s) ===
    print(f"\n[P3] Active-set momentum (t={time.monotonic()-_t0:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p3_end = min(deadline - 250, time.monotonic() + 300)
        a3, sc3 = _active_set_momentum(best_seq, p3_end, min_time_remaining=50, verbose=True)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3: {best_score:.10f}")

    # === Phase 4: L-BFGS round 2 (300s) ===
    print(f"\n[P4] L-BFGS r2 (t={time.monotonic()-_t0:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p4_end = min(deadline - 200, time.monotonic() + 300)
        a4, sc4 = _run_lbfgs(best_seq, p4_end, min_time_remaining=100, verbose=True, tag="P4")
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4: {best_score:.10f}")

    # === Phase 5: Subgradient + L-BFGS (200s) ===
    print(f"\n[P5] Subgradient (t={time.monotonic()-_t0:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p5_end = min(deadline - 100, time.monotonic() + 150)
        a5, sc5 = _subgradient(best_seq, p5_end, min_time_remaining=50)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  P5: {best_score:.10f}")

        if time.monotonic() + 20 < deadline:
            p5b_end = min(deadline - 50, time.monotonic() + 80)
            a5b, sc5b = _run_lbfgs(best_seq, p5b_end, min_time_remaining=30, verbose=False, tag="P5b")
            if sc5b < best_score:
                best_score = sc5b
                best_seq = a5b.copy()
                print(f"  P5b: {best_score:.10f}")

    # === Phase 6: Multi-n (180s) ===
    print(f"\n[P6] Multi-n (t={time.monotonic()-_t0:.0f}s)")
    n_options = [24576, 40960, 49152, 65536, 20480]
    n_options = [x for x in n_options if x != n_main]
    n_options.sort(key=lambda x: abs(x - n_main))

    for n_try in n_options[:3]:
        if time.monotonic() + 50 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if time.monotonic() + 35 > deadline:
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
            continue
        p6_end = min(deadline - 30, time.monotonic() + 50)
        a6, sc6 = _run_lbfgs(a_r, p6_end, min_time_remaining=20, verbose=False, tag=f"n{n_try}")
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  n={n_try}: {sc6:.10f}")

    # === Phase 7: Final polish (100s) ===
    print(f"\n[P7] Final (t={time.monotonic()-_t0:.0f}s)")
    if time.monotonic() + 20 < deadline:
        p7_end = min(deadline - 10, time.monotonic() + 60)
        a7, sc7 = _active_set_momentum(best_seq, p7_end, min_time_remaining=10, verbose=False)
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()

        if time.monotonic() + 15 < deadline:
            p7b_end = min(deadline - 3, time.monotonic() + 35)
            a7b, sc7b = _run_lbfgs(best_seq, p7b_end, min_time_remaining=3, verbose=False, tag="fin")
            if sc7b < best_score:
                best_score = sc7b
                best_seq = a7b.copy()

    # Coordinate shave as final touch
    if time.monotonic() + 10 < deadline:
        a_sh, sc_sh = _coordinate_shave(best_seq, n_shaves=15)
        if sc_sh < best_score:
            best_score = sc_sh
            best_seq = a_sh.copy()
            print(f"  After final shave: {best_score:.10f}")

    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) / 256
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-_t0:.1f}s")
    return [float(x) for x in final]
