def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_soft_structured(a, beta, n):
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft_structured(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)


class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _ideal_sequence(n):
    """Generate the 'ideal' sequence based on central binomial coefficients.
    b[k] = binom(2k,k)/4^k, which satisfies sum_{i+j=k} b[i]*b[j] = 1 for infinite sequence."""
    a = np.zeros(n)
    b = 1.0
    a[0] = 1.0
    for k in range(1, n):
        b *= (2.0 * 2 * k - 1) / (4.0 * k)
        a[k] = b
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def _alternating_projections(n, max_iter=300):
    """Alternating projections to find a non-negative sequence with flat spectrum."""
    N = _next_pow2(2 * n)
    a = np.ones(n) / n
    # Blend with ideal sequence for better start
    a_ideal = _ideal_sequence(n)
    a = 0.5 * a + 0.5 * a_ideal
    a = a / a.sum()

    prev_score = _score(a)
    for it in range(max_iter):
        # Time domain projection: non-negativity + normalization
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
            break
        a = a / s

        # Frequency domain projection: flatten magnitude
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-20)
        # Target: constant magnitude = geometric mean
        target_mag = np.sqrt(n / N) * 2.0  # heuristic scaling
        A_new = target_mag * phase
        a = np.fft.irfft(A_new, N)[:n]

        # Periodic score check
        if it % 50 == 49:
            sc = _score(a)
            if sc < prev_score - 0.01:
                prev_score = sc

    return a


def _random_init(n, rng):
    """Generate a random non-negative sequence with some structure."""
    kind = rng.integers(0, 5)
    if kind == 0:
        a = np.ones(n)
        a += 0.3 * rng.standard_normal(n)
    elif kind == 1:
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5 + 0.1 * rng.standard_normal(n)
    elif kind == 2:
        a = np.exp(-np.linspace(0, 3, n)) + 0.05 * rng.standard_normal(n)
    elif kind == 3:
        freq = rng.integers(3, 20)
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * t / n) + 0.1 * rng.standard_normal(n)
    else:
        a = rng.standard_normal(n) ** 2 + 0.01
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "one_sided":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t) + 0.005
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.6 * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, verbose=False, t_start=0.0):
    """L-BFGS with adaptive beta schedule - enhanced version."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = float(2.0 * n * c0_max / (a.sum() ** 2))

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    # Compute initial gap for beta scheduling
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(min(200, len(c_sorted) - 1), len(c_sorted) - 1)]
    if gap < c0_max * 0.005:
        gap = c0_max * 0.005
    if gap < 1e-15:
        gap = c0_max * 0.01

    # More fine-grained beta levels, biased toward small values
    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625,
        0.0078125, 0.00390625, 0.002, 0.001, 0.0005,
        0.00025, 0.000125, 0.0000625, 0.00003125, 0.000015625,
        0.0000078, 0.0000039, 0.000002, 0.000001, 0.0000005
    ])
    betas = gap * multipliers

    t_level_start = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # Adaptive iteration count
        size_factor = max(1.0, n / 8192.0)
        if beta > gap * 0.01:
            # Larger beta: fewer iterations needed
            max_iter = min(2000, max(50, int(remaining / (0.005 * size_factor * 50))))
        elif beta > gap * 0.001:
            max_iter = min(4000, max(100, int(remaining / (0.005 * size_factor * 30))))
        else:
            # Small beta: more iterations, but cap to avoid getting stuck
            max_iter = min(3000, max(200, int(remaining / (0.005 * size_factor * 20))))

        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-12:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    elapsed = time.monotonic() - t_start
                    print(f"    beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}, t={elapsed:.0f}s")
            else:
                if verbose and i < 5:
                    print(f"    beta={beta:.2e}: no improvement (best={best_sc:.10f})")
        except Exception as e:
            if verbose:
                print(f"    beta={beta:.2e}: error {e}")

    return best_a, best_sc


def _active_set_refine(a_init, deadline, min_time_remaining=20.0, verbose=False, t_start=0.0):
    """KKT-inspired active-set refinement with adaptive tolerance."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    no_improve = 0
    max_no_improve = 200
    tol_frac = 0.01
    min_tol = 0.00005
    eta_base = 0.03

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-12:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.7)
            if tol_frac <= min_tol * 1.2:
                break
            no_improve = 0

        # Identify active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Compute average gradient from active set
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]

        # Identify support
        threshold = 1e-6 * a.max()
        support = a > threshold
        n_support = int(support.sum())
        if n_support < 2:
            break

        # KKT: grad should be constant on support
        lam = float(np.mean(grad_avg[support]))
        residual = grad_avg[support] - lam

        # Try multiple step sizes
        found = False
        for eta in [eta_base, eta_base * 0.5, eta_base * 0.25, eta_base * 0.1]:
            a_new = a.copy()
            a_new[support] -= eta * residual
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-13:
                a = a_new
                found = True
                break

        if not found:
            # Try additive perturbation
            a_new = a + 0.001 * a * np.random.default_rng(int(time.time()*1000)).standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                sc_new = _score(a_new)
                if sc_new < current_sc:
                    a = a_new
                    found = True

        if not found:
            eta_base *= 0.9

        # Gradually tighten
        if no_improve == 0 and best_sc < _score(a_init) - 0.001:
            tol_frac = max(min_tol, tol_frac * 0.95)

    return best_a, best_sc


def _projected_gradient(a_init, deadline, min_time_remaining=20.0):
    """Projected gradient method on the probability simplex."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lb = 2.0 * n / (2.0 * n - 1)
    step_scale = 1.0
    t = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-12:
            best_sc = current_sc
            best_a = a.copy()

        # Active set for gradient
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Gradient
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        # Polyak step size
        gap = current_sc - lb
        if gap < 1e-10:
            break
        g_centered = grad - np.mean(grad)
        g_norm_sq = float(np.dot(g_centered, g_centered))
        if g_norm_sq < 1e-20:
            break
        step = step_scale * gap / (2.0 * g_norm_sq)
        step = min(step, 0.05)

        a_new = a - step * g_centered
        # Project onto simplex
        a_new = _proj_simplex(a_new)
        a = a_new
        t += 1

        if t % 50 == 0:
            step_scale *= 0.97

    return best_a, best_sc


def _proj_simplex(v, s=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0]
    if len(rho_candidates) == 0:
        return np.full(n, s / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _perturb_and_optimize(a, n, deadline, min_time_remaining=30.0, rng=None, t_start=0.0):
    """Add random perturbation and re-optimize."""
    if rng is None:
        rng = np.random.default_rng(12345)
    a_pert = a * (1.0 + 0.05 * rng.standard_normal(n))
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-10:
        return a, _score(a)
    a_pert = a_pert / s
    return _run_lbfgs(a_pert, deadline, min_time_remaining=min_time_remaining, verbose=False, t_start=t_start)


t_start_global = 0.0


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    P0: Load parent
    P1: Generate diverse initializations
    P2: L-BFGS on best initializations
    P3: Active-set refinement
    P4: Multi-n exploration
    P5: Perturbation-based restarts
    P6: Final polish
    """
    global t_start_global
    t_start_global = time.monotonic()
    deadline = t_start_global + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # === Phase 0: Load parent ===
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Phase 0: Parent loaded, n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(16384, "tapered")
        best_score = _score(best_seq)
        print(f"Phase 0: No parent. Tapered n=16384: score={best_score:.10f}")

    n_main = len(best_seq)

    # === Phase 1: Generate diverse initializations ===
    print(f"\n[Phase 1] Generating initializations (t={time.monotonic()-t_start_global:.0f}s)")
    candidates = []

    # Candidate 1: Ideal sequence
    a_ideal = _ideal_sequence(n_main)
    sc_ideal = _score(a_ideal)
    candidates.append((a_ideal, sc_ideal, "ideal"))
    print(f"  Ideal: score={sc_ideal:.10f}")

    # Candidate 2: Alternating projections
    a_altproj = _alternating_projections(n_main, max_iter=200)
    sc_altproj = _score(a_altproj)
    candidates.append((a_altproj, sc_altproj, "altproj"))
    print(f"  AltProj: score={sc_altproj:.10f}")

    # Candidate 3: Parent (if available)
    if parent is not None:
        candidates.append((parent.copy(), parent_sc, "parent"))

    # Candidate 4: Tapered
    a_taper = _make_profile(n_main, "tapered")
    sc_taper = _score(a_taper)
    candidates.append((a_taper, sc_taper, "tapered"))
    print(f"  Tapered: score={sc_taper:.10f}")

    # Candidate 5: Gaussian
    a_gauss = _make_profile(n_main, "gaussian")
    sc_gauss = _score(a_gauss)
    candidates.append((a_gauss, sc_gauss, "gaussian"))
    print(f"  Gaussian: score={sc_gauss:.10f}")

    # Candidate 6-7: Random
    for ri in range(2):
        a_rand = _random_init(n_main, rng)
        sc_rand = _score(a_rand)
        candidates.append((a_rand, sc_rand, f"random_{ri}"))

    # Sort by score
    candidates.sort(key=lambda x: x[1])
    print(f"  Best candidate: {candidates[0][2]} with score={candidates[0][1]:.10f}")

    # Update best if any candidate is better
    for a_c, sc_c, name in candidates:
        if sc_c < best_score:
            best_score = sc_c
            best_seq = a_c.copy()

    # === Phase 2: L-BFGS on top candidates ===
    print(f"\n[Phase 2] L-BFGS optimization (t={time.monotonic()-t_start_global:.0f}s)")
    # Spend most time on the best candidate
    p2_budget = min(600, (budget_s - 10) * 0.35)
    p2_end = t_start_global + p2_budget

    # Optimize top 3 candidates
    top_candidates = candidates[:3]
    remaining_for_top = p2_end - t_start_global
    time_per_candidate = remaining_for_top / len(top_candidates)

    for ci, (a_c, sc_c, name) in enumerate(top_candidates):
        if time.monotonic() + 30 > p2_end:
            break
        cand_end = min(p2_end, time.monotonic() + time_per_candidate)
        a_opt, sc_opt = _run_lbfgs(a_c, cand_end, min_time_remaining=15,
                                    verbose=(ci == 0), t_start=t_start_global)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"  {name}: improved to {best_score:.10f}")

    # === Phase 3: Active-set refinement ===
    print(f"\n[Phase 3] Active-set refinement (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p3_end = min(deadline - 400, time.monotonic() + 300)
        a3, sc3 = _active_set_refine(best_seq, p3_end, min_time_remaining=30,
                                      verbose=True, t_start=t_start_global)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  Active-set: improved to {best_score:.10f}")

    # === Phase 4: L-BFGS round 2 ===
    print(f"\n[Phase 4] L-BFGS round 2 (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p4_end = min(deadline - 200, time.monotonic() + 250)
        a4, sc4 = _run_lbfgs(best_seq, p4_end, min_time_remaining=20,
                              verbose=True, t_start=t_start_global)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  L-BFGS2: improved to {best_score:.10f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[Phase 5] Multi-n exploration (t={time.monotonic()-t_start_global:.0f}s)")
    n_options = [4096, 8192, 12288, 20480, 32768]
    n_options = [x for x in n_options if x != n_main]

    for n_try in n_options:
        if time.monotonic() + 60 > deadline:
            break
        # Check if we have time
        if time.monotonic() + 80 > deadline:
            break

        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        print(f"  n={n_try}: resampled score={sc_r:.10f}")

        if sc_r < best_score:
            best_score = sc_r
            best_seq = a_r.copy()

        if time.monotonic() + 50 > deadline:
            continue

        p5_end = min(deadline - 100, time.monotonic() + 80)
        a5, sc5 = _run_lbfgs(a_r, p5_end, min_time_remaining=15,
                              verbose=False, t_start=t_start_global)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  n={n_try}: optimized to {best_score:.10f}")

    # === Phase 6: Perturbation restarts ===
    print(f"\n[Phase 6] Perturbation restarts (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 40 < deadline:
        n_cur = len(best_seq)
        n_perturb = min(3, max(1, int((deadline - time.monotonic()) / 120)))
        for pi in range(n_perturb):
            if time.monotonic() + 40 > deadline:
                break
            p6_end = min(deadline - 30, time.monotonic() + 100)
            a_p, sc_p = _perturb_and_optimize(best_seq, n_cur, p6_end,
                                              min_time_remaining=15, rng=rng,
                                              t_start=t_start_global)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_p.copy()
                print(f"  Perturb {pi}: improved to {best_score:.10f}")

    # === Phase 7: Final polish ===
    print(f"\n[Phase 7] Final polish (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 10 < deadline:
        p7_end = min(deadline - 5, time.monotonic() + 80)
        a7, sc7 = _run_lbfgs(best_seq, p7_end, min_time_remaining=5,
                              verbose=False, t_start=t_start_global)
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()

        if time.monotonic() + 5 < deadline:
            p7b_end = min(deadline - 2, time.monotonic() + 30)
            a7b, sc7b = _active_set_refine(best_seq, p7b_end, min_time_remaining=2,
                                            verbose=False, t_start=t_start_global)
            if sc7b < best_score:
                best_score = sc7b
                best_seq = a7b.copy()

    # Final
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start_global:.1f}s")
    return [float(x) for x in final]
