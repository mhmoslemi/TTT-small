"""
Multi-scale optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses coarse-to-fine scaling, Adam optimizer on log-parameterization,
active-set QP for exact peak equalization, and multi-start exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _upsample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _adam_optimize(a_init, deadline, min_time_remaining=10.0, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on log-parameterization to minimize smoothed max of conv."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    # Adaptive beta (temperature) for smoothing
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted)-1)]
    beta = max(gap, max_c * 0.01)
    
    last_time = time.monotonic()
    it = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        it += 1
        
        # Compute gradient
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Smoothed objective gradient
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        W = float(np.dot(w, c))
        grad_z = 4.0 * n * a * (h - W)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Learning rate with decay
        lr_t = lr / (1 + 0.01 * t / 1000)
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Check score every 10 iterations
        if it % 10 == 0:
            a_new = _softmax(z)
            sc_new = _score(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
            else:
                # Restore z from best_a if we've diverged
                if sc_new > best_sc * 1.5:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    m = np.zeros_like(z)
                    v = np.zeros_like(z)
        
        # Adapt beta (decrease over time)
        if it % 100 == 0:
            beta *= 0.9
        
        # Time check
        if time.monotonic() - last_time > 2.0:
            last_time = time.monotonic()
    
    return best_a, best_sc

def _active_set_qp(a_init, deadline, min_time_remaining=10.0, K_max=50):
    """Equalize top K peaks using QP: minimize ||a - a_init||^2 
    subject to sum(a)=1, a>=0, b_k <= max_top_K for k in top K."""
    from scipy.optimize import minimize
    
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_sc = _score(a)
    
    K = 5
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline and K <= K_max:
        c = _full_convolve(a, a)
        top_k = np.argsort(c)[::-1][:K]
        T = float(c[top_k[0]])  # target: all top K <= T
        
        # We want to find a' close to a such that conv(a',a')[k] <= T for k in top_k
        # and sum(a') = 1, a' >= 0
        # This is a QP: min ||a' - a||^2 s.t. a^T P_k a' <= T (approximately linearized)
        # Linearize: 2*a^T P_k (a' - a) + a^T P_k a <= T
        # i.e., (P_k a)^T a' <= T - a^T P_k a + a^T P_k a = T
        # Wait, the linearization of a^T P_k a at a is: a^T P_k a + 2*(P_k a)^T (a' - a)
        # So constraint: a^T P_k a + 2*(P_k a)^T (a' - a) <= T
        # i.e., (P_k a)^T a' <= T - a^T P_k a + a^T P_k a = T
        # Hmm, that's not right either. Let me redo.
        
        # b_k(a') = a'^T P_k a' ≈ a^T P_k a + 2 (P_k a) · (a' - a)
        # We want b_k(a') <= T, so:
        # a^T P_k a + 2 (P_k a) · (a' - a) <= T
        # (P_k a) · a' <= (T - a^T P_k a)/2 + (P_k a) · a
        # (P_k a) · a' <= (T - a^T P_k a)/2 + a^T P_k a
        # (P_k a) · a' <= (T + a^T P_k a)/2
        
        # So the constraint is: (P_k a) · a' <= (T + b_k(a))/2
        # But b_k(a) for k in top_k is approximately T (since we're equalizing)
        # So constraint is approximately: (P_k a) · a' <= T
        
        # Let's just use: (P_k a) · a' <= T for each k in top_k
        # Plus: sum(a') = 1, a' >= 0
        # Objective: min ||a' - a||^2
        
        # This is a standard QP. Let's use scipy minimize with SLSQP.
        # But for large n, this might be slow. Let's use a simpler approach:
        # project a onto the intersection of half-spaces defined by the constraints.
        
        # Alternative: use the subgradient method more aggressively
        # Direction: average gradient of the top K active constraints
        P_ka = np.zeros(n)  # This will hold P_k @ a for the "average" active k
        for k in top_k:
            # (P_k @ a)[j] = a[k-j] (with a[m]=0 for m outside [0,n-1])
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    P_ka[j] += a[idx]
        P_ka /= K
        grad = 2.0 * P_ka
        grad -= np.mean(grad)  # To preserve sum
        
        # Line search
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            K += 5
            continue
        
        alpha = 1.0 / (gnorm + 1e-15)
        improved = False
        for _ in range(30):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < best_sc - 1e-14:
                a = a_try
                best_sc = sc_try
                best_a = a_try.copy()
                improved = True
                break
            alpha *= 0.5
        
        if improved:
            no_improve = 0
            K = min(K_max, K + 2)
        else:
            no_improve += 1
            if no_improve > 30:
                K += 5
                no_improve = 0
            if K > K_max:
                break
    
    return best_a, best_sc

def _make_init(n, kind, rng):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "random":
        a = rng.exponential(1.0, n)
        return a / a.sum()
    elif kind == "random_gauss":
        a = np.abs(rng.standard_normal(n))
        a += 0.1
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "chirp":
        t = np.arange(n)
        a = 1.0 + 0.5 * np.abs(np.sin(np.pi * t * t / n))
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20*(t-0.3)**2) + np.exp(-20*(t-0.7)**2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n
