    while k < n:
        k <<= 1
    return max(2, k)

    out_n = n + m - 1
    N = _next_pow2(out_n)

1. Sort x in descending order
2. Find the threshold τ such that sum(max(x_i - τ, 0)) = 1
3. Return max(x - τ, 0)

This takes O(n log n) per iteration.

Let me also think about the dual update. The exponent gradient update is:
λ_k(t+1) = λ_k(t) * exp(η * c[k](a_t)) / Z

where Z is the normalization constant and η is the learning rate.

This is equivalent to: log λ_k(t+1) = log λ_k(t) + η * c[k](a_t) - log Z

In practice, for numerical stability, we work with the unnormalized values and normalize after.

The learning rate η should be chosen carefully. If η is too large, the distribution concentrates too quickly. If too small, convergence is slow. A good choice is η = O(1/sqrt(T)) for T total iterations, but for practical purposes, we can use a fixed η or anneal it.

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power_law":
        t = np.linspace(0.0, 1.0, n)
        a = (t + 0.05) ** 2.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "skewed":
        t = np.linspace(0.0, 1.0, n)
        a = np.exp(-2.5 * (t - 0.3) ** 2) + 0.1 * np.exp(-2.5 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sawtooth":
        t = np.linspace(0.0, 1.0, n)
        a = t ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _qd_grad(a, lam):
    """Compute gradient of sum_k lam[k]*c[k] with respect to a.
    
    c[k] = sum_j a[j] a[k-j]
    d/d a[j] (c[k]) = a[k-j] + a[j-k] (with a[m]=0 outside [0,n-1])
    
    So grad[j] = sum_k lam[k] * (a[k-j] + a[j-k])
    """
    n = len(a)
    c_len = 2 * n - 1
    # grad[j] = sum_k lam[k] a[k-j] + sum_k lam[k] a[j-k]
    # Term 1: for fixed j, sum_k lam[k] a[k-j]. Let m = k-j: sum_m lam[m+j] a[m]
    #   This is the cross-correlation of lam and a, evaluated at j.
    # Term 2: for fixed j, sum_k lam[k] a[j-k]. Let m = k: sum_m lam[m] a[j-m]
    #   This is the convolution of lam and a, evaluated at j.
    
    # Using FFT for both terms
    N = _next_pow2(2 * c_len)
    
    # Term 2: (lam * a)[j] = sum_m lam[m] a[j-m], for j = 0, ..., n-1
    # Full convolution of lam (length c_len) and a (length n) has length c_len + n - 1
    fa = np.fft.rfft(a, N)
    fl = np.fft.rfft(lam, N)
    conv_la = np.fft.irfft(fl * fa, N)[:c_len + n - 1]
    term2 = conv_la[:n]  # (lam * a)[j] for j = 0,...,n-1
    
    # Term 1: sum_m lam[m+j] a[m] for j = 0, ..., n-1
    # = (a * lam_rev)[c_len - 1 - j] where lam_rev[m] = lam[c_len-1-m]
    lam_rev = lam[::-1]
    flr = np.fft.rfft(lam_rev, N)
    conv_alr = np.fft.irfft(fa * flr, N)[:n + c_len - 1]
    # conv_alr[k] = sum_m a[m] lam_rev[k-m] = sum_m a[m] lam[c_len-1-k+m]
    # We want: for j, sum_m a[m] lam[m+j]
    # conv_alr[k] with k = c_len-1-j: sum_m a[m] lam[c_len-1-(c_len-1-j)+m] = sum_m a[m] lam[j+m]
    # Yes! So term1[j] = conv_alr[c_len-1-j]
    term1 = np.zeros(n)
    for j in range(n):
        term1[j] = conv_alr[c_len - 1 - j]
    
    return term1 + term2

For k=0: x^T P_0 x = x_0^2 >= 0 ✓
For k=1: x^T P_1 x = x_0 x_1 (can be negative!)
For k=n-1 with n=3: x^T P_2 x = x_0 x_2 + x_1^2 (can be negative if x_0 x_2 < -x_1^2)

So P_k is NOT PSD in general, and Q = sum lam_k P_k may not be PSD either.

This means the inner problem `min a^T Q a s.t. a >= 0, sum(a) = 1` is a non-convex QP (if Q is not PSD). However, in practice, for the lam distributions we'll use (which are concentrated on a few lag values), Q will be "close to" PSD, and projected gradient should still make progress.
