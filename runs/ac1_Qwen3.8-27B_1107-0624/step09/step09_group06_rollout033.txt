"""
Multi-phase log-sum-exp + L-BFGS optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive beta schedule, multiple restarts, direct peak equalization, and multi-n search.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _log_softmax_grad_transform(grad_a, a):
    """Transform gradient from a-space to z-space (softmax parameterization)."""
    return a * (grad_a - float(np.dot(a, grad_a)))

def _smoobj_and_grad(z, beta, n):
    """Returns (f_beta, grad_z) for the smoothed objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -500, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        f_val = c_max
    else:
        w = e / s
        f_val = c_max + beta * np.log(s)
    # Gradient of f_beta w.r.t. a: 2 * convolve(a_rev, w)[n-1:2n-1]
    a_rev = a[::-1]
    h = _full_convolve(w, a_rev)[n-1:2*n-1]
    grad_a = 2.0 * h
    # Transform to z-space
    W = float(np.dot(a, grad_a))  # = sum_k w[k]*c[k] = f_val - beta*log(s) + ... hmm
    # Actually W = sum_j a[j]*grad_a[j] = sum_j a[j]*2*sum_k w[k]*a[k-j]
    # = 2*sum_k w[k]*sum_j a[j]*a[k-j] = 2*sum_k w[k]*c[k]
    grad_z = a * (grad_a - W)
    return f_val, grad_z

def _objective_wrapper(z, beta, n, score_factor):
    """Objective for L-BFGS: score_factor * f_beta."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -500, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return score_factor * c_max
    return score_factor * (c_max + beta * np.log(s))

def _grad_wrapper(z, beta, n, score_factor):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -500, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    h = _full_convolve(w, a_rev)[n-1:2*n-1]
    grad_a = 2.0 * h
    W = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - W)
    return score_factor * grad_z

def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)

def _run_lbfgs_beta(a_init, beta, n, max_iter, min_time_remaining, deadline):
    """Run L-BFGS at a given beta level."""
    z = np.log(np.maximum(a_init, 1e-300))
    z = z - z.mean()
    score_factor = 2.0 * n
    
    obj = lambda z: _objective_wrapper(z, beta, n, score_factor)
    grad = lambda z: _grad_wrapper(z, beta, n, score_factor)
    
    try:
        result = minimize(obj, z, jac=grad, method='L-BFGS-B',
                         options={'maxiter': max_iter, 'ftol': 1e-16,
                                  'gtol': 1e-14, 'maxcor': 50, 'maxls': 40,
                                  'iprint': 0})
        a_new = _softmax(result.x)
        return a_new
    except Exception as ex:
        print(f"    L-BFGS error at beta={beta:.2e}: {ex}")
        return a_init

def _nesterov_subgrad(a_init, deadline, min_time_remaining=30.0, seed=0, K=20):
    """Subgradient method with Nesterov acceleration."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())
    
    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    mu = 0.9  # momentum
    t = 0
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Active peaks
        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]
        w = np.zeros_like(c)
        w[active] = 1.0 / K
        
        # Subgradient
        a_rev = a[::-1]
        h = _full_convolve(w, a_rev)[n-1:2*n-1]
        grad_a = 2.0 * h
        # Project to tangent space
        grad_t = grad_a - float(np.dot(a, grad_a))
        gnorm = np.linalg.norm(grad_t)
        
        if gnorm < 1e-18:
            break
        
        # Step size (Polyak-like)
        gap = c_max - float(np.mean(c))
        step = 2.0 * gap / (gnorm * gnorm + 1e-30)
        step = min(step, 0.5 / (t + 1))
        
        # Nesterov step
        v = mu * v - step * grad_t
        a_cand = a + v
        a_proj = _project_simplex(a_cand)
        
        # Check if improved
        c_new = _full_convolve(a_proj, a_proj)
        sc_new = 2.0 * n * float(c_new.max())
        
        if sc_new < current_sc:
            a = a_proj
        else:
            v = 0.5 * v
            a = a_proj  # stay on simplex
        
        # Restart if stuck
        if no_improve > 1000:
            perturb = 0.001 * rng.standard_normal(n)
            a_pert = _project_simplex(a + perturb)
            break
    
    return best_a, best_sc

def _direct_peak_reduce(a_init, deadline, min_time_remaining=30.0):
    """Directly reduce the max peak by adjusting contributors."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    lr = 0.01
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
        
        # Find the peak location
        k_peak = int(np.argmax(c))
        
        # Gradient at the peak
        g = np.zeros(n)
        j_lo = max(0, k_peak - n + 1)
        j_hi = min(n - 1, k_peak)
        for j in range(j_lo, j_hi + 1):
            g[j] += 2.0 * a[k_peak - j]
        
        # Reduce a[j] where g[j] is large, increase where g[j] is small
        g_norm = g / (np.max(g) + 1e-30)
        # Move mass from high-g to low-g positions
        high_idx = np.argsort(g)[::-1][:max(10, n // 100)]
        low_idx = np.argsort(g)[:max(10, n // 100)]
        
        total_high = a[high_idx].sum()
        if total_high < 1e-15:
            break
        
        move = lr * total_high * 0.01
        a[high_idx] -= move / len(high_idx)
        a[low_idx] += move / len(low_idx)
        
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
    
    return best_a, best_sc

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_edge":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        # Try to make the convolution flat
        t = np.linspace(0, 1, n)
        # Use a Tukey-like window
        a = 1.0
        taper = max(1, n // 20)
        for i in range(taper):
            a[i] = 0.5 * (1 + np.cos(np.pi * (i + 0.5) / taper))
            a[n-1-i] = 0.5 * (1 + np.cos(np.pi * (i + 0.5) / taper))
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2) + 0.1
        return a / a.sum()
    elif kind == "inverse_trig":
        # Try a profile that compensates for the triangular shape
        t = np.linspace(0, 1, n)
        # The convolution of uniform is triangular; we want the opposite
        a = 1.0 / (0.5 + np.abs(t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
