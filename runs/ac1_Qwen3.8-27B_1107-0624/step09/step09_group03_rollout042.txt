def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    """Compute correlation: result[i] = sum_k A_vec[k] * a[k-i] (valid range)"""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _project_nonneg_sum1(v):
    """Project v to non-negative simplex (sum=1)."""
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-15:
        return np.ones_like(v) / len(v)
    return v / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec, rhs, max_cg=300, tol=1e-14):
    """Solve matvec(x) = rhs using conjugate gradient."""
    n = len(rhs)
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-15:
        return x
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _newton_active(a, deadline, max_outer=200):
    """
    Active-set Newton method with adaptive tolerance.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive tolerance: start loose, tighten over iterations
        if outer < 20:
            tol_fracs = [0.05, 0.02, 0.01, 0.005]
        elif outer < 60:
            tol_fracs = [0.005, 0.002, 0.001, 0.0005]
        else:
            tol_fracs = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
        
        best_improvement = 0
        best_a_new = None
        
        for tol_frac in tol_fracs:
            active = c >= (c_max - tol_frac * c_max)
            if not active.any():
                active[np.argmax(c)] = True
            m = int(active.sum())
            if m < 2 or m > 10 * n:
                continue
            
            A_vec = active.astype(np.float64) / m
            g = 2.0 * _corr_a_A(a, A_vec, n)
            g_t = g - np.mean(g)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                continue
            
            mu = 1e-6 / m + 1e-10
            
            # Define matvec for CG
            A_vec_ref = A_vec
            n_ref = n
            mu_ref = mu
            
            def _hess_vp(v):
                return 2.0 * _corr_a_A(v, A_vec_ref, n_ref) + mu_ref * v
            
            u = _cg_solve(_hess_vp, g_t, max_cg=300)
            w = _cg_solve(_hess_vp, np.ones(n), max_cg=300)
            w_sum = float(w.sum())
            
            if abs(w_sum) < 1e-15:
                dx = -u
            else:
                lam = float(u.sum()) / w_sum
                dx = -u + lam * w
            
            dx = dx - np.mean(dx)
            
            # Line search
            t = 1.0
            for ls in range(60):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < c_max - 1e-14:
                    improvement = c_max - f_try
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_a_new = a_try.copy()
                    break
                t *= 0.5
        
        if best_a_new is None:
            break
        
        a = best_a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        if best_improvement < 1e-15 * c_max:
            break
    
    return best_a, best_sc


def _deflation_step(a, n_attempts=100):
    """Multi-peak deflation."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for attempt in range(n_attempts):
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find peaks within 98% of max
        threshold = c_max * 0.98
        peak_mask = c >= threshold
        n_peaks = int(peak_mask.sum())
        if n_peaks == 0:
            peak_mask[np.argmax(c)] = True
            n_peaks = 1
        
        w = np.zeros_like(c)
        w[peak_mask] = 1.0 / n_peaks
        
        grad = 2.0 * _corr_a_A(a, w, n)
        g_t = grad - np.mean(grad)
        
        applied = False
        for strength in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
            a_try = a - strength * g_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                applied = True
                break
        
        if not applied:
            break
    
    return best_a, best_sc


def _lbfgs_lse_continuation(a_init, deadline, min_time_remaining=60.0):
    """
    L-BFGS with log-sum-exp continuation.
    Aggressive schedule going to very small beta.
    """
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    # Beta schedule: geometric from beta_0 to beta_end
    beta_0 = max_c * 0.01
    beta_end = max(1e-10, max_c * 1e-8)
    n_betas = 35
    log_betas = np.linspace(np.log(beta_0), np.log(beta_end), n_betas)
    betas = np.exp(log_betas)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # More iterations at smaller beta
        if beta < max_c * 0.001:
            max_iter = min(5000, max(100, int(remaining / 0.1)))
        elif beta < max_c * 0.01:
            max_iter = min(3000, max(50, int(remaining / 0.15)))
        else:
            max_iter = min(2000, max(30, int(remaining / 0.2)))
        
        try:
            def _obj(z_val):
                return _lse_obj(z_val, beta, n)
            
            def _grad(z_val):
                return _lse_grad(z_val, beta, n)
            
            result = minimize(
                _obj, z,
                jac=_grad,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 500,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # If no improvement, try smaller step
                pass
        except Exception as ex:
            print(f"    LSE L-BFGS beta={beta:.2e}: exception {ex}")
    
    return current_a


def _interpolate(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    """Perturb the sequence in log-space."""
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)

import numpy as np
import time
from scipy.optimize import minimize


_lse_state = {'beta': 1.0, 'n': 0}


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    if out_len <= 0:
        return np.zeros(n)
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lse_objective(z):
    beta = _lse_state['beta']
    n = _lse_state['n']
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z):
    beta = _lse_state['beta']
    n = _lse_state['n']
    a = _softmax(z)
    c = _full_convolve(a, a)

    e = np.exp(np.clip(shifted, -700, 0))

    if s < 1e-300 or not np.isfinite(s):

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec, rhs, max_cg=300, tol=1e-14):
    n = len(rhs)
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-15:
        return x
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


class _HessOp:

        if self.mu != 0.0:
            result = result + self.mu * v
        return result

    for outer in range(max_outer):

        c_max = float(c.max())

        if outer < 30:
            tol_fracs = [0.05, 0.02, 0.01, 0.005]
        elif outer < 80:
            tol_fracs = [0.005, 0.002, 0.001, 0.0005]
        else:
            tol_fracs = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]

        best_improvement = 0.0
        best_a_new = None

        for tol_frac in tol_fracs:
            active = c >= (c_max - tol_frac * c_max)
            if not active.any():
                active[np.argmax(c)] = True
            m = int(active.sum())
            if m < 2 or m > 10 * n:
                continue

            A_vec = active.astype(np.float64)
            g = 2.0 * _corr_a_A(a, A_vec, n)
            g_t = g - np.mean(g)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                continue

            mu = 1e-6 / m + 1e-10
            op = _HessOp(A_vec, n, mu)
            u = _cg_solve(op, g_t, max_cg=400)
            w = _cg_solve(op, np.ones(n), max_cg=400)
            w_sum = float(w.sum())

            if abs(w_sum) < 1e-15:
                dx = -u
            else:
                lam = float(u.sum()) / w_sum
                dx = -u + lam * w

            dx = dx - np.mean(dx)

            t = 1.0
            for ls in range(60):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < c_max - 1e-14:
                    improvement = c_max - f_try
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_a_new = a_try.copy()
                    break
                t *= 0.5

        if best_a_new is None:
            break

        a = best_a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if best_improvement < 1e-15 * c_max:
            break

    return best_a, best_sc


def _add_only_newton(a_init, deadline, max_iter=100):
    """Add-only active set Newton: active set only grows."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())
    active_mask = np.zeros(len(c), dtype=bool)
    active_mask[np.argmax(c)] = True

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        A_vec = active_mask.astype(np.float64)
        m = int(active_mask.sum())
        if m > 5 * n:
            # Reset to top peaks if active set too large
            c = _full_convolve(a, a)
            c_max = float(c.max())
            active_mask = np.zeros(len(c), dtype=bool)
            n_top = min(100, len(c))
            top_idx = np.argsort(c)[::-1][:n_top]
            active_mask[top_idx] = True
            A_vec = active_mask.astype(np.float64)
            m = n_top

        g = 2.0 * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-6 / m + 1e-10
        op = _HessOp(A_vec, n, mu)
        u = _cg_solve(op, g_t, max_cg=400)
        w = _cg_solve(op, np.ones(n), max_cg=400)
        w_sum = float(w.sum())

        if abs(w_sum) < 1e-15:
            dx = -u
        else:
            lam = float(u.sum()) / w_sum
            dx = -u + lam * w

        dx = dx - np.mean(dx)

        # Line search
        applied = False
        t = 1.0
        for ls in range(50):
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < c_max - 1e-14:
                a = a_try
                c = c_try
                c_max = f_try
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                applied = True
                break
            t *= 0.5

        if not applied:
            break

        # Add violators
        new_violators = c > (c_max + 1e-10 * c_max)
        n_new = int(new_violators.sum())
        if n_new == 0:
            # Try to add near-max elements
            near_max = c >= (c_max - 0.001 * c_max)
            new_violators = near_max & ~active_mask
            n_new = int(new_violators.sum())
            if n_new == 0:
                break
        # Limit additions per step
        if n_new > 50:
            viol_idx = np.where(new_violators)[0]
            viol_c = c[viol_idx]
            top_viol = viol_idx[np.argsort(viol_c)[::-1][:50]]
            new_violators = np.zeros_like(new_violators)
            new_violators[top_viol] = True

        active_mask = active_mask | new_violators

    return best_a, best_sc


def _spectral_shaping(a, alpha=0.5, deadline=None):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    P = np.abs(A) ** 2
    # Compress the power spectrum
    P_flat = P ** alpha
    # Reconstruct with modified magnitude, keep phase
    phase = np.angle(A)
    A_new = np.sqrt(P_flat) * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _deflation_step(a, n_attempts=80):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for attempt in range(n_attempts):
        if time.monotonic() > deadline_global:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        threshold = c_max * 0.98
        peak_mask = c >= threshold
        n_peaks = int(peak_mask.sum())
        if n_peaks == 0:
            peak_mask[np.argmax(c)] = True
            n_peaks = 1

        w = np.zeros_like(c)
        w[peak_mask] = 1.0 / n_peaks

        grad = 2.0 * _corr_a_A(a, w, n)
        g_t = grad - np.mean(grad)

        applied = False
        for strength in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
            a_try = a - strength * g_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                applied = True
                break

        if not applied:
            break

    return best_a, best_sc


deadline_global = np.inf


def _lbfgs_lse_continuation(a_init, deadline, min_time_remaining=60.0):
    global deadline_global
    deadline_global = deadline

    n = len(a_init)
    _lse_state['n'] = n
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    beta_0 = max_c * 0.02
    beta_end = max(1e-12, max_c * 5e-9)
    n_betas = 40
    log_betas = np.linspace(np.log(beta_0), np.log(beta_end), n_betas)
    betas = np.exp(log_betas)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        _lse_state['beta'] = float(beta)

        if beta < max_c * 0.001:
            max_iter = min(5000, max(100, int(remaining / 0.08)))
        elif beta < max_c * 0.01:
            max_iter = min(3000, max(50, int(remaining / 0.12)))
        else:
            max_iter = min(2000, max(30, int(remaining / 0.15)))

        try:
            result = minimize(
                _lse_objective, z,
                jac=_lse_gradient,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 500,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _refinement_combine(a, deadline, max_rounds=8):
    """Iterate between Newton, spectral shaping, and deflation."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    global deadline_global
    deadline_global = deadline

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Newton step
        if time.monotonic() < deadline - 30:
            an, scn = _add_only_newton(best_a, min(deadline - 20, time.monotonic() + 40), max_iter=20)
            if scn < best_sc:
                best_sc = scn
                best_a = an.copy()

        # Spectral shaping
        if time.monotonic() < deadline - 20:
            for alpha in [0.3, 0.5, 0.7]:
                a_sp = _spectral_shaping(best_a, alpha)
                sc_sp = _score(a_sp)
                if sc_sp < best_sc - 1e-12:
                    best_sc = sc_sp
                    best_a = a_sp.copy()

        # Deflation
        if time.monotonic() < deadline - 15:
            ad, scd = _deflation_step(best_a, n_attempts=15)
            if scd < best_sc:
                best_sc = scd
                best_a = ad.copy()

        # Check convergence
        if time.monotonic() < deadline:
            sc_cur = _score(best_a)
            if abs(sc_cur - best_sc) < 1e-14:
                break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS LSE, add-only Newton,
    spectral shaping, deflation, multi-scale, and restarts.
    """
    global deadline_global
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    deadline_global = deadline
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s_val = float(p_arr.sum())
        if s_val < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            if prev < 10:
                print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                      f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s_val = float(parent_raw.sum())
            if s_val > 0.01:
                parent = parent_raw / s_val
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={len(parent)}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        parent = np.ones(16384) / 16384
        update_best(parent, _score(parent), "uniform_init")

    # === Phase 1: L-BFGS LSE continuation (aggressive) ===
    print(f"\n[Phase 1] L-BFGS LSE (t={time.monotonic() - t_start:.1f}s)")
    p1_dl = min(deadline - 700, time.monotonic() + 350)
    a1 = _lbfgs_lse_continuation(parent, p1_dl, min_time_remaining=700.0)
    update_best(a1, _score(a1), "lbfgs_lse")
    print(f"  After LSE: {best_score:.10f}")

    # === Phase 2: Add-only Newton ===
    print(f"\n[Phase 2] Add-only Newton (t={time.monotonic() - t_start:.1f}s)")
    p2_dl = min(deadline - 600, time.monotonic() + 200)
    a2, sc2 = _add_only_newton(best_seq, p2_dl, max_iter=80)
    update_best(a2, sc2, "add_only_newton")
    print(f"  After Newton: {best_score:.10f}")

    # === Phase 3: Spectral shaping ===
    print(f"\n[Phase 3] Spectral shaping (t={time.monotonic() - t_start:.1f}s)")
    for alpha in [0.3, 0.5, 0.7, 0.9]:
        a_sp = _spectral_shaping(best_seq, alpha)
        sc_sp = _score(a_sp)
        update_best(a_sp, sc_sp, f"spectral_a={alpha}")
    print(f"  After spectral: {best_score:.10f}")

    # === Phase 4: Deflation ===
    print(f"\n[Phase 4] Deflation (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a4, sc4 = _deflation_step(best_seq, n_attempts=40)
        update_best(a4, sc4, "deflation")
    print(f"  After deflation: {best_score:.10f}")

    # === Phase 5: Combined refinement loop ===
    print(f"\n[Phase 5] Refinement loop (t={time.monotonic() - t_start:.1f}s)")
    p5_dl = min(deadline - 350, time.monotonic() + 200)
    a5, sc5 = _refinement_combine(best_seq, p5_dl, max_rounds=8)
    update_best(a5, sc5, "refinement")
    print(f"  After refinement: {best_score:.10f}")

    # === Phase 6: L-BFGS polish ===
    print(f"\n[Phase 6] L-BFGS polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        p6_dl = min(deadline - 250, time.monotonic() + 100)
        a6 = _lbfgs_lse_continuation(best_seq, p6_dl, min_time_remaining=250.0)
        update_best(a6, _score(a6), "lbfgs_polish")
    print(f"  After polish: {best_score:.10f}")

    # === Phase 7: Add-only Newton final ===
    print(f"\n[Phase 7] Final Newton (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        p7_dl = min(deadline - 200, time.monotonic() + 80)
        a7, sc7 = _add_only_newton(best_seq, p7_dl, max_iter=50)
        update_best(a7, sc7, "final_newton")
    print(f"  After final Newton: {best_score:.10f}")

    # === Phase 8: Multi-scale n=32768 ===
    print(f"\n[Phase 8] Multi-scale n=32768 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 180:
        a_large = _interpolate_seq(best_seq, 32768)
        sc_large = _score(a_large)
        print(f"  n=32768 init: {sc_large:.10f}")
        update_best(a_large, sc_large, "interp_32768")

        if time.monotonic() < deadline - 120:
            p8a_dl = min(deadline - 120, time.monotonic() + 100)
            a8a = _lbfgs_lse_continuation(a_large, p8a_dl, min_time_remaining=120.0)
            update_best(a8a, _score(a8a), "lbfgs_32768")

        if time.monotonic() < deadline - 60:
            p8b_dl = min(deadline - 60, time.monotonic() + 40)
            a8b, sc8b = _add_only_newton(best_seq, p8b_dl, max_iter=30)
            update_best(a8b, sc8b, "newton_32768")

        # Spectral shaping on large
        if time.monotonic() < deadline - 40:
            for alpha in [0.5, 0.7]:
                a_sp_l = _spectral_shaping(best_seq, alpha)
                sc_sp_l = _score(a_sp_l)
                update_best(a_sp_l, sc_sp_l, f"spectral_32k_a={alpha}")

    # === Phase 9: Diversified restarts ===
    print(f"\n[Phase 9] Restarts (t={time.monotonic() - t_start:.1f}s)")
    for sigma in [0.002, 0.005, 0.01, 0.03, 0.08]:
        if time.monotonic() > deadline - 30:
            break
        p = _perturb(best_seq, sigma, rng)
        rem = deadline - 25 - time.monotonic()
        if rem < 15:
            break
        # Short LSE
        ar = _lbfgs_lse_continuation(p, deadline - 15, min_time_remaining=15.0)
        update_best(ar, _score(ar), f"restart_s={sigma}")
        # Quick Newton
        if time.monotonic() < deadline - 15:
            arn, scn = _add_only_newton(ar, deadline - 8, max_iter=15)
            update_best(arn, scn, f"newton_r_s={sigma}")

    # === Phase 10: Final combined push ===
    print(f"\n[Phase 10] Final push (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 8:
        af, scf = _refinement_combine(best_seq, deadline - 3, max_rounds=5)
        update_best(af, scf, "final_push")
        # One more Newton
        if time.monotonic() < deadline - 3:
            an, scn = _add_only_newton(best_seq, deadline - 1, max_iter=20)
            update_best(an, scn, "very_final")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s_val = float(final.sum())
        if s_val < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        # Cross-check with evaluate_sequence
        try:
            eval_score = evaluate_sequence([float(x) for x in final])
            print(f"FINAL: _score={sc_final:.10f}, evaluate={eval_score:.10f}, "
                  f"n={len(final)}, time={time.monotonic() - t_start:.1f}s")
            if not np.isfinite(eval_score) or eval_score > sc_final + 1e-6:
                print("WARNING: score mismatch!")
        except Exception:
            print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
                  f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
