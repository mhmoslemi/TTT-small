    k = 1

"""
Multi-phase optimizer: LP-guided steepest descent + aggressive L-BFGS annealing
+ multi-n exploration + spectral refinements. Key innovations: (1) LP-based exact
steepest descent directions for the max-norm objective, (2) 30-stage beta schedule
with longer convergence at small beta, (3) systematic n-sweep, (4) diversity via
randomized perturbations with re-optimization.

import numpy as np
import time
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _conv_grad_single(a, k):
    n = len(a)
    grad = np.zeros(n)
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    if j_lo <= j_hi:
        idx = k - np.arange(j_lo, j_hi + 1)
        grad[j_lo:j_hi + 1] += a[idx]
    j_lo2 = max(0, k)
    j_hi2 = min(n - 1, k + n - 1)
    if j_lo2 <= j_hi2:
        idx = np.arange(j_lo2, j_hi2 + 1) - k
        grad[j_lo2:j_hi2 + 1] += a[idx]
    return grad


def _conv_grad_multi(a, active_indices):
    n = len(a)
    grad = np.zeros(n)
    for k in active_indices:
        grad += _conv_grad_single(a, k)
    return grad


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.15 * np.cos(2.0 * np.pi * 13.0 * t / n) + 0.05 * np.cos(2.0 * np.pi * 29.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    elif kind == "flat_taper":
        a = np.ones(n)
        taper = max(1, n // 100)
        for i in range(taper):
            a[i] = (i + 1) / taper
            a[n - 1 - i] = (i + 1) / taper
        return a / a.sum()
    elif kind == "rho":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _lp_solve_direction(a, K=50, time_limit=25):
    """Solve reduced LP for steepest descent direction of max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    if n > 32768:
        return None, c_max

    c_idx = np.argsort(c)[::-1]
    active = c_idx[:K]

    # LP: minimize t
    # s.t. g_k . d <= t - c[k]  (K constraints)
    #      -d_i <= a[i]         (n constraints)
    #      sum(d) = 0           (1 equality)
    # Variables: [d_0,...,d_{n-1}, t], total n+1

    rows_ub = []
    cols_ub = []
    vals_ub = []
    b_ub = []

    for i, k in enumerate(active):
        g = _conv_grad_single(a, k)
        nz = np.nonzero(np.abs(g) > 1e-15)[0]
        for j in nz:
            rows_ub.append(i)
            cols_ub.append(int(j))
            vals_ub.append(g[j])
        rows_ub.append(i)
        cols_ub.append(n)
        vals_ub.append(-1.0)
        b_ub.append(-c[k])

    offset = len(active)
    for i in range(n):
        rows_ub.append(offset + i)
        cols_ub.append(i)
        vals_ub.append(-1.0)
        b_ub.append(a[i])

    A_ub = csr_matrix((vals_ub, (rows_ub, cols_ub)), shape=(len(b_ub), n + 1))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=np.array(b_ub), A_eq=A_eq, b_eq=b_eq,
                         method='highs', options={'disp': False, 'time_limit': time_limit})
        if result.success:
            d = result.x[:n]
            d = d - np.mean(d)
            return d, c_max
    except Exception:
        pass
    return None, c_max

def _run_aggressive_lbfgs(a_init, deadline, min_remaining=60.0):
    """Aggressive L-BFGS with 30-stage beta schedule."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()

        a = a / s
    best_a = a.copy()

    if gap < 1e-15:
        gap = mb * 0.01
    base_beta = max(gap, mb * 0.003)

    # 30-stage schedule: fast initial cooling then slow
    ratios = []
    for i in range(30):
        if i < 10:
            ratios.append(0.4 ** i)
        elif i < 20:
            ratios.append(0.4 ** 10 * (0.7 ** (i - 10)))
        else:
            ratios.append(0.4 ** 10 * 0.7 ** 10 * (0.85 ** (i - 20)))
    betas = base_beta * np.array(ratios)
    betas = np.maximum(betas, mb * 1e-7)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        remaining = deadline - min_remaining - time.monotonic()

        try:
            res = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 80, 'maxls': 60})
            a_new = _softmax(res.x)

            no_improve += 1
        if no_improve > 300:
            break

        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
        if gap < c_max * 0.003 and K < K_max:
            K = min(K_max, int(K * 1.3) + 1)

        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]
        grad = _conv_grad_multi(a, active)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break

        lr = base_lr / (gnorm + 1e-18) * 0.1
        v = momentum * v - lr * grad_t
        a_cand = _project_simplex(a + v)
        c_new = _full_convolve(a_cand, a_cand)
        sc_new = 2.0 * n * float(c_new.max())
        if sc_new < cur_sc:
            a = a_cand
        else:
            v = 0.5 * v
            a = a_cand
        if no_improve % 100 == 0 and no_improve > 0:
            perturb = 0.0005 * rng.standard_normal(n)
            a_p = _project_simplex(a + perturb)
            sc_p = _score(a_p)
            if sc_p < best_sc:
                a = a_p
                best_a = a.copy()
                best_sc = sc_p
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_sc


def _shrinkage_optimize(a, deadline):
    """Try zeroing small elements to reduce effective support."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    for frac in [0.001, 0.002, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)
            print(f"  Shrinkage frac={frac}: {best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Main search: LP-guided + aggressive L-BFGS + multi-n + refinement."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                parent_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
                best_seq = parent.copy()
                best_score = parent_sc
    except Exception:
        pass

    if best_seq is None:
        n_init = 16384
        best_seq = _make_profile(n_init, "uniform")
        best_score = _score(best_seq)
        print(f"No parent, uniform n={n_init}, score={best_score:.10f}")

    # === Phase 1: LP-guided optimization ===
    print(f"\n[P1] LP-guided (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 600, t_start + 400)
    if time.monotonic() < p1_deadline:
        a_p1, sc_p1 = _run_lp_guided(best_seq, p1_deadline, min_remaining=60.0, max_iters=25)
        if sc_p1 < best_score:
            best_seq = a_p1.copy()
            best_score = sc_p1
            print(f"  NEW BEST: {best_score:.10f} [LP-guided]")

    # === Phase 2: Aggressive L-BFGS ===
    print(f"\n[P2] Aggressive L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 400, t_start + 700)
    if time.monotonic() < p2_deadline:
        a_p2, sc_p2 = _run_aggressive_lbfgs(best_seq, p2_deadline, min_remaining=60.0)
        if sc_p2 < best_score:
            best_seq = a_p2.copy()
            best_score = sc_p2
            print(f"  NEW BEST: {best_score:.10f} [aggr-LBFGS]")

    # === Phase 3: Nesterov refinement ===
    print(f"\n[P3] Nesterov refine (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 300, t_start + 900)
    if time.monotonic() < p3_deadline:
        a_p3, sc_p3 = _nesterov_refine(best_seq, p3_deadline, min_remaining=40.0,
                                       K_init=3, K_max=80, seed=seed)
        if sc_p3 < best_score:
            best_seq = a_p3.copy()
            best_score = sc_p3
            print(f"  NEW BEST: {best_score:.10f} [nesterov]")

    # === Phase 4: Second aggressive L-BFGS (even finer) ===
    print(f"\n[P4] Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 250, t_start + 1100)
    if time.monotonic() < p4_deadline:
        a_p4, sc_p4 = _run_aggressive_lbfgs(best_seq, p4_deadline, min_remaining=40.0)
        if sc_p4 < best_score:
            best_seq = a_p4.copy()
            best_score = sc_p4
            print(f"  NEW BEST: {best_score:.10f} [fine-LBFGS]")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    p5_deadline = min(deadline - 150, t_start + 1300)
    if time.monotonic() < p5_deadline:
        best_seq, best_score = _multi_n_exploration(best_seq, best_score, p5_deadline, min_remaining=60.0)

    # === Phase 6: Shrinkage ===
    print(f"\n[P6] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 40)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} [shrinkage]")
            # Re-optimize after shrinkage
            if time.monotonic() < deadline - 30:
                a_re = _run_aggressive_lbfgs(best_seq, deadline, min_remaining=20.0)
                sc_re = _score(a_re)
                if sc_re < best_score:
                    best_seq = a_re.copy()
                    best_score = sc_re
                    print(f"  NEW BEST: {best_score:.10f} [post-shrink]")

    # === Phase 7: Final Nesterov + tiny L-BFGS polish ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 15:
        a_fin, sc_fin = _nesterov_refine(best_seq, deadline - 5, min_remaining=10.0,
                                         K_init=3, K_max=40, seed=seed + 999)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final-nest]")

    remaining = deadline - 5 - time.monotonic()
    if remaining > 10:
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        c_sorted = np.sort(c_best)[::-1]
        gap = mb - c_sorted[min(10, len(c_sorted) - 1)]
        beta_fin = max(gap * 0.01, mb * 1e-5)
        prob = _LbfgsProb(beta_fin, n)
        max_iter = min(3000, max(20, int(remaining / 0.03)))
        try:
            res = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 60})
            a_fin2 = _softmax(res.x)
            sc_fin2 = _score(a_fin2)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final-LBFGS]")
        except Exception:
            pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
