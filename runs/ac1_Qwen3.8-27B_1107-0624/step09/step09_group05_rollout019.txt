        x_new = _project_simplex(x_new)

        # Evaluate objective
        c_new = _full_convolve(x_new, x_new)
        f_new = float(np.dot(w, c_new))

        if f_new < best_f:
            best_f = f_new
            best_x = x_new.copy()

        # Backtracking if objective increased significantly
        if f_new > f_init + 0.5 * (f_new - f_init) + 1e-10:
            alpha *= 0.5
            x_new = y - alpha * grad_y
            x_new = _project_simplex(x_new)

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta_mom = (t_mom - 1.0) / t_new
        y = x_new + beta_mom * (x_new - x)
        y = _project_simplex(y)
        x = x_new
        t_mom = t_new

        # Convergence check
        if it > 50 and abs(f_new - best_f) < 1e-12:
            break

    return best_x


def _active_set_qp_refine(a_init, deadline, min_time_remaining=30.0, verbose=True):
    """Alternating active-set QP refinement with FISTA solver."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    no_improve = 0
    max_no_improve = 150

    # Geometric tolerance sweep
    tol_levels = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1]
    tol_idx = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            # Tighten tolerance on improvement
            if tol_idx > 0:
                tol_idx -= 1
        else:
            no_improve += 1
            # Widen tolerance to escape
            if no_improve > max_no_improve // 3 and tol_idx < len(tol_levels) - 1:
                tol_idx += 1
            if no_improve > max_no_improve:
                break

        tol = tol_levels[tol_idx] * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Build weight vector
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active

        # QP solve with FISTA
        remaining = deadline - time.monotonic() - min_time_remaining
        qp_time = min(10.0, max(0.1, remaining * 0.05))
        max_iter_qp = min(3000, max(100, int(qp_time * 200)))
        a_new = _qp_fista(a, w, max_iter=max_iter_qp, time_limit=qp_time)

        sc_new = _score(a_new)
        if sc_new < current_sc - 1e-14:
            a = a_new
            if verbose:
                elapsed = time.monotonic() - _t0_global
                print(f"    QP it={no_improve}: score={best_sc:.10f}, "
                      f"n_active={n_active}, tol={tol_levels[tol_idx]:.3f}, t={elapsed:.0f}s")
        else:
            no_improve += 1

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0, verbose=True, tag=""):
    """L-BFGS with aggressive adaptive beta schedule."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c0_max * 0.002:
        gap = c0_max * 0.002
    if gap < 1e-10:
        gap = c0_max * 0.001

    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
        0.00003125, 0.000015625, 0.0000078, 0.0000039,
        0.000001, 0.0000005
    ])
    betas = gap * multipliers

    t_level_start = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        size_factor = max(1.0, n / 8192.0)
        est_iter_time = 0.003 * size_factor
        max_iter = min(10000, max(200, int(remaining / (est_iter_time * 25))))

        if beta < gap * 0.001:
            max_iter = min(max_iter, 3000)

        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 200,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    elapsed = time.monotonic() - _t0_global
                    print(f"    {tag} beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}, t={elapsed:.0f}s")
            else:
                if verbose and i < 5:
                    print(f"    {tag} beta={beta:.2e}: no improvement (best={best_sc:.10f})")
        except Exception as e:
            if verbose:
                print(f"    {tag} beta={beta:.2e}: error {e}")

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _symmetric_refine_half(a_init, deadline, min_time_remaining=30.0):
    """Refine assuming symmetry: a[j] = a[n-1-j].
    Optimize only the first half, then symmetrize."""
    n = len(a_init)
    if n % 2 != 0:
        return a_init, _score(a_init)

    n_half = n // 2
    a = a_init.copy()
    s = a.sum()
    a = a / s
    a = 0.5 * (a + a[::-1])  # symmetrize
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)

    # Work with half-vector parameterized by softmax
    half = a[:n_half].copy()
    z = np.log(np.maximum(half, 1e-30))
    z = z - z.mean()

    def _sym_score_from_z(zh):
        h = _softmax(zh)
        full = np.zeros(n)
        full[:n_half] = h
        full[n_half:] = h[::-1]
        full = full / full.sum()
        return _score(full), full

    # Use L-BFGS on the half-vector
    class _SymProb:
        def __call__(self, zh):
            sc, _ = _sym_score_from_z(zh)
            return sc

        def grad(self, zh):
            h = _softmax(zh)
            full = np.zeros(n)
            full[:n_half] = h
            full[n_half:] = h[::-1]
            full = full / full.sum()
            c = _full_convolve(full, full)
            c_max = float(c.max())
            # Use subgradient at the max
            idx = int(np.argmax(c))
            grad_full = np.zeros(n)
            j_lo = max(0, idx - (n - 1))
            j_hi = min(n - 1, idx)
            for j in range(j_lo, j_hi + 1):
                if idx == 2 * j:
                    grad_full[j] = full[idx - j]
                else:
                    grad_full[j] = 2.0 * full[idx - j]
            # Map back to half (with symmetry)
            grad_half = grad_full[:n_half] + grad_full[n_half:][::-1]
            W = float(np.dot(grad_half, h))
            return h * (grad_half - W)

    # Ultra-slow gradient due to Python loop; use vectorized version instead
    # Let's just use L-BFGS on the soft-max surrogate for the full vector
    # but project to symmetry after each step.

    t_start = time.monotonic()
    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    gap = c0_max * 0.01

    for beta_mult in [1.0, 0.1, 0.01, 0.001]:
        beta = gap * beta_mult
        c_curr = _full_convolve(a, a)
        w = _soft_max_weights(c_curr, beta)

        # Gradient of soft-max objective
        grad_a = 2.0 * _full_convolve(w, a[::-1])[n-1:2*n-1]
        # Symmetrize gradient
        grad_sym = 0.5 * (grad_a + grad_a[::-1])

        # Projected gradient step on symmetry-preserving simplex
        for step_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            alpha = step_mult * beta
            a_new = a - alpha * grad_sym
            # Symmetrize
            a_new = 0.5 * (a_new + a_new[::-1])
            a_new = _project_simplex(a_new)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new
                break

        if time.monotonic() + min_time_remaining > deadline:
            break

    return best_a, best_sc

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    P1: L-BFGS on parent with adaptive beta
    P2: Active-set QP with FISTA (main new contribution)
    P3: L-BFGS round 2 (tighter)
    P4: Symmetry-aware refinement
    P5: Multi-n exploration (including 32768)
    P6: Final polish
    """
    global _t0_global
    _t0_global = time.monotonic()
    deadline = _t0_global + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        n_main = len(parent)
        print(f"Parent: n={n_main}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_main = 16384
        best_seq = _make_profile(n_main, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n={n_main}: score={best_score:.10f}")

    # === Phase 1: L-BFGS ===
    print(f"\n[P1] L-BFGS main (n={n_main}, t={time.monotonic()-_t0_global:.0f}s)")
    p1_end = _t0_global + min(400, (budget_s - 10) * 0.25)
    a1, sc1 = _run_lbfgs(best_seq, p1_end, min_time_remaining=200, verbose=True, tag="P1:")
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  P1 result: {best_score:.10f}")

    # === Phase 2: Active-set QP with FISTA ===
    print(f"\n[P2] QP FISTA refinement (t={time.monotonic()-_t0_global:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p2_end = min(deadline - 400, time.monotonic() + 500)
        a2, sc2 = _active_set_qp_refine(best_seq, p2_end, min_time_remaining=60, verbose=True)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(f"  P2 result: {best_score:.10f}")

    # === Phase 3: L-BFGS round 2 (tighter) ===
    print(f"\n[P3] L-BFGS round 2 (t={time.monotonic()-_t0_global:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p3_end = min(deadline - 300, time.monotonic() + 300)
        a3, sc3 = _run_lbfgs(best_seq, p3_end, min_time_remaining=80, verbose=True, tag="P3:")
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3 result: {best_score:.10f}")

    # === Phase 4: Another QP round ===
    print(f"\n[P4] QP round 2 (t={time.monotonic()-_t0_global:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p4_end = min(deadline - 200, time.monotonic() + 200)
        a4, sc4 = _active_set_qp_refine(best_seq, p4_end, min_time_remaining=40, verbose=True)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4 result: {best_score:.10f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n (t={time.monotonic()-_t0_global:.0f}s)")
    n_main = len(best_seq)
    n_options = []
    for n_try in [8192, 12288, 16384, 20480, 24576, 32768]:
        if n_try != n_main:
            n_options.append(n_try)

    for n_try in n_options:
        if time.monotonic() + 60 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if sc_r < best_score:
            best_score = sc_r
            best_seq = a_r.copy()
            print(f"  n={n_try} (resampled): {best_score:.10f}")

        if time.monotonic() + 80 > deadline:
            continue
        p5_end = min(deadline - 40, time.monotonic() + 80)
        a5, sc5 = _run_lbfgs(a_r, p5_end, min_time_remaining=20, verbose=False, tag=f"n={n_try}:")
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  n={n_try} (optimized): {best_score:.10f}")

        # Quick QP on best n
        if time.monotonic() + 40 > deadline:
            continue
        p5b_end = min(deadline - 20, time.monotonic() + 50)
        a5b, sc5b = _active_set_qp_refine(best_seq, p5b_end, min_time_remaining=10, verbose=False)
        if sc5b < best_score:
            best_score = sc5b
            best_seq = a5b.copy()
            print(f"  n={n_try} (QP): {best_score:.10f}")

    # === Phase 6: Symmetry-aware ===
    print(f"\n[P6] Symmetry (t={time.monotonic()-_t0_global:.0f}s)")
    n_cur = len(best_seq)
    if n_cur % 2 == 0 and time.monotonic() + 40 < deadline:
        p6_end = min(deadline - 20, time.monotonic() + 80)
        a6, sc6 = _symmetric_refine_half(best_seq, p6_end, min_time_remaining=5)
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  Symmetric: {best_score:.10f}")

    # === Phase 7: Final polish ===
    print(f"\n[P7] Final polish (t={time.monotonic()-_t0_global:.0f}s)")
    if time.monotonic() + 15 < deadline:
        p7_end = min(deadline - 3, time.monotonic() + 60)
        a7, sc7 = _run_lbfgs(best_seq, p7_end, min_time_remaining=5, verbose=False, tag="P7:")
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()
            print(f"  P7 L-BFGS: {best_score:.10f}")

        if time.monotonic() + 10 < deadline:
            p7b_end = min(deadline - 2, time.monotonic() + 40)
            a7b, sc7b = _active_set_qp_refine(best_seq, p7b_end, min_time_remaining=2, verbose=False)
            if sc7b < best_score:
                best_score = sc7b
                best_seq = a7b.copy()
                print(f"  P7 QP: {best_score:.10f}")

    # Final
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-_t0_global:.1f}s")
    return [float(x) for x in final]
