    while k < n:
        k <<= 1
    return max(2, k)

    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)

        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:

    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines aggressive L-BFGS log-sum-exp annealing, Nesterov-accelerated
active-set subgradient, structural iterated local search with Fourier-domain
perturbations, block-exchange mutations, and multi-scale exploration.
Key innovations: (1) much more aggressive beta annealing schedule,
(2) Fourier-phase randomization for ILS diversity,
(3) adaptive active-set with exponentially decaying weights,
(4) block-exchange structural perturbation,
(5) careful time budgeting with early termination.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_from_conv(a, c):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * float(c.max()) / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def _active_set_subgrad(a, c, tol_frac=0.01, use_inverse_gap=True):
    n = len(a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros(n)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    n_active = int(active.sum())
    if n_active == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        n_active = 1

    if use_inverse_gap:
        eps = 1e-10 * c_max
        w = np.zeros_like(c)
        w[active] = 1.0 / (c_max - c[active] + eps)
    else:
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
    w_sum = w.sum()
    if w_sum < 1e-30:
        w[int(np.argmax(c))] = 1.0
        w_sum = 1.0
    else:
        w = w / w_sum

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad = 2.0 * h
    grad = grad - np.mean(grad)
    return grad


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "raised_cos2":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) ** 2
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(777)
        a = rng.random(n)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "hann_sq":
        t = np.linspace(0, 1, n)
        a = 0.25 * (1.0 + np.cos(2.0 * np.pi * t)) ** 2 * (1.0 + 0.5 * np.cos(4.0 * np.pi * t))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def f(self, z):
        return _smoobj(z, self.beta, self.n)

    def g(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs_anneal(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    k = min(50, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[k]
    if gap < max_c * 0.0005:
        gap = max_c * 0.0005
    base_beta = max(gap, max_c * 0.003)

    # Very aggressive annealing schedule with many steps
    ratios = [1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05, 0.035,
              0.025, 0.018, 0.012, 0.008, 0.005, 0.003, 0.002, 0.001,
              0.0006, 0.0003, 0.00015, 0.00008, 0.00004, 0.00002,
              0.00001, 0.000005, 0.000002, 0.000001]
    betas = base_beta * np.array(ratios)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    prev_sc = current_sc
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.003 * max(1.0, n / 4096.0)
        # More iterations for smaller betas (finer optimization)
        if beta < base_beta * 0.01:
            max_iter = min(8000, max(100, int(remaining / (est_time * 12))))
        elif beta < base_beta * 0.1:
            max_iter = min(5000, max(80, int(remaining / (est_time * 15))))
        else:
            max_iter = min(3000, max(50, int(remaining / (est_time * 20))))

        prob = _LbfgsProblem(beta, n)
        try:
            result = minimize(
                prob.f, z, jac=prob.g, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 80, 'maxls': 40}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-18:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if sc_new < prev_sc - 1e-10:
                    print(f"      L-BFGS beta={beta:.2e}: {sc_new:.12f}")
                prev_sc = sc_new
        except Exception:
            pass

    return current_a


def _nesterov_polish(a_init, deadline, tol_start=0.005, tol_end=0.00001,
                     lr_init=0.03, momentum=0.82):
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    v = np.zeros(n)
    tol_frac = tol_start
    lr = lr_init
    no_improve = 0
    max_stall = 150
    iter_count = 0

    while time.monotonic() < deadline:
        iters_per_batch = 20
        for _ in range(iters_per_batch):
            if time.monotonic() >= deadline:
                break
            c = _full_convolve(a, a)
            c_max = float(c.max())
            current_sc = 2.0 * n * c_max

            if current_sc < best_sc - 1e-15:
                best_sc = current_sc
                best_a = a.copy()
                no_improve = 0

            grad = _active_set_subgrad(a, c, tol_frac)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-18:
                return best_a, best_sc

            v = momentum * v - lr * grad
            alpha = 1.0
            improved = False
            for _ in range(40):
                a_try = _normalize(a + alpha * v)
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-15:
                    a = a_try
                    improved = True
                    break
                alpha *= 0.5

            if not improved:
                v = np.zeros(n)
                no_improve += 1
                if no_improve % 30 == 0:
                    tol_frac = max(tol_end, tol_frac * 0.7)
                    lr = max(1e-6, lr * 0.7)
                if no_improve >= max_stall:
                    a = best_a.copy()
                    v = np.zeros(n)
                    tol_frac = max(tol_end, tol_frac * 0.4)
                    lr = max(1e-6, lr_init * 0.3)
                    no_improve = 0
            else:
                if tol_frac > tol_end:
                    tol_frac = max(tol_end, tol_frac * 0.98)
                lr = min(lr_init * 2, lr * 1.01)

            iter_count += 1

    return best_a, _score(best_a)


def _fourier_perturb(a, rng, strength=0.3):
    """Perturb in Fourier domain: randomize phase partially."""
    n = len(a)
    N = _next_pow2(n)
    A = np.fft.rfft(a, N)
    magnitude = np.abs(A)
    phase = np.angle(A)
    # Randomize phase with probability proportional to frequency
    freqs = np.arange(len(A)) / len(A)
    mask = rng.random(len(A)) < strength * freqs[1:] if len(A) > 1 else np.zeros(1, dtype=bool)
    if len(A) > 1:
        new_phase = phase.copy()
        new_phase[mask[1:]] = rng.uniform(-np.pi, np.pi, mask[1:].sum())
        new_phase[mask[1:]] *= strength
        new_phase[1:] += strength * rng.standard_normal(len(A) - 1) * freqs[1:]
        A_new = magnitude * np.exp(1j * new_phase)
    else:
        A_new = A
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return _normalize(a)
    return a_new / s


def _block_exchange(a, rng, n_blocks=2, min_block=10):
    """Exchange two random blocks in the sequence."""
    n = len(a)
    result = a.copy()
    for _ in range(n_blocks):
        b1 = max(min_block, n // (n_blocks * 3))
        b2 = max(min_block, n // (n_blocks * 3))
        s1 = int(rng.integers(0, n - b1))
        s2 = int(rng.integers(0, n - b2))
        if abs(s1 - s2) < max(b1, b2):
            continue
        seg1 = result[s1:s1 + b1].copy()
        seg2 = result[s2:s2 + b2].copy()
        # Pad to same length for exchange
        if b1 != b2:
            max_b = max(b1, b2)
            seg1_p = np.zeros(max_b)
            seg2_p = np.zeros(max_b)
            seg1_p[:b1] = seg1
            seg2_p[:b2] = seg2
            result[s1:s1 + b1] = seg2_p[:b1]
            result[s2:s2 + b2] = seg1_p[:b2]
        else:
            result[s1:s1 + b1] = seg2
            result[s2:s2 + b2] = seg1
    return _normalize(result)


def _sparse_reweight(a, rng, n_perturb=None, strength=0.1):
    """Perturb a random subset of entries multiplicatively."""
    n = len(a)
    if n_perturb is None:

    idx = rng.choice(n, size=n_perturb, replace=False)
    factors = np.exp(strength * rng.standard_normal(n_perturb))
    a_new = a.copy()
    a_new[idx] *= factors
    return _normalize(a_new)


def _global_noise(a, rng, sigma):

    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _lp_direction_find(a, M, top_k=100):
    """
    Find a descent direction using a simplified approach inspired by the LP.
    We solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for top-k active k.
    Approximated via iterative water-filling.
    """
    n = len(a)
    c_self = _full_convolve(a, a)
    # Get top-k peak positions
    all_idx = np.argsort(c_self)[::-1]
    active_k = all_idx[:top_k]

    # Initialize b proportional to a
    b = a.copy().astype(float)

    # Iterative proportional fitting:
    # At each step, for each active constraint k, if conv(a,b)[k] > M,
    # reduce b proportionally.
    for iteration in range(50):
        c_ab = _full_convolve(a, b)
        violations = c_ab[active_k] - M
        if violations.max() < 1e-12:
            break
        # Find violated constraints
        violated = violations > 0
        if not violated.any():
            break
        # Compute gradient of violations with respect to b
        # d(conv(a,b)[k])/db[j] = a[k-j] (valid range)
        # Sum over violated constraints
        vig = active_k[violated]
        w = np.zeros(len(c_ab))
        for k in vig:
            w[k] = violations[k]  # weight by violation amount
        # grad_b[i] = sum_k w[k] * a[k-i]
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad_b = wc[n - 1:2 * n - 1]
        # Reduce b where gradient is high
        b = b - 0.1 * grad_b
        b = np.maximum(b, 1e-15)

    b = _normalize(b)
    return b

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 24576
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    n_main = n_parent if n_parent > 0 else 24576
    print(f"Starting optimization with n={n_main}, budget={budget_s}s")

    # === Phase 1: Aggressive L-BFGS annealing on parent (400s) ===
    print(f"\n[P1] L-BFGS annealing (t={time.monotonic() - t_start:.1f}s)")
    p1_deadline = t_start + 400
    a_p1 = _run_lbfgs_anneal(best_seq, p1_deadline, min_time_remaining=30)
    sc_p1 = _score(a_p1)
    print(f"  P1 result: {sc_p1:.12f}")
    if sc_p1 < best_score:
        best_seq = a_p1.copy()
        best_score = sc_p1
        print(f"  NEW BEST: {best_score:.12f} [P1] t={time.monotonic() - t_start:.1f}s")

    # === Phase 2: Nesterov polish (200s) ===
    print(f"\n[P2] Nesterov polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 800:
        p2_deadline = min(deadline - 800, time.monotonic() + 200)
        a_p2, sc_p2 = _nesterov_polish(best_seq, p2_deadline,
                                        tol_start=0.003, tol_end=0.00001,
                                        lr_init=0.02, momentum=0.82)
        print(f"  P2 result: {sc_p2:.12f}")
        if sc_p2 < best_score:
            best_seq = a_p2.copy()
            best_score = sc_p2
            print(f"  NEW BEST: {best_score:.12f} [P2] t={time.monotonic() - t_start:.1f}s")

    # === Phase 3: Structural ILS (400s) ===
    print(f"\n[P3] Structural ILS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        p3_deadline = min(deadline - 400, time.monotonic() + 400)
        a_p3, sc_p3 = _structural_ils(best_seq, p3_deadline, rng, min_time_remaining=60)
        print(f"  P3 result: {sc_p3:.12f}")
        if sc_p3 < best_score:
            best_seq = a_p3.copy()
            best_score = sc_p3
            print(f"  NEW BEST: {best_score:.12f} [P3] t={time.monotonic() - t_start:.1f}s")

    # === Phase 4: L-BFGS on ILS result (300s) ===
    print(f"\n[P4] L-BFGS on best (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        p4_deadline = min(deadline - 250, time.monotonic() + 300)
        a_p4 = _run_lbfgs_anneal(best_seq, p4_deadline, min_time_remaining=30)
        sc_p4 = _score(a_p4)
        print(f"  P4 result: {sc_p4:.12f}")
        if sc_p4 < best_score:
            best_seq = a_p4.copy()
            best_score = sc_p4
            print(f"  NEW BEST: {best_score:.12f} [P4] t={time.monotonic() - t_start:.1f}s")

    # === Phase 5: Multi-n exploration (200s) ===
    print(f"\n[P5] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = []
    base = n_main
    for mult in [1.5, 2.0, 0.75]:
        nn = int(base * mult)
        nn = (nn // 128) * 128  # round to multiple of 128
        if nn > 512 and nn not in [len(best_seq)] and nn not in n_candidates:
            n_candidates.append(nn)

    for n_try in n_candidates[:3]:
        if time.monotonic() > deadline - 150:
            break
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init={sc_init:.10f}")
        opt_dl = min(deadline - 100, time.monotonic() + max(20, (deadline - 100 - time.monotonic()) // max(1, len(n_candidates))))
        a_opt = _quick_optimize(a_init, opt_dl, min_time_remaining=20)
        sc_opt = _score(a_opt)
        print(f"  n={n_try}: opt={sc_opt:.10f}")
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.12f} (n={n_try}) [P5] t={time.monotonic() - t_start:.1f}s")

    # === Phase 6: Second ILS round on best (150s) ===
    print(f"\n[P6] ILS round 2 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        p6_deadline = min(deadline - 100, time.monotonic() + 150)
        rng2 = np.random.default_rng(int(seed) + 12345)
        a_p6, sc_p6 = _structural_ils(best_seq, p6_deadline, rng2, min_time_remaining=30)
        print(f"  P6 result: {sc_p6:.12f}")
        if sc_p6 < best_score:
            best_seq = a_p6.copy()
            best_score = sc_p6
            print(f"  NEW BEST: {best_score:.12f} [P6] t={time.monotonic() - t_start:.1f}s")

    # === Phase 7: Final ultra-fine Nesterov polish (80s) ===
    print(f"\n[P7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 10:
        p7_deadline = deadline - 5
        a_p7, sc_p7 = _nesterov_polish(best_seq, p7_deadline,
                                        tol_start=0.001, tol_end=0.000001,
                                        lr_init=0.005, momentum=0.78)
        print(f"  P7 result: {sc_p7:.12f}")
        if sc_p7 < best_score:
            best_seq = a_p7.copy()
            best_score = sc_p7
            print(f"  NEW BEST: {best_score:.12f} [P7] t={time.monotonic() - t_start:.1f}s")

        # Final L-BFGS at very low temperature
        remaining = deadline - 5 - time.monotonic()
        if remaining > 8 and best_seq is not None:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.000005
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _LbfgsProblem(beta_fin, n)
            max_iter = min(3000, max(30, int(remaining / 0.05)))
            try:
                result = minimize(prob.f, z, jac=prob.g, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-20,
                                           'gtol': 1e-16, 'maxcor': 60})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.12f} [final_lbfgs] t={time.monotonic() - t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
