Multi-scale soft-max L-BFGS optimizer with adaptive active-set refinement
for minimizing 2*n*max(conv(a,a))/sum(a)^2.

Key improvements over previous:
- Two-scale approach: coarse (n/4) optimization followed by fine (n) refinement
- More aggressive beta schedule with 30+ levels down to 1e-12
- Dynamic active-set with percentile-based tolerance
- Peak-directed perturbation when stuck
- Better time management with adaptive iteration limits
- Secondary n=24576 exploration

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_structured(a, w):
    """Gradient of sum(w[k] * c[k]) w.r.t. a, where c = conv(a,a).
    
    g[i] = 2 * sum_k w[k] * a[k-i] (with a[m]=0 for m outside [0,n-1])
    Computed as: g = 2 * (a_rev * w)[n-1 : 2n-1]
    """
    n = len(a)
    a_rev = a[::-1]
    aw = _full_convolve(a_rev, w)
    # aw has length n + len(w) - 1 = n + (2n-1) - 1 = 3n-2
    # We need indices n-1 to 2n-2 (total n values)
    grad = 2.0 * aw[n - 1:2 * n - 1]
    return grad


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    grad_a = _grad_structured(a, w)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)


def _compute_grad_from_active(a, active_indices, weights=None):
    """Compute gradient using explicit active set."""
    n = len(a)
    n_c = 2 * n - 1
    w = np.zeros(n_c)
    if weights is None:
        w[active_indices] = 1.0 / len(active_indices)
    else:
        w[active_indices] = weights
    return _grad_structured(a, w)

                z = z - z.mean()
                if verbose:
                    elapsed = time.monotonic() - t_start
                    print(f"    beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}, t={elapsed:.0f}s")
            elif verbose and i < 15:
                print(f"    beta={beta:.2e}: no improvement (best={best_sc:.10f})")
        except Exception as e:
            if verbose:
                print(f"    beta={beta:.2e}: error {e}")

    return best_a, best_sc


def _peak_shaving(a_init, deadline, min_time_remaining=20.0, verbose=False):
    """Directly target the top peaks to reduce them."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score_normalized(a)

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()

        # Find top peaks
        top_k = min(20, len(c))
        top_indices = np.argsort(c)[::-1][:top_k]
        c_top = c[top_indices]
        
        # For the top peak, compute its "responsibility" distribution
        # c[k*] = sum_i a[i]*a[k*-i]
        # The "responsibility" of a[i] for peak k* is a[i]*a[k*-i]
        k_star = top_indices[0]
        # a[i] contributes a[i]*a[k_star-i] to c[k_star]
        
        # Strategy: reduce a[i] where it contributes most to the top peaks
        # but not to other peaks (to avoid increasing other peaks)
        
        # Actually, a simpler approach: use gradient from top few peaks
        # with weights proportional to their values
        weights = c_top / c_top.sum()
        w = np.zeros(len(c))
        w[top_indices] = weights
        grad = _grad_structured(a, w)
        
        # Normalize gradient
        lam = float(np.mean(grad))
        residual = grad - lam
        
        # Line search
        improved = False
        for eta in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            a_try = a - eta * residual
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score_normalized(a_try)
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
        
        if not improved:
            # Try reducing the top peak directly
            # The peak at k* is sum_i a[i]*a[k*-i]
            # To reduce it, we can try to "spread" the mass
            # that contributes to this peak
            contribute = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    contribute[i] = a[i] * a[j]
            
            # Reduce a[i] proportionally to its contribution
            total_contrib = contribute.sum()
            if total_contrib > 0:
                for eta in [0.01, 0.005, 0.002, 0.001]:
                    a_try = a.copy()
                    a_try -= eta * contribute / total_contrib * a.max()
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    sc_try = _score_normalized(a_try)
                    if sc_try < current_sc - 1e-14:
                        a = a_try
                        improved = True
                        break
            
            if not improved:
                break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.8) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "double_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) * (1 + 0.3 * np.cos(2 * np.pi * 3 * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb_restart(a, epsilon=0.01, seed=0):
    """Add small random perturbation to escape local minima."""
    rng = np.random.default_rng(seed)
    n = len(a)
    noise = rng.normal(0, epsilon, n)
    a_new = a + noise * a.max()
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def _coarse_fine_optimize(a_coarse, n_fine, deadline, min_time_remaining=60.0, verbose=True):
    """Use coarse solution to initialize fine optimization."""
    a_fine = _resample(a_coarse, n_fine)
    sc = _score_normalized(a_fine)
    if verbose:
        print(f"  Coarse->Fine: n={n_fine}, score={sc:.10f}")
    return _run_lbfgs_v2(a_fine, deadline, min_time_remaining, verbose)
