            break

        size_factor = max(1.0, n / 4096.0)
        est_time_per_iter = 0.003 * size_factor
        max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 15))))

        # Define objective and gradient
        def obj(z_):
            a_ = _softmax_func(z_)
            c_ = _full_convolve(a_, a_)
            cm = float(c_.max())
            shifted = (c_ - cm) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            smv = cm + beta * np.log(e.sum())
            return 2.0 * n * smv

        def grad(z_):
            a_ = _softmax_func(z_)
            c_ = _full_convolve(a_, a_)
            cm = float(c_.max())
            shifted = (c_ - cm) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            w = e / e.sum()
            a_rev = a_[::-1]
            wc = _full_convolve(w, a_rev)
            grad_a = 2.0 * wc[n - 1:2 * n - 1]
            W = float(np.dot(grad_a, a_))
            return a_ * (grad_a - W)

        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 50}
            )
            a_new = _softmax_func(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    elapsed = time.monotonic() - t_start_global
                    print(f"    LBFGS beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}, t={elapsed:.0f}s")
        except Exception as e:
            if verbose:
                print(f"    LBFGS beta={beta:.2e}: error {e}")

    return best_a, best_sc


def _softmax_func(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _momentum_active_set(a_init, deadline, min_time_remaining=30.0):
    """Active-set method with momentum for faster convergence."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    v = np.zeros(n)
    mu = 0.85
    tol_frac = 0.03
    min_tol_frac = 0.001
    no_improve = 0
    lr = 0.5

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            if tol_frac > min_tol_frac * 1.5:
                tol_frac = max(min_tol_frac, tol_frac * 0.5)
                no_improve = 0
            else:
                break

        grad, _ = _grad_active_set(c, a, tol_frac)
        g = grad - np.mean(grad)

        # Momentum update
        v = mu * v + (1 - mu) * g
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        sc_new = 2.0 * n * float(_full_convolve(a_new, a_new).max())
        if sc_new < current_sc - 1e-14:
            a = a_new
            if no_improve == 0:
                lr = min(lr * 1.05, 5.0)
                tol_frac = max(min_tol_frac, tol_frac * 0.99)
        else:
            # Backtrack
            for trial_lr in [lr * 0.5, lr * 0.25, lr * 0.1]:
                a_try = a - trial_lr * v
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = 2.0 * n * float(_full_convolve(a_try, a_try).max())
                if sc_try < current_sc - 1e-14:
                    a = a_try
                    lr = trial_lr
                    break
            else:
                lr = max(lr * 0.7, 1e-6)

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, eps=0.01, seed=0):
    """Add small random perturbation and renormalize."""
    rng = np.random.default_rng(seed)
    a_p = a + eps * (rng.random(len(a)) - 0.5)
    a_p = np.maximum(a_p, 0.0)
    s = a_p.sum()
    if s > 1e-15:
        a_p = a_p / s
    return a_p


t_start_global = 0.0


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    P1: Global gradient descent (fast, 20%)
    P2: Active-set descent with momentum (refinement, 30%)
    P3: L-BFGS with soft-max (smooth optimization, 20%)
    P4: Active-set again (tighter, 15%)
    P5: Multi-n exploration (10%)
    P6: Final polish (5%)
    """
    global t_start_global
    t_start_global = time.monotonic()
    deadline = t_start_global + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(16384, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n=16384: score={best_score:.10f}")

    n_main = len(best_seq)
    total_time = float(budget_s) - 10.0

    # === Phase 1: Global gradient descent ===
    print(f"\n[P1] Global GD (n={n_main}, budget={0.20*total_time:.0f}s)")
    p1_end = t_start_global + 0.20 * total_time
    a1, sc1 = _global_gradient_descent(best_seq, p1_end, min_time_remaining=30)
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  P1 result: {best_score:.10f}")

    # === Phase 2: Active-set with momentum ===
    print(f"\n[P2] Active-set + momentum (t={time.monotonic()-t_start_global:.0f}s)")
    p2_end = t_start_global + 0.50 * total_time
    a2, sc2 = _momentum_active_set(best_seq, p2_end, min_time_remaining=60)
    if sc2 < best_score:
        best_score = sc2
        best_seq = a2.copy()
        print(f"  P2 result: {best_score:.10f}")

    # === Phase 3: L-BFGS with soft-max ===
    print(f"\n[P3] L-BFGS soft-max (t={time.monotonic()-t_start_global:.0f}s)")
    p3_end = t_start_global + 0.70 * total_time
    a3, sc3 = _lbfgs_with_softmax(best_seq, p3_end, min_time_remaining=60, verbose=True)
    if sc3 < best_score:
        best_score = sc3
        best_seq = a3.copy()
        print(f"  P3 result: {best_score:.10f}")

    # === Phase 4: Active-set again (tighter) ===
    print(f"\n[P4] Active-set refine (t={time.monotonic()-t_start_global:.0f}s)")
    p4_end = t_start_global + 0.85 * total_time
    a4, sc4 = _active_set_descent(best_seq, p4_end, min_time_remaining=30, verbose=True)
    if sc4 < best_score:
        best_score = sc4
        best_seq = a4.copy()
        print(f"  P4 result: {best_score:.10f}")

    # Quick L-BFGS after active-set
    if time.monotonic() + 20 < deadline:
        p4b_end = min(deadline - 20, time.monotonic() + 0.05 * total_time)
        a4b, sc4b = _lbfgs_with_softmax(best_seq, p4b_end, min_time_remaining=10, verbose=False)
        if sc4b < best_score:
            best_score = sc4b
            best_seq = a4b.copy()
            print(f"  P4b result: {best_score:.10f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n (t={time.monotonic()-t_start_global:.0f}s)")
    n_options = [n_main // 2, n_main, n_main * 3 // 2, n_main * 2]
    n_options = sorted(set([x for x in n_options if 2048 <= x <= 65536]))
    for n_try in n_options:
        if time.monotonic() + 60 > deadline:
            break
        if n_try == n_main:
            continue
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if time.monotonic() + 40 > deadline:
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  n={n_try} (resample only): {best_score:.10f}")
            continue

        # Quick optimization at new n
        p5_end = min(deadline - 30, time.monotonic() + 0.04 * total_time)
        a5, sc5 = _active_set_descent(a_r, p5_end, min_time_remaining=10)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  n={n_try}: {best_score:.10f}")

    # === Phase 6: Perturbation + final polish ===
    print(f"\n[P6] Final polish (t={time.monotonic()-t_start_global:.0f}s)")
    # Try a few perturbations
    for pert_seed in [1, 7, 42]:
        if time.monotonic() + 30 > deadline:
            break
        a_p = _perturb(best_seq, eps=0.005, seed=pert_seed)
        p6_end = min(deadline - 5, time.monotonic() + 0.03 * total_time)
        a_p_opt, sc_p_opt = _active_set_descent(a_p, p6_end, min_time_remaining=5)
        if sc_p_opt < best_score:
            best_score = sc_p_opt
            best_seq = a_p_opt.copy()
            print(f"  pert_{pert_seed}: {best_score:.10f}")

    # Final L-BFGS
    if time.monotonic() + 10 < deadline:
        p7_end = min(deadline - 2, time.monotonic() + 30)
        a7, sc7 = _lbfgs_with_softmax(best_seq, p7_end, min_time_remaining=2, verbose=False)
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()
            print(f"  Final LBFGS: {best_score:.10f}")

    # Final active set
    if time.monotonic() + 5 < deadline:
        p7b_end = min(deadline - 1, time.monotonic() + 15)
        a7b, sc7b = _active_set_descent(best_seq, p7b_end, min_time_remaining=1)
        if sc7b < best_score:
            best_score = sc7b
            best_seq = a7b.copy()
            print(f"  Final AS: {best_score:.10f}")

    # Final
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start_global:.1f}s")
    return [float(x) for x in final]
