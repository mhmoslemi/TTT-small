```python
"""
Multi-phase optimizer for minimum peak convolution sequences.
Uses adaptive log-sum-exp smoothing with L-BFGS, direct peak-shaving steps,
projected gradient with momentum, coordinate descent polish, and multi-n exploration.
Key improvements over prior: sharper beta schedule, peak shaving, coordinate descent,
and more aggressive convergence criteria.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_len = n + m - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_len]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _grad_conv(a, w, n):
    """Gradient of sum_k w[k]*c[k] where c = a*a (linear convolution).
    grad[j] = 2 * convolve(w, a_rev)[n-1+j]
    """
    a_rev = a[::-1]
    conv = _full_convolve(w, a_rev)
    return 2.0 * conv[n - 1:2 * n - 1]


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * t
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    elif kind == "u_shape":
        t = np.linspace(0, 1, n)
        a = 0.2 + 0.8 * (2 * t - 1) ** 2
        return a / a.sum()
    elif kind == "w_shape":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.abs(np.sin(2 * np.pi * t))
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, max_beta_scale=1.0, min_beta_scale=1e-10):
    """L-BFGS with adaptive log-sum-exp smoothing and continuation on beta."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap * 0.15, max_c * 0.002)

    # Adaptive beta schedule: geometric decay with adaptive rate
    ratios = [1.0, 0.6, 0.36, 0.216, 0.13, 0.078, 0.047, 0.028, 0.017,
              0.01, 0.006, 0.0036, 0.0022, 0.0013, 0.0008, 0.0005,
              0.0003, 0.00018, 0.0001, 0.00006, 0.000036, 0.00002,
              0.000012, 0.000007, 0.000004, 0.000002, 0.000001,
              0.0000006, 0.0000003, 0.0000001]

    for ratio in ratios:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        beta = base_beta * ratio * max_beta_scale
        if beta < base_beta * min_beta_scale:
            beta = base_beta * min_beta_scale

        est_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 8))))

        try:
            def obj_func(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    return 2.0 * n * c_max_local
                return 2.0 * n * (c_max_local + beta * np.log(s_e))

            def grad_func(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    w = np.zeros(2 * n - 1)
                    w[int(np.argmax(c_local))] = 1.0
                else:
                    w = e / s_e
                grad_c = _grad_conv(a_local, w, n)
                # Chain rule: d/dz [f(softmax(z))] = diag(softmax(z)) * (grad - a * sum(grad))
                h = a_local * (grad_c - float(np.dot(a_local, grad_c)))
                return 4.0 * n * h

            result = minimize(
                obj_func, z,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return best_a


def _run_peak_shave(a_init, deadline, min_time_remaining=30.0, max_steps=500):
    """Direct peak shaving: target the lag with maximum convolution value."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for step in range(max_steps):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        peak_lags = np.where(c >= c_max * 0.9995)[0]
        if len(peak_lags) == 0:
            peak_lags = np.array([int(np.argmax(c))])

        # Average gradient over peak lags
        w = np.zeros(2 * n - 1)
        w[peak_lags] = 1.0 / len(peak_lags)
        grad = _grad_conv(a, w, n)

        # Line search: find best step size
        best_step = None
        best_c_max = c_max
        for ls in range(25):
            step_size = 0.1 / (np.linalg.norm(grad) + 1e-15) * (0.7 ** ls)
            a_try = _proj_simplex(a - step_size * grad)
            c_try = _full_convolve(a_try, a_try)
            cm_try = float(c_try.max())
            if cm_try < best_c_max:
                best_c_max = cm_try
                best_step = (a_try, cm_try)
            if step_size < 1e-15:
                break

        if best_step is not None:
            a, c_new_max = best_step
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
        else:
            # No improvement, try smaller peak threshold
            peak_lags = np.where(c >= c_max * 0.99)[0]
            if len(peak_lags) == 0:
                break

    return best_a, best_sc


def _run_proj_grad_momentum(a_init, deadline, min_time_remaining=30.0, max_steps=3000):
    """Projected gradient with Nesterov momentum and adaptive step size."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    momentum = np.zeros(n)
    beta_m = 0.85
    patience = 0
    max_patience = 80
    top_k = 8

    for step in range(max_steps):
        if time.monotonic() + min_time_remaining > deadline:
            break
        a_curr = _softmax(z)
        c = _full_convolve(a_curr, a_curr)
        c_max = float(c.max())

        # Adaptive top-k
        if step % 50 == 0:
            gap = c_max - c_sorted_mid
            c_sorted_mid = float(np.sort(c)[::-1][min(50, len(c)-1)]) if len(c) > 50 else 0.0
            if gap < c_max * 0.001:
                top_k = min(20, len(c))
            elif gap < c_max * 0.01:
                top_k = min(10, len(c))
            else:
                top_k = 5

        active = np.argsort(c)[::-1][:top_k]
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / top_k

        grad = _grad_conv(a_curr, w, n)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        momentum = beta_m * momentum + (1.0 - beta_m) * grad
        lr = 0.01 * (1.0 + 0.05 * np.log1p(step)) / (np.linalg.norm(momentum) / grad_norm + 1e-10)
        z_new = z - lr * momentum
        z_new = np.clip(z_new, -40.0, 40.0)
        z_new = z_new - z_new.mean()
        z = z_new

        if step % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros(n)
                    patience = 0
                    beta_m *= 0.95

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _coordinate_descent(a_init, deadline, min_time_remaining=10.0, max_sweeps=200):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for sweep in range(max_sweeps):
        if time.monotonic() + min_time_remaining > deadline:
            break
        improved = False
        # Random order
        order = np.random.permutation(n)
        for idx in order:
            if time.monotonic() + min_time_remaining > deadline:
                break
            old_val = a[idx]
            if old_val < 1e-15:
                continue
            # Try a few candidate values
            c_old = _full_convolve(a, a)
            c_max_old = float(c_old.max())

            # The score depends on max(c) and sum(a)
            # Changing a[idx] by delta changes c[k] for all k
            # and changes sum by delta
            # Optimal delta: minimize 2n*max(c_new)/(sum+delta)^2
            # Approximate: try delta that reduces the peak
            for factor in [0.8, 0.6, 0.4, 0.2, 1.2, 1.5, 2.0]:
                a_try = a.copy()
                a_try[idx] = old_val * factor
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                c_try = _full_convolve(a_try, a_try)
                cm_try = float(c_try.max())
                sc_try = 2.0 * n * cm_try / (s_try * s_try)
                if sc_try < 2.0 * n * c_max_old / (a.sum() ** 2) * (1 - 1e-10):
                    a = a_try
                    improved = True
                    break

        # Check if overall improved
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
        elif not improved:
            break

    return best_a, best_sc


def _run_continuation(a_init, deadline, min_time_remaining=30.0):
    """Two-phase continuation: coarse then fine L-BFGS."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    a = a / a.sum()

    # Phase A: Coarse (large beta)
    alpha = time.monotonic()
    remaining = deadline - min_time_remaining - alpha
    if remaining < 10:
        return a
    mid_deadline = alpha + max(remaining * 0.4, 10)
    a_coarse = _run_lbfgs(a, mid_deadline, min_time_remaining=10.0,
                          max_beta_scale=1.0, min_beta_scale=1e-4)

    # Phase B: Fine (small beta)
    alpha = time.monotonic()
    remaining = deadline - min_time_remaining - alpha
    if remaining < 10:
        return a_coarse
    a_fine = _run_lbfgs(a_coarse, deadline, min_time_remaining=min_time_remaining,
                        max_beta_scale=1e-4, min_beta_scale=1e-12)
    return a_fine


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 32768
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Multi-start L-BFGS (350s) ===
    print(f"\n[Phase 1] Multi-start L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    phase1_deadline = t_start + 350
    candidates = []

    # Start from parent with continuation
    a1 = _run_continuation(parent, phase1_deadline, min_time_remaining=30.0)
    update_best(a1, _score(a1), "lbfgs_parent")
    candidates.append(a1)

    # Try other profiles
    profiles = ["cos", "u_shape", "w_shape", "bump"]
    for prof in profiles:
        if time.monotonic() > phase1_deadline - 40:
            break
        a_init = _make_profile(n_parent, prof)
        a_p = _run_lbfgs(a_init, phase1_deadline, min_time_remaining=20.0)
        sc_p = _score(a_p)
        if sc_p < best_score + 0.02:
            update_best(a_p, sc_p, f"lbfgs_{prof}")
            candidates.append(a_p)

    # Perturbed parent
    if time.monotonic() < phase1_deadline - 30:
        z_p = np.log(np.maximum(parent, 1e-30))
        z_p = z_p + 0.05 * rng.standard_normal(n_parent)
        z_p = z_p - z_p.mean()
        a_pert = _softmax(z_p)
        a_p = _run_lbfgs(a_pert, phase1_deadline, min_time_remaining=15.0)
        update_best(a_p, _score(a_p), "lbfgs_perturb")

    # === Phase 2: Peak shaving (250s) ===
    print(f"\n[Phase 2] Peak Shaving (t={time.monotonic() - t_start:.1f}s)")
    phase2_deadline = min(deadline - 800, phase1_deadline + 250)
    if best_seq is not None and time.monotonic() < phase2_deadline:
        a_ps, sc_ps = _run_peak_shave(best_seq, phase2_deadline, min_time_remaining=30.0,
                                       max_steps=2000)
        update_best(a_ps, sc_ps, "peak_shave_1")
        # Follow up with L-BFGS
        if time.monotonic() < phase2_deadline - 30:
            a_follow = _run_lbfgs(a_ps, phase2_deadline, min_time_remaining=20.0,
                                  max_beta_scale=1e-3, min_beta_scale=1e-10)
            update_best(a_follow, _score(a_follow), "lbfgs_after_ps")

    # === Phase 3: Projected gradient with momentum (250s) ===
    print(f"\n[Phase 3] Proj Grad Momentum (t={time.monotonic() - t_start:.1f}s)")
    phase3_deadline = min(deadline - 550, phase2_deadline + 250)
    if best_seq is not None and time.monotonic() < phase3_deadline:
        a_pg, sc_pg = _run_proj_grad_momentum(best_seq, phase3_deadline,
                                               min_time_remaining=30.0, max_steps=5000)
        update_best(a_pg, sc_pg, "proj_grad_1")

    # === Phase 4: Combined refinement round 2 (300s) ===
    print(f"\n[Phase 4] Refinement R2 (t={time.monotonic() - t_start:.1f}s)")
    phase4_deadline = min(deadline - 250, phase3_deadline + 300)
    if best_seq is not None and time.monotonic() < phase4_deadline:
        # L-BFGS with very tight beta
        a_lb2 = _run_lbfgs(best_seq, phase4_deadline, min_time_remaining=30.0,
                           max_beta_scale=1e-5, min_beta_scale=1e-12)
        update_best(a_lb2, _score(a_lb2), "lbfgs_r2")

        # Peak shave again
        if time.monotonic() < phase4_deadline - 30:
            a_ps2, sc_ps2 = _run_peak_shave(best_seq, phase4_deadline,
                                             min_time_remaining=20.0, max_steps=1000)
            update_best(a_ps2, sc_ps2, "peak_shave_2")

        # Proj grad again
        if time.monotonic() < phase4_deadline - 20:
            a_pg2, sc_pg2 = _run_proj_grad_momentum(best_seq, phase4_deadline,
                                                     min_time_remaining=15.0, max_steps=3000)
            update_best(a_pg2, sc_pg2, "proj_grad_2")

    # === Phase 5: Multi-n exploration (200s) ===
    print(f"\n[Phase 5] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    phase5_deadline = min(deadline - 130, phase4_deadline + 200)
    n_tries = [49152, 65536, 24576, 16384]
    for n_try in n_tries:
        if time.monotonic() > phase5_deadline - 40:
            break
        if n_try == n_parent:
            continue
        a_init_n = _resample(best_seq, n_try) if best_seq is not None else _make_profile(n_try, "tapered")
        a_n = _run_lbfgs(a_init_n, phase5_deadline, min_time_remaining=20.0)
        sc_n = _score(a_n)
        update_best(a_n, sc_n, f"n_{n_try}")

        if sc_n < best_score + 0.003 and time.monotonic() < phase5_deadline - 30:
            a_n2 = _run_lbfgs(a_n, phase5_deadline, min_time_remaining=15.0,
                             max_beta_scale=1e-4, min_beta_scale=1e-10)
            update_best(a_n2, _score(a_n2), f"n_{n_try}_r2")

    # === Phase 6: Coordinate descent polish (80s) ===
    print(f"\n[Phase 6] Coord Descent (t={time.monotonic() - t_start:.1f}s)")
    phase6_deadline = min(deadline - 50, phase5_deadline + 80)
    if best_seq is not None and time.monotonic() < phase6_deadline:
        # Subsample for coordinate descent (too slow on full sequence)
        n_cd = min(4096, len(best_seq))
        a_cd_init = _resample(best_seq, n_cd)
        a_cd, sc_cd = _coordinate_descent(a_cd_init, phase6_deadline, min_time_remaining=10.0,
                                           max_sweeps=50)
        # Resample back
        a_cd_full = _resample(a_cd, len(best_seq))
        a_cd_full = a_cd_full / a_cd_full.sum()
        # L-BFGS polish on full size
        a_cd_polish = _run_lbfgs(a_cd_full, phase6_deadline, min_time_remaining=5.0,
                                  max_beta_scale=1e-6, min_beta_scale=1e-12)
        update_best(a_cd_polish, _score(a_cd_polish), "cd_polish")

    # === Phase 7: Final ultra-tight polish (50s) ===
    print(f"\n[Phase 7] Final Polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 15:
        # One more L-BFGS with extremely tight beta
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        c_sorted = np.sort(c_b)[::-1]
        gap = c_sorted[0] - c_sorted[min(5, len(c_sorted)-1)]
        beta_fin = max(gap * 0.05, mb * 1e-10)

        remaining = deadline - 15 - time.monotonic()
        if remaining > 5:
            max_iter = min(3000, max(100, int(remaining / 0.005)))

            def final_obj(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta_fin
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    return 2.0 * n * c_max_local
                return 2.0 * n * (c_max_local + beta_fin * np.log(s_e))

            def final_grad(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta_fin
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    w = np.zeros(2 * n - 1)
                    w[int(np.argmax(c_local))] = 1.0
                else:
                    w = e / s_e
                grad_c = _grad_conv(a_local, w, n)
                h = a_local * (grad_c - float(np.dot(a_local, grad_c)))
                return 4.0 * n * h

            try:
                result = minimize(
                    final_obj, z,
                    jac=final_grad,
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                             'maxcor': 50, 'maxls': 100}
                )
                a_flb = _softmax(result.x)
                update_best(a_flb, _score(a_flb), "final_lbfgs")
            except Exception:
                pass

        # Final peak shave
        if time.monotonic() < deadline - 10:
            a_final_ps, sc_final_ps = _run_peak_shave(best_seq, deadline - 5,
                                                       min_time_remaining=5.0, max_steps=200)
            update_best(a_final_ps, sc_final_ps, "final_peak_shave")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
