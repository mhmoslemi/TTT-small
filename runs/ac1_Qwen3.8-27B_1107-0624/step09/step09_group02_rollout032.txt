```python
    n = len(m_first_col)
    N = 2 * n - 1
    # For symmetric Toeplitz, the first column is [m_0, m_1, ..., m_{n-1}]
```

```python
def solve_toeflitz(m, rhs, max_iter=500, tol=1e-10):
    """Solve T*x = rhs where T is symmetric Toeplitz with first column m."""
    n = len(rhs)
    x = np.zeros(n)

    p = r.copy()
    rs = r @ r

        pAp = p @ Ap

        rs_new = r @ r
        if np.sqrt(rs_new) < tol:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x
```

```python
"""
KKT-based Toeplitz solver for minimum peak autocorrelation sequences.
Uses the KKT conditions of the min-max problem to derive a fixed-point iteration:
  (1) Compute autocorrelation c = conv(a,a)
  (2) Form dual weights w = softmax(c/beta)  
  (3) Build symmetric Toeplitz matrix M from w (M[i,j] = 0.5*w_{|i-j|}, M[i,i] = w_0)
  (4) Solve M*x = 1 via Conjugate Gradient with FFT-based matvec
  (5) Project to non-negative simplex
  (6) Blend and update
Explores beta schedule, line search, re-excitation, and multiple n values.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _toeflitz_matvec(v, m):
    n = len(v)
    if n <= 1:
        return m[0] * v
    k = np.concatenate([m[n - 1:0:-1], m[:1], m[1:]])
    N = _next_pow2(3 * n - 2)
    F_v = np.fft.rfft(v, N)
    F_k = np.fft.rfft(k, N)
    result = np.fft.irfft(F_v * F_k, N)[:2 * n - 1]
    return result[n - 1:2 * n - 1]


def _solve_toeflitz_cg(m, rhs, max_iter=500, tol=1e-8):
    n = len(rhs)
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    if rs < 1e-30:
        return x
    for i in range(max_iter):
        Ap = _toeflitz_matvec(p, m)
        pAp = float(p @ Ap)
        if abs(pAp) < 1e-30:
            break
        if pAp < 0:
            x = rhs / m[0]
            return np.maximum(x, 0)
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = float(r @ r)
        if np.sqrt(rs_new) < tol * (np.linalg.norm(rhs) + 1e-30):
            break
        if not np.isfinite(rs_new):
            break
        ratio = rs_new / rs
        p = r + ratio * p
        rs = rs_new
    return x


def _make_initial(n, kind="cosine"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "cosine":
        a = 0.5 * (1.0 + np.cos(np.pi * t))
    elif kind == "sine":
        a = np.sin(np.pi * t)
    elif kind == "ramp":
        a = 0.3 + 0.7 * t
    elif kind == "ramp2":
        a = 0.1 + 0.9 * t
    elif kind == "bump":
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.1
    elif kind == "gauss":
        a = np.exp(-8.0 * (t - 0.5) ** 2) + 0.05
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _run_kkt(a_init, deadline, min_time_remaining=30.0, beta_start=0.05,
             beta_end=1e-5, blend=0.5, cg_tol=1e-8, cg_max_iter=300,
             line_search=True, label=""):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    total_iters = max(10, int((deadline - min_time_remaining - time.monotonic()) / 0.3))
    total_iters = min(total_iters, 500)

    for it in range(total_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break
        beta = beta_start * (beta_end / beta_start) ** (it / max(1, total_iters - 1))
        c = _full_convolve(a, a)
        c_max = float(c.max())
        w = _softmax(c / max(beta, 1e-12))
        m_col = np.empty(n)
        m_col[0] = w[0]
        m_col[1:] = 0.5 * w[1:n]
        rhs = np.ones(n)
        x = _solve_toeflitz_cg(m_col, rhs, max_iter=cg_max_iter, tol=cg_tol)
        x = np.maximum(x, 0)
        sx = x.sum()
        if sx < 1e-15:
            continue
        x = x / sx
        if line_search:
            best_t = 0.0
            best_c = c_max
            for t_try in [blend, 0.7, 0.5, 0.3, 0.15]:
                a_try = (1.0 - t_try) * a + t_try * x
                c_try = _full_convolve(a_try, a_try)
                c_try_max = float(c_try.max())
                if c_try_max < best_c:
                    best_c = c_try_max
                    best_t = t_try
            t_final = best_t if best_t > 0 else blend
        else:
            t_final = blend
        a = (1.0 - t_final) * a + t_final * x
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            break
        a = a / sa
        if it % 10 == 0 or it == total_iters - 1:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"    KKT[{label}] it={it}/{total_iters} sc={sc:.10f} beta={beta:.2e} t={time.monotonic()-t_start:.0f}s")
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _run_kkt_active(a_init, deadline, min_time_remaining=30.0, label=""):
    """KKT with active-set selection instead of softmax."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    total_iters = max(10, int((deadline - min_time_remaining - time.monotonic()) / 0.2))
    total_iters = min(total_iters, 400)
    tol_frac = 0.1
    for it in range(total_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol_frac = max(0.001, 0.1 * (0.95 ** it))
        active = c >= c_max * (1.0 - tol_frac)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / active.sum()
        m_col = np.empty(n)
        m_col[0] = w[0]
        m_col[1:] = 0.5 * w[1:n]
        rhs = np.ones(n)
        x = _solve_toeflitz_cg(m_col, rhs, max_iter=300, tol=1e-8)
        x = np.maximum(x, 0)
        sx = x.sum()
        if sx < 1e-15:
            continue
        x = x / sx
        best_t = 0.0
        best_c = c_max
        for t_try in [0.5, 0.3, 0.7, 0.2, 0.1]:
            a_try = (1.0 - t_try) * a + t_try * x
            c_try = _full_convolve(a_try, a_try)
            if float(c_try.max()) < best_c:
                best_c = float(c_try.max())
                best_t = t_try
        a = (1.0 - best_t) * a + best_t * x
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            break
        a = a / sa
        if it % 10 == 0 or it == total_iters - 1:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"    KKT-act[{label}] it={it}/{total_iters} sc={sc:.10f} tol={tol_frac:.4f} t={time.monotonic()-t_start:.0f}s")
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _run_lbfgs_polish(a_init, deadline, min_time_remaining=20.0, label=""):
    """L-BFGS polish with log-sum-exp smoothing on the score."""
    from scipy.optimize import minimize
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_sc = _score(a)
    best_a = a.copy()
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    beta_schedule = [max_c * 0.01, max_c * 0.005, max_c * 0.001, max_c * 0.0005,
                     max_c * 0.0001, max_c * 0.00005, max_c * 0.00001]
    for beta in beta_schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        def obj_func(z_local):
            a_local = _softmax(z_local)
            c_local = _full_convolve(a_local, a_local)
            c_max_local = float(c_local.max())
            shifted = (c_local - c_max_local) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s_e = e.sum()
            if s_e < 1e-300:
                return 2.0 * n * c_max_local
            return 2.0 * n * (c_max_local + beta * np.log(s_e))

        def grad_func(z_local):
            a_local = _softmax(z_local)
            c_local = _full_convolve(a_local, a_local)
            c_max_local = float(c_local.max())
            shifted = (c_local - c_max_local) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s_e = e.sum()
            if s_e < 1e-300:
                w = np.zeros_like(c_local)
                w[int(np.argmax(c_local))] = 1.0
            else:
                w = e / s_e
            a_rev = a_local[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W_val = float(np.dot(w, c_local))
            return 4.0 * n * a_local * (h - W_val)

        max_iter = min(1000, max(20, int(remaining / 0.05)))
        try:
            result = minimize(
                obj_func, z,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS[{label}] beta={beta:.2e} sc={best_sc:.10f} t={time.monotonic()-t_start:.0f}s")
        except Exception:
            pass
    return best_a, best_sc

        _update_best(a7, sc7)
        if time.monotonic() + 20 < ph7_deadline:
            a7b, sc7b = _run_lbfgs_polish(best_seq, ph7_deadline, min_time_remaining=10, label="p7_lb")
            _update_best(a7b, sc7b)

    # === Phase 8: Try different n ===
    print(f"\n[Phase 8] Multi-n exploration")
    ph8_deadline = min(deadline - 30, ph7_deadline + 80)
    if best_seq is not None and time.monotonic() < ph8_deadline:
        for n_try in [16384, 65536]:
            if time.monotonic() + 30 > ph8_deadline:
                break
            if n_try == n_parent:
                continue
            a_resamp = _resample(best_seq, n_try)
            a8, sc8 = _run_kkt(a_resamp, ph8_deadline, min_time_remaining=30,
                               beta_start=0.05, beta_end=1e-4, blend=0.5,
                               label=f"p8_n{n_try}")
            _update_best(a8, sc8)

    # === Phase 9: Final ultra-tight KKT ===
    print(f"\n[Phase 9] Final ultra-tight KKT")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a9, sc9 = _run_kkt(best_seq, deadline - 5, min_time_remaining=3,
                           beta_start=0.05, beta_end=1e-7, blend=0.3,
                           label="final")
        _update_best(a9, sc9)

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_initial(4096, "cosine")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_initial(8192, "cosine")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
