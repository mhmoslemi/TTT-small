"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous version:
- Peak-weighted active set equalization (not just top-K uniform weights)
- Direct "peak shaving" via 1D coordinate optimization on affected coordinates
- More aggressive beta cooling schedule with held-out fine-tuning phase
- Simulated-restart escape mechanism to break out of local minima
- Multi-n exploration with adaptive resampling
- Final coordinate-descent polish exploiting the 1D structure at each coordinate
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005) * beta_scale

    # Extended beta schedule with more values and finer cooling
    betas = base_beta * np.array([
        1.0, 0.6, 0.4, 0.25, 0.15,
        0.08, 0.05, 0.03, 0.015, 0.008,
        0.004, 0.002, 0.001, 0.0005,
        0.00025, 0.0001, 0.00005, 0.00002, 0.00001
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Scale iterations based on beta magnitude
        if beta < base_beta * 0.01:
            est_time_per_iter = 0.004 * max(1.0, n / 2048.0)
            max_iter = min(8000, max(100, int(remaining / (est_time_per_iter * 12))))
        elif beta < base_beta * 0.1:
            est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
            max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 14))))
        else:
            est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
            max_iter = min(3000, max(30, int(remaining / (est_time_per_iter * 15))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _peak_weighted_equalize(a_init, deadline, min_time_remaining=30.0):
    """Peak-weighted active set equalization: weights peaks by their excess
    over the median, giving more attention to the worst peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    K_start = 5
    K_max = min(200, n)
    lr = 0.03
    no_improve = 0
    K = K_start
    total_iters = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            if K < K_max:
                K = min(K_max, K + 2)

        # Get top K indices with importance weighting
        top_k_idx = np.argsort(c)[::-1][:K]
        top_k_vals = c[top_k_idx]
        c_median = np.median(c)
        # Weight by excess over median (peaks further above median get more weight)
        excess = np.maximum(top_k_vals - c_median, 1e-15)
        weights = excess / excess.sum()

        # Compute weighted average gradient of top-K peaks
        w_S = np.zeros(2 * n - 1)
        for i, idx in enumerate(top_k_idx):
            w_S[idx] = 2.0 * weights[i]
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)[n - 1:2 * n - 1]
        grad = h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-17:
            break

        # Adaptive line search
        alpha = lr / (gnorm + 1e-15)
        improved = False
        for _ in range(60):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            lr *= 0.88
            if no_improve % 30 == 0:
                K = max(K_start, K // 2)
            if no_improve >= 300:
                a = best_a.copy()
                lr = 0.03
                K = K_start
                no_improve = 0
                break
        else:
            lr = min(0.06, lr * 1.015)

        total_iters += 1

    return best_a, best_sc


def _peak_shaving(a_init, deadline, min_time_remaining=30.0):
    """Direct peak shaving: for each top peak, find the coordinate that most
    contributes to it and reduce that coordinate (redistributing mass)."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    T = len(a)

    no_improve = 0
    peak_count = min(20, n)

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Get top peaks
        top_peaks = np.argsort(c)[::-1][:peak_count]

        # For each peak, find the coordinate that contributes most
        # and try reducing it while redistributing to the least-contributing
        improved = False
        for pk in top_peaks:
            # conv(a,a)[pk] = sum_i a[i]*a[pk-i]
            # The contribution of a[j] is a[j]*a[pk-j] (if valid)
            # Find j that maximizes a[j]*a[pk-j]
            valid_j = np.arange(0, n)
            valid_pk_j = np.clip(pk - valid_j, 0, n - 1)
            # Check which are actually valid (0 <= pk-j < n)
            mask = (pk - valid_j >= 0) & (pk - valid_j < n)
            if not mask.any():
                continue
            contrib = np.zeros(n)
            contrib[valid_j[mask]] = a[valid_j[mask]] * a[pk - valid_j[mask]]
            j_max = int(np.argmax(contrib))
            if contrib[j_max] < 1e-15:
                continue

            # Find the coordinate with minimum contribution to this peak
            # to redistribute mass to
            min_contrib_idx = int(np.argmin(contrib))
            if contrib[min_contrib_idx] >= contrib[j_max] * 0.5:
                continue

            # Try reducing a[j_max] and increasing a[min_contrib_idx]
            delta = a[j_max] * 0.02  # 2% transfer
            for frac in [0.01, 0.02, 0.005, 0.05, 0.015]:
                delta = a[j_max] * frac
                if delta <= 0:
                    continue
                a_try = a.copy()
                a_try[j_max] -= delta
                a_try[min_contrib_idx] += delta
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-15:
                    a = a_try
                    current_sc = sc_try
                    improved = True
                    break

        if not improved:
            no_improve += 1
            if no_improve >= 50:
                break

    return best_a, best_sc


def _coordinate_polish(a_init, deadline, min_time_remaining=15.0):
    """Final coordinate descent polish: for each coordinate, try small
    perturbations and keep the best."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    # Precompute convolution for speed
    c = _full_convolve(a, a)

    step_sizes = np.array([0.001, 0.005, 0.01, 0.02, 0.05])

    while time.monotonic() + min_time_remaining < deadline:
        improved_any = False
        # Process coordinates in random order
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() + min_time_remaining > deadline:
                break
            c = _full_convolve(a, a)
            current_sc = 2.0 * n * float(c.max())

            for ss in step_sizes:
                # Try increasing
                a_try = a.copy()
                a_try[j] += ss
                a_try = a_try / a_try.sum()
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-15:
                    a = a_try
                    current_sc = sc_try
                    improved_any = True
                    break

                # Try decreasing
                if a[j] > ss:
                    a_try = a.copy()
                    a_try[j] -= ss
                    a_try = a_try / a_try.sum()
                    c_try = _full_convolve(a_try, a_try)
                    sc_try = 2.0 * n * float(c_try.max())
                    if sc_try < current_sc - 1e-15:
                        a = a_try
                        current_sc = sc_try
                        improved_any = True
                        break

            if current_sc < best_sc - 1e-14:
                best_sc = current_sc
                best_a = a.copy()

        if not improved_any:
            # Try smaller steps
            step_sizes = step_sizes * 0.5
            if step_sizes[0] < 1e-6:
                break

    return best_a, best_sc


def _asymmetric_perturb(a, rng, strength=0.05):
    """Break symmetry by adding an asymmetric perturbation."""
    n = len(a)
    t = np.arange(n) / n
    ramp = 0.3 * (t - 0.5)
    noise = 0.7 * rng.standard_normal(n)
    perturbation = ramp + noise
    perturbation = perturbation - perturbation.mean()
    z = np.log(np.maximum(a, 1e-30))
    z = z + strength * perturbation
    z = z - z.mean()
    return _softmax(z)


def _structure_perturb(a, rng, strength=0.03):
    """Add a structured perturbation (low-frequency mode)."""
    n = len(a)
    k = rng.integers(1, max(2, n // 100))
    t = np.arange(n)
    mode = np.sin(2.0 * np.pi * k * t / n + rng.random() * 2 * np.pi)
    z = np.log(np.maximum(a, 1e-30))
    z = z + strength * mode
    z = z - z.mean()
    return _softmax(z)


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

        a_coord, sc_coord = _coordinate_polish(a_work, p9_deadline, min_time_remaining=5.0)
        if sc_coord < best_score:
            best_seq = a_coord.copy()
            best_score = sc_coord
            a_work = a_coord.copy()
            print(f"  NEW BEST: {best_score:.10f} [coord_polish]")

    # === Phase 10: Final peak equalization + ultra-fine L-BFGS ===
    print(f"\n[P10] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 5 - time.monotonic()
    if remaining > 20:
        # Quick peak equalization
        if remaining > 40:
            p10a_deadline = min(deadline - 10, time.monotonic() + 20.0)
            a_fin_eq, sc_fin_eq = _peak_weighted_equalize(a_work, p10a_deadline, min_time_remaining=3.0)
            if sc_fin_eq < best_score:
                best_seq = a_fin_eq.copy()
                best_score = sc_fin_eq
                a_work = a_fin_eq.copy()
                print(f"  NEW BEST: {best_score:.10f} [final_equalize]")

        # Ultra-fine L-BFGS
        if time.monotonic() < deadline - 5:
            a_final = _run_lbfgs(a_work, deadline - 3, min_time_remaining=3.0, beta_scale=0.1)
            sc_final = _score(a_final)
            if sc_final < best_score:
                best_seq = a_final.copy()
                best_score = sc_final
                print(f"  NEW BEST: {best_score:.10f} [final_ultra]")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
