"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: diverse asymmetric initializations, aggressive L-BFGS continuation with
log-sum-exp smoothing, peak-shaving heuristics, active-set Newton refinement,
and multi-scale optimization (n=8192, 16384).
Key innovation: explicit asymmetric ramp initializations and a novel
"convolution balancing" step that directly equalizes peak values.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _lsobj_factory(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad_factory(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec, rhs, max_cg=200, tol=1e-13):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _newton_step(a, c, c_max, tol_frac):
    n = len(a)
    active = c >= (c_max - tol_frac * c_max)
    if not active.any():
        active[np.argmax(c)] = True
    m = int(active.sum())
    if m < 2 or m > 10 * n:
        return None, None
    A_vec = active.astype(np.float64) / m
    g = 2.0 * _corr_a_A(a, A_vec, n)
    g_t = g - np.mean(g)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-14:
        return None, None
    mu = 1e-7 / m + 1e-10
    A_vec_copy = A_vec
    n_copy = n
    mu_copy = mu

    def _hess_vp(v):
        return 2.0 * _corr_a_A(v, A_vec_copy, n_copy) + mu_copy * v

    u = _cg_solve(_hess_vp, g_t, max_cg=200)
    w = _cg_solve(_hess_vp, np.ones(n), max_cg=200)
    w_sum = float(w.sum())
    if abs(w_sum) < 1e-15:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w
    dx = dx - np.mean(dx)
    return dx, A_vec


def _active_set_newton(a_init, deadline, max_outer=100):

def _peak_shaving(a, deadline, max_iter=200):
    """
    Direct peak-shaving: iteratively identify the maximum lag and apply
    a targeted direction to reduce it while controlling the increase at other lags.
    """
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Find all near-max positions
        threshold = c_max * 0.995
        peak_mask = c >= threshold
        n_peaks = int(peak_mask.sum())
        if n_peaks == 0:
            peak_mask[np.argmax(c)] = True
            n_peaks = 1

        # Weighted peak indicator
        w = np.zeros_like(c)
        w[peak_mask] = 1.0 / n_peaks

        # Gradient of weighted peak convex value
        grad = 2.0 * _corr_a_A(a, w, n)
        g_t = grad - np.mean(grad)

        # Try multiple step sizes
        improved = False
        for step in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
            a_try = a - step * g_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            # Try randomized perturbation
            rng = np.random.default_rng(it + 12345)
            a_try = a * (1.0 + 0.02 * rng.standard_normal(n))
            a_try = _project_simplex(a_try)
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
            else:
                break

    return best_a, best_sc


def _convolution_balance(a, deadline, max_iter=100):
    """
    Novel approach: try to equalize the top K convolution values.
    Uses a water-filling / successive approximation approach.
    """
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        c_mean = float(c.mean())

        # Identify the "excess" peaks
        excess = c - c_mean
        # Create a correction target: reduce peaks, boost valleys
        # Direction: for each position j in a, the contribution to c[k] is a[j]*a[k-j]
        # We want to reduce a[j]*a[k-j] where c[k] is large
        # Approximate: gradient of max(c) is 2*corr(a, e_k_max)
        # Better: use a soft version

        # Soft weighting based on how much above mean each c[k] is
        alpha = 5.0 / max(c_max - c_mean, 1e-15)
        w = np.exp(alpha * (c - c_max))
        w = w / w.sum()

        grad = 2.0 * _corr_a_A(a, w, n)
        g_t = grad - np.mean(grad)

        improved = False
        for step in [0.0005, 0.001, 0.002, 0.005, 0.01]:
            a_try = a - step * g_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            break

    return best_a, best_sc


def _lbfgs_continuation(a_init, deadline, min_time_remaining=30.0,
                         start_beta_frac=0.01, end_beta_frac=1e-8):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 100:
        gap = c_sorted[0] - c_sorted[100]
    else:
        gap = c_sorted[0] - c_sorted[-1]
    base_beta = max(gap * 0.5, max_c * start_beta_frac)

    n_betas = 30
    log_betas = np.linspace(np.log(base_beta), np.log(max_c * end_beta_frac), n_betas)
    betas = np.exp(log_betas)

    current_a = a_init.copy()
    current_sc = _score(current_a)

    # Initialize z from current_a
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 12))))

        try:
            result = minimize(
                lambda z_val: _lsobj_factory(z_val, beta, n),
                z,
                jac=lambda z_val: _lsgrad_factory(z_val, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 200,
                    'maxls': 500,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # If no improvement, try with current z perturbed
                z = np.log(np.maximum(current_a, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _fista_smoothed(a_init, deadline, n_betas=15, min_time_remaining=30.0):
    """
    FISTA (accelerated proximal gradient) on the smoothed objective.
    Uses the log-sum-exp smoothing and the simplex projection.
    """
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - (c_sorted[min(50, len(c_sorted)-1)] if len(c_sorted) > 50 else c_sorted[-1])
    base_beta = max(gap * 0.3, max_c * 0.005)
    end_beta = max_c * 1e-7

    log_betas = np.linspace(np.log(base_beta), np.log(end_beta), n_betas)
    betas = np.exp(log_betas)

    current_a = a_init.copy()
    current_sc = _score(current_a)

    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # FISTA on the simplex with the smoothed objective
        a = current_a.copy()
        v = a.copy()
        t_fista = 1.0

        # Step size: use 1/L where L is a rough Lipschitz constant
        # L ~ 4n * max|a| (rough bound)
        L = 4.0 * n * max_c * 2.0
        step_size = 1.0 / L

        max_iter = min(3000, max(50, int(remaining / (0.005 * max(1.0, n / 8192.0)))))

        for it in range(max_iter):
            if time.monotonic() > deadline - min_time_remaining:
                break

            # Gradient of smoothed objective at v
            c_v = _full_convolve(v, v)
            c_v_max = float(c_v.max())
            shifted = (c_v - c_v_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c_v)
                w[int(np.argmax(c_v))] = 1.0
            else:
                w = e / s

            grad = 4.0 * n * v * (_corr_a_A(v, w, n) - float(np.dot(w, c_v)))

            # Gradient step
            y = v - step_size * grad

            # Project onto simplex {a >= 0, sum(a) = 1}
            y_proj = _project_simplex(y)

            # FISTA update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista)) / 2.0
            v = y_proj + ((t_fista - 1.0) / t_new) * (y_proj - a)
            v = np.maximum(v, 0)
            v = v / max(v.sum(), 1e-300)
            a = y_proj
            t_fista = t_new

        current_a = _project_simplex(a)
        sc = _score(current_a)
        if sc < current_sc:
            current_sc = sc

    return current_a


def _make_initialization(n, kind, rng=None):
    """Generate diverse initializations."""
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "linear_ramp":
        a = np.linspace(0.1, 1.0, n)
        return a / a.sum()
    elif kind == "linear_ramp_rev":
        a = np.linspace(1.0, 0.1, n)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "sin_squeeze":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t)
        return a / a.sum()
    elif kind == "exp_decay_rev":
        t = np.linspace(0, 1, n)
        a = np.exp(2.0 * t)
        return a / a.sum()
    elif kind == "double_hump":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2)
        return a / a.sum()
    elif kind == "asym_cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * np.sin(np.pi * t)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "asym_cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * t)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "quadratic_right":
        t = np.linspace(0, 1, n)
        a = t ** 0.5
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "quadratic_left":
        t = np.linspace(0, 1, n)
        a = (1 - t) ** 0.5
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "barker_like":
        # Approximate Barker code pattern [-1,-1,1,1,1,-1,1] mapped to [0,1]
        pattern = np.array([0.0, 0.0, 1.0, 1.0, 1.0, 0.0, 1.0])
        a = np.tile(pattern, n // len(pattern) + 1)[:n].astype(float)
        a = a + 0.3
        return a / a.sum()
    elif kind == "random_smooth":
        if rng is None:
            rng = np.random.default_rng(42)
        # Generate smooth random sequence via low-freq Fourier
        k = np.arange(n)
        freqs = np.array([1, 2, 3, 5, 7])
        amps = rng.uniform(0.1, 0.5, len(freqs))
        phases = rng.uniform(0, 2 * np.pi, len(freqs))
        a = np.ones(n)
        for i, f in enumerate(freqs):
            a += amps[i] * np.sin(2 * np.pi * f * k / n + phases[i])
        a = np.maximum(a, 0.05)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    1. Diverse asymmetric initializations
    2. L-BFGS aggressive continuation (main workhorse)
    3. FISTA accelerated proximal gradient
    4. Peak shaving + convolution balancing heuristics
    5. Active-set Newton refinement
    6. Multi-scale (n=8192 and n=16384)
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 40.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.12f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.12f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_initialization(n_parent, "uniform")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Try multiple n values with diverse initializations ===
    print(f"\n[Phase 1] Multi-initialization search (t={time.monotonic() - t_start:.1f}s)")

    # First, optimize the parent at its own length
    n_work = n_parent if n_parent > 0 else 16384

    # Quick L-BFGS on parent
    lbfgs_dl = min(deadline - 600, time.monotonic() + 250)
    a_opt = _lbfgs_continuation(parent, lbfgs_dl, min_time_remaining=600)
    update_best(a_opt, _score(a_opt), "lbfgs_parent")

    # Try a few diverse initializations at n_work
    init_kinds = ["linear_ramp", "asym_cos", "quadratic_right", "exp_decay"]
    for kind in init_kinds:
        if time.monotonic() > deadline - 500:
            break
        a_init = _make_initialization(n_work, kind, rng)
        sc_init = _score(a_init)
        print(f"  Init '{kind}': score={sc_init:.6f}")
        if sc_init < best_score:
            update_best(a_init, sc_init, f"init_{kind}")
        # Quick optimization
        opt_dl = min(deadline - 400, time.monotonic() + 60)
        a_opt2 = _lbfgs_continuation(a_init, opt_dl, min_time_remaining=400)
        update_best(a_opt2, _score(a_opt2), f"opt_{kind}")

    # === Phase 2: FISTA on current best ===
    print(f"\n[Phase 2] FISTA optimization (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        fista_dl = min(deadline - 350, time.monotonic() + 100)
        a_fista = _fista_smoothed(best_seq, fista_dl, n_betas=12, min_time_remaining=350)
        update_best(a_fista, _score(a_fista), "fista")

    # === Phase 3: Peak shaving ===
    print(f"\n[Phase 3] Peak shaving (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        ps_dl = min(deadline - 250, time.monotonic() + 60)
        a_ps, sc_ps = _peak_shaving(best_seq, ps_dl, max_iter=300)
        update_best(a_ps, sc_ps, "peak_shave")

    # === Phase 4: Convolution balancing ===
    print(f"\n[Phase 4] Convolution balancing (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        cb_dl = min(deadline - 200, time.monotonic() + 50)
        a_cb, sc_cb = _convolution_balance(best_seq, cb_dl, max_iter=200)
        update_best(a_cb, sc_cb, "conv_balance")

    # === Phase 5: Active-set Newton refinement ===
    print(f"\n[Phase 5] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        as_dl = min(deadline - 150, time.monotonic() + 80)
        a_as, sc_as = _active_set_newton(best_seq, as_dl, max_outer=80)
        update_best(a_as, sc_as, "newton")

    # === Phase 6: Aggressive L-BFGS polish (very small beta) ===
    print(f"\n[Phase 6] Aggressive L-BFGS polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        pol_dl = min(deadline - 100, time.monotonic() + 100)
        a_pol = _lbfgs_continuation(best_seq, pol_dl, min_time_remaining=100,
                                     start_beta_frac=0.001, end_beta_frac=1e-9)
        update_best(a_pol, _score(a_pol), "lbfgs_polish")

    # === Phase 7: Try n=8192 (smaller, more iterations possible) ===
    print(f"\n[Phase 7] Smaller n optimization (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 120:
        n_small = 8192
        a_small = _interpolate_seq(best_seq, n_small) if best_seq is not None else _make_initialization(n_small, "uniform")
        sc_small = _score(a_small)
        print(f"  n={n_small} interpolated score: {sc_small:.12f}")
        update_best(a_small, sc_small, f"interp_n{n_small}")

        small_dl = min(deadline - 80, time.monotonic() + 70)
        a_small_opt = _lbfgs_continuation(a_small, small_dl, min_time_remaining=80)
        update_best(a_small_opt, _score(a_small_opt), f"opt_n{n_small}")

        # Also try direct initialization at n_small
        for kind in ["linear_ramp", "asym_cos"]:
            if time.monotonic() > deadline - 60:
                break
            a_sr = _make_initialization(n_small, kind, rng)
            sr_dl = min(deadline - 40, time.monotonic() + 30)
            a_sr_opt = _lbfgs_continuation(a_sr, sr_dl, min_time_remaining=40)
            update_best(a_sr_opt, _score(a_sr_opt), f"n{n_small}_{kind}")

    # === Phase 8: Randomized restarts from current best ===
    print(f"\n[Phase 8] Randomized restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        n_best = len(best_seq)
        for sigma in [0.003, 0.01, 0.03]:
            if time.monotonic() > deadline - 30:
                break
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z + sigma * rng.standard_normal(n_best)
            z = z - z.mean()
            a_restart = _softmax(z)
            rd = min(deadline - 15, time.monotonic() + 25)
            a_r_opt = _lbfgs_continuation(a_restart, rd, min_time_remaining=15)
            update_best(a_r_opt, _score(a_r_opt), f"restart_s{sigma}")

    # === Phase 9: Final combined refinement ===
    print(f"\n[Phase 9] Final refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        # Peak shave
        ps_dl = min(deadline - 8, time.monotonic() + 20)
        a_fin1, sc_fin1 = _peak_shaving(best_seq, ps_dl, max_iter=100)
        update_best(a_fin1, sc_fin1, "final_ps")

        # Newton
        if time.monotonic() < deadline - 8:
            nw_dl = min(deadline - 5, time.monotonic() + 15)
            a_fin2, sc_fin2 = _active_set_newton(best_seq, nw_dl, max_outer=40)
            update_best(a_fin2, sc_fin2, "final_newton")

        # Tiny beta L-BFGS
        if time.monotonic() < deadline - 6:
            n_fin = len(best_seq)
            c_fin = _full_convolve(best_seq, best_seq)
            mb_fin = float(c_fin.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_tiny = mb_fin * 2e-8
            remaining = deadline - 6 - time.monotonic()
            if remaining > 3:
                max_iter_fin = min(2000, max(100, int(remaining / 0.05)))
                try:
                    result = minimize(
                        lambda z_val: _lsobj_factory(z_val, beta_tiny, n_fin),
                        z,
                        jac=lambda z_val: _lsgrad_factory(z_val, beta_tiny, n_fin),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter_fin, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 200}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_tiny_beta")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_initialization(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_initialization(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
