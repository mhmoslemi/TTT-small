import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

3. Growing active set with periodic QP refinement (many inner iterations)
4. Frequency-domain spectral shaping for escape from local minima
5. Multi-n exploration with resampling

import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def _grad_topk(a, k, n):
    """Gradient of top-k average of autocorrelation."""
    c = _full_convolve(a, a)
    top_k = np.argsort(c)[::-1][:k]
    p = np.zeros(2 * n - 1)
    p[top_k] = 1.0 / k
    # grad = sum_k p_k * grad((a*a)[k])
    # grad[(a*a)[k]][m] = sum_j [a[j] if j+k=m or j=m+k]
    # = (a shifted) + (a reversed shifted)
    # More precisely: d/d a[m] of sum_j a[j]a[j+k] = sum_j [delta(j,m)*a[j+k] + delta(j+k,m)*a[j]]
    # = a[m+k] (if m+k < n) + a[m-k] (if m-k >= 0)
    # For the full linear convolution: d/d a[m] of (a*a)[k] = a[m+k-n+1... ] 
    # Let me use the FFT-based approach:
    # grad[m] = sum_k p[k] * (a_rev convolved with p)[m + n - 1]
    # where a_rev = a[::-1]
    
    # Using the relation: d/d a [a^T Q_k a] = (Q_k + Q_k^T) a = 2 Q_k a (since Q_k symmetric)
    # Q_k a [m] = a[m+k] (for valid m)
    # So grad[m] = sum_k p[k] * 2 * a[m+k] (with appropriate bounds)
    # Actually: (a*a)[k] = sum_j a[j] a[j+k], so d/d a[m] = a[m+k] + a[m-k] (if in range)
    # But for the full convolution, (a*a)[k] = sum_j a[j] a[k-j]
    # d/d a[m] = a[k-m] + a[m-k] (if in range)
    
    # Let me just use the FFT trick:
    # grad = 2 * (p conv a_rev)[n-1 : 2n-1] - correction for diagonal
    a_rev = a[::-1]
    N = _next_pow2(len(p) + n - 1)
    Fp = np.fft.rfft(p, N)
    Fa = np.fft.rfft(a_rev, N)
    conv = np.fft.irfft(Fp * Fa, N)[:len(p) + n - 1]
    result = 2.0 * conv[n - 1:2 * n - 1].copy()
    # Diagonal correction: for even k, the term a[m]*a[m] appears only once
    # In the FFT conv, it appears twice, so we subtract once
    for k in range(0, 2 * n - 1, 2):
        m = k // 2
        if m < n and k < len(p) and p[k] > 0:
            result[m] -= a[m] * p[k]
    return result

def _grad_smoothed(a, beta, n):
    """Gradient of log-sum-exp smoothed max of autocorrelation."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s_e = e.sum()
    if s_e < 1e-300:
        p = np.zeros(2 * n - 1)
        p[int(np.argmax(c))] = 1.0
    else:
        p = e / s_e
    return _grad_topk_from_p(a, p, n)

def _grad_topk_from_p(a, p, n):
    """Gradient given the probability vector p over lags."""
    a_rev = a[::-1]
    N = _next_pow2(len(p) + n - 1)
    Fp = np.fft.rfft(p, N)
    Fa = np.fft.rfft(a_rev, N)
    conv = np.fft.irfft(Fp * Fa, N)[:len(p) + n - 1]
    result = 2.0 * conv[n - 1:2 * n - 1].copy()
    # Diagonal correction
    for k in range(0, 2 * n - 1, 2):
        m = k // 2
        if m < n and k < len(p) and p[k] > 0:
            result[m] -= a[m] * p[k]
    return result

def _run_adam_topk(a_init, deadline, k_schedule, min_time_remaining=60.0):
    """Adam optimizer with top-k objective and Nesterov momentum."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    m_v = np.zeros(n)  # First moment
    v_v = np.zeros(n)  # Second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    best_a = a_init.copy()
    best_sc = _score(a_init)
    t = 0
    lr = 0.01
    
    total_iters = max(1, int((deadline - min_time_remaining - time.monotonic()) / 0.05))
    # Schedule: k starts at k_schedule[0] and decreases to k_schedule[-1]
    # Linearly interpolate k over total_iters
    k_start = k_schedule[0]
    k_end = k_schedule[-1]
    
    for it in range(total_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break
        # Interpolate k
        frac = min(1.0, it / max(1, total_iters - 1))
        k = max(1, int(round(k_start + (k_end - k_start) * frac)))
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Compute gradient of top-k average
        top_k = np.argsort(c)[::-1][:k]
        p = np.zeros(2 * n - 1)
        p[top_k] = 1.0 / k
        grad = _grad_topk_from_p(a, p, n)
        
        # Adam update (in log-space)
        t += 1
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad * grad
        m_hat = m_v / (1 - beta1 ** t)
        v_hat = v_v / (1 - beta2 ** t)
        
        # Adaptive learning rate
        lr_base = 0.02 / (np.linalg.norm(grad) / (np.linalg.norm(a) + 1e-10) + 1e-10)
        lr_t = lr_base * min(1.0, 10.0 / (t + 1))
        
        z_new = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new = z_new - z_new.mean()
        z = z_new
        
        # Periodic evaluation
        if it % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a_check.copy()
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _run_qp_refinement(a_init, deadline, min_time_remaining=30.0, max_rounds=200):
    """QP refinement with growing active set and many inner iterations."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    # Active set schedule: start with 5, grow to 100
    k_schedule = [5, 10, 20, 30, 50, 80, 100, 150, 200, 300, 500]
    k_idx = 0
    
    for rnd in range(max_rounds):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Determine active set size
        k = k_schedule[min(k_idx, len(k_schedule) - 1)]
        k = min(k, 2 * n - 1)
        top_k = np.argsort(c)[::-1][:k]
        p = np.zeros(2 * n - 1)
        p[top_k] = 1.0 / k
        
        # Solve QP: min sum_k p_k (a*a)[k] s.t. sum a = 1, a >= 0
        # Using projected gradient with many inner iterations
        a_qp = a.copy()
        for inner in range(50):
            grad = _grad_topk_from_p(a_qp, p, n)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            # Step size: backtracking line search
            step = 0.5 / (gnorm + 1e-10)
            for ls in range(20):
                a_try = _proj_simplex(a_qp - step * grad)
                obj_try = float(np.dot(p, _full_convolve(a_try, a_try)))
                obj_cur = float(np.dot(p, _full_convolve(a_qp, a_qp)))
                if obj_try < obj_cur:
                    a_qp = a_try
                    break
                step *= 0.5
        
        a = a_qp
        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
        
        # Grow active set every 5 rounds
        if (rnd + 1) % 5 == 0 and k_idx < len(k_schedule) - 1:
            k_idx += 1
    
    return best_a, best_sc

def _run_lbfgs_smoothed(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive log-sum-exp smoothing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)]
    base_beta = max(gap * 0.05, max_c * 0.001)
    
    ratios = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
              0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000001]
    
    for i, ratio in enumerate(ratios):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        beta = base_beta * ratio
        max_iter = min(2000, max(30, int(remaining / 0.02)))
        
        try:
            def obj_func(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    return 2.0 * n * c_max_local
                return 2.0 * n * (c_max_local + beta * np.log(s_e))
            
            def grad_func(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    w = np.zeros_like(c_local)
                    w[int(np.argmax(c_local))] = 1.0
                else:
                    w = e / s_e
                return _grad_topk_from_p(a_local, w, n)
            
            result = minimize(
                obj_func, z,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 50, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return best_a, best_sc

def _spectral_shape(a, n_modes=10):
    """Apply spectral shaping: keep only low-frequency components."""
    n = len(a)
    N = _next_pow2(n)
    fa = np.fft.rfft(a, N)
    # Keep only first n_modes frequency bins
    mask = np.zeros(N // 2 + 1)
    mask[:n_modes + 1] = 1.0
    # Smooth transition
    transition = 5
    if n_modes + transition < N // 2:
        t = np.linspace(1, 0, transition)
        mask[n_modes + 1:n_modes + 1 + transition] = t
    fa_shaped = fa * mask
    a_shaped = np.fft.irfft(fa_shaped, N)[:n]
    a_shaped = np.maximum(a_shaped, 0)
    s = a_shaped.sum()
    if s < 1e-15:
        return a.copy()
    return a_shaped / s

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

For the linear convolution $(a*a)[\ell] = \sum_j a[j] \cdot a[\ell - j]$ where the sum is over all valid $j$ (i.e., $0 \leq j < n$ and $0 \leq \ell - j < n$).

The derivative with respect to $a[m]$:
$$\frac{\partial (a*a)[\ell]}{\partial a[m]} = a[\ell - m] + a[m - \ell]$$

where we define $a[x] = 0$ for $x < 0$ or $x \geq n$.

So the gradient of $\sum_k p_k (a*a)[k]$ w.r.t. $a[m]$ is:
$$\sum_k p_k (a[k-m] + a[m-k])$$

This can be computed as:
$$\text{grad}[m] = \sum_k p_k a[k-m] + \sum_k p_k a[m-k]$$

$\frac{\partial}{\partial a[m]} \sum_j a[j] \cdot a[k-j] = \sum_j \left(\frac{\partial a[j]}{\partial a[m]} \cdot a[k-j] + a[j] \cdot \frac{\partial a[k-j]}{\partial a[m]}\right)$
