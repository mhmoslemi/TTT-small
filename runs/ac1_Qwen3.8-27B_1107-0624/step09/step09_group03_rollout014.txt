"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Adaptive log-sum-exp continuation with dynamic beta scheduling
2. KKT-based active-set Newton with adaptive active set identification
3. Multi-scale search (n=8192, 16384, 32768) with time-proportional allocation
4. Frequency-domain informed initialization and perturbation
5. Iterative refinement loop with convergence detection
"""
import numpy as np
import time
from scipy.optimize import minimize

# Module-level state for optimizer callbacks (avoids closures)
_state = {'beta': 0.0, 'n': 0}
_newton_state = {'A_vec': None, 'n': 0, 'mu': 0.0}


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _ls_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _ls_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _obj_wrapper(z):
    return _ls_objective(z, _state['beta'], _state['n'])


def _grad_wrapper(z):
    return _ls_gradient(z, _state['beta'], _state['n'])


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _hess_matvec(v):
    return 2.0 * _corr_a_A(v, _newton_state['A_vec'], _newton_state['n']) + _newton_state['mu'] * v


def _cg_solve(matvec_fn, rhs, max_cg=200, tol=1e-13):
    x = np.zeros_like(rhs)
    r = rhs - matvec_fn(x)
    p = r.copy()
    rs = float(np.dot(r, r))
    rhs_norm = float(np.linalg.norm(rhs))
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_cg):
        Ap = matvec_fn(p)
        pAp = float(np.dot(p, Ap))
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = float(np.dot(r, r))
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _newton_active_set(a, c, c_max, tol_frac, mu):
    n = len(a)
    active = c >= (c_max - tol_frac * c_max)
    if not active.any():
        active[int(np.argmax(c))] = True
    m = int(active.sum())
    if m < 2 or m > 50 * n:
        return None

    A_vec = active.astype(np.float64) / m
    _newton_state['A_vec'] = A_vec
    _newton_state['n'] = n
    _newton_state['mu'] = mu

    g = 2.0 * _corr_a_A(a, A_vec, n)
    g_t = g - np.mean(g)
    gnorm = float(np.linalg.norm(g_t))
    if gnorm < 1e-14:
        return None

    u = _cg_solve(_hess_matvec, g_t, max_cg=300)
    w_vec = _cg_solve(_hess_matvec, np.ones(n), max_cg=300)
    w_sum = float(w_vec.sum())

    if abs(w_sum) < 1e-15:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w_vec

    dx = dx - np.mean(dx)

    # Verify descent direction
    if float(np.dot(dx, g_t)) > 0:
        dx = -g_t + np.mean(g_t)
        dx = dx - np.mean(dx)

    return dx


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "bipolar":
        a = np.ones(n)
        a[n // 4:3 * n // 4] = 0.3
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - np.mean(z)
    return _softmax(z)


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _run_newton_polish(a_init, deadline, max_outer=80):

    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()

        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for outer in range(max_outer):

        c_max = float(c.max())

        # Try multiple tolerance levels
        best_improvement = 0.0
        best_a_new = None

        tol_list = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]

            tol_list = [0.01, 0.02, 0.05, 0.1]

        mu = 1e-8

        for tol_frac in tol_list:

            t = 1.0

                a_try = np.maximum(a_try, 0.0)

                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)

                if f_try < c_max - 1e-14:
                    improvement = c_max - f_try
                    if improvement > best_improvement:

                        best_a_new = a_try.copy()
                    break
                t *= 0.5

        if best_a_new is None:
            # Gradient fallback
            g_all = 2.0 * _corr_a_A(a, np.ones(2 * n - 1) / (2 * n - 1), n)
            g_t = g_all - np.mean(g_all)
            t = 0.05

        sc = _score(a)

            best_a = a.copy()

            a = best_a

        c_check = _full_convolve(a, a)

def _run_refinement(a_init, deadline, max_rounds=6):

        a_nw = _run_newton_polish(a, nw_dl, max_outer=40)
        sc_nw = _score(a_nw)
        if sc_nw < best_sc:

            a = a_nw
        else:
            a = best_a

                    _obj_wrapper, z_fin, jac=_grad_wrapper, method='L-BFGS-B',
                    options={'maxiter': max_iter_fin, 'ftol': 1e-18, 'gtol': 1e-15,
                             'maxcor': 100, 'maxls': 200}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_tiny_beta")
            except Exception as ex:
                print(f"  Final LS error: {ex}")

        # Final Newton
        if time.monotonic() < deadline - 5:
            a_fin_nw = _run_newton_polish(best_seq, deadline - 3, max_outer=20)
            update_best(a_fin_nw, _score(a_fin_nw), "final_newton")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
