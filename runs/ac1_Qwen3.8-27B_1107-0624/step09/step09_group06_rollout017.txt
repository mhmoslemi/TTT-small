    Newton's method to solve c[k] = t for k in S, sum(b) = 1, b >= 0.
    b0: initial point (on the simplex)
    S: active set (array of lag indices)
    Returns: (b, t) or None if failed.

        F_vals = c[S] - t  # shape (|S|,)

        # Row k (0-indexed, k < |S|): d(c[S[k]] - t)/d(b[m]) = grad[k][m], and -1 for t
        # Last row: d(sum(b)-1)/d(b[m]) = 1, and 0 for t
        
        # Compute J J^T (small matrix of size (|S|+1) x (|S|+1))
        K = len(S)
        JJT = np.zeros((K + 1, K + 1))
        
        # Compute gradients for active peaks
        grads = np.zeros((K, n))

            grads[i] = _conv_grad_single(b, k)

        JJT[:K, :K] = grads @ grads.T
        # JJT[:K, K] = sum_m grads[i][m] = grads[i].sum()
        JJT[:K, K] = grads.sum(axis=1)
        JJT[K, :K] = grads.sum(axis=1)

        JJT[K, K] = n

        rhs = np.zeros(K + 1)
        rhs[:K] = -(c[S] - t)
        rhs[K] = -(b.sum() - 1.0)

        try:
            x = np.linalg.solve(JJT, rhs)

            x = np.linalg.lstsq(JJT, rhs, rcond=None)[0]

            if s_new < 1e-15:
