def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

"""
Active-set LP-based optimizer for minimizing the peak of the self-convolution.
Uses linearized active-set constraints solved via LP at each iteration,
combined with line search and periodic perturbation.
"""
import numpy as np
import time
from scipy.optimize import linprog

# ... helper functions ...

def _solve_active_lp(a, c, active_indices, eta, alpha_safety=10):
    """
    Solve the LP for the optimal perturbation direction.
    
    Minimize tau'
    Subject to:
      For k in active: sum_j 2*a[j]*delta[k-j] - tau' <= c[k]
      sum_j delta[j] = 0
      -a[j]/alpha <= delta[j] <= eta
      0 <= tau' <= 2*max(c[active])
    """
    n = len(a)
    m = len(active_indices)
    
    # Variable: x = [delta[0], ..., delta[n-1], tau']
    # Objective: minimize tau' => c_obj = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ineq @ x <= b_ineq
    # For each active lag k:
    #   sum_j 2*a[k-j]*delta[j] - tau' <= c[k]
    #   => sum_j 2*a[k-j]*delta[j] + (-1)*tau' <= c[k]
    A_ineq = np.zeros((m, n + 1))
    b_ineq = np.zeros(m)
    
    for i, k in enumerate(active_indices):
        # delta[j] appears with coefficient 2*a[k-j] if 0 <= k-j < n
        # i.e., k-n+1 <= j <= k
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A_ineq[i, j] = 2.0 * a[k - j]
        A_ineq[i, -1] = -1.0
        b_ineq[i] = c[k]
    
    # Equality constraint: sum delta[j] = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    lb = np.concatenate([(-a / alpha_safety), [0.0]])
    ub = np.concatenate([np.full(n, eta), [2.0 * max(c[active_indices])]])
    
    result = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, 
                     A_eq=A_eq, b_eq=b_eq,
                     bounds=list(zip(lb, ub)), method='highs')
    
    if not result.success:
        return None, None
    
    delta = result.x[:n]
    tau_prime = result.x[-1]
    return delta, tau_prime

Active-set LP optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: solves a linear program at each step to find the optimal
first-order perturbation direction that reduces the maximum convolution value,
using an adaptive active set of binding constraints.

import numpy as np

from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _solve_lp(a, c, active_idx, eta, alpha=10.0):
    """
    Solve: min tau'
    s.t. c[k] + 2*(a*delta)[k] <= tau' for k in active_idx
         sum(delta) = 0
         -a/alpha <= delta <= eta
         0 <= tau'
    """
    n = len(a)
    m = len(active_idx)
    
    # Objective: minimize tau' (last variable)
    f = np.zeros(n + 1)
    f[-1] = 1.0
    
    # Inequality: A_ub @ x <= b_ub
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    for i, k in enumerate(active_idx):
        # (a*delta)[k] = sum_j a[j] * delta[k-j] for valid j
        # delta[t] appears with coefficient 2*a[k-t] in row i
        # t = k-j, so j = k-t
        # Valid: 0 <= j < n and 0 <= t < n => 0 <= k-j < n => k-n+1 <= j <= k
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            A_ub[i, j] = 2.0 * a[k - j]
        A_ub[i, n] = -1.0
        b_ub[i] = c[k]
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    lb = np.empty(n + 1)
    ub = np.empty(n + 1)
    lb[:n] = -a / alpha
    ub[:n] = eta
    lb[n] = 0.0
    ub[n] = 2.0 * c.max()
    
    try:
        res = linprog(f, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=list(zip(lb, ub)), method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
    except Exception:
        return None
    
    if not res.success:
        return None
    
    delta = res.x[:n]
    return delta

def _line_search(a, delta, c_original, n):
    """Find best step size t for a + t*delta."""
    c_peak_orig = c_original.max()
    best_t = 0.0
    best_score = 2.0 * n * c_peak_orig
    best_a = a.copy()
    
    for t in [1.0, 0.7, 0.5, 0.3, 0.15, 0.075, 0.04, 0.02]:
        a_try = a + t * delta
        if a_try.min() < 0:
            continue
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s  # normalize
        c_try = _full_convolve(a_try, a_try)
        sc = 2.0 * n * c_try.max()
        if sc < best_score:
            best_score = sc
            best_t = t
            best_a = a_try.copy()
    
    return best_a, best_t, best_score

        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:

    n = len(a)
    m = len(active_idx)
    
    # x = [delta_0, ..., delta_{n-1}, tau']
    f = np.zeros(n + 1)
    f[n] = 1.0  # minimize tau'
    
    A_ub = np.zeros((m, n + 1))
    b_ub = np.empty(m)
    
    for i, k in enumerate(active_idx):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            A_ub[i, j_lo:j_hi+1] = 2.0 * a[k - j_lo : k - j_hi + 1][::-1]
        A_ub[i, n] = -1.0
        b_ub[i] = c[k]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    lb = np.empty(n + 1)
    ub = np.empty(n + 1)
    lb[:n] = -a / alpha
    ub[:n] = eta
    lb[n] = 0.0
    ub[n] = c.max() * 1.5
    
    res = linprog(f, A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq,
                  bounds=list(zip(lb, ub)),
                  method='highs')
    
    if res.success:
        return res.x[:n]
    return None
