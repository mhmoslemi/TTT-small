"""
Optimized sequence finder for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses aggressive L-BFGS with log-sum-exp smoothing, Nesterov subgradient,
peak-directed local search, and multi-n exploration.
Key improvements: finer beta schedule, directed peak reduction,
convex combination restarts, and better time allocation.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _conv_grad_single(a, k):
    n = len(a)
    grad = np.zeros(n)
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    if j_lo <= j_hi:
        idx = k - np.arange(j_lo, j_hi + 1)
        grad[j_lo:j_hi + 1] += a[idx]
    j_lo2 = max(0, k)
    j_hi2 = min(n - 1, k + n - 1)
    if j_lo2 <= j_hi2:
        idx = np.arange(j_lo2, j_hi2 + 1) - k
        grad[j_lo2:j_hi2 + 1] += a[idx]
    return grad


def _conv_grad_multi(a, active_indices):
    n = len(a)
    grad = np.zeros(n)
    for k in active_indices:
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            idx = k - np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi + 1] += a[idx]
        j_lo2 = max(0, k)
        j_hi2 = min(n - 1, k + n - 1)
        if j_lo2 <= j_hi2:
            idx = np.arange(j_lo2, j_hi2 + 1) - k
            grad[j_lo2:j_hi2 + 1] += a[idx]
    return grad


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _nesterov_subgrad(a_init, deadline, min_time_remaining=60.0,
                      K_init=5, K_max=200, seed=0):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    momentum = 0.8
    base_lr = 0.008
    lr = base_lr
    K = K_init
    t = 0
    no_improve = 0
    stall_count = 0

    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
        if gap < c_max * 0.003 and K < K_max:
            K = min(K_max, int(K * 1.4) + 1)

        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]
        grad = _conv_grad_multi(a, active)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-18:
            break

        lr = min(base_lr, lr * 1.005)
        effective_lr = lr / (gnorm + 1e-18) * 0.15

        v = momentum * v - effective_lr * grad_t
        a_cand = a + v
        a_proj = _project_simplex(a_cand)

        c_new = _full_convolve(a_proj, a_proj)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < current_sc:
            a = a_proj
        else:
            v = 0.5 * v
            lr *= 0.9
            a = a_proj

        if no_improve % 150 == 0 and no_improve > 0:
            stall_count += 1
            if stall_count > 8:
                break
            perturb = 0.002 * rng.standard_normal(n)
            a_pert = _project_simplex(a + perturb)
            c_pert = _full_convolve(a_pert, a_pert)
            sc_pert = 2.0 * n * float(c_pert.max())
            if sc_pert < best_sc:
                a = a_pert
                best_a = a.copy()
                best_sc = sc_pert
            v = np.zeros(n)
            lr = base_lr
            no_improve = 0

    return best_a, best_sc

    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        # Scale max_iter based on remaining time and problem size
        scale = max(1.0, n / 8192.0)
        if i < len(betas) // 3:
            est_time = 0.003 * scale
            max_iter = min(8000, max(100, int(remaining / (est_time * 15))))
        elif i < 2 * len(betas) // 3:
            est_time = 0.004 * scale
            max_iter = min(6000, max(50, int(remaining / (est_time * 12))))
        else:
            est_time = 0.005 * scale
            max_iter = min(5000, max(30, int(remaining / (est_time * 10))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 100,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 5 == 0:
                    print(f"    LBFGS beta_idx={i}/{len(betas)-1}: score={current_sc:.10f}")
        except Exception:
            pass

    return current_a


def _peak_directed_search(a_init, deadline, min_time_remaining=30.0, seed=0):
    """Directly target and reduce the top peaks using local Newton-like steps."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    rng = np.random.default_rng(seed)

    lr = 0.05
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            break

        # Find top K peaks
        K = max(3, min(20, int(0.002 * n)))
        c_idx = np.argsort(c)[::-1]
        active_k = c_idx[:K]
        peak_vals = c[active_k]

        # Weight peaks by how much they exceed the median
        med = np.median(c)
        weights = np.maximum(0, peak_vals - med)
        if weights.sum() < 1e-15:
            weights = np.ones(K)
        weights = weights / weights.sum()

        # Compute weighted gradient
        grad = np.zeros(n)
        for i, k in enumerate(active_k):
            g_k = _conv_grad_single(a, k)
            grad += weights[i] * g_k

        # Project to tangent space
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break

        # Step direction: reduce peaks while maintaining sum constraint
        direction = -grad_t / (gnorm + 1e-18)

        # Line search
        step = lr / (gnorm + 1e-18)
        improved = False
        for trial in range(8):
            a_try = a + step * direction
            a_try = _project_simplex(a_try)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc:
                a = a_try
                current_sc = sc_try
                improved = True
                break
            step *= 0.5

        if not improved:
            # Try a random perturbation in a useful direction
            pert_size = 0.001 * (1.0 + no_improve * 0.01)
            a_pert = _project_simplex(a + pert_size * rng.standard_normal(n))
            c_pert = _full_convolve(a_pert, a_pert)
            sc_pert = 2.0 * n * float(c_pert.max())
            if sc_pert < best_sc:
                a = a_pert
                best_a = a.copy()
                best_sc = sc_pert
                no_improve = 0
            else:
                if no_improve > 50:
                    lr *= 0.7
                if no_improve > 150:
                    break

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
