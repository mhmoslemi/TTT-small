"""
Optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Fourier-parameterized optimization (low-dim parameter space with exp parameterization)
2. Active-set Newton with CG Hessian solve and multi-tolerance active set selection
3. Frequency-domain random exploration for global diversity
4. Multi-scale strategy (n=8192, 16384, 32768)
5. Iterative refinement combining Fourier + Newton + deflation
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _convolve_grad(w, a, n):
    """Gradient of sum_k w[k]*c[k] w.r.t. a[j], where c=conv(a,a).
    Result: 2 * (correlation of w with a), restricted to valid indices.
    w has length 2n-1, a has length n.
    """
    flip_a = a[::-1]
    out_len = n + len(w) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fw = np.fft.rfft(w, N)
    result = np.fft.irfft(fa * fw, N)[:out_len]
    return 2.0 * result[n - 1:2 * n - 1]


def _hessian_apply(w, v, n):
    """Apply Hessian of sum_k w[k]*c[k] to vector v.
    H[j][i] = 2*w[i+j] (if 0<=i+j<2n-1), so (H*v)[j] = 2*sum_i w[i+j]*v[i].
    This is 2 * (correlation of w with v) at valid positions.
    w has length 2n-1, v has length n.
    """
    flip_v = v[::-1]
    out_len = n + len(w) - 1
    N = _next_pow2(out_len)
    fv = np.fft.rfft(flip_v, N)
    fw = np.fft.rfft(w, N)
    result = np.fft.irfft(fv * fw, N)[:out_len]
    return 2.0 * result[n - 1:2 * n - 1]


def _cg_solve(hess_func, rhs, max_cg=300, tol=1e-13):
    x = np.zeros_like(rhs)
    r = rhs.copy()
    p = r.copy()
    rs = np.dot(r, r)
    if rs < 1e-30:
        return x
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = hess_func(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _fourier_to_seq(params, n):
    """Convert Fourier params to non-negative sequence.
    params = [r1, p1, r2, p2, ..., rK, pK]
    a[j] = exp(sum_k r_k * cos(2*pi*k*j/n + p_k)) / sum
    """
    K = len(params) // 2
    t = np.arange(n, dtype=np.float64)
    log_a = np.zeros(n)
    for k in range(K):
        r = params[2 * k]
        p = params[2 * k + 1]
        log_a += r * np.cos(2.0 * np.pi * (k + 1) * t / n + p)
    log_a -= log_a.max()
    a = np.exp(log_a)
    s = a.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return a / s


def _seq_to_fourier(a, n, K):
    """Extract Fourier parameters from sequence (DFT-based initialization)."""
    # Use the log of the sequence to extract phases and amplitudes
    log_a = np.log(np.maximum(a, 1e-30))
    log_a_mean = log_a.mean()
    log_a_centered = log_a - log_a_mean
    params = np.zeros(2 * K)
    t = np.arange(n, dtype=np.float64)
    for k in range(K):
        freq = (k + 1)
        cos_k = np.cos(2.0 * np.pi * freq * t / n)
        sin_k = np.sin(2.0 * np.pi * freq * t / n)
        # Project log_a_centered onto cos and sin
        r_cos = 2.0 * np.dot(log_a_centered, cos_k) / n
        r_sin = 2.0 * np.dot(log_a_centered, sin_k) / n
        r_k = np.sqrt(r_cos ** 2 + r_sin ** 2)
        if r_k > 1e-15:
            p_k = np.arctan2(-r_sin, r_cos)
        else:
            p_k = 0.0
        params[2 * k] = r_k
        params[2 * k + 1] = p_k
    return params


def _fouriern_smooth_obj(params, n, beta):
    """Smoothed objective using log-sum-exp on the Fourier-parameterized sequence."""
    a = _fourier_to_seq(params, n)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _fouriern_smooth_grad(params, n, beta):
    """Gradient of smoothed objective w.r.t. Fourier parameters."""
    a = _fourier_to_seq(params, n)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # Gradient of sum_k w[k]*c[k] w.r.t. a
    grad_c = _convolve_grad(w, a, n)
    # Chain rule: d a[j] / d params
    K = len(params) // 2
    t = np.arange(n, dtype=np.float64)
    grad = np.zeros(2 * K)
    for k in range(K):
        freq = (k + 1)
        cos_k = np.cos(2.0 * np.pi * freq * t / n + params[2 * k + 1])
        sin_k = np.sin(2.0 * np.pi * freq * t / n + params[2 * k + 1])
        mean_cos = float(np.dot(a, cos_k))
        mean_sin = float(np.dot(a, sin_k))
        grad[2 * k] = float(np.dot(a * (cos_k - mean_cos), grad_c))
        grad[2 * k + 1] = -float(np.dot(a * (sin_k - mean_sin), grad_c))
    return grad

def _deflation_step(a, deadline, n_attempts=50):
    """Multi-peak deflation to reduce all significant peaks simultaneously."""
    n = len(a)

        w[peak_mask] = 1.0 / n_peaks

        g_t = grad - np.mean(grad)

                best_a = a_try.copy()
                a = a_try
                applied = True
                break

        if not applied:

                break

    return best_a, best_sc

def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "sinc_squared":
        t = np.linspace(-0.5, 0.5, n)
        a = np.sinc(t) ** 2
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _fourier_lbfgs(params_init, n, deadline, min_time_remaining=30.0):
    """L-BFGS in Fourier parameter space with log-sum-exp smoothing."""
    current_params = params_init.copy()
    a_cur = _fourier_to_seq(current_params, n)
    current_sc = _score(a_cur)
    c_cur = _full_convolve(a_cur, a_cur)
    max_c = float(c_cur.max())

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 50:
        gap = c_sorted[0] - c_sorted[50]
    else:
        gap = c_sorted[0] - c_sorted[-1]
    base_beta = max(gap * 0.5, max_c * 0.001)

    n_betas = 20
    log_betas = np.linspace(np.log(base_beta), np.log(max(max_c * 1e-8, 1e-15)), n_betas)
    betas = np.exp(log_betas)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.01 * max(1.0, n / 8192.0)
        max_iter = min(3000, max(50, int(remaining / (est_time * 5))))

        try:
            def _obj(p):
                return _fouriern_smooth_obj(p, n, beta)

            def _grad(p):
                return _fouriern_smooth_grad(p, n, beta)

            result = minimize(
                _obj, current_params, jac=_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 50, 'maxls': 200}
            )
            a_new = _fourier_to_seq(result.x, n)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_sc = sc_new
                current_params = result.x.copy()
        except Exception:
            pass

    a_final = _fourier_to_seq(current_params, n)
    a_final = np.maximum(a_final, 0)
    s_final = a_final.sum()
    if s_final > 1e-15:
        a_final = a_final / s_final
    return a_final

                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())

            if f_try < c_max - c1 * t * np.dot(g_t, dx):
                # Check curvature condition
                g_try = _convolve_grad(A_vec, a_try, n)
                g_try_t = g_try - np.mean(g_try)
                if np.dot(g_try_t, dx) < c2 * np.dot(g_t, dx):
                    found = True
                    break
            t *= 0.5

        if found:
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
            a = a_try
        else:
            # Fall back to gradient step
            g_full = _convolve_grad(np.ones(2 * n - 1) / (2 * n - 1), a, n)
            g_full_t = g_full - np.mean(g_full)
            t = 0.01
            improved = False
            for ls in range(30):
                a_try = a - t * g_full_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_sc - 1e-14:
                        best_sc = sc_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                t *= 0.5
            if not improved:
                break

    return best_a, best_sc
