"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses: multi-n exploration, aggressive L-BFGS with log-sum-exp annealing,
active-set projected gradient, symmetry-breaking line search, and direct peak shaving.
Key innovations: (1) symmetry/antisymmetry decomposition to find the optimal
asymmetry level, (2) much more aggressive beta annealing, (3) larger n exploration,
(4) direct "partner swap" moves that exploit the c[n-1] = sum a[j]*a[n-1-j] structure.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    if b is a:
        r = np.fft.irfft(fa * fa, N)[:out_n]
    else:
        fb = np.fft.rfft(b, N)
        r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _norm(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 2 + 0.01
        return a / a.sum()
    elif kind == "two_bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.3) ** 2) + np.exp(-8.0 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.4) ** 2) + 0.3 * np.exp(-4.0 * (t - 0.8) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        if rng is None:
            rng = np.random.default_rng(42)
        a = rng.random(n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _active_set_gradient(a, c, tol_frac, n_active_max=50):
    n = len(a)
    c_max = float(c.max())
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    n_act = int(active.sum())
    if n_act == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        n_act = 1
    if n_act > n_active_max:
        idxs = np.argsort(c[active])[::-1][:n_active_max]
        active_new = np.zeros_like(c, dtype=bool)
        active_new[np.where(active)[0][idxs]] = True
        active = active_new
        n_act = n_active_max
    eps = 1e-12 * c_max + 1e-30
    w = np.zeros_like(c)
    w[active] = 1.0 / (c_max - c[active] + eps)
    w = w / w.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad = 2.0 * h
    grad = grad - np.mean(grad)
    return grad


def _nesterov_polish(a_init, deadline, tol_start=0.005, tol_end=0.00003,
                     lr_init=0.04, momentum=0.85, max_stall=200):
    n = len(a_init)
    a = _norm(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    v = np.zeros(n)
    tol_frac = tol_start
    lr = lr_init
    no_improve = 0
    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        grad = _active_set_gradient(a, c, tol_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        v = momentum * v - lr * grad
        alpha = 1.0
        improved = False
        for _ in range(40):
            a_try = _norm(a + alpha * v)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            v = np.zeros(n)
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_end, tol_frac * 0.7)
                lr *= 0.7
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(tol_end, tol_frac * 0.3)
                lr = lr_init * 0.4
                no_improve = 0
        else:
            if tol_frac > tol_end:
                tol_frac = max(tol_end, tol_frac * 0.98)
            lr = min(lr_init, lr * 1.005)
    return best_a, _score(best_a)


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LBFGSProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def f(self, z):
        return _smoobj(z, self.beta, self.n)

    def g(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs_aggressive(a_init, deadline, min_remaining=20.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    k = min(50, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[k]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008,
                                  0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                                  0.00005, 0.00002, 0.00001, 0.000005, 0.000002])
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        est_time = 0.003 * max(1.0, n / 16384.0)
        max_iter = min(8000, max(50, int(remaining / (est_time * 20))))
        prob = _LBFGSProb(beta, n)
        try:
            result = minimize(prob.f, z, jac=prob.g, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxcor': 120, 'maxls': 60})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a


def _symmetry_break(a_init, deadline):
    """Exploit symmetric/antisymmetric decomposition to find optimal asymmetry level."""
    n = len(a_init)
    a = _norm(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    # Antisymmetric direction: larger on left, smaller on right
    t = np.linspace(0, 1, n)
    d = (1.0 - 2.0 * t)  # linear ramp: positive on left, negative on right
    # Ensure d has zero mean
    d = d - d.mean()
    # Project d onto the tangent space (already done) and ensure it doesn't violate a>=0 too badly
    # Try a range of asymmetry levels
    alphas = np.logspace(-2, 0.5, 20)
    for eps in alphas:
        a_try = _norm(a + eps * d)
        sc = _score(a_try)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_try.copy()
    # Also try the negative direction
    for eps in alphas:
        a_try = _norm(a - eps * d)
        sc = _score(a_try)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_try.copy()
    # Try a "quadratic" asymmetry
    d2 = (t - 0.5) ** 2 - ((t - 0.5) ** 2).mean()
    d2 = d2 - d2.mean()
    for eps in alphas[:10]:
        a_try = _norm(a + eps * d2)
        sc = _score(a_try)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_try.copy()
    return best_a, best_sc


def _partner_swap_search(a_init, deadline, n_trials=200):
    """Direct partner swaps: move mass from j to n-1-j to reduce c[n-1]."""
    n = len(a_init)
    a = _norm(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(a, a)
    peak_k = int(np.argmax(c))
    peak_val = float(c[peak_k])
    for _ in range(n_trials):
        if time.monotonic() > deadline:
            break
        # Compute sensitivity: how much does c[peak_k] change if we move mass from j to peak_k - j?
        # d(c[peak_k])/d(a[j]) = 2 * a[peak_k - j] (if valid)
        partner = np.zeros(n)
        valid = (peak_k >= 0) & (peak_k < 2 * n - 1)
        if valid:
            for j in range(n):
                pj = peak_k - j
                if 0 <= pj < n:
                    partner[j] = 2.0 * a[pj]
        # We want to reduce a[j] where partner[j] is high and increase where partner[peak_k - j] is low
        # Simple heuristic: find j where a[j] > a[peak_k - j] and partner[j] is large
        mirror_idx = np.array([peak_k - j for j in range(n)])
        valid_mask = (mirror_idx >= 0) & (mirror_idx < n)
        if not valid_mask.any():
            break
        diff = a[valid_mask] - a[mirror_idx[valid_mask]]
        # Move mass from high-diff positions (where a[j] >> a[peak_k-j])
        n_swap = max(5, n // 50)
        top_idx = np.argsort(diff)[-n_swap:]
        # Small step
        step = 0.01 * a.mean()
        a_try = a.copy()
        for idx in top_idx:
            j = int(idx)
            mj = peak_k - j
            if 0 <= mj < n and mj != j:
                amt = min(step, a_try[j] * 0.1)
                a_try[j] -= amt
                a_try[mj] += amt
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s > 1e-15:
            a_try = a_try / s
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try.copy()
            c = _full_convolve(a, a)
            peak_k = int(np.argmax(c))
            peak_val = float(c[peak_k])
        else:
            # Try smaller step
            step *= 0.5
    return best_a, best_sc


def _peak_shave(a_init, deadline, n_rounds=50):
    """Directly shave the top contributors to the peak."""
    n = len(a_init)
    a = _norm(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    for _ in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))
        # Find top contributors to c[peak_idx]
        partners = np.zeros(n)
        for j in range(n):
            pj = peak_idx - j
            if 0 <= pj < n:
                partners[j] = a[pj]
        # Also the "self" contribution
        partners[peak_idx] += a[peak_idx]  # double count at center
        # Reduce top 5% of partners, increase bottom 5%
        n_top = max(10, n // 20)
        top_positions = np.argsort(partners)[-n_top:]
        bot_positions = np.argsort(partners)[:n_top]
        total_reduce = a[top_positions].sum() * 0.02
        avg_reduce = total_reduce / n_top
        avg_increase = total_reduce / n_bot if (n_bot := len(bot_positions)) > 0 else 0
        a_try = a.copy()
        for j in top_positions:
            a_try[j] = max(0, a_try[j] - avg_reduce)
        for j in bot_positions:
            a_try[j] += avg_increase
        a_try = _norm(a_try)
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-15:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try.copy()
        else:
            # Try smaller reduction
            total_reduce *= 0.5
    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _quick_optimize(a_init, deadline, min_remaining=10.0):
    """Combined L-BFGS + polish + peak shave."""
    a = _run_lbfgs_aggressive(a_init, deadline, min_remaining=min_remaining + 10)
    if time.monotonic() < deadline - min_remaining:
        pol_deadline = min(deadline - min_remaining, time.monotonic() + 10)
        a2, sc2 = _nesterov_polish(a, pol_deadline, tol_start=0.003, tol_end=0.00005,
                                   lr_init=0.02, momentum=0.82)
        if sc2 < _score(a):
            a = a2
    if time.monotonic() < deadline - min_remaining + 5:
        shave_deadline = min(deadline - min_remaining + 3, time.monotonic() + 5)
        a3, sc3 = _peak_shave(a, shave_deadline, n_rounds=20)
        if sc3 < _score(a):
            a = a3
    return a
