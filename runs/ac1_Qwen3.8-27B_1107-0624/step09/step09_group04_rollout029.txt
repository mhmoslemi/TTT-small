**Key improvement in Phase 2: Exact line search**

When we have a direction d (with sum(d) = 0 to maintain the simplex), the function along the line a + t*d is:
f(t) = max_k [(a + t*d) * (a + t*d)]_k
     = max_k [c_k + 2t * (d * a)_k + t^2 * (d * d)_k]

where c = a*a, and I'm using the full convolution.

So f(t) = max_k [quadratic in t]. The minimum of f(t) for t >= 0 (and a + t*d >= 0) can be found by:
1. For each k, the unconstrained minimum of the quadratic is at t_k = -(d*a)_k / (d*d)_k (if (d*d)_k > 0)
2. The minimum of the max is where two (or more) quadratics intersect at their minimum
3. This is a 1D problem that can be solved by bisection or by finding the critical points

Actually, a simpler approach: use ternary search or golden section search on t in [0, t_max] where t_max is the largest t such that a + t*d >= 0.

Each evaluation of f(t) requires one FFT (O(n log n)), and we need O(log(1/epsilon)) evaluations for the search. So each line search is O(n log^2 n), which for n = 16384 is about 16384 * 14^2 ≈ 3.2M operations per line search. If we do 1000 line searches, that's 3.2B operations, about 30 seconds. Feasible.

But wait, we can do better. Instead of a full FFT for each evaluation, we can precompute (d*a) and (d*d) once, and then just evaluate the max of the quadratics:
f(t) = max_k [c_k + 2t * da_k + t^2 * dd_k]

This is O(n) per evaluation (just evaluating 2n-1 quadratics and taking the max), plus O(n) for the precomputation of da and dd (each requiring one FFT).

So the total cost per line search is O(n log n) for precomputation + O(n * n_iters) for the search, where n_iters is the number of function evaluations (say 50-100).

For n = 16384: O(16384 * 14 + 16384 * 100) ≈ 1.8M operations per line search. Very fast!

This is a key insight: **exact line search is cheap** when we can precompute the quadratic coefficients.

```python
    while k < n:

    return max(2, k)

def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    if s < 1e-15:

    return a / s
```

```python
"""
Active-set optimization with exact line search and incremental convolution updates
for minimizing 2*n*max(conv(a,a))/sum(a)^2. Key innovations over previous approach:
(1) Exact grid-based line search instead of backtracking, (2) adaptive active set
that grows to hundreds of lags, (3) incremental convolution update saving FFT calls,
(4) structured multi-scale perturbations for escape, (5) Nesterov-inspired momentum.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = float(a.sum())
    if s < 1e-15:
        return a.copy()
    return a / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 7.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.3) ** 2) + 0.8 * np.exp(-15.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _SmoProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs_short(a_init, deadline, min_time_remaining=30.0):
    """Quick L-BFGS warmup with few beta scales."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    betas = base_beta * np.array([1.0, 0.25, 0.05, 0.01, 0.002, 0.0005])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:

        max_iter = min(2000, max(30, int(remaining / (0.005 * max(1.0, n / 4096.0) * 10))))
        prob = _SmoProb(beta, n)

                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))

            step_scale = min(step_scale * 1.1, 0.5)
        else:
            step_scale = max(step_scale * 0.95, 0.01)

    return best_a, best_sc


def _fine_active_set(a_init, deadline, min_time_remaining=5.0):
    """Very fine active-set optimization with tiny epsilon."""
    n = len(a_init)
    a = a_init.copy()
    a = _normalize(a)
    best_a = a.copy()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    best_sc = 2.0 * n * c_max

    epsilon_frac = 0.001
    momentum = 0.0
    mom_dir = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() + min_time_remaining < deadline:
        it += 1

        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                a = best_a.copy()
                c = _full_convolve(a, a)
                c_max = float(c.max())
                no_improve = 0
                epsilon_frac = min(0.1, epsilon_frac * 10)
                mom_dir = np.zeros(n)
                continue

        threshold = c_max * (1.0 - epsilon_frac)
        S = np.where(c >= threshold)[0]
        if len(S) == 0:
            S = np.array([np.argmax(c)])
        # Cap active set size for stability
        if len(S) > 500:
            S = S[np.argsort(c[S])[::-1][:500]]

        w_S = np.zeros(2 * n - 1)
        w_S[S] = 1.0 / len(S)
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)
        g = 2.0 * h[n - 1:2 * n - 1]

        momentum = min(0.3, momentum + 0.0005)
        d = -g + momentum * mom_dir
        d = d - d.mean()
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            epsilon_frac = min(0.5, epsilon_frac * 2)
            continue
        d = d / d_norm

        da = _full_convolve(d, a)
        dd = _full_convolve(d, d)

        neg_mask = d < -1e-15
        if neg_mask.any():
            t_max_feas = float(np.min(-a[neg_mask] / d[neg_mask]))
        else:
            t_max_feas = 2.0
        t_max = min(t_max_feas, 0.1)

        if t_max < 1e-14:
            continue

        # Finer grid
        n_grid = 60
        t_grid = np.linspace(0, t_max, n_grid)
        ts = t_grid[:, None]
        f_grid = c[None, :] + 2 * ts * da[None, :] + ts * ts * dd[None, :]
        f_max = np.max(f_grid, axis=1)
        best_idx = int(np.argmin(f_max))
        best_t = t_grid[best_idx]
        best_f = f_max[best_idx]

        if best_t < 1e-13 or best_f >= c_max - 1e-16:
            no_improve += 1
            continue

        a = a + best_t * d
        a = np.maximum(a, 0.0)
        s_a = float(a.sum())
        if s_a > 1e-15:
            a = a / s_a
        c = c + 2 * best_t * da + best_t * best_t * dd
        c = c / (s_a * s_a)
        c_max = float(c.max())
        mom_dir = 0.95 * mom_dir + 0.05 * d

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    P1: L-BFGS warmup (smoothed objective)
    P2: Main active-set + exact line search
    P3: Fine active-set polish
    P4: Multi-n exploration
    P5: Final micro-polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS warmup ===
    print(f"\n[P1] L-BFGS warmup (t={time.monotonic()-t_start:.1f}s)")
    a_p1 = _run_lbfgs_short(best_seq, deadline, min_time_remaining=600.0)
    sc_p1 = _score(a_p1)
    if sc_p1 < best_score:
        best_seq = a_p1.copy()
        best_score = sc_p1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs]")
    else:
        print(f"  L-BFGS: no improvement (best={best_score:.10f})")

    # === Phase 2: Main active-set optimization ===
    print(f"\n[P2] Main active-set (t={time.monotonic()-t_start:.1f}s)")
    a_p2, sc_p2 = _active_set_main(best_seq, deadline, min_time_remaining=400.0)
    if sc_p2 < best_score:
        best_seq = a_p2.copy()
        best_score = sc_p2
        print(f"  NEW BEST: {best_score:.10f} [active_main]")
    else:
        print(f"  Active-main: no improvement (best={best_score:.10f})")

    # === Phase 2b: Second active-set round with different params ===
    if time.monotonic() < deadline - 300:
        print(f"\n[P2b] Active-set round 2 (t={time.monotonic()-t_start:.1f}s)")
        a_p2b, sc_p2b = _active_set_main(best_seq, deadline, min_time_remaining=300.0)
        if sc_p2b < best_score:
            best_seq = a_p2b.copy()
            best_score = sc_p2b
            print(f"  NEW BEST: {best_score:.10f} [active_main2]")
        else:
            print(f"  Active-main2: no improvement (best={best_score:.10f})")

    # === Phase 3: Fine active-set polish ===
    if time.monotonic() < deadline - 200:
        print(f"\n[P3] Fine polish (t={time.monotonic()-t_start:.1f}s)")
        a_p3, sc_p3 = _fine_active_set(best_seq, deadline, min_time_remaining=200.0)
        if sc_p3 < best_score:
            best_seq = a_p3.copy()
            best_score = sc_p3
            print(f"  NEW BEST: {best_score:.10f} [fine_polish]")
        else:
            print(f"  Fine: no improvement (best={best_score:.10f})")

    # === Phase 4: Try different n ===
    if time.monotonic() < deadline - 120:
        print(f"\n[P4] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
        n_curr = len(best_seq)
        n_candidates = sorted(set([n_curr,
                                   int(n_curr * 1.5),
                                   int(n_curr * 0.75),
                                   8192, 24576]))
        n_candidates = [x for x in n_candidates if 1024 <= x <= 32768 and x != n_curr]
        for n_try in n_candidates[:3]:
            if time.monotonic() > deadline - 80:
                break
            a_rs = _resample(best_seq, n_try)
            sc_rs = _score(a_rs)
            print(f"  n={n_try}: resampled score={sc_rs:.10f}")
            if sc_rs < best_score:
                best_seq = a_rs.copy()
                best_score = sc_rs
                print(f"  NEW BEST: {best_score:.10f} [resample n={n_try}]")
            # Quick active-set on resampled
            a_opt, sc_opt = _active_set_main(a_rs, deadline, min_time_remaining=80.0)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [active n={n_try}]")

    # === Phase 5: Final micro-polish ===
    if time.monotonic() < deadline - 20:
        print(f"\n[P5] Final micro (t={time.monotonic()-t_start:.1f}s)")
        a_fin, sc_fin = _fine_active_set(best_seq, deadline, min_time_remaining=20.0)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_micro]")

        # One more round of main active-set with very tight params
        if time.monotonic() < deadline - 5:
            a_fin2, sc_fin2 = _active_set_main(best_seq, deadline, min_time_remaining=5.0)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_active]")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
