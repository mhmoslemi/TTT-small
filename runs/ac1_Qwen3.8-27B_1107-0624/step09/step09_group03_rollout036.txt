"""
Optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Aggressive L-BFGS log-sum-exp continuation (extremely small beta)
2. Gradient-based peak shaving with backtracking line search
3. Multi-restart with adaptive noise
4. Alternative sequence lengths for diversity
5. Iterative refinement combining L-BFGS and gradient steps
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _ls_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _ls_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    out_len = n + len(w) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(w, N)
    result = np.fft.irfft(fa * fw, N)[:out_len]
    h = result[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _peak_shave_gradient(a, n):
    """Compute gradient for reducing the maximum peak using exact max."""
    c = _full_convolve(a, a)
    k_max = int(np.argmax(c))
    # Gradient of c[k_max] w.r.t. a is 2 * S_{k_max}(a)
    # S_k(a)[i] = a[k-i] if 0 <= k-i < n, else 0
    # This equals the cross-correlation approach
    w = np.zeros(2 * n - 1)
    w[k_max] = 1.0
    a_rev = a[::-1]
    out_len = n + (2 * n - 1) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(w, N)
    result = np.fft.irfft(fa * fw, N)[:out_len]
    h = result[n - 1:2 * n - 1]
    W = float(c[k_max])
    return 4.0 * h, W, c


def _peak_shave_step(a, max_steps=100, deadline=None):
    """Iterative peak shaving with backtracking line search."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for step in range(max_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find all near-max peaks (within 95% of max)
        threshold = c_max * 0.95
        peak_mask = c >= threshold
        n_peaks = int(peak_mask.sum())
        
        # Weight peaks: peaks closer to max get more weight
        w = np.zeros_like(c)
        w[peak_mask] = (c[peak_mask] / c_max) ** 4 / np.sum((c[peak_mask] / c_max) ** 4)
        
        # Compute gradient
        a_rev = a[::-1]
        out_len = n + len(w) - 1
        N = _next_pow2(out_len)
        fa = np.fft.rfft(a_rev, N)
        fw = np.fft.rfft(w, N)
        result = np.fft.irfft(fa * fw, N)[:out_len]
        h = result[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        
        grad = 4.0 * n * h
        grad_centered = grad - np.mean(grad)
        
        # Try several step sizes
        improved = False
        for alpha in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
            a_try = a - alpha * grad_centered
            a_try = np.maximum(a_try, 0)
            s_try = float(a_try.sum())
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-15:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        
        if not improved:
            # Try smaller steps
            for alpha in [0.0001, 0.0002, 0.0005]:
                a_try = a - alpha * grad_centered
                a_try = np.maximum(a_try, 0)
                s_try = float(a_try.sum())
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)

                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

    return best_a, best_sc


def _lbfgs_run(a_init, beta, deadline, max_iter=3000):
    """Run L-BFGS with given beta. Returns optimized a."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    remaining = deadline - time.monotonic()
    if remaining < 5:
        return a_init
    
    est_time_per_iter = 0.004 * max(1.0, n / 8192.0)
    actual_max_iter = min(max_iter, max(20, int(remaining / (est_time_per_iter * 8))))
    
    try:
        result = minimize(
            _ls_objective,
            z,
            args=(beta, n),
            jac=_ls_gradient,
            method='L-BFGS-B',
            options={
                'maxiter': actual_max_iter,
                'ftol': 1e-20,
                'gtol': 1e-15,
                'maxcor': 100,
                'maxls': 200,
            }
        )
        a_new = _softmax(result.x)
        return a_new
    except Exception:
        return a_init


def _lbfgs_continuation(a_init, deadline, min_remaining=60.0):

    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)

    current_sc = _score(current_a)

def _perturb(a, sigma, rng):
    """Add log-normal perturbation and renormalize."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_seq(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "u_shape":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "two_bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.15) ** 2) + np.exp(-20 * (t - 0.85) ** 2) + 0.05 * np.exp(-5 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n
