"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. L-BFGS with power-smoothed max and adaptive p
2. Nesterov-accelerated top-K gradient descent with momentum
3. Adaptive active set and line search
4. Multi-n exploration

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _grad_max_sm(a, p=100):
    """Gradient of the power-smoothed max of conv(a,a).
    S_p = (sum c_k^p)^(1/p)
    dS_p/da_j = (1/S_p^(p-1)) * sum_k c_k^(p-1) * dc_k/da_j
    where dc_k/da_j = a_{k-j} (if valid)
    """
    n = len(a)
    c = _full_convolve(a, a)
    c_p = c ** (p - 1)  # c_k^(p-1)
    S_p = (c ** p).sum() ** (1.0 / p)
    if S_p < 1e-300:
        return np.zeros(n), 0.0
    
    weights = c_p / (S_p ** (p - 1))
    
    # grad_j = sum_k weights[k] * a_{k-j} (if 0 <= k-j < n)
    # This is a cross-correlation of weights with a
    a_rev = a[::-1]
    h = _full_convolve(weights, a_rev)[n - 1:2 * n - 1]
    return 2.0 * h, float(S_p)


def _grad_topk(a, K=10):
    """Gradient of the average of top K values of conv(a,a)."""
    n = len(a)
    c = _full_convolve(a, a)
    idx = np.argpartition(c, -K)[-K:]
    weights = np.zeros(2 * n - 1)
    weights[idx] = 1.0 / K
    a_rev = a[::-1]
    h = _full_convolve(weights, a_rev)[n - 1:2 * n - 1]
    return 2.0 * h, float(c[idx].max())


def _nesterov_polish(a_init, deadline, min_time_remaining=30.0):

    with adaptive K and backtracking line search on the simplex."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    c = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c.max())
    
    K = min(5, n)
    momentum = 0.9
    lr = 0.05
    no_improve = 0
    total_steps = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            if K < min(200, n):
                K = min(K + 1, min(200, n))
        
        # Get top K
        if K >= 2 * n - 1:
            top_idx = np.arange(2 * n - 1)
        else:
            top_idx = np.argpartition(c, -K)[-K:]
        
        weights = np.zeros(2 * n - 1)
        weights[top_idx] = 1.0 / K
        
        a_rev = a[::-1]
        h = _full_convolve(weights, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)  # ensure sum-preserving
        gnorm = np.linalg.norm(grad)
        
        if gnorm < 1e-18:
            break
        
        # Nesterov step
        v = momentum * v - lr * grad / (gnorm + 1e-15)
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            v = np.zeros(n)
            continue
        a_new = a_new / s_new
        
        # Line search (backtracking)
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())
        
        if sc_new < current_sc:
            a = a_new
            total_steps += 1
        else:
            # Try smaller step
            v = momentum * v - 0.5 * lr * grad / (gnorm + 1e-15)
            a_new2 = a + v
            a_new2 = np.maximum(a_new2, 0.0)
            s_new2 = a_new2.sum()
            if s_new2 > 1e-15:
                a_new2 = a_new2 / s_new2
                c_new2 = _full_convolve(a_new2, a_new2)
                sc_new2 = 2.0 * n * float(c_new2.max())
                if sc_new2 < current_sc:
                    a = a_new2
                    v = v / 2
                    total_steps += 1
                else:
                    lr *= 0.9
                    no_improve += 1
            else:
                lr *= 0.9
                no_improve += 1
        
        if no_improve > 50:

            a = best_a.copy()
            v = np.zeros(n)
            lr = 0.05
            K = min(5, n)
            no_improve = 0
            momentum = 0.9
        
        if no_improve % 20 == 0 and no_improve > 0:
            momentum *= 0.95
            lr *= 0.8

def _direct_peak_shave(a_init, deadline, min_time_remaining=10.0):
    """Directly shave the maximum peak by adjusting specific elements."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
        
        # Find the max peak
        k_max = int(np.argmax(c))
        
        # The gradient of c_{k_max} w.r.t. a is:
        # dc_{k_max}/da_j = a_{k_max - j} (if 0 <= k_max - j < n)
        a_rev = a[::-1]
        w = np.zeros(2 * n - 1)
        w[k_max] = 1.0
        h = _full_convolve(w, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Try to reduce the peak
        best_alpha = 0
        best_sc_alpha = current_sc
        for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            a_try = a - alpha * grad / gnorm
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < best_sc_alpha:
                best_sc_alpha = sc_try
                best_alpha = alpha
        
        if best_alpha == 0:
            break
        
        a = (a - best_alpha * grad / gnorm)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
    
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "oscillatory":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + 0.8 * np.exp(-20.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

Uses: (1) L-BFGS with power-smoothed max, (2) Nesterov top-K gradient

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    best_a = a.copy()
    best_sc = _score(best_a)
    
    # p schedule: start large (smooth), decrease (sharp)
    p_schedule = [100, 50, 25, 15, 10, 7, 5, 4, 3, 2.5, 2, 1.8, 1.6, 1.4, 1.3, 1.2, 1.1, 1.05, 1.02]
    
    for pi, p in enumerate(p_schedule):
        if time.monotonic() + min_remaining > deadline:
            break
        
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break
        
        def obj(z_):
            a_ = _softmax(z_)
            c_ = _full_convolve(a_, a_)
            return float((c_ ** p).sum() ** (1.0 / p))
        
        def grad(z_):
            a_ = _softmax(z_)
            c_ = _full_convolve(a_, a_)
            S_p = (c_ ** p).sum() ** (1.0 / p)
            if S_p < 1e-300:
                return np.zeros(n)
            w = (c_ ** (p - 1)) / (S_p ** (p - 1))
            a_rev = a_[::-1]
            h = _full_convolve(w, a_rev)[n - 1:2 * n - 1]
            g_a = 2.0 * h
            dot = float(np.dot(a_, g_a))
            return a_ * (g_a - dot)
        
        max_iter = min(2000, max(10, int(remaining / 0.005)))
        
        try:
            res = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 60, 'maxls': 50}
            )
            a_new = _softmax(res.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_a = a_new.copy()
                best_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return best_a, best_sc


def _nesterov_topk(a_init, deadline, min_remaining=20.0):
    """Nesterov-accelerated gradient descent on top-K average of convolution."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    c = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c.max())
    
    K = 3
    momentum = 0.85
    lr = 0.1
    no_improve = 0
    it = 0
    
    while time.monotonic() + min_remaining < deadline:
        it += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        cur_sc = 2.0 * n * c_max
        
        if cur_sc < best_sc - 1e-16:
            best_sc = cur_sc
            best_a = a.copy()
            no_improve = 0
            K = min(K + 1, max(20, n // 100))
        
        # Top K indices
        if K >= 2 * n - 1:
            top_idx = np.arange(2 * n - 1)
        else:
            top_idx = np.argpartition(c, -K)[-K:]
        
        w = np.zeros(2 * n - 1)
        w[top_idx] = 1.0 / K
        a_rev = a[::-1]
        h = _full_convolve(w, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        
        if gnorm < 1e-18:
            break
        
        # Nesterov update
        v = momentum * v - lr * grad / (gnorm + 1e-15)
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            v = np.zeros(n)
            no_improve += 1
            continue
        a_new = a_new / s_new
        
        # Check improvement (every 3 iterations to save time)
        if it % 3 == 0 or no_improve == 0:
            c_new = _full_convolve(a_new, a_new)
            sc_new = 2.0 * n * float(c_new.max())
            if sc_new < cur_sc - 1e-17:
                a = a_new
            else:
                no_improve += 1
                if no_improve > 30:
                    lr *= 0.8
                    if no_improve > 100:
                        a = best_a.copy()
                        v = np.zeros(n)
                        lr = 0.1
                        momentum = 0.85
                        K = 3
                        no_improve = 0
                
        if no_improve % 50 == 0 and no_improve > 0:
            momentum = max(0.5, momentum * 0.95)
    
    return best_a, best_sc

def _active_set_refine(a_init, deadline, min_remaining=20.0):
    """Refine using active set: find the set of near-max peaks and
    optimize the average gradient direction with careful line search."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    K_schedule = [5, 10, 20, 50, 100, 200, 500]
    
    for K in K_schedule:
        if time.monotonic() + min_remaining > deadline:
            break
        if K >= 2 * n - 1:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Get top K
        top_idx = np.argpartition(c, -K)[-K:]
        w = np.zeros(2 * n - 1)
        w[top_idx] = 1.0 / K
        
        a_rev = a[::-1]
        h = _full_convolve(w, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        
        if gnorm < 1e-18:
            continue
        
        # Backtracking line search
        alpha = 0.01
        improved = False
        for _ in range(40):
            a_try = a - alpha * grad / gnorm
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < 2.0 * n * c_max - 1e-15:
                a = a_try
                best_sc = min(best_sc, sc_try)
                best_a = a.copy()
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            continue
    
    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, rng, strength=0.03):
    """Add a small asymmetric perturbation to break symmetry."""
    n = len(a)
    t = np.arange(n) / n
    pert = 0.5 * (t - 0.5) + 0.5 * rng.standard_normal(n)
    pert = pert - pert.mean()
    z = np.log(np.maximum(a, 1e-30))
    z = z + strength * pert
    z = z - z.mean()
    return _softmax(z)
