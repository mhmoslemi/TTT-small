    M = cp.Variable()

    active_k = set()
    
    a_val = np.array(a_init, dtype=float)
    a_val = np.maximum(a_val, 0)
    a_val /= a_val.sum()

        # Evaluate all conv values
        c = np.convolve(a_val, a_val)  # or use FFT
        k_star = np.argmax(c)
        M_val = c[k_star]

        new_active = set(np.where(c >= M_val - tol * M_val)[0])
        if new_active == active_k and iteration > 0:
            break
        active_k = new_active
        
        # Build and solve QP
        constraints = [a >= 0, cp.sum(a) == 1]
        for k in active_k:
            # conv(a,a)[k] = sum_j a[j]*a[k-j]

            break
        a_val = np.maximum(a.value, 0)
        a_val /= a_val.sum()
    
    return a_val

"""
Convex-structured optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines spectral shaping, anti-alignment deflation, active-set gradient,
and iterated local search with diverse perturbations.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _spectral_shape(a, alpha=0.05, n_iters=100):
    """Flatten the spectrum of a to reduce autocorrelation peaks."""
    n = len(a)
    N = 2 * n  # zero-padding for linear conv
    for _ in range(n_iters):
        A = np.fft.rfft(a, N)
        A_mag = np.abs(A)
        # Avoid division by zero
        A_mag_safe = np.maximum(A_mag, 1e-30)
        # Flatten: raise to power -alpha (reduces dynamic range)
        A_new = A * (A_mag_safe ** (-alpha))
        # Normalize to preserve energy-ish
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s
        if np.allclose(a_new, a, atol=1e-15):
            break
        a = a_new
    return a

def _anti_align_deflate(a, n_subtracts=10, eps=0.01):
    """Reduce the center peak by anti-aligning first and second halves."""
    n = len(a)
    half = n // 2
    a = a.copy()
    for _ in range(n_subtracts):
        # Compute correlation between first half and reversed second half
        # conv(a,a)[n-1] = sum_i a[i] * a[n-1-i]
        # For i < n/2: pairs (i, n-1-i) where i < n/2 and n-1-i >= n/2
        first = a[:half].copy()
        second = a[n-1:half-1:-1].copy()  # reversed second half
        if len(first) != len(second):
            min_len = min(len(first), len(second))
            first = first[:min_len]
            second = second[:min_len]
        corr = np.dot(first, second)
        if corr < 1e-15:
            break
        # Subtract a small multiple
        a[:min_len] -= eps * second / min_len
        a[n-1:n-1-min_len:-1] -= eps * first / min_len
        # Project
        a = _normalize(a)
    return a

def _active_set_gradient(a, c, tol_frac):
    """Compute weighted active-set gradient."""
    n = len(a)
    c_max = float(c.max())
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    n_active = int(active.sum())
    if n_active == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        n_active = 1
    
    eps = 1e-12 * c_max
    w = np.zeros_like(c)
    w[active] = 1.0 / (c_max - c[active] + eps)
    w = w / w.sum()
    
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad = 2.0 * h
    grad = grad - np.mean(grad)
    return grad

def _gradient_polish(a_init, deadline, tol_start=0.01, tol_end=0.00005,
                     lr_init=0.05, momentum=0.85):
    """Active-set gradient with momentum and adaptive step size."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    v = np.zeros(n)
    tol_frac = tol_start
    lr = lr_init
    no_improve = 0
    max_stall = 200
    
    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        grad = _active_set_gradient(a, c, tol_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        v = momentum * v - lr * grad
        alpha = 1.0
        improved = False
        for _ in range(50):
            a_try = _normalize(a + alpha * v)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            v = np.zeros(n)
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_end, tol_frac * 0.7)
                lr *= 0.8
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(tol_end, tol_frac * 0.3)
                lr = lr_init * 0.4
                no_improve = 0
        else:
            if tol_frac > tol_end:
                tol_frac = max(tol_end, tol_frac * 0.99)
            lr = min(lr_init, lr * 1.005)
    
    return best_a, _score(best_a)

def _coordinate_descent(a_init, deadline, n_sweeps=100):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_sc = _score(a)
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        # Random order for diversity
        idx = np.random.default_rng(sweep).permutation(n)
        for i in idx:
            if time.monotonic() > deadline:
                break
            # Optimize a[i] while keeping others fixed
            # conv(a,a)[k] = a[i]*a[k-i] + a[k-i]*a[i] + rest
            # = 2*a[i]*a[k-i] + rest (for k where both i and k-i are valid and i != k-i)
            # or a[i]^2 + rest (when i = k-i, i.e., k = 2i)
            # So conv(a,a)[k] as a function of a[i] is:
            #   if k = 2i: a[i]^2 + const_k
            #   elif k-i in [0,n-1] and k-i != i: 2*a[i]*a[k-i] + const_k
            #   else: const_k
            # We want to find a[i] >= 0 that minimizes max_k conv(a,a)[k]
            # subject to sum(a) = 1 (so sum of others = 1 - a[i])
            
            # This is a 1D minimization of max of linear/quadratic functions
            # For simplicity, use golden section or just try a few values

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)

def _spectral_shape(a, alpha=0.03, n_iters=200):

        A_mag = np.abs(A)
        A_mag_safe = np.maximum(A_mag, 1e-30)
        # Flatten: reduce dynamic range of |A|
        # Raising to -alpha compresses the dynamic range
        flatten_factor = A_mag_safe ** (-alpha)
        A_new = A * flatten_factor

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]

    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _quick_optimize(a_init, deadline, min_time_remaining=10.0):
    """Combined optimization: spectral shape + gradient polish."""
    a = a_init.copy()
    # Quick spectral shaping
    if time.monotonic() < deadline - min_time_remaining - 5:
        a = _spectral_shape(a, alpha=0.01, n_iters=30)
    # Gradient polish
    if time.monotonic() < deadline - min_time_remaining:
        a2, sc2 = _gradient_polish(a, deadline - min_time_remaining,
                                   tol_start=0.003, tol_end=0.0003,
                                   lr_init=0.02, momentum=0.82)
        if sc2 < _score(a):
            a = a2
    return a


def _ils_step(a_current, a_best, sc_best, deadline, rng, min_time_remaining=30.0,
              n_strategies=8):
    """One round of ILS with diverse perturbations."""
    n = len(a_current)
    current_sc = _score(a_current)

    strategies = [
        ('spectral', 0.02),
        ('anti_align', 0.01),
        ('global_small', 0.008),
        ('block', 0.03),
        ('spectral', 0.05),
        ('sparse', 0.05),
        ('shrink_expand', 0.03),
        ('global_med', 0.015),
    ]

    for i in range(n_strategies):
        if time.monotonic() + min_time_remaining > deadline:
            break
        kind, sigma = strategies[i % len(strategies)]

        a_perturbed = a_current.copy()

        if kind == 'spectral':
            a_perturbed = _spectral_shape(a_current, alpha=sigma, n_iters=50)
        elif kind == 'anti_align':
            a_perturbed = _anti_align_deflate(a_current, n_steps=30, eps=sigma)
        elif kind == 'global_small' or kind == 'global_med':
            z = np.log(np.maximum(a_current, 1e-30))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_perturbed = _softmax(z)
        elif kind == 'block':
            block_size = max(32, n // 10)
            start = int(rng.integers(0, n - block_size))
            a_perturbed = a_current.copy()
            z = np.log(np.maximum(a_perturbed[start:start + block_size], 1e-30))
            z = z + sigma * rng.standard_normal(block_size)
            a_perturbed[start:start + block_size] = np.exp(z - np.max(z))
            a_perturbed = _normalize(a_perturbed)
        elif kind == 'sparse':
            n_perturb = max(20, n // 20)
            idx = rng.choice(n, size=n_perturb, replace=False)
            a_perturbed = a_current.copy()
            factors = np.exp(sigma * rng.standard_normal(n_perturb))
            a_perturbed[idx] *= factors
            a_perturbed = _normalize(a_perturbed)
        elif kind == 'shrink_expand':
            a_perturbed = _shrink_expand(a_current, center_frac=sigma * 10, eps=0.02)

        opt_deadline = min(deadline - min_time_remaining, time.monotonic() + 6)
        a_opt = _quick_optimize(a_perturbed, opt_deadline, min_time_remaining=2)
        sc_opt = _score(a_opt)

        if sc_opt < current_sc - 1e-12:
            a_current = a_opt.copy()
            current_sc = sc_opt
        if sc_opt < sc_best - 1e-12:
            a_best = a_opt.copy()
            sc_best = sc_opt
            print(f"    ILS BEST: {sc_best:.10f} ({kind}) t={time.monotonic()-t_global:.0f}s")

    return a_current, a_best, sc_best

def propose_candidate(seed=42, budget_s=1800):
    global t_global
    t_global = time.monotonic()
    deadline = t_global + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = _score(parent)
        n_main = n_parent
        print(f"Parent: n={n_parent}, score={best_score:.10f}")
    else:
        n_main = 16384
        best_seq = _make_profile(n_main, "uniform")
        best_score = _score(best_seq)
        print(f"No parent, uniform n={n_main}, score={best_score:.10f}")

    # === Phase 1: Spectral shaping + anti-alignment on parent (200s) ===
    print(f"\n[P1] Spectral + anti-align (t={time.monotonic()-t_global:.1f}s)")
    p1_deadline = min(deadline - 800, t_global + 200)

    a1 = _spectral_shape(best_seq, alpha=0.02, n_iters=100)
    sc1 = _score(a1)
    print(f"  After spectral: {sc1:.10f}")
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1

    a2 = _anti_align_deflate(best_seq, n_steps=30, eps=0.003)
    sc2 = _score(a2)
    print(f"  After anti-align: {sc2:.10f}")
    if sc2 < best_score:
        best_seq = a2.copy()
        best_score = sc2

    # Gradient polish
    a3, sc3 = _gradient_polish(best_seq, p1_deadline, tol_start=0.005,
                               tol_end=0.0002, lr_init=0.03, momentum=0.83)
    print(f"  After polish: {sc3:.10f}")
    if sc3 < best_score:
        best_seq = a3.copy()
        best_score = sc3

    # === Phase 2: Multi-start initialization (250s) ===
    print(f"\n[P2] Multi-start (t={time.monotonic()-t_global:.1f}s)")
    p2_deadline = min(deadline - 600, t_global + 450)

    starts = []
    starts.append(('parent', best_seq.copy()))
    for kind in ['hann', 'gaussian', 'hann2', 'bimodal', 'cosine_skew']:
        starts.append((kind, _make_profile(n_main, kind)))

    for name, a_init in starts:
        if time.monotonic() > p2_deadline - 30:
            break
        a_opt = _quick_optimize(a_init, p2_deadline, min_time_remaining=20)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [{name}] t={time.monotonic()-t_global:.0f}s")

    # === Phase 3: ILS round 1 (350s) ===
    print(f"\n[P3] ILS round 1 (t={time.monotonic()-t_global:.1f}s)")
    p3_deadline = min(deadline - 400, t_global + 800)
    rounds = 0
    while time.monotonic() + 30 < p3_deadline:
        a_cur, a_best, sc_best = _ils_step(best_seq, best_seq.copy(), best_score,
                                           p3_deadline, rng, min_time_remaining=20,
                                           n_strategies=6)
        if sc_best < best_score:
            best_seq = a_best.copy()
            best_score = sc_best
        best_seq = a_cur.copy()
        rounds += 1

    print(f"  After ILS1: best={best_score:.10f}, rounds={rounds}")

    # === Phase 4: L-BFGS on smoothed objective (200s) ===
    print(f"\n[P4] L-BFGS smoothed (t={time.monotonic()-t_global:.1f}s)")
    if time.monotonic() < deadline - 250:
        lbfgs_deadline = min(deadline - 250, time.monotonic() + 200)
        a_lbfgs = _lbfgs_smoothed(best_seq, lbfgs_deadline, min_time_remaining=20)
        sc_lbfgs = _score(a_lbfgs)
        print(f"  L-BFGS: {sc_lbfgs:.10f}")
        if sc_lbfgs < best_score:
            best_seq = a_lbfgs.copy()
            best_score = sc_lbfgs
            print(f"  NEW BEST: {best_score:.10f} [lbfgs]")

    # === Phase 5: Multi-n exploration (300s) ===
    print(f"\n[P5] Multi-n (t={time.monotonic()-t_global:.1f}s)")
    p5_deadline = min(deadline - 150, time.monotonic() + 300)
    n_candidates = [n_main, n_main + 4096, n_main + 8192, n_main - 4096,
                    int(n_main * 1.2), int(n_main * 0.8), 32768, 40960]
    seen = set()
    for n_try in n_candidates:
        if time.monotonic() > p5_deadline - 30:
            break
        if n_try in seen or n_try < 1024:
            continue
        seen.add(n_try)
        a_resampled = _resample(best_seq, n_try)
        sc_init = _score(a_resampled)
        # Quick optimize
        a_opt = _quick_optimize(a_resampled, p5_deadline, min_time_remaining=10)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n]")
        else:
            print(f"  n={n_try}: {sc_opt:.10f} (init {sc_init:.10f})")

    # === Phase 6: Intensive ILS round 2 (200s) ===
    print(f"\n[P6] ILS round 2 (t={time.monotonic()-t_global:.1f}s)")
    if time.monotonic() < deadline - 100:
        p6_deadline = min(deadline - 100, time.monotonic() + 200)
        rng2 = np.random.default_rng(int(seed) + 777)
        rounds2 = 0
        while time.monotonic() + 20 < p6_deadline:
            a_cur, a_best, sc_best = _ils_step(best_seq, best_seq.copy(), best_score,
                                               p6_deadline, rng2, min_time_remaining=15,
                                               n_strategies=5)
            if sc_best < best_score:
                best_seq = a_best.copy()
                best_score = sc_best
            best_seq = a_cur.copy()
            rounds2 += 1
        print(f"  After ILS2: best={best_score:.10f}, rounds={rounds2}")

    # === Phase 7: Final ultra-fine polish (remaining time) ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_global:.1f}s)")
    if time.monotonic() < deadline - 5:
        # Very tight gradient polish
        a_fin, sc_fin = _gradient_polish(best_seq, deadline - 3,
                                         tol_start=0.001, tol_end=0.000003,
                                         lr_init=0.01, momentum=0.8)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [fin_grad]")

        # Spectral fine-tune
        if time.monotonic() < deadline - 3:
            a_fin2 = _spectral_shape(best_seq, alpha=0.005, n_iters=50)
            sc_fin2 = _score(a_fin2)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [fin_spec]")

            a_fin3 = _anti_align_deflate(best_seq, n_steps=20, eps=0.002)
            sc_fin3 = _score(a_fin3)
            if sc_fin3 < best_score:
                best_seq = a_fin3.copy()
                best_score = sc_fin3
                print(f"  NEW BEST: {best_score:.10f} [fin_aa]")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_global:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    return [float(x) for x in fallback]
