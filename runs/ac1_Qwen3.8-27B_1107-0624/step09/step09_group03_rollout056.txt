def _cg_solve(matvec, rhs, max_cg=300, tol=1e-14):
    """Conjugate gradient solver."""
    n = len(rhs)
    x = np.zeros(n)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)

def _fista_optimize(a_init, deadline, min_time_remaining=30.0):
    """
    FISTA with log-sum-exp smoothing and decreasing beta.
    Uses backtracking line search.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_score = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())

    # Beta schedule
    c_sorted = np.sort(c)[::-1]
    if len(c_sorted) > 100:
        gap = c_sorted[0] - c_sorted[100]
    else:
        gap = c_sorted[0] - c_sorted[-1]
    beta_start = max(gap * 0.4, c_max * 0.005)
    beta_end = max(c_max * 5e-9, 1e-18)

    remaining_total = deadline - time.monotonic() - min_time_remaining
    if remaining_total < 5:
        return best_a, best_score

    n_betas = max(8, min(35, int(remaining_total / 5.0)))
    log_betas = np.linspace(math.log(beta_start), math.log(beta_end), n_betas)
    betas = np.exp(log_betas)

    for bi, beta in enumerate(betas):
        if time.monotonic() > deadline - min_time_remaining:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # FISTA parameters
        x = a.copy()
        y = a.copy()
        t_k = 1.0
        # Initial step size heuristic
        step = 1.0 / (8.0 * n * beta + 1e-30)

        c_y = _full_convolve(y, y)
        f_y = 2.0 * n * float(c_y.max())

        max_iter = max(50, int(remaining / 0.02))

        for it in range(max_iter):
            if time.monotonic() > deadline - min_time_remaining:
                break

            # Compute gradient at y
            grad_y = _grad_smoothed(y, beta)

            # Backtracking
            step_local = step
            accepted = False
            for bt in range(30):
                x_new = y - step_local * grad_y
                x_new = _project_simplex(x_new)
                s_new = x_new.sum()
                if s_new < 1e-15:
                    step_local *= 0.5
                    continue
                x_new = x_new / s_new

                # Evaluate objective
                c_new = _full_convolve(x_new, x_new)
                f_new_raw = float(c_new.max())

                # Simple sufficient decrease (Armijo)
                if f_new_raw <= f_y / (2.0 * n) + 0.5 * step_local * np.dot(grad_y, x_new - y) / (2.0 * n) + 1e-30:
                    accepted = True
                    break
                step_local *= 0.5

            if not accepted:
                # Just take a small step
                x_new = y - step_local * grad_y
                x_new = _project_simplex(x_new)
                s_new = x_new.sum()
                if s_new > 1e-15:
                    x_new = x_new / s_new
                else:
                    x_new = a.copy()

            a = x_new
            c_a = _full_convolve(a, a)
            f_a = 2.0 * n * float(c_a.max())

            # Nesterov update
            t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = a + (t_k - 1.0) / t_new * (a - x)
            y = _project_simplex(y)
            t_k = t_new
            x = a.copy()
            f_y = f_a

            # Track best
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        a = best_a.copy()
        s = a.sum()
        if s > 1e-15:
            a = a / s

    return best_a, best_score


def _multiplicative_weights(a_init, deadline, min_time_remaining=30.0):
    """
    Multiplicative weights (mirror descent) on the simplex with smoothed gradient.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_score = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())

    # Beta schedule (slower decrease)
    c_sorted = np.sort(c)[::-1]
    if len(c_sorted) > 100:
        gap = c_sorted[0] - c_sorted[100]
    else:
        gap = c_sorted[0] - c_sorted[-1]
    beta_start = max(gap * 0.3, c_max * 0.01)
    beta_end = max(c_max * 1e-7, 1e-18)

    remaining_total = deadline - time.monotonic() - min_time_remaining
    if remaining_total < 5:
        return best_a, best_score

    n_betas = max(6, min(25, int(remaining_total / 4.0)))
    log_betas = np.linspace(math.log(beta_start), math.log(beta_end), n_betas)
    betas = np.exp(log_betas)

    for bi, beta in enumerate(betas):
        if time.monotonic() > deadline - min_time_remaining:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        max_iter = max(50, int(remaining / 0.03))
        # Step size: start moderate, decrease over iterations
        eta = 1.0 / (4.0 * n * beta)

        for it in range(max_iter):
            if time.monotonic() > deadline - min_time_remaining:
                break

            grad = _grad_smoothed(a, beta)
            # Multiplicative update
            log_a = np.log(np.maximum(a, 1e-30))
            log_a += -eta * grad
            log_a -= log_a.mean()
            a = _softmax(log_a)

            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        a = best_a.copy()

    return best_a, best_score


def _newton_step_for_active(a, c, c_max, active_mask):
    """
    Compute Newton step given a specific active mask.
    Returns (dx, success).
    """
    n = len(a)
    if not active_mask.any():
        active_mask = np.zeros_like(c)
        active_mask[int(np.argmax(c))] = True

    m = int(active_mask.sum())
    if m < 2 or m > 50 * n:
        return None

    A_vec = active_mask.astype(np.float64) / m
    n_eff = n

    # Gradient: g[i] = 2 * sum_k A_vec[k] * a[k-i]
    g = 2.0 * _conv_grad(a, A_vec)
    g_t = g - np.mean(g)

    # Hessian (positive part): H*v[i] = 2 * sum_k A_vec[k] * v[k-i]
    mu = 1e-8 / m + 1e-12
    A_vec_local = A_vec
    n_local = n_eff
    mu_local = mu

    def _hess_vp(v):
        return 2.0 * _conv_grad(v, A_vec_local) + mu_local * v

    # Solve (H + mu*I) u = g_t
    u = _cg_solve(_hess_vp, g_t, max_cg=300, tol=1e-14)

    # Solve (H + mu*I) w = 1 (for the Lagrange multiplier)
    w = _cg_solve(_hess_vp, np.ones(n), max_cg=300, tol=1e-14)
    w_sum = float(w.sum())

    if abs(w_sum) < 1e-15:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w

    dx = dx - np.mean(dx)
    # Ensure non-negativity direction: if a[i] is near 0, clip dx[i] to be >= 0
    tiny = 1e-15
    dx[a < tiny] = np.maximum(dx[a < tiny], 0)
    dx = dx - np.mean(dx)

    return dx


def _active_set_refine(a_init, deadline, min_time_remaining=30.0):
    """
    Active-set Newton method with multiple tolerance levels.
    Tries many different active set definitions.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_score = _score(a)

    max_outer = 200
    for outer in range(max_outer):
        if time.monotonic() > deadline - min_time_remaining:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_score = 2.0 * n * c_max

        # Try multiple tolerance levels for active set
        best_improvement = 0.0
        best_a_new = None

        tol_list = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
        # Also try "top-k" active sets
        topk_list = [10, 50, 100, 200, 500, 1000, 2000, min(5000, 2 * n)]

        # Tolerance-based active sets
        for tol_frac in tol_list:
            if time.monotonic() > deadline - min_time_remaining:
                break
            threshold = c_max * (1.0 - tol_frac)
            active = c >= threshold
            dx = _newton_step_for_active(a, c, c_max, active)
            if dx is None:
                continue

            # Line search
            t = 1.0
            for ls in range(60):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < c_max - 1e-15:
                    improvement = c_max - f_try
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_a_new = a_try.copy()
                    break
                t *= 0.5

        # Top-k active sets
        for k in topk_list:
            if time.monotonic() > deadline - min_time_remaining:
                break
            k = min(k, 2 * n - 1)
            idx = np.argsort(c)[::-1][:k]
            active = np.zeros_like(c)
            active[idx] = True
            dx = _newton_step_for_active(a, c, c_max, active)
            if dx is None:
                continue

            t = 1.0
            for ls in range(60):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < c_max - 1e-15:
                    improvement = c_max - f_try
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_a_new = a_try.copy()
                    break
                t *= 0.5

        if best_a_new is None:
            break

        a = best_a_new
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Convergence check
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-16:
            break

    return best_a, best_score


def _subgradient_explore(a_init, deadline, min_time_remaining=30.0):
    """
    Projected subgradient method with momentum for diversification.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_score = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())

    # Decreasing step size
    L0 = 4.0 * n * c_max
    max_iter = max(100, int((deadline - time.monotonic() - min_time_remaining) / 0.02))
    momentum = 0.9
    v = np.zeros(n)

    for it in range(max_iter):
        if time.monotonic() > deadline - min_time_remaining:
            break

        # Subgradient of max: pick argmax (or random from top few)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Use smoothed gradient for better direction
        k_top = min(20, len(c))
        top_idx = np.argsort(c)[::-1][:k_top]
        p = np.zeros_like(c)
        p[top_idx] = 1.0 / k_top
        grad = 4.0 * n * _conv_grad(a, p)

        # Step size
        step = 1.0 / (L0 * math.sqrt(it + 1))

        v = momentum * v - step * grad
        a_new = a + v
        a_new = _project_simplex(a_new)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        a = a_new

        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _interpolate_seq(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_initial(n, kind="uniform"):
    """Create initial sequence of given type."""
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.5 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "block":
        a = np.zeros(n)
        block = n // 3
        a[:block] = 1.0
        a[2 * block:] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        return a
    else:
        return np.ones(n) / n


def _deflation_step(a, deadline, n_attempts=40, min_time_remaining=30.0):
    """
    Multi-peak deflation: reduce all significant peaks simultaneously.
    """
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)

    for attempt in range(n_attempts):
        if time.monotonic() > deadline - min_time_remaining:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find significant peaks
        threshold = c_max * 0.98
        peak_mask = c >= threshold
        n_peaks = int(peak_mask.sum())
        if n_peaks == 0:
            peak_mask[np.argmax(c)] = True
            n_peaks = 1

        w = np.zeros_like(c)
        w[peak_mask] = 1.0 / n_peaks
        grad = 4.0 * n * _conv_grad(a, w)
        g_t = grad - np.mean(grad)

        applied = False
        for strength in [0.0001, 0.0003, 0.001, 0.003, 0.01, 0.03]:
            a_try = a - strength * g_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_score - 1e-15:
                best_score = sc_try
                best_a = a_try.copy()
                applied = True
                break

        if not applied:
            break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    1. FISTA with aggressive smoothing
    2. Multiplicative weights
    3. Active-set Newton refinement
    4. Subgradient exploration
    5. Multi-scale (n=16384, n=32768)
    6. Diversified restarts
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.12f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={len(parent)}, score={sc_parent:.12f}")
        update_best(parent, sc_parent, "parent")
    else:
        parent = _make_initial(16384, "uniform")
        update_best(parent, _score(parent), "init_uniform")

    n_parent = len(parent) if parent is not None else 16384

    # === Phase 1: FISTA on parent (main workhorse) ===
    print(f"\n[Phase 1] FISTA optimization (t={time.monotonic() - t_start:.1f}s)")
    fista_dl = min(deadline - 600, time.monotonic() + 400)
    if best_seq is not None:
        a_fista, sc_fista = _fista_optimize(best_seq, fista_dl, min_time_remaining=600.0)
        update_best(a_fista, sc_fista, "fista")

    # === Phase 2: Active-set Newton refinement ===
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        as_dl = min(deadline - 400, time.monotonic() + 250)
        a_as, sc_as = _active_set_refine(best_seq, as_dl, min_time_remaining=400.0)
        update_best(a_as, sc_as, "active_newton")

    # === Phase 3: Multiplicative weights ===
    print(f"\n[Phase 3] Multiplicative weights (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        mw_dl = min(deadline - 300, time.monotonic() + 150)
        a_mw, sc_mw = _multiplicative_weights(best_seq, mw_dl, min_time_remaining=300.0)
        update_best(a_mw, sc_mw, "mult_weights")

    # === Phase 4: Deflation ===
    print(f"\n[Phase 4] Multi-peak deflation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        def_dl = min(deadline - 250, time.monotonic() + 100)
        a_def, sc_def = _deflation_step(best_seq, def_dl, n_attempts=50, min_time_remaining=250.0)
        update_best(a_def, sc_def, "deflation")

    # === Phase 5: Active-set Newton again (after deflation) ===
    print(f"\n[Phase 5] Active-set Newton round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        as2_dl = min(deadline - 200, time.monotonic() + 150)
        a_as2, sc_as2 = _active_set_refine(best_seq, as2_dl, min_time_remaining=200.0)
        update_best(a_as2, sc_as2, "active_newton_2")

    # === Phase 6: Try n=32768 ===
    print(f"\n[Phase 6] n=32768 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 200 and best_seq is not None:
        n_large = 32768
        a_large = _interpolate_seq(best_seq, n_large)
        sc_large_init = _score(a_large)
        print(f"  n={n_large} init: {sc_large_init:.12f}")
        update_best(a_large, sc_large_init, "n32768_interp")

        # FISTA on large
        fista3_dl = min(deadline - 120, time.monotonic() + 100)
        a_large_opt, sc_large = _fista_optimize(a_large, fista3_dl, min_time_remaining=120.0)
        update_best(a_large_opt, sc_large, "n32768_fista")

        # Active-set on large
        if time.monotonic() < deadline - 60:
            as3_dl = min(deadline - 40, time.monotonic() + 50)
            a_large_as, sc_large_as = _active_set_refine(a_large_opt, as3_dl, min_time_remaining=40.0)
            update_best(a_large_as, sc_large_as, "n32768_as")

    # === Phase 7: Subgradient exploration + re-optimize ===
    print(f"\n[Phase 7] Diversified exploration (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        # Perturb and re-optimize
        for sigma in [0.003, 0.01, 0.03]:
            if time.monotonic() > deadline - 60:
                break
            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z += sigma * rng.standard_normal(n)
            z -= z.mean()
            a_pert = _softmax(z)
            sc_pert = _score(a_pert)

            if sc_pert < best_score:
                update_best(a_pert, sc_pert, f"perturb_{sigma}")

            # Quick FISTA on perturbed
            if time.monotonic() < deadline - 40:
                pert_dl = min(deadline - 30, time.monotonic() + 40)
                a_pert_opt, sc_pert_opt = _fista_optimize(a_pert, pert_dl, min_time_remaining=30.0)
                update_best(a_pert_opt, sc_pert_opt, f"perturb_fista_{sigma}")

    # === Phase 8: Final active-set polish ===
    print(f"\n[Phase 8] Final active-set polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        final_dl = deadline - 5
        a_final, sc_final = _active_set_refine(best_seq, final_dl, min_time_remaining=5.0)
        update_best(a_final, sc_final, "final_polish")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_initial(256, "uniform")
        sc_end = _score(final)
        print(f"FINAL: score={sc_end:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_initial(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
