"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses proximal gradient with Adam optimizer, adaptive soft-max beta,
spectral perturbation for escape, multi-n exploration, and L-BFGS refinement.
Key improvements over previous: Adam optimizer per-coordinate adaptive LR,
trapped beta schedule, frequency-domain perturbation, better initialization.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z(z, beta):
    """Compute gradient of 2n * beta * log(sum(exp(c_k/beta))) w.r.t. z."""
    a = _softmax(z)
    n = len(a)
    c = _full_convolve(a, a)
    c_shifted = (c - c.max()) / beta
    e = np.exp(np.clip(c_shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = e / s
    # grad_a_i = 2 * conv(w, a_rev)[n-1+i]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad_a = 2.0 * wc[n - 1:2 * n - 1]
    # grad_z_i = a_i * (grad_a_i - sum_j grad_a_j * a_j)
    ga_dot_a = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - ga_dot_a)
    return 4.0 * n * grad_z


def _compute_obj_z(z, beta):
    """Compute 2n * beta * log(sum(exp(c_k/beta))) for soft-max approximation."""
    a = _softmax(z)
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_shifted = (c - c_max) / beta
    e = np.exp(np.clip(c_shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _adam_proxgrad(z_init, deadline, max_steps=20000):
    """Proximal gradient with Adam optimizer and adaptive beta."""
    n = len(z_init)
    z = z_init.copy()
    best_z = z.copy()
    best_a = _softmax(z)
    best_sc = _score(best_a)
    
    beta_init = max(1e-4, 0.01 * _full_convolve(best_a, best_a).max())
    beta = beta_init
    beta_min = beta_init * 0.001
    
    lr = 0.01
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 1
    beta_1, beta_2 = 0.9, 0.999
    
    patience = 0
    max_patience = 200
    
    for step in range(max_steps):
        if time.monotonic() > deadline - 5:
            break
        
        # Compute gradient
        grad = _compute_grad_z(z, beta)
        
        # Adam update
        m = beta_1 * m + (1 - beta_1) * grad
        v = beta_2 * v + (1 - beta_2) * (grad ** 2)
        m_hat = m / (1 - beta_1 ** t_count)
        v_hat = v / (1 - beta_2 ** t_count)
        z_new = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        z_new = z_new - np.mean(z_new)
        z_new = np.clip(z_new, -100.0, 100.0)
        t_count += 1
        
        # Evaluate
        a_new = _softmax(z_new)
        sc_new = _score(a_new)
        
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_z = z_new.copy()
            best_a = a_new.copy()
            patience = 0
            # Decay learning rate slowly on improvement
            lr *= 0.999
        else:
            patience += 1
            if patience > max_patience:
                # Reset to best and try different beta
                z = best_z.copy()
                beta = max(beta * 0.5, beta_min)
                lr = max(lr * 0.5, 1e-6)
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 1
                patience = 0
                if beta <= beta_min * 1.1:

            ee = np.exp(np.clip(shifted, -700, 0))
            ss = ee.sum()
            if ss < 1e-300 or not np.isfinite(ss):
                return 2.0 * nn * cm
            return 2.0 * nn * (cm + b * np.log(ss))
        
        def grad_fn(z_inner, b=beta, nn=n):
            return _compute_grad_z(z_inner, b)
        
        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 80, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
        
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]
        
        if new_gap < 1e-8 * c_max_new:
            beta *= 0.08
        elif new_gap < 1e-5 * c_max_new:
            beta *= 0.15
        elif new_gap < 1e-3 * c_max_new:
            beta *= 0.3
        else:
            beta *= 0.5
        
        beta = min(beta, c_max_new * 0.0002)
    
    return current_a


def _spectral_perturb(a, amount, rng):
    """Perturb in frequency domain to escape local minima."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    # Add noise to the spectrum
    noise = rng.standard_normal(A.shape).astype(np.float64) + 1j * rng.standard_normal(A.shape).astype(np.float64)
    A_new = A + amount * noise
    # Take inverse and clip
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "skew":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.7) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-5.0 * (t - 0.5) ** 2) + 0.02
        return a / a.sum()
    elif kind == "power":
        t = np.linspace(0, 1, n)
        a = t ** 0.3 * (1 - t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _subgrad_momentum(a_init, deadline, max_steps=5000):
    """Subgradient method with momentum in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.015
    momentum = 0.9
    v = np.zeros(n)
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive active set
        for tol_frac in [0.00005, 0.0002, 0.0005, 0.001, 0.003, 0.01]:
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            if active.sum() >= 3:
                break
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        W = np.zeros_like(c)
        W[active] = 1.0 / active.sum()
        
        # Compute gradient
        a_rev = a[::-1]
        wc = _full_convolve(W, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(W, c))
        grad_a = 2.0 * a * (h - W_val)
        grad_a = grad_a - np.mean(grad_a)
        grad_norm = np.linalg.norm(grad_a)
        if grad_norm < 1e-15:
            break
        
        v = momentum * v - lr * grad_a
        z = z + v
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        
        if i % 30 == 29:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, 0.04)
            else:
                patience += 1
                lr *= 0.85
                momentum = min(momentum + 0.005, 0.95)
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)
                    patience = 0
                    lr = 0.01
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc
