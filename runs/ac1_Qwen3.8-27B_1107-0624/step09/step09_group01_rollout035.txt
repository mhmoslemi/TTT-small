"""
Improved hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous version:
1. Adaptive n exploration with better time allocation
2. More aggressive and adaptive log-sum-exp smoothing
3. Exact active-set KKT polish with constraint violation checking
4. Multi-scale perturbation for escaping local minima
5. Better time management to maximize within 1800s budget

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute the Hessian-vector product for the weighted quadratic form.
    H = sum_k W[k] * T_k where T_k is the shift matrix for lag k.
    (H v)_i = sum_k W[k] * (T_k v)_i = sum_k W[k] * v_{k-i} (with bounds)
    This is: 2 * (middle part of convolve(W, v_reversed))
    """
    v_rev = v[::-1]
    out_len = len(W) + len(v_rev) - 1
    N = _next_pow2(out_len)
    F_v = np.fft.rfft(v_rev, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_v * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve_hankel(W, n, rhs, max_iter=500, tol=1e-12):
    """Solve H x = rhs using conjugate gradient, where H is the Hankel/Toeplitz matrix
    defined by the weight vector W."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            # Try restarted CG
            x_new = x + (rs / max(pHp, 1e-30)) * p
            return x_new
        alpha = rs / pHp
        x = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * (1 + 1e-10):
            break
        if rs_new < tol * tol * rs0:
            r = r_new
            break
        p = (rs_new / rs) * p + r_new
        r = r_new
        rs = rs_new
    return x


def _compute_gradient(a, W, n):
    """Compute gradient of sum_k W[k] * (a*a)[k] = a^T H a.
    Gradient = 2 * H a where (Ha)_i = sum_k W[k] * a_{k-i}."""
    a_rev = a[::-1]
    wc = _full_convolve(W, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _kkt_newton(a_init, deadline, max_rounds=300, k_schedule=None):
    """KKT-based Newton method for the active set problem.
    At each round, select top-K active constraints, solve KKT via CG,
    take a line search step that decreases max(conv)."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    if k_schedule is None:
        k_schedule = [1, 5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        K = k_schedule[min(rnd, len(k_schedule) - 1)]
        K = min(K, 2 * n - 1)
        idx = np.argpartition(c, -K)[-K:]
        W = np.zeros(2 * n - 1)
        W[idx] = 1.0 / K

        # Compute gradient of the weighted objective
        g = _compute_gradient(a, W, n)
        g_centered = g - np.mean(g)
        gn = float(np.linalg.norm(g_centered))
        if gn < 1e-14 * n:
            break

        # Solve H dx = g for Newton direction (with constraint sum(dx)=0)
        Hg = _cg_solve_hankel(W, n, g, max_iter=300)
        H1 = _cg_solve_hankel(W, n, np.ones(n), max_iter=300)
        H1_denom = float(np.ones(n) @ H1)

        if abs(H1_denom) < 1e-25 or not np.isfinite(H1_denom):
            dx = -g_centered / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / H1_denom
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_centered / (gn + 1e-30)

        # Line search
        alpha = 1.0
        step_done = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1.0 - 0.005 * alpha) - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not step_done:
            # Fallback: subgradient step
            for trial in [0.01, 0.005, 0.001, 0.0005]:
                a_try = a - trial * g_centered / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 10:
            break

    return best_a, best_sc


def _lbfgs_smooth(a_init, deadline, min_remaining=20.0, beta_init=None):
    """L-BFGS optimization using log-sum-exp smoothing of the max.
    Progressively decreases the smoothing parameter beta."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if beta_init is None:
        c_sorted = np.sort(c_cur)[::-1]
        gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
        beta_init = max(gap * 0.5, max_c * 0.01)

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    current_a = a_init.copy()
    current_sc = _score(current_a)
    beta = beta_init

    obj_count = 0

    while beta > max_c * 5e-6:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break

        scale = max(1.0, n / 8192.0)
        est_time_per_iter = 0.003 * scale
        max_iter = min(3000, max(20, int(remaining / (est_time_per_iter * 20))))

        def obj_fn(z, beta=beta, n=n):
            nonlocal obj_count
            obj_count += 1
            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                return 2.0 * n * c_max
            return 2.0 * n * (c_max + beta * np.log(s))

        def grad_fn(z, beta=beta, n=n):
            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W_val = float(np.dot(w, c))
            return 4.0 * n * a * (h - W_val)

        try:
            result = minimize(
                lambda z: obj_fn(z), z,
                jac=lambda z: grad_fn(z),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception as e:
            print(f"  L-BFGS error: {e}")
            pass

        # Update beta
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        top_k = min(50, len(c_sorted_new))
        new_gap = c_sorted_new[0] - c_sorted_new[top_k - 1]
        rel_gap = new_gap / max(c_max_new, 1e-30)

        if rel_gap < 1e-5:
            beta = max(beta * 0.1, c_max_new * 5e-6)
        elif rel_gap < 1e-3:
            beta = max(beta * 0.2, c_max_new * 5e-6)
        elif rel_gap < 1e-1:
            beta = max(beta * 0.3, c_max_new * 5e-6)
        else:
            beta = min(beta * 0.5, c_max_new * 0.005)

        beta = max(beta, c_max_new * 5e-6)

    return current_a, current_sc


def _active_set_polish(a_init, deadline, max_rounds=100):
    """Exact active set polish: identify the true active set and solve KKT precisely.
    Uses a growing active set with violation checking."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Start with a generous active set
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    # Active set: all indices within a small tolerance of the max
    for tol_frac in [1e-4, 5e-4, 1e-3, 5e-3, 0.01, 0.02, 0.05, 0.1]:
        active = np.where(c >= c_max * (1 - tol_frac))[0]
        if len(active) >= 5:
            break
    
    W = np.zeros(2 * n - 1)
    W[active] = 1.0 / len(active)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        # Solve KKT with current active set
        g = _compute_gradient(a, W, n)
        g_centered = g - np.mean(g)
        gn = float(np.linalg.norm(g_centered))
        
        if gn < 1e-13 * n:
            break

        Hg = _cg_solve_hankel(W, n, g, max_iter=500)
        H1 = _cg_solve_hankel(W, n, np.ones(n), max_iter=500)
        H1_denom = float(np.ones(n) @ H1)

        if abs(H1_denom) < 1e-25 or not np.isfinite(H1_denom):
            # Fallback to gradient step
            for trial in [0.01, 0.005, 0.001]:
                a_try = a - trial * g_centered / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < _score(best_a):
                    a = a_try
                    break
            continue

        lam = float(np.ones(n) @ Hg) / H1_denom
        dx = -Hg + lam * H1
        dx = dx - np.mean(dx)

        # Full Newton step with line search
        alpha = 1.0
        step_done = False
        c_cur = _full_convolve(a, a)
        c_max_cur = float(c_cur.max())
        
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max_cur - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.5

        if not step_done:
            # Try smaller step
            for trial_alpha in [0.1, 0.05, 0.01]:
                a_try = a + trial_alpha * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max_cur - 1e-16:
                    a = a_try
                    step_done = True
                    break

        # Update best
        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

        # Check for violated constraints (non-active indices that now exceed max)
        c_new = _full_convolve(a, a)
        c_max_new = float(c_new.max())
        # Find indices where convolution exceeds the current active set's max by a margin
        active_max = float(c_new[active].max()) if len(active) > 0 else 0
        violated = np.where(c_new > active_max * 1.001 + 1e-16)[0]
        
        if len(violated) > 0:
            # Add violated constraints to active set
            new_active = np.union1d(active, violated)
            W = np.zeros(2 * n - 1)
            W[new_active] = 1.0 / len(new_active)
            active = new_active
        
        if not step_done:
            break

    return best_a, best_sc


def _subgrad_replace(a_init, deadline, max_steps=5000):
    """Subgradient method with momentum and adaptive learning rate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.01
    momentum = 0.85
    v = np.zeros(n)

    for i in range(max_steps):
        if time.monotonic() > deadline:

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    elif kind == "flat_wide":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "two_hump":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.3) ** 2) + np.exp(-8.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

import time
from scipy.optimize import minimize

# Global variables for L-BFGS objective/gradient (to avoid lambdas/closures)
_GLB_BETA = 1.0
_GLB_N = 0


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    v_rev = v[::-1]
    out_len = len(W) + len(v_rev) - 1
    N = _next_pow2(out_len)
    F_v = np.fft.rfft(v_rev, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_v * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve_hankel(W, n, rhs, max_iter=400, tol=1e-13):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.01:
            break
        if rs_new < tol * tol * rs0:
            r = r_new
            break
        p = (rs_new / rs) * p + r_new
        r = r_new
        rs = rs_new
        if i % 100 == 99:
            x = x + (rs / max(pHp, 1e-30)) * p
            r = rhs - _hankel_matvec(x, W, n)
            p = r.copy()
            rs = float(r @ r)
    return x


def _compute_gradient(a, W, n):
    a_rev = a[::-1]
    wc = _full_convolve(W, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _lbfgs_obj(z):
    beta = _GLB_BETA
    n = _GLB_N
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lbfgs_grad(z):
    beta = _GLB_BETA
    n = _GLB_N
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _lbfgs_smooth(a_init, deadline, min_remaining=20.0, beta_init=None):
    global _GLB_BETA, _GLB_N
    n = len(a_init)
    _GLB_N = n
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if beta_init is None:
        c_sorted = np.sort(c_cur)[::-1]
        gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
        beta_init = max(gap * 0.4, max_c * 0.008)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    current_a = a_init.copy()
    current_sc = _score(current_a)
    beta = beta_init
    iterations_used = 0

    while beta > max_c * 3e-6:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        scale = max(1.0, n / 8192.0)
        est_time_per_iter = 0.0025 * scale
        max_iter = min(4000, max(10, int(remaining / (est_time_per_iter * 15))))
        _GLB_BETA = beta
        _GLB_N = n
        try:
            result = minimize(
                _lbfgs_obj, z, jac=_lbfgs_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxcor': 100, 'maxls': 80}
            )
            iterations_used += result.niter
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        top_k = min(100, len(c_sorted_new))
        new_gap = c_sorted_new[0] - c_sorted_new[top_k - 1]
        rel_gap = new_gap / max(c_max_new, 1e-30)

        if rel_gap < 5e-5:
            beta = max(beta * 0.08, c_max_new * 3e-6)
        elif rel_gap < 5e-4:
            beta = max(beta * 0.15, c_max_new * 3e-6)
        elif rel_gap < 5e-3:
            beta = max(beta * 0.25, c_max_new * 3e-6)
        elif rel_gap < 0.05:
            beta = max(beta * 0.4, c_max_new * 3e-6)
        else:
            beta = min(beta * 0.6, c_max_new * 0.003)
        beta = max(beta, c_max_new * 3e-6)

    return current_a, current_sc


def _kkt_newton(a_init, deadline, max_rounds=400, k_schedule=None):
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    if k_schedule is None:
        k_schedule = [3, 5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        K = k_schedule[min(rnd, len(k_schedule) - 1)]
        K = min(K, 2 * n - 1)
        idx = np.argpartition(c, -K)[-K:]
        W = np.zeros(2 * n - 1)
        W[idx] = 1.0 / K

        g = _compute_gradient(a, W, n)
        g_centered = g - np.mean(g)
        gn = float(np.linalg.norm(g_centered))
        if gn < 1e-14 * n:
            break

        Hg = _cg_solve_hankel(W, n, g, max_iter=300)
        H1 = _cg_solve_hankel(W, n, np.ones(n), max_iter=300)
        H1_denom = float(np.ones(n) @ H1)

        if abs(H1_denom) < 1e-25 or not np.isfinite(H1_denom):
            dx = -g_centered / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / H1_denom
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_centered / (gn + 1e-30)

        alpha = 1.0
        step_done = False
        for ls in range(45):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1.0 - 0.003 * alpha) - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not step_done:
            for trial in [0.01, 0.005, 0.001, 0.0005]:
                a_try = a - trial * g_centered / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 12:
            break

    return best_a, best_sc
