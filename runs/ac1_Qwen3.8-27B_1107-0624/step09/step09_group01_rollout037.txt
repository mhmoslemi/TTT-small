"""
Optimized step function minimizer using smoothed gradient descent in log-space.
Minimizes 2*n*max(conv(a,a))/sum(a)^2 via adaptive log-sum-exp smoothing,
softmax parameterization, efficient FFT-based gradients, and multi-scale search.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    """Full linear convolution via FFT."""
    out_n = len(a) + len(b) - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _corr(a, w):
    """
    Compute cross-correlation corr[a,w][j] = sum_n a[n]*w[n+j]
    for j = 0, ..., len(a)-1.
    """
    n = len(a)
    # We need N >= (2n-1) + n - 1 = 3n-2
    N = _next_pow2(3 * n - 1)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    corr_full = np.fft.irfft(np.conj(A) * W, N)
    return corr_full[:n]


def _smoothed_obj_and_grad(z, beta, n):
    """
    Compute the smoothed objective and gradient w.r.t. z (log-parameters).
    f_beta = 2n * beta * log(sum exp(c[k]/beta))
    grad_z[j] = a[j] * (grad_a[j] - grad_a . a)
    where grad_a = 4n * corr(a, softmax(c/beta))
    """
    a = _softmax(z)
    c = _convolve(a, a)
    c_max_val = c.max()
    shifted = c / beta - c_max_val / beta

    grad_a_dot_a = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - grad_a_dot_a)

def _smoothed_obj(z, beta, n):

    a = _softmax(z)
    c = _convolve(a, a)
    c_max_val = c.max()
    shifted = c / beta - c_max_val / beta
    e = np.exp(shifted)
    s = e.sum()

This equals `IFFT(conj(FFT(w)) * FFT(a))` evaluated at the right index, with proper zero-padding.

Let me verify the FFT formula. For sequences of lengths `Lw` and `La`, the linear cross-correlation has length `Lw + La - 1`. We need FFT length `N >= Lw + La - 1 = (2n-1) + n - 1 = 3n-2`.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    out_n = len(a) + len(b) - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _corr_aw(a, w, N_cache=None):
    """Cross-correlation: corr[j] = sum_k a[k]*w[k+j] for j=0..n-1."""
    n = len(a)
    if N_cache is None:
        N = _next_pow2(3 * n - 1)
    else:
        N = N_cache
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    result = np.fft.irfft(np.conj(A) * W, N)
    return result[:n]

            if time.monotonic() > t_p1_end:
                break
            a = _make_profile(n, prof)
            sc = _score(a)
            candidates.append((f"{prof}_n{n}", n, a))
            if time.monotonic() > t_p1_end:
                break

    # Sort by score and take top 4
    candidates.sort(key=lambda x: _score(x[2]))
    top_candidates = candidates[:4]
    for name, n, a in top_candidates:
        sc = _score(a)
        print(f"  Candidate: {name}, score={sc:.10f}")
        update_best(a, sc)

    # === Phase 2: Smoothed L-BFGS optimization (1200s) ===
    print(f"\n[Phase 2] Smoothed L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    t_p2_end = min(deadline - 400, t_start + 1260.0)

    # Optimize the top candidates
    for ci, (name, n, a) in enumerate(top_candidates):
        if time.monotonic() > t_p2_end:
            break
        print(f"  Optimizing {name} (n={n})...")
        z_init = _z_from_a(a)
        alloc_time = (t_p2_end - time.monotonic()) / (len(top_candidates) - ci)
        alloc_time = min(alloc_time, 300.0)
        t_end = time.monotonic() + alloc_time
        z_best, sc_best = _full_smoothed_optimize(z_init, n, t_end, n_beta_levels=8)
        a_best = _softmax(z_best)
        update_best(a_best, sc_best)
        print(f"    Best so far: {best_score:.10f}")

    # === Phase 3: Re-optimize best with more iterations (300s) ===
    print(f"\n[Phase 3] Deeper optimization (t={time.monotonic()-t_start:.1f}s)")
    t_p3_end = min(deadline - 200, t_start + 1560.0)

    if best_seq is not None:
        n_best = len(best_seq)
        a_norm = best_seq / best_seq.sum()
        z = _z_from_a(a_norm)

        # More aggressive beta schedule
        c = _convolve(a_norm, a_norm)
        c_max = float(c.max())
        N_cache = _next_pow2(3 * n_best - 1)
        beta_schedule2 = c_max * np.array([0.1, 0.04, 0.015, 0.005, 0.002, 0.0008, 0.0003, 0.0001])

        for beta in beta_schedule2:
            if time.monotonic() > t_p3_end:
                break
            remaining = t_p3_end - time.monotonic()
            if remaining < 10:
                break
            time_per_iter = 0.004 * max(1.0, n_best / 8192.0)
            max_iter = min(500, max(50, int(remaining * 0.3 / time_per_iter)))
            z = _run_lbfgs_at_beta(z, beta, n_best, max_iter, t_p3_end, N_cache)
            a_new = _softmax(z)
            sc_new = _score(a_new)
            if sc_new < best_score:
                best_score = sc_new
                best_seq = a_new.copy()
                print(f"      beta={beta:.4e}, score={sc_new:.10f}")

    # === Phase 4: Active-set polish (150s) ===
    print(f"\n[Phase 4] Active-set polish (t={time.monotonic()-t_start:.1f}s)")
    t_p4_end = min(deadline - 60, t_start + 1710.0)

    if best_seq is not None:
        a_polished, sc_polished = _active_set_polish(best_seq, t_p4_end, max_rounds=300)
        update_best(a_polished, sc_polished)
        print(f"  After polish: {best_score:.10f}")

    # === Phase 5: Multi-n exploration (60s) ===
    print(f"\n[Phase 5] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    t_p5_end = deadline

    if best_seq is not None:
        n_best = len(best_seq)
        n_tries = [n_best // 2, n_best, n_best * 2]
        n_tries = [n for n in n_tries if 4096 <= n <= 131072]
        # Remove duplicates
        seen = set()
        n_tries_unique = []
        for n_t in n_tries:
            if n_t not in seen:
                seen.add(n_t)
                n_tries_unique.append(n_t)

        for n_t in n_tries_unique:
            if time.monotonic() > t_p5_end - 20:
                break
            a_res = _resample(best_seq, n_t)
            sc_res = _score(a_res)
            if sc_res < best_score:
                update_best(a_res, sc_res)
                print(f"  n={n_t}: score={sc_res:.10f}")
            else:
                print(f"  n={n_t}: score={sc_res:.10f} (not better)")

    # === Phase 6: Final quick perturbations (remaining time) ===
    print(f"\n[Phase 6] Perturbations (t={time.monotonic()-t_start:.1f}s)")

    if best_seq is not None and time.monotonic() < deadline - 5:
        n_best = len(best_seq)
        a_norm = best_seq / best_seq.sum()
        z = _z_from_a(a_norm)
        N_cache = _next_pow2(3 * n_best - 1)
        c = _convolve(a_norm, a_norm)
        c_max = float(c.max())

        for attempt in range(6):
            if time.monotonic() > deadline - 5:
                break
            sigma = 0.05 * (attempt + 1)
            z_pert = _perturb_z(z, sigma, rng)
            # Quick optimization with a couple of beta levels
            for beta in [c_max * 0.01, c_max * 0.003]:
                if time.monotonic() > deadline - 5:
                    break
                z_pert = _run_lbfgs_at_beta(z_pert, beta, n_best, 80, deadline - 5, N_cache)
            a_pert = _softmax(z_pert)
            sc_pert = _score(a_pert)
            if sc_pert < best_score:
                update_best(a_pert, sc_pert)
                z = z_pert.copy()
                print(f"  Perturb {attempt}: score={sc_pert:.10f}")

    # === Final output ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(8192, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        # Scale to ensure sum >= 0.01 (should already be satisfied)
        if s < 0.01:
            final = final * (0.1 / s)
        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
