```
def _resample_sequence(p, n_new):
    n_old=len(p)
    if n_old == n_new: return p.copy()
    x_old=np.linspace(0.0,1.0,n_old)
    x_new=np.linspace(0.0,1.0,n_new)
    q=np.interp(x_new, x_old, p)
    q=np.maximum(q,0.0)
    s=q.sum()
    if s <= 0: return np.ones(n_new)/n_new
    return q/s
```

```python
"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.

Strategy:
- Work in log-softmax coordinates so nonnegativity and sum=1 are automatic.
- Use several smooth surrogates:
  * sorted top-K averages (decreasing K),
  * log-sum-exp soft maximum with annealing.
- Scipy L-BFGS-B is run under a deadline callback.
- Multi-length resampling and short restarts are used to avoid local traps.
"""

import numpy as np
import time
from scipy.optimize import minimize

_RFFT = np.fft.rfft
_IRFFT = np.fft.irfft
_OPT_DEADLINE = 0.0


def _opt_time_callback(xk):
    if time.monotonic() > _OPT_DEADLINE:
        return True
    return False


def _next_pow2(m):
    k = 2
    while k < m:
        k <<= 1
    return k


def _square_conv(a):
    """Linear convolution a*a, length 2n-1."""
    n = int(a.size)
    m = 2 * n - 1
    N = _next_pow2(m)
    f = _RFFT(a, N)
    r = _IRFFT(f * f, N)
    return r[:m]


def _grad_mid(W, a):
    """
    For quadratic form sum_i W_i c_i where c_i = (a*a)_i,
    return H = convolution(W, reverse(a)) slice [n-1:2n-1].
    The log-space gradient is 4*n*a*(H - sum_i W_i c_i).
    """
    n = int(a.size)
    out_len = 3 * n - 2
    N = _next_pow2(out_len)
    fW = _RFFT(W, N)
    fa = _RFFT(a[::-1], N)
    return _IRFFT(fW * fa, N)[n - 1:2 * n - 1]


def _normalize_array(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    if not np.all(np.isfinite(a)):
        a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return None
    return a / s


def _score_norm(p):
    p = np.asarray(p, dtype=np.float64)
    if p.size == 0:
        return np.inf
    if not np.all(np.isfinite(p)):
        p = np.where(np.isfinite(p), p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.inf
    p = p / s
    c = _square_conv(p)
    m = float(c.max())
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * float(p.size) * m)


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return np.array([], dtype=np.float64)
    if not np.all(np.isfinite(z)):
        z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(np.max(z))
    e = np.exp(np.clip(z, -700.0, 700.0))
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones_like(z, dtype=np.float64) / float(z.size)
    return e / s


def _z_from_p(p):
    z = np.log(np.maximum(np.asarray(p, dtype=np.float64), 1e-300))
    if not np.all(np.isfinite(z)):
        z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(np.mean(z))
    return z


def _top_gap(c, top_cnt):
    if c.size <= 1:
        return 1e-30
    if c.size <= top_cnt:
        return max(float(c.max()) - float(c.min()), 1e-30)
    top = np.partition(c, -top_cnt)[-top_cnt:]
    return max(float(c.max()) - float(top.min()), 1e-30)


def _make_profile(n, kind="tapered"):
    n = int(n)
    if n <= 0:
        n = 1
    t = np.linspace(0.0, 1.0, n)
    if kind == "tapered":
        a = np.sin(np.pi * t) ** 0.7 + 1e-3
    elif kind == "bell":
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 1e-3
    elif kind == "asym":
        a = np.exp(-3.0 * (t - 0.3) ** 2) + 0.6 * np.exp(-3.0 * (t - 0.7) ** 2) + 1e-3
    elif kind == "cos":
        a = np.sin(np.pi * t) + 1e-3
    else:
        a = np.ones(n, dtype=np.float64) + 1e-3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _resample_sequence(p, n_new):
    p = np.asarray(p, dtype=np.float64)
    n_old = int(p.size)
    n_new = int(n_new)
    if n_old == n_new:
        q = p.copy()
    else:
        x_old = np.linspace(0.0, 1.0, n_old)
        x_new = np.linspace(0.0, 1.0, n_new)
        q = np.interp(x_new, x_old, p)
    q = np.maximum(q, 0.0)
    s = float(q.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n_new, dtype=np.float64) / float(n_new)
    return q / s


def _perturb_log(p, sigma, rng):
    z = _z_from_p(p)
    z = z + float(sigma) * rng.standard_normal(z.size)
    z = z - float(z.mean())
    return _softmax(z)


def _topk_obj(z, K, n, reg):
    p = _softmax(z)
    c = _square_conv(p)
    size = int(c.size)
    K = int(K)
    if K < 1:
        K = 1
    if K >= size:
        K = max(1, size // 2)
    if K >= size:
        K = size

    if K < size:
        idx = np.argpartition(c, -K)[-K:]
        S = float(c[idx].sum()) / float(K)
    else:
        S = float(c.sum()) / float(K)

    val = 2.0 * float(n) * S
    if reg > 0.0:
        zc = np.clip(z, -1000.0, 1000.0)
        val += 0.5 * float(reg) * float(np.dot(zc, zc))
    if not np.isfinite(val):
        return 1e100
    return float(val)


def _topk_grad(z, K, n, reg):
    p = _softmax(z)
    c = _square_conv(p)
    size = int(c.size)
    K = int(K)
    if K < 1:
        K = 1
    if K >= size:
        K = max(1, size // 2)
    if K >= size:
        K = size

    if K < size:
        idx = np.argpartition(c, -K)[-K:]
        w = np.zeros(size, dtype=np.float64)
        w[idx] = 1.0 / float(K)
        S = float(c[idx].sum()) / float(K)
    else:
        w = np.ones(size, dtype=np.float64) / float(K)
        S = 1.0 / float(K)

    h = _grad_mid(w, p)
    g = 4.0 * float(n) * p * (h - S)
    if not np.all(np.isfinite(g)):
        g = np.where(np.isfinite(g), g, 0.0)
    g = g - float(np.mean(g))
    if reg > 0.0:
        g = g + float(reg) * np.clip(z, -1000.0, 1000.0)
    return g


def _smooth_obj(z, beta, n, reg):
    p = _softmax(z)
    c = _square_conv(p)
    cmax = float(c.max())
    if not np.isfinite(cmax) or cmax <= 0.0:
        S = 0.0
    elif beta <= 1e-30:
        S = cmax
    else:
        t = (c - cmax) / float(beta)
        t = np.clip(t, -700.0, 0.0)
        e = np.exp(t)
        s = float(np.sum(e))
        if not np.isfinite(s) or s <= 1e-300:
            S = cmax
        else:
            S = cmax + float(beta) * float(np.log(s))
    val = 2.0 * float(n) * S
    if reg > 0.0:
        zc = np.clip(z, -1000.0, 1000.0)
        val += 0.5 * float(reg) * float(np.dot(zc, zc))
    if not np.isfinite(val):
        return 1e100
    return float(val)


def _smooth_grad(z, beta, n, reg):
    p = _softmax(z)
    c = _square_conv(p)
    cmax = float(c.max())
    size = int(c.size)

    if not np.isfinite(cmax) or cmax <= 0.0:
        w = np.zeros(size, dtype=np.float64)
        w[0] = 1.0
        S = 0.0
    elif beta <= 1e-30:
        w = np.zeros(size, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        S = cmax
    else:
        t = (c - cmax) / float(beta)
        t = np.clip(t, -700.0, 0.0)
        e = np.exp(t)
        s = float(np.sum(e))
        if not np.isfinite(s) or s <= 1e-300:
            w = np.zeros(size, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
            S = cmax
        else:
            w = e / s
            S = cmax + float(beta) * float(np.log(s))

    h = _grad_mid(w, p)
    g = 4.0 * float(n) * p * (h - S)
    if not np.all(np.isfinite(g)):
        g = np.where(np.isfinite(g), g, 0.0)
    g = g - float(np.mean(g))
    if reg > 0.0:
        g = g + float(reg) * np.clip(z, -1000.0, 1000.0)
    return g


def _run_topk_single(p, end_time, K, reg=1e-14):
    p = np.asarray(p, dtype=np.float64)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        return p.copy(), np.inf, p.copy(), np.inf
    p = p / s

    n = int(p.size)
    z = _z_from_p(p)
    best_p = p.copy()
    best_sc = _score_norm(p)
    end_p = p.copy()
    end_sc = best_sc

    if time.monotonic() < end_time - 2.0:
        global _OPT_DEADLINE
        _OPT_DEADLINE = max(time.monotonic() + 1.0, float(end_time) - 1.5)
        options = {
            "maxiter": 100000,
            "ftol": 1e-20,
            "gtol": 1e-14,
            "maxcor": 50,
            "maxls": 80,
            "disp": False
        }
        try:
            res = minimize(
                _topk_obj,
                z,
                jac=_topk_grad,
                args=(K, n, reg),
                method="L-BFGS-B",
                options=options,

            )
            if res is not None and res.x is not None:

                            best_p = p_new
        except Exception as exc:
            print(f"  topk minimize error K={K}: {exc}", flush=True)

    return best_p, best_sc, end_p, end_sc


def _run_topk_phase(p, phase_start, phase_end):
    current = np.asarray(p, dtype=np.float64).copy()
    current_sc = _score_norm(current)
    input_p = current.copy()
    input_sc = current_sc

    if not np.isfinite(current_sc) or phase_end <= phase_start + 2.0:
        return current.copy(), current_sc

    n = int(current.size)
    K = min(max(2048, n // 4), max(2, (2 * n - 1) // 3))
    if K < 2:
        K = 2
    K_list = []
    Kcur = K
    while Kcur >= 2:
        K_list.append(Kcur)
        Kcur //= 2
    if not K_list:
        K_list = [2]

    L = len(K_list)
    if L > 1:
        idx = np.arange(L, dtype=np.float64)
        w = 1.0 + idx / float(L - 1)
        fracs = w / w.sum()
    else:
        fracs = np.ones(1, dtype=np.float64)

    total_budget = float(phase_end - phase_start)

    for K, frac in zip(K_list, fracs):
        if time.monotonic() >= phase_end - 3.0:
            break
        budget = max(2.0, float(frac) * total_budget)
        k_end = min(float(phase_end), time.monotonic() + budget)
        if k_end <= time.monotonic() + 1.0:
            continue
        print(f"    topK K={K}, budget={k_end - time.monotonic():.1f}s", flush=True)
        bp, bs, ep, es = _run_topk_single(current, k_end, K)

        if np.isfinite(bs) and bs < current_sc - 1e-14:
            current = bp
            current_sc = bs
        elif (np.isfinite(es) and es <= current_sc + 1e-4 and es <= input_sc + 1e-4):
            # Allow a very small temporary worsening to feed later K-values.
            current = ep
            current_sc = es

    if current_sc > input_sc + 1e-6:
        return input_p.copy(), input_sc
    return current.copy(), current_sc

        options = {
            "maxiter": 100000,
            "ftol": 1e-20,
            "gtol": 1e-14,
            "maxcor": 50,
            "maxls": 80,
            "disp": False
        }
        try:
            res = minimize(
                _smooth_obj,
                z,
                jac=_smooth_grad,
                args=(beta, n, reg),
                method="L-BFGS-B",
                options=options,
                callback=_opt_time_callback
            )
            if res is not None and res.x is not None:
                zx = np.asarray(res.x, dtype=np.float64)
                if np.all(np.isfinite(zx)):
                    p_new = _softmax(zx)
                    sc_new = _score_norm(p_new)
                    if np.isfinite(sc_new):
                        end_p = p_new
                        end_sc = sc_new
                        if sc_new < best_sc:
                            best_sc = sc_new
                            best_p = p_new
        except Exception as exc:
            print(f"  smooth minimize error beta={beta:.3e}: {exc}", flush=True)

    return best_p, best_sc, end_p, end_sc


def _run_smooth_phase(p, phase_start, phase_end):
    current = np.asarray(p, dtype=np.float64).copy()
    current_sc = _score_norm(current)
    if not np.isfinite(current_sc) or phase_end <= phase_start + 5.0:
        return current, current_sc

    c = _square_conv(current)
    cmax = float(c.max())
    if not np.isfinite(cmax) or cmax <= 0.0:
        return current, current_sc

    top_cnt = min(80, max(1, int(c.size) - 1))
    gap = _top_gap(c, top_cnt)
    beta = max(3.0 * gap, 5e-4 * cmax, 1e-30)
    phase_len = max(1.0, float(phase_end - phase_start))
    no_improve = 0

    while time.monotonic() < phase_end - 8.0:
        round_budget = max(8.0, 0.15 * phase_len)
        round_end = min(float(phase_end) - 2.0, time.monotonic() + round_budget)
        if round_end <= time.monotonic() + 1.0:
            break

        bp, bs, ep, es = _run_smooth_single(current, round_end, beta)
        if np.isfinite(bs) and bs < current_sc - 1e-14:
            current = bp
            current_sc = bs
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5:
                break

        c = _square_conv(current)
        cmax = float(c.max())
        if not np.isfinite(cmax) or cmax <= 0.0:
            break

        gap = _top_gap(c, top_cnt)
        if gap < 1e-5 * cmax:
            beta *= 0.30
        elif gap < 1e-4 * cmax:
            beta *= 0.50
        else:
            beta *= 0.70

        beta = max(beta, 1e-6 * cmax)
        beta = min(beta, max(10.0 * gap, 1e-2 * cmax))

        if beta <= 1e-6 * cmax:
            break

    return current, current_sc


def _run_multi_n_phase(p, phase_end):
    current = np.asarray(p, dtype=np.float64).copy()
    current_sc = _score_norm(current)

    if not np.isfinite(current_sc):
        return current, current_sc

    n0 = int(current.size)
    cand_list = []
    for nc in (32768, 2 * n0, 16384, n0, 65536, max(2048, n0 // 2)):
        if 4096 <= nc <= 65536 and nc not in cand_list:
            cand_list.append(nc)

    optimized = 0
    for nc in cand_list:
        if int(nc) == int(current.size):
            continue
        if optimized >= 3:
            break
        if time.monotonic() > phase_end - 90.0:
            break

        q = _resample_sequence(current, nc)
        sc_q = _score_norm(q)
        print(f"  multi-n trial n={nc}, baseline={sc_q:.10f}", flush=True)

        Kfirst = min(4096, max(2, 2 * nc - 2))
        Ks = []
        Kcur = Kfirst
        while Kcur >= 2:
            Ks.append(Kcur)
            Kcur //= 2
        Ks = Ks[:6]

        remaining = max(10.0, float(phase_end) - time.monotonic())
        per = max(8.0, min(25.0, remaining / max(1, len(Ks) + 2)))

        for K in Ks:
            if time.monotonic() > phase_end - 30.0:
                break
            k_end = min(float(phase_end) - 5.0, time.monotonic() + per)
            if k_end <= time.monotonic() + 1.0:
                continue
            bp, bs, ep, es = _run_topk_single(q, k_end, K)
            if np.isfinite(bs) and bs < sc_q:
                q = bp
                sc_q = bs

        if np.isfinite(sc_q) and sc_q < current_sc:
            current = q
            current_sc = sc_q
            print(f"  multi-n improved to n={len(q)}, score={current_sc:.10f}", flush=True)
        optimized += 1

    return current, current_sc


def _run_restart_phase(p, end_time, seed):
    current = np.asarray(p, dtype=np.float64).copy()
    current_sc = _score_norm(current)
    if not np.isfinite(current_sc):
        return current, current_sc

    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    sigmas = (0.02, 0.045, 0.09, 0.18)

    for sigma in sigmas:
        if time.monotonic() > end_time - 25.0:
            break
        q = _perturb_log(current, sigma, rng)
        sc_q = _score_norm(q)
        if np.isfinite(sc_q) and sc_q < current_sc:
            current = q
            current_sc = sc_q

        c = _square_conv(current)
        cmax = float(c.max())
        if not np.isfinite(cmax) or cmax <= 0.0:
            break
        beta = max(3e-6 * cmax, 1e-30)

        sm_end = min(float(end_time) - 12.0, time.monotonic() + 25.0)
        if sm_end > time.monotonic() + 2.0:
            bp, bs, ep, es = _run_smooth_single(current, sm_end, beta)
            if np.isfinite(bs) and bs < current_sc:
                current = bp
                current_sc = bs

        tk_end = min(float(end_time) - 5.0, time.monotonic() + 18.0)
        if tk_end > time.monotonic() + 2.0:
            bp, bs, ep, es = _run_topk_single(current, tk_end, 8)
            if np.isfinite(bs) and bs < current_sc:
                current = bp
                current_sc = bs

    return current, current_sc


def _external_score_list(lst):
    try:
        return float(evaluate_sequence(lst))
    except Exception:
        return np.inf


def propose_candidate(seed=42, budget_s=1800.0):
    t0 = time.monotonic()
    budget = max(120.0, float(budget_s))
    available = max(60.0, budget - 60.0)
    deadline = t0 + available

    best_seq = None
    best_score = np.inf

    # Load previous state-of-the-art if available.
    try:
        par = _normalize_array(height_sequence_1)
        if par is not None:
            best_seq = par
            best_score = _score_norm(par)
            print(f"parent loaded: n={len(best_seq)}, score={best_score:.10f}", flush=True)
    except Exception:
        pass

    if best_seq is None or not np.isfinite(best_score):
        n_default = 32768
        best_sc = np.inf
        for kind in ("tapered", "bell", "asym", "cos", "uniform"):
            p = _make_profile(n_default, kind)
            sc = _score_norm(p)
            if np.isfinite(sc) and sc < best_sc:
                best_seq = p
                best_sc = sc
        best_score = best_sc
        print(f"initialized profile: n={len(best_seq)}, score={best_score:.10f}", flush=True)

    if not np.isfinite(best_score):
        best_seq = _make_profile(32768, "tapered")
        best_score = _score_norm(best_seq)

    # Keep lengths in a practical FFT-friendly range.
    if len(best_seq) < 2048 or len(best_seq) > 65536:
        target = 32768 if len(best_seq) <= 32768 else 65536
        best_seq = _resample_sequence(best_seq, target)
        best_score = _score_norm(best_seq)
        print(f"resampled start to n={len(best_seq)}, score={best_score:.10f}", flush=True)

    A = available

    # Phase boundaries.
    pre_end = min(deadline - 10.0, t0 + 0.06 * A)
    top_end = min(deadline - 10.0, t0 + 0.66 * A)
    smooth_end = min(deadline - 10.0, t0 + 0.78 * A)
    multi_end = min(deadline - 20.0, t0 + 0.90 * A)
    restart_end = min(deadline - 5.0, t0 + 0.965 * A)

    n0 = len(best_seq)
    if n0 > 32768:
        pre_end = min(pre_end, t0 + 0.04 * A)
        top_end = min(top_end, t0 + 0.58 * A)
        smooth_end = min(smooth_end, t0 + 0.70 * A)
        multi_end = min(multi_end, t0 + 0.86 * A)

    # Warm smooth phase.
    if pre_end > time.monotonic() + 15.0:
        print(f"[phase] pre-smooth until {pre_end - t0:.1f}s", flush=True)
        seq, sc = _run_smooth_phase(best_seq, t0, pre_end)
        if np.isfinite(sc) and sc < best_score:
            best_seq = seq
            best_score = sc
            print(f"  pre-smooth improved to {best_score:.10f}", flush=True)

    # Main top-K homotopy.
    if top_end > time.monotonic() + 30.0:
        print(f"[phase] top-K until {top_end - t0:.1f}s", flush=True)
        seq, sc = _run_topk_phase(best_seq, pre_end, top_end)
        if np.isfinite(sc) and sc < best_score:
            best_seq = seq
            best_score = sc
            print(f"  top-K improved to {best_score:.10f}", flush=True)

    # Annealed log-sum-exp polish.
    if smooth_end > time.monotonic() + 20.0:
        print(f"[phase] smooth anneal until {smooth_end - t0:.1f}s", flush=True)
        seq, sc = _run_smooth_phase(best_seq, top_end, smooth_end)
        if np.isfinite(sc) and sc < best_score:
            best_seq = seq
            best_score = sc
            print(f"  smooth polish improved to {best_score:.10f}", flush=True)

    # Multi-length exploration.
    if multi_end > time.monotonic() + 60.0:
        print(f"[phase] multi-n until {multi_end - t0:.1f}s", flush=True)
        seq, sc = _run_multi_n_phase(best_seq, multi_end)
        if np.isfinite(sc) and sc < best_score:
            best_seq = seq
            best_score = sc
            print(f"  multi-n improved to {best_score:.10f} (n={len(best_seq)})", flush=True)

    # Randomized restarts.
    if restart_end > time.monotonic() + 50.0:
        print(f"[phase] restarts until {restart_end - t0:.1f}s", flush=True)
        seq, sc = _run_restart_phase(best_seq, restart_end, int(seed) % 1000000)
        if np.isfinite(sc) and sc < best_score:
            best_seq = seq
            best_score = sc
            print(f"  restart improved to {best_score:.10f}", flush=True)

    # Final ultra-small smooth polish.
    if time.monotonic() < deadline - 25.0:
        c = _square_conv(best_seq)
        cmax = float(c.max())
        if np.isfinite(cmax) and cmax > 0.0:
            beta = max(2e-6 * cmax, 1e-30)
            end = min(deadline - 22.0, time.monotonic() + min(120.0, max(20.0, deadline - 22.0 - time.monotonic())))
            seq, sc, ep, es = _run_smooth_single(best_seq, end, beta)
            if np.isfinite(sc) and sc < best_score:
                best_seq = seq
                best_score = sc
                print(f"  final smooth improved to {best_score:.10f}", flush=True)

    # Final sharp top-4 L-BFGS.
    if time.monotonic() < deadline - 10.0:
        end = min(deadline - 2.0, time.monotonic() + max(10.0, min(40.0, deadline - 10.0 - time.monotonic())))
        seq, sc, ep, es = _run_topk_single(best_seq, end, 4)
        if np.isfinite(sc) and sc < best_score:
            best_seq = seq
            best_score = sc
            print(f"  final top-4 improved to {best_score:.10f}", flush=True)

    final = np.asarray(best_seq, dtype=np.float64).ravel()
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.clip(final, 0.0, 1000.0)
    s = float(final.sum())
    if not np.isfinite(s) or s < 0.01:
        final = _make_profile(32768, "tapered")
        s = float(final.sum())
        if s <= 0.0:
            final = np.ones(32768, dtype=np.float64) / 32768.0
            s = 1.0
    final = final / s

    result = [float(x) for x in final]
    internal = _score_norm(final)
    external = _external_score_list(result)
    print("=" * 70, flush=True)
    print(f"FINAL n={len(result)}", flush=True)
    print(f"FINAL internal score={internal:.10f}", flush=True)
    print(f"FINAL external score={external:.10f}" if np.isfinite(external) else
          f"FINAL external score unavailable", flush=True)
    print(f"FINAL elapsed={time.monotonic() - t0:.1f}s", flush=True)

    return result
```
