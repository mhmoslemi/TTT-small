Multi-strategy optimizer for 2*n*max(conv(a,a))/sum(a)^2.
Improvements: larger n, KKT refinement, peak-shaving, random restarts.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)

def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

def _grad_soft_structured(a, beta, n):
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)

def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft_structured(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)

class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)
    def grad(self, z):
        return _smograd(z, self.beta, self.n)

def _run_lbfgs(a_init, deadline, min_time_remaining=60.0, verbose=False):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c0_max * 0.003:
        gap = c0_max * 0.003
    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
        0.00003125, 0.000015625, 0.0000078, 0.0000039,
        0.000002, 0.000001, 0.0000005, 0.0000002
    ])
    betas = gap * multipliers
    t_level_start = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        size_factor = max(1.0, n / 4096.0)
        est_iter_time = 0.002 * size_factor
        max_iter = min(8000, max(100, int(remaining / (est_iter_time * 20))))
        if beta < gap * 0.001:

        prob = _Prob(beta, n)

    a = a / s
    best_a = a.copy()

                break
            no_improve = 0
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]
        support = a > 1e-10 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break
        lam = float(np.mean(grad_avg[support]))
        residual = grad_avg[support] - lam
        # Try multiple step sizes
        best_step_sc = current_sc
        best_step_a = None
        for eta in [0.1, 0.05, 0.02, 0.01, 0.005]:
            a_new = a.copy()
            a_new[support] -= eta * residual
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < best_step_sc:
                best_step_sc = sc_new
                best_step_a = a_new
        if best_step_a is not None:
            a = best_step_a
        else:
            tol_frac = max(min_tol, tol_frac * 0.8)
        if no_improve == 0:
            tol_frac = max(min_tol, tol_frac * 0.97)
    return best_a, best_sc

def _peak_shave(a_init, deadline, min_time_remaining=30.0, verbose=False):
    """Iteratively shave peaks by reducing elements that contribute most to the peak."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0
    max_no_improve = 200
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > max_no_improve:
            break
        # Find the peak position
        k_peak = int(np.argmax(c))
        # Find all near-peak positions
        tol = 0.01 * c_max
        near_peak = c >= (c_max - tol)
        # For each j, compute how much a[j] contributes to the near-peak convolution values
        # Contribution of a[j] to c[k] is a[j]*a[k-j] (for valid k-j)
        # Total contribution to near-peak: sum_{k in near_peak} a[j]*a[k-j]
        # = a[j] * sum_{k in near_peak, 0<=k-j<n} a[k-j]
        # Let's compute eff[j] = sum_{k in near_peak, 0<=k-j<n} a[k-j]
        # This is conv(near_peak_indicator, a_rev) evaluated at position j
        np_ind = near_peak.astype(float)
        a_rev = a[::-1]
        eff_conv = _full_convolve(np_ind, a_rev)
        eff = eff_conv[n - 1:2 * n - 1]
        # We want to reduce a[j] where eff[j] is high (contributes a lot to peaks)
        # and increase where eff[j] is low
        # But we need to maintain sum(a) = 1 and a >= 0
        # Simple approach: a_new[j] = a[j] - eta * (eff[j] - mean(eff))
        # Clip to non-negative, renormalize
        eff_centered = eff - np.mean(eff)
        best_step_sc = current_sc
        best_step_a = None
        for eta in [0.05, 0.02, 0.01, 0.005]:
            a_new = a - eta * eff_centered
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < best_step_sc:
                best_step_sc = sc_new
                best_step_a = a_new
        if best_step_a is not None:
            a = best_step_a
        else:
            # Try reducing only the top contributors
            top_k = max(1, n // 10)
            top_idx = np.argsort(eff)[-top_k:]
            for eta in [0.02, 0.01, 0.005]:
                a_new = a.copy()
                a_new[top_idx] *= (1 - eta)
                s_new = a_new.sum()
                if s_new < 1e-10:
                    continue
                a_new = a_new / s_new
                sc_new = _score(a_new)
                if sc_new < best_step_sc:
                    best_step_sc = sc_new
                    best_step_a = a_new
            if best_step_a is not None:
                a = best_step_a
            else:
                break
    return best_a, best_sc

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) / 0.3) ** 2) + 0.01
        return a / a.sum()
    elif kind == "super_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2.0
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _random_perturb(a, magnitude=0.01, seed=None):
    rng = np.random.default_rng(seed)
    n = len(a)
    noise = rng.normal(0, magnitude, n)
    a_new = a + noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines: softmax L-BFGS, KKT active-set refinement with proper gradients,
peak-shaving, symmetric optimization, multi-size search (up to 65536),
and random restarts to escape local minima.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_conv_structured(a, w, n):
    """Compute 2*conv(w, a_rev)[n-1:2n-1] efficiently via FFT."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    grad_a = _grad_conv_structured(a, w, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)


class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=50.0, verbose=False):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c0_max * 0.002:
        gap = c0_max * 0.002
    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
        0.00003125, 0.000015625, 0.0000078, 0.0000039,
        0.000002, 0.000001, 0.0000005, 0.0000002
    ])
    betas = gap * multipliers
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        size_factor = max(1.0, n / 4096.0)
        est_iter_time = 0.002 * size_factor
        max_iter = min(8000, max(100, int(remaining / (est_iter_time * 20))))
        if beta < gap * 0.001:
            max_iter = min(max_iter, 3000)
        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-15, 'maxcor': 100, 'maxls': 60}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"      beta={beta:.2e}: score={best_sc:.10f}, nit={result.nit}")
        except Exception:
            pass
    return best_a, best_sc


def _kkt_refine(a_init, deadline, min_time_remaining=30.0, verbose=False):
    """KKT active-set refinement with proper gradient via FFT."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0
    max_no_improve = 400
    tol_frac = 0.008
    min_tol = 0.00005

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.5)
            if tol_frac <= min_tol * 1.1:
                break
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros(len(c), dtype=bool)
            active[idx] = True
            n_active = 1

        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        g = _grad_conv_structured(a, w, n)

        support = a > 1e-10 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        lam = float(np.mean(g[support]))
        r = np.zeros(n)
        r[support] = g[support] - lam

        # Proper gradient: df/da[l] = 4/|S| * sum_j r[j] * S_ind[j+l]
        # Use FFT cross-correlation
        N_fft = _next_pow2(3 * n)
        R = np.fft.rfft(r, N_fft)
        S_ind = active.astype(np.float64)
        Si = np.fft.rfft(S_ind, N_fft)
        h_circ = np.fft.irfft(np.conj(R) * Si, N_fft)
        h = h_circ[:n]
        grad_f = (4.0 / n_active) * h

        # Line search with backtracking
        best_step_sc = current_sc
        best_step_a = None
        for eta in [0.05, 0.02, 0.01, 0.005, 0.002]:
            a_try = a.copy()
            a_try[support] -= eta * grad_f[support]
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_step_sc - 1e-14:
                best_step_sc = sc_try
                best_step_a = a_try

        if best_step_a is not None:
            a = best_step_a
        else:
            # Fallback: simple gradient step on a directly
            residual = g[support] - lam
            for eta in [0.01, 0.005, 0.002]:
                a_try = a.copy()
                a_try[support] -= eta * residual
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-14:
                    a = a_try
                    break
            else:
                tol_frac = max(min_tol, tol_frac * 0.85)
                continue

        if no_improve == 0:
            tol_frac = max(min_tol, tol_frac * 0.98)

        if verbose and no_improve == 0 and (no_improve + 1) % 50 == 0:
            print(f"      KKT iter {no_improve}: score={best_sc:.10f}, tol={tol_frac:.6f}")

    return best_a, best_sc


def _peak_shave(a_init, deadline, min_time_remaining=30.0, verbose=False):

            break

        eff_conv = _full_convolve(np_ind, a_rev)

            if s_try < 1e-10:

            a_try = a_try / s_try

                best_step_a = a_try

        else:

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: softmax L-BFGS, FFT-based KKT active-set refinement, peak-shaving,
symmetric optimization, multi-size search (16384-65536), random restarts.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_conv_structured(a, w, n):
    """Compute 2*conv(w, a_rev)[n-1:2n-1] via FFT."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    grad_a = _grad_conv_structured(a, w, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)


class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=50.0, verbose=False):
    """Adaptive beta L-BFGS on softmax parameterization."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c0_max * 0.002:
        gap = c0_max * 0.002
    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
        0.00003125, 0.000015625, 0.0000078, 0.0000039,
        0.000002, 0.000001, 0.0000005, 0.0000002
    ])
    betas = gap * multipliers
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        size_factor = max(1.0, n / 4096.0)
        est_iter_time = 0.002 * size_factor
        max_iter = min(8000, max(100, int(remaining / (est_iter_time * 20))))
        if beta < gap * 0.001:
            max_iter = min(max_iter, 3000)
        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-15, 'maxcor': 100, 'maxls': 60}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"      beta={beta:.2e}: score={best_sc:.10f}, nit={result.nit}")

def _kkt_refine(a_init, deadline, min_time_remaining=30.0, verbose=False):
    """KKT active-set refinement with proper FFT-based gradient."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0
    max_no_improve = 400
    tol_frac = 0.008
    min_tol = 0.00005

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.5)
            if tol_frac <= min_tol * 1.1:
                break
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros(len(c), dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        g = _grad_conv_structured(a, w, n)

        support = a > 1e-10 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        lam = float(np.mean(g[support]))
        r = np.zeros(n)
        r[support] = g[support] - lam

        # Proper gradient: df/da[l] = 4/|S| * cross-corr(r, S_ind)[l]
        N_fft = _next_pow2(3 * n)
        R = np.fft.rfft(r, N_fft)
        S_ind = active.astype(np.float64)
        Si = np.fft.rfft(S_ind, N_fft)
        h_circ = np.fft.irfft(np.conj(R) * Si, N_fft)
        h = h_circ[:n]
        grad_f = (4.0 / n_active) * h

        # Backtracking line search
        best_step_sc = current_sc
        best_step_a = None
        for eta in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            a_try = a.copy()
            a_try[support] -= eta * grad_f[support]
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_step_sc - 1e-14:
                best_step_sc = sc_try
                best_step_a = a_try

        if best_step_a is not None:
            a = best_step_a
        else:
            # Fallback: simple residual step
            residual = g[support] - lam
            for eta in [0.005, 0.002, 0.001]:
                a_try = a.copy()
                a_try[support] -= eta * residual
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-14:
                    a = a_try
                    break
            else:
                tol_frac = max(min_tol, tol_frac * 0.85)
                continue

        if no_improve == 0:
            tol_frac = max(min_tol, tol_frac * 0.98)

        if verbose and no_improve == 0 and (no_improve + 1) % 100 == 0:
            print(f"      KKT: score={best_sc:.10f}, tol={tol_frac:.6f}, n_act={n_active}")

    return best_a, best_sc

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _random_perturb(a, magnitude=0.005, seed=None):
    rng = np.random.default_rng(seed)
    n = len(a)
    noise = rng.normal(0, magnitude, n)
    a_new = a + noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"    {prof}: {best_score:.10f}")

    # === Phase 5: Final aggressive KKT ===
    print(f"\n[P5] Final KKT polish")
    if time.monotonic() + 15 < deadline:
        p5_end = deadline - 5
        a5, sc5 = _kkt_refine(best_seq, p5_end, min_time_remaining=5, verbose=True)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"    Final KKT: {best_score:.10f}")

    # === Phase 6: Final peak shave ===
    if time.monotonic() + 10 < deadline:
        p6_end = deadline - 3
        a6, sc6 = _peak_shave(best_seq, p6_end, min_time_remaining=3)
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"    Final PeakShave: {best_score:.10f}")

    # Final validation
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
    sc_final = _score(final)
    elapsed = time.monotonic() - t0
    print(f"\n{'='*60}")
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={elapsed:.1f}s")
    print(f"{'='*60}")
    return [float(x) for x in final]
