"""
FFT-accelerated optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with log-sum-exp smoothing (continuation method) combined with
direct Nesterov subgradient for fine-tuning. Explores multiple sequence lengths
and random restarts to escape local minima.
"""

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a):
    """Compute auto-convolution of a using FFT. Returns array of length 2n-1."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c


def _fft_grad(a, p):
    """Compute gradient of sum_k p[k]*(a*a)[k] w.r.t. a.
    p has length 2n-1. Returns array of length n.
    grad[m] = 2 * sum_{j=0}^{n-1} p[j+m] * a[j]
    """
    n = len(a)
    N = _next_pow2(2 * n - 1 + n - 1)  # = _next_pow2(3n-2)
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)
    R = np.fft.irfft(np.conj(fa) * fp, N)[:n]
    return 2.0 * R


def _score(a):
    """Compute the true score: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    x = np.sort(v)[::-1]
    cssv = np.cumsum(x)
    idx = np.arange(1, n + 1)
    cond = x * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def _smooth_obj(z, beta, n):
    """Smooth approximation of 2n*max(c) using log-sum-exp."""
    a = _softmax(z)
    c = _fft_conv(a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smooth_grad(z, beta, n):
    """Gradient of smooth objective w.r.t. z."""
    a = _softmax(z)
    c = _fft_conv(a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p = e / s
    grad_a = _fft_grad(a, p)
    # Chain rule for softmax parameterization
    zg = a * (grad_a - float(np.dot(a, grad_a)))
    return zg


class _SmoothProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smooth_obj(z, self.beta, self.n)

    def grad(self, z):
        return _smooth_grad(z, self.beta, self.n)


def _lbfgs_smooth(a_init, deadline, min_time_left=30.0, verbose=False):
    """Run L-BFGS with continuation on beta."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s

    c = _fft_conv(a)
    max_c = float(c.max())
    if max_c < 1e-15:
        return a

    # Determine initial beta
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < 1e-18:
        gap = max_c * 1e-6
    base_beta = max(gap * 10, max_c * 0.01)

    # Continuation schedule
    n_betas = 25
    betas = base_beta * (0.5 ** np.arange(n_betas))

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    current_score = _score(a)
    best_a = a.copy()
    best_score = current_score

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_left > deadline:
            break
        remaining = deadline - min_time_left - time.monotonic()
        if remaining < 5:
            break

        # Estimate iterations
        est_time = 0.005 * max(1.0, n / 8192.0)
        max_iter = min(3000, max(50, int(remaining / (est_time * 5))))

        prob = _SmoothProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_score - 1e-15:
                current_score = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"      beta_idx={i}: score={current_score:.12f}")
            else:
                # Try with actual z from optimizer (might have improved the smooth obj
                # but not the true obj)
                a_try = _softmax(result.x)
                sc_try = _score(a_try)
                if sc_try < best_score:
                    best_score = sc_try
                    best_a = a_try.copy()
        except Exception as e:
            if verbose:
                print(f"      L-BFGS error at beta_idx={i}: {e}")
            pass

    return best_a


def _nesterov_direct(a_init, deadline, min_time_left=30.0, K=20, seed=42):
    """Nesterov subgradient with exact max (no smoothing)."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s

    best_a = a.copy()
    best_sc = _score(a)

    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    momentum = 0.9
    base_lr = 0.002
    lr = base_lr
    no_improve = 0
    t = 0

    while time.monotonic() + min_time_left < deadline:
        t += 1
        c = _fft_conv(a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-16:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            break

        # Adaptive K
        c_sorted = np.sort(c)[::-1]
        threshold = c_max * 0.99
        active = np.where(c >= threshold)[0]
        if len(active) < 3:
            active = np.argsort(c)[::-1][:K]
        if len(active) > 500:
            active = np.argsort(c)[::-1][:500]

        # Compute gradient of sum of active peaks
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / len(active)
        grad = _fft_grad(a, p)

        # Project to simplex tangent
        grad_t = grad - float(np.mean(grad))
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-18:
            break

        # Nesterov step
        effective_lr = lr / (gnorm + 1e-18)
        v = momentum * v - effective_lr * grad_t
        a_cand = a + v
        a_new = _project_simplex(a_cand)

        c_new = _fft_conv(a_new)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new <= current_sc:
            a = a_new
            lr = min(0.01, lr * 1.005)
        else:
            v = 0.5 * v
            lr *= 0.9
            a = a_new  # stay on simplex

        if no_improve % 200 == 0 and no_improve > 0:
            # Perturb and restart momentum
            perturb = 0.0005 * rng.standard_normal(n)
            a_pert = _project_simplex(a + perturb)
            sc_pert = 2.0 * n * float(_fft_conv(a_pert).max())
            if sc_pert < best_sc:
                a = a_pert
                best_a = a_pert.copy()
                best_sc = sc_pert
            v = np.zeros(n)
            lr = base_lr
            no_improve = 0

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
