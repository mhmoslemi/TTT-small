Let me think about the **active set QP** approach more carefully:

At the optimum (for a fixed active set $S$), the KKT conditions are:
1. $\sum_i a_i = 1$
2. $a_i \geq 0$ for all $i$
3. $c_k = \sum_i a_i a_{k-i} = M$ for $k \in S$ (active)
4. $c_k \leq M$ for $k \notin S$ (inactive)
5. For $a_i > 0$: $\frac{\partial}{\partial a_i}\left(\sum_{k \in S} \lambda_k c_k + \mu \sum_i a_i\right) = 0$

where $\lambda_k \geq 0$ are Lagrange multipliers for the active constraints.

Condition 5 gives: $\sum_{k \in S} \lambda_k \cdot 2a_{k-i} + \mu = 0$ for $a_i > 0$.

This is a system of equations in the unknowns $a_i$ (on the support), $M$, $\lambda_k$, and $\mu$. It's non-linear because of the $a_{k-i}$ terms.

However, if we fix the support and active set, we can solve this iteratively:
- Given $\lambda$ and $\mu$, solve for $a$ from the linear system $\sum_{k \in S} \lambda_k a_{k-i} = -\mu/2$ for $i$ in support, plus $\sum a_i = 1$
- Given $a$, compute $c$ and update $M = \max(c_k)$
- Update the active set and $\lambda$

This is essentially the "primal-dual" approach.

"""
Multi-resolution optimizer with active-set focused smooth max,
larger-n exploration, and two-stage refinement.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _topk_soft_max(c, beta, k=20):
    """Smooth max over only the top-k values."""
    if k >= len(c):
        k = len(c)
    # Find top k indices
    if k == len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argpartition(c, -k)[-k:]
    c_top = c[idx]
    c_max = float(c_top.max())
    shifted = (c_top - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max, np.zeros(len(c))
    w_top = e / s
    w = np.zeros(len(c))
    w[idx] = w_top
    return c_max + beta * np.log(s), w


def _grad_structured(a, w, n):
    """Gradient of soft max using precomputed weights."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


class _SMOProb:
    def __init__(self, beta, n, topk=20):
        self.beta = beta
        self.n = n
        self.topk = topk

    def obj(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        val, _ = _topk_soft_max(c, self.beta, self.topk)
        return 2.0 * self.n * val

    def grad(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        _, w = _topk_soft_max(c, self.beta, self.topk)
        ga = _grad_structured(a, w, self.n)
        W = float(np.dot(ga, a))
        return a * (ga - W)

    def __call__(self, z):
        return self.obj(z)

    s = a.sum()
    a = a / s

    best_a = a.copy()

    # Use the gap between the top few values

def _mirror_descent(a_init, deadline, min_time_remaining=30.0, verbose=True):

    n = len(a_init)
    a = a_init.copy()

    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    while time.monotonic() + min_time_remaining < deadline:

        cur_sc = 2.0 * n * c_max

            best_sc = cur_sc

            no_improve += 1

        if no_improve > 500:
            break

        # Gradient of smooth max (use small beta for near-exact gradient)
        beta = c_max * 0.001
        _, w = _topk_soft_max(c, beta, 10)
        ga = _grad_structured(a, w, n)
        # Center the gradient (maintain sum=1 direction)
        ga_centered = ga - np.mean(ga)

        # Mirror descent update: a_i <- a_i * exp(-eta * ga_i)
        log_a = np.log(np.maximum(a, 1e-30))
        log_a -= eta * ga_centered
        log_a -= log_a.mean()  # maintain sum=1 in log space (approximately)

            a = a_new
            # Reduce eta gradually
            eta *= 0.999
        else:
            eta *= 0.9
            if eta < 1e-6:

def _active_set_qp(a_init, deadline, min_time_remaining=20.0):
    """Local QP on the active set to refine the solution.
    
    At optimum with active set S: for i in support(a):
    sum_{k in S} lambda_k * 2 * a[k-i] = -mu (constant)
    
    We solve this iteratively, updating S and lambda.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    max_iter = 500
    for it in range(max_iter):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        cur_sc = 2.0 * n * c_max

        if cur_sc < best_sc - 1e-14:
            best_sc = cur_sc
            best_a = a.copy()

        # Identify active set (top values)
        top_k = min(50, len(c))
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        # Refine: only truly "active" (within 1% of max)
        active_mask = c >= c_max * 0.999
        active_idx = np.where(active_mask)[0]
        if len(active_idx) < 2:
            active_idx = top_idx[:5]
        S = active_idx

        # Compute the "target gradient" from active set
        # grad_avg[i] = 2/|S| * sum_{k in S} a[k-i]
        a_rev = a[::-1]
        # Use uniform weights on active set
        w = np.zeros(2 * n - 1)
        w[S] = 1.0 / len(S)
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]

        # Support of a
        support = a > 1e-6 * a.max()
        n_sup = int(support.sum())
        if n_sup < 2:
            break

        # KKT: grad_avg should be constant on support
        # Target: grad_avg[i] = mu for i in support
        mu = float(np.mean(grad_avg[support]))
        residual = grad_avg[support] - mu

        # Newton-like update: adjust a to reduce residual
        # da = -eta * residual (on support)
        eta = 0.02
        a_new = a.copy()
        a_new[support] -= eta * residual
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < cur_sc - 1e-14:
            a = a_new
        else:
            # Try smaller step
            improved = False
            for trial_eta in [0.01, 0.005, 0.002, 0.001]:
                a_try = a.copy()
                a_try[support] -= trial_eta * residual
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:

                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < cur_sc - 1e-14:
                    a = a_try
                    improved = True
                    break
            if not improved:
                break

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":

    elif kind == "tapered":

    elif kind == "ramp":

        return a / a.sum()
    elif kind == "ramp2":

        a = t ** 1.5 + 0.01
        return a / a.sum()
    elif kind == "asymmetric_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.8) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.3) ** 2) + np.exp(-40.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "optimal_power":
        # Try different power laws
        t = np.linspace(0, 1, n)
        a = t ** 2.0 * (1 - t) ** 0.5 + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
