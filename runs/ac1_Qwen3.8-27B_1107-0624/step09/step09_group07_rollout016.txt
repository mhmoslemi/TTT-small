"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive LSE-smoothed L-BFGS, peak-tracking subgradient with BB steps,
simulated annealing restarts, and multi-n exploration.
Key improvements: p-norm hybrid objective, peak-aware active sets,
asymmetry-exploiting perturbations, and aggressive final polish.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_conv(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _lse_obj_grad(z, beta, n):
    """Full LSE smoothed objective and gradient in log-domain."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        w = e / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    ga = _grad_conv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


def _pnorm_obj_grad(z, p, n):
    """p-norm smoothed objective and gradient in log-domain."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = c / c_max
    shifted = np.clip(shifted, 1e-30, None)
    lp = shifted ** p
    sum_lp = lp.sum()
    if sum_lp < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        w = lp / sum_lp
        obj = 2.0 * n * c_max * sum_lp ** (1.0 / p)
    ga = _grad_conv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    obj_scaled = obj / (2.0 * n * c_max)
    grad_scaled = grad_z / (2.0 * n * c_max)
    return obj_scaled, grad_scaled


class _LSEProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _lse_obj_grad(z, self.beta, self.n)[0]

    def grad(self, z):
        return _lse_obj_grad(z, self.beta, self.n)[1]


class _PNormProb:
    def __init__(self, p, n):
        self.p = p
        self.n = n

    def __call__(self, z):
        return _pnorm_obj_grad(z, self.p, self.n)[0]

    def grad(self, z):
        return _pnorm_obj_grad(z, self.p, self.n)[1]


def _run_lbfgs_lse(a_init, deadline, min_time_remaining=30.0, beta_schedule=None):
    """L-BFGS with LSE smoothing and adaptive beta schedule."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a

    base_beta = max_c * 0.02

    if beta_schedule is None:
        beta_schedule = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for beta_mult in beta_schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 15:
            break

        beta = base_beta * beta_mult
        est_time = 0.004 * max(1.0, n / 8192.0)
        max_iter = min(8000, max(100, int(remaining * 0.7 / (est_time * 30))))

        prob = _LSEProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxcor': 150, 'maxls': 60}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a

def _subgradient_peak_tracking(a_init, deadline, min_time_remaining=60.0):
    """Subgradient with peak tracking and BB steps."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n_2nm1 = 2 * n - 1

    momentum = 0.6
    lr = 0.01
    lr_min = 1e-5
    lr_max = 0.05
    stall = 0
    max_stall = 100
    velocity = np.zeros(n)

    prev_a = None
    prev_g = None

    active_frac = 0.005
    peak_history = {}

    t_start = time.monotonic()
    last_lbfgs_time = t_start
    iteration = 0

    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Track peaks
        top_k_idx = np.argsort(c)[::-1][:max(5, n_2nm1 // 200)]
        for idx in top_k_idx:
            peak_history[int(idx)] = peak_history.get(int(idx), 0) + 1

        # Select active set: top peaks weighted by persistence
        n_active = max(5, min(n_2nm1, int(n_2nm1 * active_frac)))
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_active]

        # Weight by persistence (older peaks get more weight)
        w = np.zeros(n_2nm1)
        for idx in active_idx:
            weight = 1.0 + 0.1 * peak_history.get(int(idx), 0)
            w[idx] = weight
        w = w / w.sum()

        g = _grad_conv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break

        # BB step size
        if prev_a is not None and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g_centered - prev_g
            s_norm = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm > 1e-18 and abs(sy) > 1e-18:
                lr_bb = s_norm / abs(sy)
                lr = np.clip(lr_bb, lr_min, lr_max)

        velocity = momentum * velocity - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()

        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            for ls in range(30):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-15:
                    improved = True
                    break
                velocity = momentum * velocity * 0.5 - lr * 0.5 * g_centered
                a_try = a + velocity
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    velocity = np.zeros(n)
                    a_try = a
                    break
                a_try = a_try / s_try
                lr *= 0.7

        if improved:
            prev_a = a.copy()
            prev_g = g_centered.copy()
            a = a_try
            lr = min(lr_max, lr * 1.02)
            active_frac = max(0.001, active_frac * 0.97)
        else:
            velocity = np.zeros(n)
            lr = max(lr_min, lr * 0.6)
            active_frac = min(0.05, active_frac * 1.3)

        if stall > max_stall:
            active_frac = min(0.05, active_frac * 2.0)
            lr = max(lr_min, lr * 0.3)
            momentum = min(0.85, momentum + 0.05)
            stall = 0
            # Reset peak history partially
            for key in peak_history:
                peak_history[key] = max(0, peak_history[key] - 5)

        # Periodic L-BFGS
        if iteration % 400 == 0 and time.monotonic() - last_lbfgs_time > 180:
            if time.monotonic() + min_time_remaining + 120 < deadline:
                lbfgs_deadline = time.monotonic() + 100
                a_tmp = _run_lbfgs_lse(a, lbfgs_deadline, min_time_remaining=5)
                sc_tmp = _score(a_tmp)
                if sc_tmp < best_sc:
                    best_a = a_tmp.copy()
                    best_sc = sc_tmp
                    a = a_tmp.copy()
                    velocity = np.zeros(n)
                    prev_a = None
                    prev_g = None
                    peak_history = {}
                last_lbfgs_time = time.monotonic()

    return best_a, best_sc


def _peak_equalization(a_init, deadline, min_time_remaining=30.0):
    """Directly equalize top peaks by redistributing mass."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    patience = 0
    max_patience = 300

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break

        # Find top peaks
        n_peaks = max(3, min(20, int((2 * n - 1) * 0.003)))
        top_idx = np.argsort(c)[::-1][:n_peaks]
        top_vals = c[top_idx]
        peak_mean = top_vals.mean()

        # For each peak, find top contributors and reduce them
        delta = np.zeros(n)
        total_reduction = 0.0

        for k_peak in top_idx:
            excess = c[k_peak] - peak_mean
            if excess <= 0:
                continue
            i_lo = max(0, k_peak - (n - 1))
            i_hi = min(n - 1, k_peak)
            contrib = np.zeros(n)
            for i in range(i_lo, i_hi + 1):
                j = k_peak - i
                contrib[i] += a[j]
                if i != j:
                    contrib[j] += a[i]
            top_contrib = np.argsort(contrib)[::-1][:max(1, int(n * 0.001))]
            for idx in top_contrib:
                if contrib[idx] > 1e-15:
                    red = excess * 0.003 / (contrib[idx] + 1e-30)
                    delta[idx] -= red
                    total_reduction += red

        # Redistribute to low-contributing positions
        if total_reduction > 1e-15:
            # Find positions that contribute least to any peak
            peak_contrib = np.zeros(n)
            for k_peak in top_idx:
                i_lo = max(0, k_peak - (n - 1))
                i_hi = min(n - 1, k_peak)
                for i in range(i_lo, i_hi + 1):
                    j = k_peak - i
                    peak_contrib[i] += c[k_peak] * a[j]
                    if i != j:
                        peak_contrib[j] += c[k_peak] * a[i]
            low_idx = np.argsort(peak_contrib)[:int(n * 0.002)]
            for idx in low_idx:
                delta[idx] = total_reduction / len(low_idx)

        delta -= delta.mean()

        # Line search
        alpha = 1.0
        improved = False
        for ls in range(35):
            a_try = a + alpha * delta
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                patience = max(0, patience - 5)
                break
            alpha *= 0.5

        if not improved:
            patience += 5

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_exp":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t) + 0.1
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_exp2":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.5 * t) + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2) + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quarter_wave":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t / 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb_random(a, sigma, rng):
    """Gaussian perturbation in log-space."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _perturb_asym(a, rng):
    """Asymmetry-breaking perturbation: mirror + noise."""
    n = len(a)
    a_mir = a[::-1].copy()
    t = 0.1 + 0.15 * rng.random()
    a_mix = (1 - t) * a + t * a_mir
    z = np.log(np.maximum(a_mix, 1e-30))
    z = z + 0.05 * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _perturb_block(a, rng):
    """Block perturbation: scale random contiguous blocks."""
    n = len(a)
    a_new = a.copy()
    n_blocks = rng.integers(3, 10)
    for _ in range(n_blocks):
        start = rng.integers(0, n - 10)
        length = rng.integers(5, max(6, n // 50))
        scale = np.exp(rng.normal(0, 0.2))
        a_new[start:start + length] *= scale
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _simulated_annealing(a_init, deadline, min_time_remaining=60.0, rng=None):
    """SA with structured perturbations and L-BFGS refinement."""
    if rng is None:
        rng = np.random.default_rng(12345)

    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    T = 0.002
    T_min = 0.0001
    cooling = 0.995
    iterations = 0
    no_improve = 0
    max_no_improve = 50

    while time.monotonic() + min_time_remaining < deadline:
        iterations += 1
        # Choose perturbation type
        ptype = rng.random()
        if ptype < 0.4:
            a_try = _perturb_random(a, 0.02 + 0.03 * T / T_min * 0.1, rng)
        elif ptype < 0.7:
            a_try = _perturb_asym(a, rng)
        else:
            a_try = _perturb_block(a, rng)

        sc_try = _score(a_try)
        delta = sc_try - best_sc

        # Accept if better or with probability exp(-delta/T)
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            a = a_try
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        T = max(T_min, T * cooling)

        # Periodic L-BFGS from current state
        if iterations % 30 == 0 and no_improve > max_no_improve:
            if time.monotonic() + min_time_remaining + 60 < deadline:
                lb_deadline = time.monotonic() + 50
                a_refined = _run_lbfgs_lse(a, lb_deadline, min_time_remaining=5)
                sc_refined = _score(a_refined)
                if sc_refined < best_sc:
                    best_sc = sc_refined
                    best_a = a_refined.copy()
                    a = a_refined.copy()
                    no_improve = 0
                    T = max(T, 0.001)
            else:
                break

    return best_a, best_sc


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    """Ultra-fine active-set polish."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.003
    tol = 0.003
    no_improve = 0
    max_no_improve = 500

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Active set: peaks within tol of max
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active

        g = _grad_conv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-16:
            break

        alpha = min(lr, 0.3 / (gnorm + 1e-15))
        improved = False
        for ls in range(50):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 50 == 0:
                tol = max(0.00005, tol * 0.7)
                lr *= 0.7
            if no_improve > max_no_improve:
                a = best_a.copy()
                tol = max(0.00005, tol * 0.5)
                lr = 0.002
                no_improve = 0
        else:
            if tol > 0.00005:
                tol = max(0.00005, tol * 0.96)
            lr = min(0.008, lr * 1.01)

    return best_a, best_sc


def _mirror_descent(a_init, deadline, min_time_remaining=30.0):
    """Mirror descent on the probability simplex."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    eta = 0.01
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Use top peaks for gradient
        n_active = max(5, int((2 * n - 1) * 0.005))
        top_idx = np.argsort(c)[::-1][:n_active]
        w = np.zeros(2 * n - 1)
        w[top_idx] = 1.0 / n_active

        g = _grad_conv(a, w)
        g_centered = g - g.mean()

        # Mirror descent step: a_new_i ∝ a_i * exp(-eta * g_i)
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - eta * g_centered
        log_a_new = log_a_new - log_a_new.max()
        a_try = np.exp(log_a_new)
        a_try = a_try / a_try.sum()

        sc_try = _score(a_try)
        if sc_try < current_sc - 1e-15:
            a = a_try
            eta = min(0.05, eta * 1.05)
        else:
            no_improve += 1
            eta *= 0.8
            if no_improve > 200:
                break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: LSE L-BFGS, p-norm L-BFGS, peak-tracking subgradient,
    peak equalization, simulated annealing, mirror descent, multi-n exploration,
    and final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, tapered n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS with LSE (coarse to fine) ===
    print(f"\n[P1] L-BFGS LSE (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        a1 = _run_lbfgs_lse(best_seq, min(deadline - 550, time.monotonic() + 400),
                            min_time_remaining=60)
        sc1 = _score(a1)
        if sc1 < best_score:
            best_seq = a1.copy()
            best_score = sc1
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_lse] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: L-BFGS with p-norm ===
    print(f"\n[P2] L-BFGS p-norm (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        a2 = _run_lbfgs_pnorm(best_seq, min(deadline - 450, time.monotonic() + 300),
                              min_time_remaining=60)
        sc2 = _score(a2)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_pnorm] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Peak-tracking subgradient ===
    print(f"\n[P3] Peak-tracking subgradient (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a3, sc3 = _subgradient_peak_tracking(best_seq, min(deadline - 350, time.monotonic() + 350),
                                             min_time_remaining=60)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [subgrad] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Peak equalization ===
    print(f"\n[P4] Peak equalization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a4, sc4 = _peak_equalization(best_seq, min(deadline - 250, time.monotonic() + 250),
                                     min_time_remaining=30)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [peak_eq] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Simulated annealing ===
    print(f"\n[P5] Simulated annealing (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        a5, sc5 = _simulated_annealing(best_seq, min(deadline - 200, time.monotonic() + 250),
                                       min_time_remaining=50, rng=rng)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [sa] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Mirror descent ===
    print(f"\n[P6] Mirror descent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a6, sc6 = _mirror_descent(best_seq, min(deadline - 150, time.monotonic() + 150),
                                  min_time_remaining=30)
        if sc6 < best_score:
            best_seq = a6.copy()
            best_score = sc6
            print(f"  NEW BEST: {best_score:.10f} [mirror] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: L-BFGS round 2 (very small beta) ===
    print(f"\n[P7] L-BFGS fine (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        beta_fine = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
        a7 = _run_lbfgs_lse(best_seq, min(deadline - 100, time.monotonic() + 150),
                            min_time_remaining=30, beta_schedule=beta_fine)
        sc7 = _score(a7)
        if sc7 < best_score:
            best_seq = a7.copy()
            best_score = sc7
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_fine] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Multi-n exploration ===
    print(f"\n[P8] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        n_main = len(best_seq)
        n_options = [n_main // 2, n_main * 2, n_main * 3 // 2]
        n_options = [x for x in n_options if x != n_main and 4096 <= x <= 65536]

        for n_try in n_options[:3]:
            if time.monotonic() > deadline - 50:
                break
            a_res = _resample(best_seq, n_try)
            a_opt = _run_lbfgs_lse(a_res, min(deadline - 40, time.monotonic() + 40),
                                   min_time_remaining=5)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [n={n_try}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 9: Final polish ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 10:
        a_fin, sc_fin = _fine_polish(best_seq, deadline - 5, min_time_remaining=3)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [polish] t={time.monotonic()-t_start:.1f}s")

        # Tiny L-BFGS
        remaining = deadline - 5 - time.monotonic()
        if remaining > 10:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.0001
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _LSEProb(beta_fin, n)
            max_iter = min(5000, max(50, int(remaining / 0.02)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-20,
                                           'gtol': 1e-15, 'maxcor': 200, 'maxls': 80})
                a_fin2 = np.exp(result.x - result.x.max())
                a_fin2 = a_fin2 / a_fin2.sum()
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
