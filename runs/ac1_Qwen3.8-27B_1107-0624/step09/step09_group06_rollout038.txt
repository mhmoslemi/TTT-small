"""
Hybrid optimizer: Spectral equalization (Gerchberg-Saxton) + L-BFGS + Active-set methods
for minimizing 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)

    """Gerchberg-Saxton type spectral equalization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    rng = np.random.default_rng(seed)
    
    N = _next_pow2(2 * n)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute FFT
        A = np.fft.fft(a, N)
        P = np.abs(A) ** 2
        
        # Compress power spectrum
        t = it / max(1, n_iters - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * t
        mu = P.mean()
        P_new = mu + alpha * (P - mu)
        P_new = np.maximum(P_new, 1e-30)
        
        # Reconstruct
        mag = np.sqrt(P_new)
        phase = np.angle(A)
        A_new = mag * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real[:n]
        
        # Project to simplex
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)
    def grad(self, z):
        return _smograd(z, self.beta, self.n)

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0, max_beta_levels=20):
    """L-BFGS with adaptive beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    K_check = min(30, len(c_sorted))
    gap = c_sorted[0] - c_sorted[K_check - 1]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.002) * beta_scale
    
    # Generate beta schedule
    betas = []
    b = base_beta
    for i in range(max_beta_levels):
        betas.append(b)
        b *= 0.5
        if b < max_c * 1e-6:
            break
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        est_time_per_iter = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 8))))
        
        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 3 == 0:
                    print(f"    L-BFGS beta_idx={i}: score={current_sc:.10f}")
        except Exception as e:
            print(f"    L-BFGS error at beta_idx={i}: {e}")
    
    return current_a

def _peak_shave(a_init, deadline, min_time_remaining=30.0):
    """Direct peak shaving: reduce the highest peaks by adjusting a."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.01
    stall = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 200:
            break
        
        # Find active peaks
        K = max(5, int(n * 0.003))
        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]
        
        # Compute gradient of sum of active peaks
        grad = np.zeros(n)
        for k in active:
            # d/d a_j of conv(a,a)[k] = a[(k-j) mod ...] + a[(j-k) mod ...]
            # For linear conv: d/d a_j of sum_i a[i]a[k-i] = a[k-j] + a[j-k] (if valid)
            # a[k-j] is valid when 0 <= k-j <= n-1, i.e., k-(n-1) <= j <= k
            # a[j-k] is valid when 0 <= j-k <= n-1, i.e., k <= j <= k+n-1
            j_lo1 = max(0, k - n + 1)
            j_hi1 = min(n - 1, k)
            if j_lo1 <= j_hi1:
                grad[j_lo1:j_hi1+1] += a[k - np.arange(j_lo1, j_hi1+1)]
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_lo2 <= j_hi2:
                grad[j_lo2:j_hi2+1] += a[np.arange(j_lo2, j_hi2+1) - k]
        
        # Project gradient to simplex tangent
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break
        
        # Step
        a_cand = a - lr * grad_t / gnorm
        a_new = _project_simplex(a_cand)
        
        c_new = _full_convolve(a_new, a_new)
        if float(c_new.max()) < c_max:
            a = a_new
            if stall > 50:
                lr = max(0.001, lr * 0.95)
        else:
            lr *= 0.7
            if lr < 0.0001:
                break
    
    return best_a, best_sc

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _make_profile(n, kind="uniform", seed=0):
    rng = np.random.default_rng(seed)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        a = np.abs(rng.standard_normal(n))
        return a / a.sum()
    elif kind == "cos_mod":
        freq = rng.integers(3, 20)
        amp = rng.uniform(0.1, 0.5)
        t = np.arange(n)
        a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.1, 0.3)
        a = np.exp(-((t - center) / width) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

"""
Hybrid optimizer combining spectral equalization (Gerchberg-Saxton), L-BFGS with
adaptive beta schedules, peak shaving, and multi-n exploration for minimizing
2*n*max(conv(a,a))/sum(a)^2. Key innovation: frequency-domain spectral flattening
explores a different optimization landscape than gradient-based methods, potentially
finding better local minima.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _spectral_equalize(a_init, deadline, alpha_start=0.95, alpha_end=0.5,
                       n_iters=10000, seed=0, mode="compress"):
    """Gerchberg-Saxton type spectral equalization to flatten autocorrelation."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    N = _next_pow2(2 * n)
    no_improve = 0

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        A = np.fft.fft(a, N)
        P = np.abs(A) ** 2

        t_frac = it / max(1, n_iters - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * t_frac

        if mode == "compress":
            mu = P.mean()
            P_new = mu + alpha * (P - mu)
            P_new = np.maximum(P_new, 1e-30)
        elif mode == "floor":
            mu = P.mean()
            floor = alpha * mu
            P_new = np.maximum(P, floor)
            P_new = P_new * (mu / P_new.mean())
        elif mode == "soft":
            mu = P.mean()
            sigma = P.std() + 1e-30
            P_new = mu + alpha * sigma * np.tanh((P - mu) / (alpha * sigma + 1e-30))
            P_new = np.maximum(P_new, 1e-30)
        else:
            mu = P.mean()
            P_new = mu + alpha * (P - mu)
            P_new = np.maximum(P_new, 1e-30)

        mag = np.sqrt(P_new)
        phase = np.angle(A)
        A_new = mag * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real[:n]

        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc = _score(a_new)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5000:
                break

        a = a_new

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0,
               max_beta_levels=25, verbose=False):
    """L-BFGS with adaptive geometric beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    K_check = min(50, len(c_sorted))
    gap = c_sorted[0] - c_sorted[K_check - 1]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003) * beta_scale

    betas = []
    b = base_beta
    for _ in range(max_beta_levels):
        betas.append(b)
        b *= 0.6
        if b < max_c * 5e-7:
            break

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 8192.0)
        est_per_iter = 0.004 * scale
        max_iter = min(10000, max(100, int(remaining / (est_per_iter * 6))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose and i % 3 == 0:
                    print(f"    L-BFGS [{i}]: score={current_sc:.10f}")
        except Exception:
            pass

    return current_a


def _peak_shave(a_init, deadline, min_time_remaining=30.0):
    """Subgradient method focusing on active peaks with adaptive K."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.005
    K = max(5, int(n * 0.002))
    no_improve = 0
    v = np.zeros(n)
    momentum = 0.8

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            break

        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]

        grad = np.zeros(n)
        for k in active:
            j_range = np.arange(n)
            idx1 = k - j_range
            idx2 = j_range - k
            m1 = (idx1 >= 0) & (idx1 < n)
            m2 = (idx2 >= 0) & (idx2 < n)
            g = np.zeros(n)
            g[m1] += a[idx1[m1]]
            g[m2] += a[idx2[m2]]
            grad += g

        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break

        v = momentum * v - lr * grad_t / gnorm
        a_cand = a + v
        a_new = _project_simplex(a_cand)

        c_new = _full_convolve(a_new, a_new)
        if float(c_new.max()) < c_max:
            a = a_new
        else:
            v *= 0.5
            lr *= 0.95

        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(K * 2, len(c_sorted) - 1)]
        if gap < c_max * 0.003 and K < 200:
            K = min(200, int(K * 1.2) + 1)

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
