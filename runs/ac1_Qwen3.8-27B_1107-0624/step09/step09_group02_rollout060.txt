    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _make_profile(n, kind="uniform"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        a = 0.3 + 0.7 * t
        return a / a.sum()
    elif kind == "bump_center":
        a = np.exp(-6.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    elif kind == "bump_off":
        a = np.exp(-6.0 * (t - 0.3) ** 2) + 0.05
        return a / a.sum()
    elif kind == "arch":
        a = 4.0 * t * (1.0 - t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss_wide":
        a = np.exp(-2.0 * (t - 0.5) ** 2) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _spectral_init(n, rng, iters=200):
    """Spectral method: iterate between time and frequency domain to find
    a non-negative sequence whose self-convolution is flat."""
    N = _next_pow2(2 * n - 1)
    # Flat target in frequency domain
    target_fft = np.ones(N // 2 + 1)
    # Start with random non-negative
    a = np.abs(rng.standard_normal(n))
    a = a / a.sum()
    for it in range(iters):
        # Compute current spectrum
        A = np.fft.rfft(a, N)
        # Target: flat magnitude
        mag = np.abs(A)
        mag = np.maximum(mag, 1e-10)
        # Adjust: blend current magnitude with flat target
        alpha = 0.3
        new_mag = (1 - alpha) * mag + alpha * np.mean(mag)
        # Reconstruct: use current phase, new magnitude
        phase = np.angle(A)
        new_A = new_mag * np.exp(1j * phase)
        # Inverse FFT
        a_new = np.fft.irfft(new_A, N)[:n]
        # Clip and normalize
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        # Check if converged
        if np.allclose(a_new, a, atol=1e-10):
            break
        a = a_new
    return a


def _spectral_init_v2(n, rng, iters=100, flat_frac=0.5):
    """Alternative spectral init: target convolution is 'partially flat'."""
    N = _next_pow2(2 * n - 1)
    # Target: a trapezoid (blend of flat and triangular)
    conv_len = 2 * n - 1
    x = np.arange(conv_len)
    tri = np.minimum(x + 1, 2 * n - x) / n  # triangular shape, peak 1
    target_conv = (1 - flat_frac) * tri + flat_frac / conv_len
    target_conv = np.maximum(target_conv, 1e-10)
    # FFT of target
    target_fft = np.fft.rfft(target_conv, N)
    target_mag = np.abs(target_fft)
    target_mag = np.maximum(target_mag, 1e-10)
    # Start
    a = np.abs(rng.standard_normal(n)) + 0.1
    a = a / a.sum()
    for it in range(iters):
        A = np.fft.rfft(a, N)
        # Blend magnitude towards target
        alpha = 0.5
        cur_mag = np.maximum(np.abs(A), 1e-10)
        new_mag = (1 - alpha) * cur_mag + alpha * target_mag
        phase = np.angle(A)
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(new_A, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        if np.allclose(a_new, a, atol=1e-12):
            break
        a = a_new
    return a


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0, beta_start_scale=0.005, n_beta_steps=25):
    """L-BFGS in log-space with log-sum-exp smoothing and adaptive beta."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    current_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    best_a = a.copy()
    best_sc = current_sc

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap * 0.05, max_c * 0.001)

    ratios = [1.0, 0.5, 0.25, 0.15, 0.08, 0.05, 0.03, 0.02, 0.012, 0.008,
              0.005, 0.003, 0.002, 0.0012, 0.0008, 0.0005, 0.0003, 0.0002,
              0.00012, 0.00008, 0.00005, 0.00003, 0.00002, 0.00001, 0.000005]

    for i, ratio in enumerate(ratios):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        beta = base_beta * ratio
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(2000, max(50, int(remaining / (est_time * 4))))

        try:
            def obj_func(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    return 2.0 * n * c_max_local
                return 2.0 * n * (c_max_local + beta * np.log(s_e))

            def grad_func(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    w = np.zeros_like(c_local)
                    w[int(np.argmax(c_local))] = 1.0
                else:
                    w = e / s_e
                a_rev = a_local[::-1]
                wc = _full_convolve(w, a_rev)
                h = wc[n - 1:2 * n - 1]
                W_val = float(np.dot(w, c_local))
                return 4.0 * n * a_local * (h - W_val)

            result = minimize(
                obj_func, z,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 40, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return best_a


def _run_subgrad_mirror(a_init, deadline, min_time_remaining=60.0, n_iter_max=50000):
    """Mirror descent (multiplicative weights) with random peak selection
    and adaptive step size via backtracking."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())
    eta = 0.1  # initial step size
    patience = 0
    max_patience = 200
    best_eta = eta

    for it in range(n_iter_max):
        if time.monotonic() + min_time_remaining > deadline:
            break
        if it % 500 == 0 and it > 0:
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    a = best_a.copy()
                    eta = best_eta * 0.5
                    patience = 0

        # Recompute convolution periodically
        if it % 10 == 0 or it == 0:
            c = _full_convolve(a, a)
            c_max = float(c.max())

        # Select random peaks (top-k with random weighting)
        k_act = np.random.randint(1, min(11, 2 * n))
        # Sample from top portion of convolution
        threshold = np.percentile(c, 100 - k_act * 50 / (2 * n - 1))
        candidates = np.where(c >= threshold)[0]
        if len(candidates) < k_act:
            candidates = np.argsort(c)[::-1][:k_act]
        selected = np.random.choice(candidates, size=min(k_act, len(candidates)),
                                     replace=False)
        # Weight peaks by their value
        weights = c[selected] - c_min_sel
        c_min_sel = c[selected].min()
        weights = c[selected] / c[selected].sum()

        # Compute subgradient: for each selected peak k, dc_k/da = shifted a
        # dc_k/da[i] = 2*a[k-i] (for k≠2i) or 2*a[i] (for k=2i)
        # Approximate: grad[i] = 2 * sum_

k w_k * a[k-i] (with appropriate bounds)
        a_rev = a[::-1]
        w_sel = np.zeros(2 * n - 1)
        w_sel[selected] = weights
        conv_wa = _full_convolve(w_sel, a_rev)
        h = conv_wa[n - 1:2 * n - 1]
        # Correction for diagonal terms
        for idx in selected:
            if idx % 2 == 0:
                m = idx // 2
                if m < n:
                    h[m] -= a[m] * weights[selected == idx][0]
        grad = 2.0 * h

        # Mirror descent step: a <- a * exp(-eta * grad / max_grad), renormalize
        gmax = float(np.max(np.abs(grad)))
        if gmax < 1e-15:
            break
        log_a = np.log(np.maximum(a, 1e-30))
        log_a = log_a - eta * grad / gmax
        a_new = _softmax(log_a)

        # Backtracking: check if score improved
        c_new = _full_convolve(a_new, a_new)
        c_max_new = float(c_new.max())
        if c_max_new < c_max:
            a = a_new
            c = c_new
            c_max = c_max_new
        else:
            eta *= 0.7

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _run_coord_descent(a_init, deadline, min_time_remaining=60.0, n_passes_max=100):
    """Coordinate descent: for each coordinate, try a few candidate values
    and keep the best. Uses incremental convolution update."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    order = np.arange(n)
    factors = [0.0, 0.5, 0.75, 1.5, 2.0, 0.2, 3.0]

    for pass_num in range(n_passes_max):
        if time.monotonic() + min_time_remaining > deadline:
            break
        # Periodically re-shuffle
        np.random.shuffle(order)

        for i in order:
            if time.monotonic() + min_time_remaining > deadline:
                break
            if it_i_budget_check(pass_num, i) > deadline:
                break
            x_old = a[i]
            if x_old < 1e-15:
                # Try small positive value
                candidates_x = [1.0 / n]
            else:
                candidates_x = [x_old * f for f in factors]
                # Also try 0 (zero out)
                candidates_x.append(0.0)

            best_x = x_old
            best_c_max = c_max
            a[i] = 0  # temporarily remove
            # Compute base convolution (without a[i])
            # conv_base[k] = c[k] - contribution of a[i]
            # Contribution: a[i]*a[k-i] for valid k-i, plus a[i]^2 if k=2i
            contrib = np.zeros(2 * n - 1)
            for j in range(n):
                k = i + j
                if k < 2 * n - 1:
                    if j == i:
                        contrib[k] = x_old * x_old
                    else:
                        contrib[k] = x_old * a[j]
            base_c = c - contrib

            a[i] = x_old  # restore

            for x_try in candidates_x:
                if x_try < 0:
                    continue
                # New convolution: base_c + contribution of x_try
                new_contrib = np.zeros(2 * n - 1)
                for j in range(n):
                    k = i + j
                    if k < 2 * n - 1:
                        if j == i:
                            new_contrib[k] = x_try * x_try
                        else:
                            new_contrib[k] = x_try * a[j]
                new_c = base_c + new_contrib
                new_max = float(new_c.max())
                if new_max < best_c_max:
                    best_c_max = new_max
                    best_x = x_try

            # Also try binary search around best_x
            if best_x != x_old:
                # Fine-tune: try a few values around best_x
                for frac in [0.5, 0.75, 1.0, 1.25, 1.5]:
                    x_try2 = best_x * frac
                    if x_try2 < 0:
                        continue
                    new_contrib = np.zeros(2 * n - 1)
                    for j in range(n):
                        k = i + j
                        if k < 2 * n - 1:
                            if j == i:
                                new_contrib[k] = x_try2 * x_try2
                            else:
                                new_contrib[k] = x_try2 * a[j]
                    new_c = base_c + new_contrib
                    new_max = float(new_c.max())
                    if new_max < best_c_max:
                        best_c_max = new_max
                        best_x = x_try2

            if best_x != x_old:
                a[i] = best_x
                # Update c
                for j in range(n):
                    k = i + j
                    if k < 2 * n - 1:
                        if j == i:
                            c[k] += best_x * best_x - x_old * x_old
                        else:
                            c[k] += best_x * a[j] - x_old * a[j]
                c_max = float(c.max())

        # End of pass: rescale to maintain sum=1
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _coord_step_contribution(a, i, x_new, x_old):
    """Compute the change in convolution when a[i] changes from x_old to x_new."""
    n = len(a)
    delta = x_new - x_old
    # Change in conv[k]:
    # For k = i + j (j valid, j != i): delta * a[j]
    # For k = 2*i: (x_new^2 - x_old^2) - delta*x_old = delta*(x_new + x_old) - delta*x_old = delta*x_new
    # Actually: new - old = x_new*a[j] - x_old*a[j] = delta*a[j] for k=i+j, j≠i
    # For k=2i: x_new^2 - x_old^2 = delta*(x_new+x_old)
    # But in the original formula, the term at k=2i was x_old^2 (one term), now it's x_new^2
    # And the cross terms: x_new*a[j] for j != i at position i+j
    # vs x_old*a[j] for j != i at position i+j
    # So delta at position i+j (j≠i): delta * a[j]
    # Delta at position 2i: x_new^2 - x_old^2
    # But wait, position i+j where j=i is position 2i, and there we have x_new*x_new vs x_old*x_old
    # Position i+j where j≠i: the term x_new*a[j] replaces x_old*a[j], change = delta*a[j]
    # But also position j+i where j≠i: same position i+j, same change

    # Actually in the linear convolution:
    # conv[k] = sum_j a[j] * a[k-j]
    # Terms involving a[i]: j=i gives a[i]*a[k-i] (at position k=i+(k-i), so k-i = k-i)
    #              j=k-i gives a[k-i]*a[i] (at position k=(k-i)+i, same position)
    # If i != k-i (k != 2i): two terms, total contribution = 2*a[i]*a[k-i]
    #   Change: 2*delta*a[k-i]
    # If i == k-i (k = 2i): one term a[i]^2
    #   Change: x_new^2 - x_old^2

    # So the update to c is:
    # For each valid j (j != i, 0 <= j < n, k = i+j valid): c[i+j] += 2 * delta * a[j]
    # For k = 2i (if valid): c[2i] += x_new^2 - x_old^2
    # But wait, when j = i, k = 2i, and we already handled that case.
    # When j != i, k = i+j, and the contribution is 2*a[i]*a[j] (two terms in the sum).
    # Change: 2*delta*a[j]

    # Hmm but this double-counts. Let me reconsider.
    # conv[k] = sum_j a[j]*a[k-j]
    # The term a[i] appears when j=i (giving a[i]*a[k-i]) or when k-j=i, i.e., j=k-i (giving a[k-i]*a[i]).
    # If i != k-i: both terms exist, total = 2*a[i]*a[k-i], change = 2*delta*a[k-i]
    # If i == k-i (k=2i): only one term a[i]*a[i] = a[i]^2, change = x_new^2 - x_old^2

    # So for updating c when a[i] changes:
    result = np.zeros(2 * n - 1)
    for j in range(n):
        if j == i:
            k = 2 * i
            if k < 2 * n - 1:
                result[k] = x_new * x_new - x_old * x_old
        else:
            k = i + j
            if k < 2 * n - 1:
                result[k] = 2.0 * delta * a[j]
    return result


def it_i_budget_check(pass_num, i):
    """Simple check - always returns 0 here, actual check done in caller."""
    return 0


def _run_coord_descent_fast(a_init, deadline, min_time_remaining=60.0, n_passes_max=50):
    """Faster coordinate descent using vectorized contribution computation."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    rng = np.random.default_rng(12345)
    factors = np.array([0.0, 0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0])

    for pass_num in range(n_passes_max):
        if time.monotonic() + min_time_remaining > deadline:
            break
        order = np.arange(n)
        rng.shuffle(order)

        # Check time every 256 coordinates
        for coord_idx, i in enumerate(order):
            if coord_idx % 256 == 255:
                if time.monotonic() + min_time_remaining > deadline:
                    break

            x_old = a[i]
            candidates_x = x_old * factors
            candidates_x = np.maximum(candidates_x, 0)

            # Vectorized: compute new c for each candidate
            # c_new[x] = c - contrib_old + contrib_new
            # contrib_old[k] = for each j: if j==i: x_old^2 at k=2i; else: 2*x_old*a[j] at k=i+j
            # This is expensive to do for all candidates at once for large n
            # Instead, just try 2-3 candidates

            # Try zero and half
            best_x = x_old
            best_max = c_max

            # Quick check: try x=0
            if x_old > 1e-10:
                # c at x=0: remove contribution
                delta_k = np.zeros(2 * n - 1)
                for j in range(n):
                    if j == i:
                        k = 2 * i
                        if k < 2 * n - 1:
                            delta_k[k] = -x_old * x_old
                    else:
                        k = i + j
                        if k < 2 * n - 1:
                            delta_k[k] = -2.0 * x_old * a[j]
                new_c = c + delta_k
                new_max = float(new_c.max())
                if new_max < best_max:
                    best_max = new_max
                    best_x = 0.0

            # Try x = x_old * 0.5
            x_try = 0.5 * x_old
            if x_try > 0 and x_try != x_old:
                delta = x_try - x_old
                delta_k = np.zeros(2 * n - 1)
                for j in range(n):
                    if j == i:
                        k = 2 * i
                        if k < 2 * n - 1:
                            delta_k[k] = x_try * x_try - x_old * x_old
                    else:
                        k = i + j
                        if k < 2 * n - 1:
                            delta_k[k] = 2.0 * delta * a[j]
                new_c = c + delta_k
                new_max = float(new_c.max())
                if new_max < best_max:
                    best_max = new_max
                    best_x = x_try

            # Try x = x_old * 2
            x_try = 2.0 * x_old
            if x_try > 0:
                delta = x_try - x_old
                delta_k = np.zeros(2 * n - 1)
                for j in range(n):
                    if j == i:
                        k = 2 * i
                        if k < 2 * n - 1:
                            delta_k[k] = x_try * x_try - x_old * x_old
                    else:
                        k = i + j
                        if k < 2 * n - 1:
                            delta_k[k] = 2.0 * delta * a[j]
                new_c = c + delta_k
                new_max = float(new_c.max())
                if new_max < best_max:
                    best_max = new_max
                    best_x = x_try

            # Apply best
            if best_x != x_old:
                delta = best_x - x_old
                for j in range(n):
                    if j == i:
                        k = 2 * i
                        if k < 2 * n - 1:
                            c[k] += best_x * best_x - x_old * x_old
                    else:
                        k = i + j
                        if k < 2 * n - 1:
                            c[k] += 2.0 * delta * a[j]
                a[i] = best_x
                c_max = float(c.max())

        # End of pass
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _run_top_k_smoothed_grad(a_init, deadline, min_time_remaining=60.0):
    """Top-k smoothed gradient with Nesterov momentum and adaptive step."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    momentum = np.zeros(n)
    beta_m = 0.85
    patience = 0
    max_patience = 100
    eta = 1.0

    for step in range(100000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        if step % 100 == 99:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros(n)
                    patience = 0
                    eta *= 0.5
                    if eta < 1e-6:
                        break

        a_curr = _softmax(z)
        c = _full_convolve(a_curr, a_curr)
        c_max = float(c.max())

        # Top-k peaks with softmax weighting
        k_act = min(20, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k_act]
        # Exponential weights on top peaks
        peaks_vals = c[top_idx]
        temp = (peaks_vals.max() - peaks_vals.min()) + 1e-10
        w_peaks = np.exp((peaks_vals - peaks_vals.max()) / (temp * 0.1))
        w_peaks = w_peaks / w_peaks.sum()

        w = np.zeros(2 * n - 1)
        w[top_idx] = w_peaks

        # Compute gradient (same as in L-BFGS)
        a_rev = a_curr[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a_curr * (h - W_val)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Nesterov momentum
        momentum = beta_m * momentum + (1 - beta_m) * grad
        mnorm = np.linalg.norm(momentum)
        if mnorm < 1e-15:
            break

        lr = eta / (mnorm + 1e-10)
        z_new = z - lr * momentum
        z_new = np.clip(z_new, -40.0, 40.0)
        z_new = z_new - z_new.mean()
        z = z_new

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining spectral initialization, L-BFGS,
    mirror descent, top-k smoothed gradient, coordinate descent,
    and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Spectral initialization (60s) ===
    print(f"\n[Phase 1] Spectral Init (t={time.monotonic() - t_start:.1f}s)")
    p1_end = t_start + 60
    if time.monotonic() < p1_end:
        # Try spectral init at parent's n
        spec1 = _spectral_init(n_parent, rng, iters=150)
        sc1 = _score(spec1)
        if sc1 < best_score + 0.1:
            print(f"  spectral1 score={sc1:.6f}")
            update_best(spec1, sc1, "spectral1")

        spec2 = _spectral_init_v2(n_parent, rng, iters=100, flat_frac=0.3)
        sc2 = _score(spec2)
        if sc2 < best_score + 0.1:
            print(f"  spectral2 score={sc2:.6f}")
            update_best(spec2, sc2, "spectral2")

        spec3 = _spectral_init_v2(n_parent, rng, iters=100, flat_frac=0.7)
        sc3 = _score(spec3)
        if sc3 < best_score + 0.1:
            print(f"  spectral3 score={sc3:.6f}")
            update_best(spec3, sc3, "spectral3")

    # === Phase 2: L-BFGS from parent (400s) ===
    print(f"\n[Phase 2] L-BFGS parent (t={time.monotonic() - t_start:.1f}s)")
    p2_end = t_start + 460
    a_lb1 = _run_lbfgs(parent, p2_end, min_time_remaining=30.0)
    update_best(a_lb1, _score(a_lb1), "lbfgs_p")

    # L-BFGS from perturbed parent
    if time.monotonic() < p2_end - 60:
        a_pert = _perturb(parent, 0.02, rng)
        a_lb2 = _run_lbfgs(a_pert, p2_end, min_time_remaining=20.0)
        update_best(a_lb2, _score(a_lb2), "lbfgs_pert")

    # === Phase 3: Top-k smoothed gradient (300s) ===
    print(f"\n[Phase 3] Top-k Grad (t={time.monotonic() - t_start:.1f}s)")
    p3_end = t_start + 760
    if best_seq is not None:
        a_tk, sc_tk = _run_top_k_smoothed_grad(best_seq, p3_end, min_time_remaining=30.0)
        update_best(a_tk, sc_tk, "topk_grad")

    # === Phase 4: Mirror descent / subgradient (250s) ===
    print(f"\n[Phase 4] Mirror Descent (t={time.monotonic() - t_start:.1f}s)")
    p4_end = t_start + 1010
    if best_seq is not None:
        a_md, sc_md = _run_subgrad_mirror(best_seq, p4_end, min_time_remaining=30.0,
                                           n_iter_max=20000)
        update_best(a_md, sc_md, "mirror")

    # === Phase 5: Coordinate descent (250s) ===
    print(f"\n[Phase 5] Coord Descent (t={time.monotonic() - t_start:.1f}s)")
    p5_end = t_start + 1260
    if best_seq is not None:
        # Only do coord descent if n is not too large
        n_cd = len(best_seq)
        if n_cd <= 8192:
            a_cd, sc_cd = _run_coord_descent_fast(best_seq, p5_end, min_time_remaining=30.0,
                                                   n_passes_max=30)
            update_best(a_cd, sc_cd, "coord_cd")
        else:
            # For larger n, do a reduced version on a subset
            # Resample to smaller n for CD
            n_small = 4096
            a_small = _resample(best_seq, n_small)
            a_cd_s, sc_cd_s = _run_coord_descent_fast(a_small, p5_end, min_time_remaining=30.0,
                                                       n_passes_max=20)
            # Try to upsample back
            a_up = _resample(a_cd_s, n_cd)
            update_best(a_up, _score(a_up), "coord_cd_up")
            # Also keep the small one
            update_best(a_cd_s, sc_cd_s, "coord_cd_small")

    # === Phase 6: Second L-BFGS round (200s) ===
    print(f"\n[Phase 6] L-BFGS Round 2 (t={time.monotonic() - t_start:.1f}s)")
    p6_end = t_start + 1460
    if best_seq is not None and time.monotonic() < p6_end:
        a_lb3 = _run_lbfgs(best_seq, p6_end, min_time_remaining=30.0,
                            beta_start_scale=0.002, n_beta_steps=20)
        update_best(a_lb3, _score(a_lb3), "lbfgs_r2")

    # === Phase 7: Multi-n exploration (200s) ===
    print(f"\n[Phase 7] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    p7_end = t_start + 1660
    n_tries = [65536, 24576, 16384, 49152]
    for n_try in n_tries:
        if time.monotonic() > p7_end - 40:
            break
        if n_try == n_parent:
            continue
        if best_seq is not None:
            a_init_n = _resample(best_seq, n_try)
        else:
            a_init_n = _make_profile(n_try, "tapered")
        a_n = _run_lbfgs(a_init_n, p7_end, min_time_remaining=20.0,
                          beta_start_scale=0.003)
        sc_n = _score(a_n)
        update_best(a_n, sc_n, f"n_{n_try}")

    # === Phase 8: Final polish (100s) ===
    print(f"\n[Phase 8] Final Polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        # Very tight L-BFGS
        a_fin = _run_lbfgs(best_seq, deadline - 20, min_time_remaining=5.0,
                            beta_start_scale=0.001, n_beta_steps=10)
        update_best(a_fin, _score(a_fin), "final_lbfgs")

        # Top-k gradient with lower learning rate
        if time.monotonic() < deadline - 15:
            a_tk2, sc_tk2 = _run_top_k_smoothed_grad(best_seq, deadline - 15,
                                                       min_time_remaining=5.0)
            update_best(a_tk2, sc_tk2, "final_topk")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.6f}")
    return [float(x) for x in fallback]
