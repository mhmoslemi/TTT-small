"""
Optimized step function search: minimizes 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _smoothed_obj_grad(z, beta, n, active_k=None):
    """Compute smoothed objective and gradient for log-sum-exp smoothing.
    If active_k is given, only smooth over those indices (sharper approximation).
    """
    a = _softmax(z)
    c = _full_convolve(a, a)
    if active_k is None:
        c_max = float(c.max())
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
    else:
        c_max_full = float(c.max())
        c_act = c[active_k]
        c_max_act = float(c_act.max())
        # Use the global max as reference for the objective value
        shifted = (c_act - c_max_act) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            idx = int(np.argmax(c_act))
            w[active_k[idx]] = 1.0
        else:
            w_full = np.zeros_like(c)
            w_full[active_k] = e / s
            w = w_full
    # Gradient: 4*n*a*(conv(w, a_rev)[n-1:2n-1] - dot(w,c))
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    # Objective value
    c_max = float(c.max())
    if active_k is None:
        val = 2.0 * n * (c_max + beta * np.log(max(s, 1e-300)))
    else:
        # Use global max for the objective
        val = 2.0 * n * (c_max + beta * np.log(max(s, 1e-300)))
    return val, grad, c, c_max


class _SmoothedProb:
    def __init__(self, beta, n, active_k=None, update_period=50):
        self.beta = beta
        self.n = n
        self.active_k = active_k
        self.update_period = update_period
        self._call_count = 0
        self._c_cache = None
        self._c_max_cache = 0.0

    def __call__(self, z):
        self._call_count += 1
        val, _, c, c_max = _smoothed_obj_grad(z, self.beta, self.n, self.active_k)
        if self.active_k is None and self._call_count % self.update_period == 1:
            self._c_cache = c.copy()
            self._c_max_cache = c_max
        elif self.active_k is None:
            self._c_cache = c.copy()
            self._c_max_cache = c_max
        return val

    def grad(self, z):
        _, grad, c, c_max = _smoothed_obj_grad(z, self.beta, self.n, self.active_k)
        return grad

    def get_active_set(self, K=50):
        """Update active set based on current convolution."""
        if self._c_cache is None:
            return None
        c_idx = np.argsort(self._c_cache)[::-1]
        # Include indices where conv is within a small fraction of max
        threshold = self._c_max_cache - self.beta * 5
        active_mask = self._c_cache >= threshold
        active_indices = np.where(active_mask)[0]
        if len(active_indices) < K:
            active_indices = c_idx[:K]
        elif len(active_indices) > 200:
            active_indices = c_idx[:200]
        return active_indices

    # Extended fine schedule at the end

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive K: grow as peaks flatten
        if t % 10 == 0:
            c_sorted = np.sort(c)[::-1]
            gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
            if gap < c_max * 0.003 and K < K_max:
                K = min(K_max, int(K * 1.4) + 1)

        # Select active peaks
        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]

        # Compute subgradient (sum of top-K convolution gradients)
        grad = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                idx = k - np.arange(j_lo, j_hi + 1)
                grad[j_lo:j_hi + 1] += a[idx]
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_lo2 <= j_hi2:
                idx = np.arange(j_lo2, j_hi2 + 1) - k
                grad[j_lo2:j_hi2 + 1] += a[idx]

        # Project to tangent space of simplex
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-18:
            break

        # Step
        effective_lr = lr / (gnorm + 1e-18) * 0.05
        v = momentum * v - effective_lr * grad_t
        a_cand = a + v
        a_proj = _project_simplex(a_cand)

        c_new = _full_convolve(a_proj, a_proj)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new <= current_sc:
            a = a_proj
        else:
            v = 0.5 * v
            lr *= 0.95
            a = a_proj

        # Periodic restart
        if no_improve > 300 and no_improve % 100 == 0:
            perturb = 0.002 * rng.standard_normal(n)
            a_pert = _project_simplex(a + perturb)
            c_pert = _full_convolve(a_pert, a_pert)
            sc_pert = 2.0 * n * float(c_pert.max())
            if sc_pert < best_sc:
                a = a_pert
                best_a = a.copy()
                best_sc = sc_pert
            v = np.zeros(n)
            lr = base_lr
            no_improve = 0
            if no_improve > 2000:
                break

    return best_a, best_sc


def _exact_subgrad_polish(a_init, deadline, min_time_remaining=20.0):
    """Exact subgradient method using only the top active set."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    t = 0
    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()

        # Find active set (values within tiny epsilon of max)
        eps = c_max * 1e-10
        active_mask = c >= c_max - eps
        active = np.where(active_mask)[0]
        if len(active) == 0:
            break
        if len(active) > 50:
            c_idx = np.argsort(c)[::-1]
            active = c_idx[:50]

        # Subgradient of max over active set (averaged)
        grad = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                idx = k - np.arange(j_lo, j_hi + 1)
                grad[j_lo:j_hi + 1] += a[idx]
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_lo2 <= j_hi2:
                idx = np.arange(j_lo2, j_hi2 + 1) - k
                grad[j_lo2:j_hi2 + 1] += a[idx]
        grad /= len(active)

        # Project to tangent space
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break

        # Decreasing step size
        lr = 0.01 / np.sqrt(t + 1)
        a_cand = a - lr * grad_t / (gnorm + 1e-18) * 0.1
        a_proj = _project_simplex(a_cand)

        c_new = _full_convolve(a_proj, a_proj)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new <= current_sc * (1 + 1e-12):
            a = a_proj
        else:
            a = a_proj  # stay on simplex anyway

        if t > 5000:
            break

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.15 * np.cos(2.0 * np.pi * 13.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        taper = max(1, n // 100)
        for i in range(taper):
            a[i] = (i + 1) / taper
            a[n - 1 - i] = (i + 1) / taper
        return a / a.sum()
    else:
        return np.ones(n) / n


def _optimize_one_n(a_init, deadline, min_time_remaining=60.0, seed=0, verbose=True):
    """Multi-phase optimization for a single n."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(best_a)

    # Phase 1: Nesterov (coarse)
    if time.monotonic() < deadline - min_time_remaining - 120:
        p1_deadline = min(deadline - min_time_remaining - 120, time.monotonic() + 120)
        a_nes, sc_nes = _nesterov_subgrad(best_a, p1_deadline,
                                          min_time_remaining=30.0,
                                          K_init=10, K_max=200, seed=seed)
        if sc_nes < best_sc:
            best_a = a_nes.copy()
            best_sc = sc_nes
            if verbose:
                print(f"      Nesterov: {best_sc:.10f}")

    # Phase 2: L-BFGS adaptive (main)
    if time.monotonic() < deadline - min_time_remaining - 30:
        p2_deadline = min(deadline - min_time_remaining - 30, time.monotonic() + 300)
        a_lb = _lbfgs_adaptive(best_a, p2_deadline, min_time_remaining=30.0)
        sc_lb = _score(a_lb)
        if sc_lb < best_sc:
            best_a = a_lb.copy()
            best_sc = sc_lb
            if verbose:
                print(f"      L-BFGS: {best_sc:.10f}")

    # Phase 3: Exact subgradient polish
    if time.monotonic() < deadline - min_time_remaining - 10:
        p3_deadline = min(deadline - min_time_remaining - 10, time.monotonic() + 60)
        a_sg, sc_sg = _exact_subgrad_polish(best_a, p3_deadline, min_time_remaining=10.0)
        if sc_sg < best_sc:
            best_a = a_sg.copy()
            best_sc = sc_sg
            if verbose:
                print(f"      Subgrad polish: {best_sc:.10f}")

    # Phase 4: Final L-BFGS with very small beta
    if time.monotonic() < deadline - min_time_remaining:
        p4_deadline = min(deadline - min_time_remaining - 5, time.monotonic() + 120)
        a_lb2 = _lbfgs_adaptive(best_a, p4_deadline, min_time_remaining=5.0,
                                max_beta_rounds=8)
        sc_lb2 = _score(a_lb2)
        if sc_lb2 < best_sc:
            best_a = a_lb2.copy()
            best_sc = sc_lb2
            if verbose:
                print(f"      Final L-BFGS: {best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Multi-n search with adaptive L-BFGS + Nesterov + exact polish."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # Define n candidates
    n_candidates = [n_parent]
    for n_try in [16384, 8192, 24576, 32768, 12288, 65536]:
        if n_try not in n_candidates:
            n_candidates.append(n_try)

    # Sort by priority: parent first, then powers of 2, then others
    n_candidates.sort(key=lambda x: (x != n_parent, abs(x - 16384)))

    # Time allocation: give more time to parent n, less to others
    total_time = deadline - t_start - 30  # reserve 30s for final polish
    # Allocate proportional to expected value
    time_alloc = {}
    remaining_budget = total_time
    for i, n_try in enumerate(n_candidates):
        if i == 0:
            # Parent gets 40% of time
            alloc = min(remaining_budget, total_time * 0.40)
        elif i == 1:
            alloc = min(remaining_budget, total_time * 0.25)
        elif i == 2:
            alloc = min(remaining_budget, total_time * 0.15)
        else:
            alloc = min(remaining_budget, total_time * 0.08)
        time_alloc[n_try] = max(30.0, alloc)
        remaining_budget -= alloc

    # Process each n
    for n_try in n_candidates:
        elapsed = time.monotonic() - t_start
        if elapsed > deadline - 40:
            break
        alloc = time_alloc.get(n_try, 60.0)
        n_deadline = min(t_start + elapsed + alloc, deadline - 40)

        if n_try == len(best_seq):
            a_init = best_seq.copy()
        else:
            a_init = _resample(best_seq, n_try)

        sc_init = _score(a_init)
        print(f"\n  n={n_try}: init score={sc_init:.10f}, time_alloc={alloc:.0f}s")

        # Try different initializations
        best_for_n = a_init.copy()
        best_sc_n = sc_init

        # Init 1: resampled parent
        a_opt, sc_opt = _optimize_one_n(a_init, n_deadline, min_time_remaining=30.0,
                                        seed=seed, verbose=True)
        if sc_opt < best_sc_n:
            best_for_n = a_opt.copy()
            best_sc_n = sc_opt

        # Init 2: small perturbation of best
        if time.monotonic() < n_deadline - 60:
            a_pert = _project_simplex(best_for_n + 0.001 * rng.standard_normal(len(best_for_n)))
            a_opt2, sc_opt2 = _optimize_one_n(a_pert, n_deadline, min_time_remaining=30.0,
                                              seed=seed + 1, verbose=False)
            if sc_opt2 < best_sc_n:
                best_for_n = a_opt2.copy()
                best_sc_n = sc_opt2

        # Init 3: structured (cosine modulation)
        if time.monotonic() < n_deadline - 60:
            a_struct = _make_profile(n_try, "cos_mod")
            a_opt3, sc_opt3 = _optimize_one_n(a_struct, n_deadline, min_time_remaining=30.0,
                                              seed=seed + 2, verbose=False)
            if sc_opt3 < best_sc_n:
                best_for_n = a_opt3.copy()
                best_sc_n = sc_opt3

        if best_sc_n < best_score:
            best_seq = best_for_n.copy()
            best_score = best_sc_n
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) t={time.monotonic()-t_start:.1f}s")

    # Final polish on the best sequence
    remaining = deadline - time.monotonic() - 5
    if remaining > 20:
        print(f"\n  Final polish (remaining={remaining:.0f}s)")
        a_final, sc_final = _optimize_one_n(best_seq, deadline - 5,
                                            min_time_remaining=5.0,
                                            seed=seed + 100, verbose=True)
        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
