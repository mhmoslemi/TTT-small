"""
Optimized step function search using FFT-accelerated gradients exploiting
autoconvolution symmetry, Nesterov subgradient with power-sum smoothing,
L-BFGS with log-sum-exp, multi-n exploration, and coordinate descent polish.
Key innovation: O(n log n) gradient via the identity grad[j] = v[j] + v[2n-2-j]
where v = conv(a, c^p) and c = conv(a,a) (symmetric).

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _fast_power_grad(a, power=2):
    """
    Compute gradient of (sum_k c[k]^power) w.r.t. a, where c = conv(a,a).
    Uses symmetry: c is palindromic, so c^(power-1) is also palindromic.
    Returns (grad, c, score_proxy).
    """
    n = len(a)
    c = _full_convolve(a, a)
    if power == 1:
        w = c.copy()
    else:
        w = np.power(np.maximum(c, 0.0), power)
    v = _full_convolve(a, w)
    # v has length 3n-2, indices 0..3n-3
    # grad[j] = v[j] + v[2n-2-j] for j = 0..n-1
    # v[2n-2-j] for j=0..n-1 gives indices 2n-2, 2n-3, ..., n-1
    grad = v[:n] + v[n - 1:2 * n - 1][::-1]
    score_proxy = float(np.sum(np.power(np.maximum(c, 0.0), power)))
    return grad, c, score_proxy


def _lbfgs_smooth_objective(z, beta, n, cache=None):
    """Smooth objective using log-sum-exp of convolution peaks."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lbfgs_smooth_grad(z, beta, n):
    """Gradient of smooth objective w.r.t. z (log-softmax parameterization)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # dF/da[j] = 2n*beta*(v[j] + v[2n-2-j]) where v = conv(a, w), w symmetric
    v = _full_convolve(a, w)

def _nesterov_power(a_init, deadline, min_remaining=60.0, power=3,
                    K_init=20, K_max=200, seed=0):
    """
    Nesterov-accelerated subgradient using fast FFT-based power-gradient.
    Uses armijo backtracking line search.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    momentum = 0.8
    t = 0
    no_improve = 0
    stall_count = 0
    base_lr = 0.01
    lr = base_lr
    armijo_c = 0.0001
    armijo_rho = 0.5

    while time.monotonic() + min_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Fast gradient using power-sum
        grad, _, _ = _fast_power_grad(a, power=power)
        # Project to tangent space of simplex
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-18:
            break

        # Armijo backtracking
        step_lr = lr
        improved = False
        for _bt in range(10):
            v_new = momentum * v - step_lr * grad_t / (gnorm + 1e-18)
            a_cand = _project_simplex(a + v_new)
            c_new = _full_convolve(a_cand, a_cand)
            sc_new = 2.0 * n * float(c_new.max())
            if sc_new < current_sc - armijo_c * step_lr * gnorm:
                a = a_cand
                v = v_new
                improved = True
                break
            step_lr *= armijo_rho

        if not improved:
            v = v * 0.5
            a = _project_simplex(a - 0.001 * grad_t / (gnorm + 1e-18))

        # Periodic restart
        if no_improve % 300 == 0 and no_improve > 0:
            stall_count += 1
            if stall_count > 8:
                break
            perturb_scale = 0.005 * np.exp(-0.1 * stall_count)
            perturb = perturb_scale * rng.standard_normal(n)
            a_pert = _project_simplex(a + perturb)
            c_pert = _full_convolve(a_pert, a_pert)
            sc_pert = 2.0 * n * float(c_pert.max())
            if sc_pert < best_sc:
                a = a_pert
                best_a = a.copy()
                best_sc = sc_pert
            v = np.zeros(n)
            no_improve = 0

        if t % 1000 == 0:
            lr = min(base_lr * 2, lr * 1.002)
            print(f"    Nes t={t}: best={best_sc:.10f}, sc={current_sc:.10f}, "
                  f"power={power}, lr={lr:.5f}, stall={stall_count}")

    return best_a, best_sc

                         'gtol': 1e-14, 'maxcor': 70, 'maxls': 40}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 3 == 0:
                    print(f"    LBF beta/{i}: score={current_sc:.10f}")
        except Exception:
            pass

    return current_a


def _coordinate_descent_polish(a_init, deadline, min_remaining=30.0):
    """
    Limited coordinate descent: for each coordinate in random order,
    do a small local search to find the best value.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    rng = np.random.default_rng(12345)
    order = rng.permutation(n)
    # Only try a subset for speed
    n_try = min(n, 5000)

    for sweep in range(5):
        if time.monotonic() + min_remaining > deadline:
            break
        for idx in order[:n_try]:
            if time.monotonic() + min_remaining > deadline:
                break
            j = int(idx)
            old_val = a[j]
            # Try a few perturbed values
            a[j] = 0.0
            s_rest = a.sum()
            c0 = _full_convolve(a, a)
            # h[k] = a[k-j] (with bounds) - this is just a shifted copy
            h = np.zeros(2 * n - 1)
            lo = max(0, j)
            hi = min(n - 1, 2 * n - 1 - j)
            if lo <= hi:
                h[lo:hi + 1] = a[hi - j:lo - j if lo - j >= 0 else 0]
                # Actually: h[k] = a[k-j] if 0<=k-j<n else 0
                h = np.zeros(2 * n - 1)
                for k in range(max(0, j), min(2 * n - 1, j + n)):
                    h[k] = a[k - j]

            # Try values: old_val * [0.5, 0.75, 1.0, 1.25, 1.5]
            candidates = [old_val * f for f in [0.3, 0.5, 0.7, 1.0, 1.3, 1.7, 2.5, 0.0]]
            best_local = old_val
            best_local_sc = best_sc

            for x in candidates:
                x = max(0.0, x)
                a[j] = x
                s_total = s_rest + x
                if s_total < 1e-15:
                    a[j] = old_val
                    continue
                conv_x = c0 + 2.0 * x * h + x * x * np.zeros(2 * n - 1)
                conv_x[j + j if 2 * j < 2 * n - 1 else 0] += 0  # x^2 term only at k=2j
                if 2 * j < 2 * n - 1:
                    conv_x[2 * j] += x * x  # additional x^2 at k=2j from the a[j]*a[j] term
                    # Wait, c0 already has a[j]=0, so conv_x = c0 + 2*x*h + x^2*delta_{k,2j}
                    # But h already includes a[k-j] which when k=2j gives a[j]=0 (since a[j] was set to 0)
                    # So we need to add the x^2 term:
                    # Actually conv(a,a)[k] = c0[k] + x*a[k-j] + a[k-j]*x + x*a[j] * delta...
                    # Let me just recompute properly
                    pass
                sc = 2.0 * n * float(conv_x.max()) / (s_total * s_total)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_local = x

            a[j] = best_local
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
                break
            a = a / s

            if _score(a) < best_sc - 1e-13:
                best_sc = _score(a)
                best_a = a.copy()

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    else:
        return np.ones(n) / n


def _shrinkage_and_reoptimize(a, deadline, min_remaining=30.0):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    for frac in [0.001, 0.003, 0.005, 0.01, 0.02]:
        if time.monotonic() + min_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    P1: Nesterov subgradient with fast FFT power-gradient
    P2: L-BFGS with log-sum-exp smoothing and fine beta schedule
    P3: Second Nesterov pass with different power
    P4: L-BFGS ultra-fine
    P5: Multi-n exploration
    P6: Shrinkage + final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"Parent: n={len(parent)}, score={best_score:.10f}")
    else:
        n_init = 16384
        best_seq = _make_profile(n_init, "uniform")
        best_score = _score(best_seq)
        print(f"No parent, uniform n={n_init}, score={best_score:.10f}")

    n_best = len(best_seq)

    # === Phase 1: Nesterov power-subgradient (main workhorse) ===
    print(f"\n[P1] Nesterov power-subgrad (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 900, t_start + 450)
    if p1_deadline > time.monotonic() + 30:
        a_nes, sc_nes = _nesterov_power(best_seq, p1_deadline,
                                        min_remaining=30.0, power=3,
                                        K_init=20, K_max=200, seed=seed)
        if sc_nes < best_score:
            best_seq = a_nes.copy()
            best_score = sc_nes
            print(f"  NEW BEST: {best_score:.10f} [nest_p3] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Nest p3: no improvement (best={best_score:.10f})")

    # Try power=4 as well if time
    if time.monotonic() < deadline - 700:
        p1b_deadline = min(deadline - 700, t_start + 550)
        if p1b_deadline > time.monotonic() + 30:
            a_nes4, sc_nes4 = _nesterov_power(best_seq, p1b_deadline,
                                              min_remaining=30.0, power=4,
                                              K_init=10, K_max=100, seed=seed + 1)
            if sc_nes4 < best_score:
                best_seq = a_nes4.copy()
                best_score = sc_nes4
                print(f"  NEW BEST: {best_score:.10f} [nest_p4] t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: L-BFGS fine ===
    print(f"\n[P2] L-BFGS fine (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 600, t_start + 800)
    if p2_deadline > time.monotonic() + 50:
        a_lb = _run_lbfgs_repeated(best_seq, p2_deadline, min_remaining=50.0)
        sc_lb = _score(a_lb)
        if sc_lb < best_score:
            best_seq = a_lb.copy()
            best_score = sc_lb
            print(f"  NEW BEST: {best_score:.10f} [lbfgs1] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  LBF1: no improvement (best={best_score:.10f})")

    # === Phase 3: Nesterov round 2 ===
    print(f"\n[P3] Nesterov round 2 (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 400, t_start + 1000)
    if p3_deadline > time.monotonic() + 30:
        a_nes2, sc_nes2 = _nesterov_power(best_seq, p3_deadline,
                                          min_remaining=30.0, power=5,
                                          K_init=5, K_max=50, seed=seed + 2)
        if sc_nes2 < best_score:
            best_seq = a_nes2.copy()
            best_score = sc_nes2
            print(f"  NEW BEST: {best_score:.10f} [nest2_p5] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: L-BFGS ultra-fine ===
    print(f"\n[P4] L-BFGS ultra-fine (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 250, t_start + 1200)
    if p4_deadline > time.monotonic() + 50:
        a_lb2 = _run_lbfgs_repeated(best_seq, p4_deadline, min_remaining=40.0)
        sc_lb2 = _score(a_lb2)
        if sc_lb2 < best_score:
            best_seq = a_lb2.copy()
            best_score = sc_lb2
            print(f"  NEW BEST: {best_score:.10f} [lbfgs2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [n_best]
    for n_try in [12288, 16384, 20480, 24576, 32768]:
        if n_try != n_best and n_try not in n_candidates:
            n_candidates.append(n_try)
    # Sort by proximity to current best n
    n_candidates.sort(key=lambda x: abs(x - n_best))

    for n_try in n_candidates[1:3]:  # Try up to 2 other n values
        if time.monotonic() > deadline - 150:
            break
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")

        # Quick L-BFGS on new n
        if time.monotonic() < deadline - 120:
            a_opt = _run_lbfgs_repeated(a_init, deadline - 100, min_remaining=40.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] "
                      f"t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Shrinkage ===
    print(f"\n[P6] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        a_shr, sc_shr = _shrinkage_and_reoptimize(best_seq, deadline - 60,
                                                   min_remaining=30.0)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} [shrinkage] t={time.monotonic()-t_start:.1f}s")
            # Re-optimize after shrinkage
            if time.monotonic() < deadline - 40:
                a_shr2 = _run_lbfgs_repeated(best_seq, deadline, min_remaining=20.0)
                sc_shr2 = _score(a_shr2)
                if sc_shr2 < best_score:
                    best_seq = a_shr2.copy()
                    best_score = sc_shr2
                    print(f"  NEW BEST: {best_score:.10f} [post_shrink] "
                          f"t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final Nesterov + L-BFGS polish ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 20:
        a_fin, sc_fin = _nesterov_power(best_seq, deadline - 10,
                                        min_remaining=10.0, power=3,
                                        K_init=3, K_max=30, seed=seed + 100)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_nest] t={time.monotonic()-t_start:.1f}s")

    # Tiny final L-BFGS
    remaining = deadline - 8 - time.monotonic()
    if remaining > 15:
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        c_sorted = np.sort(c_best)[::-1]
        gap = mb - c_sorted[min(20, len(c_best) - 1)]
        beta_fin = max(gap * 0.005, mb * 0.000005)
        prob = _SmoothProb(beta_fin, n)
        max_iter = min(3000, max(50, int(remaining / 0.03)))
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxcor': 50})
            a_fin2 = _softmax(result.x)
            sc_fin2 = _score(a_fin2)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] "
                      f"t={time.monotonic()-t_start:.1f}s")
        except Exception:
            pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        # Verify with evaluate_sequence
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"FINAL: score={sc_final:.10f}, eval={ev_score:.10f}, "
                  f"n={len(final)}, time={time.monotonic()-t_start:.1f}s")
            if np.isfinite(ev_score) and ev_score < sc_final:
                sc_final = ev_score
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
