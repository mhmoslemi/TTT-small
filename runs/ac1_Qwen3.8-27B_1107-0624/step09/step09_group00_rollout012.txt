import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="uniform", rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian_wide":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos2":

    elif kind == "raised_cos3":

        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) ** 3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        a = rng.random(n)

    elif kind == "bimodal":

        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _active_set_subgradient(a, c, tol_frac):
    """Compute subgradient of max(c) weighted by inverse gap."""
    n = len(a)
    c_max = float(c.max())
    tol = max(tol_frac * c_max, 1e-15)
    active = c >= (c_max - tol)
    n_active = int(active.sum())
    if n_active == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        n_active = 1

    eps = 1e-12 * c_max
    w = np.zeros_like(c)
    w[active] = 1.0 / (c_max - c[active] + eps)
    w = w / w.sum()

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad = 2.0 * h
    grad = grad - np.mean(grad)
    return grad, active, c_max


def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adapt_beta(a, min_beta_frac=0.000005):
    """Adaptively compute beta based on gap between top peaks."""
    c = _full_convolve(a, a)
    c_sorted = np.sort(c)[::-1]
    c_max = c_sorted[0]
    if len(c_sorted) < 50:
        return c_max * 0.01
    gap = c_max - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < c_max * 0.0005:
        gap = c_max * 0.0005
    return max(c_max * min_beta_frac, gap * 0.5)


def _lbfgs_smoothed(a_init, deadline, min_remaining=20.0):
    """L-BFGS with adaptive beta annealing for smooth max approximation."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = _adapt_beta(current_a)
    beta_frac = 1.0

    while time.monotonic() + min_remaining < deadline:
        beta = beta * 0.55
        if beta < _adapt_beta(current_a) * 0.000005:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break
        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(8000, max(50, int(remaining / (est_time * 20))))

        try:
            result = minimize(
                lambda zz: _lse_objective(zz, beta, n),
                z,
                jac=lambda zz: _lse_gradient(zz, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-18,
                         'maxcor': 80, 'maxls': 40}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                beta = _adapt_beta(current_a)
        except Exception:
            pass

        beta_frac *= 0.5
        if beta_frac < 0.00001:
            break

    return current_a


def _nesterov_polish(a_init, deadline, tol_start=0.005, tol_end=0.00001,

    a = _normalize(a_init.copy())

    v = np.zeros(n)

    no_improve = 0

        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        grad, active, _ = _active_set_subgradient(a, c, tol_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        v = momentum * v - lr * grad
        alpha = 1.0
        improved = False
        for _ in range(40):

                break

        if not improved:
            v = np.zeros(n)
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_end, tol_frac * 0.6)
                lr *= 0.7
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(tol_end, tol_frac * 0.3)
                lr = lr_init * 0.4
                no_improve = 0
        else:
            if tol_frac > tol_end:
                tol_frac = max(tol_end, tol_frac * 0.97)
            lr = min(lr_init, lr * 1.01)

    return best_a, _score(best_a)


def _mirror_descent(a_init, deadline, min_remaining=15.0):
    """Multiplicative weights (mirror descent) on the active-set subgradient."""

        scale = np.clip(scale, -50, 50)

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

        print(f"  small_n={small_n}: {sc_small:.6f} -> {sc_small_opt:.6f}, upsampled: {sc_up:.10f}")
        if sc_up < best_score:
            best_seq = a_up.copy()
            best_score = sc_up
            print(f"  NEW BEST: {best_score:.10f} [multi_res]")

    # === Phase 1: L-BFGS with adaptive beta (400s) ===
    print(f"\n[P1] L-BFGS adaptive (t={time.monotonic()-t_start_ref:.1f}s)")
    p1_dead = min(deadline - 700, time.monotonic() + 400)
    a1 = _lbfgs_smoothed(best_seq, p1_dead, min_remaining=50)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs] t={time.monotonic()-t_start_ref:.1f}s")
    else:
        print(f"  No improvement: {sc1:.10f} vs {best_score:.10f}")

    # === Phase 2: Nesterov polish (200s) ===
    print(f"\n[P2] Nesterov polish (t={time.monotonic()-t_start_ref:.1f}s)")
    p2_dead = min(deadline - 500, time.monotonic() + 200)
    a2, sc2 = _nesterov_polish(best_seq, p2_dead, tol_start=0.004, tol_end=0.00002, lr_init=0.02, momentum=0.85)
    if sc2 < best_score:
        best_seq = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [nesterov] t={time.monotonic()-t_start_ref:.1f}s")
    else:
        print(f"  No improvement: {sc2:.10f} vs {best_score:.10f}")

    # === Phase 3: Mirror descent (100s) ===
    print(f"\n[P3] Mirror descent (t={time.monotonic()-t_start_ref:.1f}s)")
    p3_dead = min(deadline - 400, time.monotonic() + 100)
    a3, sc3 = _mirror_descent(best_seq, p3_dead, min_remaining=20)
    if sc3 < best_score:
        best_seq = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} [mirror] t={time.monotonic()-t_start_ref:.1f}s")
    else:
        print(f"  No improvement: {sc3:.10f} vs {best_score:.10f}")

    # === Phase 4: Peak-targeted ILS (300s) ===
    print(f"\n[P4] Peak-targeted ILS (t={time.monotonic()-t_start_ref:.1f}s)")
    p4_dead = min(deadline - 200, time.monotonic() + 300)
    a4, sc4 = _peak_targeted_ils(best_seq, p4_dead, rng, min_remaining=50)
    if sc4 < best_score:
        best_seq = a4.copy()
        best_score = sc4
        print(f"  NEW BEST: {best_score:.10f} [ils] t={time.monotonic()-t_start_ref:.1f}s")
    else:
        print(f"  No improvement: {sc4:.10f} vs {best_score:.10f}")

    # === Phase 5: Second L-BFGS pass on improved solution (150s) ===
    print(f"\n[P5] L-BFGS pass 2 (t={time.monotonic()-t_start_ref:.1f}s)")
    p5_dead = min(deadline - 150, time.monotonic() + 150)
    a5 = _lbfgs_smoothed(best_seq, p5_dead, min_remaining=30)
    sc5 = _score(a5)
    if sc5 < best_score:
        best_seq = a5.copy()
        best_score = sc5
        print(f"  NEW BEST: {best_score:.10f} [lbfgs2] t={time.monotonic()-t_start_ref:.1f}s")

    # === Phase 6: Multi-n exploration (150s) ===
    print(f"\n[P6] Multi-n exploration (t={time.monotonic()-t_start_ref:.1f}s)")
    n_options = []
    for nn in [n_main + 4096, n_main - 4096, n_main // 2, n_main * 2, 32768, 49152]:
        if nn >= 512 and nn != n_main and nn not in n_options:
            n_options.append(nn)

    for n_try in n_options[:4]:
        if time.monotonic() > deadline - 80:
            break
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        a_opt = _quick_opt(a_init, min(deadline - 60, time.monotonic() + 30), min_rem=5)
        sc_opt = _score(a_opt)
        if sc_opt < best_score - 1e-13:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start_ref:.1f}s")
        else:
            print(f"  n={n_try}: {sc_opt:.10f} (no improvement)")

    # === Phase 7: Another ILS round with different seed (100s) ===
    print(f"\n[P7] ILS round 2 (t={time.monotonic()-t_start_ref:.1f}s)")
    if time.monotonic() < deadline - 60:
        rng2 = np.random.default_rng(int(seed) + 7777)
        p7_dead = min(deadline - 40, time.monotonic() + 100)
        a7, sc7 = _peak_targeted_ils(best_seq, p7_dead, rng2, min_remaining=20)
        if sc7 < best_score:
            best_seq = a7.copy()
            best_score = sc7
            print(f"  NEW BEST: {best_score:.10f} [ils2] t={time.monotonic()-t_start_ref:.1f}s")

    # === Phase 8: Ultra-fine direct subgradient polish (remaining time) ===
    print(f"\n[P8] Ultra-fine polish (t={time.monotonic()-t_start_ref:.1f}s)")
    if time.monotonic() < deadline - 5:
        a8, sc8 = _direct_subgradient_polish(best_seq, deadline - 3)
        if sc8 < best_score:
            best_seq = a8.copy()
            best_score = sc8
            print(f"  NEW BEST: {best_score:.10f} [direct_polish] t={time.monotonic()-t_start_ref:.1f}s")

        # Final L-BFGS with tiny beta
        remaining = deadline - 3 - time.monotonic()
        if remaining > 8:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_tiny = mb * 0.000003
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            max_iter = min(3000, max(30, int(remaining / 0.05)))
            try:
                result = minimize(
                    lambda zz: _lse_objective(zz, beta_tiny, n),
                    z,
                    jac=lambda zz: _lse_gradient(zz, beta_tiny, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-22, 'gtol': 1e-20,
                             'maxcor': 60, 'maxls': 30}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                if sc_fin < best_score:
                    best_seq = a_fin.copy()
                    best_score = sc_fin
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start_ref:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start_ref:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
