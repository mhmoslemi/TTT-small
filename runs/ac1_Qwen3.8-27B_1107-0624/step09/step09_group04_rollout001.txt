"""
Optimized minimizer for 2*n*max(conv(a,a))/sum(a)^2 using multiplicative weights
minimax iteration, log-sum-exp L-BFGS refinement, growing active-set equalization,
and multi-scale n-adaptation. Key innovation: exploits the Hankel structure of the
autocorrelation quadratic form for O(n log n) gradient computation, and uses a
geometric gamma annealing schedule for the multiplicative weights update to
progressively focus on the worst peaks.
"""
import numpy as np
import time
from scipy.optimize import minimize

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    """Euclidean projection onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0, beta_scale=1.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003) * beta_scale

    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
        0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005,
        0.00002, 0.00001, 0.000005, 0.000002, 0.000001
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        if beta < base_beta * 0.01:
            est_time_per_iter = 0.003 * max(1.0, n / 4096.0)
            max_iter = min(8000, max(100, int(remaining / (est_time_per_iter * 20))))
        else:
            est_time_per_iter = 0.002 * max(1.0, n / 4096.0)
            max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 15))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _multiplicative_weights(a_init, deadline, min_time_remaining=120.0):
    """Minimax multiplicative weights iteration for the peak autocorrelation problem.
    Exploits the Hankel structure: H(p)[i,j] = p[i+j], so gradient = 2*conv(p, a_rev)."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    c = _full_convolve(a, a)
    c = np.maximum(c, 1e-30)

    # Geometric gamma annealing: start broad, narrow to worst peaks
    n_iters = 0
    gamma = 2.0
    alpha = 0.05
    no_improve = 0
    patience = 40

    while time.monotonic() + min_time_remaining < deadline:
        n_iters += 1

        # Update p: concentrate weight on worst peaks
        p = c ** gamma
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.ones_like(c) / len(c)
        else:
            p = p / p_sum

        # Compute gradient: grad_j = 2 * sum_k p[k] * a[k-j] (valid range)
        # This equals 2 * (p * a_reversed)[j + n - 1] for j in [0, n-1]
        a_rev = a[::-1]
        # Pad p to get the right convolution alignment
        wc = _full_convolve(p, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h

        # Make sum-preserving (project gradient to tangent space of simplex)
        grad = grad - np.mean(grad)

        # Backtracking line search
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-20:
            break

        step = alpha / (gnorm + 1e-15)
        improved = False
        for _ in range(40):
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                step *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < best_sc - 1e-16:
                a = a_try
                best_a = a.copy()
                best_sc = sc_try
                c = c_try
                improved = True
                break
            step *= 0.5

        if not improved:
            no_improve += 1
            alpha *= 0.9
            gamma *= 1.08
            if no_improve >= patience:
                # Perturb to escape
                rng = np.random.default_rng(n_iters)
                a = a + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                c = _full_convolve(a, a)
                c = np.maximum(c, 1e-30)
                no_improve = 0
                alpha = 0.05
        else:
            alpha = min(0.1, alpha * 1.005)
            if no_improve > 0:
                no_improve -= 1

        # Increase gamma (focus on worst peaks)
        gamma = min(gamma * 1.003, 200.0)

    return best_a, best_sc, n_iters


def _growing_active_set(a_init, deadline, min_time_remaining=60.0):
    """Iterative equalization with growing active set and Nesterov acceleration."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    c = _full_convolve(a, a)
    c_max = float(c.max())

    K = 5
    K_max = min(100, n)
    lr = 0.03
    no_improve = 0
    momentum = 0.9
    v = np.zeros(n)

    while time.monotonic() + min_time_remaining < deadline:
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            if K < K_max:
                K = min(K_max, K + 1)

        # Active set: top K peaks
        top_k_idx = np.argsort(c)[::-1][:K]
        top_k_vals = c[top_k_idx]

        # Also include peaks within 0.1% of max (broad active set)
        threshold = c_max * 0.9995
        broad_idx = np.where(c >= threshold)[0]
        if len(broad_idx) < K:
            # Add more from top
            additional = np.argsort(c)[::-1][K:min(K * 2, len(c))]
            active = np.unique(np.concatenate([broad_idx, additional[:K - len(broad_idx)]]))
        else:
            active = broad_idx[:min(K_max, len(broad_idx))]

        Ka = len(active)
        if Ka == 0:
            active = np.array([np.argmax(c)])
            Ka = 1

        # Weighted gradient: equal weight on active peaks
        w_S = np.zeros(2 * n - 1)
        w_S[active] = 2.0 / Ka
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)[n - 1:2 * n - 1]
        grad = h

        # Nesterov: use momentum
        grad = grad - np.mean(grad)
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            v *= 0.5
            a_new = a + v
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
        a_new = a_new / s_new

        c_new = _full_convolve(a_new, a_new)
        c_max_new = float(c_new.max())
        sc_new = 2.0 * n * c_max_new

        if sc_new < current_sc - 1e-15:
            a = a_new
            c = c_new
            c_max = c_max_new
            momentum = min(0.95, momentum * 1.002)
            lr = min(0.05, lr * 1.003)
        else:
            no_improve += 1
            v *= 0.5
            lr *= 0.92
            if no_improve % 30 == 0:
                K = max(3, K // 2)
                momentum = 0.8
            if no_improve >= 300:
                a = best_a.copy()
                c = _full_convolve(a, a)
                c_max = float(c.max())
                lr = 0.03
                K = 5
                momentum = 0.9
                no_improve = 0
                v = np.zeros(n)

    return best_a, best_sc


def _subgradient_polish(a_init, deadline, min_time_remaining=20.0):
    """Final subgradient polish with Polyak step size."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    # Estimate lower bound for Polyak step
    lb = 0.5  # conservative lower bound on optimal value
    max_iters = 3000

    for it in range(max_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()

        # Polyak step size
        gap = current_sc - 2.0 * n * lb  # use 2n*lb as proxy for lower bound
        if gap < 1e-10:
            gap = 1e-10

        # Active set: peaks within 0.001% of max
        threshold = c_max * (1 - 0.00001)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        Ka = len(active)
        w_S = np.zeros(2 * n - 1)
        w_S[active] = 2.0 / Ka
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)[n - 1:2 * n - 1]
        grad = h
        grad = grad - np.mean(grad)
        gnorm_sq = float(np.dot(grad, grad))
        if gnorm_sq < 1e-30:
            break

        eta = gap / (gnorm_sq + 1e-30)
        eta = min(eta, 1.0 / (np.max(np.abs(grad)) + 1e-15))

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _asymmetric_perturb(a, rng, strength=0.03):
    n = len(a)
    t = np.arange(n) / n
    ramp = 0.3 * (t - 0.5)
    noise = 0.7 * rng.standard_normal(n)
    pert = ramp + noise
    pert = pert - pert.mean()
    z = np.log(np.maximum(a, 1e-30))
    z = z + strength * pert
    z = z - z.mean()
    return _softmax(z)

    """Multi-phase optimizer: MW minimax iteration, L-BFGS refinement,
    growing active-set with Nesterov, subgradient polish, multi-n search."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = np.ones(n_parent) / n_parent
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Multiplicative weights minimax ===
    print(f"\n[P1] Multiplicative weights minimax (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        mw_deadline = min(deadline - 600, time.monotonic() + 200)
        a_mw, sc_mw, n_iters = _multiplicative_weights(best_seq, mw_deadline, min_time_remaining=60.0)
        print(f"  MW: {n_iters} iters, score={sc_mw:.10f}")
        if sc_mw < best_score:
            best_seq = a_mw.copy()
            best_score = sc_mw
            print(f"  NEW BEST: {best_score:.10f} [MW]")

    # === Phase 2: L-BFGS with log-sum-exp ===
    print(f"\n[P2] L-BFGS log-sum-exp (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a_l1 = _run_lbfgs(best_seq, deadline, min_time_remaining=400.0, beta_scale=1.0)
        sc_l1 = _score(a_l1)
        print(f"  L-BFGS 1: score={sc_l1:.10f}")
        if sc_l1 < best_score:
            best_seq = a_l1.copy()
            best_score = sc_l1
            print(f"  NEW BEST: {best_score:.10f} [LBFGS1]")

    # === Phase 3: Growing active set with Nesterov ===
    print(f"\n[P3] Growing active set + Nesterov (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        as_deadline = min(deadline - 300, time.monotonic() + 200)
        a_as, sc_as = _growing_active_set(best_seq, as_deadline, min_time_remaining=60.0)
        print(f"  ActiveSet: score={sc_as:.10f}")
        if sc_as < best_score:
            best_seq = a_as.copy()
            best_score = sc_as
            print(f"  NEW BEST: {best_score:.10f} [ActiveSet]")

    # === Phase 4: Asymmetric perturbation + re-optimize ===
    print(f"\n[P4] Asymmetric exploration (t={time.monotonic()-t_start:.1f}s)")
    for trial in range(3):
        if time.monotonic() > deadline - 250:
            break
        strength = 0.02 * (trial + 1)
        a_asym = _asymmetric_perturb(best_seq, rng, strength=strength)
        a_opt = _run_lbfgs(a_asym, deadline, min_time_remaining=250.0, beta_scale=0.5)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [asym trial {trial}]")
        else:
            print(f"  Asym trial {trial}: {sc_opt:.10f} (no improvement)")

    # === Phase 5: Fine L-BFGS with small beta ===
    print(f"\n[P5] Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        a_fine = _run_lbfgs(best_seq, deadline, min_time_remaining=150.0, beta_scale=0.3)
        sc_fine = _score(a_fine)
        print(f"  Fine L-BFGS: score={sc_fine:.10f}")
        if sc_fine < best_score:
            best_seq = a_fine.copy()
            best_score = sc_fine
            print(f"  NEW BEST: {best_score:.10f} [fine_lbfgs]")

    # === Phase 6: Second active set round ===
    print(f"\n[P6] Active set round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        as2_deadline = min(deadline - 100, time.monotonic() + 80)
        a_as2, sc_as2 = _growing_active_set(best_seq, as2_deadline, min_time_remaining=50.0)
        if sc_as2 < best_score:
            best_seq = a_as2.copy()
            best_score = sc_as2
            print(f"  NEW BEST: {best_score:.10f} [active2]")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_cur = len(best_seq)
    n_candidates = sorted(set([n_cur, 32768, 24576, 16384, 12288, 8192]), reverse=True)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try == n_cur:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        # Quick MW refinement
        mw_d = min(deadline - 60, time.monotonic() + 40)
        a_mw2, sc_mw2, _ = _multiplicative_weights(a_init, mw_d, min_time_remaining=30.0)
        # Quick L-BFGS
        a_opt = _run_lbfgs(a_mw2, deadline, min_time_remaining=60.0, beta_scale=0.5)
        sc_opt = _score(a_opt)
        print(f"  n={n_try}: init={sc_init:.10f}, optimized={sc_opt:.10f}")
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [multi_n={n_try}]")

    # === Phase 8: Subgradient polish ===
    print(f"\n[P8] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 20:
        sg_deadline = min(deadline - 15, time.monotonic() + 60)
        a_sg, sc_sg = _subgradient_polish(best_seq, sg_deadline, min_time_remaining=15.0)
        if sc_sg < best_score:
            best_seq = a_sg.copy()
            best_score = sc_sg
            print(f"  NEW BEST: {best_score:.10f} [subgrad]")

    # === Phase 9: Final ultra-fine L-BFGS ===
    print(f"\n[P9] Final ultra-fine (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 5 - time.monotonic()
    if remaining > 10:
        a_final = _run_lbfgs(best_seq, deadline, min_time_remaining=5.0, beta_scale=0.1)
        sc_final = _score(a_final)
        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final
            print(f"  NEW BEST: {best_score:.10f} [ultra_final]")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
