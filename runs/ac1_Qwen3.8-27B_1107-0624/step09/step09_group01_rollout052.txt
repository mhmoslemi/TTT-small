        g = 2.0 * a * (h - W_val)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        Hg = _cg_solve_hankel(W, n, g, max_iter=200)
        H1 = _cg_solve_hankel(W, n, np.ones(n), max_iter=200)
        H1_denom = float(np.ones(n) @ H1)
        if abs(H1_denom) < 1e-25 or not np.isfinite(H1_denom):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / H1_denom
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        alpha = 1.0
        step_done = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1.0 - 0.005 * alpha) - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not step_done:
            for trial in [0.1, 0.05, 0.02, 0.01, 0.005, 0.001]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 10:
            break

    return best_a, best_sc


def _smooth_phase(a_init, deadline, beta_init=None, min_beta_frac=1e-8):
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c_cur = _full_convolve(best_a, best_a)
    c_max = float(c_cur.max())
    if beta_init is None:
        beta = c_max * 0.005
    else:
        beta = beta_init

    z = np.log(np.maximum(best_a, 1e-30))
    z = z - z.mean()
    no_improve = 0

    while time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        scale = max(1.0, n / 8192.0)
        max_iter = max(20, min(5000, int((remaining - 20) / (0.006 * scale * 20))))

        def obj_wrapper(zz):
            return _smooth_obj_val(zz, beta, n)

        def grad_wrapper(zz):
            return _smooth_grad_val(zz, beta, n)

        try:
            result = minimize(obj_wrapper, z, jac=grad_wrapper, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-22, 'gtol': 1e-18,
                                       'maxcor': 100, 'maxls': 500, 'factr': 1e12})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-16:
                best_sc = sc_new
                best_a = a_new.copy()
                z = result.x.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 3:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z + 0.005 * np.random.randn(n)
                    z = z - z.mean()
                    no_improve = 0
        except Exception:
            no_improve += 1

        c_cur = _full_convolve(best_a, best_a)
        c_max = float(c_cur.max())
        c_sorted = np.sort(c_cur)[::-1]
        gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]

        if gap < 1e-10 * c_max:
            beta *= 0.15
        elif gap < 1e-7 * c_max:
            beta *= 0.3
        elif gap < 1e-5 * c_max:
            beta *= 0.5
        elif gap < 1e-3 * c_max:
            beta *= 0.7
        else:
            beta *= 0.85

        min_beta = c_max * min_beta_frac
        if beta < min_beta:
            beta = min_beta

    return best_a, best_sc


def _coord_descent_fast(a_init, deadline, K=50, max_passes=30):
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sum_a2 = float(np.dot(a, a))

        if K < 2 * n - 1:
            active = np.argpartition(c, -K)[-K:]
        else:
            active = np.arange(2 * n - 1)
        active = np.sort(active)

        improved = False
        for j in range(0, n, 1):
            if time.monotonic() > deadline:
                break
            aj = a[j]
            C0 = sum_a2 - aj * aj
            max_at_cur = C0 + aj * aj

            for k in active:
                k = int(k)
                if k == 0:
                    continue
                lk = 0.0
                jp = j + k
                jm = j - k
                if jp < n:
                    lk += a[jp]
                if jm >= 0:
                    lk += a[jm]
                ck = c[k] - aj * lk
                val = ck + lk * aj
                if val > max_at_cur:
                    max_at_cur = val

            best_aj = aj
            best_obj = max_at_cur

            # Try zero
            obj0 = C0
            for k in active:
                k = int(k)
                if k == 0:
                    continue
                lk = 0.0
                jp = j + k
                jm = j - k
                if jp < n:
                    lk += a[jp]
                if jm >= 0:
                    lk += a[jm]
                ck = c[k] - aj * lk
                val = ck
                if val > obj0:
                    obj0 = val
            if obj0 < best_obj:
                best_obj = obj0
                best_aj = 0.0

            # Try intersections with top few active lags
            for k in active[:min(10, len(active))]:
                k = int(k)
                if k == 0:
                    continue
                lk = 0.0
                jp = j + k
                jm = j - k
                if jp < n:
                    lk += a[jp]
                if jm >= 0:
                    lk += a[jm]
                ck = c[k] - aj * lk
                disc = lk * lk - 4.0 * (C0 - ck)
                if disc < 0:
                    continue
                sd = np.sqrt(disc)
                for cand in [(lk + sd) * 0.5, (lk - sd) * 0.5]:
                    if cand < 0 or cand > 10.0:
                        continue
                    obj_c = C0 + cand * cand
                    for k2 in active[:min(10, len(active))]:
                        k2 = int(k2)
                        if k2 == 0:
                            continue
                        lk2 = 0.0
                        jp2 = j + k2
                        jm2 = j - k2
                        if jp2 < n:
                            lk2 += a[jp2]
                        if jm2 >= 0:
                            lk2 += a[jm2]
                        ck2 = c[k2] - aj * lk2
                        val = ck2 + lk2 * cand
                        if val > obj_c:
                            obj_c = val
                    if obj_c < best_obj - 1e-18:
                        best_obj = obj_c
                        best_aj = cand

            if best_aj != aj:
                a[j] = best_aj
                improved = True

        s2 = a.sum()
        if s2 < 1e-15:
            break
        a = a / s2
        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
        if not improved:
            break

    return best_a, best_sc


def _resample_to(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="tapered"):
    if kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.01
    elif kind == "chirp":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * 3 * t * (t - 0.5))
        a = np.maximum(a, 1e-10)
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.7) ** 0.8
        a = np.maximum(a, 1e-10)
    else:
        a = np.ones(n)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] imp={prev-best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                par_sc = _score(parent)
                print(f"Parent loaded: n={len(parent)}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        parent = _make_profile(16384, "tapered")
        print(f"No parent, using tapered n=16384")
        update_best(parent, _score(parent), "init")

    n0 = len(parent)
    print(f"\n{'='*60}")
    print(f"Starting optimization from n={n0}, score={best_score:.10f}")
    print(f"Deadline in {deadline - t_start:.0f}s")
    print(f"{'='*60}")

    # Phase 1: Smooth optimization (aggressive)
    print(f"\n[Phase 1] Smooth optimization (main) t={time.monotonic()-t_start:.1f}s")
    t_p1 = min(deadline - 600, t_start + 400)
    a1, sc1 = _smooth_phase(parent, t_p1, beta_init=None, min_beta_frac=1e-7)
    update_best(a1, sc1, "smooth1")

    # Phase 2: KKT-Newton refinement
    print(f"\n[Phase 2] KKT-Newton t={time.monotonic()-t_start:.1f}s")
    t_p2 = min(deadline - 450, t_start + 600)
    k_sch = [3, 5, 10, 20, 50, 100, 300, 1000, 2000, 5000]
    a2, sc2 = _kkt_newton(best_seq, t_p2, k_schedule=k_sch, max_rounds=500)
    update_best(a2, sc2, "kkt1")

    # Phase 3: Smooth again from KKT result
    print(f"\n[Phase 3] Smooth (refined) t={time.monotonic()-t_start:.1f}s")
    t_p3 = min(deadline - 300, t_start + 850)
    a3, sc3 = _smooth_phase(best_seq, t_p3, beta_init=None, min_beta_frac=1e-8)
    update_best(a3, sc3, "smooth2")

    # Phase 4: KKT-Newton again with different K schedule
    print(f"\n[Phase 4] KKT-Newton (alt) t={time.monotonic()-t_start:.1f}s")
    t_p4 = min(deadline - 200, t_start + 1050)
    k_sch2 = [1, 2, 3, 5, 10, 20, 50, 100, 500]
    a4, sc4 = _kkt_newton(best_seq, t_p4, k_schedule=k_sch2, max_rounds=400)
    update_best(a4, sc4, "kkt2")

    # Phase 5: Final smooth with very tight beta
    print(f"\n[Phase 5] Final smooth polish t={time.monotonic()-t_start:.1f}s")
    t_p5 = min(deadline - 100, t_start + 1300)
    a5, sc5 = _smooth_phase(best_seq, t_p5, beta_init=None, min_beta_frac=1e-9)
    update_best(a5, sc5, "smooth3")

    # Phase 6: Coordinate descent polish
    print(f"\n[Phase 6] Coord descent t={time.monotonic()-t_start:.1f}s")
    t_p6 = min(deadline - 50, t_start + 1500)
    a6, sc6 = _coord_descent_fast(best_seq, t_p6, K=30, max_passes=10)
    update_best(a6, sc6, "coord")

    # Phase 7: KKT with K=1 (single active constraint)
    print(f"\n[Phase 7] KKT K=1 t={time.monotonic()-t_start:.1f}s")
    t_p7 = min(deadline - 10, t_start + 1600)
    a7, sc7 = _kkt_newton(best_seq, t_p7, k_schedule=[1, 1, 2, 2, 3, 3, 5, 5, 10], max_rounds=200)
    update_best(a7, sc7, "kkt_K1")

    # Phase 8: Try slightly different n
    print(f"\n[Phase 8] Multi-n t={time.monotonic()-t_start:.1f}s")
    for n_try in [n0 // 2, n0 * 2]:
        if time.monotonic() > deadline - 30:
            break
        if n_try < 1024 or n_try > 65536:
            continue
        a_res = _resample_to(best_seq, n_try)
        sc_res = _score(a_res)
        if sc_res < best_score:
            update_best(a_res, sc_res, f"resample_n{n_try}")
        t_end = min(deadline - 15, time.monotonic() + 40)
        if time.monotonic() < t_end:
            a_opt, sc_opt = _smooth_phase(a_res, t_end, beta_init=None, min_beta_frac=1e-7)
            update_best(a_opt, sc_opt, f"opt_n{n_try}")

    # Phase 9: Restart from perturbation
    print(f"\n[Phase 9] Restarts t={time.monotonic()-t_start:.1f}s")
    for ri in range(3):
        if time.monotonic() > deadline - 15:
            break
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z + 0.01 * (ri + 1) * rng.standard_normal(n0)
        z = z - z.mean()
        a_pert = _softmax(z)
        t_end = min(deadline - 5, time.monotonic() + 15)
        a_r, sc_r = _smooth_phase(a_pert, t_end, beta_init=None, min_beta_frac=1e-7)
        update_best(a_r, sc_r, f"restart_{ri}")

    # Final
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
