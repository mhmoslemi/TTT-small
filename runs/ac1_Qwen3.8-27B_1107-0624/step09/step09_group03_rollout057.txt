This is a weighted sum of shifted versions of v, which can be computed in O(n * |A|) time, or faster using FFT if |A| is large.

```python
"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Active-set Newton with exact sparse Hessian (anti-diagonal structure)
2. Adaptive active set identification
3. KKT-based convergence detection
4. Multi-scale refinement
5. Careful non-negativity handling
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _hessian_matvec(v, active_lags, weights, n):
    """
    Compute H @ v where H = sum_k weights[k] * H_k
    H_k[i,j] = 2 if i+j=k (and both in [0,n-1]), else 0.
    
    (H*v)[i] = 2 * sum_k weights[k] * v[k-i] (where 0 <= k-i < n)
    """
    result = np.zeros(n)
    for k, w in zip(active_lags, weights):
        if w < 1e-15:
            continue
        # v[k-i] is valid when 0 <= k-i < n, i.e., max(0, k-n+1) <= i <= min(n-1, k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo > i_hi:
            continue
        # v[k-i] for i in [i_lo, i_hi] means v indices from k-i_hi to k-i_lo
        j_lo = k - i_hi
        j_hi = k - i_lo
        result[i_lo:i_hi+1] += w * v[j_lo:j_hi+1]
    return 2.0 * result


def _gradient_active(a, active_lags, weights, n):
    """
    Compute g = sum_k weights[k] * grad(c[k])
    grad(c[k])[i] = 2*a[k-i] (where 0 <= k-i < n)
    """
    result = np.zeros(n)
    for k, w in zip(active_lags, weights):
        if w < 1e-15:
            continue
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo > i_hi:
            continue
        j_lo = k - i_hi
        j_hi = k - i_lo
        result[i_lo:i_hi+1] += w * a[j_lo:j_hi+1]
    return 2.0 * result


def _cg_solve(matvec_fn, rhs, n, max_iter=500, tol=1e-12):
    """Conjugate gradient solver."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Ap = matvec_fn(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x += alpha * p
        r -= alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _newton_step(a, c, n, top_k=10):
    """
    Compute one Newton step for the active-set minimax problem.
    Returns (dx, score_reduction) or (None, None) if no step found.
    """
    M = float(c.max())
    if M < 1e-30:
        return None, None
    
    # Identify active set: top lags
    sorted_idx = np.argsort(c)[::-1]
    # Include lags that are within 1% of max
    active_mask = c >= M * 0.995
    active_idx = np.where(active_mask)[0]
    
    # Limit to top_k
    if len(active_idx) > top_k:
        active_idx = sorted_idx[:top_k]
    elif len(active_idx) == 0:
        active_idx = sorted_idx[:3]
    
    active_lags = active_idx
    m = len(active_lags)
    weights = np.ones(m) / m
    
    # Compute gradient
    g = _gradient_active(a, active_lags, weights, n)
    
    # Project out the mean (to handle the sum constraint)
    g_mean = g.mean()
    g_t = g - g_mean
    
    # Solve H @ dx = -g_t (approximately, ignoring the sum constraint for now)
    # H is the Hessian of the smooth approximation
    # Add a small regularization for stability
    mu = 1e-8 / m
    
    def matvec(v):
        return _hessian_matvec(v, active_lags, weights, n) + mu * v
    
    dx = _cg_solve(matvec, -g_t, n, max_iter=300)
    
    # Ensure sum(dx) = 0 (projection onto the sum=1 hyperplane)
    dx = dx - dx.mean()
    
    # Check if the step direction is valid
    dx_norm = np.linalg.norm(dx)
    if dx_norm < 1e-15:
        return None, None
    
    # Line search
    best_t = 0
    best_improvement = 0
    c_max_cur = M
    
    t = 1.0
    for ls in range(50):
        a_try = a + t * dx
        if (a_try < -1e-15).any():
            t *= 0.5
            continue
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            t *= 0.5
            continue
        a_try = a_try / s_try
        c_try = _full_convolve(a_try, a_try)
        f_try = float(c_try.max())
        if f_try < c_max_cur - 1e-14 * t:
            if c_max_cur - f_try > best_improvement:
                best_improvement = c_max_cur - f_try
                best_t = t
                best_a_try = a_try.copy()
            break
        t *= 0.5
    
    if best_t == 0:
        return None, None
    
    return best_a_try, best_improvement

    return 2.0 * n * (c_max + beta * np.log(s))


def _lsq_smooth_grad(z, beta, n):
    """Gradient of the log-sum-exp smooth approximation."""
    a = _safe_softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    
    # Gradient of sum_k w[k]*c[k] wrt a, then chain rule through softmax
    # d/d a_j of sum_k w[k]*c[k] = 2 * sum_k w[k]*a[k-j] (valid indices)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    
    # Chain rule: d/dz = (1/n) * a * (h - sum(h*a))... 
    # Actually for softmax: d a_i / d z_j = a_i * (δ_ij - a_j)
    # So gradient wrt z = a * (grad_a - a * sum(a * grad_a))
    W = float(np.dot(w, c))  # weighted sum of c
    grad_a = 2.0 * h - 2.0 * W * a  # Hmm, this isn't quite right
    
    # Let me redo: f = sum_k w[k] * c[k]
    # df/da_j = 2 * sum_k w[k] * a[k-j] (valid)
    # = 2 * (w * a_flipped)[j]... 
    
    # Actually, let's use the formula:
    # df/da_j = 2 * sum_k w[k] * a[k-j] for 0 <= k-j < n
    # This is 2 * (correlation of w and a at various lags)
    
    # Using the h computed above: h[j] = sum_k w[k] * a[k-j] (valid)
    # So df/da = 2*h
    
    # Then df/dz_i = sum_j (df/da_j) * (da_j/dz_i)
    # da_j/dz_i = a_j * (δ_ji - a_i)
    # df/dz_i = sum_j 2*h_j * a_j * (δ_ji - a_i)
    # = 2*h_i * a_i - 2*a_i * sum_j h_j * a_j
    # = 2*a_i * (h_i - sum(h*a))
    
    ha = float(np.dot(h, a))
    grad_z = 2.0 * n * a * (h - ha)
    
    return grad_z


def _safe_softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lbfgs_refine(a_init, deadline, n_betas=15):
    """L-BFGS refinement with log-sum-exp continuation."""
    n = len(a_init)
    a = a_init.copy()
    sc = _score(a)
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[min(50, len(c_sorted)-1)] - c_sorted[0] if len(c_sorted) > 50 else c_sorted[-1] - c_sorted[0]
    base_beta = max(abs(gap) * 0.5, c_max * 0.01)
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    from scipy.optimize import minimize
    
    for i in range(n_betas):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        
        beta = base_beta * (0.3 ** i)
        est_time = 0.005 * max(1.0, n / 8192.0)
        max_iter = min(3000, max(20, int(remaining / (est_time * 5))))
        
        try:
            def obj_fn(z_val):
                return _lsq_smooth_obj(z_val, beta, n, 1.0)
            
            def grad_fn(z_val):
                return _lsq_smooth_grad(z_val, beta, n)
            
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxcor': 50}
            )
            a_new = _safe_softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < sc:
                a = a_new
                sc = sc_new
                z = np.log(np.maximum(a, 1e-30))
                z = z - z.mean()
            print(f"    L-BFGS beta={beta:.2e}: score={sc:.12f}")
        except Exception as ex:
            print(f"    L-BFGS beta={beta:.2e}: {ex}")
    
    return a, sc
