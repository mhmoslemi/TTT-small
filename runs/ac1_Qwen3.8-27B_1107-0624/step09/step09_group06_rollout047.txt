def solve_lp_direction(a, M, active_indices):
    n = len(a)
    m = len(active_indices)
    
    # Build constraint matrix A (m x n)
    # A[i, j] = a[active_indices[i] - j] if 0 <= active_indices[i] - j < n else 0
    A = np.zeros((m, n))
    for i, k in enumerate(active_indices):
        # conv(a,b)[k] = sum_j a[k-j] * b[j]
        # For j in [0, n-1], a[k-j] is valid if 0 <= k-j < n
        # i.e., k - (n-1) <= j <= k
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            A[i, j_lo:j_hi+1] = a[k - np.arange(j_lo, j_hi + 1)]
    
    # LP: max 1^T b subject to A b <= M, b >= 0
    c = -np.ones(n)  # minimize -1^T b
    A_ub = A
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x
