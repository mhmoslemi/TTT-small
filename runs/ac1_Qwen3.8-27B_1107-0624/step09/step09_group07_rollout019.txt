def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _spectral_factorize(c, n):
    """Given a target convolution c (length 2n-1), recover a (length n) via spectral factorization."""
    N = _next_pow2(2 * len(c))
    C = np.fft.rfft(c, N)
    # Square root in frequency domain (principal branch)
    X = np.sqrt(C + 0j)
    a_full = np.fft.irfft(X, N)
    a = a_full[:n]
    return a

    W = float(np.dot(a, ga))

    return obj, grad_z


class _TopkSmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):

    def grad(self, z):

def _sum_topk_obj_grad(z, K, n):

        top_idx = np.arange(len(c))

    W = float(np.dot(a, ga))
