def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _hankel_mv(v, W, n):
    """Compute H_W @ v where (H_W)_{i,j} = W[i+j]."""
    v_rev = v[::-1]
    out_len = len(W) + len(v_rev) - 1
    N = _next_pow2(out_len)
    F_W = np.fft.rfft(W, N)
    F_v = np.fft.rfft(v_rev, N)
    conv = np.fft.irfft(F_W * F_v, N)[:out_len]
    return conv[n-1:2*n-1]

def _cg_solve(W, n, rhs, max_iter=200, tol=1e-14):
    """Solve H_W x = rhs using CG with FFT-based matvec."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs_old = float(r @ r)
    rs0 = rs_old
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_mv(p, W, n)
        pHp = float(p @ Hp)
        if pHp <= 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs_old / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new) or rs_new > rs_old * 1.1:
            x = x - alpha * p
            r = r + alpha * Hp
            break
        if rs_new < tol * rs0:
            break
        p = (rs_new / rs_old) * p + r
        rs_old = rs_new
    return x

def _compute_subgrad(a, W, n):
    """Compute 2 * H_W @ a (a subgradient of max_k a^T H_k a)."""
    return 2.0 * _hankel_mv(a, W, n)

def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _hankel_mv(v, W, n):
    """Compute H_W @ v where (H_W)_{i,j} = W[i+j] for i,j in [0,n)."""
    v_rev = v[::-1]
    out_len = len(W) + len(v_rev) - 1
    N = _next_pow2(out_len)
    F_W = np.fft.rfft(W, N)
    F_v = np.fft.rfft(v_rev, N)
    conv = np.fft.irfft(F_W * F_v, N)[:out_len]
    return conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=200, tol=1e-13):
    """Solve H_W x = rhs using CG with FFT-based matvec."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs_old = float(r @ r)
    rs0 = rs_old
    if rs0 < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = _hankel_mv(p, W, n)
        pHp = float(p @ Hp)
        if pHp <= 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs_old / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new) or rs_new > rs_old * 1.05:
            x = x - alpha * p
            break
        if rs_new < tol * rs0:
            break
        p = (rs_new / rs_old) * p + r
        rs_old = rs_new
    return x


def _select_active(c, K_min=1, strategy="topk"):
    """Select active lags from autocorrelation."""
    c_max = float(c.max())
    n_lags = len(c)
    if strategy == "topk":
        K = K_min
        return np.argpartition(c, -K)[-K:], K
    elif strategy == "adaptive":
        for frac in [0.0001, 0.0005, 0.001, 0.005, 0.01, 0.05]:
            tol = frac * c_max
            active = np.where(c >= c_max - tol)[0]
            if len(active) >= max(K_min, 5):
                return active, len(active)
        idx = np.argmax(c)
        return np.array([idx]), 1
    else:
        idx = np.argmax(c)
        return np.array([idx]), 1


def _kkt_newton(a_init, deadline, k_schedule=None, max_rounds=500):
    """KKT-based Newton with adaptive active set and line search."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    if k_schedule is None:
        k_schedule = [2000, 1000, 500, 200, 100, 50, 20, 10, 5, 3, 2, 1]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        K = min(k_schedule[rnd % len(k_schedule)], 2 * n - 1)
        K = max(K, 1)
        if K == 1:
            idx = np.array([int(np.argmax(c))])
        else:
            idx = np.argpartition(c, -K)[-K:]
        W = np.zeros(2 * n - 1)
        W[idx] = 1.0 / K

        g = 2.0 * _hankel_mv(a, W, n)
        g_centered = g - np.mean(g)
        gn = float(np.linalg.norm(g_centered))
        if gn < 1e-13 * n:
            break

        Hg = _cg_solve(W, n, g_centered, max_iter=150)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=150)
        H1_denom = float(np.ones(n) @ H1)

        if abs(H1_denom) < 1e-25 or not np.isfinite(H1_denom):
            dx = -g_centered / max(gn, 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / H1_denom
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_centered / max(gn, 1e-30)

        alpha = 1.0
        step_done = False
        for _ in range(50):
            a_try = np.maximum(a + alpha * dx, 0.0)
            s_try = float(a_try.sum())
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break

        if not step_done:
            for trial in [0.05, 0.02, 0.01, 0.005, 0.001, 0.0005, 0.0001]:
                a_try = np.maximum(a - trial * g_centered / max(gn, 1e-30), 0.0)
                s_try = float(a_try.sum())
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 12:
            break

    return best_a, best_sc


def _fw_away(a_init, deadline, max_iters=20000):
    """Away-step Frank-Wolfe for simplex-constrained min max_k a^T H_k a."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    gamma0 = 1.0
    gamma = 1.0

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol_fracs = [0.0002, 0.001, 0.005, 0.01, 0.05]
        active = None
        for tf in tol_fracs:
            active = c >= (c_max - tf * c_max)
            if active.sum() >= 3:
                break
        if active is None or active.sum() == 0:
            active = np.zeros(len(c), dtype=bool)
            active[int(np.argmax(c))] = True

        W = np.zeros(len(c))
        W[active] = 1.0 / active.sum()

        g = 2.0 * _hankel_mv(a, W, n)

        i_min = int(np.argmin(g))

        away_scores = a * g
        i_away = int(np.argmax(away_scores))
        use_away = a[i_away] > 1e-8

        t = 2.0 / (gamma0 + gamma + 2.0)

        e_min = np.zeros(n)
        e_min[i_min] = 1.0

        if use_away:
            s_away = min(t * 0.5, a[i_away])
            e_away = np.zeros(n)
            e_away[i_away] = 1.0
            a_new = (1.0 - t - s_away) * a + t * e_min + s_away * e_away
        else:
            a_new = (1.0 - t) * a + t * e_min

        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        gamma += 1.0

        if i % 200 == 199:
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    return best_a, best_sc


def _spectral_polish(a_init, deadline, max_iters=500):
    """Spectral iteration: flatten autocorrelation via frequency-domain manipulation."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    N = _next_pow2(2 * n - 1)

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        P = np.real(P)

        c = _full_convolve(a, a)
        c_max = float(c.max())

        target = np.ones(2 * n - 1) * (float(np.sum(c)) / (2 * n - 1))
        target[int(np.argmax(c))] = c_max * 0.5

        blended = 0.3 * c + 0.7 * target
        blended = np.maximum(blended, 1e-15)

        T = np.fft.rfft(blended, N)
        T = np.real(T)
        T = np.maximum(T, 0.0)

        ratio = np.sqrt(T / np.maximum(P, 1e-30))
        A_new = A * ratio

        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new

        if i % 50 == 49:
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.6
    elif kind == "cos":
        a = 0.5 + 0.5 * np.cos(np.pi * t)
    elif kind == "bell":
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.01
    elif kind == "flat_taper":
        a = np.ones(n)
        edge = max(3, n // 50)
        taper = np.linspace(0, 1, edge)
        a[:edge] = taper ** 0.5
        a[-edge:] = taper ** 0.5[::-1]
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = float(a_new.sum())
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _subgrad_momentum(a_init, deadline, max_steps=5000):
    """Momentum subgradient in log-space with adaptive tolerance."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = 0.015
    momentum = 0.9
    v = np.zeros(n)
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        e = np.exp(z - np.max(z))
        a = e / e.sum()
        c = _full_convolve(a, a)
        c_max = float(c.max())

        for tf in [0.0002, 0.001, 0.003, 0.01]:
            active = c >= (c_max - tf * c_max)
            if active.sum() >= 5:
                break
        if active.sum() == 0:
            active = np.zeros(len(c), dtype=bool)
            active[int(np.argmax(c))] = True

        W = np.zeros(len(c))
        W[active] = 1.0 / active.sum()

        g = 2.0 * _hankel_mv(a, W, n)
        g_centered = g - np.mean(g)
        gn = float(np.linalg.norm(g_centered))
        if gn < 1e-14:
            break

        v = momentum * v - lr * g_centered / gn
        z = z + v
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 99:
            e2 = np.exp(z - np.max(z))
            a_chk = e2 / e2.sum()
            sc = _score(a_chk)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_chk.copy()
                patience = 0
                lr = min(lr * 1.03, 0.04)
            else:
                patience += 1
                lr *= 0.9
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)
                    patience = 0
                    lr = 0.01

    e_f = np.exp(z - np.max(z))
    a_f = e_f / e_f.sum()
    sc_f = _score(a_f)
    if sc_f < best_sc:
        best_a = a_f.copy()
        best_sc = sc_f
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _upd(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                par_sc = _score(parent)
                print(f"Parent: n={len(parent)}, score={par_sc:.10f}")
                _upd(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        parent = _make_profile(32768, "flat_taper")
        _upd(parent, _score(parent), "init")

    n0 = len(parent)

    # Phase 1: Broad KKT-Newton
    print(f"\n[Phase 1] KKT-Newton broad (t={time.monotonic()-t_start:.1f}s)")
    k_broad = [5000, 3000, 2000, 1000, 500, 200, 100, 50, 20, 10, 5, 3, 2, 1]
    t_p1 = min(deadline - 1300, t_start + 350)
    a1, sc1 = _kkt_newton(parent, t_p1, k_schedule=k_broad, max_rounds=300)
    _upd(a1, sc1, "kkt_broad")

    # Phase 2: Subgradient momentum
    print(f"\n[Phase 2] Subgrad momentum (t={time.monotonic()-t_start:.1f}s)")
    t_p2 = min(deadline - 1000, t_start + 700)
    if t_p2 > time.monotonic() + 10:
        a2, sc2 = _subgrad_momentum(best_seq, t_p2, max_steps=8000)
        _upd(a2, sc2, "subgrad")

    # Phase 3: Away-step FW
    print(f"\n[Phase 3] Away-FW (t={time.monotonic()-t_start:.1f}s)")
    t_p3 = min(deadline - 700, t_start + 1100)
    if t_p3 > time.monotonic() + 10:
        a3, sc3 = _fw_away(best_seq, t_p3, max_iters=30000)
        _upd(a3, sc3, "away_fw")

    # Phase 4: Fine KKT-Newton
    print(f"\n[Phase 4] KKT-Newton fine (t={time.monotonic()-t_start:.1f}s)")
    k_fine = [20, 10, 5, 3, 2, 1, 1, 1]
    t_p4 = min(deadline - 400, t_start + 1400)
    if t_p4 > time.monotonic() + 10:
        a4, sc4 = _kkt_newton(best_seq, t_p4, k_schedule=k_fine, max_rounds=300)
        _upd(a4, sc4, "kkt_fine")

    # Phase 5: Spectral polish
    print(f"\n[Phase 5] Spectral polish (t={time.monotonic()-t_start:.1f}s)")
    t_p5 = min(deadline - 200, t_start + 1550)
    if t_p5 > time.monotonic() + 10:
        a5, sc5 = _spectral_polish(best_seq, t_p5, max_iters=300)
        _upd(a5, sc5, "spectral")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([n0, n0 * 2, n0 // 2, 16384, 32768, 65536]))
    n_candidates = [x for x in n_candidates if 4096 <= x <= 131072]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 50:
            break
        if n_try == n0:
            continue
        a_res = _resample(best_seq, n_try)
        sc_res = _score(a_res)
        if sc_res < best_score:
            _upd(a_res, sc_res, f"resample_{n_try}")
        t_opt = min(deadline - 20, time.monotonic() + 25)
        if t_opt > time.monotonic() + 5:
            a_opt, sc_opt = _kkt_newton(a_res, t_opt,
                                         k_schedule=[50, 20, 10, 5, 3, 2, 1],
                                         max_rounds=80)
            _upd(a_opt, sc_opt, f"opt_{n_try}")

    # Phase 7: Final ultra-fine KKT
    print(f"\n[Phase 7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    k_ult = [2, 1, 1, 1, 1]
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _kkt_newton(best_seq, deadline - 2,
                                     k_schedule=k_ult, max_rounds=200)
        _upd(a_fin, sc_fin, "final_ult")

    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "flat_taper")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
