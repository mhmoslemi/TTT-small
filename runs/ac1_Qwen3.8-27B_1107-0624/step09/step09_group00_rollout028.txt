import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_unnormalized(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _softmax(z):

    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    if s < 1e-30:

    return a / s

    elif kind == "triangle":

def _active_set_gradient(a, c, n_peaks=5):
    """Compute weighted gradient using top n_peaks with inverse-gap weighting."""
    n = len(a)
    c_max = float(c.max())
    
    # Get top peaks with adaptive weighting
    c_sorted_idx = np.argsort(c)[::-1]
    n_use = min(n_peaks, len(c))
    peak_indices = c_sorted_idx[:n_use]
    peak_values = c[peak_indices]
    
    # Inverse-gap weighting
    eps = 1e-12 * c_max
    w = np.zeros_like(c)
    gaps = c_max - peak_values + eps
    weights = 1.0 / gaps
    weights = weights / weights.sum()
    for i, idx in enumerate(peak_indices):
        w[idx] = weights[i]
    
    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad = 2.0 * h
    
    # Project onto tangent space of sum(a)=1
    grad = grad - np.mean(grad)
    return grad


def _adaptive_polish(a_init, deadline, tol_start=0.01, tol_end=0.00001,
                     lr_init=0.04, momentum=0.85, max_peaks=8):
    """Active-set gradient with adaptive peak count and step size."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score_unnormalized(a)
    v = np.zeros(n)
    tol_frac = tol_start
    lr = lr_init
    no_improve = 0
    max_stall = 200
    n_peaks = max_peaks

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Adaptive number of peaks: use more peaks when they're close together
        c_sorted = np.sort(c)[::-1]
        n_valid = min(len(c_sorted), 50)
        gap_10 = c_sorted[0] - c_sorted[min(10, n_valid - 1)] if n_valid > 10 else c_max * 0.01
        gap_frac = gap_10 / c_max
        if gap_frac < 0.001:
            n_peaks = min(max_peaks, int(50 * (1 - gap_frac * 1000)))
            n_peaks = max(n_peaks, 3)
        elif gap_frac < 0.01:
            n_peaks = max(3, int(max_peaks * (1 - gap_frac * 50)))
        else:
            n_peaks = max(2, min(5, max_peaks))

        grad = _active_set_gradient(a, c, n_peaks)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Heavy ball momentum
        v = momentum * v - lr * grad
        alpha = 1.0
        improved = False
        for _ in range(40):
            a_try = _normalize(a + alpha * v)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            v = np.zeros(n)
            no_improve += 1
            if no_improve % 15 == 0:
                tol_frac = max(tol_end, tol_frac * 0.65)
                lr *= 0.7
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(tol_end, tol_frac * 0.3)
                lr = lr_init * 0.4
                no_improve = 0
        else:
            if tol_frac > tol_end:
                tol_frac = max(tol_end, tol_frac * 0.98)
            lr = min(lr_init * 1.5, lr * 1.01)

    return best_a, _score(best_a)


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def f(self, z):
        return _smoobj(z, self.beta, self.n)

    def g(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=20.0):
    """L-BFGS with adaptive beta based on peak entropy."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Compute initial beta from peak distribution
    c_sorted = np.sort(c_cur)[::-1]
    n_top = min(50, len(c_sorted))
    peak_region = c_sorted[:n_top]
    peak_entropy = -np.sum((peak_region / peak_region.sum()) * np.log(peak_region / peak_region.sum() + 1e-30))
    effective_peaks = np.exp(peak_entropy)
    
    base_beta = max_c * min(0.05, 0.01 * effective_peaks / n_top)
    base_beta = max(base_beta, max_c * 0.001)

    # Adaptive beta schedule: start with large beta, decrease geometrically
    n_betas = min(20, max(8, int((deadline - min_time_remaining - time.monotonic()) / 15)))
    betas = base_beta * (0.55 ** np.arange(n_betas))

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Estimate time per iteration
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(50, int(remaining * 0.7 / (est_time * 20))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob.f, z, jac=prob.g, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-19, 'gtol': 1e-16,
                         'maxcor': 120, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _fourier_ils_perturb(a, rng, strength=0.05):
    """Perturb in Fourier domain: add noise to magnitudes, keep phases."""
    n = len(a)
    N = _next_pow2(n)
    Fa = np.fft.rfft(a, N)
    magnitudes = np.abs(Fa)
    phases = np.angle(Fa)
    
    # Add noise to magnitudes with frequency-dependent weighting
    freqs = np.arange(len(magnitudes))
    # Low frequencies get less noise (preserve global structure)
    noise_scale = strength * (1.0 + 0.5 * freqs / max(1, len(freqs)))
    noise = noise_scale * rng.standard_normal(len(magnitudes))
    
    new_magnitudes = np.maximum(magnitudes + noise, 0)
    new_Fa = new_magnitudes * np.exp(1j * phases)
    a_new = np.fft.irfft(new_Fa, N)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(n) / n
    return a_new


def _structured_perturb(a, rng, mode=0):
    """Structured perturbation preserving more of the sequence structure."""
    n = len(a)
    if mode == 0:
        # Local block perturbation
        block_size = max(10, n // 16)
        start = int(rng.integers(0, n - block_size))
        a_p = a.copy()
        segment = a_p[start:start + block_size].copy()
        # Random smooth perturbation of the block
        t = np.linspace(0, 1, block_size)
        a1 = float(rng.uniform(0.5, 1.5))
        a2 = float(rng.uniform(0.5, 1.5))
        phase = float(rng.uniform(0, 2 * np.pi))
        modulation = a1 + a2 * np.cos(2 * np.pi * t + phase)
        a_p[start:start + block_size] = segment * modulation
        return _normalize(a_p)
    elif mode == 1:
        # Sparse perturbation with smooth kernel
        n_perturb = max(5, n // 30)
        idx = rng.choice(n, size=n_perturb, replace=False)
        a_p = a.copy()
        for i in idx:
            offset = int(rng.integers(-5, 6))
            j = (i + offset) % n
            a_p[j] += a_p[i] * 0.1 * rng.standard_normal()
        return _normalize(a_p)
    elif mode == 2:
        # Phase perturbation in Fourier domain
        N = _next_pow2(n)
        Fa = np.fft.rfft(a, N)
        phases = np.angle(Fa)
        mags = np.abs(Fa)
        # Add small phase noise (more at higher frequencies)
        freqs = np.arange(len(phases))
        phase_noise = 0.02 * (freqs / max(1, len(freqs))) * rng.standard_normal(len(phases))
        new_Fa = mags * np.exp(1j * (phases + phase_noise))
        a_new = np.fft.irfft(new_Fa, N)[:n]
        return _normalize(a_new)
    else:
        # Multiplicative perturbation with smooth factor
        t = np.linspace(0, 1, n)
        n_harmonics = int(rng.integers(2, 6))
        factor = np.ones(n)
        for h in range(n_harmonics):
            amp = float(rng.uniform(0, 0.1 * (h + 1)))
            freq = int(rng.integers(1, n // 4 + 1))
            phase = float(rng.uniform(0, 2 * np.pi))
            factor += amp * np.cos(2 * np.pi * freq * t + phase)
        factor = np.maximum(factor, 0.1)
        return _normalize(a * factor)


def _multigrid_optimize(a_fine, deadline, n_coarse=4096, min_time_remaining=50):
    """Multigrid: optimize at coarse level, then refine at fine level."""
    n_fine = len(a_fine)
    
    # Downsample
    a_coarse = _resample(a_fine, n_coarse)
    
    # Optimize at coarse level
    coarse_deadline = min(deadline - min_time_remaining, time.monotonic() + 60)
    a_coarse_opt = _run_lbfgs_adaptive(a_coarse, coarse_deadline, min_time_remaining=5)
    
    # Upsample back
    a_fine_init = _resample(a_coarse_opt, n_fine)
    
    # Combine with original (weighted average in log-space)
    z_orig = np.log(np.maximum(a_fine, 1e-30)) - np.log(np.maximum(a_fine, 1e-30)).mean()
    z_coarse = np.log(np.maximum(a_fine_init, 1e-30)) - np.log(np.maximum(a_fine_init, 1e-30)).mean()
    z_blend = 0.5 * z_orig + 0.5 * z_coarse
    a_blend = _softmax(z_blend)
    
    # Quick optimize the blend
    blend_sc = _score(a_blend)
    orig_sc = _score(a_fine)
    
    if blend_sc < orig_sc:
        return a_blend
    return a_fine


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _multiplicative_update(a, c, n_peaks=5, lr=0.02, n_steps=10):
    """Multiplicative weight update in log-space for peak reduction."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    for step in range(n_steps):
        c = _full_convolve(_softmax(z), _softmax(z))
        c_max = float(c.max())
        
        # Compute peak weights
        c_sorted_idx = np.argsort(c)[::-1]
        n_use = min(n_peaks, len(c))
        peak_indices = c_sorted_idx[:n_use]
        peaks = c[peak_indices]
        eps = 1e-12 * c_max
        w = np.zeros_like(c)
        weights = 1.0 / (c_max - peaks + eps)
        weights = weights / weights.sum()
        for i, idx in enumerate(peak_indices):
            w[idx] = weights[i]
        
        # Gradient in the multiplicative sense
        a_cur = _softmax(z)
        a_rev = a_cur[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        log_grad = h - W
        
        # Multiplicative update
        z = z - lr * log_grad
        z = z - z.mean()
    
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-scale hybrid optimizer combining adaptive L-BFGS, multiplicative
    weight updates, Fourier-domain ILS, and multigrid refinement.
    """
    global t_start
    t_start = time.monotonic()

    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "kaiser")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, kaiser n={n_parent}, score={best_score:.10f}")

    n_main = n_parent if n_parent > 0 else 16384

    # === Phase 1: Quick multi-start (150s) ===
    print(f"\n[P1] Multi-start (t={time.monotonic() - t_start:.1f}s)")
    ms_deadline = t_start + 150

    starts = [('parent', parent.copy())]
    for kind in ['kaiser', 'triangle', 'hann', 'optimal_window']:
        starts.append((kind, _make_profile(n_main, kind)))

    for name, a_init in starts:
        if time.monotonic() > ms_deadline - 30:

        a_opt = _run_lbfgs_adaptive(a_init, ms_deadline, min_time_remaining=20)

        lbfgs_deadline = min(deadline - 600, time.monotonic() + 300)
        a_main = _run_lbfgs_adaptive(best_seq, lbfgs_deadline, min_time_remaining=30)
        sc = _score(a_main)
        if sc < best_score:
            best_seq = a_main.copy()
            best_score = sc
            print(f"  NEW BEST: {best_score:.10f} [P2/lbfgs]")

    # === Phase 3: Adaptive active-set polish (150s) ===
    print(f"\n[P3] Active-set polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        pol_deadline = min(deadline - 400, time.monotonic() + 150)
        a_pol, sc_pol = _adaptive_polish(best_seq, pol_deadline,
                                          tol_start=0.005, tol_end=0.000005,
                                          lr_init=0.03, momentum=0.85, max_peaks=10)
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  NEW BEST: {best_score:.10f} [P3/polish]")

    # === Phase 4: Multiplicative weight updates (100s) ===
    print(f"\n[P4] Multiplicative updates (t={time.monotonic() - t_start:.1f}s)")

        sc_mu = _score(a_mu)
        if sc_mu < best_score:
            best_seq = a_mu.copy()
            best_score = sc_mu
            print(f"  NEW BEST: {best_score:.10f} [P4/mult]")
        # Follow up with polish
        if time.monotonic() < mu_deadline - 10:
            a_pol2, sc_pol2 = _adaptive_polish(best_seq, mu_deadline,
                                                tol_start=0.003, tol_end=0.00001,

            if sc_pol2 < best_score:
                best_seq = a_pol2.copy()
                best_score = sc_pol2

            elif sc_opt < current_sc - 1e-12:
                current_a = a_opt.copy()
                current_sc = sc_opt
            
            attempt += 1
        
        if ilsc_best < best_score:
            best_seq = ilsc_best_a.copy()
            best_score = ilsc_best
            print(f"  NEW BEST: {best_score:.10f} [P5/ils]")

    # === Phase 6: Multi-n exploration (200s) ===
    print(f"\n[P6] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [n_main, int(n_main * 1.5), int(n_main * 0.75),
                    int(n_main * 2), int(n_main * 1.25)]
    seen = set()
    n_unique = []
    for nn in n_candidates:
        nn = max(4096, min(65536, nn))
        if nn not in seen and nn != len(best_seq):
            seen.add(nn)
            n_unique.append(nn)

    for n_try in n_unique:
        if time.monotonic() > deadline - 100:
            break
        a_init = _resample(best_seq, n_try)
        a_opt = _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=30)
        # Polish
        if time.monotonic() < deadline - 40:
            a_pol3, sc_pol3 = _adaptive_polish(a_opt, min(deadline - 30, time.monotonic() + 20),
                                                tol_start=0.003, tol_end=0.00002,
                                                lr_init=0.015, momentum=0.8, max_peaks=5)
            if sc_pol3 < _score(a_opt):
                a_opt = a_pol3
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [P6]")

    # === Phase 7: Multigrid refinement (100s) ===
    print(f"\n[P7] Multigrid refinement (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        mg_deadline = min(deadline - 60, time.monotonic() + 100)
        a_mg = _multigrid_optimize(best_seq, mg_deadline,
                                    n_coarse=max(1024, len(best_seq) // 4),

            best_score = sc_mg

    print(f"\n[P8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 5:

                                          tol_start=0.001, tol_end=0.000001,
                                          lr_init=0.008, momentum=0.82, max_peaks=4)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [P8/polish]")

                    best_score = sc_fin2

            except Exception:
                pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:

        return [float(x) for x in final]

    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
