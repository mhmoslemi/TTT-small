At the optimum of $\min \max_k c_k$ subject to $\sum a_i = 1, a_i \geq 0$:

KKT conditions:
1. For all $k$ with $\lambda_k > 0$ (active peaks): $c_k = c_{\max}$

3. For all $i$ with $a_i = 0$: $\sum_k \lambda_k \cdot 2a_{k-i} \leq -\mu$ (complementary slackness)

So the condition is: on the support, the "dual convolution" $d_i = \sum_k \lambda_k a_{k-i}$ should be constant.

4. Compute $d_i = \sum_{k \in S} a_{k-i} = |S| \cdot (\text{conv}(1_S, a))[2i]$...

If I define $w_k = 1/|S|$ for $k \in S$ and 0 otherwise, then $d_i = |S| \cdot \sum_k w_k a_{k-i}$.

Now, $\sum_k w_k a_{k-i}$ is the $(i)$-th coefficient of the convolution of $w$ with $a_{\text{rev}}$ (reverse of $a$):
$\text{conv}(w, a_{\text{rev}})[i + (n-1)] = \sum_k w_k \cdot a_{\text{rev}}[i + (n-1) - k] = \sum_k w_k \cdot a[n-1 - i + k - (n-1)] = \sum_k w_k \cdot a[k - i]$

Wait: $a_{\text{rev}}[j] = a[n-1-j]$. So $a_{\text{rev}}[i + n - 1 - k] = a[n-1 - (i+n-1-k)] = a[k-i]$.

Yes! So $d_i = |S| \cdot \text{conv}(w, a_{\text{rev}})[i + n - 1]$.

The KKT condition is that $d_i = \text{const}$ for all $i$ in the support.

Algorithm:
1. Compute $c$, find $S$
2. Compute $d_i = \sum_{k \in S} a_{k-i}$ using FFT
3. Let $\bar{d} = \text{mean}(d_i : i \in \text{support})$
4. Residual: $r_i = d_i - \bar{d}$ for $i$ in support
5. Update: $a_i \leftarrow a_i - \eta \cdot r_i$ for $i$ in support
6. Project: $a_i = \max(a_i, 0)$, normalize
7. Line search on $\eta$

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a_norm = a / s

    return float(2.0 * n * c.max() / 1.0)

        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.5)
            if tol_frac <= min_tol * 1.1:
                break
            no_improve = 0

        # Identify active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Compute dual convolution d_i = sum_{k in S} a[k-i]
        # Using FFT: d_i = conv(w, a_rev)[n-1+i] where w = 1_S / n_active
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        d = wc[n - 1:2 * n - 1]  # d[i] = sum_{k in S} (1/|S|) a[k-i]

        # Identify support
        support_thresh = 1e-10 * a.max()
        support = a > support_thresh
        n_support = int(support.sum())
        if n_support < 2:
            break

        # Try to find better lambda via a few LS iterations
        # d_i = sum_{k in S} lambda_k * a[k-i]
        # We want d to be as constant as possible on support
        # Gradient of ||d[support] - mean||^2 w.r.t. lambda:
        # = 2 * M^T (M lambda - mean) where M[i,k] = a[k-i]
        # Instead of full LS, use the simple update:
        # residual on support
        d_support = d[support]
        d_mean = float(d_support.mean())
        residual = d_support - d_mean

        # Update a: a[i] -= eta * residual[i] for i in support
        # But we need to be careful: changing a changes d nonlinearly
        # Use a small step
        eta = 0.02
        best_eta = 0
        best_sc_try = current_sc
        for trial_eta in [0.05, 0.03, 0.02, 0.01, 0.005, 0.002]:
            a_try = a.copy()
            a_try[support] -= trial_eta * residual
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_eta = trial_eta

        if best_eta == 0:
            # No improvement, tighten tolerance
            tol_frac = max(min_tol, tol_frac * 0.7)
            continue

        a[support] -= best_eta * residual
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        # Gradually tighten
        if no_improve == 0:
            tol_frac = max(min_tol, tol_frac * 0.98)

        if verbose and no_improve % 50 == 0:
            elapsed = time.monotonic() - t_start_global
            print(f"    KKT: tol={tol_frac:.6f}, score={best_sc:.10f}, "
                  f"|S|={n_active}, |T|={n_support}, t={elapsed:.0f}s")

    return best_a, best_sc


def _kkt_with_optimal_lambda(a_init, deadline, min_time_remaining=30.0, verbose=True):
    """KKT refinement with optimal lambda via iterative LS."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_frac = 0.005
    min_tol = 0.0001
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            tol_frac = max(min_tol, tol_frac * 0.5)
            if tol_frac <= min_tol * 1.1:
                break
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        if n_active > 500:
            # Too many active, just use top peaks
            top_k = np.argsort(c)[::-1][:200]
            active = np.zeros_like(c, dtype=bool)
            active[top_k] = True
            n_active = 200

        support = a > 1e-10 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        # Build the linear system for optimal lambda
        # We want: M @ lambda = d_target (constant on support)
        # M[i,k] = a[k-i] for i in support, k in active
        # This is a |T| x |S| matrix with Hankel structure
        # Instead of forming M explicitly, use the structure:
        # (M @ lambda)[i] = sum_{k in S} lambda[k] * a[k-i] = conv(lambda, a_rev)[n-1+i]
        
        # Initialize lambda uniformly
        lam = np.zeros(len(c))
        lam[active] = 1.0 / n_active

        # Iterative refinement: adjust lambda to make d more constant on support
        for ls_it in range(20):
            if time.monotonic() + min_time_remaining > deadline:
                break

            # Compute d = conv(lam, a_rev)[n-1 : 2n-1]
            a_rev = a[::-1]
            xc = _full_convolve(lam, a_rev)
            d = xc[n - 1:2 * n - 1]

            # Residual on support
            d_sup = d[support]
            d_mean = d_sup.mean()
            r = d_sup - d_mean  # residual

            # Gradient w.r.t. lam[k] = sum_{i in support} r_i * a[k-i]
            # = sum_{i in support} r_i * a[(k - i)]
            # = conv(r_ext, a_rev)[n-1+k] where r_ext has r on support positions
            r_ext = np.zeros(n)
            r_ext[support] = r
            r_ext_rev = r_ext[::-1]
            # We need: grad_lam[k] = sum_{i in support} r_i * a[k-i]
            # = (r_ext * a_rev) at appropriate position
            # sum_{i} r_ext[i] * a[k-i] = conv(r_ext, a_rev)[k + n - 1]
            grad_lam_full = _full_convolve(r_ext, a_rev)
            grad_lam = grad_lam_full[n - 1:2 * n - 1]

            # Only project onto active set
            grad_active = grad_lam * active.astype(float)
            g_norm_sq = float(np.dot(grad_active, grad_active))
            if g_norm_sq < 1e-30:
                break

            # Gradient descent step on lambda
            r_norm_sq = float(np.dot(r, r))
            step = r_norm_sq / (g_norm_sq + 1e-30)
            step = min(step, 0.5)

            lam_new = lam + step * grad_active
            lam_new[~active] = 0.0
            lam_new = np.maximum(lam_new, 0.0)
            lam_sum = lam_new.sum()
            if lam_sum > 1e-30:
                lam = lam_new / lam_sum
            # Re-center to avoid drift
            lam_active = lam[active]
            if lam_active.sum() > 1e-30:
                lam[active] = lam_active / lam_active.sum()

        # Now use optimal lambda to update a
        a_rev = a[::-1]
        xc = _full_convolve(lam, a_rev)
        d = xc[n - 1:2 * n - 1]
        d_sup = d[support]
        d_mean = d_sup.mean()
        residual = d_sup - d_mean

        # Line search
        best_sc_try = current_sc
        best_eta = 0
        for trial_eta in [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]:
            a_try = a.copy()
            a_try[support] -= trial_eta * residual
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_eta = trial_eta

        if best_eta > 0:
            a[support] -= best_eta * residual
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-10:
                break
            a = a / s
        else:
            tol_frac = max(min_tol, tol_frac * 0.7)

        if no_improve == 0:
            tol_frac = max(min_tol, tol_frac * 0.97)

    return best_a, best_sc


def _peak_shave(a_init, deadline, min_time_remaining=30.0):
    """Peak shaving: for each peak, find best single-coordinate adjustment."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()

        # For each of the top peaks, try to reduce it
        peak_indices = np.argsort(c)[::-1][:5]

        improved = False
        for k_peak in peak_indices:
            k = int(k_peak)
            if c[k] < c_max * 0.99:
                break

            # Find which a[i] contribute to c[k]
            # c[k] = sum_i a[i] * a[k-i]
            # To reduce c[k], we want to reduce a[i] where a[k-i] is large
            i_lo = max(0, k - (n - 1))
            i_hi = min(n - 1, k)

            if i_lo > i_hi:
                continue

            # For each i in [i_lo, i_hi], the contribution is a[i]*a[k-i]
            # Try reducing a[i] proportionally to a[k-i]
            best_reduction = 0
            best_i = -1
            best_factor = 1.0

            for i in range(i_lo, i_hi + 1):
                contribution = a[i] * a[k - i]
                if contribution > best_reduction:
                    best_reduction = contribution
                    best_i = i
                    best_factor = 0.5  # reduce by 50%

            if best_i >= 0 and best_reduction > 0:
                # Try reducing a[best_i]
                for factor in [0.7, 0.5, 0.3, 0.1]:
                    a_try = a.copy()
                    a_try[best_i] *= factor
                    s_try = a_try.sum()
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    sc_try = 2.0 * n * float(c_try.max())
                    if sc_try < current_sc - 1e-12:
                        a = a_try
                        current_sc = sc_try
                        improved = True
                        break

        if not improved:
            break

    return best_a, best_sc


def _nesterov_optimize(a_init, deadline, min_time_remaining=30.0, verbose=True):
    """Nesterov accelerated gradient descent on softmax parameterization."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(200, len(c_sorted) - 1)]
    if gap < c0_max * 0.001:
        gap = c0_max * 0.001

    n_levels = 20
    betas = np.geomspace(gap * 0.05, gap * 0.00005, n_levels)

    # Nesterov momentum
    mu = 0.9
    lr = 0.005
    x = z.copy()
    v = np.zeros(n)
    t_nest = 0

    for bi, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        size_factor = max(1.0, n / 8192.0)
        max_iters = min(3000, max(100, int(remaining / (0.005 * size_factor * 20))))

        lr_cur = lr * min(1.0, (beta / (gap * 0.01)) ** 0.5)
        lr_cur = max(lr_cur, 1e-5)

        for it in range(max_iters):
            if time.monotonic() + min_time_remaining > deadline:
                break

            # Nesterov step
            y = x + mu * v
            g = _grad_log_domain(y, beta, n)
            v = mu * v - lr_cur * g
            x_new = x + v

            a_new = _softmax(x_new)
            c_new = _full_convolve(a_new, a_new)
            sc_new = 2.0 * n * float(c_new.max())

            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()

            # Backtracking: if objective increased too much, reduce step
            a_curr = _softmax(x)
            c_curr = _full_convolve(a_curr, a_curr)
            sc_curr = 2.0 * n * float(c_curr.max())

            if sc_new > sc_curr + 1e-6 * sc_curr:
                # Reject step
                x_new = x - 0.5 * v
                v = 0.5 * v
                lr_cur *= 0.5

            x = x_new

            if it % 200 == 0:
                x = x - x.mean()

        if verbose and (bi % 4 == 0):
            elapsed = time.monotonic() - t_start_global
            print(f"    Nesterov beta={beta:.2e} ({bi+1}/{n_levels}): "
                  f"score={best_sc:.10f}, t={elapsed:.0f}s")

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "one_sided":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * t) + 0.001
        return a / a.sum()
    elif kind == "one_sided2":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t) + 0.001
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.3) ** 2 / 0.1) + 0.001
        return a / a.sum()
    elif kind == "bimodal_asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-50.0 * (t - 0.25) ** 2) + 0.3 * np.exp(-30.0 * (t - 0.7) ** 2) + 0.001
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, magnitude, seed=0):
    """Add random perturbation and re-normalize."""
    rng = np.random.default_rng(seed)
    n = len(a)
    noise = rng.normal(0, magnitude, size=n)
    a_new = a + noise * a
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    P1: Adam on soft-max (main workhorse)
    P2: Tight KKT refinement
    P3: KKT with optimal lambda
    P4: Nesterov with different schedule
    P5: Multi-n exploration
    P6: Diverse starting points
    P7: Final polish
    """
    global t_start_global
    t_start_global = time.monotonic()
    deadline = t_start_global + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    n_main = 16384
    if parent is not None:
        n_main = len(parent)
        parent_sc = _score(parent)
        print(f"Parent: n={n_main}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(n_main, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n={n_main}: score={best_score:.10f}")

    # === Phase 1: Adam optimization (main workhorse) ===
    print(f"\n[P1] Adam optimize (n={n_main}, t={time.monotonic()-t_start_global:.0f}s)")
    p1_end = t_start_global + min(600, (budget_s - 10) * 0.35)
    a1, sc1 = _adam_optimize(best_seq, p1_end, min_time_remaining=300, verbose=True)
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  P1 result: {best_score:.10f}")

    # === Phase 2: Tight KKT refinement ===
    print(f"\n[P2] KKT refine (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p2_end = min(deadline - 400, time.monotonic() + 300)
        a2, sc2 = _kkt_refine_tight(best_seq, p2_end, min_time_remaining=50, verbose=True)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(f"  P2 result: {best_score:.10f}")

    # === Phase 3: KKT with optimal lambda ===
    print(f"\n[P3] KKT optimal lambda (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p3_end = min(deadline - 300, time.monotonic() + 250)
        a3, sc3 = _kkt_with_optimal_lambda(best_seq, p3_end, min_time_remaining=50, verbose=True)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3 result: {best_score:.10f}")

    # === Phase 4: Nesterov with different schedule ===
    print(f"\n[P4] Nesterov (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p4_end = min(deadline - 200, time.monotonic() + 250)
        a4, sc4 = _nesterov_optimize(best_seq, p4_end, min_time_remaining=50, verbose=True)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4 result: {best_score:.10f}")
        # Quick KKT after Nesterov
        if time.monotonic() + 30 < deadline:
            p4b_end = min(deadline - 100, time.monotonic() + 120)
            a4b, sc4b = _kkt_refine_tight(best_seq, p4b_end, min_time_remaining=30, verbose=False)
            if sc4b < best_score:
                best_score = sc4b
                best_seq = a4b.copy()
                print(f"  P4b KKT: {best_score:.10f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n (t={time.monotonic()-t_start_global:.0f}s)")
    n_options = [12288, 20480, 24576, 32768]
    n_options = [x for x in n_options if x != n_main and x <= 32768]
    for n_try in n_options:
        if time.monotonic() + 100 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if sc_r < best_score + 0.1:  # only optimize if competitive
            p5_end = min(deadline - 80, time.monotonic() + 120)
            a5, sc5 = _adam_optimize(a_r, p5_end, min_time_remaining=30, verbose=False)
            if sc5 < best_score:
                best_score = sc5
                best_seq = a5.copy()
                print(f"  n={n_try}: {best_score:.10f}")
            # Quick KKT
            if time.monotonic() + 30 < deadline and sc5 < best_score + 0.01:
                p5b_end = min(deadline - 50, time.monotonic() + 60)
                a5b, sc5b = _kkt_refine_tight(a5, p5b_end, min_time_remaining=20, verbose=False)
                if sc5b < best_score:
                    best_score = sc5b
                    best_seq = a5b.copy()
                    print(f"  n={n_try} KKT: {best_score:.10f}")

    # === Phase 6: Diverse perturbations ===
    print(f"\n[P6] Perturbations (t={time.monotonic()-t_start_global:.0f}s)")
    n_final = len(best_seq)
    seeds = [100, 200, 300, 400, 500]
    for s_val in seeds:
        if time.monotonic() + 80 > deadline:
            break
        a_p = _perturb(best_seq, 0.02, seed=s_val)
        sc_p = _score(a_p)
        # Only optimize if not much worse
        if sc_p < best_score + 0.05:
            p6_end = min(deadline - 40, time.monotonic() + 80)
            a6, sc6 = _kkt_refine_tight(a_p, p6_end, min_time_remaining=20, verbose=False)
            if sc6 < best_score:
                best_score = sc6
                best_seq = a6.copy()
                print(f"  perturb seed={s_val}: {best_score:.10f}")

    # === Phase 7: Final polish ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 20 < deadline:
        p7_end = min(deadline - 5, time.monotonic() + 90)
        a7, sc7 = _kkt_refine_tight(best_seq, p7_end, min_time_remaining=5, verbose=True)
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()

        if time.monotonic() + 15 < deadline:
            p7b_end = min(deadline - 3, time.monotonic() + 50)
            a7b, sc7b = _kkt_with_optimal_lambda(best_seq, p7b_end, min_time_remaining=3, verbose=False)
            if sc7b < best_score:
                best_score = sc7b
                best_seq = a7b.copy()

        # One final Adam pass
        if time.monotonic() + 15 < deadline:
            p7c_end = min(deadline - 2, time.monotonic() + 40)
            a7c, sc7c = _adam_optimize(best_seq, p7c_end, min_time_remaining=2, verbose=False)
            if sc7c < best_score:
                best_score = sc7c
                best_seq = a7c.copy()

    # Final output
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start_global:.1f}s")
    return [float(x) for x in final]
