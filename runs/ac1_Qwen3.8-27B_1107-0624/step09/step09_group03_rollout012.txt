import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

"""
Optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Spectral equalization (IFTA) for initialization
2. L-BFGS with aggressive log-sum-exp continuation (40-step geometric beta)
3. Active-set Newton with Armijo line search
4. Frequency-domain peak deflation
5. Multi-scale with adaptive random restarts
6. Final polish with ultra-small beta
"""
import numpy as np
import time
from scipy.optimize import minimize

# Global parameters for scipy callback (avoids closures)
_g_beta = 1.0
_g_n = 1024
_g_pl = 2.0  # power for power-sum smoothing alternative


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming a is already normalized to sum=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _lsobj(z):
    beta = _g_beta
    n = _g_n
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z):
    beta = _g_beta
    n = _g_n
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _pobj(z):
    """Power-sum smoothing objective (alternative to LSE)."""
    p = _g_pl
    n = _g_n
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / (c_max + 1e-30)
    sp = np.sum(shifted ** p)
    return 2.0 * n * c_max * (1.0 + sp ** (1.0 / p))


def _ifft_amplitude_equalize(a, n_iter=30, target_flat=True):
    """
    IFTA/Gerchberg-Saxton: equalize |A(f)|^2 to be flat.
    This pushes the autocorrelation toward being delta-like.
    """
    n = len(a)
    a = a.copy()
    N = _next_pow2(2 * n - 1)
    for _ in range(n_iter):
        A = np.fft.rfft(a, N)
       mag_A = np.abs(A)
        # Target: flat amplitude (all equal)
        # But we can't make DC zero since sum(a)>0
        if target_flat:
            target_mag = np.full_like(mag_A, np.mean(mag_A[1:]))  # exclude DC
            target_mag[0] = mag_A[0]  # keep DC (determined by sum)
        else:
            target_mag = np.ones_like(mag_A)
            target_mag[0] = mag_A[0]
        # Scale A to match target magnitude, preserve phase
        scale = target_mag / (mag_A + 1e-30)
        A_new = A * scale
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s
        a = a_new
    return a


def _ifft_tapered(a, n_iter=20, taper_power=2.0):
    """
    IFTA with tapered target: |A(f)|^2 should decrease away from f=0.
    This helps maintain non-negativity.
    """
    n = len(a)
    a = a.copy()
    N = _next_pow2(2 * n - 1)
    freqs = np.arange(N // 2 + 1)
    # Tapered target: maximum at low freq, decreasing
    target = np.exp(-taper_power * (freqs / (N // 2)) ** 2)
    target /= target.sum()
    peak_target = target.max()
    for _ in range(n_iter):
        A = np.fft.rfft(a, N)
        mag_A = np.abs(A)
        # Scale proportionally to target shape
        scale = (peak_target * target) / (mag_A + 1e-30)
        A_new = A * scale
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s
        a = a_new
    return a


def _corr_a_A(a, A_vec, n):
    """Compute correlation of a with A_vec, restricted to valid range."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _newton_step_active_set(a, c, c_max, tol_frac, mu_base=1e-7):
    """
    Newton step with active set defined by tolerance.
    Returns (dx, A_vec) or (None, None) if step fails.
    """
    n = len(a)
    active = c >= (c_max - tol_frac * c_max)
    if not active.any():
        active[np.argmax(c)] = True
    m = int(active.sum())
    if m < 2 or m > 20 * n:
        return None, None

    A_vec = active.astype(np.float64) / m
    g = 2.0 * _corr_a_A(a, A_vec, n)
    g_t = g - np.mean(g)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-14:
        return None, None

    mu = mu_base / m + 1e-10

    # CG solve for (H + mu*I) u = g_t where H is the Hessian of the active constraint
    def _hess_vp(v):
        return 2.0 * _corr_a_A(v, A_vec, n) + mu * v

    u = _cg_solve_matvec(_hess_vp, g_t, max_cg=300, tol=1e-12)
    w = _cg_solve_matvec(_hess_vp, np.ones(n), max_cg=300, tol=1e-12)
    w_sum = float(w.sum())

    if abs(w_sum) < 1e-15:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w

    dx = dx - np.mean(dx)
    return dx, A_vec


def _cg_solve_matvec(matvec_fn, rhs, max_cg=200, tol=1e-13):
    """Conjugate gradient solver."""
    n = len(rhs)
    x = np.zeros(n)
    r = rhs - matvec_fn(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_cg):
        Ap = matvec_fn(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            break
        if rs < 1e-30:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _active_set_optimize(a_init, deadline, max_outer=100):
    """Active-set Newton optimization."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Try different tolerance levels
        tol_list = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        best_improvement = 0.0
        best_a_new = None

        for tol_frac in tol_list:
            dx, A_vec = _newton_step_active_set(a, c, c_max, tol_frac)
            if dx is None:
                continue

            # Armijo backtracking line search
            f_cur = float(c_max)
            slope = np.dot(dx, 2.0 * _corr_a_A(a, A_vec, n) - np.mean(2.0 * _corr_a_A(a, A_vec, n)))
            if slope >= 0:
                slope = -1e-6  # Ensure descent direction estimate

            t = 1.0
            for ls in range(60):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                armijo = f_cur + 1e-4 * t * slope
                if f_try <= armijo + 1e-15:
                    improvement = f_cur - f_try
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_a_new = a_try.copy()
                    break
                t *= 0.5

            if time.monotonic() > deadline:
                break

        if best_a_new is None:
            break

        a = best_a_new
        sc = _score_normalized(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if best_improvement < 1e-14:
            break

    return best_a, best_sc


def _lbfgs_run(a_init, betas, deadline, min_time_remaining=30.0):
    """
    Run L-BFGS continuation over a sequence of beta values.
    """
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score_normalized(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    total_betas = len(betas)
    for i, beta in enumerate(betas):
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Allocate more iterations to smaller beta
        frac = (1.0 / (i + 1)) / sum(1.0 / (j + 1) for j in range(total_betas))
        alloc_time = remaining * frac * 0.8
        if alloc_time < 1.0:
            alloc_time = 1.0

        # Estimate time per L-BFGS function eval
        est_te = 0.005 * max(1.0, n / 8192.0)
        max_iter = int(min(5000, max(20, alloc_time / est_te / 5)))

        # Set globals for scipy callback
        global _g_beta, _g_n
        _g_beta = beta
        _g_n = n

        inner_deadline = time.monotonic() + alloc_time

        try:
            def _wrap_obj(z_val):
                return _lsobj(z_val)

            def _wrap_grad(z_val):
                return _lsgrad(z_val)

            result = minimize(
                _wrap_obj,
                z,
                jac=_wrap_grad,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-16,
                    'maxcor': 50,
                    'maxls': 200,
                    'iprint': -1,
                }
            )

            if time.monotonic() > inner_deadline:
                continue

            a_new = _softmax(result.x)
            sc_new = _score_normalized(a_new)
            if sc_new < current_sc - 1e-18:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()

            if i % 5 == 0 or i == total_betas - 1:
                print(f"    LSE beta={beta:.2e}: score={sc_new:.12f} "
                      f"iter_cap={max_iter} t={time.monotonic() - (deadline - min_time_remaining - remaining):.1f}s")

        except Exception as ex:
            print(f"    LSE beta={beta:.2e}: exception {ex}")

    return current_a


def _freq_deflation(a, n_iter=50, deadline=None):
    """
    Frequency-domain peak deflation: modify |A(f)| to reduce autocorrelation peaks.
    For each peak position k in the autocorrelation, the corresponding frequency
    component that reinforces it is reduced.
    """
    n = len(a)
    a = a.copy()
    N = _next_pow2(2 * n - 1)
    best_sc = _score_normalized(a)

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Identify top peaks
        top_k = min(20, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]
        peak_threshold = c[peak_indices[-1]] * 0.95
        peak_mask = c >= peak_threshold

        # In frequency domain, the autocorrelation is IFFT(|A|^2)
        # To reduce peaks at specific lags, we reduce |A| at frequencies that
        # constructively interfere at those lags.
        # Simple approach: reduce |A| uniformly at high frequencies (where
        # rapid oscillation creates side lobes in autocorrelation).
        A = np.fft.rfft(a, N)
        freqs = np.arange(N // 2 + 1)

        # Compute current |A|^2
        mag_sq = np.abs(A) ** 2

        # Target: slightly reduce high-frequency components
        # This is equivalent to a mild low-pass filter in frequency domain
        low_pass = np.exp(-0.5 * (freqs / (N // 4)) ** 2)
        # Blend: keep low freq, reduce high freq
        blend = 0.5 + 0.5 * low_pass
        A_new = A * np.sqrt(blend)

        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s
        sc_new = _score_normalized(a_new)

        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            a = a_new
        else:
            # Try even more aggressive
            A = np.fft.rfft(a, N)
            A_new = A * (low_pass ** 2)
            a_new2 = np.fft.irfft(A_new, N)[:n]
            a_new2 = np.maximum(a_new2, 0.0)
            s2 = a_new2.sum()
            if s2 > 1e-30:
                a_new2 = a_new2 / s2
                sc_new2 = _score_normalized(a_new2)
                if sc_new2 < best_sc - 1e-14:
                    best_sc = sc_new2
                    a = a_new2
                else:
                    break
            else:
                break

    return a, best_sc


def _multi_peak_deflation(a, deadline, n_attempts=60):
    """Time-domain multi-peak deflation."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score_normalized(a)

    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        threshold = c_max * 0.995
        peak_mask = c >= threshold
        if not peak_mask.any():
            peak_mask[np.argmax(c)] = True
        n_peaks = int(peak_mask.sum())

        w = np.zeros_like(c)
        w[peak_mask] = 1.0 / n_peaks

        grad = 2.0 * _corr_a_A(a, w, n)
        g_t = grad - np.mean(grad)

        applied = False
        for strength in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
            a_try = a - strength * g_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score_normalized(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                applied = True
                break

        if not applied:
            rng_local = np.random.default_rng(42 + attempt)
            a_try = a * (1.0 + 0.005 * rng_local.standard_normal(n))
            a_try = _project_simplex(a_try)
            sc_try = _score_normalized(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
            else:
                break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian_centered":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block":
        a = np.zeros(n)
        gap = n // 5
        a[gap:2 * n - gap] = 1.0
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_seq(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_mult(a, sigma, rng):
    a_new = a * (1.0 + sigma * rng.standard_normal(len(a)))
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Main optimization function.
    Uses spectral initialization, aggressive LSE continuation,
    active-set Newton, frequency deflation, and random restarts.
    """
    global _g_beta, _g_n

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.12f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.12f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        update_best(parent, _score(parent), "init_uniform")

    n = len(parent)

    # === Phase 1: Spectral initialization (IFTA) ===
    print(f"\n[Phase 1] Spectral initialization (t={time.monotonic() - t_start:.1f}s)")
    phase1_dl = min(deadline - 600, time.monotonic() + 30)
    a_ifft = _ifft_amplitude_equalize(parent, n_iter=20)
    sc_ifft = _score_normalized(a_ifft)
    print(f"  IFTA flat: score={sc_ifft:.12f}")
    update_best(a_ifft, sc_ifft, "ifft_flat")

    a_tapered = _ifft_tapered(parent, n_iter=20, taper_power=1.5)
    sc_tap = _score_normalized(a_tapered)
    print(f"  IFTA tapered: score={sc_tap:.12f}")
    update_best(a_tapered, sc_tap, "ifft_tapered")

    # Use the best of parent and IFTA results as starting point
    if best_seq is not None:
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
    else:
        a_work = parent.copy()

    # === Phase 2: Aggressive L-BFGS continuation ===
    print(f"\n[Phase 2] L-BFGS aggressive continuation (t={time.monotonic() - t_start:.1f}s)")
    c_cur = _full_convolve(a_work, a_work)
    max_c = float(c_cur.max())

    # 40-step geometric beta schedule
    n_betas = 40
    beta_start = max_c * 0.05
    beta_end = max_c * 1e-9
    log_betas = np.linspace(np.log(beta_start), np.log(max(beta_end, 1e-30)), n_betas)
    betas = np.exp(log_betas)

    phase2_dl = min(deadline - 400, time.monotonic() + 400)
    a_lbfgs = _lbfgs_run(a_work, betas, phase2_dl, min_time_remaining=400.0)
    sc_lbfgs = _score_normalized(a_lbfgs)
    print(f"  L-BFGS result: score={sc_lbfgs:.12f}")
    update_best(a_lbfgs, sc_lbfgs, "lbfgs_main")

    # === Phase 3: Active-set Newton refinement ===
    print(f"\n[Phase 3] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        a_as = best_seq / best_seq.sum()
        phase3_dl = min(deadline - 300, time.monotonic() + 120)
        a_as_opt, sc_as = _active_set_optimize(a_as, phase3_dl, max_outer=200)
        update_best(a_as_opt, sc_as, "active_set_newton")

    # === Phase 4: Frequency-domain deflation ===
    print(f"\n[Phase 4] Frequency deflation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        a_fd = best_seq / best_seq.sum()
        phase4_dl = min(deadline - 250, time.monotonic() + 60)
        a_fd_opt, sc_fd = _freq_deflation(a_fd, n_iter=80, deadline=phase4_dl)
        update_best(a_fd_opt, sc_fd, "freq_deflation")

    # === Phase 5: Multi-peak time-domain deflation ===
    print(f"\n[Phase 5] Multi-peak deflation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        a_mp = best_seq / best_seq.sum()
        phase5_dl = min(deadline - 200, time.monotonic() + 60)
        a_mp_opt, sc_mp = _multi_peak_deflation(a_mp, phase5_dl, n_attempts=100)
        update_best(a_mp_opt, sc_mp, "multi_peak_def")

    # === Phase 6: L-BFGS polish with ultra-small beta ===
    print(f"\n[Phase 6] L-BFGS ultra-small beta polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        a_pol = best_seq / best_seq.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        mc = float(c_pol.max())
        # 15 ultra-small beta values
        betas_small = np.linspace(np.log(mc * 1e-5), np.log(max(mc * 1e-10, 1e-30)), 15)
        betas_small = np.exp(betas_small)
        phase6_dl = min(deadline - 150, time.monotonic() + 100)
        a_pol_opt = _lbfgs_run(a_pol, betas_small, phase6_dl, min_time_remaining=150.0)
        sc_pol = _score_normalized(a_pol_opt)
        print(f"  Polish result: score={sc_pol:.12f}")
        update_best(a_pol_opt, sc_pol, "lbfgs_polish")

    # === Phase 7: Randomized restarts ===
    print(f"\n[Phase 7] Randomized restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        a_base = best_seq / best_seq.sum()
        restart_configs = [
            (0.003, "log", 40),
            (0.01, "log", 40),
            (0.005, "mult", 30),
            (0.02, "mult", 30),
            (0.002, "log", 50),
            (0.05, "log", 20),
        ]
        for sigma, ptype, n_betas_r in restart_configs:
            if time.monotonic() > deadline - 60:
                break
            if ptype == "log":
                a_r = _perturb_log(a_base, sigma, rng)
            else:
                a_r = _perturb_mult(a_base, sigma, rng)

            c_r = _full_convolve(a_r, a_r)
            mc_r = float(c_r.max())
            betas_r = np.exp(np.linspace(np.log(mc_r * 0.02), np.log(max(mc_r * 1e-9, 1e-30)), n_betas_r))
            r_dl = min(deadline - 40, time.monotonic() + 50)
            a_r_opt = _lbfgs_run(a_r, betas_r, r_dl, min_time_remaining=40.0)
            sc_r = _score_normalized(a_r_opt)
            label = f"restart_{ptype}_{sigma}"
            update_best(a_r_opt, sc_r, label)

    # === Phase 8: Try n=32768 ===
    print(f"\n[Phase 8] Try n=32768 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        n_large = 32768
        a_large = _interpolate_seq(best_seq, n_large)
        sc_large = _score_normalized(a_large)
        print(f"  n={n_large} interpolated score: {sc_large:.12f}")

        # Quick L-BFGS on larger n
        c_l = _full_convolve(a_large, a_large)
        mc_l = float(c_l.max())
        betas_l = np.exp(np.linspace(np.log(mc_l * 0.03), np.log(max(mc_l * 1e-8, 1e-30)), 20))
        phase8_dl = min(deadline - 100, time.monotonic() + 120)
        a_large_opt = _lbfgs_run(a_large, betas_l, phase8_dl, min_time_remaining=100.0)
        sc_large_opt = _score_normalized(a_large_opt)
        print(f"  n={n_large} optimized score: {sc_large_opt:.12f}")
        update_best(a_large_opt, sc_large_opt, "n_32768")

        # Active set on large n
        if time.monotonic() < deadline - 80:
            a_ls = a_large_opt / a_large_opt.sum()
            as_dl = min(deadline - 60, time.monotonic() + 50)
            a_ls_opt, sc_ls = _active_set_optimize(a_ls, as_dl, max_outer=80)
            update_best(a_ls_opt, sc_ls, "n_32768_newton")

    # === Phase 9: Final refinement loop ===
    print(f"\n[Phase 9] Final refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        a_fin = best_seq / best_seq.sum()
        for rnd in range(5):
            if time.monotonic() > deadline - 20:
                break
            # Newton step
            ndl = min(deadline - 15, time.monotonic() + 15)
            a_nd, sc_nd = _active_set_optimize(a_fin, ndl, max_outer=30)
            if sc_nd < _score_normalized(a_fin):
                a_fin = a_nd
            # Deflation
            ddl = min(deadline - 10, time.monotonic() + 10)
            a_dd, sc_dd = _multi_peak_deflation(a_fin, ddl, n_attempts=20)
            if sc_dd < _score_normalized(a_fin):
                a_fin = a_dd
            # Small L-BFGS polish
            c_fin = _full_convolve(a_fin, a_fin)
            mc_fin = float(c_fin.max())
            betas_fin = np.linspace(np.log(mc_fin * 1e-6), np.log(max(mc_fin * 1e-10, 1e-30)), 8)
            betas_fin = np.exp(betas_fin)
            fdl = min(deadline - 10, time.monotonic() + 10)
            a_fin = _lbfgs_run(a_fin, betas_fin, fdl, min_time_remaining=10.0)

        sc_fin = _score_normalized(a_fin)
        print(f"  Final score: {sc_fin:.12f}")
        update_best(a_fin, sc_fin, "final_refinement")

    # === Final validation and return ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        # Validate with evaluate_sequence
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"Evaluate check: {ev_score:.12f}")
        except Exception:
            pass
        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(1024, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
