```python
"""
Optimized multiplicative descent with Adam + L-BFGS polish for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Key innovations:
- Adaptive multiplicative descent (AMD) with per-coordinate Adam learning rates
- Active-set annealing (wide to narrow) for global-to-local search
- Periodic randomization to escape local minima
- FFT-based gradient computation via convolution with active set indicator
- Final L-BFGS polish in softmax parameterization with very small beta
- Multi-n and multi-profile exploration
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_from_active(a, active_mask):
    """Compute gradient of average active convolution w.r.t. a.
    g[j] = (2/|S|) * sum_{k in S} a[k-j] for valid k-j.
    Uses FFT convolution with active indicator.
    """
    n = len(a)
    c_len = len(active_mask)
    w = active_mask.astype(np.float64)
    w_sum = w.sum()
    if w_sum < 1e-15:
        return np.zeros(n)
    w = w / w_sum
    a_rev = a[::-1].copy()
    wc = _full_convolve(w, a_rev)
    # wc has length c_len + n - 1
    # We want wc[n-1+j] for j=0..n-1
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _make_profile(n, kind="uniform", rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-50.0 * (t - 0.25) ** 2) + np.exp(-50.0 * (t - 0.75) ** 2) + 0.005
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 7.0 * t / n)
             + 0.2 * np.cos(2.0 * np.pi * 3.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 13.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        # Generate smooth random signal
        freqs = rng.integers(1, 20, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        amps = rng.uniform(0.1, 1.0, size=5)
        t = np.arange(n)
        a = np.ones(n)
        for i in range(5):
            a += amps[i] * np.sin(2 * np.pi * freqs[i] * t / n + phases[i])
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian_center":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    elif kind == "tukey":
        # Tukey (tapered cosine) window
        t = np.linspace(0, 1, n)
        r = 0.3  # taper fraction
        a = np.ones(n)
        left = int(r * n)
        right = n - int(r * n)
        if left > 0:
            tt = np.linspace(0, 0.5, left)
            a[:left] = 0.5 * (1 + np.cos(np.pi * (2 * tt / r - 1)))
        if right < n:
            tt = np.linspace(0.5, 1, n - right)
            a[right:] = 0.5 * (1 + np.cos(np.pi * (2 * tt / r - 1)))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _run_adam_descent(a_init, deadline, min_remaining=30.0,
                      eta0=0.1, eta_decay=0.9999,
                      tol_start=0.1, tol_end=0.001,
                      anneal_frac=0.6, restart_every=500,
                      restart_scale=0.01, verbose=True):
    """Adaptive Multiplicative Descent with Adam and active-set annealing."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    rng = np.random.default_rng(12345)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Total iterations estimate for annealing schedule
    # We'll anneal over the first anneal_frac of available time
    total_time = max(1.0, deadline - time.monotonic())
    anneal_iterations = int(total_time * anneal_frac * 2000)  # estimate ~2000 iter/s
    anneal_iterations = max(anneal_iterations, 100)

    iteration = 0
    last_print = 0

    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max  # since sum(a)=1

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()

        # Annealing: interpolate tolerance
        frac = min(1.0, iteration / max(1, anneal_iterations))
        tol = tol_start + (tol_end - tol_start) * frac
        tol = max(tol, 1e-6)

        # Active set
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Gradient
        g = _grad_from_active(a, active)
        # Center to preserve sum
        g = g - np.mean(g)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Adaptive step size
        g_norm = np.sqrt(np.mean(g * g))
        if g_norm < 1e-15:
            break
        eta = eta0 / (1.0 + 0.1 * iteration / 1000.0)

        # Multiplicative update
        update = -eta * m_hat / (np.sqrt(v_hat) + eps)
        # Clip update to prevent overflow
        update = np.clip(update, -5.0, 5.0)
        a_new = a * np.exp(update)
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Accept if improved (with some tolerance for noise)
        sc_new = _score(a_new)
        if sc_new < current_sc + 1e-12:
            a = a_new
        else:
            # Try smaller step
            update = update * 0.5
            a_try = a * np.exp(np.clip(update, -5.0, 5.0))
            a_try = np.maximum(a_try, 1e-15)
            s_try = a_try.sum()
            if s_try > 1e-10:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc:
                    a = a_try

        iteration += 1

        # Periodic restart
        if iteration % restart_every == 0 and frac < 0.9:
            noise_scale = restart_scale * (1.0 - frac)
            noise = rng.normal(0, noise_scale, n)
            a = a + noise
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            # Reset momentum
            m *= 0.5
            v *= 0.5

        # Print progress
        if verbose and iteration - last_print > 10000:
            elapsed = time.monotonic() - (deadline - total_time)
            print(f"    AMD iter={iteration}, score={best_sc:.10f}, "
                  f"tol={tol:.4f}, n_active={n_active}, t={elapsed:.0f}s")
            last_print = iteration

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, verbose=False):
    """L-BFGS with adaptive beta schedule in softmax parameterization."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    # Beta schedule
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < c0_max * 0.005:
        gap = c0_max * 0.005

    multipliers = np.array([
        0.5, 0.25, 0.1, 0.05, 0.02, 0.01,
        0.005, 0.002, 0.001, 0.0005,
        0.0002, 0.0001, 0.00005, 0.00002,
        0.00001, 0.000005, 0.000002
    ])
    betas = gap * multipliers

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        size_factor = max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (0.01 * size_factor))))

        def obj(z):
            aa = _softmax(z)
            cc = _full_convolve(aa, aa)
            return 2.0 * n * _soft_max_value(cc, beta)

        def grad(z):
            aa = _softmax(z)
            cc = _full_convolve(aa, aa)
            w = _soft_max_weights(cc, beta)
            aa_rev = aa[::-1].copy()
            wc = _full_convolve(w, aa_rev)
            grad_a = 2.0 * wc[n - 1:2 * n - 1]
            W = float(np.dot(grad_a, aa))
            return aa * (grad_a - W)

        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 50,
                    'maxls': 40,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"    L-BFGS beta={beta:.2e}: score={best_sc:.10f}")
        except Exception:
            pass

    return best_a, best_sc


def _kkt_refine(a_init, deadline, min_remaining=20.0):
    """KKT-inspired active-set refinement with multiplicative updates."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    no_improve = 0
    tol_frac = 0.01
    min_tol = 0.0002

    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            tol_frac = max(min_tol, tol_frac * 0.7)
            if tol_frac <= min_tol * 1.2:
                break
            no_improve = 0

        # Active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Gradient from active set
        g = _grad_from_active(a, active)
        g = g - np.mean(g)

        # KKT condition: at optimum, g should be ~0 on the support
        # Use multiplicative update to drive g toward 0
        support = a > 1e-10
        if int(support.sum()) < 2:
            break

        # Adaptive step size based on gradient magnitude
        g_std = np.std(g[support])
        if g_std < 1e-12:
            break
        eta = 0.3 * c_max / (g_std + 1e-12)
        eta = min(eta, 1.0)

        update = -eta * g
        update = np.clip(update, -2.0, 2.0)
        a_new = a * np.exp(update)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < current_sc:
            a = a_new
            # Tighten tolerance on improvement
            if best_sc <= sc_new + 1e-12:
                tol_frac = max(min_tol, tol_frac * 0.98)
        else:
            # Backtrack
            eta *= 0.3
            update = -eta * g
            update = np.clip(update, -2.0, 2.0)
            a_try = a * np.exp(update)
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                break
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < current_sc:
                a = a_try
            else:
                break

    return best_a, best_sc


def _coordinate_polish(a_init, deadline, min_remaining=10.0):
    """Coordinate descent: try setting individual elements to 0 or adjusting them."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(777)

    # Focus on elements that might be reducible
    c = _full_convolve(a, a)
    c_max = float(c.max())

    # Try zeroing elements that are in "high-convolution" regions
    order = rng.permutation(n)

    for j in order:
        if time.monotonic() + min_remaining > deadline:
            break
        aj = a[j]
        if aj < 1e-8:
            continue

        # Try reducing a[j]
        for frac in [0.0, 0.5, 0.8]:
            a_try = a.copy()
            a_try[j] = aj * frac
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-12:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining AMD (Adaptive Multiplicative Descent),
    L-BFGS polish, and KKT refinement.
    """
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(16384, "hann")
        best_score = _score(best_seq)
        print(f"No parent. Hann n=16384: score={best_score:.10f}")

    n_main = len(best_seq)
    print(f"Starting score: {best_score:.10f}, n={n_main}")

    # === Phase 1: Quick initialization search (try different profiles) ===
    print(f"\n[P1] Initialization search (t={time.monotonic()-t0:.1f}s)")
    p1_deadline = t0 + min(60, (budget_s - 5) * 0.04)
    profiles = ["hann", "tapered", "tukey", "raised_cosine", "periodic",
                "random_smooth", "bimodal", "gaussian_center"]
    n_options = [n_main]
    # Add neighboring sizes
    if n_main <= 32768:
        n_options.append(int(n_main * 1.25))
    if n_main >= 4096:
        n_options.append(int(n_main * 0.75))

    for prof in profiles:
        if time.monotonic() > p1_deadline:
            break
        for n_try in n_options:
            a_p = _make_profile(n_try, prof, rng)
            sc_p = _score(a_p)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_p.copy()
                n_main = n_try
                print(f"  {prof} n={n_try}: {sc_p:.10f}")

    # === Phase 2: AMD main optimization ===
    print(f"\n[P2] AMD main (n={len(best_seq)}, t={time.monotonic()-t0:.1f}s)")
    p2_deadline = t0 + min(int((budget_s - 5) * 0.45), 800)
    a2, sc2 = _run_adam_descent(
        best_seq, p2_deadline, min_remaining=300,
        eta0=0.15, tol_start=0.15, tol_end=0.002,
        anneal_frac=0.7, restart_every=300, restart_scale=0.02,
        verbose=True
    )
    if sc2 < best_score:
        best_score = sc2
        best_seq = a2.copy()
        print(f"  P2 AMD: {best_score:.10f}")

    # === Phase 3: L-BFGS polish ===
    print(f"\n[P3] L-BFGS polish (t={time.monotonic()-t0:.1f}s)")
    if time.monotonic() + 60 < deadline:
        p3_deadline = min(deadline - 200, time.monotonic() + 400)
        a3, sc3 = _run_lbfgs(best_seq, p3_deadline, min_time_remaining=100, verbose=True)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3 L-BFGS: {best_score:.10f}")

    # === Phase 4: KKT refinement ===
    print(f"\n[P4] KKT refine (t={time.monotonic()-t0:.1f}s)")
    if time.monotonic() + 30 < deadline:
        p4_deadline = min(deadline - 100, time.monotonic() + 200)
        a4, sc4 = _kkt_refine(best_seq, p4_deadline, min_remaining=50)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4 KKT: {best_score:.10f}")

    # === Phase 5: Second AMD round (finer) ===
    print(f"\n[P5] AMD round 2 (t={time.monotonic()-t0:.1f}s)")
    if time.monotonic() + 60 < deadline:
        p5_deadline = min(deadline - 80, time.monotonic() + 250)
        a5, sc5 = _run_adam_descent(
            best_seq, p5_deadline, min_remaining=40,
            eta0=0.05, tol_start=0.03, tol_end=0.0005,
            anneal_frac=0.5, restart_every=500, restart_scale=0.005,
            verbose=True
        )
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  P5 AMD2: {best_score:.10f}")

    # === Phase 6: L-BFGS final polish ===
    print(f"\n[P6] L-BFGS final (t={time.monotonic()-t0:.1f}s)")
    if time.monotonic() + 20 < deadline:
        p6_deadline = min(deadline - 20, time.monotonic() + 150)
        a6, sc6 = _run_lbfgs(best_seq, p6_deadline, min_time_remaining=10, verbose=True)
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  P6 L-BFGS final: {best_score:.10f}")

    # === Phase 7: Coordinate polish + KKT ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t0:.1f}s)")
    if time.monotonic() + 15 < deadline:
        p7_deadline = min(deadline - 10, time.monotonic() + 60)
        a7, sc7 = _kkt_refine(best_seq, p7_deadline, min_remaining=5)
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()
            print(f"  P7 KKT: {best_score:.10f}")

    # === Phase 8: Try different n on best solution ===
    print(f"\n[P8] Multi-n (t={time.monotonic()-t0:.1f}s)")
    if time.monotonic() + 30 < deadline:
        n_current = len(best_seq)
        for n_try in [int(n_current * 0.8), int(n_current * 1.2),
                      int(n_current * 1.5), int(n_current * 0.67)]:
            if time.monotonic() + 20 > deadline:
                break
            if n_try < 1024 or n_try > 65536:
                continue
            a_r = _resample(best_seq, n_try)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  n={n_try}: {sc_r:.10f}")
            # Quick AMD on resampled
            if time.monotonic() + 30 < deadline:
                p8_end = min(deadline - 10, time.monotonic() + 40)
                a8, sc8 = _run_adam_descent(
                    a_r, p8_end, min_remaining=5,
                    eta0=0.05, tol_start=0.05, tol_end=0.005,
                    anneal_frac=0.5, restart_every=200, restart_scale=0.01,
                    verbose=False
                )
                if sc8 < best_score:
                    best_score = sc8
                    best_seq = a8.copy()
                    print(f"  n={n_try} AMD: {sc8:.10f}")

    # === Final ===
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    if sc_final < best_score:
        best_score = sc_final
    print(f"FINAL: score={best_score:.10f}, n={len(final)}, "
          f"time={time.monotonic()-t0:.1f}s")
    return [float(x) for x in final]
```
