    while k < n:

    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s

This is a strong condition! It means that at a local minimum, the elements of a that pair up to form the peak must all be equal. If there are multiple active peaks, the elements must satisfy multiple such constraints simultaneously.

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines smoothed L-BFGS, primal-dual annealing, spectral basin hopping,
and peak-chasing to escape local minima and find near-optimal sequences.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smooth_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smooth_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_remaining=30.0, beta_scale=1.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005) * beta_scale

    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625,
        0.03, 0.015, 0.0075, 0.004,
        0.002, 0.001, 0.0005, 0.0002, 0.0001,
        0.00005, 0.00002, 0.00001
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - min_remaining:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        if beta < base_beta * 0.01:
            est_time = 0.004 * max(1.0, n / 2048.0)
            max_iter = min(8000, max(50, int(remaining / (est_time * 15))))
        else:
            est_time = 0.003 * max(1.0, n / 2048.0)
            max_iter = min(5000, max(30, int(remaining / (est_time * 12))))

        def obj_fn(z_):
            return _smooth_obj(z_, beta, n)

        def grad_fn(z_):
            return _smooth_grad(z_, beta, n)

        try:
            result = minimize(
                obj_fn, z, jac=grad_fn, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-17,
                         'gtol': 1e-14, 'maxcor': 100, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _primal_dual(a_init, deadline, min_remaining=60.0):
    n = len(a_init)
    m = 2 * n - 1
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    w = np.ones(m) / m
    lr_a = 0.005
    inner_steps = 3
    max_outer = 50000

    for outer in range(max_outer):
        if time.monotonic() > deadline - min_remaining:
            break

        for _ in range(inner_steps):
            a_rev = a[::-1]
            Qa = _full_convolve(w, a_rev)[n - 1:2 * n - 1]
            grad_a = 2.0 * Qa
            grad_a = grad_a - np.mean(grad_a)
            a_try = a - lr_a * grad_a
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s > 1e-15:
                a_try = a_try / s
            a = a_try

        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        c_norm = c / (c_max + 1e-15)
        sharpen = min(1.0, 0.1 + 0.9 * outer / max_outer)
        log_w = sharpen * (c_norm - 1.0)
        w_new = np.exp(np.clip(log_w, -40, 40))
        wn = w_new.sum()
        if wn > 1e-30:
            w_new = w_new / wn
        w = 0.6 * w + 0.4 * w_new

        if outer % 500 == 0:
            lr_a = max(0.001, lr_a * 0.99)

    return best_a, best_sc


def _spectral_perturb(a, rng, epsilon=0.08):
    n = len(a)
    N = _next_pow2(n)
    A = np.fft.rfft(a, N)
    magnitude = np.abs(A)
    phase = np.angle(A)
    noise = rng.standard_normal(len(magnitude))
    new_magnitude = magnitude * (1.0 + epsilon * np.abs(noise))
    A_new = new_magnitude * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _peak_chase(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    alpha = 0.015
    no_improve = 0

    while time.monotonic() < deadline - min_remaining:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Top K peaks
        K = min(5, 2 * n - 1)
        top_k = np.argsort(c)[::-1][:K]
        w_S = np.zeros(2 * n - 1)
        w_S[top_k] = 1.0 / K

        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        step = alpha / (gnorm + 1e-15)
        improved = False
        for _ in range(40):
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                step *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            step *= 0.5

        if not improved:
            no_improve += 1
            alpha *= 0.9
            if no_improve > 300:
                break
        else:
            alpha = min(0.04, alpha * 1.01)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.3) ** 2) + 0.6 * np.exp(-15.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.sin(2.0 * np.pi * 5.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 2.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
