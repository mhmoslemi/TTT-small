Active-set QP optimizer with FFT-accelerated CG inner solver.
Exploits sequence symmetry for improved conditioning.
Multi-phase: L-BFGS init -> Active-set QP -> Symmetric QP -> Multi-n -> Polish.

import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _hvp_fft(w, v, n):
    """Compute H * v where H = 2 * sum_k w_k Q_k.
    w has length 2n-1, v has length n.
    Returns vector of length n.
    """
    v_rev = v[::-1]
    conv_len = len(w) + n - 1
    N = _next_pow2(conv_len)
    Fw = np.fft.rfft(w, N)
    Fvr = np.fft.rfft(v_rev, N)
    conv = np.fft.irfft(Fw * Fvr, N)[:conv_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _hvp_sym(w, v_sym, half_n):
    """Compute H * v for symmetric case.
    w has length 2n-1 (full), v_sym has length half_n (symmetric part).
    Returns full H*v of length n, but we only need the first half_n components.
    """
    n = 2 * half_n
    # Reconstruct full v from symmetric part
    v_full = np.zeros(n)
    v_full[:half_n] = v_sym
    v_full[half_n:] = v_sym::-1
    
    # Compute full H*v
    Hv = _hvp_fft(w, v_full, n)
    
    # Return only the first half (by symmetry, the second half is determined)
    # But we need the full gradient for the CG to work correctly on the symmetric subspace
    return Hv, v_full


def _cg_solve_hvp(w, free_mask, b_vec, n, max_iter=500, tol=1e-10):
    """Solve H_FF x = b using CG, where H_FF is the submatrix of H on the free set.
    H is defined by the HVP _hvp_fft(w, v, n).
    free_mask is a boolean array of length n.
    b_vec is a vector of length n (only free_mask components matter).
    Returns x of length n (zero on inactive set).
    """
    x = np.zeros(n)
    # r = b - H(x) (masked to free set)
    Hx = _hvp_fft(w, x, n)
    r = np.where(free_mask, b_vec - Hx, 0.0)
    p = r.copy()
    rs = float(np.dot(r, r))
    
    for it in range(max_iter):
        if np.sqrt(rs) < tol * (np.linalg.norm(b_vec[free_mask]) + 1e-30):
            break
        Hp = _hvp_fft(w, p, n)
        # Project Hp to free set
        Hp_free = np.where(free_mask, Hp, 0.0)
        pAp = float(np.dot(p, Hp_free))
        if abs(pAp) < 1e-30:
            break
        alpha = rs / pAp
        x += alpha * p
        r -= alpha * Hp_free
        # Ensure r is zero on inactive set
        r[~free_mask] = 0.0
        rs_new = float(np.dot(r, r))
        if rs < 1e-30:
            break
        beta = rs_new / rs
        p = r + beta * p
        p[~free_mask] = 0.0
        rs = rs_new
    
    return x
