"""
MW-based LP iterative optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.

Key innovation: Uses multiplicative weights (MW) to solve the LP subproblem at each
outer iteration, providing a fundamentally different search direction from gradient
methods. The LP direction spreads pressure across all lags rather than concentrating
on the active set. Combined with alternating optimization, L-BFGS polishing, and ILS.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    return 1 << (n - 1).bit_length()


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "sin":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _mw_lp_direction(a, T=500, eta=0.5, init='same'):
    """
    Run multiplicative weights LP to find direction b that maximizes sum(b)
    subject to conv(a, b) <= M componentwise, b >= 0.
    
    Returns b normalized to sum=1.
    """
    n = len(a)
    M = float(_fft_convolve(a, a).max())
    if M < 1e-15:
        return a.copy()
    
    if init == 'same':
        q = a.copy()
    elif init == 'uniform':
        q = np.ones(n) / n
    else:
        q = _normalize(a.copy())
    
    a_rev = a[::-1]
    
    for t in range(T):
        # Compute c = conv(a, q)
        c = _fft_convolve(a, q)  # length 2n-1
        c_pos = np.maximum(c, 0.0)
        c_sum = c_pos.sum()
        if c_sum < 1e-15:
            break
        
        # p = normalized "tightness" distribution over lags
        p = c_pos / c_sum  # length 2n-1
        
        # Compute pressure = A^T p: pressure[j] = sum_k p[k] * a[k-j] for valid k-j
        # = conv(p, a_rev)[n-1 : 2n-1]
        prod = _fft_convolve(p, a_rev)  # length (2n-1) + n - 1 = 3n-2
        pressure = prod[n - 1:2 * n - 1]  # length n
        pressure = np.maximum(pressure, 0.0)
        
        # MW update: decrease q_j where pressure is high
        p_mean = pressure.mean()
        if p_mean < 1e-15:
            break
        # Normalized pressure: 1 means "average pressure"
        norm_pressure = pressure / p_mean
        # Update
        log_decrease = eta * (norm_pressure - 1.0)
        log_decrease = np.clip(log_decrease, -50, 50)
        q = q * np.exp(log_decrease)
        q = np.maximum(q, 0.0)
        s = q.sum()
        if s < 1e-15:
            q = np.ones(n) / n
        else:
            q = q / s
    
    return q


def _mw_alternate(a_fixed, T=300, eta=0.5):
    """
    LW direction: find b >= 0, sum(b)=1 minimizing max(conv(a_fixed, b)).
    This is equivalent to the LP: max V s.t. V*q = b, conv(a_fixed, b)_k <= M, b_k >= 0.
    We use MW to find q that minimizes max_k conv(a_fixed, q)_k.
    """
    n = len(a_fixed)
    M_ref = float(_fft_convolve(a_fixed, a_fixed).max())
    if M_ref < 1e-15:
        return a_fixed.copy()
    
    q = _normalize(a_fixed.copy())
    a_rev = a_fixed[::-1]
    
    for t in range(T):
        c = _fft_convolve(a_fixed, q)
        c_max = float(c.max())
        c_pos = np.maximum(c, 0.0)
        c_sum = c_pos.sum()
        if c_sum < 1e-15:
            break
        p = c_pos / c_sum
        
        prod = _fft_convolve(p, a_rev)
        pressure = prod[n - 1:2 * n - 1]
        pressure = np.maximum(pressure, 1e-30)
        
        # Smoothed log update for better behavior
        log_p = np.log(pressure)
        log_p -= log_p.mean()
        log_p = np.clip(log_p, -10, 10)
        q = q * np.exp(-eta * 0.1 * log_p)
        q = np.maximum(q, 0.0)
        s = q.sum()
        if s < 1e-15:
            q = np.ones(n) / n
        else:
            q = q / s
    
    return q


def _active_set_gradient(a, c, tol_frac=0.005):
    """Compute weighted active-set gradient."""
    n = len(a)
    c_max = float(c.max())
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    n_active = int(active.sum())
    if n_active == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
    
    eps = 1e-12 * c_max
    w = np.zeros_like(c)
    w[active] = 1.0 / (c_max - c[active] + eps)
    w = w / w.sum()
    
    a_rev = a[::-1]
    wc = _fft_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad = 2.0 * h
    grad = grad - np.mean(grad)
    return grad


def _grad_polish(a_init, deadline, tol_start=0.01, tol_end=0.0005,
                 lr_init=0.05, momentum=0.85):
    """Gradient polishing with heavy-ball momentum."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    v = np.zeros(n)
    tol_frac = tol_start
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() < deadline:
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        grad = _active_set_gradient(a, c, tol_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        v = momentum * v - lr * grad
        alpha = 1.0
        improved = False
        for _ in range(40):
            a_try = _normalize(a + alpha * v)
            sc_try = 2.0 * n * float(_fft_convolve(a_try, a_try).max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            v = np.zeros(n)
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_end, tol_frac * 0.6)
                lr *= 0.7
            if no_improve >= 100:
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(tol_end, tol_frac * 0.3)
                lr = lr_init * 0.4
                no_improve = 0
        else:
            if tol_frac > tol_end:
                tol_frac = max(tol_end, tol_frac * 0.98)
            lr = min(lr_init, lr * 1.005)
    
    return best_a, _score(best_a)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _fft_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS on log-softmax parameterization with adaptive beta annealing."""
    n = len(a_init)
    c_cur = _fft_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004,
                                  0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 15))))
        
        try:
            result = minimize(
                lambda z: _smoobj(z, beta, n), z,
                jac=lambda z: _smograd(z, beta, n),
                method='L-BFGS-B',

            a_new = _softmax(result.x)

            if sc_new < current_sc - 1e-15:

                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _mw_alternating(a_init, b_init, deadline, T_mw=150, eta=0.4, max_iters=100):
    """
    Alternating MW: fix a, optimize b; fix b, optimize a; repeat.
    Converges to a fixed point where a = b is LP-self-consistent.
    """
    n = len(a_init)
    a = _normalize(a_init.copy())
    b = _normalize(b_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() + 3 > deadline:
            break
        
        # Fix a, find optimal b
        b_new = _mw_alternate(a, T=T_mw, eta=eta)
        b = b_new
        
        # Fix b, find optimal a
        a_new = _mw_alternate(b, T=T_mw, eta=eta)
        a = a_new
        
        # Check score
        sc = _score(a)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc


def _ils_mw(a_init, deadline, rng, min_time_remaining=60.0):
    """ILS with MW-based local optimization."""
    n = len(a_init)
    current_a = _normalize(a_init.copy())
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc
    
    strategies = [
        ('global', 0.01),
        ('global', 0.03),
        ('block', 0.05),
        ('sparse', 0.08),
        ('global', 0.005),
        ('block', 0.03),
    ]
    
    attempt = 0
    while time.monotonic() + min_time_remaining < deadline:
        kind, sigma = strategies[attempt % len(strategies)]
        
        if kind == 'global':
            z = np.log(np.maximum(current_a, 1e-30))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_perturbed = _softmax(z)
        elif kind == 'block':
            block_size = max(50, n // 6)
            start = int(rng.integers(0, n - block_size))
            a_perturbed = current_a.copy()
            z = np.log(np.maximum(a_perturbed[start:start + block_size], 1e-30))
            z = z + sigma * rng.standard_normal(block_size)
            a_perturbed[start:start + block_size] = np.exp(z - z.max())
            a_perturbed = _normalize(a_perturbed)
        elif kind == 'sparse':
            n_perturb = max(20, n // 10)
            idx = rng.choice(n, size=n_perturb, replace=False)
            a_perturbed = current_a.copy()
            factors = np.exp(sigma * rng.standard_normal(n_perturb))
            a_perturbed[idx] *= factors
            a_perturbed = _normalize(a_perturbed)
        else:
            a_perturbed = current_a.copy()
        
        # Quick MW optimization
        opt_deadline = min(deadline - min_time_remaining, time.monotonic() + 10)
        a_opt, sc_opt = _mw_outer_loop(a_perturbed, opt_deadline,
                                         T_mw=50, eta=0.5, blend_rate=0.1,
                                         n_restarts=1)
        
        if sc_opt < best_sc - 1e-12:
            best_sc = sc_opt
            best_a = a_opt.copy()
            current_a = a_opt.copy()
        elif sc_opt < current_sc - 1e-12:
            current_a = a_opt.copy()
        
        attempt += 1
    
    return best_a, best_sc

    best_seq = None

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 24576
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")
    
    n_main = n_parent if n_parent > 0 else 24576
    
    # === Phase 1: MW outer loop on parent (350s) ===
    print(f"\n[P1] MW outer loop (t={time.monotonic() - t_start:.1f}s)")
    p1_deadline = min(t_start + 350, deadline - 400)
    if time.monotonic() < p1_deadline:
        a_p1, sc_p1 = _mw_outer_loop(best_seq, p1_deadline,
                                      T_mw=300, eta=0.5, blend_rate=0.08,
                                      n_restarts=3)
        if sc_p1 < best_score:
            best_seq = a_p1.copy()
            best_score = sc_p1
            print(f"  MW P1: {best_score:.10f}")
    
    # === Phase 2: Alternating MW (250s) ===
    print(f"\n[P2] Alternating MW (t={time.monotonic() - t_start:.1f}s)")
    p2_deadline = min(t_start + 600, deadline - 350)
    if time.monotonic() < p2_deadline:
        a_p2, sc_p2 = _mw_alternating(best_seq, best_seq, p2_deadline,
                                       T_mw=200, eta=0.4, max_iters=80)
        if sc_p2 < best_score:
            best_seq = a_p2.copy()
            best_score = sc_p2
            print(f"  Alt MW: {best_score:.10f}")
    
    # === Phase 3: MW with different parameters (200s) ===
    print(f"\n[P3] MW variant (t={time.monotonic() - t_start:.1f}s)")
    p3_deadline = min(t_start + 800, deadline - 300)
    if time.monotonic() < p3_deadline:
        a_p3, sc_p3 = _mw_outer_loop(best_seq, p3_deadline,
                                      T_mw=500, eta=0.3, blend_rate=0.05,
                                      n_restarts=2)
        if sc_p3 < best_score:
            best_seq = a_p3.copy()
            best_score = sc_p3
            print(f"  MW P3: {best_score:.10f}")
    
    # === Phase 4: L-BFGS polish (250s) ===
    print(f"\n[P4] L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    p4_deadline = min(t_start + 1050, deadline - 250)
    if time.monotonic() < p4_deadline:
        a_p4 = _run_lbfgs(best_seq, p4_deadline, min_time_remaining=40)
        sc_p4 = _score(a_p4)
        if sc_p4 < best_score:
            best_seq = a_p4.copy()
            best_score = sc_p4
            print(f"  L-BFGS: {best_score:.10f}")
    
    # === Phase 5: Gradient polish (150s) ===
    print(f"\n[P5] Gradient polish (t={time.monotonic() - t_start:.1f}s)")
    p5_deadline = min(t_start + 1200, deadline - 200)
    if time.monotonic() < p5_deadline:
        a_p5, sc_p5 = _grad_polish(best_seq, p5_deadline,
                                    tol_start=0.005, tol_end=0.0001,
                                    lr_init=0.03, momentum=0.85)
        if sc_p5 < best_score:
            best_seq = a_p5.copy()
            best_score = sc_p5
            print(f"  Grad: {best_score:.10f}")
    
    # === Phase 6: ILS with MW (250s) ===
    print(f"\n[P6] ILS (t={time.monotonic() - t_start:.1f}s)")
    p6_deadline = min(t_start + 1450, deadline - 150)
    if time.monotonic() < p6_deadline:
        a_p6, sc_p6 = _ils_mw(best_seq, p6_deadline, rng, min_time_remaining=40)
        if sc_p6 < best_score:
            best_seq = a_p6.copy()
            best_score = sc_p6
            print(f"  ILS: {best_score:.10f}")
    
    # === Phase 7: Multi-n exploration (150s) ===
    print(f"\n[P7] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    p7_deadline = min(t_start + 1600, deadline - 100)
    n_tries = []
    for nn in [n_main * 2, n_main // 2, n_main + 8192, n_main - 4096, 32768, 49152]:
        if nn >= 1024 and nn != n_main and nn not in n_tries:
            n_tries.append(nn)
    
    for n_try in n_tries:
        if time.monotonic() > p7_deadline - 30:
            break
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        a_opt, sc_opt = _mw_outer_loop(a_init, p7_deadline,
                                        T_mw=100, eta=0.5, blend_rate=0.1,
                                        n_restarts=1)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  Multi-n ({n_try}): {best_score:.10f}")
    
    # === Phase 8: Final ultra-fine polish (remaining) ===
    print(f"\n[P8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        # Quick L-BFGS
        rem = deadline - 10
        n = len(best_seq)
        c_best = _fft_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        beta_fin = mb * 0.0005
        
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        max_iter = min(3000, max(30, int((rem - time.monotonic()) / 0.02)))
        try:
            result = minimize(
                lambda zz: _smoobj(zz, beta_fin, n), z,
                jac=lambda zz: _smograd(zz, beta_fin, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 80}
            )
            a_fin = _softmax(result.x)
            sc_fin = _score(a_fin)
            if sc_fin < best_score:
                best_seq = a_fin.copy()
                best_score = sc_fin
                print(f"  Final L-BFGS: {best_score:.10f}")
        except Exception:
            pass
        
        # Tiny gradient steps
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _grad_polish(best_seq, deadline - 2,
                                            tol_start=0.001, tol_end=0.00001,
                                            lr_init=0.005, momentum=0.8)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  Final grad: {best_score:.10f}")
    
    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
