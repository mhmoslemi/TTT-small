                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 60, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            print(f"    beta={beta:.2e}, score={best_sc:.10f}, iter={result.nit}")
        except Exception as e:
            print(f"    L-BFGS error at beta={beta:.2e}: {e}")

    return best_a, best_sc


def _run_active_set_qp(a_init, deadline, min_time_remaining, max_rounds=200):
    """Active-set QP using projected gradient with correct HVP."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    tol_frac = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002,
                0.0001, 0.00005, 0.00002, 0.00001, 0.000005,
                0.000002, 0.000001, 0.0000005, 0.0000002, 0.0000001,
                0.00000005, 0.00000002, 0.00000001, 0.000000005]
    tol_idx = 0
    rounds_per_tol = 4

    for rnd in range(max_rounds):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _autocorr(a)
        c_max = float(c.max())
        tol = tol_frac[min(tol_idx, len(tol_frac) - 1)]
        active = c >= c_max * (1.0 - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        n_active = int(active.sum())

        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / n_active

        # Inner: projected gradient on simplex for min a^T H a
        # with H = 2 * sum_k p_k Q_k, using HVP
        lr = 0.01
        momentum = np.zeros(n)
        beta_m = 0.8

        for inner in range(30):
            if time.monotonic() + min_time_remaining > deadline:
                break
            grad = _hvp_correct(p, a, n)
            # Project gradient onto tangent space of simplex
            grad = grad - np.mean(grad)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-16:
                break
            momentum = beta_m * momentum + (1 - beta_m) * grad
            lr = 0.05 / (np.linalg.norm(momentum) + 1e-10)
            a_new = _proj_simplex(a - lr * momentum)
            a = a_new

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            if rnd % 10 == 0:
                print(f"    AS-QP round {rnd}: score={best_sc:.10f}, active={n_active}")
        elif rnd > 10:
            a = best_a.copy()

        if (rnd + 1) % rounds_per_tol == 0:
            tol_idx += 1

    return best_a, best_sc


def _run_peak_shave(a_init, deadline, min_time_remaining):
    """Dedicated peak shaving: focus exclusively on the top peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(500):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = _autocorr(a)
        c_max = float(c.max())

        # Find peaks above 99.9th percentile
        thresh = np.percentile(c, 99.0)
        peaks = c >= thresh
        if peaks.sum() == 0:
            peaks[int(np.argmax(c))] = True

        p = np.zeros(2 * n - 1)
        p[peaks] = 1.0 / peaks.sum()

        # Gradient descent with line search
        grad = _hvp_correct(p, a, n)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        step = 0.1 / (gnorm + 1e-10)
        improved = False
        for ls in range(40):
            a_new = _proj_simplex(a - step * grad)
            c_new = _autocorr(a_new)
            if float(c_new.max()) < c_max * (1 - 1e-12):
                a = a_new
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                improved = True
                break
            step *= 0.5
            if step < 1e-15:
                break

        if not improved and it > 50:
            # Try small perturbation
            a_pert = a + 0.001 * np.random.randn(n)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            if _score(a_pert) < best_sc:
                a = a_pert
            else:
                a = best_a.copy()

    return best_a, best_sc


def _perturb_reopt(a_init, rng, n_perturbations, deadline, min_time_remaining):
    """Perturb the best solution and re-optimize from the perturbed point."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)

    for i in range(n_perturbations):
        if time.monotonic() + min_time_remaining > deadline:
            break
        sigma = 0.02 * (1 + i * 0.3)
        z = np.log(np.maximum(best_a, 1e-30))
        z = z + sigma * rng.standard_normal(n)
        z = z - z.mean()
        a_pert = _softmax(z)

        # Quick L-BFGS from perturbed point
        sub_deadline = min(deadline - min_time_remaining, time.monotonic() + 60)
        a_new, sc_new = _run_lbfgs_smoothed(
            a_pert, sub_deadline, 5.0,
            beta_start=0.1, beta_end=0.001, n_steps=5
        )
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            print(f"    Perturb {i}: improved to {best_sc:.10f}")
        else:
            # Try smaller perturbation
            sigma_small = sigma * 0.3
            z2 = np.log(np.maximum(best_a, 1e-30))
            z2 = z2 + sigma_small * rng.standard_normal(n)
            z2 = z2 - z2.mean()
            a_pert2 = _softmax(z2)
            a_new2, sc_new2 = _run_lbfgs_smoothed(
                a_pert2, sub_deadline, 5.0,
                beta_start=0.05, beta_end=0.001, n_steps=4
            )
            if sc_new2 < best_sc:
                best_sc = sc_new2
                best_a = a_new2.copy()
                print(f"    Perturb {i} (small): improved to {best_sc:.10f}")

    return best_a, best_sc


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * (t - 0.3))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.4) ** 2) + 0.05
        return a / a.sum()
    elif kind == "asym2":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.6) ** 2) + 0.05
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS with log-sum-exp smoothing,
    active-set QP with correct HVP, peak shaving, perturbation-based
    diversification, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 32768
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    n = len(best_seq)

    # Compute initial beta from autocorrelation spread
    c_init = _autocorr(best_seq)
    c_sorted = np.sort(c_init)[::-1]
    max_c = float(c_sorted[0])
    gap = max_c - c_sorted[min(20, len(c_sorted) - 1)]
    beta_start = max(gap * 0.5, max_c * 0.01)
    print(f"Initial beta_start={beta_start:.2e}, gap={gap:.2e}")

    # === Phase 1: L-BFGS with beta continuation (400s) ===
    print(f"\n[Phase 1] L-BFGS beta continuation (t={time.monotonic() - t_start:.1f}s)")
    phase1_deadline = t_start + 400
    a1, sc1 = _run_lbfgs_smoothed(
        best_seq, phase1_deadline, 30.0,
        beta_start=beta_start, beta_end=max_c * 1e-5, n_steps=20
    )
    update_best(a1, sc1, "lbfgs_p1")

    # === Phase 2: Active-set QP (350s) ===
    print(f"\n[Phase 2] Active-set QP (t={time.monotonic() - t_start:.1f}s)")
    phase2_deadline = min(deadline - 1050, phase1_deadline + 350)
    a2, sc2 = _run_active_set_qp(best_seq, phase2_deadline, 30.0, max_rounds=150)
    update_best(a2, sc2, "as_qp_p2")

    # === Phase 3: Peak shaving (250s) ===
    print(f"\n[Phase 3] Peak shaving (t={time.monotonic() - t_start:.1f}s)")
    phase3_deadline = min(deadline - 800, phase2_deadline + 250)
    a3, sc3 = _run_peak_shave(best_seq, phase3_deadline, 20.0)
    update_best(a3, sc3, "peak_shave_p3")

    # === Phase 4: Perturbation + re-optimization (300s) ===
    print(f"\n[Phase 4] Perturbation exploration (t={time.monotonic() - t_start:.1f}s)")
    phase4_deadline = min(deadline - 500, phase3_deadline + 300)
    a4, sc4 = _perturb_reopt(best_seq, rng, 8, phase4_deadline, 20.0)
    update_best(a4, sc4, "perturb_p4")

    # === Phase 5: Second active-set QP round (200s) ===
    print(f"\n[Phase 5] AS-QP Round 2 (t={time.monotonic() - t_start:.1f}s)")
    phase5_deadline = min(deadline - 300, phase4_deadline + 200)
    a5, sc5 = _run_active_set_qp(best_seq, phase5_deadline, 20.0, max_rounds=100)
    update_best(a5, sc5, "as_qp_p5")

    # === Phase 6: Multi-n exploration (250s) ===
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    phase6_deadline = min(deadline - 100, phase5_deadline + 250)
    n_tries = [49152, 24576, 65536, 16384]
    for n_try in n_tries:
        if time.monotonic() > phase6_deadline - 40:
            break
        if n_try == n:
            continue
        a_init_n = _resample(best_seq, n_try)
        a_n, sc_n = _run_lbfgs_smoothed(
            a_init_n, phase6_deadline, 15.0,
            beta_start=max_c * 0.05, beta_end=max_c * 1e-5, n_steps=8
        )
        sc_n = _score(a_n)
        update_best(a_n, sc_n, f"n_{n_try}")
        if sc_n < best_score + 0.002:
            a_nq, sc_nq = _run_active_set_qp(a_n, phase6_deadline, 10.0, max_rounds=40)
            update_best(a_nq, sc_nq, f"n_{n_try}_qp")

    # === Phase 7: Final ultra-tight polish (100s) ===
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 15:
        # Tight AS-QP
        a_fin, sc_fin = _run_active_set_qp(best_seq, deadline - 15, 5.0, max_rounds=60)
        update_best(a_fin, sc_fin, "final_qp")

        # Peak shave
        if time.monotonic() < deadline - 10:
            a_ps, sc_ps = _run_peak_shave(best_seq, deadline - 10, 5.0)
            update_best(a_ps, sc_ps, "final_peak")

        # Final L-BFGS with very tight beta
        if time.monotonic() < deadline - 8:
            c_b = _autocorr(best_seq)
            mb = float(c_b.max())
            a_final, sc_final = _run_lbfgs_smoothed(
                best_seq, deadline - 8, 3.0,
                beta_start=mb * 1e-4, beta_end=mb * 1e-8, n_steps=5
            )
            update_best(a_final, sc_final, "final_lbfgs")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        # Verify with evaluate_sequence
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"Evaluate check: {ev_score:.10f}")
        except Exception:
            pass
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
