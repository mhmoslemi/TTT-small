"""
Improved multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:
- More aggressive L-BFGS beta schedule with emphasis on critical transition range
- Adaptive step size for active set refinement with backtracking line search
- Fourier-domain preconditioned gradient steps for faster convergence
- Structured perturbation restarts using low-frequency perturbations
- Multi-scale initialization for diversity
- Better time management focused on main optimization

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 2
    while k < n:
        k <<= 1
    return k


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_arr(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_soft_structured(a, beta, n):
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _smoobj(z, beta, n):
    a = _softmax_arr(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax_arr(z)
    grad_a = _grad_soft_structured(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)


class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0, verbose=True, tag=""):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < c0_max * 0.002:
        gap = c0_max * 0.002

    # Refined beta schedule: more levels in the critical range
    multipliers = np.array([
        1.0, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01,
        0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003,
        0.00015, 0.00008, 0.00004, 0.00002, 0.00001,
        0.000005, 0.000002, 0.000001
    ])
    betas = gap * multipliers

    t_phase_start = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        size_factor = max(1.0, n / 8192.0)
        est_iter_time = 0.0015 * size_factor
        # More iterations for critical betas (the transition range)
        if 0.001 <= multipliers[i] <= 0.05:
            max_iter = min(12000, max(200, int(remaining / (est_iter_time * 15))))
        else:
            max_iter = min(6000, max(100, int(remaining / (est_iter_time * 25))))

        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 200,
                    'maxls': 80,
                }
            )
            a_new = _softmax_arr(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    elapsed = time.monotonic() - t_global_start
                    print(f"    [{tag}] beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}, t={elapsed:.0f}s")
            elif verbose and i < 8:
                print(f"    [{tag}] beta={beta:.2e}: no improvement (best={best_sc:.10f})")
        except Exception as e:
            if verbose:
                print(f"    [{tag}] beta={beta:.2e}: error {e}")

    return best_a, best_sc


def _active_set_step(a, tol_frac):
    """One active set step with adaptive step size via backtracking."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    current_sc = 2.0 * n * c_max

    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    n_active = int(active.sum())
    if n_active == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        n_active = 1

    a_rev = a[::-1]
    w = np.zeros(len(c))
    w[active] = 1.0 / n_active
    wc = _full_convolve(w, a_rev)
    grad_avg = 2.0 * wc[n - 1:2 * n - 1]

    support = a > 1e-10 * a.max()
    n_support = int(support.sum())
    if n_support < 2:
        return a, current_sc, n_active

    lam = float(np.mean(grad_avg[support]))
    residual = grad_avg[support] - lam

    # Backtracking line search
    best_a = a
    best_sc = current_sc
    for eta in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
        a_try = a.copy()
        a_try[support] -= eta * residual
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try.copy()
        else:
            break

    return best_a, best_sc, n_active


def _active_set_refine(a_init, deadline, min_time_remaining=30.0, verbose=False, tag=""):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    no_improve = 0
    max_no_improve = 200
    tol_frac = 0.003
    min_tol = 0.00005
    initial_tol = tol_frac

    while time.monotonic() + min_time_remaining < deadline:
        a, sc, n_active = _active_set_step(a, tol_frac)

        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
            # Relax tolerance slightly when improving
            tol_frac = max(min_tol, min(initial_tol, tol_frac * 1.1))
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                tol_frac = max(min_tol, tol_frac * 0.7)
                if verbose and no_improve % 100 == 0:
                    print(f"    [{tag}] AS: no_improve={no_improve}, tol={tol_frac:.6f}, "
                          f"best={best_sc:.10f}")

        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.5)
            if tol_frac <= min_tol * 1.05:
                break
            no_improve = 0

    return best_a, best_sc


def _fourier_preconditioned_step(a, deadline, min_time_remaining=30.0, verbose=False, tag=""):
    """Take gradient steps with Fourier-domain preconditioning.
    
    The idea: compute the subgradient in Fourier domain, apply a spectral
    preconditioner (divide by |A_hat|^2 + eps), then transform back.
    This is similar to using the inverse Hessian as a preconditioner.
    """
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    N = _next_pow2(2 * n - 1)

    no_improve = 0
    max_no_improve = 500
    tol_frac = 0.002
    min_tol = 0.0001

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                tol_frac = max(min_tol, tol_frac * 0.8)
                if tol_frac <= min_tol * 1.05:
                    break
                no_improve = 0

        if no_improve > max_no_improve:
            break

        # Compute subgradient
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]
        grad_centered = grad - np.mean(grad)

        # Fourier preconditioner
        A_hat = np.fft.rfft(a, N)
        A_hat_abs_sq = np.abs(A_hat) ** 2
        eps = 1e-10 * (A_hat_abs_sq.max() + 1e-30)

        G_hat = np.fft.rfft(grad_centered, N)
        # Preconditioned gradient: G_hat / (|A_hat|^2 + eps)
        G_prec = G_hat / (A_hat_abs_sq + eps)
        # Zero out DC component (to maintain sum=1)
        G_prec[0] = 0
        g_prec = np.fft.irfft(G_prec, N)[:n]

        # Normalize step
        g_prec_norm = float(np.linalg.norm(g_prec))
        if g_prec_norm < 1e-20:
            break

        # Scale to get a reasonable step
        step_size = 0.01 * np.linalg.norm(grad_centered) / g_prec_norm
        step = min(step_size, 0.5)

        # Take step with backtracking
        a_try = a - step * g_prec
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            step *= 0.1
            a_try = a - step * g_prec
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                break
        a_try = a_try / s_try
        sc_try = _score(a_try)

        if sc_try < current_sc - 1e-14:
            a = a_try
        else:
            # Try smaller step
            found = False
            for trial in [0.5, 0.25, 0.1, 0.05]:
                a_try2 = a - step * trial * g_prec
                a_try2 = np.maximum(a_try2, 0.0)
                s_try2 = a_try2.sum()
                if s_try2 < 1e-10:
                    continue
                a_try2 = a_try2 / s_try2
                sc_try2 = _score(a_try2)
                if sc_try2 < current_sc - 1e-14:
                    a = a_try2
                    found = True
                    break
            if not found and no_improve > 100:
                tol_frac = max(min_tol, tol_frac * 0.85)

    return best_a, best_sc


def _perturbation_restart(a_init, deadline, min_time_remaining=30.0, verbose=False, tag=""):
    """Try structured perturbations to escape local minima."""
    n = len(a_init)
    a_best = a_init.copy()
    s = a_best.sum()
    if s < 1e-15:
        return a_best, np.inf
    a_best = a_best / s
    best_sc = _score(a_best)
    rng = np.random.default_rng(12345)

    n_perturb = 8
    time_per = max(5.0, (deadline - min_time_remaining - time.monotonic()) / max(1, n_perturb))

    for p in range(n_perturb):
        if time.monotonic() + min_time_remaining + time_per > deadline:
            break

        # Create structured perturbation: low-frequency sinusoids
        freqs = rng.integers(1, max(2, n // 100), size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        amp = 0.005 * (1 + 0.3 * p / n_perturb)

        t = np.arange(n, dtype=np.float64) / n
        pert = np.zeros(n)
        for f, ph in zip(freqs, phases):
            pert += np.sin(2 * np.pi * f * t + ph)
        pert = pert / (np.max(np.abs(pert)) + 1e-30)

        a_pert = a_best + amp * pert
        a_pert = np.maximum(a_pert, 0.0)
        s_pert = a_pert.sum()
        if s_pert < 1e-10:

        a_pert = a_pert / s_pert

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

        p2_end = min(deadline - 500, time.monotonic() + 300)
        a2, sc2 = _active_set_refine(best_seq, p2_end, min_time_remaining=60,
                                     verbose=True, tag="P2")
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(f"  P2 result: {best_score:.10f}")
        else:
            print(f"  P2: no improvement (best={best_score:.10f})")

    # === Phase 3: L-BFGS round 2 ===
    elapsed = time.monotonic() - t_global_start
    print(f"\n[P3] L-BFGS round 2 (t={elapsed:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p3_end = min(deadline - 350, time.monotonic() + 300)
        a3, sc3 = _run_lbfgs(best_seq, p3_end, min_time_remaining=100, verbose=True, tag="P3")
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3 result: {best_score:.10f}")

    # === Phase 4: Fourier-preconditioned steps ===
    elapsed = time.monotonic() - t_global_start
    print(f"\n[P4] Fourier-preconditioned (t={elapsed:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p4_end = min(deadline - 250, time.monotonic() + 250)
        a4, sc4 = _fourier_preconditioned_step(best_seq, p4_end, min_time_remaining=50,
                                               verbose=True, tag="P4")
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4 result: {best_score:.10f}")
        else:
            print(f"  P4: no improvement (best={best_score:.10f})")

    # === Phase 5: Active set round 2 ===
    elapsed = time.monotonic() - t_global_start
    print(f"\n[P5] Active set round 2 (t={elapsed:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p5_end = min(deadline - 200, time.monotonic() + 180)
        a5, sc5 = _active_set_refine(best_seq, p5_end, min_time_remaining=40,
                                     verbose=False, tag="P5")
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  P5 result: {best_score:.10f}")

    # === Phase 6: Perturbation restarts ===
    elapsed = time.monotonic() - t_global_start
    print(f"\n[P6] Perturbation restarts (t={elapsed:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p6_end = min(deadline - 120, time.monotonic() + 200)
        a6, sc6 = _perturbation_restart(best_seq, p6_end, min_time_remaining=50,
                                        verbose=True, tag="P6")
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  P6 result: {best_score:.10f}")
        else:
            print(f"  P6: no improvement (best={best_score:.10f})")

    # === Phase 7: Multi-n exploration (brief) ===
    elapsed = time.monotonic() - t_global_start
    print(f"\n[P7] Multi-n (t={elapsed:.0f}s)")
    n_options = [n_main // 2, n_main * 3 // 2, n_main + 2048]
    n_options = [x for x in n_options if x != n_main and x >= 1024 and x <= 32768]
    for n_try in n_options[:2]:
        if time.monotonic() + 60 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if sc_r < best_score - 1e-4:
            if time.monotonic() + 40 < deadline:
                p7_end = min(deadline - 40, time.monotonic() + 60)
                a7, sc7 = _run_lbfgs(a_r, p7_end, min_time_remaining=10,
                                      verbose=False, tag=f"n{n_try}")
                if sc7 < best_score:
                    best_score = sc7
                    best_seq = a7.copy()
                    print(f"  n={n_try}: {best_score:.10f}")
        elif sc_r < best_score:
            best_score = sc_r
            best_seq = a_r.copy()
            print(f"  n={n_try}: resample improvement to {best_score:.10f}")

    # === Phase 8: Final polish ===
    elapsed = time.monotonic() - t_global_start
    print(f"\n[P8] Final polish (t={elapsed:.0f}s)")
    if time.monotonic() + 15 < deadline:
        p8_end = min(deadline - 3, time.monotonic() + 80)
        a8, sc8 = _run_lbfgs(best_seq, p8_end, min_time_remaining=3, verbose=False, tag="P8")
        if sc8 < best_score:
            best_score = sc8
            best_seq = a8.copy()

        if time.monotonic() + 10 < deadline:
            p8b_end = min(deadline - 2, time.monotonic() + 30)
            a8b, sc8b = _active_set_refine(best_seq, p8b_end, min_time_remaining=2,
                                           verbose=False, tag="P8b")
            if sc8b < best_score:
                best_score = sc8b
                best_seq = a8b.copy()

    # Final
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
          f"time={time.monotonic()-t_global_start:.1f}s")
    return [float(x) for x in final]
