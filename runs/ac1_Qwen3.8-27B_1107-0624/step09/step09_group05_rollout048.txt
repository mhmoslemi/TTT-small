```python
Adam-based optimizer with exact subgradient for min 2n*max(conv(a,a))/sum(a)^2.
Uses diverse initializations, multi-scale length exploration, and adaptive 
learning rate scheduling to improve upon L-BFGS-based approaches.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_normalized(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    return _score_normalized(a_norm)

def _compute_subgradient(a, tol_frac=0.0):
    """Compute subgradient of 2n*max(conv(a,a)) w.r.t. a (with sum(a)=1).
    Returns (subgrad_vector, active_indices, max_value)."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if tol_frac > 0:
        active = c >= (c_max - tol_frac * c_max)
    else:
        active = c == c_max  # exact
    
    # Make sure at least one is active
    if not active.any():
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
    
    # Subgradient: 2 * conv(w_active, a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    w = np.zeros(len(c))
    w[active] = 1.0 / active.sum()
    wc = _full_convolve(w, a_rev)
    subgrad = 2.0 * wc[n-1 : 2*n-1]
    
    return subgrad, active, c_max

def _adam_step(a, subgrad, m, v, t, lr=0.01, beta1=0.9, beta2=0.999):
    """One Adam step with projection onto probability simplex."""
    m = beta1 * m + (1 - beta1) * subgrad
    v = beta2 * v + (1 - beta2) * subgrad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    
    # Center the gradient (remove mean to preserve sum)
    subgrad_centered = subgrad - subgrad.mean()
    m_hat_centered = m_hat - m_hat.mean()
    
    step = lr * m_hat_centered / (np.sqrt(v_hat) + 1e-8)
    a_new = a - step
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a, m, v
    a_new = a_new / s
    return a_new, m, v

def _adam_optimize(a_init, deadline, min_remaining=30, lr=0.01, tol_frac=0.001, verbose=True):
    """Adam optimization with exact subgradient."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_sc = _score_normalized(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    # Adaptive learning rate schedule
    lr_current = lr
    no_improve = 0
    max_no_improve = 200
    
    start_time = time.monotonic()
    
    while time.monotonic() + min_remaining < deadline:
        subgrad, active, c_max = _compute_subgradient(a, tol_frac)
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve == max_no_improve:
            # Reduce learning rate and tolerance
            lr_current *= 0.5
            tol_frac *= 0.5
            no_improve = 0
            if lr_current < 1e-8 or tol_frac < 1e-6:
                break
        
        t += 1
        a, m, v = _adam_step(a, subgrad, m, v, t, lr=lr_current)
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    Adam t={t}, sc={best_sc:.10f}, lr={lr_current:.6f}, t={elapsed:.0f}s")
    
    return best_a, best_sc
```

```python
"""
Multi-strategy optimizer combining Adam subgradient descent, L-BFGS refinement,
harmonic perturbation search, and diverse restarts to minimize
2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Key innovations: exact subgradient Adam with adaptive LR, harmonic mode search,
and multi-length exploration with time-aware scheduling.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_normalized(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score(a):
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return _score_normalized(a / s)


def _compute_subgradient(a, tol_frac=0.0):
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if tol_frac > 0:
        active = c >= (c_max - tol_frac * c_max)
    else:
        active = c >= c_max * (1.0 - 1e-15)
    if not active.any():
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
    a_rev = a[::-1]
    w = np.zeros(len(c))
    w[active] = 1.0 / active.sum()
    wc = _full_convolve(w, a_rev)
    subgrad = 2.0 * wc[n - 1:2 * n - 1]
    return subgrad, c_max


def _adam_optimize(a_init, deadline, min_remaining=20, lr=0.005, tol_frac=0.001, verbose=True):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score_normalized(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr_cur = lr
    no_improve = 0
    max_no_improve = 500
    start_time = time.monotonic()

    while time.monotonic() + min_remaining < deadline:
        subgrad, c_max = _compute_subgradient(a, tol_frac)
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve == max_no_improve:
            lr_cur *= 0.5
            tol_frac = max(tol_frac * 0.5, 1e-7)
            no_improve = 0
            if lr_cur < 1e-9:
                break

        t += 1
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1 - beta1) * subgrad
        v = beta2 * v + (1 - beta2) * (subgrad * subgrad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        sg_centered = subgrad - np.mean(subgrad)
        m_hat_c = m_hat - np.mean(m_hat)
        step = lr_cur * m_hat_c / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    Adam t={t}, best={best_sc:.10f}, lr={lr_cur:.6f}, elapsed={elapsed:.0f}s")

    return best_a, best_sc


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


class _SmoProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        return float(2.0 * self.n * _soft_max_value(c, self.beta))

    def grad(self, z):
        a = _softmax_fn(z)
        n = self.n
        c = _full_convolve(a, a)
        w = _soft_max_weights(c, self.beta)
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad_a = 2.0 * wc[n - 1:2 * n - 1]
        W = float(np.dot(grad_a, a))
        return a * (grad_a - W)


def _lbfgs_optimize(a_init, deadline, min_remaining=30, verbose=True):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_normalized(a)

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < c0_max * 0.002:
        gap = c0_max * 0.002

    multipliers = np.array([
        1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01,
        0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005
    ])
    betas = gap * multipliers
    start_time = time.monotonic()
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        max_iter = min(5000, max(200, int(remaining / 0.05)))

        prob = _SmoProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 50}
            )
            a_new = _softmax_fn(result.x)
            sc_new = _score_normalized(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"    L-BFGS beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, t={time.monotonic()-start_time:.0f}s")
        except Exception:
            pass

    return best_a, best_sc


def _harmonic_search(a_init, deadline, min_remaining=30, verbose=True):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_normalized(a)
    rng = np.random.default_rng(12345)

    n_tried = 0
    while time.monotonic() + min_remaining < deadline:
        n_tried += 1
        freq = rng.integers(1, max(2, n // 4))
        phase = rng.uniform(0, 2 * np.pi)
        # Try several amplitudes
        for eps in [0.01, 0.02, 0.05, 0.1, 0.2]:
            mod = np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)
            for sign in [1, -1]:
                a_try = a + sign * eps * mod
                a_try = np.maximum(a_try, 0.0)

                if s_try < 1e-10:

                a_try = a_try / s_try

                    a = a_try.copy()

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new

    # === Phase 5: L-BFGS again ===
    print(f"\n[P5] L-BFGS round 2 (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p5_deadline = min(deadline - 50, time.monotonic() + 200)
        a5, sc5 = _lbfgs_optimize(best_seq, p5_deadline, min_remaining=30, verbose=True)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  P5 result: {best_score:.10f}")

    # === Phase 6: Multi-length exploration ===
    print(f"\n[P6] Multi-length exploration (t={time.monotonic()-t_start:.0f}s)")
    n_options = [12288, 20480, 24576, 32768]
    n_options = [x for x in n_options if x != n_main and x <= 32768]
    for n_try in n_options:
        if time.monotonic() + 100 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score_normalized(a_r)
        print(f"  Resampled n={n_try}: score={sc_r:.10f}")
        if time.monotonic() + 80 > deadline:
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
            continue
        p6_deadline = min(deadline - 50, time.monotonic() + 80)
        a6, sc6 = _adam_optimize(a_r, p6_deadline, min_remaining=20,
                                 lr=0.003, tol_frac=0.001, verbose=False)
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  n={n_try} Adam: {best_score:.10f}")

    # === Phase 7: Diverse initializations + Adam ===
    print(f"\n[P7] Diverse starts (t={time.monotonic()-t_start:.0f}s)")
    n_div = len(best_seq)
    diverse_kinds = ["periodic", "bimodal", "gaussian_centered", "random"]
    for kind in diverse_kinds:
        if time.monotonic() + 60 > deadline:
            break
        a_d = _make_initialization(n_div, kind, rng)
        # Perturb from best to guide
        a_d = 0.5 * a_d + 0.5 * best_seq
        a_d = np.maximum(a_d, 0)
        a_d = a_d / a_d.sum()
        sc_d = _score_normalized(a_d)
        if sc_d < best_score + 0.3:
            p7_deadline = min(deadline - 30, time.monotonic() + 60)
            a7, sc7 = _adam_optimize(a_d, p7_deadline, min_remaining=10,
                                     lr=0.003, tol_frac=0.001, verbose=False)
            if sc7 < best_score:
                best_score = sc7
                best_seq = a7.copy()
                print(f"  {kind}: {best_score:.10f}")

    # === Phase 8: Final Adam polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 10 < deadline:
        p8_deadline = min(deadline - 2, time.monotonic() + 100)
        a8, sc8 = _adam_optimize(best_seq, p8_deadline, min_remaining=2,
                                 lr=0.0005, tol_frac=0.0001, verbose=False)
        if sc8 < best_score:
            best_score = sc8
            best_seq = a8.copy()
            print(f"  P8 result: {best_score:.10f}")

    # Final output
    final = np.clip(best_seq, 0.0, 1000.0)
    s_final = float(final.sum())
    if s_final < 0.01:
        final = np.ones(256) / 256
    sc_final = float(2.0 * len(final) * max(_full_convolve(final, final)) / (s_final ** 2))
    print(f"\n{'='*60}")
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    return [float(x) for x in final]
```
