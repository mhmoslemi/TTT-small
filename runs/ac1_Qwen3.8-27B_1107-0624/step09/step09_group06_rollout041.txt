At the optimum, the KKT conditions require that the weighted sum of active peak gradients is in the normal cone of the simplex at the optimal point. If we can find the right weights, we can directly compute the optimal direction.

The "peak merging" algorithm:
1. Identify the top 2 peaks (k1, k2) with values c1 ≈ c2
2. Find a direction d (in the tangent space of the simplex) such that:
   - d @ grad_{k1} = d @ grad_{k2} (equal change for both peaks)
   - This is a linear constraint on d
   - Also require d @ grad_k < 0 for other active peaks k
3. Take a step in direction d
4. Repeat

This is essentially solving a small LP at each step, but with a much smaller problem (just the top few peaks).

import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _conv_grad_single(a, k):
    n = len(a)
    grad = np.zeros(n)
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    if j_lo <= j_hi:
        idx = k - np.arange(j_lo, j_hi + 1)
        grad[j_lo:j_hi + 1] = a[idx]
    j_lo2 = max(0, k)
    j_hi2 = min(n - 1, k + n - 1)
    if j_lo2 <= j_hi2 and j_lo2 <= j_hi2:
        idx2 = np.arange(j_lo2, j_hi2 + 1) - k
        grad[j_lo2:j_hi2 + 1] += a[idx2]
    return grad


def _conv_grad_multi(a, active_indices):
    n = len(a)
    a_rev = a[::-1]
    grad = np.zeros(n)
    for k in active_indices:
        if 0 <= k < n:

            start = max(0, k - n + 1)
            end = min(n - 1, k)
            if start <= end:

        if k >= 0:
            start2 = k if k < n else n
            end2 = min(n - 1, k + n - 1)
            if k < n and start2 <= end2:

        # Alternative: use the reversed array
        # a[j-k] for j in [k, n-1] is a_rev[n-1-(j-k)] = a_rev[n-1-j+k]
    return grad

        if j_lo2 <= j_hi2:
            idx2 = np.arange(j_lo2, j_hi2 + 1) - k
            grad[j_lo2:j_hi2 + 1] += w * a[idx2]
    return grad


def _fista_subgrad(a_init, deadline, min_time_remaining=30.0, K=20, seed=0):
    """FISTA-accelerated proximal subgradient method."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    a_prev = a.copy()
    t = 1
    L = 4.0  # Lipschitz constant estimate
    no_improve = 0
    stall_count = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() + min_time_remaining < deadline:
        # FISTA momentum
        if t > 1:
            alpha = (t - 1) / (t + 2)
            y = a + alpha * (a - a_prev)
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys < 1e-15:
                y = a
            else:
                y = y / ys
        else:
            y = a.copy()

        # Compute subgradient at y
        c = _full_convolve(y, y)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            stall_count += 1
            if stall_count > 8:
                break
            # Restart with perturbation
            perturb = 0.005 * rng.standard_normal(n)
            a_pert = _project_simplex(a + perturb)
            a = a_pert
            a_prev = a.copy()
            t = 1
            L = 4.0
            no_improve = 0
            continue

        # Adaptive K
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
        if gap < c_max * 0.01 and K < 200:
            K = min(200, int(K * 1.5) + 1)

        # Select active peaks with weights proportional to value
        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]
        active_vals = c[active]
        # Softmax weights over active peaks
        w_shift = (active_vals - active_vals.max()) / (c_max * 0.01 + 1e-10)
        w_exp = np.exp(w_shift)
        w_norm = w_exp / w_exp.sum()

        # Compute weighted gradient
        grad = _weighted_conv_grad(y, -w_norm, active)
        # Project to tangent space
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-18:
            break

        # Step size for FISTA
        lr = 1.0 / (L + t * 0.001)

        # Proximal step (project onto simplex)
        a_new = _project_simplex(y - lr * grad_t)

        # Check for improvement, backtrack if needed
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new > current_sc * 1.01:
            # Backtrack
            lr *= 0.5
            a_new = _project_simplex(y - lr * grad_t)

        a_prev = a
        a = a_new
        t += 1

        # Increase L if we had to backtrack
        if lr < 1.0 / L:
            L = max(L * 1.1, 2.0 * t / (lr * t * t + 1))

        if t % 1000 == 0:
            print(f"    FISTA t={t}: best={best_sc:.10f}, cur={sc_new:.10f}, K={K}, L={L:.2f}")

    return best_a, best_sc


def _smoothed_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smoothed_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # Gradient: 4n * sum_k w[k] * grad_k
    # grad_k[j] = a[k-j] + a[j-k] (with bounds)
    # Vectorized: conv(w, a_reversed) gives sum_k w[k] * a[k-j]
    #             conv(w_reversed, a) gives sum_k w[k] * a[j-k]
    a_rev = a[::-1]
    w_conv_a = _full_convolve(w, a_rev)
    # w_conv_a[m] = sum_k w[k] * a_rev[m-k] = sum_k w[k] * a[n-1-(m-k)] = sum_k w[k] * a[n-1-m+k]
    # For the part we want: sum_k w[k] * a[k-j] for j in [0,n-1]
    # a[k-j] with k in [0, 2n-2], j in [0, n-1]
    # Let m = k-j, then k = m+j. w[k] = w[m+j].
    # sum_k w[k]*a[k-j] = sum_m w[m+j]*a[m] for m in [0,n-1] (with w[m+j]=0 if out of range)
    # This is the cross-correlation of w and a, which is conv(w, flip(a))[n-1:2n-1]
    # Actually: (w * a_rev)[n-1+j] = sum_k w[k] * a_rev[n-1+j-k] = sum_k w[k] * a[k-j]
    # Wait: a_rev[m] = a[n-1-m]. So (w*a_rev)[p] = sum_k w[k]*a_rev[p-k] = sum_k w[k]*a[n-1-p+k]
    # For p = n-1+j: sum_k w[k]*a[n-1-(n-1-j)+k] = sum_k w[k]*a[j+k]
    # Hmm, that gives a[j+k] not a[k-j]. Let me use a different approach.
    
    # Direct: sum_k w[k] * a[k-j] for valid k,j
    # = cross-correlation of w and a at lag j
    # In numpy: np.correlate(w, a, mode='full')[n-1:2n-1] or similar
    
    # Let me just use the FFT approach:
    # Part 1: sum_k w[k] * a[k-j] for j in [0, n-1]
    # This is (w correlated with a) at lag j, which equals (w * flip(a))[(n-1)+j]
    # Actually: (w * flip(a))[m] = sum_k w[k] * flip(a)[m-k] = sum_k w[k] * a[n-1-m+k]
    # For m = n-1-j: sum_k w[k] * a[n-1-(n-1-j)+k] = sum_k w[k] * a[j+k]
    # That's a[j+k], not a[k-j]. 
    
    # Let me think differently. I want h[j] = sum_k w[k] * a[k-j] for j = 0, ..., n-1
    # Let b = flip(a), so b[m] = a[n-1-m]
    # (w * b)[n-1-j] = sum_k w[k] * b[n-1-j-k] = sum_k w[k] * a[n-1-(n-1-j-k)] = sum_k w[k] * a[j+k]
    # Still getting a[j+k].
    
    # Alternative: Let c = flip(w), c[m] = w[n_w-1-m] where n_w = 2n-1
    # (c * a)[j] = sum_m c[m] * a[j-m] = sum_m w[n_w-1-m] * a[j-m]
    # Let k = n_w-1-m, m = n_w-1-k
    # = sum_k w[k] * a[j-(n_w-1-k)] = sum_k w[k] * a[j - 2n + 2 + k]
    # For j in [0, n-1]: a[j - 2n + 2 + k]... this is getting messy.
    
    # Simplest: just compute directly using the known formula
    # grad_k[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    # sum_k w[k] * grad_k[j] = sum_k w[k]*a[k-j] + sum_k w[k]*a[j-k]
    # First term: for fixed j, sum over k where 0<=k-j<n, i.e., j<=k<j+n
    #   = sum_{k=j}^{j+n-1} w[k]*a[k-j] = (w conv a)[j] in the "cross-correlation" sense
    #   This is np.correlate(w, a, 'full')[2n-2-j] ... let me just use FFT
    
    # FFT approach for sum_k w[k]*a[k-j]:
    # Extend a to length 2n-1 (pad with zeros), compute DFT
    # The result is the "anti-diagonal" of the outer product
    # Actually, let me just use the original approach from the previous code:
    
    # From previous code:
    # wc = _full_convolve(w, a_rev) gives wc[m] = sum_k w[k]*a_rev[m-k] = sum_k w[k]*a[n-1-m+k]
    # h = wc[n-1:2n-1] gives h[j] = wc[n-1+j] = sum_k w[k]*a[n-1-(n-1-j)+k] = sum_k w[k]*a[j+k]
    # That's sum_k w[k]*a[j+k], which is NOT what we want.
    
    # We want sum_k w[k]*a[k-j]. Let me try:
    # _full_convolve(w_rev, a) where w_rev = w[::-1]
    # = sum_m w_rev[m]*a[p-m] = sum_m w[n_w-1-m]*a[p-m]
    # Let k = n_w-1-m, so m = n_w-1-k
    # = sum_k w[k]*a[p-(n_w-1-k)] = sum_k w[k]*a[p-2n+2+k]
    # For p = 2n-2-j: sum_k w[k]*a[2n-2-j-2n+2+k] = sum_k w[k]*a[k-j] ✓
    
    # So: sum_k w[k]*a[k-j] = (_full_convolve(w[::-1], a))[2n-2-j] for j in [0, n-1]
    
    n_w = len(w)  # 2n-1
    w_rev = w[::-1]
    wc1 = _full_convolve(w_rev, a)
    # wc1[2n-2-j] = sum_k w[k]*a[k-j]
    part1 = wc1[2*n-3 - np.arange(n)]  # wc1[2n-3], wc1[2n-4], ..., wc1[n-1]
    # Wait, for j=0: wc1[2n-2-0] = wc1[2n-2]. But wc1 has length n_w + n - 1 = (2n-1)+n-1 = 3n-2
    # Index 2n-2 is valid.
    # For j=n-1: wc1[2n-2-(n-1)] = wc1[n-1]. Valid.
    
    # Part 2: sum_k w[k]*a[j-k] for j in [0, n-1]
    # = (w conv a)[j] in the standard convolution sense... 
    # Actually sum_k w[k]*a[j-k] is the cross-correlation, which equals:
    # (w * flip(a))[n_w-1+(j)] ... let me just use:
    # sum_k w[k]*a[j-k] for valid k (0<=k<n_w, 0<=j-k<n)
    # = sum_{k=max(0,j-n+1)}^{min(j, n_w-1)} w[k]*a[j-k]
    # This is (w conv a)[j] if we think of w as a filter applied to a:
    # (w*a)[j] = sum_k w[k]*a[j-k] ✓
    # So part2[j] = _full_convolve(w, a)[j]
    
    wc2 = _full_convolve(w, a)
    part2 = wc2[:n]  # part2[j] = wc2[j] = sum_k w[k]*a[j-k]
    
    # Total gradient of smoothed objective w.r.t. a:
    # 4n * (part1 + part2)
    grad_a = 4.0 * n * (part1 + part2)
    
    # Convert to gradient w.r.t. z (since a = softmax(z)):
    # d/dz_j = a[j] * (grad_a[j] - sum_i a[i]*grad_a[i])
    W = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - W)
    
    return grad_z


class _LbfgsObj:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoothed_obj(z, self.beta, self.n)

    def grad(self, z):
        return _smoothed_grad(z, self.beta, self.n)


def _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta based on observed progress."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = max_c - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

    # Adaptive beta schedule: start large, decrease geometrically
    n_betas = 25
    betas = base_beta * (0.65 ** np.arange(n_betas))
    # Ensure minimum beta
    min_beta = max_c * 0.000005
    betas = np.maximum(betas, min_beta)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        # Adaptive max iterations
        est_time = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(8000, max(100, int(remaining / (est_time * 15))))

        prob = _LbfgsObj(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 100,
                    'maxls': 40,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS beta={beta:.2e} (idx={i}): score={current_sc:.12f}")
            elif i > 3:
                # If no improvement for several betas, try a different strategy
                print(f"    L-BFGS stall at beta={beta:.2e}, score={current_sc:.12f}")
        except Exception as e:
            print(f"    L-BFGS error at beta={beta:.2e}: {e}")
            continue

    return current_a


def _peak_merging(a_init, deadline, min_time_remaining=30.0):
    """Peak merging: iteratively find directions that equalize top peaks."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    lr = 0.05
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            lr *= 0.5
            no_improve = 0
            if lr < 1e-6:
                break

        # Find top K peaks
        K = min(50, max(3, int(n * 0.01)))
        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]
        peak_vals = c[active]

        # Find direction using weighted gradient (weights = normalized peak values)
        w_shift = (peak_vals - peak_vals.min()) / (peak_vals.max() - peak_vals.min() + 1e-15)
        w = w_shift / w_shift.sum()

        # Weighted gradient
        grad = _weighted_conv_grad(a, w, active)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-18:
            break

        # Step in negative gradient direction (to decrease peaks)
        direction = -grad_t / gnorm
        a_new = _project_simplex(a + lr * direction)

        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < current_sc:
            a = a_new
        else:
            # Try smaller step
            a_new = _project_simplex(a + 0.1 * lr * direction)
            c_new2 = _full_convolve(a_new, a_new)
            sc_new2 = 2.0 * n * float(c_new2.max())
            if sc_new2 < current_sc:
                a = a_new
                lr *= 0.7
            else:
                lr *= 0.5

        if no_improve % 500 == 0:
            print(f"    PeakMerge: best={best_sc:.10f}, lr={lr:.6f}, no_improve={no_improve}")

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_initial(n, kind="uniform", seed=42):
    rng = np.random.default_rng(seed)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        freqs = [3, 7, 13, 21]
        a = np.ones(n)
        for f in freqs:
            a += 0.1 * np.cos(2.0 * np.pi * f * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        # Random low-frequency signal
        freq = rng.uniform(0.5, 3.0)
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * t / n + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.3) ** 2) + np.exp(-12.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    else:
        return np.ones(n) / n


def _coordinate_descent_1d(a, j, c, deadline_per_coord=0.01):
    """Optimize a single coordinate a[j] given all others fixed.
    Returns the optimal value of a[j]."""
    n = len(a)
    old = a[j]
    
    # Determine affected peaks and their linear/quadratic coefficients
    # c[k](x) = c[k] + 2*(x-old)*a[k-j] for k != 2j, k-j in [0,n-1]
    # c[2j](x) = c[2j] + x^2 - old^2 (if 2j < 2n-1)
    
    # The function to minimize: max_k c[k](x) = max(constant_part, affected_part)
    # constant_part = max of c[k] for unaffected k
    
    # Affected peaks: k in [j, j+n-1] (where k-j is in [0, n-1])
    k_lo = j
    k_hi = min(2*n - 2, j + n - 1)
    
    if k_lo > k_hi:
        return old
    
    affected_k = np.arange(k_lo, k_hi + 1)
    # For each affected k, the change is:
    # delta_k = 2*(x-old)*a[k-j] for k != 2j
    # delta_{2j} = x^2 - old^2
    
    # Coefficients: m_k = 2*a[k-j] (slope in x), b_k = c[k] - 2*old*a[k-j]
    # For k=2j: it's x^2 + c[2j] - old^2
    
    slopes = 2.0 * a[affected_k - j]  # m_k = 2*a[k-j]
    intercepts = c[affected_k] - 2.0 * old * a[affected_k - j]  # b_k
    
    # Handle k=2j separately (quadratic term)
    if 2*j <= 2*n - 2:
        idx_2j = 2*j - k_lo
        if 0 <= idx_2j < len(affected_k):
            # c[2j](x) = c[2j] + x^2 - old^2
            # The linear approximation would be: c[2j] + 2*(x-old)*old = c[2j] + 2x*old - 2old^2
            # But actual is: c[2j] + x^2 - old^2
            # Set slope to 0 and handle quadratic separately
            # For the 1D minimization, we'll just include it as a separate term
            pass
    
    # Unaffected max
    if k_lo > 0:
        c_unaffected_lo = c[:k_lo]
    else:
        c_unaffected_lo = np.array([])
    if k_hi < 2*n-1:
        c_unaffected_hi = c[k_hi+1:]
    else:
        c_unaffected_hi = np.array([])
    c_unaffected = np.concatenate([c_unaffected_lo, c_unaffected_hi])
    if len(c_unaffected) > 0:
        c_unaff_max = float(c_unaffected.max())
    else:
        c_unaff_max = 0.0
    
    # 1D minimization: find x >= 0 minimizing max(c_unaff_max, max_k (slopes[k]*x + intercepts[k]), x^2 + c[2j] - old^2)
    # Use a simple approach: evaluate at a few candidate points
    
    # Candidate x values:
    # 1. x = 0
    # 2. x = old (current)
    # 3. Intersections of pairs of lines (top few)
    # 4. x where the quadratic equals the max line
    
    # For efficiency, just use golden section on [0, x_max]
    # x_max: large enough that the function is increasing
    x_max = max(2.0 * old, 0.1 / (np.max(np.abs(slopes)) + 1e-10))
    x_max = min(x_max, 10.0 * old + 0.01)
    x_max = max(x_max, 1e-6)
    
    # Golden section search
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    
    def f_1d(x):
        val_line = slopes * x + intercepts
        val_max = max(c_unaff_max, float(val_line.max()))
        if 2*j <= 2*n - 1:
            val_quad = x * x + c[2*j] - old * old
            val_max = max(val_max, val_quad)
        return val_max
    
    a_lo = 0.0
    a_hi = x_max
    c_pt = a_hi - phi * (a_hi - a_lo)
    d_pt = a_lo + phi * (a_hi - a_lo)
    f_c = f_1d(c_pt)
    f_d = f_1d(d_pt)
    
    for _ in range(40):
        if f_c < f_d:
            a_hi = d_pt
            d_pt = c_pt
            f_d = f_c
            c_pt = a_hi - phi * (a_hi - a_lo)
            f_c = f_1d(c_pt)
        else:
            a_lo = c_pt
            c_pt = d_pt
            f_c = f_d
            d_pt = a_lo + phi * (a_hi - a_lo)
            f_d = f_1d(d_pt)
    
    x_opt = (a_lo + a_hi) / 2.0
    return max(0.0, x_opt)


def _propose_candidate_impl(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_initial(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    def _update_best(a, source):
        nonlocal best_seq, best_score
        sc = _score(a)
        if sc < best_score:
            best_seq = a.copy()
            best_score = sc
            print(f"  NEW BEST: {best_score:.10f} [{source}] t={time.monotonic()-t_start:.1f}s")
        return sc

    # === Phase 1: FISTA subgradient (fast initial improvement) ===
    print(f"\n[P1] FISTA subgradient (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 600, t_start + 300)
    if time.monotonic() < p1_deadline:
        a_fista, sc_fista = _fista_subgrad(best_seq, p1_deadline, min_time_remaining=60, K=30, seed=seed)
        _update_best(a_fista, "fista")

    # === Phase 2: Adaptive L-BFGS (main refinement) ===
    print(f"\n[P2] Adaptive L-BFGS round 1 (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 400, t_start + 600)
    if time.monotonic() < p2_deadline:
        a_lb = _run_lbfgs_adaptive(best_seq, p2_deadline, min_time_remaining=60)
        _update_best(a_lb, "lbfgs1")

    # === Phase 3: Peak merging ===
    print(f"\n[P3] Peak merging (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 300, t_start + 800)
    if time.monotonic() < p3_deadline:
        a_pm, sc_pm = _peak_merging(best_seq, p3_deadline, min_time_remaining=60)
        _update_best(a_pm, "peakmerge")

    # === Phase 4: Second FISTA pass ===
    print(f"\n[P4] FISTA round 2 (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 250, t_start + 1000)
    if time.monotonic() < p4_deadline:
        a_fista2, sc_fista2 = _fista_subgrad(best_seq, p4_deadline, min_time_remaining=50, K=10, seed=seed+1)
        _update_best(a_fista2, "fista2")

    # === Phase 5: Second L-BFGS (finer) ===
    print(f"\n[P5] Adaptive L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    p5_deadline = min(deadline - 200, t_start + 1200)
    if time.monotonic() < p5_deadline:
        a_lb2 = _run_lbfgs_adaptive(best_seq, p5_deadline, min_time_remaining=50)
        _update_best(a_lb2, "lbfgs2")

    # === Phase 6: Peak merging + L-BFGS cycle ===
    print(f"\n[P6] PM+L-BFGS cycle (t={time.monotonic()-t_start:.1f}s)")
    p6_deadline = min(deadline - 150, t_start + 1400)
    for cyc in range(3):
        if time.monotonic() > p6_deadline - 60:
            break
        a_pm2, _ = _peak_merging(best_seq, min(p6_deadline - 60 - cyc*30, t_start + 1450), min_time_remaining=30)
        _update_best(a_pm2, f"pm_cyc{cyc}")
        if time.monotonic() > p6_deadline - 30:
            break
        a_lb3 = _run_lbfgs_adaptive(best_seq, p6_deadline, min_time_remaining=30)
        _update_best(a_lb3, f"lbfgs_cyc{cyc}")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    p7_deadline = deadline - 80
    n_candidates = [16384, 12288, 8192, 24576, 32768]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)
    n_candidates = [x for x in n_candidates if x <= 49152]  # cap size

    for n_try in n_candidates:
        if time.monotonic() > p7_deadline:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")
        if time.monotonic() < p7_deadline - 30:
            a_opt = _run_lbfgs_adaptive(a_init, p7_deadline, min_time_remaining=30)
            _update_best(a_opt, f"multi_n_{n_try}")

    # === Phase 8: Final ultra-fine L-BFGS ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 10 - time.monotonic()
    if remaining > 20:
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        c_sorted = np.sort(c_best)[::-1]
        gap = mb - c_sorted[min(5, len(c_sorted)-1)]
        beta_fin = max(gap * 0.005, mb * 0.000001)
        prob = _LbfgsObj(beta_fin, n)
        max_iter = min(5000, max(50, int(remaining / 0.02)))
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-20,
                                       'gtol': 1e-18, 'maxcor': 100})
            a_fin = _softmax(result.x)
            _update_best(a_fin, "final_lbfgs")
        except Exception:
            pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_initial(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_initial(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]


def propose_candidate(seed=42, budget_s=1800):
    """Main entry point: hybrid convex optimizer for autocorrelation minimization."""
    try:
        return _propose_candidate_impl(seed=seed, budget_s=budget_s)
    except Exception as e:
        print(f"ERROR in propose_candidate: {e}")
        import traceback
        traceback.print_exc()
        # Fallback: return a reasonable sequence
        n = 4096
        a = np.ones(n) / n
        return [float(x) for x in a]
