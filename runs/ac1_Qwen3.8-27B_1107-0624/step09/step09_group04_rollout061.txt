"""
Optimized minimizer for 2*n*max(conv(a,a))/sum(a)^2 using:
- Multi-resolution warm start
- Adaptive log-sum-exp smoothing with L-BFGS
- Active-set peak equalization via direct QP-like iterations
- Multi-probe random restarts to escape local minima
- FFT-based fast convolution throughout
Key insight: the center of full conv is sum(a_j * a_{n-1-j}), NOT sum(a_j^2),
so off-center peaks can dominate and the minimum is strictly below 2.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _fast_score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-30:
        return np.inf, a.copy()
    c = _full_conv(a, a)
    return float(2.0 * n * c.max() / (s * s)), c


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_conv(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_conv(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_conv(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_c = 4.0 * n * (h - W)
    return a * grad_c


class _SmoothProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0, max_beta_mults=None):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc, _ = _fast_score(current_a)
    c_cur = _full_conv(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003) * beta_scale

    if max_beta_mults is None:
        max_beta_mults = [1.0, 0.5, 0.25, 0.1, 0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for beta_mult in max_beta_mults:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        beta = base_beta * beta_mult

        est_time_per_iter = 0.003 * max(1.0, n / 4096.0)
        if beta < base_beta * 0.01:
            max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 20))))
        else:
            max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 12))))

        prob = _SmoothProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 100, 'maxls': 60}
            )
            a_new = _softmax(result.x)
            sc_new, _ = _fast_score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_equalize(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc, _ = _fast_score(a)

    K = 5
    K_max = min(200, n)
    lr = 0.03
    no_improve = 0
    max_no_improve = 300

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_conv(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            if K < K_max:
                K = min(K_max, K + 2)
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        top_k_idx = np.argsort(c)[::-1][:K]
        w_S = np.zeros(2 * n - 1)
        w_S[top_k_idx] = 1.0 / K
        a_rev = a[::-1]
        h = _full_conv(w_S, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        alpha = lr / (gnorm + 1e-15)
        improved = False
        for _ in range(40):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_conv(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            lr *= 0.8
            if no_improve % 30 == 0:
                K = max(3, K // 2)
            if no_improve > max_no_improve:
                a = best_a.copy()
                lr = 0.03
                K = 5
                no_improve = 0
                if no_improve > max_no_improve * 2:
                    break
        else:
            lr = min(0.08, lr * 1.02)

    return best_a, best_sc


def _subgradient_polish(a_init, deadline, min_time_remaining=10.0):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc, _ = _fast_score(a)

    max_iters = 8000
    avg_a = a.copy()
    W = 0.0
    gamma_init = 0.02

    for it in range(max_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_conv(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()

        threshold = c_max * (1 - 0.0005)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        w_S = np.zeros(2 * n - 1)
        w_S[active] = 2.0 / len(active)
        a_rev = a[::-1]
        h = _full_conv(w_S, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        gamma = gamma_init / np.sqrt(it + 1)
        eta = min(gamma, 0.4 / (np.max(np.abs(grad)) + 1e-15))

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        W += 1.0
        avg_a = (avg_a * (W - 1) + a_new) / W
        a = a_new

    sc_avg, _ = _fast_score(avg_a)
    if sc_avg < best_sc:
        best_a = avg_a
        best_sc = sc_avg

    return best_a, best_sc


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.8 + 0.1 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15 * (t - 0.25) ** 2) + 0.8 * np.exp(-15 * (t - 0.75) ** 2) + 0.02
        return a / a.sum()
    elif kind == "fourier_mixed":
        if rng is None:
            rng = np.random.default_rng(0)
        t = np.arange(n)
        a = np.ones(n)
        for freq in [3, 7, 13, 21]:
            a += 0.2 * np.sin(2 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "waterfall":
        t = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + 20.0 * t) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, rng, strength=0.05):
    n = len(a)
    t = np.arange(n) / n
    ramp = 0.3 * (t - 0.5)
    noise = 0.7 * rng.standard_normal(n)
    perturbation = ramp + noise
    perturbation = perturbation - perturbation.mean()
    z = np.log(np.maximum(a, 1e-30))
    z = z + strength * perturbation
    z = z - z.mean()
    return _softmax(z)


def _peak_reduction(a, deadline, min_time_remaining=20.0):
    """Aggressively reduce the single highest peak using direct line search
    in the direction that most reduces it while keeping other peaks controlled."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc, _ = _fast_score(a)

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_conv(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()

        peak_idx = int(np.argmax(c))
        # Direction to reduce peak at peak_idx: -a[peak_idx - j] (shifted)
        # grad at peak_idx: d(a*a)/da_j = a_{peak_idx - j} (for valid indices)
        direction = np.zeros(n)
        for j in range(n):
            k = peak_idx - j
            if 0 <= k < 2 * n - 1:
                direction[j] = a[k] if 0 <= k < n else 0.0
        # Actually, d(a*a)_k / da_j = a_{k-j} if 0 <= k-j < n, else 0
        direction = np.zeros(n)
        for j in range(n):
            kk = peak_idx - j
            if 0 <= kk < n:
                direction[j] = a[kk]

        direction = direction - direction.mean()
        gnorm = np.linalg.norm(direction)
        if gnorm < 1e-18:
            break

        alpha = 0.05 / (gnorm + 1e-15)
        improved = False
        for _ in range(30):
            a_try = a - alpha * direction
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try, _ = _fast_score(a_try)
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            break
        else:
            alpha = min(0.1, alpha * 1.1)

    return best_a, best_sc

        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final, _ = _fast_score(final)
        print(f"FINAL: internal score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"FINAL: evaluate_sequence={ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_fast_score(fallback)[0]:.10f}")
    return [float(x) for x in fallback]
