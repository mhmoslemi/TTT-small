def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _conv_gradient(w, a):
    """Gradient of sum_k w[k]*C[k] w.r.t. a, where C = convolve(a,a)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n-1:2*n-1]

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -500, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -500, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

class _SmoProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)
    def gradient(self, z):
        return _smograd(z, self.beta, self.n)

def _run_lbfgs_smooth(a_init, deadline, min_time_remaining=60.0, beta_scale=1.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003) * beta_scale
    
    # Beta schedule: start coarse, end fine
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        # Time per iteration estimate
        est_time = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(20, int(remaining / (est_time * 10))))
        
        prob = _SmoProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 60}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

def _active_set_optimize(a_init, deadline, min_time_remaining=30.0):
    """Active-set projected gradient with adaptive K and momentum."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    K = 5
    lr = 0.005
    momentum_coef = 0.7
    v = np.zeros(n)
    no_improve = 0
    max_no_improve = 300
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            # Slowly grow K
            if K < min(40, n // 4):
                K += 1
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        
        # Adaptive K
        if no_improve > 50:
            K = max(3, K // 2)
            lr = min(0.01, lr * 1.5)
            no_improve = 0
            v[:] = 0
        
        # Get top K peaks
        top_k = np.argpartition(c, -K)[-K:]
        top_k = top_k[np.argsort(c[top_k])[::-1]]
        
        # Weight: uniform over active set
        w = np.zeros(2*n-1)
        w[top_k] = 1.0 / K
        grad = _conv_gradient(w, a)
        
        # Project to tangent space (sum-preserving)
        grad = grad - grad.mean()
        
        # Non-negativity: zero out gradient at boundaries
        small_mask = a < 1e-10
        if small_mask.any():
            grad[small_mask] = 0
        
        # Momentum
        v = momentum_coef * v - lr * grad
        v = v - v.mean()
        v[small_mask] = 0
        
        # Line search (backtracking)
        vn = float(np.dot(v, v))
        if vn < 1e-30:
            break
        
        alpha = 1.0
        improved = False
        for ls in range(25):
            a_try = a + alpha * v
            if (a_try < -1e-10).any():
                alpha *= 0.5
                continue
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14 * max(1.0, current_sc):
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            lr *= 0.85
            if lr < 1e-8:
                break
    
    return best_a, best_sc
