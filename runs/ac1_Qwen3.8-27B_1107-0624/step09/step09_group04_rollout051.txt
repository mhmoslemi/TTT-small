        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Scale iterations based on beta
        if beta < base_beta * 0.01:
            max_iter = min(8000, max(100, int(remaining / 0.02)))
        else:
            max_iter = min(5000, max(50, int(remaining / 0.03)))
        
        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

def _active_set_equalize_v2(a_init, deadline, min_time_remaining=30.0):
    """Improved active set equalization with BB step size and adaptive K."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    K = 5
    K_max = min(300, n)
    lr = 0.03
    no_improve = 0
    prev_a = None
    prev_grad = None
    bb_alpha = lr
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            K = min(K_max, K + 2)
        
        # Adaptive active set threshold
        c_sorted = np.sort(c)[::-1]
        adaptive_thresh = c_max * (1 - max(0.00001, 1.0 / (K * 100)))
        top_k_idx = np.argsort(c)[::-1][:K]
        
        # Compute average gradient of top-K peaks
        w_S = np.zeros(2 * n - 1)
        w_S[top_k_idx] = 1.0 / K
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Barzilai-Borwein step size
        if prev_a is not None and prev_grad is not None:
            s_vec = a - prev_a
            y_vec = grad - prev_grad
            sy = np.dot(s_vec, y_vec)
            yy = np.dot(y_vec, y_vec)
            if yy > 1e-20 and sy > -1e-20:
                bb_alpha = max(1e-10, sy / yy)
                bb_alpha = min(bb_alpha, 0.1)
        
        alpha = bb_alpha / (gnorm + 1e-15)
        improved = False
        for _ in range(40):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                prev_a = a.copy()
                prev_grad = grad.copy()
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            lr *= 0.9
            if no_improve % 30 == 0:
                K = max(5, K // 2)
            if no_improve >= 300:
                a = best_a.copy()
                lr = 0.03
                K = 5
                no_improve = 0
                bb_alpha = lr
                if no_improve > 600:
                    break
        else:
            lr = min(0.08, lr * 1.02)
    
    return best_a, best_sc

def _nesterov_subgradient(a_init, deadline, min_time_remaining=20.0):
    """Nesterov-accelerated subgradient method."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    y = a.copy()  # Nesterov momentum variable
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    t_val = 1.0
    momentum = 0.9
    
    for it in range(50000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        c = _full_convolve(y, y)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = y.copy()
        
        # Active set
        active = np.where(c >= c_max * (1 - 0.0001))[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        # Subgradient
        w_S = np.zeros(2 * n - 1)
        w_S[active] = 1.0 / len(active)
        y_rev = y[::-1]
        h = _full_convolve(w_S, y_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Step size
        t_val += 1.0
        eta = 0.005 / (t_val ** 0.5)
        eta = min(eta, 0.5 / (np.max(np.abs(grad)) + 1e-15))
        
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Nesterov update
        mu = (1 + momentum) / (1 + momentum + t_val * 0.01)
        y = a_new + mu * (a_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        
        a = a_new
    
    return best_a, best_sc

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = t + 0.1 * np.sin(2 * np.pi * 3 * t) + 1e-10
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
