def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n_out = len(a) + len(b) - 1
    N = _next_pow2(n_out)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n_out]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_hvp(v, W, n):
    """Hankel matrix-vector product: (H@v)[i] = 2*sum_j W[i+j]*v[j] = 2*conv(W, v_rev)[i+n-1]"""
    v_rev = v[::-1]
    cv = _fft_convolve(W, v_rev)
    return 2.0 * cv[n - 1:2 * n - 1]


def _cg_solve_hankel(W, n, rhs, max_iter=200, reg=0.0):
    """Solve H@x = rhs where H[i,j] = 2*W[i+j]. Uses regularized CG for robustness."""
    n_int = n
    x = np.zeros(n_int)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    W_reg = W.copy() if reg > 0 else W
    for it in range(max_iter):
        if reg > 0:
            Hp = _hankel_hvp(p, W, n_int) + reg * p
        else:
            Hp = _hankel_hvp(p, W, n_int)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        if pHp > 0:
            alpha = rs / pHp
        else:
            alpha = -rs / max(-pHp, 1e-30)
            if alpha < 0:
                break
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            break
        x = x_new
        r = r_new
        p = (rs_new / max(rs, 1e-30)) * p + r
        rs = rs_new
        if rs < 1e-26 * rs0:
            break
    return x


def _kkt_newton(a_init, deadline, max_rounds=500, k_schedule=None, verbose=False):

        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        K = min(k_schedule[K_cycle % len(k_schedule)], 2 * n - 1)
        K_cycle += 1

        W[idx] = 1.0 / K

        wc = _fft_convolve(W, a_rev)

            x_cg = _cg_solve_hankel(W_test, n, rhs, max_iter=150, reg=reg)
            dx = x_cg - np.mean(x_cg)

            dx = None

        sc = _score(a)

def _entropic_lbfgs(a_init, deadline, min_remaining=30.0, verbose=False):
    """L-BFGS on softmax parameterization with entropic smooth max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _fft_convolve(current_a, current_a)
    c_max_cur = float(c_cur.max())
    beta = c_max_cur * 0.05

    for outer in range(50):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 15:
            break

        def obj_fn(z_val, b=beta, n_val=n):
            a_val = _softmax(z_val)
            c_val = _fft_convolve(a_val, a_val)
            cm = float(c_val.max())
            shifted = (c_val - cm) / b
            e = np.exp(np.clip(shifted, -700, 0))
            s_val = e.sum()
            if s_val < 1e-300:
                return 2.0 * n_val * cm
            return 2.0 * n_val * (cm + b * np.log(s_val))

        def grad_fn(z_val, b=beta, n_val=n):
            a_val = _softmax(z_val)
            c_val = _fft_convolve(a_val, a_val)
            cm = float(c_val.max())
            shifted = (c_val - cm) / b
            e = np.exp(np.clip(shifted, -700, 0))
            s_val = e.sum()
            if s_val < 1e-300:
                w = np.zeros_like(c_val)
                w[int(np.argmax(c_val))] = 1.0
            else:
                w = e / s_val
            a_rev = a_val[::-1]
            wc = _fft_convolve(w, a_rev)
            g_a = 2.0 * wc[n_val - 1:2 * n_val - 1]
            inner = float(np.dot(a_val, g_a))
            g_z = a_val * (g_a - inner)
            return 2.0 * n_val * g_z

        est_cost = 0.003 * max(1.0, n / 16384.0)
        max_iter = min(2000, max(20, int(remaining / (est_cost * 25))))

        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',

        c_new = _fft_convolve(current_a, current_a)

        gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]

        elif gap < 1e-5 * c_max_new:

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "hanning":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "flat_taper":
        t = np.linspace(0, 1, n)
        taper = 0.15
        a = np.ones(n)
        for i in range(n):
            if t[i] < taper:
                a[i] = 0.5 + 0.5 * np.cos(np.pi * t[i] / taper)
            elif t[i] > 1 - taper:
                a[i] = 0.5 + 0.5 * np.cos(np.pi * (t[i] - (1 - taper)) / taper)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _dual_refine(a_init, deadline, max_rounds=200):
    """Refine using dual variable (active set weights) optimization."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Use top-K with K increasing
        K = min(1 + rnd, 10000, 2 * n - 1)
        idx = np.argpartition(c, -K)[-K:]
        
        # Use smoothed weights instead of uniform
        c_active = c[idx]
        w_active = np.exp((c_active - c_max) / (0.01 * c_max + 1e-30))
        w_active = w_active / w_active.sum()
        
        W = np.zeros(2 * n - 1)
        W[idx] = w_active

        # Solve H @ a = lambda (approximately) via CG
        a_rev = a[::-1]
        # Use the current a to estimate the system
        # Actually, let's use the KKT condition:
        # sum_k W[k] * 2 * R_k a = lambda * 1
        # => 2 * conv(W, a_rev)[n-1:2n-1] = lambda * 1
        # This is already satisfied approximately if a is near optimal.
        # Instead, use gradient descent on the KKT residual.
        
        target = 2.0 * _fft_convolve(W, a_rev)[n - 1:2 * n - 1]
        resid = target - np.mean(target)

            break

        # The direction to reduce KKT violation:
        # We want 2*conv(W', a_rev) closer to constant.

        grad = 2.0 * _fft_convolve(W, a[::-1])[n - 1:2 * n - 1]
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Try Newton step
        dx = _cg_solve_hankel(W, n, -grad, max_iter=100, reg=1e-10)
        dx = dx - np.mean(dx)
        if grad @ dx < -1e-30:
            direction = dx
        else:
            direction = -grad / gnorm
            direction = direction - np.mean(direction)

        # Line search
        alpha = 1.0
        step_done = False
        for ls in range(40):
            a_try = a + alpha * direction
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _fft_convolve(a_try, a_try)
            if float(c_try.max()) < c_max - 1e-18:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-18:
                break

        if not step_done:
            a_try = a - 0.01 * grad / max(gnorm, 1e-30)
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                c_try = _fft_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-18:
                    a = a_try

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, _score(best_a)
