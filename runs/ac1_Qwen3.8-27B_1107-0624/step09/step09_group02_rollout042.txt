"""
Coarse-to-fine optimizer for minimum peak autocorrelation sequences.
Uses progressive resolution refinement, L-BFGS with log-sum-exp smoothing,
adaptive weighted conjugate gradient, asymmetric initializations, and
block coordinate descent for final polishing.
Key insight: the optimal sequence is asymmetric; the symmetric (palindromic)
constraint forces score >= 2, so asymmetry is essential for score < 2.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _hvp_pv(v, p, n):
    b = v[::-1]
    conv_len = len(b) + len(p) - 1
    N = _next_pow2(conv_len)
    Fb = np.fft.rfft(b, N)
    Fp = np.fft.rfft(p, N)
    conv = np.fft.irfft(Fb * Fp, N)[:conv_len]
    result = 2.0 * conv[n - 1:2 * n - 1].copy()
    for k in range(0, 2 * n - 1, 2):
        m = k // 2
        if m < n and k < len(p):
            result[m] -= v[m] * p[k]
    return result


def _make_profile(n, kind="tapered", asym=0.0):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
    elif kind == "tapered_asym":
        shift = t + asym * (t * (1 - t))
        shift = np.clip(shift, 0.001, 0.999)
        a = np.sin(np.pi * shift) ** 0.5
        a = np.maximum(a, 1e-10)
    elif kind == "cos":
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
    elif kind == "ramp_taper":
        a = np.sin(np.pi * t) ** 0.5 * (0.5 + 0.5 * t)
        a = np.maximum(a, 1e-10)
    elif kind == "bump_asym":
        a = np.exp(-4.0 * (t - 0.45) ** 2) + 0.05
    elif kind == "bump_asym2":
        a = np.exp(-4.0 * (t - 0.55) ** 2) + 0.05
    elif kind == "power_taper":
        a = (t * (1 - t)) ** 0.4
        a = np.maximum(a, 1e-10)
    else:
        a = np.ones(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 1e-15)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, max_beta_factor=1.0, min_beta_factor=1e-9):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap * 0.15, max_c * 0.005) * max_beta_factor

    ratios = [1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001,
              0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001,
              0.000005, 0.000002, 0.000001, 0.0000005, 0.0000002, 0.0000001]
    ratios = [max(r, min_beta_factor / 1.0) for r in ratios]

    for i, ratio in enumerate(ratios):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        beta = base_beta * ratio
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(20, int(remaining / (est_time * 4))))

        try:
            def obj_func(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    return 2.0 * n * c_max_local
                return 2.0 * n * (c_max_local + beta * np.log(s_e))

            def grad_func(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    w = np.zeros_like(c_local)
                    w[int(np.argmax(c_local))] = 1.0
                else:
                    w = e / s_e
                a_rev = a_local[::-1]
                wc = _full_convolve(w, a_rev)
                h = wc[n - 1:2 * n - 1]
                W_val = float(np.dot(w, c_local))
                return 4.0 * n * a_local * (h - W_val)

            result = minimize(
                obj_func, z,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 80}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return best_a


def _run_proj_grad(a_init, deadline, min_time_remaining=30.0, momentum_decay=0.9):

                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros(n)
                    patience = 0
                    beta_m *= 0.95

    a_final = _softmax(z)
    sc_final = _score(a_final)

def _run_weighted_cg(a_init, deadline, min_time_remaining=30.0):
    Weighted conjugate gradient for the smoothed QP:
    minimize sum_k w[k]*c[k] subject to sum(a)=1, a>=0
    using the FFT-based HVP and simplex projection.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())

    for outer in range(200):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Adaptive weights: emphasize peaks
        temp = max(c_max * 0.01, 1e-10)
        w = np.exp((c - c_max) / temp)
        w = w / w.sum()

        # Conjugate gradient in the simplex
        d = np.zeros(n)
        r = _hvp_pv(a, w, n)  # gradient of the weighted objective
        # Project gradient to the tangent space of the simplex
        r = r - r.mean()
        alpha_prev = 0
        rnorm_sq = float(r @ r)
        if rnorm_sq < 1e-20:
            break

        for cg_iter in range(30):
            if time.monotonic() + min_time_remaining > deadline:
                break
            # Line search
            alpha = 0.1
            a_cur_projected = _proj_simplex(a - alpha * r)
            for ls in range(10):
                a_try = _proj_simplex(a - alpha * r)
                if np.linalg.norm(a_try - a) > 1e-15:
                    break
                alpha *= 0.5
            a = a_try
            r_new = _hvp_pv(a, w, n)
            r_new = r_new - r_new.mean()
            rnorm_new_sq = float(r_new @ r_new)
            if rnorm_new_sq < 1e-20:
                break
            beta_cg = rnorm_new_sq / (rnorm_sq + 1e-20)
            d = r_new + beta_cg * d
            r = r_new
            rnorm_sq = rnorm_new_sq
            alpha = 0.1

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _run_block_coord(a_init, deadline, min_time_remaining=30.0, block_size=64):
    Block coordinate descent: optimize blocks of coordinates one at a time.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()

    n_blocks = (n + block_size - 1) // block_size

    for epoch in range(50):
        if time.monotonic() + min_time_remaining > deadline:
            break
        # Shuffle block order
        order = np.random.default_rng(epoch * 1234 + 7).permutation(n_blocks)
        for bi in order:
            if time.monotonic() + min_time_remaining > deadline:
                break
            start = bi * block_size
            end = min(start + block_size, n)
            if end - start < 2:
                continue

            # Optimize the block using a few L-BFGS iterations
            z_block = np.log(np.maximum(a[start:end], 1e-30))
            z_rest = np.log(np.maximum(a[:start].copy(), 1e-30))
            if end < n:
                z_rest2 = np.log(np.maximum(a[end:].copy(), 1e-30))
            else:
                z_rest2 = np.array([])

            for inner in range(5):
                if time.monotonic() + min_time_remaining > deadline:
                    break
                # Reconstruct full sequence
                a_full = np.concatenate([
                    _softmax(z_rest) if len(z_rest) > 0 else np.array([]),
                    _softmax(z_block),
                    _softmax(z_rest2) if len(z_rest2) > 0 else np.array([])
                ])
                if len(a_full) != n:
                    break
                # Normalize
                a_full = a_full / a_full.sum()

                c = _full_convolve(a_full, a_full)
                c_max = float(c.max())
                k_act = min(5, len(c))
                top_k = np.argsort(c)[::-1][:k_act]
                p = np.zeros(2 * n - 1)
                p[top_k] = 1.0 / k_act

                grad_full = _hvp_pv(a_full, p, n)
                grad_block = grad_full[start:end]
                gnorm = np.linalg.norm(grad_block)
                if gnorm < 1e-14:
                    break
                lr = 0.05 / (gnorm + 1e-10)
                z_block = z_block - lr * grad_block
                z_block = np.clip(z_block, -50, 50)

            # Update
            a_new_block = _softmax(z_block)
            a_new_rest1 = _softmax(z_rest) if len(z_rest) > 0 else np.array([])

                sc = _score(a_candidate)
                if sc <= best_sc + 1e-12:
                    a = a_candidate
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        best_a = a.copy()

    return best_a, best_sc


def _run_peak_shave(a_init, deadline, min_time_remaining=30.0):
    """Direct peak shaving: iteratively reduce the dominant convolution peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(500):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Find peaks above 99.5% of max
        threshold = c_max * 0.995
        peaks = c >= threshold
        if peaks.sum() == 0:
            peaks[int(np.argmax(c))] = True
        p = np.zeros(2 * n - 1)
        p[peaks] = 1.0 / peaks.sum()

        grad = _hvp_pv(a, p, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 0.1 / (gnorm + 1e-10)
        improved = False
        for ls in range(40):
            a_new = _proj_simplex(a - step * grad)
            c_new = _full_convolve(a_new, a_new)
            if float(c_new.max()) < c_max * (1 - 1e-8):
                a = a_new
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                improved = True
                break
            step *= 0.5

        if not improved:
            # Try a small random perturbation to escape
            z = np.log(np.maximum(a, 1e-30))
            z = z + 0.01 * np.random.default_rng(it).standard_normal(n)
            z = z - z.mean()
            a = _softmax(z)

    return best_a, best_sc

        current_a = a1

        # Method 2: If time allows, try from asymmetric init
        remaining_stage = stage_deadline - 30 - time.monotonic()
        if remaining_stage > 20:
            a2 = _run_lbfgs(a_asym, time.monotonic() + remaining_stage,
                            min_time_remaining=5, max_beta_factor=0.8, min_beta_factor=1e-7)
            sc2 = _score(a2)
            if sc2 < _score(current_a):
                update_best(a2, sc2, f"lbfgs_asym_n{n_stage}")
                current_a = a2

        # Method 3: Projected gradient polish
        remaining_stage = stage_deadline - 15 - time.monotonic()
        if remaining_stage > 10:
            a3, sc3 = _run_proj_grad(current_a, time.monotonic() + remaining_stage,
                                     min_time_remaining=3, momentum_decay=0.85)
            update_best(a3, sc3, f"pg_n{n_stage}")
            if sc3 < _score(current_a):
                current_a = a3

    # === Intensive optimization at target n ===
    n_target = min(32768, n_stages[-1] if n_stages else 32768)
    print(f"\n[Intensive] n={n_target} (t={time.monotonic() - t_start:.1f}s)")

    if len(current_a) != n_target:
        current_a = _resample(current_a, n_target)
    update_best(current_a, _score(current_a), "before_intensive")

    # Phase A: Extended L-BFGS with very tight beta
    phase_a_deadline = min(deadline - 300, time.monotonic() + 400)
    a_ext = _run_lbfgs(current_a, phase_a_deadline, min_time_remaining=20,
                       max_beta_factor=0.3, min_beta_factor=1e-10)
    update_best(a_ext, _score(a_ext), "lbfgs_tight")
    if _score(a_ext) < _score(current_a):
        current_a = a_ext

    # Phase B: Weighted CG
    phase_b_deadline = min(deadline - 200, phase_a_deadline + 200)
    a_cg, sc_cg = _run_weighted_cg(current_a, phase_b_deadline, min_time_remaining=15)
    update_best(a_cg, sc_cg, "weighted_cg")
    if sc_cg < _score(current_a):
        current_a = a_cg

    # Phase C: Peak shaving
    phase_c_deadline = min(deadline - 150, phase_b_deadline + 200)
    a_ps, sc_ps = _run_peak_shave(current_a, phase_c_deadline, min_time_remaining=10)
    update_best(a_ps, sc_ps, "peak_shave")
    if sc_ps < _score(current_a):
        current_a = a_ps

    # Phase D: Block coordinate descent
    phase_d_deadline = min(deadline - 100, phase_c_deadline + 200)
    a_bc, sc_bc = _run_block_coord(current_a, phase_d_deadline, min_time_remaining=10,
                                   block_size=128)
    update_best(a_bc, sc_bc, "block_coord")
    if sc_bc < _score(current_a):
        current_a = a_bc

    # Phase E: Asymmetric L-BFGS (try to break any residual symmetry)
    phase_e_deadline = min(deadline - 80, phase_d_deadline + 150)
    a_asym2, sc_asym2 = _run_asymmetric_lbfgs(current_a, phase_e_deadline, min_time_remaining=10)
    update_best(a_asym2, sc_asym2, "asym_lbfgs")
    if sc_asym2 < _score(current_a):
        current_a = a_asym2

    # Phase F: Try n=65536 if time allows
    remaining = deadline - 60 - time.monotonic()
    if remaining > 120:
        print(f"\n[Large n] Trying n=65536 (remaining={remaining:.1f}s)")
        a_large_init = _resample(current_a, 65536)
        a_large = _run_lbfgs(a_large_init, deadline - 30, min_time_remaining=15,
                             max_beta_factor=0.5, min_beta_factor=1e-9)
        update_best(a_large, _score(a_large), "n65536")

        if _score(a_large) < _score(current_a):
            current_a = a_large
            # Polish at large n
            if time.monotonic() < deadline - 30:
                a_large2, sc_l2 = _run_proj_grad(current_a, deadline - 20,
                                                 min_time_remaining=5, momentum_decay=0.8)
                update_best(a_large2, sc_l2, "n65536_pg")

    # Phase G: Final ultra-tight polish
    remaining = deadline - 15 - time.monotonic()
    if remaining > 10:
        print(f"\n[Final Polish] (remaining={remaining:.1f}s)")
        n_fin = len(current_a)
        z = np.log(np.maximum(current_a, 1e-30))
        z = z - z.mean()
        c_b = _full_convolve(current_a, current_a)
        mb = float(c_b.max())
        beta_fin = mb * 1e-7

        def final_obj(z_local):
            a_local = _softmax(z_local)
            c_local = _full_convolve(a_local, a_local)
            c_max_local = float(c_local.max())
            shifted = (c_local - c_max_local) / beta_fin
            e = np.exp(np.clip(shifted, -700, 0))
            s_e = e.sum()
            if s_e < 1e-300:
                return 2.0 * n_fin * c_max_local
            return 2.0 * n_fin * (c_max_local + beta_fin * np.log(s_e))

        def final_grad(z_local):
            a_local = _softmax(z_local)
            c_local = _full_convolve(a_local, a_local)
            c_max_local = float(c_local.max())
            shifted = (c_local - c_max_local) / beta_fin
            e = np.exp(np.clip(shifted, -700, 0))
            s_e = e.sum()
            if s_e < 1e-300:
                w = np.zeros_like(c_local)
                w[int(np.argmax(c_local))] = 1.0
            else:
                w = e / s_e
            a_rev = a_local[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n_fin - 1:2 * n_fin - 1]
            W_val = float(np.dot(w, c_local))
            return 4.0 * n_fin * a_local * (h - W_val)

        try:
            max_iter = min(2000, max(50, int(remaining / 0.02)))
            result = minimize(
                final_obj, z,
                jac=final_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 50}
            )
            a_flb = _softmax(result.x)
            update_best(a_flb, _score(a_flb), "final_lbfgs")
        except Exception:
            pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
