"""
KKT-based optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations:
- FFT-accelerated matrix-free KKT linear solves (no softmax parameterization)
- Nesterov accelerated projected subgradient for fast initial convergence
- Iterative active-set refinement with exact KKT projection
- Adaptive active set growth
- L-BFGS polish and multi-n exploration
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.sparse.linalg import LinearOperator, gmres


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    c_max = float(c.max())
    return float(2.0 * n * c_max / (s * s))


def _normalize(a):
    s = float(a.sum())
    if s < 1e-15:
        return a
    return a / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_subgrad(a, S, n):
    """Compute average subgradient: (2/|S|) * sum_{k in S} grad(c_k).
    
    Result: g[j] = (2/|S|) * sum_{i=0}^{n-1} a[i] * 1[i+j in S]
    = (2/|S|) * corr(S, a)[j]  where corr[j] = sum_i S[i+j]*a[i]
    """
    S_rev = S[2*n-2::-1]
    N = _next_pow2(3*n - 1)
    conv = np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(S_rev, N))[:3*n-2]
    corr = conv[2*n-2:3*n-2]
    return (2.0 / len(S)) * corr


def _kkt_matvec_factory(S, n):
    """Create a function that computes A*v for the KKT system.
    
    A is n x n:
    Row j (0-indexed, for j=0,...,n-2): A[j,i] = S[i+j+1] - S[i]
    Row n-1: A[n-1,i] = 1
    
    (Av)[j] = corr[j+1] - S[:n] dot v  for j = 0,...,n-2
    (Av)[n-1] = sum(v)
    """
    S_rev = S[2*n-2::-1].astype(np.float64)
    S_n = S[:n].astype(np.float64)
    N = _next_pow2(3*n - 1)
    F_Sr = np.fft.rfft(S_rev, N)
    
    def matvec(v):
        F_v = np.fft.rfft(v, N)
        conv = np.fft.irfft(F_v * F_Sr)[:3*n-2]
        corr = conv[2*n-2:3*n-2]
        Sv = float(np.dot(S_n, v))
        Av = np.empty(n)
        Av[:n-1] = corr[1:n] - Sv
        Av[n-1] = float(np.sum(v))
        return Av
    
    return matvec


def _kkt_solve(a, S, n, max_iter=200, rtol=1e-10):
    """Solve the KKT linear system for given active set S.
    
    System: For j=1,...,n-1: sum_i a_i (S[i+j] - S[i]) = 0
            sum_i a_i = 1
    
    Returns solution or None if it fails.
    """
    matvec_fn = _kkt_matvec_factory(S, n)
    A_op = LinearOperator((n, n), matvec=matvec_fn, dtype=np.float64)
    b = np.zeros(n)
    b[n-1] = 1.0
    
    try:
        x, info = gmres(A_op, b, rtol=rtol, maxiter=max_iter)
        if info > 0:
            return None
    except Exception:
        return None
    
    # Check non-negativity
    if np.any(x < -1e-10):
        return None
    
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s < 1e-10:
        return None
    x = x / s
    
    return x


def _kkt_solve_with_support(a, S, n, support, max_iter=200):
    """Solve KKT system restricted to a given support set.
    
    Unknowns: a_j for j in support, plus lambda (eliminated via differences).
    """
    idx = np.where(support)[0]
    n_support = len(idx)
    if n_support < 3:
        return None
    
    # Build the reduced system
    # For pairs (j, j0) in support: sum_i a_i (S[i+j] - S[i+j0]) = 0
    # Plus sum of a_i = 1
    
    j0 = idx[0]
    
    # We need: for each j in support (except j0):
    #   sum_{i in support} a_i * (S[i+j] - S[i+j0]) = 0
    # And: sum_{i in support} a_i = 1
    
    # Matrix size: n_support x n_support
    # Row r (for j = idx[r], r=0,...,n_support-2):
    #   sum_{s=0}^{n_support-1} a[idx[s]] * (S[idx[s]+j] - S[idx[s]+j0])
    # Last row: sum of a = 1
    
    # This is O(n_support^2) to form, which is fine for moderate support sizes
    if n_support > 900:
        # Use matrix-free approach with GMRES on the full system
        return _kkt_solve(a, S, n, max_iter=max_iter)
    
    A_red = np.zeros((n_support, n_support))
    for r in range(n_support - 1):
        j = idx[r]
        for s in range(n_support):
            i = idx[s]
            A_red[r, s] = float(S[i + j]) - float(S[i + j0])
    A_red[n_support - 1, :] = 1.0
    
    b_red = np.zeros(n_support)
    b_red[n_support - 1] = 1.0
    
    try:
        a_red = np.linalg.solve(A_red, b_red)
    except np.linalg.LinAlgError:
        try:
            a_red, residuals, rank, sv = np.linalg.lstsq(A_red, b_red, rcond=None)
        except Exception:
            return None
    
    if np.any(a_red < -1e-10):
        return None
    
    a_new = np.zeros(n)
    a_new[idx] = np.maximum(a_red, 0.0)
    s = float(a_new.sum())
    if s < 1e-10:
        return None
    a_new = a_new / s
    
    return a_new

        if gap_to_lb < 1e-10:
            break
        
        step = step_base * gap_to_lb / (float(np.dot(g, g)) + 1e-20)
        step = min(step, 0.5 / (t + 1) * 10)
        step = max(step, 1e-8)
        
        v = beta * v - step * g
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            v = np.zeros(n)
            continue
        a_new = a_new / s_new
        a = a_new
        t += 1
        
        if t % 500 == 0:
            step_base *= step_decay
        
        # Periodic KKT solve
        if t % 200 == 0 and time.monotonic() + min_time_remaining + 5 < deadline:
            a_kkt = _kkt_solve(a, S, n, max_iter=100, rtol=1e-8)
            if a_kkt is not None:
                sc_kkt = _score(a_kkt)
                if sc_kkt < current_sc - 1e-12:
                    a = a_kkt
                    v = np.zeros(n)
                    if verbose:
                        print(f"    KKT solve improved: {sc_kkt:.10f}")
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic()
            print(f"    t={t}, score={current_sc:.10f}, best={best_sc:.10f}")
    
    return best_a, best_sc


def _adaptive_kkt_refine(a, deadline, min_time_remaining=30.0, verbose=True):
    """Iterative active-set KKT refinement with growing active set.
    
    Alternates between:
    1. Solving KKT for current active set
    2. Checking if solution is valid (all c_k <= max)
    3. Expanding active set if violated
    """
    n = len(a)
    a = a.copy()
    a = _normalize(a)
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c_max
    
    K = 10  # initial active set size
    max_K = min(500, n)
    
    for iteration in range(200):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        # Identify active set: top K peaks
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
        
        c_sorted = np.sort(c)[::-1]
        # Active set: top K values
        thresh = c_sorted[min(K, len(c_sorted)-1)]
        active = c >= thresh
        S = active.astype(np.float64)
        
        # Solve KKT
        a_new = _kkt_solve(a, S, n, max_iter=300, rtol=1e-10)
        
        if a_new is not None:
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                a = a_new
                # Check if all c_k are within the active set max
                c_new = _full_convolve(a, a)
                c_new_max = float(c_new.max())
                active_max = float(c_new[active].max()) if active.sum() > 0 else c_new_max
                if c_new_max > active_max + 1e-12:
                    # Violated: grow active set
                    K = min(int(active.sum()) + K // 2 + 5, max_K)
                else:
                    # Slight improvement, try to shrink K
                    if K > 10:
                        K = max(10, K - 2)
            else:
                # KKT solve didn't help, try growing active set
                K = min(K + 5, max_K)
        else:
            # KKT solve failed, use subgradient step instead
            g = _grad_subgrad(a, S, n)
            g = g - np.mean(g)
            g_norm = float(np.dot(g, g))
            if g_norm < 1e-20:
                break
            step = 0.01 / np.sqrt(g_norm)
            a_try = a - step * g
            a_try = np.maximum(a_try, 0.0)
            s_try = float(a_try.sum())
            if s_try > 1e-10:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-14:
                    a = a_try
        
        if verbose and iteration % 20 == 0:
            print(f"    KKT iter {iteration}: K={K}, score={best_sc:.10f}")
        
        # Check convergence
        if K >= max_K and iteration > 50:
            # Try to converge with current K
            pass
    
    return best_a, best_sc


def _lbfgs_polish(a_init, deadline, min_time_remaining=10.0, verbose=False):
    """L-BFGS polish in log-space (softmax parameterization)."""
    n = len(a_init)
    a = a_init.copy()
    a = _normalize(a)
    
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())
    best_a = a.copy()
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    c_sorted = np.sort(c0)[::-1]
    gap = float(c_sorted[0] - c_sorted[min(50, len(c_sorted)-1)])
    if gap < c0.max() * 0.003:
        gap = float(c0.max() * 0.003)
    
    multipliers = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [gap * m for m in multipliers]
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        N = _next_pow2(3*n - 1)
        
        def smoobj(z_local):
            a_local = _softmax(z_local)
            fa = np.fft.rfft(a_local, N)
            c_local = np.fft.irfft(fa * fa, N)[:2*n-1]
            return 2.0 * n * _soft_max_value(c_local, beta)
        
        def smograd(z_local):
            a_local = _softmax(z_local)
            fa = np.fft.rfft(a_local, N)
            c_local = np.fft.irfft(fa * fa, N)[:2*n-1]
            # Soft-max weights
            c_max = float(c_local.max())
            shifted = (c_local - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                w = np.zeros_like(c_local)
                w[int(np.argmax(c_local))] = 1.0
            else:
                w = e / s
            # Gradient via convolution
            a_rev = a_local[::-1]
            wa = np.fft.irfft(np.fft.rfft(w, N) * np.fft.rfft(a_rev, N))[:3*n-2]
            grad_a = 2.0 * wa[n-1:2*n-1]
            W = float(np.dot(grad_a, a_local))
            return a_local * (grad_a - W)
        
        max_iter = min(2000, max(100, int(30.0 / max(0.01, n/4096.0))))
        
        try:
            result = minimize(
                smoobj, z, jac=smograd, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxcor': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"      L-BFGS beta={beta:.2e}: score={best_sc:.10f}")
        except Exception:
            pass
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    elif kind == "quartic":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.4*((2*t-1)**4)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _direct_kkt_fixed(a, S, n, n_iter=50):
    """Fixed-point iteration for KKT: alternate between computing active set
    and solving the linear system. Works for moderate n."""
    a = a.copy()
    a = _normalize(a)
    n_s = n
    
    for it in range(n_iter):
        c = _full_convolve(a, a)
        a_new = _kkt_solve(a, S, n_s, max_iter=200, rtol=1e-10)
        if a_new is None:
            break
        if np.allclose(a_new, a, atol=1e-12):
            break
        a = a_new
    
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase KKT-based optimizer:
    1. Load parent or initialize
    2. Nesterov projected subgradient with periodic KKT solves
    3. Adaptive active-set KKT refinement  
    4. L-BFGS polish
    5. Multi-n exploration
    6. Final KKT + L-BFGS polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass
    
    if parent is not None:
        n_main = len(parent)
        best_seq = parent.copy()
        best_score = _score(best_seq)
        print(f"Parent: n={n_main}, score={best_score:.10f}")
    else:
        n_main = 16384
        best_seq = _make_profile(n_main, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n={n_main}: score={best_score:.10f}")
    
    # === Phase 1: Nesterov subgradient with KKT ===
    print(f"\n[P1] Nesterov + KKT (n={n_main}, t={time.monotonic()-t_start:.0f}s)")
    p1_end = t_start + min(400, (budget_s - 150) * 0.25)
    try:
        a1, sc1 = _nesterov_projected_subgrad(best_seq, p1_end, min_time_remaining=100, verbose=True)
        if sc1 < best_score:
            best_score = sc1
            best_seq = a1.copy()
            print(f"  P1 result: {best_score:.10f}")
    except Exception as e:
        print(f"  P1 error: {e}")
    
    # === Phase 2: Adaptive KKT refinement ===
    print(f"\n[P2] KKT refine (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p2_end = min(deadline - 300, time.monotonic() + 350)
        try:
            a2, sc2 = _adaptive_kkt_refine(best_seq, p2_end, min_time_remaining=50, verbose=True)
            if sc2 < best_score:
                best_score = sc2
                best_seq = a2.copy()
                print(f"  P2 result: {best_score:.10f}")
        except Exception as e:
            print(f"  P2 error: {e}")
    
    # === Phase 3: L-BFGS polish ===
    print(f"\n[P3] L-BFGS polish (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p3_end = min(deadline - 200, time.monotonic() + 300)
        try:
            a3, sc3 = _lbfgs_polish(best_seq, p3_end, min_time_remaining=60, verbose=True)
            if sc3 < best_score:
                best_score = sc3
                best_seq = a3.copy()
                print(f"  P3 result: {best_score:.10f}")
        except Exception as e:
            print(f"  P3 error: {e}")
    
    # === Phase 4: Second KKT round ===
    print(f"\n[P4] KKT round 2 (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 40 < deadline:
        p4_end = min(deadline - 150, time.monotonic() + 200)
        try:
            a4, sc4 = _adaptive_kkt_refine(best_seq, p4_end, min_time_remaining=30, verbose=False)
            if sc4 < best_score:
                best_score = sc4
                best_seq = a4.copy()
                print(f"  P4 result: {best_score:.10f}")
        except Exception as e:
            print(f"  P4 error: {e}")
    
    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n (t={time.monotonic()-t_start:.0f}s)")
    n_options = [8192, 12288, 20480, 24576]
    n_options = [x for x in n_options if x != n_main]
    for n_try in n_options:
        if time.monotonic() + 100 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        print(f"  n={n_try} resampled: {sc_r:.10f}")
        
        if sc_r < best_score:
            best_score = sc_r
            best_seq = a_r.copy()
        
        if time.monotonic() + 80 > deadline:
            continue
        
        # Quick KKT refinement at this n
        p5_end = min(deadline - 80, time.monotonic() + 60)
        try:
            a5, sc5 = _adaptive_kkt_refine(a_r, p5_end, min_time_remaining=20, verbose=False)
            if sc5 < best_score:
                best_score = sc5
                best_seq = a5.copy()
                print(f"  n={n_try} KKT: {best_score:.10f}")
        except Exception:
            pass
    
    # === Phase 6: Diverse initializations ===
    print(f"\n[P6] Diverse profiles (t={time.monotonic()-t_start:.0f}s)")
    n_for = len(best_seq)
    profiles = ["periodic", "bimodal", "gaussian", "quartic", "raised_cosine"]
    for prof in profiles:
        if time.monotonic() + 60 > deadline:
            break
        a_p = _make_profile(n_for, prof)
        sc_p = _score(a_p)
        if sc_p < best_score + 0.3:
            p6_end = min(deadline - 40, time.monotonic() + 50)
            try:
                a_p2, sc_p2 = _lbfgs_polish(a_p, p6_end, min_time_remaining=15, verbose=False)
                if sc_p2 < best_score:
                    best_score = sc_p2
                    best_seq = a_p2.copy()
                    print(f"  {prof}: {best_score:.10f}")
            except Exception:
                pass
    
    # === Phase 7: Final polish ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 20 < deadline:
        p7_end = min(deadline - 5, time.monotonic() + 80)
        try:
            a7, sc7 = _lbfgs_polish(best_seq, p7_end, min_time_remaining=10, verbose=False)
            if sc7 < best_score:
                best_score = sc7
                best_seq = a7.copy()
        except Exception:
            pass
        
        if time.monotonic() + 15 < deadline:
            p7b_end = min(deadline - 3, time.monotonic() + 30)
            try:
                a7b, sc7b = _adaptive_kkt_refine(best_seq, p7b_end, min_time_remaining=5, verbose=False)
                if sc7b < best_score:
                    best_score = sc7b
                    best_seq = a7b.copy()
            except Exception:
                pass
    
    # Final validation
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    
    # Final score check
    sc_final = _score(final)
    print(f"\n{'='*60}")
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    
    return [float(x) for x in final]
