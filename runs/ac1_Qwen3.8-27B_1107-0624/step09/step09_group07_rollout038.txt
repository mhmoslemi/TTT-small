import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _topk_smooth_obj_grad(z, beta, K, n):
    z_max = z.max()
    a = np.exp(z - z_max)
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


class _SmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _topk_smooth_obj_grad(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _topk_smooth_obj_grad(z, self.beta, self.K, self.n)[1]


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=False):

    a = _normalize(a_init.copy())
    current_sc = _score(a)

    if max_c < 1e-15:
        return current_a

        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.008)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    if aggressive:
        schedule = [
            (1.0, 0.08), (0.7, 0.06), (0.5, 0.04), (0.3, 0.03),

        ]
    for beta_mult, K_frac in schedule:

            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        beta = base_beta * beta_mult
        K = max(5, int((2 * n - 1) * K_frac))
        est_time = 0.005 * max(1.0, n / 4096.0)

                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))

def _spectral_smooth(a_init, deadline, min_time_remaining=30.0):
    """Spectral smoothing: iteratively modify the power spectrum to flatten convolution."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    N = _next_pow2(2 * n - 1)
    alpha = 0.3
    no_improve = 0
    max_no_improve = 50
    iteration = 0
    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        c = np.fft.irfft(P, N)[:2 * n - 1]
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            alpha = min(0.8, alpha * 1.1)
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                alpha *= 0.7
                if alpha < 0.01:
                    break
                no_improve = 0
        c_mean = c.mean()
        ratio = c / (c_mean + 1e-30)
        ratio_clipped = np.clip(ratio, 0.5, 3.0)
        P_mod = P * (1.0 - alpha * (ratio_clipped - 1.0))
        P_mod = np.maximum(P_mod, 0.0)
        P_mod_sum = P_mod.sum()
        if P_mod_sum < 1e-15:
            break
        P_mod = P_mod * (P.sum() / P_mod_sum)
        mag_new = np.sqrt(P_mod)
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < current_sc:
            a = a_new
        else:
            blend = 0.5
            a = blend * a_new + (1 - blend) * a
            a = np.maximum(a, 0.0)
            a = a / a.sum()
    return best_a, best_sc

                a[j] = a_save_j - d
            if a[i] < -1e-15 or a[j] < -1e-15:
                a[i] = a_save_i
                a[j] = a_save_j
                break
            a[i] = max(0.0, a[i])
            a[j] = max(0.0, a[j])
            sc_trial = _score(a)
            if sc_trial < best_sc - 1e-14:
                best_sc = sc_trial
                best_a = a.copy()
                no_improve = 0
                break
            else:
                a[i] = a_save_i
                a[j] = a_save_j
        else:
            a[i] = a_save_i
            a[j] = a_save_j
            no_improve += 1
        if no_improve > max_no_improve:
            # Re-evaluate peak and updating contributing set
            c = _full_convolve(a, a)
            c_max = float(c.max())
            k_peak = int(np.argmax(c))
            i_lo = max(0, k_peak - (n - 1))
            i_hi = min(n - 1, k_peak)
            contributing = list(range(i_lo, i_hi + 1))
            no_improve = 0
            max_no_improve = int(max_no_improve * 0.7)
            if max_no_improve < 100:
                break
    return best_a, best_sc


def _lp_step(a_init, step_limit, deadline, min_time_remaining=10.0):
    """LP-based proximal step: solve linearized problem with box constraint."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    c_0 = _full_convolve(a, a)
    c_max_0 = float(c_0.max())
    # Build the constraint matrix for the LP
    # min t s.t. c_0[k] + 2*conv(a, delta)[k] <= t, sum(delta)=0, |delta[i]| <= step_limit
    # Variables: delta[0..n-1], t
    # Using scipy linprog
    # Actually, this LP is too large for n=16384. Let's use a simplified version:
    # Only enforce constraints for the top M peaks
    M = min(20, 2 * n - 1)
    c_sorted = np.argsort(c_0)[::-1]
    top_M = c_sorted[:M]
    # Build A_ub: for each top peak k, the constraint is:
    # sum_i (2*a[i] if i+j=k for some j) * delta[j] - t <= -c_0[k]
    # conv(a, delta)[k] = sum_i a[i]*delta[k-i] (where 0<=i<n, 0<=k-i<n)
    # So coeff of delta[j] in conv(a,delta)[k] is a[k-j] if 0<=k-j<n
    # Full constraint: 2*sum_j a[k-j]*delta[j] - t <= -c_0[k]
    A_ub = np.zeros((M, n + 1))
    b_ub = np.zeros(M)
    for m_idx, k in enumerate(top_M):
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                A_ub[m_idx, j] = 2.0 * a[kj]
        A_ub[m_idx, n] = -1.0
        b_ub[m_idx] = -c_0[k]
    # Box constraint: |delta[i]| <= step_limit
    # Add as: delta[i] <= step_limit and -delta[i] <= step_limit
    I_mat = np.eye(n)
    A_box = np.vstack([I_mat, -I_mat])
    b_box = np.full(2 * n, step_limit)
    A_ub_full = np.vstack([A_ub, A_box])
    b_ub_full = np.concatenate([b_ub, b_box])
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    try:
        result = minimize(
            lambda x: x[n],  # minimize t
            np.zeros(n + 1),
            jac=lambda x: c_obj,
            method='SLSQP',
            bounds=[(−step_limit, step_limit)] * n + [(0, None)],
            options={'maxiter': 200, 'ftol': 1e-12}
        )
        if result.success and result.fun < c_max_0:
            delta = result.x[:n]
            delta -= delta.mean()
            a_new = a + delta
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 0.01:
                a_new = a_new / s
                return a_new
    except Exception:
        pass
    return a


def _lp_step_simple(a_init, step_limit):
    """Simple LP step using only top peaks and SLSQP (faster, less constrained)."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    c_0 = _full_convolve(a, a)
    c_max_0 = float(c_0.max())
    M = min(10, 2 * n - 1)
    c_sorted = np.argsort(c_0)[::-1]
    top_M = c_sorted[:M]
    # Build constraint: for each top peak, gradient dot delta <= reduction_target
    reductions = c_0[top_M] - c_max_0 + 0.1 * c_max_0
    # Actually we want: c_0[k] + grad_c_k . delta <= c_max_0 (keep same max)
    # Or better: c_0[k] + grad_c_k . delta <= c_max_new where c_max_new < c_max_0
    target = c_max_0 * 0.99  # Try to reduce by 1%
    constraints = []
    for k in top_M:
        g_k = _grad_maxconv(np.ones(n) * 0 + a, np.zeros(2 * n - 1))
        # Recompute properly
        w_k = np.zeros(2 * n - 1)
        w_k[k] = 1.0
        g_k = _grad_maxconv(a, w_k)
        g_k_centered = g_k - g_k.mean()
        constraints.append({'type': 'ineq',
                           'fun': lambda d, gk=g_k_centered, t=target, k=k:
                              (c_0[k] + float(np.dot(gk, d))) - t})
    if not constraints:
        return a
    from scipy.optimize import minimize as sp_min
    res = sp_min(
        lambda d: 0.5 * float(np.dot(d, d)),
        np.zeros(n),
        constraints=constraints,
        bounds=[(-step_limit, step_limit)] * n,
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-10}
    )
    if res.success:
        delta = res.x
        delta -= delta.mean()
        a_new = a + delta
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0.01:
            a_new = a_new / s
            sc = _score(a_new)
            if sc < _score(a):
                return a_new
    return a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.4 + 0.6 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "optimal_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.5 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2) + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "super_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.where(np.abs(t - 0.5) < 0.3, 1.0, 0.3)
        a = a + 0.01 * np.random.default_rng(0).standard_normal(n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    """Fine active-set polish with adaptive tolerance."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.003
    tol = 0.003
    no_improve = 0
    n_2nm1 = 2 * n - 1
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        w = np.zeros(n_2nm1)
        w[active] = 1.0 / n_active
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break
        alpha = min(lr, 0.3 / (gnorm + 1e-15))
        improved = False
        for ls in range(35):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                tol = max(0.0001, tol * 0.6)
                lr *= 0.7
            if no_improve > 400:
                a = best_a.copy()
                tol = max(0.0001, tol * 0.5)
                lr = 0.002
                no_improve = 0
        else:
            if tol > 0.0001:
                tol = max(0.0001, tol * 0.96)
            lr = min(0.008, lr * 1.01)
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining spectral smoothing, Nesterov multiplicative
    weights, L-BFGS with continuation, coordinate descent, and fine polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, tapered n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Quick L-BFGS on parent ===
    print(f"\n[P1] Quick L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        a1 = _run_lbfgs(best_seq, min(deadline - 550, time.monotonic() + 120),
                        min_time_remaining=10)
        sc1 = _score(a1)
        if sc1 < best_score:
            best_seq = a1.copy()
            best_score = sc1
            print(f"  P1 BEST: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  P1 no improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Spectral smoothing ===
    print(f"\n[P2] Spectral smoothing (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 450:
        a2, sc2 = _spectral_smooth(best_seq, min(deadline - 400, time.monotonic() + 180),
                                   min_time_remaining=20)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  P2 BEST: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  P2 no improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Nesterov multiplicative weights ===
    print(f"\n[P3] Nesterov MW (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        a3, sc3 = _nesterov_mult_weights(best_seq, min(deadline - 300, time.monotonic() + 200),
                                         min_time_remaining=20)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  P3 BEST: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  P3 no improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Aggressive L-BFGS ===
    print(f"\n[P4] Aggressive L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        a4 = _run_lbfgs(best_seq, min(deadline - 200, time.monotonic() + 250),
                        min_time_remaining=20, aggressive=True)
        sc4 = _score(a4)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  P4 BEST: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  P4 no improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Spectral smoothing round 2 ===
    print(f"\n[P5] Spectral smoothing 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a5, sc5 = _spectral_smooth(best_seq, min(deadline - 160, time.monotonic() + 120),
                                   min_time_remaining=15)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  P5 BEST: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Coordinate descent ===
    print(f"\n[P6] Coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        a6, sc6 = _coordinate_descent(best_seq, min(deadline - 120, time.monotonic() + 130),
                                      min_time_remaining=15)
        if sc6 < best_score:
            best_seq = a6.copy()
            best_score = sc6
            print(f"  P6 BEST: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  P6 no improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Multi-restart exploration ===
    print(f"\n[P7] Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 120:
        # Perturbed restarts
        restart_sigmas = [0.02, 0.04, 0.06, 0.1, 0.03, 0.08, 0.05, 0.12]
        n_restart = min(6, max(2, int((deadline - 120 - time.monotonic()) / 35)))
        for i in range(n_restart):
            if time.monotonic() > deadline - 60:
                break
            sigma = restart_sigmas[i % len(restart_sigmas)]
            a_pert = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs(a_pert, min(deadline - 50, time.monotonic() + 30),
                             min_time_remaining=5)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  P7 BEST: {best_score:.10f} [restart_{i}, sig={sigma}] "
                      f"t={time.monotonic()-t_start:.1f}s")
        # Different n values
        n_main = len(best_seq)
        n_options = [n_main // 2, n_main * 3 // 2, n_main * 5 // 4]
        n_options = [x for x in n_options if x != n_main and 4096 <= x <= 49152]
        for n_try in n_options[:2]:
            if time.monotonic() > deadline - 40:
                break
            a_res = _resample(best_seq, n_try)
            a_opt = _run_lbfgs(a_res, min(deadline - 30, time.monotonic() + 35),
                               min_time_remaining=5)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  P7 BEST: {best_score:.10f} [n={n_try}] "
                      f"t={time.monotonic()-t_start:.1f}s")
        # Different profiles
        profiles = ["super_gauss", "plateau", "cos_taper"]
        for prof_kind in profiles:
            if time.monotonic() > deadline - 30:
                break
            a_prof = _make_profile(len(best_seq), prof_kind)
            a_p = _run_lbfgs(a_prof, min(deadline - 20, time.monotonic() + 25),
                             min_time_remaining=5)
            sc_p = _score(a_p)
            if sc_p < best_score:
                best_seq = a_p.copy()
                best_score = sc_p
                print(f"  P7 BEST: {best_score:.10f} [profile={prof_kind}] "
                      f"t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final fine polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _fine_polish(best_seq, deadline - 2, min_time_remaining=2)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  P8 BEST: {best_score:.10f} [polish] t={time.monotonic()-t_start:.1f}s")
        # Tiny L-BFGS
        remaining = deadline - 2 - time.monotonic()
        if remaining > 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            c_sorted = np.sort(c_best)[::-1]
            gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
            beta_fin = max(gap * 0.01, mb * 0.0001, 1e-10)
            K_fin = max(5, int(n * 0.002))
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _SmoProb(beta_fin, K_fin, n)
            max_iter = min(5000, max(30, int(remaining / 0.02)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-15, 'maxcor': 200})
                a_fin2 = np.exp(result.x - result.x.max())
                a_fin2 = a_fin2 / a_fin2.sum()
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  P8 BEST: {best_score:.10f} [final_lbfgs] "
                          f"t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        # Return in a form that evaluate_sequence can handle
        # Scale so sum is reasonable (the evaluator normalizes by sum^2,
        # so scale doesn't matter, but keep values in [0, 1000])
        return [float(x) for x in final]
    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
