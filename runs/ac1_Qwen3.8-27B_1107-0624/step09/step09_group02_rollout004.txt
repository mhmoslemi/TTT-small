def _next_pow2(n):

    fa = np.fft.rfft(a, N)

            else:
                def obj_func(z_local):
                    return _obj_grad_smoothed(z_local, n, beta)[0]

                def grad_func(z_local):
                    return _obj_grad_smoothed(z_local, n, beta)[1]

            result = minimize(
                obj_func, z,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax_z(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                stagnation_count = 0
            else:
                stagnation_count += 1
                # Try to escape: small perturbation
                if stagnation_count > 5:
                    z = z + 0.01 * np.random.standard_normal(n)
                    z = z - z.mean()
                    stagnation_count = 0
        except Exception:
            pass

    return best_a


def _peak_shave(a_init, deadline, min_time_remaining=10.0):
    """Direct peak shaving: iteratively reduce the top autocorrelation values
    using Newton-like updates on the top-k constraints."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    k_schedule = [3, 5, 8, 12, 20, 30, 50, 80, 120, 200, 400]

    for k_act in k_schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        k_act = min(k_act, 2 * n - 1)

        for inner in range(30):
            if time.monotonic() + min_time_remaining > deadline:
                break
            c = _full_convolve(a, a)
            c_max = float(c.max())
            # Top-k active set
            top_k = np.argsort(c)[::-1][:k_act]
            w = np.zeros(2 * n - 1)
            w[top_k] = 1.0 / k_act

            # Gradient
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W_val = float(np.dot(w, c))
            grad_a = 2.0 * (h - W_val)

            # Line search in log-space
            gnorm = np.linalg.norm(grad_a)
            if gnorm < 1e-15:
                break

            # Step in negative gradient direction, maintain positivity via log-space
            z = np.log(np.maximum(a, 1e-30))
            lr = 0.1 / (gnorm + 1e-10)
            improved = False
            for ls in range(20):
                z_try = z - lr * grad_a * a  # multiplicative update
                a_try = _softmax_z(z_try)
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max * (1 - 1e-10):
                    a = a_try
                    z = z_try
                    sc = _score(a)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()
                    improved = True
                    break
                lr *= 0.5
                if lr < 1e-15:
                    break

            if not improved:
                break

    return best_a, best_sc


def _spectral_init(n):
    """Initialize sequence to have a flat power spectrum (for flat autocorrelation).
    Uses a randomized phase approach."""
    # Target: |DFT(a)|^2 should be as flat as possible
    # Start with random phases and iterate
    rng = np.random.default_rng(12345)
    # Random complex spectrum with unit magnitude
    N = n
    phases = rng.uniform(0, 2 * np.pi, N)
    H = np.exp(1j * phases)
    a = np.fft.irfft(H, N)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _tapered_init(n, power=0.5):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _cosine_init(n):
    t = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _hamming_init(n):
    a = np.hamming(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax_z(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: adaptive L-BFGS with log-sum-exp smoothing,
    peak shaving, multi-length exploration, and spectral initialization.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")
        return

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _tapered_init(n_parent)
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Adaptive L-BFGS from parent (200s) ===
    print(f"\n[Phase 1] Adaptive L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    phase1_deadline = min(t_start + 200, deadline - 600)
    a1 = _run_lbfgs_adaptive(parent, phase1_deadline, min_time_remaining=20.0)
    update_best(a1, _score(a1), "lbfgs_parent")

    # === Phase 2: Peak shaving (100s) ===
    print(f"\n[Phase 2] Peak Shaving (t={time.monotonic() - t_start:.1f}s)")
    phase2_deadline = min(t_start + 300, deadline - 500)
    if best_seq is not None:
        a2, sc2 = _peak_shave(best_seq, phase2_deadline, min_time_remaining=15.0)
        update_best(a2, sc2, "peak_shave_1")

    # === Phase 3: L-BFGS with regularization (150s) ===
    print(f"\n[Phase 3] L-BFGS + Reg (t={time.monotonic() - t_start:.1f}s)")
    phase3_deadline = min(t_start + 450, deadline - 400)
    if best_seq is not None:
        a3 = _run_lbfgs_adaptive(best_seq, phase3_deadline, min_time_remaining=15.0, use_reg=True)
        update_best(a3, _score(a3), "lbfgs_reg")

    # === Phase 4: Multi-length exploration (250s) ===
    print(f"\n[Phase 4] Multi-n Exploration (t={time.monotonic() - t_start:.1f}s)")
    phase4_deadline = min(t_start + 700, deadline - 300)
    n_tries = [65536, 49152, 24576, 16384, 8192]
    n_tries = [n for n in n_tries if n != n_parent]

    for n_try in n_tries:
        if time.monotonic() > phase4_deadline - 30:
            break
        # Initialize by resampling current best
        if best_seq is not None:
            a_init_n = _resample(best_seq, n_try)
        else:
            a_init_n = _tapered_init(n_try)

        remaining = phase4_deadline - 30 - time.monotonic()
        if remaining < 20:
            break

        a_n = _run_lbfgs_adaptive(a_init_n, t_start + time.monotonic() - t_start + remaining,
                                   min_time_remaining=10.0)
        sc_n = _score(a_n)
        update_best(a_n, sc_n, f"n_{n_try}_lbfgs")

        # Quick peak shave on the new length
        if sc_n < best_score + 0.01 and time.monotonic() < phase4_deadline - 15:
            a_ns, sc_ns = _peak_shave(a_n, phase4_deadline, min_time_remaining=8.0)
            update_best(a_ns, sc_ns, f"n_{n_try}_shave")

    # === Phase 5: Different initializations (150s) ===
    print(f"\n[Phase 5] Different Inits (t={time.monotonic() - t_start:.1f}s)")
    phase5_deadline = min(t_start + 850, deadline - 200)
    inits = [
        ("hamming", lambda n: _hamming_init(n)),
        ("cosine", lambda n: _cosine_init(n)),
        ("tapered_1", lambda n: _tapered_init(n, power=1.0)),
        ("spectral", lambda n: _spectral_init(n)),
    ]

    for name, init_fn in inits:
        if time.monotonic() > phase5_deadline - 20:
            break
        a_init = init_fn(n_parent)
        # Quick L-BFGS
        rem = phase5_deadline - 20 - time.monotonic()
        if rem < 15:
            break
        a_p = _run_lbfgs_adaptive(a_init, t_start + (time.monotonic() - t_start) + rem,
                                   min_time_remaining=5.0)
        update_best(a_p, _score(a_p), f"init_{name}")

    # === Phase 6: Perturbation + L-BFGS restarts (150s) ===
    print(f"\n[Phase 6] Perturb + Restart (t={time.monotonic() - t_start:.1f}s)")
    phase6_deadline = min(t_start + 1000, deadline - 150)
    for trial in range(5):
        if time.monotonic() > phase6_deadline - 20:
            break
        if best_seq is None:
            break
        sigma = [0.02, 0.01, 0.03, 0.005, 0.015][trial]
        a_pert = _perturb(best_seq, sigma, rng)
        rem = phase6_deadline - 20 - time.monotonic()
        if rem < 15:
            break
        a_p = _run_lbfgs_adaptive(a_pert, t_start + (time.monotonic() - t_start) + rem,
                                   min_time_remaining=5.0)
        update_best(a_p, _score(a_p), f"perturb_{trial}")

    # === Phase 7: Final tight L-BFGS (100s) ===
    print(f"\n[Phase 7] Final Tight L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    phase7_deadline = min(t_start + 1100, deadline - 60)
    if best_seq is not None and time.monotonic() < phase7_deadline:
        a7 = _run_lbfgs_adaptive(best_seq, phase7_deadline, min_time_remaining=10.0)
        update_best(a7, _score(a7), "final_lbfgs")

    # === Phase 8: Final peak shave (60s) ===
    print(f"\n[Phase 8] Final Peak Shave (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a8, sc8 = _peak_shave(best_seq, deadline - 10, min_time_remaining=5.0)
        update_best(a8, sc8, "final_shave")

    # === Phase 9: Ultra-tight L-BFGS finish (remaining time) ===
    print(f"\n[Phase 9] Ultra-tight Finish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a9 = _run_lbfgs_adaptive(best_seq, deadline - 5, min_time_remaining=3.0)
        update_best(a9, _score(a9), "ultra_final")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _tapered_init(8192)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        # Verify with evaluate_sequence
        try:
            eval_sc = evaluate_sequence([float(x) for x in final])
            print(f"EVALUATE: score={eval_sc:.10f}")
        except Exception as e:
            print(f"Evaluation error: {e}")
        return [float(x) for x in final]

    fallback = _tapered_init(16384)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
