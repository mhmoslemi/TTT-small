import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    c = _full_convolve(a_norm, a_norm)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

    """Gradient of soft_max(conv(a,a)) w.r.t. a."""
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)

    best_a = a.copy()

    no_improve = 0
    max_no_improve = 500

        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        # Compute soft weights
        w = np.exp(-(c - c_max + 1e-15) / sigma)
        w = w / w.sum()  # normalize

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        n_support = int(support.sum())
        if n_support < 2:

        residual = grad[support] - lam

        # Step size with backtracking

        improved = False
        for trial_eta in [eta, eta*0.5, eta*0.25, eta*0.125, eta*0.0625, eta*0.03125]:

            if s_new < 1e-10:

            a_new = a_new / s_new

                a = a_new
                improved = True
                break

def _nesterov_subgradient(a_init, deadline, min_time_remaining=30.0, verbose=True):
    """Nesterov-accelerated subgradient method for the max convolution problem."""
    n = len(a_init)
    a = a_init.copy()
    s = float(a.sum())
    a = a / s
    y = a.copy()  # Nesterov momentum variable
    best_a = a.copy()
    best_sc = _score(a)

    t0 = time.monotonic()
    t_iter = 0
    alpha = 1.0  # step size parameter

    while time.monotonic() + min_time_remaining < deadline:
        c_y = _full_convolve(y, y)
        c_max_y = float(c_y.max())

        # Subgradient at y: pick the argmax of conv(y,y)
        k_star = int(np.argmax(c_y))

        # Gradient of c[k_star] w.r.t. y
        grad = np.zeros(n)
        j_lo = max(0, k_star - (n - 1))
        j_hi = min(n - 1, k_star)
        for j in range(j_lo, j_hi + 1):
            grad[j] = 2.0 * y[k_star - j]

        # Normalize gradient
        grad_centered = grad - np.mean(grad)
        g_norm_sq = float(np.dot(grad_centered, grad_centered))
        if g_norm_sq < 1e-20:
            break

        # Nesterov step
        t_iter += 1
        beta_n = (1.0 + np.sqrt(1.0 + 4.0 * t_iter * t_iter)) / 2.0
        beta_prev = (1.0 + np.sqrt(1.0 + 4.0 * (t_iter - 1) * (t_iter - 1))) / 2.0

        step = alpha / np.sqrt(t_iter + 1)
        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        y = a_new + (beta_prev / beta_n - 1.0) * (a_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-10:
            y = a_new
        else:
            y = y / s_y

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()

        a = a_new
        alpha *= 0.9999  # slow decay

        if verbose and t_iter % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    nesterov: sc={best_sc:.12f}, t={t_iter}, t={elapsed:.1f}s")

    return best_a, best_sc

"""

Key improvements over previous:
1. L-BFGS with finer beta schedule and extended iterations at small beta
2. Soft active-set method with adaptive sigma and Armijo backtracking
3. Averaged subgradient with momentum for global mixing
4. Block coordinate descent for final polish

6. Careful time management to maximize polishing

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    c = _full_convolve(a_norm, a_norm)
    return float(2.0 * n * c.max())


def _score_unnormalized(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    c = _full_convolve(a_norm, a_norm)
    return float(2.0 * n * c.max()), float(c.max()), c


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_soft(a, beta, n):
    """Gradient of soft_max(conv(a,a)) w.r.t. a using structure."""
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)


class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs_phased(a_init, deadline, min_time_remaining=30.0, verbose=True):
    """L-BFGS with aggressive adaptive beta schedule and extended small-beta phase."""
    n = len(a_init)
    a = a_init.copy()
    s = float(a.sum())
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    # Compute initial gap
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c0_max * 0.004:
        gap = c0_max * 0.004

    # Fine beta schedule with emphasis on small values
    multipliers = np.array([
        0.2, 0.1, 0.05, 0.025, 0.0125,
        0.00625, 0.003125, 0.00156, 0.0008,
        0.0004, 0.0002, 0.0001, 0.00005,
        0.000025, 0.0000125, 0.00000625,
        0.000003, 0.0000015, 0.0000008,
        0.0000004, 0.0000002, 0.0000001
    ])
    betas = gap * multipliers

    t0 = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        size_factor = max(1.0, n / 8192.0)

        if beta < gap * 0.01:

        else:
            max_iter = min(5000, max(200, int(remaining / (est_iter_time * 15))))

        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 200,
                    'maxls': 50,
                    'iprint': -1,
                }
            )

            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))

                if verbose:

            else:
                if verbose and i < 8:
                    print(f"    beta={beta:.2e}: no improvement (best={best_sc:.12f})")
        except Exception as e:
            if verbose:
                print(f"    beta={beta:.2e}: error {e}")

    return best_a, best_sc


def _soft_active_set(a_init, deadline, min_time_remaining=30.0, verbose=True):
    """Active-set method with soft Gaussian weights and continuation.
    
    Uses a smooth weight function w[k] = exp(-(c[k]-c_max+eps)/sigma)
    that transitions from soft (wide sigma) to hard (narrow sigma).
    Armijo backtracking for step size.
    """
    n = len(a_init)
    a = a_init.copy()
    s = float(a.sum())
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    t0 = time.monotonic()
    # Initial sigma relative to the range of c
    c_init = _full_convolve(a, a)
    c_range = float(c_init.max()) - float(np.percentile(c_init, 10))
    if c_range < 1e-15:
        c_range = float(c_init.max()) * 0.01
    sigma = 0.1 * c_range
    no_improve = 0
    max_no_improve = 800

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        # Compute soft weights
        w = np.exp(-(c - c_max + 1e-18) / sigma)
        w_sum = w.sum()
        if w_sum < 1e-100:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum

        # Compute gradient from soft active set
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        # Identify support (generous threshold to include small but non-zero elements)
        support_thresh = 1e-12 * a.max()
        support = a > support_thresh
        n_support = int(support.sum())
        if n_support < 2:
            break

        # KKT target: grad should be constant on support
        lam = float(np.mean(grad[support]))
        residual = grad[support] - lam

        # Armijo backtracking line search
        r_norm = float(np.sqrt(np.dot(residual, residual)))
        if r_norm < 1e-18:
            # Gradient is already (nearly) constant on support
            # Try to reduce the max by scaling
            no_improve += 5
            sigma *= 0.9
            continue

        improved = False
        for trial_alpha in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625]:
            step = trial_alpha * r_norm
            scale = 1.0 / (a[support] + step + 1e-30)
            a_new = a.copy()
            a_new[support] = a[support] - step * residual / (r_norm + 1e-30)
            # Simplified: just subtract proportional to residual
            a_new[support] = a[support] - trial_alpha * residual
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc * (1.0 - 1e-10):
                a = a_new
                improved = True
                break

        if not improved:
            # Try proportional step
            max_res = float(np.max(np.abs(residual)))
            if max_res > 1e-18:
                for frac in [0.001, 0.0005, 0.0001, 0.00005]:
                    target_step = frac * max_res
                    a_new = a.copy()
                    a_new[support] -= target_step * residual / max_res
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new < 1e-10:
                        continue
                    a_new = a_new / s_new
                    sc_new = _score(a_new)
                    if sc_new < current_sc * (1.0 - 1e-10):
                        a = a_new
                        improved = True
                        break

        if improved:
            sigma *= 0.97
        else:
            sigma *= 0.85
            if sigma < c_range * 1e-6:
                # Use hard active set
                tol = 0.0005 * c_max
                active = c >= (c_max - tol)
                n_active = max(1, int(active.sum()))
                w = np.zeros(len(c))
                w[active] = 1.0 / n_active
                a_rev = a[::-1]
                wc = _full_convolve(w, a_rev)
                grad = 2.0 * wc[n - 1:2 * n - 1]

                        if s_new < 1e-10:
                            continue
                        a_new = a_new / s_new
                        sc_new = _score(a_new)
                        if sc_new < current_sc * (1.0 - 1e-10):
                            a = a_new
                            improved = True
                            break
                if not improved:
                    no_improve += 20

        if verbose and (no_improve % 200 == 0 or (improved and no_improve == 0 and (int((time.monotonic()-t0)*2) % 5 == 0))):
            elapsed = time.monotonic() - t0
            print(f"    soft_active: sc={best_sc:.12f}, sigma={sigma:.2e}, "

def _momentum_subgradient(a_init, deadline, min_time_remaining=30.0, verbose=True):
    """Averaged subgradient with Nesterov momentum.
    
    Uses soft-weighted average of all peak gradients for a smoother direction.
    """
    n = len(a_init)
    a = a_init.copy()
    s = float(a.sum())
    a = a / s
    y = a.copy()  # momentum variable
    best_a = a.copy()
    best_sc = _score(a)

    t0 = time.monotonic()
    t_iter = 0
    step_base = 0.01
    momentum = 0.9

    while time.monotonic() + min_time_remaining < deadline:
        c_y = _full_convolve(y, y)
        c_max_y = float(c_y.max())

        # Soft subgradient: weighted average over peaks
        sigma = 0.005 * c_max_y
        w = np.exp(-(c_y - c_max_y + 1e-18) / sigma)
        w = w / w.sum()

        # Gradient: conv(w, a_rev)[n-1:2n-1] * 2
        y_rev = y[::-1]
        wc = _full_convolve(w, y_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        # Center the gradient (to preserve sum)
        grad_centered = grad - np.mean(grad)
        g_norm = float(np.sqrt(np.dot(grad_centered, grad_centered)))
        if g_norm < 1e-20:
            break

        # Nesterov step
        t_iter += 1
        step = step_base / np.sqrt(1.0 + 0.1 * t_iter)
        raw_step = step * grad_centered / g_norm

        a_new = a - raw_step
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Nesterov momentum
        beta_t = (1.0 + np.sqrt(1.0 + 4.0 * t_iter)) / 2.0
        if t_iter > 1:
            beta_prev = (1.0 + np.sqrt(1.0 + 4.0 * (t_iter - 1))) / 2.0
            y = a_new + (beta_prev / beta_t - 1.0) * (a_new - a)
        else:
            y = a_new.copy()
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-10:
            y = a_new
        else:
            y = y / s_y

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()

        a = a_new
        step_base *= 0.99995

        if verbose and t_iter % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    momentum_subgrad: sc={best_sc:.12f}, t={t_iter}, t={elapsed:.1f}s")

    return best_a, best_sc


def _block_coordinate_descent(a_init, deadline, min_time_remaining=30.0, verbose=True):
    """Optimize blocks of coordinates using L-BFGS on the block.
    
    For a block of indices I, we optimize a[I] while keeping a[~I] fixed.
    The objective is max_k conv(a,a)[k], which for fixed a[~I] is a
    non-smooth convex function of a[I].
    """
    n = len(a_init)
    a = a_init.copy()
    s = float(a.sum())
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    block_size = max(16, n // 256)
    n_blocks = (n + block_size - 1) // block_size
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    passes = 0
    while time.monotonic() + min_time_remaining < deadline:
        passes += 1
        block_order = rng.permutation(n_blocks)

        for bi in block_order:
            if time.monotonic() + min_time_remaining > deadline:
                break

            start = bi * block_size
            end = min(n, start + block_size)
            if end - start < 2:
                continue

            block = a[start:end].copy()
            if block.sum() < 1e-15:
                continue

            # Optimize this block using the soft objective
            # We fix the rest and optimize the block
            rest = a.copy()
            rest[start:end] = 0

            def block_obj(x):
                a_trial = rest.copy()
                a_trial[start:end] = np.maximum(x, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    return 1e10
                a_t = a_trial / s_t
                c_t = _full_convolve(a_t, a_t)
                return float(c_t.max())

            def block_grad(x):
                a_trial = rest.copy()
                a_trial[start:end] = np.maximum(x, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    return np.zeros(end - start)
                a_t = a_trial / s_t
                c_t = _full_convolve(a_t, a_t)
                c_max = float(c_t.max())
                # Soft gradient
                beta = 0.001 * c_max
                w = np.exp(-(c_t - c_max + 1e-18) / beta)
                w = w / w.sum()
                a_t_rev = a_t[::-1]
                wc = _full_convolve(w, a_t_rev)
                grad_full = 2.0 * wc[n - 1:2 * n - 1]
                # Project to block
                grad_block = grad_full[start:end]
                # Account for normalization
                W = float(np.dot(grad_full, a_t))
                grad_normed = (grad_full - W * a_t) / s_t
                return grad_normed[start:end]

            x0 = block.copy()
            try:
                res = minimize(block_obj, x0, jac=block_grad, method='L-BFGS-B',
                              bounds=[(0.0, None)] * (end - start),
                              options={'maxiter': 50, 'ftol': 1e-15, 'gtol': 1e-12})
                x_new = np.maximum(res.x, 0.0)
                a_new = a.copy()
                a_new[start:end] = x_new
                sc_new = _score(a_new)
                if sc_new < best_sc - 1e-14:
                    best_sc = sc_new
                    best_a = a_new.copy()
                    a = a_new
            except Exception:
                pass

        if verbose and passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    block_cd: pass={passes}, sc={best_sc:.12f}, t={elapsed:.1f}s")

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

        p3_time = min(250, (budget_s - 10) * 0.18)
        p3_end = min(deadline - 200, time.monotonic() + p3_time)
        a3, sc3 = _run_lbfgs_phased(best_seq, p3_end, min_time_remaining=80, verbose=True)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3 result: {best_score:.12f}")

    # === Phase 4: Soft active-set again (tighter) ===
    print(f"\n[P4] Soft active-set round 2 (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p4_time = min(150, (budget_s - 10) * 0.12)
        p4_end = min(deadline - 100, time.monotonic() + p4_time)
        a4, sc4 = _soft_active_set(best_seq, p4_end, min_time_remaining=30, verbose=True)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4 result: {best_score:.12f}")

    # === Phase 5: Momentum subgradient ===
    print(f"\n[P5] Momentum subgradient (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p5_time = min(100, (budget_s - 10) * 0.06)
        p5_end = min(deadline - 80, time.monotonic() + p5_time)
        a5, sc5 = _momentum_subgradient(best_seq, p5_end, min_time_remaining=20, verbose=True)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  P5 result: {best_score:.12f}")
        # Quick L-BFGS after subgradient
        if time.monotonic() + 20 < deadline:
            p5b_end = min(deadline - 60, time.monotonic() + 60)
            a5b, sc5b = _run_lbfgs_phased(best_seq, p5b_end, min_time_remaining=20, verbose=False)
            if sc5b < best_score:
                best_score = sc5b
                best_seq = a5b.copy()
                print(f"  P5b L-BFGS: {best_score:.12f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n exploration (t={time.monotonic()-t_start:.0f}s)")
    n_options = [n_main + 4096, n_main - 4096, n_main + 8192]
    n_options = [x for x in n_options if x >= 4096 and x <= 32768 and x != n_main]
    for n_try in n_options:
        if time.monotonic() + 60 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        print(f"  n={n_try}: initial score={sc_r:.12f}")
        if sc_r < best_score + 0.1:
            p6_end = min(deadline - 50, time.monotonic() + 80)
            a6, sc6 = _run_lbfgs_phased(a_r, p6_end, min_time_remaining=20, verbose=False)
            if sc6 < best_score:
                best_score = sc6
                best_seq = a6.copy()
                print(f"  n={n_try}: improved to {best_score:.12f}")

    # === Phase 7: Final polish ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 15 < deadline:
        p7_end = min(deadline - 5, time.monotonic() + 80)
        a7, sc7 = _run_lbfgs_phased(best_seq, p7_end, min_time_remaining=5, verbose=False)
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()
            print(f"  P7 L-BFGS: {best_score:.12f}")

        if time.monotonic() + 10 < deadline:
            p7b_end = min(deadline - 3, time.monotonic() + 30)
            a7b, sc7b = _soft_active_set(best_seq, p7b_end, min_time_remaining=3, verbose=False)
            if sc7b < best_score:
                best_score = sc7b
                best_seq = a7b.copy()
                print(f"  P7b active: {best_score:.12f}")

    # Final validation
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"\n{'='*60}")
    print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
