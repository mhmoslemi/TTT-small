      k = 1
      while k < n:
          k <<= 1
      return max(2, k)

      n, m = len(a), len(b)
      out_n = n + m - 1
      N = _next_pow2(out_n)
      fa = np.fft.rfft(a, N)
      fb = np.fft.rfft(b, N)
      r = np.fft.irfft(fa * fb, N)[:out_n]

      a = _normalize(a_init.copy())
      best_a = a.copy()
      best_sc = _score(a)

      active_frac = 0.05
      eta = 0.01  # Step size

      w_ema = None
      stall = 0
      max_stall = 100
      total_iter = 0

          total_iter += 1
          c = _full_convolve(a, a)
          c_max = float(c.max())
          current_sc = 2.0 * n * c_max  # Since sum(a) = 1
          
          if current_sc < best_sc - 1e-15:
              best_sc = current_sc
              best_a = a.copy()
              stall = 0
          else:
              stall += 1
          
          # Active set
          n_2nm1 = 2 * n - 1
          n_active = max(5, int(n_2nm1 * active_frac))
          top_idx = np.argpartition(c, -n_active)[-n_active:]
          active_vals = c[top_idx]
          
          # Weights: exponential of (c - c_max) / temperature
          # This gives more weight to peaks closer to the max
          temp = max(c_max * 0.01, 1e-10)
          w = np.zeros(n_2nm1)
          shifted = (active_vals - c_max) / temp
          e = np.exp(np.clip(shifted, -50, 0))
          w[top_idx] = e / e.sum()
          
          # Gradient
          g = _grad_maxconv(a, w)
          g_centered = g - g.mean()
          gnorm = np.linalg.norm(g_centered)
          if gnorm < 1e-18:
              break
          
          # MWU step: a_new = a * exp(-eta * g_centered) / Z
          # Adaptive eta: use Barzilai-Borwein or fixed
          # For MWU, a good choice is eta ~ 1 / max(|g_centered|)
          eta_adaptive = 0.5 / (np.max(np.abs(g_centered)) + 1e-10)
          eta_use = min(eta, eta_adaptive)  # Conservative step
          
          log_scale = -eta_use * g_centered
          log_scale = log_scale - log_scale.max()  # Numerical stability
          scale = np.exp(log_scale)
          a_new = a * scale
          s_new = a_new.sum()
          if s_new < 1e-10:
              break
          a_new = a_new / s_new
          
          # Check improvement
          c_new = _full_convolve(a_new, a_new)
          sc_new = 2.0 * n * float(c_new.max())
          
          if sc_new < current_sc:
              a = a_new

              eta = min(0.5, eta * 1.02)
              # Shrink active set
              active_frac = max(0.002, active_frac * 0.995)
          else:
              # Roll back, decrease step size
              eta = max(1e-6, eta * 0.5)
              active_frac = min(0.1, active_frac * 1.2)
          
          # Stall handling
          if stall > max_stall:

              eta = max(1e-6, eta * 0.3)
              stall = 0
              # Occasional restart from best
              if stall > max_stall * 3:  # This won't trigger since we reset stall
                  a = best_a.copy()
      
      return best_a, best_sc
