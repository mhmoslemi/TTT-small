def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _next_pow2(n):
    k = 1

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

        active = c >= (c_max - tol_frac * c_max)

            active[max(0, idx-1):min(len(c), idx+2)] = True

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, verbose=False):

        0.001, 0.0005, 0.00025, 0.000125, 0.0000625,

    t_level_start = time.monotonic()

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "decreasing":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 0.5 + 0.01
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.25) ** 2) + np.exp(-30.0 * (t - 0.75) ** 2) + 0.01
        return a / a.sum()
    elif kind == "spectrum_flat":
        # Try to create a sequence with flatter power spectrum
        # Use random phase in frequency domain, take inverse DFT, clip to non-negative
        rng = np.random.default_rng(12345)
        N = _next_pow2(2 * n)
        A = np.zeros(N // 2 + 1, dtype=complex)
        A[0] = np.sqrt(n)
        for k in range(1, N // 2 + 1):
            mag = 1.0 / np.sqrt(n)
            phase = rng.uniform(0, 2 * np.pi)
            A[k] = mag * np.exp(1j * phase)
        A_full = np.zeros(N, dtype=complex)
        A_full[:N // 2 + 1] = A
        A_full[N // 2 + 1:] = np.conj(A_full[1:N // 2][::-1])
        a_full = np.fft.ifft(A_full).real
        a = a_full[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            return np.ones(n) / n
        return a / s
    else:
        return np.ones(n) / n

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

        if sc_new <= sc + 1e-14:
            a = a_new
            sc = sc_new
            c = c_new
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
        
        # Update lambda (dual ascent)
        # Dual gradient for lambda_k: c[k] - t (where t = max c)
        c_max_new = float(c.max())
        dual_grad = c - c_max_new
        lambda_vec = np.maximum(lambda_vec + step_dual * dual_grad, 0.0)
        lsum = lambda_vec.sum()
        if lsum > 1e-20:
            lambda_vec = lambda_vec / lsum
        
        # Adapt dual step
        if sc_new < sc - 1e-12:
            step_dual = min(1.0, step_dual * 1.05)
        else:
            step_dual = max(0.01, step_dual * 0.8)
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t_start
            print(f"      DA iter {it}: sc={best_sc:.10f}, step_d={step_dual:.4f}, t={elapsed:.0f}s")
    
    return best_a, best_sc

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    P1: L-BFGS main optimization
    P2: Gradient flow with exact subgradient
    P3: Active-set Newton refinement
    P4: Dual ascent
    P5: L-BFGS round 2
    P6: Multi-n exploration
    P7: Final polish (active set + L-BFGS)
    """
    t_global_start = time.monotonic()
    deadline = t_global_start + max(1.0, float(budget_s) - 8.0)
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(16384, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n=16384: score={best_score:.10f}")
    
    n_main = len(best_seq)
    
    # === Phase 1: L-BFGS ===
    print(f"\n[P1] L-BFGS main (n={n_main}, t={time.monotonic()-t_global_start:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p1_end = min(t_global_start + 400, deadline - 400)
        a1, sc1 = _run_lbfgs(best_seq, p1_end, min_time_remaining=200, verbose=True)
        if sc1 < best_score:
            best_score = sc1
            best_seq = a1.copy()
            print(f"  P1: {best_score:.10f}")
    
    # === Phase 2: Gradient flow ===
    print(f"\n[P2] Gradient flow (t={time.monotonic()-t_global_start:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p2_end = min(deadline - 500, time.monotonic() + 250)
        a2, sc2 = _gradient_flow(best_seq, p2_end, min_time_remaining=100, verbose=True)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(f"  P2: {best_score:.10f}")
        # L-BFGS after gradient flow
        if time.monotonic() + 40 < deadline:
            p2b_end = min(p2_end + 80, deadline - 400)
            a2b, sc2b = _run_lbfgs(best_seq, p2b_end, min_time_remaining=60, verbose=False)
            if sc2b < best_score:
                best_score = sc2b
                best_seq = a2b.copy()
                print(f"  P2b: {best_score:.10f}")
    
    # === Phase 3: Active-set Newton ===
    print(f"\n[P3] Active-set Newton (t={time.monotonic()-t_global_start:.0f}s)")
    if time.monotonic() + 40 < deadline:
        p3_end = min(deadline - 300, time.monotonic() + 200)
        a3, sc3 = _newton_active_set(best_seq, p3_end, min_time_remaining=60, verbose=True)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3: {best_score:.10f}")
    
    # === Phase 4: Dual ascent ===
    print(f"\n[P4] Dual ascent (t={time.monotonic()-t_global_start:.0f}s)")
    if time.monotonic() + 40 < deadline:
        p4_end = min(deadline - 200, time.monotonic() + 120)
        a4, sc4 = _dual_ascent(best_seq, p4_end, min_time_remaining=50, verbose=False)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4: {best_score:.10f}")
    
    # === Phase 5: L-BFGS round 2 ===
    print(f"\n[P5] L-BFGS round 2 (t={time.monotonic()-t_global_start:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p5_end = min(deadline - 150, time.monotonic() + 200)
        a5, sc5 = _run_lbfgs(best_seq, p5_end, min_time_remaining=60, verbose=True)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  P5: {best_score:.10f}")
    
    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n (t={time.monotonic()-t_global_start:.0f}s)")
    n_options = [n_main // 2, int(n_main * 1.5), n_main * 2]
    n_options = sorted(set([x for x in n_options if 2048 <= x <= 65536 and x != n_main]))
    for n_try in n_options:
        if time.monotonic() + 80 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        p6_end = min(deadline - 100, time.monotonic() + 100)
        a6, sc6 = _run_lbfgs(a_r, p6_end, min_time_remaining=30, verbose=False)
        # Quick active set
        if time.monotonic() + 30 < p6_end:
            a6b, sc6b = _newton_active_set(a6, min(p6_end, time.monotonic() + 40), min_time_remaining=10, verbose=False)
            if sc6b < sc6:
                a6, sc6 = a6b, sc6b
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  n={n_try}: {best_score:.10f}")
    
    # === Phase 7: Final polish ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_global_start:.0f}s)")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        half = remaining * 0.5
        p7a_end = time.monotonic() + half
        a7, sc7 = _newton_active_set(best_seq, p7a_end, min_time_remaining=5, verbose=False)
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()
        
        if time.monotonic() + 10 < deadline:
            p7b_end = deadline - 3
            a7b, sc7b = _run_lbfgs(best_seq, p7b_end, min_time_remaining=3, verbose=False)
            if sc7b < best_score:
                best_score = sc7b
                best_seq = a7b.copy()
        
        # One more active set pass
        if time.monotonic() + 8 < deadline:
            a7c, sc7c = _newton_active_set(best_seq, deadline - 1, min_time_remaining=1, verbose=False)
            if sc7c < best_score:
                best_score = sc7c
                best_seq = a7c.copy()
    
    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_global_start:.1f}s")
    
    # Also verify with evaluate_sequence if available
    try:
        ev_sc = evaluate_sequence([float(x) for x in final])
        print(f"evaluate_sequence: {ev_sc:.10f}")
        if np.isfinite(ev_sc) and ev_sc < sc_final:
            sc_final = ev_sc
    except Exception:
        pass
    
    return [float(x) for x in final]
