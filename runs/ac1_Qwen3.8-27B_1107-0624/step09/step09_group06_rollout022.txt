def _next_pow2(n):

def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)

        gap = c_sorted_tmp[0] - c_sorted_tmp[min(K, len(c_sorted_tmp) - 1)]

            K = min(K_max, int(K * 1.3) + 1)
        
        c_idx = np.argsort(c)[::-1]

            j_hi = min(n - 1, k)

            j_lo2 = max(0, k)

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, alpha=0.5):
    """Power-law profile: a_j proportional to ((j+0.5)/(n-0.5))^alpha"""
    t = (np.arange(n) + 0.5) / (n - 0.5)
    a = t ** alpha
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_profile_symmetric(n, alpha=0.5):
    """Symmetric power-law: small at edges, large in center."""
    t = np.linspace(0, 1, n)
    a = (1.0 - 2.0 * np.abs(t - 0.5)) ** alpha + 0.1
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _try_initializations(n, budget):
    """Try several initializations and return the best."""
    candidates = []
    # Power-law profiles with different exponents
    for alpha in [0.0, 0.25, 0.5, 0.75, 1.0, 1.5]:
        a = _make_profile(n, alpha)
        candidates.append(a)
        a = _make_profile(n, -alpha)  # reversed
        candidates.append(a)
    # Symmetric profiles
    for alpha in [0.5, 1.0, 2.0]:
        candidates.append(_make_profile_symmetric(n, alpha))
    # Uniform
    candidates.append(np.ones(n) / n)
    
    best_a = candidates[0]
    best_sc = _score(best_a)
    for a in candidates[1:]:
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc
