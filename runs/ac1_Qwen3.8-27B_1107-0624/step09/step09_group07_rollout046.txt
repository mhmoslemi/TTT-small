def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    """Score assuming sum(a)=1, so score = 2*n*max_c."""
    n = len(a)

    return 2.0 * n * float(c.max())

def _topk_smoobj_grad(z, beta, K, n):

def _spectral_shaping(a_init, deadline, min_time_remaining=30.0):
    """Flatten the magnitude spectrum to reduce autocorrelation peak."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    N = _next_pow2(2 * n)
    alpha_values = [0.7, 0.5, 0.3, 0.6, 0.4, 0.8, 0.2, 0.5, 0.3, 0.4]
    alpha_idx = 0

    for it in range(200):
        if time.monotonic() + min_time_remaining < deadline:
            break

        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Preserve DC
        dc = mag[0]

        # Flatten non-DC components
        alpha = alpha_values[alpha_idx % len(alpha_values)]
        if it % 3 == 0:
            alpha_idx += 1

        mag_work = mag.copy()
        mag_work[0] = dc
        if len(mag_work) > 1:
            # Normalize non-DC to have same mean
            non_dc = mag_work[1:]
            if non_dc.max() > 1e-30:
                # Compress: raise to alpha (alpha < 1 compresses)
                scaled = non_dc / non_dc.max()
                shaped = scaled ** alpha
                # Scale back to preserve total energy approximately
                shaped = shaped * non_dc.max() * (non_dc.mean() / max(shaped.mean(), 1e-30))
                mag_work[1:] = shaped

        A_new = mag_work * np.exp(1j * phase)
        A_new[0] = dc  # ensure DC is preserved

        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Try mixing
            for t_mix in [0.3, 0.5, 0.7]:
                a_mix = (1 - t_mix) * a + t_mix * a_new
                a_mix = np.maximum(a_mix, 0.0)
                s_mix = a_mix.sum()
                if s_mix < 1e-10:
                    continue
                a_mix = a_mix / s_mix
                sc_mix = _score(a_mix)
                if sc_mix < best_sc - 1e-14:
                    best_sc = sc_mix
                    best_a = a_mix.copy()
                    a = a_mix.copy()
                    break

    return best_a, best_sc


def _active_set_newton(a_init, deadline, min_time_remaining=30.0):

    n = len(a_init)

    best_a = a.copy()

        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                break

        # Find active set: peaks within 1% of max
        tol = 0.01
        active = c >= c_max * (1 - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 50:
            # Take top 50
            top_idx = np.argsort(c)[::-1][:50]
            active = np.zeros(n_2nm1, dtype=bool)
            active[top_idx] = True
            n_active = 50

        # Weighted gradient
        active_vals = c[active]
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w = np.zeros(n_2nm1)
            w[active] = (active_vals - active_vals.min()) / spread + 0.05
            w = w / w.sum()
        else:
            w = np.zeros(n_2nm1)
            w[active] = 1.0 / n_active

        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break

        # Newton correction: approximate Hessian H = 2 * sum_k w[k] * B_k
        # H * d for a direction d can be computed efficiently
        # For the step direction, we use: d_newton = -H^{-1} g (approximately)
        # Since H is a sum of permutation matrices weighted by w,
        # H*d = 2 * sum_k w[k] * B_k * d
        # We approximate the inverse with a few power iterations (conjugate gradient)

        # For efficiency, just use a damped gradient step with momentum
        step = momentum * velocity - lr * g_centered
        velocity = step

        a_try = a + step
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()

        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            for ls in range(20):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-15:
                    improved = True
                    break
                step *= 0.5
                a_try = a + step
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    break
                a_try = a_try / s_try

        if improved:
            a = a_try
            lr = min(0.05, lr * 1.02)
            tol = max(0.002, tol * 0.98)
        else:
            velocity = np.zeros(n)
            lr = max(1e-5, lr * 0.6)
            tol = min(0.03, tol * 1.3)

    return best_a, best_sc

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        n_active = max(3, min(n_2nm1, int(n_2nm1 * active_frac)))
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_active]

        active_vals = c[active_idx]
        w = np.zeros(n_2nm1)
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w[active_idx] = (active_vals - active_vals.min()) / spread + 0.05
        else:
            w[active_idx] = 1.0 / n_active
        w = w / w.sum()

        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break

        if prev_a is not None and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g_centered - prev_g
            s_norm = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm > 1e-18 and abs(sy) > 1e-18:
                lr_bb = s_norm / abs(sy)
                lr = np.clip(lr_bb, lr_min, lr_max)

        velocity = momentum * velocity - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()

        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            for ls in range(20):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-15:
                    improved = True
                    break
                velocity = momentum * velocity * 0.5 - lr * 0.5 * g_centered
                a_try = a + velocity
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    velocity = np.zeros(n)
                    a_try = a
                    break
                a_try = a_try / s_try
                lr *= 0.7

        if improved:
            prev_a = a.copy()
            prev_g = g_centered.copy()
            a = a_try
            lr = min(lr_max, lr * 1.03)
            active_frac = max(0.001, active_frac * 0.97)
        else:
            velocity = np.zeros(n)
            lr = max(lr_min, lr * 0.6)
            active_frac = min(0.05, active_frac * 1.4)

        if stall > max_stall:
            active_frac = min(0.05, active_frac * 2.0)
            lr = max(lr_min, lr * 0.3)
            momentum = min(0.85, momentum + 0.05)
            stall = 0

        # Periodic L-BFGS restart
        if iteration % 400 == 0 and time.monotonic() - last_lbfgs_time > 100:
            if time.monotonic() + min_time_remaining + 60 < deadline:
                lbfgs_deadline = time.monotonic() + 50
                a_tmp = _run_lbfgs_adaptive(a, lbfgs_deadline, min_time_remaining=5)
                sc_tmp = _score(a_tmp)
                if sc_tmp < best_sc:
                    best_a = a_tmp.copy()
                    best_sc = sc_tmp
                    a = a_tmp.copy()
                    velocity = np.zeros(n)
                    prev_a = None
                    prev_g = None
                    last_lbfgs_time = time.monotonic()

    return best_a, best_sc


def _multi_scale_refine(a_init, deadline, min_time_remaining=30.0):
    """Optimize at coarser scale, upsample, and refine."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)

    # Try several coarser scales
    coarse_ns = [n // 4, n // 8]
    for cn in coarse_ns:
        if cn < 64:
            continue
        if time.monotonic() + min_time_remaining + 30 < deadline:
            continue
        # Downsample
        a_coarse = _resample(a_init, cn)
        # Optimize at coarse scale
        a_coarse_opt = _run_lbfgs_adaptive(a_coarse, min(deadline - min_time_remaining - 10, time.monotonic() + 25),
                                           min_time_remaining=5)
        # Upsample back
        a_up = _resample(a_coarse_opt, n)
        sc = _score(a_up)
        if sc < best_sc:
            best_a = a_up
            best_sc = sc
            # Refine at full scale
            a_ref = _run_lbfgs_adaptive(a_up, min(deadline - min_time_remaining, time.monotonic() + 20),
                                        min_time_remaining=5)
            sc_ref = _score(a_ref)
            if sc_ref < best_sc:
                best_a = a_ref
                best_sc = sc_ref

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "optimal_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.25) ** 2) + np.exp(-30.0 * (t - 0.75) ** 2) + 0.03
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "super_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _freq_perturb(a, rng, n_harmonics=5):
    """Perturb in frequency domain: add random phase rotations to low harmonics."""
    n = len(a)
    N = _next_pow2(2 * n)
    A = np.fft.rfft(a, N)
    # Add small random phase shifts to first few harmonics
    for k in range(1, min(n_harmonics + 1, len(A))):
        angle = rng.uniform(-0.1, 0.1)
        A[k] *= np.exp(1j * angle)
        if k < N - k:
            A[N - k] *= np.exp(-1j * angle)
    a_new = np.fft.irfft(A, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _fine_polish(a_init, deadline, min_time_remaining=5.0):
    """Very fine active-set polish."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.003
    tol = 0.003
    no_improve = 0
    n_2nm1 = 2 * n - 1

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        w = np.zeros(n_2nm1)
        w[active] = 1.0 / n_active

        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-17:
            break

        alpha = min(lr, 0.3 / (gnorm + 1e-15))
        improved = False
        for ls in range(30):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-16:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 10 == 0:
                tol = max(0.0001, tol * 0.8)
                lr *= 0.8
            if no_improve > 200:
                a = best_a.copy()
                tol = max(0.0001, tol * 0.5)
                lr = 0.002
                no_improve = 0
        else:
            if tol > 0.0001:
                tol = max(0.0001, tol * 0.96)
            lr = min(0.008, lr * 1.01)

    return best_a, best_sc


def _stochastic_peak_reduction(a_init, deadline, min_time_remaining=30.0, rng=None):
    """Stochastically select peaks to reduce, adding randomness to escape local minima."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n_2nm1 = 2 * n - 1

    lr = 0.005
    stall = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                break

        # Stochastic selection: with probability 0.7, use top 10; else random 10 from top 50
        if rng.random() < 0.7:
            n_sel = 10
            top_idx = np.argsort(c)[::-1][:n_sel]
        else:
            n_sel = 10
            top_50 = np.argsort(c)[::-1][:50]
            top_idx = rng.choice(top_50, size=n_sel, replace=False)

        w = np.zeros(n_2nm1)
        w[top_idx] = 1.0 / n_sel

        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break

        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        improved = False
        for ls in range(20):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if improved:
            lr = min(0.02, lr * 1.01)
        else:
            lr = max(1e-5, lr * 0.7)

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS, spectral shaping, active-set methods,
    stochastic peak reduction, multi-scale refinement, and fine polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                print(f"Loaded parent: n={len(parent)}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"Parent score: {best_score:.10f}, n={len(best_seq)}")
    else:
        n_init = 16384
        best_seq = _make_profile(n_init, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Init tapered n={n_init}, score={best_score:.10f}")

    n_main = len(best_seq)

    # === Phase 1: L-BFGS adaptive ===
    print(f"\n[P1] L-BFGS adaptive (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        a1 = _run_lbfgs_adaptive(best_seq, min(deadline - 450, time.monotonic() + 300),
                                 min_time_remaining=60)
        sc1 = _score(a1)
        if sc1 < best_score:
            best_seq = a1.copy()
            best_score = sc1
            print(f"  NEW BEST: {best_score:.10f} [lbfgs1] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Spectral shaping ===
    print(f"\n[P2] Spectral shaping (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a2, sc2 = _spectral_shaping(best_seq, min(deadline - 350, time.monotonic() + 150),
                                    min_time_remaining=30)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [spectral] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: L-BFGS after spectral ===
    print(f"\n[P3] L-BFGS after spectral (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        a3 = _run_lbfgs_adaptive(best_seq, min(deadline - 300, time.monotonic() + 200),
                                 min_time_remaining=50)
        sc3 = _score(a3)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [lbfgs2] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Active-set Newton ===
    print(f"\n[P4] Active-set Newton (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        a4, sc4 = _active_set_newton(best_seq, min(deadline - 200, time.monotonic() + 150),
                                     min_time_remaining=30)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [newton] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Subgradient BB ===
    print(f"\n[P5] Subgradient BB (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a5, sc5 = _subgradient_bb(best_seq, min(deadline - 150, time.monotonic() + 120),
                                  min_time_remaining=30)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [subgrad] t={time.monotonic()-t_start:.1f}s")
        else:

    # === Phase 6: Stochastic peak reduction ===
    print(f"\n[P6] Stochastic peak reduction (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        a6, sc6 = _stochastic_peak_reduction(best_seq, min(deadline - 100, time.monotonic() + 80),
                                             min_time_remaining=20, rng=rng)
        if sc6 < best_score:
            best_seq = a6.copy()
            best_score = sc6

    print(f"\n[P7] Multi-scale (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 120:
        a7, sc7 = _multi_scale_refine(best_seq, min(deadline - 80, time.monotonic() + 60),

            best_seq = a7.copy()

    print(f"\n[P8] Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        restart_budget = min(150, (deadline - 70 - time.monotonic()) * 0.6)
        n_restart = max(2, int(restart_budget / 30))
        for i in range(n_restart):
            if time.monotonic() > deadline - 50:
                break
            if i % 3 == 0:
                sigma = [0.02, 0.04, 0.06, 0.03][ (i // 3) % 4]
                a_pert = _perturb(best_seq, sigma, rng)
            elif i % 3 == 1:
                a_pert = _freq_perturb(best_seq, rng, n_harmonics=3 + i)
            else:
                # Mix with a different profile
                prof_kind = ["cos_taper", "super_taper", "bimodal"][i % 3]
                a_prof = _make_profile(n_main, prof_kind)
                mix_t = 0.1 + 0.1 * (i % 3)
                a_pert = (1 - mix_t) * best_seq + mix_t * a_prof
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()

            a_r = _run_lbfgs_adaptive(a_pert, min(deadline - 40, time.monotonic() + 25),
                                      min_time_remaining=5)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.10f} [restart_{i}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 9: Spectral + L-BFGS combined ===
    print(f"\n[P9] Spectral + L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        a_sp, sc_sp = _spectral_shaping(best_seq, min(deadline - 40, time.monotonic() + 30),
                                        min_time_remaining=5)
        if sc_sp < best_score:
            best_seq = a_sp.copy()
            best_score = sc_sp
            print(f"  NEW BEST: {best_score:.10f} [spec2] t={time.monotonic()-t_start:.1f}s")
        a_final_lb = _run_lbfgs_adaptive(best_seq, min(deadline - 20, time.monotonic() + 20),
                                         min_time_re
