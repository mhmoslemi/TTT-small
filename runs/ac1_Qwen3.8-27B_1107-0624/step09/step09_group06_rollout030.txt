```python
Spectral-domain iterative refinement for minimizing peak auto-convolution ratio.
Uses Gerchberg-Saxton-style spectral flattening combined with L-BFGS inner loop
and multi-scale refinement.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _project_nonneg_sum1(a):
    """Project a to {x >= 0, sum(x) = 1}."""
    return _project_simplex(a)

def _project_simplex(x):
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)
```

```python
import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)

    """Gerchberg-Saxton style spectral flattening."""
    n = len(a)

        a = a / s
```

```python
Spectral-domain iterative refinement for minimizing peak auto-convolution ratio.
Combines: (1) theoretically-optimal spectrum initialization, (2) Gerchberg-Saxton
spectral flattening, (3) verified-correct L-BFGS with log-sum-exp smoothing,

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _spectral_init(n):
    """Initialize using theoretically optimal power spectrum (flat target convolution)."""
    num = 2 * n - 1
    N = _next_pow2(num)
    c_target = np.zeros(N)
    c_target[:num] = 1.0 / num
    C = np.fft.fft(c_target)
    C_real = np.maximum(C.real, 0.0)
    A_mag = np.sqrt(C_real)
    A = A_mag.astype(complex)
    a = np.fft.ifft(A).real[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a

    best_a = a.copy()

    alpha = 0.5
    it = 0

        A = np.fft.fft(a, N)

        target = 1.0 / num
        error = C_auto - target
        E = np.fft.fft(error)
        A2_new = np.abs(A) ** 2 - alpha * E.real
        A2_new = np.maximum(A2_new, 0.0)
        A_new = np.sqrt(A2_new) * np.exp(1j * np.angle(A))
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)

        self.beta = beta

    def __call__(self, z):

    def grad(self, z):

def _run_lbfgs(a_init, deadline, min_remaining=30.0, beta_scale=1.0):
    """L-BFGS with adaptive beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005) * beta_scale

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03125, 0.015625, 0.0078, 0.0039,
                                  0.00195, 0.00098, 0.00049, 0.000245,
                                  0.00012, 0.00006, 0.00003, 0.000015,
                                  0.0000075, 0.000004])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        if i < len(betas) // 3:
            max_iter = min(8000, max(100, int(remaining / 0.02)))
        elif i < 2 * len(betas) // 3:
            max_iter = min(6000, max(50, int(remaining / 0.03)))
        else:

        prob = _LbfgsProb(beta, n)

                prob, z, jac=prob.grad, method='L-BFGS-B',

                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))

            print(f"    L-BFGS error at beta[{i}]: {ex}")

    z = z - z.mean()
    m_z = np.zeros(n)
    v_z = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    best_z = z.copy()
    best_sc = _score(a_init)
    it = 0
    current_sc = _score(a_init)

    beta = max(beta_smooth, _full_convolve(a_init, a_init).max() * 0.01)

    while time.monotonic() + min_remaining < deadline:
        it += 1
        obj = _smoobj(z, beta, n)
        grad = _smograd(z, beta, n)
        t = it
        m_z = beta1 * m_z + (1 - beta1) * grad
        v_z = beta2 * v_z + (1 - beta2) * grad ** 2
        m_hat = m_z / (1 - beta1 ** t)
        v_hat = v_z / (1 - beta2 ** t)
        z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z_new = z_new - z_new.mean()
        a_cand = _softmax(z_new)
        sc_cand = _score(a_cand)
        if sc_cand < best_sc:
            best_sc = sc_cand
            best_z = z_new.copy()
        if sc_cand < current_sc:
            z = z_new
            current_sc = sc_cand
        else:
            z = 0.5 * z + 0.5 * best_z
            z = z - z.mean()
        if it % 100 == 0:
            beta *= 0.9
            lr *= 0.98
            a_cur = _softmax(z)
            sc_cur = _score(a_cur)
            if sc_cur < best_sc:
                best_sc = sc_cur
                best_z = z.copy()
            print(f"    Adam it={it}: best={best_sc:.10f}, lr={lr:.6f}, beta={beta:.2e}")
        if it > 20000:
            break

    a_best = _softmax(best_z)
    return a_best, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.75) ** 2) + 0.1
        return a / a.sum()
    elif kind == "flat_taper":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        taper = max(1, n // 40)
        for i in range(taper):
            a[i] = (i + 1) / taper
            a[n - 1 - i] = (i + 1) / taper
        return a / a.sum()
    else:
        return np.ones(n) / n


def _conv_peak_info(a):
    """Get peak info about the convolution."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    return c, c_max, gap

    except Exception:
        pass

        parent_sc = _score(parent)

    print(f"\n[P1] Spectral init + flatten (t={time.monotonic()-t_start:.1f}s)")

    # === Phase 3: Adam optimization (300s) ===

    if time.monotonic() < deadline - 300:
```
