- Simulated annealing ILS with diverse perturbation strategies
- Multi-resolution coarse-to-fine optimization
- Harmonic structure initialization (window + cosine harmonics)
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _project_simplex(a):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(a)
    if n == 0:
        return a
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(a - theta, 0.0)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos2":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "blackman":
        t = np.linspace(0, 1, n)
        a = 0.42 + 0.5 * np.cos(2.0 * np.pi * t) + 0.08 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        a = np.random.default_rng(777).random(n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _freq_domain_init(n, rng, phase_type="quadratic"):
    """Initialize from frequency domain with specified phase for flat convolution."""
    N = _next_pow2(2 * n - 1)
    f = np.arange(N, dtype=np.float64)
    
    if phase_type == "quadratic":
        alpha = rng.uniform(0.5, 2.0)
        phi = alpha * (f / N) ** 2 * 2.0 * np.pi
    elif phase_type == "cubic":
        alpha = rng.uniform(0.3, 1.5)
        phi = alpha * (f / N) ** 3 * 2.0 * np.pi
    elif phase_type == "random":
        phi = rng.random(N) * 2.0 * np.pi
    elif phase_type == "log":
        phi = np.log1p(f + 1) * rng.uniform(1.0, 3.0)
    elif phase_type == "chirp_linear":
        alpha = rng.uniform(1.0, 5.0)
        phi = alpha * f / N * 2.0 * np.pi
    else:
        phi = rng.random(N) * 2.0 * np.pi
    
    best_a = None
    best_score = np.inf
    
    for trial in range(20):
        if phase_type == "random":
            phi = rng.random(N) * 2.0 * np.pi
        elif trial > 0:
            phi = phi + rng.uniform(-0.5, 0.5)
        
        A = np.exp(1j * phi)
        # Ensure Hermitian symmetry for real output
        if N % 2 == 0:
            A[N//2:] = np.conj(A[1:N//2][::-1])
        else:
            A[(N//2+1):] = np.conj(A[1:(N//2+1)][::-1])
        
        a_full = np.fft.irfft(A, N)
        a = a_full[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.05 * n:
            a = a / s
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    if best_a is not None:
        return best_a
    return np.ones(n) / n


def _harmonic_init(n, rng, n_harmonics=8):
    """Initialize with window + cosine harmonics structure."""
    t = np.linspace(0, 1, n)
    window = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    window = np.maximum(window, 1e-10)
    
    best_a = None
    best_score = np.inf
    for trial in range(15):
        harmonics = np.zeros(n)
        for k in range(n_harmonics):
            freq = rng.integers(2, max(3, n // 3))
            phase = rng.random() * 2.0 * np.pi
            amp = rng.uniform(0.02, 0.15) / (k + 1)
            harmonics += amp * np.cos(2.0 * np.pi * freq * t + phase)
        
        a = window * (1.0 + harmonics)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    if best_a is not None:
        return best_a
    return _make_profile(n, "hann")


def _variance_flatten_gradient(a, c, tol_frac=0.02):
    """
    Gradient that flattens the convolution on the active set.
    Minimizes variance of c[active] to reduce the max.
    """
    n = len(a)
    c_max = float(c.max())
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    n_active = max(1, int(active.sum()))
    
    c_active_mean = c[active].mean()
    residual = np.zeros_like(c)
    residual[active] = (c[active] - c_active_mean) / n_active
    
    a_rev = a[::-1]
    rc = _full_convolve(residual, a_rev)
    grad = 2.0 * rc[n - 1:2 * n - 1]
    grad = grad - np.mean(grad)
    return grad


def _active_set_gradient(a, c, tol_frac=0.01):
    """Standard active-set gradient (inverse-gap weighted)."""
    n = len(a)
    c_max = float(c.max())
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    n_active = int(active.sum())
    if n_active == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        n_active = 1
    
    eps = 1e-12 * c_max
    w = np.zeros_like(c)
    w[active] = 1.0 / (c_max - c[active] + eps)
    w = w / w.sum()
    
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    grad = grad - np.mean(grad)
    return grad


def _adam_optimize(a_init, deadline, lr=0.01, beta1=0.9, beta2=0.999,
                   tol_start=0.05, tol_end=0.001, mode="variance"):
    """Adam optimizer with adaptive smoothing."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    tol_frac = tol_start
    no_improve = 0
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    beta = max(c_max * 0.01, 1e-10)
    
    while time.monotonic() < deadline:
        t_count += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            break
        
        if mode == "variance":
            grad = _variance_flatten_gradient(a, c, tol_frac)
        else:
            grad = _active_set_gradient(a, c, tol_frac)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_count)
        v_hat = v / (1.0 - beta2 ** t_count)
        
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        step = step / (np.linalg.norm(step) + 1e-15)
        
        improved = False
        for scale in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            a_try = _project_simplex(a - scale * step)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < sc - 1e-13:
                a = a_try
                improved = True
                break
        
        if not improved:
            if no_improve % 50 == 0:
                tol_frac = max(tol_end, tol_frac * 0.8)
                lr *= 0.85
            if no_improve > 300:
                a = best_a.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                tol_frac = max(tol_end, tol_start * 0.3)
                lr = 0.01
                no_improve = 0
        else:
            if tol_frac > tol_end:
                tol_frac = max(tol_end, tol_frac * 0.99)
            if t_count % 100 == 0:
                beta *= 0.9
    
    return best_a, _score(best_a)


def _nesterov_polish(a_init, deadline, tol_start=0.01, tol_end=0.0001,
                     lr_init=0.03, momentum=0.85, mode="active"):
    """Nesterov momentum active-set gradient with line search."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    v = np.zeros(n)
    tol_frac = tol_start
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 400:
            break
        
        if mode == "variance":
            grad = _variance_flatten_gradient(a, c, tol_frac)
        else:
            grad = _active_set_gradient(a, c, tol_frac)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        v = momentum * v - lr * grad
        improved = False
        for _ in range(40):
            a_try = _project_simplex(a + v)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            v *= 0.5
        
        if not improved:
            v = np.zeros(n)
            if no_improve % 30 == 0:
                tol_frac = max(tol_end, tol_frac * 0.7)
                lr *= 0.8
            if no_improve > 250:
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(tol_end, tol_start * 0.3)
                lr = lr_init * 0.5
                no_improve = 0
        else:
            if tol_frac > tol_end:
                tol_frac = max(tol_end, tol_frac * 0.98)
            lr = min(lr_init * 1.5, lr * 1.005)
    
    return best_a, _score(best_a)


def _lbfgs_smooth(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lbfgs_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS on log-softmax with adaptive beta annealing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)
    
    betas = base_beta * np.array([
        1.0, 0.7, 0.5, 0.3, 0.2, 0.12, 0.08, 0.05, 0.03, 0.02,
        0.012, 0.008, 0.005, 0.003, 0.002, 0.001, 0.0006, 0.0003,
        0.00015, 0.00008, 0.00004, 0.00002
    ])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        est_per_iter = 0.003 * max(1.0, n / 16384.0)
        max_iter = min(8000, max(30, int(remaining / (est_per_iter * 12))))
        
        try:
            result = minimize(
                _lbfgs_smooth, z, args=(beta, n),
                jac=_lbfgs_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 120, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _newton_active_step(a, max_iter=30, tol=1e-4):
    """Newton-type step to equalize active convolution values using CG."""
    n = len(a)
    a = a.copy()
    
    for it in range(max_iter):
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol_abs = tol * c_max
        active = c >= (c_max - tol_abs)
        n_active = int(active.sum())
        
        if n_active <= 1:
            break
        
        c_mean = c[active].mean()
        std = (c[active] - c_mean).std()
        if std < 1e-12 * c_max:
            break
        
        residual = np.zeros_like(c)
        residual[active] = (c[active] - c_mean) / n_active
        
        # Gradient of sum(residual^2) w.r.t. a
        a_rev = a[::-1]
        rc = _full_convolve(residual, a_rev)
        grad = 2.0 * rc[n - 1:2 * n - 1]
        grad = grad - np.mean(grad)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Line search
        improved = False
        for scale in [0.1, 0.05, 0.01, 0.005, 0.001, 0.0005]:
            a_try = _project_simplex(a - scale * grad / gnorm * 10.0)
            c_try = _full_convolve(a_try, a_try)
            sc_try = c_try.max()
            if sc_try < c_max * (1.0 - 1e-10):
                a = a_try
                improved = True
                break
        
        if not improved:
            break
    
    return a


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _quick_optimize(a_init, deadline, min_time_remaining=15.0):
    """Combined optimization: L-BFGS + Adam + Nesterov polish."""
    if time.monotonic() > deadline - min_time_remaining:
        return a_init
    
    # Phase A: L-BFGS
    lbfgs_deadline = min(deadline - min_time_remaining - 10, time.monotonic() + (deadline - min_time_remaining) * 0.5)
    if time.monotonic() < lbfgs_deadline - 10:
        a = _run_lbfgs(a_init, lbfgs_deadline, min_time_remaining=10)
    else:
        a = a_init.copy()
    
    # Phase B: Adam with variance flattening
    adam_deadline = min(deadline - min_time_remaining, time.monotonic() + 20)
    if time.monotonic() < adam_deadline - 2:
        a2, sc2 = _adam_optimize(a, adam_deadline, lr=0.008, mode="variance",
                                 tol_start=0.03, tol_end=0.003)
        if sc2 < _score(a):
            a = a2
    
    # Phase C: Nesterov polish
    nest_deadline = min(deadline - min_time_remaining, time.monotonic() + 15)
    if time.monotonic() < nest_deadline - 2:
        a3, sc3 = _nesterov_polish(a, nest_deadline, tol_start=0.005,
                                   tol_end=0.0005, lr_init=0.02, momentum=0.85,
                                   mode="active")
        if sc3 < _score(a):
            a = a3
    
    return a


def _ils_anneal(a_init, deadline, rng, min_time_remaining=50.0,
                T_start=0.002, T_end=0.0001, total_iter=200):
    """Simulated annealing ILS with diverse perturbations."""
    n = len(a_init)
    current_a = _normalize(a_init.copy())
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc
    
    strategies = [
        ('global_small', 0.004),
        ('global_med', 0.015),
        ('global_large', 0.04),
        ('block', 0.05),
        ('sparse', 0.1),
        ('mirror', 0.02),
        ('freq_restart', 0.0),
        ('gradient_noise', 0.01),
    ]
    
    attempt = 0
    t0 = time.monotonic()
    
    while time.monotonic() + min_time_remaining < deadline and attempt < total_iter:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / max(1.0, deadline - min_time_remaining - t0))
        T = T_start * (T_end / T_start) ** elapsed_frac
        
        kind, sigma = strategies[attempt % len(strategies)]
        
        if kind == 'global_small' or kind == 'global_med' or kind == 'global_large':
            z = np.log(np.maximum(current_a, 1e-30))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_perturbed = _softmax(z)
        elif kind == 'block':
            block_size = max(30, n // 6)
            start = int(rng.integers(0, n - block_size))
            a_perturbed = current_a.copy()
            z = np.log(np.maximum(a_perturbed[start:start + block_size], 1e-30))
            z = z + sigma * rng.standard_normal(block_size)
            a_perturbed[start:start + block_size] = np.exp(z - np.max(z))
            a_perturbed = _normalize(a_perturbed)
        elif kind == 'sparse':
            n_perturb = max(20, n // 10)
            idx = rng.choice(n, size=n_perturb, replace=False)
            a_perturbed = current_a.copy()
            factors = np.exp(sigma * rng.standard_normal(n_perturb))
            a_perturbed[idx] *= factors
            a_perturbed = _normalize(a_perturbed)
        elif kind == 'mirror':
            seg_len = max(20, n // 15)
            start = int(rng.integers(0, n - seg_len))
            a_perturbed = current_a.copy()
            seg = a_perturbed[start:start + seg_len].copy()
            a_perturbed[start:start + seg_len] = seg[::-1]
            z = np.log(np.maximum(a_perturbed, 1e-30))
            z = z + 0.005 * rng.standard_normal(n)
            z = z - z.mean()
            a_perturbed = _softmax(z)
        elif kind == 'freq_restart':
            a_perturbed = _freq_domain_init(n, rng, "random")
        elif kind == 'gradient_noise':
            c = _full_convolve(current_a, current_a)
            grad = _active_set_gradient(current_a, c, 0.02)
            z = np.log(np.maximum(current_a, 1e-30))
            z = z + sigma * rng.standard_normal(n)
            z = z - 0.5 * sigma * grad / (np.linalg.norm(grad) + 1e-15)
            a_perturbed = _softmax(z)
        else:
            a_perturbed = current_a.copy()
        
        opt_deadline = min(deadline - min_time_remaining, time.monotonic() + 6)
        a_opt = _quick_optimize(a_perturbed, opt_deadline, min_time_remaining=3)
        sc_opt = _score(a_opt)
        
        # Simulated annealing acceptance
        if sc_opt < current_sc:
            current_a = a_opt.copy()
            if sc_opt < best_sc - 1e-12:
                best_sc = sc_opt
                best_a = a_opt.copy()
                print(f"    ILS-SA NEW BEST: {best_sc:.10f} ({kind}) t={time.monotonic()-t0:.0f}s")
        else:
            delta = sc_opt - current_sc
            accept_prob = np.exp(-delta / max(T, 1e-10))
            if rng.random() < accept_prob:
                current_a = a_opt.copy()
        
        attempt += 1
    
    return best_a, best_sc


def _multi_scale_optimize(a_best, n_base, deadline, min_time_remaining=200.0):
    """Multi-resolution: downsample, optimize, upsample, refine."""
    scales = [4, 2]
    result = a_best.copy()
    
    for factor in scales:
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        n_coarse = max(256, n_base // factor)
        if n_coarse >= n_base:
            continue
        
        a_coarse = _resample(result, n_coarse)
        
        opt_deadline = min(deadline - min_time_remaining, time.monotonic() + 30)
        a_coarse_opt = _quick_optimize(a_coarse, opt_deadline, min_time_remaining=5)
        a_up = _resample(a_coarse_opt, n_base)
        
        # Blend: keep the better of original and upsampled
        sc_orig = _score(result)
        sc_up = _score(a_up)
        if sc_up < sc_orig:
            result = a_up.copy()
            print(f"    Multi-scale: n={n_base}/{factor} improved: {sc_orig:.10f} -> {sc_up:.10f}")
    
    return result


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-scale optimizer with frequency-domain initialization, variance-flattening
    gradient, active-set Newton, Adam, SA-ILS, and multi-resolution refinement.
    """
    global t_start
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "hann")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, hann n={n_parent}, score={best_score:.10f}")
    
    n_main = n_parent if n_parent > 0 else 16384
    print(f"Main n = {n_main}")
    
    # === Phase 1: Diverse multi-start (250s) ===
    print(f"\n[P1] Multi-start (t={time.monotonic() - t_start:.1f}s)")
    ms_deadline = min(deadline - 1200, t_start + 250)
    
    starts = []
    starts.append(('parent', parent.copy()))
    starts.append(('hann', _make_profile(n_main, "hann")))
    starts.append(('blackman', _make_profile(n_main, "blackman")))
    starts.append(('raised_cos2', _make_profile(n_main, "raised_cos2")))
    starts.append(('gaussian', _make_profile(n_main, "gaussian")))
    
    # Frequency domain inits
    fd1 = _freq_domain_init(n_main, rng, "quadratic")
    starts.append(('fd_quad', fd1))
    fd2 = _freq_domain_init(n_main, rng, "random")
    starts.append(('fd_rand', fd2))
    
    # Harmonic init
    hrm = _harmonic_init(n_main, rng, n_harmonics=6)
    starts.append(('harmonic', hrm))
    
    for name, a_init in starts:
        if time.monotonic() > ms_deadline - 30:
            break
        sc_init = _score(a_init)
        a_opt = _quick_optimize(a_init, ms_deadline, min_time_remaining=40)
        sc_opt = _score(a_opt)
        print(f"  {name}: init={sc_init:.10f}, opt={sc_opt:.10f}")
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [ms/{name}] t={time.monotonic() - t_start:.1f}s")
    
    # === Phase 2: Intensive L-BFGS on best (300s) ===
    print(f"\n[P2] Intensive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 800:
        lbfgs_deadline = min(deadline - 800, time.monotonic() + 300)
        a_main = _run_lbfgs(best_seq, lbfgs_deadline, min_time_remaining=60)
        sc = _score(a_main)
        if sc < best_score:
            best_seq = a_main.copy()
            best_score = sc
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_int] t={time.monotonic() - t_start:.1f}s")
    
    # === Phase 3: Adam + Newton polish (200s) ===
    print(f"\n[P3] Adam + Newton (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        adam_deadline = min(deadline - 500, time.monotonic() + 100)
        a_adam, sc_adam = _adam_optimize(best_seq, adam_deadline, lr=0.005,
                                         mode="variance", tol_start=0.02, tol_end=0.002)
        if sc_adam < best_score:
            best_seq = a_adam.copy()
            best_score = sc_adam
            print(f"  NEW BEST: {best_score:.10f} [adam_var] t={time.monotonic() - t_start:.1f}s")
        
        # Newton step
        if time.monotonic() < deadline - 350:
            newton_deadline = min(deadline - 350, time.monotonic() + 50)
            a_newton = best_seq.copy()
            for _ in range(5):
                if time.monotonic() > newton_deadline:
                    break
                a_newton = _newton_active_step(a_newton, max_iter=20, tol=1e-4)
            sc_newton = _score(a_newton)
            if sc_newton < best_score:
                best_seq = a_newton.copy()
                best_score = sc_newton
                print(f"  NEW BEST: {best_score:.10f} [newton] t={time.monotonic() - t_start:.1f}s")
    
    # === Phase 4: Nesterov polish (150s) ===
    print(f"\n[P4] Nesterov polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        nest_deadline = min(deadline - 300, time.monotonic() + 150)
        a_pol, sc_pol = _nesterov_polish(best_seq, nest_deadline,
                                         tol_start=0.005, tol_end=0.0001,
                                         lr_init=0.02, momentum=0.85, mode="active")
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  NEW BEST: {best_score:.10f} [nesterov] t={time.monotonic() - t_start:.1f}s")
    
    # === Phase 5: SA-ILS (350s) ===
    print(f"\n[P5] Simulated Annealing ILS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        ils_deadline = min(deadline - 200, time.monotonic() + 350)
        a_ils, sc_ils = _ils_anneal(best_seq, ils_deadline, rng,
                                     min_time_remaining=60,
                                     T_start=0.003, T_end=0.00005,
                                     total_iter=300)
        if sc_ils < best_score:
            best_seq = a_ils.copy()
            best_score = sc_ils
            print(f"  NEW BEST: {best_score:.10f} [sa_ils] t={time.monotonic() - t_start:.1f}s")
    
    # === Phase 6: Multi-scale (150s) ===
    print(f"\n[P6] Multi-scale (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        ms_deadline2 = min(deadline - 150, time.monotonic() + 150)
        a_ms = _multi_scale_optimize(best_seq, len(best_seq), ms_deadline2, min_time_remaining=50)
        sc_ms = _score(a_ms)
        if sc_ms < best_score:
            best_seq = a_ms.copy()
            best_score = sc_ms
            print(f"  NEW BEST: {best_score:.10f} [multi_scale] t={time.monotonic() - t_start:.1f}s")
    
    # === Phase 7: Multi-n exploration (150s) ===
    print(f"\n[P7] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [len(best_seq), 24576, 32768, 16384, 20480, 28672, 40960]
    seen_n = set()
    n_unique = []
    for nn in n_candidates:
        if nn not in seen_n and nn != len(best_seq) and nn >= 4096:
            seen_n.add(nn)
            n_unique.append(nn)
    
    for n_try in n_unique[:4]:
        if time.monotonic() > deadline - 100:
            break
        a_try = _resample(best_seq, n_try)
        sc_try = _score(a_try)
        a_opt = _quick_optimize(a_try, deadline, min_time_remaining=30)
        sc_opt = _score(a_opt)
        print(f"  n={n_try}: init={sc_try:.10f}, opt={sc_opt:.10f}")
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic() - t_start:.1f}s")
    
    # === Phase 8: Second SA-ILS round (100s) ===
    print(f"\n[P8] SA-ILS round 2 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        rng2 = np.random.default_rng(int(seed) + 7777)
        ils2_deadline = min(deadline - 80, time.monotonic() + 100)
        a_ils2, sc_ils2 = _ils_anneal(best_seq, ils2_deadline, rng2,
                                       min_time_remaining=30,
                                       T_start=0.001, T_end=0.00002,
                                       total_iter=100)
        if sc_ils2 < best_score:
            best_seq = a_ils2.copy()
            best_score = sc_ils2
            print(f"  NEW BEST: {best_score:.10f} [sa_ils2] t={time.monotonic() - t_start:.1f}s")
    
    # === Phase 9: Final ultra-fine polish (80s) ===
    print(f"\n[P9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 10:
        # Fine Nesterov
        fin_deadline = min(deadline - 10, time.monotonic() + 40)
        a_fin, sc_fin = _nesterov_polish(best_seq, fin_deadline,
                                         tol_start=0.002, tol_end=0.00005,
                                         lr_init=0.01, momentum=0.82, mode="variance")
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_nest] t={time.monotonic() - t_start:.1f}s")
        
        # Fine L-BFGS
        remaining = deadline - 10 - time.monotonic()
        if remaining > 15:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.00002
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            max_iter = min(5000, max(50, int(remaining / 0.02)))
            try:
                result = minimize(_lbfgs_smooth, z, args=(beta_fin, n),
                                  jac=_lbfgs_grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-14, 'maxcor': 120})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic() - t_start:.1f}s")
            except Exception:
                pass
        
        # Final Newton
        remaining = deadline - 10 - time.monotonic()
        if remaining > 5:
            a_fin3 = _newton_active_step(best_seq, max_iter=30, tol=1e-5)
            sc_fin3 = _score(a_fin3)
            if sc_fin3 < best_score:
                best_seq = a_fin3.copy()
                best_score = sc_fin3
                print(f"  NEW BEST: {best_score:.10f} [final_newton] t={time.monotonic() - t_start:.1f}s")
    
    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
