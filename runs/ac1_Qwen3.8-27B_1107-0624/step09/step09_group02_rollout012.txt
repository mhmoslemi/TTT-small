def next_pow2(n):
    k = 1

    out_n = n + m - 1

    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def active_set_refine(a_init, deadline, min_remaining=20.0):
    """Iteratively refine using active-set QP with increasing number of constraints."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = score(a)
    
    # Start with top-1, gradually increase
    for m in [1, 2, 3, 5, 8, 12, 20, 30, 50, 80, 120, 200]:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        
        c = full_convolve(a, a)
        c_max = float(c.max())
        # Select active set: top m values
        top_idx = np.argsort(c)[::-1][:m]
        p = np.zeros(2*n-1)
        p[top_idx] = 1.0 / m
        
        # Solve: min sum_k p_k c[k] s.t. sum(a) = 1, a >= 0
        # Using projected gradient with backtracking
        lr = 0.1
        for inner in range(100):
            if inner % 50 == 50 and time.monotonic() + min_remaining > deadline:
                break
            grad = gradient_of_weighted_corr(a, p, n)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            # Projected gradient step
            lr = 0.5 / (gnorm * (1 + 0.01 * inner))
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a
            else:
                a_new = a_new / s
            # Check improvement
            obj_new = float(np.dot(p, full_convolve(a_new, a_new)))
            obj_old = float(np.dot(p, full_convolve(a, a)))
            if obj_new < obj_old:
                a = a_new
            else:
                lr *= 0.5
                if lr < 1e-15:
                    break
        
        sc = score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def mul_weight_update(a_init, deadline, min_remaining=20.0):
    """Multiplicative weight update for flattening the convolution."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = score(a)
    
    # Target: make c flat
    c = full_convolve(a, a)
    target = float(c.mean())  # target level
    
    eta = 0.1  # step size
    
    for it in range(500):
        if time.monotonic() + min_remaining > deadline:
            break
        c = full_convolve(a, a)
        c_max = float(c.max())
        c_min_pos = float(c[c > 0].min()) if (c > 0).any() else 0
        
        # Adaptive target: between current max and mean
        target = c_max * 0.95  # try to bring max down by 5%
        
        # Compute weights: w[k] = target / c[k] (clipped)
        w = np.ones_like(c)
        pos = c > 1e-30
        w[pos] = target / c[pos]
        w = np.clip(w, 0.01, 100.0)
        
        # Gradient of the weighted correlation
        grad = gradient_of_weighted_corr(a, w, n)
        
        # Multiplicative update in log-space
        z = np.log(a)
        z_new = z - eta * grad / (np.linalg.norm(grad) + 1e-10)
        z_new = z_new - z_new.mean()
        a_new = softmax(z_new)
        
        sc_new = score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce step size
            eta *= 0.8
            if eta < 1e-8:
                eta = 0.1  # reset
                a = best_a.copy()
    
    return best_a

def perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return softmax(z)

def resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.abs(np.sin(np.pi * t)) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wedge":
        t = np.linspace(0, 1, n)
        a = np.abs(t - 0.5) * 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def compute_gradient(a, p, n):
    """Compute gradient of sum_k p[k]*c[k] where c = conv(a,a)."""
    ext_n = 2*n - 1  # length of p
    conv_len = n + ext_n - 1  # = 3n-2
    N = next_pow2(conv_len)
    
    a_rev = a[::-1]  # length n
    a_rev_pad = np.zeros(N)
    a_rev_pad[:n] = a_rev
    
    p_pad = np.zeros(N)
    p_pad[:ext_n] = p
    
    A = np.fft.rfft(a_rev_pad)
    P = np.fft.rfft(p_pad)
    conv = np.fft.irfft(A * P, N)
    
    return 2.0 * conv[n-1:2*n-1]
