Key innovations:
- FFT-based gradient computation for both variance and smooth-max objectives

import numpy as np

def _next_pow2(n):
    return 1 << (n - 1).bit_length() if n > 1 else 2


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_fft(a, N):
    """Compute score using FFT convolution."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:2*n-1]
    return float(2.0 * n * c.max() / (s * s))

    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

    """Objective: 2*n*sum(c^2) with FFT-based gradient."""
    def __init__(self, n, N):
        self.n = n
        self.N = N

        a_pad[:self.n] = a
        A = np.fft.rfft(a_pad, self.N)

        # For rfft, we need to handle the real structure

        c = np.fft.irfft(A * A, self.N)[:2*self.n-1]
        return c

        return float(2.0 * self.n * np.sum(c**2))

If `F(a) = f(a/s)` where `s = sum(a)` and `f(a_norm) = 2*n*max(conv(a_norm, a_norm))`, then:
`dF/da[m] = (f_i(a_norm)/s - (a[m]/s^2) * sum_j a[j] * f_j(a_norm))` where `f_i = df_i/da_i`.

Simplifying: `dF/da[m] = (1/s) * f[m] - (a[m]/s^2) * sum(a*f)`
= `(1/s) * (f[m] - a[m] * sum(a*f)/s)`

Where `f = df/da_norm` is the gradient of `f` with respect to the normalized vector.

        a = a / a.sum()

        a = a / a.sum()

        # Transform to z-gradient

"""
FFT-accelerated optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
- FFT-based gradient computation (O(N log N)) for both variance and smooth-max objectives
- Gerchberg-Saxton spectral flattening for high-quality initialization
- Phased optimization: variance -> smooth max -> top-K max -> active-set polish
- Log-domain (softmax) parametrization for clean constrained optimization
- Multi-restart with diverse initializations from different spectral profiles
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    if n <= 1:
        return 2
    return 1 << (n - 1).bit_length()


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_fft(a, N):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:2 * n - 1]
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _convolve_fft(a, N):
    n = len(a)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2 * n - 1]


def _grad_weighted_conv(a, w, N):
    """Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a, using FFT.
    Returns array of length n."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    w_pad = np.zeros(N)
    w_pad[:len(w)] = w
    A = np.fft.fft(a_pad, N)
    W = np.fft.fft(w_pad, N)
    grad_pad = 2.0 * np.real(np.fft.ifft(W * np.conj(A)))
    return grad_pad[:n]


def _grad_variance(a, N):
    """Gradient of sum_k c[k]^2 where c=conv(a,a), using FFT.
    dF/da[m] = 4*Re[IFFT(A^2*conj(A))[m]]"""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = np.fft.fft(a_pad, N)
    B = A * A * np.conj(A)
    grad_pad = 4.0 * np.real(np.fft.ifft(B))
    return grad_pad[:n]


class _LogDomainObj:
    """Wraps a normalized-a objective into the log (softmax) domain."""

    def __init__(self, obj_func, obj_grad, n, N):
        self.f = obj_func
        self.g = obj_grad
        self.n = n
        self.N = N

    def _softmax(self, z):
        z_shifted = z - z.max()
        a = np.exp(z_shifted)
        a = a / a.sum()
        return a

    def __call__(self, z):
        a = self._softmax(z)
        return self.f(a)

    def grad(self, z):
        a = self._softmax(z)
        g = self.g(a)
        return a * (g - float(np.dot(a, g)))


class _VarianceInner:
    """F(a) = 2*n*sum(c^2) for normalized a."""

    def __init__(self, n, N):
        self.n = n
        self.N = N

    def f(self, a):
        c = _convolve_fft(a, self.N)
        return float(2.0 * self.n * np.sum(c ** 2))

    def g(self, a):
        return _grad_variance(a, self.N)


class _SmoothMaxInner:
    """F(a) = 2*n*beta*log-sum-exp(c/beta) for normalized a."""

    def __init__(self, n, N, beta):
        self.n = n
        self.N = N
        self.beta = beta

    def _weights(self, c):
        c_max = c.max()
        shifted = np.clip((c - c_max) / self.beta, -700, 0)
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        return w, c_max + self.beta * np.log(s) if s > 1e-300 else c_max

    def f(self, a):
        c = _convolve_fft(a, self.N)
        _, val = self._weights(c)
        return float(2.0 * self.n * val)

    def g(self, a):
        c = _convolve_fft(a, self.N)
        w, _ = self._weights(c)
        return _grad_weighted_conv(a, w, self.N)


class _TopKMaxInner:
    """F(a) = 2*n*beta*log(sum of top-K exp(c/beta)) for normalized a."""

    def __init__(self, n, N, beta, K):
        self.n = n
        self.N = N
        self.beta = beta
        self.K = K

    def _topk_weights(self, c):
        K = min(self.K, len(c))
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
        c_max = c_top.max()
        shifted = np.clip((c_top - c_max) / self.beta, -700, 0)
        e = np.exp(shifted)
        s = e.sum()
        w_full = np.zeros(len(c))
        if s < 1e-300:
            w_full[int(np.argmax(c))] = 1.0
            val = c.max()
        else:
            w_full[top_idx] = e / s
            val = c_max + self.beta * np.log(s)
        return w_full, val

    def f(self, a):
        c = _convolve_fft(a, self.N)
        _, val = self._topk_weights(c)
        return float(2.0 * self.n * val)

    def g(self, a):
        c = _convolve_fft(a, self.N)
        w, _ = self._topk_weights(c)
        return _grad_weighted_conv(a, w, self.N)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.3) ** 2) + np.exp(-15.0 * (t - 0.7) ** 2) + 0.02
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.01
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "optimal_shape":
        # Based on small-n analysis: bimodal tends to be better
        t = np.linspace(0, 1, n)
        a = 0.4 + 0.6 * np.cos(2 * np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _gs_flatten(a, N, n_iter=100, alpha=0.5):
    """Gerchberg-Saxton spectral flattening."""
    n = len(a)
    a = _normalize(a.copy())
    # Target: flat autocorrelation of length 2n-1 with sum 1
    c_target = np.zeros(N)
    c_len = 2 * n - 1
    c_target[:c_len] = 1.0 / c_len
    S_target = np.fft.fft(c_target)

    for it in range(n_iter):
        A = np.fft.fft(a, N)
        S = np.abs(A) ** 2
        # Blend towards target spectrum (magnitude squared)
        S_abs = np.abs(S_target)
        phase = np.angle(A)
        # Geometric blend of magnitudes
        mag_target = np.sqrt(S_abs)
        mag_current = np.sqrt(S)
        # Avoid log of zeros
        mag_target = np.maximum(mag_target, 1e-30)
        mag_current = np.maximum(mag_current, 1e-30)
        # Log-domain blend
        log_mag = (1 - alpha) * np.log(mag_current) + alpha * np.log(mag_target)
        new_mag = np.exp(log_mag)
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new

    return a


def _run_lbfgs_logdomain(z, obj, deadline, min_remaining=5.0, maxiter=None):
    """L-BFGS in log domain with time budget."""
    current_obj = obj(z)
    best_z = z.copy()
    best_val = current_obj

    try:
        el_result = minimize(
            obj, z, jac=obj.grad, method='L-BFGS-B',
            options={
                'maxiter': maxiter if maxiter else 5000,
                'ftol': 1e-18,
                'gtol': 1e-14,
                'maxcor': 100,
                'maxls': 50,
                'iprint': -1,
            }
        )
        val = obj(el_result.x)
        if val < best_val:
            best_z = el_result.x.copy()
            best_val = val
    except Exception:
        pass

    return best_z, best_val


def _perturb_z(z, sigma, rng):
    """Perturb in log domain."""
    return z + sigma * rng.standard_normal(len(z))


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

            z_try2, _ = _run_lbfgs_logdomain(z_try2, tk_obj2, t_rs2 + per_restart_time * 0.4,
                                             min_remaining=1, maxiter=maxiter2)
            a_try2 = _from_z(z_try2)
            _try_update(a_try2, f"restart_{i}_tight")

    # === Phase 6: Different n exploration ===
    print(f"\n[P6] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        n_options = [16384, 12288, 24576, 8192]
        n_options = [x for x in n_options if x != n][:2]
        for n_try in n_options:
            if time.monotonic() > deadline - 40:
                break
            N_try = _next_pow2(2 * n_try + 10)
            a_res = _resample(best_a, n_try)
            # Quick optimize
            c_res = np.convolve(a_res, a_res)
            beta = float(c_res.max()) * 0.03
            K = max(10, int((2 * n_try - 1) * 0.03))
            tk_inner = _TopKMaxInner(n_try, N_try, beta, K)
            tk_obj = _LogDomainObj(tk_inner.f, tk_inner.g, n_try, N_try)
            z_res = _to_z(a_res)
            t_n = time.monotonic()
            maxiter_n = min(1000, max(50, int(30 / 0.003)))
            z_res, _ = _run_lbfgs_logdomain(z_res, tk_obj, t_n + 30, min_remaining=2, maxiter=maxiter_n)
            a_opt = _from_z(z_res)
            a_opt = a_opt / a_opt.sum()
            sc_opt = _score(a_opt)
            if sc_opt < best_score - 1e-10:
                best_a = a_opt.copy()
                best_score = sc_opt
                n = n_try
                N = N_try
                print(f"  NEW BEST: {best_score:.10f} [n={n_try}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final polish ===
    print(f"\n[P7] Final Polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - time.monotonic()
    if remaining > 10:
        c_fin = _convolve_fft(_normalize(best_a), N)
        c_max_fin = float(c_fin.max())
        beta_fin = c_max_fin * 0.01
        K_fin = max(5, int((2 * n - 1) * 0.005))
        tk_inner = _TopKMaxInner(n, N, beta_fin, K_fin)
        tk_obj = _LogDomainObj(tk_inner.f, tk_inner.g, n, N)
        z_fin = _to_z(best_a)
        maxiter_fin = min(3000, max(50, int((remaining - 5) / 0.003)))
        z_fin, _ = _run_lbfgs_logdomain(z_fin, tk_obj, deadline, min_remaining=2, maxiter=maxiter_fin)
        a_fin = _from_z(z_fin)
        _try_update(a_fin, "final_polish")

    # Final validation
    print(f"\n{'=' * 60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
