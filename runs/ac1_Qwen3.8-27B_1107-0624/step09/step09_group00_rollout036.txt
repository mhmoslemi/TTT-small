    w[active] = 1.0
    w = w / w.sum()
    # Same computation as pnorm gradient but with 0-1 weights
    a_rev = a[::-1]
    part1 = _full_convolve(w, a_rev)[n - 1:2 * n - 1]
    part2 = _full_convolve(w, a)[:n]
    return part1 + part2

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hanning":
        t = np.arange(n) / (n - 1)
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        for sigma in [0.2, 0.25, 0.3, 0.35]:
            a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
        return a
    elif kind == "raised_cos2":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = 1.0 - 1.93 * np.cos(2 * np.pi * t) + 0.92 * np.cos(4 * np.pi * t) \
            - 0.254 * np.cos(6 * np.pi * t) + 0.049 * np.cos(8 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "optimal_param":
        # Search over a parametric family
        t = np.linspace(0, 1, n)
        best_a = None
        best_sc = np.inf
        for alpha in [0.3, 0.5, 0.7, 1.0, 1.3, 1.7, 2.0, 2.5, 3.0, 4.0]:
            a = np.power(np.maximum(1.0 - np.abs(2.0 * t - 1.0), 0.0), alpha)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            sc = _score_normalized(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        return best_a
    elif kind == "cos_power":
        t = np.linspace(0, 1, n)
        best_a = None
        best_sc = np.inf
        for gamma in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 6.0]:
            a = np.power(np.maximum(np.cos(np.pi * (t - 0.5)), 0.0), gamma)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            sc = _score_normalized(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        return best_a
    else:
        return np.ones(n) / n

def _pnorm_lbfgs(a_init, p, deadline, max_iter=500):
    """L-BFGS optimization of p-norm objective using softmax parameterization."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    c = _full_convolve(a, a)
    best_a = a.copy()
    best_sc = _score_normalized(a)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    def obj(zv):
        av = _softmax(zv)
        cv = _full_convolve(av, av)
        cmax = cv.max()
        if cmax < 1e-300:
            return 0.0
        shifted = cv / cmax
        return cmax + (1.0 / p) * np.log(np.mean(np.power(shifted, p)))

    def grad(zv):
        av = _softmax(zv)
        cv = _full_convolve(av, av)
        cmax = cv.max()
        if cmax < 1e-300:
            return np.zeros(len(zv))
        w = np.power(np.maximum(cv, 0.0), p - 1)
        w_sum = w.sum()
        if w_sum < 1e-300:
            idx = int(np.argmax(cv))
            w = np.zeros_like(cv)
            w[idx] = 1.0
        else:
            w = w / w_sum
        grad_a = _gradient_pnorm(av, cv, p)
        # softmax gradient: grad_z[i] = a_i * (grad_a[i] - sum(a*grad_a))
        correction = np.dot(av, grad_a)
        grad_z = av * (grad_a - correction)
        return grad_z * 2.0 * n  # scale for the 2n*max_c objective

    try:
        result = minimize(obj, z, jac=grad, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxcor': 80, 'maxls': 30})
        a_new = _softmax(result.x)
        sc_new = _score_normalized(a_new)
        if sc_new < best_sc:
            best_a = a_new
            best_sc = sc_new
    except Exception:
        pass

    return best_a, best_sc

def _active_set_optimize(a_init, deadline, tol_start=0.01, tol_end=0.00001,
                         lr_init=0.03, momentum=0.85, max_iters=5000):
    """Active-set gradient descent with heavy-ball momentum."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score_normalized(a)
    v = np.zeros(n)
    tol_frac = tol_start
    lr = lr_init
    no_improve = 0
    max_stall = 200

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        grad = _gradient_active_set(a, c, tol_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Project gradient onto tangent space (sum=0)
        grad = grad - np.mean(grad)

        # Heavy ball momentum
        v = momentum * v - lr * grad

        # Line search
        alpha = 1.0
        improved = False
        for _ in range(40):
            a_try = a + alpha * v
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            v = np.zeros(n)
            if no_improve % 20 == 0:
                tol_frac = max(tol_end, tol_frac * 0.7)
                lr = max(1e-6, lr * 0.8)
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(tol_end, tol_frac * 0.5)
                lr = lr_init * 0.5
                no_improve = 0
        else:
            if tol_frac > tol_end:
                tol_frac = max(tol_end, tol_frac * 0.99)
            lr = min(lr_init, lr * 1.005)

    return best_a, _score_normalized(best_a)

def _lp_direction(a, M, n_constraints=None):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M, b >= 0.
    Optionally use only top K constraints."""
    n = len(a)
    c = _full_convolve(a, a)

    if n_constraints is not None and n_constraints < len(c):
        # Use only top n_constraints peaks
        idx = np.argsort(c)[::-1][:n_constraints]
    else:
        idx = np.arange(len(c))

    # Build constraint matrix (sparse): A[k, j] = contribution of b[j] to c[k]
    # c[k] = sum_j a[j] * b[k-j] (valid indices)
    # So A[k, j] = a[j] if 0 <= k-j < n, else 0
    # Equivalently, A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # Wait: c[k] = sum_m a[m] * b[k-m], so coefficient of b[j] in c[k] is a[k-j] (if 0<=k-j<n)

    rows = []
    cols = []
    vals = []
    for ki in idx:
        k = int(ki)
        # b[j] appears in c[k] with coefficient a[k-j] if 0 <= k-j < n
        # j ranges: max(0, k-n+1) to min(k, n-1)
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            js = np.arange(j_lo, j_hi + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(a[k - js].tolist())

    if len(rows) == 0:
        return a.copy()

    A = csr_matrix((vals, (rows, cols)), shape=(len(idx), n))

    # LP: max sum(b) s.t. A@b <= M, b >= 0
    # linprog minimizes, so: min -sum(b) s.t. A@b <= M, b >= 0
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(len(idx), M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': 30, 'presolve': True})

    if res.status == 0 and res.x is not None:
        b = np.maximum(res.x, 0)
        s = b.sum()
        if s > 1e-10:
            return b / s
    return a.copy()

def _lp_iteration(a_init, deadline, n_lps=5, n_constraints=200):
    """Do several LP-based direction finding iterations."""
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score_normalized(a)

    for i in range(n_lps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        M = float(c.max())
        b = _lp_direction(a, M, n_constraints=n_constraints)
        if b is None or np.allclose(b, a):
            break
        # Blend
        for t in [1.0, 0.7, 0.5, 0.3]:
            a_try = _normalize((1 - t) * a + t * b)
            sc_try = _score_normalized(a_try)
            if sc_try < best_sc - 1e-15:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try.copy()
                break
        else:
            # No improvement, try smaller step
            a_try = _normalize(a + 0.1 * (b - a))
            sc_try = _score_normalized(a_try)
            if sc_try < best_sc - 1e-15:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try.copy()

    return best_a, _score_normalized(best_a)

def _stochastic_ils(a_init, deadline, seed=42, min_remaining=60):
    """Stochastic ILS with diverse perturbation strategies."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    current_a = _normalize(a_init.copy())
    current_sc = _score_normalized(current_a)
    best_a = current_a.copy()
    best_sc = current_sc
    attempt = 0

    while time.monotonic() + min_remaining < deadline:
        r = rng.random()
        if r < 0.2:
            # Small global perturbation
            sigma = 0.01
            z = np.log(np.maximum(current_a, 1e-30)) + sigma * rng.standard_normal(n)
            a_pert = _softmax(z)
        elif r < 0.35:
            # Medium global perturbation
            sigma = 0.05
            z = np.log(np.maximum(current_a, 1e-30)) + sigma * rng.standard_normal(n)
            a_pert = _softmax(z)
        elif r < 0.5:
            # Block perturbation
            block = max(20, n // 6)
            start = int(rng.integers(0, n - block))
            a_pert = current_a.copy()
            z_block = np.log(np.maximum(a_pert[start:start+block], 1e-30))
            z_block += 0.05 * rng.standard_normal(block)
            a_pert[start:start+block] = np.exp(z_block - np.max(z_block))
            a_pert = _normalize(a_pert)
        elif r < 0.65:
            # Spectral perturbation (add a cosine mode)
            a_pert = current_a.copy().astype(float)
            freq = int(rng.integers(1, n // 4))
            phase = rng.uniform(0, 2 * np.pi)
            amplitude = 0.02 * rng.random()
            t = np.arange(n)
            a_pert += amplitude * np.cos(2 * np.pi * freq * t / n + phase)
            a_pert = _normalize(a_pert)
        elif r < 0.8:
            # Sparse perturbation
            n_pert = max(20, n // 10)
            idx = rng.choice(n, size=n_pert, replace=False)
            a_pert = current_a.copy()
            a_pert[idx] *= np.exp(0.1 * rng.standard_normal(n_pert))
            a_pert = _normalize(a_pert)
        else:
            # Endpoint emphasis: boost or reduce endpoints
            a_pert = current_a.copy()
            frac = max(5, n // 20)
            factor = 0.5 + rng.random()
            a_pert[:frac] *= factor
            a_pert[-frac:] *= factor
            a_pert = _normalize(a_pert)

        # Quick optimization
        opt_deadline = min(deadline - min_remaining, time.monotonic() + 5)
        a_opt, sc_opt = _active_set_optimize(a_pert, opt_deadline,
                                              tol_start=0.01, tol_end=0.001,
                                              lr_init=0.02, momentum=0.8,
                                              max_iters=200)
        if sc_opt < best_sc - 1e-13:
            best_sc = sc_opt
            best_a = a_opt.copy()
            current_a = a_opt.copy()
            print(f"    ILS NEW BEST: {best_sc:.10f} t={time.monotonic()-t_start:.0f}s")
        elif sc_opt < current_sc - 1e-13:
            current_a = a_opt.copy()

        attempt += 1

    return best_a, best_sc

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: parametric search, p-norm continuation via L-BFGS,
    active-set refinement, LP direction finding, ILS, and multi-n exploration.
    """
    global t_start
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _normalize(_make_profile(n_parent, "hann"))
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, using hann n={n_parent}, score={best_score:.10f}")

    n_main = n_parent

    # === Phase 1: Parametric family search (40s) ===
    print(f"\n[P1] Parametric search (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = t_start + 40
    for kind in ['hann', 'hanning', 'triangle', 'raised_cos2', 'flat_top',
                 'optimal_param', 'cos_power', 'gaussian']:
        if time.monotonic() > p1_deadline:
            break
        a_try = _normalize(_make_profile(n_main, kind))
        sc_try = _score(a_try)
        if sc_try < best_score:
            best_score = sc_try
            best_seq = a_try.copy()
            print(f"  {kind}: {sc_try:.10f} NEW BEST")
        else:
            print(f"  {kind}: {sc_try:.10f}")

    # === Phase 2: p-norm continuation (400s) ===
    print(f"\n[P2] p-norm continuation (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = t_start + 440
    a_curr = _normalize(best_seq.copy())

    p_schedule = [2, 3, 4, 6, 8, 12, 16, 24, 32, 48, 64, 96, 128, 192, 256, 384, 512]
    for p in p_schedule:
        if time.monotonic() > p2_deadline - 30:
            break
        remaining = p2_deadline - 30 - time.monotonic()
        max_iter = min(800, max(50, int(remaining / 0.05)))
        time.perf_counter()
        t0 = time.monotonic()
        a_curr, sc_curr = _pnorm_lbfgs(a_curr, p, p2_deadline - 30, max_iter=max_iter)
        dt = time.monotonic() - t0
        sc_check = _score(a_curr)
        print(f"  p={p}: score={sc_check:.10f} (time={dt:.1f}s)")
        if sc_check < best_score:
            best_score = sc_check
            best_seq = a_curr.copy()
            print(f"  NEW BEST: {best_score:.10f} [p={p}]")

    # === Phase 3: Active-set refinement (300s) ===
    print(f"\n[P3] Active-set refinement (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = t_start + 740
    a_curr = _normalize(best_seq.copy())
    a_opt, sc_opt = _active_set_optimize(a_curr, p3_deadline,
                                          tol_start=0.005, tol_end=0.000005,
                                          lr_init=0.02, momentum=0.85)
    if sc_opt < best_score:
        best_score = sc_opt
        best_seq = a_opt.copy()
        print(f"  NEW BEST: {best_score:.10f} [active_set]")
    else:
        print(f"  No improvement: {sc_opt:.10f} vs {best_score:.10f}")

    # === Phase 4: LP direction finding (200s) ===
    print(f"\n[P4] LP direction finding (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = t_start + 940
    if time.monotonic() < p4_deadline - 30:
        a_curr = _normalize(best_seq.copy())
        # Try LP with different constraints counts
        for n_con in [100, 200, 500]:
            if time.monotonic() > p4_deadline - 30:
                break
            t0 = time.monotonic()
            a_lp, sc_lp = _lp_iteration(a_curr, p4_deadline - 60, n_lps=3,
                                         n_constraints=n_con)
            dt = time.monotonic() - t0
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
                a_curr = a_lp.copy()
                print(f"  LP({n_con} constraints): NEW BEST {best_score:.10f} (t={dt:.1f}s)")
            else:
                print(f"  LP({n_con} constraints): {sc_lp:.10f} (t={dt:.1f}s)")

    # === Phase 5: ILS (350s) ===
    print(f"\n[P5] ILS (t={time.monotonic()-t_start:.1f}s)")
    p5_deadline = t_start + 1290
    if time.monotonic() < p5_deadline - 60:
        a_ils, sc_ils = _stochastic_ils(best_seq, p5_deadline, seed=int(seed)+1,
                                         min_remaining=60)
        if sc_ils < best_score:
            best_score = sc_ils
            best_seq = a_ils.copy()
            print(f"  NEW BEST: {best_score:.10f} [ILS]")

    # === Phase 6: Multi-n exploration (250s) ===
    print(f"\n[P6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    p6_deadline = t_start + 1540
    n_candidates = sorted(set([8192, 12288, 16384, 24576, 32768, 40960, 49152]),
                          key=lambda x: abs(x - len(best_seq)))
    for n_try in n_candidates:
        if time.monotonic() > p6_deadline - 30:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        # Quick optimize
        a_opt, sc_opt = _active_set_optimize(a_init, min(p6_deadline - 30,
                                                          time.monotonic() + 30),
                                              tol_start=0.01, tol_end=0.001,
                                              lr_init=0.02, momentum=0.8,
                                              max_iters=500)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: NEW BEST {best_score:.10f}")
        else:
            print(f"  n={n_try}: {sc_opt:.10f}")

    # === Phase 7: Final combined polish (200s) ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    # Active set with very tight tolerance
    a_final, sc_final = _active_set_optimize(best_seq, deadline - 5,
                                              tol_start=0.001, tol_end=0.0000005,
                                              lr_init=0.005, momentum=0.8,
                                              max_iters=5000)
    if sc_final < best_score:
        best_score = sc_final
        best_seq = a_final.copy()
        print(f"  NEW BEST: {best_score:.10f} [final_active]")

    # One more p-norm L-BFGS with high p
    remaining = deadline - 5 - time.monotonic()
    if remaining > 20:
        a_final2, sc_final2 = _pnorm_lbfgs(best_seq, 512, deadline - 5, max_iter=300)
        if sc_final2 < best_score:
            best_score = sc_final2
            best_seq = a_final2.copy()
            print(f"  NEW BEST: {best_score:.10f} [final_pnorm]")

        # Tiny active-set polish
        remaining2 = deadline - 5 - time.monotonic()
        if remaining2 > 10:
            a_final3, sc_final3 = _active_set_optimize(best_seq, deadline - 3,
                                                        tol_start=0.0005,
                                                        tol_end=0.0000001,
                                                        lr_init=0.003, momentum=0.75,
                                                        max_iters=3000)
            if sc_final3 < best_score:
                best_score = sc_final3
                best_seq = a_final3.copy()
                print(f"  NEW BEST: {best_score:.10f} [ultra_fine]")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
