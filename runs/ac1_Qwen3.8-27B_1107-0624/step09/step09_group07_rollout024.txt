"""
Improved multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements:
- Power-sum smoothing (alternative to log-sum-exp) for better L-BFGS behavior
- Nesterov-accelerated active-set gradient descent with adaptive tolerance
- Coarse-to-fine multi-resolution optimization
- Diverse population-based restarts with structural mutations
- Multi-n exploration (8192, 16384, 32768)
- Careful final polish with coordinate descent on top variables
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _power_smoobj_grad(z, p, K, n):
    """Power-sum smoothing in log-space. Returns (obj, grad_z)."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c
    
    cp = c_top ** p
    s = cp.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        obj = 2.0 * n * (s / K) ** (1.0 / p)
        # Weights proportional to c[k]^(p-1)
        if p > 1:
            w_top = cp / c_top  # = c_top^(p-1)
            w_top = w_top / w_top.sum()
        else:
            w_top = np.ones(K) / K
        w = np.zeros(2 * n - 1)
        w[top_idx] = w_top
    
    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


class _PowerSmoProb:
    def __init__(self, p, K, n):
        self.p = p
        self.K = K
        self.n = n

    def __call__(self, z):
        return _power_smoobj_grad(z, self.p, self.K, self.n)[0]

    def grad(self, z):
        return _power_smoobj_grad(z, self.p, self.K, self.n)[1]


class _LogSmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[1]


def _topk_smoobj_grad(z, beta, K, n):
    """Top-K log-sum-exp smoothed objective and gradient in log-domain."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c
    
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    
    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, use_power=True):
    """L-BFGS with adaptive smoothing (power-sum or log-sum-exp)."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.008)
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # Schedule for power-sum: (p, K_fraction)
    if use_power:
        schedule = [
            (4.0, 0.1), (8.0, 0.08), (16.0, 0.06), (32.0, 0.05),
            (64.0, 0.04), (128.0, 0.03), (256.0, 0.02), (512.0, 0.015),
            (1024.0, 0.01), (2048.0, 0.008), (4096.0, 0.005), (8192.0, 0.003),
        ]
    else:
        schedule = [
            (1.0, 0.1), (0.7, 0.08), (0.5, 0.06), (0.3, 0.04),
            (0.2, 0.03), (0.1, 0.02), (0.05, 0.015), (0.03, 0.01),
            (0.01, 0.008), (0.005, 0.005), (0.002, 0.003), (0.001, 0.002),
        ]
    
    for param, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        K = max(5, int((2 * n - 1) * K_frac))
        
        if use_power:
            prob = _PowerSmoProb(param, K, n)
        else:
            beta = base_beta * param
            prob = _LogSmoProb(beta, K, n)
        
        est_time = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 25))))
        
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 40}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _nesterov_active_set(a_init, deadline, min_time_remaining=60.0):

    tol_max = 0.05
    
    a_prev = a.copy()

    max_stall = 100

        iteration += 1

        y = a + gamma * (a - a_prev)
        y = np.maximum(y, 0.0)

            y = a.copy()
            sy = a.sum()

        cy = _full_convolve(y, y)

                    break
                a_try = a_try / s_try
        
        if improved:

                tol = max(tol_min, tol * 0.99)

def _coordinate_descent_polish(a_init, deadline, min_time_remaining=10.0):
    """Coordinate descent on the top contributing variables for peak reduction."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    patience = 0
    max_patience = 500
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        
        # Find top peak
        k_peak = int(np.argmax(c))
        
        # Compute which variables most contribute to this peak
        # c[k] = sum_i a[i]*a[k-i], so dc[k]/da[i] = a[k-i] + a[i-k]
        i_lo = max(0, k_peak - (n - 1))
        i_hi = min(n - 1, k_peak)
        contrib = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
            j = k_peak - i
            contrib[i] = a[j]
            if i != j and j >= 0 and j < n:
                contrib[j] += a[i]
        
        # Try reducing top contributors and redistributing
        n_top = max(5, int(n * 0.005))
        top_idx = np.argsort(contrib)[::-1][:n_top]
        low_idx = np.argsort(contrib)[:n_top]
        
        # Optimal redistribution: reduce top, increase bottom
        for trial in range(3):
            frac = [0.001, 0.003, 0.0005][trial]
            delta = np.zeros(n)
            red_total = 0.0
            for idx in top_idx:
                if a[idx] > 1e-10:
                    delta[idx] = -frac * a[idx]
                    red_total += -delta[idx]
            # Redistribute to low contributors
            red_total = max(red_total, 1e-15)
            for idx in low_idx:
                delta[idx] = red_total / n_top
            
            a_try = a + delta
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                break
    
    return best_a, best_sc

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.4 + 0.6 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_exp":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * t) + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + 0.1
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2) + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss_center":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_strong":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * t) + 0.02
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power_taper":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 0.8 + (t) ** 0.3 * 0.1 + 0.01
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _structural_mutate(a, rng, mutation_type=0):
    """Apply structural mutations to diversify the population."""
    n = len(a)
    a = a.copy()
    
    if mutation_type == 0:
        # Flip a random segment
        start = rng.integers(0, n // 2)
        length = rng.integers(n // 10, n // 3)
        end = min(start + length, n)
        a[start:end] = a[start:end][::-1]
    elif mutation_type == 1:
        # Cross-correlate with a shifted version
        shift = rng.integers(n // 10, n // 2)
        a_new = a.copy()
        a_new[shift:] = a[:-shift] * 0.5 + a[shift:] * 0.5
        a = a_new
    elif mutation_type == 2:
        # Add structured noise (sine modulation)
        freq = rng.uniform(2.0, 10.0)
        phase = rng.uniform(0, 2 * np.pi)
        modulation = 1.0 + 0.2 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        a = a * modulation
    elif mutation_type == 3:
        # Block permutation
        n_blocks = rng.integers(4, 10)
        block_size = n // n_blocks
        indices = list(range(n_blocks))
        rng.shuffle(indices)
        a_new = np.zeros(n)
        for new_i, old_i in enumerate(indices):
            s = old_i * block_size
            e = min(s + block_size, n)
            a_new[new_i * block_size:(new_i + 1) * block_size] = a[s:e]
        a = a_new
    else:
        # Random Gaussian perturbation in log-space
        z = np.log(np.maximum(a, 1e-30))
        z = z + 0.1 * rng.standard_normal(n)
        z = z - z.max()
        a = np.exp(z)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def _coarse_to_fine(a_init, deadline, min_time_remaining=60.0):
    """Multi-resolution: optimize on coarser grid, then refine."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    # Coarse resolutions
    resolutions = []
    nr = n
    while nr >= n // 8 and nr >= 1024:
        resolutions.append(nr)
        nr //= 2
    resolutions.sort()  # smallest first
    
    # Also try the original
    if n not in resolutions:
        resolutions.append(n)
    
    for res_n in resolutions:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 20:
            break
        
        if res_n == n:
            a_cur = _normalize(best_a.copy())
        else:
            a_cur = _resample(best_a, res_n)
        
        # Optimize at this resolution
        a_opt = _run_lbfgs(a_cur, min(deadline - min_time_remaining, time.monotonic() + remaining * 0.4),
                           min_time_remaining=5)
        sc_opt = _score(a_opt)
        
        # Upsample to next resolution
        if res_n < n:
            a_up = _resample(a_opt, min(res_n * 2, n))
            a_up_sc = _score(a_up)
            if a_up_sc < best_sc:
                best_a = a_up
                best_sc = a_up_sc
        else:
            if sc_opt < best_sc:
                best_a = _normalize(a_opt)
                best_sc = sc_opt
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer with power-sum smoothing, Nesterov acceleration,
    coarse-to-fine resolution, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Try to load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, tapered n={n_parent}, score={best_score:.10f}")
    
    n_main = len(best_seq)
    
    # === Phase 1: Power-sum L-BFGS from parent ===
    print(f"\n[P1] Power-sum L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p1_dl = min(deadline - 1200, time.monotonic() + 500)
    a1 = _run_lbfgs(best_seq, p1_dl, min_time_remaining=5, use_power=True)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [power_lbfgs] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 2: Nesterov active-set ===
    print(f"\n[P2] Nesterov active-set (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        p2_dl = min(deadline - 500, time.monotonic() + 500)
        a2, sc2 = _nesterov_active_set(best_seq, p2_dl, min_time_remaining=5)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [nesterov] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 3: More L-BFGS rounds with different configs ===
    print(f"\n[P3] L-BFGS rounds (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        p3_dl = min(deadline - 350, time.monotonic() + 350)
        a3, sc3 = _power_smo_lbfgs_round(best_seq, p3_dl, min_time_remaining=5)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_rounds] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 4: Coordinate descent polish ===
    print(f"\n[P4] Coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        p4_dl = min(deadline - 150, time.monotonic() + 200)
        a4, sc4 = _coordinate_descent_polish(best_seq, p4_dl, min_time_remaining=5)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [coord] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 5: Fine polish ===
    print(f"\n[P5] Fine polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        p5_dl = min(deadline - 100, time.monotonic() + 150)
        a5, sc5 = _fine_polish(best_seq, p5_dl, min_time_remaining=5)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [polish] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 6: Diverse restarts with structural mutations ===
    print(f"\n[P6] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        restart_budget = min(300, (deadline - 80 - time.monotonic()) * 0.6)
        n_restart = max(3, int(restart_budget / 50))
        
        strategies = [
            # (mutation_type, sigma, method)
            (0, 0.05, "lbfgs"), (1, 0.03, "lbfgs"), (2, 0.08, "nesterov"),
            (3, 0.04, "lbfgs"), (0, 0.1, "lbfgs"), (2, 0.05, "lbfgs"),
            (1, 0.06, "nesterov"), (0, 0.02, "lbfgs"),
        ]
        
        for i in range(n_restart):
            if time.monotonic() > deadline - 40:
                break
            strat = strategies[i % len(strategies)]
            mut_type, sigma, method = strat
            
            a_pert = _structural_mutate(best_seq, rng, mut_type)
            a_pert = _perturb(a_pert, sigma, rng)
            
            if method == "lbfgs":
                t_rem = min(deadline - 30, time.monotonic() + 45)
                a_r = _run_lbfgs(a_pert, t_rem, min_time_remaining=3)
            else:
                t_rem = min(deadline - 30, time.monotonic() + 40)
                a_r, _ = _nesterov_active_set(a_pert, t_rem, min_time_remaining=3)
            
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.10f} [restart_{i}_{method}] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 7: Different profiles ===
    print(f"\n[P7] Profile exploration (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        profiles = ["asym_exp", "asym_gauss", "power_taper", "asym_strong", "bimodal"]
        for prof_kind in profiles:
            if time.monotonic() > deadline - 30:
                break
            a_prof = _make_profile(n_main, prof_kind)
            t_p = min(deadline - 20, time.monotonic() + 40)
            a_p = _run_lbfgs(a_prof, t_p, min_time_remaining=3, use_power=True)
            sc_p = _score(a_p)
            if sc_p < best_score:
                best_seq = a_p.copy()
                best_score = sc_p
                print(f"  NEW BEST: {best_score:.10f} [profile_{prof_kind}] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 8: Multi-n exploration ===
    print(f"\n[P8] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 50:
        n_options = [8192, 32768]
        n_options = [x for x in n_options if x != n_main and x >= 1024]
        
        for n_try in n_options[:2]:
            if time.monotonic() > deadline - 20:
                break
            a_res = _resample(best_seq, n_try)
            t_r = min(deadline - 10, time.monotonic() + 45)
            a_opt = _run_lbfgs(a_res, t_r, min_time_remaining=3, use_power=True)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [n={n_try}] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 9: Final fine polish ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3:
        a_fin, sc_fin = _fine_polish(best_seq, deadline - 1, min_time_remaining=1)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")
        
        # Tiny L-BFGS at the end
        remaining = deadline - 1 - time.monotonic()
        if remaining > 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            K_fin = max(5, int(n * 0.002))
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _PowerSmoProb(4096.0, K_fin, n)
            max_iter = min(2000, max(20, int(remaining / 0.02)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-14, 'maxcor': 50})
                a_fin2 = np.exp(result.x - result.x.max())
                a_fin2 = a_fin2 / a_fin2.sum()
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass
    
    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
