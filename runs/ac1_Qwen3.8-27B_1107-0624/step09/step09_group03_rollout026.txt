"""

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    """Compute the cross-correlation restricted to valid indices.
    (a * A_vec)[k] for k in [n-1, 2n-2], which equals the gradient of c[k] w.r.t. a (up to factor 2).
    Returns array of length n: result[j] = sum_k A_vec[k] * a[k-j] (valid range)
    """
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _lsobj_factory(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad_factory(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec, rhs, max_cg=500, tol=1e-14):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

    A_vec_copy = A_vec
    n_copy = n

    def _hess_vp(v):
        return 2.0 * _corr_a_A(v, A_vec_copy, n_copy) + mu_copy * v

def _line_search(a, dx, c_max, max_ls=60):
    """Armijo line search minimizing max(conv(a,a))."""
    n = len(a)
    best_a = a.copy()
    best_f = c_max
    t = 1.0
    for _ in range(max_ls):

        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-15:

        a_try = a_try / s_try

        f_try = float(c_try.max())

            best_a = a_try.copy()
            best_f = f_try
            break

    return best_a, best_f

            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Try multiple tolerance levels for active set
        best_improvement = 0
        best_a_new = None
        best_c_new = c_max
        
        # Threshold-based active sets
        tol_list = [0.0001, 0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05]
        for tol_frac in tol_list:
            active = c >= (c_max - tol_frac * c_max)
            m_act = int(active.sum())
            if m_act < 2 or m_act > 3000:
                continue
            
            dx = _newton_step_precise(a, c, c_max, active)
            if dx is None:
                continue
            
            a_try, f_try = _line_search(a, dx, c_max)
            if f_try < best_c_new - 1e-15:
                best_c_new = f_try
                best_a_new = a_try
                best_improvement = c_max - f_try
        
        # Greedy add: try adding one position to the current tight active set
        tight_active = c >= (c_max - 0.001 * c_max)
        m_tight = int(tight_active.sum())
        if 2 <= m_tight < 500:
            # Find the best position to add
            candidates = np.where(~tight_active)[0]
            if len(candidates) > 0:
                # Sample candidates (too many to try all)
                rng_add = np.random.default_rng(rnd + 42)
                n_sample = min(20, len(candidates))
                sampled = rng_add.choice(candidates, size=n_sample, replace=False)
                
                for k_add in sampled:
                    active_try = tight_active.copy()
                    active_try[k_add] = True
                    dx = _newton_step_precise(a, c, c_max, active_try)
                    if dx is None:
                        continue
                    a_try, f_try = _line_search(a, dx, c_max)
                    if f_try < best_c_new - 1e-15:
                        best_c_new = f_try
                        best_a_new = a_try
                        best_improvement = c_max - f_try
        
        # Greedy remove: try removing one position from the active set
        if 3 <= m_tight < 100:
            active_positions = np.where(tight_active)[0]
            # Sort by "importance" (how close to max they are)
            c_vals = c[active_positions]
            order = np.argsort(c_vals)  # least important first
            n_remove_try = min(5, len(order))
            
            for idx in order[:n_remove_try]:
                k_remove = active_positions[idx]
                active_try = tight_active.copy()
                active_try[k_remove] = False
                if int(active_try.sum()) < 2:
                    continue
                dx = _newton_step_precise(a, c, c_max, active_try)
                if dx is None:
                    continue
                a_try, f_try = _line_search(a, dx, c_max)
                if f_try < best_c_new - 1e-15:
                    best_c_new = f_try
                    best_a_new = a_try
                    best_improvement = c_max - f_try
        
        if best_a_new is not None:
            a = best_a_new
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
            if best_improvement < 1e-14:
                stall += 1
        else:
            stall += 1
        
        if stall > 30:
            # Try a small perturbation to escape
            rng_esc = np.random.default_rng(rnd + 1000)
            sigma = 0.002 * (1 + rnd / 50.0)
            z = np.log(np.maximum(a, 1e-30))
            z = z + sigma * rng_esc.standard_normal(n)
            z = z - z.mean()
            a = _softmax(z)
            stall = 0
    
    return best_a, best_sc


def _lbfgs_pass(a_init, deadline, min_time_remaining=30.0, n_betas=20):
    """L-BFGS with log-sum-exp continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 30:
        gap = c_sorted[0] - c_sorted[30]
    else:
        gap = c_sorted[0] - c_sorted[-1]
    base_beta = max(gap * 0.2, max_c * 0.005)
    
    log_betas = np.linspace(np.log(base_beta), max(np.log(max_c * 1e-8), -30), n_betas)
    betas = np.exp(log_betas)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        
        est_time = 0.004 * max(1.0, n / 8192.0)
        max_iter = min(3000, max(20, int(remaining / (est_time * 8))))
        
        try:
            def _obj(z_val):
                return _lsobj_factory(z_val, beta, n)
            
            def _grad(z_val):
                return _lsgrad_factory(z_val, beta, n)
            
            result = minimize(
                _obj, z, jac=_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-17, 'gtol': 1e-13,
                         'maxcor': 80, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.12f}")
        except Exception as ex:
            print(f"    L-BFGS beta={beta:.2e}: error {ex}")
    
    return current_a


def _deflation_pass(a, deadline, n_attempts=50):
    """Multi-peak deflation."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find peaks
        threshold = c_max * (0.98 + 0.02 * attempt / max(n_attempts, 1))
        peak_mask = c >= threshold
        n_peaks = int(peak_mask.sum())
        if n_peaks == 0:
            peak_mask[np.argmax(c)] = True
            n_peaks = 1
        
        w = np.zeros_like(c)
        w[peak_mask] = 1.0 / n_peaks
        grad = 2.0 * _corr_a_A(a, w, n)
        g_t = grad - np.mean(grad)
        
        applied = False
        for strength in [0.0002, 0.0005, 0.001, 0.002, 0.005, 0.01]:
            a_try = a - strength * g_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                applied = True
                break
        
        if not applied:
            break
    
    return best_a, best_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _structured_perturb(a, rng):
    """Apply structured perturbations (block shifts, ramps, etc.)"""
    n = len(a)
    kind = rng.integers(0, 5)
    if kind == 0:
        # Random block swap
        b = n // 4
        i = rng.integers(0, n - b)
        j = rng.integers(0, n - b)
        a_try = a.copy()
        a_try[i:i+b], a_try[j:j+b] = a_try[j:j+b].copy(), a_try[i:i+b].copy()
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s > 0:
            a_try = a_try / s
        return a_try
    elif kind == 1:
        # Ramp perturbation
        t = np.linspace(0, 1, n)
        amp = 0.05 * rng.uniform(0.5, 1.5)
        a_try = a * (1 + amp * (t - 0.5))
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s > 0:
            a_try = a_try / s
        return a_try
    elif kind == 2:
        # Sine perturbation
        freq = rng.uniform(1, 10)
        amp = 0.03 * rng.uniform(0.5, 1.5)
        t = np.linspace(0, 1, n)
        a_try = a * (1 + amp * np.sin(2 * np.pi * freq * t))
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s > 0:
            a_try = a_try / s
        return a_try
    elif kind == 3:
        # Gaussian bump addition
        center = rng.uniform(0.1, 0.9)
        width = rng.uniform(0.02, 0.1)
        t = np.linspace(0, 1, n)
        bump = np.exp(-0.5 * ((t - center) / width) ** 2)
        amp = 0.1 * rng.uniform(0.5, 1.5)
        a_try = a + amp * bump
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s > 0:
            a_try = a_try / s
        return a_try
    else:
        # Log-space perturbation
        return _perturb(a, 0.03, rng)


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = np.exp(-2 * (t - 0.3) ** 2)
        return a / a.sum()
    elif kind == "bipolar_soft":
        t = np.linspace(0, 1, n)
        a = np.exp(-4 * (t - 0.25) ** 2) + np.exp(-4 * (t - 0.75) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    1. Quick L-BFGS from parent
    2. Refined Newton with active set enumeration
    3. Deflation
    4. Structured restarts
    5. Multi-n (8192, 16384, 32768)
    6. Final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 35.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.12f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass
    
    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={len(parent)}, score={sc_parent:.12f}")
        update_best(parent, sc_parent, "parent")
    else:
        parent = _make_profile(16384, "uniform")
        update_best(parent, _score(parent), "init")
    
    # === Phase 1: Quick L-BFGS to settle (200s) ===
    print(f"\n[Phase 1] L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    lbfgs_dl = min(deadline - 600, time.monotonic() + 200)
    a_lbfgs = _lbfgs_pass(parent, lbfgs_dl, min_time_remaining=600.0, n_betas=15)
    update_best(a_lbfgs, _score(a_lbfgs), "lbfgs1")
    
    # === Phase 2: Refined Newton with active set refinement (400s) ===
    print(f"\n[Phase 2] Refined Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        nw_dl = min(deadline - 400, time.monotonic() + 400)
        a_nw, sc_nw = _active_set_refinement(best_seq, nw_dl, max_rounds=500, label="main")
        update_best(a_nw, sc_nw, "newton_main")
    
    # === Phase 3: Deflation (100s) ===
    print(f"\n[Phase 3] Deflation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        def_dl = min(deadline - 300, time.monotonic() + 100)
        a_def, sc_def = _deflation_pass(best_seq, def_dl, n_attempts=80)
        update_best(a_def, sc_def, "deflation")
    
    # === Phase 4: Second Newton round (200s) ===
    print(f"\n[Phase 4] Newton round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        nw2_dl = min(deadline - 250, time.monotonic() + 200)
        a_nw2, sc_nw2 = _active_set_refinement(best_seq, nw2_dl, max_rounds=300, label="round2")
        update_best(a_nw2, sc_nw2, "newton_r2")
    
    # === Phase 5: Structured restarts (200s) ===
    print(f"\n[Phase 5] Structured restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        n_best = len(best_seq)
        for i in range(20):
            if time.monotonic() > deadline - 200:
                break
            a_pert = _structured_perturb(best_seq, rng)
            # Quick Newton on perturbed
            quick_dl = min(deadline - 200, time.monotonic() + 8)
            a_q, sc_q = _active_set_refinement(a_pert, quick_dl, max_rounds=30, label=f"restart_{i}")
            update_best(a_q, sc_q, f"restart_{i}")
            
            # Also try L-BFGS on perturbed
            if time.monotonic() < deadline - 200:
                lbfgs_q_dl = min(deadline - 200, time.monotonic() + 5)
                a_lq = _lbfgs_pass(a_pert, lbfgs_q_dl, min_time_remaining=200.0, n_betas=8)
                update_best(a_lq, _score(a_lq), f"restart_lbfgs_{i}")
    
    # === Phase 6: Try n=32768 (150s) ===
    print(f"\n[Phase 6] n=32768 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 120:
        n_large = 32768
        a_large = _interpolate_seq(best_seq, n_large) if best_seq is not None else _make_profile(n_large, "uniform")
        sc_large_init = _score(a_large)
        print(f"  n={n_large} init score: {sc_large_init:.12f}")
        
        if time.monotonic() < deadline - 120:
            lbfgs_l_dl = min(deadline - 120, time.monotonic() + 60)
            a_large_lbfgs = _lbfgs_pass(a_large, lbfgs_l_dl, min_time_remaining=120.0, n_betas=10)
            update_best(a_large_lbfgs, _score(a_large_lbfgs), "n32768_lbfgs")
        
        if time.monotonic() < deadline - 120:
            nw_l_dl = min(deadline - 120, time.monotonic() + 40)
            a_large_nw, sc_large_nw = _active_set_refinement(a_large_lbfgs, nw_l_dl, max_rounds=100)
            update_best(a_large_nw, sc_large_nw, "n32768_newton")
    
    # === Phase 7: Try n=8192 (100s) ===
    print(f"\n[Phase 7] n=8192 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 50:
        n_small = 8192
        a_small = _interpolate_seq(best_seq, n_small) if best_seq is not None else _make_profile(n_small, "uniform")
        sc_small_init = _score(a_small)
        print(f"  n={n_small} init score: {sc_small_init:.12f}")
        
        if time.monotonic() < deadline - 50:
            lbfgs_s_dl = min(deadline - 50, time.monotonic() + 30)
            a_small_lbfgs = _lbfgs_pass(a_small, lbfgs_s_dl, min_time_remaining=50.0, n_betas=10)
            update_best(a_small_lbfgs, _score(a_small_lbfgs), "n8192_lbfgs")
        
        if time.monotonic() < deadline - 50:
            nw_s_dl = min(deadline - 50, time.monotonic() + 25)
            a_small_nw, sc_small_nw = _active_set_refinement(a_small_lbfgs, nw_s_dl, max_rounds=80)
            update_best(a_small_nw, sc_small_nw, "n8192_newton")
    
    # === Phase 8: Final polish on best (remaining time) ===
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        # Final Newton
        fin_dl = min(deadline - 5, time.monotonic() + 40)
        a_fin, sc_fin = _active_set_refinement(best_seq, fin_dl, max_rounds=100, label="final")
        update_best(a_fin, sc_fin, "final_newton")
        
        # Very small beta L-BFGS
        if time.monotonic() < deadline - 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 1e-8
            remaining = deadline - 5 - time.monotonic()
            if remaining > 3:
                max_iter_fin = min(1500, max(30, int(remaining / 0.05)))
                try:
                    def _obj_f(z_val):
                        return _lsobj_factory(z_val, beta_final, n)
                    def _grad_f(z_val):
                        return _lsgrad_factory(z_val, beta_final, n)
                    result = minimize(_obj_f, z, jac=_grad_f, method='L-BFGS-B',
                                     options={'maxiter': max_iter_fin, 'ftol': 1e-18,
                                              'gtol': 1e-14, 'maxcor': 80})
                    a_fin2 = _softmax(result.x)
                    update_best(a_fin2, _score(a_fin2), "final_tiny_beta")
                except Exception:
                    pass
    
    # Also try a few fresh starts with different profiles
    print(f"\n[Phase 9] Fresh starts (t={time.monotonic() - t_start:.1f}s)")
    for profile_kind in ["asymmetric", "bipolar_soft", "raised_cosine"]:
        if time.monotonic() > deadline - 20:
            break
        n_try = len(best_seq) if best_seq is not None else 16384
        a_fresh = _make_profile(n_try, profile_kind)
        # Quick optimization
        fresh_dl = min(deadline - 20, time.monotonic() + 15)
        a_fresh_opt = _lbfgs_pass(a_fresh, fresh_dl, min_time_remaining=20.0, n_betas=8)
        update_best(a_fresh_opt, _score(a_fresh_opt), f"fresh_{profile_kind}")
    
    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
