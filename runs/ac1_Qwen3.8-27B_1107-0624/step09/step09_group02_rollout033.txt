    # So dc[2]/da[1] = a[1] + a[0]? That gives a0+a1, but the correct answer is 2*a1.
    # 
    # I think the issue is that when k=2m (here k=2, m=1), the two terms are the SAME term.
    # c[k] = sum_i a[i]*a[k-i]. For k=2: i=0: a[0]*a[2], i=1: a[1]*a[1], i=2: a[2]*a[0]
    # dc[2]/da[1] = d/d a[1] (a[0]*a[2] + a[1]^2 + a[2]*a[0]) = 2*a[1]
    # 
    # So the correct formula is:
    # dc[k]/da[m] = a[k-m] + a[m-(k-0)]... no.
    # dc[k]/da[m]: c[k] = sum_i a[i]*a[k-i]
    # d/d a[m] = sum_i [delta(i,m)*a[k-i] + a[i]*delta(k-i,m)]
    # = a[k-m] + a[m] * delta(k-m, m)... no.
    # = a[k-m] (from i=m) + sum_i a[i]*delta(k-i,m) (from the second factor)
    # = a[k-m] + a[k-m] (from i=k-m in the second sum, if k-m valid)
    # Wait: sum_i a[i]*delta(k-i,m) = a[k-m] (when i=k-m)
    # So dc[k]/da[m] = a[k-m] + a[k-m] = 2*a[k-m] when both i=m and i=k-m are valid and m != k-m
    # When m = k-m (i.e., k=2m): dc[k]/da[m] = a[m] + a[m] = 2*a[m] (same term counted twice? No!)
    # 
    # Let me be very careful. c[k] = sum_{i=0}^{n-1} a[i]*a[k-i] * [0<=k-i<n]
    # dc[k]/da[m] = sum_i [delta_{i,m} * a[k-i] * [0<=k-i<n] + a[i] * delta_{k-i,m} * [0<=i<n]]
    # = a[k-m] * [0<=k-m<n] + a[m] * [0<=m<n] * [0<=k-m<n]  (from the second term, i=k-m... wait)
    # 
    # Second term: sum_i a[i] * delta_{k-i,m} = a[k-m] (when i = k-m, and we need 0<=k-m<n)
    # So dc[k]/da[m] = a[k-m] * [0<=k-m<n] + a[k-m] * [0<=k-m<n] = 2*a[k-m] * [0<=k-m<n]
    # 
    # Wait that can't be right either. Let me redo.
    # c[k] = sum_{i: 0<=i<n AND 0<=k-i<n} a[i] * a[k-i]
    # dc[k]/da[m] = sum_{i: valid} [delta_{i,m} * a[k-i] + a[i] * delta_{k-i,m}]
    # First part: a[k-m] if m is valid (0<=m<n, always true) and k-m is valid (0<=k-m<n)
    # Second part: a[i] where k-i=m, i.e., i=k-m. Valid if 0<=k-m<n. Value: a[k-m].
    # Total: 2*a[k-m] if 0<=k-m<n, else a[k-m] if only one is valid... 
    # Actually both conditions are the same: 0<=k-m<n. So dc[k]/da[m] = 2*a[k-m] if 0<=k-m<n, else 0.
    # 
    # Let me verify: n=3, k=2, m=1: 0<=2-1=1<3 ✓, so dc[2]/da[1] = 2*a[1]. ✓
    # n=3, k=2, m=0: 0<=2-0=2<3 ✓, so dc[2]/da[0] = 2*a[2]. 
    # Check: c[2] = a[0]a[2]+a[1]^2+a[2]a[0] = 2a0a2+a1^2. dc[2]/da[0] = 2a2. ✓
    
    # So the gradient of sum_k w[k]*c[k] w.r.t. a[m] is:
    # sum_k w[k] * 2*a[k-m] * [0<=k-m<n]
    # = 2 * sum_{k: 0<=k-m<n} w[k]*a[k-m]
    # = 2 * (w * a_rev)[n-1+m]  where a_rev[j] = a[n-1-j]
    #   because (w*a_rev)[n-1+m] = sum_k w[k]*a_rev[n-1+m-k] = sum_k w[k]*a[n-1-(n-1+m-k)] = sum_k w[k]*a[k-m]
    #   (with the understanding that a[j]=0 for j outside [0,n-1])
    
    # So: grad_a[m] = p * 2 * (w * a_rev)[n-1+m]
    
    # Great, so the gradient is simply 2p times the convolution!
    return p * 2.0 * conv

def _run_power_homotopy(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with power-mean homotopy: start with p=2, increase to p=64."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    # Power schedule
    p_schedule = [2, 4, 8, 16, 32, 64, 128, 256]
    
    for p in p_schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(2000, max(20, int(remaining / (est_time * 8))))
        
        try:
            def obj_func(z_local, p=p):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                # Power mean objective (normalized)
                s_c = float(c_local.sum())
                if s_c < 1e-300:
                    return 2.0 * n * float(c_local.max())
                pm = float(np.mean(c_local ** p)) ** (1.0 / p)
                return 2.0 * n * pm
            
            def grad_func(z_local, p=p):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                # d/dz [2n * (mean(c^p))^(1/p)] 
                # = 2n * (1/p) * (mean(c^p))^(1/p - 1) * p * mean(c^{p-1}) * dc/dz
                # = 2n * (mean(c^p))^(1/p - 1) * mean(c^{p-1}) * dc/dz
                # where dc/dz[m] = dc/da[m] * a[m] (chain rule for softmax)
                # dc/da[m] = 2 * sum_k c[k]^{p-1}... wait no, we're differentiating the power mean, not sum c^p
                
                # Let's use: f = 2n * (1/(2n-1) * sum c^p)^(1/p)
                # df/dc[k] = 2n * (1/p) * (1/(2n-1))^{1/p - 1} * (sum c^p)^{1/p - 1} * p * c[k]^{p-1} / (2n-1)
                # = 2n * (1/(2n-1)) * (sum c^p / (2n-1))^{1/p - 1} * c[k]^{p-1}
                # = 2n * (mean(c^p))^{1/p - 1} * c[k]^{p-1} / (2n-1)
                
                # Simpler: let g = sum c^p, f = 2n * (g/(2n-1))^{1/p}
                # df/dg = 2n * (1/p) * (g/(2n-1))^{1/p - 1} * 1/(2n-1)
                # df/dc[k] = df/dg * p * c[k]^{p-1} = 2n * (1/(2n-1)) * (g/(2n-1))^{1/p-1} * c[k]^{p-1}
                # df/da[m] = sum_k df/dc[k] * dc[k]/da[m] = 2 * sum_k df/dc[k] * c[k]^{p-1} * a[k-m] * [0<=k-m<n]
                #           = 2 * (w * a_rev)[n-1+m] where w[k] = df/dc[k]
                # df/dz[m] = df/da[m] * a[m]
                
                L = 2 * n - 1
                g = float(np.sum(c_local ** p))
                if g < 1e-300:
                    return np.zeros_like(z_local)
                factor = 2.0 * n / L * (g / L) ** (1.0 / p - 1.0)
                w = factor * c_local ** (p - 1)
                # df/da = 2 * (w * a_rev)[n-1:n-1+n]
                a_rev = a_local[::-1]
                conv = _full_convolve(w, a_rev)
                grad_a = 2.0 * conv[n - 1:2 * n - 1]
                # Chain rule: grad_z = grad_a * a
                grad_z = grad_a * a_local
                return grad_z
            
            result = minimize(
                obj_func, z,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 50, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            # Even if score didn't improve, update z for next p level
            a_new = _softmax(result.x)
            z = np.log(np.maximum(a_new, 1e-30))
            z = z - z.mean()
        except Exception as e:
            print(f"  Power p={p} failed: {e}")
    
    return best_a

def _run_active_set(a_init, deadline, min_time_remaining=30.0):
    """Adaptive active-set gradient descent with geometrically shrinking tolerance."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    momentum = np.zeros(n)
    beta_m = 0.85
    patience = 0
    max_patience = 40
    
    for step in range(5000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        a_curr = _softmax(z)
        c = _full_convolve(a_curr, a_curr)
        c_max = float(c.max())
        
        # Adaptive active set: start with wide, narrow down
        progress = min(1.0, step / 2000.0)
        tol_frac = 0.01 * (1.0 - progress) + 0.0001 * progress
        active = c >= c_max * (1.0 - tol_frac)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        p_weights = np.zeros(2 * n - 1)
        p_weights[active] = 1.0 / active.sum()
        
        # Gradient
        conv = _full_convolve(p_weights, a_curr[::-1])
        grad_a = 2.0 * conv[n - 1:2 * n - 1]
        grad_z = grad_a * a_curr
        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-18:
            break
        
        # Adam-like step
        momentum = beta_m * momentum + (1 - beta_m) * grad_z
        lr = 0.01 / (np.linalg.norm(momentum) / grad_norm + 1e-10)
        z_new = z - lr * momentum
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new = z_new - z_new.mean()
        z = z_new
        
        # Periodic evaluation
        if step % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros(n)
                    patience = 0
                    beta_m = max(0.5, beta_m * 0.95)
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _run_peak_shave(a_init, deadline, min_time_remaining=10.0):
    """Direct peak shaving: reduce the highest autocorrelation values."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(100):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find the top few peaks
        k_peaks = np.argsort(c)[::-1][:min(5, len(c))]
        p = np.zeros(2 * n - 1)
        p[k_peaks] = 1.0 / len(k_peaks)
        
        # Gradient of sum of peak values
        conv = _full_convolve(p, a[::-1])
        grad_a = 2.0 * conv[n - 1:2 * n - 1]
        gnorm = np.linalg.norm(grad_a)
        if gnorm < 1e-16:
            break
        
        # Line search
        step = 0.05 / (gnorm + 1e-10)
        improved = False
        for ls in range(40):
            a_try = _proj_simplex(a - step * grad_a)
            c_try = _full_convolve(a_try, a_try)
            if float(c_try.max()) < c_max * (1 - 1e-12):
                a = a_try
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                improved = True
                break
            step *= 0.5
            if step < 1e-14:
                break
        if not improved:
            break
    
    return best_a, best_sc

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power1":
        t = np.linspace(0.01, 1, n)
        a = t ** 0.5
        return a / a.sum()
    elif kind == "power2":
        t = np.linspace(0.01, 1, n)
        a = t ** 1.0
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.75) ** 2)
        return a / a.sum()
    elif kind == "w_shape":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.abs(np.sin(2 * np.pi * t))
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def _run_lbfgs_lse(a_init, deadline, min_time_remaining=20.0):
    """L-BFGS with log-sum-exp smoothing (improved version)."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap * 0.05, max_c * 0.0005)
    
    ratios = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
              0.0005, 0.0002, 0.0001, 0.00005]
    
    for ratio in ratios:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        beta = base_beta * ratio
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(2000, max(20, int(remaining / (est_time * 6))))
        
        try:
            def obj_func(z_local, beta=beta):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    return 2.0 * n * c_max_local
                return 2.0 * n * (c_max_local + beta * np.log(s_e))
            
            def grad_func(z_local, beta=beta):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    w = np.zeros_like(c_local)
                    w[int(np.argmax(c_local))] = 1.0
                else:
                    w = e / s_e
                conv = _full_convolve(w, a_local[::-1])
                grad_a = 2.0 * conv[n - 1:2 * n - 1]
                return grad_a * a_local
            
            result = minimize(
                obj_func, z,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 50, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
            z = np.log(np.maximum(_softmax(result.x), 1e-30))
            z = z - z.mean()
        except Exception:
            pass
    
    return best_a

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer for minimum peak autocorrelation sequences.

    """

        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)

        if s < 0.01:
            return

            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()

"""
Multi-strategy peak-autocorrelation minimizer using power-mean homotopy,
log-sum-exp L-BFGS, adaptive active-set gradient descent, and peak shaving.
Key improvements over previous approach:
- Power-mean (p-norm) homotopy for smoother convergence landscape
- Adaptive active-set that geometrically shrinks
- Two-stage optimization (flatten then peak-reduce)
- Better time allocation and diverse initializations
- Correct gradient chain-rule for softmax parameterization
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _grad_apex(w, a, n):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = a*a.
    Result: 2 * (w * a_rev)[n-1 : 2n-1]
    """
    a_rev = a[::-1]
    conv = _full_convolve(w, a_rev)
    return 2.0 * conv[n - 1:2 * n - 1]


def _run_power_homotopy(a_init, deadline, min_time_remaining=25.0):
    """L-BFGS with power-mean homotopy: p goes from 2 to 256."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    L = 2 * n - 1

    p_schedule = [2, 4, 8, 16, 32, 64, 128, 256, 512]

    for p in p_schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(1500, max(15, int(remaining / (est_time * 10))))

        try:
            def obj_pm(z_local, p=p):
                a_l = _softmax(z_local)
                c_l = _full_convolve(a_l, a_l)
                s_c = float(np.sum(c_l ** p))
                if s_c < 1e-300:
                    return 2.0 * n * float(c_l.max())
                return 2.0 * n * (s_c / L) ** (1.0 / p)

            def grad_pm(z_local, p=p):
                a_l = _softmax(z_local)
                c_l = _full_convolve(a_l, a_l)
                s_c = float(np.sum(c_l ** p))
                if s_c < 1e-300:
                    return np.zeros_like(z_local)
                # df/dc[k] = (2n/L) * (s_c/L)^(1/p - 1) * c[k]^(p-1)
                factor = (2.0 * n / L) * (s_c / L) ** (1.0 / p - 1.0)
                w = factor * np.power(np.maximum(c_l, 0.0), p - 1)
                grad_a = _grad_apex(w, a_l, n)
                # Chain rule for softmax: grad_z = a * grad_a (up to constant)
                return grad_a * a_l

            result = minimize(
                obj_pm, z,
                jac=grad_pm,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 50, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
            # Always update z for next p level
            z = np.log(np.maximum(_softmax(result.x), 1e-30))
            z = z - z.mean()
        except Exception as e:
            pass

    return best_a


def _run_lbfgs_lse(a_init, deadline, min_time_remaining=20.0):
    """L-BFGS with log-sum-exp smoothing and adaptive beta."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    base_beta = max(gap * 0.03, max_c * 0.0003)

    ratios = [1.0, 0.4, 0.15, 0.06, 0.02, 0.008, 0.003, 0.001,
              0.0004, 0.00015, 0.00006, 0.00002, 0.000008, 0.000003]

    for ratio in ratios:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        beta = base_beta * ratio
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(2000, max(15, int(remaining / (est_time * 7))))

        try:
            def obj_lse(z_local, beta=beta):
                a_l = _softmax(z_local)
                c_l = _full_convolve(a_l, a_l)
                c_mx = float(c_l.max())
                shifted = (c_l - c_mx) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    return 2.0 * n * c_mx
                return 2.0 * n * (c_mx + beta * np.log(s_e))

            def grad_lse(z_local, beta=beta):
                a_l = _softmax(z_local)
                c_l = _full_convolve(a_l, a_l)
                c_mx = float(c_l.max())
                shifted = (c_l - c_mx) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    w = np.zeros_like(c_l)
                    w[int(np.argmax(c_l))] = 1.0
                else:
                    w = e / s_e
                grad_a = _grad_apex(w, a_l, n)
                return grad_a * a_l

            result = minimize(
                obj_lse, z,
                jac=grad_lse,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 50, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
            z = np.log(np.maximum(_softmax(result.x), 1e-30))
            z = z - z.mean()
        except Exception:
            pass

    return best_a


def _run_active_set(a_init, deadline, min_time_remaining=20.0):
    """Adaptive active-set gradient descent with geometrically shrinking tolerance."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    momentum = np.zeros(n)
    beta_m = 0.85
    patience = 0
    max_patience = 50
    step_count = 0

    while time.monotonic() + min_time_remaining < deadline and step_count < 8000:
        step_count += 1
        a_curr = _softmax(z)
        c = _full_convolve(a_curr, a_curr)
        c_max = float(c.max())

        # Adaptive active set: wide early, narrow late
        progress = min(1.0, step_count / 3000.0)
        tol_frac = 0.02 * (1.0 - progress) ** 2 + 0.00005 * progress
        active = c >= c_max * (1.0 - tol_frac)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        p_w = np.zeros(2 * n - 1)
        p_w[active] = 1.0 / active.sum()

        grad_a = _grad_apex(p_w, a_curr, n)
        grad_z = grad_a * a_curr
        gnorm = np.linalg.norm(grad_z)
        if gnorm < 1e-18:
            break

        momentum = beta_m * momentum + (1.0 - beta_m) * grad_z
        mnorm = np.linalg.norm(momentum)
        lr = 0.015 * (1.0 + 0.05 * min(step_count, 100)) / (mnorm / (gnorm + 1e-15) + 1e-10)
        z_new = z - lr * momentum
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new = z_new - z_new.mean()
        z = z_new

        if step_count % 25 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros(n)
                    patience = 0
                    beta_m = max(0.5, beta_m * 0.93)

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _run_peak_shave(a_init, deadline, min_time_remaining=10.0):
    """Direct peak shaving via projected gradient on top peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(200):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Top peaks with adaptive count
        k_top = min(max(3, int(0.001 * (2 * n - 1))), 20)
        idx = np.argsort(c)[::-1][:k_top]
        p = np.zeros(2 * n - 1)
        p[idx] = 1.0 / k_top

        grad_a = _grad_apex(p, a, n)
        gnorm = np.linalg.norm(grad_a)
        if gnorm < 1e-16:
            break

        step = 0.04 / (gnorm + 1e-10)
        improved = False
        for ls in range(40):
            a_try = _proj_simplex(a - step * grad_a)
            c_try = _full_convolve(a_try, a_try)
            if float(c_try.max()) < c_max * (1.0 - 1e-13):
                a = a_try
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                improved = True
                break
            step *= 0.5
            if step < 1e-14:
                break
        if not improved:
            break

    return best_a, best_sc


def _run_variance_reduce(a_init, deadline, min_time_remaining=15.0):

        est_time = 0.004 * max(1.0, n / 4096.0)

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power_low":
        t = np.linspace(0.001, 1, n)
        a = t ** 0.3
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2) + 0.05
        return a / a.sum()
    elif kind == "w_shape":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.abs(np.sin(2 * np.pi * t))
        return a / a.sum()
    elif kind == "middle_heavy":
        t = np.linspace(0, 1, n)
        a = np.exp(-5.0 * (t - 0.5) ** 2) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining power-mean homotopy, LSE L-BFGS,
    adaptive active-set gradient, variance reduction, and peak shaving.
    Explores multiple sequence lengths and diverse initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 32768
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc)
    except Exception:
        pass

    if parent is None:
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent))

    # Phase 1: Variance reduction (flatten autocorrelation) - 120s
    print(f"\n[Phase 1] Variance reduction (t={time.monotonic()-t_start:.1f}s)")
    p1_dl = t_start + 120
    a_var, sc_var = _run_variance_reduce(parent, p1_dl, min_time_remaining=10.0)
    update_best(a_var, sc_var)

    # Phase 2: Power-mean homotopy - 300s
    print(f"[Phase 2] Power-mean homotopy (t={time.monotonic()-t_start:.1f}s)")
    p2_dl = t_start + 420
    if best_seq is not None:
        a_pm = _run_power_homotopy(best_seq, p2_dl, min_time_remaining=20.0)
        update_best(a_pm, _score(a_pm))

    # Phase 3: LSE L-BFGS - 300s
    print(f"[Phase 3] LSE L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p3_dl = t_start + 720
    if best_seq is not None:
        a_lse = _run_lbfgs_lse(best_seq, p3_dl, min_time_remaining=20.0)
        update_best(a_lse, _score(a_lse))

    # Phase 4: Adaptive active-set - 250s
    print(f"[Phase 4] Active-set gradient (t={time.monotonic()-t_start:.1f}s)")
    p4_dl = t_start + 970
    if best_seq is not None:
        a_as, sc_as = _run_active_set(best_seq, p4_dl, min_time_remaining=15.0)
        update_best(a_as, sc_as)

    # Phase 5: Peak shaving - 120s
    print(f"[Phase 5] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    p5_dl = t_start + 1090
    if best_seq is not None:
        a_ps, sc_ps = _run_peak_shave(best_seq, p5_dl, min_time_remaining=10.0)
        update_best(a_ps, sc_ps)

    # Phase 6: Round 2 - Power homotopy + LSE - 200s
    print(f"[Phase 6] Round 2 optimization (t={time.monotonic()-t_start:.1f}s)")
    p6_dl = t_start + 1290
    if best_seq is not None:
        a_pm2 = _run_power_homotopy(best_seq, p6_dl - 80, min_time_remaining=10.0)
        update_best(a_pm2, _score(a_pm2))
        if best_seq is not None and time.monotonic() < p6_dl - 10:
            a_lse2 = _run_lbfgs_lse(best_seq, p6_dl, min_time_remaining=8.0)
            update_best(a_lse2, _score(a_lse2))

    # Phase 7: Multi-n exploration - 200s
    print(f"[Phase 7] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    p7_dl = t_start + 1490
    n_tries = []
    if n_parent <= 16384:
        n_tries = [n_parent * 2, n_parent * 2 + 8192, 49152, 32768]
    elif n_parent <= 32768:
        n_tries = [n_parent + 8192, n_parent + 16384, 49152, 24576, 40960]
    else:
        n_tries = [n_parent + 8192, 49152, 40960, 24576, 65536]

    for n_try in n_tries:
        if time.monotonic() > p7_dl - 25:
            break
        if n_try == n_parent or n_try < 2048 or n_try > 100000:
            continue
        a_init_n = _resample(best_seq, n_try) if best_seq is not None else _make_profile(n_try, "tapered")
        a_n = _run_lbfgs_lse(a_init_n, p7_dl, min_time_remaining=10.0)
        sc_n = _score(a_n)
        update_best(a_n, sc_n)

    # Phase 8: Diverse initializations - 120s
    print(f"[Phase 8] Diverse inits (t={time.monotonic()-t_start:.1f}s)")
    p8_dl = t_start + 1610
    profiles = ["middle_heavy", "bimodal", "cos2", "w_shape"]
    for prof in profiles:
        if time.monotonic() > p8_dl - 15:
            break
        a_init_p = _make_profile(n_parent, prof)
        a_p = _run_lbfgs_lse(a_init_p, p8_dl, min_time_remaining=8.0)
        sc_p = _score(a_p)
        if sc_p < best_score * 1.02:
            update_best(a_p, sc_p)

    # Phase 9: Perturbation + local optimize - 80s
    print(f"[Phase 9] Perturbation search (t={time.monotonic()-t_start:.1f}s)")
    p9_dl = t_start + 1690
    if best_seq is not None:
        for sigma in [0.01, 0.005, 0.02]:
            if time.monotonic() > p9_dl - 10:
                break
            a_pert = _perturb(best_seq, sigma, rng)
            a_opt = _run_lbfgs_lse(a_pert, p9_dl, min_time_remaining=8.0)
            update_best(a_opt, _score(a_opt))

    # Phase 10: Final polish - 95s
    print(f"[Phase 10] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_fin, sc_fin = _run_active_set(best_seq, deadline - 20, min_time_remaining=5.0)
        update_best(a_fin, sc_fin)
        if time.monotonic() < deadline - 15:
            a_ps2, sc_ps2 = _run_peak_shave(best_seq, deadline - 15, min_time_remaining=5.0)
            update_best(a_ps2, sc_ps2)
        if time.monotonic() < deadline - 10:
            a_fin2 = _run_lbfgs_lse(best_seq, deadline - 5, min_time_remaining=5.0)
            update_best(a_fin2, _score(a_fin2))

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
