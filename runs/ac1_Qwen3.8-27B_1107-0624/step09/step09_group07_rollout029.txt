"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive log-sum-exp smoothing, active-set subgradient with BB steps,
diverse restarts, and multi-n exploration. Key improvements: adaptive beta
scheduling, power-mean optimization, more aggressive restarts, and
peak-targeted coordinate refinement.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _lsexp_smooth_obj_grad(z, beta, K, n):
    """Top-K log-sum-exp smoothed objective and gradient."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())

    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c

    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()

    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))

    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


def _power_mean_obj_grad(z, p, n):
    """Power-mean smoothed objective and gradient. p=inf gives max."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_pos = np.maximum(c, 1e-30)
    cm = float(c.max())
    
    if p < 1000:
        cp = c_pos ** p
        sp = cp.sum()
        if sp < 1e-300 or not np.isfinite(sp):
            obj = 2.0 * n * cm
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            obj = 2.0 * n * sp ** (1.0 / p)
            w = cp / sp
    else:
        obj = 2.0 * n * cm
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0

    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


class _SmoothProb:
    def __init__(self, beta, K, n, use_power_mean=False, p=50):
        self.beta = beta
        self.K = K
        self.n = n
        self.use_power_mean = use_power_mean
        self.p = p

    def __call__(self, z):
        if self.use_power_mean:
            return _power_mean_obj_grad(z, self.p, self.n)[0]
        return _lsexp_smooth_obj_grad(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        if self.use_power_mean:
            return _power_mean_obj_grad(z, self.p, self.n)[1]
        return _lsexp_smooth_obj_grad(z, self.beta, self.K, self.n)[1]


def _run_lbfgs_phased(a_init, deadline, min_remaining=30.0, verbose=False):
    """L-BFGS with adaptive beta/K schedule and power-mean option."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(200, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.01)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Extended schedule: start with full smoothing, end with sharp
    # (beta_mult, K_frac, use_pm, p)
    schedule = [
        (2.0, 1.0, False, 0),
        (1.0, 0.5, False, 0),
        (0.5, 0.3, False, 0),
        (0.3, 0.2, True, 100),
        (0.2, 0.15, True, 80),
        (0.1, 0.1, True, 60),
        (0.05, 0.06, True, 50),
        (0.03, 0.04, True, 40),
        (0.015, 0.025, False, 0),
        (0.008, 0.015, False, 0),
        (0.004, 0.01, False, 0),
        (0.002, 0.006, False, 0),
        (0.001, 0.004, False, 0),
        (0.0005, 0.002, False, 0),
        (0.0002, 0.001, False, 0),
    ]

    for beta_mult, K_frac, use_pm, p in schedule:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 15:
            break

        if use_pm:
            beta = base_beta * beta_mult
            K = max(10, int((2 * n - 1) * K_frac))
            prob = _SmoothProb(beta, K, n, use_power_mean=True, p=p)
        else:
            beta = base_beta * beta_mult
            K = max(10, int((2 * n - 1) * K_frac))
            prob = _SmoothProb(beta, K, n)

        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(100, int(remaining / (est_time * 25))))

        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 150, 'maxls': 60}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"    beta_mult={beta_mult}, K_frac={K_frac}, score={current_sc:.10f}")
        except Exception:
            pass

    return current_a


def _active_set_refine(a_init, deadline, min_remaining=30.0):
    """Active-set subgradient with BB steps, targeting top peaks."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    n_2nm1 = 2 * n - 1
    momentum = 0.5
    lr = 0.01
    lr_min = 1e-6
    lr_max = 0.05
    stall = 0
    max_stall = 100
    velocity = np.zeros(n)
    prev_a = None
    prev_g = None
    n_top = 3

    iteration = 0
    t_start = time.monotonic()
    last_improve_time = t_start

    while time.monotonic() + min_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
            last_improve_time = time.monotonic()
        else:
            stall += 1

        # Adaptive n_top: start with few, increase if stalling
        if stall > 50:
            n_top = min(10, n_top + 1)
        elif stall == 0:
            n_top = max(2, n_top - 1)

        # Select top peaks
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_top]

        # Weighted by height
        active_vals = c[active_idx]
        w = np.zeros(n_2nm1)
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w[active_idx] = (active_vals - active_vals.min()) / spread + 0.2
        else:
            w[active_idx] = 1.0 / n_top
        w = w / w.sum()

        # Gradient
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break

        # BB step
        if prev_a is not None and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g_centered - prev_g
            s_norm = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm > 1e-18 and abs(sy) > 1e-18:
                lr_bb = s_norm / abs(sy)
                lr = np.clip(lr_bb, lr_min, lr_max)

        # Momentum step with line search
        velocity = momentum * velocity - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()

        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            for ls in range(30):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-15:
                    improved = True
                    break
                vel_factor = 0.5
                velocity = momentum * velocity * vel_factor - lr * vel_factor * g_centered
                a_try = a + velocity
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    velocity = np.zeros(n)
                    a_try = a
                    break
                a_try = a_try / s_try
                lr *= 0.7

        if improved:
            prev_a = a.copy()
            prev_g = g_centered.copy()
            a = a_try
            lr = min(lr_max, lr * 1.02)
            momentum = min(0.7, momentum + 0.005)
        else:
            velocity = np.zeros(n)
            lr = max(lr_min, lr * 0.6)
            momentum = max(0.3, momentum - 0.05)

        # Reset on long stall
        if stall > max_stall:
            a = best_a.copy()
            velocity = np.zeros(n)
            prev_a = None
            prev_g = None
            lr = 0.01
            momentum = 0.5
            n_top = 3
            stall = 0
            max_stall = min(200, max_stall + 20)

        # Periodic L-BFGS micro-restart

            if time.monotonic() + min_remaining + 30 < deadline:
                a_tmp = _run_lbfgs_phased(a, time.monotonic() + 25, min_remaining=5)
                sc_tmp = _score(a_tmp)
                if sc_tmp < best_sc:
                    best_a = a_tmp.copy()
                    best_sc = sc_tmp
                    a = a_tmp.copy()
                    velocity = np.zeros(n)
                    prev_a = None
                    prev_g = None
                    last_improve_time = time.monotonic()
                    stall = 0

    return best_a, best_sc


def _peak_direct_reduce(a_init, deadline, min_remaining=30.0):
    """Directly reduce the highest peaks by adjusting contributing elements."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    patience = 0
    max_patience = 300

    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break

        # Find top 3 peaks
        peak_indices = np.argsort(c)[::-1][:3]

        for k_peak in peak_indices:
            if time.monotonic() + min_remaining > deadline:
                break

            # Compute contributions to this peak
            i_lo = max(0, k_peak - (n - 1))
            i_hi = min(n - 1, k_peak)
            contrib = np.zeros(n)
            for i in range(i_lo, i_hi + 1):
                j = k_peak - i
                contrib[i] += a[j]
                if i != j:
                    contrib[j] += a[i]

            # Reduce top contributors, redistribute to bottom
            n_reduce = max(2, int(n * 0.003))
            top_contrib = np.argsort(contrib)[::-1][:n_reduce]
            bottom_contrib = np.argsort(contrib)[:n_reduce]

            reduction = c_max * 0.003
            delta = np.zeros(n)
            for idx in top_contrib:
                if contrib[idx] > 1e-15:
                    delta[idx] = -reduction / (2.0 * contrib[idx] + 1e-30)
            for idx in bottom_contrib:
                delta[idx] = reduction * 0.5 / (n_reduce * max(a[idx], 1e-10) + 1e-30)

            delta -= delta.mean()

            # Line search
            alpha = 1.0
            improved = False
            for ls in range(25):
                a_try = a + alpha * delta
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-15:
                    a = a_try
                    current_sc = sc_try
                    improved = True
                    break
                alpha *= 0.5

            if not improved:
                patience += 5

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "optimal_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "blackman":
        t = np.linspace(0, 1, n)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(z, sigma, rng):
    """Perturb in log-space."""
    z_new = z + sigma * rng.standard_normal(len(z))
    z_new = z_new - z_new.max()
    e = np.exp(z_new)
    return e / e.sum()


def _perturb_log(z, sigma, rng):
    """Perturb in log-space, return log-space."""
    z_new = z + sigma * rng.standard_normal(len(z))
    return z_new - z_new.mean()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer with adaptive smoothing, active-set refinement,
    diverse restarts, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, tapered n={n_parent}, score={best_score:.10f}")

    n_main = len(best_seq)

    # === Phase 1: Coarse L-BFGS with full smoothing ===
    print(f"\n[P1] L-BFGS coarse (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - (budget_s - 15) * 0.65, time.monotonic() + (budget_s - 15) * 0.25)
    a1 = _run_lbfgs_phased(best_seq, p1_deadline, min_remaining=120, verbose=True)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [P1] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Active-set refinement ===
    print(f"\n[P2] Active-set (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        p2_deadline = min(deadline - 400, time.monotonic() + 350)
        a2, sc2 = _active_set_refine(best_seq, p2_deadline, min_remaining=30)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [P2] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Peak direct reduce ===
    print(f"\n[P3] Peak reduce (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        p3_deadline = min(deadline - 250, time.monotonic() + 200)
        a3, sc3 = _peak_direct_reduce(best_seq, p3_deadline, min_remaining=30)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [P3] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Second L-BFGS from improved point ===
    print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        p4_deadline = min(deadline - 180, time.monotonic() + 200)
        a4 = _run_lbfgs_phased(best_seq, p4_deadline, min_remaining=30)
        sc4 = _score(a4)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [P4] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Active-set round 2 ===
    print(f"\n[P5] Active-set round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 180:
        p5_deadline = min(deadline - 100, time.monotonic() + 150)
        a5, sc5 = _active_set_refine(best_seq, p5_deadline, min_remaining=20)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [P5] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-restart exploration ===
    print(f"\n[P6] Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        z_best = np.log(np.maximum(best_seq, 1e-30))
        z_best = z_best - z_best.mean()

        sigmas = [0.02, 0.03, 0.05, 0.04, 0.06, 0.03, 0.08, 0.02, 0.04, 0.05,
                  0.07, 0.03, 0.06, 0.04, 0.05, 0.02]
        n_restarts = min(16, max(3, int((deadline - 100 - time.monotonic()) / 35)))

        for i in range(n_restarts):
            if time.monotonic() > deadline - 60:
                break
            sigma = sigmas[i % len(sigmas)]
            a_pert = _perturb_log(z_best, sigma, rng)
            a_pert = np.exp(a_pert - a_pert.max())
            a_pert = a_pert / a_pert.sum()

            rl_deadline = min(deadline - 50, time.monotonic() + 30)
            a_r = _run_lbfgs_phased(a_pert, rl_deadline, min_remaining=5)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                z_best = np.log(np.maximum(best_seq, 1e-30))
                z_best = z_best - z_best.mean()
                print(f"  NEW BEST: {best_score:.10f} [restart_{i}, sigma={sigma}] t={time.monotonic()-t_start:.1f}s")

        # Try different profiles at same n
        profiles = ["cos_taper", "optimal_gauss", "raised_cos", "blackman", "hann"]
        for prof_kind in profiles:
            if time.monotonic() > deadline - 50:
                break
            a_prof = _make_profile(n_main, prof_kind)
            pr_deadline = min(deadline - 40, time.monotonic() + 25)
            a_p = _run_lbfgs_phased(a_prof, pr_deadline, min_remaining=5)
            sc_p = _score(a_p)
            if sc_p < best_score:
                best_seq = a_p.copy()
                best_score = sc_p
                print(f"  NEW BEST: {best_score:.10f} [profile_{prof_kind}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        n_options = [16384, 12288, 20480, 8192, 24576]
        n_options = [x for x in n_options if x != len(best_seq) and x >= 4096]

        for n_try in n_options[:3]:
            if time.monotonic() > deadline - 30:
                break
            a_res = _resample(best_seq, n_try)
            an_deadline = min(deadline - 20, time.monotonic() + 35)
            a_opt = _run_lbfgs_phased(a_res, an_deadline, min_remaining=5)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [n={n_try}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final active-set polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _active_set_refine(best_seq, deadline - 2, min_remaining=2)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final] t={time.monotonic()-t_start:.1f}s")

        # Tiny L-BFGS
        remaining = deadline - 2 - time.monotonic()
        if remaining > 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = max(mb * 0.0005, 1e-10)
            K_fin = max(5, int(n * 0.002))
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _SmoothProb(beta_fin, K_fin, n)
            max_iter = min(5000, max(50, int(remaining / 0.02)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-15, 'maxcor': 200, 'maxls': 80})
                a_fin2 = np.exp(result.x - result.x.max())
                a_fin2 = a_fin2 / a_fin2.sum()
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        # Scale to ensure good sum (the evaluator normalizes, but let's be safe)
        # Actually the score is scale-invariant, so we can return as-is.
        # But let's make sure the sum is reasonable for the evaluator.
        if s > 1000 * len(final):
            final = final / s * len(final)

        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
