def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_spectral_init(n, rng, dc_factor=3.0):
    """Create a sequence with nearly flat power spectrum and random phases."""
    N = _next_pow2(2 * n - 1)
    A = np.zeros(N // 2 + 1, dtype=complex)
    A[0] = dc_factor * np.sqrt(n)
    for k in range(1, N // 2 + 1):
        magnitude = 1.0
        phase = rng.uniform(0, 2 * np.pi)
        A[k] = magnitude * np.exp(1j * phase)
    a_full = np.fft.irfft(A * np.sqrt(N), N)
    a = a_full[:n]
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos2":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        if rng is None:
            rng = np.random.default_rng(42)
        a = rng.random(n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _active_set_gradient(a, c, tol_frac):
    """Weighted active-set gradient for L-infinity autocorrelation."""
    n = len(a)
    c_max = float(c.max())
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    n_active = int(active.sum())
    if n_active == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        n_active = 1
    eps = 1e-12 * c_max
    w = np.zeros_like(c)
    w[active] = 1.0 / (c_max - c[active] + eps)
    w = w / w.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad = 2.0 * h
    grad = grad - np.mean(grad)
    return grad


def _nesterov_polish(a_init, deadline, tol_start=0.01, tol_end=0.00003,
                     lr_init=0.04, momentum=0.85):
    """Active-set gradient with heavy-ball momentum."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    v = np.zeros(n)
    tol_frac = tol_start
    lr = lr_init
    no_improve = 0
    max_stall = 200

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        grad = _active_set_gradient(a, c, tol_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        v = momentum * v - lr * grad
        alpha = 1.0
        improved = False
        for _ in range(40):
            a_try = _normalize(a + alpha * v)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            v = np.zeros(n)
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_end, tol_frac * 0.7)
                lr *= 0.7
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(tol_end, tol_frac * 0.3)
                lr = lr_init * 0.4
                no_improve = 0
        else:
            if tol_frac > tol_end:
                tol_frac = max(tol_end, tol_frac * 0.99)
            lr = min(lr_init, lr * 1.005)

    return best_a, _score(best_a)


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


class _LbfgsProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def f(self, z):
        return _smoobj(z, self.beta, self.n)

    def g(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS on log-softmax with aggressive geometric beta annealing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.002:
        gap = max_c * 0.002
    base_beta = max(gap, max_c * 0.005)

    # Geometric schedule: halve beta each time
    n_levels = 20
    betas = base_beta * (0.5 ** np.arange(n_levels))

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # More iterations for smaller beta (finer optimization)
        est_per_iter = 0.003 * max(1.0, n / 16384.0)
        iters_for_level = max(30, min(8000, int(remaining * 0.4 / (est_per_iter * 20))))
        # For early levels (large beta), use fewer iterations
        scale = min(1.0, (n_levels - i) / 5.0)
        max_iter = int(iters_for_level * scale)
        max_iter = max(20, min(max_iter, 10000))

        prob = _LbfgsProblem(beta, n)
        try:
            result = minimize(
                prob.f, z, jac=prob.g, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 120, 'maxls': 60}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            print(f"    L-BFGS level {i+1}/{len(betas)}: beta={beta:.2e}, score={current_sc:.10f}, iters={result.niter}")
        except Exception as e:
            print(f"    L-BFGS level {i+1} failed: {e}")

    return current_a


def _quick_optimize(a_init, deadline, min_time_remaining=15.0):
    """Combined L-BFGS + polish."""
    a = _run_lbfgs(a_init, deadline, min_time_remaining=min_time_remaining + 10)
    if time.monotonic() < deadline - min_time_remaining:
        a2, sc2 = _nesterov_polish(
            a, min(deadline - min_time_remaining, time.monotonic() + 12),
            tol_start=0.004, tol_end=0.0003, lr_init=0.02, momentum=0.82)
        if sc2 < _score(a):
            a = a2
    return a


def _lp_direction(a, M, max_iter=200):
    """
    Solve (approximately) the LP:
        maximize sum(b) subject to conv(a,b) <= M, b >= 0
    using projected subgradient + multiplicative updates.
    Returns the optimal b (normalized).
    """
    n = len(a)
    b = np.ones(n) / n
    best_b = b.copy()
    best_sum = b.sum()

    for it in range(max_iter):
        c = _full_convolve(a, b)
        c_max = float(c.max())
        if c_max <= M * 1.001:
            break
        # Multiplicative update: reduce b where it contributes most to violated constraints
        # Sensitivity: how much does b[j] contribute to each c[k]?
        # c[k] = sum_j a[j] * b[k-j], so d c[k] / d b[j] = a[k-j] (when valid)
        # Approximate: for each j, total sensitivity = sum_k a[|k-j|] * max(0, c[k] - M)
        violation = np.maximum(c - M, 0)
        if violation.sum() < 1e-20:
            break
        # Compute sensitivity using convolution
        v_rev = violation[::-1]
        sens = _full_convolve(a, v_rev)
        sens = sens[n - 1:2 * n - 1]
        # Multiplicative update
        eta = 0.5
        update = (1.0 + eta * (1.0 / n - sens / max(sens.sum(), 1e-20)))
        update = np.maximum(update, 0.01)
        b = b * update
        b = b / b.sum() * best_sum  # keep sum roughly constant
        if b.sum() > best_sum * 0.9:
            best_b = b.copy()
            best_sum = b.sum()

    return best_b / best_b.sum()


def _simulated_ils(a_init, deadline, rng, min_time_remaining=60.0, T_init=0.01, T_min=0.0001):
    """
    Simulated Annealing Iterated Local Search.
    Uses SA acceptance criterion to escape local minima.
    """
    n = len(a_init)
    current_a = _normalize(a_init.copy())
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc

    T = T_init
    cooling = (T_min / T_init) ** (1.0 / max(100, int((deadline - min_time_remaining - time.monotonic()) / 10)))

    strategies = [
        ('global_small', 0.004),
        ('global_med', 0.015),
        ('global_large', 0.04),
        ('block', 0.03),
        ('sparse', 0.06),
        ('mirror', 0.02),
        ('peak_target', 0.0),
        ('spectral_jump', 0.0),
    ]

    attempt = 0
    no_improve_count = 0

    while time.monotonic() + min_time_remaining < deadline:
        kind, sigma = strategies[attempt % len(strategies)]

        if kind in ('global_small', 'global_med', 'global_large'):
            z = np.log(np.maximum(current_a, 1e-30))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_perturbed = _softmax(z)
        elif kind == 'block':
            block_size = max(30, n // 6)
            start = int(rng.integers(0, n - block_size))
            a_perturbed = current_a.copy()
            z = np.log(np.maximum(a_perturbed[start:start + block_size], 1e-30))
            z = z + sigma * rng.standard_normal(block_size)
            a_perturbed[start:start + block_size] = np.exp(z - np.max(z))
            a_perturbed = _normalize(a_perturbed)
        elif kind == 'sparse':
            n_perturb = max(20, n // 10)
            idx = rng.choice(n, size=n_perturb, replace=False)
            a_perturbed = current_a.copy()
            factors = np.exp(sigma * rng.standard_normal(n_perturb))
            a_perturbed[idx] *= factors
            a_perturbed = _normalize(a_perturbed)
        elif kind == 'mirror':
            seg_len = max(20, n // 15)
            start = int(rng.integers(0, n - seg_len))
            a_perturbed = current_a.copy()
            seg = a_perturbed[start:start + seg_len].copy()
            a_perturbed[start:start + seg_len] = seg[::-1]
            z = np.log(np.maximum(a_perturbed, 1e-30))
            z = z + 0.004 * rng.standard_normal(n)
            z = z - z.mean()
            a_perturbed = _softmax(z)
        elif kind == 'peak_target':
            # Target the largest peaks: perturb in direction that reduces top peaks
            c = _full_convolve(current_a, current_a)
            c_max = float(c.max())
            top_k = min(20, len(c))
            top_idx = np.argsort(c)[::-1][:top_k]
            # Create a perturbation that reduces contributions to top peaks
            a_perturbed = current_a.copy()
            for k in top_idx[:5]:
                # Gradient of c[k] w.r.t. a[j] is a[k-j] (for valid indices)
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                if j_end >= j_start:
                    contrib = a_perturbed[j_start:j_end + 1].copy()
                    # Reduce where contribution is highest
                    a_perturbed[j_start:j_end + 1] -= 0.02 * contrib
            a_perturbed = _normalize(a_perturbed)
            z = np.log(np.maximum(a_perturbed, 1e-30))
            z = z + 0.005 * rng.standard_normal(n)
            z = z - z.mean()
            a_perturbed = _softmax(z)
        elif kind == 'spectral_jump':
            # Jump in the spectral domain
            N = _next_pow2(2 * n - 1)
            A = np.fft.rfft(current_a, N)
            # Perturb phases randomly
            for k in range(1, N // 2 + 1):
                if rng.random() < 0.3:
                    A[k] *= np.exp(1j * rng.uniform(-0.5, 0.5))
            a_full = np.fft.irfft(A, N)
            a_perturbed = a_full[:n]
            a_perturbed = np.maximum(a_perturbed, 1e-10)
            a_perturbed = a_perturbed / a_perturbed.sum()
        else:
            a_perturbed = current_a.copy()

        opt_deadline = min(deadline - min_time_remaining, time.monotonic() + 6)
        a_opt = _quick_optimize(a_perturbed, opt_deadline, min_time_remaining=2)
        sc_opt = _score(a_opt)

        # SA acceptance
        delta = sc_opt - current_sc
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            current_a = a_opt.copy()
            current_sc = sc_opt
            no_improve_count = 0
        else:
            no_improve_count += 1

        if sc_opt < best_sc - 1e-12:
            best_sc = sc_opt
            best_a = a_opt.copy()
            no_improve_count = 0
            print(f"    SA-ILS NEW BEST: {best_sc:.10f} ({kind}) t={time.monotonic() - t_start:.0f}s")

        # Restart if stuck too long
        if no_improve_count > 30:
            # Jump to a new random point near the best
            z = np.log(np.maximum(best_a, 1e-30))
            z = z + 0.1 * rng.standard_normal(n)
            z = z - z.mean()
            current_a = _softmax(z)
            current_sc = _score(current_a)
            no_improve_count = 0

        T = max(T_min, T * cooling)
        attempt += 1

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: spectral init -> L-BFGS annealing -> SA-ILS ->
    multi-n exploration -> final ultra-fine polish.
    """
    global t_start
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 24576
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    n_main = n_parent if n_parent > 0 else 24576

    # === Phase 1: Diverse multi-start (250s) ===
    print(f"\n[P1] Multi-start (t={time.monotonic() - t_start:.1f}s)")
    ms_deadline = t_start + 250

    starts = [('parent', parent.copy())]
    for kind in ['triangle', 'hann', 'gaussian', 'raised_cos2']:
        starts.append((kind, _make_profile(n_main, kind, rng)))
    # Spectral initializations with different DC factors
    for dc in [2.0, 3.0, 5.0]:
        starts.append((f'spectral_dc{dc}', _make_spectral_init(n_main, rng, dc_factor=dc)))

    for name, a_init in starts:
        if time.monotonic() > ms_deadline - 30:
            break
        sc_init = _score(a_init)
        a_opt = _quick_optimize(a_init, ms_deadline, min_time_remaining=40)
        sc_opt = _score(a_opt)
        print(f"  {name}: init={sc_init:.10f}, opt={sc_opt:.10f}")
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [ms/{name}]")

    # === Phase 2: Intensive L-BFGS (400s) ===
    print(f"\n[P2] Intensive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 700:
        lbfgs_deadline = min(deadline - 700, time.monotonic() + 400)
        a_main = _run_lbfgs(best_seq, lbfgs_deadline, min_time_remaining=60)
        sc = _score(a_main)
        if sc < best_score:
            best_seq = a_main.copy()
            best_score = sc
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_int]")

    # === Phase 3: Nesterov polish (100s) ===
    print(f"\n[P3] Nesterov polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        pol_deadline = min(deadline - 500, time.monotonic() + 100)
        a_pol, sc_pol = _nesterov_polish(best_seq, pol_deadline,
                                         tol_start=0.003, tol_end=0.00002,
                                         lr_init=0.015, momentum=0.83)
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  NEW BEST: {best_score:.10f} [nesterov]")

    # === Phase 4: Simulated Annealing ILS (400s) ===
    print(f"\n[P4] SA-ILS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        ils_deadline = min(deadline - 350, time.monotonic() + 400)
        a_ils, sc_ils = _simulated_ils(best_seq, ils_deadline, rng, min_time_remaining=50)
        if sc_ils < best_score:
            best_seq = a_ils.copy()
            best_score = sc_ils
            print(f"  NEW BEST: {best_score:.10f} [sa_ils]")

    # === Phase 5: Multi-n exploration (250s) ===
    print(f"\n[P5] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = sorted(set([
        n_main, 16384, 32768, 12288, 20480, 24576,
        32768 + 4096, 16384 + 2048, 24576 + 8192
    ]))
    # Filter to reasonable range
    n_candidates = [nn for nn in n_candidates if 8192 <= nn <= 49152]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init={sc_init:.10f}")
        a_opt = _quick_optimize(a_init, deadline, min_time_remaining=35)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n]")

    # === Phase 6: Second SA-ILS round (150s) ===
    print(f"\n[P6] SA-ILS round 2 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        ils2_deadline = min(deadline - 100, time.monotonic() + 150)
        rng2 = np.random.default_rng(int(seed) + 777)
        a_ils2, sc_ils2 = _simulated_ils(best_seq, ils2_deadline, rng2, min_time_remaining=30,
                                         T_init=0.005, T_min=0.00005)
        if sc_ils2 < best_score:
            best_seq = a_ils2.copy()
            best_score = sc_ils2
            print(f"  NEW BEST: {best_score:.10f} [sa_ils2]")

    # === Phase 7: Final ultra-fine polish (80s) ===
    print(f"\n[P7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _nesterov_polish(best_seq, deadline - 3,
                                         tol_start=0.001, tol_end=0.000002,
                                         lr_init=0.004, momentum=0.78)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish]")

        remaining = deadline - 3 - time.monotonic()
        if remaining > 15:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.000005
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _LbfgsProblem(beta_fin, n)
            max_iter = min(5000, max(50, int(remaining / 0.025)))
            try:
                result = minimize(prob.f, z, jac=prob.g, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-20,
                                           'gtol': 1e-16, 'maxcor': 120})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs]")
            except Exception:
                pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
