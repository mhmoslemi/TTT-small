"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) adaptive soft-max L-BFGS with restarts, (2) projected subgradient 
with top-k averaging, (3) targeted peak shaving, (4) multi-length exploration.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, n if n > 1 else 2)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def _lower_bound(n):
    """Theoretical lower bound: 2n/(2n-1)."""
    return 2.0 * n / (2.0 * n - 1)

def _make_profile(n, kind="tapered", seed=None):

    elif kind == "tapered":

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    a_new = np.interp(x_new, x_old, a, period=old_n) if old_n > 1 else a
    # Actually, for non-periodic resampling:

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:

    return a_new

        current_sc = 2.0 * n * float(c.max())

            no_improve = 0

            no_improve += 1

def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)

def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

def _grad_soft(a, beta):
    """Gradient of soft_max(conv(a,a)) using structure."""
    n = len(a)
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]

def _smoobj(z, beta, n):
    """Log-softmax parameterized objective."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)

def _smograd(z, beta, n):
    """Gradient of the log-softmax parameterized objective."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    grad_a = _grad_soft(a, beta)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)

def _lbfgs_round(a_init, deadline, beta_start, beta_end, n_betas, min_time=20.0):
    """L-BFGS with geometric beta schedule."""
    n = len(a_init)
    best_a = a_init.copy()
    s = best_a.sum()
    best_a = best_a / s
    best_sc = _score(best_a)
    
    z = np.log(np.maximum(best_a, 1e-30))
    z = z - z.mean()
    
    betas = np.geomspace(beta_start, beta_end, n_betas)
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        
        def obj(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a, a)
            return 2.0 * n * _soft_max_value(c, beta)
        
        def grad(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            grad_a = _grad_soft(a, beta)
            W = float(np.dot(grad_a, a))
            return a * (grad_a - W)
        
        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': 5000, 'ftol': 1e-16, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 50}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    beta={beta:.2e}: score={best_sc:.10f}, nit={result.nit}")
        except Exception as e:
            pass
    
    return best_a, best_sc

def _peak_shave(a, deadline, min_time=10.0):
    """Targeted peak shaving: identify the highest peaks and reduce them."""
    n = len(a)
    a = a.copy()
    a = _simplex_project(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    eta = 0.01
    no_improve = 0
    
    while time.monotonic() + min_time < deadline:
        c = _full_convolve(a, a)
        current_sc = 2.0 * n * float(c.max())
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                eta *= 0.5
                if eta < 1e-6:
                    break
                no_improve = 0
        
        # Find the argmax
        k_star = int(np.argmax(c))
        
        # Gradient of c[k_star] w.r.t. a
        # d(c[k])/d(a[j]) = a[k-j] if 0<=k-j<n, plus a[j-k] if 0<=j-k<n and j-k != k-j
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[k_star] = 1.0
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]
        
        # Center the gradient
        grad_c = grad - np.mean(grad)
        
        # Try the step
        a_new = a - eta * grad_c
        a_new = _simplex_project(a_new)
        sc_new = _score(a_new)
        
        if sc_new < current_sc:
            a = a_new
        else:
            # Try smaller

            a_new = _simplex_project(a_new)
            sc_new = _score(a_new)
            if sc_new < current_sc:
                a = a_new
            else:
                eta *= 0.7
    
    return best_a, best_sc

Multi-phase optimizer using mirror descent, soft-max L-BFGS, and top-k subgradient.
Key innovation: multiplicative (mirror) updates that naturally maintain non-negativity,
combined with adaptive active-set tracking.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "mild_taper":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.3 * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "semicircle":
        t = np.linspace(0, 1, n)
        a = np.sqrt(np.maximum(1.0 - (2 * t - 1) ** 2, 0))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.2 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.1 * np.cos(2.0 * np.pi * 5.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _grad_at_k(a, k, n):
    """Gradient of c[k] = sum_i a[i]*a[k-i] w.r.t. a, using FFT."""
    a_rev = a[::-1]
    w = np.zeros(2 * n - 1)
    w[k] = 1.0
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]

def _grad_multi(a, indices, n):
    """Average gradient over multiple convolution indices."""
    a_rev = a[::-1]
    w = np.zeros(2 * n - 1)
    w[indices] = 1.0 / len(indices)
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]

def _mirror_descent(a_init, deadline, min_time=30.0):
    """Mirror descent with multiplicative updates.
    
    Uses the entropy mirror map: a_new ∝ a * exp(-eta * grad_centered)
    This naturally maintains positivity.
    """
    n = len(a_init)
    a = _simplex_project(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    lb = 2.0 * n / (2.0 * n - 1)
    
    eta = 0.1
    no_improve = 0
    max_no_improve = 500
    k_adaptive_base = 10
    
    while time.monotonic() + min_time < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            eta *= 0.5
            if eta < 1e-8:
                break
            no_improve = 0
        
        # Adaptive top-k: include more peaks when they're close
        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active < 1:
            active_idx = [int(np.argmax(c))]
        elif n_active > 500:
            # Limit to top 500
            topk = np.argpartition(c, -500)[-500:]
            active_idx = list(topk)
        else:
            active_idx = list(np.where(active)[0])
        
        # Compute gradient
        grad = _grad_multi(a, active_idx, n)
        grad_c = grad - np.mean(grad)
        
        # Mirror descent update: a_new ∝ a * exp(-eta * grad_c)
        log_update = -eta * grad_c
        log_update = np.clip(log_update, -50, 50)  # Numerical safety
        a_new = a * np.exp(log_update)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Check improvement
        sc_new = _score(a_new)
        if sc_new < current_sc:
            a = a_new
            # Slightly increase eta on improvement
            eta = min(eta * 1.02, 1.0)
        else:
            # Try smaller step
            a_try = a * np.exp(0.5 * log_update)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc:
                    a = a_try
                    eta *= 0.8
                else:
                    eta *= 0.7
            else:
                eta *= 0.7
    
    return best_a, best_sc

def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)

def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

def _grad_soft(a, beta):
    n = len(a)
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]

def _lbfgs_round(a_init, deadline, beta_vals, min_time=20.0):
    """L-BFGS with given beta schedule."""
    n = len(a_init)
    best_a = _simplex_project(a_init.copy())
    best_sc = _score(best_a)
    
    z = np.log(np.maximum(best_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(beta_vals):
        if time.monotonic() + min_time > deadline:
            break
        
        def obj(z_val):
            a_val = np.exp(z_val - z_val.max())
            a_val = a_val / a_val.sum()
            c_val = _full_convolve(a_val, a_val)
            return 2.0 * n * _soft_max_value(c_val, beta)
        
        def grad(z_val):
            a_val = np.exp(z_val - z_val.max())
            a_val = a_val / a_val.sum()
            grad_a_val = _grad_soft(a_val, beta)
            W = float(np.dot(grad_a_val, a_val))
            return a_val * (grad_a_val - W)
        
        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 40}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return best_a, best_sc

def _topk_subgradient(a_init, k, deadline, min_time=20.0):
    """Projected subgradient with top-k averaging."""
    n = len(a_init)
    a = _simplex_project(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    lb = 2.0 * n / (2.0 * n - 1)
    
    step_scale = 0.3
    no_improve = 0
    
    while time.monotonic() + min_time < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            break
        
        # Top-k indices
        actual_k = min(k, len(c))
        topk_idx = np.argpartition(c, -actual_k)[-actual_k:]
        
        grad = _grad_multi(a, list(topk_idx), n)
        grad_c = grad - np.mean(grad)
        
        gap = current_sc - lb
        if gap < 1e-12:
            break
        g_norm_sq = float(np.dot(grad_c, grad_c))
        if g_norm_sq < 1e-20:
            break
        
        step = step_scale * gap / g_norm_sq
        step = min(step, 0.3)
        
        a_new = a - step * grad_c
        a_new = _simplex_project(a_new)
        
        sc_new = _score(a_new)
        if sc_new <= current_sc * 1.001:
            a = a_new
        else:
            a_new = a - 0.1 * step * grad_c
            a_new = _simplex_project(a_new)
            a = a_new
            step_scale *= 0.95
        
        if no_improve % 100 == 50:
            step_scale *= 0.9
    
    return best_a, best_sc

def _peak_shave(a_init, deadline, min_time=10.0):
    """Iteratively shave the highest peak."""
    n = len(a_init)
    a = _simplex_project(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    eta = 0.005
    no_improve = 0
    
    while time.monotonic() + min_time < deadline:
        c = _full_convolve(a, a)
        current_sc = 2.0 * n * float(c.max())
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                eta *= 0.5
                if eta < 1e-7:
                    break
                no_improve = 0
        
        k_star = int(np.argmax(c))
        grad = _grad_at_k(a, k_star, n)
        grad_c = grad - np.mean(grad)
        
        a_new = a - eta * grad_c
        a_new = _simplex_project(a_new)
        sc_new = _score(a_new)
        
        if sc_new < current_sc:
            a = a_new
        else:
            eta *= 0.7
    
    return best_a, best_sc

def _perturb(a, epsilon, seed=None):
    rng = np.random.default_rng(seed)
    a_pert = a + epsilon * rng.standard_normal(len(a))
    return _simplex_project(a_pert)
