"""
Hybrid optimizer combining KKT-Newton with fast incremental coordinate descent
for minimizing 2*n*max(conv(a,a))/sum(a)^2. Uses adaptive active sets,
incremental convolution tracking, and multi-phase optimization.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.5) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve_hankel(W, n, rhs, max_iter=200, tol=1e-12):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.05:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _compute_gradient(a, W, n):
    a_rev = a[::-1]
    wc = _full_convolve(W, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(W, _full_convolve(a, a)))
    return 2.0 * a * (h - W_val)


def _kkt_newton(a_init, deadline, max_rounds=200, k_schedule=None):
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    if k_schedule is None:
        k_schedule = [1, 3, 5, 10, 20, 50, 100, 200, 500]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        K = k_schedule[rnd % len(k_schedule)]
        K = min(K, 2 * n - 1)
        idx = np.argpartition(c, -K)[-K:]
        W = np.zeros(2 * n - 1)
        W[idx] = 1.0 / K

        g = _compute_gradient(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        Hg = _cg_solve_hankel(W, n, g, max_iter=150)
        H1 = _cg_solve_hankel(W, n, np.ones(n), max_iter=150)
        H1_denom = float(np.ones(n) @ H1)
        if abs(H1_denom) < 1e-25 or not np.isfinite(H1_denom):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / H1_denom
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        alpha = 1.0
        step_done = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1.0 - 0.005 * alpha) - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not step_done:
            for trial in [0.05, 0.02, 0.01, 0.005, 0.001]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 10:
            break

    return best_a, best_sc


def _coord_descent(a_init, deadline, max_sweeps=500, verbose=True, t0=0.0):
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    c = _full_convolve(a, a).astype(np.float64)
    S = float(a.sum())
    best_a = a.copy()
    best_score = float(2.0 * n * c.max() / (S * S))
    no_improve = 0
    sweep = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() < deadline and sweep < max_sweeps:
        S = float(a.sum())
        current_max_c = float(c.max())
        base_sc = 2.0 * n * current_max_c / (S * S)
        coords_updated = 0

        if sweep < 5:
            order = np.arange(n)
        elif sweep < 15:
            order = np.arange(n - 1, -1, -1)
        else:
            order = rng.permutation(n)

        for idx in range(n):
            j = int(order[idx])
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            if old_val < 1e-15:
                # Try turning on this coordinate
                trial_val = S / n * 0.01
                delta = trial_val
                c[j:j + n] += 2.0 * delta * a
                if 2 * j < 2 * n - 1:
                    c[2 * j] += delta * delta
                new_S = S + delta
                new_max_c = float(c.max())
                new_sc = 2.0 * n * new_max_c / (new_S * new_S)
                c[j:j + n] -= 2.0 * delta * a
                if 2 * j < 2 * n - 1:
                    c[2 * j] -= delta * delta
                if new_sc < base_sc - 1e-14:
                    c[j:j + n] += 2.0 * delta * a
                    if 2 * j < 2 * n - 1:
                        c[2 * j] += delta * delta
                    a[j] = trial_val
                    S = new_S
                    base_sc = new_sc
                    current_max_c = new_max_c
                    coords_updated += 1
                continue

            # Determine step size based on convergence
            if no_improve < 10:
                multipliers = [0.0, 0.3, 0.6, 1.4, 2.0]
            elif no_improve < 30:
                multipliers = [0.5, 0.8, 1.2, 1.5]
            else:
                multipliers = [0.9, 1.1]

            best_delta = 0.0
            best_sc_coord = base_sc

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                delta = new_val - old_val
                if abs(delta) < old_val * 1e-14:
                    continue

                # Apply update
                c[j:j + n] += 2.0 * delta * a
                if 2 * j < 2 * n - 1:
                    c[2 * j] += delta * delta
                new_S = S + delta

                if new_S < 1e-15:
                    c[j:j + n] -= 2.0 * delta * a
                    if 2 * j < 2 * n - 1:
                        c[2 * j] -= delta * delta
                    continue

                new_max_c = float(c.max())
                new_sc = 2.0 * n * new_max_c / (new_S * new_S)

                # Revert
                c[j:j + n] -= 2.0 * delta * a
                if 2 * j < 2 * n - 1:
                    c[2 * j] -= delta * delta

                if new_sc < best_sc_coord - 1e-15:
                    best_sc_coord = new_sc
                    best_delta = delta

            # Apply best
            if best_delta != 0.0:
                c[j:j + n] += 2.0 * best_delta * a
                if 2 * j < 2 * n - 1:
                    c[2 * j] += best_delta * best_delta
                a[j] += best_delta
                S += best_delta
                base_sc = best_sc_coord
                coords_updated += 1

        sweep += 1

        # Verify every 10 sweeps
        if sweep % 10 == 0:
            sc_verify = _score(a)
            if sc_verify < best_score - 1e-12:
                best_score = sc_verify
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            if verbose and sweep % 20 == 0:
                print(f"    CoordDesc s{sweep}: sc={sc_verify:.10f} best={best_score:.10f} "
                      f"upd={coords_updated} t={time.monotonic()-t0:.1f}s")

            if no_improve > 15:
                break

    # Final check
    sc_final = _score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score


def _subgrad_momentum(a_init, deadline, max_steps=5000, t0=0.0):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.015
    momentum = 0.9
    v = np.zeros(n)

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        for tol_frac in [0.0001, 0.0005, 0.001, 0.003, 0.01]:
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            if active.sum() >= 3:
                break
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        W = np.zeros_like(c)
        W[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(W, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(W, c))
        grad = 2.0 * a * (h - W_val)
        grad = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        v = momentum * v - lr * grad
        z = z + v
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 30 == 29:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, 0.04)
            else:
                patience += 1
                lr *= 0.85
                momentum = min(momentum + 0.005, 0.95)
                if patience > 25:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)
                    patience = 0
                    lr = 0.01

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)

        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
