def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]

    def grad(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[1]


def _run_lbfgs_continuation(a_init, deadline, min_time_remaining=60.0):
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a

    c_sorted = np.sort(c_cur)[::-1]
    n_c = 2 * n - 1
    gap_idx = min(50, n_c - 1)
    gap = max_c - c_sorted[gap_idx]
    if gap < max_c * 0.0005:
        gap = max_c * 0.0005
    base_beta = max(gap, max_c * 0.01)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    n_stages = 14
    for stage in range(n_stages):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 15:
            break

        t_frac = stage / max(1, n_stages - 1)
        beta_mult = (1.0 - t_frac) ** 2
        beta = max(base_beta * beta_mult, base_beta * 0.0005)
        K_frac = 0.15 - 0.13 * t_frac
        K = max(5, int(n_c * K_frac))

        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(30, int(remaining / (est_time * 25))))

        prob = _TopkSmoProb(beta, K, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxcor': 120, 'maxls': 60}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _nesterov_subgrad(a_init, deadline, min_time_remaining=60.0):
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n_c = 2 * n - 1

    x = a.copy()
    y = a.copy()
    k = 0
    momentum = 0.9
    momentum_decay = 0.999
    momentum_grow = 1.002
    lr = 0.01
    lr_min = 1e-6
    lr_max = 0.05
    stall = 0
    max_stall = 100
    active_tol = 0.02
    active_tol_min = 0.001
    active_tol_decay = 0.995

    t_start = time.monotonic()
    last_eval = t_start

    while time.monotonic() + min_time_remaining < deadline:
        k += 1
        c = _full_convolve(y, y)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive active set: peaks within active_tol of max
        threshold = c_max * (1.0 - active_tol)
        active = c >= threshold
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        # Cap active set
        if n_active > 200:
            top_idx = np.argpartition(c, -200)[-200:]
            active = np.zeros(n_c, dtype=bool)
            active[top_idx] = True
            n_active = 200

        # Weighted subgradient
        active_vals = c[active]
        spread = active_vals.max() - active_vals.min()
        w = np.zeros(n_c)
        if spread > 1e-18:
            w[active] = (active_vals - active_vals.min()) / spread + 0.05
        else:
            w[active] = 1.0 / n_active
        w = w / w.sum()

        g = _grad_maxconv(y, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break

        # Polyak-like step size
        lower_bound = 2.0 * n / n_c  # theoretical lower bound on 2n*max_c
        polyak_lr = max(0.0, (current_sc - lower_bound)) / (gnorm * gnorm + 1e-30)
        lr = min(lr_max, max(lr_min, 0.5 * lr + 0.5 * min(polyak_lr, lr_max)))

        # Nesterov step
        alpha = (1.0 + np.sqrt(1.0 + 4.0 * k * k)) / 2.0
        momentum = min(0.99, momentum * momentum_grow)
        y_new = y - lr * g_centered
        y_new = np.maximum(y_new, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-10:
            y_new = y
        else:
            y_new = y_new / s_new

        x_new = y_new + (momentum - 1.0) / alpha * (y_new - y)
        x_new = np.maximum(x_new, 0.0)
        s_x = x_new.sum()
        if s_x > 1e-10:
            x_new = x_new / s_x
        else:
            x_new = y_new.copy()

        # Check improvement with line search
        improved = False
        for ls in range(15):
            c_try = _full_convolve(y_new, y_new)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-16:
                improved = True
                break
            lr *= 0.5
            if lr < lr_min:
                break
            y_try = y - lr * g_centered
            y_try = np.maximum(y_try, 0.0)
            s_try = y_try.sum()
            if s_try > 1e-10:
                y_try = y_try / s_try
                y_new = y_try
            else:
                y_new = y
                break

        if improved:
            x = x_new
            y = y_new
            active_tol = max(active_tol_min, active_tol * active_tol_decay)
        else:
            if stall > max_stall:
                y = best_a.copy()
                x = best_a.copy()
                lr = lr_min * 2
                active_tol = min(0.05, active_tol * 2.0)
                stall = 0
            else:
                lr = max(lr_min, lr * 0.7)
                momentum = max(0.8, momentum * 0.99)

    return best_a, best_sc


def _spectral_shaping(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)

    patience = 0
    max_patience = 50
    cutoff_start = 0.1
    alpha_start = 0.3

    for iteration in range(200):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break

        # Try different cutoff frequencies
        best_local_sc = current_sc
        best_local_a = a.copy()
        n_freq = N // 2 + 1
        for cutoff_frac in [0.05, 0.08, 0.1, 0.15, 0.2]:
            for alpha in [0.1, 0.3, 0.5]:
                if time.monotonic() + min_time_remaining > deadline:
                    break
                freq = np.fft.rfftfreq(N) * N
                mask = np.exp(-alpha * (freq / (cutoff_frac * n_freq)) ** 2)
                A_modified = A * mask
                # Preserve DC component
                A_modified[0] = A[0]
                a_try = np.fft.irfft(A_modified, N)[:n]
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_local_sc - 1e-16:
                    best_local_sc = sc_try
                    best_local_a = a_try.copy()
                    A = np.fft.rfft(a_try, N)

        if best_local_sc < current_sc - 1e-16:
            a = best_local_a
        else:
            patience += 5

    return best_a, best_sc


def _peak_shaving(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n_c = 2 * n - 1

    patience = 0
    max_patience = 100

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break

        # Find top 3 peaks
        top_peaks = np.argpartition(c, -3)[-3:]
        for k_peak in top_peaks:
            if time.monotonic() + min_time_remaining > deadline:
                break
            # Compute contributions
            i_lo = max(0, k_peak - (n - 1))
            i_hi = min(n - 1, k_peak)
            if i_lo > i_hi:
                continue
            contrib = np.zeros(n)
            for i in range(i_lo, i_hi + 1):
                j = k_peak - i
                contrib[i] += a[j]
                if i != j:
                    contrib[j] += a[i]

            n_reduce = max(2, int(n * 0.003))
            top_contrib = np.argsort(contrib)[::-1][:n_reduce]
            low_contrib = np.argsort(contrib)[:n_reduce]

            # Compute reduction
            reduction = c_max * 0.003
            delta = np.zeros(n)
            for idx in top_contrib:
                if contrib[idx] > 1e-15:
                    delta[idx] = -reduction / (2.0 * contrib[idx] + 1e-30)
            total_reduce = -delta.sum()
            for idx in low_contrib:
                delta[idx] = total_reduce / (n_reduce * (a[idx] + 1e-10) + 1e-30)

            delta -= delta.mean()

            # Line search
            alpha = 1.0
            improved = False
            for ls in range(25):
                a_try = a + alpha * delta
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-16:
                    a = a_try
                    improved = True
                    break
                alpha *= 0.5

            if not improved:
                patience += 5

    return best_a, best_sc


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n_c = 2 * n - 1

    lr = 0.003
    no_improve = 0
    tol = 0.003

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-16:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 15 == 0:
                tol = max(0.0001, tol * 0.8)
                lr *= 0.8
            if no_improve > 200:
                a = best_a.copy()
                tol = max(0.00005, tol * 0.5)
                lr = 0.001
                no_improve = 0

        # Very tight active set
        threshold = c_max * (1.0 - tol)
        active = c >= threshold
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 50:
            top_idx = np.argpartition(c, -50)[-50:]
            active = np.zeros(n_c, dtype=bool)
            active[top_idx] = True
            n_active = 50

        w = np.zeros(n_c)
        w[active] = 1.0 / n_active
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-17:
            break

        alpha = min(lr, 0.3 / (gnorm + 1e-15))
        improved = False
        for ls in range(35):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-17:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if improved:
            lr = min(0.005, lr * 1.02)

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _perturb_structured(a, n_blocks, rng):
    """Perturb in blocks for more structured diversity."""
    n = len(a)
    block_size = max(1, n // n_blocks)
    a_new = a.copy()
    for i in range(0, n, block_size):
        end = min(i + block_size, n)
        scale = 1.0 + rng.standard_normal() * 0.1
        a_new[i:end] *= max(0.0, scale)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, tapered n={n_parent}, score={best_score:.10f}")

    n = len(best_seq)

    # === Phase 1: Coarse L-BFGS continuation ===
    print(f"\n[P1] L-BFGS continuation (t={time.monotonic()-t_start:.1f}s)")
    p1_budget = (deadline - t_start) * 0.25
    a1 = _run_lbfgs_continuation(best_seq, t_start + p1_budget, min_time_remaining=600)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_coarse] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Nesterov subgradient ===
    print(f"\n[P2] Nesterov subgradient (t={time.monotonic()-t_start:.1f}s)")
    p2_budget = min(400, (deadline - t_start) * 0.2)
    a2, sc2 = _nesterov_subgrad(best_seq, t_start + p2_budget, min_time_remaining=500)
    if sc2 < best_score:
        best_seq = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [nesterov] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Spectral shaping ===
    print(f"\n[P3] Spectral shaping (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        p3_budget = min(150, (deadline - t_start) * 0.1)
        a3, sc3 = _spectral_shaping(best_seq, t_start + p3_budget, min_time_remaining=350)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [spectral] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Peak shaving ===
    print(f"\n[P4] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        p4_budget = min(200, (deadline - t_start) * 0.12)
        a4, sc4 = _peak_shaving(best_seq, t_start + p4_budget, min_time_remaining=300)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [peakshave] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: L-BFGS refinement ===
    print(f"\n[P5] L-BFGS refinement (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        p5_budget = min(200, (deadline - t_start) * 0.12)
        a5 = _run_lbfgs_continuation(best_seq, t_start + p5_budget, min_time_remaining=250)
        sc5 = _score(a5)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_refine] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Nesterov again from improved point ===
    print(f"\n[P6] Nesterov round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        p6_budget = min(200, (deadline - t_start) * 0.12)
        a6, sc6 = _nesterov_subgrad(best_seq, t_start + p6_budget, min_time_remaining=200)
        if sc6 < best_score:
            best_seq = a6.copy()
            best_score = sc6
            print(f"  NEW BEST: {best_score:.10f} [nesterov2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Multi-restart ===
    print(f"\n[P7] Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        restart_budget = deadline - 150 - time.monotonic()
        n_restart = max(3, int(restart_budget / 45))
        sigmas = [0.02, 0.04, 0.06, 0.03, 0.08, 0.025, 0.05, 0.015, 0.07, 0.035]
        for i in range(n_restart):
            if time.monotonic() > deadline - 120:
                break
            sigma = sigmas[i % len(sigmas)]
            a_pert = _perturb(best_seq, sigma, rng)
            # Short L-BFGS from perturbed point
            rb = min(35, deadline - 120 - time.monotonic())
            if rb > 10:
                a_r = _run_lbfgs_continuation(a_pert, t_start + deadline - 120 - time.monotonic() + t_start, min_time_remaining=120 - (deadline - t_start - time.monotonic()))
                # Simpler: just give it a short deadline
                a_r = _run_lbfgs_continuation(a_pert, min(deadline - 120, time.monotonic() + rb), min_time_remaining=5)
                sc_r = _score(a_r)
                if sc_r < best_score:
                    best_seq = a_r.copy()
                    best_score = sc_r
                    print(f"  NEW BEST: {best_score:.10f} [restart_{i}] t={time.monotonic()-t_start:.1f}s")

        # Try structured perturbation
        if time.monotonic() < deadline - 100:
            a_struct = _perturb_structured(best_seq, 64, rng)
            rb = min(30, deadline - 100 - time.monotonic())
            if rb > 10:
                a_s = _run_lbfgs_continuation(a_struct, min(deadline - 90, time.monotonic() + rb), min_time_remaining=5)
                sc_s = _score(a_s)
                if sc_s < best_score:
                    best_seq = a_s.copy()
                    best_score = sc_s
                    print(f"  NEW BEST: {best_score:.10f} [structured] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Multi-n exploration ===
    print(f"\n[P8] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        n_main = len(best_seq)
        n_options = []
        if n_main > 8192:
            n_options.append(8192)
        if n_main < 24576:
            n_options.append(24576)
        if n_main != 16384:
            n_options.append(16384)

        for n_try in n_options[:2]:
            if time.monotonic() > deadline - 50:
                break
            a_res = _resample(best_seq, n_try)
            rb = min(40, deadline - 50 - time.monotonic())
            if rb > 10:
                a_opt = _run_lbfgs_continuation(a_res, min(deadline - 40, time.monotonic() + rb), min_time_remaining=5)
                sc_opt = _score(a_opt)
                if sc_opt < best_score:
                    best_seq = a_opt.copy()
                    best_score = sc_opt
                    print(f"  NEW BEST: {best_score:.10f} [n={n_try}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 9: Final fine polish ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 10:
        a_fin, sc_fin = _fine_polish(best_seq, deadline - 5, min_time_remaining=3)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 10: Ultra-fine L-BFGS ===
    remaining = deadline - 3 - time.monotonic()
    if remaining > 5:
        n_fin = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        n_c = 2 * n_fin - 1
        beta_fin = max(mb * 0.0001, 1e-10)
        K_fin = max(3, int(n_c * 0.001))
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        prob = _TopkSmoProb(beta_fin, K_fin, n_fin)
        max_iter = min(5000, max(10, int(remaining / 0.02)))
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxcor': 100})
            a_ultra = np.exp(result.x - result.x.max())
            a_ultra = a_ultra / a_ultra.sum()
            sc_ultra = _score(a_ultra)
            if sc_ultra < best_score:
                best_seq = a_ultra.copy()
                best_score = sc_ultra
                print(f"  NEW BEST: {best_score:.10f} [ultra] t={time.monotonic()-t_start:.1f}s")
        except Exception:
            pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
