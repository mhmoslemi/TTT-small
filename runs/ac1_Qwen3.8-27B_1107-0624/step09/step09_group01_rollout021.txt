"""
Optimized step function sequence minimizer using adaptive log-sum-exp smoothing,
iterative active-set Newton, multi-scale exploration, and gradient-enhanced L-BFGS-B.
Targets improvement below 1.5027 by more aggressive temperature annealing and
better KKT-based refinement near the optimum.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smooth_max_and_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return c_max, w
    w = e / s
    return c_max + beta * np.log(s), w


def _grad_conv_wrt_a(w, a, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _softmax_grad(grad_a, a):
    return grad_a * a - np.sum(grad_a * a) * a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    elif kind == "ripple":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2 * np.pi * t) + 0.05 * np.cos(6 * np.pi * t)
        a = np.maximum(a, 1e-8)
        return a / a.sum()
    elif kind == "optimal_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = a * (1.0 + 0.05 * np.cos(4 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _optimize_l_bfgs(z_init, n, beta, deadline, max_iter=5000):
    def obj(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        val, _ = _smooth_max_and_weights(c, beta)
        return 2.0 * n * val

    def grad(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        _, w = _smooth_max_and_weights(c, beta)
        ga = _grad_conv_wrt_a(w, a, n)
        return 2.0 * n * _softmax_grad(ga, a)

    remaining = deadline - time.monotonic()
    if remaining < 3:
        a = _softmax(z_init)
        return z_init, _score(a)

    est_iter_time = 0.003 * max(1.0, n / 16384.0)
    max_iter = min(max_iter, max(20, int(remaining / (est_iter_time * 8))))

    try:
        result = minimize(
            obj, z_init, jac=grad,
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-14,
                     'maxcor': 50, 'maxls': 200}
        )
        a = _softmax(result.x)
        return np.asarray(result.x), _score(a)
    except Exception:
        a = _softmax(z_init)
        return z_init, _score(a)

        return ga_test - ga_base - np.mean(ga_test - ga_base)

    for rnd in range(max_rounds):

            dx = x - np.mean(x)
            dx_n = float(np.linalg.norm(dx))
            if dx_n < 1e-20 or not np.isfinite(dx_n):
                dx = -ga_c / max(ga_norm, 1e-30)

        # Line search
        step_done = False
        alpha_ls = 1.0
        for ls in range(50):
            a_try = a + alpha_ls * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha_ls *= 0.5
                continue
            a_try = a_try / s_try

                    step_done = True
                    break

            if no_improve > 5:
                if k_idx < len(k_schedule) - 1:
                    k_idx += 1
                    no_improve = 0
                else:
                    break
        else:
            no_improve = 0

        sc = _score(a)
        if sc < best_sc - 1e-15:

            best_a = a.copy()

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-300))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _adaptive_beta(c, c_max, current_beta):
    """Adaptively update beta based on the spread of top convolution values."""
    if c_max < 1e-30:
        return current_beta * 0.5
    c_sorted = np.sort(c)[::-1]
    n_c = len(c_sorted)
    if n_c < 10:
        return current_beta * 0.5

    gap_1 = c_sorted[0] - c_sorted[1]
    gap_10 = c_sorted[0] - c_sorted[9]
    gap_100 = c_sorted[0] - c_sorted[min(99, n_c - 1)]
    gap_500 = c_sorted[0] - c_sorted[min(499, n_c - 1)]

    # If top values are very close, we can reduce beta aggressively
    if gap_10 < 1e-8 * c_max:
        return max(current_beta * 0.2, c_max * 1e-10)
    elif gap_10 < 1e-6 * c_max:
        return max(current_beta * 0.3, c_max * 1e-8)
    elif gap_10 < 1e-4 * c_max:
        return max(current_beta * 0.4, c_max * 1e-6)
    elif gap_100 < 1e-3 * c_max:
        return max(current_beta * 0.5, c_max * 1e-5)
    elif gap_500 < 1e-2 * c_max:
        return max(current_beta * 0.6, c_max * 1e-4)
    else:
        return max(current_beta * 0.7, c_max * 1e-3)
