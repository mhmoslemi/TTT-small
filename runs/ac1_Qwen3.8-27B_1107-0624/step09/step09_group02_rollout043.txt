"""
Convex optimizer for minimum peak linear autocorrelation sequences.
Uses: (1) multiplicative pressure equalization (mirror descent),
(2) active-set convex QP with L-BFGS inner solve,
(3) Nesterov-accelerated projected subgradient,
(4) L-BFGS with log-sum-exp smoothing for final polish.
Key insight: max_k C[k] is convex on a>=0, enabling KKT-based active-set methods.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(x):
    k = 1
    while k < x:
        k <<= 1
    return max(2, k)


def _full_conv(a, b):
    n, m = len(a), len(b)
    out = n + m - 1
    N = _next_pow2(out)
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:out]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_conv(a, a)
    return 2.0 * n * float(c.max()) / (s * s)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    r = np.where(u * np.arange(1, n + 1) > (cs - 1))[0]
    if len(r) == 0:
        return np.ones(n) / n
    r = r[-1]
    th = (cs[r] - 1) / (r + 1)
    return np.maximum(v - th, 0)


def _grad_topK(a, top_indices):
    """Gradient of sum_{k in top} C[k] w.r.t. a, via FFT. O(n log n)."""
    n = len(a)
    ind = np.zeros(2 * n - 1)
    ind[top_indices] = 1.0
    ind_rev = ind[::-1]
    conv = _full_conv(a, ind_rev)
    return 2.0 * conv[n - 1:2 * n - 1][::-1]


def _grad_weighted(a, weights):
    """Gradient of sum_k w[k]*C[k] w.r.t. a. weights has length 2n-1."""
    n = len(a)
    ind = weights.astype(np.float64)
    ind_rev = ind[::-1]
    conv = _full_conv(a, ind_rev)
    return 2.0 * conv[n - 1:2 * n - 1][::-1]


def _run_pressure_eq(a_init, deadline, min_remaining=120.0):
    """Multiplicative pressure equalization (mirror descent on simplex with entropy reg)."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    p_schedule = [1.0, 2.0, 3.0, 5.0, 8.0, 13.0, 21.0, 34.0, 55.0, 89.0, 144.0, 233.0, 377.0, 610.0]
    alpha = 0.4

    for p in p_schedule:
        if time.monotonic() + min_remaining > deadline:
            break
        n_inner = 300
        stalled = 0
        for it in range(n_inner):
            if time.monotonic() + min_remaining > deadline:
                break
            c = _full_conv(a, a)
            c_max = float(c.max())
            if c_max < 1e-30:
                break
            w = c ** p
            w_sum = w.sum()
            if w_sum < 1e-30:
                break
            w = w / w_sum

            P = _grad_weighted(a, w) * 0.5
            P = np.maximum(P, 1e-30)
            P_mean = P.mean()

            ratio = (P_mean / P) ** alpha
            a = a * ratio
            a = np.maximum(a, 0)
            s2 = a.sum()
            if s2 < 1e-15:
                a = best_a.copy()
                break
            a = a / s2

            if it % 20 == 0:
                sc = _score(a)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best_a = a.copy()
                    stalled = 0
                else:
                    stalled += 1
                    if stalled > 15:
                        break

        sc = _score(a)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()

    return best_a


def _run_fista(a_init, deadline, min_remaining=120.0):
    """Nesterov-accelerated projected subgradient on the simplex."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    y = a.copy()
    t_k = 1.0
    K_top = 80
    f_lb = 1.5000
    patience = 0
    max_patience = 200

    for it in range(500000):
        if time.monotonic() + min_remaining > deadline:
            break

        c = _full_conv(y, y)
        c_max = float(c.max())
        f_y = 2.0 * n * c_max

        top_idx = np.argsort(c)[::-1][:K_top]
        active_thresh = c_max * 0.9995
        active_mask = c[top_idx] >= active_thresh
        active_idx = top_idx[active_mask]
        if len(active_idx) < 1:
            active_idx = top_idx[:1]

        g = _grad_topK(y, active_idx)
        g_norm_sq = float(np.dot(g, g))
        if g_norm_sq < 1e-20:
            break

        gap = f_y - f_lb
        if gap > 1e-10:
            lr = gap / g_norm_sq
        else:
            lr = 1.0 / (np.sqrt(g_norm_sq) + 1e-10)
        lr = min(lr, 0.5)

        x_new = _proj_simplex(y - lr * g)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = _proj_simplex(y)
        a = x_new
        t_k = t_new

        if it % 10 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    a = best_a.copy()
                    y = a.copy()
                    t_k = 1.0
                    patience = 0

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _active_set_lbfgs(a_init, deadline, min_remaining=60.0):
    """Active-set method: identify peak lags, solve smoothed QP with L-BFGS in log-space."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    c_cur = _full_conv(a, a)
    max_c = float(c_cur.max())
    gap_c = c_cur.max() - c_cur.min()

    # Beta schedule for active set tolerance
    fracs = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]

    for frac in fracs:
        if time.monotonic() + min_remaining > deadline:
            break
        c = _full_conv(a, a)
        c_max = float(c.max())
        thresh = c_max * (1.0 - frac)
        active = np.where(c >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # L-BFGS in log-space to minimize (1/|K|) sum_{k in K} C[k]
        K = active
        nK = len(K)
        z0 = np.log(np.maximum(a, 1e-30))
        z0 = z0 - z0.mean()

        def obj_and_grad(z_local):
            a_local = _softmax(z_local)
            g = _grad_topK(a_local, K)
            val = float(np.dot(a_local, g) * 0.5)
            # Since grad C[k] = 2*contrib, the objective sum C[k]/nK has grad = grad_topK/nK
            # And the value is (1/nK) * sum_k C[k]
            # For L-BFGS we just need the gradient of the objective
            grad = g / nK
            return val, grad

        def obj_func(z_local):
            a_local = _softmax(z_local)
            g = _grad_topK(a_local, K)
            return float(np.dot(a_local, g) * 0.5)

        def grad_func(z_local):
            a_local = _softmax(z_local)
            g = _grad_topK(a_local, K)
            return g / nK

        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        est_time = 0.005 * max(1.0, n / 8192.0)
        max_iter = min(2000, max(50, int(remaining / (est_time * 3))))

        try:
            result = minimize(
                obj_func, z0, jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 50, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new.copy()
            else:
                a = best_a.copy()
        except Exception:
            pass

    return best_a, best_sc


def _run_lbfgs_smoothed(a_init, deadline, min_remaining=60.0):

    n = len(a_init)

    best_a = a.copy()
    best_sc = _score(a)

    c_cur = _full_conv(a, a)

    c_sorted = np.sort(c_cur)[::-1]

    base_beta = max(gap * 0.1, max_c * 0.005)

    ratios = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
              0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    for ratio in ratios:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        beta = base_beta * ratio
        est_time = 0.004 * max(1.0, n / 8192.0)
        max_iter = min(3000, max(50, int(remaining / (est_time * 4))))

        def obj_func(z_local):
            a_local = _softmax(z_local)
            c_local = _full_conv(a_local, a_local)
            c_max_local = float(c_local.max())
            shifted = (c_local - c_max_local) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s_e = e.sum()
            if s_e < 1e-300:
                return 2.0 * n * c_max_local
            return 2.0 * n * (c_max_local + beta * np.log(s_e))

        def grad_func(z_local):
            a_local = _softmax(z_local)
            c_local = _full_conv(a_local, a_local)
            c_max_local = float(c_local.max())
            shifted = (c_local - c_max_local) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s_e = e.sum()
            if s_e < 1e-300:
                w = np.zeros_like(c_local)

            else:
                w = e / s_e
            a_rev = a_local[::-1]
            wc = _full_conv(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W_val = float(np.dot(w, c_local))
            return 4.0 * n * a_local * (h - W_val)

        try:
            result = minimize(
                obj_func, z, jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,

            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return best_a, best_sc

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(8192) / 8192.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
