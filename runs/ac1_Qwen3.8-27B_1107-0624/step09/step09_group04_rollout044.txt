    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _logsumexp_approx(c, beta):
    """Smooth approximation of max(c) using log-sum-exp."""
    c_max = np.max(c)
    if beta < 1e-15:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)

def _smooth_objective(z, beta, n):
    """Smooth objective in log-space (softmax parameterization)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    sm = _logsumexp_approx(c, beta)
    return 2.0 * n * sm

def _smooth_gradient(z, beta, n):
    """Gradient of smooth objective w.r.t. z (log-space)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # Gradient of smooth max w.r.t. a:
    # d(smooth_max)/da_j = sum_k w_k * d(a*a)_k/da_j = 2 * sum_k w_k * a[k-j] (shifted)
    a_rev = a[::-1]
    h = _full_convolve(w, a_rev)
    grad_a = 2.0 * h[n-1:2*n-1]
    # Chain rule: d/dz = da/dz * d/da
    # da/dz = diag(a) - a*a^T (Jacobian of softmax)
    # So grad_z = a * (grad_a - sum(a * grad_a))
    W = float(np.dot(w, c))
    # Actually let me recompute more carefully
    # grad_a[j] = 2 * (w * a_rev)[n-1+j] ... no
    # Let me use the formula: d(a*a)_k / da_j = 2 * a[k-j] if 0 <= k-j <= n-1
    # So grad_a[j] = sum_k w_k * 2 * a[k-j] (where a[k-j] = 0 if k-j not in [0,n-1])
    # This is: grad_a = 2 * (w * a_rev_flipped)[n-1 : 2n-1] where * is cross-correlation
    # Actually: grad_a[j] = 2 * sum_k w_k * a[k-j]
    # Let b[k] = w_k, then grad_a = 2 * (b cross-correlate a)
    # cross-correlation: (b ⋆ a)[j] = sum_k b[k] a[k-j]
    # This is the same as full convolution of b with a reversed, shifted:
    # (b * a_rev)[j + (n-1)] = sum_k b[k] a_rev[n-1-k+j-n+1] = sum_k b[k] a[j-k]
    # Hmm, let me just use the FFT approach
    # grad_a = 2 * ifft(fft(w) * conj(fft(a))) ... no
    # Actually the simplest: grad_a[j] = 2 * sum_{k: 0<=k-j<=n-1} w[k] * a[k-j]
    # = 2 * sum_{i: 0<=i<=n-1} w[j+i] * a[i]  (substituting k=j+i)
    # This is the cross-correlation of w and a at lag j
    # = (w * flip(a))[j + n - 1] in full convolution
    a_flip = a[::-1]
    conv_wa = _full_convolve(w, a_flip)
    grad_a = 2.0 * conv_wa[n-1:2*n-1]
    
    # Now grad_z = a * (grad_a - dot(a, grad_a))
    # Because softmax Jacobian: d a_i / d z_j = a_i (delta_ij - a_j)
    # So (J^T g)_j = a_j (g_j - sum_i a_i g_i)
    W_a = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - W_a)
    return grad_z

def _lbfgs_smooth(a_init, deadline, min_time_remaining=30.0, beta_start=None, beta_end=None, n_betas=10):
    """Run L-BFGS with a schedule of decreasing beta values."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    gap = max(c_sorted[min(20, len(c_sorted)-1)] - c_sorted[-1], c_max * 0.001)
    
    if beta_start is None:
        beta_start = max(gap, c_max * 0.01)
    if beta_end is None:
        beta_end = beta_start * 1e-6
    
    # Geometric schedule
    ratios = (beta_end / beta_start) ** (1.0 / max(1, n_betas - 1))
    betas = [beta_start * ratios**i for i in range(n_betas)]
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    best_a = a.copy()
    best_sc = _score(a)
    
    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Scale iterations based on beta and n
        if beta < beta_start * 0.01:
            max_iter = min(3000, max(50, int(remaining / 0.05)))
        else:
            max_iter = min(2000, max(30, int(remaining / 0.08)))
        
        def obj(z):
            return _smooth_objective(z, beta, n)
        
        def grad(z):
            return _smooth_gradient(z, beta, n)
        
        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_a = a_new
                best_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return best_a

def _active_set_newton(a_init, deadline, min_time_remaining=30.0):
    """Active set Newton method for peak equalization.
    Solves the KKT system for the active set using Newton's method."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = np.maximum(a, 0)
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    
    K = 5  # Active set size
    max_newton_iters = 50
    line_search_factor = 0.5
    
    for outer in range(20):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
        
        # Identify active set: top K peaks
        # Use a threshold: peaks within 0.1% of max
        threshold = c_max * (1 - 0.001)
        active = np.where(c >= threshold)[0]
        if len(active) < 2:
            active = np.argsort(c)[::-1][:K]
        active = active[:min(30, len(active))]  # Cap at 30
        m = len(active)
        
        if m < 2:
            break
        
        # Newton iteration on the KKT system
        # Variables: (a, t, lambda, mu)
        # Constraints:
        #   (a*a)_{k_i} - t = 0 for i = 1,...,m
        #   sum(a) - 1 = 0
        #   lambda_i * ((a*a)_{k_i} - t) = 0 (complementarity)
        #   a_j * nu_j = 0 (complementarity)
        
        # Simplified: assume all a_j > 0 (interior point)
        # Then we solve: (a*a)_{k_i} = t for all i, sum(a) = 1
        # Using Lagrange multipliers: grad_a [sum_i lam_i * ((a*a)_{k_i} - t)] + mu * 1 = 0
        
        # This is: sum_i lam_i * 2 * shift(a, k_i) + mu * 1 = 0
        # where shift(a, k)_j = a[k-j] if valid
        
        # In matrix form: 2 * A(lam) + mu * 1 = 0
        # where A(lam)_j = sum_i lam_i * a[k_i - j]
        
        # The system is:
        # (a*a)_{k_i} = t (m equations)
        # sum_j [sum_i lam_i * 2*a[k_i-j]] + mu*n = 0 (1 equation, from taking dot with 1)
        # sum(a) = 1 (1 equation)
        
        # Unknowns: a (n), t (1), lam (m), mu (1) = n+m+2 unknowns
        # Equations: m + 1 + 1 = m+2 ... not enough!
        # We need n equations from the gradient condition.
        # Actually: sum_i lam_i * 2 * shift(a, k_i) + mu * 1 = 0 gives n equations.
        # Total: m + n + 1 = n + m + 1 equations for n + m + 2 unknowns.
        # One equation short - we can fix mu = -1 (gauge freedom) or add another condition.
        
        # Let me use a different approach: just do gradient descent on the active set constraints.
        
        break  # Newton is too complex, fall back to gradient approach
    
    # Fallback: gradient-based active set equalization
    return _active_set_gradient(a, best_a, best_sc, deadline, min_time_remaining)

def _active_set_gradient(a_init, best_a, best_sc, deadline, min_time_remaining=30.0):
    """Gradient-based active set equalization with adaptive step size."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    
    best_a_local = best_a.copy()
    best_sc_local = best_sc
    lr = 0.01
    no_improve = 0
    K = 5
    
    for it in range(5000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc_local - 1e-14:
            best_sc_local = current_sc
            best_a_local = a.copy()
            no_improve = 0
            K = min(50, K + 1)
        
        # Active set: top K peaks or peaks within 0.05% of max
        threshold = c_max * (1 - 0.0005)
        active = np.where(c >= threshold)[0]
        if len(active) < 2:
            active = np.argsort(c)[::-1][:K]
        active = active[:min(K, len(active))]
        m = len(active)
        
        # Compute gradient: average of gradients of active peaks
        w = np.zeros(2*n-1)
        w[active] = 1.0 / m
        a_rev = a[::-1]
        h = _full_convolve(w, a_rev)
        grad = 2.0 * h[n-1:2*n-1]
        
        # Project to simplex direction (zero mean)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Adaptive step size
        alpha = lr / (gnorm * np.max(np.abs(grad)) + 1e-15)
        alpha = min(alpha, 0.5)
        
        improved = False
        for ls in range(40):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < current_sc * (1 - 1e-12):
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            lr *= 0.9
            if no_improve >= 100:
                a = best_a_local.copy()
                lr = 0.01
                no_improve = 0
                break
        else:
            lr = min(0.05, lr * 1.005)
    
    return a, best_sc_local  # Actually return best_a_local
    # Fix: return best_a_local, best_sc_local

def _spline_init(n, m=20, rng=None):
    """Generate a smooth sequence using B-spline parameterization."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    # Random control points
    t = np.linspace(0, 1, n)
    # Create a smooth positive function
    freqs = rng.integers(1, 5, size=min(m//4, 10))
    amps = rng.uniform(0.1, 1.0, size=len(freqs))
    phases = rng.uniform(0, 2*np.pi, size=len(freqs))
    
    f = np.ones(n) * 0.5
    for i, (fr, amp, ph) in enumerate(zip(freqs, amps, phases)):
        f += amp * np.sin(2*np.pi*fr*t + ph)
    
    f = np.maximum(f, 0.01)
    return f / f.sum()
