import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_weighted(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _smooth_objective_grad(z, beta, K, n):
    """Top-K log-sum-exp smoothed objective in log-space."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c
    
    # Adaptive: if peaks are very close, use effective beta = min(beta, spread)
    if len(c_top) > 1:
        spread = c_top.max() - c_top.min()
        eff_beta = min(beta, max(spread * 0.3, beta * 0.01))
    else:
        eff_beta = beta
    
    shifted = (c_top - c_max) / eff_beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + eff_beta * np.log(s))
    
    ga = _grad_weighted(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z, a


def _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=30.0, max_seconds=None):
    """L-BFGS with adaptive beta and K."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc
    
    # Initial beta based on peak spread
    c_sorted = np.sort(c_cur)[::-1]
    n_check = min(50, len(c_sorted))
    gap = c_sorted[0] - c_sorted[n_check - 1]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap * 0.5, max_c * 0.005)
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # Schedule: (beta_multiplier, K_fraction)
    # More gradual than before
    schedule = [
        (1.0, 0.15), (0.8, 0.12), (0.6, 0.10), (0.4, 0.08),
        (0.3, 0.06), (0.2, 0.05), (0.12, 0.04), (0.08, 0.03),
        (0.05, 0.025), (0.03, 0.02), (0.015, 0.015),
        (0.008, 0.012), (0.004, 0.010), (0.002, 0.008),
        (0.001, 0.006), (0.0005, 0.004), (0.0002, 0.003),
    ]
    
    t0 = time.monotonic()
    if max_seconds is None:
        max_seconds = deadline - time.monotonic() - min_time_remaining
    
    for phase, (beta_mult, K_frac) in enumerate(schedule):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 15:
            break
        
        beta = base_beta * beta_mult
        K = max(5, int((2 * n - 1) * K_frac))
        
        class Prob:
            def __init__(self):
                self._last_a = None
            def f(self, z):
                obj, grad, a = _smooth_objective_grad(z, beta, K, n)
                self._last_a = a
                return obj
            def g(self, z):
                obj, grad, a = _smooth_objective_grad(z, beta, K, n)
                self._last_a = a
                return grad
        
        prob = Prob()
        est_time_per_iter = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 30))))
        
        try:
            result = minimize(
                prob.f, z, jac=prob.g, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 150, 'maxls': 80}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    Phase {phase}: beta_mult={beta_mult:.4f}, K={K}, score={current_sc:.10f}")
        except Exception as e:
            print(f"    Phase {phase} error: {e}")
    
    # Final exact evaluation
    sc_final = _score(current_a)
    return current_a, sc_final


def _subgradient_adaptive(a_init, deadline, min_time_remaining=60.0):
    """Adaptive active-set subgradient with momentum and BB steps."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    n_2nm1 = 2 * n - 1
    momentum = 0.5
    lr = 0.01
    lr_min = 1e-6
    lr_max = 0.05
    stall = 0
    max_stall = 100
    velocity = np.zeros(n)
    prev_a = None
    prev_g = None
    
    iteration = 0
    t_start = time.monotonic()
    last_lbfgs_time = t_start
    
    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        
        # Adaptive active set: peaks within adaptive tolerance of max
        # Start tight, loosen if stuck
        tol_frac = 0.001 + 0.003 * (stall / max_stall)
        tol_frac = min(tol_frac, 0.05)
        threshold = c_max * (1 - tol_frac)
        active_mask = c >= threshold
        n_active = int(active_mask.sum())
        if n_active < 3:
            # Ensure at least 3 active
            idx_sorted = np.argsort(c)[::-1]
            active_mask = np.zeros(n_2nm1, dtype=bool)
            active_mask[idx_sorted[:3]] = True
            n_active = 3
        
        # Weights: emphasize peaks closer to max
        active_vals = c[active_mask]
        w = np.zeros(n_2nm1)
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w[active_mask] = ((active_vals - active_vals.min()) / spread) ** 2 + 0.05
        else:
            w[active_mask] = 1.0 / n_active
        w = w / w.sum()
        
        g = _grad_weighted(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break
        
        # BB step
        if prev_a is not None and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g_centered - prev_g
            s_norm = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm > 1e-18 and abs(sy) > 1e-18:
                lr_bb = s_norm / abs(sy)
                lr = np.clip(0.7 * lr + 0.3 * lr_bb, lr_min, lr_max)
        
        # Momentum + gradient step
        velocity = momentum * velocity - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        
        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            for ls in range(30):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-16:
                    improved = True
                    break
                velocity = 0.5 * (momentum * velocity - lr * g_centered)
                a_try = a + velocity
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    velocity = np.zeros(n)
                    a_try = a
                    break
                a_try = a_try / s_try
                lr *= 0.75
        
        if improved:
            prev_a = a.copy()
            prev_g = g_centered.copy()
            a = a_try
            # Gradually increase momentum
            if iteration % 200 == 0:
                momentum = min(0.8, momentum + 0.02)
        else:
            velocity = np.zeros(n)
            lr = max(lr_min, lr * 0.5)
            stall += 1
        
        if stall > max_stall:
            # Reset: go back to best, try different direction
            a = best_a.copy()
            velocity = np.zeros(n)
            lr = 0.01
            stall = 0
            prev_a = None
            prev_g = None
        
        # Periodic L-BFGS restart
        if iteration % 300 == 0 and time.monotonic() - last_lbfgs_time > 60:
            if time.monotonic() + min_time_remaining + 60 < deadline:
                lbfgs_deadline = min(deadline - min_time_remaining - 10, time.monotonic() + 50)
                a_tmp, sc_tmp = _run_adaptive_lbfgs(a, lbfgs_deadline, min_time_remaining=5)
                if sc_tmp < best_sc:
                    best_a = a_tmp.copy()
                    best_sc = sc_tmp
                    a = a_tmp.copy()
                    velocity = np.zeros(n)
                    prev_a = None
                    prev_g = None
                last_lbfgs_time = time.monotonic()
    
    return best_a, best_sc


def _peak_equalization(a_init, deadline, min_time_remaining=30.0):
    """Directly reduce top peaks by targeted mass transfer."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    patience = 0
    max_patience = 300
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        
        # Find the top peaks
        n_peaks = max(5, int(n * 0.005))
        peak_idx = np.argsort(c)[::-1][:n_peaks]
        peak_vals = c[peak_idx]
        
        if len(peak_vals) < 2:
            break
        
        # For each peak, compute which a[i] contribute most
        # Gradually reduce the highest peak
        k_peak = peak_idx[0]
        i_lo = max(0, k_peak - (n - 1))
        i_hi = min(n - 1, k_peak)
        
        contrib = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
            j = k_peak - i
            contrib[i] += a[j]
            if i != j:
                contrib[j] += a[i]
        
        # Also check which of these contribute to OTHER peaks (to avoid hurting them)
        other_contrib = np.zeros(n)
        for k_other in peak_idx[1:min(len(peak_idx), 5)]:
            i_lo2 = max(0, k_other - (n - 1))
            i_hi2 = min(n - 1, k_other)
            for i in range(i_lo2, i_hi2 + 1):
                j = k_other - i
                other_contrib[i] += a[j]
                if i != j:
                    other_contrib[j] += a[i]
        
        # We want to reduce elements that contribute to the main peak but NOT to other peaks
        # Score: high contrib to main peak, low contrib to other peaks
        net_score = contrib - 0.5 * other_contrib
        
        # Reduce top net-score elements, add to bottom net-score elements
        n_transfer = max(3, int(n * 0.003))
        sorted_idx = np.argsort(net_score)[::-1]
        reduce_idx = sorted_idx[:n_transfer]
        add_idx = sorted_idx[-n_transfer:]
        
        # Compute transfer amount
        reduction = c_max * 0.002
        delta = np.zeros(n)
        for idx in reduce_idx:
            if contrib[idx] > 1e-15 and a[idx] > 1e-15:
                # How much to reduce to lower c[k_peak] by reduction
                # dc[k_peak]/da[idx] = 2*a[k_peak-idx] (approximately)
                deriv = 2.0 * a[max(0, min(n-1, k_peak - idx))]
                if deriv > 1e-15:
                    delta[idx] = -reduction / (deriv * n_transfer)
        
        # Redistribute
        total_reduced = -delta[reduce_idx].sum()
        if total_reduced > 0:
            for idx in add_idx:
                delta[idx] = total_reduced / n_transfer
        
        # Preserve sum
        delta -= delta.mean()
        
        # Line search
        alpha = 1.0
        improved = False
        for ls in range(35):
            a_try = a + alpha * delta
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-16:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            patience += 5
            # Try smaller reduction
            reduction *= 0.5
    
    return best_a, best_sc


def _fine_polish(a_init, deadline, min_time_remaining=5.0):
    """Very fine active-set polish."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.002
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        # Active set: peaks within 0.1% of max
        threshold = c_max * 0.999
        active = c >= threshold
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active
        
        g = _grad_weighted(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break
        
        alpha = min(lr, 1.0 / (gnorm * gnorm + 1e-10))
        improved = False
        for ls in range(50):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try

            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-16:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve > 100:
                a = best_a.copy()
                lr *= 0.5
                no_improve = 0
            if lr < 1e-8:
                break
        else:
            lr = min(0.005, lr * 1.02)
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "optimal_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.01
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_weighted(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _smooth_obj(z, beta, K, n):
    """Compute smoothed objective value."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        c_top = c
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smooth_grad(z, beta, K, n):
    """Compute gradient of smoothed objective."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
    ga = _grad_weighted(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return grad_z


def _run_lbfgs_phase(a_init, beta, K, n, max_iter, deadline, min_remaining=10.0):
    """Run L-BFGS for one phase with fixed beta and K."""
    if time.monotonic() + min_remaining > deadline:
        return a_init, _score(a_init)
    
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    remaining = deadline - time.monotonic() - min_remaining
    if remaining < 10:
        return a_init, _score(a_init)
    
    max_iter = min(max_iter, max(20, int(remaining / 0.005)))
    
    def obj_func(z_eval):
        return _smooth_obj(z_eval, beta, K, n)
    
    def grad_func(z_eval):
        return _smooth_grad(z_eval, beta, K, n)
    
    try:
        result = minimize(
            obj_func, z, jac=grad_func, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                     'maxcor': 100, 'maxls': 60}
        )
        a_new = np.exp(result.x - result.x.max())
        a_new = a_new / a_new.sum()
        sc_new = _score(a_new)
        return a_new, sc_new
    except Exception:
        return a_init, _score(a_init)


def _continuation_lbfgs(a_init, deadline, min_time_remaining=60.0):
    """Continuation method: start with large beta, decrease geometrically."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return best_a, best_sc
    
    # Determine initial beta from peak structure
    c_sorted = np.sort(c_cur)[::-1]
    n_check = min(100, len(c_sorted))
    gap = c_sorted[0] - c_sorted[n_check - 1]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    init_beta = max(gap * 0.5, max_c * 0.01)
    
    # Continuation schedule: geometric decrease of beta
    beta_levels = []
    beta = init_beta
    K = max(10, int((2 * n - 1) * 0.15))
    while beta > max_c * 0.00005:
        beta_levels.append((beta, K))
        beta *= 0.6
        K = max(5, int(K * 0.8))
    
    # Ensure we have enough levels
    if len(beta_levels) < 8:
        while len(beta_levels) < 12:
            b, k = beta_levels[-1]
            b *= 0.5
            k = max(3, int(k * 0.85))
            beta_levels.append((b, k))
    
    for phase_idx, (beta, K) in enumerate(beta_levels):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - time.monotonic() - min_time_remaining
        if remaining < 20:
            break
        
        # More iterations for later phases (finer optimization)
        if phase_idx < len(beta_levels) // 2:
            max_iter = 3000
        else:
            max_iter = 5000
        
        a_new, sc_new = _run_lbfgs_phase(a, beta, K, n, max_iter, deadline, min_time_remaining)
        if sc_new < best_sc - 1e-15:
            best_a = a_new.copy()
            best_sc = sc_new
            a = a_new
        else:
            # If no improvement, still use the new point (might help next phase)
            a = a_new
            a = _normalize(a)
        
        if (phase_idx + 1) % 4 == 0:
            print(f"    Continuation phase {phase_idx+1}/{len(beta_levels)}: "
                  f"beta={beta:.2e}, K={K}, score={_score(a):.10f}")
    
    final_sc = _score(best_a)
    return best_a, final_sc


def _subgradient_exact(a_init, deadline, min_time_remaining=30.0):
    """Exact subgradient with adaptive active set and careful line search."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    n_2nm1 = 2 * n - 1
    momentum = 0.3
    lr = 0.005
    velocity = np.zeros(n)
    stall = 0
    max_stall = 150
    prev_a = None
    prev_g = None
    iteration = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        
        # Adaptive active set: peaks within tolerance of max
        # Tolerance starts tight and loosens with stalls
        tol_frac = min(0.01, 0.0005 + 0.0002 * stall)
        threshold = c_max * (1.0 - tol_frac)
        active_mask = c >= threshold
        n_active = int(active_mask.sum())
        
        if n_active < 2:
            idx_sorted = np.argsort(c)[::-1]
            active_mask = np.zeros(n_2nm1, dtype=bool)
            active_mask[idx_sorted[:max(2, n_active)]] = True
            n_active = max(2, n_active)
        
        # Equal weights for active peaks (simpler and often better)
        w = np.zeros(n_2nm1)
        w[active_mask] = 1.0 / n_active
        
        g = _grad_weighted(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-16:
            break
        
        # Barzilai-Borwein step size
        if prev_a is not None and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g_centered - prev_g
            s_norm = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm > 1e-18 and abs(sy) > 1e-18:
                lr_bb = s_norm / abs(sy)
                lr = np.clip(0.5 * lr + 0.5 * lr_bb, 1e-6, 0.02)
        
        # Momentum step with line search
        velocity = momentum * velocity - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        
        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            for ls in range(35):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-16:
                    improved = True
                    break
                # Halve the step
                velocity = 0.5 * (momentum * velocity - lr * g_centered)
                a_try = a + velocity
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    velocity = np.zeros(n)
                    break
                a_try = a_try / s_try
                lr *= 0.7
                if lr < 1e-8:
                    break
        
        if improved:
            prev_a = a.copy()
            prev_g = g_centered.copy()
            a = a_try
            lr = min(0.02, lr * 1.01)
        else:
            velocity = np.zeros(n)
            lr = max(1e-7, lr * 0.5)
            stall += 1
        
        if stall > max_stall:
            a = best_a.copy()
            velocity = np.zeros(n)
            lr = 0.005
            stall = 0
            prev_a = None
            prev_g = None
            momentum = min(0.7, momentum + 0.05)
    
    return best_a, best_sc


def _pairwise_swap(a_init, deadline, min_time_remaining=20.0):
    """Try pairwise coordinate swaps to reduce the peak."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    # Identify the peak position
    k_peak = int(np.argmax(c))
    
    # Find which pairs (i, j) contribute to the peak
    # c[k_peak] = sum_i a[i] * a[k_peak - i]
    # Swapping a[i] and a[j] changes c[k_peak] by:
    # (a[j]*a[k_peak-i] + a[i]*a[k_peak-j]) - (a[i]*a[k_peak-i] + a[j]*a[k_peak-j])
    # = (a[j] - a[i]) * (a[k_peak-i] - a[k_peak-j])
    
    # We want to REDUCE c[k_peak], so we want:
    # (a[j] - a[i]) * (a[k_peak-i] - a[k_peak-j]) < 0
    # i.e., if a[j] > a[i], we want a[k_peak-i] > a[k_peak-j]
    # (move mass from a position that contributes a lot to the peak 
    #  to one that contributes less)
    
    i_lo = max(0, k_peak - (n - 1))
    i_hi = min(n - 1, k_peak)
    
    # Compute contribution of each i to the peak
    contrib = np.zeros(n)
    for i in range(i_lo, i_hi + 1):
        j = k_peak - i
        contrib[i] = a[j]
        if i != j:
            contrib[j] += a[i]
    
    # Sort by contribution (high to low)
    sorted_idx = np.argsort(contrib)[::-1]
    
    patience = 0
    max_patience = 200
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        
        k_peak = int(np.argmax(c))
        i_lo = max(0, k_peak - (n - 1))
        i_hi = min(n - 1, k_peak)
        
        contrib = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
            j = k_peak - i
            contrib[i] = a[j]
            if i != j:
                contrib[j] += a[i]
        
        # Try swapping high-contrib with low-contrib elements
        n_high = max(3, int(n * 0.002))
        n_low = max(3, int(n * 0.002))
        sorted_idx = np.argsort(contrib)[::-1]
        high_set = set(sorted_idx[:n_high].tolist())
        low_set = set(sorted_idx[-n_low:].tolist())
        
        found_improvement = False
        for i in sorted_idx[:n_high]:
            if found_improvement:
                break
            for j in sorted_idx[-n_low:]:
                if j == i:
                    continue
                # Try partial swap: move fraction f of a[i] to a[j]
                # c[k_peak] change = (a[j] + f*(a[i]-a[j])) * (a[k_peak-i] - f*(...)) 
                # This is complex; just try a few fractions
                for f in [0.1, 0.3, 0.5]:
                    delta_i = -f * a[i]
                    delta_j = f * a[i]
                    a_try = a.copy()
                    a_try[i] += delta_i
                    a_try[j] += delta_j
                    if a_try[i] < 0 or a_try[j] < 0:
                        continue
                    s_try = a_try.sum()
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    sc_try = 2.0 * n * float(c_try.max())
                    if sc_try < current_sc - 1e-15:
                        a = a_try
                        found_improvement = True
                        break
                if found_improvement:
                    break
        
        if not found_improvement:
            patience += 5
    
    return best_a, best_sc


def _fine_polish(a_init, deadline, min_time_remaining=5.0):
    """Very fine polish with tiny steps and tight active set."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.001
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        # Very tight active set
        threshold = c_max * (1.0 - 0.0001)
        active = c >= threshold
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active
        
        g = _grad_weighted(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-16:
            break
        
        # Very careful line search
        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        improved = False
        for ls in range(60):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-17:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve > 50:
                a = best_a.copy()
                lr *= 0.5
                no_improve = 0
            if lr < 1e-10:
                break
        else:
            lr = min(0.003, lr * 1.01)
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.2 + 0.8 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n
