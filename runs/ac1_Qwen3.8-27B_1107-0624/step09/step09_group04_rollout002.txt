"""
Optimized step-function coefficient search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses multi-phase approach: L-BFGS with log-sum-exp smoothing, momentum-based active-set
equalization with adaptive step size, and iterative refinement. Key improvements over
previous: momentum in active-set, adaptive K cycling, better time allocation, and
peaks-weighted gradient with exponential proximity weighting.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0, label=""):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003) * beta_scale

    betas = base_beta * np.array([
        1.0, 0.7, 0.5, 0.35, 0.25,
        0.18, 0.12, 0.08, 0.05, 0.03,
        0.02, 0.012, 0.008, 0.005,
        0.003, 0.002, 0.001, 0.0006,
        0.0003, 0.0001, 0.00005, 0.00002
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:

        prob = _LbfgsProblem(float(beta), n)

def _active_set_equalize(a_init, deadline, min_time_remaining=30.0, use_momentum=True, label=""):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    K_values = [3, 5, 8, 12, 20, 35, 50, 80, 120, 200, 300, 500]
    K_idx = 0
    lr = 0.015
    momentum = 0.0
    velocity = np.zeros(n)
    no_improve = 0
    total_iters = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            if K_idx < len(K_values) - 1 and total_iters % 50 == 0:
                K_idx += 1
        K = K_values[K_idx]

        # Exponential weighting of peaks near the max
        threshold = c_max * (1 - 0.005)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        if len(active) > K * 3:
            # Take top K by value
            top_idx = np.argsort(c)[::-1][:K]
            active = top_idx
        elif len(active) < 2:
            top_idx = np.argsort(c)[::-1][:max(2, K // 10)]
            active = top_idx

        # Weighted gradient: weight peaks by exp((c_k - c_max)/tau)
        tau = max(c_max * 0.001, 1e-10)
        w_active = np.exp((c[active] - c_max) / tau)
        w_active = w_active / w_active.sum()

        w_S = np.zeros(2 * n - 1)
        w_S[active] = w_active
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        if use_momentum:
            momentum = 0.85 if no_improve > 20 else 0.7
            velocity = momentum * velocity - lr * grad / (gnorm + 1e-15)
            step = velocity
        else:
            step = -grad / (gnorm + 1e-15)

        # Line search with backtracking
        alpha = 1.0
        improved = False
        for _ in range(40):
            a_try = a + alpha * step
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        total_iters += 1
        if not improved:
            no_improve += 1
            lr *= 0.95
            if no_improve % 50 == 0:
                # Cycle K
                K_idx = (K_idx + 1) % len(K_values)
            if no_improve >= 300:
                a = best_a.copy()
                velocity = np.zeros(n)
                lr = 0.015
                K_idx = min(K_idx, 5)
                no_improve = 0
                if total_iters > 50000:
                    break
        else:
            lr = min(0.03, lr * 1.005)

    return best_a, best_sc

        if sc_try < current_sc - 1e-15:
            a = a_try
        else:
            no_improve += 1
            lr *= 0.97
            if no_improve > 500:
                break

        it += 1

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.sin(2.0 * np.pi * 5.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 11.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _truncated_cosine(n, alpha=1.0):
    """Create a truncated cosine profile: a[j] = cos(alpha * pi * j / (n-1)) for |j - (n-1)/2| < (n-1)/2."""
    j = np.arange(n)
    center = (n - 1) / 2.0
    t = alpha * np.pi * (j - center) / (n - 1)
    a = np.cos(t)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    t_start = time.monotonic()

            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s

    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")

        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS warmup ===
    print(f"\n[P1] L-BFGS warmup (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 1300.0, time.monotonic() + 200.0)

                best_seq = a_opt.copy()
                best_score = sc_opt

            else:

                best_score = sc_fin2
                print(f"  Final active-set: NEW BEST {best_score:.10f}")

    # === Phase 9: Ultra-polish (if time remains) ===
    if time.monotonic() < deadline - 5:
        print(f"\n[P9] Ultra-polish (t={time.monotonic()-t_start:.1f}s)")
        a_up, sc_up = _peaks_weighted_polish(best_seq, deadline - 4.0, min_time_remaining=4.0)
        if sc_up < best_score:

            best_score = sc_up

    # Final validation

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:

        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:

        except Exception as e:

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
