    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)

def _lse_score(a, beta):
    """Log-sum-exp smoothed score."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a = a / s
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s_e = e.sum()
    if s_e < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s_e))

def _lse_grad_z(z, beta):
    """Gradient of LSE score w.r.t. z (softmax parameterization)."""
    n = len(z)
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s_e = e.sum()
    if s_e < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s_e
    # Gradient of sum_k w_k * q_k(a) w.r.t. a, then chain rule through softmax
    # q_k(a) = sum_j a_j * a_{k-j}
    # d/d a_i [sum_k w_k * q_k(a)] = sum_k w_k * (a_{k-i} + a_{i-k}) [with bounds]
    # = sum_k w_k * 2 * a_{k-i} (when both valid)
    # This is equivalent to: 2 * conv(w, a_rev)[i] where a_rev is reversed a
    # Wait, let me think more carefully.
    # d q_k / d a_i = a_{k-i} + a_{i-k} (when valid)
    # Sum over k: sum_k w_k * a_{k-i} + sum_k w_k * a_{i-k}
    # First term: sum_k w_k * a_{k-i} = (w * a_rev_conv)[i] where this is the cross-correlation
    # Actually, sum_k w_k * a_{k-i}: let j = k-i, then sum_j w_{j+i} * a_j
    # This is like a cross-correlation of w and a.
    # In FFT terms: if we define the cross-correlation as (w ⋆ a)[i] = sum_j w[j] * a[j-i]
    # Then sum_k w_k * a_{k-i} = (w ⋆ a) ... hmm, let me just compute it directly.
    
    # Actually, the simplest way:
    # grad_a[i] = sum_k w_k * (a[k-i] + a[i-k]) [only valid indices]
    # = sum_k w_k * a[k-i] + sum_k w_k * a[i-k]
    # First sum: for fixed i, k ranges such that 0 <= k-i < n, i.e., i <= k < i+n
    #   = sum_{k=i}^{min(i+n-1, 2n-2)} w_k * a[k-i]
    #   Let j = k-i: sum_{j=0}^{min(n-1, 2n-2-i)} w_{j+i} * a_j
    #   This is the cross-correlation (w, a) at lag i, but only valid part.
    #   
    # Second sum: for fixed i, k ranges such that 0 <= i-k < n, i.e., i-n+1 <= k <= i
    #   = sum_{k=max(0,i-n+1)}^{min(i, 2n-2)} w_k * a[i-k]
    #   Let j = i-k: sum_{j=max(0,i-n+1) to min(i,2n-2)} w_{i-j} * a_j
    #   This is the cross-correlation (w, a) at lag -i, but only valid part.
    #
    # Combined: grad_a[i] = (w ⋆ a)[i] + (w ⋆ a)[-i] = (w ⋆ a)[i] + (a ⋆ w)[i]
    # where (w ⋆ a)[i] = sum_j w[j] * a[j+i] (cross-correlation)
    #
    # Actually, by symmetry of cross-correlation: (w ⋆ a)[i] = (a ⋆ w)[-i]
    # So grad_a[i] = 2 * (w ⋆ a)[i] (when the cross-correlation is symmetric, which it's not in general)
    
    # Let me just compute it using FFT:
    # Define: cross(i) = sum_k w[k] * a[k-i] for valid k, k-i
    # This is the "non-standard" cross-correlation.
    # In terms of convolution: cross(i) = (reverse(w) * a)[n-1+i] ... hmm, this is getting complicated.
    #
    # Let me use a different approach:
    # grad_a = 2 * conv(w_padded, a_padded) [middle part]
    # where w_padded and a_padded are appropriately padded.
    
    # Actually, the simplest correct approach:
    # grad_a[i] = sum_k w[k] * (a[k-i] if 0<=k-i<n else 0) + sum_k w[k] * (a[i-k] if 0<=i-k<n else 0)
    # = sum_{j: k=j+i, 0<=j<n} w[j+i] * a[j] + sum_{j: k=i-j, 0<=j<n} w[i-j] * a[j]
    # = sum_j w[j+i] * a[j] + sum_j w[i-j] * a[j]
    #
    # First term: cross-correlation of w and a at lag i (shifted).
    # = (flip(w) conv a)[something]
    #
    # Let me just use the FFT approach:
    # sum_j w[j+i] * a[j] = coefficient of x^0 in W(x+i) * A(x) ... no, this is getting too complicated.
    
    # Let me use a direct FFT-based computation:
    # Define W reversed: w_r[j] = w[(n-1-j) mod (2n-1)] ... no.
    
    # OK let me just do it with the formula:
    # grad_a[i] = sum_j a[j] * (w[j+i] + w[i-j]) [with bounds]
    #
    # w[j+i]: this is w shifted by i. For j=0 to n-1-i: w[j+i]. For j > n-1-i: w[j+i] is out of bounds (since w has length 2n-1, j+i can be up to n-1+n-1 = 2n-2, which is valid).
    # Wait, w has length 2n-1 (indices 0 to 2n-2). j+i ranges from i to n-1+i, which is at most n-1+(n-1) = 2n-2. So it's always valid!
    #
    # w[i-j]: for j=0 to min(i, 2n-2): w[i-j]. For j > i: w[i-j] would be negative index, invalid.
    # So w[i-j] is valid for j = 0 to i, and i-j ranges from i to 0.
    # But also i-j must be in [0, 2n-2], so i-j >= 0 means j <= i, and i-j <= 2n-2 is always true for j >= 0.
    #
    # So: grad_a[i] = sum_{j=0}^{n-1} a[j] * w[j+i] + sum_{j=0}^{min(i, 2n-2)} a[j] * w[i-j]
    #
    # First sum: let's call it S1[i] = sum_{j=0}^{n-1} a[j] * w[j+i]
    # This is the cross-correlation of a and w at lag i: S1 = (a ⋆ w)
    # In FFT: S1 = ifft(fft(a, N) * conj(fft(w, N))) ... no, that's not right either.
    #
    # Actually, the cross-correlation (a ⋆ w)[i] = sum_j a[j] * w[j+i] = sum_j a[j] * conj(w[j+i]) (for real)
    # = ifft(fft(a, N) * conj(fft(w, N)))[i] for appropriate N.
    #
    # Hmm, but w has length 2n-1 and a has length n. Let me pad both to length N >= 3n-2.
    #
    # Second sum: S2[i] = sum_{j=0}^{min(i, 2n-2)} a[j] * w[i-j]
    # This is the convolution (a * w)[i] (only the part where both are valid).
    # = ifft(fft(a, N) * fft(w, N))[i] for appropriate N.
    
    N = _next_pow2(3 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    # S1 = cross-correlation = ifft(fft(a) * conj(fft(w)))
    s1 = np.fft.irfft(fa * np.conj(fw), N)
    # S2 = convolution = ifft(fft(a) * fft(w))
    s2 = np.fft.irfft(fa * fw, N)
    grad_a = s1[:n] + s2[:n]
    
    # Now chain rule through softmax:
    # a = softmax(z), so da/dz[i,j] = a[i]*(delta_ij - a[j])
    # d F / d z[j] = sum_i (dF/da[i]) * (da[i]/dz[j]) = sum_i grad_a[i] * a[i] * (delta_ij - a[j])
    # = grad_a[j] * a[j] - a[j] * sum_i grad_a[i] * a[i]
    # = a[j] * (grad_a[j] - dot(grad_a, a))
    dot_ga_a = float(np.dot(grad_a, a))
    return a * (grad_a - dot_ga_a)

def _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with cooling-reheating beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    current_a = a_init.copy()
    s = current_a.sum()
    if s > 1e-15:
        current_a = current_a / s
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # Beta schedule: cooling-reheating
    # Start high, cool down, reheat, repeat with lower minimum
    c_sorted = np.sort(c_cur)[::-1]
    gap = max_c - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)
    
    # Cooling-reheating schedule
    all_betas = []
    for cycle in range(3):
        scale = 0.5 ** cycle
        # Cool from base_beta * scale to base_beta * scale * 0.01
        n_steps = 8
        cool = base_beta * scale * np.logspace(0, -2, n_steps)
        all_betas.extend(cool)
        # Reheat back
        reheat = base_beta * scale * np.logspace(-2, 0, 3)[::-1]
        all_betas.extend(reheat)
    
    # Add a final fine-tuning phase
    final_betas = base_beta * 0.25 * np.logspace(0, -3, 5)
    all_betas.extend(final_betas)
    
    for i, beta in enumerate(all_betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Estimate time per iteration
        est_time = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 5))))
        
        def obj(z_val):
            a_val = _softmax(z_val)
            return _lse_score(a_val, beta)
        
        def grad(z_val):
            return _lse_grad_z(z_val, beta)
        
        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 8 == 0:
                    print(f"      LSE beta={beta:.6f}: score={current_sc:.10f} (iter {i}/{len(all_betas)})")
        except Exception as ex:
            print(f"      L-BFGS error at beta={beta:.6f}: {ex}")
    
    return current_a
