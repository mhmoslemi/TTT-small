"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. L-BFGS with adaptive log-sum-exp smoothing
2. Newton-based active set peak equalization (key improvement over gradient descent)
3. Adaptive active set management (add/remove peaks dynamically)
4. Subgradient polish
5. Multi-n exploration
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005) * beta_scale

    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625,
        0.03, 0.015, 0.0075, 0.004,
        0.002, 0.001, 0.0005, 0.0002, 0.0001,
        0.00005, 0.00002, 0.00001, 0.000005
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        if beta < base_beta * 0.01:
            est_time_per_iter = 0.004 * max(1.0, n / 2048.0)
            max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 15))))
        else:
            est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
            max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 12))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 80,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _compute_gradients(a, S):
    """Compute gradients of (a*a)_k for k in S. Returns (K, n) matrix."""
    n = len(a)
    K = len(S)
    grads = np.zeros((K, n))
    for i, k in enumerate(S):
        idx = np.arange(n)
        valid = (k - idx >= 0) & (k - idx < n)
        grads[i, valid] = 2.0 * a[k - idx[valid]]
    return grads


def _newton_active_set(a, S, max_iters=50, tol=1e-12):
    """
    Newton iteration to equalize peaks in active set S.
    Solves: (a*a)_{k_i} = t for all k_i in S, sum(a) = 1
    Uses damped Newton with non-negativity enforcement.
    Returns updated a.
    """
    n = len(a)
    K = len(S)
    if K == 0:
        return a
    if K == 1:
        return a  # Nothing to equalize

    a = a.copy()
    a = a / a.sum()

    best_a = a.copy()
    best_range = np.inf

    for it in range(max_iters):
        c = _full_convolve(a, a)
        peak_vals = c[S]
        t = peak_vals[0]
        residuals_peaks = peak_vals - t  # K values, first is 0
        residual_sum = a.sum() - 1.0

        if abs(residual_sum) > 1e-6:
            a = a / a.sum()
            residual_sum = 0.0

        peak_range = peak_vals.max() - peak_vals.min()
        if peak_range < best_range:
            best_range = peak_range
            best_a = a.copy()

        if peak_range < tol * max(t, 1e-15):
            break

        # Build Jacobian components
        grads = _compute_gradients(a, S)  # (K, n)

        # J is (K+1) x (n+1):
        # Rows 0..K-1: [grads[i], -1]
        # Row K: [ones(n), 0]
        #
        # J J^T is (K+1) x (K+1):
        # (J J^T)[i,j] = grads[i].grads[j] + 1  for i,j < K
        # (J J^T)[i,K] = sum(grads[i])  for i < K  (since -1 * 0 + grads[i].1)
        # Wait: J_i = [grads_i, -1], J_K = [1, 0]
        # J_i . J_j = grads_i . grads_j + (-1)(-1) = grads_i.grads_j + 1
        # J_i . J_K = grads_i . 1 + (-1)*0 = sum(grads_i)
        # J_K . J_K = 1 . 1 + 0 = n

        GtG = grads @ grads.T  # (K, K)
        Gt1 = grads.sum(axis=1)  # (K,)

        JJT = np.zeros((K + 1, K + 1))
        JJT[:K, :K] = GtG + 1.0
        JJT[:K, K] = Gt1
        JJT[K, :K] = Gt1
        JJT[K, K] = float(n)

        # Residual vector
        r = np.zeros(K + 1)
        r[:K] = c[S] - t
        r[K] = a.sum() - 1.0

        # Solve JJT * delta = -r
        try:
            delta = np.linalg.solve(JJT, -r)
        except np.linalg.LinAlgError:
            try:
                delta = np.linalg.lstsq(JJT, -r, rcond=None)[0]
            except:
                break

        delta_a = delta[:n]
        a_trial = a + delta_a

        # Damping: find alpha such that a_trial stays non-negative
        # a + alpha * delta_a >= 0 for all j
        # For negative delta_a[j]: alpha <= -a[j] / delta_a[j]
        neg_mask = delta_a < 0
        if neg_mask.any():
            max_alpha = np.min(-a[neg_mask] / delta_a[neg_mask])
            alpha = min(1.0, max_alpha * 0.99)
            alpha = max(alpha, 0.01)  # don't take too small a step
        else:
            alpha = 1.0

        a_new = a + alpha * delta_a
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        # Check improvement
        c_new = _full_convolve(a_new, a_new)
        peak_new = c_new[S]
        range_new = peak_new.max() - peak_new.min()

        if range_new < peak_range * 0.5 or alpha < 0.1:
            a = a_new
        else:
            a = a_new  # Accept anyway, Newton should converge

    return best_a if best_range < peak_range else a


def _adaptive_active_set_optimize(a_init, deadline, min_time_remaining=30.0):
    """
    Optimally identify and equalize active peaks using Newton's method.
    Adaptively grows/shrinks the active set.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    no_improve = 0
    K_min = 3
    K_max = min(60, n)

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            break

        # Identify active set: peaks within adaptive tolerance of max
        # Start with a small tolerance, expand if Newton doesn't converge
        for rel_tol in [1e-8, 1e-6, 1e-4, 1e-3, 0.01, 0.05]:
            threshold = c_max * (1.0 - rel_tol)
            S = np.where(c >= threshold)[0]
            if len(S) <= K_max:
                break
            S = np.argsort(c)[::-1][:K_max]

        if len(S) < K_min:
            S = np.argsort(c)[::-1][:K_min]

        # Apply Newton
        a_new = _newton_active_set(a, S, max_iters=30, tol=1e-13)

        # Verify
        c_check = _full_convolve(a_new, a_new)
        sc_check = 2.0 * n * float(c_check.max())

        if sc_check < current_sc - 1e-15:
            a = a_new
        else:
            # Try with smaller active set
            top_half = S[:max(K_min, len(S) // 2)]
            a_new2 = _newton_active_set(a, top_half, max_iters=20, tol=1e-13)
            c_check2 = _full_convolve(a_new2, a_new2)
            sc_check2 = 2.0 * n * float(c_check2.max())
            if sc_check2 < current_sc - 1e-15:

            else:

                a = a + 1e-5 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()

    return best_a, best_sc


def _subgradient_polish(a_init, deadline, min_time_remaining=10.0):
    """Final polish using subgradient method with careful step size."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    max_iters = 8000
    eta = 0.05

    for it in range(max_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break

        if it % 100 == 0:
            c = _full_convolve(a, a)
            c_max = float(c.max())
            current_sc = 2.0 * n * c_max
            if current_sc < best_sc - 1e-15:
                best_sc = current_sc
                best_a = a.copy()

            # Adaptive step size
            eta = max(0.001, 0.05 / (1.0 + it * 0.01))

        # Find active peaks
        c = _full_convolve(a, a)
        c_max = float(c.max())
        threshold = c_max * (1 - 0.001)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        # Subgradient
        w_S = np.zeros(2 * n - 1)
        w_S[active] = 1.0 / len(active)
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = eta / (gnorm + 1e-15)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new

    # Final check
    c = _full_convolve(a, a)
    sc = 2.0 * n * float(c.max())
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc

    return best_a, best_sc

def _make_profile(n, kind="uniform"):

        return np.ones(n) / n
    elif kind == "tapered":

        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)

    elif kind == "asymmetric_taper":

        return a / a.sum()
    elif kind == "bimodal_asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.3) ** 2) + 0.7 * np.exp(-12.0 * (t - 0.7) ** 2) + 0.05

    elif kind == "sine_mod":

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _newton_with_reprojection(a, S, max_iters=100):
    """
    Newton method with reprojection onto simplex.
    More robust than _newton_active_set for cases where Newton step
    would violate non-negativity severely.
    """
    n = len(a)
    K = len(S)
    if K < 2:
        return a

    a = a.copy() / a.sum()
    best_a = a.copy()
    best_range = np.inf

    for it in range(max_iters):
        c = _full_convolve(a, a)
        peaks = c[S]
        t = peaks.mean()
        range_current = peaks.max() - peaks.min()

        if range_current < best_range:
            best_range = range_current
            best_a = a.copy()

        if range_current < 1e-14:
            break

        # Gradient of each peak
        grads = _compute_gradients(a, S)

        # We want to find delta_a such that:
        #   sum_j delta_a[j] * grads[i,j] = -(c[S[i]] - t)  for each i
        #   sum_j delta_a[j] = 0
        # Minimum norm solution.

        # Build the (K+1) x (n) matrix: [grads; ones]
        # Solve: M @ delta_a = -[c[S] - t; 0]
        # Using least squares: delta_a = -M^T (M M^T)^{-1} [c[S]-t; 0]

        M = np.vstack([grads, np.ones((1, n))])  # (K+1, n)
        MMT = M @ M.T  # (K+1, K+1)
        rhs = np.zeros(K + 1)
        rhs[:K] = c[S] - t
        # rhs[K] = 0 (sum constraint already satisfied)

        try:
            sol = np.linalg.solve(MMT, rhs)
        except:
            sol = np.linalg.lstsq(MMT, rhs, rcond=None)[0]

        delta_a = -(M.T @ sol)

        # Damped step
        norm_db = np.linalg.norm(delta_a)
        if norm_db < 1e-16:
            break

        # Find max alpha for non-negativity
        neg = delta_a < 0
        if neg.any():
            max_alpha = np.min(-a[neg] / delta_a[neg]) * 0.95
            alpha = min(1.0, max(1e-6, max_alpha))
        else:
            alpha = min(1.0, 0.1 / (norm_db / max(1e-15, a.sum())))
            alpha = min(alpha, 1.0)

        a_new = a + alpha * delta_a
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS smoothing, Newton active-set
    equalization, subgradient polish, and multi-n exploration.
    Key improvement: Newton-based active set equalization for fast convergence.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS warmup ===
    print(f"\n[P1] L-BFGS warmup (t={time.monotonic()-t_start:.1f}s)")
    a_p1 = _run_lbfgs(parent, deadline, min_time_remaining=900.0, beta_scale=1.0)
    sc_p1 = _score(a_p1)
    if sc_p1 < best_score:
        best_seq = a_p1.copy()
        best_score = sc_p1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_warmup]")
    else:
        print(f"  No improvement (best={best_score:.10f})")

    # === Phase 2: Newton active-set equalization (KEY IMPROVEMENT) ===
    print(f"\n[P2] Newton active-set equalization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        pol_deadline = min(deadline - 500, time.monotonic() + 300)
        a_eq, sc_eq = _adaptive_active_set_optimize(best_seq, pol_deadline, min_time_remaining=100.0)
        if sc_eq < best_score:
            best_seq = a_eq.copy()
            best_score = sc_eq
            print(f"  NEW BEST: {best_score:.10f} [newton_as]")
        else:
            print(f"  No improvement (best={best_score:.10f})")

    # === Phase 3: Second L-BFGS pass to clean up ===
    print(f"\n[P3] L-BFGS cleanup (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a_p3 = _run_lbfgs(best_seq, deadline, min_time_remaining=400.0, beta_scale=0.5)
        sc_p3 = _score(a_p3)
        if sc_p3 < best_score:
            best_seq = a_p3.copy()
            best_score = sc_p3
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_cleanup]")

    # === Phase 4: Another Newton round ===
    print(f"\n[P4] Newton round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        pol_deadline2 = min(deadline - 300, time.monotonic() + 200)
        a_eq2, sc_eq2 = _adaptive_active_set_optimize(best_seq, pol_deadline2, min_time_remaining=100.0)
        if sc_eq2 < best_score:
            best_seq = a_eq2.copy()
            best_score = sc_eq2
            print(f"  NEW BEST: {best_score:.10f} [newton_as2]")

    # === Phase 5: Asymmetric perturbation + re-optimize ===
    print(f"\n[P5] Asymmetric exploration (t={time.monotonic()-t_start:.1f}s)")
    n = len(best_seq)
    for trial in range(3):
        if time.monotonic() > deadline - 250:
            break
        strength = 0.02 * (trial + 1)
        a_asym = _asymmetric_perturb(best_seq, rng, strength=strength)
        a_opt = _run_lbfgs(a_asym, deadline, min_time_remaining=200.0, beta_scale=0.5)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (trial={trial}) [asym]")
        # Quick Newton on the asymmetric result
        if time.monotonic() < deadline - 100:
            c_tmp = _full_convolve(a_opt, a_opt)
            S_tmp = np.argsort(c_tmp)[::-1][:5]
            a_newt = _newton_active_set(a_opt, S_tmp, max_iters=20)
            sc_newt = _score(a_newt)
            if sc_newt < best_score:
                best_seq = a_newt.copy()
                best_score = sc_newt
                print(f"  NEW BEST: {best_score:.10f} (trial={trial}) [asym+newton]")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([n, 32768, 16384, 24576]), reverse=True)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=60.0, beta_scale=0.5)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n]")
            # Quick Newton on multi-n result
            if time.monotonic() < deadline - 50:
                c_tmp = _full_convolve(a_opt, a_opt)
                S_tmp = np.argsort(c_tmp)[::-1][:5]
                a_newt = _newton_active_set(a_opt, S_tmp, max_iters=20)
                sc_newt = _score(a_newt)
                if sc_newt < best_score:
                    best_seq = a_newt.copy()
                    best_score = sc_newt
                    print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n+newton]")

    # === Phase 7: Subgradient polish ===
    print(f"\n[P7] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        sg_deadline = min(deadline - 20, time.monotonic() + 100)
        a_sg, sc_sg = _subgradient_polish(best_seq, sg_deadline, min_time_remaining=15.0)
        if sc_sg < best_score:
            best_seq = a_sg.copy()
            best_score = sc_sg
            print(f"  NEW BEST: {best_score:.10f} [subgrad]")

    # === Phase 8: Final Newton polish ===
    print(f"\n[P8] Final Newton polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 10 - time.monotonic()
    if remaining > 30:
        c_final = _full_convolve(best_seq, best_seq)
        S_final = np.argsort(c_final)[::-1][:min(10, len(c_final))]
        a_fin = _newton_active_set(best_seq, S_final, max_iters=50, tol=1e-14)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_newton]")

        # One more Newton with wider active set
        if time.monotonic() < deadline - 5:
            c_final2 = _full_convolve(best_seq, best_seq)
            c_max2 = c_final2.max()
            S_wide = np.where(c_final2 >= c_max2 * 0.999)[0]
            if len(S_wide) > 2:
                a_fin2 = _newton_active_set(best_seq, S_wide[:min(20, len(S_wide))], max_iters=30)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_newton_wide]")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
