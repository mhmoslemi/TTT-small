```python
def run_active_set_lp(f, K=20, max_iter=100, step_sizes=[0.1, 0.05, 0.01]):
    n = len(f)
    c = _full_convolve(f, f)
    M = c.max()
    f_current = f.copy()
    
    for iteration in range(max_iter):
        # Find active set: top K lags
        top_k = np.argsort(c)[::-1][:K]
        
        # Build constraint matrix A (K × n)
        A = np.zeros((K, n))
        b_constraints = np.zeros(K)
        for idx, k in enumerate(top_k):
            for j in range(n):
                ki = k - j
                if 0 <= ki < n:
                    A[idx, j] = f_current[ki]
            b_constraints[idx] = M  # (f*g)(k) ≤ M
        
        # Solve LP: max sum(g) s.t. A@g ≤ b, g ≥ 0
        c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
        res = linprog(c_obj, A_ub=A, b_ub=b_constraints, bounds=[(0, None)]*n, method='highs')
        
        if res.status != 0:
            break
        
        g = res.x
        g = g / g.sum()  # normalize
        
        # Take a step
        for t in step_sizes:
            f_try = (1-t)*f_current + t*g
            c_try = _full_convolve(f_try, f_try)
            if c_try.max() < M:
                f_current = f_try
                c = c_try
                M = c.max()
                break
    
    return f_current
```

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _grad_p(a, p, n):
    """Compute H_p * a = 2 * sum_k p[k] * H_k * a efficiently via FFT.
    Result[m] = 2 * sum_j a[j] * p[m+j] for valid indices.
    """
    b = a[::-1]
    conv_len = len(b) + len(p) - 1
    N = _next_pow2(conv_len)
    Fb = np.fft.rfft(b, N)
    Fp = np.fft.rfft(p, N)
    conv = np.fft.irfft(Fb * Fp, N)[:conv_len]
    return 2.0 * conv[n - 1:2 * n - 1]

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _run_lbfgs(a_init, deadline, min_time_remaining=15.0):
    """L-BFGS in log-space with adaptive beta continuation for log-sum-exp smoothed max."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap * 0.15, max_c * 0.005)

    ratios = [1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002,
              0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001,
              0.000005, 0.000002, 0.000001, 0.0000005, 0.0000002,
              0.0000001, 0.00000005, 0.00000002]

    for i, ratio in enumerate(ratios):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        beta = base_beta * ratio
        est_time = 0.004 * max(1.0, n / 32768.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 3))))

        def _obj(z_local):
            a_local = _softmax(z_local)
            c_local = _full_convolve(a_local, a_local)
            c_max_local = float(c_local.max())
            shifted = (c_local - c_max_local) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s_e = e.sum()
            if s_e < 1e-300:
                return 2.0 * n * c_max_local
            return 2.0 * n * (c_max_local + beta * np.log(s_e))

        def _jac(z_local):
            a_local = _softmax(z_local)
            c_local = _full_convolve(a_local, a_local)
            c_max_local = float(c_local.max())
            shifted = (c_local - c_max_local) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s_e = e.sum()
            if s_e < 1e-300:
                w = np.zeros_like(c_local)
                w[int(np.argmax(c_local))] = 1.0
            else:
                w = e / s_e
            a_rev = a_local[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W_val = float(np.dot(w, c_local))
            return 4.0 * n * a_local * (h - W_val)

        z = np.log(np.maximum(a, 1e-30))
        z = z - z.mean()

        try:
            result = minimize(
                _obj, z,
                jac=_jac,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-19, 'gtol': 1e-15,
                         'maxcor': 60, 'maxls': 120}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-16:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new
        except Exception:
            pass

    return best_a


def _run_active_set_lp(a_init, deadline, min_time_remaining=30.0, max_rounds=200):
    """Active-set LP: find direction g minimizing max cross-correlation with current f."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() + min_time_remaining > deadline:
            break
        t0 = time.monotonic()

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set: lags within adaptive tolerance of max
        tol_frac = 0.001 * (1.0 + 0.1 * rnd)
        active = c >= (c_max - tol_frac * c_max)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        K = int(active.sum())
        active_lags = np.where(active)[0]

        # Build LP: min M' s.t. (f*g)(k) <= M' for active k, sum(g)=1, g>=0
        # Variables: g[0..n-1], M'  => total n+1 variables
        # Constraints: for each active lag k: sum_j f[k-j]*g[j] - M' <= 0
        A_rows = []
        for k in active_lags:
            row = np.zeros(n + 1)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                row[j] = a[k - j]
            row[n] = -1.0  # -M'
            A_rows.append(row)
        A = np.array(A_rows)
        b = np.zeros(len(A_rows))

        # Equality: sum g = 1
        A_eq_row = np.zeros(n + 1)
        A_eq_row[:n] = 1.0
        A_eq = A_eq_row.reshape(1, -1)
        b_eq = np.array([1.0])

        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0  # minimize M'

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                          bounds=[(0, None)] * n + [(None, None)],
                          method='highs', options={'time_limit': 10})
            if res.status != 0:
                continue
            g = res.x[:n]
            g = g / max(g.sum(), 1e-30)
        except Exception:
            continue

        # Line search for step size
        best_t = 0.0
        best_c_try = c_max
        for t in [0.3, 0.2, 0.1, 0.05, 0.03, 0.01, 0.5, 0.7]:
            a_try = (1 - t) * a + t * g
            c_try = _full_convolve(a_try, a_try)
            c_try_max = float(c_try.max())
            if c_try_max < best_c_try:
                best_c_try = c_try_max
                best_t = t

        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
            sc = _score(a)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a.copy()

        # Periodic check
        if (rnd + 1) % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ActiveSet LP: round {rnd}, score={best_sc:.10f}, "
                  f"K={K}, time/round={elapsed:.2f}s")

    return best_a, best_sc


def _run_grad_descent(a_init, deadline, min_time_remaining=15.0):

    for step in range(10000):

        else:
            lr = 0.03 * (1.0 + 0.05 * step) / (mnorm / gnorm + 1e-10)

        z_new = z - lr * momentum
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()
        z = z_new

        if step % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)

def _coarse_to_fine(a_init, n_final, deadline, min_time_remaining=60.0):
    """Multi-resolution: optimize at coarser levels then refine."""
    scales = [4, 2, 1]
    current = a_init.copy()
    for scale in scales:
        n_sub = max(64, n_final // scale)
        if n_sub >= n_final:
            a_sub = current.copy()
        else:
            a_sub = _resample(current, n_sub)
        a_opt = _run_lbfgs(a_sub, deadline, max(5.0, min_time_remaining / 3))
        if n_sub < n_final:
            current = _resample(a_opt, n_final)
        else:
            current = a_opt
        sc = _score(current)
        print(f"      C2F scale={scale}: n={n_sub}, score={sc:.10f}")
    return current

        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                _update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 32768
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        _update_best(parent, _score(parent), "init")

    # ============ Phase 1: Multi-start L-BFGS (350s) ============
    print(f"\n[Phase 1] Multi-start L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    phase1_deadline = t_start + 350

    # Primary: optimize parent
    a1 = _run_lbfgs(parent, phase1_deadline, min_time_remaining=30.0)
    _update_best(a1, _score(a1), "lbfgs_parent")

    # Secondary starts
    if time.monotonic() < phase1_deadline - 80:
        for prof in ["cos", "power", "ramp"]:
            if time.monotonic() > phase1_deadline - 40:
                break
            a_init = _make_profile(n_parent, prof)
            a_p = _run_lbfgs(a_init, phase1_deadline, min_time_remaining=20.0)
            _update_best(a_p, _score(a_p), f"lbfgs_{prof}")

    # Perturbation of best
    if time.monotonic() < phase1_deadline - 30 and best_seq is not None:
        a_pert = _perturb(best_seq, 0.02, rng)
        a_p = _run_lbfgs(a_pert, phase1_deadline, min_time_remaining=10.0)
        _update_best(a_p, _score(a_p), "lbfgs_perturb")

    # ============ Phase 2: Active-set LP (350s) ============
    print(f"\n[Phase 2] Active-Set LP (t={time.monotonic() - t_start:.1f}s)")
    phase2_deadline = min(deadline - 700, phase1_deadline + 350)
    if best_seq is not None and time.monotonic() < phase2_deadline:
        a_lp, sc_lp = _run_active_set_lp(best_seq, phase2_deadline,
                                          min_time_remaining=30.0, max_rounds=150)
        _update_best(a_lp, sc_lp, "active_lp_1")

    # ============ Phase 3: L-BFGS polish (250s) ============
    print(f"\n[Phase 3] L-BFGS Polish (t={time.monotonic() - t_start:.1f}s)")
    phase3_deadline = min(deadline - 450, phase2_deadline + 250)
    if best_seq is not None and time.monotonic() < phase3_deadline:
        a_lb2 = _run_lbfgs(best_seq, phase3_deadline, min_time_remaining=20.0)
        _update_best(a_lb2, _score(a_lb2), "lbfgs_polish_1")

    # ============ Phase 4: Gradient descent (150s) ============
    print(f"\n[Phase 4] Gradient Descent (t={time.monotonic() - t_start:.1f}s)")
    phase4_deadline = min(deadline - 300, phase3_deadline + 150)
    if best_seq is not None and time.monotonic() < phase4_deadline:
        a_pg, sc_pg = _run_grad_descent(best_seq, phase4_deadline,
                                         min_time_remaining=15.0)
        _update_best(a_pg, sc_pg, "grad_desc_1")

    # ============ Phase 5: Second active-set LP round (200s) ============
    print(f"\n[Phase 5] Active-Set LP Round 2 (t={time.monotonic() - t_start:.1f}s)")
    phase5_deadline = min(deadline - 150, phase4_deadline + 200)
    if best_seq is not None and time.monotonic() < phase5_deadline:
        a_lp2, sc_lp2 = _run_active_set_lp(best_seq, phase5_deadline,
                                             min_time_remaining=20.0, max_rounds=100)
        _update_best(a_lp2, sc_lp2, "active_lp_2")

    # ============ Phase 6: Multi-n exploration (200s) ============
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    phase6_deadline = min(deadline - 100, phase5_deadline + 200)
    n_tries = [16384, 49152, 65536]
    for n_try in n_tries:
        if time.monotonic() > phase6_deadline - 30:
            break
        if abs(n_try - n_parent) < n_parent // 4:
            continue
        a_init_n = _resample(best_seq, n_try) if best_seq is not None else _make_profile(n_try, "tapered")
        a_n = _run_lbfgs(a_init_n, phase6_deadline, min_time_remaining=20.0)
        sc_n = _score(a_n)
        _update_best(a_n, sc_n, f"n_{n_try}")

    # ============ Phase 7: Coarse-to-fine (150s) ============
    print(f"\n[Phase 7] Coarse-to-Fine (t={time.monotonic() - t_start:.1f}s)")
    phase7_deadline = min(deadline - 50, phase6_deadline + 150)
    if best_seq is not None and time.monotonic() < phase7_deadline:
        n_best = len(best_seq)
        a_c2f = _coarse_to_fine(best_seq, n_best, phase7_deadline, min_time_remaining=20.0)
        _update_best(a_c2f, _score(a_c2f), "coarse_to_fine")

    # ============ Phase 8: Final combined polish (50s) ============
    print(f"\n[Phase 8] Final Polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        # Very tight L-BFGS
        a_fin = _run_lbfgs(best_seq, deadline - 5, min_time_remaining=3.0)
        _update_best(a_fin, _score(a_fin), "final_lbfgs")

        # Quick gradient step
        if time.monotonic() < deadline - 5:
            a_fd, sc_fd = _run_grad_descent(best_seq, deadline - 3, min_time_remaining=2.0)
            _update_best(a_fd, sc_fd, "final_grad")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    sc_fb = _score(fallback)
    print(f"FALLBACK: score={sc_fb:.10f}, n={len(fallback)}")
    return [float(x) for x in fallback]
