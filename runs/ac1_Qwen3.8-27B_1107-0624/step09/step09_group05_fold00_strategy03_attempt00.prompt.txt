def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax(z):

def _soft_max_value(c, beta):

    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))

def _active_set_refine(a_init, deadline, min_time_remaining=30.0):
    """KKT-inspired active-set refinement.
    
    At the optimum, for active peaks S and support T of a:
    mean_{k in S} a[k-j] = const for all j in T
    
    We iteratively:
    1. Identify active set S (peaks near max)
    2. Compute the "target" gradient from S
    3. Adjust a to equalize the gradient on the support
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    no_improve = 0
    max_no_improve = 300
    tol_frac = 0.005
    min_tol = 0.0001

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.5)
            if tol_frac <= min_tol * 1.1:
                break
            no_improve = 0

        # Identify active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Compute average gradient from active set
        # grad[j] = 2/n_active * sum_{k in active} a[k-j] (for valid k-j)
        # = 2/n_active * conv(w_active, a_rev)[n-1+j]
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]

        # Identify support of a
        support = a > 1e-8 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        # KKT condition: grad_avg should be constant on support
        # Target: make grad_avg[j] = lambda for j in support
        # lambda = mean of grad_avg over support
        lam = float(np.mean(grad_avg[support]))
        # Residual: how much each support element deviates
        residual = grad_avg[support] - lam

        # Update: move a in direction that reduces residual
        # da[j] = -eta * (grad_avg[j] - lam) for j in support
        # This is like a projected gradient step toward KKT satisfaction
        eta = 0.05
        a_new = a.copy()
        a_new[support] -= eta * residual
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Line search
        sc_new = _score(a_new)
        if sc_new >= current_sc:
            # Try smaller step
            found = False
            for trial_eta in [0.02, 0.01, 0.005, 0.002]:
                a_try = a.copy()
                a_try[support] -= trial_eta * residual
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-12:
                    a_new = a_try
                    sc_new = sc_try
                    found = True
                    break
            if not found:
                # No improvement, tighten tolerance
                tol_frac = max(min_tol, tol_frac * 0.8)
                continue

        a = a_new

        # Gradually tighten
        if no_improve == 0:
            tol_frac = max(min_tol, tol_frac * 0.97)

    return best_a, best_sc


def _subgradient_method(a_init, deadline, min_time_remaining=30.0):
    """Pure subgradient method with Polyak step size."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Lower bound: 2n/(2n-1)
    lb = 2.0 * n / (2 * n - 1)
    sc_best = best_sc

    t = 0
    step_scale = 0.5

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-12:
            best_sc = current_sc
            best_a = a.copy()
            sc_best = current_sc

        # Active set
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Gradient (average over active set)
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        # Polyak step size
        gap = current_sc - lb
        if gap < 1e-10:
            break
        g_norm_sq = float(np.dot(grad - np.mean(grad), grad - np.mean(grad)))
        if g_norm_sq < 1e-20:
            break
        step = step_scale * gap / g_norm_sq
        step = min(step, 0.1)

        grad_c = grad - np.mean(grad)
        a_new = a - step * grad_c
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new
        t += 1

        # Decay step scale
        if t % 100 == 0:
            step_scale *= 0.95

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "one_sided":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t) + 0.005
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.6 * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _coordinate_descent_pass(a, n_coord=50):
    """One pass of coordinate descent: optimize individual elements.
    
    For a[j], the relevant convolution values are c[k] for k in [j, j+n-1].
    The dependency is: c[k] includes a[j]*a[k-j] and a[k-j]*a[j] (if distinct).
    """
    n = len(a)
    a = a.copy()
    s = a.sum()
    a = a / s

    # Random order
    order = np.random.default_rng(999).permutation(n)
    order = order[:n]  # all elements

    for idx in order:
        if idx is None:
            continue
        j = int(idx)
        # Find which c[k] values depend on a[j]
        # c[k] depends on a[j] when k-j is in [0, n-1], i.e., k in [j, j+n-1]
        # Also when k-j < 0 but j is in valid range... actually for full conv:
        # c[k] = sum_i a[i]*a[k-i], a[j] appears when i=j (needs k-j in [0,n-1])
        # or when k-i=j, i.e., i=k-j (needs k-j in [0,n-1], same condition)
        # So k ranges from j to j+n-1, clipped to [0, 2n-2]
        k_lo = max(0, j)
        k_hi = min(2 * n - 2, j + n - 1)

        if k_lo > k_hi:
            continue

        # For each k in [k_lo, k_hi], the contribution of a[j] to c[k] is:
        # a[j]*a[k-j] if k-j != j (i.e., k != 2j)
        # a[j]^2 if k == 2j
        # Plus a[k-j]*a[j] if k-j != j (same as above, double counted)
        # Total: 2*a[j]*a[k-j] if k != 2j, or a[j]^2 if k == 2j

        # Let's compute the base (without a[j]) and the coefficient of a[j]
        aj = a[j]
        # For the objective max_k c[k], we want to reduce the max
        # Simple approach: try setting a[j] to 0 and see if max decreases
        a[j] = 0.0
        s_temp = a.sum()
        if s_temp < 1e-10:
            a[j] = aj
            continue
        a_norm = a / s_temp
        c_temp = _full_convolve(a_norm, a_norm)
        max_temp = float(c_temp.max())
        score_temp = 2.0 * n * max_temp

        a[j] = aj
        s = a.sum()
        a_norm_orig = a / s
        c_orig = _full_convolve(a_norm_orig, a_norm_orig)
        max_orig = float(c_orig.max())
        score_orig = 2.0 * n * max_orig

        if score_temp < score_orig - 1e-12:
            # Setting to zero helps
            a[j] = 0.0
        else:
            # Try a small reduction
            a[j] = aj * 0.9
            s = a.sum()
            a_try = a / s
            c_try = _full_convolve(a_try, a_try)
            score_try = 2.0 * n * float(c_try.max())
            if score_try < score_orig - 1e-12:
                a = a_try
            else:
                a[j] = aj

    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a
