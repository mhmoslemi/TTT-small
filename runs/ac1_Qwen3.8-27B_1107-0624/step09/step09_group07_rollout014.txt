"""
Multi-scale optimizer with adaptive smoothing and Frank-Wolfe hybrid.
Improvements over previous:
- Adaptive K/beta based on real-time peak structure analysis
- Frank-Wolfe steps for global structure correction
- Multi-scale (coarse-to-fine) refinement

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]

def _adaptive_smooth_obj_grad(z, beta, K, n, use_power_mean=False, p=100):
    """Compute objective and gradient with adaptive smoothing."""
    z_max = z.max()
    a = np.exp(z - z_max)
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if use_power_mean:
        # Power mean smoothing
        c_pos = np.maximum(c, 1e-30)
        c_p = c_pos ** p
        s_p = c_p.sum()
        if s_p < 1e-300 or not np.isfinite(s_p):
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
            obj = 2.0 * n * c_max
        else:
            M_p = (s_p / (2 * n - 1)) ** (1.0 / p)
            w = (c_pos ** (p - 1)) / s_p
            obj = 2.0 * n * M_p
    else:
        # Top-K log-sum-exp
        if K < len(c):
            top_idx = np.argpartition(c, -K)[-K:]
            c_top = c[top_idx]
        else:
            top_idx = np.arange(len(c))
            c_top = c
        
        shifted = (c_top - c_max) / beta
        shifted = np.clip(shifted, -700, 0)
        e = np.exp(shifted)
        s_e = e.sum()
        
        if s_e < 1e-300 or not np.isfinite(s_e):
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
            obj = 2.0 * n * c_max
        else:
            e_full = np.zeros(2 * n - 1)
            e_full[top_idx] = e
            w = e_full / s_e
            obj = 2.0 * n * (c_max + beta * np.log(s_e))
    
    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z

class _SmoProb:
    def __init__(self, beta, K, n, use_power_mean=False, p=100):
        self.beta = beta
        self.K = K
        self.n = n
        self.use_power_mean = use_power_mean
        self.p = p
    
    def __call__(self, z):
        return _adaptive_smooth_obj_grad(z, self.beta, self.K, self.n, self.use_power_mean, self.p)[0]
    
    def grad(self, z):
        return _adaptive_smooth_obj_grad(z, self.beta, self.K, self.n, self.use_power_mean, self.p)[1]

def _adaptive_k_beta(c, n):
    """Determine K and beta adaptively based on peak structure."""
    c_max = c.max()
    # Find peaks within a certain fraction of max
    threshold = c_max * 0.99
    n_active = int((c >= threshold).sum())
    n_active = max(5, min(n_active, int(0.05 * (2*n-1))))
    
    # Beta based on gap
    if n_active < len(c):
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[n_active-1] - c_sorted[min(n_active+10, len(c)-1)]
    else:
        gap = c_max * 0.01
    
    beta = max(gap * 0.5, c_max * 0.002)
    return n_active, beta

def _frank_wolfe(a_init, deadline, min_time_remaining=30.0):

    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    iteration = 0
    gamma = 1.0  # Momentum parameter
    
    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
        
        # Subgradient: gradient of c[k*] where k* = argmax
        k_star = int(np.argmax(c))
        e_k = np.zeros(2 * n - 1)
        e_k[k_star] = 1.0
        g = _grad_maxconv(a, e_k)
        g_centered = g - g.mean()
        
        # Frank-Wolfe step: move towards e_j where j = argmin g
        j = int(np.argmin(g_centered))
        v = np.zeros(n)
        v[j] = 1.0
        v_centered = v - 1.0/n  # Centered for simplex
        
        # Step size: 2/(gamma * (iteration + 2)) or line search
        t_fw = 2.0 / (gamma * (iteration + 1))
        
        # Line search
        a_try = (1 - t_fw) * a + t_fw * v
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try > 1e-10:
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc:
                a = a_try
        
        # Also try gradient step
        alpha = 1.0 / (iteration + 1)
        a_grad = a - alpha * g_centered / (np.linalg.norm(g_centered) + 1e-15)
        a_grad = np.maximum(a_grad, 0)
        s_grad = a_grad.sum()
        if s_grad > 1e-10:
            a_grad = a_grad / s_grad
            c_grad = _full_convolve(a_grad, a_grad)
            sc_grad = 2.0 * n * float(c_grad.max())
            if sc_grad < 2.0 * n * float(c.max()):
                a = a_grad
        
        if iteration % 1000 == 0:
            print(f"    FW iter {iteration}: sc={best_sc:.10f}")
    
    return best_a, best_sc

def _peak_equalize(a_init, deadline, min_time_remaining=20.0):
    """Directly equalize peaks by solving a local problem."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    patience = 0
    max_patience = 500
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        
        # Find top peaks
        n_peaks = max(3, int(0.01 * (2*n-1)))
        top_idx = np.argsort(c)[::-1][:n_peaks * 2]
        top_idx = top_idx[c[top_idx] > c_max * 0.95]
        if len(top_idx) == 0:
            top_idx = [int(np.argmax(c))]
        
        # For each top peak, find the top contributors
        # and try to reduce them while adding to low areas
        delta = np.zeros(n)
        for k in top_idx:
            i_lo = max(0, k - (n-1))
            i_hi = min(n-1, k)
            contrib = np.zeros(n)
            for i in range(i_lo, i_hi + 1):
                j = k - i
                contrib[i] += a[j]
                if i != j:
                    contrib[j] += a[i]
            
            # Reduce top 5 contributors
            n_red = max(2, int(n * 0.001))
            top_c = np.argsort(contrib)[::-1][:n_red]
            for idx in top_c:
                if contrib[idx] > 1e-15:
                    delta[idx] -= 0.01 / (contrib[idx] * n * 10)
            
            # Add to bottom contributors
            bot_c = np.argsort(contrib)[:n_red]
            for idx in bot_c:
                delta[idx] += 0.005 / (n_red * (a[idx] + 1e-10))
        
        delta -= delta.mean()
        
        # Line search
        alpha = 1.0
        improved = False
        for ls in range(20):
            a_try = a + alpha * delta
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-16:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            patience += 5
        else:
            patience = 0
    
    return best_a, best_sc

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t) + 0.1
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.35) ** 2) + np.exp(-15.0 * (t - 0.65) ** 2) + 0.03
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        taper = int(n * 0.1)
        for i in range(taper):
            f = i / taper
            a[i] = f
            a[n-1-i] = f
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    """Fine active-set polish with adaptive tolerance."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.003
    tol = 0.01
    no_improve = 0
    max_no_improve = 500
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-16:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        # Active set: peaks within tol of max
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        
        # Weighted by excess
        w = np.zeros(2 * n - 1)
        excess = c[active] - (c_max - tol * c_max)
        max_excess = excess.max()
        if max_excess > 1e-18:
            w[active] = excess / max_excess + 0.1
        else:
            w[active] = 1.0 / n_active
        w = w / w.sum()
        
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break
        
        alpha = min(lr, 0.3 / (gnorm + 1e-15))
        improved = False
        for ls in range(30):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-16:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve % 50 == 0:
                tol = max(0.0005, tol * 0.7)
                lr *= 0.8
            if no_improve > max_no_improve:
                a = best_a.copy()
                tol = max(0.0005, tol * 0.5)
                lr = 0.002
                no_improve = 0
        else:
            if tol > 0.0005:
                tol = max(0.0005, tol * 0.98)
            lr = min(0.008, lr * 1.005)
    
    return best_a, best_sc
