- b_k = b[k] for k in K
- b_max = max_b = b[k_0] (where k_0 is the argmax)

We want: a = a_0 - (1/2) * (G λ + μ * 1)

where G is the n × |K| matrix with columns g_k, and λ is the vector of Lagrange multipliers.

The constraint b[k] = b_max for all k in K:
b[k](a) = b[k_0] for all k in K

Since b[k] is quadratic: b[k](a) = a^T M_k a where M_k is the matrix for lag k.

This is getting complicated because b[k] is quadratic, not linear. The linearization approach would be:
b[k](a_0 + δ) ≈ b[k](a_0) + g_k^T δ

Setting this equal to b_max:
b[k](a_0) + g_k^T δ = b_max
g_k^T δ = b_max - b[k](a_0)

And δ = -(1/2) * (G λ + μ * 1)

So: g_k^T (-(1/2) * (G λ + μ * 1)) = b_max - b[k](a_0)
-(1/2) * (g_k^T G λ + μ * g_k^T 1) = b_max - b[k](a_0)

This gives us the linear system:
[ -1/2 * G^T G   -1/2 * G^T 1 ] [λ]   [b_max - b_K(a_0)]
[ -1/2 * 1^T G    -1/2 * n     ] [μ] = [    1               ]

where the last equation is sum(δ) = 0 (to maintain sum a = 1), i.e., 1^T δ = 0.

Wait, sum(δ) = -(1/2) * (1^T G λ + μ * n) = 0

So the system is:
G^T G * λ + G^T 1 * μ = 2 * (b_K(a_0) - b_max)
1^T G * λ + n * μ = 0

This is a (|K|+1) × (|K|+1) system that we can solve directly!

After finding λ and μ, we compute:
δ = -(1/2) * (G λ + μ * 1)
a_new = a_0 + δ

Then we check: is a_new >= 0? Are b[k] <= b_max for k not in K? If not, we need to adjust.

This is essentially a "linearized" Newton step that makes all active peaks equal (to first order). It's different from the full Newton method (which would use the Hessian of the max function), but it should work well in practice.

"""
Optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. LSE-smoothed Nesterov-accelerated gradient descent
2. Direct KKT active-set linear steps (solving small (|K|+1)x(|K|+1) systems)
3. Subgradient Nesterov with entropy mirror descent
4. Multi-scale search and diversified restarts
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _project_simplex(v):
    """Project v onto the simplex {a: sum a = 1, a >= 0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)

def _ls_obj_grad(z, beta, n):
    """LSE-smoothed objective and gradient in z-space."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        obj = 2.0 * n * c_max
        w = np.zeros(2*n-1)
        w[int(np.argmax(c))] = 1.0
    else:
        obj = 2.0 * n * (c_max + beta * np.log(s))
        w = e / s
    # Gradient of b weighted by w
    a_rev = a[::-1]
    out_len = n + (2*n-1) - 1
    N = _next_pow2(out_len)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a_rev, N)
    corr = np.fft.irfft(fw * fa, N)[:out_len]
    grad_a = 2.0 * corr[n-1:2*n-1]
    grad_z = a * (grad_a - np.mean(grad_a))
    return obj, grad_z

def _nesterov_lse(a_init, deadline, min_time=5.0):
    """LSE-smoothed Nesterov-accelerated gradient descent in log-space."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted)-1)]
    base_beta = max(gap * 0.15, c_max * 0.003)
    
    n_betas = 18
    betas = np.geomspace(base_beta, max(c_max * 1e-9, 1e-18), n_betas)
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    best_a = a.copy()
    best_sc = _score(a)
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        
        remaining = deadline - min_time - time.monotonic()
        if remaining < 3:
            break
        est_time_per_iter = 0.008 * max(1.0, n / 16384.0)
        max_iter = min(5000, max(100, int(remaining / est_time_per_iter)))
        
        # Nesterov state
        t = 1.0
        y = z.copy()
        z_prev = z.copy()
        step = 0.5 / (np.linalg.norm(_ls_obj_grad(z, beta, n)[1]) + 1e-10)
        
        obj_y, _ = _ls_obj_grad(y, beta, n)
        
        for it in range(max_iter):
            if time.monotonic() + min_time > deadline:
                break
            
            obj_y, grad_y = _ls_obj_grad(y, beta, n)
            gn = np.linalg.norm(grad_y)
            if gn < 1e-15:
                break
            
            # Adaptive step
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            step = min(step * 1.02, 1.5 / (gn * t_new))
            
            z_new = y - step * grad_y
            a_new = _softmax(z_new)
            obj_new, _ = _ls_obj_grad(z_new, beta, n)
            
            if obj_new > obj_y + 1e-12:
                step *= 0.5
                if step < 1e-15:
                    break
                continue
            
            momentum = t / t_new
            y_new = z_new + momentum * (z_new - z_prev)
            z_prev = z_new
            z = z_new
            y = y_new
            t = t_new
        
        a_new = _softmax(z)
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            z = np.log(np.maximum(a_new, 1e-30))
            z = z - z.mean()
    
    return best_a

def _kkt_step(a, K_indices, n):
    """
    Compute a KKT linear step that equalizes active peaks.
    Returns (delta, success) where delta is the step direction.
    """
    m = len(K_indices)
    if m == 0:
        return None, False
    if m == 1:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        k0 = K_indices[0]
        # Gradient of b[k0]
        g = np.zeros(n)
        for i in range(n):
            val = k0 - i
            if 0 <= val < n:
                g[i] = 2.0 * a[val]
        g_proj = g - np.mean(g)
        gn = np.linalg.norm(g_proj)
        if gn < 1e-15:
            return None, False
        return -g_proj / gn, True
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    # Build (m+1) x (m+1) system
    # (GTG)[j,j'] = 4 * c[|K[j]-K[j']|] if |K[j]-K[j']| < n
    # (GT1)[j] = 2 * sum_{m:max(0,K[j]-n+1)}^{min(K[j],n-1)} a[m]
    
    # Precompute prefix sums of a for efficient GT1
    prefix = np.concatenate([[0.0], np.cumsum(a)])
    
    GTG = np.zeros((m, m))
    GT1 = np.zeros(m)
    for j in range(m):
        kt = K_indices[j]
        lo = max(0, kt - n + 1)
        hi = min(kt, n - 1)
        GT1[j] = 2.0 * (prefix[hi+1] - prefix[lo])
        for jp in range(m):
            ksp = K_indices[jp]
            lag = abs(kt - ksp)
            if lag < (2*n - 1):  # c has length 2n-1
                GTG[j, jp] = 4.0 * c[lag]
    
    # RHS
    c_vals = np.array([c[k] for k in K_indices])
    rhs1 = 2.0 * (c_vals - c_max)
    
    # Assemble system
    A_sys = np.zeros((m+1, m+1))
    A_sys[:m, :m] = GTG
    A_sys[:m, m] = GT1
    A_sys[m, :m] = GT1
    A_sys[m, m] = float(n)
    
    b_sys = np.zeros(m+1)
    b_sys[:m] = rhs1
    b_sys[m] = 0.0
    
    # Add small regularization for numerical stability
    A_sys[:m, :m] += 1e-12 * np.eye(m)
    
    try:
        sol = np.linalg.solve(A_sys, b_sys)
    except np.linalg.LinAlgError:
        sol = np.linalg.lstsq(A_sys, b_sys, rcond=None)[0]
    
    lam = sol[:m]
    mu = sol[m]
    
    # Compute delta = -(1/2) * (G λ + μ * 1)
    G_lam = np.zeros(n)
    for j in range(m):
        kt = K_indices[j]
        i_lo = max(0, kt - n + 1)
        i_hi = min(kt, n - 1)
        if i_lo <= i_hi:
            indices = kt - np.arange(i_lo, i_hi + 1)
            G_lam[i_lo:i_hi+1] += lam[j] * 2.0 * a[indices]
    
    delta = -0.5 * (G_lam + mu)
    delta -= delta.mean()  # Ensure sum = 0
    
    return delta, True

"""
Optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. LSE-smoothed Nesterov-accelerated gradient descent (log-parameterization)
2. Direct KKT active-set linear steps (solving small (|K|+1)x(|K|+1) systems)
3. Subgradient Nesterov with simplex projection
4. Multi-scale search (n=16384, n=32768) and diversified restarts

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)

def _grad_weighted(a, w):
    """Compute sum_k w[k] * grad_b[k](a). Result has shape (n,)."""
    n = len(a)
    a_rev = a[::-1]
    out_len = n + len(w) - 1
    N = _next_pow2(out_len)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a_rev, N)
    result = np.fft.irfft(fw * fa, N)[:out_len]
    return 2.0 * result[n - 1:2 * n - 1]

def _ls_obj_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        obj = 2.0 * n * c_max
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        obj = 2.0 * n * (c_max + beta * np.log(s))
        w = e / s
    grad_a = _grad_weighted(a, w)
    grad_z = a * (grad_a - np.mean(grad_a))
    return obj, grad_z

def _nesterov_lse(a_init, deadline, min_time=5.0):

    s = a.sum()

    z = np.log(np.maximum(a, 1e-30))

    best_a = a.copy()

        _, g0 = _ls_obj_grad(z, beta, n)

def _kkt_linear_step(a, K_indices, n):
    m = len(K_indices)
    if m == 0:
        return None
    if m == 1:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        k0 = K_indices[0]
        w = np.zeros(2 * n - 1)
        w[k0] = 1.0
        g = _grad_weighted(a, w)
        g_proj = g - np.mean(g)
        gn = np.linalg.norm(g_proj)
        if gn < 1e-14:
            return None
        return -g_proj / gn
    c = _full_convolve(a, a)
    c_max = float(c.max())
    prefix = np.concatenate([[0.0], np.cumsum(a)])
    GTG = np.zeros((m, m))
    GT1 = np.zeros(m)
    for j in range(m):
        kt = K_indices[j]
        lo = max(0, kt - n + 1)
        hi = min(kt, n - 1)
        GT1[j] = 2.0 * (prefix[hi + 1] - prefix[lo])
        for jp in range(m):
            ksp = K_indices[jp]
            lag = abs(kt - ksp)
            if lag < (2 * n - 1):
                GTG[j, jp] = 4.0 * c[lag]
    c_vals = np.array([c[k] for k in K_indices])
    rhs1 = 2.0 * (c_vals - c_max)
    A_sys = np.zeros((m + 1, m + 1))
    A_sys[:m, :m] = GTG
    A_sys[:m, m] = GT1
    A_sys[m, :m] = GT1
    A_sys[m, m] = float(n)
    A_sys[:m, :m] += 1e-10 * np.eye(m)
    b_sys = np.zeros(m + 1)
    b_sys[:m] = rhs1
    try:
        sol = np.linalg.solve(A_sys, b_sys)
    except np.linalg.LinAlgError:
        sol = np.linalg.lstsq(A_sys, b_sys, rcond=None)[0]
    lam = sol[:m]
    mu = sol[m]
    G_lam = np.zeros(n)
    for j in range(m):
        kt = K_indices[j]
        i_lo = max(0, kt - n + 1)
        i_hi = min(kt, n - 1)
        if i_lo <= i_hi:
            indices = kt - np.arange(i_lo, i_hi + 1)
            G_lam[i_lo:i_hi + 1] += lam[j] * 2.0 * a[indices]
    delta = -0.5 * (G_lam + mu)
    delta -= delta.mean()
    return delta

def _interpolate_seq(a, n_new):

        a5 = _kkt_refine(best_seq, dl5, min_time=200.0, max_outer=200)
        update_best(a5, _score(a5), "kkt_pass2")

    # Phase 6: Try n=32768
    print(f"\n[Phase 6] n=32768 (t={time.monotonic() - t_start:.0f}s)")
    if time.monotonic() < deadline - 150 and best_seq is not None:
        n_large = 32768
        a_large = _interpolate_seq(best_seq, n_large)
        sc_large = _score(a_large)
        print(f"  n=32768 init: {sc_large:.10f}")
        update_best(a_large, sc_large, "interp_32768")
        dl6a = min(deadline - 100, t_start + 1550)
        a6a = _nesterov_lse(a_large, dl6a, min_time=100.0)
        update_best(a6a, _score(a6a), "lse_32768")
        if time.monotonic() < deadline - 80 and best_seq is not None and len(best_seq) == n_large:
            dl6b = min(deadline - 40, t_start + 1700)
            a6b = _kkt_refine(best_seq, dl6b, min_time=40.0, max_outer=100)
            update_best(a6b, _score(a6b), "kkt_32768")

    # Phase 7: Diversified restarts
    print(f"\n[Phase 7] Restarts (t={time.monotonic() - t_start:.0f}s)")
    if best_seq is not None:
        for sigma, label in [(0.003, "tiny"), (0.01, "small"), (0.03, "med")]:
            if time.monotonic() > deadline - 60:
                break
            p_start = _perturb(best_seq, sigma, rng)
            dl_r = min(deadline - 30, t_start + 1800)
            # Quick LSE + KKT
            dl_r1 = min(dl_r - 30, time.monotonic() + 60)
            a_r = _nesterov_lse(p_start, dl_r1, min_time=30.0)
            update_best(a_r, _score(a_r), f"restart_{label}_lse")
            if time.monotonic() < deadline - 20 and best_seq is not None:
                dl_r2 = min(deadline - 10, time.monotonic() + 30)
                a_r2 = _kkt_refine(best_seq, dl_r2, min_time=10.0, max_outer=30)
                update_best(a_r2, _score(a_r2), f"restart_{label}_kkt")

    # Phase 8: Final KKT polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.0f}s)")
    if best_seq is not None:
        dl_final = deadline - 5
        a_fin = _kkt_refine(best_seq, dl_final, min_time=5.0, max_outer=50)
        update_best(a_fin, _score(a_fin), "final_kkt")

    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]
    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
