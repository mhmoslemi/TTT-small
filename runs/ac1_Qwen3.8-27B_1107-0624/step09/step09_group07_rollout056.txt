    n = len(a)

    # c[k] = sum_i a[i]*a[k-i]
    # dc[k]/da[j] = a[k-j] + a[j-k] (with appropriate bounds)
    
    # Build gradient matrix G (K x n): G[k,j] = dc[k]/da[j]
    G = np.zeros((K, n))
    for row, k in enumerate(active_idx):
        for j in range(n):

                val += a[k-j]
            if 0 <= j - k < n:
                val += a[j-k]

    best_a = a.copy()

        c = _conv(a, a)
        top_idx = np.argpartition(c, -K)[-K:]
        top_vals = c[top_idx]
        current_topk_mean = top_vals.mean()
        
        # Gradient of (1/K) * sum_{j in topK} c[j]
        # grad[i] = (1/K) * sum_{j in topK} (a[j-i] + a[i-j]) (with bounds)
        grad = np.zeros(n)
        for j in top_idx:
            lo = max(0, j - n + 1)
            hi = min(n - 1, j)
            for i in range(lo, hi + 1):
                grad[i] += a[j - i]
            # The "other" part: a[i-j] for valid i-j
            # This is the same as the first part with i and j-i swapped
            # Actually, for c[j] = sum_i a[i]*a[j-i], dc[j]/da[i] = a[j-i] (if valid) + a[i-j] (if valid)
            # When i ranges over valid values for the first term, j-i ranges over valid values for the second
            # So actually dc[j]/da[i] = a[j-i] * 1(0<=j-i<n) + a[i-j] * 1(0<=i-j<n)
            # But note: when we sum over i, the second term is just the first term with i -> j-i
            # So dc[j]/da[i] = a[j-i] * 1(0<=j-i<n) + a[i-j] * 1(0<=i-j<n)

        #              + a[i-j] * 1(max(0,j) <= i <= min(n-1,j+n-1))  [but i<n so: j <= i <= min(n-1, j+n-1)]
        # The second term: i-j >= 0 means i >= j, and i-j < n means i < j+n. Since i < n, this is i >= j and i < n.
        # So: dc[j]/da[i] = a[j-i] * 1(max(0,j-n+1) <= i <= min(n-1,j)) + a[i-j] * 1(j <= i <= n-1)

                a = a_try
                improved = True
                break
            alpha *= 0.5

def _find_direction(a, active_idx, bound=0.1):
    """Find direction d that minimizes max_k G[k,:]@d.
    LP: min t s.t. G[k,:]@d - t <= 0, sum(d)=0, -a <= d <= bound
    """
    n = len(a)
    K = len(active_idx)
    
    # Build G (K x n): G[k,j] = dc[k]/da[j]
    G = np.zeros((K, n))
    for row, k in enumerate(active_idx):
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            G[row, lo1:hi1+1] += a[k - np.arange(lo1, hi1+1)]
        if k <= n - 1:
            G[row, k:] += a[:n-k]
    
    # Build sparse constraint matrix
    # Variables: [d (n), t (1)]
    # Constraints:
    # 1) G@x - t <= 0  (K constraints)
    # 2) -d <= a  (n constraints) => -I_n @ d <= a
    # 3) d <= bound (n constraints) => I_n @ d <= bound
    
    # A_ub has (K + 2n) rows and (n+1) cols
    # Row i (0..K-1): G[i,:] for first n cols, -1 for last col
    # Row i (K..K+n-1): -e_{i-K} for first n cols, 0 for last
    # Row i (K+n..K+2n-1): e_{i-K-n} for first n cols, 0 for last
    
    # Build as CSR
    rows = []
    cols = []
    vals = []
    
    # Part 1: G rows
    for i in range(K):
        for j in range(n):
            if G[i, j] != 0:
                rows.append(i)
                cols.append(j)
                vals.append(G[i, j])
        rows.append(i)
        cols.append(n)  # t component
        vals.append(-1.0)
    
    # Part 2: -d <= a
    offset = K
    for i in range(n):
        rows.append(offset + i)
        cols.append(i)
        vals.append(-1.0)
    
    # Part 3: d <= bound
    offset = K + n
    for i in range(n):
        rows.append(offset + i)
        cols.append(i)
        vals.append(1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(K + 2*n, n + 1))
