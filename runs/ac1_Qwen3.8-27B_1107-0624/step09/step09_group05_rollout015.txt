    return (2.0 / n_active) * wc[n - 1:2 * n - 1]


def _estimate_lipschitz(a, S_indicator, iterations=30):
    """Estimate the spectral norm of the Hessian (2/|S|)*C via power iteration."""
    n = len(a)
    n_active = float(S_indicator.sum())
    if n_active < 1:
        return 2.0
    rng = np.random.default_rng(12345)
    v = rng.random(n)
    v = v / (np.linalg.norm(v) + 1e-30)
    L = 0.0
    for _ in range(iterations):
        a_rev = v[::-1]
        Cv = _full_convolve(S_indicator, a_rev)[n - 1:2 * n - 1]
        Hv = (2.0 / n_active) * Cv
        Hv_norm = np.linalg.norm(Hv)
        if Hv_norm < 1e-30:
            return 2.0
        L = float(np.dot(v, Hv))
        v = Hv / Hv_norm
    return max(L, 1e-10)


def _solve_subproject_constraint_free(a_init, S_indicator, max_iter=3000):
    """Solve the convex subproblem using projected gradient with adaptive step.
    
    Minimize (1/|S|) * sum_{k in S} conv(a,a)[k] s.t. a >= 0, sum(a) = 1.
    Uses projected gradient with backtracking line search.
    """
    n = len(a)
    n_active = float(S_indicator.sum())
    if n_active < 1:
        return a_init.copy()

    a = _proj_simplex(a_init.copy())
    grad = _grad_from_indicator(a, S_indicator)

    # Estimate Lipschitz constant
    L = _estimate_lipschitz(a, S_indicator, iterations=20)
    step = 1.0 / (L * 1.05)

    # Compute initial objective
    c = _full_convolve(a, a)
    S_idx = np.where(S_indicator > 0)[0]
    obj = float(np.mean(c[S_idx]))

    a_prev = a.copy()
    grad_prev = grad.copy()

    for it in range(max_iter):
        a_prev = a.copy()
        grad_prev = grad.copy()

        # Barzilai-Borwein step size
        if it >= 1:
            s_vec = a - a_prev
            y_vec = grad - grad_prev
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            yy = float(np.dot(y_vec, y_vec))
            if yy > 1e-20 and sy > 1e-15:
                step_bb = 0.5 * (ss / sy + sy / yy)
                step = np.clip(step_bb, 0.05 / L, 3.0 / L)

        # Gradient step with projection
        a_new = _proj_simplex(a - step * grad)
        grad_new = _grad_from_indicator(a_new, S_indicator)

        # Backtracking if no improvement
        c_new = _full_convolve(a_new, a_new)
        obj_new = float(np.mean(c_new[S_idx]))
        backtracks = 0
        while obj_new > obj + 1e-15 and backtracks < 10:
            step *= 0.5
            a_new = _proj_simplex(a - step * grad)
            grad_new = _grad_from_indicator(a_new, S_indicator)
            c_new = _full_convolve(a_new, a_new)
            obj_new = float(np.mean(c_new[S_idx]))
            backtracks += 1

        a = a_new
        grad = grad_new
        obj = obj_new

        # Convergence check
        if np.linalg.norm(a - a_prev) < 1e-12:
            break

    return a


def _active_set_outer(a_init, deadline, min_time_remaining=30.0, verbose=True):
    """Active set outer loop: iteratively identify active set and solve subproblem."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_frac = 0.01
    min_tol_frac = 0.0005
    no_improve = 0
    max_no_improve = 50

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(min_tol_frac, tol_frac * 0.7)
            if tol_frac <= min_tol_frac * 1.05:
                break
            no_improve = 0

        # Identify active set
        tol = tol_frac * c_max
        threshold = c_max - tol
        S_indicator = (c >= threshold).astype(np.float64)
        n_active = int(S_indicator.sum())
        if n_active == 0:
            S_indicator[int(np.argmax(c))] = 1.0
            n_active = 1

        # Solve subproblem
        sub_iter = max(100, min(2000, int(5000000 / n)))
        a_new = _solve_subproblem(a, S_indicator, max_iter=sub_iter)

        # Evaluate
        sc_new = _score(a_new)
        if sc_new < current_sc - 1e-14:
            a = a_new
            if verbose:
                print(f"    AS-QP: sc={sc_new:.10f}, |S|={n_active}, tol={tol_frac:.5f}")
        else:
            # Try with smaller active set (just the top peak)
            S_indicator = np.zeros(2 * n - 1)
            top_k = int(np.argmin(np.abs(np.arange(len(c)) - (n - 1))))
            S_indicator[top_k] = 1.0
            a_new2 = _solve_subproblem(a, S_indicator, max_iter=sub_iter // 2)
            sc_new2 = _score(a_new2)
            if sc_new2 < current_sc - 1e-14:
                a = a_new2
            else:
                # Tighten tolerance
                tol_frac = max(min_tol_frac, tol_frac * 0.85)

    return best_a, best_sc


def _solve_subproblem(a_init, S_indicator, max_iter=2000):
    """Solve the convex subproblem for a given active set."""
    a = _proj_simplex(a_init.copy())
    n = len(a)
    n_active = float(S_indicator.sum())
    if n_active < 1:
        return a

    grad = _grad_from_indicator(a, S_indicator)
    L = _estimate_lipschitz(a, S_indicator, iterations=15)
    step = 1.0 / (L * 1.1)

    # Use momentum (Nesterov-like) for faster convergence
    momentum = 0.0
    mu = 0.9
    y = a.copy()

    for it in range(max_iter):
        grad_y = _grad_from_indicator(y, S_indicator)
        y_new = _proj_simplex(y - step * grad_y)

        momentum_new = ((1 + np.sqrt(1 + 4 * momentum * momentum)) / 2)
        y_next = y_new + ((momentum - 1) / momentum_new) * (y_new - y)
        y_next = _proj_simplex(y_next)

        # Adaptive step: reduce if objective increases
        c_old = float(np.mean(_full_convolve(y, y)[np.where(S_indicator > 0)[0]]))
        c_new = float(np.mean(_full_convolve(y_next, y)[np.where(S_indicator > 0)[0]]))
        if c_new > c_old + 1e-14:
            step *= 0.7
            y_next = _proj_simplex(y - step * grad_y)

        y = y_next
        momentum = momentum_new

        if np.linalg.norm(y_next - y) < 1e-12:
            break

    return y


def _grad_soft_structured(a, beta):
    """Gradient of 2*n*soft_max(conv(a,a)) using FFT structure."""
    n = len(a)
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft_structured(a, beta)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)


class _SmoothProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, verbose=True):
    """L-BFGS with adaptive beta schedule on log-softmax parametrization."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    # Compute initial gap for beta scheduling
    c_work = c0.copy()
    c_sorted = np.sort(c_work)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < c0_max * 0.002:
        gap = c0_max * 0.002
    if gap < 1e-15:
        gap = c0_max * 0.01

    # Adaptive beta levels - more fine-grained at the end
    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.004, 0.002,
        0.001, 0.0005, 0.00025, 0.000125,
        0.0000625, 0.00003125, 0.000015625, 0.0000078,
        0.000004, 0.000002, 0.000001, 0.0000005
    ])
    betas = gap * multipliers

    t_start = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        size_factor = max(1.0, n / 8192.0)
        est_iter_time = 0.001 * size_factor
        max_iter = min(5000, max(200, int(remaining / (est_iter_time * 30))))

        if beta < gap * 0.001:
            max_iter = min(max_iter, 1500)

        prob = _SmoothProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 150,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    elapsed = time.monotonic() - t_start
                    print(f"    L-BFGS beta={beta:.2e}: score={best_sc:.10f}, "
                          f"nit={result.nit}, t={elapsed:.0f}s")
        except Exception:
            pass

    return best_a, best_sc


def _kkt_refine(a_init, deadline, min_time_remaining=30.0, verbose=False):
    """KKT-inspired refinement: enforce gradient constancy on support."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    no_improve = 0
    tol_frac = 0.003
    min_tol = 0.0003

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 80:
            tol_frac = max(min_tol, tol_frac * 0.8)
            if tol_frac <= min_tol * 1.1:
                break
            no_improve = 0

        # Active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active = np.zeros(len(c), dtype=bool)
            active[int(np.argmax(c))] = True
            n_active = 1

        # Average gradient over active set
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]

        # Support of a
        support = a > 1e-10
        n_support = int(support.sum())
        if n_support < 3:
            break

        # KKT: grad_avg should be constant on support
        # Compute deviation
        lam = float(np.mean(grad_avg[support]))
        residual = grad_avg[support].copy()
        residual -= lam

        # Projected gradient step
        eta = 0.1
        a_new = a.copy()
        a_new[support] -= eta * residual
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new >= current_sc - 1e-15:
            # Try smaller steps
            found = False
            for trial_eta in [0.05, 0.02, 0.01, 0.005]:
                a_try = a.copy()
                a_try[support] -= trial_eta * residual
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-14:
                    a_new = a_try
                    sc_new = sc_try
                    found = True
                    break
            if not found:
                tol_frac = max(min_tol, tol_frac * 0.9)
                continue

        a = a_new
        if no_improve == 0:
            tol_frac = max(min_tol, tol_frac * 0.97)

    return best_a, best_sc


def _subgradient_nesterov(a_init, deadline, min_time_remaining=30.0):
    """Subgradient method with Nesterov acceleration."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lb = 2.0 * n / (2 * n - 1)

    t_k = 1.0
    a_yn = a.copy()
    step_scale = 1.0

    for iteration in range(100000):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a_yn, a_yn)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-12:
            best_sc = current_sc
            best_a = a_yn.copy()

        # Active set
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active = np.zeros(len(c), dtype=bool)
            active[int(np.argmax(c))] = True
            n_active = 1

        # Gradient
        a_rev = a_yn[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        # Center gradient (removes the lambda * 1 component)
        grad_c = grad - np.mean(grad)

        # Polyak step size with Nesterov
        gap = current_sc - lb
        if gap < 1e-12:
            break
        g_norm_sq = float(np.dot(grad_c, grad_c))
        if g_norm_sq < 1e-20:
            break
        step = step_scale * gap / g_norm_sq
        step = min(step, 0.05)

        a_new = a_yn - step * grad_c
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        a_mix = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        a_mix = np.maximum(a_mix, 0.0)
        s_mix = a_mix.sum()
        if s_mix < 1e-10:
            a_mix = a_new
        else:
            a_mix = a_mix / s_mix

        a = a_yn
        a_yn = a_mix
        t_k = t_new

        if iteration % 200 == 0:
            step_scale *= 0.97

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.8
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-50.0 * (t - 0.25) ** 2) + np.exp(-50.0 * (t - 0.75) ** 2) + 0.005
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.25 * np.sin(2.0 * np.pi * 7.0 * t / n)
             + 0.12 * np.cos(2.0 * np.pi * 3.0 * t / n))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.2
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "power_law":
        t = np.linspace(0, 1, n)
        center = np.abs(t - 0.5) * 2
        a = (1.0 - center) ** 1.5 + 0.001
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    P1: Active Set QP (convex subproblem solving)
    P2: L-BFGS with adaptive beta
    P3: KKT refinement
    P4: L-BFGS round 2
    P5: Subgradient with Nesterov
    P6: Multi-n exploration
    P7: Final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(16384, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n=16384: score={best_score:.10f}")

    n_main = len(best_seq)

    # === Phase 1: Active Set QP ===
    print(f"\n[P1] Active Set QP (n={n_main}, t={time.monotonic()-t_start:.0f}s)")
    p1_end = t_start + min(500.0, (budget_s - 10) * 0.30)
    if time.monotonic() + 60 < p1_end:
        a1, sc1 = _active_set_outer(best_seq, p1_end, min_time_remaining=200, verbose=True)
        if sc1 < best_score:
            best_score = sc1
            best_seq = a1.copy()
            print(f"  P1 result: {best_score:.10f}")

    # === Phase 2: L-BFGS ===
    print(f"\n[P2] L-BFGS (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p2_end = min(deadline - 400, time.monotonic() + 350)
        a2, sc2 = _run_lbfgs(best_seq, p2_end, min_time_remaining=150, verbose=True)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(f"  P2 result: {best_score:.10f}")

    # === Phase 3: KKT refinement ===
    print(f"\n[P3] KKT refine (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p3_end = min(deadline - 300, time.monotonic() + 200)
        a3, sc3 = _kkt_refine(best_seq, p3_end, min_time_remaining=50, verbose=True)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3 result: {best_score:.10f}")

    # === Phase 4: L-BFGS round 2 ===
    print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p4_end = min(deadline - 250, time.monotonic() + 250)
        a4, sc4 = _run_lbfgs(best_seq, p4_end, min_time_remaining=100, verbose=True)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4 result: {best_score:.10f}")

    # === Phase 5: Subgradient with Nesterov ===
    print(f"\n[P5] Subgradient+Nesterov (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p5_end = min(deadline - 200, time.monotonic() + 150)
        a5, sc5 = _subgradient_nesterov(best_seq, p5_end, min_time_remaining=50)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  P5 result: {best_score:.10f}")
        # Quick L-BFGS after subgradient
        if time.monotonic() + 40 < deadline:
            p5b_end = min(deadline - 100, time.monotonic() + 100)
            a5b, sc5b = _run_lbfgs(best_seq, p5b_end, min_time_remaining=30, verbose=False)
            if sc5b < best_score:
                best_score = sc5b
                best_seq = a5b.copy()
                print(f"  P5b result: {best_score:.10f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n (t={time.monotonic()-t_start:.0f}s)")
    n_options = [12288, 20480, 24576]
    n_options = [x for x in n_options if x != n_main]
    for n_try in n_options:
        if time.monotonic() + 80 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        print(f"  n={n_try}: resampled score={sc_r:.10f}")
        if time.monotonic() + 60 > deadline:
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
            continue
        p6_end = min(deadline - 80, time.monotonic() + 120)
        a6, sc6 = _run_lbfgs(a_r, p6_end, min_time_remaining=30, verbose=False)
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  n={n_try} optimized: {best_score:.10f}")

    # === Phase 7: Final polish ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p7_end = min(deadline - 15, time.monotonic() + 100)
        a7, sc7 = _run_lbfgs(best_seq, p7_end, min_time_remaining=5, verbose=False)
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()

        if time.monotonic() + 15 < deadline:
            p7b_end = min(deadline - 5, time.monotonic() + 40)
            a7b, sc7b = _active_set_outer(best_seq, p7b_end, min_time_remaining=3, verbose=False)
            if sc7b < best_score:
                best_score = sc7b
                best_seq = a7b.copy()

        if time.monotonic() + 10 < deadline:
            p7c_end = min(deadline - 3, time.monotonic() + 20)
            a7c, sc7c = _kkt_refine(best_seq, p7c_end, min_time_remaining=2, verbose=False)
            if sc7c < best_score:
                best_score = sc7c
                best_seq = a7c.copy()

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    # Ensure no NaN or inf
    final = np.nan_to_num(final, nan=0.0, posinf=1000.0, neginf=0.0)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
          f"time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
