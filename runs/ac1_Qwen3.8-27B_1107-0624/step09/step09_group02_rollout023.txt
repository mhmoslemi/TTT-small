Optimized minimum peak autocorrelation sequence search.
Uses: (1) Log-space L-BFGS with adaptive log-sum-exp smoothing,
(2) Primal-dual active-set QP with FFT-based CG inner solver exploiting Hankel structure,

Key: H[i,j] = 2*p[i+j] is Hankel, so H*v computed via FFT cross-correlation.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_mv(p, v, N_fft):
    """Compute H*v where H[j,m] = 2*p[j+m]. Uses cross-correlation via FFT."""
    Fp = np.fft.rfft(p, N_fft)
    Fv = np.fft.rfft(v, N_fft)
    result = np.fft.irfft(Fp * np.conj(Fv), N_fft)
    return 2.0 * result[:len(v)]


def _hankel_mv_constrained(p, v, active_mask, N_fft):
    """Compute H_SS * v where only active entries are kept (FAC for CG)."""
    v_zeroed = np.where(active_mask, v, 0.0)
    result = _hankel_mv(p, v_zeroed, N_fft)
    return np.where(active_mask, result, 0.0)


def _cg_solve_hankel(p, rhs, active_mask, N_fft, max_iter=500, tol=1e-10):
    """Solve H_SS * x = rhs using CG with FFT-based matrix-vector products.
    active_mask: boolean array marking the active set.
    Returns x supported on active set."""
    r = rhs.copy()
    p_vec = r.copy()
    rs = float(np.dot(r, r))
    if rs < 1e-30:
        return rhs.copy()
    for _ in range(max_iter):
        Ap = _hankel_mv_constrained(p, p_vec, active_mask, N_fft)
        alpha = rs / (float(np.dot(p_vec, Ap)) + 1e-30)
        rhs += alpha * p_vec
        r -= alpha * Ap
        rs_new = float(np.dot(r, r))
        if np.sqrt(rs_new) < tol * np.sqrt(float(np.dot(rhs, rhs)) + 1e-30):
            break
        p_vec = r + (rs_new / (rs + 1e-30)) * p_vec
        rs = rs_new
    return np.where(active_mask, rhs, 0.0)


def _solve_active_qp(p_dual, a_init, N_fft, max_rounds=200, tol=1e-12):
    """Active-set QP: min a'H*a s.t. sum(a)=1, a>=0.
    H[i,j] = 2*p_dual[i+j]. Returns optimal a."""
    n = len(a_init)
    active = np.ones(n, dtype=bool)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    a /= s

    for rnd in range(max_rounds):
        # Solve H_SS * x = 1 on active set
        rhs = np.where(active, 1.0, 0.0)
        x = _cg_solve_hankel(p_dual, rhs, active, N_fft, max_iter=300, tol=1e-11)
        x_sum = float(x[active].sum())
        if x_sum < 1e-15:
            break
        x = x / x_sum
        a = x

        # Check primal feasibility: a[i] >= 0 for i in active
        min_active = float(a[active].min())
        if min_active < -1e-12:
            # Remove the most negative
            idx_in_active = np.where(active)[0]
            worst = idx_in_active[np.argmin(a[idx_in_active])]
            active[worst] = False
            active_count = active.sum()
            if active_count < 2:
                active = np.ones(n, dtype=bool)
                a = np.ones(n) / n
                continue
            continue

        # Check dual feasibility: mu_i = 2*(H*a)_i - lambda >= 0 for i not in active
        lambda_val = 2.0 / x_sum
        Ha = _hankel_mv(p_dual, a, N_fft)
        mu = 2.0 * Ha - lambda_val
        # For i not in active, mu[i] should be >= 0
        inactive_mu = mu[~active]
        if inactive_mu.size > 0:
            min_mu = float(inactive_mu.min())
            if min_mu < -1e-10:
                # Add the most violating
                idx_inactive = np.where(~active)[0]
                worst = idx_inactive[np.argmin(inactive_mu)]
                active[worst] = True
                continue

        # Converged
        break

    a = np.where(active, np.maximum(a, 0), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    return a

            c_local = _full_convolve(a_local, a_local)
            c_max_local = float(c_local.max())
            shifted = (c_local - c_max_local) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s_e = e.sum()
            if s_e < 1e-300:
                return 2.0 * n * c_max_local
            return 2.0 * n * (c_max_local + beta * np.log(s_e))

        def grad_func(z_local):
            a_local = _softmax(z_local)
            c_local = _full_convolve(a_local, a_local)
            c_max_local = float(c_local.max())
            shifted = (c_local - c_max_local) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s_e = e.sum()
            if s_e < 1e-300:
                w = np.zeros_like(c_local)
                w[int(np.argmax(c_local))] = 1.0
            else:
                w = e / s_e
            # Gradient in z-space: 4*n*a*(h - W) where h = conv(w, a_rev) centered
            a_rev = a_local[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W_val = float(np.dot(w, c_local))
            return 4.0 * n * a_local * (h - W_val)

        est_time = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 3))))

        try:
            result = minimize(
                obj_func, z, jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 80}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return best_a


def _primal_dual(a_init, deadline, min_time_remaining=60.0, n_topk=50):
    """Primal-dual with top-k dual and active-set QP (CG inner)."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    a /= a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    N_fft = _next_pow2(3 * n - 2)
    c = _full_convolve(a, a)
    max_c = float(c.max())
    base_temp = max_c * 0.02

    for outer in range(500):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Dual update: top-k with softmax
        k_act = min(n_topk, len(c))
        sorted_idx = np.argsort(c)[::-1]
        p = np.zeros(len(c))
        top_vals = c[sorted_idx[:k_act]]
        temp = base_temp * (0.95 ** min(outer, 100))
        top_vals_norm = top_vals / (temp + 1e-30)
        top_vals_norm = top_vals_norm - top_vals_norm.max()
        e = np.exp(np.clip(top_vals_norm, -700, 70))
        p[sorted_idx[:k_act]] = e / e.sum()

        # Primal: active-set QP
        a_new = _solve_active_qp(p, a, N_fft, max_rounds=100, tol=1e-12)
        sc = _score(a_new)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
        elif outer > 10:
            a = best_a.copy()

    return best_a, best_sc


def _proj_grad_polish(a_init, deadline, min_time_remaining=30.0):
    """Projected gradient with Nesterov momentum on log-space."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    a /= a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z -= z.mean()
    momentum = np.zeros(n)
    beta_m = 0.85
    patience = 0
    max_patience = 30

    for step in range(10000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        a_curr = _softmax(z)
        c = _full_convolve(a_curr, a_curr)
        c_max = float(c.max())

        # Top-k active set for gradient
        k_act = min(20, len(c))
        top_k = np.argsort(c)[::-1][:k_act]
        p = np.zeros(len(c))
        p[top_k] = 1.0 / k_act

        # Gradient via Hessian-vector product
        N_fft = _next_pow2(3 * n - 2)
        a_rev = a_curr[::-1]
        wc = _full_convolve(p, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(p, c))
        grad = 4.0 * n * a_curr * (h - W_val)

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        momentum = beta_m * momentum + (1 - beta_m) * grad
        lr = 0.01 / (np.linalg.norm(momentum) / (gn + 1e-15) + 0.5)
        z = z - lr * momentum
        z = np.clip(z, -60.0, 60.0)
        z -= z.mean()

        if step % 15 == 0:
            sc = _score(_softmax(z))
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = _softmax(z).copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z -= z.mean()
                    momentum *= 0.5
                    patience = 0

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.3) ** 2) + 0.05 * np.exp(-6.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "edges":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.05) ** 2) + np.exp(-10.0 * (t - 0.95) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS (log-sum-exp), primal-dual active-set QP (FFT-CG),
    projected gradient polishing, and multi-n exploration.
    Exploits Hankel structure of the Hessian for fast matrix-vector products.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 40.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 32768
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: L-BFGS from parent (400s) ===
    print(f"\n[Phase 1] L-BFGS log-space (t={time.monotonic() - t_start:.1f}s)")
    phase1_deadline = t_start + 400
    a1 = _lbfgs_logspace(parent, phase1_deadline, min_time_remaining=30.0)
    update_best(a1, _score(a1), "lbfgs_parent")

    # === Phase 2: Primal-dual active-set QP (500s) ===
    print(f"\n[Phase 2] Primal-Dual Active-Set QP (t={time.monotonic() - t_start:.1f}s)")
    phase2_deadline = min(deadline - 500, phase1_deadline + 500)
    if time.monotonic() < phase2_deadline and best_seq is not None:
        n_best = len(best_seq)
        # For large n, limit the QP to manageable size
        if n_best > 65536:
            a_work = _resample(best_seq, 49152)
        else:
            a_work = best_seq.copy()
        a_pd, sc_pd = _primal_dual(a_work, phase2_deadline, min_time_remaining=30.0,
                                    n_topk=100)
        update_best(a_pd, sc_pd, "primal_dual")

        # If we resampled, also try to resample back
        if a_work.length != n_best if hasattr(a_work, 'length') else len(a_work) != n_best:
            a_pd_back = _resample(a_pd, n_best)
            update_best(a_pd_back, _score(a_pd_back), "pd_resample_back")

    # === Phase 3: Second L-BFGS round from improved point (300s) ===
    print(f"\n[Phase 3] L-BFGS Round 2 (t={time.monotonic() - t_start:.1f}s)")
    phase3_deadline = min(deadline - 350, phase2_deadline + 300)
    if time.monotonic() < phase3_deadline and best_seq is not None:
        a2 = _lbfgs_logspace(best_seq, phase3_deadline, min_time_remaining=20.0)
        update_best(a2, _score(a2), "lbfgs_r2")

    # === Phase 4: Projected gradient polishing (200s) ===
    print(f"\n[Phase 4] Proj Grad Polish (t={time.monotonic() - t_start:.1f}s)")
    phase4_deadline = min(deadline - 200, phase3_deadline + 200)
    if time.monotonic() < phase4_deadline and best_seq is not None:
        a_pg, sc_pg = _proj_grad_polish(best_seq, phase4_deadline, min_time_remaining=15.0)
        update_best(a_pg, sc_pg, "proj_grad")

    # === Phase 5: Multi-n exploration (250s) ===
    print(f"\n[Phase 5] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    phase5_deadline = min(deadline - 120, phase4_deadline + 250)
    n_tries = [49152, 24576, 65536, 16384]
    n_best = len(best_seq) if best_seq is not None else 32768
    for n_try in n_tries:
        if time.monotonic() > phase5_deadline - 40:
            break
        if n_try == n_best:
            continue
        if best_seq is not None:
            a_init_n = _resample(best_seq, n_try)
        else:
            a_init_n = _make_profile(n_try, "tapered")
        a_n = _lbfgs_logspace(a_init_n, phase5_deadline, min_time_remaining=15.0)
        sc_n = _score(a_n)
        update_best(a_n, sc_n, f"n_{n_try}")

        # Quick primal-dual on this n if promising
        if sc_n < best_score + 0.005 and time.monotonic() < phase5_deadline - 30:
            a_nq, sc_nq = _primal_dual(a_n, phase5_deadline, min_time_remaining=10.0,
                                        n_topk=50)
            update_best(a_nq, sc_nq, f"n_{n_try}_pd")

    # === Phase 6: Final tight primal-dual (100s) ===
    print(f"\n[Phase 6] Final PD (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_fin = len(best_seq)
        if n_fin <= 49152:
            a_fin, sc_fin = _primal_dual(best_seq, deadline - 20,
                                          min_time_remaining=10.0, n_topk=200)
            update_best(a_fin, sc_fin, "final_pd")

    # === Phase 7: Ultra-tight L-BFGS (remainder) ===
    print(f"\n[Phase 7] Ultra-tight L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_fin2 = _lbfgs_logspace(best_seq, deadline - 10, min_time_remaining=5.0)
        update_best(a_fin2, _score(a_fin2), "final_lbfgs")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(32768, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
