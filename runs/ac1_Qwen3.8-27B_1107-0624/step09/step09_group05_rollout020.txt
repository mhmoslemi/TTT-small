```python
def solve_lp_active_set(a, M, active_constraints, max_iter=100):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in active, b >= 0, sum(b) = 1"""
    n = len(a)
    # Let's use scipy.optimize.linprog
    # Variables: b[0], ..., b[n-1]
    # Objective: maximize sum(b) -> minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints: (a*b)[k] <= M for k in active_constraints
    # (a*b)[k] = sum_j a[j]*b[k-j] (for valid j, k-j)
    # In terms of b: sum_j A[k,j] * b[j] <= M
    # where A[k,j] = a[k-j] if 0 <= k-j < n else 0
    
    A_ub = np.zeros((len(active_constraints), n))
    for i, k in enumerate(active_constraints):
        j_lo = max(0, k - (n-1))
        j_hi = min(n-1, k)
        for j in range(j_lo, j_hi+1):
            A_ub[i, j] = a[k-j]
    
    b_ub = np.full(len(active_constraints), M)
    
    # sum(b) = 1
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    
    # b >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    return result
```

The autocorrelation a*a in the time domain has DFT equal to |A(f)|^2 (where A is the DFT of a). The max of a*a is bounded by n * max(|A(f)|^2) / n = max(|A(f)|^2) (by Parseval, roughly).     z = z - np.max(z)
```python
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    c_max = float(c.max())
    if beta < 1e-30:

    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:

    return c_max + beta * np.log(s)

        size_factor = max(1.0, n / 4096.0)
        est_iter_time = 0.003 * size_factor
        max_iter = min(6000, max(200, int(remaining / (est_iter_time * 25))))

        if beta < gap * 0.001:
            max_iter = min(max_iter, 3000)

        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 120,
                    'maxls': 80,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"    beta={beta:.2e} ({i+1}/{len(betas)}): score={best_sc:.10f}, nit={result.nit}")
        except Exception:
            pass

    return best_a, best_sc


def _lp_find_direction(a, M, active_indices, n):
    """Solve LP: max b[0]+...+b[n-1] s.t. (a*b)[k] <= M for k in active, b>=0, sum(b)=1.
    Returns b vector or None if infeasible."""
    k_list = list(active_indices)
    if len(k_list) == 0:
        return np.ones(n) / n

    rows = []
    cols = []
    vals = []
    for i, k in enumerate(k_list):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            v = a[k - j]
            if v > 1e-12:
                rows.append(i)
                cols.append(j)
                vals.append(v)

    if len(vals) == 0:
        return np.ones(n) / n

    A_ub = csr_matrix((vals, (rows, cols)), shape=(len(k_list), n))
    b_ub = np.full(len(k_list), M)

    result = linprog(
        -np.ones(n),
        A_ub=A_ub, b_ub=b_ub,
        bounds=[(0, None)] * n,
        method='highs',
        options={'presolve': True, 'time_limit': 30}
    )

    if result.status == 0:
        b = result.x
        s = b.sum()
        if s > 1e-10:
            b = b / s
        return b
    return None


def _lp_refinement(a_init, deadline, min_time_remaining=60.0, verbose=False):
    """LP-based active set refinement.
    
    Iteratively:
    1. Compute convolution, find active peaks
    2. Solve LP for best feasible direction
    3. Update a toward the LP solution with line search
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    start_k = 5
    end_k = 80
    no_improve_count = 0
    max_no_improve = 50

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-12:
            best_sc = current_sc
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        if no_improve_count > max_no_improve:
            break

        # Visualize active set size: start small, grow over iterations
        progress = no_improve_count / max_no_improve
        k_active = int(start_k + (end_k - start_k) * min(1.0, (1 + no_improve_count) / max_no_improve))
        k_active = min(k_active, 2 * n - 2)

        # Find top k_active peaks
        top_idx = np.argsort(c)[::-1][:k_active]
        # Smooth the active set: include neighbors of top peaks
        active_mask = np.zeros(2 * n - 1, dtype=bool)
        for idx in top_idx:
            lo = max(0, idx - 2)
            hi = min(2 * n - 2, idx + 2)
            active_mask[lo:hi + 1] = True
        active_indices = np.where(active_mask)[0]
        if len(active_indices) > 200:
            active_indices = top_idx[:80]

        # Solve LP
        M = c_max * (1 + 1e-6)  # Slight relaxation for numerical safety
        b_dir = _lp_find_direction(a, M, active_indices, n)

        if b_dir is None:
            # LP infeasible - tighten M slightly and retry
            M = c_max * (1 + 1e-4)
            b_dir = _lp_find_direction(a, M, active_indices, n)
            if b_dir is None:
                # Fall back to: move mass from peaks to valleys
                break

        b_dir = np.maximum(b_dir, 0.0)
        b_dir = b_dir / b_dir.sum()

        # Symmetrize the direction
        b_dir = 0.5 * (b_dir + b_dir[::-1])
        b_dir = b_dir / b_dir.sum()

        # Line search for step size
        best_alpha = None
        for alpha in [0.5, 0.3, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03, 0.02]:
            a_try = (1 - alpha) * a + alpha * b_dir
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-13:
                best_alpha = alpha
                a = a_try
                break

        if best_alpha is None:
            no_improve_count += 5
            # Try a smaller active set
            continue

        # Symmetrize a
        a = 0.5 * (a + a[::-1])
        a = a / a.sum()

        if verbose and no_improve_count % 20 == 0:
            print(f"    LP iter: score={best_sc:.10f}, k_active={k_active}, alpha={best_alpha}")

    return best_a, best_sc


def _kkt_refine(a_init, deadline, min_time_remaining=30.0):
    """KKT-inspired active-set refinement with improved step selection."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    no_improve = 0
    max_no_improve = 200
    tol_frac = 0.003

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(0.0001, tol_frac * 0.7)
            if tol_frac <= 0.00011:
                break
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]

        support = a > 1e-10 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        lam = float(np.mean(grad_avg[support]))
        residual = grad_avg[support] - lam

        # Adaptive step size
        r_norm = np.sqrt(np.sum(residual ** 2))
        if r_norm < 1e-15:
            break
        eta = 0.1 * (c_max - (c_max - tol)) / (r_norm * r_norm + 1e-15)
        eta = min(eta, 0.3)
        eta = max(eta, 0.01)

        a_new = a.copy()
        a_new[support] -= eta * residual
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a_new = 0.5 * (a_new + a_new[::-1])
        a_new = a_new / a_new.sum()

        sc_new = _score(a_new)
        if sc_new >= current_sc - 1e-14:
            # Try smaller steps
            found = False
            for trial_eta in [eta * 0.5, eta * 0.25, eta * 0.1, 0.005, 0.002]:
                a_try = a.copy()
                a_try[support] -= trial_eta * residual
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                a_try = 0.5 * (a_try + a_try[::-1])
                a_try = a_try / a_try.sum()
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-13:
                    a_new = a_try
                    sc_new = sc_try
                    found = True
                    break
            if not found:
                tol_frac = max(0.0001, tol_frac * 0.85)
                continue

        a = a_new
        if no_improve == 0:
            tol_frac = max(0.0001, tol_frac * 0.95)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-50.0 * (t - 0.25) ** 2) + np.exp(-50.0 * (t - 0.75) ** 2) + 0.005
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.2 * np.sin(2.0 * np.pi * 13.0 * t / n)
             + 0.1 * np.cos(2.0 * np.pi * 7.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "super_tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.0
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        # Taper edges
        edge = max(1, n // 10)
        for i in range(edge):
            a[i] *= 0.5 * (1 + np.cos(np.pi * i / edge))
            a[n-1-i] *= 0.5 * (1 + np.cos(np.pi * i / edge))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _symmetrize(a):
    a = 0.5 * (a + a[::-1])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def _qpp_direction(a, active_indices, n):
    """Solve QP: find lambda >= 0, sum=1, minimizing variation of (lambda * a_rev) on support.
    This gives the optimal KKT multiplier for the active set."""
    c = _full_convolve(a, a)
    c_max = c[active_indices].max()
    
    support = a > 1e-10 * a.max()
    sup_idx = np.where(support)[0]
    
    if len(sup_idx) < 2 or len(active_indices) < 2:
        return None
    
    # Build matrix M where M[i, j] = a[active_indices[j] - sup_idx[i]] (if valid)
    m = len(active_indices)
    p = len(sup_idx)
    M = np.zeros((p, m))
    for i, si in enumerate(sup_idx):
        for j, ki in enumerate(active_indices):
            diff = ki - si
            if 0 <= diff < n:
                M[i, j] = a[diff]
    
    # Center: we want M * lambda to be as constant as possible
    # Equivalent to: minimize ||M*lambda - mean(M*lambda)*1||^2
    # = lambda^T M^T (I - J/p) M lambda
    J = np.ones((p, p)) / p
    Q = M.T @ (np.eye(p) - J) @ M
    Q = 0.5 * (Q + Q.T)  # Ensure symmetry
    
    # Solve: min lambda^T Q lambda s.t. lambda >= 0, sum(lambda) = 1
    # Use projected gradient or simple optimization
    from scipy.optimize import minimize as min_func
    
    def qp_obj(lam):
        return lam @ Q @ lam
    
    def qp_grad(lam):
        return 2 * Q @ lam
    
    lam0 = np.ones(m) / m
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - 1.0,
                    'jac': lambda x: np.ones(m)}]
    bounds = [(0, None)] * m
    
    try:
        res = min_func(qp_obj, lam0, jac=qp_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 200, 'ftol': 1e-12})
        if res.success:
            lam = np.maximum(res.x, 0)
            lam = lam / lam.sum()
            return lam
    except Exception:
        pass
    
    return None


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    P1: Symmetrize + quick L-BFGS
    P2: LP-based active set refinement (main innovation)
    P3: KKT refinement
    P4: L-BFGS with very small beta
    P5: Multi-n exploration
    P6: Final polish
    """
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        # Symmetrize parent
        best_seq = _symmetrize(parent.copy())
        best_score = _score(best_seq)
        print(f"Parent (symmetrized): n={len(best_seq)}, score={best_score:.10f}")
    else:
        best_seq = _make_profile(16384, "tapered")
        best_seq = _symmetrize(best_seq)
        best_score = _score(best_seq)
        print(f"No parent. Tapered symmetrized n=16384: score={best_score:.10f}")

    n_main = len(best_seq)
    print(f"Starting: n={n_main}, score={best_score:.10f}, time={time.monotonic()-t0:.1f}s")

    # === Phase 1: Quick L-BFGS to get close to a good local min ===
    print(f"\n[P1] L-BFGS warmup (n={n_main})")
    p1_end = t0 + min(200, (budget_s - 10) * 0.12)
    a1, sc1 = _run_lbfgs(best_seq, p1_end, min_time_remaining=200, verbose=True)
    if sc1 < best_score:
        best_score = sc1
        best_seq = _symmetrize(a1)
        best_score = _score(best_seq)
        print(f"  P1 result: {best_score:.10f}")

    # === Phase 2: LP-based active set refinement (MAIN INNOVATION) ===
    print(f"\n[P2] LP-based refinement (t={time.monotonic()-t0:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p2_end = min(deadline - 400, time.monotonic() + 300)
        a2, sc2 = _lp_refinement(best_seq, p2_end, min_time_remaining=100, verbose=True)
        if sc2 < best_score:

            best_seq = _symmetrize(a2)
            best_score = _score(best_seq)
            print(f"  P2 result: {best_score:.10f}")

    # === Phase 3: KKT refinement ===
    print(f"\n[P3] KKT refinement (t={time.monotonic()-t0:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p3_end = min(deadline - 300, time.monotonic() + 150)
        a3, sc3 = _kkt_refine(best_seq, p3_end, min_time_remaining=100)
        if sc3 < best_score:
            best_score = sc3
            best_seq = _symmetrize(a3)
            best_score = _score(best_seq)
            print(f"  P3 result: {best_score:.10f}")

    # === Phase 4: L-BFGS with very small beta (fine-grain) ===
    print(f"\n[P4] L-BFGS fine (t={time.monotonic()-t0:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p4_end = min(deadline - 200, time.monotonic() + 250)
        a4, sc4 = _run_lbfgs(best_seq, p4_end, min_time_remaining=150, verbose=True)
        if sc4 < best_score:
            best_score = sc4
            best_seq = _symmetrize(a4)
            best_score = _score(best_seq)
            print(f"  P4 result: {best_score:.10f}")

    # === Phase 5: Another LP round ===
    print(f"\n[P5] LP round 2 (t={time.monotonic()-t0:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p5_end = min(deadline - 150, time.monotonic() + 200)
        a5, sc5 = _lp_refinement(best_seq, p5_end, min_time_remaining=100, verbose=True)
        if sc5 < best_score:
            best_score = sc5
            best_seq = _symmetrize(a5)
            best_score = _score(best_seq)
            print(f"  P5 result: {best_score:.10f}")

    # === Phase 6: KKT + L-BFGS alternation ===
    print(f"\n[P6] KKT+L-BFGS (t={time.monotonic()-t0:.0f}s)")
    if time.monotonic() + 30 < deadline:
        for round_num in range(3):
            if time.monotonic() + 40 > deadline:
                break
            # KKT step
            p6a_end = min(deadline - 100 - (2 - round_num) * 30, time.monotonic() + 60)
            if time.monotonic() + 20 < p6a_end:
                a6a, sc6a = _kkt_refine(best_seq, p6a_end, min_time_remaining=30)
                if sc6a < best_score:
                    best_score = sc6a
                    best_seq = _symmetrize(a6a)
                    best_score = _score(best_seq)
                    print(f"  P6 round {round_num} KKT: {best_score:.10f}")
            # L-BFGS step

            p6b_end = min(deadline - 60 - (2 - round_num) * 20, time.monotonic() + 80)
            if time.monotonic() + 20 < p6b_end:
                a6b, sc6b = _run_lbfgs(best_seq, p6b_end, min_time_remaining=40, verbose=False)

                    best_score = _score(best_seq)

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n (t={time.monotonic()-t0:.0f}s)")
    n_options = [8192, 12288, 20480]
    n_options = [x for x in n_options if x != n_main]
    for n_try in n_options:

            break
        a_r = _resample(best_seq, n_try)
        a_r = _symmetrize(a_r)
        sc_r = _score(a_r)
        if sc_r < best_score + 0.1:
            p7_end = min(deadline - 40, time.monotonic() + 60)
            a7, sc7 = _run_lbfgs(a_r, p7_end, min_time_remaining=30, verbose=False)
            a7 = _symmetrize(a7)
            sc7 = _score(a7)
            if sc7 < best_score:
                best_score = sc7
                best_seq = a7
                print(f"  n={n_try}: {best_score:.10f}")

    # === Phase 8: Final polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t0:.0f}s)")
    if time.monotonic() + 15 < deadline:
        p8_end = min(deadline - 5, time.monotonic() + 80)
        a8, sc8 = _run_lbfgs(best_seq, p8_end, min_time_remaining=5, verbose=False)
        a8 = _symmetrize(a8)
        sc8 = _score(a8)
        if sc8 < best_score:
            best_score = sc8
            best_seq = a8

        if time.monotonic() + 10 > deadline:
            pass
        else:
            p8b_end = min(deadline - 3, time.monotonic() + 30)
            a8b, sc8b = _kkt_refine(best_seq, p8b_end, min_time_remaining=3)
            a8b = _symmetrize(a8b)
            sc8b = _score(a8b)
            if sc8b < best_score:
                best_score = sc8b
                best_seq = a8b

    # Final
    print(f"\n{'=' * 60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    if sc_final < best_score:
        best_score = sc_final
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t0:.1f}s")
    return [float(x) for x in final]
```
