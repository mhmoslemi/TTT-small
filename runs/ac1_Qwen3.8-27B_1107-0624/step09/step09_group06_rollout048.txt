import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a)=1, so just 2n*max(c)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _grad_conv_all(a, c, top_indices):

    # Second term: cross-correlation of w and reversed a
    w = np.zeros(c_len)
    w[top_indices] = 1.0  # uniform weights for active set

    # ifft(conj(Wa) * Aa)[i] = sum_j w[j] a[j+i] (other cross-correlation)
    
    # Let me just use: 
    # term2[i] = sum_k w[k] a[i-k] (with zero padding)
    # = (w flipped) convolved with a, evaluated at i + (c_len - 1)

    Wf = np.fft.rfft(w_flip, N)
    term2 = np.fft.irfft(Wf * Aa, N)

def _grad_conv_fft(a, weights):
    """Compute sum_k weights[k] * grad c[k] using FFT.
    weights has length 2n-1 (length of full convolution)."""
    n = len(a)
    c_len = 2 * n - 1
    N = _next_pow2(2 * c_len)
    Wa = np.fft.rfft(weights, N)
    Aa = np.fft.rfft(a, N)
    
    # grad c[k] at position i: a[k-i] + a[i-k] (zero-padded)
    # sum_k w[k] a[k-i]: This is the cross-correlation (w ⊛ a) at lag i
    # = ifft(conj(FFT(w)) * FFT(a))[i] ... let me verify
    # ifft(conj(W) * A)[i] = sum_j w[j] a[j+i] (by definition of cross-corr)
    # We want sum_k w[k] a[k-i]. Let j = k-i, k = j+i: sum_j w[j+i] a[j]
    # This is ifft(W * conj(A))[i+n-1] ... this is getting confusing.
    
    # Let me use a different approach. Define:
    # h1[i] = sum_k w[k] a[k-i] for i = 0, ..., n-1
    # h2[i] = sum_k w[k] a[i-k] for i = 0, ..., n-1
    # h1[i] = sum_j w[j+i] a[j] (letting j = k-i, valid when 0<=j<n and 0<=j+i<c_len)
    # h2[i] = sum_j w[i-j] a[j] (letting j = i-k, valid when 0<=j<n and 0<=i-j<c_len)
    
    # h1 = cross_correlation(w, a) shifted: h1[i] = (w cross a)[-i] 
    # where (w cross a)[m] = sum_j w[j+m] a[j]
    
    # Using FFT: (w cross a)[m] = ifft(FFT(w) * conj(FFT(a)))[m + offset]
    # Actually, the standard cross-correlation via FFT:
    # corr[m] = sum_j x[j] y[j+m] = ifft(conj(X) * Y)[m] (with appropriate indexing)
    
    # Let me just use numpy correlate for clarity, it should be fast enough for our sizes
    # Actually for n=16384, direct computation over K active peaks is O(K*n) which is fine
    
    # For efficiency, let's use the FFT approach properly:
    # h1[i] = sum_j w[j+i] a[j] for j=0..n-1 (with w out of range = 0)
    # This is the "anti-convolution" or we can write:
    # h1 = np.correlate(w, a, mode='full')[c_len - n + 1 : c_len + 1]  # not quite right
    
    # Let me just do it the straightforward way with the active indices
    # This is O(K*n) but K is small (<=200)
    top_indices = np.where(weights != 0)[0]
    grad = np.zeros(n)
    for k in top_indices:
        wk = weights[k]
        if wk == 0:
            continue
        # a[k-i] for valid i
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += wk * a[k - np.arange(i_lo, i_hi + 1)]
        # a[i-k] for valid i
        i_lo2 = max(0, k)
        i_hi2 = min(n - 1, k + n - 1)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2 + 1] += wk * a[np.arange(i_lo2, i_hi2 + 1) - k]
    return grad


class _SmoothedObj:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / self.beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * self.n * c_max
        return 2.0 * self.n * (c_max + self.beta * np.log(s))

    def grad(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / self.beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w_idx = int(np.argmax(c))
            w = np.zeros_like(c)
            w[w_idx] = 1.0
        else:
            w = e / s
        # Compute gradient: 4n * sum_k w[k] * grad_c[k]
        # Then project to simplex tangent space (subtract mean)
        grad_c = _grad_conv_fft(a, w)
        grad_obj = 4.0 * self.n * grad_c
        # Project: the gradient in z-space needs the softmax Jacobian
        # df/dz = J^T (df/da) where J is the softmax Jacobian
        # J_ij = a_i (delta_ij - a_j)
        # So df/dz_i = sum_j J_ji * df/da_j = a_i * (df/da_i - sum_j a_j * df/da_j)
        a_mean_grad = float(np.dot(a, grad_obj))
        grad_z = a * (grad_obj - a_mean_grad)
        return grad_z


def _run_lbfgs_smoothed(a_init, deadline, min_remaining=30.0):
    """L-BFGS on the log-sum-exp smoothed objective with adaptive beta."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap * 0.5, max_c * 0.005)

    # More aggressive schedule: start larger, decrease faster
    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078, 0.0039, 0.00195,
        0.000975, 0.00049, 0.000245, 0.00012,
        0.00006, 0.00003, 0.000015, 0.0000075
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break

        # More iterations for smaller beta (harder to optimize)
        frac = i / len(betas)
        max_iter = int(200 + 1500 * (1 - frac) + 800 * frac)

        prob = _SmoothedObj(beta, n)
        try:
            result = minimize(
                prob, z.copy(), jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 100,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 3 == 0:
                    print(f"    [LBFGS] beta_idx={i}, beta={beta:.2e}, score={current_sc:.10f}, time={time.monotonic()-(deadline-min_remaining):.0f}s")
        except Exception as e:

def _lp_peak_equalize(a, deadline, min_remaining=60.0):
    """LP-based peak equalization using CVXPY.
    Solves: min t s.t. grad_c[k_i].d <= t, sum(d)=0, a+d>=0
    for the top K peaks."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score_normalized(a)

    # Determine K adaptively
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    # K = number of peaks within 1% of max
    K = int(np.sum(c_sorted > c_max * 0.995))
    K = max(5, min(K, 100))

    lr_base = 0.5
    no_improve = 0

    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = _score_normalized(a)

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break

        # Select active peaks
        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]
        active_vals = c[active]
        # Only include peaks close to max
        active = active[active_vals > c_max * 0.99]
        if len(active) < 3:
            active = c_idx[:5]
        K_actual = len(active)

        # Compute gradients for active peaks
        # grad_c[k] as a vector of length n
        grads = np.zeros((K_actual, n))
        for idx, k in enumerate(active):
            grad_k = np.zeros(n)
            # (grad c[k])_i = a[k-i] + a[i-k]
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad_k[i_lo:i_hi + 1] = a[k - np.arange(i_lo, i_hi + 1)]
            i_lo2 = max(0, k)
            i_hi2 = min(n - 1, k + n - 1)
            if i_lo2 <= i_hi2:
                grad_k[i_lo2:i_hi2 + 1] += a[np.arange(i_lo2, i_hi2 + 1) - k]
            grads[idx] = grad_k

        # Solve LP: min t + mu*||d||^1 s.t. grads@ d <= t, sum(d)=0, d >= -a
        # Use l1 regularization for step size control
        mu = 0.01
        d = cp.Variable(n)
        t = cp.Variable()
        constraints = [
            grads @ d <= t,
            cp.sum(d) == 0,
            d >= -a
        ]
        objective = cp.Minimize(t + mu * cp.norm1(d))
        prob = cp.Problem(objective, constraints)

        try:
            # Use a fast LP/QP solver
            prob.solve(solver=cp.CLARABEL, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except Exception:
                break

        if d.value is None or not np.all(np.isfinite(d.value)):
            break

        d_val = d.value
        # Line search
        found = False
        for alpha in [lr_base, lr_base * 0.5, lr_base * 0.25, lr_base * 0.1, lr_base * 0.05]:
            a_try = a + alpha * d_val
            if np.any(a_try < -1e-15):
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score_normalized(a_try)
            if sc_try < current_sc - 1e-15:
                a = a_try
                found = True
                break

        if not found:
            lr_base *= 0.7
            if lr_base < 0.001:
                break

    return best_a, best_sc


def _nesterov_subgrad(a_init, deadline, min_remaining=60.0, K_init=10, K_max=200, seed=0):
    """Nesterov-accelerated subgradient method."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = float(2.0 * n * c0.max())

    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    momentum = 0.8
    base_lr = 0.003
    lr = base_lr
    K = K_init
    t = 0
    no_improve = 0
    stall_count = 0

    while time.monotonic() + min_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive K
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
        if gap < c_max * 0.003 and K < K_max:
            K = min(K_max, int(K * 1.2) + 1)

        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]

        # Subgradient
        grad = _grad_conv_fft(a, np.ones(len(c)) * 0 + np.zeros(len(c)))
        # Just compute directly for active set
        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad[i_lo:i_hi + 1] += a[k - np.arange(i_lo, i_hi + 1)]
            i_lo2 = max(0, k)
            i_hi2 = min(n - 1, k + n - 1)
            if i_lo2 <= i_hi2:
                grad[i_lo2:i_hi2 + 1] += a[np.arange(i_lo2, i_hi2 + 1) - k]

        # Project to tangent space
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break

        eff_lr = lr / (gnorm + 1e-18) * 0.1
        v = momentum * v - eff_lr * grad_t
        a_cand = a + v
        a_proj = _project_simplex(a_cand)

        c_new = _full_convolve(a_proj, a_proj)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new >= current_sc:
            v = 0.5 * v
            lr *= 0.95

        a = a_proj

        if no_improve % 100 == 0 and no_improve > 0:
            stall_count += 1
            if stall_count > 8:
                break
            perturb = 0.0005 * rng.standard_normal(n)
            a_pert = _project_simplex(a + perturb)
            c_pert = _full_convolve(a_pert, a_pert)
            sc_pert = 2.0 * n * float(c_pert.max())
            if sc_pert < best_sc:
                a = a_pert
                best_a = a.copy()
                best_sc = sc_pert
            v = np.zeros(n)
            lr = base_lr
            no_improve = 0

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "gaussian":
        t = np.linspace(-3, 3, n)
        a = np.exp(-0.5 * t ** 2)
        a = a / a.sum()
        return a
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sinc_like":
        t = np.linspace(-4, 4, n)
        a = np.sinc(t / 4.0) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":
        a = np.ones(n)
        taper = max(1, n // 30)
        for i in range(taper):
            a[i] = (i + 1) / taper
            a[n - 1 - i] = (i + 1) / taper
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.3) ** 2) + np.exp(-20 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _optimize_single(a_init, deadline, min_remaining=60.0, seed=0):
    """Full optimization pipeline for a single starting point."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score_normalized(a)

    # Step 1: L-BFGS with smoothing
    if time.monotonic() + min_remaining + 50 < deadline:
        t1 = min(deadline - min_remaining - 50, time.monotonic() + 200)
        a1 = _run_lbfgs_smoothed(a, t1, min_remaining=30.0)
        sc1 = _score_normalized(a1)
        if sc1 < best_sc:
            best_a, best_sc = a1, sc1
            a = a1

    # Step 2: LP peak equalization
    if time.monotonic() + min_remaining + 30 < deadline:
        t2 = min(deadline - min_remaining - 30, time.monotonic() + 150)
        a2, sc2 = _lp_peak_equalize(a, t2, min_remaining=30.0)
        if sc2 < best_sc:
            best_a, best_sc = a2, sc2
            a = a2

    # Step 3: Another L-BFGS pass
    if time.monotonic() + min_remaining + 30 < deadline:
        t3 = min(deadline - min_remaining - 30, time.monotonic() + 150)
        a3 = _run_lbfgs_smoothed(a, t3, min_remaining=30.0)
        sc3 = _score_normalized(a3)
        if sc3 < best_sc:
            best_a, best_sc = a3, sc3
            a = a3

    # Step 4: Nesterov subgradient
    if time.monotonic() + min_remaining + 20 < deadline:
        t4 = min(deadline - min_remaining - 20, time.monotonic() + 100)
        a4, sc4 = _nesterov_subgrad(a, t4, min_remaining=20.0,
                                     K_init=5, K_max=100, seed=seed)
        if sc4 < best_sc:
            best_a, best_sc = a4, sc4
            a = a4

    # Step 5: Final LP
    if time.monotonic() + min_remaining + 10 < deadline:
        t5 = min(deadline - min_remaining - 10, time.monotonic() + 60)
        a5, sc5 = _lp_peak_equalize(a, t5, min_remaining=10.0)
        if sc5 < best_sc:
            best_a, best_sc = a5, sc5

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Main optimization function. Tries multiple strategies and n values."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score_normalized(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score_normalized(parent)
        print(f"No parent. Uniform n={n_parent}, score={best_score:.10f}")

    print(f"\n{'='*60}")
    print(f"Starting optimization. Initial best: {best_score:.10f} at n={len(best_seq)}")
    print(f"Deadline in {deadline - t_start:.0f}s")
    print(f"{'='*60}\n")

    # === Phase 1: Quick exploration of different initial points ===
    print("[Phase 1] Quick exploration (diverse starts)")
    p1_deadline = min(deadline - 600, t_start + 200)
    
    initial_kinds = ["uniform", "gaussian", "raised_cosine", "sinc_like",
                     "tapered", "flat_taper", "bimodal"]
    
    # Include parent as one of the candidates
    candidates = [(parent, "parent")]
    for kind in initial_kinds:
        a_try = _make_profile(n_parent, kind)
        candidates.append((a_try, kind))
    
    for a_try, kind in candidates:
        if time.monotonic() > p1_deadline:
            break
        sc_try = _score_normalized(a_try)
        print(f"  {kind}: score={sc_try:.10f}")
        if sc_try < best_score:
            best_seq = a_try.copy()
            best_score = sc_try

    # === Phase 2: Main optimization on best candidate ===
    print(f"\n[Phase 2] Main optimization (t={time.monotonic()-t_start:.0f}s)")
    p2_deadline = min(deadline - 300, t_start + 800)
    a_opt, sc_opt = _optimize_single(best_seq, p2_deadline, min_remaining=60.0, seed=seed)
    if sc_opt < best_score:
        best_seq = a_opt
        best_score = sc_opt
        print(f"  NEW BEST: {best_score:.10f} [phase2] t={time.monotonic()-t_start:.0f}s")
    else:
        print(f"  No improvement in phase 2. Best: {best_score:.10f}")

    # === Phase 3: Second round with perturbation ===
    print(f"\n[Phase 3] Perturbed restart (t={time.monotonic()-t_start:.0f}s)")
    p3_deadline = min(deadline - 200, t_start + 1100)
    n = len(best_seq)
    for perturb_scale in [0.01, 0.005, 0.001]:
        if time.monotonic() > p3_deadline - 60:
            break
        a_pert = best_seq + perturb_scale * rng.standard_normal(n)
        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert > 1e-10:
            a_pert = a_pert / s_pert
        a_p2, sc_p2 = _optimize_single(a_pert, min(p3_deadline, time.monotonic() + 150),
                                        min_remaining=30.0, seed=seed + 1)
        if sc_p2 < best_score:
            best_seq = a_p2
            best_score = sc_p2
            print(f"  NEW BEST: {best_score:.10f} [perturb={perturb_scale}] t={time.monotonic()-t_start:.0f}s")

    # === Phase 4: Multi-n exploration ===
    print(f"\n[Phase 4] Multi-n search (t={time.monotonic()-t_start:.0f}s)")
    p4_deadline = min(deadline - 100, t_start + 1400)
    n_candidates = [len(best_seq)]
    # Add some other sizes
    for n_try in [12288, 8192, 24576, 32768]:
        if n_try not in n_candidates and 4096 <= n_try <= 65536:
            n_candidates.append(n_try)
    n_candidates.sort()

    for n_try in n_candidates:
        if time.monotonic() > p4_deadline - 80:
            break
        if n_try == len(best_seq):
            continue
        # Resample best to new size
        a_resamp = _resample(best_seq, n_try)
        sc_resamp = _score_normalized(a_resamp)
        print(f"  n={n_try}: resampled score={sc_resamp:.10f}")
        
        if time.monotonic() < p4_deadline - 40:
            a_opt_n, sc_opt_n = _optimize_single(a_resamp, min(p4_deadline, time.monotonic() + 200),
                                                  min_remaining=30.0, seed=seed + n_try)
            if sc_opt_n < best_score:
                best_seq = a_opt_n
                best_score = sc_opt_n
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) t={time.monotonic()-t_start:.0f}s")

    # === Phase 5: Final intensive polish ===
    print(f"\n[Phase 5] Final polish (t={time.monotonic()-t_start:.0f}s)")
    p5_deadline = deadline - 20
    if time.monotonic() < p5_deadline - 30:
        # Final L-BFGS with very small beta
        n_fin = len(best_seq)
        c_fin = _full_convolve(best_seq, best_seq)
        mb = float(c_fin.max())
        c_sorted = np.sort(c_fin)[::-1]
        gap = mb - c_sorted[min(10, len(c_sorted) - 1)]
        beta_fin = max(gap * 0.01, mb * 1e-5)
        
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        prob = _SmoothedObj(beta_fin, n_fin)
        remaining = p5_deadline - 10 - time.monotonic()
        if remaining > 10:
            max_iter = min(2000, max(50, int(remaining / 0.08)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-20,
                                           'gtol': 1e-17, 'maxcor': 100, 'maxls': 60})
                a_fin = _softmax(result.x)
                sc_fin = _score_normalized(a_fin)
                if sc_fin < best_score:
                    best_seq = a_fin
                    best_score = sc_fin
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.0f}s")
            except Exception:
                pass

        # Final LP if time permits
        if time.monotonic() < p5_deadline - 20:
            a_lp, sc_lp = _lp_peak_equalize(best_seq, p5_deadline - 10, min_remaining=10.0)
            if sc_lp < best_score:
                best_seq = a_lp
                best_score = sc_lp
                print(f"  NEW BEST: {best_score:.10f} [final_lp] t={time.monotonic()-t_start:.0f}s")

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s_final = float(final.sum())
    if s_final < 0.01:
        final = _make_profile(256, "uniform")
    # Compute actual score using the evaluate_sequence formula
    n_final = len(final)
    c_final = _full_convolve(final, final)
    final_score = 2.0 * n_final * float(c_final.max()) / (s_final ** 2)
    print(f"FINAL: score={final_score:.10f}, n={n_final}, sum={s_final:.4f}, time={time.monotonic()-t_start:.1f}s")
    print(f"{'='*60}")
    return [float(x) for x in final]
