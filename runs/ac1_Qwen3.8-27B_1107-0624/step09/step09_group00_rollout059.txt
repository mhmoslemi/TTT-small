"""
Active-set QP optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key differences from previous approach:

- Support set management (allows sparse solutions)

- Diverse structural restarts (symmetric, periodic, block-constant)
- Multi-n exploration with careful resampling

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _q_matvec_direct(v, active_lags, n):
    """Compute Q*v where Q[i,j] = 2*(i+j in active_lags). Direct method O(|K|*n)."""
    result = np.zeros(n)
    for k in active_lags:
        if k < 0 or k >= 2 * n - 1:
            continue
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            j_min = k - i_max
            j_max = k - i_min
            result[i_min:i_max + 1] += 2.0 * v[j_min:j_max + 1][::-1]
    return result

def _q_matvec_fft(v, active_set_mask, n):
    """Compute Q*v using FFT. active_set_mask is boolean array of length 2n-1."""
    N = _next_pow2(3 * n - 2)
    v_pad = np.zeros(N)
    v_pad[:n] = v
    b = np.zeros(2 * n - 1)
    b[active_set_mask] = 1.0
    b_rev = b[::-1]
    b_pad = np.zeros(N)
    b_pad[:len(b_rev)] = b_rev
    result = np.fft.irfft(np.fft.rfft(v_pad) * np.fft.rfft(b_pad), N)
    # result[i] corresponds to conv[v, b_rev][i]
    # We want: (Qv)[i] = 2 * conv[v, b_rev][2n-2-i] for i=0..n-1
    idx = 2 * n - 2 - np.arange(n)
    return 2.0 * result[idx]

def _solve_qp(active_lags, n, support, a_warm, max_cg_iter=300, reg=1e-10, use_fft=False):
    """
    Solve: min (1/2) a^T Q a subject to 1^T a = 1, a >= 0 on given support.
    Returns the solution on the support.
    """
    S = len(support)
    if S == 0:
        return np.array([])
    
    sup = np.array(support, dtype=int)
    
    def Qv(v_full):
        """Compute (Q_S v)[i] for i in support, given full-length vector."""
        if use_fft:
            mask = np.zeros(2 * n - 1, dtype=bool)
            valid_lags = active_lags[(active_lags >= 0) & (active_lags < 2 * n - 1)]
            mask[valid_lags] = True
            full = np.zeros(n)
            full[sup] = v_full
            Qfull = _q_matvec_fft(full, mask, n)
            return Qfull[sup]
        else:
            full = np.zeros(n)
            full[sup] = v_full
            Qfull = _q_matvec_direct(full, active_lags, n)
            return Qfull[sup]
    
    # Solve (Q_S + reg*I) x = 1 using CG
    b = np.ones(S)
    x = a_warm.copy() if len(a_warm) == S else b / S
    # Ensure non-negative starting point
    x = np.maximum(x, 0)
    x = x / max(x.sum(), 1e-30)
    
    r = b - (Qv(x) + reg * x)
    p = r.copy()
    rs_old = float(r @ r)
    
    for it in range(max_cg_iter):
        Ap = Qv(p) + reg * p
        pAp = float(p @ Ap)
        if abs(pAp) < 1e-30:
            break
        alpha = rs_old / pAp
        x_new = x + alpha * p
        r_new = b - (Qv(x_new) + reg * x_new)
        rs_new = float(r_new @ r_new)
        if np.sqrt(rs_new) < 1e-12 * np.sqrt(rs_old + 1e-30):
            x = x_new
            break
        p = r_new + (rs_new / max(rs_old, 1e-30)) * p
        x = x_new
        r = r_new
        rs_old = rs_new
    
    # Enforce non-negativity and normalize
    x = np.maximum(x, 0)
    s = x.sum()
    if s < 1e-30:
        x = np.ones(S) / S
    else:
        x = x / s
    
    return x

            cv_max = float(cv.max())
            shifted = (cv - cv_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                w = np.zeros_like(cv)
                w[int(np.argmax(cv))] = 1.0
            else:
                w = e / s
            a_rev = av[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, cv))
            return 4.0 * n * av * (h - W)
        
        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(4000, max(30, int(remaining / (est_time * 20))))
        
        try:
            result = minimize(obj, z, jac=grad, method='L-BFGS-B',
                            options={'maxiter': max_iter, 'ftol': 1e-18, 
                                   'gtol': 1e-15, 'maxcor': 100, 'maxls': 50})
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return a, current_sc

def _gradient_polish(a_init, deadline, min_time_remaining=10.0):
    """Simple projected gradient with backtracking for final polish."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.01
    
    while time.monotonic() < deadline - min_time_remaining:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
        
        # Active set gradient
        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        
        # Gradient
        grad = np.zeros(n)
        active_lags = np.where(active)[0]
        for k in active_lags:
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            if i_min <= i_max:
                j_min = k - i_max
                j_max = k - i_min
                grad[i_min:i_max + 1] += 2.0 * a[j_min:j_max + 1][::-1]
        
        # Project onto sum=1
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking
        alpha = lr
        improved = False
        for _ in range(30):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            lr *= 0.7
            if lr < 1e-8:
                break
    
    return best_a, best_sc

def _make_symmetric(a):
    """Make sequence symmetric."""
    n = len(a)
    result = np.zeros(n)
    for i in range(n):
        result[i] = (a[i] + a[n - 1 - i]) / 2.0
    return _normalize(result)

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _diverse_init(n, kind, rng):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        a = rng.random(n)
        # Smooth with a window
        w = 10
        kernel = np.ones(w) / w
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block":
        nblocks = rng.integers(3, 12)
        a = np.repeat(rng.random(nblocks), n // nblocks + 1)[:n]
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exponential":
        r = rng.uniform(0.7, 0.99)
        t = np.arange(n)
        a = r ** t
        a = a + r ** (n - 1 - t)  # Make somewhat symmetric
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def propose_candidate(seed=42, budget_s=1800):
    """
    Active-set QP + L-BFGS + gradient polish optimizer.
    Combines exact QP solving, smooth optimization, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 24576
        parent = np.ones(n_parent) / n_parent
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")
    
    n_main = n_parent if n_parent > 0 else 24576
    print(f"Starting optimization, n_main={n_main}, deadline in {budget_s-15:.0f}s")
    
    # === Phase 1: QP optimization from parent (300s) ===
    print(f"\n[P1] QP from parent (t={time.monotonic()-t_start:.1f}s)")
    qp_deadline = min(t_start + 300, deadline - 400)
    a_qp, sc_qp = _active_set_qp_optimize(best_seq, qp_deadline, min_time_remaining=50)
    if sc_qp < best_score:
        best_seq = a_qp.copy()
        best_score = sc_qp
        print(f"  QP improved: {best_score:.10f}")
    else:
        print(f"  QP: {sc_qp:.10f} (no improvement over {best_score:.10f})")
    
    # === Phase 2: L-BFGS smooth optimization (250s) ===
    print(f"\n[P2] L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        lbfgs_deadline = min(t_start + 550, deadline - 500)
        a_lbf, sc_lbf = _smooth_lbfgs_optimize(best_seq, lbfgs_deadline, min_time_remaining=60)
        if sc_lbf < best_score:
            best_seq = a_lbf.copy()
            best_score = sc_lbf
            print(f"  L-BFGS improved: {best_score:.10f}")
        else:
            print(f"  L-BFGS: {sc_lbf:.10f}")
    
    # === Phase 3: QP again on L-BFGS result (150s) ===
    print(f"\n[P3] QP round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        qp2_deadline = min(t_start + 700, deadline - 400)
        a_qp2, sc_qp2 = _active_set_qp_optimize(best_seq, qp2_deadline, min_time_remaining=50)
        if sc_qp2 < best_score:
            best_seq = a_qp2.copy()
            best_score = sc_qp2
            print(f"  QP2 improved: {best_score:.10f}")
    
    # === Phase 4: Diverse restarts with QP+L-BFGS (400s) ===
    print(f"\n[P4] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        kinds = ['triangle', 'hann', 'gaussian', 'bimodal', 'random_smooth', 
                 'block', 'exponential', 'uniform']
        for kind in kinds:
            if time.monotonic() > deadline - 350:
                break
            a_init = _diverse_init(n_main, kind, rng)
            # Mix with current best
            a_init = _normalize(0.5 * a_init + 0.5 * best_seq)
            
            # Quick QP
            sub_deadline = min(deadline - 350, time.monotonic() + 40)
            a_try, sc_try = _active_set_qp_optimize(a_init, sub_deadline, min_time_remaining=5,
                                                     eps_init=0.2, eps_min=0.01, max_iter=50)
            if sc_try < best_score - 1e-10:
                # Follow up with L-BFGS
                lb_deadline = min(deadline - 350, time.monotonic() + 30)
                a_try2, sc_try2 = _smooth_lbfgs_optimize(a_try, lb_deadline, min_time_remaining=5)
                if sc_try2 < best_score:
                    best_seq = a_try2.copy()
                    best_score = sc_try2
                    print(f"  NEW BEST: {best_score:.10f} [{kind}] t={time.monotonic()-t_start:.0f}s")
            elif sc_try < best_score:
                print(f"  [{kind}]: {sc_try:.10f} (close)")
    
    # === Phase 5: Random perturbations + optimize (300s) ===
    print(f"\n[P5] Random perturbations (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        for attempt in range(30):
            if time.monotonic() > deadline - 280:
                break
            sigma = 0.01 * (1.0 + attempt * 0.1)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z + sigma * rng.standard_normal(n_main)
            a_pert = np.exp(z - z.max())
            a_pert = a_pert / a_pert.sum()
            
            sub_deadline = min(deadline - 280, time.monotonic() + 25)
            a_try, sc_try = _active_set_qp_optimize(a_pert, sub_deadline, min_time_remaining=3,
                                                     eps_init=0.1, eps_min=0.005, max_iter=40)
            if sc_try < best_score - 1e-10:
                best_seq = a_try.copy()
                best_score = sc_try
                print(f"  NEW BEST: {best_score:.10f} [perturb {attempt}] t={time.monotonic()-t_start:.0f}s")
    
    # === Phase 6: Multi-n exploration (250s) ===
    print(f"\n[P6] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([n_main, n_main // 2, n_main * 3 // 2, 
                                n_main + 2048, n_main - 2048,
                                16384, 20480, 28672, 32768, 25000, 30000]))
    n_candidates = [nn for nn in n_candidates if 4096 <= nn <= 65536 and nn != n_main]
    
    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        a_resamp = _resample(best_seq, n_try)
        sc_init = _score(a_resamp)
        
        sub_deadline = min(deadline - 150, time.monotonic() + 30)
        a_opt, sc_opt = _active_set_qp_optimize(a_resamp, sub_deadline, min_time_remaining=5,
                                                 eps_init=0.15, eps_min=0.01, max_iter=40)
        if sc_opt < best_score:
            # Follow up
            lb_deadline = min(deadline - 150, time.monotonic() + 20)
            a_opt2, sc_opt2 = _smooth_lbfgs_optimize(a_opt, lb_deadline, min_time_remaining=5)
            if sc_opt2 < best_score:
                best_seq = a_opt2.copy()
                best_score = sc_opt2
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) t={time.monotonic()-t_start:.0f}s")
        else:
            print(f"  n={n_try}: init={sc_init:.8f}, opt={sc_opt:.8f}")
    
    # === Phase 7: Symmetric optimization (100s) ===
    print(f"\n[P7] Symmetric (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 120:
        a_sym = _make_symmetric(best_seq)
        sc_sym = _score(a_sym)
        sub_deadline = min(deadline - 120, time.monotonic() + 50)
        a_sym_opt, sc_sym_opt = _active_set_qp_optimize(a_sym, sub_deadline, min_time_remaining=10,
                                                         eps_init=0.1, eps_min=0.005, max_iter=60)
        if sc_sym_opt < best_score:
            best_seq = a_sym_opt.copy()
            best_score = sc_sym_opt
            print(f"  Symmetric NEW BEST: {best_score:.10f}")
        else:
            print(f"  Symmetric: {sc_sym_opt:.10f}")
    
    # === Phase 8: Final combined polish (100s) ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 20:
        # QP with very tight eps
        pol_deadline = min(deadline - 15, time.monotonic() + 50)
        a_pol, sc_pol = _active_set_qp_optimize(best_seq, pol_deadline, min_time_remaining=10,
                                                 eps_init=0.02, eps_min=0.0005, max_iter=100)
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  Polish QP: {best_score:.10f}")
        
        # Gradient polish
        if time.monotonic() < deadline - 10:
            grad_deadline = min(deadline - 5, time.monotonic() + 20)
            a_grad, sc_grad = _gradient_polish(best_seq, grad_deadline, min_time_remaining=5)
            if sc_grad < best_score:
                best_seq = a_grad.copy()
                best_score = sc_grad
                print(f"  Gradient polish: {best_score:.10f}")
        
        # One more L-BFGS with very small beta
        if time.monotonic() < deadline - 8:
            lb_deadline2 = min(deadline - 5, time.monotonic() + 15)
            a_lb2, sc_lb2 = _smooth_lbfgs_optimize(best_seq, lb_deadline2, min_time_remaining=5)
            if sc_lb2 < best_score:
                best_seq = a_lb2.copy()
                best_score = sc_lb2
                print(f"  Final L-BFGS: {best_score:.10f}")
    
    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
            # Scale to have reasonable sum
            final = final * 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    # Fallback
    fallback = np.ones(256) / 256 * 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
