"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. L-BFGS warmup with log-sum-exp smoothing
2. Direct gradient descent on simplex with backtracking line search
3. KKT-based active set equalization
4. Multi-n exploration
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)
    def grad(self, z):
        return _smograd(z, self.beta, self.n)

def _run_lbfgs(a_init, deadline, min_time_remaining=60.0, beta_scale=1.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005) * beta_scale
    
    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625,
        0.03, 0.015, 0.0075, 0.004,
        0.002, 0.001, 0.0005
    ])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        if beta < base_beta * 0.01:
            max_iter = min(3000, max(30, int(remaining / 0.05)))
        else:
            max_iter = min(2000, max(20, int(remaining / 0.03)))
        
        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 60, 'maxls': 40}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS warmup, active-set gradient descent with parabolic line search,
momentum, adaptive active set, and multi-phase refinement.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)
    def grad(self, z):
        return _smograd(z, self.beta, self.n)

    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005) * beta_scale
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075, 0.004, 0.002, 0.001])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        max_iter = min(3000, max(30, int(remaining / 0.04)))
        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 60, 'maxls': 40})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _active_set_optimize(a_init, deadline, min_time_remaining=30.0):
    """Active set gradient descent with parabolic line search and momentum."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    # Parameters
    K_min = 5
    K_max = min(100, n)
    K = K_min
    lr = 0.005
    momentum = 0.3
    v = np.zeros(n)  # momentum accumulator
    no_improve = 0
    max_no_improve = 300
    eps_rel = 0.001  # relative threshold for active set
    
    # Parabolic line search history
    prev_alpha = None
    prev_sc = None
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                # Reset
                a = best_a.copy()
                v = np.zeros(n)
                lr = 0.005
                K = K_min
                no_improve = 0
                if no_improve > max_no_improve * 3:
                    break
                continue
        
        # Adaptive active set
        # Find peaks within (1 - eps_rel) of max
        threshold = c_max * (1.0 - eps_rel)
        active = np.where(c >= threshold)[0]
        # But limit to top K
        if len(active) > K:
            active = np.argsort(c)[::-1][:K]
        K_actual = len(active)
        
        # Gradually increase K
        if no_improve == 0 and K < K_max:
            K = min(K_max, K + 2)
        
        # Compute subgradient
        w_S = np.zeros(2 * n - 1)
        w_S[active] = 1.0 / K_actual
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)[n-1:2*n-1]
        grad = h  # This is (1/|S|) * sum a[k-i] for k in S
        # The actual gradient of 2n*max(a*a) is 2n * 2 * (1/|S|) * sum a[k-i]
        # But for direction, we just need (1/|S|) * sum a[k-i]
        
        # Project to tangent of simplex
        grad = grad - np.mean(grad)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Momentum update
        v = momentum * v + (1 - momentum) * grad
        v = v - np.mean(v)
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-18:
            break
        
        # Parabolic line search
        # Try the current lr first
        alphas_to_try = [lr]
        if prev_alpha is not None:
            alphas_to_try.insert(0, lr * 1.5)
            alphas_to_try.append(lr * 0.5)
        
        best_alpha = lr
        best_try_sc = current_sc
        improved = False
        
        for alpha in alphas_to_try:
            a_try = a - alpha * v
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < best_try_sc:
                best_try_sc = sc_try
                best_alpha = alpha
        
        if best_try_sc < current_sc - 1e-13:
            # Re-evaluate with best_alpha
            a_try = a - best_alpha * v
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            a_try = a_try / s_try
            a = a_try
            lr = best_alpha
            prev_alpha = best_alpha
            prev_sc = best_try_sc
        else:
            # Try smaller steps
            alpha = lr
            for _ in range(20):
                alpha *= 0.5
                a_try = a - alpha * v
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    break
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc * (1 - 1e-10):
                    a = a_try
                    lr = alpha
                    prev_alpha = alpha
                    prev_sc = sc_try
                    improved = True
                    break
            
            if not improved:
                lr *= 0.8
                momentum *= 0.95
                if lr < 1e-12:
                    break
    
    return best_a, best_sc


def _variance_optimize(a_init, deadline, min_time_remaining=30.0):
    """Minimize the variance of the convolution (smooth proxy for max)."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.01
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        current_sc = 2.0 * n * float(c.max())
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        
        # Gradient of variance = (1/(2n-1)) * sum_k (c_k - mean(c)) * 2*a_{k-i} (reversed)
        c_mean = c.mean()
        w = 2.0 * (c - c_mean) / (2 * n - 1)
        a_rev = a[::-1]
        h = _full_convolve(w, a_rev)[n-1:2*n-1]
        grad = h - np.mean(h)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        alpha = lr / (gnorm + 1e-15)
        improved = False
        for _ in range(20):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc * (1 - 1e-10):
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            lr *= 0.9
            if lr < 1e-12:
                break
    
    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
