def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(np.max(c)) / (s * s))

            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"    beta={beta:.6f}: score={sc_new:.10f}")
        except Exception:
            pass

    return current_a


def _compute_weighted_gradient(w, a):
    """Compute sum_k w[k] * 2*a[k-j] for j=0..n-1 using FFT."""
    n = len(a)
    a_rev = a[::-1].copy()
    N_conv = _next_pow2(3 * n - 2)
    fw = np.fft.rfft(w, N_conv)
    fa = np.fft.rfft(a_rev, N_conv)
    h_full = np.fft.irfft(fw * fa, N_conv)
    return 2.0 * h_full[n - 1:2 * n - 1]


def _solve_peak_weights(peak_indices, a, c):
    """Find optimal weights for top-K peaks to maximize min rate of decrease."""
    K = len(peak_indices)
    if K == 1:
        return np.array([1.0])
    # Compute G[i,j] = g_i[j] where g_i = grad c_{k_i} = 2*a[k_i - j]
    # We need the matrix M[i,j] = g_i dot g_j = sum_l (2*a[k_i-l]) * (2*a[k_j-l])
    # = 4 * sum_l a[k_i-l] * a[k_j-l]
    # This is 4 * (c_{k_i - k_j}) if |k_i - k_j| < n, else 0... 
    # Actually: sum_l a[k_i-l]*a[k_j-l] = sum_m a[m + k_i - k_j]*a[m] = c_{k_i - k_j} (with appropriate bounds)
    # More precisely, if we define the full autocorrelation, this is the cross-correlation at lag k_i - k_j.
    # For our purposes, let's just compute the gram matrix directly for small K.
    
    # Compute gradients for each peak
    grads = []
    for k in peak_indices:
        idx = k - np.arange(len(a))
        mask = (idx >= 0) & (idx < len(a))
        g = np.zeros(len(a))
        g[mask] = 2.0 * a[idx[mask]]
        grads.append(g)
    grads = np.array(grads)  # shape (K, n)
    
    # Gram matrix: M[i,j] = grads[i] dot grads[j]
    M = grads @ grads.T  # shape (K, K)
    
    # We want to maximize min_i (sum_j w_j * M[i,j]) subject to w >= 0, sum w = 1
    # Dual: minimize sum_j y_j subject to M^T y >= 1 (componentwise), y >= 0
    # Actually: primal is max_w min_i (w^T M[:, i]) s.t. w in simplex
    # By minimax: = min_{y in simplex} max_i (y^T M[i, :])
    # Dual of primal: min_y sum_j y_j s.t. sum_j y_j * M[i,j] >= 1 for all i, y_j >= 0
    # Let's use the symmetric form: find w in simplex maximizing t s.t. w^T M[:,i] >= t for all i
    # This is: max t s.t. M^T w >= t*1, w >= 0, 1^T w = 1
    # Convert to LP: max t (equivalently min -t)
    # Variables: w_0,...,w_{K-1}, t (K+1 vars)
    # Constraints: -M^T w + t <= 0 (K constraints)
    #              -sum(w) <= 0 and sum(w) <= 0 (equality: sum(w) = 1)
    #              -w_i <= 0 (K constraints)
    #              -t <= 0 (1 constraint, t >= 0)
    
    # For LP solver: minimize -t
    # c_obj = [0, 0, ..., 0, -1] (K+1 entries)
    # A_ub: for constraint -M^T w + t <= 0: row is [-M[i,:], 1]
    #       for -sum(w) <= 0: row is [-1, -1, ..., -1, 0]
    #       for sum(w) <= 0 (i.e., sum(w) >= 0, redundant with w>=0 but helps): [-(-1),..., 0] = [1,...,1, 0]
    #         Actually let's use equality for sum(w)=1
    # A_eq: [1, 1, ..., 1, 0] = 1
    # bounds: w_i in [0, 1], t in [0, inf)
    
    K = len(peak_indices)
    c_obj = np.zeros(K + 1)
    c_obj[-1] = -1.0  # minimize -t = maximize t
    
    A_ub = np.zeros((2 * K, K + 1))
    # -M^T w + t <= 0  =>  [-M[i,:], 1] dot [w, t] <= 0
    A_ub[:K, :K] = -M.T
    A_ub[:K, K] = 1.0
    # -sum(w) <= 0  =>  [-1,...,-1, 0] dot [w,t] <= 0
    A_ub[K:2*K, :K] = -1.0
    A_ub[K:2*K, K] = 0.0
    
    A_eq = np.zeros((1, K + 1))
    A_eq[0, :K] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, 1)] * K + [(0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=np.zeros(2 * K),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if res.success:
            w_opt = res.x[:K]
            w_opt = np.maximum(w_opt, 0)
            s = w_opt.sum()
            if s > 1e-10:
                w_opt /= s
            return w_opt
    except Exception:
        pass
    
    # Fallback: uniform weights
    return np.ones(K) / K


def _peak_equalize(a_init, deadline, min_time_remaining=20.0):
    """LP-weighted peak equalization with line search."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    K_max = min(20, n)
    K = 3
    lr = 0.03
    no_improve = 0
    momentum = np.zeros(n)
    mu = 0.9
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(np.max(c))
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            K = min(K_max, K + 1)
        else:
            no_improve += 1
        
        # Identify top K peaks
        top_k_idx = np.argsort(c)[::-1][:K]
        top_k_vals = c[top_k_idx]
        
        # Solve for optimal weights
        w_opt = _solve_peak_weights(top_k_idx, a, c)
        
        # Compute weighted gradient
        w_full = np.zeros(2 * n - 1)
        w_full[top_k_idx] = w_opt
        grad = _compute_weighted_gradient(w_full, a)
        
        # Project to sum-zero direction
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Momentum update
        momentum = mu * momentum - grad / gnorm
        direction = momentum - np.mean(momentum)
        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-18:
            direction = -grad
            dnorm = gnorm
        
        # Line search
        alpha = lr / (dnorm + 1e-15)
        improved = False
        for _ in range(40):
            a_try = a + alpha * direction
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(np.max(c_try))
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if improved:
            lr = min(0.06, lr * 1.02)
            mu = min(0.95, mu + 0.01)
        else:
            lr *= 0.8
            no_improve += 1
            if no_improve % 30 == 0:
                K = max(3, K // 2)
            if no_improve >= 300:
                a = best_a.copy()
                lr = 0.03
                K = 3
                no_improve = 0
                momentum = np.zeros(n)
                break
    
    return best_a, best_sc


def _subgrad_momentum(a_init, deadline, min_time_remaining=15.0):
    """Subgradient method with Nesterov momentum."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    momentum = np.zeros(n)
    mu = 0.9
    base_lr = 0.02
    K = 5
    
    for it in range(10000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(np.max(c))
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-16:
            best_sc = current_sc
            best_a = a.copy()
        
        # Top K peaks
        top_k_idx = np.argsort(c)[::-1][:K]
        w_full = np.zeros(2 * n - 1)
        w_full[top_k_idx] = 1.0 / K
        grad = _compute_weighted_gradient(w_full, a)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Nesterov momentum
        momentum = mu * momentum - base_lr * grad / (gnorm + 1e-15)
        direction = momentum - np.mean(momentum)
        
        a_new = a + direction
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        
        # Periodic score check
        if (it + 1) % 10 == 0:
            c_new = _full_convolve(a, a)
            sc_new = 2.0 * n * float(np.max(c_new))
            if sc_new > best_sc * 1.001:
                # Reset if diverging
                a = best_a.copy()
                momentum = np.zeros(n)
                mu = 0.8
                base_lr *= 0.8
    
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.1
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.where(np.abs(t - 0.5) < 0.4, 1.0, 0.3)
        return a / a.sum()
    elif kind == "sine":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.sin(2 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase optimizer with LP-weighted peak equalization and multi-n search."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None

        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()

            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()

        best_seq = parent.copy()

        print(f"No parent. Uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS warmup ===
    p1_deadline = t_start + (deadline - t_start) * 0.35
    print(f"\n[P1] L-BFGS warmup (budget to t={p1_deadline - t_start:.0f}s)")

    sc_p1 = _score(a_p1)
    if sc_p1 < best_score:
        best_seq = a_p1.copy()
        best_score = sc_p1
        print(f"  Phase 1 best: {best_score:.10f}")
    else:
        print(f"  Phase 1: no improvement (score={sc_p1:.10f})")

    # === Phase 2: Peak equalization ===
    p2_deadline = t_start + (deadline - t_start) * 0.55
    print(f"\n[P2] Peak equalization (budget to t={p2_deadline - t_start:.0f}s)")
    a_eq, sc_eq = _peak_equalize(best_seq, p2_deadline, min_time_remaining=30.0)

        best_score = sc_eq
        print(f"  NEW BEST: {best_score:.10f} [equalize]")
    else:
        print(f"  Equalize: no improvement (best={best_score:.10f})")

    # === Phase 3: L-BFGS fine tuning ===
    p3_deadline = t_start + (deadline - t_start) * 0.70
    print(f"\n[P3] Fine L-BFGS (budget to t={p3_deadline - t_start:.0f}s)")
    a_fine = _run_lbfgs(best_seq, p3_deadline, min_time_remaining=20.0, beta_scale=0.3)
    sc_fine = _score(a_fine)
    if sc_fine < best_score:
        best_seq = a_fine.copy()
        best_score = sc_fine
        print(f"  NEW BEST: {best_score:.10f} [fine_lbfgs]")
    else:
        print(f"  Fine L-BFGS: no improvement (score={sc_fine:.10f})")

    # === Phase 4: Try n=32768 ===
    p4_deadline = t_start + (deadline - t_start) * 0.85
    print(f"\n[P4] Multi-n exploration (budget to t={p4_deadline - t_start:.0f}s)")
    n_current = len(best_seq)
    n_candidates = []
    if n_current <= 16384:
        n_candidates = [32768, 16384]
    elif n_current <= 32768:
        n_candidates = [16384, 49152]
    else:
        n_candidates = [n_current // 2, n_current * 2]
    
    for n_try in n_candidates:
        if time.monotonic() > p4_deadline - 60:
            break
        if n_try == n_current or n_try < 4096 or n_try > 65536:
            continue
        a_init = _resample(best_seq, n_try)

        if time.monotonic() < p4_deadline - 10:

    # Final validation

    try:

        print(f"evaluate_sequence: {ev_score:.10f}")
    except Exception as e:
        print(f"evaluate_sequence error: {e}")
    
    return [float(x) for x in final]
