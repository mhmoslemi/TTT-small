"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Nesterov-accelerated log-sum-exp smoothing, KKT-Newton with adaptive
active sets, CMA-ES on cosine-basis shape parameters, and multi-scale
exploration. Exploits antisymmetric structure for better initialization.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_init(n, style="antisymm"):
    t = np.linspace(0, 1, n)
    if style == "antisymm":
        # Envelope * (1 + alpha * antisymmetric_modulation)
        envelope = np.sin(np.pi * t) ** 0.7
        # Antisymmetric: cos(π(2t-1)) = -cos(2πt) ... 
        # Actually cos(π(2t-1)) = cos(2πt - π) = -cos(2πt)
        # This is symmetric. Let me use sin(2πt) which is antisymmetric around t=0.5
        mod = 1.0 + 0.3 * np.sin(2 * np.pi * t)
        a = envelope * mod
    elif style == "tapered":
        a = np.sin(np.pi * t) ** 0.6
    elif style == "shifted":
        # Concentrate mass in first half
        a = np.exp(-4*(t - 0.35)**2 / 0.2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _compute_grad_smoothed(a, beta, n):
    """Compute gradient of log-sum-exp smoothed objective w.r.t. z (log-space)."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    
    # Gradient: 4 * a_j * sum_k w_k * (a_{k-j} - c_k) for valid k-j
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W_val = float(np.dot(w, c))
    grad = 4.0 * a * (h - W_val)
    # Center the gradient (removes the null direction from scale invariance)
    grad = grad - np.mean(grad)
    return grad, w, c_max


def _nesterov_smooth(a_init, deadline, max_iter=5000, beta0=0.01, beta_decay=0.95):
    """Nesterov-accelerated gradient descent on log-sum-exp smoothed objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    beta = beta0
    t_nesterov = 1.0
    z_prev = z.copy()
    lr = 0.01
    
    no_improve = 0
    step_count = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Nesterov lookahead
        momentum = (t_nesterov + 1.0) / (t_nesterov + 1.0 + np.sqrt(1.0 + 4.0 * t_nesterov * t_nesterov))

        # y_k = z_k + (t_k - 1)/t_{k+1} * (z_k - z_{k-1})
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nesterov * t_nesterov)) / 2.0
        momentum = (t_nesterov - 1.0) / t_new
        y = z + momentum * (z - z_prev)
        
        # Compute gradient at y
        a_y = _softmax(y)
        grad, w, c_max = _compute_grad_smoothed(a_y, beta, n)

            sc_new = _score(a_new)
            
            if sc_new < best_sc + 0.001 * beta:  # Allow slight increase
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-10:
                break
        
        if not step_done:
            # Try smaller step
            alpha = lr * 0.01
            z_new = y - alpha * grad
            z_new = np.clip(z_new, -100, 100)
            z_new = z_new - z_new.mean()

        z = z_new
        t_nesterov = t_new

                    no_improve = 0

            break
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:

        idx = np.argpartition(c, -K)[-K:]

            break

        # H * v = 2 * (hankel_matvec(v, W))

        # Use a simple step if CG is too slow

                continue

def _cosine_basis_eval(t, max_order):
    """Evaluate cosine basis functions at points t."""
    M = max_order + 1
    B = np.zeros((len(t), M))
    for m in range(M):
        B[:, m] = np.cos(m * np.pi * t)
    return B


def _evaluate_shape(c, t, max_order):
    """Evaluate a sequence from shape parameters c."""
    B = _cosine_basis_eval(t, max_order)
    z = B @ c
    return _softmax(z)

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
1. Antisymmetric-aware initialization (exploits that optimal sequences are asymmetric)
2. Nesterov-accelerated log-sum-exp optimization with adaptive step size
3. IRLS (Iteratively Reweighted Least Squares) for the max objective
4. CMA-ES exploration on cosine-basis shape parameters
5. Multi-phase cycling with diverse active set strategies
6. Adaptive beta annealing based on progress (not purely geometric)
"""
import numpy as np
import time
from scipy.optimize import minimize, differential_evolution


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_subgrad(a, W, n):
    """Compute subgradient at a given weight vector W over convolution lags."""
    a_rev = a[::-1]
    wc = _full_convolve(W, a_rev)
    h = wc[n-1:2*n-1]
    c = _full_convolve(a, a)
    W_val = float(np.dot(W, c))
    grad = 2.0 * a * (h - W_val)
    return grad


def _make_init(n, style="antisymm", rng=None):
    t = np.linspace(0, 1, n)
    if style == "antisymm":
        envelope = np.sin(np.pi * t) ** 0.7
        mod = 1.0 + 0.25 * np.sin(2 * np.pi * t)
        a = envelope * mod
    elif style == "tapered":
        a = np.sin(np.pi * t) ** 0.6
    elif style == "shifted":
        a = np.exp(-6.0 * (t - 0.4) ** 2 / 0.15) + 0.01
    elif style == "double_peak":
        a = np.exp(-20*(t-0.3)**2) + np.exp(-20*(t-0.7)**2) + 0.01
    elif style == "ramp":
        a = t * np.sin(np.pi * t) + 0.01
    else:
        a = np.ones(n)
    if rng is not None:
        a = a * (1.0 + 0.02 * rng.standard_normal(n))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _nesterov_smooth(a_init, deadline, max_iter=10000, beta0=0.005, beta_decay=0.93):
    """Nesterov-accelerated gradient descent on log-sum-exp smoothed objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    best_a = a_init.copy()
    best_sc = _score(a_init)

    beta = beta0
    t_nest = 1.0
    z_prev = z.copy()
    lr = 0.008
    no_improve = 0
    step_count = 0
    last_beta_change = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Nesterov step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        momentum = (t_nest - 1.0) / t_new
        y = z + momentum * (z - z_prev)

        # Gradient at y
        a_y = _softmax(y)
        c = _full_convolve(a_y, a_y)
        c_max = float(c.max())
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        a_rev = a_y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * a_y * (h - W_val)
        grad = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad)

        if grad_norm < 1e-14:
            break

        # Backtracking line search
        alpha = lr
        a_y_sc = 2.0 * n * c_max
        step_done = False
        for ls in range(25):
            z_try = y - alpha * grad
            z_try = np.clip(z_try, -100.0, 100.0)
            z_try = z_try - z_try.mean()
            a_try = _softmax(z_try)
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_max)
            # Smooth objective value
            shifted_try = (c_try - c_max_try) / beta
            e_try = np.exp(np.clip(shifted_try, -700, 0))
            s_try = e_try.sum()
            if s_try > 1e-300:
                obj_try = 2.0 * n * (c_max_try + beta * np.log(s_try))
            else:
                obj_try = 2.0 * n * c_max_try

            if obj_try < a_y_sc * (1.0 - 1e-4 * alpha) + 1e-12:
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-12:
                break

        if not step_done:
            alpha = max(lr * 0.001, 1e-10)
            z_try = y - alpha * grad
            z_try = np.clip(z_try, -100.0, 100.0)
            z_try = z_try - z_try.mean()
            a_try = _softmax(z_try)
            sc_try = _score(a_try)
            if sc_try < best_sc:
                step_done = True

        z_prev = z
        z = z_try if step_done else z
        t_nest = t_new
        step_count += 1

        # Periodic check
        if step_count % 15 == 14:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_a = a_check.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.03)
            else:
                no_improve += 1
                lr *= 0.92
                if no_improve > 15:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    z_prev = z.copy()
                    t_nest = 1.0
                    no_improve = 0
                    lr = max(lr, 0.003)

        # Adaptive beta annealing
        if no_improve == 0 and step_count - last_beta_change > 50:
            beta *= beta_decay
            last_beta_change = step_count
        elif no_improve > 8:
            beta *= 0.97
            last_beta_change = step_count

        if beta < 1e-9:
            break

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _irls(a_init, deadline, max_rounds=500):
    """Iteratively Reweighted Least Squares for the max-autocorrelation problem.
    
    Idea: At each iteration, approximate the max by a weighted sum where weights
    are concentrated on the current maximizers, then solve the resulting
    quadratic minimization (which has a linear system).
    """
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Build weights: softmax over convolution values
        # Use adaptive temperature
        temp = c_max * 0.0j  # start with hard max
        t_fractions = [0.001, 0.005, 0.01, 0.02, 0.05]
        for tf in t_fractions:
            w = np.exp((c - c_max) / (tf * c_max + 1e-30))
            w_sum = w.sum()
            if w_sum > 1.1:  # at least 2 active
                break
        w = np.exp((c - c_max) / (tf * c_max + 1e-30))
        w = w / w.sum()

        # Solve the IRLS step: minimize sum_k w_k * b_k^2 (approximately)
        # This gives a linear system: sum_k w_k * (H_k) a = 0
        # where H_k is the Hankel matrix for lag k
        # We solve: (sum_k w_k * H_k) a = 0 with constraints sum(a)=1, a>=0
        
        # More practically: use a gradient step on the weighted sum
        # grad = sum_k w_k * 2 * b_k * db_k/da
        # db_k/da_j = a_{k-j} (for valid k-j)
        a_rev = a[::-1]
        wc = _full_convolve(w * c, a_rev)
        h = wc[n-1:2*n-1]
        
        # The gradient of sum_k w_k * b_k is: 2 * (w * something)
        # Actually for IRLS we minimize sum_k w_k * b_k (linear in a for fixed others)
        # grad of b_k w.r.t. a_j = a_{k-j} (symmetric part)
        # So grad of sum_k w_k * b_k = 2 * (W_conv * a) where W_conv is the "weighted" convolution
        
        # Let's just use: direction = -sum_k w_k * (a * shift_k(a))
        # This is: -a * (W-reversed-convolved with a), centered
        
        g = 2.0 * a * h
        g = g - np.mean(g)
        g_norm = np.linalg.norm(g)
        
        if g_norm < 1e-14:
            break
        
        # Newton-like step using the Hessian approximation
        # H * v = 2 * hankel_matvec(v, w)
        def hankel_matvec(v):
            V = v[::-1]
            out_len = len(V) + len(w) - 1
            N = _next_pow2(out_len)
            F_V = np.fft.rfft(V, N)
            F_w = np.fft.rfft(w, N)
            conv = np.fft.irfft(F_V * F_w, N)[:out_len]
            return 2.0 * conv[n-1:2*n-1]
        
        # CG solve
        x = np.zeros(n)
        r = -g.copy()
        p = r.copy()
        rs = float(r @ r)
        rs0 = rs
        if rs0 > 1e-30:
            for i in range(80):
                Hp = hankel_matvec(p)
                pHp = float(p @ Hp)
                if abs(pHp) < 1e-30 or not np.isfinite(pHp):
                    break
                alpha_cg = rs / pHp
                x = x + alpha_cg * p
                r_new = r - alpha_cg * Hp
                rs_new = float(r_new @ r_new)
                if not np.isfinite(rs_new) or rs_new > rs * 1.05:
                    break
                r = r_new
                p = (rs_new / rs) * p + r
                rs = rs_new
                if rs < 1e-12 * rs0:
                    break
        
        # Project onto sum=1
        x = x - np.mean(x)
        
        # Line search
        alpha = 1.0
        step_done = False
        for ls in range(40):
            a_try = a + alpha * x
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            if float(c_try.max()) < c_max * (1.0 - 0.002 * alpha) - 1e-15:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break
        
        if not step_done:
            a_try = a - 0.005 * g / max(g_norm, 1e-30)
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                if float(_full_convolve(a_try, a_try).max()) < c_max:
                    a = a_try
                    step_done = True
        
        sc = _score(a)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 15 and not step_done:
                break

    return best_a, best_sc
