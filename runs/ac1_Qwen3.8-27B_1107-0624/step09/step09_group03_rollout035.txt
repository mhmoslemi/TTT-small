"""
Optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2.
Strategy:
1. Adaptive active-set Newton with KKT-based active set refinement
2. Aggressive L-BFGS log-sum-exp continuation (beta down to 1e-9 * max_c)
3. Nesterov-accelerated subgradient (new)
4. Multi-resolution: solve at n=4096, interpolate to n=16384, refine
5. Iterative refinement loop with diverse strategies
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    """Compute correlation: result[i] = sum_j a[j] * A_vec[i+j] (valid range)"""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n-1:2*n-1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _hess_vec(v, A_vec, n, mu):
    """Apply (H + mu*I) to v, where H is the weighted Hessian of active constraints."""
    return 2.0 * _corr_a_A(v, A_vec, n) + mu * v


def _cg_solve(matvec_fn, rhs, max_cg=300, tol=1e-12):
    n = len(rhs)
    x = np.zeros(n)
    r = rhs - matvec_fn(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_cg):
        Ap = matvec_fn(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

"""
Optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive active-set Newton, aggressive L-BFGS log-sum-exp continuation,
Nesterov-accelerated subgradient, and multi-resolution refinement.
Key improvements over previous: adaptive KKT-based active set, cubic line search,
Nesterov momentum, and more aggressive beta continuation to 1e-10.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_c(a, c):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * float(c.max()) / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _hess_vec(v, A_vec, n, mu):
    return 2.0 * _corr_a_A(v, A_vec, n) + mu * v


def _cg_solve(matvec_fn, rhs, max_cg=300, tol=1e-12):
    n = len(rhs)
    x = np.zeros(n)
    r = rhs - matvec_fn(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_cg):
        Ap = matvec_fn(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            break
        if rs < 1e-30:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_step(z_init, beta, n, max_iter, deadline):
    """Run L-BFGS for one beta value."""
    def _obj(z_val):
        return _lsobj(z_val, beta, n)

    def _grad(z_val):
        return _lsgrad(z_val, beta, n)

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return z_init

    est_time_per_iter = 0.004 * max(1.0, n / 8192.0)
    max_iter = min(max_iter, max(50, int(remaining / (est_time_per_iter * 8))))

    result = minimize(
        _obj, z_init, jac=_grad, method='L-BFGS-B',
        options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-14,
                 'maxcor': 150, 'maxls': 200}
    )
    return result.x


def _lbfgs_continuation(a_init, deadline, min_remaining=50.0):
    """Aggressive L-BFGS log-sum-exp continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 100:
        gap = c_sorted[0] - c_sorted[100]
    elif len(c_sorted) > 10:
        gap = c_sorted[0] - c_sorted[10]
    else:
        gap = c_sorted[0] - c_sorted[-1]
    base_beta = max(gap * 0.3, max_c * 0.001)

    # Very aggressive: go to max_c * 5e-11
    n_betas = 30
    log_lo = np.log(base_beta)
    log_hi = np.log(max_c * 5e-11)
    if log_hi > log_lo:
        log_hi = log_lo
    betas = np.exp(np.linspace(log_lo, log_hi, n_betas))

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break

        max_iter = min(5000, max(100, int(remaining / 0.5)))
        z = _lbfgs_step(z, beta, n, max_iter, deadline)
        a_new = _softmax(z)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        sc_new = _score(a_new)
        if sc_new < current_sc:
            current_a = a_new
            current_sc = sc_new
            z = np.log(np.maximum(a_new, 1e-30))
            z = z - z.mean()
            if i % 5 == 0 or i == n_betas - 1:
                print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.14f} ({time.monotonic() - (deadline - min_remaining):.0f}s elapsed)")

    return current_a


def _newton_dx(a, c, c_max, tol_frac, mu_base):
    """Compute Newton direction with given active set tolerance."""
    n = len(a)
    active = c >= (c_max - tol_frac * c_max)
    if not active.any():
        active[int(np.argmax(c))] = True
    m = int(active.sum())
    if m < 1:
        return None
    if m > 30 * n:
        active = c >= (c_max - max(tol_frac * 10, 0.1) * c_max)
        m = int(active.sum())
        if m < 1:
            active[int(np.argmax(c))] = True
            m = 1

    A_vec = active.astype(np.float64) / m
    g = 2.0 * _corr_a_A(a, A_vec, n)
    g_centered = g - np.mean(g)
    gnorm = np.linalg.norm(g_centered)
    if gnorm < 1e-15:
        return None

    mu = mu_base / m + 1e-12
    A_vec_ref = A_vec

    def _matvec(v):
        return _hess_vec(v, A_vec_ref, n, mu)

    u = _cg_solve(_matvec, g_centered, max_cg=400, tol=1e-11)
    w = _cg_solve(_matvec, np.ones(n), max_cg=400, tol=1e-11)
    w_sum = float(w.sum())

    if abs(w_sum) < 1e-15:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w

    dx = dx - np.mean(dx)
    return dx


def _cubic_line_search(a, dx, f0, c0, n):
    """Cubic interpolation line search. Returns (t, f_new, c_new)."""
    # Find max t such that a + t*dx >= 0
    neg_mask = dx < -1e-30
    if neg_mask.any():
        t_max = np.min(-a[neg_mask] / dx[neg_mask]) * 0.999
    else:
        t_max = 1.0
    t_max = min(t_max, 5.0)

    if t_max <= 0:
        t_max = 1e-6

    # Try several points using cubic interpolation
    t_candidates = [t_max, t_max * 0.5, t_max * 0.25, t_max * 0.1,
                    t_max * 0.05, t_max * 0.01]
    # Also try some fixed values
    t_candidates.extend([0.5, 0.1, 0.05, 0.01, 0.005, 0.001])
    t_candidates = [t for t in t_candidates if 0 < t <= t_max]
    if not t_candidates:
        t_candidates = [t_max]

    best_t = None
    best_f = np.inf
    best_c = None

    for t in t_candidates:
        a_try = a + t * dx
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            continue
        a_try = a_try / s_try
        c_try = _full_convolve(a_try, a_try)
        f_try = float(c_try.max())
        if f_try < best_f:
            best_f = f_try
            best_t = t
            best_c = c_try

    if best_t is None:
        # Try very small step
        t = min(t_max, 1e-6)
        a_try = a + t * dx
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            return 0.0, f0, c0
        a_try = a_try / s_try
        c_try = _full_convolve(a_try, a_try)
        f_try = float(c_try.max())
        return t, f_try, c_try

    return best_t, best_f, best_c

def _nesterov_subgrad(a_init, deadline, max_iter=20000):
    """Nesterov-accelerated subgradient method on the simplex."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)

    y = a.copy()
    alpha0 = 1.0  # initial step size

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Compute subgradient at y
        c = _full_convolve(y, y)
        c_max = float(c.max())
        # Softmax weights for near-peaks
        beta_sg = c_max * 0.001
        shifted = (c - c_max) / beta_sg
        w = np.exp(np.clip(shifted, -50, 0))
        w = w / w.sum()

        g = 2.0 * _corr_a_A(y, w, n)
        g_centered = g - np.mean(g)

        # Nesterov momentum
        t_k = 1.0 + 2.0 * np.sqrt(1.0 + 4.0 * (it + 1) ** 2)
        t_k = (t_k - 1.0) / 2.0
        if it > 0:
            t_prev = (1.0 + 2.0 * np.sqrt(1.0 + 4.0 * it ** 2) - 1.0) / 2.0
            beta_mom = (t_prev - 1.0) / (t_k + 1.0)
        else:
            beta_mom = 0.0

        # Step size
        alpha = alpha0 / (np.sqrt(it + 1) + 1.0)

        y_new = y + beta_mom * (y - a)
        step = -alpha * g_centered
        a_new = y_new + step
        a_new = _project_simplex(a_new)

        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            # Reduce step size on improvement
            alpha0 = max(alpha0 * 0.95, 0.01)
        else:
            # Increase step size slightly if no improvement
            alpha0 = min(alpha0 * 1.02, 2.0)

        a = a_new
        y = y_new

        if it % 5000 == 0:
            print(f"    Nesterov iter {it}: score={best_sc:.14f}, alpha0={alpha0:.4f}")

    return best_a


def _interpolate_seq(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _refinement_loop(a, deadline, max_rounds=20):
    """Interleave Newton and L-BFGS for refinement."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    c_best = _full_convolve(a, a)
    cmax_best = float(c_best.max())

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Newton phase
        as_dl = min(deadline - 60, time.monotonic() + min(120, remaining * 0.4))
        a_n, sc_n = _active_set_newton(best_a, as_dl, max_outer=50, mu_base=1e-7)
        if sc_n < best_sc:
            best_sc = sc_n
            best_a = a_n
            c_best = _full_convolve(best_a, best_a)
            cmax_best = float(c_best.max())
            print(f"    Ref round {rnd}: Newton improved to {best_sc:.14f}")

        # L-BFGS phase
        lb_dl = min(deadline - 30, time.monotonic() + min(80, remaining * 0.3))
        c_cur = _full_convolve(best_a, best_a)
        mc = float(c_cur.max())
        z = np.log(np.maximum(best_a, 1e-30))
        z = z - z.mean()
        beta_for_step = mc * 1e-7
        z = _lbfgs_step(z, beta_for_step, n, 2000, lb_dl)
        a_l = _softmax(z)
        a_l = np.maximum(a_l, 0)
        s_l = a_l.sum()
        if s_l > 1e-15:
            a_l = a_l / s_l
            sc_l = _score(a_l)
            if sc_l < best_sc:
                best_sc = sc_l
                best_a = a_l
                print(f"    Ref round {rnd}: L-BFGS improved to {best_sc:.14f}")

        # Convergence check
        if time.monotonic() > deadline - 30:
            break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block2":
        a = np.zeros(n)
        block = n // 3
        a[:block] = 1.0
        a[2 * block:] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        return a
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-3 * (t - 0.5) ** 2)
        return a / a.sum()
    return np.ones(n) / n


def _perturb_simplex(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    1. L-BFGS continuation (aggressive beta schedule)
    2. Active-set Newton (adaptive tolerance, cubic line search)
    3. Nesterov-accelerated subgradient
    4. Iterative refinement loop
    5. Final ultra-fine L-BFGS polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.14f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.4e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.14f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        update_best(parent, _score(parent), "init_uniform")

    n_work = len(best_seq)

    # === Phase 1: L-BFGS continuation (300s) ===
    print(f"\n[Phase 1] L-BFGS continuation (t={time.monotonic() - t_start:.1f}s)")
    lbfgs_dl = min(deadline - 700, time.monotonic() + 300)
    a_p1 = _lbfgs_continuation(best_seq, lbfgs_dl, min_remaining=700.0)
    update_best(a_p1, _score(a_p1), "phase1_lbfgs")

    # === Phase 2: Active-set Newton (400s) ===
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    newton_dl = min(deadline - 500, time.monotonic() + 400)
    a_p2, sc_p2 = _active_set_newton(best_seq, newton_dl, max_outer=200, mu_base=1e-6)
    update_best(a_p2, sc_p2, "phase2_newton")

    # === Phase 3: L-BFGS with very small beta (250s) ===
    print(f"\n[Phase 3] L-BFGS very small beta (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        c_cur = _full_convolve(best_seq, best_seq)
        mc = float(c_cur.max())
        # Start from very small beta
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_start = mc * 1e-5
        a_p3a = _softmax(z)
        remaining = deadline - 400 - time.monotonic()
        if remaining > 30:
            # Do several L-BFGS steps with decreasing beta
            betas_p3 = np.logspace(np.log10(1e-5), np.log10(1e-10), 15) * mc
            for b in betas_p3:
                rem = deadline - 400 - time.monotonic()
                if rem < 10:
                    break
                z = _lbfgs_step(z, b, n_work, 3000, deadline - 400)
                a_try = _softmax(z)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    break
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_score:
                    update_best(a_try, sc_try, f"phase3_beta={b:.1e}")

    # === Phase 4: Nesterov subgradient (150s) ===
    print(f"\n[Phase 4] Nesterov subgradient (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        nes_dl = min(deadline - 250, time.monotonic() + 150)
        a_p4 = _nesterov_subgrad(best_seq, nes_dl, max_iter=10000)
        update_best(a_p4, _score(a_p4), "phase4_nesterov")

    # === Phase 5: Refinement loop (200s) ===
    print(f"\n[Phase 5] Refinement loop (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        ref_dl = min(deadline - 150, time.monotonic() + 200)
        a_p5, sc_p5 = _refinement_loop(best_seq, ref_dl, max_rounds=15)
        update_best(a_p5, sc_p5, "phase5_refine")

    # === Phase 6: Final ultra-fine polish (remaining time) ===
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        n_fin = len(best_seq)
        c_fin = _full_convolve(best_seq, best_seq)
        mc_fin = float(c_fin.max())
        z_fin = np.log(np.maximum(best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()

        # Try progressively smaller beta
        remaining = deadline - 5 - time.monotonic()
        if remaining > 5:
            # Ultra-fine beta L-BFGS
            betas_final = np.array([1e-8, 5e-9, 2e-9, 1e-9, 5e-10, 2e-10, 1e-10]) * mc_fin
            for bf in betas_final:
                rem = deadline - 5 - time.monotonic()
                if rem < 5:
                    break
                max_it = min(3000, max(50, int(rem / 0.3)))
                z_fin = _lbfgs_step(z_fin, bf, n_fin, max_it, deadline - 5)
                a_try = _softmax(z_fin)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    break
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_score:
                    update_best(a_try, sc_try, f"final_beta={bf:.1e}")
                    z_fin = np.log(np.maximum(a_try, 1e-30))
                    z_fin = z_fin - z_fin.mean()

        # Final Newton with very tight tolerance
        if time.monotonic() < deadline - 3:
            rem = deadline - 3 - time.monotonic()
            if rem > 5:
                a_fn, sc_fn = _active_set_newton(best_seq, deadline - 3,
                                                 max_outer=50, mu_base=1e-8)
                update_best(a_fn, sc_fn, "final_newton")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.14f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.14f}")
    return [float(x) for x in fallback]
