def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]

def _cg_solve_hankel(W, n, rhs, max_iter=500, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.05:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

def _compute_gradient(a, W, n):
    a_rev = a[::-1]
    wc = _full_convolve(W, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(W, _full_convolve(a, a)))
    return 2.0 * a * (h - W_val)

def _lbfgs_smooth(a_init, deadline, min_remaining=30.0, beta_frac=0.001, max_beta_iters=50):
    """L-BFGS with log-sum-exp smooth approximation and adaptive beta."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    beta = max(gap * 0.5, max_c * beta_frac)
    
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    
    for beta_iter in range(max_beta_iters):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        
        scale = max(1.0, n / 8192.0)
        est_time_per_iter = 0.003 * scale
        max_iter = min(10000, max(50, int(remaining / (est_time_per_iter * 50))))
        
        def _obj_and_grad(z, beta=beta, n=n):
            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                obj = 2.0 * n * c_max
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                obj = 2.0 * n * (c_max + beta * np.log(s))
                w = e / s
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W_val = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W_val)
            return obj, grad
        
        def obj_fn(z):
            return _obj_and_grad(z)[0]
        
        def grad_fn(z):
            return _obj_and_grad(z)[1]
        
        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 80, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-18:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
        
        # Update beta
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        # Use top-10 gap
        top_k = min(10, len(c_sorted_new))
        new_gap = c_sorted_new[0] - c_sorted_new[top_k - 1]
        
        if new_gap < 1e-8 * c_max_new:
            beta *= 0.05
        elif new_gap < 1e-6 * c_max_new:
            beta *= 0.1
        elif new_gap < 1e-4 * c_max_new:
            beta *= 0.2
        else:
            beta *= 0.35
        
        beta = min(beta, max_c_new * 1e-4)
        if beta < c_max_new * 1e-10:
            break
    
    return current_a
