def _next_pow2(n):
    k = 1

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

def _cg_solve(matvec, rhs, max_cg, tol):

    x = np.zeros(n)
    r = rhs - matvec(x)
    p = r.copy()

    for _ in range(max_cg):

        if pAp < 1e-30:

        r = r - alpha * Ap

        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new

    w_vec = _cg_solve(_hess_vp, np.ones(n), 300, 1e-13)
    w_sum = float(w_vec.sum())

    if abs(w_sum) < 1e-15:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w_vec

    dx = dx - np.mean(dx)
    return dx


def _active_set_newton(a_init, deadline, max_outer):
    """KKT-based active-set Newton refinement."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        best_improvement = 0.0
        best_a_new = None

        if outer < max_outer // 2:
            tol_list = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001]
        else:
            tol_list = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]

        for tol_frac in tol_list:
            dx = _newton_step(a, c, c_max, tol_frac, 1e-8 / max(1, int((c >= c_max * 0.999).sum())))
            if dx is None:
                continue

            t = 1.0
            for ls in range(60):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < c_max - 1e-15 * t:
                    improvement = c_max - f_try
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_a_new = a_try.copy()
                    break
                t *= 0.5
                if t < 1e-15:
                    break

        if best_a_new is None:
            # GMRES-like fallback: use the smooth gradient
            c_sorted = np.sort(c)[::-1]
            if len(c_sorted) > 10:
                gap = c_sorted[0] - c_sorted[10]
            else:
                gap = c_sorted[0] - c_sorted[-1]
            beta = max(gap * 0.5, c_max * 0.001)
            w = np.exp(np.clip((c - c_max) / beta, -700, 0))
            w = w / w.sum()
            g = _gradient_weighted(w, a, n)
            g_t = g - np.mean(g)
            t = 0.01
            applied = False
            for ls in range(40):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_sc - 1e-14:
                        best_a_new = a_try.copy()
                        best_improvement = 1.0
                        applied = True
                        break
                t *= 0.7
            if not applied:
                break

        a = best_a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Check convergence
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _lbfgs_continue(a_init, deadline, min_time_remaining):
    """L-BFGS with adaptive log-sum-exp continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    idx = min(20, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[idx]
    base_beta = max(gap * 0.3, max_c * 0.005)

    # Adaptive schedule: geometric from base_beta to very small
    min_beta = max_c * 3e-8
    n_betas = 30
    log_betas = np.linspace(np.log(base_beta), np.log(min_beta), n_betas)
    betas = np.exp(log_betas)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time = 0.004 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 12))))

        beta_val = beta
        n_val = n

        def _obj(z_val):
            return _lsobj(z_val, beta_val, n_val)

        def _grad(z_val):
            return _lsgrad(z_val, beta_val, n_val)

        try:
            result = minimize(
                _obj, z, jac=_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 120, 'maxls': 400}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-17:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            if i % 5 == 0 or sc_new < current_sc + 1e-16:
                print(f"    L-BFGS beta={beta:.2e} (i={i}): score={sc_new:.14f}")
        except Exception as ex:
            print(f"    L-BFGS beta={beta:.2e}: exception {ex}")

    return current_a


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _multi_peak_deflation(a, deadline, n_attempts):
    """Targeted deflation of significant peaks."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        for thresh_frac in [0.995, 0.99, 0.98, 0.97]:
            threshold = c_max * thresh_frac
            peak_mask = c >= threshold
            n_peaks = int(peak_mask.sum())
            if n_peaks == 0:
                peak_mask[np.argmax(c)] = True
                n_peaks = 1

            w = np.zeros_like(c)
            # Weight peaks by how close they are to max
            w[peak_mask] = (c[peak_mask] - threshold) / (c_max - threshold + 1e-30)
            w = w / w.sum()

            grad = _gradient_weighted(w, a, n)
            g_t = grad - np.mean(grad)

            applied = False
            for strength in [0.0003, 0.0007, 0.0015, 0.003, 0.006, 0.012, 0.025]:
                a_try = a - strength * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-15:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = best_a
                    applied = True
                    break

            if applied:
                break

        if not applied:
            # Random perturbation escape
            rng_local = np.random.default_rng(hash((attempt, n)) % (2**31))
            found = False
            for sigma in [0.002, 0.005, 0.01, 0.02]:
                a_try = _perturb(a, sigma, rng_local)
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-15:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = best_a
                    found = True
                    break
            if not found:
                break

    return best_a, best_sc


def _refinement_loop(a, deadline, max_rounds):
    """Iterate between Newton and deflation."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Newton step
        as_dl = min(deadline - 5, time.monotonic() + 60)
        a_newton, sc_newton = _active_set_newton(a, as_dl, max_outer=40)
        if sc_newton < best_sc:
            best_sc = sc_newton
            best_a = a_newton.copy()
            a = a_newton

        # Deflation step
        def_dl = min(deadline - 5, time.monotonic() + 30)
        a_def, sc_def = _multi_peak_deflation(a, def_dl, n_attempts=15)
        if sc_def < best_sc:
            best_sc = sc_def
            best_a = a_def.copy()
            a = a_def

        sc_current = _score(a)
        if abs(sc_current - best_sc) < 1e-15:
            break

    return best_a, best_sc


def _make_initial(n):
    """Create a reasonable initial sequence."""
    # Use a raised cosine taper to reduce boundary effects
    t = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining L-BFGS continuation, KKT-based
    active-set Newton, multi-peak deflation, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.14f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={len(parent)}, score={sc_parent:.14f}")
        update_best(parent, sc_parent, "parent")
    else:
        parent = _make_initial(16384)
        update_best(parent, _score(parent), "uniform_init")

    n_parent = len(parent)

    # === Phase 1: L-BFGS continuation on parent size ===
    print(f"\n[Phase 1] L-BFGS continuation n={n_parent} (t={time.monotonic() - t_start:.1f}s)")
    lbfgs_dl = min(deadline - 600, time.monotonic() + 400)
    a_lbfgs = _lbfgs_continue(parent, lbfgs_dl, min_time_remaining=600.0)
    update_best(a_lbfgs, _score(a_lbfgs), "lbfgs_phase1")

    # === Phase 2: Active-set Newton refinement ===
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        as_dl = min(deadline - 450, time.monotonic() + 180)
        a_as, sc_as = _active_set_newton(best_seq, as_dl, max_outer=200)
        update_best(a_as, sc_as, "as_newton_phase2")

    # === Phase 3: Multi-peak deflation ===
    print(f"\n[Phase 3] Multi-peak deflation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        def_dl = min(deadline - 350, time.monotonic() + 100)
        a_def, sc_def = _multi_peak_deflation(best_seq, def_dl, n_attempts=80)
        update_best(a_def, sc_def, "deflation_phase3")

    # === Phase 4: Refinement loop ===
    print(f"\n[Phase 4] Refinement loop (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        ref_dl = min(deadline - 250, time.monotonic() + 120)
        a_ref, sc_ref = _refinement_loop(best_seq, ref_dl, max_rounds=6)
        update_best(a_ref, sc_ref, "refinement_phase4")

    # === Phase 5: L-BFGS polish ===
    print(f"\n[Phase 5] L-BFGS polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        lbfgs2_dl = min(deadline - 200, time.monotonic() + 100)
        a_pol = _lbfgs_continue(best_seq, lbfgs2_dl, min_time_remaining=200.0)
        update_best(a_pol, _score(a_pol), "lbfgs_polish_phase5")

    # === Phase 6: Try n=32768 ===
    print(f"\n[Phase 6] Try n=32768 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        n_large = 32768
        if best_seq is not None:
            a_large = _interpolate_seq(best_seq, n_large)
        else:
            a_large = _make_initial(n_large)
        sc_init = _score(a_large)
        print(f"  n={n_large} init score: {sc_init:.14f}")
        update_best(a_large, sc_init, "n32768_init")

        lbfgs3_dl = min(deadline - 150, time.monotonic() + 90)
        a_large_opt = _lbfgs_continue(a_large, lbfgs3_dl, min_time_remaining=150.0)
        sc_large = _score(a_large_opt)
        update_best(a_large_opt, sc_large, "n32768_lbfgs")

        # Quick Newton on large
        if time.monotonic() < deadline - 100:
            as3_dl = min(deadline - 100, time.monotonic() + 50)
            a_large_as, sc_large_as = _active_set_newton(a_large_opt, as3_dl, max_outer=50)
            update_best(a_large_as, sc_large_as, "n32768_newton")

    # === Phase 7: Try n=65536 if time allows ===
    print(f"\n[Phase 7] Try n=65536 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 120:
        n_xl = 65536
        if best_seq is not None:
            a_xl = _interpolate_seq(best_seq, n_xl)
        else:
            a_xl = _make_initial(n_xl)
        sc_xl_init = _score(a_xl)
        print(f"  n={n_xl} init score: {sc_xl_init:.14f}")
        update_best(a_xl, sc_xl_init, "n65536_init")

        if time.monotonic() < deadline - 60:
            lbfgs4_dl = min(deadline - 50, time.monotonic() + 55)
            a_xl_opt = _lbfgs_continue(a_xl, lbfgs4_dl, min_time_remaining=50.0)
            sc_xl = _score(a_xl_opt)
            update_best(a_xl_opt, sc_xl, "n65536_lbfgs")

    # === Phase 8: Diversified restarts ===
    print(f"\n[Phase 8] Diversified restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        for sigma, label in [(0.003, "micro"), (0.01, "small"), (0.03, "med"), (0.08, "large"), (0.15, "big")]:
            if time.monotonic() > deadline - 40:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _lbfgs_continue(p_start, deadline, min_time_remaining=35.0)
            update_best(a_r, _score(a_r), f"restart_{label}")

    # === Phase 9: Final intensive polish ===
    print(f"\n[Phase 9] Final intensive polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        # Final refinement loop
        ref_final_dl = min(deadline - 10, time.monotonic() + 50)
        a_fin, sc_fin = _refinement_loop(best_seq, ref_final_dl, max_rounds=10)
        update_best(a_fin, sc_fin, "final_ref")

        # Ultra-small beta L-BFGS
        if time.monotonic() < deadline - 8:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 1e-8
            remaining = deadline - 8 - time.monotonic()
            if remaining > 2:
                max_iter_fin = min(3000, max(50, int(remaining / 0.08)))
                beta_val = beta_final
                n_val = n

                def _obj_final(z_val):
                    return _lsobj(z_val, beta_val, n_val)

                def _grad_final(z_val):
                    return _lsgrad(z_val, beta_val, n_val)

                try:
                    result = minimize(
                        _obj_final, z, jac=_grad_final, method='L-BFGS-B',
                        options={'maxiter': max_iter_fin, 'ftol': 1e-20,
                                 'gtol': 1e-16, 'maxcor': 150, 'maxls': 500}
                    )
                    a_fin2 = _softmax(result.x)
                    update_best(a_fin2, _score(a_fin2), "final_tiny_beta")
                except Exception:
                    pass

        # One more Newton pass
        if time.monotonic() < deadline - 5:
            a_nn, sc_nn = _active_set_newton(best_seq, deadline - 2, max_outer=80)
            update_best(a_nn, sc_nn, "final_newton")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_initial(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.14f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_initial(256)
    print(f"FALLBACK: score={_score(fallback):.14f}")
    return [float(x) for x in fallback]
