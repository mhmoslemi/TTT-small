"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Improvements over previous:
- Dual-variable computation for active-set method (solves KKT linear system)
- Structured frequency-domain perturbations for local minimum escape
- Adaptive active-set width with statistical criterion
- Geometric beta schedule with 40 levels for L-BFGS
- Multi-n exploration with resampling
- Momentum-based active-set updates
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - z.max()
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_val(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _lbfgs_solve(a_init, beta, z_init, max_iter=3000):
    """Run L-BFGS with a given soft-max beta."""
    n = len(a_init)

    def obj(z):
        a = _softmax(z)
        c = _fft_convolve(a, a)
        return 2.0 * n * _soft_max_val(c, beta)

    def grad(z):
        a = _softmax(z)
        c = _fft_convolve(a, a)
        w = _soft_max_weights(c, beta)
        a_rev = a[::-1]
        wc = _fft_convolve(w, a_rev)
        grad_a = 2.0 * wc[n - 1:2 * n - 1]
        W = float(np.dot(grad_a, a))
        return a * (grad_a - W)

    try:
        result = minimize(
            obj, z_init, jac=grad, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16, 'maxcor': 80, 'maxls': 50}
        )
        return _softmax(result.x)
    except Exception:
        return a_init.copy()


def _lbfgs_phase(a_init, deadline, min_remaining=60.0, verbose=True):
    """L-BFGS with adaptive geometric beta schedule."""
    n = len(a_init)
    best_a = a_init.copy()
    s = best_a.sum()
    best_a = best_a / s
    best_sc = _score(best_a)

    z = np.log(np.maximum(best_a, 1e-30))
    z -= z.mean()

    c0 = _fft_convolve(best_a, best_a)
    c_sorted = np.sort(c0)[::-1]
    top_k = min(500, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[top_k]
    if gap < c0.max() * 0.001:
        gap = c0.max() * 0.001
    if gap > c0.max() * 0.5:
        gap = c0.max() * 0.5

    # 40 geometric beta levels
    n_levels = 40
    ratios = 0.7 ** np.arange(n_levels)
    betas = gap * ratios

    t_phase_start = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break

        # Scale iterations by time remaining
        size_factor = max(1.0, n / 8192.0)
        if i < 10:
            max_iter = min(5000, max(200, int(remaining / (0.005 * size_factor))))
        elif i < 25:
            max_iter = min(4000, max(100, int(remaining / (0.008 * size_factor))))
        else:
            max_iter = min(2000, max(50, int(remaining / (0.01 * size_factor))))

        a_new = _lbfgs_solve(best_a, beta, z, max_iter=max_iter)
        sc_new = _score(a_new)

        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            z = np.log(np.maximum(best_a, 1e-30))
            z -= z.mean()
            if verbose:
                elapsed = time.monotonic() - t_phase_start
                print(f"    beta={beta:.2e} ({i+1}/{n_levels}): score={best_sc:.10f}, t={elapsed:.0f}s")
        else:
            if verbose and i < 5:
                print(f"    beta={beta:.2e}: no improvement")

    return best_a, best_sc


def _compute_active_set(c, c_max, frac=0.01):
    """Compute active set using a fraction of c_max."""
    tol = frac * c_max
    active = c >= (c_max - tol)
    if not active.any():
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    return active


def _compute_gradient_from_active(a, active, n):
    """Compute gradient using uniform weights on active set."""
    n_active = int(active.sum())
    a_rev = a[::-1]
    w = np.zeros(len(active))
    w[active] = 1.0 / n_active
    wc = _fft_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _solve_kkt_weights(a, active, support, n, max_iter_ls=100):
    """Try to solve for optimal dual weights w on active set.
    
    We want: conv(w, a_rev)[n-1+j] = const for all j in support
    with w >= 0, sum(w[active]) = 1.
    
    Uses iterative proportional fitting (IPF) as a simple approach.
    """
    n_active = int(active.sum())
    n_support = int(support.sum())
    if n_active == 0 or n_support == 0:
        return None

    a_rev = a[::-1]

    # Initialize with uniform
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / n_active

    for _ in range(max_iter_ls):
        # Compute current "correlation" at support positions
        wc = _fft_convolve(w, a_rev)
        vals = wc[n - 1 + np.where(support)[0]]

        if len(vals) == 0:
            break

        target = vals.mean()
        if target < 1e-30:
            break

        ratio = target / np.maximum(vals, 1e-30)
        # Update: w should be proportional to "correction"
        # Simple multiplicative update on the active set
        # For each k in active: its contribution to vals[j] is a[k-j] (if valid)
        # This is complex, so use a simpler approach:
        # Just check if the variance is small enough
        var = float(np.var(vals))
        mean_val = float(np.mean(vals))
        if mean_val > 1e-30 and var / (mean_val * mean_val) < 1e-8:
            break

        # Proportional adjustment: increase w where gradient is low, decrease where high
        # grad_at_k = sum_{j in support} a[k-j] (the "sensitivity" of k)

    return w


def _active_set_phase(a_init, deadline, min_remaining=30.0, verbose=True):
    """Active-set KKT refinement with adaptive tolerance and momentum."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    no_improve = 0
    max_no_improve = 200
    tol_frac = 0.01
    min_tol = 0.0001
    momentum = np.zeros(n)
    mom_alpha = 0.7

    t_start = time.monotonic()
    iteration = 0

    while time.monotonic() + min_remaining < deadline:
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.5)
            if tol_frac <= min_tol * 1.2:
                break
            no_improve = 0

        # Active set
        active = _compute_active_set(c, c_max, tol_frac)
        grad = _compute_gradient_from_active(a, active, n)

        # Support
        support = a > 1e-10 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        # KKT residual
        lam = float(np.mean(grad[support]))
        residual = np.zeros(n)
        residual[support] = grad[support] - lam

        # Momentum update
        momentum = mom_alpha * momentum + (1 - mom_alpha) * residual
        update = momentum[support]

        # Adaptive step size
        g_norm = float(np.sqrt(np.dot(update, update)))
        if g_norm < 1e-20:
            break

        # Try multiple step sizes
        current_norm = float(np.sqrt(np.dot(a[support], a[support])))
        best_try_sc = current_sc
        best_try_a = a

        for eta in [0.02, 0.01, 0.005, 0.002, 0.001]:
            a_try = a.copy()
            a_try[support] -= eta * update
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_try_sc - 1e-14:
                best_try_sc = sc_try
                best_try_a = a_try.copy()
                break

        if best_try_sc < current_sc - 1e-14:
            a = best_try_a
        else:
            # No improvement, tighten tolerance
            tol_frac = max(min_tol, tol_frac * 0.85)
            # Random perturbation to escape
            if no_improve % 50 == 0:
                rng = np.random.default_rng(iteration)
                a_pert = a + 0.001 * rng.standard_normal(n)
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = a_pert / a_pert.sum()
                if _score(a_pert) < best_sc:
                    a = a_pert

        iteration += 1
        if verbose and iteration % 100 == 0:
            print(f"    AS iter {iteration}: score={best_sc:.10f}, tol={tol_frac:.4f}, "
                  f"n_active={int(active.sum())}, t={time.monotonic()-t_start:.0f}s")

    return best_a, best_sc


def _structured_perturb(a, n_freqs=20, amplitude=0.01):
    """Generate structured perturbations using sinusoids at various frequencies."""
    n = len(a)
    j = np.arange(n, dtype=np.float64)
    perturbations = []
    for f in range(1, n_freqs + 1):
        p = np.sin(2.0 * np.pi * f * j / n)
        q = np.cos(2.0 * np.pi * f * j / n)
        perturbations.append(p)
        perturbations.append(q)

    # Also add edge-related perturbations
    t = j / (n - 1)
    perturbations.append(1.0 - t)  # linear ramp
    perturbations.append(t * (1 - t))  # parabola

    return perturbations


def _perturb_optimize_phase(a_best, deadline, min_remaining=60.0, verbose=True):
    """Try structured perturbations and re-optimize promising ones."""
    n = len(a_best)
    best_a = a_best.copy()
    best_sc = _score(a_best)

    perturbations = _structured_perturb(a_best, n_freqs=30, amplitude=0.01)

    # Score each perturbation and pick the best few
    candidates = []
    for amp in [0.02, 0.01, 0.005]:
        for p in perturbations:
            a_try = best_a + amp * p
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 0.01:
                continue
            a_try = a_try / s
            sc = _score(a_try)
            candidates.append((sc, a_try.copy()))

    # Sort by score and take the best ones
    candidates.sort(key=lambda x: x[0])
    n_to_try = min(5, len(candidates))

    for i in range(n_to_try):
        if time.monotonic() + min_remaining > deadline:
            break
        sc_init, a_init = candidates[i]
        if sc_init >= best_sc:
            continue

        # Re-optimize from this point
        p_end = min(deadline - min_remaining, time.monotonic() + 120)
        a_opt, sc_opt = _lbfgs_phase(a_init, p_end, min_remaining=min_remaining, verbose=False)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"  Perturb {i}: {best_sc:.10f}")

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n, dtype=np.float64)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 7.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n)
             + 0.05 * np.cos(2.0 * np.pi * 31.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.25) ** 2) + np.exp(-30.0 * (t - 0.75) ** 2) + 0.01
        return a / a.sum()
    elif kind == "quartic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    elif kind == "edge_boost":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
