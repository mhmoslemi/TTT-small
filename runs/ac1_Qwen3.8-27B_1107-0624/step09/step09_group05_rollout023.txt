def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

        m = 50

    # A_active[i, j] = a[active_indices[i] - j] if 0 <= active_indices[i] - j < n, else 0
    A = np.zeros((m, n))

            idx = k - j
            if 0 <= idx < n:

    # Objective: minimize t => c = [0, 0, ..., 0, 1]
    #

    #   2*A@da - t <= -c_active  =>  [2*A, -1] @ [da, t] <= -c_active  (m constraints)
    #   da >= -a  =>  -da <= a  =>  [-I, 0] @ [da, t] <= a  (n constraints)

    #   sum(da) = 0  =>  [ones(n), 0] @ [da, t] = 0  (1 constraint)
    
    c_obj = np.zeros(n + 1)

    A_ub = np.zeros((m + n, n + 1))

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:
- FISTA (Nesterov accelerated) with projection as fast alternative to L-BFGS
- Aggressive Polyak-step active-set refinement with adaptive tolerance
- Targeted peak shaving heuristic
- Larger n exploration (up to 32768)
- Linearized active-set LP solve for final convergence
- Better time management focusing on final refinement

import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _proj_simplex(v):
    """Project v onto {a >= 0, sum(a) = 1} using bisection."""
    n = len(v)
    lo = float(v.min()) - 1.0
    hi = float(v.max())
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.maximum(v - mid, 0.0).sum()
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    a = np.maximum(v - tau, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def _soft_max_val(c, beta):
    c_max = float(c.max())
    if beta < 1e-15:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_smooth(a, beta, n):
    """Gradient of 2n*soft_max(conv(a,a)) w.r.t. a."""
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return 2.0 * n * grad


def _fista(a_init, beta, deadline, verbose=False):
    """FISTA with projection onto simplex."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s

    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())

    # Estimate L (Lipschitz constant of gradient)
    # grad[j] = 4n * sum_k w[k]*a[k-j], ||grad|| <= 4n * max(a) * sum(|w|)
    # For step size, use conservative L
    L = 8.0 * n * max(1.0, c_max / beta * 0.1)
    L = min(L, 1000.0 * n)

    x_curr = a.copy()
    x_prev = a.copy()
    t_val = 1.0
    n_iter = 0

    while time.monotonic() + 2.0 < deadline:
        # Nesterov extrapolation
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = x_curr + ((t_val - 1.0) / t_next) * (x_curr - x_prev)

        # Gradient at y
        c_y = _full_convolve(y, y)
        w_y = _soft_max_weights(c_y, beta)
        y_rev = y[::-1]
        wc_y = _full_convolve(w_y, y_rev)
        grad_y = 4.0 * n * wc_y[n - 1:2 * n - 1]

        # Backtracking line search (limited)
        step = 1.0 / L
        best_trial = None
        f_y = 2.0 * n * _soft_max_val(c_y, beta)
        for _bt in range(8):
            x_trial = _proj_simplex(y - step * grad_y)
            c_xt = _full_convolve(x_trial, x_trial)
            f_xt = 2.0 * n * _soft_max_val(c_xt, beta)
            grad_dot = 4.0 * n * float(np.dot(wc_y[n - 1:2 * n - 1], x_trial - y))
            quad = 0.5 * (1.0 / step) * float(np.dot(x_trial - y, x_trial - y))
            if f_xt <= f_y + grad_dot + quad + 1e-9:
                best_trial = x_trial
                break
            step *= 0.5
            L *= 2.0
        if best_trial is None:
            best_trial = _proj_simplex(y - step * grad_y)

        x_prev = x_curr.copy()
        x_curr = best_trial
        t_val = t_next
        n_iter += 1

        sc = _score(x_curr)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = x_curr.copy()

        if verbose and n_iter % 500 == 0:
            print(f"    FISTA it={n_iter}, L={L:.1f}, sc={best_sc:.10f}")

    return best_a, best_sc


class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        return 2.0 * self.n * _soft_max_val(c, self.beta)

    def grad(self, z):
        a = _softmax(z)
        grad_a = _grad_smooth(a, self.beta, self.n)
        W = float(np.dot(grad_a, a))
        return a * (grad_a - W)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0, verbose=True):
    """L-BFGS with adaptive beta schedule."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c0_max * 0.003:
        gap = c0_max * 0.003

    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
        0.00003125, 0.000015625, 0.0000078, 0.0000039,
        0.000002, 0.000001, 0.0000005, 0.00000025
    ])
    betas = gap * multipliers

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        size_factor = max(1.0, n / 4096.0)
        est_iter_time = 0.002 * size_factor
        max_iter = min(10000, max(200, int(remaining / (est_iter_time * 15))))

        if beta < gap * 0.001:
            max_iter = min(max_iter, 3000)

        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 80,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"    beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}")
        except Exception as e:
            if verbose and i < 5:
                print(f"    beta={beta:.2e}: error {e}")

    return best_a, best_sc


def _active_set_refine_aggressive(a_init, deadline, min_time_remaining=30.0):
    """Aggressive active-set refinement with Polyak step and adaptive tolerance."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lb = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound

    tol_frac = 0.008
    min_tol = 0.00005
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.5)
            if tol_frac <= min_tol * 1.1:
                # Try wider active set
                tol_frac = 0.02
            no_improve = 0

        # Identify active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        # Cap active set size for stability
        if n_active > 2000:
            threshold = np.sort(c)[::-1][2000]
            active = c >= threshold
            n_active = int(active.sum())

        # Compute average gradient from active set
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]

        # Support of a
        support = a > 1e-10 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        # KKT: grad_avg should be constant on support
        lam = float(np.mean(grad_avg[support]))
        residual = np.zeros(n)
        residual[support] = grad_avg[support] - lam

        # Polyak-like step size
        gap = current_sc - lb
        if gap < 1e-12:
            break
        r_norm_sq = float(np.dot(residual, residual))
        if r_norm_sq < 1e-20:
            break
        step = 0.5 * gap / (r_norm_sq * n)
        step = min(step, 0.05)
        step = max(step, 0.001)

        # Try the step
        a_new = a.copy()
        a_new[support] -= step * residual[support]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new >= current_sc - 1e-13:
            # Try smaller steps
            found = False
            for trial_frac in [0.5, 0.25, 0.1, 0.05, 0.02]:
                a_try = a.copy()
                a_try[support] -= step * trial_frac * residual[support]
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-13:
                    a_new = a_try
                    sc_new = sc_try
                    found = True
                    break
            if not found:
                tol_frac = max(min_tol, tol_frac * 0.85)
                continue

        a = a_new

        if no_improve == 0:
            tol_frac = max(min_tol, tol_frac * 0.96)

    return best_a, best_sc


def _peak_shaving(a_init, deadline, min_time_remaining=30.0):
    """Targeted peak shaving: reduce the max peak by adjusting a."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        # Find contributions to c[k_star]
        # c[k_star] = sum_j a[j] * a[k_star - j] for valid indices
        # d(c[k_star])/d(a[j]) = 2 * a[k_star - j] (for valid k_star - j)

        # Identify top contributors
        contrib = np.zeros(n)
        j_lo = max(0, k_star - (n - 1))
        j_hi = min(n - 1, k_star)
        for j in range(j_lo, j_hi + 1):
            contrib[j] = 2.0 * a[k_star - j]

        total_contrib = float(contrib.sum())
        if total_contrib < 1e-15:
            break

        # Normalize contributions to get "importance"
        importance = contrib / total_contrib

        # We want to reduce a[j] proportional to importance, but not too much
        # Try reducing the top contributors
        top_k = min(200, n)
        top_idx = np.argsort(contrib)[::-1][:top_k]

        # Compute how much other peaks would be affected
        # For each j in top_idx, reduce a[j] by delta
        # This reduces c[k_star] by 2*a[k_star-j]*delta
        # But may increase other c[k] values
        # Simple: just try a small reduction

        delta_base = 0.01 * a.mean()
        a_new = a.copy()
        for j in top_idx:
            a_new[j] -= delta_base * importance[j] * 10.0
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        current_sc = _score(a)

        if sc_new < current_sc - 1e-13:
            a = a_new
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
        else:
            # Try smaller delta
            improved = False
            for scale in [0.5, 0.25, 0.1, 0.05]:
                a_try = a.copy()
                for j in top_idx:
                    a_try[j] -= delta_base * scale * importance[j] * 10.0
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-13:
                    a = a_try
                    if sc_try < best_sc:
                        best_sc = sc_try
                        best_a = a_try.copy()
                    improved = True
                    break
            if not improved:
                # Try reducing from the bottom (least important for k_star)
                # to maintain sum without increasing k_star much
                bot_idx = np.argsort(contrib)[:top_k]
                a_try = a.copy()
                boost = delta_base * 0.5
                for j in top_idx[:50]:
                    a_try[j] -= boost
                a_try[bot_idx[:50]] += boost * 50.0 / len(bot_idx[:50])
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    break
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-13:
                    a = a_try
                    if sc_try < best_sc:
                        best_sc = sc_try
                        best_a = a_try.copy()
                    improved = True
            if not improved:
                break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.6 * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _random_perturb(a, strength=0.01, seed=0):
    """Add structured random perturbation."""
    rng = np.random.default_rng(seed)
    n = len(a)
    # Smooth random perturbation
    noise = rng.normal(0, 1, n)
    # Smooth it
    kernel_size = max(3, n // 500)
    kernel = np.ones(kernel_size) / kernel_size
    noise_smooth = np.convolve(noise, kernel, mode='same')
    noise_smooth = noise_smooth / (np.std(noise_smooth) + 1e-15)
    a_new = a + strength * noise_smooth
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def _active_set_lp_solve(a_init, max_active=200):
    """Solve linearized minimax LP for the active set. Returns improved a."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c = _full_convolve(a, a)
    c_max = float(c.max())

    # Identify active set (top peaks)
    tol = 0.005 * c_max
    active = c >= (c_max - tol)
    n_active = int(active.sum())
    if n_active > max_active:
        threshold = np.sort(c)[::-1][max_active]
        active = c >= threshold
        n_active = int(active.sum())
    if n_active < 2:
        return a

    active_indices = np.where(active)[0]

    # Build the m x n matrix A where A[i,j] = a[k_i - j] if valid
    m = len(active_indices)
    # For efficiency, use the convolution structure
    # A[i, j] = a[active_indices[i] - j] for 0 <= active_indices[i] - j < n
    # This is a slice of a for each row
    A_matrix = np.zeros((m, n))
    for i, k in enumerate(active_indices):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        idx = k - np.arange(j_lo, j_hi + 1)
        A_matrix[i, j_lo:j_hi + 1] = a[idx]

    # LP: minimize t
    # Variables: [da_0, ..., da_{n-1}, t]
    # Constraints:
    #   2 * A_matrix @ da - t <= -c[active_indices]  (m inequalities)
    #   -da <= a  (i.e., da >= -a)  (n inequalities)
    #   sum(da) = 0  (1 equality)
    #   da free, t free

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # Inequality 1: [2*A, -1] @ [da, t] <= -c_active
    A_ub = np.zeros((m + n, n + 1))
    b_ub = np.zeros(m + n)
    A_ub[:m, :n] = 2.0 * A_matrix
    A_ub[:m, n] = -1.0
    b_ub[:m] = -c[active_indices]

    # Inequality 2: -da <= a
    A_ub[m:, :n] = -np.eye(n)
    b_ub[m:] = a

    # Equality: sum(da) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] * (n + 1)

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'time_limit': 60})
        if result.success:
            da = result.x[:n]
            a_new = a + da
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                if _score(a_new) < _score(a) - 1e-12:
                    return a_new
    except Exception:
        pass

    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer with aggressive active-set refinement.
    Phases:
    P1: L-BFGS main optimization (adaptive beta)
    P2: Aggressive active-set refinement
    P3: L-BFGS round 2 (tighter beta)
    P4: Peak shaving + FISTA
    P5: Multi-n exploration
    P6: Random perturbations + re-optimize
    P7: Final active-set LP + polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(16384, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n=16384: score={best_score:.10f}")

    n_main = len(best_seq)

    # === Phase 1: Main L-BFGS ===
    print(f"\n[P1] L-BFGS main (n={n_main}, t={time.monotonic()-t_start:.0f}s)")
    p1_end = t_start + min(400, (budget_s - 10) * 0.25)
    a1, sc1 = _run_lbfgs(best_seq, p1_end, min_time_remaining=200, verbose=True)
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  P1 result: {best_score:.10f}")

    # === Phase 2: Aggressive Active-Set Refinement ===
    print(f"\n[P2] Active-set refine (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p2_end = min(deadline - 400, time.monotonic() + 300)
        a2, sc2 = _active_set_refine_aggressive(best_seq, p2_end, min_time_remaining=50)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(f"  P2 result: {best_score:.10f}")

    # === Phase 3: L-BFGS round 2 ===
    print(f"\n[P3] L-BFGS round 2 (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p3_end = min(deadline - 300, time.monotonic() + 250)
        a3, sc3 = _run_lbfgs(best_seq, p3_end, min_time_remaining=100, verbose=True)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3 result: {best_score:.10f}")

    # === Phase 4: Peak shaving + FISTA ===
    print(f"\n[P4] Peak shaving + FISTA (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p4a_end = min(deadline - 200, time.monotonic() + 150)
        a4a, sc4a = _peak_shaving(best_seq, p4a_end, min_time_remaining=30)
        if sc4a < best_score:
            best_score = sc4a
            best_seq = a4a.copy()
            print(f"  P4a (peak shaving): {best_score:.10f}")

        if time.monotonic() + 30 < deadline:
            # FISTA for a quick polish
            c_tmp = _full_convolve(best_seq, best_seq)
            gap_tmp = float(c_tmp.max()) * 0.001
            p4b_end = min(deadline - 100, time.monotonic() + 100)
            a4b, sc4b = _fista(best_seq, gap_tmp, p4b_end, verbose=False)
            if sc4b < best_score:
                best_score = sc4b
                best_seq = a4b.copy()
                print(f"  P4b (FISTA): {best_score:.10f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n (t={time.monotonic()-t_start:.0f}s)")
    n_options = [12288, 20480, 24576, 32768]
    n_options = [x for x in n_options if x != n_main]
    for n_try in n_options:
        if time.monotonic() + 100 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        print(f"  n={n_try}: resampled score={sc_r:.10f}")
        if sc_r < best_score:
            best_score = sc_r
            best_seq = a_r.copy()
        if time.monotonic() + 80 > deadline:
            continue
        p5_end = min(deadline - 60, time.monotonic() + 80)
        a5, sc5 = _run_lbfgs(a_r, p5_end, min_time_remaining=20, verbose=False)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  n={n_try} optimized: {best_score:.10f}")

    # === Phase 6: Random perturbations ===
    print(f"\n[P6] Random perturbations (t={time.monotonic()-t_start:.0f}s)")
    n_best = len(best_seq)
    for pert_seed in [1, 7, 42, 123]:
        if time.monotonic() + 50 > deadline:
            break
        a_pert = _random_perturb(best_seq, strength=0.005, seed=pert_seed)
        sc_pert = _score(a_pert)
        p6_end = min(deadline - 40, time.monotonic() + 50)
        a6, sc6 = _run_lbfgs(a_pert, p6_end, min_time_remaining=10, verbose=False)
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  Perturbation seed={pert_seed}: {best_score:.10f}")

    # === Phase 7: Final polish ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 20 < deadline:
        # Active-set LP solve
        p7a_end = time.monotonic() + 90
        while time.monotonic() + 10 < min(deadline, p7a_end):
            a_lp = _active_set_lp_solve(best_seq, max_active=150)
            sc_lp = _score(a_lp)
            if sc_lp < best_score - 1e-12:
                best_score = sc_lp
                best_seq = a_lp.copy()
                print(f"  LP solve improved: {best_score:.10f}")
            else:
                break

        # Final active-set refinement
        if time.monotonic() + 10 < deadline:
            p7b_end = min(deadline - 5, time.monotonic() + 60)
            a7b, sc7b = _active_set_refine_aggressive(best_seq, p7b_end, min_time_remaining=5)
            if sc7b < best_score:
                best_score = sc7b
                best_seq = a7b.copy()
                print(f"  Final active-set: {best_score:.10f}")

        # Very final L-BFGS
        if time.monotonic() + 15 < deadline:
            p7c_end = min(deadline - 3, time.monotonic() + 40)
            a7c, sc7c = _run_lbfgs(best_seq, p7c_end, min_time_remaining=3, verbose=False)
            if sc7c < best_score:
                best_score = sc7c
                best_seq = a7c.copy()
                print(f"  Final L-BFGS: {best_score:.10f}")

    # Finalize
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    # Verify with evaluate_sequence if available
    try:
        eval_score = evaluate_sequence([float(x) for x in final])
        print(f"Verified with evaluate_sequence: {eval_score:.10f}")
    except Exception:
        pass

    return [float(x) for x in final]
