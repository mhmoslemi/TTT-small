```python
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy import sparse

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s
```

```python
def _fista_topk(z_init, n, beta, K, max_iter, deadline):
    """FISTA for top-K smoothed max convolution objective."""
    z = z_init.copy()
    z = z - z.max()
    y = z.copy()
    t = 1.0
    best_obj = np.inf
    best_z = z.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute a from y
        e = np.exp(y - y.max())
        a = e / e.sum()
        
        # Compute c = conv(a, a)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Top-K weights
        if K < len(c):
            top_idx = np.argpartition(c, -K)[-K:]
            c_top = c[top_idx]
        else:
            top_idx = np.arange(len(c))
            c_top = c
        
        shifted = (c_top - c_max) / beta
        shifted = np.clip(shifted, -500, 0)
        exps = np.exp(shifted)
        s = exps.sum()
        
        if s < 1e-300:
            w_clean = np.zeros(len(c))
            w_clean[top_idx[0]] = 1.0
            obj = 2.0 * n * c_max
        else:
            w_full = np.zeros(len(c))
            w_full[top_idx] = exps / s
            w_clean = w_full
            obj = 2.0 * n * (c_max + beta * np.log(s))
        
        if obj < best_obj:
            best_obj = obj
            best_z = y.copy()
        
        # Gradient of obj w.r.t. a
        ga = 2.0 * n * _grad_maxconv(a, w_clean)
        
        # Chain rule: grad_z = a * (ga - dot(a, ga))
        W = float(np.dot(a, ga))
        gz = a * (ga - W)
        
        # Lipschitz estimate (use a conservative bound or backtracking)
        # For now, use a fixed step based on problem size
        L = 4.0 * n * np.max(np.abs(ga)) + 1e-6  # rough estimate
        step = 1.0 / L
        
        # FISTA update
        x_new = y - step * gz
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - z)
        z = x_new
        t = t_new
    
    # Final: convert best_z to a
    e = np.exp(best_z - best_z.max())
    a_best = e / e.sum()
    return a_best, _score(a_best)
```

Let me also think about a key mathematical insight. The optimal sequence for minimizing max autocorrelation of a non-negative sequence is related to the roots of certain polynomials. Specifically, if we let A(z) = sum a[i] z^i, then the autocorrelation is related to |A(z)|^2 on the unit circle. Minimizing the max of |A(z)|^2 is related to the "minimax" polynomial approximation problem.

For non-negative a, A(z) has all real non-negative coefficients. The problem of minimizing max_{|z|=1} |A(z)|^2 is related to the "Fejér" or "Riesz" problems in approximation theory.

The optimal solution (in the limit) is related to the "flat-top" window or the "Blackman-Harris" window, which have very low sidelobes.

However, for our problem, we're minimizing the MAX of the convolution (which is the autocorrelation), not the sidelobes. The peak at lag 0 is always the "main lobe" and we want ALL lags (including lag 0) to be small.

This is equivalent to minimizing the L-infinity norm of the periodogram |A(e^{iω})|^2, which for non-negative a is a constrained approximation problem.

The optimal solution is known to be related to the "eigenfunction" of a certain integral operator. For the continuous analog, it's the prolate spheroidal wave function (Slepian function). For the discrete case, it's the discrete prolate spheroidal sequence (DPSS).

This is a great insight! The optimal sequence should be close to a DPSS (or Slepian) function!

The DPSS are the eigenvectors of the time-frequency concentration operator. The first DPSS (for a given time-bandwidth product) is the most "concentrated" in both time and frequency, which means it has a nearly flat spectrum (and hence nearly flat autocorrelation).

Let me use DPSS as an initialization!

The DPSS can be computed as the eigenvector of the matrix:
Q = N × N matrix where Q[i,j] = sin(2*pi*W*(i-j)) / (pi*(i-j)) for i≠j, and Q[i,i] = 2*W

where W is the time-bandwidth product (as a fraction of N).

For our problem, we want the sequence to have a "flat" spectrum, which means W should be close to N/2 (the maximum possible bandwidth). But for non-negative sequences, the effective bandwidth is limited.

Actually, I think the connection to DPSS is:
- We want min_a max_k c[k] = min_a max_ω |A(e^{iω})|^2 (by Parseval, up to constants)
- This is equivalent to finding a non-negative a that minimizes the peak of its periodogram
- The unconstrained optimal solution (allowing negative values) is the first DPSS
- The constrained solution (a >= 0) is what we're looking for

So the first DPSS gives us a great starting point! Let me compute it.

For a DPSS with time support [0, N-1] and bandwidth W (cycles per unit), the eigenvector v satisfies:
sum_j sin(2*pi*W*(i-j)) / (pi*(i-j)) * v[j] = lambda * v[i]

The largest eigenvalue is the "time-bandwidth product" W (approximately), and the corresponding eigenvector is the first DPSS.

For our problem, we want the sequence to be "as flat as possible" in frequency, which means we want a large W. The maximum W for a real sequence of length N is N/2 (Nyquist). But practically, W = 0.4*N or so gives a good "flat" spectrum.

Let me compute the DPSS using the eigendecomposition of the sinc matrix. For N=16384, this is a 16384×16384 matrix, which is too large for a full eigendecomposition. But we only need the top eigenvector, which can be found by power iteration or Lanczos.

Actually, for a Toeplitz matrix like this, we can use FFT-based matvec:
(Qv)[i] = sum_j sinc(2W(i-j)) * v[j]

This is a convolution (since sinc(2W(i-j)) depends only on i-j), so it can be computed with FFT in O(N log N)!

So the matvec for the DPSS eigenproblem is:
Qv = IFFT(FFT(sinc_kernel) * FFT(v))

where sinc_kernel[k] = sin(2*pi*W*k) / (pi*k) for k≠0, and 2W for k=0.

We can find the top eigenvector by power iteration with a few dozen iterations (since the gap between the top eigenvalue and the rest is large for W >> 1).

```python
"""
Multi-phase optimizer: FISTA + Active-set LP + Subgradient polish
for minimizing 2*n*max(conv(a,a))/sum(a)^2 with non-negative a.
Key innovations:
- Nesterov-accelerated FISTA with adaptive top-K smoothing
- Active-set LP for conditional gradient "jumps"
- Multi-scale initialization (optimize at n/4, upscale)
- DPSS-inspired initialization
- Adaptive beta/K annealing with stall detection

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_conv_weighted(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c=conv(a,a). Length n."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _topk_smooth_obj_grad(z, beta, K, n):
    """Compute smoothed objective and gradient in log-domain."""
    e = np.exp(z - z.max())
    a = e / e.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())

    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c

    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -500, 0)
    exps = np.exp(shifted)
    s = exps.sum()

    if s < 1e-300:
        w_full = np.zeros(len(c))
        w_full[top_idx[np.argmax(c_top)]] = 1.0
        obj = 2.0 * n * c_max
    else:
        w_full = np.zeros(len(c))
        w_full[top_idx] = exps / s
        obj = 2.0 * n * (c_max + beta * np.log(s))

    ga = _grad_conv_weighted(a, w_full)
    W = float(np.dot(a, ga))
    gz = a * (ga - W)
    return obj, gz


def _fista_optimize(a_init, deadline, min_time_remaining=60.0, n_phases=8):
    """FISTA with adaptive top-K smoothing and beta/K annealing."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.01)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    schedule = [
        (1.5, 0.15), (1.0, 0.12), (0.7, 0.09), (0.5, 0.07),
        (0.3, 0.05), (0.2, 0.04), (0.1, 0.03), (0.05, 0.02),
        (0.03, 0.015), (0.01, 0.01), (0.005, 0.008), (0.002, 0.005),
        (0.001, 0.003), (0.0005, 0.002),
    ]

    time_per_phase = max(10.0, (deadline - min_time_remaining - time.monotonic()) / n_phases)
    n_sched = min(len(schedule), n_phases)

    for phase in range(n_sched):
        if time.monotonic() + min_time_remaining > deadline:
            break
        phase_deadline = min(deadline - min_time_remaining, time.monotonic() + time_per_phase)
        if phase_deadline - time.monotonic() < 5:
            break

        beta_mult, K_frac = schedule[phase]
        beta = base_beta * beta_mult
        K = max(5, int((2 * n - 1) * K_frac))

        est_per_iter = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(20000, max(100, int((phase_deadline - time.monotonic()) / est_per_iter)))

        z_y = z.copy()
        z_x = z.copy()
        t_nest = 1.0
        step = 0.1
        best_local_z = z_y.copy()
        best_local_obj = _score(a)
        stall = 0
        max_stall = 200

        for it in range(max_iter):
            if time.monotonic() > phase_deadline:
                break

            obj, gz = _topk_smooth_obj_grad(z_y, beta, K, n)

            if obj < best_local_obj - 1e-12:
                best_local_obj = obj
                best_local_z = z_y.copy()
                stall = 0
            else:
                stall += 1
                if stall > max_stall:
                    z_y = best_local_z.copy()
                    z_x = z_y.copy()
                    t_nest = 1.0
                    step = max(0.01, step * 0.5)
                    stall = 0
                    max_stall = max(50, max_stall // 2)
                    continue

            gnorm = np.linalg.norm(gz)
            if gnorm < 1e-14:
                break

            # Backtracking line search
            alpha = step
            improved = False
            for ls in range(15):
                z_new = z_y - alpha * gz
                e_new = np.exp(z_new - z_new.max())
                a_new = e_new / e_new.sum()
                c_new = _full_convolve(a_new, a_new)
                sc_new = 2.0 * n * float(c_new.max())
                if sc_new < obj - 1e-15 or ls == 14:
                    improved = True
                    break
                alpha *= 0.5

            if not improved:
                break

            z_new_check = z_y - alpha * gz
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
            z_momentum = z_new_check + ((t_nest - 1.0) / t_new) * (z_new_check - z_x)

            # Check if momentum point is good
            e_m = np.exp(z_momentum - z_momentum.max())
            a_m = e_m / e_m.sum()
            c_m = _float_c_max(a_m)
            if c_m * 2.0 * n < obj + 0.01 * beta:
                z_x = z_new_check
                z_y = z_momentum
            else:
                z_x = z_new_check
                z_y = z_new_check
                t_nest = 1.0

            z = z_y.copy()
            step = min(0.5, alpha * 1.1)

        a_check = _z_to_a(best_local_z)
        sc_check = _score(a_check)

            current_a = a_check
            current_sc = sc_check
            z = np.log(np.maximum(current_a, 1e-30))
            z = z - z.mean()

    return current_a


def _float_c_max(a):
    c = _full_convolve(a, a)
    return float(c.max())


def _z_to_a(z):
    e = np.exp(z - z.max())
    return e / e.sum()


def _active_set_lp(a, M, active_idx, n):
    """Solve LP: max sum(b) s.t. conv(a,b)[k] <= M for k in active_idx, b >= 0."""
    # Build constraint matrix A: A[k,j] = a[k-j] if 0<=k-j<=n-1 else 0
    n_constraints = len(active_idx)
    rows = []
    cols = []
    data = []

    for i, k in enumerate(active_idx):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            val = a[k - j]
            if val > 0:
                rows.append(i)
                cols.append(j)
                data.append(val)

    if len(data) == 0:
        return None

    A_csc = sp.csc_matrix((data, (rows, cols)), shape=(n_constraints, n))

    # Objective: maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)

    # Bounds: b >= 0
    bounds = [(0, None)] * n

    # Solve
    try:
        res = linprog(c_obj, A_ub=A_csc, b_ub=np.full(n_constraints, M),
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max(5, int(30))})
        if res.status == 0 and res.x is not None:
            b = np.maximum(res.x, 0)
            s = b.sum()
            if s > 1e-10:
                return b / s
    except Exception:
        pass
    return None

def _subgradient_polish(a_init, deadline, min_time_remaining=30.0):

                    break

def _make_dpss_init(n, num_iter=80):
    """Generate DPSS-inspired initialization using power iteration."""
    W = 0.4 * n
    idx = np.arange(n)

    # Sinc kernel for Toeplitz matrix
    k_diff = idx[:, None] - idx[None, :]
    # We can't form the full n×n matrix for large n, so use FFT-based matvec
    kernel_len = 2 * n - 1
    k = np.arange(-n + 1, n)
    sinc_kernel = np.zeros(kernel_len)
    sinc_kernel[n] = 2 * W
    mask = k != 0
    sinc_kernel[mask + n - 1] = np.sin(2 * np.pi * W * k[mask]) / (np.pi * k[mask])

    rng = np.random.default_rng(123)
    v = rng.standard_normal(n)
    v = v / np.linalg.norm(v)

    for _ in range(num_iter):
        # Qv = conv(sinc_kernel, v) evaluated at positions n-1 to 2n-2
        Qv = _full_convolve(sinc_kernel, v)[n - 1:2 * n - 1]
        norm = np.linalg.norm(Qv)
        if norm < 1e-15:
            break
        v_new = Qv / norm
        # Power iteration convergence check
        if np.linalg.norm(v_new - v) < 1e-10:
            v = v_new
            break
        v = v_new

    # Make non-negative: take absolute value
    v = np.abs(v)
    v = np.maximum(v, 1e-10)
    return v / v.sum()


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_linear":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 0.8 + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_exp":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t) + 0.1
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2) + 0.03
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "dpss":
        return _make_dpss_init(n)
    elif kind == "gauss_off":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.4) ** 2) + 0.01
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _peak_targeted_reduction(a_init, deadline, min_time_remaining=30.0):
    """Directly target and reduce the top peaks."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    patience = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > 100:
                break

        # Find top peaks
        n_peaks = min(10, len(c))
        top_peaks = np.argpartition(c, -n_peaks)[-n_peaks:]
        top_peaks = top_peaks[np.argsort(c[top_peaks])[::-1]]

        # Build combined weight on top peaks
        w = np.zeros(len(c))
        peak_vals = c[top_peaks]
        pmax = peak_vals[0]
        if pmax > 0:
            w[top_peaks] = np.exp((peak_vals - pmax) / (pmax * 0.05))
        w = w / w.sum()

        g = _grad_conv_weighted(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-15:
            break

        # Line search
        alpha = min(0.02, 0.5 / (gnorm + 1e-10))
        improved = False
        for ls in range(25):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = 2.0 * n * _float_c_max(a_try)
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            patience += 5

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: FISTA + Active-set LP + Subgradient polish.
    Uses Nesterov acceleration, adaptive smoothing, and LP-based jumps.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        best_seq = _make_profile(n_parent, "tapered")
        best_score = _score(best_seq)
        print(f"No parent, tapered n={n_parent}, score={best_score:.10f}")

    n = len(best_seq)

    # === Phase 1: FISTA on parent ===
    print(f"\n[P1] FISTA (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        a1 = _fista_optimize(best_seq, deadline, min_time_remaining=600, n_phases=8)
        sc1 = _score(a1)
        if sc1 < best_score:
            best_seq = a1.copy()
            best_score = sc1
            print(f"  NEW BEST: {best_score:.10f} [fista1] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Active-set LP ===
    print(f"\n[P2] Active-set LP (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        a2, sc2 = _lp_based_optimize(best_seq, min(deadline - 450, time.monotonic() + 300),
                                     min_time_remaining=50)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [lp1] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: FISTA round 2 ===
    print(f"\n[P3] FISTA round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a3 = _fista_optimize(best_seq, min(deadline - 350, time.monotonic() + 250),
                             min_time_remaining=40, n_phases=6)
        sc3 = _score(a3)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [fista2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Peak-targeted reduction ===
    print(f"\n[P4] Peak reduction (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a4, sc4 = _peak_targeted_reduction(best_seq, min(deadline - 250, time.monotonic() + 200),
                                           min_time_remaining=30)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [peakred] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Active-set LP round 2 ===
    print(f"\n[P5] Active-set LP 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        a5, sc5 = _lp_based_optimize(best_seq, min(deadline - 200, time.monotonic() + 200),
                                     min_time_remaining=40)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [lp2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-restart with diverse profiles ===
    print(f"\n[P6] Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        n = len(best_seq)
        profiles = ["dpss", "asym_linear", "asym_exp", "gauss_off", "bimodal", "cos_taper"]
        sigmas = [0.03, 0.05, 0.08, 0.12, 0.06, 0.1]

        for i, (prof_kind, sigma) in enumerate(zip(profiles, sigmas)):
            if time.monotonic() > deadline - 60:
                break
            # Try DPSS or profile directly
            a_prof = _make_profile(n, prof_kind)
            a_r = _fista_optimize(a_prof, min(deadline - 50, time.monotonic() + 60),
                                  min_time_remaining=5, n_phases=4)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.10f} [profile_{prof_kind}] t={time.monotonic()-t_start:.1f}s")

            # Try perturbed best
            a_pert = _perturb(best_seq, sigma, rng)
            a_p = _fista_optimize(a_pert, min(deadline - 40, time.monotonic() + 40),
                                  min_time_remaining=5, n_phases=3)
            sc_p = _score(a_p)
            if sc_p < best_score:
                best_seq = a_p.copy()
                best_score = sc_p
                print(f"  NEW BEST: {best_score:.10f} [pert_{i}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        n_main = len(best_seq)
        n_options = [16384, 12288, 20480, 24576]
        n_options = [x for x in n_options if x != n_main and x <= 32768]

        for n_try in n_options[:2]:
            if time.monotonic() > deadline - 40:
                break
            a_res = _resample(best_seq, n_try)
            a_opt = _fista_optimize(a_res, min(deadline - 30, time.monotonic() + 60),
                                    min_time_remaining=10, n_phases=5)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [n={n_try}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final subgradient polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 10:
        a_fin, sc_fin = _subgradient_polish(best_seq, deadline - 5, min_time_remaining=5)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 9: Tiny LP polish ===
    print(f"\n[P9] LP polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_lp, sc_lp = _lp_based_optimize(best_seq, deadline - 2, min_time_remaining=2)
        if sc_lp < best_score:
            best_seq = a_lp.copy()
            best_score = sc_lp
            print(f"  NEW BEST: {best_score:.10f} [lp_final] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        # Use the actual evaluate function for validation
        val_sc = evaluate_sequence([float(x) for x in final])
        print(f"VALIDATED: score={val_sc:.10f}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
