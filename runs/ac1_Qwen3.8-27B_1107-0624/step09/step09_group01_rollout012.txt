    e = np.exp(np.clip(z, -700, 700))

def _grad_weighted(a, W):

    Result = 2 * conv(W, a[::-1])[n-1 : 2n-1]
    """
    n = len(a)
    a_rev = a[::-1]
    wc = _conv(W, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]

def _smoothed_obj_grad(z, beta, n):
    """Objective and gradient of smoothed log-sum-exp approximation."""
    a = _softmax(z)
    c = _conv(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0.0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        W = np.zeros_like(c)
        W[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        W = e / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    grad_a = _grad_weighted(a, W)
    # Convert to z-gradient: grad_z = a * (grad_a - mean(a*grad_a)) for softmax
    grad_z = a * (grad_a - np.mean(a * grad_a))
    grad_z = 2.0 * n * grad_z
    return obj, grad_z

    z = z - z.mean()

    current_a = a_init.copy()
    current_sc = _score(current_a)

        def obj_fn(z_in, beta=beta, n=n):

            c_in = _conv(a_in, a_in)
            cm = float(c_in.max())
            sh = (c_in - cm) / beta
            ee = np.exp(np.clip(sh, -700, 0.0))

            c_in = _conv(a_in, a_in)

            gz = a_in * (ga - np.mean(a_in * ga))

        scale = max(1.0, n / 8192.0)

        elif new_gap < 1e-4 * c_max_new:
            beta *= 0.15
        else:

                x = x + alpha * p_vec
                r = r - alpha * Hp
                rs_new = float(r @ r)
                if not np.isfinite(rs_new) or rs_new > rs * 1.1:
                    break
                p_vec = (rs_new / rs) * p_vec + r
                rs = rs_new
                if rs < 1e-14 * rs0:
                    break
            dx = -x
            dx = dx - np.mean(dx)
            dxn = float(np.linalg.norm(dx))
            if dxn < 1e-30 or not np.isfinite(dxn):
                dx = -grad_a / gn
                dxn = 1.0

        # Line search
        alpha = 1.0
        step_done = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _conv(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not step_done:
            for trial in [0.1, 0.05, 0.01, 0.005, 0.001]:
                a_try = a - trial * grad_a / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _conv(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 10:
            break

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.02
        return a / a.sum()
    elif kind == "u_shape":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * (2 * t - 1) ** 2
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.where(np.abs(t - 0.5) < 0.4, 1.0, 0.3)
        # Smooth transition
        a = np.convolve(a, np.ones(100) / 100, mode='same')
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"imp={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                par_sc = _score(parent)
                print(f"Parent: n={len(parent)}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        parent = _make_profile(32768, "tapered")
        par_sc = _score(parent)
        print(f"No parent, using tapered n=32768, score={par_sc:.10f}")
        update_best(parent, par_sc, "init")

    n0 = len(parent)
    print(f"Starting optimization from n={n0}, score={best_score:.10f}")

    # Phase 1: Subgradient with momentum (fast, good for initial progress)
    print(f"\n[Phase 1] Subgrad+momentum (t={time.monotonic() - t_start:.1f}s)")
    t_p1 = min(deadline - 1200, t_start + 400)
    a_sg, sc_sg = _subgrad_momentum(best_seq, t_p1, max_steps=30000)
    update_best(a_sg, sc_sg, "subgrad_1")

    # Phase 2: Newton active set
    print(f"\n[Phase 2] Newton active-set (t={time.monotonic() - t_start:.1f}s)")
    t_p2 = min(deadline - 900, t_start + 700)
    a_np, sc_np = _newton_active_set(best_seq, t_p2, max_rounds=300)
    update_best(a_np, sc_np, "newton_1")

    # Phase 3: Smoothed L-BFGS
    print(f"\n[Phase 3] Smoothed L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    t_p3 = min(deadline - 500, t_start + 1200)
    a_lb, sc_lb = _run_smoothed_lbfgs(best_seq, t_p3, min_remaining=400.0)
    update_best(a_lb, sc_lb, "lbfgs_1")

    # Phase 4: Cycle (subgrad -> newton -> lbfgs)
    print(f"\n[Phase 4] Cycle (t={time.monotonic() - t_start:.1f}s)")
    cycle_end = min(deadline - 200, t_start + 1600)
    for cyc in range(6):
        if time.monotonic() > cycle_end:
            break
        # Subgradient burst
        t_sub = min(cycle_end, time.monotonic() + 30)
        a_c1, sc_c1 = _subgrad_momentum(best_seq, t_sub, max_steps=5000)
        update_best(a_c1, sc_c1, f"cyc_sg_{cyc}")

        # Newton burst
        t_new = min(cycle_end, time.monotonic() + 40)
        a_c2, sc_c2 = _newton_active_set(best_seq, t_new, max_rounds=80)
        update_best(a_c2, sc_c2, f"cyc_np_{cyc}")

        # L-BFGS burst
        t_lbf = min(cycle_end, time.monotonic() + 60)
        if time.monotonic() < cycle_end - 20:
            a_c3, sc_c3 = _run_smoothed_lbfgs(best_seq, cycle_end, min_remaining=50.0)
            update_best(a_c3, sc_c3, f"cyc_lb_{cyc}")

    # Phase 5: Multi-n exploration
    print(f"\n[Phase 5] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_options = []
    for n_try in [8192, 16384, 32768, 65536, n0 // 2, n0 * 2]:
        if n_try not in n_options and 4096 <= n_try <= 131072:
            n_options.append(n_try)

    for n_try in sorted(n_options, key=abs, reverse=False):
        if time.monotonic() > deadline - 60:
            break
        a_res = _resample(best_seq, n_try)
        sc_res = _score(a_res)
        label = f"resample_n{n_try}"
        if sc_res < best_score:
            update_best(a_res, sc_res, label)

        if time.monotonic() < deadline - 30 and n_try <= 65536:
            t_end = min(deadline - 20, time.monotonic() + 15)
            a_opt, sc_opt = _subgrad_momentum(a_res, t_end, max_steps=2000)
            update_best(a_opt, sc_opt, f"opt_n{n_try}")

    # Phase 6: Restarts with perturbation
    print(f"\n[Phase 6] Restarts (t={time.monotonic() - t_start:.1f}s)")
    for restart in range(5):
        if time.monotonic() > deadline - 15:
            break
        sigma = 0.01 * (restart + 1)
        a_pert = _perturb(best_seq, sigma, rng)
        t_end = min(deadline - 10, time.monotonic() + 8)
        a_r, sc_r = _subgrad_momentum(a_pert, t_end, max_steps=1000)
        update_best(a_r, sc_r, f"restart_{restart}")

    # Phase 7: Final ultra-fine polish
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 2:
        a_fin, sc_fin = _newton_active_set(best_seq, deadline - 1, max_rounds=100)
        update_best(a_fin, sc_fin, "final_newton")

    if time.monotonic() < deadline - 1:
        a_fin2, sc_fin2 = _subgrad_momentum(best_seq, deadline - 0.5, max_steps=2000)
        update_best(a_fin2, sc_fin2, "final_subgrad")

    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
