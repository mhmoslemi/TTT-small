def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr(a, w, n):
    """Compute sum_i a[i]*w[i+k] for k=0..n-1 (zero-padded)."""
    flip_a = a[::-1]
    out_len = n + len(w) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fw = np.fft.rfft(w, N)
    result = np.fft.irfft(fa * fw, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _ls_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _ls_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec_fn, rhs, max_cg=300, tol=1e-14):
    n = len(rhs)
    x = np.zeros(n)
    r = rhs - matvec_fn(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_cg):
        Ap = matvec_fn(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _newton_step_active(a, c, c_max, active_tol, mu_reg):
    """Compute Newton step for active set defined by tolerance."""
    n = len(a)

        return None
    
    A_vec = active.astype(np.float64) / m
    g = 2.0 * _corr(a, A_vec, n)
    g_t = g - np.mean(g)

    if gnorm < 1e-15:
        return None
    
    A_vec_local = A_vec
    n_local = n
    
    def _hess_vp(v):
        return 2.0 * _corr(v, A_vec_local, n_local) + mu_reg * v
    
    u = _cg_solve(_hess_vp, g_t, max_cg=300)

    w_sum = float(w.sum())
    
    if abs(w_sum) < 1e-15:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w
    
    dx = dx - np.mean(dx)
    return dx

def _lbfgs_continuation(a_init, deadline, min_remaining=60.0):

def _peak_shaving(a, deadline, max_iters=100):
    """
    Direct peak shaving: iteratively reduce the single highest peak
    using the most efficient gradient direction.
    """

    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:

        n_peaks = len(peaks)

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-2 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "two_hump":
        t = np.linspace(0, 1, n)
        a = np.exp(-8 * (t - 0.3) ** 2) + 0.8 * np.exp(-8 * (t - 0.7) ** 2)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = np.exp(-3 * (t - 0.35) ** 2) + 0.5 * np.exp(-5 * (t - 0.75) ** 2)
        return a / a.sum()
    elif kind == "plateau":
        a = np.zeros(n)
        a[n//4:3*n//4] = 1.0
        # Smooth edges
        edge = n // 16
        for i in range(edge):
            frac = i / edge
            a[n//4 + i] = frac
            a[3*n//4 - 1 - i] = frac
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

        nonlocal best_seq, best_score, best_n
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            best_n = len(p_arr)
            print(f"  NEW BEST: {best_score:.12f} (n={best_n}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s")
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.12f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        update_best(parent, _score(parent), "init")
    
    # === Phase 1: L-BFGS continuation (300s) ===
    print(f"\n[Phase 1] L-BFGS continuation (t={time.monotonic()-t_start:.1f}s)")
    p1_dl = min(deadline - 500, time.monotonic() + 300)
    a1 = _lbfgs_continuation(parent, p1_dl, min_remaining=500.0)
    update_best(a1, _score(a1), "lbfgs_p1")
    
    # === Phase 2: Active-set Newton (250s) ===
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        p2_dl = min(deadline - 350, time.monotonic() + 250)
        a2, sc2 = _active_set_refine(best_seq, p2_dl, max_outer=200)
        update_best(a2, sc2, "as_newton")
    
    # === Phase 3: Peak shaving (100s) ===
    print(f"\n[Phase 3] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        p3_dl = min(deadline - 250, time.monotonic() + 100)
        a3, sc3 = _peak_shaving(best_seq, p3_dl, max_iters=200)
        update_best(a3, sc3, "peak_shave")
    
    # === Phase 4: Combined refinement loop (200s) ===
    print(f"\n[Phase 4] Refinement loop (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        p4_dl = min(deadline - 200, time.monotonic() + 200)
        a_cur = best_seq.copy()
        sc_cur = best_score
        for _ in range(6):
            if time.monotonic() > p4_dl:
                break
            # Newton step
            a_n, sc_n = _active_set_refine(a_cur, min(p4_dl, time.monotonic() + 40), max_outer=20)
            if sc_n < sc_cur:
                a_cur = a_n
                sc_cur = sc_n
            # Peak shave
            a_p, sc_p = _peak_shaving(a_cur, min(p4_dl, time.monotonic() + 30), max_iters=30)
            if sc_p < sc_cur:
                a_cur = a_p
                sc_cur = sc_p
            update_best(a_cur, sc_cur, "refine_loop")
    
    # === Phase 5: Multi-scale upscale to 32768 (300s) ===
    print(f"\n[Phase 5] Multi-scale n=32768 (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        n_large = 32768
        a_large = _interpolate_seq(best_seq, n_large)
        sc_large = _score(a_large)
        print(f"  n={n_large} interpolated: {sc_large:.12f}")
        update_best(a_large, sc_large, "interp_32k")
        
        # L-BFGS on large n
        p5a_dl = min(deadline - 120, time.monotonic() + 150)
        a_large1 = _lbfgs_continuation(a_large, p5a_dl, min_remaining=120.0)
        update_best(a_large1, _score(a_large1), "lbfgs_32k")
        
        # Active-set on large n
        p5b_dl = min(deadline - 80, time.monotonic() + 100)
        a_large2, sc_large2 = _active_set_refine(a_large1, p5b_dl, max_outer=50)
        update_best(a_large2, sc_large2, "as_32k")
    
    # === Phase 6: Multi-scale upscale to 65536 (150s) ===
    print(f"\n[Phase 6] Multi-scale n=65536 (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_xl = 65536
        a_xl = _interpolate_seq(best_seq, n_xl)
        sc_xl = _score(a_xl)
        print(f"  n={n_xl} interpolated: {sc_xl:.12f}")
        update_best(a_xl, sc_xl, "interp_64k")
        
        p6_dl = min(deadline - 40, time.monotonic() + 80)
        a_xl_opt = _lbfgs_continuation(a_xl, p6_dl, min_remaining=40.0)
        update_best(a_xl_opt, _score(a_xl_opt), "lbfgs_64k")
    
    # === Phase 7: Final L-BFGS polish (100s) ===
    print(f"\n[Phase 7] Final L-BFGS polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        p7_dl = min(deadline - 30, time.monotonic() + 100)
        a_pol = _lbfgs_continuation(best_seq, p7_dl, min_remaining=30.0)
        update_best(a_pol, _score(a_pol), "final_lbfgs")
    
    # === Phase 8: Diversified restarts (remaining time) ===
    print(f"\n[Phase 8] Diversified restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 20:
            # Try small perturbations
            for sigma in [0.002, 0.005, 0.01, 0.02]:
                if time.monotonic() > deadline - 15:
                    break
                p_start = _perturb(best_seq, sigma, rng)
                r_dl = min(deadline - 10, time.monotonic() + min(30, remaining / 4))
                a_r = _lbfgs_continuation(p_start, r_dl, min_remaining=10.0)
                update_best(a_r, _score(a_r), f"restart_{sigma}")
            
            # Try different structural initializations at same n
            if time.monotonic() < deadline - 30:
                n_try = best_n
                if n_try > 32768:
                    n_try = 32768
                for kind in ["two_hump", "asymmetric", "plateau"]:
                    if time.monotonic() > deadline - 15:
                        break
                    a_struct = _make_profile(n_try, kind)
                    # Blend with best
                    a_blend = 0.7 * best_seq[:n_try] + 0.3 * a_struct
                    a_blend = np.maximum(a_blend, 0)
                    a_blend = a_blend / a_blend.sum()
                    r_dl = min(deadline - 10, time.monotonic() + 20)
                    a_r2 = _lbfgs_continuation(a_blend, r_dl, min_remaining=10.0)
                    update_best(a_r2, _score(a_r2), f"struct_{kind}")
    
    # === Phase 9: Final active-set polish on best ===
    print(f"\n[Phase 9] Final Newton polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        p9_dl = min(deadline - 3, time.monotonic() + 30)
        a_fin, sc_fin = _active_set_refine(best_seq, p9_dl, max_outer=30)
        update_best(a_fin, sc_fin, "final_newton")
    
    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
