   - Identify the top K peaks in the convolution
   - For each peak at position k, identify which elements of `a` contribute most to it
   - Try reducing those elements (while maintaining sum=1)
   - This is a greedy local search

   For a peak at position k, `c[k] = sum_i a[i]*a[k-i]`. The largest contributors are the pairs `(i, k-i)` with the largest product. Reducing `a[i]` for the largest `a[i]*a[k-i]` would reduce `c[k]` the most.

   But reducing `a[i]` also affects other convolution entries. So we need to be careful.

   **Simplified peak shaving**: For the maximum peak at position k*, try:
   - For each i in the range [0, n-1] where k-i is also in [0, n-1]:

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:
- Enhanced active-set with backtracking line search and adaptive tolerance
- L-BFGS with extended beta schedule and restart mechanism
- Peak-shaving local search to escape local minima
- Multi-n exploration with guided resampling
- Squared parameterization alternative for L-BFGS
- Final multi-start polish combining all techniques

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = max(2, 1 << (n - 1).bit_length())
    return k


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_with_conv(a):
    n = len(a)
    s = float(a.sum())
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s)), c


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_structured(a, beta, n):
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


class _SoftProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        return 2.0 * self.n * _soft_max_value(c, self.beta)

    def grad(self, z):
        a = _softmax(z)
        grad_a = _grad_structured(a, self.beta, self.n)
        W = float(np.dot(grad_a, a))
        return a * (grad_a - W)


class _SqProb:
    """Squared parameterization: a = (x^2)/sum(x^2)."""
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, x):
        a = x * x
        s = a.sum()
        if s < 1e-300:
            return 1e10
        a = a / s
        c = _full_convolve(a, a)
        return 2.0 * self.n * _soft_max_value(c, self.beta)

    def grad(self, x):
        a = x * x
        s = a.sum()
        if s < 1e-300:
            return np.zeros_like(x)
        a = a / s
        grad_a = _grad_structured(a, self.beta, self.n)
        # da/dx_j = 2*x_j/s - 2*a_j*x_j/s^2 * s = 2*x_j*(1/s - a_j/s)
        # = 2*x_j*(1 - a_j)/s... let me be careful
        # a_j = x_j^2 / s, da_j/dx_k = (2*x_k*delta_jk * s - x_j^2 * 2*x_k) / s^2
        # = 2*x_k*(x_j^2*delta_jk - x_j^2... 
        # Actually: da/dx_k = 2*x_k*e_k/s - (x^2/s^2)*2*x_k = 2*x_k*(e_k - a_k... 
        # Let me use: a = x^2/sum(x^2). da_k/dx_j = 2*x_j*delta_{kj}/S - x_k^2*2*x_j/S^2
        # = 2*x_j*(delta_{kj} - a_k)/S
        # So grad_x_j = sum_k grad_a_k * 2*x_j*(delta_{kj} - a_k)/S
        # = 2*x_j*(grad_a_j - sum_k grad_a_k * a_k)/S
        # = 2*x_j*(grad_a_j - dot(grad_a, a))/S
        ga_dot_a = float(np.dot(grad_a, a))
        return 2.0 * x * (grad_a - ga_dot_a) / s


def _active_set_v2(a_init, deadline, verbose=True):
    """Enhanced active-set with backtracking and adaptive tolerance."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol = 0.008
    min_tol = 0.0003
    no_improve = 0
    max_no_improve = 150
    step_base = 0.05

    while time.monotonic() > deadline - 20:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol = max(min_tol, tol * 0.5)
            step_base = max(0.005, step_base * 0.7)
            if tol <= min_tol * 1.5 and step_base < 0.008:
                break
            no_improve = 0

        threshold = c_max - tol * c_max
        active = c >= threshold
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros(len(c), dtype=bool)
            active[idx] = True
            n_active = 1

        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        support = a > 1e-9 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        lam = float(np.mean(grad[support]))
        residual = grad[support] - lam
        res_norm = float(np.sqrt(np.sum(residual ** 2)))
        if res_norm < 1e-12:
            tol = max(min_tol, tol * 0.5)
            continue

        found_step = False
        for eta in [step_base, step_base * 0.5, step_base * 0.25,
                    step_base * 0.125, step_base * 0.0625, 0.001, 0.0005]:
            a_new = a.copy()
            a_new[support] -= eta * residual
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                a = a_new
                found_step = True
                break

        if not found_step:
            tol = max(min_tol, tol * 0.7)
            if no_improve > max_no_improve * 2:
                break

    return best_a, best_sc


def _peak_shaving(a_init, deadline, verbose=True):
    """Peak shaving: identify top peaks and try to reduce them."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    while time.monotonic() > deadline - 30:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        # Find top peaks
        top_k = min(5, len(c))
        peaks = np.argsort(c)[::-1][:top_k]
        peaks = peaks[peaks < 2 * n - 1]

        improved = False
        for pk in peaks:
            pk = int(pk)
            c_pk = c[pk]
            if c_pk < c_max * 0.95:
                break

            # Find which a[i] contribute most to c[pk]
            # c[pk] = sum_i a[i]*a[pk-i] for valid i
            i_lo = max(0, pk - (n - 1))
            i_hi = min(n - 1, pk)

            if i_lo > i_hi:
                continue

            # Get contributions
            idx = np.arange(i_lo, i_hi + 1)
            contributions = a[idx] * a[pk - idx]
            total_contrib = float(contributions.sum())
            if total_contrib < 1e-15:
                continue

            # Try reducing the top contributor
            top_idx_local = int(np.argmax(contributions))
            j = int(idx[top_idx_local])

            # Try reducing a[j] and redistributing to elements that
            # don't contribute to this peak
            a_j = a[j]
            if a_j < 1e-12:
                continue

            # Find "safe" elements to receive the mass (those that don't
            # contribute to the top peaks)
            top_peak_set = set(int(p) for p in peaks[:3])
            safe = np.ones(n, dtype=bool)
            for p in top_peak_set:
                p = int(p)
                ilo = max(0, p - (n - 1))
                ihi = min(n - 1, p)
                if j >= ilo and j <= ihi:
                    # a[j] contributes to this peak, but other elements might be safe
                    pass
                # Mark elements that contribute to peak p
                for ii in range(ilo, ihi + 1):
                    if ii != j:
                        safe[ii] = False

            safe[j] = True  # We're reducing j, so it's "safe" in a sense
            safe_candidates = np.where(safe & (a < 0.5 * a.max()))[0]

            if len(safe_candidates) == 0:
                safe_candidates = np.where(a < 0.3 * a.max())[0]
                if len(safe_candidates) == 0:
                    continue

            # Try reducing a[j] by small fraction and adding to a safe element
            for frac in [0.01, 0.005, 0.002, 0.001]:
                delta = a_j * frac
                for sc_idx in safe_candidates[:5]:
                    a_try = a.copy()
                    a_try[j] -= delta
                    a_try[sc_idx] += delta
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_sc - 1e-12:
                        best_sc = sc_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            break

    return best_a, best_sc


def _run_lbfgs_enhanced(a_init, deadline, min_time_remaining=60.0, verbose=True,
                        use_squred=False):
    """L-BFGS with adaptive beta, extended schedule, and restart."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    if use_squred:
        z = np.sqrt(np.maximum(a, 1e-30))
    else:
        z = np.log(np.maximum(a, 1e-30))
        z = z - z.mean()

    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(200, len(c_sorted) - 1)]
    if gap < c0_max * 0.002:
        gap = c0_max * 0.002

    # Extended fine-grain beta schedule
    multipliers = np.array([
        1.0, 0.5, 0.3, 0.15, 0.08, 0.04,
        0.02, 0.01, 0.005, 0.003, 0.0015,
        0.0008, 0.0004, 0.0002, 0.0001,
        0.00005, 0.00003, 0.00002, 0.00001,
        0.000005, 0.000003, 0.000002
    ])
    betas = gap * multipliers

    t_start = time.monotonic()
    stall_count = 0

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        size_factor = max(1.0, n / 8192.0)
        est_iter_time = 0.003 * size_factor
        max_iter = min(10000, max(200, int(remaining / (est_iter_time * 30))))

        if beta < gap * 0.001:
            max_iter = min(max_iter, 3000)
        if beta < gap * 0.0001:
            max_iter = min(max_iter, 1500)

        if use_squred:
            prob = _SqProb(beta, n)
        else:
            prob = _SoftProb(beta, n)

        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 150,
                    'maxls': 80,
                }
            )
            if use_squred:
                a_new = result.x * result.x
                s_new = a_new.sum()
                a_new = a_new / s_new
            else:
                a_new = _softmax(result.x)

            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                if use_squred:
                    z = np.sqrt(np.maximum(a_new, 1e-30))
                else:
                    z = np.log(np.maximum(a_new, 1e-30))
                    z = z - z.mean()
                stall_count = 0
                if verbose:
                    elapsed = time.monotonic() - t_start
                    print(f"    beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}, t={elapsed:.0f}s")
            else:
                stall_count += 1
                if verbose and i < 15:
                    print(f"    beta={beta:.2e}: no improvement (best={best_sc:.10f})")
                # If stalled for a while, skip to next beta
                if stall_count > 3:
                    stall_count = 0
        except Exception as e:
            if verbose:
                print(f"    beta={beta:.2e}: error {e}")

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.3) ** 2) + np.exp(-40.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "one_sided":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t) + 0.005
        return a / a.sum()
    elif kind == "asymmetric_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.8) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "shifted_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * (t + 0.05) ** 0.9) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma=0.01, rng=None):
    """Add small random perturbation and re-normalize."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a)
    noise = rng.normal(0, sigma, n)
    a_new = a * (1.0 + noise)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new

    if time.monotonic() + 20 < deadline:
        p6_end = min(deadline - 130, time.monotonic() + 100)
        a6, sc6 = _active_set_v2(best_seq, p6_end, verbose=True)
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  P6: {best_score:.10f}")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n (t={time.monotonic()-t_start_global:.0f}s)")
    n_options = [8192, 12288, 20480, 24576]
    n_options = [x for x in n_options if x != n_main]
    for n_try in n_options:
        if time.monotonic() + 60 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if sc_r < best_score:
            best_score = sc_r
            best_seq = a_r.copy()
            print(f"  n={n_try} resample: {best_score:.10f}")
        if time.monotonic() + 50 > deadline:
            continue
        p7_end = min(deadline - 80, time.monotonic() + 80)
        a7, sc7 = _run_lbfgs_enhanced(a_r, p7_end, min_time_remaining=40,
                                       verbose=False, use_squred=False)
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()
            print(f"  n={n_try} optimized: {best_score:.10f}")
        # Quick active-set
        if time.monotonic() + 30 > deadline:
            continue
        p7b_end = min(deadline - 60, time.monotonic() + 50)
        a7b, sc7b = _active_set_v2(best_seq, p7b_end, verbose=False)
        if sc7b < best_score:
            best_score = sc7b
            best_seq = a7b.copy()
            print(f"  n={n_try} active-set: {best_score:.10f}")

    # === Phase 8: Perturbation + restart ===
    print(f"\n[P8] Perturbation restart (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 40 < deadline:
        for trial in range(3):
            if time.monotonic() + 20 > deadline:
                break
            a_p = _perturb(best_seq, sigma=0.005, rng=rng)
            sc_p = _score(a_p)
            if sc_p < best_score + 0.01:  # Only proceed if not too worse
                p8_end = min(deadline - 40, time.monotonic() + 60)
                a8, sc8 = _run_lbfgs_enhanced(a_p, p8_end, min_time_remaining=20,
                                               verbose=False, use_squred=False)
                if sc8 < best_score:
                    best_score = sc8
                    best_seq = a8.copy()
                    print(f"  Perturb {trial+1}: {best_score:.10f}")

    # === Phase 9: Final active-set + L-BFGS polish ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 15 < deadline:
        p9_end = min(deadline - 20, time.monotonic() + 80)
        a9, sc9 = _active_set_v2(best_seq, p9_end, verbose=True)
        if sc9 < best_score:
            best_score = sc9
            best_seq = a9.copy()
            print(f"  P9 active-set: {best_score:.10f}")

    if time.monotonic() + 10 < deadline:
        p9b_end = min(deadline - 5, time.monotonic() + 40)
        a9b, sc9b = _run_lbfgs_enhanced(best_seq, p9b_end, min_time_remaining=5,
                                         verbose=False, use_squred=False)
        if sc9b < best_score:
            best_score = sc9b
            best_seq = a9b.copy()
            print(f"  P9 L-BFGS: {best_score:.10f}")

    # Final
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
          f"time={time.monotonic()-t_start_global:.1f}s")
    return [float(x) for x in final]
