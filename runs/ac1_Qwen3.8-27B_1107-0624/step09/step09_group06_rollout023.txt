Optimized flat-spectrum step function search using adaptive L-BFGS with
log-sum-exp smoothing, subgradient polish, and multi-scale exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

"""
Adaptive L-BFGS with log-sum-exp smoothing + Fista subgradient polish for
minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses FFT-based convolution/correlation for efficient gradient computation.
Explores multiple n values and initializations with time-budgeted phases.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _fft_corr_w_a(w, a):
    """Compute g[i] = sum_k w[k]*a[k-i] for i=0..len(a)-1.
    w has length 2n-1, a has length n. Uses FFT."""
    n = len(a)
    m = len(w)
    N = _next_pow2(n + m - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fw * np.conj(fa), N)
    return c[:n]


def _score_norm(a):
    """Score assuming a is already normalized (sum=1)."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - z.max()
    e = np.exp(np.clip(z, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)

def _project_simplex(x):
    """Project x onto probability simplex {a>=0, sum(a)=1}."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _smooth_obj_grad_z(z, beta, n):
    """Compute f_beta(softmax(z)) and gradient w.r.t. z.
    f_beta = (2n/beta) * log(sum_k exp(beta * c[k])) where c=conv(a,a).

    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = float(c.max())

    if s < 1e-300:

        w[int(np.argmax(c))] = 1.0
        log_s = 0.0
    else:
        w = e / s
        log_s = float(np.log(s))

    obj = 2.0 * n * (c_max + beta * log_s)

    mean_ga = float(np.dot(a, grad_a))

def _adaptive_lbfgs(a_init, deadline, reserve=30.0):
    """Run L-BFGS with adaptive beta schedule. Returns best (a, score)."""
    n = len(a_init)
    c = _fft_convolve(a_init, a_init)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    K = min(30, 2 * n - 1)
    gap = c_sorted[0] - c_sorted[K - 1]
    if gap < c_max * 1e-6:
        gap = c_max * 1e-6

    # Beta schedule: start large, decrease geometrically
    beta_start = max(gap * 3.0, c_max * 0.05)
    beta_end = max(c_max * 1e-8, 1e-15)
    n_betas = 18

    best_a = a_init.copy()
    best_sc = _score_norm(a_init)

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    for i in range(n_betas):
        if time.monotonic() + reserve > deadline:
            break
        fraction = i / max(1, n_betas - 1)
        beta = beta_start * (beta_end / beta_start) ** fraction

        remaining = deadline - reserve - time.monotonic()
        if remaining < 3:
            break

        # Scale maxiter based on beta: finer beta needs more iterations
        if fraction < 0.3:
            max_iter = min(2000, max(200, int(remaining / 0.003)))
        elif fraction < 0.7:
            max_iter = min(3000, max(300, int(remaining / 0.003)))
        else:
            max_iter = min(5000, max(500, int(remaining / 0.004)))

        prob = _SmoothProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16,
                         'gtol': 1e-14, 'maxcor': 40, 'maxls': 40}
            )
            a_new = _softmax(result.x)
            sc_new = _score_norm(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        if i % 4 == 0:
            print(f"      LBFGS beta_idx={i}/{n_betas}, beta={beta:.3e}, score={best_sc:.10f}, t={time.monotonic():.1f}")

    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score_norm(a)

    y = a.copy()
    t = 1.0
    base_alpha = 0.02

    for it in range(20000):
        if time.monotonic() + reserve > deadline:
            break

        c_y = _fft_convolve(y, y)
        c_max_y = float(c_y.max())
        # Active set: top K peaks
        K = max(5, int(n * 0.003))
        idx = np.argsort(c_y)[::-1][:K]
        # Subgradient: average of gradients of active peaks
        # grad_k[i] = 2*a[k-i] for valid k-i
        # We compute sum of grad_k for active k, then average
        # grad_avg[i] = (1/K) * sum_{k in active} 2*y[k-i] for valid k-i

        # = (2/K) * (sum of w[k]*y[k-i] where w is uniform over active)
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / K
        g = _fft_corr_w_a(w, y)
        subgrad = 2.0 * n * g  # gradient of max(c) w.r.t. a (scaled by 2n for objective)

        # Step size
        alpha = base_alpha / (1.0 + 0.01 * it)

        # Fista update
        a_new = _project_simplex(y - alpha * subgrad)
        sc_new = _score_norm(a_new)

        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        y = _project_simplex(y)
        a = a_new
        t = t_new

        if it % 500 == 0 and it > 0:
            print(f"      Subgrad it={it}: best={best_sc:.10f}, cur={sc_new:.10f}")

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_random_profile(n, rng, kind="exp"):
    if kind == "exp":
        a = rng.exponential(1.0, size=n)
    elif kind == "gauss":
        a = np.abs(rng.standard_normal(n)) + 0.1
    elif kind == "cos_ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t) + 0.1 * rng.standard_normal(n)
        a = np.maximum(a, 0.01)
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.3) ** 2) + np.exp(-20 * (t - 0.7) ** 2) + 0.05 + 0.02 * rng.standard_normal(n)
        a = np.maximum(a, 0.001)
    else:
        a = rng.dirichlet(np.ones(n))
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def _try_n_and_init(n, a_init, deadline, reserve=30.0, label=""):
    """Run full optimization pipeline for a given n and initialization."""
    print(f"    [{label}] n={n}, init_score={_score_norm(a_init):.10f}")
    best_a = a_init.copy()
    best_sc = _score_norm(a_init)

    # Phase A: Adaptive L-BFGS
    if time.monotonic() + reserve + 60 < deadline:
        a_lb, sc_lb = _adaptive_lbfgs(a_init, deadline, reserve=reserve + 30)
        if sc_lb < best_sc:
            best_a = a_lb.copy()
            best_sc = sc_lb
            print(f"    [{label}] After LBFGS: {best_sc:.10f}")

    # Phase B: Subgradient polish
    if time.monotonic() + reserve + 10 < deadline:
        a_sg, sc_sg = _subgrad_polish(best_a, deadline, reserve=reserve)
        if sc_sg < best_sc:
            best_a = a_sg.copy()
            best_sc = sc_sg
            print(f"    [{label}] After subgrad: {best_sc:.10f}")

    # Phase C: Final fine L-BFGS
    if time.monotonic() + reserve + 10 < deadline:
        a_fin, sc_fin = _adaptive_lbfgs(best_a, deadline, reserve=reserve)
        if sc_fin < best_sc:
            best_a = a_fin.copy()
            best_sc = sc_fin
            print(f"    [{label}] After final LBFGS: {best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score_norm(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = np.ones(n_parent) / n_parent
        best_seq = parent.copy()
        best_score = _score_norm(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # Define search space: (n, initialization_type, label)
    search_plan = []

    # Primary: parent-based at various n
    for n_try in [16384, 8192, 24576, 32768]:
        if n_try != n_parent:
            a_init = _resample(parent, n_try)
        else:
            a_init = parent.copy()
        search_plan.append((n_try, a_init, f"parent_n{n_try}"))

    # Random initializations
    search_plan.append((16384, _make_random_profile(16384, rng, "exp"), "rand_exp_16k"))
    search_plan.append((16384, _make_random_profile(16384, rng, "bimodal"), "rand_bimodal_16k"))
    search_plan.append((8192, _make_random_profile(8192, rng, "gauss"), "rand_gauss_8k"))
    search_plan.append((24576, _make_random_profile(24576, rng, "cos_ramp"), "rand_cos_24k"))

    # Execute search plan with time budgeting
    total_plans = len(search_plan)
    for i, (n_try, a_init, label) in enumerate(search_plan):
        elapsed = time.monotonic() - t_start
        remaining_total = deadline - time.monotonic()
        if remaining_total < 120:
            print(f"  Stopping search: only {remaining_total:.0f}s left")
            break

        # Allocate time: earlier plans get more time
        plans_left = total_plans - i
        frac = 1.0 / plans_left
        alloc = min(remaining_total * 0.85 * (1.0 + 0.5 * (1.0 - i / total_plans)),
                    remaining_total - 60)
        alloc = max(alloc, 60.0)
        plan_deadline = time.monotonic() + alloc

        reserve = max(20.0, (deadline - time.monotonic()) * 0.03)
        print(f"\n  Plan {i+1}/{total_plans}: {label} (budget {alloc:.0f}s)")

        a_best, sc_best = _try_n_and_init(n_try, a_init, plan_deadline,
                                           reserve=reserve, label=label)
        if sc_best < best_score:
            best_score = sc_best
            best_seq = a_best.copy()
            print(f"  *** NEW BEST: {best_score:.10f} ({label}) ***")

    # Final polish on the overall best
    print(f"\n  Final polish phase (remaining {deadline - time.monotonic():.0f}s)")
    if time.monotonic() + 30 < deadline and best_seq is not None:
        n_fin = len(best_seq)
        # Fine L-BFGS with very small beta
        c_fin = _fft_convolve(best_seq, best_seq)
        c_fin_max = float(c_fin.max())
        c_fin_sorted = np.sort(c_fin)[::-1]
        gap_fin = c_fin_sorted[0] - c_fin_sorted[min(20, len(c_fin_sorted) - 1)]
        if gap_fin < c_fin_max * 1e-7:
            gap_fin = c_fin_max * 1e-7

        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()

        betas_final = [gap_fin * 0.5, gap_fin * 0.1, gap_fin * 0.03,
                       gap_fin * 0.01, gap_fin * 0.003, gap_fin * 0.001]
        for beta_f in betas_final:
            if time.monotonic() + 15 > deadline:
                break
            prob = _SmoothProb(beta_f, n_fin)
            max_iter = min(3000, max(100, int((deadline - 15 - time.monotonic()) / 0.005)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-16,
                                           'gtol': 1e-15, 'maxcor': 50, 'maxls': 50})
                a_new = _softmax(result.x)
                sc_new = _score_norm(a_new)
                if sc_new < best_score:
                    best_score = sc_new
                    best_seq = a_new.copy()
                    z = np.log(np.maximum(a_new, 1e-30))
                    z = z - z.mean()
                    print(f"    Final polish beta={beta_f:.2e}: {best_score:.10f}")
            except Exception:
                pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        # Normalize to have a reasonable sum (score is scale-invariant)
        sc_final = _score_norm(final / s)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score_norm(fallback):.10f}")
    return [float(x) for x in fallback]
