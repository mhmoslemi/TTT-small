Active-set linear solver + L-BFGS polishing for minimum peak self-convolution.
Key innovation: exploits the linear structure of the optimality conditions
(M*a = c*1 where M[i,j] = active[i+j]) to solve for exact peak-equalizing
solutions via conjugate gradient with FFT-based matrix products, combined
with adaptive active-set management and L-BFGS polishing for final refinement.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="tapered"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        a = 0.3 + 0.7 * t
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _matvec_cg(d_active, v, n, N_fft, Fd):
    """Compute M*v where M[i,j] = d[i+j] (d is indicator of active set).
    Uses FFT: y[i] = (d * v_rev)[n-1+i]
    """
    v_rev = v[::-1]
    Fv = np.fft.rfft(v_rev, N_fft)
    conv = np.fft.irfft(Fd * Fv, N_fft)[:2 * n]
    return conv[n - 1:2 * n - 1].copy()


def _solve_active_set(active_set, n, a_init, max_iter=200, tol=1e-12):
    """Solve M*a = c*1, sum(a)=1 using block CG.
    M[i,j] = |{r: active[r] = i+j}|
    """
    m = len(active_set)
    if m == 0:
        return a_init.copy(), 0.0

    N_fft = _next_pow2(3 * n)
    d = np.zeros(2 * n - 1)
    for k in active_set:
        if 0 <= k < 2 * n - 1:
            d[k] = 1.0
    Fd = np.fft.rfft(d, N_fft)

    alpha_h = 2.0 * (2 * n - 2) - np.arange(n, dtype=np.float64)
    mask = (alpha_h >= 0) & (alpha_h < 2 * n - 1)
    d_sum = np.zeros(n)
    d_sum[mask] = d.astype(np.float64)[alpha_h[mask].astype(int)]

    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    r = d_sum - 0  # We'll solve M*a = c*1 by introducing c
    # Actually let's use the augmented system approach:
    # Minimize ||M*a - c*1||^2 + penalty*(sum(a)-1)^2
    # Or better: solve the KKT system directly.

    # Simpler: iterate a = M^{-1}(c*1) subject to sum(a)=1
    # Use the fact that at solution, (Ma)_i = c for all i
    # So c = mean(Ma) = (1/n) * sum_i (Ma)_i
    # And we need sum(a) = 1.

    # Let's use: minimize f(a) = sum_i (d_sum_i * c - (Ma)_i)^2 + lambda*(sum(a)-1)^2
    # where c = mean(Ma)
    # This is getting circular. Let me use a different approach.

    # Direct approach: solve the saddle-point system
    # [M, -1][a]   [0]
    # [1^T, 0][c] = [1]
    # Using block elimination: from first eq: M*a = c*1, so c = (1^T M a)/n... no
    # c*1 = M*a means c = (Ma)_i for all i (they must all be equal at solution)
    # So we want M*a to be a constant vector.
    # Minimize ||M*a - mean(Ma)*1||^2 subject to sum(a)=1, a>=0

    # For now, use a simple iterative method:
    # a_{new} = normalize(M^{-1} * c * 1) where c is chosen so sum = 1

    # Use CG to solve M*x = rhs for various rhs
    def matvec(v):
        return _matvec_cg(d, v, n, N_fft, Fd)

    # Solve M*x = 1 (all-ones vector) using CG
    rhs = np.ones(n)
    x = np.zeros(n)
    r = rhs - matvec(x)
    p = r.copy()
    rsold = float(r @ r)
    c_val = 1.0

    for it in range(max_iter):
        Ap = matvec(p)
        pAp = float(p @ Ap)
        if abs(pAp) < 1e-30:
            break
        alpha = rsold / pAp
        x += alpha * p
        r -= alpha * Ap
        rsnew = float(r @ r)
        if np.sqrt(rsnew) < tol * (np.sqrt(rsold) + 1e-30):
            break
        p = r + (rsnew / rsold) * p
        rsold = rsnew

    # Now x ≈ M^{-1} * 1, so M*x = 1, meaning c=1
    # We need sum(x) = 1, so a = x / sum(x)
    if x.sum() > 0:
        a_sol = x / x.sum()
    else:
        a_sol = a_init.copy()

    a_sol = np.maximum(a_sol, 0)
    s = a_sol.sum()
    if s > 1e-15:
        a_sol = a_sol / s
    else:
        a_sol = a_init.copy()

    return a_sol, c_val


def _run_active_set_optimizer(a_init, deadline, min_time_remaining=30.0):
    """Iterative active-set method with progressive tightening."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    # Start with top-1 active set, gradually increase
    tolerance_schedule = [0.5, 0.3, 0.2, 0.15, 0.1, 0.08, 0.06, 0.05,
                          0.04, 0.03, 0.025, 0.02, 0.015, 0.01, 0.008,
                          0.006, 0.005, 0.004, 0.003, 0.002, 0.001, 0.0005]
    tol_idx = 0

    for outer in range(500):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tolerance_schedule[min(tol_idx, len(tolerance_schedule) - 1)]
        active_mask = c >= c_max * (1.0 - tol)
        active_set = set(np.where(active_mask)[0].tolist())

        if len(active_set) == 0:
            active_set = {int(np.argmax(c))}

        # Solve the active set problem
        a_sol, _ = _solve_active_set(active_set, n, a, max_iter=100)

        # Check improvement
        sc = _score(a_sol)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_sol.copy()
            a = a_sol.copy()
        else:
            # If no improvement, try a blend
            for t in [0.5, 0.3, 0.1]:
                a_blend = (1 - t) * a + t * a_sol
                a_blend = np.maximum(a_blend, 0)
                a_blend = a_blend / a_blend.sum()
                sc_blend = _score(a_blend)
                if sc_blend < best_sc:
                    best_sc = sc_blend
                    best_a = a_blend.copy()
                    a = a_blend.copy()
                    break
            else:
                # No improvement, tighten tolerance or update active set
                tol_idx += 0.5
                if tol_idx >= len(tolerance_schedule) - 1:
                    # Try perturbation
                    rng = np.random.default_rng(outer)
                    a_pert = a + 0.01 * rng.standard_normal(n)
                    a_pert = np.maximum(a_pert, 0)
                    a_pert = a_pert / a_pert.sum()
                    a = a_pert

    return best_a, best_sc


def _run_subgradient(a_init, deadline, min_time_remaining=30.0):
    """Subgradient method on the exact max, with averaging."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    # Running average for Polyak averaging
    a_avg = a.copy()
    k = 0
    step_scale = 1.0

    for it in range(20000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        k += 1

        c = _full_convolve(a, a)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))

        # Subgradient of max at the peak: grad = 2 * shift(a, peak_idx)
        # (a*a)[k] = sum_j a[j]*a[k-j], d/da[i] = 2*a[k-i] (if valid)
        grad = np.zeros(n)
        k_val = peak_idx
        # grad[i] = 2*a[k-i] if 0 <= k-i < n else 0
        i_start = max(0, k_val - n + 1)
        i_end = min(k_val, n - 1)
        j_start = k_val - i_end
        j_end = k_val - i_start
        grad[i_start:i_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]

        # Project gradient onto tangent space (remove mean for sum constraint)
        grad_centered = grad - grad.mean()

        # Step size
        step = step_scale / np.sqrt(k + 1.0)
        step_scale = min(step_scale, c_max * 10.0)

        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        # Polyak averaging
        a_avg = (k * a_avg + a_new) / (k + 1)

        a = a_new

        if it % 100 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
            sc_avg = _score(a_avg)
            if sc_avg < best_sc:
                best_sc = sc_avg
                best_a = a_avg.copy()

    return best_a, best_sc


def _run_lbfgs_tight(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with very tight log-sum-exp smoothing for final polishing."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())

    # Very tight betas for polishing
    betas = [max_c * 1e-3, max_c * 5e-4, max_c * 2e-4, max_c * 1e-4,
             max_c * 5e-5, max_c * 2e-5, max_c * 1e-5, max_c * 5e-6,
             max_c * 2e-6, max_c * 1e-6]

    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        max_iter = min(1000, max(20, int(remaining / 0.05)))

        try:
            def obj_func(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    return 2.0 * n * c_max_local
                return 2.0 * n * (c_max_local + beta * np.log(s_e))

            def grad_func(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    w = np.zeros_like(c_local)
                    w[int(np.argmax(c_local))] = 1.0
                else:
                    w = e / s_e
                a_rev = a_local[::-1]
                wc = _full_convolve(w, a_rev)
                h = wc[n - 1:2 * n - 1]
                W_val = float(np.dot(w, c_local))
                return 4.0 * n * a_local * (h - W_val)

            result = minimize(
                obj_func, z,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 50, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return best_a, best_sc


def _run_peak_shaving(a_init, deadline, min_time_remaining=30.0):
    """Iteratively shave the peak by solving local subproblems."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(500):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))

        # Gradient of C(peak_idx)
        grad = np.zeros(n)
        i_start = max(0, peak_idx - n + 1)
        i_end = min(peak_idx, n - 1)
        j_start = peak_idx - i_end
        j_end = peak_idx - i_start
        grad[i_start:i_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]

        # Center the gradient
        grad_c = grad - grad.mean()

        # Line search
        best_step = 0
        best_c_max = c_max
        for step in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:
            a_try = a - step * grad_c
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            c_try = _full_convolve(a_try, a_try)
            cm = float(c_try.max())
            if cm < best_c_max:
                best_c_max = cm
                best_step = step

        if best_step > 0:
            a_new = a - best_step * grad_c
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            sc = _score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
            a = a_new
        else:
            # No improvement possible with single peak; try top-2
            top2 = np.argsort(c)[::-1][:2]
            if len(top2) >= 2 and c[top2[1]] > c_max * 0.99:
                grad2 = np.zeros(n)
                for pk in top2:
                    i_s = max(0, pk - n + 1)
                    i_e = min(pk, n - 1)
                    j_s = pk - i_e
                    j_e = pk - i_s
                    grad2[i_s:i_e + 1] += 2.0 * a[j_s:j_e + 1][::-1]
                grad2 /= 2.0
                grad2_c = grad2 - grad2.mean()
                for step in [0.01, 0.02, 0.05, 0.1]:
                    a_try = a - step * grad2_c
                    a_try = np.maximum(a_try, 0)
                    s = a_try.sum()
                    if s < 1e-15:
                        continue
                    a_try = a_try / s
                    c_try = _full_convolve(a_try, a_try)
                    cm = float(c_try.max())
                    if cm < c_max:
                        a = a_try
                        sc = _score(a)
                        if sc < best_sc:
                            best_sc = sc
                            best_a = a.copy()
                        break

    return best_a, best_sc


def _run_multi_active_avg(a_init, deadline, min_time_remaining=30.0):
    """Optimize using average of top-k peak gradients with adaptive k."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    momentum = np.zeros(n)
    beta_m = 0.95
    k_schedule = [3, 5, 7, 10, 15, 20, 30, 50, 30, 20, 10, 5, 3, 2, 1]
    k_idx = 0

    for it in range(10000):
        if time.monotonic() + min_time_remaining > deadline:
            break

        a_curr = _softmax(z)
        c = _full_convolve(a_curr, a_curr)
        c_max = float(c.max())

        k_act = k_schedule[min(k_idx, len(k_schedule) - 1)]
        top_k = np.argsort(c)[::-1][:k_act]

        # Average gradient of top-k peaks
        grad = np.zeros(n)
        for pk in top_k:
            i_start = max(0, pk - n + 1)
            i_end = min(pk, n - 1)
            j_start = pk - i_end
            j_end = pk - i_start
            grad[i_start:i_end + 1] += 2.0 * a_curr[j_start:j_end + 1][::-1]
        grad /= k_act

        grad_c = grad - grad.mean()
        gn = np.linalg.norm(grad_c)
        if gn < 1e-15:
            break

        momentum = beta_m * momentum + (1 - beta_m) * grad_c
        lr = 0.05 / (np.linalg.norm(momentum) / gn + 1e-10)
        z_new = z - lr * momentum
        z_new = np.clip(z_new, -50, 50)
        z = z_new - z_new.mean()

        if it % 5 == 0:
            sc = _score(a_curr)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_curr.copy()

        if it % 200 == 199:
            k_idx = min(k_idx + 1, len(k_schedule) - 1)

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining active-set linear solving, subgradient methods,
    peak shaving, L-BFGS polishing, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 32768
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Active-set optimizer (400s) ===
    print(f"\n[Phase 1] Active-set optimizer (t={time.monotonic() - t_start:.1f}s)")
    phase1_deadline = t_start + 400
    if time.monotonic() < phase1_deadline:
        a_as, sc_as = _run_active_set_optimizer(parent, phase1_deadline, min_time_remaining=30.0)
        update_best(a_as, sc_as, "active_set")

    # === Phase 2: L-BFGS tight polishing (300s) ===
    print(f"\n[Phase 2] L-BFGS tight polish (t={time.monotonic() - t_start:.1f}s)")
    phase2_deadline = min(deadline - 800, phase1_deadline + 300)
    if best_seq is not None and time.monotonic() < phase2_deadline:
        a_lb, sc_lb = _run_lbfgs_tight(best_seq, phase2_deadline, min_time_remaining=30.0)
        update_best(a_lb, sc_lb, "lbfgs_tight")

    # === Phase 3: Multi-active average gradient (250s) ===
    print(f"\n[Phase 3] Multi-active avg grad (t={time.monotonic() - t_start:.1f}s)")
    phase3_deadline = min(deadline - 550, phase2_deadline + 250)
    if best_seq is not None and time.monotonic() < phase3_deadline:
        a_ma, sc_ma = _run_multi_active_avg(best_seq, phase3_deadline, min_time_remaining=20.0)
        update_best(a_ma, sc_ma, "multi_active")

    # === Phase 4: Peak shaving (200s) ===
    print(f"\n[Phase 4] Peak shaving (t={time.monotonic() - t_start:.1f}s)")
    phase4_deadline = min(deadline - 350, phase3_deadline + 200)
    if best_seq is not None and time.monotonic() < phase4_deadline:
        a_ps, sc_ps = _run_peak_shaving(best_seq, phase4_deadline, min_time_remaining=20.0)
        update_best(a_ps, sc_ps, "peak_shave")

    # === Phase 5: Subgradient (150s) ===
    print(f"\n[Phase 5] Subgradient (t={time.monotonic() - t_start:.1f}s)")
    phase5_deadline = min(deadline - 200, phase4_deadline + 150)
    if best_seq is not None and time.monotonic() < phase5_deadline:
        a_sg, sc_sg = _run_subgradient(best_seq, phase5_deadline, min_time_remaining=15.0)
        update_best(a_sg, sc_sg, "subgrad")

    # === Phase 6: Second round of combined (200s) ===
    print(f"\n[Phase 6] Combined round 2 (t={time.monotonic() - t_start:.1f}s)")
    phase6_deadline = min(deadline - 150, phase5_deadline + 200)
    if best_seq is not None and time.monotonic() < phase6_deadline:
        a_lb2, sc_lb2 = _run_lbfgs_tight(best_seq, phase6_deadline, min_time_remaining=30.0)
        update_best(a_lb2, sc_lb2, "lbfgs_r2")
        if time.monotonic() < phase6_deadline - 30:
            a_as2, sc_as2 = _run_active_set_optimizer(best_seq, phase6_deadline, min_time_remaining=15.0)
            update_best(a_as2, sc_as2, "active_set_r2")

    # === Phase 7: Multi-n exploration (150s) ===
    print(f"\n[Phase 7] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    phase7_deadline = min(deadline - 100, phase6_deadline + 150)
    n_tries = [40960, 24576, 16384, 65536]
    for n_try in n_tries:
        if time.monotonic() > phase7_deadline - 20:
            break
        if n_try == n_parent:
            continue
        if best_seq is not None:
            a_init_n = _resample(best_seq, n_try)
        else:
            a_init_n = _make_profile(n_try, "tapered")
        a_n, sc_n = _run_lbfgs_tight(a_init_n, phase7_deadline, min_time_remaining=15.0)
        update_best(a_n, sc_n, f"n_{n_try}")

    # === Phase 8: Final ultra-tight polish (100s) ===
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_fin, sc_fin = _run_lbfgs_tight(best_seq, deadline - 10, min_time_remaining=5.0)
        update_best(a_fin, sc_fin, "final_lbfgs")

        if time.monotonic() < deadline - 15:
            a_ps2, sc_ps2 = _run_peak_shaving(best_seq, deadline - 15, min_time_remaining=5.0)
            update_best(a_ps2, sc_ps2, "final_shave")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
