- We want to make c as flat as possible
- This means we want |A(f)|^2 to be as close to a constant as possible (in the DFT domain, a constant corresponds to a delta in the time domain, which gives a flat autocorrelation)

3. Reconstruct: a_new = IFFT(|A|^alpha * exp(i*angle(A)))

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses frequency-domain spectral flattening combined with gradient-based
optimization and active-set subgradient methods.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_normalized(a):
    """Score for a normalized sequence (sum = 1). Returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_raw(a):
    """Score for any non-negative sequence."""
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    return _score_normalized(a_norm)


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a).
    w has length 2n-1. Returns array of length n."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


class _TopkSmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _topk_smoobj(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _topk_smoobj(z, self.beta, self.K, self.n)[1]


def _topk_smoobj(z, beta, K, n):
    """Top-K log-sum-exp smoothed objective and gradient."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        c_top = c
    
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    
    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


def _spectral_flatten(a, n_iters=20, alpha_schedule=None):
    """Iteratively flatten the power spectrum of a to reduce autocorrelation peaks."""
    n = len(a)
    a = _normalize(a.copy())
    N = _next_pow2(2 * n - 1)
    
    if alpha_schedule is None:
        alpha_schedule = [0.8, 0.7, 0.6, 0.5, 0.5, 0.4, 0.4, 0.3, 0.3, 0.25,
                         0.25, 0.2, 0.2, 0.15, 0.15, 0.1, 0.1, 0.05, 0.05, 0.0]
    
    best_a = a.copy()
    best_sc = _score_normalized(a)
    
    for i in range(n_iters):
        if i >= len(alpha_schedule):
            break
        alpha = alpha_schedule[i]
        
        # Compute spectrum
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Flatten: compress dynamic range
        if alpha > 0:
            mag_new = mag ** alpha
        else:
            # Complete flattening
            mag_new = np.ones_like(mag)
            mag_new[0] = mag[0]  # Keep DC component (preserves sum)
        
        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_recon = np.fft.ifft(A_new).real[:n]
        
        # Clip and normalize
        a_new = np.maximum(a_recon, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        
        sc = _score_normalized(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try partial update
            for t in [0.5, 0.3, 0.1]:
                a_mix = (1 - t) * a + t * a_new
                a_mix = np.maximum(a_mix, 0.0)
                a_mix = a_mix / a_mix.sum()
                sc_mix = _score_normalized(a_mix)
                if sc_mix < best_sc:
                    best_sc = sc_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    break
    
    return best_a, best_sc


def _active_set_subgradient(a_init, deadline, min_time_remaining=60.0):
    """Active-set subgradient with adaptive steps and momentum."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score_normalized(a)
    
    n_2nm1 = 2 * n - 1
    active_frac = 0.005
    momentum = 0.5
    lr = 0.01
    lr_min = 1e-5
    lr_max = 0.05
    stall = 0
    max_stall = 100
    velocity = np.zeros(n)
    
    prev_a = None
    prev_g = None
    iteration = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Select active peaks - focus on the very top ones
        n_active = max(2, min(n_2nm1, int(n_2nm1 * active_frac)))
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_active]
        
        # Exponential weighting on active peaks
        active_vals = c[active_idx]
        spread = active_vals.max() - active_vals.min()
        w = np.zeros(n_2nm1)
        if spread > 1e-18:
            w[active_idx] = np.exp(-3.0 * (active_vals.max() - active_vals) / spread)
        else:
            w[active_idx] = 1.0 / n_active
        w = w / w.sum()
        
        # Gradient
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break
        
        # BB step size
        if prev_a is not None and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g_centered - prev_g
            s_norm = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm > 1e-18 and abs(sy) > 1e-18:
                lr_bb = s_norm / abs(sy)
                lr = np.clip(lr_bb, lr_min, lr_max)
        
        # Momentum step with backtracking
        velocity = momentum * velocity - lr * g_centered
        alpha = 1.0
        improved = False
        for ls in range(20):
            a_try = a + alpha * velocity
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if improved:
            prev_a = a.copy()
            prev_g = g_centered.copy()
            lr = min(lr_max, lr * 1.02)
            active_frac = max(0.001, active_frac * 0.99)
        else:
            velocity = np.zeros(n)
            lr = max(lr_min, lr * 0.7)
            active_frac = min(0.05, active_frac * 1.2)
        
        # Adapt on stall
        if stall > max_stall:
            active_frac = min(0.05, active_frac * 2.0)
            lr = max(lr_min, lr * 0.5)
            momentum = min(0.8, momentum + 0.05)
            # Reset to best
            a = best_a.copy()
            velocity = np.zeros(n)
            prev_a = None
            prev_g = None
            stall = 0
    
    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with top-K smoothing and adaptive annealing."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score_normalized(a)
    current_a = a.copy()
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # Annealing schedule
    schedule = [
        (1.0, 0.1), (0.7, 0.07), (0.5, 0.05), (0.3, 0.03),
        (0.2, 0.02), (0.1, 0.01), (0.05, 0.005), (0.02, 0.003),
        (0.01, 0.002), (0.005, 0.001),
    ]
    
    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        beta = base_beta * beta_mult
        K = max(5, int((2 * n - 1) * K_frac))
        
        est_time = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(4000, max(50, int(remaining / (est_time * 25))))
        
        prob = _TopkSmoProb(beta, K, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 40}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score_normalized(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.3) ** 2) + np.exp(-15.0 * (t - 0.7) ** 2) + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(20 * np.pi * t * (t - 1))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    """Very fine active-set polish."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score_normalized(a)
    
    lr = 0.003
    tol = 0.003
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        # Active set: peaks within tol of max
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active
        
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-16:
            break
        
        alpha = min(lr, 0.3 / (gnorm + 1e-15))
        improved = False
        for ls in range(30):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve % 15 == 0:
                tol = max(0.0005, tol * 0.7)
                lr *= 0.7
            if no_improve > 200:
                a = best_a.copy()
                tol = max(0.0005, tol * 0.5)
                lr = 0.002
                no_improve = 0
        else:
            tol = max(0.0005, tol * 0.98)
            lr = min(0.008, lr * 1.01)
    
    return best_a, best_sc

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines spectral flattening (novel), L-BFGS with top-K smoothing,
active-set subgradient with BB steps, and fine polish.
Key innovation: frequency-domain spectral shaping to flatten the power
spectrum, reducing autocorrelation peaks, combined with gradient methods.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_normalized(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a.copy()
    return a / s


def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


class _TopkSmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _topk_smoobj(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _topk_smoobj(z, self.beta, self.K, self.n)[1]


def _topk_smoobj(z, beta, K, n):
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        c_top = c
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


def _spectral_flatten(a, n_iters=25, blend=None):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = _normalize(a.copy())
    N = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = _score_normalized(a)
    if blend is None:
        blend = 0.6
    alphas = []
    for i in range(n_iters):
        frac = i / max(1, n_iters - 1)
        alpha = 1.0 - frac * 0.7
        alphas.append(alpha)
    for i in range(n_iters):
        alpha = alphas[i]
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        mag_max = mag.max()
        if mag_max < 1e-15:
            break
        mag_norm = mag / mag_max
        if alpha >= 0.999:
            mag_new = mag
        else:
            mag_new = mag_max * (mag_norm ** alpha)
        mag_new[0] = mag[0]
        A_new = mag_new * np.exp(1j * phase)
        a_recon = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_recon, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        # Blend with current
        a_try = (1.0 - blend) * a + blend * a_new
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            continue
        a_try = a_try / s_try
        sc = _score_normalized(a_try)
        if sc < best_sc:
            best_sc = sc
            best_a = a_try.copy()
            a = a_try
    return best_a, best_sc


def _spectral_perturb(a, rng, strength=0.3):
    """Add structured frequency-domain noise to escape local minima."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
    # Add random phase perturbation in the mid frequencies
    noise_phase = rng.standard_normal(N // 2) * strength
    noise_phase[0] = 0.0
    noise_mag = 1.0 + strength * 0.1 * rng.standard_normal(N // 2)
    noise_mag = np.maximum(noise_mag, 0.5)
    noise_mag[0] = 1.0
    # Apply to positive frequencies
    new_phase = phase.copy()
    new_mag = mag.copy()
    new_phase[:N // 2] += noise_phase
    new_mag[:N // 2] *= noise_mag
    # Mirror for real signal
    new_phase[N // 2:] = new_phase[1:N // 2 + 1][::-1] if N % 2 == 0 else new_phase[1:N // 2 + 1][::-1]
    new_mag[N // 2:] = new_mag[1:N // 2 + 1][::-1] if N % 2 == 0 else new_mag[1:N // 2 + 1][::-1]
    A_new = new_mag * np.exp(1j * new_phase)
    a_recon = np.fft.ifft(A_new).real[:n]
    a_new = np.maximum(a_recon, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _active_set_subgradient(a_init, deadline, min_time_remaining=60.0):
    """Active-set subgradient with BB steps, momentum, and adaptive active set."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score_normalized(a)
    n_2nm1 = 2 * n - 1
    active_frac = 0.003
    momentum = 0.5
    lr = 0.008
    lr_min = 5e-6
    lr_max = 0.04
    stall = 0
    max_stall = 80
    velocity = np.zeros(n)
    prev_a = None
    prev_g = None
    iteration = 0
    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        n_active = max(2, min(n_2nm1, int(n_2nm1 * active_frac)))
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_active]
        active_vals = c[active_idx]
        w = np.zeros(n_2nm1)
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w[active_idx] = np.exp(-4.0 * (active_vals.max() - active_vals) / spread)
        else:
            w[active_idx] = 1.0 / n_active
        w = w / w.sum()
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break
        if prev_a is not None and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g_centered - prev_g
            s_norm = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm > 1e-18 and abs(sy) > 1e-18:
                lr_bb = s_norm / abs(sy)
                lr = np.clip(lr_bb, lr_min, lr_max)
        velocity = momentum * velocity - lr * g_centered
        alpha = 1.0
        improved = False
        for ls in range(20):
            a_try = a + alpha * velocity
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        if improved:
            prev_a = a.copy()
            prev_g = g_centered.copy()
            lr = min(lr_max, lr * 1.02)
            active_frac = max(0.001, active_frac * 0.995)
        else:
            velocity = np.zeros(n)
            lr = max(lr_min, lr * 0.7)
            active_frac = min(0.04, active_frac * 1.15)
        if stall > max_stall:
            active_frac = min(0.04, active_frac * 2.0)
            lr = max(lr_min, lr * 0.4)
            momentum = min(0.8, momentum + 0.05)
            a = best_a.copy()
            velocity = np.zeros(n)
            prev_a = None
            prev_g = None
            stall = 0
    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with top-K smoothing and annealing schedule."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score_normalized(a)
    current_a = a.copy()
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.004)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    schedule = [
        (1.0, 0.08), (0.7, 0.06), (0.5, 0.04), (0.3, 0.025),
        (0.2, 0.015), (0.1, 0.008), (0.05, 0.005), (0.02, 0.003),
        (0.01, 0.002), (0.005, 0.001), (0.002, 0.0005),
    ]
    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 8:
            break
        beta = base_beta * beta_mult
        K = max(3, int((2 * n - 1) * K_frac))
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(30, int(remaining / (est_time * 30))))
        prob = _TopkSmoProb(beta, K, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 30}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score_normalized(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.3) ** 2) + np.exp(-12.0 * (t - 0.7) ** 2) + 0.03
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power_taper":
        t = np.linspace(0, 1, n)
        a = (t * (1 - t)) ** 0.4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    """Very fine active-set polish with small steps."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score_normalized(a)
    lr = 0.002
    tol = 0.002
    no_improve = 0
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-16:
            break
        alpha = min(lr, 0.2 / (gnorm + 1e-15))
        improved = False
        for ls in range(25):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            no_improve += 1
            if no_improve % 10 == 0:
                tol = max(0.0003, tol * 0.7)
                lr *= 0.7
            if no_improve > 150:
                a = best_a.copy()
                tol = max(0.0003, tol * 0.5)
                lr = 0.001
                no_improve = 0
        else:
            tol = max(0.0003, tol * 0.985)
            lr = min(0.006, lr * 1.01)
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
