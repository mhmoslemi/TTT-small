```python
"""
Optimizer for minimizing
    2*n*max(conv(a,a)) / sum(a)^2
over non-negative coefficient vectors a.

The method combines:
  1. KKT-style active-set weighting:
       For current a, choose weights w on the largest convolution peaks so
       that the projected gradient of the weighted peak polynomial is as
       flat as possible.  This gives a much better mirror-descent direction
       than uniform/inverse-gap active-set weighting.
  2. Mirror descent in log-space with safe line searches.
  3. Adaptive L-BFGS on a smoothed max-projector (log-sum-exp) over all
     convolution coefficients.
  4. Diversified multi-start, iterated local search, and multi-length
     exploration.
  5. Cheap global symmetry/shift post-processing.

Everything is scale-invariant, so we optimize with sum(a)=1.
"""

import numpy as np
import time
from scipy.optimize import minimize

try:
    from scipy.optimize import lsq_linear
except Exception:
    lsq_linear = None


# ----------------------------------------------------------------------
# Basic numerical helpers
# ----------------------------------------------------------------------

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(1, k)


def _auto_convolve(a):
    """Full linear convolution a*a, optimized by one real FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    out = 2 * n - 1
    N = _next_pow2(out)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:out]
    # The exact convolution of a nonnegative vector is nonnegative.
    c = np.where(c < 0.0, 0.0, c)
    return c


def _full_convolve(a, b):
    """Full linear convolution a*b."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    out = a.size + b.size - 1
    if out <= 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(out)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out]


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones_like(a) / float(a.size)
    return a / s


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return np.array([], dtype=np.float64)
    z = z - float(np.max(z))
    e = np.exp(np.clip(z, -745.0, 700.0))
    s = float(e.sum())
    if (not np.isfinite(s)) or s <= 1e-300:
        return np.ones_like(z) / float(z.size)
    return e / s


def _logp(a):
    return np.log(np.maximum(a, 1e-300))


def _score_normalized(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    c = _auto_convolve(a)
    return float(2.0 * n * float(c.max()))


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    s = float(a.sum())
    if s <= 1e-300:
        return np.inf
    return _score_normalized(a / s)


def _make_profile(n, kind="uniform"):
    if n <= 0:
        return np.array([1.0], dtype=np.float64)

    t = np.linspace(0.0, 1.0, n)

    if kind == "uniform":
        a = np.ones(n, dtype=np.float64)
    elif kind == "triangle":
        a = 1.0 - np.abs(2.0 * t - 1.0)
    elif kind == "hann":
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    elif kind == "gaussian":
        a = np.exp(-8.0 * (t - 0.5) ** 2)
    elif kind == "raised_cos2":
        a = (0.5 * (1.0 + np.cos(2.0 * np.pi * t))) ** 2
    elif kind == "edge":
        a = 0.5 + 0.5 * np.cos(np.pi * t)
    elif kind == "comb2":
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * 2.0 * t)
    elif kind == "comb3":
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * 3.0 * t)
    elif kind == "comb5":
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * 5.0 * t)
    elif kind == "multispec":
        a = np.ones(n, dtype=np.float64)
        rng = np.random.default_rng(12345)
        for k in range(1, 12):
            amp = 1.0 / (k + 1.5)
            ph = rng.uniform(0.0, 2.0 * np.pi)
            a += amp * np.cos(2.0 * np.pi * float(k) * t + ph)
    else:
        a = np.ones(n, dtype=np.float64)

    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    if s <= 0.0:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    return a / s


def _resample(a, new_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        new_n = max(1, int(new_n))
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    if a.size == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    return _normalize(np.interp(x_new, x_old, a))

# KKT-style active-set mirror descent
# ----------------------------------------------------------------------

def _kkt_weights_and_residual(a, c, tol_frac, max_m, use_kkt=True):
    """
    Compute a descent direction r.

    If use_kkt is True, solve a small constrained least-squares problem
    for weights w on selected active peaks minimizing
        ||T w - 1/n||^2
    subject to sum(w)=1, w>=0, where T w is the projected gradient of the
    weighted peak polynomial.  The returned residual is r = T w - 1/n.

    If use_kkt is False, use inverse-gap weights on the selected peaks.
    """
    n = len(a)
    L = len(c)
    cmax = float(c.max())
    if cmax <= 1e-300:
        return None

    max_m = max(1, min(int(max_m), n, L))

    if max_m >= L:
        idx = np.arange(L, dtype=np.int64)
    else:
        idx = np.argpartition(c, -max_m)[:max_m]

    thresh = cmax - max(tol_frac * cmax, 1e-15)
    above = np.where(c >= thresh)[0]

    if above.size > max_m:
        # Keep only the highest candidates.
        if max_m >= L:
            idx = np.arange(L, dtype=np.int64)
        else:
            idx = np.argpartition(c, -max_m)[:max_m]
    else:
        # Make sure we include at least a small top set.
        small_m = min(20, L)
        if small_m >= L:
            small = np.arange(L, dtype=np.int64)
        else:
            small = np.argpartition(c, -small_m)[:small_m]
        idx = np.unique(np.concatenate((above, small)))
        if idx.size > max_m:
            order = np.argsort(c[idx])[::-1][:max_m]
            idx = idx[order]

    m = int(idx.size)
    if m == 0:
        return None
    if m == 1:
        wv = np.ones(1, dtype=np.float64)
    elif use_kkt and lsq_linear is not None:
        # Sample rows to keep the QP fast while retaining enough geometry.
        s = min(n, max(1000, 20 * m))
        half = max(1, s // 2)

        if n <= half:
            top_rows = np.arange(n, dtype=np.int64)
        else:
            top_rows = np.argsort(-a)[:half]

        s2 = max(0, s - top_rows.size)
        if s2 > 0:
            uniform_rows = np.linspace(0, n - 1, s2, dtype=np.int64)
            rows = np.unique(np.concatenate((top_rows, uniform_rows)))
        else:
            rows = top_rows

        rows = rows[:s]
        A = np.zeros((rows.size, m), dtype=np.float64)

        for ii, k in enumerate(idx):
            kk = int(k)
            if kk < n:
                start, end = 0, kk
            else:
                start, end = kk - n + 1, n - 1

            mask = (rows >= start) & (rows <= end)
            if mask.any():
                A[mask, ii] = a[kk - rows[mask]]

        b = np.full(rows.size, 1.0 / n, dtype=np.float64)
        wv = None
        try:
            Aeq = np.ones((1, m), dtype=np.float64)
            beq = np.array([1.0], dtype=np.float64)
            sol = lsq_linear(
                A,
                b,
                bounds=(np.zeros(m, dtype=np.float64), np.ones(m, dtype=np.float64)),
                A_eq=Aeq,
                b_eq=beq,
                method="trf",
            )
            if sol.success and sol.x is not None:
                wv = np.asarray(sol.x, dtype=np.float64)
        except Exception:
            wv = None

        if wv is None or not np.all(np.isfinite(wv)):
            wv = np.ones(m, dtype=np.float64)
    else:
        # Fallback: inverse-gap weighting over selected peaks.
        cact = c[idx]
        gap = (cmax - cact) + (1e-12 * cmax + 1e-300)
        wv = 1.0 / np.maximum(gap, 1e-300)

    wv = np.maximum(wv, 0.0)
    sw = float(wv.sum())
    if sw <= 1e-300:
        wv = np.ones(m, dtype=np.float64)
        sw = float(m)
    else:
        wv = wv / sw

    # Full residual r = T w - 1/n.
    r = np.zeros(n, dtype=np.float64)
    for ii, k in enumerate(idx):
        wk = float(wv[ii])
        if wk <= 0.0:
            continue
        kk = int(k)
        if kk < n:
            r[0:kk + 1] += wk * a[kk::-1]
        else:
            start = kk - n + 1
            stop = kk - n
            r[start:n] += wk * a[n - 1:stop:-1]

    r -= 1.0 / n
    r -= float(r.mean())
    return r, idx, wv


def _kkt_mirror_polish(a_init, deadline, verbose=False,
                       start_tol=0.01, start_m=60,
                       max_iters=100000, max_m_cap=240):
    a = _normalize(a_init)
    n = len(a)
    z = _logp(a)

    c0 = _auto_convolve(a)
    best_a = a.copy()
    best_sc = float(2.0 * n * float(c0.max()))

    tol = float(start_tol)
    max_m = min(int(start_m), int(max_m_cap), n)
    no_improve = 0
    it = 0

    etas_frac = (
        1e-5, 3e-5, 1e-4, 3e-4,
        1e-3, 3e-3, 0.01, 0.03,
        0.1, 0.3, 1.0
    )

    while time.monotonic() < deadline and it < max_iters:
        c = _auto_convolve(a)
        cur_sc = float(2.0 * n * float(c.max()))

        if cur_sc < best_sc - 1e-12:
            best_sc = cur_sc
            best_a = a.copy()
            no_improve = 0
            if verbose and it < 50:
                print(f"    KKT it={it} score={best_sc:.10f} t={time.monotonic():.0f}s")

        res = None
        use_kkt_this = (lsq_linear is not None) and ((no_improve < 12) or (it % 3 == 0))
        try:
            res = _kkt_weights_and_residual(a, c, tol, max_m, use_kkt=use_kkt_this)
        except Exception:
            res = None

        if res is None:
            break

        r, idx, wv = res
        rstd = float(np.std(r))
        if (not np.isfinite(rstd)) or rstd <= 1e-15:
            break

        eta0 = 1.0 / rstd
        best_eta_sc = cur_sc
        best_eta_a = None

        for f in etas_frac:
            eta = eta0 * f
            if eta <= 0.0 or not np.isfinite(eta):
                continue
            zt = z - eta * r
            zt -= float(zt.mean())
            zt = np.clip(zt, -700.0, 700.0)
            at = _softmax(zt)
            sc2 = float(2.0 * n * float(_auto_convolve(at).max()))
            if sc2 < best_eta_sc - 1e-13:
                best_eta_sc = sc2
                best_eta_a = at

        if best_eta_a is not None:
            a = best_eta_a
            z = _logp(a)
            if tol > 2e-4:
                tol = max(2e-4, tol * 0.97)
            if max_m < max_m_cap:
                max_m = min(max_m_cap, max_m + 2)
        else:
            no_improve += 1
            if no_improve % 10 == 0:
                tol = max(5e-5, tol * 0.7)
                max_m = min(max_m_cap, max_m + 10)
            if no_improve >= 60:
                a = best_a.copy()
                z = _logp(a)
                no_improve = 0
                tol = max(2e-5, tol * 0.5)
                max_m = min(max_m_cap, max_m + 20)

        it += 1

    return best_a, best_sc

def _smooth_auto_obj(z, beta, n):
    z = z - float(np.mean(z))
    a = _softmax(z)
    c = _auto_convolve(a)
    cmax = float(c.max())
    shifted = (c - cmax) / beta
    e = np.exp(np.clip(shifted, -745.0, 0.0))
    s = float(e.sum())
    if (not np.isfinite(s)) or s <= 1e-300:
        return 2.0 * n * cmax
    return 2.0 * n * (cmax + beta * float(np.log(s)))


def _smooth_auto_grad(z, beta, n):
    z = z - float(np.mean(z))
    a = _softmax(z)
    c = _auto_convolve(a)
    cmax = float(c.max())
    shifted = (c - cmax) / beta
    e = np.exp(np.clip(shifted, -745.0, 0.0))
    s = float(e.sum())

    if (not np.isfinite(s)) or s <= 1e-300:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p = e / s

    a_rev = a[::-1]
    h = _full_convolve(p, a_rev)[n - 1:2 * n - 1]
    W = float(np.dot(p, c))
    grad = 4.0 * n * a * (h - W)
    grad -= float(grad.mean())
    return grad


class _LbfgsProblem:
    def __init__(self, beta, n):
        self.beta = float(beta)
        self.n = int(n)

    def f(self, z):
        return _smooth_auto_obj(z, self.beta, self.n)

    def g(self, z):
        return _smooth_auto_grad(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    a = _normalize(a_init)
    n = len(a)
    c = _auto_convolve(a)
    cmax = float(c.max())
    if cmax <= 1e-300:
        return a

    cs = np.sort(c)[::-1]
    gap = float(cs[0] - cs[min(50, len(cs) - 1)])
    base_beta = max(gap, cmax * 0.003)

    ratios = (
        1.0, 0.6, 0.35, 0.2, 0.1, 0.05, 0.025, 0.01,
        0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
        0.00005, 0.00002, 0.00001, 0.000004
    )

    z = _logp(a)
    z -= float(z.mean())

    current_a = a.copy()
    current_sc = float(2.0 * n * cmax)

    n2gr = _next_pow2(3 * n - 2)
    per_iter_est = 0.00000012 * float(n2gr) + 0.0008

    total = len(ratios)
    for i, ratio in enumerate(ratios):
        rem = deadline - float(min_time_remaining) - time.monotonic()
        if rem < 8.0:
            break

        remaining_betas = float(total - i)
        time_per_beta = rem / max(1.0, remaining_betas)
        max_iter = int(min(800, max(20, time_per_beta / max(per_iter_est, 0.002))))

        beta = base_beta * ratio
        if beta <= 0.0 or not np.isfinite(beta):
            continue

        try:
            prob = _LbfgsProblem(beta, n)
            res = minimize(
                prob.f,
                z,
                jac=prob.g,
                method="L-BFGS-B",
                options={
                    "maxiter": max_iter,
                    "ftol": 1e-14,
                    "gtol": 1e-12,
                    "maxcor": 40,
                    "maxls": 30,
                },
            )
            a_new = _softmax(res.x)
            sc_new = float(2.0 * n * float(_auto_convolve(a_new).max()))
            if sc_new < current_sc - 1e-12:
                current_a = a_new
                current_sc = sc_new
                z = _logp(a_new)
                z -= float(z.mean())
        except Exception:
            pass

    return current_a

def _quick_kkt_local(a_init, max_time, use_lbfgs=False):
    a = _normalize(a_init)
    end = time.monotonic() + max(0.5, float(max_time))

    if use_lbfgs:
        kend = min(end - 5.0, time.monotonic() + max(2.0, 0.35 * max_time))
        a, sc = _kkt_mirror_polish(
            a, kend,
            verbose=False,
            start_tol=0.005,
            start_m=50,
            max_iters=300,
            max_m_cap=140,
        )
        if time.monotonic() < end - 5.0:
            a2 = _run_lbfgs(a, end, min_time_remaining=5.0)
            if _score_normalized(a2) < _score_normalized(a):
                a = a2
    else:
        a, sc = _kkt_mirror_polish(
            a, end,
            verbose=False,
            start_tol=0.005,
            start_m=50,
            max_iters=300,
            max_m_cap=140,
        )

    return a


def _quick_optimize(a_init, max_time, use_lbfgs=True):
    return _quick_kkt_local(a_init, max_time, use_lbfgs=use_lbfgs)


def _stochastic_ils(a_init, deadline, rng, min_time_remaining=30.0):

    attempt = 0
    while time.monotonic() < deadline - float(min_time_remaining):
        kind, sigma = strategies[attempt % len(strategies)]

        elif kind == "sparse":

            factors = np.exp(sigma * rng.standard_normal(k))

            a_p[idx] *= factors
            a_p = _normalize(a_p)
        elif kind == "mirror":
            seg = max(16, n // 20)
            start = int(rng.integers(0, n - seg))
            a_p = current.copy()
            a_p[start:start + seg] = a_p[start:start + seg][::-1]
            z = _logp(a_p)
            z = z + 0.003 * rng.standard_normal(n)
            z -= float(z.mean())
            a_p = _softmax(z)
        elif kind == "shift":
            sh = int(rng.integers(-n // 4, n // 4 + 1))
            a_p = np.roll(current, sh)
            a_p = _normalize(a_p)
        else:
            a_p = current.copy()

        max_t = min(deadline - float(min_time_remaining) - time.monotonic(), 7.0)
        if max_t < 1.5:
            break

        a_opt = _quick_kkt_local(a_p, max_t, use_lbfgs=False)
        sc_opt = _score_normalized(a_opt)

        if sc_opt < best_sc - 1e-12:
            best_sc = sc_opt
            best = a_opt.copy()
            current = a_opt.copy()
            current_sc = sc_opt
            print(f"    ILS NEW BEST: {best_sc:.10f} ({kind}) t={time.monotonic():.0f}s")
        elif sc_opt < current_sc - 1e-12:
            current = a_opt.copy()
            current_sc = sc_opt

    return best, best_sc


def _try_local_transforms(a):
    a = _normalize(a)
    best = a.copy()
    best_sc = _score_normalized(best)
    n = len(a)

    cands = [(a + a[::-1]) * 0.5]
    for sh in (1, -1, n // 20, -n // 20, n // 10, -n // 10, n // 5, -n // 5):
        if sh != 0:
            cands.append(np.roll(a, sh))

    for cand in cands:
        cn = _normalize(cand)
        sc = _score_normalized(cn)
        if np.isfinite(sc) and sc < best_sc - 1e-14:
            best_sc = sc
            best = cn.copy()

    return best

# Main search driver

def propose_candidate(seed=42, budget_s=1800):
    t0 = time.monotonic()
    deadline = t0 + max(60.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_a = None
    best_sc = np.inf

    # Load parent if present.
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                candidate = parent_raw / s
                sc = _score_normalized(candidate)
                if np.isfinite(sc):
                    best_a = candidate
                    best_sc = sc
    except Exception:
        pass

    if best_a is None:
        n0 = 24576
        best_a = _make_profile(n0, "uniform")
        best_sc = _score_normalized(best_a)

    # If the parent is unexpectedly huge, reduce its length so all
    # polynomial operations remain fast.
    if len(best_a) > 49152:
        best_a = _resample(best_a, 32768)
        best_sc = _score_normalized(best_a)

    try:
        print(f"Start: n={len(best_a)}, score={best_sc:.10f}")
    except Exception:
        pass

    # Phase 0: KKT polish on the parent.
    if time.monotonic() < deadline - 1300.0:
        d = min(deadline - 1300.0, t0 + 180.0)
        a, sc = _kkt_mirror_polish(best_a, d, verbose=True,
                                   start_tol=0.008, start_m=60,
                                   max_iters=100000, max_m_cap=200)
        a = _try_local_transforms(a)
        sc = _score_normalized(a)
        if sc < best_sc:
            best_a = a
            best_sc = sc
            print(f"  P0 NEW BEST: {best_sc:.10f} t={time.monotonic()-t0:.0f}s")

    # Phase 1: L-BFGS on the best found so far.
    if time.monotonic() < deadline - 950.0:
        d = min(deadline - 950.0, t0 + 420.0)
        a = _run_lbfgs(best_a, d, min_time_remaining=80.0)
        a = _try_local_transforms(a)
        sc = _score_normalized(a)
        if sc < best_sc:
            best_a = a
            best_sc = sc
            print(f"  P1 NEW BEST: {best_sc:.10f} t={time.monotonic()-t0:.0f}s")

    # Phase 2: second KKT polish.
    if time.monotonic() < deadline - 720.0:
        d = min(deadline - 720.0, t0 + 620.0)
        a, sc = _kkt_mirror_polish(best_a, d, verbose=True,
                                   start_tol=0.004, start_m=90,
                                   max_iters=100000, max_m_cap=220)
        a = _try_local_transforms(a)
        sc = _score_normalized(a)
        if sc < best_sc:
            best_a = a
            best_sc = sc
            print(f"  P2 NEW BEST: {best_sc:.10f} t={time.monotonic()-t0:.0f}s")

    # Phase 3: diversified multi-start.
    if time.monotonic() < deadline - 450.0:
        n0 = len(best_a)
        starts = [("parent", best_a.copy())]
        for kind in ("uniform", "gaussian", "edge", "comb3", "comb5",
                     "raised_cos2", "multispec", "hann"):
            starts.append((kind, _make_profile(n0, kind)))

        remaining = deadline - time.monotonic() - 450.0
        if remaining > 60.0:
            per = max(15.0, min(55.0, remaining / float(len(starts))))
            for name, a0 in starts:
                if time.monotonic() > deadline - 450.0:
                    break
                if per < 10.0:
                    break
                a0 = _normalize(a0)
                a = _quick_optimize(a0, per, use_lbfgs=(per >= 30.0))
                a = _try_local_transforms(a)
                sc = _score_normalized(a)
                if sc < best_sc:
                    best_a = a
                    best_sc = sc
                    print(f"  P3 NEW BEST: {best_sc:.10f} [{name}] t={time.monotonic()-t0:.0f}s")

    # Phase 4: iterated local search.
    if time.monotonic() < deadline - 240.0:
        d = min(deadline - 240.0, time.monotonic() + 300.0)
        a, sc = _stochastic_ils(best_a, d, rng, min_time_remaining=80.0)
        a = _try_local_transforms(a)
        sc = _score_normalized(a)
        if sc < best_sc:
            best_a = a
            best_sc = sc
            print(f"  P4 NEW BEST: {best_sc:.10f} t={time.monotonic()-t0:.0f}s")

    # Phase 5: multi-length exploration.
    if time.monotonic() < deadline - 90.0:
        base_n = len(best_a)
        n_opts = [
            base_n // 2,
            base_n,
            max(4096, (3 * base_n) // 2),
            max(4096, min(65536, base_n * 2)),
            16384, 24576, 32768, 49152,
        ]
        seen = set()
        n_opts = [int(nn) for nn in n_opts if 4096 <= int(nn) <= 65536 and (seen.add(int(nn)) or True)]

        per_n = 28.0
        for nn in n_opts:
            if time.monotonic() > deadline - 80.0:
                break
            a0 = _resample(best_a, nn)
            a = _quick_optimize(a0, per_n, use_lbfgs=False)
            a = _try_local_transforms(a)
            sc = _score_normalized(a)
            if sc < best_sc:
                best_a = a
                best_sc = sc
                print(f"  P5 NEW BEST: {best_sc:.10f} (n={nn}) t={time.monotonic()-t0:.0f}s")

    # Phase 6: final polish.
    if time.monotonic() < deadline - 4.0:
        if time.monotonic() < deadline - 45.0:
            d = min(deadline - 45.0, time.monotonic() + 80.0)
            a, sc = _kkt_mirror_polish(best_a, d, verbose=True,
                                       start_tol=0.0015, start_m=100,
                                       max_iters=100000, max_m_cap=250)
            a = _try_local_transforms(a)
            sc = _score_normalized(a)
            if sc < best_sc:
                best_a = a
                best_sc = sc
                print(f"  P6a NEW BEST: {best_sc:.10f} t={time.monotonic()-t0:.0f}s")

        if time.monotonic() < deadline - 6.0:
            a = _run_lbfgs(best_a, deadline - 6.0, min_time_remaining=4.0)
            a = _try_local_transforms(a)
            sc = _score_normalized(a)
            if sc < best_sc:
                best_a = a
                best_sc = sc
                print(f"  P6b NEW BEST: {best_sc:.10f} t={time.monotonic()-t0:.0f}s")

    # Final stabilization: keep strictly positive entries, avoid clipping issues.
    final = np.asarray(best_a, dtype=np.float64).ravel()
    final = np.maximum(final, 0.0)
    final = final + 1e-18
    s = float(final.sum())
    if s <= 0.0:
        final = _make_profile(256, "uniform")
    else:
        final = final / s
    final = np.clip(final, 0.0, 1000.0)

    final_sc = _score_normalized(final)
    try:
        print(f"FINAL internal score={final_sc:.10f}, n={len(final)}, time={time.monotonic()-t0:.1f}s")
    except Exception:
        pass

    return [float(x) for x in final]
```
