Coordinate descent + L-BFGS hybrid optimizer for minimizing
2*n*max(conv(a,a))/sum(a)^2.
Key idea: cyclic coordinate descent where each 1D subproblem is solved
via bisection on the subgradient, followed by L-BFGS polishing.

import numpy as np
import time
from scipy.optimize import minimize

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-11:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

"""
Coordinate descent + L-BFGS hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses exact 1D subproblem solves via golden section search, multiple structured
initializations, and L-BFGS polishing with smooth max approximation.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _make_profile(n, kind="uniform", rng=None):

        return np.ones(n) / n
    elif kind == "tapered":

        return a / a.sum()
    elif kind == "cos_mod":

        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t / n)

    elif kind == "bimodal":

        a = np.exp(-20.0 * (t - 0.2) ** 2) + np.exp(-20.0 * (t - 0.8) ** 2) + 0.1
        return a / a.sum()
    elif kind == "u_shape":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * (2.0 * t - 1.0) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":

    elif kind == "cos_multi":

        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 5.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 13.0 * t / n)

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _eval_max_b(a, b, i, x):

    b is the current full convolution (with a[i] = current value).
    Returns the max value of the new convolution."""

    k_hi = min(2 * n - 2, i + n - 1)

def _solve_1d_coord(a, b, i, s, max_iter=40):
    """Solve 1D subproblem: find x >= 0 minimizing 2n*max_b/(s_new)^2.
    Uses golden section search."""
    n = len(a)
    old = a[i]
    s_prime = s - old  # sum of all other coords

    def objective(x):
        s_new = s_prime + x
        if s_new < 1e-15:
            return np.inf
        M = _eval_max_b(a, b, i, x)
        return 2.0 * n * M / (s_new * s_new)

    # Quick scan to bracket the minimum
    candidates = [0.0, old * 0.25, old * 0.5, old * 0.75, old,
                  old * 1.25, old * 1.5, old * 2.0, s_prime * 0.5]
    candidates = [max(0.0, c) for c in candidates]
    # Remove duplicates
    candidates = list(set(candidates))

    best_x = old
    best_obj = objective(old)

    for x in candidates:
        if x == old:
            continue
        obj = objective(x)
        if obj < best_obj:
            best_obj = obj
            best_x = x

    # If best is at boundary, check if we can do better
    if best_x == 0.0 or best_x == max(candidates):
        # Try a few more points
        for x in [best_x * 0.5, best_x * 2.0, s_prime / 10.0]:
            x = max(0.0, x)
            obj = objective(x)
            if obj < best_obj:
                best_obj = obj
                best_x = x

    # Golden section refinement
    # Bracket: [lo, hi]
    lo = 0.0
    hi = max(best_x * 2.0, s_prime * 0.5, old * 3.0 + 1e-10)
    # Make sure best_x is inside [lo, hi]
    if best_x < lo:
        lo = 0.0
    if best_x > hi:
        hi = best_x * 2.0

    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi - lo < 1e-15:
        return max(0.0, best_x)

    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = objective(c)
    fd = objective(d)

    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = objective(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = objective(d)
        if hi - lo < 1e-14 * max(1.0, abs(best_x)):
            break

    x_result = (lo + hi) / 2.0
    obj_result = objective(x_result)
    if obj_result < best_obj:
        best_x = x_result

    return max(0.0, best_x)

    """Cyclic coordinate descent with exact 1D solves."""
    n = len(a)
    a = a.copy().astype(np.float64)
    b = b.copy().astype(np.float64)
    s = float(a.sum())
    best_a = a.copy()
    best_score = 2.0 * n * float(b.max()) / (s * s)

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline - min_time:
            break
        sweep_improved = 0
        for i in range(n):
            if deadline is not None and sweep % 10 == 0 and time.monotonic() > deadline - min_time:
                break
            old = a[i]
            new = _solve_1d_coord(a, b, i, s)
            if abs(new - old) > 1e-14:
                # Update a[i]
                a[i] = new
                s += (new - old)
                # Update b
                d = new - old
                k_lo = max(0, i)
                k_hi = min(2 * n - 2, i + n - 1)
                if k_lo <= k_hi:
                    a_idx = np.arange(k_lo, k_hi + 1) - i
                    b[k_lo:k_hi + 1] += 2.0 * d * a[a_idx]
                    if k_lo <= 2 * i <= k_hi:
                        b[2 * i] -= d * d
                sweep_improved += 1

        cur_score = 2.0 * n * float(b.max()) / (s * s)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()

        if verbose and (sweep % 10 == 0 or sweep == 0):
            print(f"    CD sweep {sweep}: score={cur_score:.10f}, best={best_score:.10f}, improved={sweep_improved}")

        if sweep_improved == 0 and sweep > 2:
            if verbose:
                print(f"    CD converged at sweep {sweep}")
            break

    return best_a, best_score


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time=30.0):
    """L-BFGS with smooth max approximation and adaptive beta."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03125, 0.015, 0.0075, 0.00375,
                                  0.0019, 0.00095])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - min_time:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(3000, max(30, int(remaining / (est_time_per_iter * 8))))

        def obj_fn(z_):
            return _smoobj(z_, beta, n)

        def grad_fn(z_):
            return _smograd(z_, beta, n)

        try:
            result = minimize(
                obj_fn, z, jac=grad_fn, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-15, 'maxcor': 60, 'maxls': 40}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 3 == 0:
                    print(f"    L-BFGS beta_idx={i}: score={current_sc:.10f}")
        except Exception:
            pass

    return current_a


def _nesterov_quick(a_init, deadline, min_time=30.0, K=20, seed=0):
    """Quick Nesterov subgradient for escaping local minima."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    momentum = 0.8
    lr = 0.003
    t = 0
    no_improve = 0

    while time.monotonic() < deadline - min_time:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            lr = min(0.01, lr * 1.01)
        else:
            no_improve += 1
            if no_improve > 200:
                break
            lr *= 0.95

        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]

        # Subgradient of sum of active peaks
        grad = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                idx = k - np.arange(j_lo, j_hi + 1)
                grad[j_lo:j_hi + 1] += a[idx]
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_lo2 <= j_hi2:
                idx = np.arange(j_lo2, j_hi2 + 1) - k
                grad[j_lo2:j_hi2 + 1] += a[idx]

        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break

        eff_lr = lr / (gnorm + 1e-18) * 0.1
        v = momentum * v - eff_lr * grad_t
        a_cand = a + v
        a = _project_simplex(a_cand)

        if t % 500 == 0:
            print(f"    Nesterov t={t}: best={best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Coordinate descent + L-BFGS hybrid optimizer.
    Phase 1: Try multiple initializations
    Phase 2: Coordinate descent (main workhorse)
    Phase 3: L-BFGS polishing
    Phase 4: Multi-n exploration
    Phase 5: Final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(best_seq)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Quick coordinate descent on parent (50s) ===
    print(f"\n[P1] Quick CD on parent (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 1500, t_start + 50)
    if t_start < p1_deadline:
        b_init = _full_convolve(best_seq, best_seq)
        a_cd, sc_cd = _coord_descent(best_seq, b_init, max_sweeps=20,
                                     deadline=p1_deadline, min_time=5.0, verbose=False)
        if sc_cd < best_score:
            best_seq = a_cd.copy()
            best_score = sc_cd
            print(f"  CD improved: {best_score:.10f}")
        else:
            print(f"  CD: no improvement (best={best_score:.10f})")

    # === Phase 2: Try different initializations with CD (200s) ===
    print(f"\n[P2] Multi-init CD (t={time.monotonic()-t_start:.1f}s)")
    n = len(best_seq)
    inits = ["cos_mod", "bimodal", "u_shape", "flat_taper", "cos_multi", "tapered"]
    p2_deadline = min(deadline - 1200, t_start + 250)

    for kind in inits:
        if time.monotonic() > p2_deadline - 20:
            break
        a_init = _make_profile(n, kind, rng=rng)
        b_init = _full_convolve(a_init, a_init)
        sc_init = 2.0 * n * float(b_init.max()) / (a_init.sum() ** 2)
        a_cd, sc_cd = _coord_descent(a_init, b_init, max_sweeps=15,
                                     deadline=p2_deadline, min_time=10.0, verbose=False)
        if sc_cd < best_score:
            best_seq = a_cd.copy()
            best_score = sc_cd
            print(f"  NEW BEST from '{kind}': {best_score:.10f}")
        else:
            print(f"  '{kind}': init={sc_init:.6f}, cd={sc_cd:.6f} (no improvement)")

    # === Phase 3: Main coordinate descent (800s) ===
    print(f"\n[P3] Main CD (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 600, t_start + 1050)
    b_main = _full_convolve(best_seq, best_seq)
    a_main, sc_main = _coord_descent(best_seq, b_main, max_sweeps=200,
                                     deadline=p3_deadline, min_time=60.0, verbose=True)
    if sc_main < best_score:
        best_seq = a_main.copy()
        best_score = sc_main
        print(f"  NEW BEST: {best_score:.10f} [main_cd]")
    else:
        print(f"  CD: no improvement (best={best_score:.10f})")

    # === Phase 4: L-BFGS polishing (300s) ===
    print(f"\n[P4] L-BFGS polish (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 300, t_start + 1350)
    a_lb = _run_lbfgs(best_seq, p4_deadline, min_time=30.0)
    sc_lb = _score(a_lb)
    if sc_lb < best_score:
        best_seq = a_lb.copy()
        best_score = sc_lb
        print(f"  NEW BEST: {best_score:.10f} [lbfgs]")
    else:
        print(f"  L-BFGS: no improvement (best={best_score:.10f})")

    # === Phase 5: Second CD round after L-BFGS (200s) ===
    print(f"\n[P5] CD round 2 (t={time.monotonic()-t_start:.1f}s)")
    p5_deadline = min(deadline - 100, t_start + 1550)
    b_5 = _full_convolve(best_seq, best_seq)
    a_cd2, sc_cd2 = _coord_descent(best_seq, b_5, max_sweeps=50,
                                   deadline=p5_deadline, min_time=30.0, verbose=False)
    if sc_cd2 < best_score:
        best_seq = a_cd2.copy()
        best_score = sc_cd2
        print(f"  NEW BEST: {best_score:.10f} [cd2]")

    # === Phase 6: Multi-n exploration (200s) ===
    print(f"\n[P6] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    p6_deadline = min(deadline - 50, t_start + 1750)
    n_candidates = [n, n // 2, n * 2, 8192, 32768]
    n_candidates = [x for x in n_candidates if 1024 <= x <= 65536]
    seen = set()
    unique_ns = []
    for nn in n_candidates:
        if nn not in seen:
            seen.add(nn)
            unique_ns.append(nn)

    for n_try in unique_ns:
        if time.monotonic() > p6_deadline - 30:
            break
        if n_try == len(best_seq):
            continue
        a_try = _resample(best_seq, n_try)
        b_try = _full_convolve(a_try, a_try)
        sc_try = 2.0 * n_try * float(b_try.max()) / (a_try.sum() ** 2)
        print(f"  n={n_try}: resampled score={sc_try:.10f}")

        a_cd_n, sc_cd_n = _coord_descent(a_try, b_try, max_sweeps=20,
                                         deadline=p6_deadline, min_time=20.0, verbose=False)
        if sc_cd_n < best_score:
            best_seq = a_cd_n.copy()
            best_score = sc_cd_n
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_cd]")

        # Quick L-BFGS on this n
        if time.monotonic() < p6_deadline - 20:
            a_lb_n = _run_lbfgs(a_cd_n if sc_cd_n < sc_try else a_try,
                                p6_deadline, min_time=15.0)
            sc_lb_n = _score(a_lb_n)
            if sc_lb_n < best_score:
                best_seq = a_lb_n.copy()
                best_score = sc_lb_n
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_lbfgs]")

    # === Phase 7: Final Nesterov + L-BFGS polish ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 10 - time.monotonic()
    if remaining > 30:
        a_nes, sc_nes = _nesterov_quick(best_seq, deadline, min_time=10.0,
                                        K=15, seed=seed + 500)
        if sc_nes < best_score:
            best_seq = a_nes.copy()
            best_score = sc_nes
            print(f"  NEW BEST: {best_score:.10f} [nesterov]")

    remaining = deadline - 5 - time.monotonic()
    if remaining > 15:
        a_fin = _run_lbfgs(best_seq, deadline, min_time=5.0)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_lbfgs]")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
