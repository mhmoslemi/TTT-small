        else:
            lr = min(0.08, lr * 1.01)
    
    return best_a, best_sc


def _project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _logsumexp_smooth_optimize(a_init, deadline, min_time_remaining=30.0, n_betas=12):
    """Optimize using log-sum-exp smooth approximation of max."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.01)
    
    betas = base_beta * np.logspace(0, -n_betas + 1, n_betas)
    betas = sorted(betas, reverse=True)
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        est_time_per_eval = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(4000, max(50, int(remaining / (est_time_per_eval * 12))))
        
        def smo_obj(z_):
            a_ = _softmax(z_)
            c_ = _full_convolve(a_, a_)
            c_max = float(c_.max())
            shifted = (c_ - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                return 2.0 * n * c_max
            return 2.0 * n * (c_max + beta * np.log(s))
        
        def smo_grad(z_):
            a_ = _softmax(z_)
            c_ = _full_convolve(a_, a_)
            c_max = float(c_.max())
            shifted = (c_ - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e / s
            a_rev = a_[::-1]
            N = _next_pow2(3 * n)
            W = np.fft.rfft(w, N)
            AR = np.fft.rfft(a_rev, N)
            h = np.fft.irfft(W * AR, N)[n - 1:2 * n - 1]
            g = 2.0 * h
            # Weighted average for the smooth term
            Wc = float(np.dot(w, c_))
            h2 = np.zeros_like(h)
            # Need: sum_k w[k] * (c[k] - Wc) * 2*a[k-j]
            # Actually the gradient of beta*log(sum exp((c-cmax)/beta)) w.r.t. a[j]
            # = beta * sum_k exp((c[k]-cmax)/beta)/sum * dc[k]/da[j] / sum
            # = sum_k w[k] * 2*a[k-j]
            # Which is just 2*conv(w, a_rev)[n-1:2n-1]
            # So the full gradient is 4*n * a * (conv(w,a_rev)[n-1:2n-1] - W * something)
            # Let me just use the standard formula
            grad_a = 4.0 * n * a_ * (h - Wc * a_ / a_.sum())
            # Hmm, this isn't quite right. Let me use a simpler approach.
            # The gradient of 2n*(cmax + beta*log(s)) w.r.t. a[j]:
            # d(cmax)/da[j] * 2n + 2n*beta * d(log(s))/da[j]
            # d(log(s))/da[j] = (1/s) * sum_k exp((c[k]-cmax)/beta)/beta * dc[k]/da[j]
            # = sum_k w[k] * dc[k]/da[j] = 2 * sum_k w[k] * a[k-j] = 2*h[j]
            # d(cmax)/da[j] is the subgradient of max, which is 2*a[k*-j] where k* is argmax
            # For the smooth approximation, we can use the weighted version
            # Actually for the smooth max, the gradient is just 2n * sum_k w[k] * dc[k]/da[j]
            # = 2n * 2 * h[j] = 4n * h[j]
            # But we also need to account for the softmax parameterization
            g_a = 4.0 * n * h
            grad_z = g_a * a_ - a_ * np.dot(g_a, a_)
            return grad_z
        
        try:
            result = minimize(
                smo_obj, z, jac=smo_grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 80,
                    'maxls': 40,
                }
            )
            a_new = _softmax(result.x)
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new
                z = np.log(np.maximum(a, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return best_a, best_sc


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.where(np.abs(t - 0.5) < 0.3, 1.0, 0.5 * (1 + np.cos(np.pi * (np.abs(t - 0.5) - 0.3) / 0.2))[np.abs(t - 0.5) >= 0.3])
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.sin(2.0 * np.pi * 11.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        if rng is None:
            rng = np.random.default_rng(42)
        a = np.exp(rng.standard_normal(n) * 0.5)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _refine(a_init, deadline, min_time_remaining=60.0):
    """Multi-method refinement: L_p norm + active set + logsumexp smooth."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    # L_p norm with increasing p
    p_schedule = [4, 8, 16, 32, 64, 128, 256, 512]
    a_lp, sc_lp = _run_lp_lbfgs(a, p_schedule, deadline, min_time_remaining=min_time_remaining)
    if sc_lp < best_sc:
        best_sc = sc_lp
        best_a = a_lp
        a = a_lp
    
    # Active set equalization
    if time.monotonic() + min_time_remaining < deadline:
        a_eq, sc_eq = _active_set_equalize(a, deadline, min_time_remaining=min_time_remaining)
        if sc_eq < best_sc:
            best_sc = sc_eq
            best_a = a_eq
            a = a_eq
    
    # Log-sum-exp smooth
    if time.monotonic() + min_time_remaining < deadline:
        a_lse, sc_lse = _logsumexp_smooth_optimize(a, deadline, min_time_remaining=min_time_remaining)
        if sc_lse < best_sc:
            best_sc = sc_lse
            best_a = a_lse
            a = a_lse
    
    # Final active set
    if time.monotonic() + min_time_remaining < deadline:
        a_eq2, sc_eq2 = _active_set_equalize(a, deadline, min_time_remaining=min_time_remaining)
        if sc_eq2 < best_sc:
            best_sc = sc_eq2
            best_a = a_eq2
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase optimizer: L_p norm, active set, log-sum-exp, multi-n, diverse restarts."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")
    
    # === Phase 1: Refine parent with L_p + active set ===
    print(f"\n[P1] Refine parent (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 600, t_start + 400)
    a_p1, sc_p1 = _refine(best_seq, p1_deadline, min_time_remaining=30)
    if sc_p1 < best_score:
        best_score = sc_p1
        best_seq = a_p1
        print(f"  NEW BEST: {best_score:.10f} [refine_parent]")
    
    # === Phase 2: L_p norm with extended schedule ===
    print(f"\n[P2] Extended L_p (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 400, t_start + 700)
    p_schedule_long = [8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096]
    a_p2, sc_p2 = _run_lp_lbfgs(best_seq, p_schedule_long, p2_deadline, min_time_remaining=30)
    if sc_p2 < best_score:
        best_score = sc_p2
        best_seq = a_p2
        print(f"  NEW BEST: {best_score:.10f} [lp_extended]")
    
    # === Phase 3: Active set with large K ===
    print(f"\n[P3] Active set large K (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 300, t_start + 1000)
    a_p3, sc_p3 = _active_set_equalize(best_seq, p3_deadline, min_time_remaining=30, max_K=200)
    if sc_p3 < best_score:
        best_score = sc_p3
        best_seq = a_p3
        print(f"  NEW BEST: {best_score:.10f} [active_large_K]")
    
    # === Phase 4: Multi-n exploration ===
    print(f"\n[P4] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_current = len(best_seq)
    n_candidates = sorted(set([n_current, 16384, 24576, 32768, 8192]), reverse=True)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 200:
            break
        if n_try == n_current:
            a_init_n = best_seq.copy()
        else:
            a_init_n = _resample(best_seq, n_try)
        
        sc_init = _score(a_init_n)
        print(f"  n={n_try}: init score={sc_init:.10f}")
        
        # Quick refine
        p4_deadline = min(deadline - 100, time.monotonic() + 120)
        a_opt, sc_opt = _refine(a_init_n, p4_deadline, min_time_remaining=20)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n]")
    
    # === Phase 5: Diverse restarts ===
    print(f"\n[P5] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_seq)
    kinds = ["tapered", "raised_cosine", "plateau", "sine_mod", "bimodal"]
    for kind in kinds:
        if time.monotonic() > deadline - 80:
            break
        a_div = _make_profile(n_best, kind, rng)
        # Blend with current best for warm start
        blend = 0.3 * a_div + 0.7 * best_seq
        blend = blend / blend.sum()
        
        p5_deadline = min(deadline - 50, time.monotonic() + 100)
        a_opt, sc_opt = _refine(blend, p5_deadline, min_time_remaining=15)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt
            print(f"  NEW BEST: {best_score:.10f} [restart_{kind}]")
        else:
            print(f"  Restart {kind}: {sc_opt:.10f} (no improvement)")
    
    # === Phase 6: Final intensive refinement ===
    print(f"\n[P6] Final refinement (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 10 - time.monotonic()
    if remaining > 30:
        # L_p with very large p
        p_schedule_final = [128, 256, 512, 1024, 2048, 4096, 8192]
        a_fin, sc_fin = _run_lp_lbfgs(best_seq, p_schedule_final, deadline - 20, min_time_remaining=10)
        if sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin
            print(f"  NEW BEST: {best_score:.10f} [final_lp]")
        
        # Active set final
        if time.monotonic() < deadline - 15:
            a_fin2, sc_fin2 = _active_set_equalize(best_seq, deadline - 5, min_time_remaining=5, max_K=150)
            if sc_fin2 < best_score:
                best_score = sc_fin2
                best_seq = a_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_active]")
        
        # Log-sum-exp final
        if time.monotonic() < deadline - 8:
            a_fin3, sc_fin3 = _logsumexp_smooth_optimize(best_seq, deadline - 3, min_time_remaining=3)
            if sc_fin3 < best_score:
                best_score = sc_fin3
                best_seq = a_fin3
                print(f"  NEW BEST: {best_score:.10f} [final_lse]")
    
    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
