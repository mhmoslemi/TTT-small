def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve_hankel(W, n, rhs, max_iter=300, tol=1e-12):

    r = rhs.copy()
    p = r.copy()

    rs0 = rs

        return x

            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new

    return x

        best_result = None
        for restart_i in range(3 if ti < len(temp_schedule) // 2 else 1):
            if time.monotonic() + min_remaining > deadline:
                break
            remaining_r = deadline - min_remaining - time.monotonic()
            if remaining_r < 10:
                break
            max_iter_r = max(20, min(max_iter, int(remaining_r / (est_iter_time * 50))))
            z_start = z.copy()
            if restart_i > 0:
                rng_r = np.random.default_rng(42 + ti * 100 + restart_i)
                z_start = z_start + 0.01 * rng_r.standard_normal(n)
                z_start = z_start - z_start.mean()
            try:
                result = minimize(
                    obj_fn, z_start, jac=grad_fn,
                    method='L-BFGS-B',
                    options={'maxiter': max_iter_r, 'ftol': 1e-18, 'gtol': 1e-14,
                             'maxcor': 80, 'maxls': 100}
                )
                a_try = _softmax(result.x)
                sc_try = _score(a_try)
                if best_result is None or sc_try < current_sc:
                    best_result = a_try
                    current_sc = sc_try
            except Exception:
                pass

        if best_result is not None:
            current_a = best_result
            z = np.log(np.maximum(current_a, 1e-30))
            z = z - z.mean()

        if ti % 3 == 2:
            print(f"    L-BFGS temp {ti}: beta={beta_t:.6e}, score={current_sc:.10f}")

    return current_a, _score(current_a)


def _coordinate_descent(a_init, deadline, n_iterations=5000):
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())
    active_idx = np.where(c >= c_max * 0.999)[0]

    rng = np.random.default_rng(12345)
    no_improve = 0

    for it in range(n_iterations):
        if time.monotonic() > deadline:
            break
        if it % 100 == 0:
            idx = np.argpartition(c, -max(10, len(active_idx)))[-max(10, len(active_idx)):]
            active_idx = idx

        j = rng.integers(0, n)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        obj_before = c_max

        for delta in [0.01, -0.01, 0.005, -0.005, 0.001, -0.001]:
            a[j] = a[j] + delta
            if a[j] < 0:
                a[j] = 0
            s = a.sum()
            if s < 1e-15:
                a[j] -= delta
                continue
            a_norm = a / s
            c_try = _full_convolve(a_norm, a_norm)
            if c_try.max() < obj_before - 1e-18:
                obj_before = c_try.max()
                a = a_norm.copy()
                a[j] += delta * 0.3
                if a[j] < 0:
                    a[j] = 0
                a = a / a.sum()
            else:
                a[j] -= delta

        if it % 200 == 199:
            sc = _score(a)
            if sc < best_sc - 1e-16:
                best_sc = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 10:
                break

    return best_a, best_sc


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="gaussian"):
    t = np.linspace(0, 1, n)
    if kind == "gaussian":
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.01
    elif kind == "tukey_50":
        r = 0.5
        a = np.ones(n)
        edge = int(n * r / 2)
        if edge > 0:
            te = np.linspace(0, 1, edge)
            taper = 0.5 + 0.5 * np.cos(np.pi * te)
            a[:edge] = taper
            a[-edge:] = taper[::-1]
    elif kind == "gaussian_wide":
        a = np.exp(-2.0 * (t - 0.5) ** 2) + 0.02
    elif kind == "raised_cos":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-8)
    else:
        a = np.ones(n)
    return a / a.sum()


def propose_candidate():
    t_start = time.monotonic()
    total_budget = 1800.0
    deadline = t_start + total_budget - 30.0

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"imp={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                par_sc = _score(parent)
                print(f"Parent loaded: n={len(parent)}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        parent = _make_profile(16384, "gaussian")
        print(f"No parent, using Gaussian n=16384")
        update_best(parent, _score(parent), "init")

    n_parent = len(parent)
    print(f"\n{'='*60}")
    print(f"Starting optimization. Best so far: {best_score:.10f}, n={n_parent}")
    print(f"Deadline in {deadline - time.monotonic():.1f}s")
    print(f"{'='*60}")

    # Phase 1: Newton on parent with soft active set
    print(f"\n[Phase 1] Soft Newton on n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    p1_end = min(deadline - 800, t_start + 200)
    a_p1, sc_p1 = _newton_soft(parent, p1_end, max_rounds=200, width_init=0.02, width_min=1e-5)
    update_best(a_p1, sc_p1, "newton_parent")

    # Phase 2: L-BFGS plateau on best
    print(f"\n[Phase 2] L-BFGS plateau on n={len(best_seq)} (t={time.monotonic()-t_start:.1f}s)")
    p2_end = min(deadline - 500, t_start + 550)
    a_p2, sc_p2 = _lbfgs_plateau(best_seq, p2_end, min_remaining=300, n_temps=14)
    update_best(a_p2, sc_p2, "lbfgs_plateau")

    # Phase 3: Try larger n
    print(f"\n[Phase 3] Larger n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_tries = []
    for n_try in [65536, 131072]:
        if n_try > n_parent:
            n_tries.append(n_try)

    for n_try in n_tries:
        if time.monotonic() > deadline - 300:
            break
        a_res = _resample(best_seq, n_try)
        sc_res = _score(a_res)
        print(f"  Resampled to n={n_try}, score={sc_res:.10f}")

        if sc_res < best_score:
            update_best(a_res, sc_res, f"resample_n{n_try}")

        # Quick Newton
        t_new_end = min(deadline - 200, time.monotonic() + 60)
        if time.monotonic() < t_new_end:
            a_nw, sc_nw = _newton_soft(a_res, t_new_end, max_rounds=50, width_init=0.01)
            update_best(a_nw, sc_nw, f"newton_n{n_try}")

        # L-BFGS
        t_lb_end = min(deadline - 150, time.monotonic() + 120)
        if time.monotonic() < t_lb_end:
            a_lb, sc_lb = _lbfgs_plateau(best_seq, t_lb_end, min_remaining=100, n_temps=8)
            update_best(a_lb, sc_lb, f"lbfgs_n{n_try}")

    # Phase 4: Second Newton + L-BFGS cycle on best
    print(f"\n[Phase 4] Refinement cycle (t={time.monotonic()-t_start:.1f}s)")
    for cyc in range(4):
        if time.monotonic() > deadline - 100:
            break
        t_new = min(deadline - 80, time.monotonic() + 40)
        a_c1, sc_c1 = _newton_soft(best_seq, t_new, max_rounds=50,
                                    width_init=max(1e-5, best_score - 1.0 + 1e-4),
                                    width_min=1e-8)
        update_best(a_c1, sc_c1, f"cycle_newton_{cyc}")

        t_lb = min(deadline - 60, time.monotonic() + 60)
        if time.monotonic() < t_lb:
            a_c2, sc_c2 = _lbfgs_plateau(best_seq, t_lb, min_remaining=40,
                                          beta_init=None, n_temps=6)
            update_best(a_c2, sc_c2, f"cycle_lbfgs_{cyc}")

    # Phase 5: Intensive final L-BFGS
    print(f"\n[Phase 5] Final intensive L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        a_fin, sc_fin = _lbfgs_plateau(best_seq, deadline - 10, min_remaining=5,
                                        beta_init=None, n_temps=20)
        update_best(a_fin, sc_fin, "final_lbfgs")

    # Phase 6: Coordinate descent micro-polish
    print(f"\n[Phase 6] Coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_cd, sc_cd = _coordinate_descent(best_seq, deadline - 2, n_iterations=2000)
        update_best(a_cd, sc_cd, "coord_desc")

    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "gaussian")
            s = final.sum()
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "gaussian")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
