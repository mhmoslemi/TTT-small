def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax(z):

def _proj_simplex(v):
    """Project v onto the probability simplex {a: sum(a)=1, a>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _hvp_pv(v, p, n):
    """Compute H_p * v where H_p = 2 * sum_k p_k H_k.
    p has length 2n-1 (autocorrelation lags), v has length n.
    Uses FFT for the convolution-based Hessian-vector product.
    """
    b = v[::-1]  # reversal of v
    conv_len = len(b) + len(p) - 1
    N = _next_pow2(conv_len)
    Fb = np.fft.rfft(b, N)
    Fp = np.fft.rfft(p, N)
    conv = np.fft.irfft(Fb * Fp, N)[:conv_len]
    # H_p * v[m] = 2 * conv[n-1+m] - correction
    result = 2.0 * conv[n - 1:2 * n - 1].copy()
    # Diagonal correction: for each k where 2m=k, subtract v[m]
    # (H_k * v)[m] should be v[k-m] not 2*v[k-m] when 2m=k
    # The correction is: for k=0,2,4,...,2(n-1): subtract v[k//2]
    for k in range(0, 2 * n - 1, 2):
        m = k // 2
        if m < n and k < len(p):
            result[m] -= v[m] * p[k]
    return result


def _grad_r_k(a, k, n):
    """Compute dR(k)/da where R(k) = (a*a)[k] is the k-th lag of linear autocorrelation."""
    grad = np.zeros(n)
    for m in range(n):
        j = k - m
        if 0 <= j < n:
            if j == m:
                grad[m] = a[m]
            else:
                grad[m] = 2.0 * a[j]
    return grad


def _grad_smoothed(a, p, n):
    """Compute gradient of sum_k p_k * R(k) w.r.t. a."""
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                if j == m:
                    grad[m] += p[k] * a[m]
                else:
                    grad[m] += 2.0 * p[k] * a[j]
    return grad


def _grad_smoothed_fast(a, p, n):
    """Fast gradient computation using FFT-based Hessian-vector product trick.
    grad_k[m] = (H_k * a)[m] essentially.
    """
    # grad = H_p * a / 2 (since H_p = 2 * sum p_k H_k, and grad of sum p_k R(k) = H_p * a / 2 * 2 = H_p * a)
    # Wait: R(k) = a^T Q_k a, so dR(k)/da = (Q_k + Q_k^T) a = 2 Q_k a (since Q_k symmetric)
    # So d/d a [sum p_k R(k)] = sum p_k * 2 Q_k a = H_p * a (where H_p = 2 sum p_k Q_k)
    return _hvp_pv(a, p, n)

def _run_fw(a_init, deadline, min_time_remaining=60.0):
    """Frank-Wolfe / Conditional Gradient method for the smoothed problem."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for k in range(3000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Top-k active set
        k_act = min(10, len(c))
        top_k = np.argsort(c)[::-1][:k_act]
        p = np.zeros(2 * n - 1)
        p[top_k] = 1.0 / k_act

        grad = _grad_smoothed_fast(a, p, n)
        # FW step: find s = argmin grad, move toward e_s
        s = int(np.argmin(grad))
        # LAR step size
        # t = (grad[s] - grad[a.argmax()]) / ||a - e_s||^2 approximately
        t_fw = 2.0 / (k + 2)
        a_new = (1 - t_fw) * a + t_fw * np.zeros(n)
        a_new[s] += t_fw

        # Check if this improves
        c_new = _full_convolve(a_new, a_new)
        c_max_new = float(c_new.max())
        if c_max_new < c_max:
            a = a_new
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        else:
            # Try smaller step
            for factor in [0.5, 0.25, 0.1]:
                a_try = (1 - t_fw * factor) * a + t_fw * factor * np.zeros(n)
                a_try[s] += t_fw * factor
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try
                    break

    return best_a, best_sc


def _active_set_qp(a_init, deadline, min_time_remaining=60.0, max_rounds=100):
    """Active-set QP with progressive tightening using projected gradient inner solve."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    tol_schedule = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                    0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
                    0.000001, 0.0000005, 0.0000002, 0.0000001]
    tol_idx = 0
    rounds_per_tol = 5

    for rnd in range(max_rounds):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / active.sum()

        # Solve QP: min sum_k p_k R(k) s.t. a in simplex
        # Using projected gradient with adaptive step
        lr = 0.01
        for inner in range(50):
            grad = _grad_smoothed_fast(a, p, n)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            lr = 0.1 / (gnorm * (1.0 + 0.01 * inner))
            a_try = _proj_simplex(a - lr * grad)
            a = a_try

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
        elif rnd > 5:
            a = best_a.copy()

        if (rnd + 1) % rounds_per_tol == 0:
            tol_idx += 1

    return best_a, best_sc


def _flatten_direct(a_init, deadline, min_time_remaining=60.0):

            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        if peaks.sum() == 0:
            peaks[int(np.argmax(c))] = True
        p = np.zeros(2 * n - 1)
        p[peaks] = 1.0 / peaks.sum()

        # Gradient of the peak objective
        grad = _grad_smoothed_fast(a, p, n)
        # Step in negative gradient direction, project to simplex
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:

        step = 0.05 / (gnorm + 1e-10)

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * t
        return a / a.sum()
    elif kind == "ramp2":
        t = np.linspace(0, 1, n)
        a = 0.1 + 0.9 * t
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.3) ** 2) + 0.1
        return a / a.sum()
    elif kind == "bump2":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.7) ** 2) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s
