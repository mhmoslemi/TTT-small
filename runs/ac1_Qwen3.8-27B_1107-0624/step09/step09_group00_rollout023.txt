So C_S[i,j] = rho[|S[i] - S[j]|] = c[n-1 + |S[i] - S[j]|].

Yes, so C_S is determined by the autocorrelation values at the lags that are differences of active indices.

Now, for the implementation:
- I need to compute C_S efficiently. Since it's a function of the differences, I can compute it in O(m^2) time by looking up c[n-1+lag] for each pair.
- Then solve the (m+1) x (m+1) system.

"""
Active-set Newton optimizer exploiting symmetry for minimizing
2*n*max(conv(a,a))/sum(a)^2.

Algorithm:
1. Enforce WLOG symmetry (a_j = a_{n-1-j})
2. Active-set Newton: solve KKT system on working set of top-m lags
3. L-BFGS refinement with log-sum-exp smoothing
4. Multi-n exploration
5. Final Newton polish
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize_pos(a):
    """Normalize to sum=1 and clip at 0."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _enforce_symmetry(a):
    """Make a symmetric: a[j] = a[n-1-j]."""
    return (a + a[::-1]) / 2.0

def _active_set_newton(a, m=100, tol_frac=0.01):
    """
    One step of active-set Newton.
    Returns direction d (satisfying sum(d)=0, symmetric).
    """
    n = len(a)
    c = _full_convolve(a, a)
    t = float(c.max())
    
    # Select working set: top m lags
    sorted_idx = np.argsort(c)[::-1][:m]
    sorted_idx = np.sort(sorted_idx)  # sort by lag for cleaner Toeplitz
    
    active_c = c[sorted_idx]
    active_t = float(active_c.max())
    
    # b[k] = active_t - active_c[k] (we want to raise the lower ones to active_t)
    # Actually, we want to equalize all to a common value and then decrease.
    # For the Newton step, let's set t_new = active_t (keep the max)
    # and solve for d such that c_new[sorted_idx[k]] = active_t for all k.
    # This means: 2 * (d · shift_k(a)) = active_t - c[sorted_idx[k]] = b[k]
    
    b = active_t - active_c
    
    # If all are already equal, b = 0, and we need a different approach
    if np.max(np.abs(b)) < 1e-15:
        # All active lags are already equal; use a steepest descent on t
        # d = -projection of the "peak indicator" onto sum=0 subspace
        peak_mask = (c >= active_t - 1e-10)
        # Gradient of max c is the shift of a at the peak lag
        peak_idx = int(np.argmax(c))
        offset = n - 1 - peak_idx
        if offset >= 0:
            grad = np.zeros(n)
            grad[:n-offset] = a[offset:]
        else:
            grad[-offset:] = a[:n+offset]
        d = -grad
        d = _enforce_symmetry(d)
        d -= d.mean()
        return d
    
    # Build C_S: m x m matrix, C_S[i,j] = c[n-1 + |sorted_idx[i] - sorted_idx[j]|]
    lags = sorted_idx
    m_actual = len(lags)
    
    # Compute differences
    diff = np.abs(lags[:, None] - lags[None, :])  # m x m
    C_S = c[n - 1 + diff]  # Look up autocorrelation at those lags
    
    # Solve the (m+1) x (m+1) system:
    # [C_S + reg*I   1_m] [y]   [b]
    # [1_m^T          n  ] [nu] = [0]
    
    reg = 1e-12 * t
    A = np.zeros((m_actual + 1, m_actual + 1))
    A[:m_actual, :m_actual] = C_S + reg * np.eye(m_actual)
    A[:m_actual, m_actual] = 1.0
    A[m_actual, :m_actual] = 1.0
    A[m_actual, m_actual] = float(n)
    
    rhs = np.zeros(m_actual + 1)
    rhs[:m_actual] = b
    
    try:
        sol = np.linalg.solve(A, rhs)
    except np.linalg.LinAlgError:
        sol = np.linalg.lstsq(A, rhs, rcond=None)[0]
    
    y = sol[:m_actual]
    nu = sol[m_actual]
    
    # d = G^T * y + nu * 1
    # G^T * y = sum_i y[i] * shift_{lags[i]}(a)
    d = np.zeros(n)
    for i in range(m_actual):
        if abs(y[i]) < 1e-30:
            continue
        k = lags[i]
        offset = n - 1 - k
        if offset >= 0:
            d[:n - offset] += y[i] * a[offset:]
        else:
            d[-offset:] += y[i] * a[:n + offset]
    d += nu  # nu * ones(n)
    
    # Enforce symmetry
    d = _enforce_symmetry(d)
    # Re-project to sum=0
    d -= d.mean()
    
    # Check that d is a descent direction: it should decrease t
    # Verify: the change in c at the peak lag should be negative
    # (This is checked in the line search)
    
    return d

def _line_search(a, d, c_init, t_init, max_iter=50):
    """
    Backtracking line search. Returns (a_new, alpha).
    We want to decrease the score (which is proportional to max(c) when sum=1).
    """
    n = len(a)
    alpha = 1.0
    current_t = t_init
    
    for _ in range(max_iter):
        a_try = a + alpha * d
        a_try = _enforce_symmetry(a_try)
        a_try = _normalize_pos(a_try)
        
        c_try = _full_convolve(a_try, a_try)
        t_try = float(c_try.max())
        
        if t_try < current_t * (1 - 1e-10):  # Sufficient decrease
            return a_try, alpha
        
        alpha *= 0.5
    
    return a, 0.0

def _newton_optimize(a_init, max_iter=500, m=100, time_limit=None, t_start=None):
    """
    Active-set Newton optimization loop.
    """
    n = len(a_init)
    a = _enforce_symmetry(_normalize_pos(a_init.copy()))
    best_a = a.copy()
    best_score = _score(a)
    
    for it in range(max_iter):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        
        c = _full_convolve(a, a)
        t = float(c.max())
        current_score = 2.0 * n * t  # since sum(a) = 1
        
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
        
        # Check convergence
        c_sorted = np.sort(c)[::-1]
        spread = c_sorted[0] - c_sorted[min(10, len(c_sorted)-1)]
        if spread < 1e-12 * t:
            break
        
        # Newton step
        d = _active_set_newton(a, m=m)
        
        # Check if d is meaningful
        if np.linalg.norm(d) < 1e-15:
            break
        
        # Line search
        a_new, alpha = _line_search(a, d, c, t)
        
        if alpha == 0:
            # Try smaller m
            d = _active_set_newton(a, m=max(10, m // 2))
            a_new, alpha = _line_search(a, d, c, t)
            if alpha == 0:
                m = max(10, m // 2)
                continue
        
        a = a_new
        
        # Optional: if score worsened significantly, revert
        if it > 0 and _score(a) > best_score * 1.01:
            a = best_a.copy()
    
    return best_a, best_score

def _make_profile(n, kind="symmetric_hann"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "hann":
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif kind == "triangle":
        a = 1.0 - np.abs(2 * t - 1)
    elif kind == "gaussian":
        a = np.exp(-4 * (t - 0.5) ** 2)
    elif kind == "raised_cos2":
        a = 0.5 * (1 + np.cos(2 * np.pi * t)) ** 2
    elif kind == "symmetric_hann":
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    else:
        a = np.ones(n)
    a = _enforce_symmetry(a)
    a = np.maximum(a, 1e-15)
    return a / a.sum()

def _smoobj(z, beta, n):
    a = np.exp(z - z.max())
    a = a / a.sum()
    a = _enforce_symmetry(a)
    a = np.maximum(a, 0)
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = np.exp(z - z.max())
    a = a / a.sum()
    a = _enforce_symmetry(a)
    a = np.maximum(a, 0)
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    # Symmetrize gradient
    grad = _enforce_symmetry(grad)
    return grad

def _lbfgs_refine(a_init, deadline, min_remaining=30):
    """L-BFGS refinement with adaptive beta annealing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.002)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015,
                                   0.008, 0.004, 0.002, 0.001, 0.0005,
                                   0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    
    # Parameterize in log-space for positivity
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        
        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 15))))
        
        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 80, 'maxls': 40}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            a_new = _enforce_symmetry(a_new)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return _enforce_symmetry(a.copy())
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = _enforce_symmetry(a_new)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

```python
"""
Active-set Newton optimizer exploiting WLOG symmetry for minimizing
2*n*max(conv(a,a))/sum(a)^2.

Key innovations over previous approach:
1. Enforces symmetry (a_j = a_{n-1-j}) which is WLOG by convexity
2. Uses 2nd-order active-set Newton method solving the KKT system
   (exploiting Toeplitz structure of the Gram matrix)
3. Working-set strategy with adaptive size
4. Steepest-descent fallback when Newton step fails
5. Multi-phase: Newton -> L-BFGS -> Newton -> multi-n -> final polish
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize_pos(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _enforce_symmetry(a):
    return (a + a[::-1]) / 2.0


def _compute_direction(a, lags, b, n):
    """Compute Newton direction from solved system."""
    m = len(lags)
    d = np.zeros(n)
    for i in range(m):
        if abs(b[i]) < 1e-30:
            continue
        s = int(lags[i])
        j_lo = max(0, s - n + 1)
        j_hi = min(n - 1, s)
        js = np.arange(j_lo, j_hi + 1)
        idx = s - js
        d[js] += b[i] * a[idx]
    return d


def _active_set_newton_dir(a, m_work=100, use_t_new=True):
    """
    Compute one Newton direction on the active set.
    Returns direction d (symmetric, sum=0) or None if failed.
    """
    n = len(a)
    c = _full_convolve(a, a)
    t = float(c.max())

    # Select working set: top m_work lags
    top_idx = np.argsort(c)[::-1][:m_work]
    lags = np.sort(top_idx)
    active_c = c[lags]
    active_t = float(active_c.max())

    b = active_t - active_c

    if np.max(np.abs(b)) < 1e-14 * t:
        # All working set lags already equal; use steepest descent on peak
        peak_idx = int(np.argmax(c))
        s = peak_idx
        j_lo = max(0, s - n + 1)
        j_hi = min(n - 1, s)
        grad = np.zeros(n)
        js = np.arange(j_lo, j_hi + 1)
        idx = s - js
        grad[js] = a[idx]
        d = -grad
        d = _enforce_symmetry(d)
        d -= d.mean()
        return d if np.linalg.norm(d) > 1e-15 else None

    m_actual = len(lags)

    # Build Gram matrix C_S[i,j] = c[n-1 + |lags[i] - lags[j]|]
    diff = np.abs(lags[:, None].astype(np.int64) - lags[None, :].astype(np.int64))
    C_S = c[n - 1 + diff]

    # Solve (m+1) x (m+1) KKT system:
    # [C_S + reg*I   1] [y]   [b]
    # [1^T            n] [nu] = [0]
    reg = max(1e-14 * t, 1e-30)
    A = np.zeros((m_actual + 1, m_actual + 1))
    A[:m_actual, :m_actual] = C_S + reg * np.eye(m_actual)
    A[:m_actual, m_actual] = 1.0
    A[m_actual, :m_actual] = 1.0
    A[m_actual, m_actual] = float(n)

    rhs = np.zeros(m_actual + 1)
    rhs[:m_actual] = b

    try:
        sol = np.linalg.solve(A, rhs)
    except np.linalg.LinAlgError:
        try:
            sol = np.linalg.lstsq(A, rhs, rcond=None)[0]
        except Exception:
            return None

    y = sol[:m_actual]
    nu = sol[m_actual]

    # d = G^T * y + nu * ones
    d = _compute_direction(a, lags, y, n)
    d += nu

    # Enforce symmetry and sum=0
    d = _enforce_symmetry(d)
    d -= d.mean()

    if np.linalg.norm(d) < 1e-15:
        return None
    return d


def _steepest_descent_dir(a, m_smooth=20):
    """Steepest descent direction using smoothed max."""
    n = len(a)
    c = _full_convolve(a, a)
    t = float(c.max())
    # Use inverse-gap weighting on top peaks
    c_sorted = np.sort(c)[::-1]
    threshold = c_sorted[min(m_smooth - 1, len(c_sorted) - 1)]
    active = c >= threshold
    eps = 1e-12 * t
    w = np.zeros_like(c)
    w[active] = 1.0 / (t - c[active] + eps)
    w = w / w.sum()

    # Gradient of smoothed max
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 2.0 * n * a * (h - W)
    d = -grad
    d = _enforce_symmetry(d)
    d -= d.mean()
    return d


def _line_search(a, d, t_init, c_fft=None, max_step=2.0):
    """Backtracking line search along direction d."""
    n = len(a)
    alpha = min(1.0, max_step)
    d_norm = np.linalg.norm(d)
    if d_norm < 1e-30:
        return a, 0.0
    d_scaled = d / d_norm

    for _ in range(40):
        a_try = a + alpha * d_scaled
        a_try = _enforce_symmetry(a_try)
        a_try = _normalize_pos(a_try)
        c_try = _full_convolve(a_try, a_try)
        t_try = float(c_try.max())

        if t_try < t_init * (1.0 - 1e-10):
            return a_try, alpha
        alpha *= 0.5

    return a, 0.0


def _newton_optimize(a_init, max_iter=300, m_work=100, time_limit=None):
    """Main active-set Newton optimization loop."""
    n = len(a_init)
    a = _enforce_symmetry(_normalize_pos(a_init.copy()))
    best_a = a.copy()
    best_score = _score(a)
    stall_count = 0
    m_current = m_work

    for it in range(max_iter):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        c = _full_convolve(a, a)
        t = float(c.max())
        current_score = 2.0 * n * t

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Check convergence: top lags are very flat
        c_sorted = np.sort(c)[::-1]
        spread = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
        if spread < 1e-13 * t:
            break

        # Get Newton direction
        d = _active_set_newton_dir(a, m_work=m_current)

        if d is None:
            # Fallback to steepest descent
            d = _steepest_descent_dir(a)

        if d is None or np.linalg.norm(d) < 1e-15:
            break

        a_new, alpha = _line_search(a, d, t)

        if alpha == 0.0:
            # Try steepest descent
            d_sd = _steepest_descent_dir(a)
            if d_sd is not None and np.linalg.norm(d_sd) > 1e-15:
                a_new, alpha = _line_search(a, d_sd, t)

        if alpha == 0.0:
            stall_count += 5
            if stall_count > 100:
                # Reset and try smaller working set
                a = best_a.copy()
                m_current = max(10, m_current // 2)
                stall_count = 0
            continue

        a = a_new

        # Adapt working set size
        if stall_count > 30:
            m_current = max(10, m_current // 2)
            stall_count = 0
        elif stall_count == 0 and m_current < m_work:
            m_current = min(m_work, m_current * 2)

    return best_a, best_score


def _make_profile(n, kind="hann"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "hann":
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif kind == "triangle":
        a = 1.0 - np.abs(2 * t - 1)
    elif kind == "gaussian":
        a = np.exp(-3.0 * (t - 0.5) ** 2)
    elif kind == "raised_cos2":
        a = 0.5 * (1 + np.cos(2 * np.pi * t)) ** 2
    elif kind == "flat_top":
        # Flat-top window: minimizes scalloping loss
        a0 = 0.21557895
        a1 = 0.41663158
        a2 = 0.277263158
        a3 = 0.083578947
        a4 = 0.006947368
        a = a0 - a1 * np.cos(2 * np.pi * t) + a2 * np.cos(4 * np.pi * t) \
            - a3 * np.cos(6 * np.pi * t) + a4 * np.cos(8 * np.pi * t)
    else:
        a = np.ones(n)
    a = _enforce_symmetry(a)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _smoobj(z, beta, n):
    a = np.exp(z - z.max())
    a = a / a.sum()
    a = _enforce_symmetry(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
