Let me think about the gradient computation more carefully.

If a = softmax(z), then:
∂a/∂z = a * (I - a*a^T) (the Jacobian of softmax)

And ∂(a*a)[k]/∂a = 2 * a_reversed_shifted (as computed above).

So ∂(a*a)[k]/∂z = (∂(a*a)[k]/∂a) * (∂a/∂z)

This is more complex. The previous code handles this by:
1. Computing w = softmax of (conv - max)/beta (the weights for the smooth max)
2. Computing the gradient of the weighted sum of conv values with respect to a
3. Then mapping back through the softmax Jacobian

Actually, let me look at the previous code's gradient more carefully:

```python
def _smograd(z, beta, n):

    wc = _full_convolve(w, a_rev)

Let me verify this. The objective is:
f(z) = 2n * (c_max + beta * log(sum exp((c - c_max)/beta)))

where c = conv(a,a) and a = softmax(z).

∂f/∂z = 2n * beta * (sum_k w_k * ∂c_k/∂z) / (sum_k exp((c_k-c_max)/beta))
       = 2n * sum_k w_k * ∂c_k/∂z

where w_k = exp((c_k-c_max)/beta) / sum_j exp((c_j-c_max)/beta).

Now, ∂c_k/∂z_j = sum_i ∂c_k/∂a_i * ∂a_i/∂z_j

∂c_k/∂a_i = 2*a[k-i] (if 0 <= k-i < n) [as derived above]

∂a_i/∂z_j = a_i * (δ_{i,j} - a_j)

So ∂c_k/∂z_j = sum_i 2*a[k-i] * a_i * (δ_{i,j} - a_j)
= 2*a[k-j]*a_j - 2*a_j * sum_i a[k-i]*a_i
= 2*a[j]*a[k-j] - 2*a[j]*c_k

Summing over k with weights w_k:
sum_k w_k * ∂c_k/∂z_j = 2*a[j] * sum_k w_k * a[k-j] - 2*a[j] * sum_k w_k * c_k
= 2*a[j] * (w * a_rev)[j+n-1] - 2*a[j] * W

where (w * a_rev) is the convolution of w with a reversed, and W = sum_k w_k * c_k.

So ∂f/∂z_j = 2n * 2 * a[j] * ((w * a_rev)[j+n-1] - W)
= 4n * a[j] * ((w * a_rev)[j+n-1] - W)

This matches the code! Good.

Now, the key computation is (w * a_rev)[j+n-1] for all j, which is a convolution that can be done in O(n log n) with FFT.

Let me also consider using `scipy.optimize.minimize` with L-BFGS-B, which is what the previous code does. The main advantage of L-BFGS-B is that it handles the optimization in the z-space (unconstrained) and the softmax ensures a >= 0, sum(a) = 1.

One issue with the softmax parameterization: for very large n, some a[j] might become extremely small (underflow in exp), which could cause numerical issues. To mitigate this, I can:
- Use a "soft" softmax: a[j] = (exp(z_j) + eps) / (sum(exp(z) + eps*n))
- Or use the log-sum-exp trick carefully.

Actually, the standard softmax with the max subtraction should be fine for n up to 100k or so.

```python
"""
Multi-scale L-BFGS optimizer with aggressive beta cooling and active-set
refinement for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative
sequences. Uses FFT-based gradient computation and exploits the structure
of the self-convolution gradient.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    """FFT-based full convolution."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    """Compute the objective: 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    """Project x onto the probability simplex."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _smooth_obj(z, beta, n):
    """Smooth approximation of 2n * max(conv(softmax(z), softmax(z)))."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smooth_grad(z, beta, n):
    """Gradient of smooth objective w.r.t. z."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _SmoothProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smooth_obj(z, self.beta, self.n)

    def grad(self, z):
        return _smooth_grad(z, self.beta, self.n)


def _run_lbfgs(a_init, time_budget, beta_start=None, beta_end=None, max_total_iter=50000):
    """Run L-BFGS with geometric beta cooling schedule."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    
    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    
    if beta_start is None:
        beta_start = max_c * 0.05
    if beta_end is None:
        beta_end = max_c * 1e-7
    
    # Geometric schedule
    n_betas = 20
    ratios = (beta_end / beta_start) ** (1.0 / (n_betas - 1))
    betas = [beta_start * (ratios ** i) for i in range(n_betas)]
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    t_start = time.monotonic()
    for i, beta in enumerate(betas):
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining < 3:
            break
        
        prob = _SmoothProblem(beta, n)
        
        # Estimate iterations based on remaining time
        # Each iteration takes roughly 2 FFT convolutions of size ~2n
        t0 = time.monotonic()
        # Quick timing
        _ = _full_convolve(current_a, current_a)
        t_conv = time.monotonic() - t0
        est_iter_time = t_conv * 6  # rough estimate per L-BFGS iteration
        
        max_iter = max(20, min(5000, int(remaining / est_iter_time * 0.7)))
        
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-15,
                    'gtol': 1e-10,
                    'maxcor': 50,
                    'maxls': 30,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            elif sc_new < current_sc:
                # Tiny improvement, still update
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception as e:
            pass
        
        if i % 5 == 0:
            print(f"      beta={beta:.3e}: score={current_sc:.10f}, iter={max_iter}, t={time.monotonic()-t_start:.1f}s")
    
    return current_a, current_sc


def _subgradient_descent(a_init, time_budget, n_top_k=100):
    """Subgradient method focusing on top-k peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    v = np.zeros(n)
    momentum = 0.9
    lr = 0.01
    t_start = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t_start < time_budget - 5:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
        
        # Top-k peak indices
        k_sel = min(n_top_k, 2*n-1)
        top_idx = np.argsort(c)[::-1][:k_sel]
        
        # Subgradient: sum of top-k gradients
        a_rev = a[::-1]
        indicator = np.zeros(2*n-1)
        indicator[top_idx] = 1.0
        grad_conv = _full_convolve(indicator, a_rev)
        grad = 2.0 * grad_conv[n-1:2*n-1]
        
        # Project to tangent space (sum = 0)
        grad = grad - grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Nesterov step
        v = momentum * v - lr * grad / gnorm
        a_new = _project_simplex(a + v)
        
        # Check improvement
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())
        
        if sc_new < current_sc:
            a = a_new
        else:
            lr *= 0.95
            v *= 0.5
            a = a_new  # still move
        
        if no_improve % 100 == 0:
            # Random perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            a = _project_simplex(a + 0.001 * rng.standard_normal(n))
            lr = 0.01
            v = np.zeros(n)
            no_improve = 0
    
    return best_a, best_sc


def _resample(a, new_n):
    """Resample sequence to new length via interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _active_set_optimize(a_init, time_budget):
    """Optimize by focusing on the active set of peak lags.
    
    Uses a bisection-like approach: try to equalize the top peaks.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    t_start = time.monotonic()
    K = 20  # number of peaks to equalize
    
    while time.monotonic() - t_start < time_budget - 10:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
        
        # Identify active set (peaks within 0.1% of max)
        threshold = c_max * 0.999
        active_mask = c >= threshold
        active_idx = np.where(active_mask)[0]
        
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(c)])
        
        # Grow active set if too small
        if len(active_idx) < K:
            sorted_idx = np.argsort(c)[::-1]
            active_idx = sorted_idx[:min(K, 2*n-1)]
        
        # Compute subgradient of max over active set
        a_rev = a[::-1]
        indicator = np.zeros(2*n-1)
        indicator[active_idx] = 1.0
        grad_conv = _full_convolve(indicator, a_rev)
        grad = 2.0 * grad_conv[n-1:2*n-1] / len(active_idx)
        
        # Project to simplex tangent
        grad = grad - grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Adam-like step
        lr = 0.01 / (1 + (time.monotonic() - t_start))  # decaying lr
        a_new = _project_simplex(a - lr * grad / gnorm)
        
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())
        
        if sc_new < current_sc * 1.001:  # accept small increases
            a = a_new
        else:
            lr *= 0.5
            a = a_new
    
    return best_a, best_sc


def _make_initial(a_init, n_target):
    """Create initial sequence of target length from a seed sequence."""
    if len(a_init) == n_target:
        return a_init.copy()
    return _resample(a_init, n_target)
