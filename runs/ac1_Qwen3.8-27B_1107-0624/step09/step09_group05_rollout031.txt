"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous:
- Power-mean (p-mean) smoothing as alternative to log-sum-exp (better conditioned)
- Newton-based active-set equalization for fast convergence to KKT
- Ramp-profile initialization (naturally low center autocorrelation)
- Wider multi-n exploration (up to 65536)
- Efficient gradient via FFT structure: grad[j] = 2*conv(w, a_rev)[n-1+j]
- Multi-restart with diverse structures
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsm_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _lsm_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _pm_value(c, p):
    """Power mean of order p."""
    if p < 1e-6:
        return float(c.mean()) ** (1.0 / p)
    cp = c ** p
    val = (cp.mean()) ** (1.0 / p)
    return float(val)


def _pm_weights(c, p):
    """Weights for power mean gradient: c[k]^(p-1) / (m * M_p^(p-1))."""
    m = len(c)
    Mp = _pm_value(c, p)
    if Mp < 1e-300:
        return np.ones(m) / m
    w = c ** (p - 1)
    w = w / (m * Mp ** (p - 1))
    return w


def _grad_via_conv(w, a, n):
    """Compute gradient vector: 2 * conv(w, a_rev)[n-1 : 2n-1]."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


class _LsmProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        return float(2.0 * self.n * _lsm_value(c, self.beta))

    def grad(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        w = _lsm_weights(c, self.beta)
        grad_a = _grad_via_conv(w, a, self.n)
        W = float(np.dot(grad_a, a))
        return a * (grad_a - W)


class _PmProb:
    def __init__(self, p, n):
        self.p = p
        self.n = n

    def __call__(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_safe = np.maximum(c, 1e-30)
        return float(2.0 * self.n * _pm_value(c_safe, self.p))

    def grad(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_safe = np.maximum(c, 1e-30)
        w = _pm_weights(c_safe, self.p)
        grad_a = _grad_via_conv(w, a, self.n)
        W = float(np.dot(grad_a, a))
        return a * (grad_a - W)


def _run_lbfgs_lsm(a_init, deadline, min_time_remaining=30.0, verbose=False):
    """L-BFGS with log-sum-exp, adaptive beta schedule."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c0_max * 0.001:
        gap = c0_max * 0.001
    if gap < 1e-15:
        gap = c0_max * 0.001

    multipliers = np.array([
        1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005,
        0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005,
        0.00002, 0.00001, 0.000005, 0.000002, 0.000001,
        0.0000005, 0.0000002, 0.0000001
    ])
    betas = gap * multipliers

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        size_factor = max(1.0, n / 4096.0)
        est_time_per_iter = 0.003 * size_factor
        max_iter = min(10000, max(200, int(remaining / (est_time_per_iter * 15))))
        if beta < gap * 0.001:
            max_iter = min(max_iter, 3000)

        prob = _LsmProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 100, 'maxls': 60}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"    lsm beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}")
        except Exception:
            pass

    return best_a, best_sc

def _newton_equalize(a_init, deadline, min_time_remaining=20.0, verbose=False):
    """Newton-based active-set equalization.
    
    At the optimum, c[k] should be equal for all k in the active set S.
    We use Newton's method on the system c[k_i] = c[k_0] for i = 1,...,|S|-1.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_active = 0.005
    max_iter = 500
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200:
            break

        # Identify active set
        threshold = c_max * (1.0 - tol_active)
        active = c >= threshold
        n_active = int(active.sum())
        if n_active < 2:
            # Only one active point; just do gradient step
            idx = int(np.argmax(c))
            active = np.zeros(len(c), dtype=bool)
            active[idx] = True
            n_active = 1
            grad_a = _grad_at_single(c, a, n, idx)
            grad_c = grad_a - np.mean(grad_a)
            step = 0.01
            a_new = a - step * grad_c
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            if _score(a_new) < current_sc:
                a = a_new
            continue

        # Active indices
        active_idx = np.where(active)[0]

        # We want to equalize c[k] for k in active_idx.
        # Newton: solve J * delta = -(c[active] - c_mean)
        # where J[i,j] = d(c[active_idx[i]])/d(a[j])
        # This is a large system. Use conjugate gradient or just gradient step.

        # Compute average gradient over active set
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]

        # The residual: c[k] - c_bar for k in active
        c_bar = float(c[active].mean())
        residual = c[active] - c_bar

        # Damped Newton-like step: move a to reduce variance of active c values
        # Direction: for each active k with c[k] > c_bar, push a in direction
        # that decreases c[k]; for c[k] < c_bar, push in direction that increases c[k].
        # This is: d = sum_k sign(c[k]-c_bar) * grad(c[k]) / (n_active * max_grad)
        # Simplified: use the gradient of (c[k] - c_bar)^2 summed over active k.
        # = 2 * sum_k (c[k]-c_bar) * grad(c[k])
        w2 = np.zeros(len(c))
        w2[active] = residual / n_active
        wc2 = _full_convolve(w2, a_rev)
        direction = 2.0 * wc2[n - 1:2 * n - 1]

        # Remove mean (to stay on simplex)
        direction = direction - np.mean(direction)

        # Line search
        sc_reduce = 0.0
        for eta in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            a_try = a + eta * direction / (np.max(np.abs(direction)) + 1e-15)
            a_try = np.maximum(a_try, 0.0)
            st = a_try.sum()
            if st < 1e-10:
                continue
            a_try = a_try / st
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-12:
                a = a_try
                sc_reduce = current_sc - sc_try
                break

        if sc_reduce == 0.0:
            # No improvement; tighten active set
            tol_active *= 0.8
            if tol_active < 0.0001:
                break

    return best_a, best_sc


def _grad_at_single(c, a, n, k):
    """Gradient of c[k] w.r.t. a."""
    grad = np.zeros(n)
    j_lo = max(0, k - (n - 1))
    j_hi = min(n - 1, k)
    for j in range(j_lo, j_hi + 1):
        grad[j] = 2.0 * a[k - j]
    return grad


def _subgradient_active(a_init, deadline, min_time_remaining=20.0):
    """Subgradient method with active-set averaging and Polyak steps."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lb = 2.0 * n / (2.0 * n - 1.0)  # Very rough lower bound
    step_scale = 0.3
    t = 0
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-12:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 500:
            break

        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active = np.zeros(len(c), dtype=bool)
            active[int(np.argmax(c))] = True
            n_active = 1

        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        gap = current_sc - lb
        if gap < 1e-10:
            break
        grad_c = grad - np.mean(grad)
        g_norm_sq = float(np.dot(grad_c, grad_c))
        if g_norm_sq < 1e-20:
            break
        step = step_scale * gap / g_norm_sq
        step = min(step, 0.05)

        a_new = a - step * grad_c
        a_new = np.maximum(a_new, 0.0)
        st = a_new.sum()
        if st < 1e-10:
            break
        a_new = a_new / st
        a = a_new
        t += 1
        if t % 200 == 0:
            step_scale *= 0.93

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.arange(1, n + 1, dtype=np.float64)
        a = t / t.sum()
        return a
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(3.0 * t)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-50.0 * (t - 0.15) ** 2) + np.exp(-50.0 * (t - 0.85) ** 2) + 0.02
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 7.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "one_sided":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * t) + 0.003
        return a / a.sum()
    elif kind == "quadratic_ramp":
        t = np.arange(n, dtype=np.float64) / n
        a = t ** 1.5 + 0.01
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-5.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    elif kind == "sawtooth":
        t = np.arange(n, dtype=np.float64)
        period = max(1, n // 16)
        a = (t % period).astype(np.float64) / period + 0.1
        return a / a.sum()
    elif kind == "zigzag":
        t = np.arange(n, dtype=np.float64) / n
        a = 1.0 - 0.5 * np.abs(2.0 * t.astype(np.float64) - 1.0)
        a = np.maximum(a, 0.05)
        # Add a linear trend
        a += 0.3 * t
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _quick_optimize(a, deadline, min_remaining=10.0):
    """Quick optimization: short L-BFGS with a few beta levels."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < c_max * 0.001:
        gap = c_max * 0.001

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    best_a = a.copy()
    best_sc = _score(a)

    for beta in [gap * 0.1, gap * 0.02, gap * 0.005]:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        max_iter = min(1000, max(50, int(remaining / 0.01)))
        prob = _LsmProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13,
                         'maxcor': 50, 'maxls': 40}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return best_a, best_sc


def _fourier_reparam_optimize(a_init, deadline, min_time_remaining=20.0):
    """Optimize in Fourier space: a[i] = exp(sum w_k cos(2*pi*k*i/n + phi_k)).
    
    This constrains a to be smooth/periodic, which can help escape local minima.
    """
    n = len(a_init)
    # Use a limited number of Fourier modes
    n_modes = min(32, n // 4)

    # Initialize from current a
    a = a_init.copy()
    s = a.sum()
    a = a / s

    # Fit Fourier coefficients to current a
    # a[i] ≈ exp(f(i)) where f(i) = sum_k A_k cos(2*pi*k*i/n) + B_k sin(2*pi*k*i/n)
    log_a = np.log(np.maximum(a, 1e-30))
    log_a = log_a - log_a.mean()

    # Use DFT to get initial coefficients
    spectrum = np.fft.rfft(log_a)
    # Keep only first n_modes
    coeffs = np.zeros(n_modes * 2)
    for k in range(n_modes):
        coeffs[2*k] = 2.0 * np.real(spectrum[k]) / n
        coeffs[2*k+1] = -2.0 * np.imag(spectrum[k]) / n

    def _fourier_obj(params):
        A = params[0::2]
        B = params[1::2]
        t = np.arange(n, dtype=np.float64)
        f = np.zeros(n)
        for k in range(len(A)):
            if k == 0:
                f += A[0]
            else:
                f += A[k] * np.cos(2.0 * np.pi * k * t / n) + B[k] * np.sin(2.0 * np.pi * k * t / n)
        a_f = np.exp(f - f.max())
        a_f = a_f / a_f.sum()
        c = _full_convolve(a_f, a_f)
        return float(c.max())

    # Optimize with Nelder-Mead (derivative-free, works in low-dim)
    best_a = a.copy()
    best_sc = _score(a)
    best_params = coeffs.copy()

    x0 = coeffs.copy()
    try:
        result = minimize(
            _fourier_obj, x0, method='Nelder-Mead',
            options={'maxiter': 2000, 'xatol': 1e-10, 'fatol': 1e-12,
                     'adaptive': True}
        )
        # Evaluate result
        p = result.x
        A = p[0::2]
        B = p[1::2]
        t = np.arange(n, dtype=np.float64)
        f = np.zeros(n)
        for k in range(len(A)):
            if k == 0:
                f += A[0]
            else:
                f += A[k] * np.cos(2.0 * np.pi * k * t / n) + B[k] * np.sin(2.0 * np.pi * k * t / n)
        a_f = np.exp(f - f.max())
        a_f = a_f / a_f.sum()
        sc_f = _score(a_f)
        if sc_f < best_sc:
            best_sc = sc_f
            best_a = a_f.copy()
    except Exception:
        pass

    return best_a, best_sc


def _coordinate_zero_test(a, deadline, min_remaining=5.0):
    """Test if zeroing any element improves the score."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    current_sc = 2.0 * n * c_max

    # Only test a subset (random) for speed
    rng = np.random.default_rng(123)
    test_indices = rng.choice(n, size=min(n, 5000), replace=False)

    improved = False
    for j in test_indices:
        if time.monotonic() + min_remaining > deadline:
            break
        aj = a[j]
        if aj < 1e-8:
            continue
        a[j] = 0.0
        s_new = a.sum()
        if s_new < 1e-10:
            a[j] = aj
            continue
        a_norm = a / s_new
        c_new = _full_convolve(a_norm, a_norm)
        sc_new = 2.0 * n * float(c_new.max())
        a[j] = aj
        if sc_new < current_sc - 1e-10:
            a[j] = 0.0
            a = a / a.sum()
            current_sc = sc_new
            improved = True

    if improved:
        return a, current_sc
    return a, _score(a)


t_start_global = 0.0


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    P1: Multi-restart with diverse profiles, quick L-BFGS
    P2: Power-mean optimization on best candidate
    P3: L-BFGS with adaptive beta (main phase)
    P4: Newton active-set equalization
    P5: Subgradient polish
    P6: Multi-n exploration
    P7: Fourier reparam + final L-BFGS
    P8: Coordinate zero test + final polish
    """
    global t_start_global
    t_start_global = time.monotonic()
    deadline = t_start_global + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(16384, "ramp")
        best_score = _score(best_seq)
        print(f"No parent. Ramp n=16384: score={best_score:.10f}")

    n_main = len(best_seq)
    np.random.seed(seed)

    # === Phase 1: Multi-restart quick optimization ===
    print(f"\n[P1] Multi-restart (n={n_main}, t={time.monotonic()-t_start_global:.0f}s)")
    p1_budget = min(200, (budget_s - 10) * 0.12)
    p1_end = t_start_global + p1_budget

    profiles = ["ramp", "exp_ramp", "one_sided", "quadratic_ramp", "zigzag"]
    if parent is None:
        profiles += ["tapered", "uniform"]

    for prof in profiles:
        if time.monotonic() + 30 > p1_end:
            break
        a_p = _make_profile(n_main, prof)
        sc_p = _score(a_p)
        if sc_p < best_score + 0.1:
            a_opt, sc_opt = _quick_optimize(a_p, p1_end, min_remaining=15)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"  {prof}: {best_score:.10f}")
        elif parent is not None and prof in ["ramp", "one_sided"]:
            # Blend with parent
            a_blend = 0.5 * best_seq + 0.5 * a_p
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()
            sc_blend = _score(a_blend)
            if sc_blend < best_score:
                a_opt, sc_opt = _quick_optimize(a_blend, p1_end, min_remaining=15)
                if sc_opt < best_score:
                    best_score = sc_opt
                    best_seq = a_opt.copy()
                    print(f"  blend_{prof}: {best_score:.10f}")

    # Also optimize the parent directly
    if parent is not None:
        p1b_end = min(p1_end, time.monotonic() + 60)
        if time.monotonic() + 20 < p1b_end:
            a_opt, sc_opt = _quick_optimize(best_seq, p1b_end, min_remaining=10)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()

    # === Phase 2: Power-mean optimization ===
    print(f"\n[P2] Power-mean (t={time.monotonic()-t_start_global:.0f}s)")
    p2_end = min(deadline - 400, time.monotonic() + 250)
    a2, sc2 = _run_lbfgs_pm(best_seq, p2_end, min_time_remaining=100, verbose=True)
    if sc2 < best_score:
        best_score = sc2
        best_seq = a2.copy()
        print(f"  P2 result: {best_score:.10f}")

    # === Phase 3: Main L-BFGS with log-sum-exp ===
    print(f"\n[P3] L-BFGS main (t={time.monotonic()-t_start_global:.0f}s)")
    p3_end = min(deadline - 300, time.monotonic() + 400)
    a3, sc3 = _run_lbfgs_lsm(best_seq, p3_end, min_time_remaining=150, verbose=True)
    if sc3 < best_score:
        best_score = sc3
        best_seq = a3.copy()
        print(f"  P3 result: {best_score:.10f}")

    # === Phase 4: Newton active-set equalization ===
    print(f"\n[P4] Newton equalize (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p4_end = min(deadline - 200, time.monotonic() + 200)
        a4, sc4 = _newton_equalize(best_seq, p4_end, min_time_remaining=50, verbose=True)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4 result: {best_score:.10f}")

    # === Phase 5: Subgradient + L-BFGS ===
    print(f"\n[P5] Subgradient (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p5_end = min(deadline - 150, time.monotonic() + 120)
        a5, sc5 = _subgradient_active(best_seq, p5_end, min_time_remaining=50)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  P5 subgrad: {best_score:.10f}")
        # Follow up with L-BFGS
        if time.monotonic() + 20 < deadline:
            p5b_end = min(deadline - 100, time.monotonic() + 80)
            a5b, sc5b = _run_lbfgs_lsm(best_seq, p5b_end, min_time_remaining=30, verbose=False)
            if sc5b < best_score:
                best_score = sc5b
                best_seq = a5b.copy()
                print(f"  P5b lsm: {best_score:.10f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n exploration (t={time.monotonic()-t_start_global:.0f}s)")
    n_options = [8192, 12288, 16384, 20480, 24576, 32768]
    n_options = [x for x in n_options if x != n_main]
    for n_try in n_options:
        if time.monotonic() + 60 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if sc_r < best_score:
            best_score = sc_r
            best_seq = a_r.copy()
            print(f"  n={n_try} resampled: {best_score:.10f}")
        # Quick optimize at this n
        if time.monotonic() + 50 > deadline:
            continue
        p6_end = min(deadline - 40, time.monotonic() + 80)
        a6, sc6 = _quick_optimize(a_r, p6_end, min_remaining=20)
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  n={n_try} optimized: {best_score:.10f}")

    # === Phase 7: Fourier reparam + L-BFGS ===
    print(f"\n[P7] Fourier reparam (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p7_end = min(deadline - 50, time.monotonic() + 100)
        a7, sc7 = _fourier_reparam_optimize(best_seq, p7_end, min_time_remaining=30)
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()
            print(f"  P7 fourier: {best_score:.10f}")

    # === Phase 8: Final polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 10 < deadline:
        # Coordinate zero test
        p8a_end = min(deadline - 30, time.monotonic() + 40)
        best_seq, sc_ca = _coordinate_zero_test(best_seq, p8a_end, min_remaining=10)
        if sc_ca < best_score:
            best_score = sc_ca
            print(f"  P8 coord_zero: {best_score:.10f}")

        # Final L-BFGS
        p8b_end = min(deadline - 5, time.monotonic() + 60)
        if time.monotonic() + 10 < p8b_end:
            a8, sc8 = _run_lbfgs_lsm(best_seq, p8b_end, min_time_remaining=5, verbose=False)
            if sc8 < best_score:
                best_score = sc8
                best_seq = a8.copy()

        # Final Newton
        if time.monotonic() + 5 < deadline:
            a8c, sc8c = _newton_equalize(best_seq, deadline - 2, min_time_remaining=2)
            if sc8c < best_score:
                best_score = sc8c
                best_seq = a8c.copy()

    # Final validation
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) / 256
    sc_final = _score(final)
    # Verify with evaluate_sequence
    try:
        eval_score = evaluate_sequence([float(x) for x in final])
        if np.isfinite(eval_score) and eval_score < sc_final:
            sc_final = eval_score
    except Exception:
        pass

    print(f"\n{'='*60}")
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start_global:.1f}s")
    return [float(x) for x in final]
