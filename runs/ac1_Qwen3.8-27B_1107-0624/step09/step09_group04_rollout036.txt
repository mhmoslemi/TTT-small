    n = len(v)

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            if K < K_max:
                K = min(K_max, K + 1)
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.8
            if no_improve >= 300:
                a = best_a.copy()
                lr = 0.02
                K = K_start
                no_improve = 0
                if no_improve > 600:
                    break

        top_k_idx = np.argsort(b)[::-1][:K]
        w_S = np.zeros(2 * n - 1)
        w_S[top_k_idx] = 2.0 / K
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)
        grad = h[n - 1:2 * n - 1][::-1]
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        alpha = lr / (gnorm + 1e-15)
        improved = False
        for _ in range(40):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            if no_improve % 50 == 0:
                K = max(K_start, K // 2)
        else:
            lr = min(0.05, lr * 1.02)

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005) * beta_scale

    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625,
        0.03, 0.015, 0.0075, 0.004,
        0.002, 0.001, 0.0005, 0.0002, 0.0001,
        0.00005, 0.00002, 0.00001
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        if beta < base_beta * 0.01:
            est_time_per_iter = 0.004 * max(1.0, n / 2048.0)
            max_iter = min(6000, max(50, int(remaining / (est_time_per_iter * 15))))
        else:
            est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
            max_iter = min(4000, max(30, int(remaining / (est_time_per_iter * 12))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-17, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _lp_peak_shave(a_init, deadline, min_time_remaining=60.0):
    """LP-based iterative peak reduction with adaptive active set."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        b = _full_convolve(a, a)
        c_max = float(b.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 80:
                break

        eps = max(5e-5, no_improve * 2e-5)
        active = np.where(b >= c_max * (1.0 - eps))[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        if len(active) > 150:
            thresh = np.sort(b)[::-1][149]
            active = np.where(b >= thresh)[0]

        d = _lp_direction(a, active, b[active])
        if d is None:
            break
        d -= d.mean()
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-15:
            break

        neg_mask = d < -1e-15
        if neg_mask.any():
            gamma_max = float(np.min(a[neg_mask] / (-d[neg_mask])))
        else:
            gamma_max = 1.0
        gamma_max = min(gamma_max, 1.0 / (dnorm + 1e-15))

        gamma = min(0.5 * gamma_max, 0.01)
        improved = False
        for _ in range(30):
            a_try = a + gamma * d
            a_try = np.maximum(a_try, 0.0)
            s_try = float(a_try.sum())
            if s_try < 1e-15:
                gamma *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-12:
                a = a_try
                improved = True
                break
            gamma *= 0.5

        if not improved:
            gamma = min(0.01 * gamma_max, 1e-4)
            a_try = a + gamma * d
            a_try = np.maximum(a_try, 0.0)
            s_try = float(a_try.sum())
            if s_try > 1e-15:
                a = a_try / s_try

    return best_a, best_sc


def _newton_refine(a_init, deadline, min_time_remaining=30.0):
    """Newton-based fine-tuning with reduced active set for Hessian accuracy."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        b = _full_convolve(a, a)
        c_max = float(b.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 120:
                break

        # Use top peaks for Newton (smaller set = better Hessian conditioning)
        K = min(10, n)
        top_idx = np.argsort(b)[::-1][:K]
        # Only include peaks that are close to max
        thresh = c_max * 0.999
        top_idx = top_idx[b[top_idx] >= thresh]
        if len(top_idx) == 0:
            top_idx = np.array([np.argmax(b)])

        d = _newton_step(a, top_idx, lam=0.001, max_cg=80)
        if d is None:
            break
        d -= d.mean()
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-15:
            break

        neg_mask = d < -1e-15
        if neg_mask.any():
            gamma_max = float(np.min(a[neg_mask] / (-d[neg_mask])))
        else:
            gamma_max = 1.0
        gamma_max = min(gamma_max, 1.0)

        gamma = min(0.3 * gamma_max, 0.005)
        improved = False
        for _ in range(40):
            a_try = a + gamma * d
            a_try = np.maximum(a_try, 0.0)
            s_try = float(a_try.sum())
            if s_try < 1e-15:
                gamma *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-12:
                a = a_try
                improved = True
                break
            gamma *= 0.5

        if not improved:
            gamma = min(0.01 * gamma_max, 1e-5)
            a_try = a + gamma * d
            a_try = np.maximum(a_try, 0.0)
            s_try = float(a_try.sum())
            if s_try > 1e-15:
                a = a_try / s_try

    return best_a, best_sc


def _subgradient_polish(a_init, deadline, min_time_remaining=10.0):
    """Final subgradient polish with careful step size."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    max_iters = 8000
    for it in range(max_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break

        b = _full_convolve(a, a)
        c_max = float(b.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()

        threshold = c_max * (1 - 0.0001)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        w_S = np.zeros(2 * n - 1)
        w_S[active] = 2.0 / len(active)
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)
        grad = h[n - 1:2 * n - 1][::-1]
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        eta = 0.005 / (it + 1.0) if it > 50 else 0.005
        eta = min(eta, 0.3 / (np.max(np.abs(grad)) + 1e-15))

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        if (it + 1) % 5 == 0:
            c_new = _full_convolve(a_new, a_new)
            sc_new = 2.0 * n * float(c_new.max())
            if sc_new < current_sc:
                a = a_new

    return best_a, best_sc


def _asymmetric_perturb(a, rng, strength=0.05):
    n = len(a)
    t = np.arange(n) / n
    ramp = 0.3 * (t - 0.5)
    noise = 0.7 * rng.standard_normal(n)
    perturbation = ramp + noise
    perturbation = perturbation - perturbation.mean()
    z = np.log(np.maximum(a, 1e-30))
    z = z + strength * perturbation
    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.7) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 7.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase optimizer: L-BFGS warmup, LP peak shaving, Newton refinement,
    multi-n exploration, and final polish."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS warmup ===
    print(f"\n[P1] L-BFGS warmup (t={time.monotonic()-t_start:.1f}s)")
    a_p1 = _run_lbfgs(parent, deadline, min_time_remaining=900.0, beta_scale=1.0)
    sc_p1 = _score(a_p1)
    if sc_p1 < best_score:
        best_seq = a_p1.copy()
        best_score = sc_p1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_warmup]")

    # === Phase 2: LP peak shaving ===
    print(f"\n[P2] LP peak shaving (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        lp_deadline = min(deadline - 500, time.monotonic() + 350)
        a_lp, sc_lp = _lp_peak_shave(best_seq, lp_deadline, min_time_remaining=60.0)
        if sc_lp < best_score:
            best_seq = a_lp.copy()
            best_score = sc_lp
            print(f"  NEW BEST: {best_score:.10f} [lp_shave]")
        else:
            print(f"  LP: no improvement (best={best_score:.10f})")

    # === Phase 3: Newton refinement ===
    print(f"\n[P3] Newton refinement (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        nw_deadline = min(deadline - 400, time.monotonic() + 250)
        a_nw, sc_nw = _newton_refine(best_seq, nw_deadline, min_time_remaining=60.0)
        if sc_nw < best_score:
            best_seq = a_nw.copy()
            best_score = sc_nw
            print(f"  NEW BEST: {best_score:.10f} [newton]")
        else:
            print(f"  Newton: no improvement (best={best_score:.10f})")

    # === Phase 4: Second LP round ===
    print(f"\n[P4] LP round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        lp2_deadline = min(deadline - 350, time.monotonic() + 200)
        a_lp2, sc_lp2 = _lp_peak_shave(best_seq, lp2_deadline, min_time_remaining=60.0)
        if sc_lp2 < best_score:
            best_seq = a_lp2.copy()
            best_score = sc_lp2
            print(f"  NEW BEST: {best_score:.10f} [lp2]")

    # === Phase 5: Active-set equalization ===
    print(f"\n[P5] Active-set equalization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        eq_deadline = min(deadline - 300, time.monotonic() + 150)
        a_eq, sc_eq = _active_set_equalize(best_seq, eq_deadline, min_time_remaining=60.0)
        if sc_eq < best_score:
            best_seq = a_eq.copy()
            best_score = sc_eq
            print(f"  NEW BEST: {best_score:.10f} [equalize]")

    # === Phase 6: Newton round 2 ===
    print(f"\n[P6] Newton round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        nw2_deadline = min(deadline - 250, time.monotonic() + 120)
        a_nw2, sc_nw2 = _newton_refine(best_seq, nw2_deadline, min_time_remaining=50.0)
        if sc_nw2 < best_score:
            best_seq = a_nw2.copy()
            best_score = sc_nw2

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_cur = len(best_seq)
    n_candidates = sorted(set([n_cur, 2 * n_cur, n_cur // 2]), reverse=True)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == n_cur:
            continue
        if n_try > 65536 or n_try < 1024:

        a_init = _resample(best_seq, n_try)

        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt

            if sc_lp3 < best_score:

    # === Phase 8: Fine L-BFGS ===
    print(f"\n[P8] Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        a_fine = _run_lbfgs(best_seq, deadline, min_time_remaining=80.0, beta_scale=0.3)
        sc_fine = _score(a_fine)
        if sc_fine < best_score:
            best_seq = a_fine.copy()
            best_score = sc_fine
            print(f"  NEW BEST: {best_score:.10f} [fine_lbfgs]")

    # === Phase 9: Final LP + Newton ===
    print(f"\n[P9] Final LP+Newton (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        a_f1, sc_f1 = _lp_peak_shave(best_seq, deadline, min_time_remaining=40.0)
        if sc_f1 < best_score:
            best_seq = a_f1.copy()
            best_score = sc_f1

        if time.monotonic() < deadline - 15:
            a_f2, sc_f2 = _newton_refine(best_seq, deadline, min_time_remaining=15.0)

                best_seq = a_f2.copy()
                best_score = sc_f2
                print(f"  NEW BEST: {best_score:.10f} [final_newton]")

    # === Phase 10: Subgradient polish ===
    print(f"\n[P10] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 10:
        a_sg, sc_sg = _subgradient_polish(best_seq, deadline, min_time_remaining=10.0)
        if sc_sg < best_score:
            best_seq = a_sg.copy()
            best_score = sc_sg
            print(f"  NEW BEST: {best_score:.10f} [subgrad]")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:

        sc_final = _score(final)

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
