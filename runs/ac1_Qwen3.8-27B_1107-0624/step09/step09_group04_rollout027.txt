            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            # Check all top-K peaks improved
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            lr *= 0.9
            if no_improve % 50 == 0:
                K = max(3, K // 2)
            if no_improve >= max_no_improve:
                a = best_a.copy()
                lr = 0.03
                K = 3
                no_improve = 0
                # Try a different K
                K = min(K_max, max(5, K // 3))
                if no_improve > max_no_improve * 2:
                    break
        else:
            lr = min(0.06, lr * 1.005)

    return best_a, best_sc


def _freq_perturb(a, rng, strength=0.05):
    """Perturb in frequency domain for structured exploration."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    # Add noise to magnitude (preserving DC)
    mag = np.abs(A)
    phase = np.angle(A)
    # Add log-normal noise to magnitude
    noise = rng.lognormal(0, strength, size=mag.shape)
    mag_new = mag * noise
    mag_new[0] = mag[0]  # Preserve DC (sum)
    A_new = mag_new * np.exp(1j * phase)
    # Also add small phase noise
    phase_noise = rng.standard_normal(A.shape) * strength * 0.1
    A_new = A_new * np.exp(1j * phase_noise)
    A_new[0] = A[0]
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _time_perturb(a, rng, strength=0.05):
    """Perturb in time domain with structured noise."""
    n = len(a)
    t = np.arange(n) / n
    # Multi-scale noise: low-freq ramp + mid-freq sine + high-freq noise
    ramp = 0.2 * (t - 0.5)
    mid = 0.3 * np.sin(2 * np.pi * rng.integers(3, 15) * t + rng.uniform(0, 2 * np.pi))
    high = 0.5 * rng.standard_normal(n)
    perturbation = ramp + mid + high
    perturbation = perturbation - perturbation.mean()

    z = np.log(np.maximum(a, 1e-30))
    z = z + strength * perturbation
    z = z - z.mean()
    return _softmax(z)


def _subgradient_polish(a_init, t_start, time_budget):
    """Final polish using subgradient with adaptive step size and momentum."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    momentum = 0.5
    v = np.zeros(n)
    max_iters = 10000
    it = 0

    while it < max_iters and time.monotonic() - t_start < time_budget:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()

        # Active set: peaks within 0.001% of max
        threshold = c_max * (1 - 1e-5)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        K = len(active)
        w_S = np.zeros(2 * n - 1)
        w_S[active] = 2.0 / K
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Adaptive step size
        eta = 0.02 / (it + 1) ** 0.5
        eta = min(eta, 0.3 / (np.max(np.abs(grad)) + 1e-15))

        # Momentum
        v = momentum * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        # Accept if improved (check every 3 iterations)
        if (it + 1) % 3 == 0:
            c_new = _full_convolve(a_new, a_new)
            sc_new = 2.0 * n * float(c_new.max())
            if sc_new < current_sc:
                a = a_new
            else:
                v *= 0.5  # Dampen momentum on rejection
        it += 1

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quadratic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.8) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + 0.6 * np.exp(-20.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    elif kind == "flat_center":
        t = np.linspace(0, 1, n)
        a = np.where(np.abs(t - 0.5) < 0.4, 1.0, 0.3)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase optimizer with adaptive L-BFGS, active-set equalization,
    frequency-domain perturbation, and multi-scale search."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, tapered n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS warmup with adaptive beta ===
    print(f"\n[P1] Adaptive L-BFGS warmup (t={time.monotonic()-t_start:.1f}s)")
    p1_budget = deadline - time.monotonic() - 1200.0
    if p1_budget > 60:
        a_p1 = _run_lbfgs_adaptive(parent, time.monotonic(), p1_budget)
        sc_p1 = _score(a_p1)
        if sc_p1 < best_score:
            best_seq = a_p1.copy()
            best_score = sc_p1
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_adaptive]")
        else:
            print(f"  L-BFGS: score={sc_p1:.10f} (no improvement)")

    # === Phase 2: Active-set equalization ===
    print(f"\n[P2] Active-set equalization (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - time.monotonic()
    p2_budget = min(300.0, remaining * 0.2)
    if p2_budget > 30:
        a_eq, sc_eq = _active_set_equalize(best_seq, time.monotonic(), p2_budget)
        if sc_eq < best_score:
            best_seq = a_eq.copy()
            best_score = sc_eq
            print(f"  NEW BEST: {best_score:.10f} [equalize]")
        else:
            print(f"  Equalize: score={sc_eq:.10f} (no improvement, best={best_score:.10f})")

    # === Phase 3: Frequency-domain perturbation + re-optimize ===
    print(f"\n[P3] Frequency perturbation exploration (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - time.monotonic()
    n_trials = 3
    for trial in range(n_trials):
        if time.monotonic() > deadline - 300:
            break
        strength = 0.02 * (trial + 1)
        a_pert = _freq_perturb(best_seq, rng, strength=strength)
        # Also try time-domain perturbation
        if trial == 1:
            a_pert = _time_perturb(best_seq, rng, strength=strength)
        elif trial == 2:
            a_pert = _freq_perturb(best_seq, rng, strength=strength * 0.5)
            a_pert = _time_perturb(a_pert, rng, strength=strength * 0.3)

        sc_pert = _score(a_pert)
        print(f"  Trial {trial}: perturbed score={sc_pert:.10f}")

        # Re-optimize from perturbed point
        opt_budget = min(200.0, (deadline - time.monotonic()) * 0.15)
        if opt_budget > 20:
            a_opt = _run_lbfgs_adaptive(a_pert, time.monotonic(), opt_budget)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [freq_perturb_{trial}]")
            else:
                print(f"  Trial {trial}: optimized score={sc_opt:.10f} (no improvement)")

    # === Phase 4: Fine L-BFGS with very small beta ===
    print(f"\n[P4] Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - time.monotonic()
    p4_budget = min(300.0, remaining * 0.2)
    if p4_budget > 30:
        a_fine = _run_lbfgs_adaptive(best_seq, time.monotonic(), p4_budget)
        sc_fine = _score(a_fine)
        if sc_fine < best_score:
            best_seq = a_fine.copy()
            best_score = sc_fine
            print(f"  NEW BEST: {best_score:.10f} [fine_lbfgs]")
        else:
            print(f"  Fine L-BFGS: score={sc_fine:.10f} (no improvement)")

    # === Phase 5: Second active-set round ===
    print(f"\n[P5] Active-set round 2 (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - time.monotonic()
    p5_budget = min(200.0, remaining * 0.2)
    if p5_budget > 30:
        a_eq2, sc_eq2 = _active_set_equalize(best_seq, time.monotonic(), p5_budget)
        if sc_eq2 < best_score:
            best_seq = a_eq2.copy()
            best_score = sc_eq2
            print(f"  NEW BEST: {best_score:.10f} [equalize2]")
        else:
            print(f"  Equalize2: score={sc_eq2:.10f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_current = len(best_seq)
    n_candidates = [n_current]
    for n_try in [16384, 24576, 8192, 32768]:
        if n_try != n_current and n_try not in n_candidates:
            n_candidates.append(n_try)
    n_candidates = sorted(n_candidates, key=lambda x: -abs(x - n_current))[:3]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == n_current:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")
        opt_budget = min(100.0, (deadline - time.monotonic()) * 0.1)
        if opt_budget > 15:
            a_opt = _run_lbfgs_adaptive(a_init, time.monotonic(), opt_budget)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n]")
            else:
                print(f"  n={n_try}: optimized score={sc_opt:.10f}")

    # === Phase 7: Subgradient polish ===
    print(f"\n[P7] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - time.monotonic()
    p7_budget = min(150.0, remaining * 0.4)
    if p7_budget > 10:
        a_sg, sc_sg = _subgradient_polish(best_seq, time.monotonic(), p7_budget)
        if sc_sg < best_score:
            best_seq = a_sg.copy()
            best_score = sc_sg
            print(f"  NEW BEST: {best_score:.10f} [subgrad]")
        else:
            print(f"  Subgrad: score={sc_sg:.10f}")

    # === Phase 8: Final ultra-fine optimization ===
    print(f"\n[P8] Final ultra-fine (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Very small beta L-BFGS
        a_final = _run_lbfgs_adaptive(best_seq, time.monotonic(), remaining * 0.4)
        sc_final = _score(a_final)
        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final
            print(f"  NEW BEST: {best_score:.10f} [final_ultra]")

        # Tiny active-set
        if time.monotonic() < deadline - 5:
            a_fin2, sc_fin2 = _active_set_equalize(best_seq, time.monotonic(),
                                                   max(5.0, deadline - time.monotonic() - 3))
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_equalize]")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
