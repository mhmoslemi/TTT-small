        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(W, c))
        g = 2.0 * a * (h - W_val)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-16 * n:
            break
        dx = -g_t / gn
        alpha = 1.0
        step_done = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            if float(c_try.max()) < c_max * (1.0 - 0.005 * alpha) - 1e-18:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-18:
                break
        if not step_done:
            for trial in [0.05, 0.02, 0.01, 0.005, 0.001]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-18:
                    a = a_try
                    step_done = True
                    break
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if not step_done and no_improve > 10:
            break
    return best_a, best_sc


def _subgrad_momentum(a_init, deadline, max_steps=3000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() < 3:
            active = np.zeros_like(c, dtype=bool)
            top_k = np.argpartition(c, -max(3, min(50, 2 * n - 1)))[-max(3, min(50, 2 * n - 1)):]
            active[top_k] = True
        W = np.zeros_like(c)
        W[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(W, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(W, c))
        grad = 2.0 * a * (h - W_val)
        grad = grad - np.mean(grad)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-18:
            break
        v = momentum * v - lr * grad
        z = z + v
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        if i % 100 == 99:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.1, 0.03)
            else:
                patience += 1
                lr *= 0.8
                momentum = min(momentum + 0.01, 0.95)
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)
                    patience = 0
                    lr = 0.008
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _try_n_values(best_seq, best_score, deadline, min_remaining=60):
    n_current = len(best_seq)
    n_options = set()
    for n_try in [n_current, n_current // 2, n_current * 2, 4096, 8192, 16384, 32768, 65536]:
        if 1024 <= n_try <= 131072:
            n_options.add(n_try)
    for n_try in sorted(n_options):
        if time.monotonic() > deadline - min_remaining:
            break
        if n_try == n_current:
            continue
        x_old = np.linspace(0, 1, n_current)
        x_new = np.linspace(0, 1, n_try)
        a_res = np.interp(x_new, x_old, best_seq)
        a_res = np.maximum(a_res, 0)
        s = a_res.sum()
        if s < 1e-15:
            continue
        a_res = a_res / s
        sc = _score(a_res)
        if sc < best_score - 1e-10:
            best_score = sc
            best_seq = a_res.copy()
            print(f"  n={n_try} resample: {sc:.10f}")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            imp = prev - best_score if np.isfinite(prev) else 0
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] imp={imp:.2e} t={time.monotonic()-t_start:.1f}s")

    # Try parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                par_sc = _score(parent)
                print(f"Parent: n={len(parent)}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    # Try uniform sequences of various lengths
    print(f"[Init] Trying uniform sequences...")
    for n_try in [4096, 8192, 16384, 32768, 65536]:
        if time.monotonic() > deadline - 30:
            break
        a_uni = np.ones(n_try)
        sc_uni = _score(a_uni)
        update_best(a_uni, sc_uni, f"uniform_n{n_try}")
        if time.monotonic() - t_start > 5:
            break

    if best_seq is None:
        best_seq = np.ones(16384)
        best_score = _score(best_seq)
        print(f"  Fallback uniform: score={best_score:.10f}")

    print(f"\n[Current best: {best_score:.10f}, n={len(best_seq)}] t={time.monotonic()-t_start:.1f}s")

    # Phase 1: Smoothed L-BFGS optimization
    print(f"\n[Phase 1] Smoothed L-BFGS with annealing")
    if time.monotonic() < deadline - 120:
        a_opt, sc_opt = _optimize_smooth(best_seq, min(deadline - 90, t_start + 400), beta_init=0.05)
        update_best(a_opt, sc_opt, "smooth_lbfgs")

    # Phase 2: KKT-Newton refinement
    print(f"\n[Phase 2] KKT-Newton")
    if time.monotonic() < deadline - 90:
        a_kkt, sc_kkt = _kkt_newton(best_seq, min(deadline - 60, t_start + 700),
                                     max_rounds=200, k_schedule=[3, 5, 10, 20, 50, 100, 200, 500, 1000])
        update_best(a_kkt, sc_kkt, "kkt")

    # Phase 3: Subgradient with momentum
    print(f"\n[Phase 3] Subgradient momentum")
    if time.monotonic() < deadline - 60:
        a_sg, sc_sg = _subgrad_momentum(best_seq, min(deadline - 40, t_start + 1000), max_steps=5000)
        update_best(a_sg, sc_sg, "subgrad")

    # Phase 4: Cycled optimization
    print(f"\n[Phase 4] Cycled refinement")
    for cyc in range(6):
        if time.monotonic() > deadline - 60:
            break
        t_sub = min(deadline - 50, time.monotonic() + 60)
        a_c, sc_c = _subgrad_momentum(best_seq, t_sub, max_steps=2000)
        update_best(a_c, sc_c, f"cyc_sg_{cyc}")
        if time.monotonic() > deadline - 60:
            break
        t_new = min(deadline - 50, time.monotonic() + 60)
        a_c2, sc_c2 = _kkt_newton(best_seq, t_new, max_rounds=80,
                                   k_schedule=[3, 5, 10, 20, 50, 100, 500])
        update_best(a_c2, sc_c2, f"cyc_kkt_{cyc}")
        if time.monotonic() > deadline - 60:
            break
        t_sm = min(deadline - 50, time.monotonic() + 80)
        a_c3, sc_c3 = _optimize_smooth(best_seq, t_sm, beta_init=0.01, beta_min=0.0005)
        update_best(a_c3, sc_c3, f"cyc_smooth_{cyc}")

    # Phase 5: Try different n
    print(f"\n[Phase 5] Multi-n exploration")
    if time.monotonic() < deadline - 50:
        best_seq, best_score = _try_n_values(best_seq, best_score, deadline, min_remaining=40)

    # Phase 6: Final polish
    print(f"\n[Phase 6] Final polish")
    if time.monotonic() < deadline - 10:
        a_fin, sc_fin = _kkt_newton(best_seq, deadline - 5, max_rounds=50,
                                     k_schedule=[1, 2, 3, 5, 10, 20, 50, 100])
        update_best(a_fin, sc_fin, "final_kkt")
        if time.monotonic() < deadline - 5:
            a_fin2, sc_fin2 = _optimize_smooth(best_seq, deadline - 3, beta_init=0.005, beta_min=0.001)
            update_best(a_fin2, sc_fin2, "final_smooth")

    # Final output
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        # Scale to ensure sum >= 0.01 (for evaluator)
        if s < 0.01:
            final = final * (0.1 / s)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, sum={final.sum():.4f}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    fallback = np.ones(8192)
    print(f"FALLBACK uniform: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
