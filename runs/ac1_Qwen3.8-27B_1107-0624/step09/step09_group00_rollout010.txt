def _adam_optimize(a_init, deadline, min_remaining=30.0):
    """Adam optimization with annealed smooth objective."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    
    # Initial beta from gap analysis
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted)-1)]
    if gap < 1e-15:
        gap = max_c * 0.01
    base_beta = max(gap * 0.5, max_c * 0.005)
    
    # Annealing schedule
    betas = base_beta * np.array([1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                                   0.05, 0.035, 0.025, 0.015, 0.01, 0.007,
                                   0.005, 0.003, 0.002, 0.001, 0.0007, 0.0005,
                                   0.0003, 0.0002, 0.0001, 0.00005])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr_base = 0.01
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    
    for bi, beta in enumerate(betas):
        if time.monotonic() > deadline - min_remaining:
            break
        
        # Estimate time per iteration
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        
        # Estimate iteration time
        est_t = 0.008 * max(1.0, n / 8192.0)
        n_iter = min(5000, max(20, int(remaining / (est_t * 5))))
        
        for _ in range(n_iter):
            if time.monotonic() > deadline - min_remaining:
                break
            
            t += 1
            lr = lr_base / (1.0 + 0.001 * t)
            
            # Compute gradient
            grad = _smooth_max_grad(z, beta, n, k=1)
            
            # Adam update
            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * (grad * grad)
            m_hat = m / (1 - b1**t)
            v_hat = v / (1 - b2**t)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Periodically check score
            if t % 200 == 0:
                a_new = _softmax(z)
                sc = _score(a_new)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a_new.copy()
                current_a = a_new
                current_sc = sc
    
    # Final check
    a_new = _softmax(z)
    sc = _score(a_new)
    if sc < best_sc:
        best_a = a_new.copy()
        best_sc = sc
    
    return best_a, best_sc

def _active_deflation(a_init, deadline, min_remaining=30.0):
    """Active-set peak deflation with line search."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.02
    tol_frac = 0.01
    no_improve = 0
    
    while time.monotonic() < deadline - min_remaining:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        
        # Inverse-gap weighting
        eps_w = 1e-12 * c_max
        w = np.zeros_like(c)
        w[active] = 1.0 / (c_max - c[active] + eps_w)
        w = w / w.sum()
        
        # Compute gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n-1:2*n-1]
        grad = grad - np.mean(grad)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        grad = grad / gnorm
        
        # Line search with backtracking
        alpha = lr
        improved = False
        a_old_sc = current_sc
        for _ in range(80):
            a_try = _normalize(a - alpha * grad)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(0.0005, tol_frac * 0.7)
                lr *= 0.7
            if no_improve >= 200:
                a = best_a.copy()
                tol_frac = max(0.0005, tol_frac * 0.3)
                lr = 0.01
                no_improve = 0
        else:
            if tol_frac > 0.0005:
                tol_frac *= 0.99
            lr = min(0.05, lr * 1.005)
    
    return best_a, _score(best_a)

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5)**2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos2":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))**2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(12345)
        a = rng.random(n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20*(t-0.25)**2) + np.exp(-20*(t-0.75)**2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.exp(-2*(np.abs(2*t-1))**8)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _diverse_ils(a_init, deadline, rng, min_remaining=60.0):

    pool = [a_init.copy()]
    scores = [_score(a_init)]
    best_a = a_init.copy()
    best_sc = scores[0]
    pool_size = 5

        # Pick a random solution from pool

        base = pool[idx].copy()
        
        # Perturb
        strategy = attempt % 6
        if strategy == 0:  # Global small
            z = np.log(np.maximum(base, 1e-30))

            z -= z.mean()
            a_perturbed = _softmax(z)

            start = int(rng.integers(0, n - seg))
            a_perturbed = base.copy()
            a_perturbed[start:start+seg] = a_perturbed[start:start+seg][::-1]
            z = np.log(np.maximum(a_perturbed, 1e-30))
            z += 0.003 * rng.standard_normal(n)
            z -= z.mean()
            a_perturbed = _softmax(z)
        else:  # Bimodal perturbation
            a_perturbed = base.copy()
            center = int(rng.uniform(0.2, 0.8) * n)
            width = max(10, n // 20)
            window = np.exp(-((np.arange(-width, width+1)) / (width/3))**2)
            lo = max(0, center - width)
            hi = min(n, center + width + 1)
            w_seg = window[lo-(center-width):hi-(center-width)]
            a_perturbed[lo:hi] += 0.01 * w_seg
            a_perturbed = _normalize(a_perturbed)
        
        # Optimize
        opt_deadline = min(deadline - min_remaining, time.monotonic() + 12)
        a_opt = _quick_optimize(a_perturbed, opt_deadline)
        sc_opt = _score(a_opt)
        
        # Update pool
        if sc_opt < best_sc - 1e-12:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"    DILS NEW BEST: {best_sc:.10f} (strategy={strategy}) t={time.monotonic()-t_start:.0f}s")
            pool.insert(0, a_opt.copy())
            scores.insert(0, sc_opt)
            if len(pool) > pool_size:
                pool = pool[:pool_size]
                scores = scores[:pool_size]
        elif len(pool) < pool_size:
            pool.append(a_opt.copy())
            scores.append(sc_opt)
        elif sc_opt < scores[-1]:
            pool[-1] = a_opt.copy()
            scores[-1] = sc_opt
        
        attempt += 1
    
    return best_a, best_sc
