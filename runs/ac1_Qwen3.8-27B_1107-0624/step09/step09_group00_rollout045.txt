def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):

    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:

    return e / s

def _make_profile(n, kind="uniform"):

def _active_set_gradient(a, c, tol_frac):
    """Compute weighted active-set gradient for L-infinity autocorrelation."""
    n = len(a)
    c_max = float(c.max())
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    n_active = int(active.sum())
    if n_active == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        n_active = 1

    eps = 1e-12 * c_max
    w = np.zeros_like(c)
    w[active] = 1.0 / (c_max - c[active] + eps)
    w = w / w.sum()

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad = 2.0 * h

    grad = grad - np.mean(grad)
    return grad


def _mw_direction_finding(a, max_iter=30, gamma=1.0):
    """
    Multiplicative Weights approach to find LP-optimal direction.
    Solves approximately: max sum(g) s.t. (a*g)[k] <= M, g >= 0
    Returns the normalized g (the direction to move towards).
    """
    n = len(a)
    c = _full_convolve(a, a)
    M = float(c.max())
    
    if M < 1e-15:
        return a.copy()
    
    # Initialize y (dual variables) uniformly over top peaks
    n_conv = len(c)
    c_norm = c / M  # normalized so peaks are at 1
    c_sorted = np.sort(c_norm)[::-1]
    
    # Focus on top K peaks
    K = min(n_conv, max(50, n // 50))
    top_k_frac = c_sorted[K - 1] if K < len(c_sorted) else 0.0
    active_mask = c_norm >= top_k_frac * 0.8
    
    y = np.zeros(n_conv)
    y[active_mask] = 1.0
    y = y / y.sum()
    
    # Initialize g
    g = a.copy()
    g_sum = 1.0
    
    eta = 0.5
    
    n_a = len(a)
    for it in range(max_iter):
        # Compute (a * g) / M = A @ g / M
        ag = _full_convolve(a, g)
        ag_norm = ag / M
        
        # Update y: concentrate on violated/near-violated constraints
        violation = np.maximum(0, ag_norm - 1.0)
        y_new = np.zeros(n_conv)
        y_new[active_mask] = 1.0 + gamma * violation[active_mask]
        # Also add weight to new peaks
        new_active = ag_norm >= np.max(ag_norm) * 0.95
        y_new[new_active] += 0.5
        y_new = np.maximum(y_new, 1e-10)
        y_new = y_new / y_new.sum()
        y = y_new
        
        # Update g: mirror descent / multiplicative weights
        # dual_obj[j] = sum_k y[k] * a[k-j] / M = (y * reverse(a))[j + n - 1] / M
        # Actually: sum_k y[k] * A[k,j] / M = sum_k y[k] * a[k-j] / M
        # This is the cross-correlation of y and a (reversed)
        a_rev = a[::-1] / M
        dual_obj = _full_convolve(y, a_rev)
        dual_obj = dual_obj[n_a - 1:2 * n_a - 1]  # extract the n elements
        
        # g[j] proportional to exp(-eta * dual_obj[j])
        g_new = np.exp(-eta * np.clip(dual_obj, -50, 50))
        g_new = np.maximum(g_new, 1e-30)
        g_new = g_new / g_new.sum()
        
        # Check convergence
        if it > 5:
            ag_old_max = _full_convolve(a, g).max() / M
            ag_new_max = _full_convolve(a, g_new).max() / M
            if abs(ag_new_max - ag_old_max) < 1e-6:
                g = g_new
                break
        
        g = g_new
    
    # Scale g to match sum of a
    g = g / g.sum() * a.sum()
    return g


def _nesterov_active_polish(a_init, deadline, tol_start=0.01, tol_end=0.00005,

        c_max = float(c.max())

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_problem(beta, n):
    def f(z):
        return _smoobj(z, beta, n)
    def g(z):
        return _smograd(z, beta, n)
    return f, g


def _run_lbfgs_continuation(a_init, deadline, min_time_remaining=30.0):

        return a_init

        gap = max_c * 0.001

    lr = 0.3
    no_improve = 0
    
    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        current_sc = 2.0 * n * float(c.max())
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        # Get MW direction
        g_dir = _mw_direction_finding(a, max_iter=max_mw_iter)
        g_dir = _normalize(g_dir)
        
        # Line search along direction
        improved = False
        for t in [lr, lr*0.5, lr*0.25, lr*0.125, lr*0.06]:
            a_try = _normalize((1 - t) * a + t * g_dir)
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
        
        if not improved:
            lr *= 0.6
            no_improve += 1
            if no_improve >= 30:
                break
        else:
            lr = min(0.5, lr * 1.05)
    
    return best_a, _score(best_a)


def _combined_polish(a_init, deadline):
    """Combine MW polish, L-BFGS, and active-set polish."""
    n = len(a_init)
    a = a_init.copy()
    sc = _score(a)
    
    # MW polish
    if time.monotonic() < deadline - 10:
        a_mw, sc_mw = _mw_polish(a, time.monotonic() + min(30, deadline - time.monotonic() - 10))
        if sc_mw < sc:
            a = a_mw
            sc = sc_mw
    
    # L-BFGS
    if time.monotonic() < deadline - 15:
        a_lb = _run_lbfgs_continuation(a, deadline, min_time_remaining=15)
        sc_lb = _score(a_lb)
        if sc_lb < sc:
            a = a_lb
            sc = sc_lb
    
    # Active-set polish
    if time.monotonic() < deadline - 5:
        a_as, sc_as = _nesterov_active_polish(a, deadline - 2,
                                              tol_start=0.003, tol_end=0.0001,
                                              lr_init=0.03, momentum=0.85)
        if sc_as < sc:
            a = a_as
            sc = sc_as
    
    return a, _score(a)


def _stochastic_ils(a_init, deadline, rng, min_time_remaining=60.0):
    """Stochastic Iterated Local Search with diverse perturbation strategies."""
    n = len(a_init)
    current_a = _normalize(a_init.copy())
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc

    strategies = [
        ('global_small', 0.004),
        ('global_med', 0.015),
        ('global_large', 0.04),
        ('block', 0.05),
        ('sparse', 0.1),
        ('mirror', 0.02),
        ('peak_target', 0.0),
        ('edge_boost', 0.0),
    ]

    attempt = 0
    while time.monotonic() + min_time_remaining < deadline:
        kind, sigma = strategies[attempt % len(strategies)]

        if kind == 'global_small' or kind == 'global_med' or kind == 'global_large':
            z = np.log(np.maximum(current_a, 1e-30))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_perturbed = _softmax(z)
        elif kind == 'block':
            block_size = max(32, n // 6)
            start = int(rng.integers(0, n - block_size))
            a_perturbed = current_a.copy()
            z = np.log(np.maximum(a_perturbed[start:start + block_size], 1e-30))
            z = z + sigma * rng.standard_normal(block_size)
            a_perturbed[start:start + block_size] = np.exp(z - np.max(z))
            a_perturbed = _normalize(a_perturbed)
        elif kind == 'sparse':
            n_perturb = max(20, n // 10)
            idx = rng.choice(n, size=n_perturb, replace=False)
            a_perturbed = current_a.copy()
            factors = np.exp(sigma * rng.standard_normal(n_perturb))
            a_perturbed[idx] *= factors
            a_perturbed = _normalize(a_perturbed)
        elif kind == 'mirror':
            seg_len = max(16, n // 15)
            start = int(rng.integers(0, n - seg_len))
            a_perturbed = current_a.copy()
            seg = a_perturbed[start:start + seg_len].copy()
            a_perturbed[start:start + seg_len] = seg[::-1]
            z = np.log(np.maximum(a_perturbed, 1e-30))
            z = z + 0.003 * rng.standard_normal(n)
            z = z - z.mean()
            a_perturbed = _softmax(z)
        elif kind == 'peak_target':
            # Target the peak of the convolution
            a_perturbed = current_a.copy()
            c = _full_convolve(a_perturbed, a_perturbed)
            peak_idx = int(np.argmax(c))
            # Reduce elements that contribute to the peak
            j_start = max(0, peak_idx - (n - 1))
            j_end = min(n - 1, peak_idx)
            for j in range(j_start, j_end + 1):
                k = peak_idx - j
                if 0 <= k < n:
                    a_perturbed[j] *= (1.0 - 0.1 * rng.random())
                    a_perturbed[k] *= (1.0 - 0.1 * rng.random())
            a_perturbed = _normalize(a_perturbed)
        elif kind == 'edge_boost':
            # Boost edges (where convolution has fewer terms)
            a_perturbed = current_a.copy()
            edge_width = n // 8
            a_perturbed[:edge_width] *= (1.0 + 0.2 * rng.random(edge_width))
            a_perturbed[-edge_width:] *= (1.0 + 0.2 * rng.random(edge_width))
            a_perturbed = _normalize(a_perturbed)
        else:
            a_perturbed = current_a.copy()

        opt_deadline = min(deadline - min_time_remaining, time.monotonic() + 6)
        a_opt, sc_opt = _combined_polish(a_perturbed, opt_deadline)

        if sc_opt < best_sc - 1e-13:
            best_sc = sc_opt
            best_a = a_opt.copy()
            current_a = a_opt.copy()
            print(f"    ILS NEW BEST: {best_sc:.10f} ({kind}) t={time.monotonic()-t_start:.0f}s", flush=True)
        elif sc_opt < current_sc - 1e-13:
            current_a = a_opt.copy()

        attempt += 1

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    P1: Diverse initializations + quick polish (250s)
    P2: Intensive L-BFGS continuation (300s)
    P3: MW-direction polish (150s)
    P4: Active-set fine polish (100s)
    P5: ILS diversification (300s)
    P6: Multi-n exploration (250s)
    P7: Final ultra-fine polish (remaining)
    """
    global t_start
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}", flush=True)
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 24576
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}", flush=True)

    n_main = n_parent if n_parent > 0 else 24576

    # === Phase 1: Multi-start quick optimization (250s) ===
    print(f"\n[P1] Multi-start (t={time.monotonic() - t_start:.1f}s)", flush=True)
    ms_deadline = min(t_start + 250, deadline - 600)

    starts = []
    starts.append(('parent', parent.copy()))
    for kind in ['hann', 'raised_cos2', 'u_shaped', 'edge_enhanced']:
        starts.append((kind, _make_profile(n_main, kind)))

    for name, a_init in starts:
        if time.monotonic() > ms_deadline - 30:
            break
        sc_init = _score(a_init)
        a_opt, sc_opt = _combined_polish(a_init, min(ms_deadline - 10, time.monotonic() + 40))
        print(f"  {name}: init={sc_init:.10f}, opt={sc_opt:.10f}", flush=True)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [ms/{name}]", flush=True)

    # === Phase 2: Intensive L-BFGS continuation (300s) ===
    print(f"\n[P2] Intensive L-BFGS (t={time.monotonic() - t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 500:
        lbfgs_deadline = min(deadline - 500, time.monotonic() + 300)
        a_main = _run_lbfgs_continuation(best_seq, lbfgs_deadline, min_time_remaining=60)
        sc = _score(a_main)
        if sc < best_score:
            best_seq = a_main.copy()
            best_score = sc
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_int]", flush=True)

    # === Phase 3: MW-direction polish (150s) ===
    print(f"\n[P3] MW polish (t={time.monotonic() - t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 400:
        mw_deadline = min(deadline - 400, time.monotonic() + 150)
        a_mw, sc_mw = _mw_polish(best_seq, mw_deadline, max_mw_iter=25)
        if sc_mw < best_score:
            best_seq = a_mw.copy()
            best_score = sc_mw
            print(f"  NEW BEST: {best_score:.10f} [mw_polish]", flush=True)

    # === Phase 4: Active-set fine polish (100s) ===
    print(f"\n[P4] Active-set polish (t={time.monotonic() - t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 350:
        as_deadline = min(deadline - 350, time.monotonic() + 100)
        a_as, sc_as = _nesterov_active_polish(best_seq, as_deadline,
                                              tol_start=0.002, tol_end=0.00001,
                                              lr_init=0.015, momentum=0.88)
        if sc_as < best_score:
            best_seq = a_as.copy()
            best_score = sc_as
            print(f"  NEW BEST: {best_score:.10f} [active_set]", flush=True)

    # === Phase 5: ILS diversification (300s) ===
    print(f"\n[P5] ILS (t={time.monotonic() - t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 300:
        ils_deadline = min(deadline - 300, time.monotonic() + 300)
        a_ils, sc_ils = _stochastic_ils(best_seq, ils_deadline, rng, min_time_remaining=60)
        if sc_ils < best_score:
            best_seq = a_ils.copy()
            best_score = sc_ils
            print(f"  NEW BEST: {best_score:.10f} [ils]", flush=True)

    # === Phase 6: Multi-n exploration (250s) ===
    print(f"\n[P6] Multi-n (t={time.monotonic() - t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 200:
        n_candidates = []
        for factor in [0.75, 1.25, 1.5, 0.5, 2.0]:
            nn = int(n_main * factor)
            nn = max(4096, min(65536, nn))
            if nn != n_main and nn not in n_candidates:
                n_candidates.append(nn)

        for n_try in n_candidates:
            if time.monotonic() > deadline - 120:
                break
            a_init = _resample(best_seq, n_try)
            sc_init = _score(a_init)
            print(f"  n={n_try}: init={sc_init:.10f}", flush=True)
            a_opt, sc_opt = _combined_polish(a_init, min(deadline - 80, time.monotonic() + 50))
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n]", flush=True)

    # === Phase 7: Second ILS + final polish ===
    print(f"\n[P7] Final phases (t={time.monotonic() - t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 80:
        ils2_deadline = min(deadline - 60, time.monotonic() + 80)
        rng2 = np.random.default_rng(int(seed) + 777)
        a_ils2, sc_ils2 = _stochastic_ils(best_seq, ils2_deadline, rng2, min_time_remaining=40)
        if sc_ils2 < best_score:
            best_seq = a_ils2.copy()
            best_score = sc_ils2
            print(f"  NEW BEST: {best_score:.10f} [ils2]", flush=True)

    # Ultra-fine final polish
    if time.monotonic() < deadline - 5:
        final_deadline = deadline - 3
        a_fin, sc_fin = _combined_polish(best_seq, final_deadline)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final]", flush=True)

    # Final validation
    print(f"\n{'=' * 60}", flush=True)
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s", flush=True)
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
