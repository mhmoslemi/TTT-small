import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def _freq_domain_opt(a_init, iterations=10000, beta_init=0.5, beta_min=0.01):
    n = len(a_init)
    a = a_init.copy()
    N = _next_pow2(2 * n - 1)
    
    best_a = a.copy()
    best_score = _score(a)
    
    for it in range(iterations):
        # Compute FFT
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        
        # Current autocorrelation
        r = np.fft.ifft(np.abs(A)**2).real
        r = r[:2*n-1]
        
        # Current score
        current_score = 2.0 * n * r.max()
        
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
        
        # Target: flat |A|, i.e., |A|^2 = constant = N * r[0] / N = r[0]
        # Actually, if |A|^2 is constant = C, then r = IFFT(|A|^2) = C * delta[0]
        # So r[0] = C*N, r[k] = 0 for k != 0
        # We can't achieve this (would require a = delta), but we can get closer.
        
        # Soft update: blend |A| towards flat
        abs_A = np.abs(A)
        flat_level = np.mean(abs_A**2)  # mean of |A|^2
        # New |A|^2: blend between current and flat
        # |A_new|^2 = |A|^2 * (flat_level / |A|^2)^beta = |A|^(2-2*beta) * flat_level^beta
        beta = beta_init * (beta_min / beta_init) ** (it / iterations)
        
        # More stable: |A_new| = |A| * (sqrt(flat_level) / |A|)^beta
        # = |A|^(1-beta) * sqrt(flat_level)^beta
        target_abs = np.sqrt(flat_level)
        # Avoid division by zero
        abs_A_safe = np.maximum(abs_A, 1e-15)
        new_abs = abs_A_safe ** (1 - beta) * target_abs ** beta
        
        # Reconstruct A with new amplitude, same phase
        phase = A / abs_A_safe  # unit complex numbers (phase)
        A_new = new_abs * phase
        
        # IFFT to get time domain
        a_new = np.fft.ifft(A_new).real
        a_new = a_new[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        a = a_new
    
    return best_a, best_score

"""
Multi-strategy optimizer combining frequency-domain flat spectrum design,
log-domain L-BFGS with adaptive top-K smoothing, simulated annealing,
and cross-entropy method for global exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _log_normalize(z):
    """Normalize in log domain: a = exp(z) / sum(exp(z))"""
    z_max = z.max()
    e = np.exp(z - z_max)
    return e / e.sum()


def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _flat_spectrum_iter(a_init, n_iters=50000, beta_start=0.1, beta_end=0.8):
    """Iterative flat-spectrum method in frequency domain."""
    n = len(a_init)
    a = a_init.copy()
    N = _next_pow2(2 * n - 1)
    
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(n_iters):
        # Anneal beta
        frac = it / max(1, n_iters - 1)
        beta = beta_start + (beta_end - beta_start) * frac
        
        # FFT
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        
        abs_A = np.abs(A)
        abs_A_mean = np.mean(abs_A)
        
        # Blend towards flat
        # |A_new| = |A|^(1-beta) * mean(|A|)^beta
        log_abs = np.log(np.maximum(abs_A, 1e-30))
        log_abs_mean = np.log(max(abs_A_mean, 1e-30))
        log_abs_new = (1 - beta) * log_abs + beta * log_abs_mean
        abs_new = np.exp(log_abs_new)
        
        # Reconstruct
        if np.max(abs_A) > 1e-30:
            phase = A / np.maximum(abs_A, 1e-30)
        else:
            phase = np.ones(N, dtype=complex)
        A_new = abs_new * phase
        
        # IFFT
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        a = a_new
        
        # Check score periodically
        if it % 500 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc = _score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    
    return best_a, best_sc


def _topk_smoobj_grad(z, beta, K, n):
    """Top-K log-sum-exp smoothed objective and gradient in log-domain."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        c_top = c
    
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    
    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


class _TopkSmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[1]


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with top-K smoothing and annealing."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.008)
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    schedule = [
        (1.0, 0.1), (0.7, 0.08), (0.5, 0.06), (0.3, 0.04),
        (0.2, 0.03), (0.1, 0.02), (0.05, 0.015), (0.03, 0.01),
        (0.01, 0.008), (0.005, 0.005), (0.002, 0.003), (0.001, 0.002),
    ]
    
    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        beta = base_beta * beta_mult
        K = max(5, int((2 * n - 1) * K_frac))
        
        est_time = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 20))))
        
        prob = _TopkSmoProb(beta, K, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 50}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

    if rng is None:
        rng = np.random.default_rng(42)

        a_new = a_new / a_new.sum()
        
        sc_new = _score(a_new)
        sc_cur = _score(a)  # Hmm, this is expensive
        # Actually, we should track current score

"""
Multi-strategy optimizer combining frequency-domain flat-spectrum design,
log-domain L-BFGS with aggressive top-K annealing, Nesterov-accelerated
subgradient, and coordinate-descent final polish.
Key innovations: frequency-domain diverse initialization, sharper K annealing,
dual-method subgradient with adaptive active sets, and per-coordinate micro-optimization.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_from_conv(a, c):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * float(c.max()) / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _topk_smoobj_grad(z, beta, K, n):
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        c_top = c
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


class _TopkSmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[1]


def _flat_spectrum_iter(a_init, n_iters=20000, beta_start=0.05, beta_end=0.6, deadline=None, min_time=5.0):
    n = len(a_init)
    a = a_init.copy()
    N = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = _score(a)
    t0 = time.monotonic()
    for it in range(n_iters):
        if deadline is not None and time.monotonic() + min_time > deadline:
            break
        frac = it / max(1, n_iters - 1)
        beta = beta_start + (beta_end - beta_start) * (frac ** 0.7)
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        abs_A = np.abs(A)
        abs_A_mean = np.sqrt(np.mean(abs_A ** 2))
        log_abs = np.log(np.maximum(abs_A, 1e-30))
        log_abs_mean = np.log(max(abs_A_mean, 1e-30))
        log_abs_new = (1.0 - beta) * log_abs + beta * log_abs_mean
        abs_new = np.exp(log_abs_new)
        phase = A / np.maximum(abs_A, 1e-30)
        A_new = abs_new * phase
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        if it % 1000 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    sc = _score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=False):
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    if aggressive:
        schedule = [
            (0.5, 0.05), (0.3, 0.03), (0.2, 0.02), (0.1, 0.015),
            (0.05, 0.01), (0.03, 0.008), (0.01, 0.005), (0.005, 0.003),
            (0.002, 0.002), (0.001, 0.001), (0.0005, 0.001),
        ]
    else:
        schedule = [
            (1.0, 0.1), (0.7, 0.07), (0.5, 0.05), (0.3, 0.03),
            (0.2, 0.02), (0.1, 0.015), (0.05, 0.01), (0.03, 0.008),
            (0.01, 0.005), (0.005, 0.003), (0.002, 0.002), (0.001, 0.001),
        ]
    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        beta = base_beta * beta_mult
        K = max(5, int((2 * n - 1) * K_frac))
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(4000, max(30, int(remaining / (est_time * 25))))
        prob = _TopkSmoProb(beta, K, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 150, 'maxls': 50}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a


def _subgradient_nesterov(a_init, deadline, min_time_remaining=60.0):
    """Nesterov-accelerated subgradient with adaptive active set."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n_2nm1 = 2 * n - 1
    active_frac = 0.005
    lr = 0.03
    lr_min = 1e-5
    lr_max = 0.15
    momentum = 0.8
    stall = 0
    max_stall = 60
    velocity = np.zeros(n)
    y = a.copy()  # Nesterov lookahead
    prev_a = None
    prev_g = None
    iteration = 0
    t_start = time.monotonic()
    last_lbfgs_time = t_start
    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        # Nesterov lookahead
        a_eval = y
        c = _full_convolve(a_eval, a_eval)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = _normalize(a_eval.copy())
            stall = 0
        else:
            stall += 1
        # Active set
        n_active = max(3, min(n_2nm1, int(n_2nm1 * active_frac)))
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_active]
        active_vals = c[active_idx]
        w = np.zeros(n_2nm1)
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            # Exponential weighting: emphasis on tallest peaks
            w[active_idx] = np.exp(2.0 * (active_vals - active_vals.min()) / max(spread, 1e-18))
        else:
            w[active_idx] = 1.0 / n_active
        w = w / w.sum()
        # Gradient at lookahead point
        g = _grad_maxconv(a_eval, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break
        # BB step
        if prev_a is not None and prev_g is not None:
            s_vec = a_eval - prev_a
            y_vec = g_centered - prev_g
            s_norm = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm > 1e-18 and abs(sy) > 1e-18:
                lr_bb = s_norm / abs(sy)
                lr = np.clip(lr_bb, lr_min, lr_max)
        # Nesterov update
        a_new = a - lr * g_centered
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new
            # Update momentum
            beta_m = (1.0 + np.sqrt(1.0 + 4.0 * momentum ** 2)) / 2.0
            velocity = momentum * velocity - lr * g_centered
            a = a_new
            y = a + velocity * (1.0 / beta_m)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-10:
                y = y / s_y
        else:
            a = best_a.copy()
            y = a.copy()
            velocity = np.zeros(n)
        # Line search (simplified)
        sc_try = _score(a)
        if sc_try > current_sc * 1.05:
            a = best_a.copy()
            y = a.copy()
            velocity = np.zeros(n)
            lr = max(lr_min, lr * 0.5)
        prev_a = a_eval.copy()
        prev_g = g_centered.copy()
        # Adapt
        if stall > max_stall:
            active_frac = min(0.05, active_frac * 2.5)
            lr = max(lr_min, lr * 0.5)
            momentum = min(0.9, momentum + 0.05)
            stall = 0
        if stall == 0:
            active_frac = max(0.002, active_frac * 0.97)
        # Periodic L-BFGS
        if iteration % 400 == 0 and time.monotonic() - last_lbfgs_time > 90:
            if time.monotonic() + min_time_remaining + 60 < deadline:
                lbfgs_dl = time.monotonic() + 50
                a_tmp = _run_lbfgs(a, lbfgs_dl, min_time_remaining=5, aggressive=True)
                sc_tmp = _score(a_tmp)
                if sc_tmp < best_sc:
                    best_a = a_tmp.copy()
                    best_sc = sc_tmp
                    a = a_tmp.copy()
                    y = a.copy()
                    velocity = np.zeros(n)
                    prev_a = None
                    prev_g = None
                    last_lbfgs_time = time.monotonic()
    return best_a, best_sc


def _coordinate_descent_polish(a_init, deadline, min_time_remaining=10.0, n_passes_max=500):
    """Per-coordinate descent for final polish."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    # Precompute: for coordinate i, the "influence" on each conv position
    # d(conv[k])/da[i] = a[k-i] + a[2n-2-k+i] (approximately)
    # Instead, just do random coordinate updates
    rng = np.random.default_rng(12345)
    delta = 1e-4
    no_improve = 0
    max_no_improve = 200
    passes = 0
    while time.monotonic() + min_time_remaining < deadline and passes < n_passes_max:
        passes += 1
        improved = False
        for _ in range(n):
            i = rng.integers(n)
            for sign in [1, -1]:
                a_try = a.copy()
                a_try[i] = max(0.0, a_try[i] + sign * delta)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < best_sc - 1e-15:
                    a = a_try
                    best_a = a_try.copy()
                    best_sc = sc_try
                    improved = True
                    break
        if not improved:
            no_improve += 1
            delta *= 0.7
            if no_improve > max_no_improve:
                break
        else:
            no_improve = 0
            delta = min(delta * 1.05, 1e-3)
    return best_a, best_sc

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":

    elif kind == "optimal_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    best_seq = None
    best_score = np.inf
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_init = 16384
        best_seq = _make_profile(n_init, "tapered")
        best_score = _score(best_seq)
        print(f"No parent, tapered n={n_init}, score={best_score:.10f}")
    n = len(best_seq)
    # === Phase 1: Frequency-domain diverse init (60s) ===
    print(f"\n[P1] Frequency-domain init (t={time.monotonic()-t_start:.1f}s)")
    p1_dl = min(deadline - 1200, t_start + 60)
    a_fs1, sc_fs1 = _flat_spectrum_iter(best_seq, n_iters=15000, beta_start=0.05, beta_end=0.5, deadline=p1_dl, min_time=30)
    if sc_fs1 < best_score:
        best_seq = a_fs1.copy()
        best_score = sc_fs1
        print(f"  Flat-spectrum: {best_score:.10f} [fs_hard]")
    a_fs2, sc_fs2 = _flat_spectrum_soft(best_seq, n_iters=10000, beta_start=0.03, beta_end=0.3, deadline=p1_dl, min_time=30)
    if sc_fs2 < best_score:
        best_seq = a_fs2.copy()
        best_score = sc_fs2
        print(f"  Flat-spectrum soft: {best_score:.10f} [fs_soft]")
    # Also try from a different profile
    a_alt = _asymmetric_profile(n, skew=0.2)
    a_fs3, sc_fs3 = _flat_spectrum_iter(a_alt, n_iters=8000, beta_start=0.1, beta_end=0.6, deadline=p1_dl, min_time=20)
    if sc_fs3 < best_score:
        best_seq = a_fs3.copy()
        best_score = sc_fs3
        print(f"  Flat-spectrum (asym): {best_score:.10f} [fs_asym]")
    print(f"  After P1: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    # === Phase 2: Aggressive L-BFGS (400s) ===
    print(f"\n[P2] Aggressive L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p2_dl = min(deadline - 800, time.monotonic() + 400)
    a2 = _run_lbfgs(best_seq, p2_dl, min_time_remaining=60, aggressive=True)
    sc2 = _score(a2)
    if sc2 < best_score:
        best_seq = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_agg] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")
    # === Phase 3: Nesterov subgradient (300s) ===
    print(f"\n[P3] Nesterov subgradient (t={time.monotonic()-t_start:.1f}s)")
    p3_dl = min(deadline - 500, time.monotonic() + 300)
    a3, sc3 = _subgradient_nesterov(best_seq, p3_dl, min_time_remaining=50)
    if sc3 < best_score:
        best_seq = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} [nesterov] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")
    # === Phase 4: Second L-BFGS (300s) ===
    print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    p4_dl = min(deadline - 350, time.monotonic() + 300)
    a4 = _run_lbfgs(best_seq, p4_dl, min_time_remaining=50, aggressive=True)
    sc4 = _score(a4)
    if sc4 < best_score:
        best_seq = a4.copy()
        best_score = sc4
        print(f"  NEW BEST: {best_score:.10f} [lbfgs2] t={time.monotonic()-t_start:.1f}s")
    # === Phase 5: Peak-focused subgradient (200s) ===
    print(f"\n[P5] Peak-focused subgradient (t={time.monotonic()-t_start:.1f}s)")
    p5_dl = min(deadline - 250, time.monotonic() + 200)
    a5, sc5 = _peak_focused_subgradient(best_seq, p5_dl, min_time_remaining=40)
    if sc5 < best_score:
        best_seq = a5.copy()
        best_score = sc5
        print(f"  NEW BEST: {best_score:.10f} [peak_subgrad] t={time.monotonic()-t_start:.1f}s")
    # === Phase 6: L-BFGS round 3 (200s) ===
    print(f"\n[P6] L-BFGS round 3 (t={time.monotonic()-t_start:.1f}s)")
    p6_dl = min(deadline - 150, time.monotonic() + 200)
    a6 = _run_lbfgs(best_seq, p6_dl, min_time_remaining=30, aggressive=True)
    sc6 = _score(a6)
    if sc6 < best_score:
        best_seq = a6.copy()
        best_score = sc6
        print(f"  NEW BEST: {best_score:.10f} [lbfgs3] t={time.monotonic()-t_start:.1f}s")
    # === Phase 7: Multi-restart + perturbation (150s) ===
    print(f"\n[P7] Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    p7_dl = min(deadline - 80, time.monotonic() + 150)
    sigmas = [0.02, 0.04, 0.06, 0.03, 0.08, 0.015, 0.05, 0.1]
    n_restarts = max(3, int((p7_dl - time.monotonic()) / 25))
    for i in range(min(n_restarts, 12)):
        if time.monotonic() > p7_dl - 10:
            break
        sigma = sigmas[i % len(sigmas)]
        a_pert = _perturb_log(best_seq, sigma, rng)
        # Quick flat-spectrum + L-BFGS
        a_fs, sc_fs = _flat_spectrum_iter(a_pert, n_iters=3000, beta_start=0.1, beta_end=0.5,
                                          deadline=min(p7_dl - 5, time.monotonic() + 10), min_time=5)
        a_r = _run_lbfgs(a_fs, min(p7_dl - 5, time.monotonic() + 18), min_time_remaining=5, aggressive=True)
        sc_r = _score(a_r)
        if sc_r < best_score:
            best_seq = a_r.copy()
            best_score = sc_r
            print(f"  NEW BEST: {best_score:.10f} [restart_{i}_sig={sigma}] t={time.monotonic()-t_start:.1f}s")
    # === Phase 8: Coordinate descent polish (60s) ===
    print(f"\n[P8] Coordinate descent polish (t={time.monotonic()-t_start:.1f}s)")
    p8_dl = min(deadline - 30, time.monotonic() + 60)
    a8, sc8 = _coordinate_descent_polish(best_seq, p8_dl, min_time_remaining=20, n_passes_max=200)
    if sc8 < best_score:
        best_seq = a8.copy()
        best_score = sc8
        print(f"  NEW BEST: {best_score:.10f} [coord_desc] t={time.monotonic()-t_start:.1f}s")
    # === Phase 9: Final L-BFGS micro-optimization (30s) ===
    print(f"\n[P9] Final micro L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p9_dl = min(deadline - 5, time.monotonic() + 30)
    a9 = _run_lbfgs(best_seq, p9_dl, min_time_remaining=3, aggressive=True)
    sc9 = _score(a9)
    if sc9 < best_score:
        best_seq = a9.copy()
        best_score = sc9
        print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
    # === Final validation ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
