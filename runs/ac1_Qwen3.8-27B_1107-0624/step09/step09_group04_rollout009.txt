    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    n = len(a)
    s = float(a.sum())

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    c = _full_convolve(a, a)
    c_max = float(c.max())

    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

    return 2.0 * n * (c_max + beta * np.log(s))

def _qp_active_step(a, active_idx, epsilon):
    """Solve SOCP to find optimal perturbation d that reduces active peaks.
    Returns the new sequence a + d (projected to be valid)."""
    if not HAS_CVXPY:
        return None
    
    n = len(a)
    c = _full_convolve(a, a)
    
    # Build coefficient vectors for each active lag
    # dconv[k] (linearized) = sum_j d[j] * coeff[k, j]
    # where coeff[k, j] = a[k-j] (if valid) + a[j-k] (if valid)
    
    d = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        cp.norm(d, 2) <= epsilon,
        cp.sum(d) == 0
    ]
    
    for k in active_idx:
        coeff = np.zeros(n)
        k = int(k)
        for j in range(n):
            val = 0.0
            if 0 <= k - j < n:
                val += a[k - j]
            if 0 <= j - k < n:
                val += a[j - k]
            coeff[j] = val
        constraints.append(c[k] + coeff @ d <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_time=30)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_time=30)
        except Exception:
            return None
    
    if d.value is None:
        return None
    
    a_new = a + d.value
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return None
    a_new = a_new / s
    return a_new

def _active_set_momentum(a_init, deadline, min_time_remaining=30.0):
    """Active-set gradient with Nesterov momentum and adaptive step size."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    momentum = np.zeros(n)
    mu = 0.5  # momentum coefficient
    lr = 0.03
    no_improve = 0
    K = 5

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            K = min(min(100, n), K + 2)
            mu = min(0.85, mu * 1.01)
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.8
                K = max(3, K // 2)
                mu = 0.4
            if no_improve > 500:
                a = best_a.copy()
                lr = 0.03
                K = 5
                mu = 0.5
                momentum = np.zeros(n)
                no_improve = 0
                if no_improve > 1000:
                    break

        # Get active set
        top_k_idx = np.argsort(c)[::-1][:K]
        w_S = np.zeros(2 * n - 1)
        w_S[top_k_idx] = 1.0 / K
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Nesterov momentum
        momentum = mu * momentum + grad
        direction = -momentum
        direction = direction - direction.mean()
        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-16:
            break

        # Line search with backtracking
        alpha = lr / (dnorm + 1e-15)
        improved = False
        for _ in range(40):
            a_try = a + alpha * direction
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = 2.0 * n * float(_full_convolve(a_try, a_try).max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if improved:
            lr = min(0.08, lr * 1.02)

    return best_a, best_sc


def _spectral_flatten(a, num_iters=50, alpha=0.1):
    """Frequency-domain spectral flattening: reduce peaks in autocorrelation
    by adjusting the amplitude spectrum while preserving phase."""
    n = len(a)
    a_cur = a.copy()
    N = _next_pow2(2 * n)
    best_a = a_cur.copy()
    best_sc = _score(best_a)

    for it in range(num_iters):
        # FFT of current sequence (zero-padded to length N)
        A = np.fft.fft(a_cur, N)
        # Circular autocorrelation (length N)
        r = np.fft.ifft(np.abs(A) ** 2).real
        # Take the relevant part: r[0:n-1] are lags 0 to n-1, r[N-n+1:N] are lags -(n-1) to -1
        r_pos = r[:n]
        r_neg = r[N - n + 1:N][::-1]  # lags 1 to n-1 (negatively indexed)
        # Full autocorrelation of length 2n-1
        r_full = np.concatenate([r_neg, r_pos])

        c_max = float(r_full.max())
        if c_max < 1e-15:
            break

        # Find peaks
        threshold = c_max * 0.95
        peak_mask = r_full >= threshold

        # Compute frequency-domain correction
        # For each peak at lag m, subtract alpha * exp(2*pi*i*m*k/N) / N from |A|^2
        # Simplified: just reduce the power at frequencies that contribute to peaks
        A_abs = np.abs(A)
        A_phase = np.angle(A)

        # Compute the "interference" pattern in frequency domain
        # The peak at lag m in time domain corresponds to a modulation at freq m/N
        correction = np.zeros(N, dtype=complex)
        peak_lags = np.where(peak_mask)[0]
        for m in peak_lags:
            if m >= n:
                m = m - (2 * n - 1)
            # Add a complex exponential correction
            k_freq = np.arange(N)
            correction += alpha * np.exp(2j * np.pi * m * k_freq / N) / len(peak_lags)

        # Apply correction to amplitude
        A_new = A - correction
        # Reconstruct
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a_cur = a_new
        else:
            # Revert and reduce alpha
            alpha *= 0.7
            if alpha < 0.001:
                break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 5.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.3) ** 2) + 0.6 * np.exp(-15.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _diversify(a, rng, strength=0.03):
    """Create a diversified version of a using random perturbation."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    # Random perturbation with structure
    noise = rng.standard_normal(n) * 0.5
    t = np.arange(n) / n
    ramp = 0.3 * np.sin(2 * np.pi * rng.uniform(1, 5) * t)
    perturbation = noise + ramp
    perturbation = perturbation - perturbation.mean()
    z = z + strength * perturbation
    z = z - z.mean()
    return _softmax(z)


def _direct_qp_step(a, deadline, min_time_remaining=5.0):
    """Single QP step using CVXPY to find optimal perturbation.
    Falls back to gradient step if QP is too slow or unavailable."""
    if not HAS_CVXPY:
        return None

    n = len(a)
    if n > 32768:
        return None  # Too large for QP

    c = _full_convolve(a, a)
    c_max = float(c.max())

    # Pick top 5 peaks
    K = 5
    top_idx = np.argsort(c)[::-1][:K]

    # Build gradient matrix (K x n)
    G = np.zeros((K, n))
    for i, k in enumerate(top_idx):
        k = int(k)
        for j in range(n):
            val = 0.0
            if 0 <= k - j < n:
                val += a[k - j]
            if 0 <= j - k < n:
                val += a[j - k]
            G[i, j] = val

    epsilon = 0.02 * np.max(a) * 5  # small perturbation

    d = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        cp.norm(d, 2) <= epsilon,
        cp.sum(d) == 0
    ]
    for i in range(K):
        constraints.append(c[int(top_idx[i])] + G[i] @ d <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)
    t0 = time.monotonic()
    try:
        prob.solve(solver=cp.SCS, max_time=20, verbose=False)
    except Exception:
        return None

    if time.monotonic() - t0 > 25:
        return None

    if d.value is None:
        return None

    a_new = a + d.value
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return None
    a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase optimizer combining L-BFGS, active-set momentum descent,
    spectral flattening, QP steps, and multi-n search."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    def _try_update(a_new, label):
        nonlocal best_seq, best_score
        if a_new is None:
            return
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            return
        sc = _score(a_new)
        if sc < best_score:
            best_seq = a_new.copy()
            best_score = sc
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [{label}] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  {label}: score={sc:.10f} (no improvement) t={time.monotonic()-t_start:.1f}s")

    # === Phase 1: L-BFGS warmup ===
    print(f"\n[P1] L-BFGS warmup (t={time.monotonic()-t_start:.1f}s)")
    a_p1 = _run_lbfgs(parent, deadline, min_time_remaining=900.0, beta_scale=1.0)
    _try_update(a_p1, "lbfgs_warmup")

    # === Phase 2: Active set with momentum ===
    print(f"\n[P2] Active set momentum (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 700:
        a_eq, sc_eq = _active_set_momentum(best_seq, deadline, min_time_remaining=700.0)
        _try_update(a_eq, "active_momentum")

    # === Phase 3: Spectral flattening ===
    print(f"\n[P3] Spectral flattening (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        a_spec, sc_spec = _spectral_flatten(best_seq, num_iters=30, alpha=0.05)
        _try_update(a_spec, "spectral")

    # === Phase 4: Fine L-BFGS ===
    print(f"\n[P4] Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        a_fine = _run_lbfgs(best_seq, deadline, min_time_remaining=500.0, beta_scale=0.3)
        _try_update(a_fine, "fine_lbfgs")

    # === Phase 5: QP active set steps ===
    print(f"\n[P5] QP steps (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400 and HAS_CVXPY:
        for qi in range(5):
            if time.monotonic() > deadline - 420:
                break
            a_qp = _direct_qp_step(best_seq, deadline, min_time_remaining=400.0)
            _try_update(a_qp, f"qp_{qi}")
            # Follow with short L-BFGS
            if a_qp is not None and time.monotonic() < deadline - 410:
                a_qp2 = _run_lbfgs(a_qp, deadline, min_time_remaining=410.0, beta_scale=0.2)
                _try_update(a_qp2, f"qp_{qi}_polish")

    # === Phase 6: Diversified restart ===
    print(f"\n[P6] Diversified restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        for trial in range(3):
            if time.monotonic() > deadline - 360:
                break
            a_div = _diversify(best_seq, rng, strength=0.02 + 0.01 * trial)
            a_div_opt = _run_lbfgs(a_div, deadline, min_time_remaining=350.0, beta_scale=0.5)
            _try_update(a_div_opt, f"diversify_{trial}")

    # === Phase 7: Active set round 2 ===
    print(f"\n[P7] Active set round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a_eq2, sc_eq2 = _active_set_momentum(best_seq, deadline, min_time_remaining=200.0)
        _try_update(a_eq2, "active_momentum2")

    # === Phase 8: Multi-n exploration ===
    print(f"\n[P8] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([len(best_seq), 16384, 32768, 8192]), reverse=True)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=100.0, beta_scale=0.5)
        _try_update(a_opt, f"multi_n_{n_try}")

    # === Phase 9: Final L-BFGS polish ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 5 - time.monotonic()
    if remaining > 20:
        a_final = _run_lbfgs(best_seq, deadline, min_time_remaining=5.0, beta_scale=0.1)
        _try_update(a_final, "final_lbfgs")

        # Final active set
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _active_set_momentum(best_seq, deadline - 2, min_time_remaining=2.0)
            _try_update(a_fin2, "final_active")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
