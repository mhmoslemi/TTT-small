        # Get top few peaks
        n_peaks = max(3, int(0.005 * n))
        peak_idx = np.argpartition(c, -n_peaks)[-n_peaks:]

        k_star = int(np.argmax(c))
        
        # Leverage: how much each position contributes to c[k_star]
        # ∂c[k_star]/∂a[m] = 2*a[k_star - m] (with a[t]=0 for invalid t)
        # But we also care about the effect on OTHER peaks
        # Simplification: just use the leverage on k_star
        
        m_lo = max(0, k_star - (n-1))
        m_hi = min(n-1, k_star)
        leverage = np.zeros(n)
        for m in range(m_lo, m_hi + 1):
            leverage[m] = a[k_star - m]
        
        # Find top-leverage (to reduce) and bottom-leverage (to increase)
        n_move = max(1, int(0.001 * n))
        # Only consider positions in the valid range for k_star
        valid = np.arange(m_lo, m_hi + 1)
        lev_valid = leverage[valid]
        
        if lev_valid.max() - lev_valid.min() < 1e-15:
            break
        
        # Sort by leverage
        sorted_idx = valid[np.argsort(lev_valid)]
        reduce_positions = sorted_idx[-n_move:]  # highest leverage
        increase_positions = sorted_idx[:n_move]  # lowest leverage
        
        # Compute the swap
        avg_reduce = np.mean(a[reduce_positions])
        avg_increase = np.mean(a[increase_positions])
        
        # Move mass from high-leverage to low-leverage
        total_mass = np.sum(a[reduce_positions])
        n_r = len(reduce_positions)
        n_i = len(increase_positions)
        
        # Reduce each reduce position proportionally
        frac = 0.1  # fraction of mass to move
        delta = np.zeros(n)
        for m in reduce_positions:
            delta[m] = -frac * a[m]
        for m in increase_positions:
            delta[m] = frac * a[m] * n_r / n_i  # balance mass
        
        # Normalize to preserve sum
        delta -= delta.mean()
        
        # Line search
        alpha = 1.0
        improved = False
        for _ in range(30):
            a_try = a + alpha * delta
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 10
    
    return best_a, best_sc

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n_out = len(a) + len(b) - 1
    N = _next_pow2(n_out)
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:n_out]

    n = len(a)

    if s < 1e-15:

    return 2.0 * n * float(c.max()) / (s * s)
