def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    if s < 1e-15:

    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]

        c_top = c

    s = e.sum()
    
    if s < 1e-300 or not np.isfinite(s):

        w[int(np.argmax(c))] = 1.0

        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    
    ga = _grad_maxconv(a, w)
    return obj, ga, w

        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):

        a = a / a.sum()
        obj, _, _ = _topk_smoothed(a, self.beta, self.K)
        return obj

    def grad(self, z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        _, ga, _ = _topk_smoothed(a, self.beta, self.K)
        W = float(np.dot(a, ga))
        return a * (ga - W)

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations:
- Adaptive L-BFGS with dynamic beta/K based on peak distribution
- Peak-focused coordinate descent with O(n) per-coordinate evaluation
- Subgradient method with Polyak step sizes and Nesterov momentum
- Asymmetric pairwise adjustments exploiting non-symmetric structure
- Multi-n exploration and diverse spectral initialization

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _topk_smoothed(a, beta, K):
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    ga = _grad_maxconv(a, w)
    return obj, ga


class _SmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        obj, _ = _topk_smoothed(a, self.beta, self.K)
        return obj

    def grad(self, z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        _, ga = _topk_smoothed(a, self.beta, self.K)
        W = float(np.dot(a, ga))
        return a * (ga - W)


def _adaptive_beta_K(a):
    """Compute adaptive beta and K from current peak distribution."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    if len(c_sorted) > 20:
        gap = c_max - c_sorted[20]
    else:
        gap = c_max - c_sorted[-1]
    if gap < c_max * 0.002:
        gap = c_max * 0.002
    beta = max(gap * 0.5, c_max * 0.003)
    # K: number of peaks within 3% of max
    K = max(5, int(np.sum(c >= c_max * 0.97)))
    K = min(K, len(c))
    return beta, K, c


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):

    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()

    beta, K, c_cur = _adaptive_beta_K(a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Phased schedule with adaptive restarts
    phases = [
        (1.0, 1.0), (0.6, 0.8), (0.4, 0.6), (0.25, 0.45),
        (0.15, 0.3), (0.08, 0.2), (0.05, 0.12), (0.03, 0.07),
        (0.015, 0.04), (0.008, 0.02), (0.004, 0.01), (0.002, 0.006),
        (0.001, 0.003), (0.0005, 0.002),

    for beta_mult, K_mult in phases:

        remaining = deadline - min_time_remaining - time.monotonic()

def _coord_descent(a_init, deadline, min_time_remaining=30.0):

    n = len(a_init)

    best_a = a.copy()
    best_sc = _score(a)

    n_coords = max(200, min(2000, n // 8))
    fracs = np.array([0.3, 0.5, 0.6, 0.7, 0.8, 0.85, 0.9, 0.95,
                      1.05, 1.1, 1.15, 1.2, 1.3, 1.5, 2.0, 0.4, 0.65])

    no_improve_rounds = 0
    max_no_improve = 50

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        current_sc = 2.0 * n * float(c.max())

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve_rounds = 0
        else:
            no_improve_rounds += 1
            if no_improve_rounds > max_no_improve:
                break

        # Find top peaks
        n_peaks = max(20, min(200, n // 64))
        peak_idx = np.argpartition(c, -n_peaks)[-n_peaks:]

        # Compute responsibility for each coordinate
        resp = np.zeros(n)
        for k in peak_idx:
            i_lo = max(0, k - (n - 1))
            i_hi = min(n - 1, k)
            if i_hi >= i_lo:
                idx_a = np.arange(i_lo, i_hi + 1)
                idx_b = k - idx_a
                prods = a[idx_a] * a[idx_b]
                np.add.at(resp, idx_a, prods)
                np.add.at(resp, idx_b, prods)

        top_coords = np.argsort(resp)[::-1][:n_coords]
        round_improved = False

        for i in top_coords:
            if time.monotonic() + 10 > deadline - min_time_remaining:
                break
            x0 = a[i]
            if x0 < 1e-8 or x0 > 0.45:
                continue

            # Compute D: convolution without a[i]'s contribution
            j_arr = np.arange(2 * n - 1) - i
            valid_j = (j_arr >= 0) & (j_arr < n)
            D = c.copy()
            cross_valid = valid_j & (j_arr != i)
            if np.any(cross_valid):
                D[cross_valid] -= 2.0 * x0 * a[j_arr[cross_valid]]
            if 2 * i < 2 * n - 1:

            best_sc_i = current_sc
            best_x_i = x0

            for frac in fracs:
                x = x0 * frac
                if x < 0 or (1.0 - x) <= 1e-10:

                    c_new[cross_valid] += 2.0 * s * x * a[j_arr[cross_valid]]

                    best_x_i = x

def _subgradient_polyak(a_init, deadline, min_time_remaining=60.0):
    """Subgradient with Polyak step and adaptive active set."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    # Polyak: lr = (f - f*) / ||g||^2
    f_lower = 2.0 * n / (2 * n - 1)  # theoretical lower bound
    momentum = 0.6
    velocity = np.zeros(n)
    lr_scale = 1.0
    stall = 0
    max_stall = 100
    active_frac = 0.005

    t_start = time.monotonic()

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive active set
        n_active = max(10, min(2 * n - 1, int((2 * n - 1) * active_frac)))
        peak_idx = np.argpartition(c, -n_active)[-n_active:]
        peak_vals = c[peak_idx]
        spread = peak_vals.max() - peak_vals.min()
        w = np.zeros(2 * n - 1)
        if spread > 1e-18:
            w[peak_idx] = (peak_vals - peak_vals.min()) / spread + 0.05
        else:
            w[peak_idx] = 1.0 / n_active
        w = w / w.sum()

        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-16:
            break

        # Polyak step
        gap = current_sc - f_lower
        if gap > 1e-15:
            lr = lr_scale * gap / (gnorm * gnorm)
        else:
            lr = 0.01
        lr = min(lr, 0.1)

        velocity = momentum * velocity - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            velocity = np.zeros(n)
            a_try = a
        else:
            a_try = a_try / s_try

        # Line search
        improved = False
        alpha = 1.0
        for _ in range(20):
            a_ls = a + alpha * velocity
            a_ls = np.maximum(a_ls, 0.0)
            s_ls = a_ls.sum()
            if s_ls < 1e-10:
                alpha *= 0.5
                continue
            a_ls = a_ls / s_ls
            c_ls = _full_convolve(a_ls, a_ls)
            sc_ls = 2.0 * n * float(c_ls.max())
            if sc_ls < current_sc - 1e-15:
                a = a_ls
                improved = True
                break
            alpha *= 0.5

        if improved:
            lr_scale = min(3.0, lr_scale * 1.1)
            active_frac = max(0.001, active_frac * 0.97)
        else:
            velocity = np.zeros(n)
            lr_scale = max(0.1, lr_scale * 0.5)

        if stall > max_stall:
            active_frac = min(0.05, active_frac * 2.0)
            momentum = min(0.9, momentum + 0.03)
            stall = 0

    return best_a, best_sc

                new_ai = total * frac
                new_aj = total * (1.0 - frac)
                a[i] = new_ai
                a[j] = new_aj
                c_try = _full_convolve(a, a)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < best_sc_i - 1e-14:
                    best_sc_i = sc_try
                    best_split = (new_ai, new_aj)

            if best_split != (ai, aj):
                a[i] = best_split[0]
                a[j] = best_split[1]
                round_improved = True

            # Reset
            a[i] = ai
            a[j] = aj

        if not round_improved:
            no_improve += 10

    return best_a, best_sc


def _make_profile(n, kind="tapered", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        for alpha in [1.5, 2.0, 3.0, 4.0, 5.0]:
            a = np.exp(-alpha * t) + 0.05
            return a / a.sum()
    elif kind == "exp_both_sides":
        t = np.linspace(0, 1, n)
        c = 0.3
        a = np.exp(-2.0 * (t - c) ** 2) + 0.05
        return a / a.sum()
    elif kind == "power_law":
        t = np.linspace(0, 1, n)
        a = (1.0 + 2.0 * t) ** (-1.5) + 0.05
        return a / a.sum()
    elif kind == "asym_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.35) ** 2) + np.exp(-2.0 * (t - 0.7) ** 2) * 0.3 + 0.03
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.25) ** 2) + np.exp(-15.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    elif kind == "random_smooth":
        if rng is None:
            rng = np.random.default_rng(42)
        a = np.abs(rng.standard_normal(n))
        # Smooth it
        kernel = np.ones(21) / 21
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    """Very fine active-set subgradient polish."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.003
    tol = 0.003
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active

        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-16:
            break

        alpha = min(lr, 0.3 / (gnorm + 1e-15))
        improved = False
        for _ in range(30):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 15 == 0:
                tol = max(0.0001, tol * 0.7)
                lr *= 0.7
            if no_improve > 200:
                a = best_a.copy()
                tol = max(0.0001, tol * 0.5)
                lr = 0.002
                no_improve = 0
        else:
            if tol > 0.0001:
                tol = max(0.0001, tol * 0.97)
            lr = min(0.008, lr * 1.01)

    return best_a, best_sc
