Target: minimize 2*n*max(conv(a,a))/sum(a)^2 to below 1.5027.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()

    return e / s


def _hankel_matvec(x, W, n):
    """Compute 2 * (M @ x) where M[i,j] = W[i+j]."""
    V = x[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    """Solve M @ x = rhs using CG, where M[i,j] = W[i+j]."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Mp = _hankel_matvec(p, W, n)
        pHp = float(p @ Mp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Mp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            # Try smaller step
            alpha *= 0.5
            x_new = x + alpha * p
            r_new = r - alpha * Mp
            rs_new = float(r_new @ r_new)
            if not np.isfinite(rs_new):
                break
        x = x_new
        r = r_new
        rr = float(r @ r)
        if rr < tol * tol * rs0:
            break
        p = (rr / rs) * p + r
        rs = rr
    return x


def _kkt_solve(W, n, max_iter=1000, tol=1e-14):
    """
    Solve the KKT system for given weight vector W.
    Returns (a, M) where a is the solution and M is the common value.
    """
    # System: M @ a = [lambda/2, ..., lambda/2; 1] where M[i,j] = W[i+j]
    # Augmented system:
    # [M  -1/2] [a]   [0]
    # [1^T  0  ] [lam] = [1]
    
    # Actually, let's solve: M @ a = (lambda/2) * 1, sum(a) = 1
    # Let a = a0 + t * v where a0 is any vector with sum 1
    # and M @ v = 1 (particular solution), v is the "direction"
    # Then lambda/2 = (M @ a0)/1_avg... 
    
    # Simpler: solve the (n+1) x (n+1) system using the structure.
    # M @ a = (lambda/2) * ones(n)
    # ones(n)^T @ a = 1
    
    # Let's use the approach: find particular solution to M @ x = ones(n)
    # Then a = x * c + d * 1 where c and d are determined by sum(a)=1
    # and the fact that M @ a = (lambda/2) * 1 means (lambda/2) = c * (M@x)_avg... 
    
    # Actually the cleanest approach: 
    # M @ a = mu * 1  (where mu = lambda/2)
    # 1^T @ a = 1
    # => a = mu * M^{-1} @ 1 (if M is invertible)
    # => mu * 1^T @ M^{-1} @ 1 = 1
    # => mu = 1 / (1^T @ M^{-1} @ 1)
    # => a = (M^{-1} @ 1) / (1^T @ M^{-1} @ 1)
    
    # So we need to solve M @ y = 1 and compute 1^T @ y.
    
    y = _cg_solve(W, n, np.ones(n), max_iter=max_iter, tol=tol)
    denom = float(np.ones(n) @ y)
    if abs(denom) < 1e-25 or not np.isfinite(denom):
        return None, np.inf
    
    a = y / denom
    # M = the common value of (a*a)[k] for active k
    c = _full_convolve(a, a)
    M = float(c.max())  # This should equal mu for the active set
    
    return a, M


def _kkt_activate(a, W_curr, K_list, deadline):
    """
    Active set iteration: start with top-K lags, solve KKT, check, expand.
    """
    n = len(a)
    a = np.maximum(a.copy(), 0)
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for K in K_list:
        if time.monotonic() > deadline:
            break
        K = min(K, 2 * n - 1)
        c = _full_convolve(a, a)
        idx = np.argpartition(c, -K)[-K:]
        W = np.zeros(2 * n - 1)
        W[idx] = 1.0 / K
        
        a_new, M_val = _kkt_solve(W, n, max_iter=2000, tol=1e-15)
        if a_new is None:
            continue
        
        # Check non-negativity
        if np.min(a_new) < -1e-6:
            # Clip and retry
            a_new = np.maximum(a_new, 0)
            a_new = _normalize(a_new)
        
        a_new = np.maximum(a_new, 0)
        a_new = _normalize(a_new)
        
        sc = _score(a_new)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_sc

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

        return np.inf
