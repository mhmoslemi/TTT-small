Minimizes 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS log-sum-exp continuation
with very fine beta schedule, active-set gradient polishing, and multi-scale search.

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)

    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new

    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _ls_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta

    """Run L-BFGS for a given beta level."""
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

        for step in [0.001, 0.005, 0.01, 0.05, 0.1, 0.5]:
            a_try = a + step * d
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            sc = _score(a_try)
            if sc < best_sc - 1e-15:
                a = a_try
                best_sc = sc
                best_a = a_try.copy()
                improved = True
                break
        
        if not improved:
            # Try smaller steps
            for step in [0.0001, 0.0005, 0.002]:
                a_try = a + step * d
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try / s
                sc = _score(a_try)
                if sc < best_sc - 1e-15:
                    a = a_try
                    best_sc = sc
                    best_a = a_try.copy()
                    improved = True
                    break
            if not improved:
                break
    
    return best_a, best_sc
