"""
Multi-scale optimization with LP-based peak equalization for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses a cascade of increasingly precise
optimizations: multi-scale init, Nesterov subgradient, LP peak equalization,
and fine L-BFGS with adaptive temperature.
"""
import numpy as np
import time
import scipy.sparse
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _conv_grad(a, k):
    """Gradient of b_k = conv(a,a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    # b_k = sum_j a[j] * a[k-j]
    # db_k/da[j] = a[k-j] + a[j-k] (when indices are valid)
    for j in range(n):
        val = 0.0
        if 0 <= k - j < n:
            val += a[k - j]
        if 0 <= j - k < n:
            val += a[j - k]
        grad[j] = val
    return grad


def _conv_grad_fast(a, k):
    """Vectorized gradient of b_k."""
    n = len(a)
    grad = np.zeros(n)
    # First term: a[k-j] for j where 0 <= k-j < n, i.e., k-n+1 <= j <= k
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    if j_lo <= j_hi:
        js = np.arange(j_lo, j_hi + 1)
        grad[js] += a[k - js]
    # Second term: a[j-k] for j where 0 <= j-k < n, i.e., k <= j <= k+n-1
    j_lo2 = max(0, k)
    j_hi2 = min(n - 1, k + n - 1)
    if j_lo2 <= j_hi2:
        js = np.arange(j_lo2, j_hi2 + 1)
        grad[js] += a[js - k]
    return grad


def _conv_grad_multi(a, indices):
    """Sum of gradients for multiple k values."""
    n = len(a)
    grad = np.zeros(n)
    for k in indices:
        grad += _conv_grad_fast(a, k)
    return grad

"""
Precise multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Strategy: Start from parent, apply L-BFGS with fine beta schedule,
peak surgery with exact line search, multi-n exploration, and final polish.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _conv_grad_fast(a, k):
    n = len(a)
    grad = np.zeros(n)
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    if j_lo <= j_hi:
        js = np.arange(j_lo, j_hi + 1)
        grad[js] += a[k - js]
    j_lo2 = max(0, k)
    j_hi2 = min(n - 1, k + n - 1)
    if j_lo2 <= j_hi2:
        js = np.arange(j_lo2, j_hi2 + 1)
        grad[js] += a[js - k]
    return grad


def _conv_grad_multi(a, indices):
    n = len(a)
    grad = np.zeros(n)
    for k in indices:
        grad += _conv_grad_fast(a, k)
    return grad


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)

        beta_schedule = base_beta * multipliers

    current_a = a_init.copy()

    s = a.sum()

    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0
    
    # Adaptive K for active set

    while time.monotonic() + min_remaining < deadline:

        M = float(b.max())
        current_sc = 2.0 * n * M

        if current_sc < best_sc - 1e-16:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:

        # Adaptive K
        b_sorted = np.sort(b)[::-1]
        gap = b_sorted[0] - b_sorted[min(K, len(b_sorted)-1)]
        if gap < M * 0.001 and K < n // 128:
            K = min(n // 128, int(K * 1.5) + 1)

        # Active set
        active = np.argsort(b)[::-1][:K]
        
        # Gradient
        g = _conv_grad_multi(a, active)
        g_t = g - g.mean()
        gn = np.linalg.norm(g_t)
        if gn < 1e-18:
            break

        # Backtracking line search
        alpha = 0.01
        improved = False
        for _ in range(50):
            a_try = _project_simplex(a - alpha * g_t / gn)
            b_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(b_try.max())
            if sc_try < current_sc - 1e-18:
                a = a_try
                improved = True
                break
            alpha *= 0.5
            if alpha < 1e-12:
                break
        
        if not improved:
            # Try a smaller active set
            if K > 3:
                K = max(3, K // 2)
            else:
                break
    
    return best_a, best_sc
