"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
- Frequency-domain damped Newton method exploiting convolution structure
- Adaptive active-set management with automatic tolerance scheduling
- Support refinement (pruning small elements to improve conditioning)
- KKT linear system solver for exact active-set refinement
- Multi-n exploration with time-budgeted resampling
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _softmax_logz(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_soft_structured(a, beta, n):
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _smoobj(z, beta, n):
    a = _softmax_logz(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax_logz(z)
    grad_a = _grad_soft_structured(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)


class _SoftProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _freq_domain_newton(a_init, deadline, min_time_remaining=120.0, verbose=True):
    """Frequency-domain damped Newton method for flattening conv(a,a).
    
    Key idea: In the DFT domain, conv(a,a) has transform A(omega)^2.
    The Newton step for the linearized problem 2*A*dA = -R gives
    dA = -R / (2*A). We regularize when A is small.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    a = _normalize(a)

    N = _next_pow2(2 * n - 1)

    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = float(2.0 * n * c0.max())

    if verbose:
        print(f"    FreqNewton start: n={n}, score={best_sc:.10f}")

    # Adaptive tolerance for active set
    tol_frac = 0.02
    min_tol_frac = 0.002
    no_improve_count = 0
    max_no_improve = 200

    # Damping factor
    damping = 1.0

    c = _full_convolve(a, a)
    A = np.fft.rfft(a, N)
    A_abs_max = float(np.abs(A).max())

    for it in range(500000):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        if no_improve_count > max_no_improve:
            tol_frac = max(min_tol_frac, tol_frac * 0.5)
            no_improve_count = 0
            if tol_frac <= min_tol_frac * 1.1:
                break

        # Active set
        threshold = c_max - tol_frac * c_max
        active = c >= threshold
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Residual: how much each active c[k] exceeds the mean of active values
        active_mean = float(np.mean(c[active]))
        r = np.zeros(len(c), dtype=np.float64)
        r[active] = c[active] - active_mean

        # Frequency domain Newton step
        R = np.fft.rfft(r, N)
        # dA = -R / (2*A + eps)
        eps = 1e-10 * A_abs_max
        denom = 2.0 * A + eps * np.sign(A + 1e-30)
        # Handle zero-crossings carefully
        mask = np.abs(denom) > 1e-15 * A_abs_max
        dA = np.zeros_like(A)
        dA[mask] = -R[mask] / denom[mask]
        # For real-valued result, enforce Hermitian symmetry (rfft already handles this)
        dA[0] = float(dA[0].real) if N > 1 else dA[0]

        # Line search
        best_t = 0.0
        best_trial_sc = current_sc
        for t_factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            t = t_factor * damping
            A_trial = A + t * dA
            a_trial = np.fft.irfft(A_trial, N)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                continue
            a_trial = a_trial / s_trial
            c_trial = _full_convolve(a_trial, a_trial)
            sc_trial = 2.0 * n * float(c_trial.max())
            if sc_trial < best_trial_sc - 1e-15:
                best_t = t
                best_trial_sc = sc_trial
                best_a_trial = a_trial
                best_c_trial = c_trial
                best_A_trial = A_trial

        if best_t == 0.0:
            # No improvement found, reduce damping
            damping *= 0.5
            if damping < 1e-6:
                break
            continue

        # Accept step
        a = best_a_trial
        c = best_c_trial
        A = best_A_trial
        A_abs_max = max(A_abs_max, float(np.abs(A).max()))
        damping = min(1.0, damping * 1.1)

        # Support refinement every 500 iterations
        if it % 500 == 499 and it > 0:
            # Try zeroing out elements below threshold
            threshold_a = 1e-6 * a.max()
            small = a < threshold_a
            n_small = int(small.sum())
            if 0 < n_small < n // 2:
                a_test = a.copy()
                a_test[small] = 0.0
                a_test = _normalize(a_test)
                sc_test = _score(a_test)
                if sc_test < best_sc:
                    a = a_test
                    c = _full_convolve(a, a)
                    A = np.fft.rfft(a, N)
                    best_sc = sc_test
                    best_a = a.copy()

        # Gradually tighten tolerance
        if no_improve_count == 0 and it % 100 == 0:
            tol_frac = max(min_tol_frac, tol_frac * 0.98)

        if verbose and (it % 5000 == 0 or it < 5):
            elapsed = time.monotonic()
            print(f"    FreqNewton it={it}: score={best_sc:.10f}, "
                  f"tol={tol_frac:.5f}, n_active={n_active}, t={best_t:.4f}")

    return best_a, best_sc


def _kkt_refinement(a_init, deadline, min_time_remaining=60.0, verbose=True):
    """Solve KKT linear system for exact active-set refinement.
    
    At optimum with active set S and support T:
    - For k in S: sum_{j in T} a[j]*a[k-j] = M
    - For j in T: 2*sum_{k in S} mu_k * a[k-j] = lambda
    
    We solve for a[j] (j in T) and mu_k (k in S) and lambda.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    a = _normalize(a)

    best_a = a.copy()
    best_sc = _score(a)

    tol_frac = 0.005
    min_tol_frac = 0.0005
    no_improve = 0

    for it in range(10000):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            tol_frac = max(min_tol_frac, tol_frac * 0.5)
            if tol_frac <= min_tol_frac * 1.1:
                break
            no_improve = 0

        # Active set
        threshold = c_max - tol_frac * c_max
        active = c >= threshold
        active_idx = np.where(active)[0]
        n_act = len(active_idx)

        if n_act == 0:
            break

        # Support
        support = a > 1e-10 * a.max()
        support_idx = np.where(support)[0]
        n_sup = len(support_idx)

        if n_sup < 2 or n_act < 1:
            break

        # Build the KKT system
        # Unknowns: a[j] for j in support (n_sup), mu_k for k in active (n_act), lambda (1)
        # Total: n_sup + n_act + 1
        # Equations:
        #   For each k in active (n_act eqs): sum_j a[j]*a[k-j] - M = 0
        #   For each j in support (n_sup eqs): 2*sum_k mu_k*a[k-j] - lambda = 0  
        #   sum_j a[j] = 1 (1 eq)
        #   Sum of mu_k = 1 (from dL/dM = 0) (1 eq)
        # Total equations: n_act + n_sup + 2
        # Total unknowns: n_sup + n_act + 2 (a[j], mu_k, lambda, M)
        
        # This is a large system. For efficiency, note that the first set of equations
        # is QUADRATIC in a. So we can't solve it directly as a linear system.
        # Instead, we use the approach: fix mu_k and lambda, solve for a.
        
        # From the second set: for each j in support:
        #   2*sum_{k in active} mu_k * a[k-j] = lambda
        # This is a linear system in a! Let's write it as:
        #   sum_k G[j,k_idx] * a[k-j] = lambda/2
        # where G[j, k_idx] = mu_k
        # 
        # But a[k-j] is only defined when 0 <= k-j < n.
        # So for each j in support, the equation involves a at positions k-j for k in active.
        # This gives a linear constraint on a.
        
        # Simpler approach: use the gradient condition directly.
        # The "flattening direction" da should satisfy:
        # conv(da, a)[k] = -(c[k] - M_target)/2 for k in active (approximately)
        # where M_target is the target flat value.
        
        # In frequency domain:
        N = _next_pow2(2 * n - 1)
        A = np.fft.rfft(a, N)
        
        # Target: make c[k] = M_target for k in active
        M_target = float(np.mean(c[active]))
        r = np.zeros(len(c))
        r[active] = c[active] - M_target
        
        R = np.fft.rfft(r, N)
        eps = 1e-8 * np.abs(A).max()
        mask = np.abs(2.0 * A) > eps
        dA = np.zeros_like(A)
        dA[mask] = -R[mask] / (2.0 * A[mask])
        
        a_new = np.fft.irfft(A + dA, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            tol_frac = max(min_tol_frac, tol_frac * 0.8)
            continue
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < current_sc - 1e-14:
            a = a_new
        else:
            # Damped step
            t = 0.5
            improved = False
            for _ in range(5):
                a_try = np.fft.irfft(A + t * dA, N)[:n]
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-14:
                    a = a_try
                    improved = True
                    break
                t *= 0.5
            if not improved:
                tol_frac = max(min_tol_frac, tol_frac * 0.9)

        if no_improve == 0:
            tol_frac = max(min_tol_frac, tol_frac * 0.97)

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0, verbose=True):
    """L-BFGS with adaptive beta schedule on softmax parametrization."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c0_max * 0.003:
        gap = c0_max * 0.003

    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
        0.00003125, 0.000015625, 0.0000078, 0.0000039,
        0.000002, 0.000001, 0.0000005
    ])
    betas = gap * multipliers

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        size_factor = max(1.0, n / 4096.0)
        est_iter_time = 0.002 * size_factor
        max_iter = min(8000, max(100, int(remaining / (est_iter_time * 20))))
        if beta < gap * 0.001:
            max_iter = min(max_iter, 2000)

        prob = _SoftProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-15, 'maxcor': 100, 'maxls': 60}
            )
            a_new = _softmax_logz(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"    LBFGS beta={beta:.2e}: score={best_sc:.10f}, nit={result.nit}")
            else:
                if verbose and i < 5:
                    print(f"    LBFGS beta={beta:.2e}: no improvement (best={best_sc:.10f})")
        except Exception as e:
            if verbose:
                print(f"    LBFGS beta={beta:.2e}: error {e}")

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-60.0 * (t - 0.2) ** 2) + np.exp(-60.0 * (t - 0.8) ** 2) + 0.005
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "one_sided":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t) + 0.005
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.6 * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    elif kind == "wide_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "double_hump":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(4 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _subgradient_method(a_init, deadline, min_time_remaining=30.0):
    """Subgradient method with Polyak step size for the max-flattening problem."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score(a)

    lb = 2.0 * n / (2 * n - 1)
    t = 0
    step_scale = 0.3

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-12:
            best_sc = current_sc
            best_a = a.copy()

        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        gap = current_sc - lb
        if gap < 1e-10:
            break
        grad_c = grad - np.mean(grad)
        g_norm_sq = float(np.dot(grad_c, grad_c))
        if g_norm_sq < 1e-20:
            break
        step = step_scale * gap / g_norm_sq
        step = min(step, 0.05)

        a_new = a - step * grad_c
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new
        t += 1

        if t % 200 == 0:
            step_scale *= 0.95

    return best_a, best_sc


_t_start_global = 0.0


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining frequency-domain Newton, L-BFGS,
    KKT refinement, and multi-n exploration.
    """
    global _t_start_global
    _t_start_global = time.monotonic()
    deadline = _t_start_global + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except NameError:
        pass
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(16384, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n=16384: score={best_score:.10f}")

    n_main = len(best_seq)

    # === Phase 1: Frequency-domain Newton (main workhorse) ===
    print(f"\n[P1] Freq-domain Newton (n={n_main}, t={time.monotonic()-_t_start_global:.0f}s)")
    p1_end = _t_start_global + (budget_s - 10) * 0.40
    a1, sc1 = _freq_domain_newton(best_seq, p1_end, min_time_remaining=(deadline - time.monotonic()), verbose=True)
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  P1 result: {best_score:.10f}")
    else:
        print(f"  P1: no improvement (best={best_score:.10f})")

    # === Phase 2: L-BFGS polish ===
    print(f"\n[P2] L-BFGS polish (t={time.monotonic()-_t_start_global:.0f}s)")
    p2_end = min(deadline - 300, time.monotonic() + 250)
    a2, sc2 = _run_lbfgs(best_seq, p2_end, min_time_remaining=200, verbose=True)
    if sc2 < best_score:
        best_score = sc2
        best_seq = a2.copy()
        print(f"  P2 result: {best_score:.10f}")

    # === Phase 3: KKT refinement ===
    print(f"\n[P3] KKT refinement (t={time.monotonic()-_t_start_global:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p3_end = min(deadline - 200, time.monotonic() + 200)
        a3, sc3 = _kkt_refinement(best_seq, p3_end, min_time_remaining=100, verbose=True)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3 result: {best_score:.10f}")

    # === Phase 4: Freq Newton again (tighter) ===
    print(f"\n[P4] Freq Newton round 2 (t={time.monotonic()-_t_start_global:.0f}s)")
    if time.monotonic() + 40 < deadline:
        p4_end = min(deadline - 150, time.monotonic() + 250)
        a4, sc4 = _freq_domain_newton(best_seq, p4_end, min_time_remaining=100, verbose=True)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4 result: {best_score:.10f}")

    # === Phase 5: Subgradient + L-BFGS ===
    print(f"\n[P5] Subgradient + LBFGS (t={time.monotonic()-_t_start_global:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p5_end = min(deadline - 100, time.monotonic() + 100)
        a5, sc5 = _subgradient_method(best_seq, p5_end, min_time_remaining=50)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
        if time.monotonic() + 20 < deadline:
            p5b_end = min(deadline - 80, time.monotonic() + 100)
            a5b, sc5b = _run_lbfgs(best_seq, p5b_end, min_time_remaining=60, verbose=False)
            if sc5b < best_score:
                best_score = sc5b
                best_seq = a5b.copy()

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n exploration (t={time.monotonic()-_t_start_global:.0f}s)")
    n_options = [8192, 12288, 20480, 32768]
    n_options = [x for x in n_options if abs(x - n_main) > n_main * 0.1]
    for n_try in n_options:
        if time.monotonic() + 60 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if sc_r < best_score:
            best_score = sc_r
            best_seq = a_r.copy()
        if time.monotonic() + 50 > deadline:
            continue
        # Quick freq Newton on resampled
        p6_end = min(deadline - 50, time.monotonic() + 80)
        a6, sc6 = _freq_domain_newton(a_r, p6_end, min_time_remaining=40, verbose=False)
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  n={n_try}: {best_score:.10f}")

    # === Phase 7: Diverse profiles ===
    print(f"\n[P7] Diverse profiles (t={time.monotonic()-_t_start_global:.0f}s)")
    profiles = ["wide_taper", "double_hump", "cubic_taper", "gaussian"]
    n_for_profiles = len(best_seq)
    for prof in profiles:
        if time.monotonic() + 40 > deadline:
            break
        a_p = _make_profile(n_for_profiles, prof)
        sc_p = _score(a_p)
        if sc_p < best_score + 0.3:
            p7_end = min(deadline - 30, time.monotonic() + 60)
            a_p_opt, sc_p_opt = _freq_domain_newton(a_p, p7_end, min_time_remaining=20, verbose=False)
            if sc_p_opt < best_score:
                best_score = sc_p_opt
                best_seq = a_p_opt.copy()
                print(f"  {prof}: {best_score:.10f}")

    # === Phase 8: Final polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-_t_start_global:.0f}s)")
    if time.monotonic() + 15 < deadline:
        p8_end = min(deadline - 5, time.monotonic() + 120)
        a8, sc8 = _freq_domain_newton(best_seq, p8_end, min_time_remaining=5, verbose=False)
        if sc8 < best_score:
            best_score = sc8
            best_seq = a8.copy()
        if time.monotonic() + 10 < deadline:
            p8b_end = min(deadline - 3, time.monotonic() + 60)
            a8b, sc8b = _run_lbfgs(best_seq, p8b_end, min_time_remaining=5, verbose=False)
            if sc8b < best_score:
                best_score = sc8b
                best_seq = a8b.copy()

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-_t_start_global:.1f}s")
    
    # Verify with evaluate_sequence
    try:
        eval_score = evaluate_sequence([float(x) for x in final])
        print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"Warning: evaluate_sequence failed: {e}")

    return [float(x) for x in final]
