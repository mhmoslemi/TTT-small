    return np.maximum(x - theta, 0.0)


def _conv_grad_multi(a, active_k):
    """Gradient of sum_{k in active_k} (a*a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_k:
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            idx = k - np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi + 1] += a[idx]
        j_lo2 = max(0, k)
        j_hi2 = min(n - 1, k + n - 1)
        if j_lo2 <= j_hi2:
            idx = np.arange(j_lo2, j_hi2 + 1) - k
            grad[j_lo2:j_hi2 + 1] += a[idx]
    return grad


def _smooth_obj_grad(z, beta, n):
    """Log-sum-exp smoothed objective and gradient in softmax parameterization."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        w = e / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return obj, grad


class _SmoothProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        obj, _ = _smooth_obj_grad(z, self.beta, self.n)
        return obj

    def grad(self, z):
        _, g = _smooth_obj_grad(z, self.beta, self.n)
        return g


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n, endpoint=False)
        a = np.sin(np.pi * (t + 0.5 / n)) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exponential_taper":
        t = np.linspace(0, 1, n, endpoint=False)
        a = np.exp(-2.0 * (2 * t - 1) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _peel_step(a, n_peels=30, move_frac=0.002, top_k_frac=0.01):
    """Greedy peeling: reduce worst peaks by moving mass from contributors to non-contributors."""
    a = a.copy()
    n = len(a)
    K = max(3, int(n * top_k_frac))
    
    for _ in range(n_peels):
        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))
        
        contrib_k = _conv_grad_multi(a, [k_star])
        weighted = contrib_k * a
        j_star = int(np.argmax(weighted))
        
        top_k = np.argsort(c)[::-1][:K]
        total_contrib = _conv_grad_multi(a, top_k)
        # Prefer moving to low-contribution, low-value positions
        j_prime = int(np.argmin(total_contrib + 0.01 * a))
        
        if j_star == j_prime or a[j_star] <= 1e-15:
            continue
        
        delta = move_frac * a[j_star]
        a[j_star] -= delta
        a[j_prime] += delta
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
    
    return a


def _nesterov_subgrad(a_init, deadline, min_time_remaining=30.0,
                       K_init=10, K_max=200, seed=0,
                       base_lr=0.008, momentum=0.8):
    """Nesterov-accelerated subgradient with backtracking line search."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = float(2.0 * n * c0.max())
    
    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    lr = base_lr
    K = K_init
    t = 0
    no_improve = 0
    stall_count = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive K
        if t % 50 == 0:
            c_sorted = np.sort(c)[::-1]
            gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
            if gap < c_max * 0.01 and K < K_max:
                K = min(K_max, int(K * 1.3) + 1)
        
        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]
        grad = _conv_grad_multi(a, active)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        
        if gnorm < 1e-18:
            break
        
        # Backtracking line search
        step = lr / (gnorm + 1e-18)
        improved = False
        for _bt in range(10):
            v = momentum * v - step * grad_t
            a_cand = _project_simplex(a + v)
            c_new = _full_convolve(a_cand, a_cand)
            sc_new = 2.0 * n * float(c_new.max())
            if sc_new < current_sc + 1e-15:
                a = a_cand
                current_sc = sc_new
                improved = True
                break
            step *= 0.5
        
        if not improved:
            v = 0.5 * v
            a = _project_simplex(a - 0.001 * lr * grad_t / (gnorm + 1e-18))
            a = a / a.sum()
        
        # Periodic peeling restart
        if no_improve >= 150 and no_improve % 150 == 0:
            stall_count += 1
            if stall_count > 8:
                break
            a_peel = _peel_step(a, n_peels=20, move_frac=0.001)
            sc_peel = 2.0 * n * float(_full_convolve(a_peel, a_peel).max())
            if sc_peel < best_sc:
                a = a_peel
                best_a = a.copy()
                best_sc = sc_peel
            v = np.zeros(n)
            no_improve = 0
        
        if t % 200 == 0:
            print(f"    Nesterov t={t}: best={best_sc:.10f}, K={K}, stall={stall_count}")
    
    return best_a, best_sc


def _run_lbfgs_schedule(a_init, deadline, min_time_remaining=30.0, max_phases=12):
    """L-BFGS with geometric beta reduction and adaptive iterations."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = max_c - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.005:
        gap = max_c * 0.005
    beta_start = max(gap, max_c * 0.02)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    beta = beta_start
    phase = 0
    
    while phase < max_phases and time.monotonic() + min_time_remaining < deadline:
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Estimate iterations based on time
        est_time_per_iter = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(30, int(remaining * 0.6 / (est_time_per_iter * 5))))
        
        prob = _SmoothProblem(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-14,
                    'maxcor': 60,
                    'maxls': 40,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-12:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS phase {phase}: score={current_sc:.10f}, beta={beta:.6f}")
            else:
                # No improvement, try smaller beta anyway
                pass
        except Exception:
            pass
        
        beta *= 0.45
        phase += 1
    
    return current_a


def _run_lbfgs_fine_beta(a_init, deadline, min_time_remaining=20.0):
    """Very fine L-BFGS for final polish."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = max_c - c_sorted[min(10, len(c_sorted) - 1)]
    beta_start = max(gap * 0.5, max_c * 0.001)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    betas = [beta_start, beta_start * 0.4, beta_start * 0.16, beta_start * 0.064,
             beta_start * 0.025, beta_start * 0.01, beta_start * 0.004,
             beta_start * 0.0016]
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        
        est_time_per_iter = 0.004 * max(1.0, n / 8192.0)
        max_iter = min(3000, max(20, int(remaining * 0.7 / (est_time_per_iter * 8))))
        
        prob = _SmoothProblem(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-17,
                    'gtol': 1e-15,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-13:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    Fine L-BFGS beta_idx={i}: score={current_sc:.10f}")
        except Exception:
            pass
    
    return current_a


def _multi_start_optimize(a_init, n, deadline, budget_per_start=60.0, n_starts=3):
    """Try multiple perturbations and keep the best result."""
    best_a = a_init.copy()
    best_sc = _score(a_init)
    rng = np.random.default_rng(12345)
    
    for i in range(n_starts):
        if time.monotonic() + budget_per_start > deadline:
            break
        # Add small random perturbation
        pert = 0.002 * rng.standard_normal(n)
        a_try = _project_simplex(a_init + pert)
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        base_a = a_try if sc_try < best_sc else a_init.copy()
        
        sub_deadline = min(deadline - 30, time.monotonic() + budget_per_start)
        a_opt, sc_opt = _nesterov_subgrad(base_a, sub_deadline, min_time_remaining=10,
                                           K_init=5, K_max=100, seed=100 + i)
        if sc_opt < best_sc:
            best_a = a_opt.copy()
            best_sc = sc_opt
    
    return best_a, best_sc
