"""
Optimized minimizer for 2*n*max(conv(a,a))/sum(a)^2 using:
- FFT-based convolution for speed
- Log-sum-exp smoothing with aggressive annealing
- L-BFGS with multiple beta scales
- Nesterov accelerated gradient for fine-tuning
- Structured Fourier-mode perturbations for multi-restart
- Active-set peak equalization polish
- Multi-n exploration
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0, max_iter_override=None):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003) * beta_scale

    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625,
        0.03125, 0.015625, 0.0078125, 0.00390625,
        0.002, 0.001, 0.0005, 0.0002, 0.0001,
        0.00005, 0.00002, 0.00001, 0.000005
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        if max_iter_override is not None:
            max_iter = max_iter_override
        elif beta < base_beta * 0.01:
            est_time_per_iter = 0.003 * max(1.0, n / 16384.0)
            max_iter = min(8000, max(100, int(remaining / (est_time_per_iter * 10))))
        else:
            est_time_per_iter = 0.002 * max(1.0, n / 16384.0)
            max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 8))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 60,
                    'iprint': -1,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _nesterov_polish(a_init, deadline, min_time_remaining=20.0, max_iter=5000):
    """Nesterov accelerated gradient on the log-sum-exp objective."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    c = _full_convolve(a, a)
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    beta = max(gap * 0.01, c[0] * 0.0005)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    m = 1.0
    z_old = z.copy()
    lr = 0.5
    patience = 0

    for it in range(max_iter):
        if time.monotonic() + min_time_remaining > deadline:
            break

        z_new = z + m * (z - z_old)
        g = _smograd(z_new, beta, n)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        z_try = z_new - lr * g / (gnorm + 1e-15)
        a_try = _softmax(z_try)
        sc_try = _score(a_try)

        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try.copy()
            patience = 0
        else:
            patience += 1
            if patience > 200:
                break

        m_old = m
        m = (1 + np.sqrt(1 + 4 * m_old * m_old)) / 2
        z_old = z.copy()
        z = z_try
        lr = np.min(lr * 1.001, 2.0 / (it + 10))

    return best_a, best_sc


def _active_set_equalize(a_init, deadline, min_time_remaining=20.0):
    """Iteratively equalize top peaks using analytical gradient structure."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    K = 5
    max_K = min(120, n)
    lr = 0.03
    no_improve = 0
    a_rev_base = a[::-1]

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            a_rev_base = a[::-1]
            if K < max_K:
                K = min(max_K, K + 2)

        top_k_idx = np.argsort(c)[::-1][:K]
        w_S = np.zeros(2 * n - 1)
        w_S[top_k_idx] = 1.0 / K
        h = _full_convolve(w_S, a_rev_base)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        alpha = lr / (gnorm + 1e-15)
        improved = False
        for _ in range(40):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                a_rev_base = a[::-1]
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            lr *= 0.9
            if no_improve % 30 == 0:
                K = max(5, K // 2)
            if no_improve >= 300:
                a = best_a.copy()
                lr = 0.03
                K = 5
                no_improve = 0
                if no_improve > 600:
                    break
        else:
            lr = min(0.08, lr * 1.01)

    return best_a, best_sc


def _fourier_perturb(a, rng, num_modes=5, max_mode=50):
    """Add structured Fourier perturbations to break symmetry."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    modes = rng.choice(range(1, max_mode + 1), size=min(num_modes, max_mode), replace=False)
    for m in modes:
        amp = rng.uniform(0.01, 0.08)
        phase = rng.uniform(0, 2 * np.pi)
        t = np.arange(n)
        z += amp * np.cos(2 * np.pi * m * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.sin(2.0 * np.pi * 5.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _hessian_vec_prod(w, v):
    """Compute H_f @ v where (H_f)_{ij} = 8n * w_{i+j} (Hankel matrix)."""
    n = len(v)
    w_rev = w[::-1]
    conv = _full_convolve(w_rev, v)
    # (H_f v)_i = 8n * (w_rev * v)_{2n-2-i}
    # conv has length (2n-1) + n - 1 = 3n-2
    # We need indices 2n-2, 2n-3, ..., n-1 (for i=0, 1, ..., n-1)
    start = 2 * n - 2
    result = 8.0 * n * conv[start - n + 1:start + 1][::-1]
    return result


def _newton_refine(a_init, deadline, min_time_remaining=30.0):
    """Newton-type refinement using structure-exploited Hessian."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    # Use small beta for near-exact max
    c = _full_convolve(a, a)
    c_max = float(c.max())
    beta = c_max * 0.001

    for newton_iter in range(100):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()

        # Compute weights
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        # Gradient (projected to simplex tangent)
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * (h - W)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Newton direction via a few CG steps on H_f
        # H_f is Hankel: (H_f)_{ij} = 8n * w_{i+j}
        # We solve (H_f + mu*I) d = -grad
        mu = gnorm * 0.1  # damping
        d = np.zeros(n)
        r = -grad
        p = r.copy()
        rr = float(np.dot(r, r))
        if rr < 1e-30:
            break

        for cg_iter in range(20):
            Hp = _hessian_vec_prod(w, p)
            Hpp = float(np.dot(p, Hp))
            if Hpp < 1e-20:
                Hpp = 1e-20
            alpha_cg = rr / Hpp
            d += alpha_cg * p
            r_new = r - alpha_cg * (Hp + mu * p)
            rr_new = float(np.dot(r_new, r_new))
            if rr_new < 1e-28:
                r = r_new
                break
            p = r_new + (rr_new / rr) * p
            r = r_new
            rr = rr_new

        # Project direction to simplex tangent
        d = d - np.mean(d)
        # Line search
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-15:
            break
        step = 1.0 / (dnorm + 1e-10)
        improved = False
        for _ in range(30):
            a_try = a - step * d * 0.1
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                step *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            step *= 0.7

        if not improved:
            # Fall back to gradient step
            step_g = 0.01 / (gnorm + 1e-10)
            a_try = a - step_g * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc:
                    a = a_try

    return best_a, best_sc

            best_seq = a_eq.copy()
            best_score = sc_eq
            print(f"  NEW BEST: {best_score:.10f} [equalize]")
        else:
            print(f"  No improvement (score={sc_eq:.10f})")

    # === Phase 5: Nesterov polish ===
    print(f"\n[P5] Nesterov polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        nst_deadline = min(deadline - 150, time.monotonic() + 100)
        a_nst, sc_nst = _nesterov_polish(best_seq, nst_deadline, min_time_remaining=50.0)
        if sc_nst < best_score:
            best_seq = a_nst.copy()
            best_score = sc_nst
            print(f"  NEW BEST: {best_score:.10f} [nesterov]")

    # === Phase 6: Fourier perturbation restarts ===
    print(f"\n[P6] Fourier restarts (t={time.monotonic()-t_start:.1f}s)")
    n = len(best_seq)
    num_restarts = 3
    for r in range(num_restarts):
        if time.monotonic() > deadline - 150:
            break
        a_pert = _fourier_perturb(best_seq, rng, num_modes=4 + r, max_mode=30 + 10 * r)
        a_opt = _run_lbfgs(a_pert, deadline, min_time_remaining=150.0, beta_scale=0.5,
                           max_iter_override=2000)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [restart_{r}]")
        else:
            print(f"  Restart {r}: score={sc_opt:.10f} (no improvement)")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([n, 16384, 24576, 8192]), reverse=True)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=80.0, beta_scale=0.4,
                           max_iter_override=3000)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n]")
        else:
            print(f"  n={n_try}: score={sc_opt:.10f} (no improvement, init={sc_init:.10f})")

    # === Phase 8: Final ultra-fine L-BFGS ===
    print(f"\n[P8] Final ultra-fine (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 10 - time.monotonic()
    if remaining > 20:
        a_final = _run_lbfgs(best_seq, deadline, min_time_remaining=10.0, beta_scale=0.1)
        sc_final = _score(a_final)
        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final
            print(f"  NEW BEST: {best_score:.10f} [final_ultra]")

        # Final active set
        if time.monotonic() < deadline - 5:
            a_fin, sc_fin = _active_set_equalize(best_seq, deadline - 3, min_time_remaining=3.0)
            if sc_fin < best_score:
                best_seq = a_fin.copy()
                best_score = sc_fin
                print(f"  NEW BEST: {best_score:.10f} [final_eq]")

        # Final Newton
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _newton_refine(best_seq, deadline - 2, min_time_remaining=2.0)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_newton]")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
