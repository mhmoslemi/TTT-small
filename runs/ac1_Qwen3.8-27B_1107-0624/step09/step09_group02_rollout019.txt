def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    w = np.zeros_like(c_local)
                    w[int(np.argmax(c_local))] = 1.0
                else:
                    w = e / s_e
                a_rev = a_local[::-1]
                wc = _full_convolve(w, a_rev)
                h = wc[n - 1:2 * n - 1]
                W_val = float(np.dot(w, c_local))
                return 4.0 * n * a_local * (h - W_val)

            result = minimize(
                obj_func, z,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                if i > 10 and time.monotonic() + min_time_remaining + 30 > deadline:
                    break
        except Exception:
            pass

    return best_a


def _run_active_set_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """Active-set QP with L-BFGS inner solver and continuation on tolerance."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    tol_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                    0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
                    0.000001, 0.0000005, 0.0000002, 0.0000001]
    tol_idx = 0
    rounds_per_tol = 3
    total_rounds = 0
    max_rounds = 300

    while time.monotonic() + min_time_remaining < deadline and total_rounds < max_rounds:
        total_rounds += 1
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]

        c = _full_convolve(a, a)
        c_max = float(c.max())
        active = c >= (c_max - tol * c_max)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        n_active = int(active.sum())
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / n_active

        # Solve QP: min sum_k p_k * c_k s.t. a in simplex
        # Using L-BFGS in log-space
        z = np.log(np.maximum(a, 1e-30))
        z = z - z.mean()

        try:
            def qp_obj(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                return float(np.dot(p[:len(c_local)], c_local))

            def qp_grad(z_local):
                a_local = _softmax(z_local)
                grad = _hvp_pv(a_local, p, n)
                return grad

            inner_deadline = deadline - min_time_remaining - (deadline - time.monotonic()) * 0.1
            if time.monotonic() + 10 > inner_deadline:
                inner_deadline = deadline - min_time_remaining
            remaining_inner = inner_deadline - time.monotonic()
            if remaining_inner < 2:
                break
            max_iter_inner = min(2000, max(50, int(remaining_inner / 0.01)))

            result = minimize(
                qp_obj, z,
                jac=qp_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter_inner, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 40, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new
            else:
                a = best_a.copy()

        except Exception:
            pass

        if (total_rounds) % rounds_per_tol == 0:
            tol_idx += 1

    return best_a, best_sc


def _run_nesterov_subgrad(a_init, deadline, min_time_remaining=30.0):
    """Nesterov-accelerated projected subgradient for the max objective."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    y = a.copy()
    momentum = 0.0
    beta_m = 0.9
    lr = 0.01

    for step in range(10000):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(y, y)
        c_max = float(c.max())
        # Active set: top lags
        k_active = min(20, len(c))
        top_k = np.argsort(c)[::-1][:k_active]
        p = np.zeros(2 * n - 1)
        p[top_k] = 1.0 / k_active

        grad = _hvp_pv(y, p, n)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            break

        # Nesterov momentum
        momentum = beta_m * momentum
        y_new = y - lr * (grad + momentum * (y - a))
        a_old = a.copy()
        a = _proj_simplex(y_new)
        y = a - momentum * (a - a_old)

        # Adaptive step size
        if step > 0 and step % 50 == 0:
            c_check = _full_convolve(a, a)
            sc_check = _score(a)
            if sc_check < best_sc - 1e-14:
                best_sc = sc_check
                best_a = a.copy()
            # Decay learning rate
            lr = lr * 0.95
            beta_m = min(0.99, beta_m + 0.001)

    a_final = _proj_simplex(y)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _run_multi_active_refine(a_init, deadline, min_time_remaining=30.0):
    """Multi-scale active set refinement: start with broad active set, tighten."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    # Progressive tightening with L-BFGS on the smoothed problem
    c = _full_convolve(a, a)
    c_max = float(c.max())
    base_beta = max(c_max * 0.01, c_max * 1e-8)

    beta_scales = [1.0, 0.1, 0.01, 0.001, 0.0001, 0.00001, 0.000001,
                   0.0000001, 0.00000001, 0.000000001]

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    for scale in beta_scales:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        beta = max(base_beta * scale, c_max * 1e-15)

        try:
            def refine_obj(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    return 2.0 * n * c_max_local
                return 2.0 * n * (c_max_local + beta * np.log(s_e))

            def refine_grad(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    w = np.zeros_like(c_local)
                    w[int(np.argmax(c_local))] = 1.0
                else:
                    w = e / s_e
                a_rev = a_local[::-1]
                wc = _full_convolve(w, a_rev)
                h = wc[n - 1:2 * n - 1]
                W_val = float(np.dot(w, c_local))
                return 4.0 * n * a_local * (h - W_val)

            max_iter = min(3000, max(50, int(remaining / 0.02)))
            result = minimize(
                refine_obj, z,
                jac=refine_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 50, 'maxls': 150}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-16:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining aggressive L-BFGS with log-sum-exp,
    active-set QP with L-BFGS inner solver, Nesterov subgradient, and
    multi-n exploration. Focuses on tight polishing of the best candidate.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 32768
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    n = n_parent
    print(f"\n{'='*60}")
    print(f"Starting optimization with n={n}, current best={best_score:.10f}")
    print(f"{'='*60}\n")

    # === Phase 1: Aggressive L-BFGS from parent (400s) ===
    print(f"[Phase 1] Aggressive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    phase1_deadline = t_start + 400
    a1 = _run_lbfgs_smoothed(parent, phase1_deadline, min_time_remaining=30.0,
                             beta_scale_min=1e-12)
    update_best(a1, _score(a1), "lbfgs_parent_aggr")

    # === Phase 2: Multi-active refine (300s) ===
    print(f"[Phase 2] Multi-active refine (t={time.monotonic() - t_start:.1f}s)")
    phase2_deadline = min(deadline - 700, phase1_deadline + 300)
    if best_seq is not None and time.monotonic() < phase2_deadline:
        a2, sc2 = _run_multi_active_refine(best_seq, phase2_deadline,
                                           min_time_remaining=30.0)
        update_best(a2, sc2, "multi_active_1")

    # === Phase 3: Active-set QP with L-BFGS (300s) ===
    print(f"[Phase 3] Active-set QP (t={time.monotonic() - t_start:.1f}s)")
    phase3_deadline = min(deadline - 500, phase2_deadline + 300)
    if best_seq is not None and time.monotonic() < phase3_deadline:
        a3, sc3 = _run_active_set_lbfgs(best_seq, phase3_deadline,
                                         min_time_remaining=30.0)
        update_best(a3, sc3, "active_qp_1")

    # === Phase 4: Another round of aggressive L-BFGS (200s) ===
    print(f"[Phase 4] L-BFGS round 2 (t={time.monotonic() - t_start:.1f}s)")
    phase4_deadline = min(deadline - 350, phase3_deadline + 200)
    if best_seq is not None and time.monotonic() < phase4_deadline:
        a4 = _run_lbfgs_smoothed(best_seq, phase4_deadline, min_time_remaining=30.0,
                                 beta_scale_min=1e-13)
        update_best(a4, _score(a4), "lbfgs_r2")

    # === Phase 5: Nesterov subgradient (100s) ===
    print(f"[Phase 5] Nesterov subgradient (t={time.monotonic() - t_start:.1f}s)")
    phase5_deadline = min(deadline - 250, phase4_deadline + 100)
    if best_seq is not None and time.monotonic() < phase5_deadline:
        a5, sc5 = _run_nesterov_subgrad(best_seq, phase5_deadline,
                                         min_time_remaining=20.0)
        update_best(a5, sc5, "nesterov_1")

    # === Phase 6: Active-set QP round 2 (150s) ===
    print(f"[Phase 6] Active-set QP round 2 (t={time.monotonic() - t_start:.1f}s)")
    phase6_deadline = min(deadline - 150, phase5_deadline + 150)
    if best_seq is not None and time.monotonic() < phase6_deadline:
        a6, sc6 = _run_active_set_lbfgs(best_seq, phase6_deadline,
                                         min_time_remaining=20.0)
        update_best(a6, sc6, "active_qp_2")

    # === Phase 7: Multi-n exploration (150s) ===
    print(f"[Phase 7] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    phase7_deadline = min(deadline - 100, phase6_deadline + 150)
    n_tries = [16384, 65536, 49152]
    for n_try in n_tries:
        if time.monotonic() > phase7_deadline - 30:
            break
        if n_try == n_parent:
            continue
        a_init_n = _resample(best_seq, n_try)
        a_n = _run_lbfgs_smoothed(a_init_n, phase7_deadline,
                                   min_time_remaining=20.0, beta_scale_min=1e-10)
        sc_n = _score(a_n)
        update_best(a_n, sc_n, f"n_{n_try}")

    # === Phase 8: Final ultra-tight polish (remaining time) ===
    print(f"[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight L-BFGS
        remaining = deadline - 5 - time.monotonic()
        if remaining > 10:
            a_fin = _run_lbfgs_smoothed(best_seq, deadline - 5,
                                         min_time_remaining=3.0, beta_scale_min=1e-14)
            update_best(a_fin, _score(a_fin), "final_lbfgs")

        # Final active set
        if time.monotonic() < deadline - 5:
            a_fin2, sc_fin2 = _run_active_set_lbfgs(best_seq, deadline - 3,
                                                     min_time_remaining=2.0)
            update_best(a_fin2, sc_fin2, "final_qp")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
