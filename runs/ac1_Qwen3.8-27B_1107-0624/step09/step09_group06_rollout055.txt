"""
Corrected-gradient L-BFGS optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key fix: correct gradient computation using FFT cross-correlation.
Uses adaptive beta smooth-max schedule with multiple restarts.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smooth_max_value_and_weights(c, beta):
    """Compute smooth max and weights for log-sum-exp."""
    c_shifted = (c - c.max()) / beta
    e = np.exp(np.clip(c_shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return c.max(), w
    w = e / s
    return c.max() + beta * np.log(s), w

def _smooth_score_and_grad(a, beta):
    """Compute smooth score and its gradient w.r.t. a.
    
    Score = 2n * S_max(a) where S_max is the smooth max of conv(a,a).
    Gradient uses FFT cross-correlation for efficiency.
    """
    n = len(a)
    c = _full_convolve(a, a)
    smax, w = _smooth_max_value_and_weights(c, beta)
    score = 2.0 * n * smax
    
    # Gradient: 2n * 2 * (w ⊛ a)[:n] = 4n * (w ⊛ a)[:n]
    # Cross-correlation via FFT: (w ⊛ a)[d] = ifft(conj(FFT(w)) * FFT(a))[d]
    # We need d = 0, ..., n-1
    # Required FFT length: at least len(w) + len(a) - 1 = (2n-1) + n - 1 = 3n-2
    N = _next_pow2(3 * n - 1)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(np.conj(W) * A, N)
    h = corr[:n]  # (w ⊛ a)[0], ..., (w ⊛ a)[n-1]
    
    grad = 4.0 * n * h
    return score, grad

def _smooth_score_and_grad_z(z, beta):
    """Score and gradient w.r.t. z (softmax parameterization).
    
    a = softmax(z), score = 2n * beta * log(sum exp(c_k/beta))
    grad_z = (I - a a^T) @ grad_a
    """
    n = len(z)
    a = _softmax(z)
    c = _full_convolve(a, a)
    smax, w = _smooth_max_value_and_weights(c, beta)
    score = 2.0 * n * smax
    
    N = _next_pow2(3 * n - 1)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(np.conj(W) * A, N)
    h = corr[:n]
    
    grad_a = 4.0 * n * h
    # Jacobian of softmax: grad_z = (I - a a^T) @ grad_a = grad_a - a * (a @ grad_a)
    grad_z = grad_a - a * np.dot(a, grad_a)
    return score, grad_z

class _SmoothProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    
    def __call__(self, z):
        return _smooth_score_and_grad_z(z, self.beta)[0]
    
    def grad(self, z):
        return _smooth_score_and_grad_z(z, self.beta)[1]

def _run_lbfgs(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with adaptive beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    # Use a gap-based initial beta
    num_top = min(50, len(c_sorted))
    gap = max_c - c_sorted[num_top - 1]
    if gap < 1e-15:
        gap = max_c * 0.01
    base_beta = max(gap * 0.1, max_c * 0.001)
    
    # Beta schedule: geometric reduction
    n_betas = 12
    ratio = 0.5
    betas = [base_beta * (ratio ** i) for i in range(n_betas)]
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        
        # Scale max_iter based on remaining time and problem size
        est_time_per_iter = 0.008 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 15))))

        try:

                prob, z, jac=prob.grad, method='L-BFGS-B',

                    'ftol': 1e-16,

                    'maxcor': 100,
                    'maxls': 40,

            a_new = _softmax(result.x)

            if sc_new < current_sc - 1e-14:

                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * t / 2) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_center":
        a = np.ones(n)
        center = n // 2
        width = n // 4
        a[max(0, center-width):center+width] = 0.5
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
