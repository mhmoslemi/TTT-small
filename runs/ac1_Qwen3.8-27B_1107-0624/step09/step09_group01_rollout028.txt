    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve_hankel(W, n, rhs, max_iter=300, tol=1e-12):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.05:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _compute_kkt_gradient(a, W, n):
    a_rev = a[::-1]
    wc = _full_convolve(W, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(W, _full_convolve(a, a)))
    return 2.0 * a * (h - W_val)


def _kkt_newton(a_init, deadline, max_rounds=200, k_schedule=None):
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    if k_schedule is None:
        k_schedule = [3, 5, 10, 20, 50, 100, 200, 500, 1000, 2000]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        c_mean = c.sum() / len(c)

        K = k_schedule[min(rnd, len(k_schedule) - 1)]
        K = min(K, 2 * n - 1)

        # Probability-weighted active set using softmax
        idx = np.argpartition(c, -K)[-K:]
        c_top = c[idx]
        c_top_max = float(c_top.max())
        spread = max(c_top_max - float(c_top.min()), 1e-15)
        beta_w = spread * 0.3
        w_top = np.exp((c_top - c_top_max) / beta_w)
        w_top = w_top / w_top.sum()

        W = np.zeros(2 * n - 1)
        W[idx] = w_top

        g = _compute_kkt_gradient(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        Hg = _cg_solve_hankel(W, n, g, max_iter=200)
        H1 = _cg_solve_hankel(W, n, np.ones(n), max_iter=200)
        H1_denom = float(np.ones(n) @ H1)
        if abs(H1_denom) < 1e-25 or not np.isfinite(H1_denom):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / H1_denom
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        alpha = 1.0
        step_done = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1.0 - 0.003 * alpha) - 1e-18:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not step_done:
            for trial in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-18:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 12:
            break

    return best_a, best_sc


def _smooth_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smooth_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _lbfgs_round(a_init, deadline, min_remaining=30.0, max_passes=10):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    c_max = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    beta = max(gap * 0.3, c_max * 1e-5)
    beta = min(beta, c_max * 0.05)

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    current_a = a_init.copy()
    current_sc = _score(current_a)

    for pass_num in range(max_passes):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 15:
            break

        scale = max(1.0, n / 16384.0)
        est_per_iter = 0.003 * scale
        max_iter = min(5000, max(100, int(remaining / (est_per_iter * 25))))

        obj_p = functools.partial(_smooth_obj, beta=beta, n=n)
        grad_p = functools.partial(_smooth_grad, beta=beta, n=n)

        try:
            result = minimize(
                obj_p, z, jac=grad_p,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 150}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS pass {pass_num}: score={sc_new:.10f}, beta={beta:.3e}")
            else:
                print(f"    L-BFGS pass {pass_num}: no improvement (sc={sc_new:.10f})")
        except Exception as e:
            print(f"    L-BFGS pass {pass_num} error: {e}")

        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(10, len(c_sorted_new) - 1)]
        new_ratio = new_gap / max(c_max_new, 1e-30)

        if new_ratio < 1e-9:
            beta *= 0.1
        elif new_ratio < 1e-6:
            beta *= 0.2
        elif new_ratio < 1e-4:
            beta *= 0.35
        else:
            beta *= 0.5

        beta = min(beta, c_max_new * 1e-6)
        if beta < c_max_new * 1e-12:
            break

    return current_a


def _subgrad_sbm(a_init, deadline, max_steps=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.015
    momentum = 0.85
    v = np.zeros(n)

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set with probability weighting
        tol_frac = 0.001
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() < 5:
            idx = np.argpartition(c, -10)[-10:]
            active = np.zeros_like(c)
            active[idx] = True
        K_active = int(active.sum())
        c_active = c[active]
        beta_w = max(float(c_active.max() - c_active.min()), 1e-15) * 0.3
        w_act = np.exp((c_active - c_active.max()) / beta_w)
        w_act = w_act / w_act.sum()
        W = np.zeros_like(c)
        W[active] = w_act

        a_rev = a[::-1]
        wc = _full_convolve(W, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(W, c))
        grad = 4.0 * a * (h - W_val)
        grad = grad - np.mean(grad)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break

        # Sign-based momentum (SBM)
        v = momentum * v + np.sign(grad) * lr / max(grad_norm, 1e-30)
        z = z + v
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 99:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, 0.04)
            else:
                patience += 1
                lr *= 0.85
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)
                    patience = 0
                    lr = 0.01

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.02
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                par_sc = _score(parent)
                print(f"Parent: n={len(parent)}, score={par_sc:.10f}")
                _update(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        parent = _make_profile(16384, "tapered")
        print(f"No parent, using tapered n=16384")
        _update(parent, _score(parent), "init")

    n0 = len(parent)
    print(f"\nStarting optimization with n={n0}, score={best_score:.10f}")

    # Phase 1: KKT-Newton (coarse)
    print(f"\n[Phase 1] KKT-Newton coarse (t={time.monotonic() - t_start:.1f}s)")
    t_p1_end = min(deadline - 900, t_start + 180)
    k_sch1 = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]
    a1, sc1 = _kkt_newton(best_seq, t_p1_end, max_rounds=300, k_schedule=k_sch1)
    _update(a1, sc1, "kkt_coarse")
    print(f"  After KKT coarse: {best_score:.10f}")

    # Phase 2: L-BFGS round 1 (main workhorse)
    print(f"\n[Phase 2] L-BFGS round 1 (t={time.monotonic() - t_start:.1f}s)")
    t_p2_end = min(deadline - 600, t_start + 500)
    a2 = _lbfgs_round(best_seq, t_p2_end, min_remaining=30.0, max_passes=15)
    _update(a2, _score(a2), "lbfgs_1")
    print(f"  After L-BFGS 1: {best_score:.10f}")

    # Phase 3: KKT-Newton refinement
    print(f"\n[Phase 3] KKT-Newton fine (t={time.monotonic() - t_start:.1f}s)")
    t_p3_end = min(deadline - 400, t_start + 700)
    k_sch2 = [1, 2, 3, 5, 10, 20, 50, 100, 200]
    a3, sc3 = _kkt_newton(best_seq, t_p3_end, max_rounds=200, k_schedule=k_sch2)
    _update(a3, sc3, "kkt_fine")
    print(f"  After KKT fine: {best_score:.10f}")

    # Phase 4: L-BFGS round 2
    print(f"\n[Phase 4] L-BFGS round 2 (t={time.monotonic() - t_start:.1f}s)")
    t_p4_end = min(deadline - 250, t_start + 1100)
    a4 = _lbfgs_round(best_seq, t_p4_end, min_remaining=30.0, max_passes=15)
    _update(a4, _score(a4), "lbfgs_2")
    print(f"  After L-BFGS 2: {best_score:.10f}")

    # Phase 5: Subgradient with SBM
    print(f"\n[Phase 5] Subgrad SBM (t={time.monotonic() - t_start:.1f}s)")
    t_p5_end = min(deadline - 150, t_start + 1300)
    a5, sc5 = _subgrad_sbm(best_seq, t_p5_end, max_steps=3000)
    _update(a5, sc5, "subgrad")
    print(f"  After subgrad: {best_score:.10f}")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_options = sorted(set([n0, n0 * 2, 16384, 32768, 65536]))
    n_options = [x for x in n_options if 2048 <= x <= 65536]
    for n_try in n_options:
        if time.monotonic() > deadline - 80:
            break
        if n_try == len(best_seq):
            continue
        a_res = _resample(best_seq, n_try)
        sc_res = _score(a_res)
        if sc_res < best_score:
            _update(a_res, sc_res, f"resample_n{n_try}")
        # Quick L-BFGS on resampled
        if time.monotonic() < deadline - 40 and n_try <= 32768:
            a_opt = _lbfgs_round(a_res, deadline - 10, min_remaining=5.0, max_passes=5)
            _update(a_opt, _score(a_opt), f"opt_n{n_try}")

    # Phase 7: Final KKT polish
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        k_final = [1, 2, 3, 5, 8, 10, 20, 50]
        a_fin, sc_fin = _kkt_newton(best_seq, deadline - 1, max_rounds=100,
                                     k_schedule=k_final)
        _update(a_fin, sc_fin, "final_kkt")

        # One more L-BFGS micro-pass
        if time.monotonic() < deadline - 2:
            a_micro = _lbfgs_round(best_seq, deadline, min_remaining=0.5, max_passes=3)
            _update(a_micro, _score(a_micro), "micro_lbfgs")

    # Phase 8: Restart with perturbation
    print(f"\n[Phase 8] Restarts (t={time.monotonic() - t_start:.1f}s)")
    for restart in range(3):
        if time.monotonic() > deadline - 3:
            break
        sigma = 0.01 * (restart + 1)
        a_pert = _perturb(best_seq, sigma, rng)
        t_end = min(deadline - 1, time.monotonic() + 15)
        a_r, sc_r = _kkt_newton(a_pert, t_end, max_rounds=30,
                                 k_schedule=[3, 10, 50, 200])
        _update(a_r, sc_r, f"restart_{restart}")

    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        # Verify with evaluate_sequence
        try:
            eval_sc = evaluate_sequence([float(x) for x in final])
            print(f"FINAL: score={sc_final:.10f}, eval={eval_sc:.10f}, "
                  f"n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        except Exception:
            print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
                  f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
