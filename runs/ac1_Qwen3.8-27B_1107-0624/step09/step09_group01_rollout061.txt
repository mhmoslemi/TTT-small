Optimized asymmetric sequence search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key insight: asymmetric sequences can achieve scores below 2 by making the center
of the linear convolution (a dot a_rev) small while keeping all other lags controlled.
Uses KKT active-set method with FFT-based Toeplitz solves, log-sum-exp smoothing,

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _kkt_matvec(v, W_rev, n):
    """Compute B*v where B*v[i] = conv(v, W_rev)[n-1-i]"""
    c = _full_convolve(v, W_rev)
    return c[n - 1:2 * n - 1][::-1]


def _kkt_solve_cg(W, n, x0=None, max_iter=300, tol=1e-13):
    """Solve B*a = 1 using CG where B*a[i] = sum_j W[i+j]*a[j]"""
    W_rev = W[::-1]
    rhs = np.ones(n)
    if x0 is not None:
        x = x0.copy()
    else:
        x = np.ones(n) / n
    Bx = _kkt_matvec(x, W_rev, n)
    r = rhs - Bx
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        s = x.sum()
        return x / s if s > 1e-15 else x
    for i in range(max_iter):
        Bp = _kkt_matvec(p, W_rev, n)
        pHp = float(p @ Bp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        if not np.isfinite(alpha):
            break
        x_new = x + alpha * p
        r_new = r - alpha * Bp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    s = x.sum()
    if not np.isfinite(s) or abs(s) < 1e-30:

    return x

            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

            tol = tol_frac * c_max

            if active.sum() >= 3:
                break
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        W = np.zeros_like(c)
        W[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(W, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(W, c))
        grad = 2.0 * a * (h - W_val)
        grad = grad - np.mean(grad)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break

        v = momentum * v - lr * grad
        z = z + v
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 99:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, 0.06)
            else:
                patience += 1
                lr *= 0.85
                momentum = min(momentum + 0.005, 0.95)
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)
                    patience = 0
                    lr = 0.02
