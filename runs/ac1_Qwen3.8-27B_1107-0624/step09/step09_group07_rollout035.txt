"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Strategy:
  - Phase 1: L-BFGS with adaptive sharp log-sum-exp smoothing
  - Phase 2: Active-set subgradient with exact subgradient + backtracking + oscillation damping
  - Phase 3: Subgradient coordinate descent (per-coordinate 1D minimax via bisection)

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_w(a, w):
    """Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a. w has length 2n-1."""
    n = len(a)
    wa = _full_convolve(w, a[::-1])
    return 2.0 * wa[n - 1:2 * n - 1]


def _active_subgrad(a, c, active_frac=0.005):
    """Compute subgradient of max(c) as average over active peaks."""
    n_2nm1 = len(c)
    n_active = max(1, int(n_2nm1 * active_frac))
    c_max = float(c.max())
    # Active: peaks within a small tolerance of max
    tol = max(c_max * 0.001, 1e-15)
    active = c >= (c_max - tol)
    n_act = int(active.sum())
    if n_act == 0:
        active[int(np.argmax(c))] = True
        n_act = 1
    w = np.zeros(n_2nm1)
    w[active] = 1.0 / n_act
    return _grad_w(a, w), n_act


def _logsumexp_smooth_grad(z, beta, n):
    """Log-sum-exp smoothed objective and gradient in log-domain."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        w = e / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    ga = _grad_w(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


class _SmoothProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _logsumexp_smooth_grad(z, self.beta, self.n)[0]

    def grad(self, z):
        return _logsumexp_smooth_grad(z, self.beta, self.n)[1]


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive log-sum-exp smoothing and annealing."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a

    # Estimate a good initial beta from the peak gap
    c_sorted = np.sort(c_cur)[::-1]
    n_check = min(50, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[n_check]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap * 0.5, max_c * 0.003)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Annealing schedule: from smooth to sharp
    schedule = [
        (1.5, 200), (1.0, 300), (0.7, 300), (0.5, 300),
        (0.3, 300), (0.2, 300), (0.1, 300), (0.05, 300),
        (0.02, 200), (0.01, 200), (0.005, 200), (0.002, 150),
    ]

    for beta_mult, max_iter in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        beta = base_beta * beta_mult
        est_time = 0.004 * max(1.0, n / 8192.0)
        actual_max_iter = min(max_iter, max(30, int(remaining / (est_time * 15))))

        prob = _SmoothProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': actual_max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxcor': 80, 'maxls': 40}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                sc_new = _score(a_new)
                if sc_new < current_sc - 1e-16:
                    current_a = a_new
                    current_sc = sc_new
                    z = np.log(np.maximum(a_new, 1e-30))
                    z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_subgrad_opt(a_init, deadline, min_time_remaining=30.0):
    """Active-set subgradient with oscillation detection and adaptive steps."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    momentum = 0.5
    velocity = np.zeros(n)
    lr = 0.03
    stall = 0
    max_stall = 100
    peak_history = []
    oscillation_damping = 1.0

    t_start = time.monotonic()
    iteration = 0

    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Detect oscillation: check if c_max is oscillating
        peak_history.append(c_max)
        if len(peak_history) > 20:
            peak_history.pop(0)
        if len(peak_history) == 20:
            # Check for oscillation pattern
            diffs = np.diff(peak_history)
            sign_changes = int(np.sum(np.abs(np.diff(np.sign(diffs))) > 0))
            if sign_changes > 10:
                oscillation_damping = min(0.95, oscillation_damping * 0.98)
                momentum = min(0.7, momentum + 0.02)
            else:
                oscillation_damping = min(1.0, oscillation_damping * 1.01)

        # Compute subgradient over active peaks
        tol = max(c_max * 0.0005, 1e-15)
        active = c >= (c_max - tol)
        n_act = int(active.sum())
        if n_act == 0:
            active[int(np.argmax(c))] = True
            n_act = 1

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_act
        g = _grad_w(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break

        # Adaptive learning rate based on progress
        if stall > 0:
            lr = max(1e-5, lr * 0.95)
        else:
            lr = min(0.08, lr * 1.02)

        # Momentum step with oscillation damping
        velocity = momentum * velocity * oscillation_damping - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()

        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            # Backtracking line search
            for ls in range(20):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-16:
                    improved = True
                    break
                velocity = velocity * 0.5
                a_try = a + velocity
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    break
                a_try = a_try / s_try

        if improved:
            a = a_try
            # Adapt active set tolerance
            if n_act < 50:
                tol *= 1.1
            elif n_act > 200:
                tol *= 0.9
        else:
            velocity = np.zeros(n)
            lr = max(1e-5, lr * 0.7)

        # Stall recovery
        if stall > max_stall:
            # Try a random perturbation in the direction of reducing variance
            rng_local = np.random.default_rng(iteration)
            pert = rng_local.standard_normal(n) * 0.01
            pert -= pert.mean()
            a_try2 = a + pert
            a_try2 = np.maximum(a_try2, 0)
            s2 = a_try2.sum()
            if s2 > 1e-10:
                a_try2 = a_try2 / s2
                sc2 = _score(a_try2)
                if sc2 < best_sc:
                    a = a_try2
                    best_a = a.copy()
                    best_sc = sc2
            velocity = np.zeros(n)
            momentum = 0.3
            stall = 0
            max_stall = int(max_stall * 1.3)

    return best_a, best_sc

                c_new = c_affected + (t_mid - old_ai) * deriv
                c_new_max = float(c_new.max())
                overall_max = max(c_new_max, c_max)

                active_new = c_new >= (c_new_max - 1e-13 * max(c_new_max, 1e-30))
                if active_new.sum() == 0:
                    active_new[int(np.argmax(c_new))] = True
                subgrad = float(deriv[active_new].mean())

                if subgrad < -1e-12:
                    t_lo = t_mid
                elif subgrad > 1e-12:
                    t_hi = t_mid
                else:
                    t_opt = t_mid
                    break
            else:
                t_opt = (t_lo + t_hi) / 2.0

            t_opt = max(0.0, min(t_opt, 10.0 / n))
            if abs(t_opt - old_ai) > 1e-14:
                a[i] = t_opt
                improved = True

        if not improved:
            no_improve += 1
            continue

        # Evaluate
        sc_new = _score(a)
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            a = best_a.copy()

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * (t - 0.5)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2) + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * t) + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS, active-set subgradient,
    and coordinate descent for minimizing the peak auto-convolution score.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        best_seq = _make_profile(n_parent, "tapered")
        best_score = _score(best_seq)
        print(f"No parent, tapered n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS (10 min) ===
    p1_time = min(600, (deadline - t_start) * 0.33)
    print(f"\n[P1] L-BFGS (budget={p1_time:.0f}s)")
    a1 = _run_lbfgs(best_seq, t_start + p1_time, min_time_remaining=30)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs1] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Active-set subgradient (5 min) ===
    p2_time = min(300, (deadline - t_start) * 0.17)
    if time.monotonic() < deadline - 200:
        print(f"\n[P2] Active-set subgradient (budget={p2_time:.0f}s)")
        a2, sc2 = _active_subgrad_opt(best_seq, t_start + p2_time, min_time_remaining=30)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [subgrad] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Coordinate descent (4 min) ===
    if time.monotonic() < deadline - 180:
        p3_time = min(240, (deadline - t_start) * 0.13)
        print(f"\n[P3] Coordinate descent (budget={p3_time:.0f}s)")
        a3, sc3 = _coord_descent_fast(best_seq, t_start + p3_time, min_time_remaining=20)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [coord] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: L-BFGS round 2 (5 min) ===
    if time.monotonic() < deadline - 150:
        p4_time = min(300, (deadline - t_start) * 0.17)
        print(f"\n[P4] L-BFGS round 2 (budget={p4_time:.0f}s)")
        a4 = _run_lbfgs(best_seq, t_start + p4_time, min_time_remaining=30)
        sc4 = _score(a4)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [lbfgs2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Active-set + coordinate polish (5 min) ===
    if time.monotonic() < deadline - 100:
        p5_time = min(300, (deadline - t_start) * 0.17)
        print(f"\n[P5] Polish (budget={p5_time:.0f}s)")
        # Active-set
        a5, sc5 = _active_subgrad_opt(best_seq, t_start + p5_time * 0.6, min_time_remaining=20)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [polish_subgrad] t={time.monotonic()-t_start:.1f}s")
        # Coordinate
        if time.monotonic() < deadline - 50:
            a5b, sc5b = _coord_descent_fast(best_seq, t_start + p5_time, min_time_remaining=20)
            if sc5b < best_score:
                best_seq = a5b.copy()
                best_score = sc5b
                print(f"  NEW BEST: {best_score:.10f} [polish_coord] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-restart (remaining time) ===
    if time.monotonic() < deadline - 60:
        print(f"\n[P6] Multi-restart (t={time.monotonic()-t_start:.1f}s)")
        n_main = len(best_seq)
        restart_budget = deadline - 60 - time.monotonic()
        n_restarts = max(2, int(restart_budget / 60))

        sigmas = [0.02, 0.03, 0.05, 0.08, 0.04, 0.06, 0.1, 0.03]
        for i in range(n_restarts):
            if time.monotonic() > deadline - 40:
                break
            sigma = sigmas[i % len(sigmas)]
            a_pert = _perturb(best_seq, sigma, rng)
            # Quick L-BFGS
            a_r = _run_lbfgs(a_pert, min(deadline - 30, time.monotonic() + 40), min_time_remaining=5)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.10f} [restart_{i}] t={time.monotonic()-t_start:.1f}s")

        # Try different profiles
        profiles = ["cos_taper", "gauss", "bimodal", "asym"]
        for prof_kind in profiles:
            if time.monotonic() > deadline - 30:
                break
            a_prof = _make_profile(n_main, prof_kind)
            a_p = _run_lbfgs(a_prof, min(deadline - 20, time.monotonic() + 25), min_time_remaining=5)
            sc_p = _score(a_p)
            if sc_p < best_score:
                best_seq = a_p.copy()
                best_score = sc_p
                print(f"  NEW BEST: {best_score:.10f} [profile_{prof_kind}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final polish ===
    if time.monotonic() < deadline - 5:
        print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.1f}s)")
        a_fin, sc_fin = _coord_descent_fast(best_seq, deadline - 3, min_time_remaining=3)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
