import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _symmetrize(a):
    return (a + a[::-1]) / 2.0


def _grad_smoothed_fast(a, p, n):
    """Compute H_p * a using FFT where H_p = 2 * sum_k p_k H_k."""
    b = a[::-1]
    conv_len = len(b) + len(p) - 1
    N = _next_pow2(conv_len)
    Fb = np.fft.rfft(b, N)
    Fp = np.fft.rfft(p, N)
    conv = np.fft.irfft(Fb * Fp, N)[:conv_len]
    result = 2.0 * conv[n - 1:2 * n - 1].copy()
    for k in range(0, 2 * n - 1, 2):
        m = k // 2
        if m < n and k < len(p):
            result[m] -= a[m] * p[k]
    return result

def _resample_linear(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _resample_fft(a, n_new):
    """FFT-based resampling (better for smooth signals)."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    N = _next_pow2(max(n_old, n_new))
    A = np.fft.rfft(a, N)
    a_new = np.fft.irfft(A, n_new)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

    """Compute max/mean ratio of autocorrelation."""
    m = c.mean()
    if m < 1e-30:

    return float(c.max() / m)

    z = z - z.mean()
    G = np.zeros(n)  # Adagrad accumulator
    base_lr = 0.5
    ln_base = np.log(base_lr)

    c = _full_convolve(a, a)
    initial_pf = _peaking_factor(c)

    for step in range(20000):

        # Adaptive threshold: start loose, tighten as peaking factor decreases

        # Threshold: values above (1 - eps) of max are active

                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    w = np.zeros_like(c_local)
                    w[int(np.argmax(c_local))] = 1.0
                else:
                    w = e / s_e
                a_rev = a_local[::-1]
                wc = _full_convolve(w, a_rev)
                h = wc[n - 1:2 * n - 1]
                W_val = float(np.dot(w, c_local))
                return 4.0 * n * a_local * (h - W_val)

            result = minimize(
                obj_func, z,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                c_cur = _full_convolve(a_new, a_new)
                max_c = float(c_cur.max())
        except Exception:
            pass
        # Re-center beta
        c_cur = _full_convolve(_softmax(z), _softmax(z))
        max_c = float(c_cur.max())

    return best_a


def _run_exact_subgrad(a_init, deadline, min_time=10.0):
    """
    Exact subgradient method with top-k active set and backtracking line search.
    For final polishing when the active set is small and well-defined.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    # Use top-k lags as active set
    top_k_values = [3, 5, 8, 10, 15]
    for top_k in top_k_values:
        if time.monotonic() + min_time > deadline:
            break

        for it in range(500):
            if time.monotonic() + min_time > deadline:
                break

            c = _full_convolve(a, a)
            c_max = float(c.max())
            # Active set: top k lags
            top_idx = np.argsort(c)[::-1][:top_k]
            p = np.zeros(2 * n - 1)
            p[top_idx] = 1.0 / top_k

            grad = _grad_smoothed_fast(a, p, n)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break

            # Backtracking line search
            step = 0.1
            found = False
            for ls in range(30):
                a_try = _proj_simplex(a - step * grad)
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) <= c_max + 1e-12:
                    a = a_try
                    found = True
                    break
                step *= 0.5
                if step < 1e-15:
                    break

            if not found and step < 1e-14:
                break

            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _run_fw_topk(a_init, deadline, min_time=10.0):
    """Frank-Wolfe with top-k active set and strong Wolfe line search."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(2000):
        if time.monotonic() + min_time > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        top_k = min(5, len(c))
        top_idx = np.argsort(c)[::-1][:top_k]
        p = np.zeros(2 * n - 1)
        p[top_idx] = 1.0 / top_k

        grad = _grad_smoothed_fast(a, p, n)
        # FW vertex: s = argmin grad
        s = int(np.argmin(grad))
        e_s = np.zeros(n)
        e_s[s] = 1.0

        # LAR step size
        gamma = 2.0 / (it + 2.0)
        a_new = (1 - gamma) * a + gamma * e_s

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_sc


def _random_restart(a_base, sigma, rng):
    """Perturb a_base in log-space and normalize."""
    z = np.log(np.maximum(a_base, 1e-30))
    z = z + sigma * rng.standard_normal(len(z))
    z = z - z.mean()
    return _softmax(z)


def _short_optimize(a_init, deadline, min_time=5.0):
    """Quick optimization: few L-BFGS iterations + a few exact subgrad steps."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    beta = max_c * 0.001

    try:
        def obj_func(z_local):
            a_local = _softmax(z_local)
            c_local = _full_convolve(a_local, a_local)
            c_max_local = float(c_local.max())
            shifted = (c_local - c_max_local) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s_e = e.sum()
            if s_e < 1e-300:
                return 2.0 * n * c_max_local
            return 2.0 * n * (c_max_local + beta * np.log(s_e))

        def grad_func(z_local):
            a_local = _softmax(z_local)
            c_local = _full_convolve(a_local, a_local)
            c_max_local = float(c_local.max())
            shifted = (c_local - c_max_local) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s_e = e.sum()
            if s_e < 1e-300:
                w = np.zeros_like(c_local)
                w[int(np.argmax(c_local))] = 1.0
            else:
                w = e / s_e
            a_rev = a_local[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W_val = float(np.dot(w, c_local))
            return 4.0 * n * a_local * (h - W_val)

        remaining = deadline - min_time - time.monotonic()
        if remaining > 3:
            max_iter = min(300, max(10, int(remaining / 0.01)))
            result = minimize(
                obj_func, z, jac=grad_func, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxcor': 30}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
    except Exception:
        pass

    # Quick exact subgradient polish
    if time.monotonic() + min_time < deadline:
        a_qs, sc_qs = _run_exact_subgrad(best_a, deadline, min_time=min_time)
        if sc_qs < best_sc:
            best_sc = sc_qs
            best_a = a_qs

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: L-BFGS with adaptive beta from parent (0-350s)
    Phase 2: Adagrad peak shaving with cooling (350-600s)
    Phase 3: Multi-n exploration (600-900s)
    Phase 4: Random restarts + short re-optimization (900-1400s)
    Phase 5: Final polish with exact subgradient (1400-1770s)
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 32768
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: L-BFGS with adaptive beta (0-350s) ===
    print(f"\n[Phase 1] L-BFGS adaptive (t={time.monotonic() - t_start:.1f}s)")
    phase1_deadline = t_start + 350
    a1 = _run_lbfgs_adaptive(parent, phase1_deadline, min_time=30.0)
    update_best(a1, _score(a1), "lbfgs_parent")

    # Try symmetrized version
    a1_sym = _symmetrize(a1)
    a1_sym = _proj_simplex(a1_sym)
    sc_sym = _score(a1_sym)
    if sc_sym < best_score:
        update_best(a1_sym, sc_sym, "lbfgs_sym")
        a1 = a1_sym

    # Second L-BFGS pass with tighter schedule
    if time.monotonic() < phase1_deadline - 40:
        a1b = _run_lbfgs_adaptive(a1, phase1_deadline, min_time=20.0)
        update_best(a1b, _score(a1b), "lbfgs_r2")
        a1 = a1b

    # Various initial profiles at same n
    if time.monotonic() < phase1_deadline - 30:
        for prof in ["kaiser", "optimal_4pt", "cos"]:
            if time.monotonic() > phase1_deadline - 25:
                break
            a_init = _make_profile(n_parent, prof)
            # Blend with parent
            a_blend = 0.7 * parent + 0.3 * a_init
            a_blend = _proj_simplex(a_blend)
            a_p = _run_lbfgs_adaptive(a_blend, phase1_deadline, min_time=15.0)
            update_best(a_p, _score(a_p), f"lbfgs_{prof}")

    # === Phase 2: Adagrad peak shaving (350-600s) ===
    print(f"\n[Phase 2] Adagrad peak shaving (t={time.monotonic() - t_start:.1f}s)")
    phase2_deadline = min(deadline - 800, phase1_deadline + 250)
    a2, sc2 = _run_adagrad_peak_shave(best_seq, phase2_deadline, min_time=30.0)
    update_best(a2, sc2, "adagrad")

    # L-BFGS after Adagrad
    if time.monotonic() < phase2_deadline - 20:
        a2b = _run_lbfgs_adaptive(a2, phase2_deadline, min_time=15.0)
        update_best(a2b, _score(a2b), "adagrad+lbfgs")

    # Exact subgradient polish
    if time.monotonic() < phase2_deadline - 10:
        a2c, sc2c = _run_exact_subgrad(a2b if 'a2b' in dir() else a2,
                                        phase2_deadline, min_time=10.0)
        update_best(a2c, sc2c, "adagrad+exact")

    # === Phase 3: Multi-n exploration (600-900s) ===
    print(f"\n[Phase 3] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    phase3_deadline = min(deadline - 550, phase2_deadline + 300)
    n_tries = [n_parent * 5 // 4, n_parent * 3 // 4, n_parent * 7 // 5, n_parent * 9 // 8]
    # Deduplicate and filter
    n_tries = sorted(set([n for n in n_tries if n >= 4096 and n <= 65536]))

    for n_try in n_tries:
        if time.monotonic() > phase3_deadline - 40:
            break
        if n_try == n_parent:
            continue
        print(f"  Trying n={n_try}...")
        a_resampled = _resample_linear(best_seq, n_try)
        # Also try FFT resampling
        a_resampled2 = _resample_fft(best_seq, n_try)
        
        # Optimize from linear resample
        a_n = _run_lbfgs_adaptive(a_resampled, phase3_deadline, min_time=20.0)
        sc_n = _score(a_n)
        update_best(a_n, sc_n, f"n_{n_try}_lin")

        # Try FFT resample if time allows
        if time.monotonic() < phase3_deadline - 40:
            a_n2 = _run_lbfgs_adaptive(a_resampled2, phase3_deadline, min_time=15.0)
            update_best(a_n2, _score(a_n2), f"n_{n_try}_fft")

    # Also try fresh profiles at different n
    if time.monotonic() < phase3_deadline - 30:
        for n_fresh in [24576, 40960]:
            if time.monotonic() > phase3_deadline - 25:
                break
            a_fresh = _make_profile(n_fresh, "tapered")
            a_f = _run_lbfgs_adaptive(a_fresh, phase3_deadline, min_time=15.0)
            update_best(a_f, _score(a_f), f"fresh_n{n_fresh}")

    # === Phase 4: Random restarts + short re-optimization (600s-1400s) ===
    print(f"\n[Phase 4] Random restarts (t={time.monotonic() - t_start:.1f}s)")
    phase4_deadline = min(deadline - 350, phase3_deadline + 500)
    n_restarts = 8
    time_per_restart = max(10.0, (phase4_deadline - time.monotonic() - 30) / n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > phase4_deadline - 30:
            break
        # Different perturbation scales
        sigma = 0.02 + 0.03 * r / n_restarts
        a_pert = _random_restart(best_seq, sigma, rng)
        # Also try symmetrized perturb
        if r % 2 == 1:
            a_pert = _symmetrize(a_pert)
            a_pert = _proj_simplex(a_pert)
        
        sub_deadline = min(phase4_deadline - 20, time.monotonic() + time_per_restart)
        a_r, sc_r = _short_optimize(a_pert, sub_deadline, min_time=5.0)
        update_best(a_r, sc_r, f"restart_{r}_s{sigma:.3f}")

    # Also try a few completely different profiles
    if time.monotonic() < phase4_deadline - 30:
        for prof in ["kaiser", "optimal_4pt"]:
            if time.monotonic() > phase4_deadline - 25:
                break
            a_fresh = 0.5 * best_seq + 0.5 * _make_profile(len(best_seq), prof)
            a_fresh = _proj_simplex(a_fresh)
            sub_dl = min(phase4_deadline - 10, time.monotonic() + 60)
            a_r, sc_r = _short_optimize(a_fresh, sub_dl, min_time=5.0)
            update_best(a_r, sc_r, f"blend_{prof}")

    # === Phase 5: Final polish (1400s-1770s) ===
    print(f"\n[Phase 5] Final polish (t={time.monotonic() - t_start:.1f}s)")
    phase5_deadline = deadline - 10

    # Symmetrize and re-optimize
    a_sym = _symmetrize(best_seq)
    a_sym = _proj_simplex(a_sym)
    sc_sym = _score(a_sym)
    if sc_sym < best_score:
        update_best(a_sym, sc_sym, "final_sym")

    # L-BFGS with very tight beta
    if time.monotonic() < phase5_deadline - 60:
        a_fin = _run_lbfgs_adaptive(best_seq, phase5_deadline - 30, min_time=30.0)
        update_best(a_fin, _score(a_fin), "final_lbfgs")

    # Exact subgradient with progressively tighter active set
    if time.monotonic() < phase5_deadline - 30:
        a_qs, sc_qs = _run_exact_subgrad(best_seq, phase5_deadline - 10, min_time=10.0)
        update_best(a_qs, sc_qs, "final_exact")

    # One more L-BFGS pass
    if time.monotonic() < phase5_deadline - 15:
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        beta_fin = mb * 1e-9
        remaining = phase5_deadline - 15 - time.monotonic()
        if remaining > 5:
            max_iter = min(500, max(20, int(remaining / 0.005)))
            try:
                def final_obj(z_local):
                    a_local = _softmax(z_local)
                    c_local = _full_convolve(a_local, a_local)
                    c_max_local = float(c_local.max())
                    shifted = (c_local - c_max_local) / beta_fin
                    e = np.exp(np.clip(shifted, -700, 0))
                    s_e = e.sum()
                    if s_e < 1e-300:
                        return 2.0 * n * c_max_local
                    return 2.0 * n * (c_max_local + beta_fin * np.log(s_e))

                def final_grad(z_local):
                    a_local = _softmax(z_local)
                    c_local = _full_convolve(a_local, a_local)
                    c_max_local = float(c_local.max())
                    shifted = (c_local - c_max_local) / beta_fin
                    e = np.exp(np.clip(shifted, -700, 0))
                    s_e = e.sum()
                    if s_e < 1e-300:
                        w = np.zeros_like(c_local)
                        w[int(np.argmax(c_local))] = 1.0
                    else:
                        w = e / s_e
                    a_rev = a_local[::-1]
                    wc = _full_convolve(w, a_rev)
                    h = wc[n - 1:2 * n - 1]
                    W_val = float(np.dot(w, c_local))
                    return 4.0 * n * a_local * (h - W_val)

                result = minimize(
                    final_obj, z, jac=final_grad, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16, 'maxcor': 40}
                )
                a_flb = _softmax(result.x)
                update_best(a_flb, _score(a_flb), "final_tight_lbfgs")
            except Exception:
                pass

    # Final tiny perturbations
    if time.monotonic() < phase5_deadline - 5:
        for tiny_sigma in [0.001, 0.0005, 0.002]:
            a_tiny = _random_restart(best_seq, tiny_sigma, rng)
            sc_tiny = _score(a_tiny)
            if sc_tiny < best_score:
                update_best(a_tiny, sc_tiny, f"tiny_{tiny_sigma}")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(8192, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
