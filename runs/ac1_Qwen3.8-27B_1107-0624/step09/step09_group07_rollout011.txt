    return idx, c[idx]


def _kkt_residual(a, c, active_idx, lambdas):
    """Compute KKT residual: how non-constant is the weighted gradient on support."""
    n = len(a)
    support = a > 1e-10
    if support.sum() < 2:
        return 0.0
    w = np.zeros(2 * n - 1)
    for k, lam in zip(active_idx, lambdas):
        w[k] = lam
    g = _grad_w(a, w)
    g_support = g[support]
    return float(np.std(g_support) / (np.abs(g_support).mean() + 1e-15))


def _lp_dual_direction(a, c, active_idx, M):
    """Solve dual LP to get optimal weights for active peaks.
    Returns lambdas (weights) and objective value."""
    n = len(a)
    m = len(active_idx)
    if m == 0:
        return np.array([1.0]), 0.0
    if m == 1:
        return np.array([1.0]), M
    
    # Dual: min M * sum(lambda)
    # s.t. sum_k lambda[k] * a[active_idx[k] - j] >= 1 for all j
    #      lambda >= 0
    #
    # The constraint matrix B[j, k] = a[active_idx[k] - j] if 0 <= active_idx[k] - j < n
    
    # For efficiency, use FFT-based matrix-vector product
    # But for linprog, we need the explicit matrix
    # For small m, just build it directly
    
    # B has shape (n, m): B[j, k] = a[active_idx[k] - j] if valid
    B = np.zeros((n, m))
    for k_idx, k_val in enumerate(active_idx):
        # B[j, k_idx] = a[k_val - j] for j where 0 <= k_val - j < n
        j_lo = max(0, k_val - (n - 1))
        j_hi = min(n - 1, k_val)
        B[j_lo:j_hi + 1, k_idx] = a[k_val - j_lo : k_val - j_hi : -1]
    
    # linprog: min c^T x s.t. A_ub @ x <= b_ub, A_eq @ x = b_eq, bounds
    # We have: min M * sum(lambda) = M * [1,1,...,1] @ lambda
    # s.t. -B @ lambda <= -1 (equivalent to B @ lambda >= 1)
    #      lambda >= 0
    
    c_obj = M * np.ones(m)
    A_ub = -B
    b_ub = -np.ones(n)
    bounds = [(0, None)] * m
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, res.fun
        else:
            # Fallback: uniform weights
            return np.ones(m) / m, M
    except Exception:
        return np.ones(m) / m, M


def _active_set_descent(a_init, deadline, min_time_remaining=30.0, verbose=True):
    """Active-set gradient descent with peak equalization and adaptive steps."""
    n = len(a_init)
    a = _normalize(np.maximum(a_init.copy(), 0.0))
    best_a = a.copy()
    best_sc = _score(a)
    
    # Parameters
    lr = 0.05
    lr_min = 1e-6
    lr_max = 0.5
    n_top = 10
    t = 0.5
    momentum = 0.8
    velocity = np.zeros(n)
    stall = 0
    max_stall = 100
    prev_dir = None
    prev_obj = None
    
    iteration = 0
    t_start = time.monotonic()
    
    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _conv(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > max_stall * 3:
            break
        
        # Get top peaks
        active_idx, active_vals = _peaks_info(c, n_top)
        
        # Compute weights: emphasize peaks that are highest
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-15:
            weights = (active_vals - active_vals.min()) / spread + 0.1
        else:
            weights = np.ones(n_top) / n_top
        weights = weights / weights.sum()
        
        # Build weight vector
        w = np.zeros(2 * n - 1)
        w[active_idx] = weights
        
        # Gradient
        g = _grad_w(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        
        if gnorm < 1e-18:
            break
        
        # Adaptive step size (simplified BB)
        if prev_dir is not None and prev_obj is not None:
            delta_g = g_centered - prev_dir
            s_norm = float(np.dot(a - best_a if prev_obj > best_sc else g_centered, g_centered))
            # Simple BB: use magnitude of gradient change
            g_change = np.linalg.norm(delta_g)
            if g_change > 1e-18:
                lr_new = gnorm / (g_change + 1e-15)
                lr = np.clip(lr_new * 0.01, lr_min, lr_max)
        
        # Update with momentum
        velocity = momentum * velocity - lr * g_centered / gnorm
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        
        if s_try < 1e-10:
            velocity = np.zeros(n)
            continue
        
        a_try = a_try / s_try
        
        # Line search
        improved = False
        alpha = 1.0
        for ls in range(10):
            c_try = _conv(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                improved = True
                break
            velocity = velocity * 0.5 - lr * 0.5 * g_centered / (gnorm + 1e-15)
            a_try = a + velocity
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                break
            a_try = a_try / s_try
            lr = max(lr_min, lr * 0.7)
        
        if improved:
            prev_dir = g_centered.copy()
            prev_obj = current_sc
            a = a_try
            # Adapt
            n_top = min(50, n_top + 1) if iteration % 100 == 0 else n_top
        else:
            velocity = np.zeros(n)
            lr = max(lr_min, lr * 0.5)
            n_top = max(5, n_top // 2)
        
        # Periodic reset
        if stall > max_stall:
            a = best_a.copy()
            velocity = np.zeros(n)
            lr = 0.05
            n_top = 10
            stall = 0
            max_stall = max(30, max_stall // 2)
        
        # Log progress
        if verbose and iteration % 5000 == 0:
            print(f"    iter {iteration}: sc={current_sc:.10f}, best={best_sc:.10f}, "
                  f"lr={lr:.2e}, n_top={n_top}, stall={stall}")
    
    return best_a, best_sc


def _lbfgs_smooth(a_init, deadline, min_time_remaining=20.0, beta=None, K=None):
    """L-BFGS with log-sum-exp smoothing."""
    n = len(a_init)
    a = _normalize(np.maximum(a_init.copy(), 0.0))
    
    c = _conv(a, a)
    c_max = float(c.max())
    
    if beta is None:
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[min(50, len(c_sorted)-1)] - c_sorted[min(100, len(c_sorted)-1)]
        beta = max(gap, c_max * 0.01) if gap > 0 else c_max * 0.01
    
    if K is None:
        K = max(10, int((2*n-1) * 0.02))
    
    # Log parameterization
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    def objective(z):
        a_s = np.exp(z - z.max())
        a_s = a_s / a_s.sum()
        c_s = _conv(a_s, a_s)
        # Log-sum-exp of top K
        if K < len(c_s):
            top = np.argpartition(c_s, -K)[-K:]
            c_top = c_s[top]
            c_max_s = float(c_top.max())
            shifted = np.clip((c_top - c_max_s) / beta, -700, 0)
            e = np.exp(shifted)
            s = e.sum()
            if s < 1e-300:
                return 2.0 * n * c_max_s
            return 2.0 * n * (c_max_s + beta * np.log(s))
        else:
            c_max_s = float(c_s.max())
            shifted = np.clip((c_s - c_max_s) / beta, -700, 0)
            e = np.exp(shifted)
            s = e.sum()
            if s < 1e-300:
                return 2.0 * n * c_max_s
            return 2.0 * n * (c_max_s + beta * np.log(s))
    
    def gradient(z):
        a_s = np.exp(z - z.max())
        a_s = a_s / a_s.sum()
        c_s = _conv(a_s, a_s)
        if K < len(c_s):
            top = np.argpartition(c_s, -K)[-K:]
            c_top = c_s[top]
            c_max_s = float(c_top.max())
        else:
            top = np.arange(len(c_s))
            c_top = c_s
            c_max_s = float(c_s.max())
        
        shifted = np.clip((c_top - c_max_s) / beta, -700, 0)
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c_s))] = 1.0
        else:
            e_full = np.zeros(2 * n - 1)
            e_full[top] = e
            w = e_full / s
        
        ga = _grad_w(a_s, w)
        W = float(np.dot(a_s, ga))
        grad_z = a_s * (ga - W)
        return grad_z
    
    try:
        max_iter = min(5000, max(50, int((deadline - time.monotonic() - min_time_remaining) / 0.01)))
        result = minimize(objective, z, jac=gradient, method='L-BFGS-B',
                         options={'maxiter': max_iter, 'ftol': 1e-18,
                                  'gtol': 1e-14, 'maxcor': 100, 'maxls': 50})
        a_new = np.exp(result.x - result.x.max())
        a_new = a_new / a_new.sum()
        return a_new
    except Exception:
        return a_init.copy()


def _make_profile(n, kind="tapered", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.5) ** 4)
