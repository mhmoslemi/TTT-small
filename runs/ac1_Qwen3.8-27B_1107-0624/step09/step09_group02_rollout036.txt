import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def _hvp_pv(v, p, n):
    b = v[::-1]
    conv_len = len(b) + len(p) - 1
    N = _next_pow2(conv_len)
    Fb = np.fft.rfft(b, N)
    Fp = np.fft.rfft(p, N)
    conv = np.fft.irfft(Fb * Fp, N)[:conv_len]
    result = 2.0 * conv[n - 1:2 * n - 1].copy()
    for k in range(0, 2 * n - 1, 2):
        m = k // 2
        if m < n and k < len(p):
            result[m] -= v[m] * p[k]
    return result

def _central_binom_profile(n):
    """Generate the 'ideal half-flat' sequence: a[k] = binom(2k,k)/4^k."""
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2*k - 1) / (2*k)
    s = a.sum()
    if s > 0:
        a /= s
    return a

def _run_lbfgs_tight(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with very tight beta schedule, designed for near-optimal starting points."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    current_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    best_a = a.copy()
    best_sc = current_sc

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap * 0.05, max_c * 0.0001)

    ratios = [1.0, 0.3, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005,

def _active_set_optimize(a_init, deadline, min_time_remaining=30.0, max_rounds=200):
    """Active-set QP optimization with progressive tightening."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    # Progressive tolerance schedule
    for rnd in range(max_rounds):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Active set: lags within tolerance of max
        tol = c_max * max(1e-10, 0.01 * (0.95 ** rnd))
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / active.sum()
        
        # Solve: minimize sum_k p_k * b[k] s.t. a in simplex
        # Using accelerated projected gradient
        for inner in range(min(100, max(10, int(3.0 / max(tol/c_max, 1e-6))))):
            if time.monotonic() + min_time_remaining > deadline:
                break
            grad = _hvp_pv(a, p, n)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            lr = 0.1 / (gnorm * (1.0 + 0.005 * inner))
            a = _proj_simplex(a - lr * grad)
        
        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
        elif rnd > 10:
            a = best_a.copy()
    
    return best_a, best_sc

def _tail_boost(a_init, deadline, min_time_remaining=30.0):
    """Specifically try to boost the tail of the convolution to reduce the peak."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(100):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Weight the tail lags more heavily
        p = np.zeros(2 * n - 1)
        # Give equal weight to all lags but with extra emphasis on the peak
        peak_idx = int(np.argmax(c))
        p[peak_idx] = 2.0
        # Also weight the tail
        tail_start = n
        for k in range(tail_start, 2*n-1):
            p[k] = 1.0 + (k - tail_start) / n
        p = p / p.sum()
        
        # Gradient step
        grad = _hvp_pv(a, p, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Line search
        lr = 0.05 / (gnorm + 1e-10)
        improved = False
        for ls in range(20):
            a_new = _proj_simplex(a - lr * grad)
            c_new = _full_convolve(a_new, a_new)
            if float(c_new.max()) < c_max * (1 - 1e-8):

                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                improved = True
                break
            lr *= 0.7

    return best_a, best_sc

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5

"""
Optimizer for minimum peak linear autocorrelation using central binomial coefficient
initialization (the 'ideal half-flat' sequence), with multi-phase refinement via
L-BFGS with log-sum-exp smoothing, adaptive active-set QP, tail boosting,
and multi-length exploration. The key insight is that the sequence a[k]=binom(2k,k)/4^k
gives a convolution with flat prefix, providing a near-optimal starting point
with asymptotic score ≈ pi/2, from which aggressive optimization can push below 1.503.
"""
import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _hvp_pv(v, p, n):
    b = v[::-1]
    conv_len = len(b) + len(p) - 1
    N = _next_pow2(conv_len)
    Fb = np.fft.rfft(b, N)
    Fp = np.fft.rfft(p, N)
    conv = np.fft.irfft(Fb * Fp, N)[:conv_len]
    result = 2.0 * conv[n - 1:2 * n - 1].copy()
    for k in range(0, 2 * n - 1, 2):
        m = k // 2
        if m < n and k < len(p):
            result[m] -= v[m] * p[k]
    return result


def _central_binom_profile(n):
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2 * k - 1) / (2 * k)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "adaptive":
        a = _central_binom_profile(n).copy()
        alpha = 0.06 * np.sqrt(n)
        t = np.linspace(0, 1, n)
        boost = np.exp(alpha * (0.5 - t))
        a = a * boost
        a = np.maximum(a, 1e-15)
        return a / a.sum()
    else:
        return _central_binom_profile(n)

def _run_lbfgs_tight(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    best_a = a.copy()
    best_sc = _score(a)

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap * 0.05, max_c * 0.0001)

    ratios = [1.0, 0.3, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005,
              0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]

    for i, ratio in enumerate(ratios):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        beta = base_beta * ratio
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 4))))

        def obj_func(z_local):
            a_local = _softmax(z_local)
            c_local = _full_convolve(a_local, a_local)
            c_max_local = float(c_local.max())
            shifted = (c_local - c_max_local) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s_e = e.sum()
            if s_e < 1e-300:
                return 2.0 * n * c_max_local
            return 2.0 * n * (c_max_local + beta * np.log(s_e))

        def grad_func(z_local):
            a_local = _softmax(z_local)
            c_local = _full_convolve(a_local, a_local)
            c_max_local = float(c_local.max())
            shifted = (c_local - c_max_local) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s_e = e.sum()
            if s_e < 1e-300:
                w = np.zeros_like(c_local)
                w[int(np.argmax(c_local))] = 1.0
            else:
                w = e / s_e
            a_rev = a_local[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W_val = float(np.dot(w, c_local))
            return 4.0 * n * a_local * (h - W_val)

        try:
            result = minimize(
                obj_func, z, jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 50, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return best_a


def _active_set_optimize(a_init, deadline, min_time_remaining=30.0, max_rounds=200):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = c_max * max(1e-10, 0.01 * (0.93 ** rnd))
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / active.sum()

        inner_max = min(150, max(15, int(2.0 / max(tol / c_max, 1e-6))))
        for inner in range(inner_max):
            if time.monotonic() + min_time_remaining > deadline:
                break
            grad = _hvp_pv(a, p, n)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            lr = 0.1 / (gnorm * (1.0 + 0.003 * inner))
            a = _proj_simplex(a - lr * grad)

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
        elif rnd > 15:
            a = best_a.copy()

    return best_a, best_sc

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

            a_n = _run_lbfgs_tight(a_init_n, phase6_deadline, min_time_remaining=15.0)
            sc_n = _score(a_n)
            update_best(a_n, sc_n, f"n_{n_try}")

            if sc_n < best_score + 0.002 and time.monotonic() < phase6_deadline - 20:
                a_nq, sc_nq = _active_set_optimize(a_n, phase6_deadline, min_time_remaining=10.0, max_rounds=60)
                update_best(a_nq, sc_nq, f"n_{n_try}_qp")

    # === Phase 7: Final polish (100s) ===
    print(f"\n[Phase 7] Final Polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        a7, sc7 = _active_set_optimize(best_seq, deadline - 15, min_time_remaining=5.0, max_rounds=150)
        update_best(a7, sc7, "final_qp")

        if time.monotonic() < deadline - 12:
            a7b = _run_lbfgs_tight(best_seq, deadline - 8, min_time_remaining=5.0)
            update_best(a7b, _score(a7b), "final_lbfgs")

        if time.monotonic() < deadline - 10:
            a7c, sc7c = _tail_boost(best_seq, deadline - 8, min_time_remaining=4.0)
            update_best(a7c, sc7c, "final_tail")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _central_binom_profile(32768)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
