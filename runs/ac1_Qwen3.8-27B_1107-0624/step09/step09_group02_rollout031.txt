def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _grad_smoothed_fft(a, w, n):
    """Compute gradient of sum_k w[k] * c[k] w.r.t. a, where c = convolve(a,a).
    Uses FFT-based method: grad = 2 * (reverse_w * a)[n-1:2n-1] with diagonal correction.
    """
    a_rev = a[::-1]
    w_rev = w[::-1]
    conv = _full_convolve(w_rev, a_rev)
    result = 2.0 * conv[n - 1:2 * n - 1].copy()
    # Diagonal correction: for even k=2m, the term a[m]*a[m] appears once not twice
    for m in range(n):
        k = 2 * m
        if k < 2 * n - 1:
            result[m] -= a[m] * w[k]
    return result


def _make_window(n, kind="raised_cos"):
    t = np.linspace(0, 1, n)
    if kind == "raised_cos":
        a = np.cos(np.pi * t / 2.0) ** 2
    elif kind == "raised_cos_half":
        a = np.cos(np.pi * t / 2.0) ** 1
    elif kind == "hann":
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif kind == "hamming":
        a = 0.54 + 0.46 * np.cos(2 * np.pi * t)
    elif kind == "gauss":
        a = np.exp(-4.0 * (t - 0.5) ** 2)
    elif kind == "gauss_wide":
        a = np.exp(-2.0 * (t - 0.5) ** 2)
    elif kind == "prolate":
        # Approximation to prolate spheroidal wave function
        a = np.sqrt(np.maximum(1 - (2 * t - 1) ** 2, 0))
    elif kind == "super_gauss":
        a = np.exp(-8.0 * (t - 0.5) ** 4)
    elif kind == "uniform":
        a = np.ones(n)
    elif kind == "cos_cubed":
        a = np.cos(np.pi * t / 2.0) ** 3
    elif kind == "cos_quart":
        a = np.cos(np.pi * t / 2.0) ** 4
    else:
        a = np.cos(np.pi * t / 2.0) ** 2
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 1e-15)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _run_lbfgs_smoothed(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS on log-sum-exp smoothed peak autocorrelation, log-space parameterization."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = max(c_sorted[0] - c_sorted[min(2, len(c_sorted) - 1)], max_c * 0.001)
    base_beta = max(gap * 0.05, max_c * 0.0005)

    ratios = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
              0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001,
              0.000005, 0.000001, 0.0000005]

    for ratio in ratios:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        beta = base_beta * ratio
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 5))))

        try:
            def obj_func(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    return 2.0 * n * c_max_local
                return 2.0 * n * (c_max_local + beta * np.log(s_e))

            def grad_func(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    w = np.zeros_like(c_local)
                    w[int(np.argmax(c_local))] = 1.0
                else:
                    w = e / s_e
                grad_c = _grad_smoothed_fft(a_local, w, n)
                # Chain rule through softmax: d/dz = a * (I - outer(a,a)) @ grad_a
                # For softmax: da/dz = a*(I - 1*a^T), so dz objective grad = a * (grad_a - a @ grad_a)
                ga = grad_c
                return 2.0 * n * a_local * (ga - float(np.dot(a_local, ga)))

            result = minimize(
                obj_func, z,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 50, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return best_a


def _run_topk_grad(a_init, deadline, min_time_remaining=30.0):
    """Gradient descent with top-k active set and backtracking line search."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    momentum = np.zeros(n)
    beta_m = 0.85
    patience = 0
    max_patience = 80

    for step in range(10000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        a_curr = _softmax(z)
        c = _full_convolve(a_curr, a_curr)
        c_max = float(c.max())

        # Adaptive top-k
        k_act = min(max(3, n // 100), len(c))
        top_k = np.argsort(c)[::-1][:k_act]
        p = np.zeros(2 * n - 1)
        # Weight by excess over median
        med_c = np.median(c)
        p[top_k] = np.maximum(c[top_k] - med_c, 1e-10)
        p = p / p.sum()

        grad_a = _grad_smoothed_fft(a_curr, p, n)
        # Convert to z-gradient
        grad_z = a_curr * (grad_a - float(np.dot(a_curr, grad_a)))
        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-18:
            break

        momentum = beta_m * momentum + (1 - beta_m) * grad_z
        # Backtracking line search
        ls = 0.1
        z_new = z - ls * momentum
        improved = False
        for _ls in range(20):
            z_try = z - ls * momentum
            a_try = _softmax(z_try)
            c_try = _full_convolve(a_try, a_try)
            if float(c_try.max()) < c_max * (1 + 1e-10):
                z = z_try
                improved = True
                break
            ls *= 0.5
        if not improved:
            z = z - 0.01 * momentum / (grad_norm + 1e-10)
            z = np.clip(z, -50, 50)
            z = z - z.mean()

        if step % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros(n)
                    beta_m *= 0.95
                    patience = 0

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _run_coord_descent(a_init, deadline, min_time_remaining=30.0):
    """Coordinate-wise descent: optimize one coordinate at a time to reduce peak."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    # Precompute which c[k] depend on each a[m]
    # c[k] depends on a[m] when max(0, k-n+1) <= m <= min(k, n-1)

    for sweep in range(200):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        improved_sw = False

        # Random order for diversity
        order = np.arange(n)
        np.random.shuffle(order)

        for m in order:
            if time.monotonic() + min_time_remaining > deadline:
                break
            # Find which lags k depend on a[m]
            k_lo = max(0, m - (n - 1))  # k - m >= 0 and k - m <= n-1 => k in [m, m+n-1] but also k >= 0
            k_hi = min(2 * n - 2, m + n - 1)
            # Actually: c[k] = sum_i a[i]*a[k-i], a[m] appears when i=m (needs 0<=k-m<n) or k-i=m (needs 0<=k-m<n)
            # So k ranges from m to m+n-1 (intersected with [0, 2n-2])
            k_lo = max(0, m)
            k_hi = min(2 * n - 2, m + n - 1)

            if k_lo > k_hi:
                continue

            # Current value
            am = a[m]
            if am < 1e-15:
                continue

            # For each k in [k_lo, k_hi], c[k] = A_k * a[m]^2 + B_k * a[m] + C_k
            # where the coefficient of a[m]^2 is 1 if k=2m, else 0
            # the coefficient of a[m] is 2*a[k-m] if k!=2m, or a[m] (i.e., coefficient 1 for a[m]^2) if k=2m

            # We want to find a[m] >= 0 that minimizes max over k in [k_lo, k_hi] of c[k](a[m])
            # subject to the sum constraint: sum(a) = 1 => we need to renormalize
            # For simplicity, just optimize a[m] and renormalize at the end of the sweep

            # Evaluate the peak as function of a[m]
            # c[k](x) = diag_k * x^2 + 2*off_k * x + rest_k
            # where diag_k = 1 if k==2m else 0
            # off_k = a[k-m] if 0 <= k-m < n and k-m != m else 0 (for the cross term)
            # rest_k = c[k] - (terms involving a[m])

            # Current c[k] values (with a[m] = am)
            c_vals = c[k_lo:k_hi + 1].copy()

            # Compute the "base" (c[k] without a[m] contribution) and coefficients
            # c[k] = sum_i a[i]*a[k-i]
            # Terms with a[m]: i=m gives a[m]*a[k-m] (if 0<=k-m<n)
            #               k-i=m gives a[k-m]*a[m] (same term, counted once unless k=2m)
            # So c[k] = base_k + (1 if k==2m else 2) * a[m] * a[k-m] (if 0<=k-m<n and k-m!=m)
            #        = base_k + a[m]^2 (if k==2m)

            # Let's compute base_k = c[k] - contribution from a[m]
            base = np.zeros(k_hi - k_lo + 1)
            coeff_linear = np.zeros(k_hi - k_lo + 1)
            coeff_quad = np.zeros(k_hi - k_lo + 1)

            for idx, k in enumerate(range(k_lo, k_hi + 1)):
                jm = k - m
                if 0 <= jm < n:
                    if jm == m:
                        # k = 2m: c[k] includes a[m]^2
                        coeff_quad[idx] = 1.0
                        base[idx] = c[k] - am * am
                    else:
                        # c[k] includes 2*a[m]*a[jm]
                        coeff_linear[idx] = 2.0 * a[jm]
                        base[idx] = c[k] - 2.0 * am * a[jm]
                else:
                    base[idx] = c[k]

            # Now minimize max_k (base[k] + coeff_linear[k]*x + coeff_quad[k]*x^2) for x >= 0
            # This is a 1D convex optimization (max of quadratics)
            # Use ternary search or golden section

            def peak_of_x(x):
                vals = base + coeff_linear * x + coeff_quad * x * x
                return float(vals.max())

            # Find a good range for x
            x_lo = 0.0
            x_hi = am * 3.0
            if peak_of_x(x_lo) < peak_of_x(am):
                x_hi = am
                x_lo = 0.0
            else:
                x_lo = 0.0
                x_hi = am * 2.0

            # Golden section search
            gr = (np.sqrt(5) - 1) / 2
            a_gs = x_lo + gr * (x_hi - x_lo)
            b_gs = x_lo + (1 - gr) * (x_hi - x_lo)
            fa = peak_of_x(a_gs)
            fb = peak_of_x(b_gs)
            for _ in range(40):
                if fa < fb:
                    x_hi = b_gs
                    b_gs = a_gs
                    fb = fa
                    a_gs = x_lo + gr * (x_hi - x_lo)
                    fa = peak_of_x(a_gs)
                else:
                    x_lo = a_gs
                    a_gs = b_gs
                    fa = fb
                    b_gs = x_lo + (1 - gr) * (x_hi - x_lo)
                    fb = peak_of_x(b_gs)

            x_opt = (x_lo + x_hi) / 2
            # Also check x=0
            if peak_of_x(0) < peak_of_x(x_opt):
                x_opt = 0.0

            # Update
            if abs(x_opt - am) > 1e-10:
                a[m] = x_opt
                improved_sw = True

        # Renormalize
        s = a.sum()
        if s > 1e-15:
            a = a / s

        if sweep % 5 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _run_gradient_primal(a_init, deadline, min_time_remaining=30.0):
    """Projected gradient on the peak with adaptive top-k smoothing."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    momentum = np.zeros(n)
    beta_m = 0.9
    patience = 0
    max_patience = 60

    for step in range(5000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Top-k with adaptive k
        frac = min(0.1, max(0.01, 1.0 / (step + 10)))
        k_act = min(max(5, int(n * frac)), len(c))
        top_k = np.argsort(c)[::-1][:k_act]
        p = np.zeros(2 * n - 1)
        # Soft weighting: more weight to higher values
        c_top = c[top_k]
        w_top = (c_top - c_top.min() + 1e-10)
        p[top_k] = w_top
        p = p / p.sum()

        grad = _grad_smoothed_fft(a, p, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        momentum = beta_m * momentum + (1 - beta_m) * grad
        # Backtracking
        step_size = 0.05 / (np.linalg.norm(momentum) + 1e-10)
        for _ in range(30):
            a_new = _proj_simplex(a - step_size * momentum)
            c_new = _full_convolve(a_new, a_new)
            if float(c_new.max()) <= c_max * (1 + 1e-8):
                a = a_new
                break
            step_size *= 0.5

        if step % 10 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    a = best_a.copy()
                    momentum = np.zeros(n)
                    beta_m *= 0.9
                    patience = 0

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc
