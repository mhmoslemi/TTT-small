Let me also think about the "iterative peak equalization" method more carefully.

**Iterative Peak Equalization (IPE):**

At each iteration:
1. Compute $c = a*a$
2. Find the top $K$ peaks: indices $k_1, ..., k_K$
3. Compute the "excess" of each peak: $e_j = c_{k_j} - \bar{c}$ where $\bar{c}$ is the mean of the convolution
4. Compute the gradient direction: $d = -\sum_j \alpha_j \nabla c_{k_j}$ where $\alpha_j \propto e_j^p$ for some $p > 0$
5. Center $d$: $d \leftarrow d - \frac{1}{n}\sum_i d_i$ (to maintain the sum)
6. Find the optimal step size $\alpha^*$ that minimizes $\max_{k \in \{k_1,...,k_K\}} (a + \alpha d)*(a + \alpha d)_k$
   - This is a 1D convex optimization problem
   - Solve by bisection or by finding the intersection of the parabolas
7. Update: $a \leftarrow a + \alpha^* d$
8. Project onto $a \geq 0$ (if needed)
9. Re-normalize

For step 6, the exact solution:
For each active peak $k_j$, the value after the update is:
$Q_j(\alpha) = A_j \alpha^2 + B_j \alpha + C_j$
where $A_j = (d*d)_k_j$, $B_j = 2(d*a)_k_j$, $C_j = (a*a)_k_j$

We want to minimize $\max_j Q_j(\alpha)$ over $\alpha \geq 0$ (and subject to $a_i + \alpha d_i \geq 0$).

Since each $Q_j$ is a convex quadratic (if $A_j > 0$) or affine (if $A_j = 0$), the function $f(\alpha) = \max_j Q_j(\alpha)$ is convex. We can find the minimum by:
- Computing the subdifferential and finding where 0 is in the subdifferential
- Or using bisection on the derivative (where differentiable)
- Or simply using a 1D minimizer (golden section search)

The most robust approach is golden section search or bisection on the derivative.

The derivative (where it exists) is: $f'(\alpha) = Q_j'(\alpha)$ where $j = \arg\max_i Q_i(\alpha)$.

So we can use a "smart bisection":
1. Compute $\alpha_{lo} = 0$ and $\alpha_{hi} = 1$ (or some upper bound)
2. At each step, evaluate $f(\alpha_{mid})$ and determine which peak is active
3. If the active peak's derivative is negative, increase $\alpha_{lo}$; otherwise decrease $\alpha_{hi}$
4. Repeat until convergence

Actually, since $f$ is convex, we can just use a standard 1D convex minimization method. Let me use a combination of bisection and golden section.

"""
Optimized step-function sequence generator using multi-phase approach:
- L-BFGS with top-K log-sum-exp smoothing
- Iterative Peak Equalization (IPE) with exact 1D line search

- Diverse restart strategy

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _topk_smooth_obj_grad(z, beta, K, n):
    z_max = z.max()
    a = np.exp(z - z_max)
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    total_len = 2 * n - 1
    if K < total_len:
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(total_len)
        c_top = c
    
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(total_len)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(total_len)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    
    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


class _TopkSmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _topk_smooth_obj_grad(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _topk_smooth_obj_grad(z, self.beta, self.K, self.n)[1]


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):

    current_a = a.copy()

    if max_c < 1e-15:
        return current_a

    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    z = z - z.mean()
    
    schedule = [
        (1.0, 0.15), (0.7, 0.10), (0.5, 0.07), (0.3, 0.05),

    ]
    
    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        if K_frac == 3:
            K = 3
        else:

        if K_frac != 3:
            beta = base_beta * beta_mult
        else:
            beta = base_beta * 0.0001
        
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(50, int(remaining / (est_time * 25))))
        
        prob = _TopkSmoProb(beta, K, n)
        try:

def _exact_line_search(a, d, active_idx, max_alpha=1.0):
    """Find alpha > 0 that minimizes max over active peaks of (a+alpha*d)*(a+alpha*d)_k."""
    n = len(a)
    d_conv = _full_convolve(d, d)
    da_conv = _full_convolve(d, a)
    a_conv = _full_convolve(a, a)
    
    # Get the relevant values
    A = d_conv[active_idx]  # (d*d)_k for active k
    B = 2.0 * da_conv[active_idx]  # 2*(d*a)_k for active k
    C = a_conv[active_idx]  # (a*a)_k for active k
    
    def f(alpha):
        vals = A * alpha * alpha + B * alpha + C
        return float(vals.max())
    
    def f_deriv(alpha):
        vals = A * alpha * alpha + B * alpha + C
        active = np.argmax(vals)
        return float(2.0 * A[active] * alpha + B[active])
    
    # Also check non-negativity constraint
    if np.any(d < 0):
        max_alpha_nn = min(max_alpha, -a[np.argmin(d)] / d[np.argmin(d)] * 0.99)
        max_alpha = max(max_alpha_nn, 0)
    
    if max_alpha <= 0:
        return 0.0
    
    # Use bisection on the derivative
    lo = 0.0
    hi = max_alpha
    
    f_lo = f_deriv(lo)
    f_hi = f_deriv(hi)
    
    # If derivative at 0 is positive, the function is increasing from the start
    if f_lo >= 0:
        return 0.0
    # If derivative at hi is negative, the function is still decreasing
    if f_hi <= 0:
        return hi
    
    # Bisection
    for _ in range(50):
        mid = (lo + hi) / 2
        f_mid = f_deriv(mid)
        if f_mid < 0:
            lo = mid
        else:
            hi = mid
    
    best_alpha = (lo + hi) / 2
    best_val = f(best_alpha)
    
    # Check a few nearby points for robustness
    for trial in [0, best_alpha * 0.5, best_alpha * 0.8, best_alpha * 1.2, best_alpha * 1.5]:
        trial = min(max(trial, 0.0), max_alpha)
        v = f(trial)
        if v < best_val:
            best_val = v
            best_alpha = trial
    
    return best_alpha


def _peak_equalization(a_init, deadline, min_time_remaining=30.0):
    """Iterative peak equalization with exact line search."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    no_improve = 0
    max_no_improve = 200
    K = max(3, int((2 * n - 1) * 0.005))  # Top K peaks
    K_min = 3
    patience = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            patience = 0
        else:
            no_improve += 1
            patience += 1
        
        if patience > max_no_improve:
            break
        
        # Select active peaks
        n_total = 2 * n - 1
        n_active = min(K, n_total)
        active_idx = np.argpartition(c, -n_active)[-n_active:]
        
        # Weights: emphasize higher peaks
        c_active = c[active_idx]
        c_min_active = c_active.min()
        c_range = c_max - c_min_active
        if c_range > 1e-18:
            w = np.power((c_active - c_min_active) / c_range + 0.01, 2.0)
        else:
            w = np.ones(n_active) / n_active
        w = w / w.sum()
        
        # Build full weight vector
        w_full = np.zeros(n_total)
        w_full[active_idx] = w
        
        # Gradient of weighted sum of active peaks
        g = _grad_maxconv(a, w_full)
        # Center (maintain sum)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        
        if gnorm < 1e-18:
            break
        
        # Direction (negative gradient)
        d = -g_centered / gnorm
        
        # Exact line search
        alpha = _exact_line_search(a, d, active_idx, max_alpha=1.0)
        
        # Update
        if alpha > 1e-10:
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                break
            a_try = a_try / s_try
            
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-15:
                a = a_try
            else:
                # Try smaller step
                for factor in [0.5, 0.25, 0.1]:
                    a_try2 = a + alpha * factor * d
                    a_try2 = np.maximum(a_try2, 0.0)
                    s_try2 = a_try2.sum()
                    if s_try2 < 1e-10:
                        break
                    a_try2 = a_try2 / s_try2
                    sc_try2 = _score(a_try2)
                    if sc_try2 < current_sc - 1e-15:
                        a = a_try2
                        break
        else:
            no_improve += 5
            # Reduce K and try again
            K = max(K_min, K // 2)
        
        # Adapt K
        if patience % 20 == 0 and patience > 0:
            K = max(K_min, int(K * 0.8))
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

$d = -\arg\min_{\|d\|=1, \sum d_i = 0} \max_{k \in \mathcal{A}} \nabla c_k \cdot d$

This is a convex optimization problem in $d$. It's equivalent to finding the direction in the hyperplane $\sum d_i = 0$ that minimizes the maximum of a set of linear functions. This is a linear program!

Specifically:
minimize $t$
subject to $\nabla c_k \cdot d \geq -t$ for all $k \in \mathcal{A}$
$\|d\| \leq 1$
$\sum d_i = 0$

This is a SOCP (second-order cone program). For small $|\mathcal{A}|$, this is very fast to solve.

If multiple peaks are active, the optimal $d$ is a combination that "balances" the descent rates across all active peaks. This is similar to the Frank-Wolfe approach.

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous approach:
- Exact 1D line search on active-set (parabolic peak tracking)
- Iterative Peak Equalization (IPE) with adaptive active set sizing
- More aggressive L-BFGS annealing schedule (K down to 3)
- Momentum-assisted peak reduction for escaping plateaus
- Multi-length exploration with targeted sub-optimization
- Diverse small-perturbation restarts with quick L-BFGS recovery
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _topk_smoobj_grad(z, beta, K, n):
    z_max = z.max()
    a = np.exp(z - z_max)
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    total_len = 2 * n - 1
    if K < total_len:
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(total_len)
        c_top = c
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(total_len)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(total_len)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


class _TopkSmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[1]


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.004)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    schedule = [
        (1.0, 0.2), (0.7, 0.12), (0.5, 0.08), (0.35, 0.05),
        (0.2, 0.03), (0.1, 0.02), (0.05, 0.012), (0.03, 0.008),
        (0.015, 0.005), (0.008, 0.003), (0.004, 0.002), (0.002, 0.001),
        (0.001, 5), (0.0005, 3), (0.0002, 3),
    ]

    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        if K_frac <= 5:
            K = int(K_frac)
            beta = base_beta * beta_mult
        else:
            K = max(3, int((2 * n - 1) * K_frac))
            beta = base_beta * beta_mult

        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(10000, max(50, int(remaining / (est_time * 30))))

        prob = _TopkSmoProb(beta, K, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 200, 'maxls': 60}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

def _exact_line_search(a, d, active_idx, max_alpha=1.0):
    n = len(a)
    d_conv = _full_convolve(d, d)
    da_conv = _full_convolve(d, a)
    a_conv = _full_convolve(a, a)

    A = d_conv[active_idx].astype(np.float64)
    B = 2.0 * da_conv[active_idx].astype(np.float64)
    C = a_conv[active_idx].astype(np.float64)

    def f_of_alpha(alpha):
        vals = A * alpha * alpha + B * alpha + C
        return float(vals.max())

    def f_deriv_of_alpha(alpha):
        vals = A * alpha * alpha + B * alpha + C
        active = int(np.argmax(vals))
        return float(2.0 * A[active] * alpha + B[active])

    # Non-negativity: a_i + alpha * d_i >= 0
    neg_mask = d < -1e-15
    if np.any(neg_mask):
        ratios = a[neg_mask] / (-d[neg_mask])
        max_alpha_nn = float(ratios.min()) * 0.999
        max_alpha = min(max_alpha, max(0.0, max_alpha_nn))

    if max_alpha <= 1e-12:
        return 0.0

    d0 = f_deriv_of_alpha(0.0)
    if d0 >= 0:
        return 0.0

    d_end = f_deriv_of_alpha(max_alpha)
    if d_end <= 0:
        return max_alpha

    # Bisection on derivative (convex function => derivative is nondecreasing where differentiable)
    lo = 0.0
    hi = max_alpha
    for _ in range(60):
        mid = (lo + hi) / 2.0
        dm = f_deriv_of_alpha(mid)
        if dm < 0:
            lo = mid
        else:
            hi = mid

    best_alpha = (lo + hi) / 2.0
    best_val = f_of_alpha(best_alpha)

    # Local refinement
    for factor in [0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0]:
        trial = min(best_alpha * factor, max_alpha)
        v = f_of_alpha(trial)
        if v < best_val:
            best_val = v
            best_alpha = trial

    return best_alpha


def _peak_equalization(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    K = max(5, int((2 * n - 1) * 0.003))
    K_min = 3
    patience = 0
    max_patience = 300
    momentum = np.zeros(n)
    mom_coeff = 0.3

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            break

        n_total = 2 * n - 1
        n_active = min(K, n_total)
        active_idx = np.argpartition(c, -n_active)[-n_active:]

        # Weighted: emphasize peaks above median of active set
        c_active = c[active_idx]
        c_med = np.median(c_active)
        excess = np.maximum(c_active - c_med, 0)
        if excess.max() > 1e-18:
            w_active = (excess / excess.max()) ** 1.5 + 0.05
        else:
            w_active = np.ones(n_active)
        w_active = w_active / w_active.sum()

        w_full = np.zeros(n_total)
        w_full[active_idx] = w_active

        g = _grad_maxconv(a, w_full)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)

        if gnorm < 1e-18:
            break

        d = -g_centered / gnorm

        # Add momentum
        momentum = mom_coeff * momentum + (1.0 - mom_coeff) * d
        mnorm = np.linalg.norm(momentum)
        if mnorm > 1e-18:
            d_mom = momentum / mnorm
        else:
            d_mom = d

        # Try both pure gradient and momentum directions
        best_alpha = 0.0
        best_dir = d
        best_dir_sc = np.inf

        for trial_dir in [d, d_mom]:
            alpha = _exact_line_search(a, trial_dir, active_idx, max_alpha=1.0)
            if alpha > 1e-12:
                a_try = a + alpha * trial_dir
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_dir_sc:
                    best_dir_sc = sc_try
                    best_dir = trial_dir
                    best_alpha = alpha

        if best_alpha > 1e-12 and best_dir_sc < current_sc - 1e-15:
            a = a + best_alpha * best_dir
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        else:
            # Fallback: small step
            small_alpha = 0.01 / max(gnorm, 1e-10)
            a_try = a - small_alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-10:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-15:
                    a = a_try

        # Adapt K
        if patience % 30 == 0 and patience > 0:
            K = max(K_min, int(K * 0.85))
        if patience % 50 == 0 and K == K_min:
            K = max(K_min, 5)
            mom_coeff = min(0.5, mom_coeff + 0.05)

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.5 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-25.0 * (t - 0.3) ** 2) + np.exp(-25.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    K = 3
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        active_idx = np.argpartition(c, -K)[-K:]
        w_full = np.zeros(2 * n - 1)
        w_full[active_idx] = 1.0 / K

        g = _grad_maxconv(a, w_full)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)

        if gnorm < 1e-18:
            break

        d = -g_centered / gnorm
        alpha = _exact_line_search(a, d, active_idx, max_alpha=0.3)

        improved = False
        for trial_alpha in [alpha, alpha * 0.5, alpha * 0.2, 0.005, 0.001, 0.0003]:
            if trial_alpha < 1e-14:
                break
            a_try = a + trial_alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 300:
                a = best_a.copy()
                no_improve = 0

    return best_a, best_sc


def _hybrid_optimize(a_init, deadline, min_time_remaining=30.0):
    """Alternate between L-BFGS and peak equalization."""
    n = len(a_init)
    best_a = _normalize(a_init.copy()).copy()
    best_sc = _score(best_a)

    t_end = deadline - min_time_remaining
    block_time = 60.0

    while time.monotonic() < t_end:
        # L-BFGS block
        lbfgs_end = min(time.monotonic() + block_time, t_end)
        a_lb = _run_lbfgs(best_a, lbfgs_end, min_time_remaining=5)
        sc_lb = _score(a_lb)
        if sc_lb < best_sc - 1e-15:
            best_a = a_lb
            best_sc = sc_lb

        # Peak equalization block
        pe_end = min(time.monotonic() + block_time * 0.5, t_end)
        a_pe, sc_pe = _peak_equalization(best_a, pe_end, min_time_remaining=5)
        if sc_pe < best_sc - 1e-15:
            best_a = a_pe
            best_sc = sc_pe

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS with top-K smoothing, Iterative Peak
    Equalization with exact line search, hybrid alternating optimization,
    multi-length exploration, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, tapered n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS with aggressive annealing ===
    print(f"\n[P1] L-BFGS aggressive (t={time.monotonic()-t_start:.1f}s)")
    p1_dead = min(t_start + 350, deadline - 900)
    a1 = _run_lbfgs(best_seq, p1_dead, min_time_remaining=30)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs1] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Peak Equalization ===
    print(f"\n[P2] Peak Equalization (t={time.monotonic()-t_start:.1f}s)")
    p2_dead = min(t_start + 650, deadline - 600)
    a2, sc2 = _peak_equalization(best_seq, p2_dead, min_time_remaining=30)
    if sc2 < best_score:
        best_seq = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [peak_eq] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Hybrid (alternate L-BFGS + Peak Eq) ===
    print(f"\n[P3] Hybrid (t={time.monotonic()-t_start:.1f}s)")
    p3_dead = min(t_start + 950, deadline - 400)
    a3, sc3 = _hybrid_optimize(best_seq, p3_dead, min_time_remaining=30)
    if sc3 < best_score:
        best_seq = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} [hybrid] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Fine polish ===
    print(f"\n[P4] Fine polish (t={time.monotonic()-t_start:.1f}s)")
    p4_dead = min(t_start + 1150, deadline - 300)
    a4, sc4 = _fine_polish(best_seq, p4_dead, min_time_remaining=10)
    if sc4 < best_score:
        best_seq = a4.copy()
        best_score = sc4
        print(f"  NEW BEST: {best_score:.10f} [polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_main = len(best_seq)
    n_options = [32768, 24576, 8192]
    n_options = [x for x in n_options if x != n_main]

    for n_try in n_options:
        if time.monotonic() > deadline - 250:
            break
        a_res = _resample(best_seq, n_try)
        sub_dead = min(time.monotonic() + 70, deadline - 200)
        a_opt = _run_lbfgs(a_res, sub_dead, min_time_remaining=10)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [n={n_try}] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  n={n_try}: {sc_opt:.10f} (no imp) t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Diverse restarts ===
    print(f"\n[P6] Restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        sigmas = [0.015, 0.025, 0.04, 0.01, 0.035, 0.02, 0.05, 0.012]
        remaining_restarts = deadline - 100 - time.monotonic()
        n_restarts = min(len(sigmas), max(2, int(remaining_restarts / 35)))

        for i in range(n_restarts):
            if time.monotonic() > deadline - 80:
                break
            sigma = sigmas[i]
            a_pert = _perturb(best_seq, sigma, rng)
            sub_dead = min(time.monotonic() + 30, deadline - 60)
            a_r = _run_lbfgs(a_pert, sub_dead, min_time_remaining=5)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.10f} [r{i},s={sigma}] t={time.monotonic()-t_start:.1f}s")
            else:
                print(f"  R{i}(s={sigma}): {sc_r:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final polish + tiny L-BFGS ===
    print(f"\n[P7] Final (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3:
        a_fin, sc_fin = _fine_polish(best_seq, deadline - 2, min_time_remaining=2)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")

        remaining = deadline - 2 - time.monotonic()
        if remaining > 10:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = max(mb * 0.0003, 1e-10)
            K_fin = 3
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _TopkSmoProb(beta_fin, K_fin, n)
            max_iter = min(5000, max(50, int(remaining / 0.02)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-15, 'maxcor': 100})
                a_fin2 = np.exp(result.x - result.x.max())
                a_fin2 = a_fin2 / a_fin2.sum()
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
