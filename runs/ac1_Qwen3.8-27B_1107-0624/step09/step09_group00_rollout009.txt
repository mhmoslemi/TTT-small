def _lp_direction(a, c, n, n_constraints=300, M=None):
    """Find LP direction: max sum(b) s.t. (a*b)_k <= M for top lags, b>=0."""
    if M is None:
        M = float(c.max())
    # Select top lags by convolution value
    idx = np.argsort(c)[::-1][:n_constraints]
    idx = np.sort(idx)

    # Build constraint matrix: row i corresponds to lag idx[i]
    # (a*b)_k = sum_j a[j] * b[k-j] for valid j
    C = np.zeros((len(idx), n))
    for i, k in enumerate(idx):
        j_low = max(0, k - n + 1)
        j_high = min(n, k + 1)
        for j in range(j_low, j_high):
            C[i, j] = a[k - j]

    # LP: max 1^T b  s.t.  C b <= M, b >= 0
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=C, b_ub=np.full(len(idx), M),
                  bounds=[(0, None)] * n, method='highs')

    if res.status == 0 and res.x is not None:
        b = np.maximum(res.x, 0)
        s = b.sum()
        if s > 1e-10:
            return b / s
    return None

    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


T_START = 0.0


def propose_candidate(seed=42, budget_s=1800):
    global T_START
    T_START = time.monotonic()
    deadline = T_START + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    n_main = n_parent if n_parent > 0 else 16384

    # === Phase 1: Quick multi-start (150s) ===
    print(f"\n[P1] Multi-start (t={time.monotonic() - T_START:.1f}s)")
    ms_deadline = T_START + 150
    starts = [('parent', parent.copy())]
    for kind in ['hann', 'triangle', 'gaussian', 'raised_cos2', 'bimodal']:
        starts.append((kind, _make_profile(n_main, kind)))

    for name, a_init in starts:
        if time.monotonic() > ms_deadline - 20:
            break
        a_opt = _run_lbfgs(a_init, ms_deadline, min_time_remaining=30, n_betas=6)
        sc_opt = _score(a_opt)
        print(f"  {name}: opt={sc_opt:.10f}")
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [ms/{name}]")

    # === Phase 2: Intensive LBFGS (200s) ===
    print(f"\n[P2] Intensive LBFGS (t={time.monotonic() - T_START:.1f}s)")
    lbfgs_deadline = min(deadline - 600, time.monotonic() + 200)
    a_main = _run_lbfgs(best_seq, lbfgs_deadline, min_time_remaining=30, n_betas=15)
    sc_main = _score(a_main)
    if sc_main < best_score:
        best_seq = a_main.copy()
        best_score = sc_main
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_int]")

    # === Phase 3: LP-guided iterations (350s) ===
    print(f"\n[P3] LP-guided (t={time.monotonic() - T_START:.1f}s)")
    lp_deadline = min(deadline - 400, time.monotonic() + 350)
    a_lp, sc_lp = _lp_guided_iteration(best_seq, lp_deadline,
                                        n_lp_rounds=8, min_time_remaining=200)
    if sc_lp < best_score:
        best_seq = a_lp.copy()
        best_score = sc_lp
        print(f"  NEW BEST: {best_score:.10f} [lp_guided]")

    # === Phase 4: ILS (200s) ===
    print(f"\n[P4] ILS (t={time.monotonic() - T_START:.1f}s)")
    ils_deadline = min(deadline - 250, time.monotonic() + 200)
    a_ils, sc_ils = _stochastic_ils(best_seq, ils_deadline, rng, min_time_remaining=60)
    if sc_ils < best_score:
        best_seq = a_ils.copy()
        best_score = sc_ils
        print(f"  NEW BEST: {best_score:.10f} [ils]")

    # === Phase 5: Multi-n (150s) ===
    print(f"\n[P5] Multi-n (t={time.monotonic() - T_START:.1f}s)")
    n_candidates = [24576, 32768, 12288, 20480, 16384, 49152]
    seen = set()
    n_unique = []
    for nn in n_candidates:
        if nn not in seen and nn != len(best_seq):
            seen.add(nn)
            n_unique.append(nn)

    for n_try in n_unique:
        if time.monotonic() > deadline - 150:
            break
        a_init = _resample(best_seq, n_try)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=30, n_betas=6)
        sc_opt = _score(a_opt)
        print(f"  n={n_try}: score={sc_opt:.10f}")
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try})")

    # === Phase 6: Second LP round (150s) ===
    print(f"\n[P6] LP round 2 (t={time.monotonic() - T_START:.1f}s)")
    if time.monotonic() < deadline - 100:
        lp2_deadline = min(deadline - 80, time.monotonic() + 150)
        a_lp2, sc_lp2 = _lp_guided_iteration(best_seq, lp2_deadline,
                                              n_lp_rounds=4, min_time_remaining=60)
        if sc_lp2 < best_score:
            best_seq = a_lp2.copy()
            best_score = sc_lp2
            print(f"  NEW BEST: {best_score:.10f} [lp2]")

    # === Phase 7: Final ultra-polish (remaining) ===
    print(f"\n[P7] Final polish (t={time.monotonic() - T_START:.1f}s)")
    if time.monotonic() < deadline - 5:
        pol_deadline = min(deadline - 5, time.monotonic() + 60)
        a_fin, sc_fin = _nesterov_polish(best_seq, pol_deadline,
                                          tol_start=0.002, tol_end=0.000005,
                                          lr_init=0.01, momentum=0.85)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [nesterov_final]")

        remaining = deadline - 5 - time.monotonic()
        if remaining > 15:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.00005
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()

            def f_final(z_in):
                return _smoobj(z_in, beta_fin, n)

            def g_final(z_in):
                return _smograd(z_in, beta_fin, n)

            max_iter = min(5000, max(100, int(remaining / 0.02)))
            try:
                result = minimize(f_final, z, jac=g_final, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-15, 'maxcor': 100})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs]")
            except Exception:
                pass

    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - T_START:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
