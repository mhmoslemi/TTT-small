Hybrid optimizer combining Adam gradient descent, MCMC exploration via
mass-transfer proposals, adaptive peak equalization, and multi-n search.
Key innovation: O(n) MCMC updates using incremental convolution tracking,
enabling millions of exploration steps to escape local minima that trap

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_normalized(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max())


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a.copy()
    return a / s


def _adam_optimize(a_init, deadline, min_time=30.0, lr_init=0.004,
                   temp_scale=0.003, top_k_max=30):
    """
    Adam optimizer with adaptive soft-peak selection.
    Works with normalized sequences (sum=1).
    """
    n = len(a_init)
    a = a_init.copy()
    assert abs(a.sum() - 1.0) < 1e-6

    best_a = a.copy()
    best_sc = _score_normalized(a)

    m_t = np.zeros(n)
    v_t = np.zeros(n)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    t = 0
    no_improve = 0
    rng = np.random.default_rng(12345)

    # Periodic reset schedule
    reset_interval = 300

    while time.monotonic() < deadline - min_time:
        t += 1
        lr = lr_init / (1.0 + 0.0005 * t)

        c = _fft_conv(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > reset_interval:
            m_t[:] = 0
            v_t[:] = 0
            t = 0
            # Small random perturbation in log-space
            z = np.log(np.maximum(a, 1e-15))
            z += 0.02 * rng.standard_normal(n)
            a = np.exp(z - np.max(z))
            a = _normalize(a)
            no_improve = 0

        sorted_c = np.sort(c)[::-1]

            w[int(np.argmax(c))] = 1.0

        else:
            a_new = a_new / s_new

        c_new = _fft_conv(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < sc:
            a = a_new
        else:
            # Try half step
            a_half = a - 0.5 * step
            a_half = np.maximum(a_half, 0.0)
            s_half = a_half.sum()
            if s_half > 1e-15:
                a_half = a_half / s_half
                c_half = _fft_conv(a_half, a_half)
                sc_half = 2.0 * n * float(c_half.max())
                if sc_half < sc:
                    a = a_half

    return best_a, best_sc


def _mcmc_explore(a_init, deadline, min_time=30.0, n_steps_target=2000000):
    """
    MCMC exploration using mass-transfer proposals with O(n) updates.
    Preserves sum(a)=1 and a>=0 exactly.
    """
    n = len(a_init)
    a = a_init.copy()
    c = _fft_conv(a, a).copy()
    c_max = float(c.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c_max

    rng = np.random.default_rng(99999)

    # Temperature schedule
    T_init = c_max * 0.01
    T_min = c_max * 1e-6
    cooling = 0.99995

    # Proposal scale
    sigma = 1.0 / (n * 5)

    T = T_init
    accepted = 0
    total = 0
    last_best_update = 0

    while time.monotonic() < deadline - min_time:
        # Batch of 100 steps for efficiency
        batch = 100
        for _ in range(batch):
            if time.monotonic() > deadline - min_time:
                    break
            total += 1

            # Pick two distinct indices
            j = rng.integers(0, n)
            k = rng.integers(0, n)
            if j == k:
                k = (k + 1) % n

            # Propose transfer
            max_delta = min(a[j], a[k]) * 0.5
            if max_delta < 1e-15:
                continue
            delta = rng.normal(0, sigma) * max(c_max, 1e-10)
            delta = np.clip(delta, -max_delta, max_delta)
            if abs(delta) < 1e-18:
                continue

            # Compute change in convolution: delta_c[m] = 2*delta*(a[m-j] - a[m-k]) (approx)
            # More precisely:
            # c[m] = sum_i a[i]*a[m-i]
            # Changing a[j] += delta, a[k] -= delta:
            # delta_c[m] = delta*(a[m-j] + a[j + (m-j)]) ... let me be careful
            # Term i=j: a[j]*a[m-j] -> (a[j]+delta)*a[m-j], change = delta*a[m-j]
            # Term where m-i=j, i=m-j: a[m-j]*a[j] -> a[m-j]*(a[j]+delta), change = delta*a[m-j]
            #   (only if m-j != j, i.e., m != 2j; if m=2j, the term is a[j]^2 -> (a[j]+delta)^2, 
            #    change = 2*a[j]*delta + delta^2, but we counted 2*delta*a[j] above, so extra delta^2)
            # Similarly for a[k] -= delta:
            #   change = -delta*a[m-k] - delta*a[m-k] = -2*delta*a[m-k]
            #   (with similar edge case for m=2k)
            #
            # For simplicity and speed:
            # delta_c[m] = 2*delta*a[m-j] - 2*delta*a[m-k] (ignoring O(delta^2) terms and edge cases)

            # a[m-j] for m=0..2n-2: this is a shifted version of a
            # a[m-j] is valid when 0 <= m-j < n, i.e., j <= m <= j+n-1
            # a[m-k] is valid when k <= m <= k+n-1

            # Vectorized computation
            shift_j = np.zeros(2 * n - 1)
            shift_k = np.zeros(2 * n - 1)
            # a[m-j] for m in [j, j+n-1]
            shift_j[j:j+n] = a
            # a[m-k] for m in [k, k+n-1]
            shift_k[k:k+n] = a

            delta_c = 2.0 * delta * (shift_j - shift_k)
            c_new = c + delta_c
            c_new_max = float(c_new.max())
            sc_new = 2.0 * n * c_new_max
            sc_old = 2.0 * n * c_max

            # Metropolis test
            delta_F = sc_new - sc_old
            if delta_F <= 0:
                accept = True
            else:
                accept = (rng.random() < np.exp(-delta_F / max(T, 1e-30)))

            if accept:
                a[j] += delta
                a[k] -= delta
                c = c_new
                c_max = c_new_max
                accepted += 1

                if sc_new < best_sc - 1e-14:
                    best_sc = sc_new
                    best_a = a.copy()
                    last_best_update = total

        # Cool down
        T *= cooling
        if T < T_min:
            T = T_min

        # Progress report every ~100k steps
        if total % 200000 == 0:
            el = time.monotonic()
            print(f"    MCMC: steps={total}, accept_rate={accepted/max(total,1):.3f}, "
                  f"best={best_sc:.10f}, T={T:.2e}")

    return best_a, best_sc


def _peak_equalize(a_init, deadline, min_time=10.0):
    """
    Direct peak equalization: solve linearized system to equalize top peaks.
    Multiple iterations with decreasing residual tolerance.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_normalized(a)

    rng = np.random.default_rng(777)
    lr_base = 0.5
    no_improve = 0

    while time.monotonic() < deadline - min_time:
        c = _fft_conv(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break

        # Identify top K peaks
        K = min(20, 2 * n - 1)
        top_k = np.argsort(c)[::-1][:K]
        top_vals = c[top_k]

        # Target: equalize to the minimum of the top K (reduce the max)
        target = float(np.median(top_vals[:min(10, len(top_vals))]))

        # Build gradient: average of shifted a's at peak locations
        w = np.zeros(2 * n - 1)
        w[top_k] = 1.0 / K
        a_rev = a[::-1]
        h = _fft_conv(w, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad -= grad.mean()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Line search
        lr = lr_base / (gnorm * np.sqrt(n) + 1e-15)
        lr = min(lr, 0.1)
        improved = False
        for _ in range(20):
            a_try = a - lr * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                lr *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _fft_conv(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < sc - 1e-14:
                a = a_try
                improved = True
                break
            lr *= 0.5

        if not improved:
            lr_base *= 0.9

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + 0.8 * np.exp(-20.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 5.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 11.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid optimizer:
    P1: Adam gradient (quick convergence)
    P2: MCMC exploration (escape local minima)
    P3: Peak equalization (fine-tune active set)
    P4: Adam gradient (exploit)
    P5: Multi-n search
    P6: Final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score_normalized(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score_normalized(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Quick Adam optimization ===
    print(f"\n[P1] Adam warmup (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 1200, t_start + 200)
    a1, sc1 = _adam_optimize(parent, p1_deadline, min_time=5.0, lr_init=0.005)
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [adam_warmup]")
    else:
        print(f"  No improvement: {sc1:.10f} vs {best_score:.10f}")

    # === Phase 2: MCMC exploration ===
    print(f"\n[P2] MCMC exploration (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 700, t_start + 900)
    a2, sc2 = _mcmc_explore(best_seq, p2_deadline, min_time=30.0)
    if sc2 < best_score:
        best_seq = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [mcmc]")
    else:
        print(f"  MCMC best: {sc2:.10f} (no improvement over {best_score:.10f})")

    # === Phase 3: Peak equalization ===
    print(f"\n[P3] Peak equalization (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 400, t_start + 1200)
    a3, sc3 = _peak_equalize(best_seq, p3_deadline, min_time=20.0)
    if sc3 < best_score:
        best_seq = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} [equalize]")
    else:
        print(f"  Equalize: {sc3:.10f} (no improvement)")

    # === Phase 4: Adam exploitation ===
    print(f"\n[P4] Adam exploit (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 200, t_start + 1500)
    a4, sc4 = _adam_optimize(best_seq, p4_deadline, min_time=20.0, lr_init=0.002)
    if sc4 < best_score:
        best_seq = a4.copy()
        best_score = sc4
        print(f"  NEW BEST: {best_score:.10f} [adam_exploit]")
    else:
        print(f"  Adam exploit: {sc4:.10f} (no improvement)")

    # === Phase 5: Multi-n search ===
    print(f"\n[P5] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_parent = len(best_seq)
    n_candidates = sorted(set([n_parent, 16384, 20000, 24576, 12288]),
                          key=lambda x: abs(x - n_parent))
    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == n_parent:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score_normalized(a_init)
        print(f"  n={n_try}: initial score={sc_init:.10f}")
        # Quick optimize
        a_opt, sc_opt = _adam_optimize(a_init, deadline, min_time=50.0, lr_init=0.003)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n]")
        else:
            print(f"  n={n_try}: best={sc_opt:.10f} (no improvement)")

    # === Phase 6: Final polish ===
    print(f"\n[P6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 5 - time.monotonic()
    if remaining > 20:
        a_fin, sc_fin = _peak_equalize(best_seq, deadline - 3, min_time=3.0)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_equalize]")

        remaining2 = deadline - 3 - time.monotonic()
        if remaining2 > 15:
            a_fin2, sc_fin2 = _adam_optimize(best_seq, deadline - 2, min_time=2.0, lr_init=0.001)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_adam]")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        # Compute final score using the actual evaluate function
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"FINAL: evaluate_sequence={ev_score:.10f}, n={len(final)}, "
                  f"time={time.monotonic()-t_start:.1f}s")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
            sc_final = _score_normalized(_normalize(final))
            print(f"FINAL (internal): {sc_final:.10f}, n={len(final)}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score_normalized(fallback):.10f}")
    return [float(x) for x in fallback]
