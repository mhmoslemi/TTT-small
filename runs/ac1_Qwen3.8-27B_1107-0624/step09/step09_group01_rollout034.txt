"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp smoothed gradient descent with Nesterov momentum in log-space,
adaptive temperature with reheating, active-set Newton refinement via FFT-based
Hankel solvers, parametric initialization (sum of Gaussians), and multi-scale
exploration. Designed to beat 1.5027 within 1800s budget.
"""
import numpy as np
import time
import sys


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute H @ v where H_{i,k} = W[i+k] for i,k in [0,n-1]."""
    v_rev = v[::-1]
    N = _next_pow2(2 * n - 1)
    F_v = np.fft.rfft(v_rev, N)
    F_W = np.fft.rfft(W[:2 * n - 1], N)
    conv = np.fft.irfft(F_v * F_W, N)[:2 * n - 1]
    return conv[n - 1:2 * n - 1]


def _cg_hankel(W, n, rhs, max_iter=200, tol=1e-10):
    """Solve H x = rhs using conjugate gradient."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = max(rs, 1e-30)
    if rs < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new) or rs_new > rs * 1.01:
            break
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _compute_grad_smoothed(a, p, n):
    """Compute gradient of sum_k p_k*(a*a)_k w.r.t. a."""
    a_rev = a[::-1]
    conv = _fft_convolve(p, a_rev)
    return 2.0 * conv[n - 1:2 * n - 1]


def _parametric_init(n, n_components=5, rng=None):
    """Initialize as sum of Gaussians."""
    if rng is None:
        rng = np.random.default_rng(42)
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    for _ in range(n_components):
        mu = rng.uniform(0.1, 0.9)
        sig = rng.uniform(0.05, 0.3)
        amp = rng.uniform(0.5, 2.0)
        a += amp * np.exp(-0.5 * ((x - mu) / sig) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _tapered_init(n, power=0.7):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _asymmetric_init(n, skew=0.3):
    """Create an asymmetric tapered sequence."""
    t = np.linspace(0, 1, n)
    center = 0.5 + skew
    a = np.exp(-3.0 * (t - center) ** 2) + 0.01
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _optimize_logspace(a_init, deadline, beta_init=None, max_iter=100000,
                       lr_init=None, momentum_init=0.9, print_interval=5000):
    """
    Log-sum-exp smoothed gradient descent in log-space with Nesterov momentum.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-30)
    a = a / a.sum()
    z = np.log(a)
    z = z - z.mean()

    c0 = _fft_convolve(a, a)
    c0_max = float(c0.max())
    if beta_init is None:
        beta = c0_max * 0.05
    else:
        beta = beta_init

    if lr_init is None:
        lr = 0.5 / max(n // 1000, 1)
    momentum = momentum_init
    v = np.zeros(n)

    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0
    best_beta = beta

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Compute attention weights
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300:
            p = np.zeros_like(c)
            p[int(np.argmax(c))] = 1.0
        else:
            p = e / s

        # Gradient
        G = _compute_grad_smoothed(a, p, n)
        G_centered = G - np.dot(G, a)
        grad_z = a * G_centered
        grad_norm = float(np.linalg.norm(grad_z))

        if grad_norm < 1e-18 * n:
            break

        # Nesterov momentum update
        v = momentum * v - lr * grad_z / (grad_norm + 1e-30)
        z = z + v
        z = np.clip(z, -100, 100)
        z = z - z.mean()

        # Adaptive learning rate
        if it % 100 == 99:
            a_check = _softmax(z)
            c_check = _fft_convolve(a_check, a_check)
            sc_check = float(2.0 * n * c_check.max() / a_check.sum()**2)
            if sc_check < best_sc - 1e-12:
                best_sc = sc_check
                best_a = a_check.copy()
                no_improve = 0
                lr = min(lr * 1.02, 2.0 / max(n // 500, 1))
            else:
                no_improve += 1
                lr *= 0.9
                if no_improve > 200:
                    # Reheat
                    beta = c_max * 0.1
                    lr = 0.5 / max(n // 1000, 1)
                    v = np.zeros(n)
                    no_improve = 0

        # Cool
        if no_improve % 50 == 0 and no_improve > 0:
            beta *= 0.92
        if beta < c_max * 1e-8:
            beta = c_max * 1e-8

        if it % print_interval == print_interval - 1:
            print(f"    logspace iter {it}: cmax={c_max:.8f}, beta={beta:.6f}, "
                  f"best_sc={best_sc:.10f}, lr={lr:.4e}")

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _active_set_newton(a_init, deadline, max_rounds=200, k_schedule=None):
    """
    Active-set Newton method: iterate on (a, W) pairs where W is a probability
    over active convolution positions. Solve the resulting linear system.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    if k_schedule is None:
        k_schedule = [1, 2, 3, 5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Adaptive tolerance for active set
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(200, len(c_sorted) - 1)]
        tol = max(gap * 0.5, c_max * 1e-6)
        active = c >= (c_max - tol)

        K = k_schedule[min(rnd, len(k_schedule) - 1)]
        if active.sum() > K * 2:
            active = c >= (c_max - max(gap * 1.0, c_max * 0.05))
            if active.sum() > K:
                idx = np.argpartition(c, -K)[-K:]
                active = np.zeros_like(c, dtype=bool)
                active[idx] = True

        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        W = np.zeros_like(c)
        W[active] = 1.0 / active.sum()

        # Compute gradient direction
        G = _compute_grad_smoothed(a, W, n)
        G_centered = G - np.dot(G, a)
        gn = float(np.linalg.norm(G_centered))
        if gn < 1e-15 * n:
            break

        # Try Newton step: solve H x = -G_centered (or similar)
        # H is the Hessian of the quadratic form a^T W_conv a
        # For the active set W, the "Hessian" applied to v is:
        # (H v)_i = 2 * (W * v_rev)[i+n-1] - (projection onto sum=1 hyperplane)
        try:
            Hg = _cg_hankel(W[:2*n-1], n, G_centered, max_iter=100)
            H1 = _cg_hankel(W[:2*n-1], n, np.ones(n), max_iter=100)
            H1_denom = float(np.ones(n) @ H1)
            if abs(H1_denom) > 1e-20 and np.isfinite(H1_denom):
                lam = float(np.ones(n) @ Hg) / H1_denom
                dx = -(Hg - lam * H1)
                dx = dx - np.mean(dx)
            else:
                dx = -G_centered / (gn + 1e-30)
        except Exception:
            dx = -G_centered / (gn + 1e-30)

        dx_norm = float(np.linalg.norm(dx))
        if dx_norm < 1e-30 or not np.isfinite(dx_norm):
            dx = -G_centered / (gn + 1e-30)
            dx_norm = float(np.linalg.norm(dx))

        # Line search
        alpha = 1.0
        step_done = False
        for _ in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _fft_convolve(a_try, a_try)
            if float(c_try.max()) < c_max * (1.0 - 0.005 * alpha) - 1e-15:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-18:
                break

        if not step_done:
            # Subgradient fallback
            for trial in [0.1, 0.05, 0.01, 0.005, 0.001]:
                a_try = a - trial * G_centered / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _fft_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-15:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 15:
            break

    return best_a, best_sc


def _resample_sequence(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)
