```python
def _qp_peak_shave(a, deadline, min_remaining=10.0):
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * float(_full_convolve(a, a).max())
    
    max_active = 300  # maximum active set size
    min_active = 3
    tol_rel = 1e-4  # relative tolerance for active set
    
    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        score = 2.0 * n * c_max
        
        if score < best_score * (1 - 1e-10):
            best_score = score
            best_a = a.copy()
        
        # Find active set
        threshold = c_max * (1 - tol_rel)
        S = np.where(c >= threshold)[0]
        if len(S) > max_active:
            S = np.argsort(c)[::-1][:max_active]
        
        # Solve QP
        delta, M_new = _solve_qp(a, S, c)
        if delta is None:
            break
        
        # Check if the predicted improvement is real
        # a_new = a + delta should have sum = sum(a) (by construction)
        # and conv(a_new, a_new)[k_i] ≈ M_new for k_i in S
        
        # Line search
        improved = False
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a_try = a + frac * delta
            if np.any(a_try < 0):
                continue
            s_try = a_try.sum()
            if abs(s_try - a.sum()) > 1e-10:
                a_try = a_try * (a.sum() / s_try)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < score * (1 - 1e-12):
                a = a_try
                improved = True
                break
        
        if not improved:
            # Reduce tolerance or break
            tol_rel *= 0.5
            if tol_rel < 1e-8:
                break
    
    return best_a, best_score
```

```python
            return 0.0
        return float(prefix[hi+1] - prefix[lo])
    
    # Build G (m+1 x m+1)
    G = np.zeros((m+1, m+1))
    b = np.zeros(m+1)
    
    for i in range(m):
        b[i] = -0.5 * c_S[i]
        lo_i = max(0, S[i] - (n-1))
        hi_i = min(S[i], n-1)
        for j in range(i, m):
            lo_j = max(0, S[j] - (n-1))
            hi_j = min(S[j], n-1)
            lo = max(lo_i, lo_j)
            hi = min(hi_i, hi_j)
            G[i,j] = range_sum(prefix_a2, lo, hi) + 0.25
            G[j,i] = G[i,j]
        G[i, m] = range_sum(prefix_a, lo_i, hi_i)
        G[m, i] = G[i, m]
    
    G[m, m] = float(n)
    b[m] = 0.0
    
    # Solve
    try:
        y = np.linalg.solve(G, b)
    except np.linalg.LinAlgError:
        y = np.linalg.lstsq(G, b, rcond=None)[0]
    
    # Compute delta
    # delta[l] = a[l] * W[l] + y[m]
    # W[l] = sum_{i: l in [lo_i, hi_i]} y[i]
    
    W = np.zeros(n)
    for i in range(m):
        lo_i = max(0, S[i] - (n-1))
        hi_i = min(S[i], n-1)
        W[lo_i:hi_i+1] += y[i]
    
    delta = a * W + y[m]
    
    # Verify sum(delta) ≈ 0
    # (should be 0 by construction)
    
    M_new = -0.5 * float(np.sum(y[:m]))
    
    return delta, M_new
```

```python
"""
Optimizer using QP-based active set peak shaving combined with L-BFGS.
Key innovation: exact linearized QP solve for equalizing active peaks via
minimum-norm solution, which is more efficient and precise than gradient-based
peak shaving for the final polish. Uses prefix sums for O(m^2) Gram matrix
construction and difference arrays for O(n) delta computation.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _qp_solve(a, S, c):
    """Solve the linearized QP: find min-norm delta such that
    conv(a+delta, a+delta)[k_i] = M_new for k_i in S, sum(delta)=0.
    Returns (delta, M_new) or (None, None) on failure."""
    n = len(a)
    m = len(S)
    if m == 0:
        return None, None
    S = np.asarray(S, dtype=int)
    c_S = c[S].astype(float)
    
    # Prefix sums for efficient range queries
    prefix_a = np.concatenate([[0.0], np.cumsum(a)])
    a2 = a * a
    prefix_a2 = np.concatenate([[0.0], np.cumsum(a2)])
    
    def range_sum(prefix, lo, hi):
        lo = max(0, lo)
        hi = min(n - 1, hi)
        if lo > hi:
            return 0.0
        return float(prefix[hi + 1] - prefix[lo])
    
    # Valid range for peak at position k: l in [max(0, k-n+1), min(k, n-1)]
    lo_bounds = np.maximum(0, S - (n - 1))
    hi_bounds = np.minimum(S, n - 1)
    
    # Build Gram matrix G = A*A^T of size (m+1) x (m+1)
    G = np.zeros((m + 1, m + 1))
    b_vec = np.zeros(m + 1)
    
    for i in range(m):
        b_vec[i] = -0.5 * c_S[i]
        for j in range(i, m):
            lo = max(lo_bounds[i], lo_bounds[j])
            hi = min(hi_bounds[i], hi_bounds[j])
            val = range_sum(prefix_a2, lo, hi) + 0.25
            G[i, j] = val
            G[j, i] = val
        row_sum = range_sum(prefix_a, int(lo_bounds[i]), int(hi_bounds[i]))
        G[i, m] = row_sum
        G[m, i] = row_sum
    
    G[m, m] = float(n)
    b_vec[m] = 0.0
    
    # Solve G * y = b
    try:
        chol = np.linalg.cholesky(G + 1e-14 * np.eye(m + 1))
        y = np.linalg.solve(chol.T, np.linalg.solve(chol, b_vec))
    except np.linalg.LinAlgError:
        try:
            y = np.linalg.solve(G, b_vec)
        except np.linalg.LinAlgError:
            y, _, _, _ = np.linalg.lstsq(G, b_vec, rcond=None)
    
    # Compute delta[l] = a[l] * W[l] + y[m]
    # W[l] = sum_{i: lo_i <= l <= hi_i} y[i]
    # Use difference array for O(m + n) computation
    W = np.zeros(n + 1)
    for i in range(m):
        W[int(lo_bounds[i])] += y[i]
        W[int(hi_bounds[i]) + 1] -= y[i]
    W = np.cumsum(W)[:n]
    
    delta = a * W + y[m]
    M_new = -0.5 * float(np.sum(y[:m]))
    
    return delta, M_new


def _qp_peak_shave(a_init, deadline, min_remaining=15.0, max_active=200, tol_rel=1e-5):
    """Iterative QP-based peak shaving. Returns (best_a, best_score)."""
    n = len(a_init)
    a = a_init.copy().astype(float)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * float(_full_convolve(a, a).max())
    
    tol = tol_rel
    no_improve = 0
    max_no_improve = 500
    
    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        score = 2.0 * n * c_max
        
        if score < best_score * (1.0 - 1e-12):
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        
        # Find active set
        threshold = c_max * (1.0 - tol)
        S = np.where(c >= threshold)[0]
        if len(S) == 0:
            S = np.array([int(np.argmax(c))])
        if len(S) > max_active:
            S = np.argsort(c)[::-1][:max_active]
        
        m = len(S)
        
        # Solve QP
        delta, M_new = _qp_solve(a, S, c)
        if delta is None:
            break
        
        # Check if M_new is reasonable (should be <= c_max)
        if M_new > c_max * 1.01:
            tol *= 0.7
            continue
        
        # Line search
        improved = False
        for frac in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            a_try = a + frac * delta
            if np.any(a_try < -1e-15):
                continue
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try * (a.sum() / a_try.sum())
            # Actually just normalize to preserve sum
            a_try = a_try / a_try.sum() * a.sum()
            
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < score * (1.0 - 1e-13):
                a = a_try
                improved = True
                break
        
        if not improved:
            tol *= 0.8
            if tol < 1e-10:
                # Try expanding active set
                tol = max(tol_rel, 1e-4)
                max_active = min(500, max_active + 50)
                no_improve = max(0, no_improve - 100)
                if no_improve > max_no_improve:
                    break
    
    return best_a, best_score


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003) * beta_scale

    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625,
        0.03125, 0.015625, 0.0078125, 0.00390625,
        0.00195, 0.000977, 0.000488, 0.000244,
        0.000122, 0.000061, 0.00003, 0.000015, 0.0000075
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(30, int(remaining / (est_time_per_iter * 10))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-17,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _asymmetric_perturb(a, rng, strength=0.05):
    n = len(a)
    t = np.arange(n) / n
    ramp = 0.3 * (t - 0.5)
    noise = 0.7 * rng.standard_normal(n)
    perturbation = ramp + noise
    perturbation = perturbation - perturbation.mean()
    z = np.log(np.maximum(a, 1e-30))
    z = z + strength * perturbation
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase optimizer: QP peak shaving + L-BFGS + QP refinement.
    Key innovation: exact linearized QP for active-set peak equalization."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    n = len(best_seq)

    # === Phase 1: Initial QP peak shaving ===
    print(f"\n[P1] QP peak shaving (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 800.0, time.monotonic() + 350.0)
    a_q1, sc_q1 = _qp_peak_shave(best_seq, p1_deadline, min_remaining=30.0,
                                  max_active=150, tol_rel=1e-5)
    if sc_q1 < best_score:
        best_seq = a_q1.copy()
        best_score = sc_q1
        print(f"  QP1: {best_score:.10f} (improved)")
    else:
        print(f"  QP1: no improvement (best={best_score:.10f})")

    # === Phase 2: L-BFGS with diverse betas ===
    print(f"\n[P2] L-BFGS optimization (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 500.0, time.monotonic() + 350.0)
    a_l1 = _run_lbfgs(best_seq, p2_deadline, min_time_remaining=30.0, beta_scale=0.5)
    sc_l1 = _score(a_l1)
    if sc_l1 < best_score:
        best_seq = a_l1.copy()
        best_score = sc_l1
        print(f"  L-BFGS1: {best_score:.10f} (improved)")
    else:
        print(f"  L-BFGS1: no improvement (best={best_score:.10f})")

    # === Phase 3: Asymmetric perturbation + QP ===
    print(f"\n[P3] Asymmetric perturbation + QP (t={time.monotonic()-t_start:.1f}s)")
    for trial in range(3):
        if time.monotonic() > deadline - 350:
            break
        strength = 0.02 * (trial + 1)
        a_asym = _asymmetric_perturb(best_seq, rng, strength=strength)
        # Quick L-BFGS warmup
        p3a_deadline = min(deadline - 350.0, time.monotonic() + 80.0)
        a_warm = _run_lbfgs(a_asym, p3a_deadline, min_time_remaining=30.0, beta_scale=0.3)
        # Then QP
        p3b_deadline = min(deadline - 300.0, time.monotonic() + 80.0)
        a_q, sc_q = _qp_peak_shave(a_warm, p3b_deadline, min_remaining=20.0,
                                    max_active=200, tol_rel=1e-5)
        sc_total = _score(a_q)
        if sc_total < best_score:
            best_seq = a_q.copy()
            best_score = sc_total
            print(f"  Asym+QP trial {trial}: {best_score:.10f} (improved)")
        else:
            print(f"  Asym+QP trial {trial}: {sc_total:.10f} (no improvement)")

    # === Phase 4: QP with larger active set ===
    print(f"\n[P4] QP with larger active set (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 300.0, time.monotonic() + 250.0)
    a_q2, sc_q2 = _qp_peak_shave(best_seq, p4_deadline, min_remaining=30.0,
                                  max_active=400, tol_rel=5e-6)
    if sc_q2 < best_score:
        best_seq = a_q2.copy()
        best_score = sc_q2
        print(f"  QP2: {best_score:.10f} (improved)")
    else:
        print(f"  QP2: no improvement (best={best_score:.10f})")

    # === Phase 5: Fine L-BFGS ===
    print(f"\n[P5] Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p5_deadline = min(deadline - 200.0, time.monotonic() + 200.0)
    a_l2 = _run_lbfgs(best_seq, p5_deadline, min_time_remaining=30.0, beta_scale=0.2)
    sc_l2 = _score(a_l2)
    if sc_l2 < best_score:
        best_seq = a_l2.copy()
        best_score = sc_l2
        print(f"  L-BFGS2: {best_score:.10f} (improved)")
    else:
        print(f"  L-BFGS2: no improvement (best={best_score:.10f})")

    # === Phase 6: Final QP polish ===
    print(f"\n[P6] Final QP polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 30.0 - time.monotonic()
    if remaining > 20:
        a_q3, sc_q3 = _qp_peak_shave(best_seq, time.monotonic() + remaining,
                                      min_remaining=20.0, max_active=500, tol_rel=1e-6)
        if sc_q3 < best_score:
            best_seq = a_q3.copy()
            best_score = sc_q3
            print(f"  QP3: {best_score:.10f} (improved)")
        else:
            print(f"  QP3: no improvement (best={best_score:.10f})")

    # === Phase 7: One more L-BFGS if time remains ===
    remaining = deadline - 10.0 - time.monotonic()
    if remaining > 30:
        print(f"\n[P7] Final L-BFGS (t={time.monotonic()-t_start:.1f}s)")
        a_l3 = _run_lbfgs(best_seq, time.monotonic() + remaining,
                           min_time_remaining=10.0, beta_scale=0.1)
        sc_l3 = _score(a_l3)
        if sc_l3 < best_score:
            best_seq = a_l3.copy()
            best_score = sc_l3
            print(f"  L-BFGS3: {best_score:.10f} (improved)")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
