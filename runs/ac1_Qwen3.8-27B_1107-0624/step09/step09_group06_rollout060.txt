def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _smooth_obj_and_grad(z, beta, n):
    """Compute smooth objective and gradient w.r.t. z (log-parameterization)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        w = e / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    # Gradient: grad_a[i] = 4n * (w * a_rev)[i + n - 1]
    a_rev = a[::-1]
    N = _next_pow2(len(w) + n - 1)
    Wf = np.fft.rfft(w, N)
    Af = np.fft.rfft(a_rev, N)
    g_full = np.fft.irfft(Wf * Af, N)
    g = g_full[n - 1:2 * n - 1]
    grad_a = 4.0 * n * g
    dot_val = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - dot_val)
    return obj, grad_z, a

def _adam_optimize(a_init, deadline, min_time_remaining=30.0, seed=0):
    """Adam optimizer with smooth log-sum-exp objective."""
    n = len(a_init)

    t = 0

    c_sorted = np.sort(c_cur)[::-1]

    stall_count = 0

            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0

        m = beta1 * m + (1 - beta1) * grad_z

class _LbfgsSmooth:

    def grad(self, z):

            # Check convergence: if L-BFGS converged (not maxiter), reduce beta
            if result.status == 1:  # convergence
                c_check = _full_convolve(a, a)
                mb_check = float(c_check.max())
                c_s = np.sort(c_check)[::-1]
                gap_check = mb_check - c_s[min(50, len(c_s) - 1)]
                if gap_check < beta and beta > min_beta:
                    beta = max(min_beta, gap_check * 0.7)
                else:
                    beta = max(min_beta, beta * 0.5)
            else:
                # Didn't fully converge - reduce beta slightly
                beta = max(min_beta, beta * 0.7)

        except Exception as e:
            print(f"    L-BFGS error: {e}")
            beta = max(min_beta, beta * 0.5)

        if beta <= min_beta * 1.01:
            break

    return a


def _pairwise_local_search(a_init, deadline, min_time_remaining=30.0, seed=0):
    """Local search using pairwise mass transfers with exact 1D optimization."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(seed)

    no_improve = 0
    max_no_improve = 500

    while time.monotonic() + min_time_remaining < deadline:
        # Get current convolution
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        # Identify active peaks (top K)
        K = max(10, min(200, n // 100))
        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]

        # Try random pairwise transfers
        n_pairs = max(50, min(500, n // 50))
        improved = False

        for _ in range(n_pairs):
            p = rng.integers(0, n)
            q = rng.integers(0, n)
            if p == q:
                continue
            # Choose direction: move mass from higher a to lower a (or random)
            if rng.random() < 0.7:
                # Move from larger to smaller
                if a[p] < a[q]:
                    p, q = q, p
            if a[q] < 1e-15:
                continue

            # Compute the change in c for transfer t from q to p
            # c'[k] = c[k] + 2t*(a[k-p] - a[k-q]) + t^2*(delta stuff)
            # For the active peaks, find optimal t by bisection

            # Slopes for active peaks
            slopes = np.zeros(K)
            for i, k in enumerate(active):
                val_p = a[k - p] if 0 <= k - p < n else 0.0
                val_q = a[k - q] if 0 <= k - q < n else 0.0
                slopes[i] = 2.0 * (val_p - val_q)

            # We want to minimize max(active c'[k])
            # c'[k_i] = c[k_i] + slopes[i] * t (ignoring quadratic for now)
            # The optimal t makes the max as small as possible
            # This is: minimize max_i (c[k_i] + slopes[i] * t)
            # Optimal t is where the two "tightest" constraints are equal

            c_active = c[active]
            # Binary search for optimal t in [-a[p], a[q]]
            t_lo = max(-a[p] * 0.5, -0.01)
            t_hi = min(a[q] * 0.5, 0.01)

            if t_lo >= t_hi:
                continue

            # Use ternary search (objective is convex in t)
            for _iter in range(30):
                t1 = t_lo + (t_hi - t_lo) / 3.0
                t2 = t_hi - (t_hi - t_lo) / 3.0

                # Evaluate max of active peaks at t1, t2
                c1 = c_active + slopes * t1
                c2 = c_active + slopes * t2
                # Also check non-active peaks near max
                if c1.max() < c2.max():
                    t_hi = t2
                else:
                    t_lo = t1

            t_opt = (t_lo + t_hi) / 2.0

            # Apply and verify
            a[p] += t_opt
            a[q] -= t_opt
            if a[p] < 0 or a[q] < 0:
                a[p] -= t_opt
                a[q] += t_opt
                continue

            s_new = a.sum()
            if s_new < 1e-10:
                continue
            a = a / s_new

            c_new = _full_convolve(a, a)
            sc_new = 2.0 * n * float(c_new.max())
            if sc_new < current_sc - 1e-15:
                improved = True
                current_sc = sc_new
            else:
                # Revert
                a[p] -= t_opt
                a[q] += t_opt
                a = a / a.sum()

        if not improved:
            no_improve += 100

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(42)
        a = np.abs(rng.standard_normal(n)) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _annealing_restart(a_init, deadline, min_time_remaining=30.0, seed=0):
    """Simulated annealing style: perturb and accept/reject."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(seed)

    T = 0.01  # Initial "temperature" (perturbation size)
    T_min = 0.001
    T_decay = 0.995
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        # Generate perturbation
        perturb = T * rng.standard_normal(n)
        a_cand = a + perturb
        a_cand = _project_simplex(a_cand)

        sc_cand = _score(a_cand)
        sc_cur = _score(a)

        if sc_cand < sc_cur:
            a = a_cand
            if sc_cand < best_sc:
                best_sc = sc_cand
                best_a = a.copy()
                no_improve = 0
            T = max(T_min, T * T_decay)
        else:
            # Metropolis acceptance
            delta = sc_cand - sc_cur
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                a = a_cand
                T = max(T_min, T * T_decay)
            else:
                T = max(T_min, T * T_decay)
                no_improve += 1

        if no_improve > 1000:
            break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS adaptive + Adam + pairwise local search +
    multi-n exploration + annealing restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS with adaptive beta (main workhorse) ===
    print(f"\n[P1] L-BFGS adaptive (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 900, t_start + 500)
    if time.monotonic() < p1_deadline:
        a_lb = _lbfgs_adaptive(best_seq, p1_deadline, min_time_remaining=60.0)
        sc_lb = _score(a_lb)
        if sc_lb < best_score:
            best_seq = a_lb.copy()
            best_score = sc_lb
            print(f"  NEW BEST: {best_score:.10f} [lbfgs1] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  L-BFGS: no improvement (best={best_score:.10f})")

    # === Phase 2: Adam optimizer ===
    print(f"\n[P2] Adam optimizer (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 600, t_start + 800)
    if time.monotonic() < p2_deadline:
        a_adam, sc_adam = _adam_optimize(best_seq, p2_deadline,
                                         min_time_remaining=60.0, seed=seed)
        if sc_adam < best_score:
            best_seq = a_adam.copy()
            best_score = sc_adam
            print(f"  NEW BEST: {best_score:.10f} [adam] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Adam: no improvement (best={best_score:.10f})")

    # === Phase 3: L-BFGS again (finer) ===
    print(f"\n[P3] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 400, t_start + 1000)
    if time.monotonic() < p3_deadline:
        a_lb2 = _lbfgs_adaptive(best_seq, p3_deadline, min_time_remaining=60.0)
        sc_lb2 = _score(a_lb2)
        if sc_lb2 < best_score:
            best_seq = a_lb2.copy()
            best_score = sc_lb2
            print(f"  NEW BEST: {best_score:.10f} [lbfgs2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Pairwise local search ===
    print(f"\n[P4] Pairwise local search (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 300, t_start + 1200)
    if time.monotonic() < p4_deadline:
        a_ls, sc_ls = _pairwise_local_search(best_seq, p4_deadline,
                                             min_time_remaining=60.0, seed=seed)
        if sc_ls < best_score:
            best_seq = a_ls.copy()
            best_score = sc_ls
            print(f"  NEW BEST: {best_score:.10f} [local_search] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Local search: no improvement (best={best_score:.10f})")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [n_parent]
    for n_try in [16384, 20480, 24576, 32768, 12288, 8192]:
        if n_try not in n_candidates:
            n_candidates.append(n_try)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")
        if time.monotonic() < deadline - 120:
            a_opt = _lbfgs_adaptive(a_init, deadline - 60, min_time_remaining=30.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Annealing restart ===
    print(f"\n[P6] Annealing restart (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        a_ann, sc_ann = _annealing_restart(best_seq, deadline - 30,
                                           min_time_remaining=20.0, seed=seed + 999)
        if sc_ann < best_score:
            best_seq = a_ann.copy()
            best_score = sc_ann
            print(f"  NEW BEST: {best_score:.10f} [annealing] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final L-BFGS polish ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 10:
        a_fin = _lbfgs_adaptive(best_seq, deadline, min_time_remaining=5.0)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Quick Adam polish ===
    remaining = deadline - 5 - time.monotonic()
    if remaining > 20:
        a_ad2, sc_ad2 = _adam_optimize(best_seq, deadline,
                                       min_time_remaining=5.0, seed=seed + 777)
        if sc_ad2 < best_score:
            best_seq = a_ad2.copy()
            best_score = sc_ad2
            print(f"  NEW BEST: {best_score:.10f} [final_adam] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
