def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1

    e = np.exp(np.clip(z, -700, 700))

    a_rev = a[::-1]

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_soft_structured(a, beta, n):
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft_structured(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _effective_peak_width(c, c_max):
    """Estimate the effective width of the peak region in the convolution."""
    if c_max < 1e-15:
        return 1
    tol = 0.5 * c_max
    active = c >= tol
    if active.sum() == 0:
        return 1
    idx = np.where(active)[0]
    return int(idx[-1] - idx[0] + 1)


def _perturb_z(z, rng, strength=0.1):
    """Apply asymmetric perturbation to z (log-domain)."""
    n = len(z)
    # Asymmetric: larger perturbation on one side
    noise = rng.standard_normal(n) * strength
    # Add a linear component to break symmetry
    lin = np.linspace(-1, 1, n) * strength * 0.5
    z_new = z + noise + lin
    z_new = z_new - z_new.mean()
    return z_new

    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Nesterov momentum
    a_prev = a.copy()
    momentum = 0.0
    max_momentum = 0.3

    no_improve = 0
    max_no_improve = 500

    # Multiple tolerance levels to try
    tol_frac = 0.01
    min_tol = 0.0002

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            # Decay momentum on improvement
            momentum *= 0.5
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.5)
            if tol_frac <= min_tol * 1.1:
                break
            no_improve = 0
            # Reset momentum
            momentum = 0.0
            a_prev = a.copy()

        # Identify active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Compute gradient from active set (weighted by proximity to max)
        weights = (c[active] - (c_max - tol)) / (tol * n_active) + 0.1
        weights = weights / weights.sum()

        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = weights
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        # Support
        support = a > 1e-10 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        # Nesterov acceleration
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-10:
            y = y / s_y
        # Recompute gradient at y for momentum
        a_rev_y = y[::-1]
        wc_y = _full_convolve(w, a_rev_y)
        grad_y = 2.0 * wc_y[n - 1:2 * n - 1]

        # Target: equalize gradient on support
        lam = float(np.mean(grad_y[support]))
        residual = grad_y[support] - lam

        # Step
        eta = 0.08
        a_new = a.copy()
        a_new[support] -= eta * residual
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Line search
        sc_new = _score(a_new)
        if sc_new >= current_sc:
            found = False
            for trial_eta in [0.04, 0.02, 0.01, 0.005, 0.002]:
                a_try = a.copy()
                a_try[support] -= trial_eta * residual
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-14:
                    a_new = a_try
                    sc_new = sc_try
                    found = True
                    break
            if not found:
                tol_frac = max(min_tol, tol_frac * 0.9)
                momentum = 0.0
                continue

        # Update momentum
        momentum = min(max_momentum, momentum * 0.95 + 0.05)
        a_prev = a.copy()
        a = a_new

        # Gradually tighten tolerance on improvement
        if no_improve == 0:
            tol_frac = max(min_tol, tol_frac * 0.99)

    return best_a, best_sc


def _direct_equalization(a_init, deadline, min_time_remaining=20.0):
    """Direct convolution peak equalization via iterative adjustment.
    
    Strategy: find peaks, then adjust a to reduce them while not increasing others.
    Uses a water-filling inspired approach.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol = 0.005
    min_tol = 0.0005
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                tol = max(min_tol, tol * 0.7)
                if tol <= min_tol * 1.1:
                    break
                no_improve = 0

        # Find all peaks
        tol_abs = tol * c_max
        peaks = np.where(c >= c_max - tol_abs)[0]
        n_peaks = len(peaks)
        if n_peaks == 0:
            break

        # For each peak k, the "offending" products are a[j]*a[k-j] for valid j
        # We want to reduce the sum of products at peaks
        # Strategy: for each peak, find the j that contributes most and reduce a[j] or a[k-j]
        
        # Compute contribution matrix: for each peak, which (j, k-j) pairs contribute
        # Aggregate: how much does each a[j] contribute to the total peak energy?
        peak_energy = np.zeros(n)
        for k in peaks:
            j_lo = max(0, k - (n - 1))
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                other = k - j
                contrib = a[j] * a[other]
                peak_energy[j] += contrib
                if j != other:
                    peak_energy[other] += contrib

        # Normalize
        peak_total = peak_energy.sum()
        if peak_total < 1e-15:
            break
        peak_frac = peak_energy / peak_total

        # Reduce proportional to contribution
        eta = 0.05
        a_new = a.copy()
        a_new -= eta * peak_frac * a
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < current_sc - 1e-14:
            a = a_new
        else:
            # Try smaller step
            found = False
            for trial_eta in [0.02, 0.01, 0.005]:
                a_try = a.copy()
                a_try -= trial_eta * peak_frac * a
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-14:
                    a = a_try
                    found = True
                    break
            if not found:
                tol = max(min_tol, tol * 0.85)
                continue

        if no_improve == 0:
            tol = max(min_tol, tol * 0.995)

    return best_a, best_sc


def _subgradient_polish(a_init, deadline, min_time_remaining=20.0):
    """Subgradient method with Polyak step for final polish."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lb = 2.0 * n / (2 * n - 1)
    t = 0
    step_scale = 0.3

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()

        # Active set
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Gradient
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        # Polyak step
        gap = current_sc - lb
        if gap < 1e-10:
            break
        g_centered = grad - np.mean(grad)
        g_norm_sq = float(np.dot(g_centered, g_centered))
        if g_norm_sq < 1e-20:
            break
        step = step_scale * gap / g_norm_sq
        step = min(step, 0.08)

        a_new = a - step * g_centered
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new
        t += 1

        if t % 200 == 0:
            step_scale *= 0.92

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer with asymmetric perturbation and improved active-set.
    
    Phases:
    P1: L-BFGS with adaptive beta and perturbation (45% time)
    P2: Active-set v2 with momentum (15% time)
    P3: Direct equalization (10% time)
    P4: L-BFGS round 2 (15% time)
    P5: Multi-n exploration (10% time)
    P6: Final polish (5% time)
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(16384, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n=16384: score={best_score:.10f}")

    n_main = len(best_seq)

    # === Phase 1: Main L-BFGS with perturbation ===
    print(f"\n[P1] L-BFGS main (n={n_main})")
    p1_end = t_start + (budget_s - 10) * 0.45
    a1, sc1 = _run_lbfgs_enhanced(best_seq, p1_end, min_time_remaining=300, verbose=True)
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  P1 result: {best_score:.10f}")

    # === Phase 2: Active-set v2 ===
    print(f"\n[P2] Active-set v2 (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p2_end = min(deadline - (budget_s - 10) * 0.40, time.monotonic() + (budget_s - 10) * 0.15)
        a2, sc2 = _active_set_v2(best_seq, p2_end, min_time_remaining=60, verbose=False)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(f"  P2 result: {best_score:.10f}")

    # === Phase 3: Direct equalization ===
    print(f"\n[P3] Direct equalization (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p3_end = min(deadline - (budget_s - 10) * 0.30, time.monotonic() + (budget_s - 10) * 0.10)
        a3, sc3 = _direct_equalization(best_seq, p3_end, min_time_remaining=50)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3 result: {best_score:.10f}")

    # === Phase 4: L-BFGS round 2 ===
    print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p4_end = min(deadline - (budget_s - 10) * 0.15, time.monotonic() + (budget_s - 10) * 0.15)
        a4, sc4 = _run_lbfgs_enhanced(best_seq, p4_end, min_time_remaining=60, verbose=True)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4 result: {best_score:.10f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n (t={time.monotonic()-t_start:.0f}s)")
    n_options = [12288, 14336, 16384, 18432, 20480]
    n_options = [x for x in n_options if x != n_main]
    # Sort by proximity to current n
    n_options.sort(key=lambda x: abs(x - n_main))

    for n_try in n_options[:3]:  # Try at most 3 different n values
        if time.monotonic() + 60 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        print(f"  n={n_try}: resampled score={sc_r:.10f}")
        if time.monotonic() + 40 > deadline:
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
            continue
        p5_end = min(deadline - (budget_s - 10) * 0.08, time.monotonic() + 80)
        a5, sc5 = _run_lbfgs_enhanced(a_r, p5_end, min_time_remaining=30, verbose=False)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  n={n_try} optimized: {best_score:.10f}")

    # === Phase 6: Final polish ===
    print(f"\n[P6] Final polish (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 15 < deadline:
        p6_end = min(deadline - 3, time.monotonic() + 60)
        a6, sc6 = _active_set_v2(best_seq, p6_end, min_time_remaining=5, verbose=False)
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()

        if time.monotonic() + 10 < deadline:
            p6b_end = min(deadline - 2, time.monotonic() + 30)
            a6b, sc6b = _subgradient_polish(best_seq, p6b_end, min_time_remaining=3)
            if sc6b < best_score:
                best_score = sc6b
                best_seq = a6b.copy()

    # Final
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
