import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve_hankel(W, n, rhs, max_iter=500, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _compute_gradient(a, W, n):
    a_rev = a[::-1]
    wc = _full_convolve(W, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(W, _full_convolve(a, a)))
    return 2.0 * a * (h - W_val)


def _newton_kkt_v2(a_init, deadline, max_rounds=300):
    """KKT-Newton with tolerance-based adaptive active set."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0
    tol_frac = 0.01  # initial tolerance fraction

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Tolerance-based active set
        active = c >= (c_max - tol_frac * c_max)
        if active.sum() < 3:
            # Fall back to top-K
            K = max(3, int(active.sum() * 3))
            K = min(K, 2 * n - 1)
            idx = np.argpartition(c, -K)[-K:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True

        K = int(active.sum())
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / K

        g = _compute_gradient(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            # Try tighter tolerance
            tol_frac *= 0.5
            if tol_frac < 1e-8:
                break
            continue

        Hg = _cg_solve_hankel(W, n, g, max_iter=300)
        H1 = _cg_solve_hankel(W, n, np.ones(n), max_iter=300)
        H1_denom = float(np.ones(n) @ H1)
        if abs(H1_denom) < 1e-25 or not np.isfinite(H1_denom):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / H1_denom
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        # Line search with cubic interpolation
        alpha = 1.0
        step_done = False
        for ls in range(80):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.4
            if alpha < 1e-18:
                break

        if not step_done:
            # Subgradient fallback
            for trial in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
            # Tighten tolerance when making progress
            tol_frac = max(tol_frac * 0.85, 1e-9)
        else:
            no_improve += 1
            # Loosen tolerance when stuck (explore more)
            if no_improve > 3:
                tol_frac = min(tol_frac * 1.5, 0.1)

        if not step_done and no_improve > 15:
            break

    return best_a, best_sc


def _lbfgs_smooth_v2(a_init, deadline, min_remaining=60.0):
    """L-BFGS with adaptive log-sum-exp smoothing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    sorted_c = np.sort(c_cur)[::-1]
    # Estimate initial beta from the gap structure
    gap_10 = sorted_c[0] - sorted_c[min(10, len(sorted_c) - 1)]
    gap_50 = sorted_c[0] - sorted_c[min(50, len(sorted_c) - 1)]
    beta = max(gap_50 * 0.5, max_c * 0.001)

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    current_a = a_init.copy()
    current_sc = _score(current_a)

    while beta > max_c * 5e-6:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break

        scale = max(1.0, n / 8192.0)
        est_iter = 0.003 * scale
        max_iter = min(8000, max(50, int(remaining / (est_iter * 40))))

        def obj_fn(z, beta=beta, n=n):
            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                return 2.0 * n * c_max
            return 2.0 * n * (c_max + beta * np.log(s))

        def grad_fn(z, beta=beta, n=n):
            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W_val = float(np.dot(w, c))
            return 4.0 * n * a * (h - W_val)

        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 80, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        # Adapt beta
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        sorted_new = np.sort(c_new)[::-1]
        new_gap = sorted_new[0] - sorted_new[min(20, len(sorted_new) - 1)]

        if new_gap < 1e-6 * c_max_new:
            beta *= 0.1
        elif new_gap < 1e-4 * c_max_new:
            beta *= 0.15
        elif new_gap < 1e-3 * c_max_new:
            beta *= 0.25
        else:
            beta *= 0.4

        beta = min(beta, c_max_new * 0.0002)

    return current_a


def _subgrad_momentum_v2(a_init, deadline, max_steps=5000):
    """Subgradient with momentum and adaptive active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.015
    momentum = 0.9
    v = np.zeros(n)

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive tolerance
        for tol_frac in [1e-5, 5e-5, 1e-4, 5e-4, 1e-3, 3e-3, 0.01]:
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            if active.sum() >= 5:
                break
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        W = np.zeros_like(c)
        W[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(W, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(W, c))
        grad = 2.0 * a * (h - W_val)
        grad = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        v = momentum * v - lr * grad
        z = z + v
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 30 == 29:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, 0.04)
            else:
                patience += 1
                lr *= 0.85
                momentum = min(momentum + 0.005, 0.95)
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)
                    patience = 0
                    lr = 0.01

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.7) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    elif kind == "pow_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edge_heavy":
        t = np.linspace(0, 1, n)
        a = 1.0 / (0.1 + np.sin(np.pi * t))
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _targeted_perturb(a, rng, strength=0.01):
    """Perturb more where the gradient suggests improvement."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    # Find which positions contribute most to the max convolution
    # The center lag (n-1) involves a[i]*a[n-1-i]
    # Positions with high a[i]*a[n-1-i] contribute most
    cross = a * a[::-1]
    cross_max = cross.max()
    if cross_max < 1e-20:
        return _perturb(a, 0.01, rng)
    # Perturb more at positions with high cross contribution
    weights = (cross / cross_max) ** 2
    weights = weights + 0.1
    weights = weights / weights.sum() * n
    noise = rng.standard_normal(n) * strength * weights
    z = np.log(np.maximum(a, 1e-30))
    z = z + noise
    z = z - z.mean()
    return _softmax(z)


def propose_candidate():
    t_start = time.monotonic()
    budget = 1800.0
    deadline = t_start + budget - 30.0
    rng = np.random.default_rng(12345)

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"imp={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                par_sc = _score(parent)
                print(f"Parent: n={len(parent)}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        parent = _make_profile(32768, "tapered")
        print(f"No parent, using tapered n=32768")
        update_best(parent, _score(parent), "init")

    n0 = len(parent)
    print(f"\n{'='*60}")
    print(f"Starting optimization. Initial n={n0}, score={best_score:.10f}")
    print(f"{'='*60}")

    # Phase 1: KKT-Newton with adaptive active set (long phase)
    print(f"\n[Phase 1] KKT-Newton adaptive (t={time.monotonic() - t_start:.1f}s)")
    t_p1_end = min(deadline - 1200, t_start + 400)
    a_kkt, sc_kkt = _newton_kkt_v2(parent, t_p1_end, max_rounds=500)
    update_best(a_kkt, sc_kkt, "kkt_adaptive")

    # Phase 2: L-BFGS with adaptive smoothing
    print(f"\n[Phase 2] L-BFGS smooth (t={time.monotonic() - t_start:.1f}s)")
    t_p2_end = min(deadline - 900, t_start + 700)
    a_lb = _lbfgs_smooth_v2(best_seq, t_p2_end, min_remaining=500.0)
    update_best(a_lb, _score(a_lb), "lbfgs_smooth")

    # Phase 3: Cycle KKT + Subgrad + L-BFGS
    print(f"\n[Phase 3] Cycling (t={time.monotonic() - t_start:.1f}s)")
    cycle_end = min(deadline - 500, t_start + 1200)
    for cyc in range(10):
        if time.monotonic() > cycle_end:
            break
        # Subgrad
        t_sub = min(cycle_end, time.monotonic() + 40)
        a_s, sc_s = _subgrad_momentum_v2(best_seq, t_sub, max_steps=3000)
        update_best(a_s, sc_s, f"cyc_sg_{cyc}")

        # KKT
        t_kkt = min(cycle_end, time.monotonic() + 50)
        a_k, sc_k = _newton_kkt_v2(best_seq, t_kkt, max_rounds=150)
        update_best(a_k, sc_k, f"cyc_kkt_{cyc}")

        # L-BFGS burst
        t_lb = min(cycle_end, time.monotonic() + 40)
        if time.monotonic() < deadline - 200:
            a_l = _lbfgs_smooth_v2(best_seq, t_lb, min_remaining=150.0)
            update_best(a_l, _score(a_l), f"cyc_lb_{cyc}")

        # Targeted perturbation every 3 cycles
        if (cyc + 1) % 3 == 0:
            a_p = _targeted_perturb(best_seq, rng, strength=0.005)
            t_p = min(cycle_end, time.monotonic() + 20)
            a_pk, sc_pk = _newton_kkt_v2(a_p, t_p, max_rounds=80)
            update_best(a_pk, sc_pk, f"cyc_pert_{cyc}")

    # Phase 4: Multi-n exploration
    print(f"\n[Phase 4] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_options = []
    for n_try in [n0 // 2, n0, n0 * 2, 16384, 32768, 49152, 65536]:
        if n_try not in n_options and 4096 <= n_try <= 65536:
            n_options.append(n_try)

    for n_try in sorted(n_options):
        if time.monotonic() > deadline - 100:
            break
        a_res = _resample(best_seq, n_try)
        sc_res = _score(a_res)
        if sc_res < best_score:
            update_best(a_res, sc_res, f"resample_n{n_try}")

        if time.monotonic() < deadline - 60:
            # Quick KKT on resampled
            t_q = min(deadline - 40, time.monotonic() + 15)
            a_q, sc_q = _newton_kkt_v2(a_res, t_q, max_rounds=30)
            update_best(a_q, sc_q, f"kqt_n{n_try}")

    # Phase 5: Diverse restarts
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    profiles = ["tapered", "pow_taper", "cos", "asym", "bell"]
    for prof in profiles:
        if time.monotonic() > deadline - 30:
            break
        a_new_profile = _make_profile(n0, prof)
        # Blend with current best
        blend = 0.3 * a_new_profile + 0.7 * best_seq
        blend = blend / blend.sum()
        t_r = min(deadline - 20, time.monotonic() + 10)
        a_r, sc_r = _newton_kkt_v2(blend, t_r, max_rounds=40)
        update_best(a_r, sc_r, f"blend_{prof}")

    # Phase 6: Final ultra-fine polish
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        # Very tight KKT with small active sets
        a_fin, sc_fin = _newton_kkt_v2(best_seq, deadline - 2, max_rounds=200)
        update_best(a_fin, sc_fin, "final_kkt")

        # Tiny subgrad steps
        if time.monotonic() < deadline - 2:
            a_ss, sc_ss = _subgrad_momentum_v2(best_seq, deadline - 1, max_steps=1000)
            update_best(a_ss, sc_ss, "final_sg")

    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
