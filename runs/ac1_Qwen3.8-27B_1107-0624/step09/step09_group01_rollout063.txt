Optimized asymmetric step-function search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key insight: asymmetric profiles (ramps) achieve much lower scores than symmetric ones.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** p
    if exp_decay > 0:
        a *= np.exp(-exp_decay * i / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s

    a = np.sin(np.pi * t) ** 0.6
    a = np.maximum(a, 1e-10)

        rel_gap = gap / c_max_cur if c_max_cur > 0 else 1.0

        if rel_gap < 1e-6:
            beta *= 5.0
        elif rel_gap < 1e-4:
            beta *= 2.0
        elif rel_gap < 1e-2:
            beta *= 1.5
        else:
            beta /= 1.5

        beta = max(beta, 10.0)
        beta = min(beta, 1e6)

    return current_a, current_sc


def _cg_solve_hankel(W, n, rhs, max_iter=200):
    """Conjugate gradient solve for Hankel matrix W."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        # H @ p where H is Hankel with symbol W
        p_rev = p[::-1]
        out_len = len(p_rev) + len(W) - 1
        N = _next_pow2(out_len)
        F_p = np.fft.rfft(p_rev, N)
        F_W = np.fft.rfft(W, N)
        conv = np.fft.irfft(F_p * F_W, N)[:out_len]
        Hp = 2.0 * conv[n - 1:2 * n - 1]
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new) or rs_new > rs * 1.05:
            break
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < 1e-24 * rs0:
            break
    return x


def _kkt_newton(a_init, deadline, max_rounds=100, k_schedule=None):
    """KKT-based Newton with adaptive top-K active sets."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    if k_schedule is None:
        k_schedule = [3, 5, 10, 20, 50, 100, 200, 500, 1000]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        K = k_schedule[rnd % len(k_schedule)]
        K = min(K, 2 * n - 1)
        idx = np.argpartition(c, -K)[-K:]
        W = np.zeros(2 * n - 1)
        W[idx] = 1.0 / K

        # Compute gradient
        a_rev = a[::-1]
        wc = _full_convolve(W, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(W, c))
        g = 2.0 * a * (h - W_val)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        # Newton step via CG on Hankel
        Hg = _cg_solve_hankel(W, n, g, max_iter=100)
        H1 = _cg_solve_hankel(W, n, np.ones(n), max_iter=100)
        H1_denom = float(np.ones(n) @ H1)
        if abs(H1_denom) < 1e-25 or not np.isfinite(H1_denom):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / H1_denom
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        # Line search
        alpha = 1.0
        step_done = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not step_done:
            for trial in [0.1, 0.05, 0.01, 0.005]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 6:
            break

    return best_a, best_sc


def _subgrad_momentum(a_init, deadline, max_steps=3000):
    """Subgradient method with momentum in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.015
    momentum = 0.85
    v = np.zeros(n)

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set
        for tol_frac in [0.0005, 0.001, 0.005, 0.01, 0.05]:
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            if active.sum() >= 3:
                break
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        W = np.zeros_like(c)
        W[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(W, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(W, c))
        grad = 2.0 * a * (h - W_val)
        grad = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        v = momentum * v - lr * grad
        z = z + v
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 30 == 29:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, 0.04)
            else:
                patience += 1
                lr *= 0.85
                momentum = min(momentum + 0.01, 0.93)
                if patience > 25:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)
                    patience = 0
                    lr = 0.01

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"imp={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # --- Generate diverse starting points ---
    print(f"[Init] Generating starting profiles...")
    starts = []

    # Ramp profiles with different powers
    for p_val in [0.5, 0.7, 1.0, 1.5]:
        for n_val in [8192, 16384, 32768]:
            a = _ramp_profile(n_val, p=p_val)
            sc = _score(a)
            starts.append((a, sc, f"ramp_p{p_val}_n{n_val}"))

    # Exponential ramps
    for decay in [0.5, 1.0, 2.0]:
        a = _ramp_profile(16384, p=1.0, exp_decay=decay)
        sc = _score(a)
        starts.append((a, sc, f"exp_ramp_d{decay}"))

    # Asymmetric bumps
    for center in [0.6, 0.7, 0.8]:
        a = _asym_bump_profile(16384, center=center, width=0.4)
        sc = _score(a)
        starts.append((a, sc, f"bump_c{center}"))

    # Tapered (symmetric) for comparison
    a = _tapered_profile(16384)
    sc = _score(a)
    starts.append((a, sc, "tapered"))

    # Uniform
    for n_val in [8192, 32768]:
        a = np.ones(n_val) / n_val
        sc = _score(a)
        starts.append((a, sc, f"uniform_n{n_val}"))

    # Load parent if available
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                par_sc = _score(parent)
                starts.append((parent, par_sc, "parent"))
                print(f"  Parent: n={len(parent)}, score={par_sc:.10f}")
    except Exception:
        pass

    # Sort by score and keep top candidates
    starts.sort(key=lambda x: x[1])
    print(f"  Top starts:")
    for a, sc, label in starts[:5]:
        print(f"    {label}: {sc:.10f} (n={len(a)})")

    # Update best with the best starting point
    for a, sc, label in starts[:3]:
        update_best(a, sc, label + "_init")

    # --- Phase 1: L-BFGS on top starts ---
    print(f"\n[Phase 1] L-BFGS optimization (t={time.monotonic() - t_start:.1f}s)")
    phase1_end = min(deadline - 600, t_start + 400)

    for idx, (a_init, sc_init, label) in enumerate(starts[:4]):
        if time.monotonic() > phase1_end:
            break
        t_alloc = min(phase1_end, time.monotonic() + 100)
        a_opt, sc_opt = _run_lbfgs_smoothed(a_init, t_alloc, min_remaining=20.0)
        update_best(a_opt, sc_opt, f"lbfgs_{label}")

    # --- Phase 2: KKT-Newton polish ---
    print(f"\n[Phase 2] KKT-Newton polish (t={time.monotonic() - t_start:.1f}s)")
    phase2_end = min(deadline - 300, t_start + 700)
    k_sch = [3, 5, 10, 20, 50, 100, 200, 500]
    a_kkt, sc_kkt = _kkt_newton(best_seq, phase2_end, max_rounds=200, k_schedule=k_sch)
    update_best(a_kkt, sc_kkt, "kkt_polish")

    # --- Phase 3: Subgradient refinement ---
    print(f"\n[Phase 3] Subgradient (t={time.monotonic() - t_start:.1f}s)")
    phase3_end = min(deadline - 200, t_start + 900)
    a_sg, sc_sg = _subgrad_momentum(best_seq, phase3_end, max_steps=5000)
    update_best(a_sg, sc_sg, "subgrad")

    # --- Phase 4: Cycle L-BFGS + KKT ---
    print(f"\n[Phase 4] Cycled optimization (t={time.monotonic() - t_start:.1f}s)")
    phase4_end = min(deadline - 100, t_start + 1400)
    for cyc in range(5):
        if time.monotonic() > phase4_end:
            break
        t_sub = min(phase4_end, time.monotonic() + 60)
        a_c, sc_c = _run_lbfgs_smoothed(best_seq, t_sub, min_remaining=15.0)
        update_best(a_c, sc_c, f"cyc_lbfgs_{cyc}")

        t_kkt = min(phase4_end, time.monotonic() + 40)
        a_k, sc_k = _kkt_newton(best_seq, t_kkt, max_rounds=50,
                                 k_schedule=[1, 2, 3, 5, 10, 20, 50])
        update_best(a_k, sc_k, f"cyc_kkt_{cyc}")

    # --- Phase 5: Multi-n exploration ---
    print(f"\n[Phase 5] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    phase5_end = min(deadline - 50, t_start + 1700)
    n0 = len(best_seq)
    n_options = sorted(set([n0 // 4, n0 // 2, n0, n0 * 2, 8192, 16384, 32768, 65536]))
    n_options = [n for n in n_options if 2048 <= n <= 65536]

    for n_try in n_options:
        if time.monotonic() > phase5_end:
            break
        # Resample best to new n
        a_res = _resample(best_seq, n_try)
        sc_res = _score(a_res)
        if sc_res < best_score:
            update_best(a_res, sc_res, f"resample_n{n_try}")

        # Also try ramp at this n
        a_ramp = _ramp_profile(n_try, p=1.0)
        sc_ramp = _score(a_ramp)
        if sc_ramp < best_score:
            update_best(a_ramp, sc_ramp, f"ramp_n{n_try}")

        # Quick L-BFGS on the best of resampled and ramp
        a_start_n = a_res if sc_res < sc_ramp else a_ramp
        if time.monotonic() < phase5_end - 30:
            t_end = min(phase5_end - 10, time.monotonic() + 30)
            a_opt_n, sc_opt_n = _run_lbfgs_smoothed(a_start_n, t_end, min_remaining=5.0)
            update_best(a_opt_n, sc_opt_n, f"opt_n{n_try}")

    # --- Phase 6: Final polish ---
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 3:
        a_fin, sc_fin = _kkt_newton(best_seq, deadline - 1, max_rounds=100,
                                     k_schedule=[1, 2, 3, 5, 10, 20])
        update_best(a_fin, sc_fin, "final_kkt")

        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _run_lbfgs_smoothed(best_seq, deadline - 0.5, min_remaining=1.0)
            update_best(a_fin2, sc_fin2, "final_lbfgs")

    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _ramp_profile(8192, p=1.0)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _ramp_profile(16384, p=1.0)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
