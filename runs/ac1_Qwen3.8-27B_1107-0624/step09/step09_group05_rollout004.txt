"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Adam with soft-max, FISTA, peak-chasing heuristics, KKT refinement,
and perturbation-restart to escape local minima.

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _norm1(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)

def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

def _grad_soft_structured(a, beta, n):
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)

def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft_structured(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)

def _active_set_gradient(a, tol_frac=0.002):
    """Compute gradient using uniform weights on active set."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    active = c >= (c_max - tol_frac * c_max)
    n_active = int(active.sum())
    if n_active == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        n_active = 1
    a_rev = a[::-1]
    w = np.zeros(len(c))
    w[active] = 1.0 / n_active
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad, c, active

def _adam_optimize(a_init, beta, deadline, lr=None, verbose=False):
    """Adam optimizer on the softmax-parameterized smoothing objective."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    if lr is None:
        lr = 0.01
    
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999

    best_z = z.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    t = 0
    c = _full_convolve(a, a)
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted)-1)]
    if gap < 1e-15:
        gap = c_max[0] * 0.01 if c_max[0] > 0 else 0.01
    
    while time.monotonic() + 5 < deadline:
        # Compute gradient
        g = _smograd(z, beta, n)
        t += 1
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            lr *= 0.85
        
        if verbose and t % 1000 == 0:

def _fista_optimize(a_init, beta, deadline, verbose=False):

    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)

        return a_init, _score(a_init)
    a = a / s
    
    a_prev = a.copy()
    t_k = 1.0
    
    # Lipschitz constant estimate
    c = _full_convolve(a, a)

    # L ≈ 2 * n * (2 * max(a)) / beta (rough bound)
    L = 4.0 * n * a.max() / max(beta, 1e-10)
    L = max(L, 1.0)
    
    best_a = a.copy()
    best_sc = _score(a)
    
    t = 0
    while time.monotonic() + 3 < deadline:
        # Gradient step

        a_new = a - (1.0 / L) * grad
        # Prox: project onto {a >= 0, sum(a) = 1} (approximate)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # FISTA update
        t_new = (1 + np.sqrt(1 + 4 * t_k * t_k)) / 2
        a = a_new + ((t_k - 1) / t_new) * (a_new - a_prev)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        a_prev = a_new
        t_k = t_new

            best_sc = sc
            best_a = a.copy()
        
        t += 1
        # Adapt L

            L *= 0.95
            if L < 1.0:
                L = 1.0

def _peak_chase(a_init, deadline, verbose=False):
    """Targeted peak reduction: iteratively reduce elements contributing to top peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    
    t = 0
    no_improve = 0
    step_size = 0.01
    
    while time.monotonic() + 3 < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            step_size *= 0.7
            if step_size < 1e-5:
                break
            no_improve = 0
        
        # Find top K peaks
        K = min(20, len(c))
        top_k = np.argsort(c)[::-1][:K]
        
        # For each top peak, compute contribution of each element
        # and reduce elements that contribute most to multiple peaks
        a_rev = a[::-1]
        
        # Weighted gradient: pay more attention to higher peaks
        weights = np.zeros(len(c))
        for i, k in enumerate(top_k):
            w = 1.0 / (i + 1)
            weights[k] += w
        weights /= weights.sum()
        
        wc = _full_convolve(weights, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]
        
        # Center the gradient (it should be constant at optimum)
        support = a > 1e-10
        if support.sum() > 0:
            lam = float(np.mean(grad[support]))
            grad_centered = grad - lam
        else:
            grad_centered = grad - np.mean(grad)
        
        # Step
        a_new = a - step_size * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            step_size *= 0.5
            continue
        a_new = a_new / s_new
        a = a_new
        t += 1
    
    return best_a, best_sc

def _kkt_refine(a_init, deadline, verbose=False):
    """KKT-inspired active-set refinement with adaptive tolerance."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    
    tol_frac = 0.003
    min_tol = 0.00005
    no_improve = 0
    eta = 0.1
    
    while time.monotonic() + 3 < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            tol_frac = max(min_tol, tol_frac * 0.5)
            if tol_frac <= min_tol * 1.05:
                break
            no_improve = 0
            eta = max(0.01, eta * 0.7)
        
        # Active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        
        # Average gradient from active set
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]
        
        # KKT: grad_avg should be constant on support
        support = a > 1e-10 * a.max()
        if support.sum() < 2:
            break
        
        lam = float(np.mean(grad_avg[support]))
        residual = grad_avg[support] - lam
        
        # Line search
        a_new = a.copy()
        found = False
        for trial_eta in [eta, eta*0.5, eta*0.25, eta*0.1, 0.001]:
            a_try = a.copy()
            a_try[support] -= trial_eta * residual
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < sc - 1e-12:
                a = a_try
                found = True
                break
        
        if not found:
            eta = max(0.001, eta * 0.5)
            # Tighten tol
            tol_frac = max(min_tol, tol_frac * 0.85)
        
        # Gradually tighten
        if no_improve == 0:
            tol_frac = max(min_tol, tol_frac * 0.98)
    
    return best_a, best_sc

def _lbfgs_run(a_init, deadline, beta_start=None, verbose=False):

    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    
    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c0_max * 0.003:
        gap = c0_max * 0.003
    
    if beta_start is None:
        beta_start = gap
    
    # Beta schedule
    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,

    ])
    betas = beta_start * multipliers

            self.beta = beta
            self.n = n
        def __call__(self, z):

            c = _full_convolve(a, a)
            return 2.0 * n * _soft_max_value(c, self.beta)
        def grad(self, z):
            a = _softmax(z)
            grad_a = _grad_soft_structured(a, self.beta, self.n)
            W = float(np.dot(grad_a, a))
            return a * (grad_a - W)
    
    t0 = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() + 10 > deadline:
            break
        
        remaining = deadline - time.monotonic()
        size_factor = max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (0.002 * size_factor * 20))))
        
        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={

def _perturb_restart(a, n_perturbations=5, deadline=None, verbose=False):

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()

        w[int(np.argmax(c))] = 1.0

def _pnorm_value(c, p):
    """p-norm of c (always positive since c >= 0 for non-negative a)."""
    if p == np.inf:
        return float(c.max())
    return float(np.sum(c ** p) ** (1.0 / p))


def _pnorm_grad(c, a, p, n):
    """Gradient of ||c||_p where c = conv(a,a), returned as grad w.r.t. a."""
    if p == np.inf:
        idx = int(np.argmax(c))
        w = np.zeros(len(c))
        w[idx] = 1.0
    else:
        cp = c ** (p - 1)
        norm_p = float(np.sum(cp)) ** (1.0 / p)
        w = cp / (p * norm_p ** (p - 1))
    # gradient: conv(w, a_rev)[n-1:2n-1] * 2
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]

    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]

        self.n = n
        self.pnorm = pnorm
        self.p = p

    def __call__(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        if self.pnorm:
            return 2.0 * self.n * _pnorm_value(c, self.p)
        return 2.0 * self.n * _soft_max_value(c, self.beta)

        a = _softmax(z)
        c = _full_convolve(a, a)

        else:

        return a * (grad_a - W)

            a_try[support] -= trial_eta * residual
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < sc - 1e-14:
                a = a_try
                found = True
                break

        if not found:
            eta = max(0.001, eta * 0.7)
            tol_frac = max(min_tol, tol_frac * 0.9)
            if tol_frac <= min_tol * 1.1 and eta < 0.002:
                break
        else:
            if no_improve == 0:
                tol_frac = max(min_tol, tol_frac * 0.97)

    if verbose:
        print(f"  KKT: {max_iter} iterations, final={best_sc:.10f}")
    return best_a, best_sc


def _peak_chase(a_init, deadline, verbose=False):
    """Targeted peak reduction using weighted top-K peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)
    step_size = 0.02
    no_improve = 0
    t = 0

    while time.monotonic() + 3 < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
            # Slightly increase step when improving
            step_size = min(0.05, step_size * 1.02)
        else:
            no_improve += 1
            step_size *= 0.95

        if no_improve > 300:
            break
        if step_size < 1e-6:
            break

        # Top-K peaks with geometric weighting
        K = min(30, len(c))
        top_k = np.argsort(c)[::-1][:K]

        weights = np.zeros(len(c))
        for i, k in enumerate(top_k):
            weights[k] = 0.7 ** i
        weights /= weights.sum()

        a_rev = a[::-1]
        wc = _full_convolve(weights, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        # Center on support
        support = a > 1e-10
        if support.sum() > 0:
            lam = float(np.mean(grad[support]))
            grad_c = grad - lam
        else:
            grad_c = grad - np.mean(grad)

        # Line search
        a_new = a.copy()
        found = False
        for trial in [step_size, step_size * 0.5, step_size * 0.25]:
            a_try = a - trial * grad_c
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < sc - 1e-12:
                a = a_try
                found = True
                break
        if not found:
            step_size *= 0.7

        a_new = a
        t += 1

    if verbose:
        print(f"  PeakChase: {t} iterations, final={best_sc:.10f}")
    return best_a, best_sc


def _lbfgs_run(a_init, deadline, beta_start=None, use_pnorm=False, p=20, verbose=False):
    """L-BFGS with adaptive schedule (soft-max or p-norm)."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c0_max * 0.003:
        gap = c0_max * 0.003

    if beta_start is None:
        beta_start = gap

    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.002,
        0.001, 0.0005, 0.00025, 0.0001, 0.00005, 0.00002
    ])
    betas = beta_start * multipliers

    t0 = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() + 10 > deadline:
            break

        remaining = deadline - time.monotonic()
        size_factor = max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (0.003 * size_factor * 15))))

        prob = _SMProb(beta, n, pnorm=use_pnorm, p=p)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"    L-BFGS beta={beta:.2e}: {best_sc:.10f}")
            else:
                # Don't update z if no improvement
                pass
        except Exception:
            pass

    return best_a, best_sc


def _adam_smooth(a_init, beta, deadline, lr=0.005, verbose=False):
    """Adam optimizer on softmax-parameterized soft-max objective."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_z = z.copy()
    best_a = a.copy()
    best_sc = _score(a)
    t = 0
    best_lr = lr

    while time.monotonic() + 3 < deadline:
        g = _SMProb(beta, n, pnorm=False, p=20).grad(z)
        t += 1

        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = _softmax(z_new)
        sc_new = _score(a_new)

        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            best_lr = lr

        z = z_new

        # Cosine-like decay
        if t % 2000 == 0:
            lr = max(0.0001, best_lr * (1.0 - 0.1 * (t // 2000)))

    return best_a, best_sc


def _iterated_local_search(a_init, deadline, n_restarts=5, verbose=False):
    """ILS: optimize, perturb, re-optimize. Keep best."""
    n = len(a_init)
    rng = np.random.default_rng(777)
    best_a = a_init.copy()
    best_sc = _score(a_init)

    for i in range(n_restarts):
        if time.monotonic() + 60 > deadline:
            break

        # Generate perturbation
        perturbation_scale = 0.005 * (1 + i * 0.5)

        strategy = i % 4
        if strategy == 0:
            # Gaussian noise
            noise = rng.normal(0, perturbation_scale, size=n)
            a_pert = best_a + noise
        elif strategy == 1:
            # Random subset zeroing
            mask = rng.random(n) < 0.1
            a_pert = best_a.copy()
            a_pert[mask] *= 0.5
        elif strategy == 2:
            # Random subset boosting
            idx = rng.choice(n, size=n // 10, replace=False)
            a_pert = best_a.copy()
            a_pert[idx] *= (1.0 + perturbation_scale * rng.random(len(idx)))
        else:
            # Smooth noise (low-pass)
            noise = rng.normal(0, perturbation_scale, size=n)
            # Simple moving average
            k = max(3, n // 100)
            kernel = np.ones(k) / k
            noise = np.convolve(noise, kernel, mode='same')
            a_pert = best_a + noise

        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert

        sc_pert = _score(a_pert)
        if verbose:
            print(f"    ILS restart {i+1}: perturbed score={sc_pert:.10f}")

        # Optimize from perturbed point
        opt_end = min(deadline - 30, time.monotonic() + 60)
        a_opt, sc_opt = _lbfgs_run(a_pert, opt_end, beta_start=0.01 * a_pert.max() * n, verbose=False)

        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    ILS restart {i+1}: improved to {best_sc:.10f}")

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.7
    elif kind == "tapered2":
        a = np.sin(np.pi * t) ** 0.5
    elif kind == "tapered3":
        a = np.sin(np.pi * t) ** 1.0
    elif kind == "raised_cosine":
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif kind == "bimodal":
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
    elif kind == "periodic":
        tt = np.arange(n, dtype=float)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * tt / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * tt / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * tt / n))
    elif kind == "gaussian":
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
    elif kind == "taylor":
        # Taylor window approximation (8 dB sidelobe)
        a = (1.0 - 1.93 * np.cos(2 * np.pi * t)
             + 1.29 * np.cos(4 * np.pi * t)
             - 0.388 * np.cos(6 * np.pi * t)
             + 0.068 * np.cos(8 * np.pi * t))
    elif kind == "hamming":
        a = 0.54 - 0.46 * np.cos(2 * np.pi * t)
    elif kind == "blackman":
        a = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    elif kind == "optimal_taper":
        # Slightly modified hann for better PSL
        a = np.sin(np.pi * t) ** 0.8
        a = 1.0 - 0.15 * (1.0 - a)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _gradual_n_search(best_a, best_sc, deadline, verbose=True):
    """Try gradually changing n and optimizing."""
    current_n = len(best_a)
    n_tries = [current_n - 2048, current_n + 2048, current_n + 4096]
    n_tries = [x for x in n_tries if 4096 <= x <= 49152]
    # Sort by closeness to current
    n_tries.sort(key=lambda x: abs(x - current_n))

    for n_try in n_tries:
        if time.monotonic() + 50 > deadline:
            break
        a_r = _resample(best_a, n_try)
        sc_r = _score(a_r)
        if verbose:
            print(f"    n={n_try}: resampled score={sc_r:.10f}")

        opt_end = min(deadline - 20, time.monotonic() + 50)
        a_opt, sc_opt = _lbfgs_run(a_r, opt_end, beta_start=None, verbose=False)
        # Also try KKT
        if time.monotonic() + 10 < opt_end:
            a_kkt, sc_kkt = _kkt_refine(a_opt, opt_end, verbose=False)
            if sc_kkt < sc_opt:
                a_opt, sc_opt = a_kkt, sc_kkt

        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    n={n_try}: improved to {best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    1. Window shape search (parametric initialization)
    2. L-BFGS with soft-max (coarse to fine beta)
    3. KKT active-set refinement
    4. Peak chasing
    5. L-BFGS with p-norm (alternative smoothing)
    6. Iterated local search (perturbation + restart)
    7. Multi-n exploration
    8. Final KKT polish
    """
    t_global_start = time.monotonic()
    deadline = t_global_start + max(1.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        # Try several initializations
        n_init = 16384
        candidates = []
        for kind in ["tapered", "tapered2", "raised_cosine", "taylor", "hamming"]:
            a_c = _make_profile(n_init, kind)
            sc_c = _score(a_c)
            candidates.append((sc_c, a_c, kind))
            print(f"  Init {kind}: score={sc_c:.10f}")
        candidates.sort(key=lambda x: x[0])
        best_seq = candidates[0][1].copy()
        best_score = candidates[0][0]
        print(f"Best init: {candidates[0][2]}, score={best_score:.10f}")

    n_main = len(best_seq)

    # === Stage 1: Window shape exploration ===
    print(f"\n[S1] Window shape search (n={n_main})")
    s1_deadline = min(deadline - 400, t_global_start + 120)
    for kind in ["tapered3", "optimal_taper", "blackman", "periodic", "gaussian"]:
        if time.monotonic() + 30 > s1_deadline:
            break
        a_c = _make_profile(n_main, kind)
        sc_c = _score(a_c)
        if sc_c < best_score + 0.1:  # Only optimize if competitive
            s1_sub_end = min(s1_deadline, time.monotonic() + 40)
            a_opt, sc_opt = _lbfgs_run(a_c, s1_sub_end, beta_start=None, verbose=False)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"  {kind}: improved to {best_score:.10f}")

    # === Stage 2: Main L-BFGS ===
    print(f"\n[S2] L-BFGS main (t={time.monotonic()-t_global_start:.0f}s)")
    s2_deadline = min(deadline - 300, time.monotonic() + 300)
    a2, sc2 = _lbfgs_run(best_seq, s2_deadline, beta_start=None, verbose=True)
    if sc2 < best_score:
        best_score = sc2
        best_seq = a2.copy()
        print(f"  S2: {best_score:.10f}")

    # === Stage 3: KKT refinement ===
    print(f"\n[S3] KKT refine (t={time.monotonic()-t_global_start:.0f}s)")
    s3_deadline = min(deadline - 250, time.monotonic() + 150)
    a3, sc3 = _kkt_refine(best_seq, s3_deadline, verbose=True)
    if sc3 < best_score:
        best_score = sc3
        best_seq = a3.copy()
        print(f"  S3: {best_score:.10f}")

    # === Stage 4: Peak chasing ===
    print(f"\n[S4] Peak chase (t={time.monotonic()-t_global_start:.0f}s)")
    s4_deadline = min(deadline - 200, time.monotonic() + 100)
    a4, sc4 = _peak_chase(best_seq, s4_deadline, verbose=True)
    if sc4 < best_score:
        best_score = sc4
        best_seq = a4.copy()
        print(f"  S4: {best_score:.10f}")

    # === Stage 5: L-BFGS with p-norm ===
    print(f"\n[S5] L-BFGS p-norm (t={time.monotonic()-t_global_start:.0f}s)")
    s5_deadline = min(deadline - 150, time.monotonic() + 120)
    a5, sc5 = _lbfgs_run(best_seq, s5_deadline, beta_start=1e-4, use_pnorm=True, p=50, verbose=True)
    if sc5 < best_score:
        best_score = sc5
        best_seq = a5.copy()
        print(f"  S5: {best_score:.10f}")

    # === Stage 6: KKT again ===
    print(f"\n[S6] KKT refine 2 (t={time.monotonic()-t_global_start:.0f}s)")
    s6_deadline = min(deadline - 100, time.monotonic() + 80)
    a6, sc6 = _kkt_refine(best_seq, s6_deadline, verbose=True)
    if sc6 < best_score:
        best_score = sc6
        best_seq = a6.copy()
        print(f"  S6: {best_score:.10f}")

    # === Stage 7: Iterated Local Search ===
    print(f"\n[S7] ILS (t={time.monotonic()-t_global_start:.0f}s)")
    s7_deadline = min(deadline - 60, time.monotonic() + 200)
    a7, sc7 = _iterated_local_search(best_seq, s7_deadline, n_restarts=4, verbose=True)
    if sc7 < best_score:
        best_score = sc7
        best_seq = a7.copy()
        print(f"  S7: {best_score:.10f}")

    # === Stage 8: Multi-n exploration ===
    print(f"\n[S8] Multi-n (t={time.monotonic()-t_global_start:.0f}s)")
    s8_deadline = min(deadline - 30, time.monotonic() + 120)
    best_seq, best_score = _gradual_n_search(best_seq, best_score, s8_deadline, verbose=True)

    # === Stage 9: Final polish ===
    print(f"\n[S9] Final polish (t={time.monotonic()-t_global_start:.0f}s)")
    if time.monotonic() + 15 < deadline:
        s9_deadline = min(deadline - 5, time.monotonic() + 60)
        a9, sc9 = _lbfgs_run(best_seq, s9_deadline, beta_start=1e-6, verbose=False)
        if sc9 < best_score:
            best_score = sc9
            best_seq = a9.copy()

    if time.monotonic() + 10 < deadline:
        s9b_deadline = min(deadline - 2, time.monotonic() + 30)
        a9b, sc9b = _kkt_refine(best_seq, s9b_deadline, verbose=False)
        if sc9b < best_score:
            best_score = sc9b
            best_seq = a9b.copy()

    # Final safety
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_global_start:.1f}s")
    return [float(x) for x in final]
