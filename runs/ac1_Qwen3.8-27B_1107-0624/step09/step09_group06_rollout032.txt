Now I can use conjugate gradient to solve $H x = g$ where $g$ is the negative gradient (projected to the tangent space of the simplex).

The CG algorithm:
1. $r_0 = g - H x_0$ (start with $x_0 = 0$)
2. $p_0 = r_0$
3. For $i = 0, 1, 2, ...$:
   a. $\alpha_i = (r_i^T r_i) / (p_i^T H p_i)$
   b. $x_{i+1} = x_i + \alpha_i p_i$
   c. $r_{i+1} = r_i - \alpha_i H p_i$
   d. $\beta_i = (r_{i+1}^T r_{i+1}) / (r_i^T r_i)$
   e. $p_{i+1} = r_{i+1} + \beta_i p_i$
   f. Check convergence: $\|r_{i+1}\| < \text{tol} \|g\|$

Each CG iteration requires one application of $H$ (which is $O(n \log n)$ via FFT). If we need, say, 50 CG iterations, the total cost is $O(n \log n \cdot 50)$, which is very fast.

After getting the Newton step $\delta$, we need to find a step size $t > 0$ such that $a + t\delta$ is in the simplex. We can do this by:
- $t_{\max} = \min_{j: \delta_j < 0} (-a_j / \delta_j)$ (the largest $t$ before any component goes negative)
- Also need $\sum (a + t\delta)_j = 1$, which requires $\sum \delta_j = 0$ (we should project the step to the tangent space)

2. Project to tangent space: $g_t = g - \text{mean}(g)$

"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements:
1. FISTA with adaptive log-sum-exp (faster than subgradient)
2. Active-set Newton with FFT-accelerated CG (Levenberg-Marquardt)
3. Aggressive beta annealing with L-BFGS
4. Systematic multi-n search with resampling
5. Sparsification and symmetry exploitation

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n_a, n_b = len(a), len(b)
    out_n = n_a + n_b - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return float(2.0 * n * (c_max + beta * np.log(s)))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _fista_smooth(a_init, beta, deadline, max_iter=200000):
    """FISTA on the log-sum-exp smooth approximation, parameterized on simplex."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    c = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c.max())
    best_a = a.copy()

    # Lipschitz constant estimate: L ~ 4n^2/beta (rough upper bound)
    # Use a more conservative estimate
    L = 4.0 * n * n / (beta + 1e-10)
    # Scale down since our actual Lipschitz constant is much smaller
    L = min(L, 4.0 * n / beta * 10.0)
    lr = 1.0 / L

    x = a.copy()
    z = a.copy()
    t_k = 1.0
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z = x + (t_k - 1.0) / t_new * (x - z)
        z = _project_simplex(z)

        # Compute gradient of smooth objective at z
        c_z = _full_convolve(z, z)
        c_max_z = float(c_z.max())
        shifted = (c_z - c_max_z) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s_e = e.sum()
        if s_e < 1e-300:
            w = np.zeros_like(c_z)
            w[int(np.argmax(c_z))] = 1.0
        else:
            w = e / s_e

        # Gradient: 4n * z * (w * z_rev - W) where W = w . c_z
        z_rev = z[::-1]
        wz = _full_convolve(w, z_rev)
        h = wz[n - 1:2 * n - 1]
        W = float(np.dot(w, c_z))
        grad = 4.0 * n * z * (h - W)

        # Project gradient to tangent space
        grad_t = grad - np.mean(grad)

        # Backtracking line search
        alpha = lr
        f_x = 2.0 * n * c_max_z + 2.0 * n * beta * np.log(s_e) if s_e > 1e-300 else 2.0 * n * c_max_z
        for bt in range(20):
            x_new = _project_simplex(x - alpha * grad_t)
            c_new = _full_convolve(x_new, x_new)
            c_max_new = float(c_new.max())
            shifted_new = (c_new - c_max_new) / beta
            e_new = np.exp(np.clip(shifted_new, -700, 0))
            s_new = e_new.sum()
            if s_new < 1e-300:
                f_new = 2.0 * n * c_max_new
            else:
                f_new = 2.0 * n * (c_max_new + beta * np.log(s_new))
            # Armijo condition
            if f_new <= f_x - 0.5 * alpha * np.dot(grad_t, x - x_new):
                break
            alpha *= 0.5

        x = x_new
        t_k = t_new

        # Track best
        sc = 2.0 * n * float(c_new.max())
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 5000:
            break

        if it % 20000 == 0 and it > 0:
            print(f"      FISTA it={it}: best={best_sc:.10f}, beta={beta:.6g}")

    return best_a, best_sc


def _apply_hessian(delta, active_indices, n):
    """Apply H = 2 * sum_{k in A} P_k to delta using FFT cross-correlation."""
    s = np.zeros(2 * n - 1)
    s[active_indices] = 1.0
    t = delta[::-1]
    N = _next_pow2(n + 2 * n - 1)
    Ft = np.fft.rfft(t, N)
    Fs = np.fft.rfft(s, N)
    conv = np.fft.irfft(Ft * Fs, N)
    # (H delta)[j] = 2 * conv[j + n - 1] for j = 0, ..., n-1
    return 2.0 * conv[n - 1:2 * n - 1]


def _newton_active_set(a_init, deadline, max_newton_steps=200, cg_max_iter=100):
    """Active-set Newton method with CG solver and LM regularization."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Determine initial active set
    c = _full_convolve(a, a)
    c_max = float(c.max())
    # Take peaks within 1% of max
    threshold = c_max * 0.99
    active_indices = np.where(c >= threshold)[0]
    if len(active_indices) < 3:
        # Take top 5
        active_indices = np.argsort(c)[::-1][:5]

    mu = 0.1  # LM regularization
    no_improve = 0
    active_set_stable = 0

    for NS in range(max_newton_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break

        # Update active set: peaks within 0.5% of max
        threshold = c_max * 0.995
        new_active = np.where(c >= threshold)[0]
        if len(new_active) < 3:
            new_active = np.argsort(c)[::-1][:5]
        if len(new_active) > 500:
            new_active = np.argsort(c)[::-1][:500]

        # Check if active set changed
        if not _sets_equal(active_indices, new_active):
            active_indices = new_active
            active_set_stable = 0
        else:
            active_set_stable += 1

        # Compute gradient
        K = len(active_indices)
        grad = np.zeros(n)
        for k in active_indices:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                idx = k - np.arange(j_lo, j_hi + 1)
                grad[j_lo:j_hi + 1] += 2.0 * a[idx]

        # Project to tangent space
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-14:
            break

        # Solve (H + mu*I) delta = -grad_t using CG
        delta = np.zeros(n)
        r = -grad_t  # residual = b - H*delta = -grad_t (since delta=0)
        p = r.copy()
        rz = np.dot(r, r)

        converged = False
        for cg_it in range(cg_max_iter):
            Hp = _apply_hessian(p, active_indices, n) + mu * p
            pHp = np.dot(p, Hp)
            if pHp < 1e-18:
                break
            alpha_cg = rz / pHp
            delta += alpha_cg * p
            r_new = r - alpha_cg * Hp
            rz_new = np.dot(r_new, r_new)
            if np.sqrt(rz_new) < 1e-10 * np.sqrt(rz) or np.sqrt(rz_new) < 1e-12:
                converged = True
                r = r_new
                break
            beta_cg = rz_new / rz
            p = r_new + beta_cg * p
            r = r_new
            rz = rz_new

        if not converged and np.linalg.norm(r) > 1e-6 * gnorm:
            mu *= 2.0
            continue

        # Project delta to tangent space
        delta -= np.mean(delta)

        # Line search
        dmax = np.inf
        neg_mask = delta < 0
        if np.any(neg_mask):
            dmax = np.min(-a[neg_mask] / delta[neg_mask]) * 0.99

        t_step = min(1.0, dmax)
        accepted = False
        for ls in range(20):
            t_try = t_step * (0.5 ** ls)
            a_try = a + t_try * delta
            if np.any(a_try < -1e-15):
                continue
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-15:
                a = a_try
                accepted = True
                mu = max(0.01, mu * 0.5)
                break
            elif sc_try < current_sc + 0.1 * t_try * gnorm:
                # Accept even if not improving (for indefinite Hessian)
                a = a_try
                accepted = True
                break

        if not accepted:
            mu *= 2.0
            # Perturb slightly
            rng = np.random.default_rng(NS)
            a = _project_simplex(a + 0.001 * rng.standard_normal(n))

        if NS % 10 == 0:
            print(f"      Newton step {NS}: sc={current_sc:.10f}, |A|={K}, mu={mu:.4f}, cg_conv={converged}")

    return best_a, best_sc


def _sets_equal(a, b):
    if len(a) != len(b):
        return False
    return np.array_equal(np.sort(a), np.sort(b))


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new > 1e-15:
        a_new = a_new / s_new
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "gaussian":
        t = np.linspace(-3, 3, n)
        a = np.exp(-0.3 * t ** 2)
        return a / a.sum()
    elif kind == "beta_dist":
        t = np.linspace(0, 1, n)
        a = t ** 2 * (1 - t) ** 2 + 1e-10
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _run_lbfgs_verbose(a_init, deadline, beta_list=None, label="L-BFGS"):
    """L-BFGS with log-sum-exp and adaptive beta schedule."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    if beta_list is None:
        c = _full_convolve(current_a, current_a)
        c_max = float(c.max())
        c_sorted = np.sort(c)[::-1]
        gap = c_max - c_sorted[min(20, len(c_sorted) - 1)]
        if gap < c_max * 0.001:
            gap = c_max * 0.001
        base_beta = max(gap, c_max * 0.005)
        beta_list = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02,
                                           0.01, 0.005, 0.002, 0.001,
                                           0.0005, 0.0002, 0.0001])

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(beta_list):
        if time.monotonic() + 30 > deadline:
            break
        remaining = deadline - time.monotonic() - 10
        if remaining < 5:
            break

        est_time = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 5))))

        def obj(z_in):
            return _smoobj(z_in, beta, n)

        def grad(z_in):
            return _smograd(z_in, beta, n)

        try:
            result = minimize(obj, z, jac=grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-16,
                                       'gtol': 1e-14, 'maxcor': 100, 'maxls': 60})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 3 == 0:
                    print(f"      {label} beta{i}: sc={current_sc:.10f}")
            else:
                # Restart with perturbation
                rng = np.random.default_rng(i)
                z = z + 0.01 * rng.standard_normal(n)
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _shrinkage_and_reopt(a, deadline):
    """Try zeroing small elements and re-optimizing."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted = np.argsort(a)

    for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
        if time.monotonic() + 60 > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            # Re-optimize the shrunk version
            if time.monotonic() + 30 < deadline:
                a_reopt = _run_lbfgs_verbose(best_a, deadline - 10,
                                             label="shrink_lbfgs")
                sc_reopt = _score(a_reopt)
                if sc_reopt < best_sc:
                    best_sc = sc_reopt
                    best_a = a_reopt

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Combines FISTA, active-set Newton, L-BFGS,
    multi-n exploration, and sparsification.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"Parent: n={n_parent}, score={best_score:.10f}")
    else:
        n_parent = 16384
        best_seq = _make_profile(n_parent, "uniform")
        best_score = _score(best_seq)
        print(f"No parent. Uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: FISTA coarse (large beta) ===
    print(f"\n[P1] FISTA coarse (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 1200, t_start + 300)
    if time.monotonic() < p1_deadline:
        c = _full_convolve(best_seq, best_seq)
        c_max = float(c.max())
        beta_coarse = c_max * 0.1
        a1, sc1 = _fista_smooth(best_seq, beta_coarse, p1_deadline, max_iter=100000)
        if sc1 < best_score:
            best_seq = a1.copy()
            best_score = sc1
            print(f"  FISTA coarse: NEW BEST {best_score:.10f}")
        else:
            print(f"  FISTA coarse: no improvement (best={best_score:.10f})")

    # === Phase 2: FISTA medium ===
    print(f"\n[P2] FISTA medium (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 900, t_start + 500)
    if time.monotonic() < p2_deadline:
        c = _full_convolve(best_seq, best_seq)
        c_max = float(c.max())
        c_sorted = np.sort(c)[::-1]
        gap = c_max - c_sorted[min(20, len(c_sorted)-1)]
        beta_med = max(gap * 0.5, c_max * 0.02)
        a2, sc2 = _fista_smooth(best_seq, beta_med, p2_deadline, max_iter=100000)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  FISTA medium: NEW BEST {best_score:.10f}")

    # === Phase 3: Active-set Newton ===
    print(f"\n[P3] Active-set Newton (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 500, t_start + 800)
    if time.monotonic() < p3_deadline:
        a3, sc3 = _newton_active_set(best_seq, p3_deadline,
                                      max_newton_steps=300, cg_max_iter=80)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  Newton: NEW BEST {best_score:.10f}")
        else:
            print(f"  Newton: no improvement (best={best_score:.10f})")

    # === Phase 4: L-BFGS fine ===
    print(f"\n[P4] L-BFGS fine (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 400, t_start + 1000)
    if time.monotonic() < p4_deadline:
        a4 = _run_lbfgs_verbose(best_seq, p4_deadline, label="LBFGS-fine")
        sc4 = _score(a4)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  L-BFGS fine: NEW BEST {best_score:.10f}")

    # === Phase 5: Second Newton pass ===
    print(f"\n[P5] Newton pass 2 (t={time.monotonic()-t_start:.1f}s)")
    p5_deadline = min(deadline - 300, t_start + 1100)
    if time.monotonic() < p5_deadline:
        a5, sc5 = _newton_active_set(best_seq, p5_deadline,
                                      max_newton_steps=200, cg_max_iter=100)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  Newton 2: NEW BEST {best_score:.10f}")

    # === Phase 6: Ultra-fine L-BFGS ===
    print(f"\n[P6] Ultra-fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p6_deadline = min(deadline - 200, t_start + 1300)
    if time.monotonic() < p6_deadline:
        n = len(best_seq)
        c = _full_convolve(best_seq, best_seq)
        c_max = float(c.max())
        c_sorted = np.sort(c)[::-1]
        gap = c_max - c_sorted[min(10, len(c_sorted)-1)]
        beta_fin = max(gap * 0.1, c_max * 0.0005)
        beta_list = beta_fin * np.array([1.0, 0.5, 0.25, 0.125, 0.06, 0.03, 0.015])
        a6 = _run_lbfgs_verbose(best_seq, p6_deadline,
                                 beta_list=beta_list, label="LBFGS-ultra")
        sc6 = _score(a6)
        if sc6 < best_score:
            best_seq = a6.copy()
            best_score = sc6
            print(f"  L-BFGS ultra: NEW BEST {best_score:.10f}")

    # === Phase 7: Multi-n search ===
    print(f"\n[P7] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [n_parent]
    for n_try in [16384, 12288, 20480, 8192, 24576, 32768]:
        if n_try not in n_candidates:
            n_candidates.append(n_try)

    n_candidates.sort(key=lambda x: abs(x - n_parent))

    for n_try in n_candidates:
        if time.monotonic() + 80 > deadline:
            break
        if n_try == len(best_seq):
            continue
        print(f"  Trying n={n_try}...")
        a_res = _resample(best_seq, n_try)
        sc_res = _score(a_res)
        if sc_res < best_score:
            print(f"    Resampled score={sc_res:.10f} (improved!)")

        # Quick L-BFGS on resampled
        p7_deadline = min(deadline - 60, time.monotonic() + 60)
        if time.monotonic() < p7_deadline:
            a_opt = _run_lbfgs_verbose(a_res, p7_deadline, label=f"n={n_try}")
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try})")

        # Quick Newton on best at this n
        if time.monotonic() + 40 < deadline:
            a_nw, sc_nw = _newton_active_set(a_res, min(deadline - 50, time.monotonic() + 40),
                                             max_newton_steps=50, cg_max_iter=60)
            if sc_nw < best_score:
                best_seq = a_nw.copy()
                best_score = sc_nw
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}, newton)")

    # === Phase 8: Sparsification ===
    print(f"\n[P8] Sparsification (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() + 40 < deadline:
        a_shr, sc_shr = _shrinkage_and_reopt(best_seq, deadline - 20)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  Sparsification: NEW BEST {best_score:.10f}")

    # === Phase 9: Final Newton polish ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() + 20 < deadline:
        a_fin, sc_fin = _newton_active_set(best_seq, deadline - 5,
                                           max_newton_steps=50, cg_max_iter=100)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  Final Newton: NEW BEST {best_score:.10f}")

    # === Phase 10: Try alternative initializations if time ===
    if time.monotonic() + 60 < deadline:
        print(f"\n[P10] Alternative starts (t={time.monotonic()-t_start:.1f}s)")
        for kind in ["gaussian", "beta_dist", "cos_mod"]:
            if time.monotonic() + 40 > deadline:
                break
            a_alt = _make_profile(len(best_seq), kind)
            # Quick FISTA
            c_alt = _full_convolve(a_alt, a_alt)
            beta_alt = float(c_alt.max()) * 0.05
            a_alt_opt, sc_alt = _fista_smooth(a_alt, beta_alt,
                                              min(deadline - 20, time.monotonic() + 30),
                                              max_iter=30000)
            if sc_alt < best_score:
                best_seq = a_alt_opt.copy()
                best_score = sc_alt
                print(f"  Alt start '{kind}': NEW BEST {best_score:.10f}")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
