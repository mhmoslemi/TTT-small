import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _cg_solve(matvec, rhs, max_cg=500, tol=1e-15):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w
    dx = dx - np.mean(dx)
    if not np.all(np.isfinite(dx)):
        return None, None
    return dx, A_vec


def _active_set_newton_refined(a_init, deadline, max_outer=200):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        if outer < 30:
            tol_list = [0.01, 0.005, 0.002, 0.001]
        elif outer < 60:
            tol_list = [0.001, 0.0005, 0.0002, 0.0001]
        else:
            tol_list = [0.0001, 0.00005, 0.00002, 0.00001]

        applied = False
        for tol_frac in tol_list:
            dx, A_vec = _newton_step_precise(a, c, c_max, tol_frac)
            if dx is None:
                continue
            dx_norm = np.linalg.norm(dx)
            if dx_norm < 1e-16:
                continue
            t = 1.0
            for ls in range(60):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < c_max - 1e-14 * t * c_max:
                    a = a_try
                    c = c_try
                    c_max = f_try
                    sc = _score(a)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()
                    applied = True
                    break
                t *= 0.5
            if applied:
                break

        if not applied:
            g_all = 2.0 * _corr_a_A(a, np.ones(2 * n - 1) / (2 * n - 1), n)
            g_t = g_all - np.mean(g_all)
            gn = np.linalg.norm(g_t)
            if gn < 1e-16:
                break
            t = 0.01
            applied2 = False
            for ls in range(40):
                a_try = a - t * g_t / gn
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_sc:
                        best_sc = sc_try
                        best_a = a_try.copy()
                        a = a_try
                        applied2 = True
                        break
                t *= 0.5
            if not applied2:
                break

        if abs(c_max - float(_full_convolve(a, a).max())) < 1e-17:
            break
    return best_a, best_sc


def _spectral_flatten(a, deadline, n_iterations=200, flatten_power=2.0):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    N = _next_pow2(2 * n - 1)
    a_cur = a.copy()

    for it in range(n_iterations):
        if time.monotonic() > deadline:
            break
        F = np.fft.rfft(a_cur, N)
        mag = np.abs(F)
        mag = np.maximum(mag, 1e-30)
        target = np.ones_like(mag)
        target[0] = mag[0]
        blend = 0.5
        mag_new = mag * (1 - blend) + target * blend
        mag_new_pow = mag_new ** flatten_power
        mag_new_pow = mag_new_pow / mag_new_pow.max()
        phase = np.angle(F)
        F_new = mag_new_pow * np.exp(1j * phase)
        F_new[0] = mag[0]
        a_new = np.fft.irfft(F_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a_cur = a_new
    return best_a, best_sc


def _lbfgs_aggressive(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 100:
        gap = c_sorted[0] - c_sorted[100]
    else:
        gap = c_sorted[0] - c_sorted[-1]
    base_beta = max(gap * 0.5, max_c * 0.01)
    min_beta = max_c * 1e-9
    n_betas = 35
    log_betas = np.linspace(np.log(base_beta), np.log(min_beta), n_betas)
    betas = np.exp(log_betas)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.002 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 20))))
        try:
            def obj_fn(z_val):
                return _lsobj(z_val, beta, n)
            def grad_fn(z_val):
                return _lsgrad(z_val, beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 200, 'maxls': 500}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-18:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            if i % 5 == 0:
                print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.14f}")
        except Exception as ex:
            print(f"    L-BFGS beta={beta:.2e}: exception {ex}")
    return current_a


def _newton_polish(a_init, deadline, n_rounds=1000):
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    a = a_init.copy()

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol_frac = max(0.000001, 0.001 / (1 + rnd * 0.1))
        dx, A_vec = _newton_step_precise(a, c, c_max, tol_frac)
        if dx is None:
            if rnd > 50:
                break
            continue
        dx_norm = np.linalg.norm(dx)
        if dx_norm < 1e-18:
            break
        t = 1.0
        applied = False
        for ls in range(80):
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < c_max - 1e-15 * t * c_max:
                a = a_try
                c = c_try
                c_max = f_try
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                applied = True
                break
            t *= 0.5
        if not applied and rnd > 100:
            break
    return best_a, best_sc
