Now let me think about the peak-shaving QP approach in Phase 4.

Given the current sequence a and the autocorrelation c = conv(a,a), let P be the set of top-K peak positions. We want to find a small perturbation δ (with sum(δ) = 0) such that:
- a + δ ≥ 0
- max_{k in P} c[k] + 2*(∂c_k/∂a)·δ is minimized (first-order approximation)

This is a linear program (or QP if we include second-order terms):
minimize max_{k in P} (c[k] + 2 * g_k · δ)
subject to sum(δ) = 0, a + δ ≥ 0, ||δ|| ≤ ε

Where g_k = ∂c_k/∂a.

This is a small LP that can be solved efficiently. But for n=24576, even a small LP might be slow. Let me use a simpler approach:

**Greedy peak shaving**: 
1. Find the maximum peak c[k*]
2. Compute the gradient g = ∂c_{k*}/∂a
3. Find the direction that most reduces c[k*] while keeping sum(a) = 0 and a ≥ 0
   - This is: minimize g·δ subject to sum(δ)=0, a+δ≥0, ||δ||_∞ ≤ ε
   - The solution is: δ = -ε * sign(g - mean(g)), clipped to maintain a+δ≥0
4. Update a ← a + δ
5. Repeat

This is essentially a projected gradient step on the single peak. It's simple and fast.

For multiple peaks, we can use a weighted combination of gradients.

"""
Active-set optimizer with adaptive temperature, peak-shaving, 
frequency-inspired initialization, and diverse search strategies
for minimizing 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

# Two-level initialization
def _make_two_level(n, m_frac, ratio):
    """Create a two-level sequence: alpha for first m, beta for rest."""
    m = max(1, int(n * m_frac))
    a = np.zeros(n)
    a[:m] = 1.0
    a[m:] = ratio
    return _normalize(a)

# Frequency-shaped initialization
def _make_freq_shaped(n, n_peaks=3, seed=0):
    """Create a sequence whose power spectrum has controlled peaks."""
    rng = np.random.default_rng(seed)
    # Random phase
    phase = rng.standard_normal(n) * np.pi
    # Power spectrum: flat with some structure
    power = np.ones(n) + 0.1 * rng.standard_normal(n)
    power = np.maximum(power, 0.1)
    # Construct sequence
    A = np.sqrt(power) * np.exp(1j * phase)
    a = np.fft.irfft(A, n)
    a = np.maximum(a, 0)
    return _normalize(a)

# Cosine-modulated initialization
def _make_cosine_mod(n, freq, amp):
    """a[j] = 1 + amp * cos(2*pi*freq*j/n), clipped to non-negative."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2 * np.pi * freq * j / n)
    a = np.maximum(a, 0.01)
    return _normalize(a)

# Active-set gradient with softmax weighting
def _active_set_grad(a, c, temperature):
    """Compute gradient using softmax weighting of peaks."""
    n = len(a)
    c_max = float(c.max())
    z = (c - c_max) / temperature
    e = np.exp(np.clip(z, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    grad = 2.0 * h
    grad -= np.mean(grad)  # Project onto sum(a)=const
    return grad

def _adam_active_optimize(a_init, deadline, init_temp=0.01, min_temp=1e-5,
                           beta1=0.9, beta2=0.999, lr=0.01, min_time=5.0):
    """Adam-style optimization with adaptive temperature for active-set gradient."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_c = float(_full_convolve(a, a).max())
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    temp = init_temp
    stall = 0
    
    while time.monotonic() < deadline - min_time:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        if c_max < best_c - 1e-14:
            best_c = c_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        grad = _active_set_grad(a, c, temp)
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Adaptive step with line search
        alpha = lr / (np.sqrt(v_hat) + 1e-8)
        for _ in range(20):
            a_try = _normalize(a - alpha * m_hat)
            c_try_max = float(_full_convolve(a_try, a_try).max())
            if c_try_max < c_max - 1e-15:
                a = a_try
                break
            alpha *= 0.5
        
        # Anneal temperature
        if stall % 50 == 0:
            temp = max(min_temp, temp * 0.85)
        
        if stall > 500:
            break
    
    return best_a, 2.0 * n * best_c

def _peak_shave(a_init, deadline, n_peaks=10, min_time=5.0):
    """Iteratively shave the top peaks of the autocorrelation."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_c = float(_full_convolve(a, a).max())
    
    while time.monotonic() < deadline - min_time:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        if c_max < best_c - 1e-14:
            best_c = c_max
            best_a = a.copy()
        
        # Find top peaks
        if n_peaks >= len(c):
            peak_idx = np.argsort(c)[::-1]
        else:
            peak_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        
        # Compute weighted gradient
        peak_vals = c[peak_idx]
        weights = peak_vals / peak_vals.sum()
        
        grad = np.zeros(n)
        for idx, w in zip(peak_idx, weights):
            # grad += w * (a[k-j] + a[j-k]) for all j
            # This is w * conv(delta_k, a_rev)[n-1:2n-1] ... 
            # Simpler: use the full formula
            # ∂c[idx]/∂a[j] = a[idx-j] + a[j-idx] (with bounds)
            k = idx
            # a[k-j] for j in range(n): valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
            lo1 = max(0, k - n + 1)
            hi1 = min(n - 1, k)
            if lo1 <= hi1:
                grad[lo1:hi1+1] += w * a[k-hi1:k-lo1+1][::-1]
            
            # a[j-k] for j in range(n): valid when 0 <= j-k < n, i.e., k <= j <= k+n-1
            lo2 = max(0, k)
            hi2 = min(n - 1, k + n - 1)
            if lo2 <= hi2:
                grad[lo2:hi2+1] += w * a[lo2-k:hi2-k+1]
        
        # Project
        grad -= np.mean(grad)
        
        # Step
        step = 0.01
        improved = False
        for _ in range(30):
            a_try = _normalize(a - step * grad)
            c_try_max = float(_full_convolve(a_try, a_try).max())
            if c_try_max < c_max - 1e-15:
                a = a_try
                improved = True
                break
            step *= 0.5
        
        if not improved:
            break
    
    return best_a, 2.0 * n * best_c

def _lbfgs_smooth_objective(a_init, deadline, min_time=5.0):
    """L-BFGS with log-sum-exp smoothing."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-15:
        return a
    
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted)-1)]
    if gap < c_max * 0.001:
        gap = c_max * 0.001
    base_beta = max(gap * 0.5, c_max * 0.003)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075, 0.004, 0.002, 0.001])
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    current_sc = _score(a)
    
    for beta in betas:
        if time.monotonic() > deadline - min_time:
            break
        
        def f(z):
            aa = _softmax(z)
            cc = _full_convolve(aa, aa)
            cm = float(cc.max())
            shifted = (cc - cm) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                return 2.0 * n * cm
            return 2.0 * n * (cm + beta * np.log(s))
        
        def g(z):
            aa = _softmax(z)
            cc = _full_convolve(aa, aa)
            cm = float(cc.max())
            shifted = (cc - cm) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                w = np.zeros_like(cc)
                w[int(np.argmax(cc))] = 1.0
            else:
                w = e / s
            ar = aa[::-1]
            wc = _full_convolve(w, ar)
            h = wc[n-1:2*n-1]
            W = float(np.dot(w, cc))
            return 4.0 * n * aa * (h - W)
        
        try:
            remaining = deadline - min_time - time.monotonic()
            max_iter = min(3000, max(50, int(remaining / 0.02)))
            result = minimize(f, z, jac=g, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'ftol': 1e-16,
                                      'gtol': 1e-14, 'maxcor': 50})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc:
                a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return a

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return _normalize(a_new)

    if kind == "uniform":

        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.5) ** 2)

    elif kind == "random":
        rng = np.random.default_rng(777)
        a = rng.random(n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n
