        z=np.zeros_like(z)

            break

            break

```python
"""
Asymmetric multiscale minimizer for the L-infinity convolution-square problem.

The program treats the problem in scaled form:
    y = n * conv(a, a),  score = 2 * max(y),
with sum(a) = 1.  It combines:
  - log-domain L-BFGS on a log-sum-exp smoothed peak objective,
  - fixed active-weight L-BFGS cycles,
  - target-convolution matching with peak clipping,
  - a short momentum polish,
  - stochastic iterated local search,
  - coarse-to-fine multiscale exploration from asymmetric starts.

The parent sequence is used when available.  The final returned sequence is
normalized and expected to have manageable length for the final evaluator.
"""

import time
import numpy as np
from scipy.optimize import minimize

_T_START = 0.0


def _elapsed():
    return time.monotonic() - _T_START


def _next_pow2(m):
    k = 1
    while k < m:
        k <<= 1
    return max(2, k)


def _self_conv(a):
    """
    Full linear convolution a * a, using FFT.  Tiny negative numerical
    values are clipped to zero because the true values are non-negative.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    out = 2 * n - 1
    N = _next_pow2(out)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:out]
    c = np.where(c > 0.0, c, 0.0)
    return c


def _corr_lag(a, w):
    """
    For a length n and w length 2n-1, return h_i = sum_k w_k a_{k-i}
    for i=0,...,n-1.  This is the cross-correlation needed for gradients.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w[::-1], N)
    h = np.fft.irfft(fa * fw, N)
    idx = 2 * n - 2 - np.arange(n)
    return h[idx]


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-300:
        return np.ones(a.size, dtype=np.float64) / max(1, a.size)
    return a / s


def _score(a):
    """
    Scale-invariant score:
        2 * n * max(conv(a,a)) / sum(a)^2.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _self_conv(a)
    y = (float(n) / (s * s)) * c
    if y.size == 0:
        return np.inf
    return float(2.0 * float(y.max()))


def _log_normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 1e-300)
    z = np.log(a)
    return z - float(np.mean(z))


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - float(np.max(z))
    e = np.exp(np.clip(z, -700.0, 700.0))
    s = float(e.sum())
    if s < 1e-300:
        return np.ones(z.size, dtype=np.float64) / max(1, z.size)
    return e / s


def _candidate_key(item):
    return item[0]


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    b = np.interp(x_new, x_old, a)
    return _normalize(b)


def _weights_from_beta(y, beta):
    L = y.size
    M = float(y.max())
    if M <= 1e-300 or beta <= 1e-300:
        w = np.zeros(L, dtype=np.float64)
        w[int(np.argmax(y))] = 1.0
        return w
    sh = np.minimum((y - M) / beta, 0.0)
    e = np.exp(np.clip(sh, -700.0, 0.0))
    s = float(e.sum())
    if s < 1e-300:
        w = np.zeros(L, dtype=np.float64)
        w[int(np.argmax(y))] = 1.0
        return w
    return e / s


def _weights_tau(y, tau):
    return _weights_from_beta(y, tau)


def _objective_smooth(z, beta, n, reg):
    """
    Smoothed score with log-sum-exp peak:
        F = 2 * (y_max + beta * log(sum(exp((y-y_max)/beta))))
    where y = n * conv(a, a), a = softmax(z).
    """
    a = _softmax(z)
    y = float(n) * _self_conv(a)
    ym = float(y.max())
    if beta is None or beta <= 1e-300:
        S = ym
    else:
        sh = np.minimum((y - ym) / beta, 0.0)
        e = np.exp(np.clip(sh, -700.0, 0.0))
        s = float(e.sum())
        if s < 1e-300:
            S = ym
        else:
            S = ym + beta * float(np.log(s))
    val = 2.0 * S
    if reg:
        val += 0.5 * reg * float(np.dot(z, z))
    return float(val)


def _gradient_smooth(z, beta, n, reg):
    a = _softmax(z)
    y = float(n) * _self_conv(a)
    w = _weights_from_beta(y, beta)
    h = _corr_lag(a, w)
    W = float(np.dot(w, y))
    grad = 4.0 * a * (float(n) * h - W)
    if reg:
        grad = grad + reg * z
    grad = grad - float(np.mean(grad))
    return grad


def _objective_fixed_w(z, w, n, reg):
    a = _softmax(z)
    y = float(n) * _self_conv(a)
    val = 2.0 * float(np.dot(w, y))
    if reg:
        val += 0.5 * reg * float(np.dot(z, z))
    return float(val)


def _gradient_fixed_w(z, w, n, reg):
    a = _softmax(z)
    y = float(n) * _self_conv(a)
    h = _corr_lag(a, w)
    W = float(np.dot(w, y))
    grad = 4.0 * a * (float(n) * h - W)
    if reg:
        grad = grad + reg * z
    grad = grad - float(np.mean(grad))
    return grad


def _objective_target(z, target_y, n, reg):
    a = _softmax(z)
    y = float(n) * _self_conv(a)
    d = y - target_y
    val = 0.5 * float(np.dot(d, d))
    if reg:
        val += 0.5 * reg * float(np.dot(z, z))
    return float(val)


def _gradient_target(z, target_y, n, reg):
    a = _softmax(z)
    y = float(n) * _self_conv(a)
    d = y - target_y
    h = _corr_lag(a, d)
    W = float(np.dot(d, y))
    grad = 2.0 * a * (float(n) * h - W)
    if reg:
        grad = grad + reg * z
    grad = grad - float(np.mean(grad))
    return grad


def _run_lbfgs(z0, objfun, gradfun, args, deadline, min_time_remaining=0.0,
               chunk_iters=60, gtol=1e-7, ftol=1e-12, maxcor=50, maxls=20):
    """
    Run L-BFGS for several fixed-size chunks until an absolute deadline.
    The objective is used to decide whether to accept a chunk.  This is a
    workhorse routine; callers deal with the true score separately.
    """
    z = np.asarray(z0, dtype=np.float64).ravel().copy()
    if not np.all(np.isfinite(z)):
        z = np.zeros_like(z)
    z = z - float(np.mean(z))
    no_progress = 0

    while time.monotonic() < deadline - min_time_remaining:
        if time.monotonic() > deadline - 0.5:
            break
        try:
            before = float(objfun(z, *args))
        except Exception:
            break
        if not np.isfinite(before):
            break

        z_prev = z.copy()
        try:
            res = minimize(
                objfun, z, jac=gradfun, args=args,
                method="L-BFGS-B",
                options={
                    "maxiter": int(chunk_iters),
                    "ftol": float(ftol),
                    "gtol": float(gtol),
                    "maxcor": int(maxcor),
                    "maxls": int(maxls),
                }
            )
        except Exception:
            z = z_prev
            no_progress += 1
            if no_progress >= 3:
                break
            continue

        x = res.x
        if not np.all(np.isfinite(x)):
            z = z_prev
            no_progress += 1
            if no_progress >= 3:
                break
            continue

        z = x - float(np.mean(x))
        try:
            after = float(objfun(z, *args))
        except Exception:
            after = np.inf

        if np.isfinite(after) and after <= before + 1e-12:
            if before - after < 1e-15:
                no_progress += 1
            else:
                no_progress = 0
        else:
            z = z_prev
            no_progress += 1

        if no_progress >= 4:
            break

    return z


def _lbfgs_refine(a_init, objfun, gradfun, args, deadline,
                  min_time_remaining=10.0, seg_seconds=12.0,
                  keep_worsen=False, max_worsen=0.04,
                  chunk_iters=60, gtol=1e-7):
    """
    Repeated L-BFGS segments, restarting or continuing based on the true
    score.  For local refinement use keep_worsen=False.  For target matching
    a little temporary worsening can be useful; keep_worsen=True then keeps
    the current point unless it gets too bad.
    """
    a = _normalize(a_init)
    best_a = a.copy()
    best_sc = _score(a)
    z_work = _log_normalize(a)
    no_eff = 0

    while time.monotonic() < deadline - min_time_remaining:
        rem = deadline - min_time_remaining - time.monotonic()
        if rem < 2.0:
            break
        seg = min(float(seg_seconds), rem)
        seg = max(2.0, 0.9 * seg)
        seg_deadline = min(deadline - min_time_remaining, time.monotonic() + seg)
        z_old = z_work.copy()

        z_work = _run_lbfgs(
            z_work, objfun, gradfun, args,
            deadline=seg_deadline, min_time_remaining=0.0,
            chunk_iters=chunk_iters, gtol=gtol
        )

        a_try = _softmax(z_work)
        sc_try = np.inf
        if np.all(np.isfinite(a_try)):
            sc_try = _score(a_try)

        improved = False
        if np.isfinite(sc_try) and sc_try < best_sc - 1e-12:
            best_a = a_try.copy()
            best_sc = sc_try
            improved = True
            if not keep_worsen:
                z_work = _log_normalize(best_a)
            # if keep_worsen, continue from z_work
        else:
            if not keep_worsen:
                z_work = _log_normalize(best_a)
            elif not np.isfinite(sc_try) or sc_try > best_sc + max_worsen:
                z_work = _log_normalize(best_a)
            # else continue from z_work

        if not improved and np.allclose(z_work, z_old, rtol=1e-12, atol=1e-14):
            no_eff += 1
        else:
            no_eff = 0

        if no_eff >= 4:
            break

    return best_a


def _smooth_lbfgs_refine(a_init, deadline, min_time_remaining=20.0,
                         beta0_factor=0.05, beta_end_factor=1e-5):
    """
    Anneal the log-sum-exp temperature and refine by true score.
    """
    a = _normalize(a_init)
    best_sc = _score(a)
    if time.monotonic() >= deadline - min_time_remaining:
        return a

    n = a.size
    y = float(n) * _self_conv(a)
    M = float(y.max())
    if M <= 1e-300:
        return a

    y_sorted = np.sort(y)[::-1]
    k = min(80, y_sorted.size)
    gap = float(y_sorted[0] - y_sorted[k - 1])
    if gap < M * 0.003:
        gap = M * 0.003

    beta0 = max(M * beta0_factor, 0.5 * gap)
    min_beta = M * beta_end_factor

    factors = [
        1.0, 0.6, 0.4, 0.25, 0.15, 0.08, 0.045, 0.025,
        0.014, 0.008, 0.0045, 0.0025, 0.0012, 0.0006,
        0.0003, 0.00012, 0.00005, 0.00002, 0.000008
    ]

    for i, f in enumerate(factors):
        if time.monotonic() >= deadline - min_time_remaining:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10.0:
            break
        beta = max(min_beta, beta0 * f)
        beta_deadline = min(
            deadline - min_time_remaining,
            time.monotonic() + max(5.0, min(80.0, remaining / max(1, len(factors) - i) * 1.3))
        )
        a2 = _lbfgs_refine(
            a, _objective_smooth, _gradient_smooth,
            (beta, n, 1e-12),
            deadline=beta_deadline,
            min_time_remaining=1.0,
            seg_seconds=min(15.0, max(5.0, 0.5 * (beta_deadline - time.monotonic()))),
            keep_worsen=False,
            max_worsen=0.01,
            chunk_iters=50,
            gtol=1e-7
        )
        sc2 = _score(a2)
        if sc2 < best_sc - 1e-12:
            a = a2
            best_sc = sc2
            print(f"  [smooth-beta {beta / max(M,1e-15):.5f}] score={best_sc:.10f} t={_elapsed():.1f}s")

    return a


def _fixed_w_refine(a_init, deadline, min_time_remaining=20.0):
    """
    Refine using several fixed active-weight objectives.
    """
    a = _normalize(a_init)
    best_sc = _score(a)
    if time.monotonic() >= deadline - min_time_remaining:
        return a

    n = a.size
    tau_fracs = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004, 0.00015]

    for i, tau in enumerate(tau_fracs):
        if time.monotonic() >= deadline - min_time_remaining:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10.0:
            break
        y = float(n) * _self_conv(a)
        M = float(y.max())
        if M <= 1e-300:
            break
        w = _weights_tau(y, max(tau * M, 1e-6))
        wsum = float(w.sum())
        if wsum <= 1e-300:
            w = np.ones_like(y) / max(1, y.size)
        else:
            w = w / wsum

        tau_deadline = min(
            deadline - min_time_remaining,
            time.monotonic() + max(5.0, min(80.0, remaining / max(1, len(tau_fracs) - i) * 1.25))
        )
        a2 = _lbfgs_refine(
            a, _objective_fixed_w, _gradient_fixed_w,
            (w, n, 1e-12),
            deadline=tau_deadline,
            min_time_remaining=1.0,
            seg_seconds=min(15.0, max(5.0, 0.5 * (tau_deadline - time.monotonic()))),
            keep_worsen=False,
            max_worsen=0.01,
            chunk_iters=50,
            gtol=1e-7
        )
        sc2 = _score(a2)
        if sc2 < best_sc - 1e-12:
            a = a2
            best_sc = sc2
            print(f"  [fixed-w tau={tau:.5f}] score={best_sc:.10f} t={_elapsed():.1f}s")

    return a


def _make_target(y, M_target, n):
    """
    Build a target convolution y_target with sum n and capped maximum.
    Deficit from clipping is spread uniformly.
    """
    L = y.size
    cap = np.minimum(y, M_target)
    s = float(cap.sum())
    deficit = float(n) - s
    target = cap + deficit / max(1, L)
    target = np.maximum(target, 0.0)
    s2 = float(target.sum())
    if s2 <= 1e-300:
        return np.full(L, float(n) / max(1, L), dtype=np.float64)
    target = target * (float(n) / s2)
    return target


def _target_refine(a_init, target_y, deadline, min_time_remaining=10.0):
    """
    Optimize against a prescribed convolution target.
    """
    n = len(a_init)
    return _lbfgs_refine(
        a_init, _objective_target, _gradient_target,
        (target_y, n, 1e-10),
        deadline=deadline,
        min_time_remaining=min_time_remaining,
        seg_seconds=12.0,
        keep_worsen=True,
        max_worsen=0.05,
        chunk_iters=60,
        gtol=1e-6
    )


def _target_improve(a_init, deadline, min_time_remaining=30.0):
    """
    Try progressively lower peak caps and a flat target.
    """
    a = _normalize(a_init)
    best_sc = _score(a)
    n = a.size
    if time.monotonic() >= deadline - min_time_remaining:
        return a

    y_avg = float(n) / max(1, 2 * n - 1)
    M = best_sc / 2.0
    y = float(n) * _self_conv(a)

    factors = [0.998, 0.995, 0.99, 0.98, 0.96, 0.94, 0.90, 0.85]
    improved_once = False

    for i, f in enumerate(factors):
        if time.monotonic() >= deadline - min_time_remaining:
            break
        M_t = M * f
        if M_t <= y_avg * 1.03:
            break
        target = _make_target(y, M_t, n)
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 15.0:
            break
        seg = max(10.0, min(60.0, remaining / max(1, len(factors) - i) * 1.15))
        t_deadline = min(deadline - min_time_remaining, time.monotonic() + seg)
        a_try = _target_refine(a, target, t_deadline, 1.5)
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            a = a_try
            best_sc = sc_try
            M = best_sc / 2.0
            y = float(n) * _self_conv(a)
            improved_once = True
            print(f"  [target cap {f:.3f}] score={best_sc:.10f} t={_elapsed():.1f}s")
        else:
            # If the first two tight caps do not help, deeper caps often do
            # not either; save time for ILS and other tools.
            if not improved_once and i >= 2:
                break

    if time.monotonic() < deadline - min_time_remaining:
        target_flat = np.full(2 * n - 1, y_avg, dtype=np.float64)
        t_deadline = min(deadline - min_time_remaining, time.monotonic() + max(10.0, min(45.0, 0.35 * (deadline - min_time_remaining - time.monotonic()))))
        a_try = _target_refine(a, target_flat, t_deadline, 1.0)
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            a = a_try
            best_sc = sc_try
            print(f"  [target flat] score={best_sc:.10f} t={_elapsed():.1f}s")

    return a


def _momentum_polish(a_init, deadline, min_time_remaining=20.0,
                     tau_start=0.02, tau_end=1e-5,
                     lr_init=1.0, momentum=0.85):
    """
    Damped natural-gradient polish with adaptive peak weights.  Accepts only
    true score decreases, so it conserves the current best sequence.
    """
    a = _normalize(a_init)
    best_a = a.copy()
    best_sc = _score(a)
    current_a = a.copy()
    current_sc = best_sc
    n = a.size
    z = _log_normalize(current_a)
    v = np.zeros(n, dtype=np.float64)
    tau = float(tau_start)
    lr = float(lr_init)
    no_improve = 0

    while time.monotonic() < deadline - min_time_remaining:
        y = float(n) * _self_conv(current_a)
        M = float(y.max())
        if M <= 1e-300:
            break
        w = _weights_tau(y, max(tau * M, 1e-7))
        h = _corr_lag(current_a, w)
        W = float(np.dot(w, y))
        grad = 4.0 * current_a * (float(n) * h - W)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        v = momentum * v - lr * grad

        improved = False
        alphas = (1.0, 1.5, 2.0, 0.5, 0.25, 0.125, 0.0625, 0.03125)
        for alpha in alphas:
            z_try = z + alpha * v
            if not np.all(np.isfinite(z_try)):
                continue
            z_try = z_try - float(np.mean(z_try))
            a_try = _softmax(z_try)
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-13:
                current_a = a_try
                z = z_try
                current_sc = sc_try
                improved = True
                break

        if improved:
            if current_sc < best_sc - 1e-12:
                best_sc = current_sc
                best_a = current_a.copy()
                print(f"  [momentum] score={best_sc:.10f} t={_elapsed():.1f}s")
            no_improve = 0
            if tau > tau_end:
                tau = max(tau_end, tau * 0.98)
            lr = min(4.0 * lr_init, lr * 1.01)
        else:
            v = np.zeros(n, dtype=np.float64)
            no_improve += 1
            if no_improve % 5 == 0:
                tau = max(tau_end, tau * 0.75)
                lr = max(0.01, lr * 0.7)
            if no_improve >= 60:
                current_a = best_a.copy()
                current_sc = best_sc
                z = _log_normalize(current_a)
                v = np.zeros(n, dtype=np.float64)
                tau = max(tau_end, tau * 0.35)
                lr = 0.5 * lr_init
                no_improve = 0

    return best_a


def _random_asymmetric_profile(n, rng):
    a = rng.random(n)
    t = np.linspace(0.0, 1.0, n)
    tilt = 1.0 + rng.uniform(-0.9, 0.9) * t
    a = a * tilt + 1e-9
    return _normalize(a)


def _diverse_profiles(n, rng, limit=12):
    t = np.linspace(0.0, 1.0, n)
    outs = []

    outs.append(_random_asymmetric_profile(n, rng))

    for lam in (2.0, 5.0, 10.0):
        outs.append(_normalize(np.exp(-lam * t) + 1e-9))
        outs.append(_normalize(np.exp(-lam * (1.0 - t)) + 1e-9))

    for c in (0.3, 0.7):
        for sd in (0.12, 0.25):
            outs.append(_normalize(np.exp(-((t - c) / sd) ** 2) + 1e-9))

    for c in (0.3, 0.7):
        outs.append(_normalize(1.0 / (1.0 + np.exp(-12.0 * (t - c))) + 1e-9))

    for p in (0.5, 1.0, 2.0, 4.0):
        outs.append(_normalize(t ** p + 1e-9))
        outs.append(_normalize((1.0 - t) ** p + 1e-9))

    if len(outs) > limit:
        outs = outs[:limit]
    return outs


def _quick_refine(a_init, deadline, min_time_remaining=2.0):
    """
    Compact local refinement used by ILS and small-scale searches.
    """
    a = _normalize(a_init)
    sc = _score(a)
    if time.monotonic() >= deadline - min_time_remaining:
        return a
    n = a.size

    y = float(n) * _self_conv(a)
    M = float(y.max())
    if M > 1e-300 and time.monotonic() < deadline - min_time_remaining:
        beta = M * 0.05
        t_deadline = min(deadline - min_time_remaining, time.monotonic() + max(2.0, 0.45 * (deadline - min_time_remaining - time.monotonic())))
        a1 = _lbfgs_refine(
            a, _objective_smooth, _gradient_smooth,
            (beta, n, 1e-12),
            deadline=t_deadline,
            min_time_remaining=1.0,
            seg_seconds=5.0,
            keep_worsen=False,
            max_worsen=0.01,
            chunk_iters=40,
            gtol=1e-6
        )
        sc1 = _score(a1)
        if sc1 < sc - 1e-12:
            a = a1
            sc = sc1

    if time.monotonic() < deadline - min_time_remaining:
        y = float(n) * _self_conv(a)
        M = float(y.max())
        if M > 1e-300:
            w = _weights_tau(y, max(M * 0.01, 1e-6))
            wsum = float(w.sum())
            if wsum <= 1e-300:
                w = np.ones_like(y) / max(1, y.size)
            else:
                w = w / wsum
            t_deadline = min(deadline - min_time_remaining, time.monotonic() + max(2.0, 0.55 * (deadline - min_time_remaining - time.monotonic())))
            a2 = _lbfgs_refine(
                a, _objective_fixed_w, _gradient_fixed_w,
                (w, n, 1e-12),
                deadline=t_deadline,
                min_time_remaining=1.0,
                seg_seconds=5.0,
                keep_worsen=False,
                max_worsen=0.01,
                chunk_iters=40,
                gtol=1e-6
            )
            sc2 = _score(a2)
            if sc2 < sc - 1e-12:
                a = a2
                sc = sc2

    return a


def _multiscale_explore(deadline, rng, n_main):
    """
    Coarse-to-fine search for diverse asymmetric profiles.
    """
    if n_main <= 1024:
        return []

    sizes = []
    for s in (256, 512, 1024, 2048, 4096, 8192):
        if s < n_main:
            sizes.append(s)
    if not sizes:
        return []

    current = []
    for si, s in enumerate(sizes):
        if time.monotonic() > deadline - 60:
            break

        starts = []
        if si == 0:
            starts.extend(_diverse_profiles(s, rng, limit=8))
        else:
            for _, a in current[:3]:
                starts.append(_resample(a, s))
            starts.append(_random_asymmetric_profile(s, rng))

        for a in starts:
            if time.monotonic() > deadline - 40:
                break
            if s <= 512:
                add = 3.0
            elif s <= 1024:
                add = 5.0
            elif s <= 2048:
                add = 8.0
            elif s <= 4096:
                add = 12.0
            else:
                add = 18.0
            t_end = min(deadline - 30.0, time.monotonic() + add)
            a_opt = _quick_refine(a, t_end, 1.0)
            sc = _score(a_opt)
            current.append((sc, a_opt))

        current.sort(key=_candidate_key)
        current = current[:3]

    candidates = []
    for _, a in current[:3]:
        if time.monotonic() > deadline - 20:
            break
        a_main = _resample(a, n_main)
        t_end = min(deadline - 10.0, time.monotonic() + max(15.0, min(40.0, 0.25 * (deadline - time.monotonic()))))
        a_opt = _quick_refine(a_main, t_end, 1.5)
        candidates.append(a_opt)

    return candidates


def _ils(a_best, deadline, rng, min_time_remaining=30.0):
    """
    Stochastic iterated local search with annealed acceptance.
    """
    n = len(a_best)
    current_a = _normalize(a_best).copy()
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc
    y_avg = float(n) / max(1, 2 * n - 1)

    temp = 0.02
    it = 0

    while time.monotonic() < deadline - min_time_remaining:
        rem = deadline - min_time_remaining - time.monotonic()
        if rem < 8.0:
            break

        decision = rng.random()

        # Target-based escape: build a capped target from the current best
        # and optimize from the current point.
        if decision < 0.30:
            yb = float(n) * _self_conv(best_a)
            Mb = best_sc / 2.0
            f = [0.998, 0.995, 0.99, 0.98, 0.96, 0.94][int(rng.integers(6))]
            Mt = Mb * f
            if Mt <= y_avg * 1.03:
                f = 0.998
                Mt = Mb * f
                if Mt <= y_avg * 1.03:
                    Mt = y_avg * 1.08
            target = _make_target(yb, Mt, n)
            short = max(3.0, min(10.0, 0.35 * rem))
            t_end = min(deadline - min_time_remaining, time.monotonic() + short)
            a_opt = _target_refine(current_a, target, t_end, 1.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_sc - 1e-12:
                best_sc = sc_opt
                best_a = a_opt.copy()
                current_a = a_opt.copy()
                current_sc = sc_opt
                print(f"  [ils-target {f:.3f}] NEW BEST={best_sc:.10f} t={_elapsed():.1f}s")
            else:
                if sc_opt < current_sc + temp:
                    current_a = a_opt.copy()
                    current_sc = sc_opt
            it += 1
            temp = max(0.0004, temp * 0.96)
            continue

        # Other perturbations, followed by quick local refinement.
        if decision < 0.45:
            sigma = [0.002, 0.005, 0.01, 0.02, 0.04][int(rng.integers(5))]
            z = _log_normalize(current_a)
            z = z + sigma * rng.standard_normal(n)
            a_pert = _softmax(z)
        elif decision < 0.60:
            block_size = max(10, n // 12)
            start = int(rng.integers(0, n - block_size))
            z = _log_normalize(current_a)
            sigma = [0.005, 0.015, 0.03][int(rng.integers(3))]
            z[start:start + block_size] += sigma * rng.standard_normal(block_size)
            a_pert = _softmax(z)
        elif decision < 0.75:
            b = _random_asymmetric_profile(n, rng)
            mu = [0.08, 0.15, 0.25, 0.40][int(rng.integers(4))]
            a_pert = _normalize((1.0 - mu) * current_a + mu * b)
        elif decision < 0.85:
            seg_len = max(10, n // 30)
            start = int(rng.integers(0, n - seg_len))
            a_pert = current_a.copy()
            a_pert[start:start + seg_len] = a_pert[start:start + seg_len][::-1]
            z = _log_normalize(a_pert)
            z += 0.004 * rng.standard_normal(n)
            a_pert = _softmax(z)
        else:
            z = _log_normalize(best_a)
            sigma = [0.003, 0.008, 0.015, 0.03][int(rng.integers(4))]
            z = z + sigma * rng.standard_normal(n)
            a_pert = _softmax(z)

        short = max(4.0, min(14.0, 0.35 * rem))
        opt_end = min(deadline - min_time_remaining, time.monotonic() + short)
        a_opt = _quick_refine(a_pert, opt_end, 1.0)
        sc_opt = _score(a_opt)

        if sc_opt < best_sc - 1e-12:
            best_sc = sc_opt
            best_a = a_opt.copy()
            current_a = a_opt.copy()
            current_sc = sc_opt
            temp = max(0.0004, temp * 0.85)
            print(f"  [ils] NEW BEST={best_sc:.10f} t={_elapsed():.1f}s")
        elif sc_opt < current_sc + temp:
            current_a = a_opt.copy()
            current_sc = sc_opt

        if current_sc > best_sc + 0.05 and it % 5 == 0:
            current_a = best_a.copy()
            current_sc = best_sc

        it += 1
        temp = max(0.0004, temp * 0.96)

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for a non-negative sequence minimizing
        2 * n * max(conv(a, a)) / sum(a)^2.
    The function returns the best list found before the time budget expires.
    """
    global _T_START
    _T_START = time.monotonic()
    deadline = _T_START + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_a = None
    best_sc = np.inf
    n_main = 24576

    # Load parent if present.
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            if float(parent_raw.sum()) > 0.01:
                parent = parent_raw
    except Exception:
        parent = None

    if parent is not None:
        parent_len = parent.size
        if 4096 <= parent_len <= 32768:
            n_main = parent_len
        else:
            n_main = 24576
        if parent_len != n_main:
            parent = _resample(parent, n_main)
        else:
            parent = _normalize(parent)

    if parent is not None:
        best_a = parent.copy()
        best_sc = _score(best_a)
        print(f"Parent: n={best_a.size}, local_score={best_sc:.10f}")
    else:
        a0 = _random_asymmetric_profile(n_main, rng)
        best_a = a0
        best_sc = _score(best_a)
        print(f"No parent: random n={best_a.size}, local_score={best_sc:.10f}")

    def maybe_update(a_cand):
        nonlocal best_a, best_sc
        if a_cand is None:
            return
        a_cand = np.asarray(a_cand, dtype=np.float64).ravel()
        a_cand = np.maximum(a_cand, 0.0)
        s = float(a_cand.sum())
        if s < 1e-12:
            return
        a_cand = a_cand / s
        sc = _score(a_cand)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_a = a_cand.copy()
            best_sc = sc
            print(f"NEW BEST: {best_sc:.10f} t={_elapsed():.1f}s")

    # Phase 0: multiscale global exploration.
    if deadline - time.monotonic() > 420:
        ms_end = min(deadline - 320, time.monotonic() + 260)
        print(f"[Phase 0] Multiscale exploration t={_elapsed():.1f}s")
        cands = _multiscale_explore(ms_end, rng, len(best_a))
        for c in cands:
            maybe_update(c)

    # Phase 1: main smooth L-BFGS anneal on best.
    if deadline - time.monotonic() > 480:
        ph1_end = min(deadline - 380, time.monotonic() + 430)
        print(f"[Phase 1] Smooth L-BFGS t={_elapsed():.1f}s")
        maybe_update(_smooth_lbfgs_refine(
            best_a, ph1_end,
            min_time_remaining=30,
            beta0_factor=0.05,
            beta_end_factor=1e-5
        ))

    # Phase 2: fixed active-weight L-BFGS.
    if deadline - time.monotonic() > 320:
        ph2_end = min(deadline - 250, time.monotonic() + 260)
        print(f"[Phase 2] Fixed-weight L-BFGS t={_elapsed():.1f}s")
        maybe_update(_fixed_w_refine(best_a, ph2_end, min_time_remaining=30))

    # Phase 3: target clipping / square-root-like matching.
    if deadline - time.monotonic() > 220:
        ph3_end = min(deadline - 160, time.monotonic() + 170)
        print(f"[Phase 3] Target matching t={_elapsed():.1f}s")
        maybe_update(_target_improve(best_a, ph3_end, min_time_remaining=25))

    # Phase 4: stochastic ILS.
    if deadline - time.monotonic() > 160:
        ph4_end = min(deadline - 80, time.monotonic() + 220)
        print(f"[Phase 4] ILS t={_elapsed():.1f}s")
        maybe_update(_ils(best_a, ph4_end, rng, min_time_remaining=30))

    # Phase 5: momentum polish.
    if deadline - time.monotonic() > 80:
        ph5_end = min(deadline - 35, time.monotonic() + 120)
        print(f"[Phase 5] Momentum polish t={_elapsed():.1f}s")
        maybe_update(_momentum_polish(best_a, ph5_end, min_time_remaining=15,
                                      tau_start=0.02, tau_end=1e-5,
                                      lr_init=1.0, momentum=0.85))

    # Phase 6: final low-temperature smooth L-BFGS.
    if deadline - time.monotonic() > 12:
        fin_end = max(time.monotonic(), min(deadline - 4.0, time.monotonic() + (deadline - time.monotonic() - 4.0)))
        print(f"[Phase 6] Final low-beta L-BFGS t={_elapsed():.1f}s")
        maybe_update(_smooth_lbfgs_refine(
            best_a, fin_end,
            min_time_remaining=2,
            beta0_factor=0.012,
            beta_end_factor=2e-6
        ))

    # Final validation and return.
    if best_a is None:
        best_a = _random_asymmetric_profile(256, rng)

    final = np.asarray(best_a, dtype=np.float64).ravel()
    final = np.clip(final, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _normalize(_ones_safe(len(final)))

    local_final = _score(final)
    print(f"FINAL local: score={local_final:.10f}, n={final.size}, t={_elapsed():.1f}s")

    # Optionally use the provided evaluator for a single sanity check.
    try:
        if final.size <= 30000 and (time.monotonic() < deadline - 5.0):
            actual = evaluate_sequence([float(x) for x in final])
            print(f"FINAL actual evaluate_sequence: {actual}")
    except Exception as e:
        print(f"evaluate_sequence sanity check skipped: {e}")

    return [float(x) for x in final]
```
