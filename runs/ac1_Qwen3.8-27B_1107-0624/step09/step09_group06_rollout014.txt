For a pair swap (i, j, delta):
delta_c[k] = delta*(a[k-j] + a[j-k] - a[k-i] - a[i-k]) + delta^2*(delta_{i,j,k} terms)

The delta^2 term is nonzero only when {i,j} = {k-i, k-j} or similar. It's a small correction.

So I can compute delta_c in O(n) without doing a full FFT. Then new_c = c + delta_c, and new_max = max(new_c).

This makes each trial O(n) instead of O(n log n), which is a factor of ~14x faster.

Smooth L-BFGS + Fast Pair-Swap Coordinate Descent + Nesterov Subgradient.
Key innovation: O(n) analytic pair-swap evaluation enabling thousands of
fine local adjustments, combined with adaptive smooth-max L-BFGS and
multi-n exploration to escape local minima.

import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _pair_swap_eval(c, a, i, j, delta, n):
    """Compute new convolution after moving delta mass from index i to index j.
    O(n) analytic evaluation without FFT."""
    new_c = c.copy()
    # Change: a[i] -= delta, a[j] += delta
    # delta_c[k] = delta * (a[k-j] + a[j-k] - a[k-i] - a[i-k])

    lo = i
    hi = min(i + n - 1, 2 * n - 2)
    new_c[lo:hi+1] -= delta * a[:(hi - lo + 1)]
    
    # -a[i-k] term: k from max(0, i-n+1) to i
    lo = max(0, i - n + 1)
    hi = i
    new_c[lo:hi+1] -= delta * a[hi::-1][: (hi - lo + 1)]
    
    # +a[k-j] term: k from j to min(j+n-1, 2n-2)
    lo = j
    hi = min(j + n - 1, 2 * n - 2)
    new_c[lo:hi+1] += delta * a[:(hi - lo + 1)]
    
    # +a[j-k] term: k from max(0, j-n+1) to j
    lo = max(0, j - n + 1)
    hi = j
    new_c[lo:hi+1] += delta * a[hi::-1][: (hi - lo + 1)]
    
    return new_c

        total_iter += 1
        if total_iter % 500 == 0:
            print(f"    PairSwap iter={total_iter}: best={best_sc:.10f}, cur={2.0*n*float(_full_convolve(a,a).max()):.10f}")
    
    # Final check
    final_sc = 2.0 * n * float(_full_convolve(a, a).max())
    if final_sc < best_sc:
        best_a = a.copy()
        best_sc = final_sc
    
    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, n_betas=12, max_iter_base=3000):
    """Adaptive-beta L-BFGS on softmax parameterization."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
                                   0.0156, 0.0078, 0.0039, 0.002, 0.001, 0.0005,
                                   0.0002, 0.0001, 0.00005, 0.00002])[:n_betas]
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        max_iter = min(max_iter_base, max(20, int(remaining / (0.003 * max(1.0, n / 8192.0) * 5))))
        
        def obj(z_in):
            return _smoobj(z_in, beta, n)
        
        def grad(z_in):
            return _smograd(z_in, beta, n)
        
        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-14,
                    'maxcor': 50,
                    'maxls': 40,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 3 == 0:
                    print(f"    L-BFGS beta={beta:.2e} (idx={i}): score={current_sc:.10f}")
        except Exception:
            pass
    
    return current_a


def _nesterov_subgrad(a_init, deadline, min_time_remaining=60.0, seed=0):
    """Nesterov-accelerated subgradient with adaptive active set."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())
    
    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    momentum = 0.8
    base_lr = 0.01
    lr = base_lr
    K = 10
    t = 0
    no_improve = 0
    stall_count = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive K
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
        if gap < c_max * 0.003 and K < 200:
            K = min(200, int(K * 1.4) + 1)
        
        # Select active peaks
        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]
        
        # Compute subgradient (sum of gradients at active peaks)
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi+1] += a[k - np.arange(lo, hi + 1)]
            lo2 = max(0, k)
            hi2 = min(n - 1, k + n - 1)
            if lo2 <= hi2:
                grad[lo2:hi2+1] += a[np.arange(lo2, hi2 + 1) - k]
        
        # Project to tangent space
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break
        
        effective_lr = lr / (gnorm + 1e-18) * 0.1
        v = momentum * v - effective_lr * grad_t
        a_cand = a + v
        a_proj = _project_simplex(a_cand)
        
        c_new = _full_convolve(a_proj, a_proj)
        sc_new = 2.0 * n * float(c_new.max())
        
        if sc_new < current_sc:
            a = a_proj
        else:
            v = 0.5 * v
            lr *= 0.92
            a = a_proj
        
        if no_improve > 0 and no_improve % 200 == 0:
            stall_count += 1
            if stall_count > 8:
                break
            perturb = 0.002 * rng.standard_normal(n)
            a_pert = _project_simplex(a + perturb)
            c_pert = _full_convolve(a_pert, a_pert)
            sc_pert = 2.0 * n * float(c_pert.max())
            if sc_pert < best_sc:
                a = a_pert
                best_a = a.copy()
                best_sc = sc_pert
            v = np.zeros(n)
            lr = base_lr
            no_improve = 0
    
    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann_like":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        a[0] = max(a[0], 1e-8)
        a[-1] = max(a[-1], 1e-8)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.3 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":
        taper = max(1, n // 32)
        a = np.ones(n)
        for i in range(taper):
            frac = (i + 1) / taper
            a[i] = frac * frac
            a[n - 1 - i] = frac * frac
        return a / a.sum()
    elif kind == "bspline3":
        # Approximate 3rd order B-spline (cubic)
        t = np.linspace(0, 1, n)
        a = np.zeros(n)
        m = 4
        for i in range(n):
            x = (i + 0.5) / n * m
            if 0 < x < 3:
                a[i] = (1/6) * ((3-x)**3 - 4*(2-x)**3 + 6*(1-x)**3 + (-x)**3)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """Main search: L-BFGS + Pair-Swap + Nesterov hybrid with multi-n exploration."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")
    
    def _try_improve(a, deadline_override=None):
        """Run full optimization pipeline on sequence a. Returns best found."""
        nonlocal best_seq, best_score
        n = len(a)
        current_best = best_seq.copy() if best_seq is not None else a.copy()
        current_best_sc = best_score
        
        def _update(a_new, sc_new, tag=""):
            nonlocal best_seq, best_score, current_best, current_best_sc
            if sc_new < current_best_sc:
                current_best = a_new.copy()
                current_best_sc = sc_new
            if sc_new < best_score:
                best_seq = a_new.copy()
                best_score = sc_new
                print(f"    NEW BEST: {sc_new:.10f} [{tag}] t={time.monotonic()-t_start:.1f}s")
        
        dl = deadline_override if deadline_override is not None else deadline
        
        # Step 1: Quick L-BFGS
        if time.monotonic() < dl - 120:
            a1 = _run_lbfgs(a, dl, min_time_remaining=60, n_betas=8, max_iter_base=2000)
            sc1 = _score(a1)
            _update(a1, sc1, "lbfgs1")
            a = a1
        
        # Step 2: Pair-swap local search (KEY INNOVATION)
        if time.monotonic() < dl - 120:
            a2, sc2 = _pair_swap_refine(a, dl, min_time_remaining=60, seed=seed)
            _update(a2, sc2, "pairswap1")
            if sc2 < _score(a):
                a = a2
        
        # Step 3: L-BFGS refinement after pair-swap
        if time.monotonic() < dl - 100:
            a3 = _run_lbfgs(a, dl, min_time_remaining=60, n_betas=10, max_iter_base=3000)
            sc3 = _score(a3)
            _update(a3, sc3, "lbfgs2")
            a = a3
        
        # Step 4: Second pair-swap round
        if time.monotonic() < dl - 80:
            a4, sc4 = _pair_swap_refine(a, dl, min_time_remaining=60, seed=seed + 1)
            _update(a4, sc4, "pairswap2")
            if sc4 < _score(a):
                a = a4
        
        # Step 5: Nesterov subgradient
        if time.monotonic() < dl - 60:
            a5, sc5 = _nesterov_subgrad(a, dl, min_time_remaining=40, seed=seed + 2)
            _update(a5, sc5, "nesterov")
            if sc5 < _score(a):
                a = a5
        
        # Step 6: Final L-BFGS polish
        if time.monotonic() < dl - 30:
            a6 = _run_lbfgs(a, dl, min_time_remaining=20, n_betas=6, max_iter_base=1500)
            sc6 = _score(a6)
            _update(a6, sc6, "final_lbfgs")
            a = a6
        
        # Step 7: Final pair-swap polish
        if time.monotonic() < dl - 15:
            a7, sc7 = _pair_swap_refine(a, dl, min_time_remaining=10, seed=seed + 3)
            _update(a7, sc7, "final_pairswap")
            a = a7
        
        return current_best, current_best_sc
    
    # === Main Phase: Optimize parent ===
    print(f"\n[Main] Optimizing parent n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    best_seq = _try_improve(best_seq)[0]
    
    # === Multi-n exploration ===
    print(f"\n[Multi-n] Exploring other lengths (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [16384, 32768, 65536, 8192, 24576]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)
    
    time_per_n = max(60, (deadline - time.monotonic()) / (len(n_candidates) + 2))
    
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try == len(best_seq):
            continue
        
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: resampled score={sc_init:.10f}")
        
        n_deadline = min(deadline - 30, time.monotonic() + time_per_n)
        best_seq = _try_improve(a_init, n_deadline)[0]
    
    # === Try diverse profiles ===
    print(f"\n[Diverse] Trying different profiles (t={time.monotonic()-t_start:.1f}s)")
    profiles = ["tapered", "hann_like", "flat_taper", "raised_cos"]
    n_div = len(best_seq)
    
    for prof in profiles:
        if time.monotonic() > deadline - 40:
            break
        a_prof = _make_profile(n_div, prof)
        # Blend with best
        a_blend = 0.5 * a_prof + 0.5 * best_seq
        a_blend = a_blend / a_blend.sum()
        
        sc_blend = _score(a_blend)
        print(f"  Profile '{prof}': blend score={sc_blend:.10f}")
        
        if sc_blend < best_score + 0.01:  # Only optimize if competitive
            short_dl = min(deadline - 30, time.monotonic() + 60)
            best_seq = _try_improve(a_blend, short_dl)[0]
    
    # === Shrinkage ===
    print(f"\n[Shrink] Trying shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        a_shr = best_seq.copy()
        n = len(a_shr)
        scores_before = 2.0 * n * float(_full_convolve(a_shr, a_shr).max())
        
        for frac in [0.001, 0.002, 0.005, 0.01, 0.02]:
            if time.monotonic() > deadline - 20:
                break
            n_zero = max(1, int(n * frac))
            a_try = a_shr.copy()
            sorted_idx = np.argsort(a_try)
            a_try[sorted_idx[:n_zero]] = 0.0
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_score:
                best_seq = a_try.copy()
                best_score = sc_try
                a_shr = a_try
                print(f"  Shrink frac={frac}: NEW BEST {sc_try:.10f}")
    
    # === Final validation ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
