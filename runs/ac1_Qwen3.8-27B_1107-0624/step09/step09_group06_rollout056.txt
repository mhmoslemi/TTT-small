def _next_pow2(n):
    k = 1

    return max(2, k)

"""
Autocorrelation flatness optimizer: combines mirror descent (KL projection) 
for coarse optimization with L-BFGS on log-sum-exp for fine polishing.
Uses phase-retrieval-inspired initialization and adaptive temperature annealing.
Key improvements: KL mirror descent for fast coarse convergence, more aggressive
low-temperature L-BFGS, and structured initialization from parent.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _conv_and_grad(a, beta):
    """Compute objective and gradient of the smoothed objective w.r.t. a (on simplex).
    Returns (objective_value, grad_a) where grad_a is dF/da."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0.0))
    S = float(e.sum())
    if S < 1e-300 or not np.isfinite(S):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        w = e / S
        obj = 2.0 * n * (c_max + beta * np.log(S))
    # Gradient: 2n * (P + Q) where Q = conv(w,a)[:n], P[j] = sum_k w[k]*a[k-j]
    wa = _full_convolve(w, a)
    Q = wa[:n]
    w_rev = w[::-1]
    wa_r = _full_convolve(w_rev, a)
    # P[j] = wa_r[2n-2-j] for j=0..n-1
    P = wa_r[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * n * (P + Q)
    return obj, grad_a


def _z_gradient(z, beta, n):
    """Gradient of F(softmax(z)) w.r.t. z."""
    a = _exp_normalize(z)
    _, grad_a = _conv_and_grad(a, beta)
    # dF/dz_k = a_k * (grad_a_k - sum_j a_j * grad_a_j)
    mean_ag = float(np.dot(a, grad_a))
    return a * (grad_a - mean_ag)


def _z_objective(z, beta, n):
    a = _exp_normalize(z)
    obj, _ = _conv_and_grad(a, beta)
    return obj


def _exp_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _mirror_descent(a, deadline, min_remaining=60.0, seed=0):
    """KL mirror descent on the log-sum-exp objective with annealing."""
    n = len(a)
    rng = np.random.default_rng(seed)
    a = a.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Start with moderate temperature
    c0 = _full_convolve(a, a)
    c_max0 = float(c0.max())
    c_sorted = np.sort(c0)[::-1]
    gap0 = c_max0 - c_sorted[min(20, len(c_sorted) - 1)]
    T = max(gap0, c_max0 * 0.01)

    eta = 0.5 / (2.0 * n * c_max0 + 1e-10)
    best_eta = eta

    no_improve = 0
    t_iter = 0
    stall_count = 0

    while time.monotonic() + min_remaining < deadline:
        t_iter += 1

        _, grad_a = _conv_and_grad(a, T)

        # Mirror descent step: a_new = a * exp(-eta * grad) / sum(...)
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - eta * grad_a
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(np.clip(log_a_new, -700, 700))
        a_new /= a_new.sum()

        # Evaluate
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            a = a_new  # still move (mirror descent is monotone in the smooth case)
            no_improve += 1

        # Adaptive temperature: decrease when progress is being made
        if t_iter % 50 == 0:
            c_cur = _full_convolve(a, a)
            c_max_cur = float(c_cur.max())
            c_s = np.sort(c_cur)[::-1]
            gap_cur = c_max_cur - c_s[min(20, len(c_s) - 1)]
            T = max(gap_cur * 0.5, c_max_cur * 0.001)
            # Decrease step size slowly
            eta = 0.5 / (2.0 * n * c_max_cur + 1e-10) * 0.8

        # Stall detection
        if no_improve > 200:
            stall_count += 1
            if stall_count > 15:
                break
            eta *= 0.5
            # Random restart perturbation
            if stall_count % 5 == 0:
                pert = 0.005 * rng.standard_normal(n)
                a_pert = np.maximum(a + pert, 0)
                a_pert /= a_pert.sum()
                sc_pert = _score(a_pert)
                if sc_pert < best_sc:
                    a = a_pert
                    best_a = a.copy()
                    best_sc = sc_pert
                    no_improve = 0
                    stall_count = 0
                else:
                    # revert
                    a = best_a.copy()
            no_improve = 0

        if t_iter % 500 == 0:
            print(f"    Mirror t={t_iter}: best={best_sc:.10f}, T={T:.2e}, eta={eta:.2e}")

    return best_a, best_sc


def _lbfgs_anneal(a_init, deadline, min_remaining=30.0, n_temps=14):
    """L-BFGS with geometric temperature annealing."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c0 = _full_convolve(a, a)
    c_max0 = float(c0.max())
    c_sorted = np.sort(c0)[::-1]
    gap0 = c_max0 - c_sorted[min(30, len(c_sorted) - 1)]
    if gap0 < c_max0 * 0.001:
        gap0 = c_max0 * 0.001
    T0 = max(gap0, c_max0 * 0.005)

    # Geometric schedule from T0 down to very small
    T_end = c_max0 * 0.00005
    ratios = np.geomspace(T0, max(T_end, T0 * 1e-6), n_temps)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    for ti, beta in enumerate(ratios):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break

        # Estimate iterations: each iteration ~ 3 convolutions * 2 (obj+grad) * 2 (line search)
        # ~ 0.3ms per iter for n=16384
        est_ms = 0.3 * max(1.0, n / 16384.0)
        max_iter = int(remaining * 1000 / (est_ms * 8))
        max_iter = max(50, min(max_iter, 50000))

        # For very small beta, allow more iterations
        if beta < T0 * 0.01:
            max_iter = max(max_iter, int(remaining * 1000 / (est_ms * 15)))
            max_iter = min(max_iter, 80000)

        class _Obj:
            def __call__(self, z):
                return _z_objective(z, beta, n)

            def grad(self, z):
                return _z_gradient(z, beta, n)

        prob = _Obj()
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 100,
                    'maxls': 80,
                }
            )
            a_new = _exp_normalize(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new
                z = np.log(np.maximum(a, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS tier {ti}/{n_temps}: beta={beta:.2e}, score={best_sc:.10f}")
        except Exception as ex:
            print(f"    L-BFGS tier {ti} error: {ex}")

    return best_a, best_sc


def _stochastic_subgrad(a, deadline, min_remaining=30.0, seed=0):
    """Stochastic subgradient with Nesterov momentum on random subsets of peaks."""
    n = len(a)
    rng = np.random.default_rng(seed)
    a = a.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c0 = _full_convolve(a, a)
    c_max0 = float(c0.max())

    v = np.zeros(n)
    momentum = 0.9
    base_lr = 0.02 / (n * c_max0)
    lr = base_lr
    K = 20
    no_improve = 0
    t_iter = 0

    while time.monotonic() + min_remaining < deadline:
        t_iter += 1

        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive K
        c_sorted = np.sort(c)[::-1]
        if t_iter % 100 == 0:
            gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
            if gap < c_max * 0.01 and K < 200:
                K = min(200, int(K * 1.4) + 1)

        # Random subset of top peaks
        top_idx = np.argsort(c)[::-1][:K]
        sample_size = min(K, max(5, K // 3))
        if t_iter % 2 == 0 and sample_size < K:
            active = rng.choice(top_idx, size=sample_size, replace=False)
        else:
            active = top_idx

        # Compute subgradient: average of dc_k/da for k in active
        grad = np.zeros(n)
        for k in active:
            # dc_k/da_j = a[k-j] + a[j-k]
            # Use vectorized computation
            j_idx = np.arange(n)
            # a[k-j]: valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
            j_lo1 = max(0, k - n + 1)
            j_hi1 = min(n - 1, k)
            if j_lo1 <= j_hi1:
                idx = k - np.arange(j_lo1, j_hi1 + 1)
                grad[j_lo1:j_hi1 + 1] += a[idx]
            # a[j-k]: valid when 0 <= j-k < n, i.e., k <= j <= k+n-1
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_lo2 <= j_hi2:
                idx = np.arange(j_lo2, j_hi2 + 1) - k
                grad[j_lo2:j_hi2 + 1] += a[idx]

        grad /= len(active)

        # Project to tangent space
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-20:
            break

        # Nesterov step
        v = momentum * v - lr * grad_t / gnorm
        a_cand = a + v
        a_proj = np.maximum(a_cand, 0.0)
        s_proj = a_proj.sum()
        if s_proj > 1e-15:
            a_proj = a_proj / s_proj
        else:
            a_proj = a.copy()

        a = a_proj

        # Adaptive learning rate
        if t_iter % 50 == 0:
            c_cur = _full_convolve(a, a)
            lr = 0.02 / (n * float(c_cur.max()) + 1e-10)

        if no_improve > 300:
            if no_improve > 1500:
                break
            lr *= 0.7
            no_improve = 0

        if t_iter % 1000 == 0:
            print(f"    Stoch t={t_iter}: best={best_sc:.10f}, K={K}, lr={lr:.2e}")

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_initial(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":
        a = np.ones(n)
        taper = max(1, n // 100)
        for i in range(taper):
            w = (i + 1) / taper
            a[i] = w
            a[n - 1 - i] = w
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.15 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    else:
        return np.ones(n) / n


def _phase_retrieval_init(n, iterations=100, seed=0):
    """Initialize using phase retrieval targeting a flat convolution."""
    rng = np.random.default_rng(seed)
    N = _next_pow2(2 * n - 1)

    # Target: flat convolution of length 2n-1
    t = np.ones(2 * n - 1) / (2 * n - 1)
    T = np.fft.rfft(t, N)
    # Ensure non-negative magnitude target
    T_mag = np.abs(T)

    # Initialize with random phases
    phases = rng.standard_normal(len(T_mag))
    # Ensure real output: phases must satisfy conjugate symmetry
    # For rfft, we just initialize and take real part
    A = T_mag * np.exp(1j * phases)
    # Fix DC to be real and positive
    A[0] = np.abs(A[0])

    a_best = None
    sc_best = np.inf

    for it in range(iterations):
        a_full = np.fft.irfft(A, N)
        # Keep only first n elements
        a = np.maximum(a_full[:n], 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
        # Score
        sc = _score(a)
        if sc < sc_best:
            sc_best = sc
            a_best = a.copy()
        # Update phases
        A_new = np.fft.rfft(np.r_[a, np.zeros(N - n)], N)
        A = T_mag * np.exp(1j * np.angle(A_new))
        A[0] = np.abs(A[0])

    if a_best is not None:
        return a_best
    return np.ones(n) / n


def _dual_ascent_polish(a, deadline, min_remaining=20.0):
    """Final polish: alternate between reducing top peaks and spreading mass."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.05
    no_improve = 0

    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                lr *= 0.5
                if lr < 1e-6:
                    break
                no_improve = 0

        # Find top peaks
        K = max(3, min(20, n // 1000))
        top_k = np.argsort(c)[::-1][:K]

        # Compute pressure: which indices contribute most to top peaks
        pressure = np.zeros(n)
        for k in top_k:
            j_lo1 = max(0, k - n + 1)
            j_hi1 = min(n - 1, k)
            if j_lo1 <= j_hi1:
                idx = k - np.arange(j_lo1, j_hi1 + 1)
                pressure[j_lo1:j_hi1 + 1] += a[idx]
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_lo2 <= j_hi2:
                idx = np.arange(j_lo2, j_hi2 + 1) - k
                pressure[j_lo2:j_hi2 + 1] += a[idx]

        # Move mass from high-pressure to low-pressure indices
        p_sorted = np.sort(pressure)
        n_move = max(3, n // 200)
        high_idx = np.argsort(pressure)[::-1][:n_move]
        low_idx = np.argsort(pressure)[:n_move]

        # Compute move amount
        total_high = a[high_idx].sum()
        if total_high < 1e-15:
            break
        move = lr * total_high * 0.1
        move = min(move, 0.5 * a[high_idx].min())

        if move < 1e-15:
            break

        a[high_idx] -= move / n_move
        # Distribute to low-pressure indices proportionally to their current value
        low_vals = a[low_idx].copy()
        low_sum = low_vals.sum()
        if low_sum > 1e-15:
            a[low_idx] += move * low_vals / low_sum

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a / s

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Main search: mirror descent + stochastic subgrad + L-BFGS annealing."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        n = n_parent
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"Parent: n={n}, score={best_score:.10f}")
    else:
        n = 16384
        best_seq = _make_initial(n, "uniform")
        best_score = _score(best_seq)
        print(f"No parent, uniform n={n}, score={best_score:.10f}")

    # === Phase 1: Mirror descent (coarse, ~300s) ===
    print(f"\n[P1] Mirror descent (t={time.monotonic() - t_start:.1f}s)")
    p1_deadline = min(deadline - 900, t_start + 300)
    if time.monotonic() < p1_deadline:
        a_md, sc_md = _mirror_descent(best_seq, p1_deadline, min_remaining=30.0, seed=seed)
        if sc_md < best_score:
            best_seq = a_md.copy()
            best_score = sc_md
            print(f"  NEW BEST: {best_score:.10f} [mirror] t={time.monotonic() - t_start:.1f}s")
        else:
            print(f"  Mirror: no improvement (best={best_score:.10f})")

    # === Phase 2: L-BFGS annealing (main workhorse, ~600s) ===
    print(f"\n[P2] L-BFGS anneal (t={time.monotonic() - t_start:.1f}s)")
    p2_deadline = min(deadline - 500, t_start + 900)
    if time.monotonic() < p2_deadline:
        a_lb, sc_lb = _lbfgs_anneal(best_seq, p2_deadline, min_remaining=60.0, n_temps=16)
        if sc_lb < best_score:
            best_seq = a_lb.copy()
            best_score = sc_lb
            print(f"  NEW BEST: {best_score:.10f} [lbfgs] t={time.monotonic() - t_start:.1f}s")

    # === Phase 3: Stochastic subgradient (escape local minima, ~300s) ===
    print(f"\n[P3] Stochastic subgrad (t={time.monotonic() - t_start:.1f}s)")
    p3_deadline = min(deadline - 300, t_start + 1200)
    if time.monotonic() < p3_deadline:
        a_sg, sc_sg = _stochastic_subgrad(best_seq, p3_deadline, min_remaining=40.0, seed=seed + 1)
        if sc_sg < best_score:
            best_seq = a_sg.copy()
            best_score = sc_sg
            print(f"  NEW BEST: {best_score:.10f} [stoch] t={time.monotonic() - t_start:.1f}s")

    # === Phase 4: L-BFGS second pass (finer, ~200s) ===
    print(f"\n[P4] L-BFGS fine (t={time.monotonic() - t_start:.1f}s)")
    p4_deadline = min(deadline - 150, t_start + 1400)
    if time.monotonic() < p4_deadline:
        a_lb2, sc_lb2 = _lbfgs_anneal(best_seq, p4_deadline, min_remaining=40.0, n_temps=10)
        if sc_lb2 < best_score:
            best_seq = a_lb2.copy()
            best_score = sc_lb2
            print(f"  NEW BEST: {best_score:.10f} [lbfgs2] t={time.monotonic() - t_start:.1f}s")

    # === Phase 5: Dual ascent polish (~100s) ===
    print(f"\n[P5] Dual ascent polish (t={time.monotonic() - t_start:.1f}s)")
    p5_deadline = min(deadline - 50, t_start + 1550)
    if time.monotonic() < p5_deadline:
        a_da, sc_da = _dual_ascent_polish(best_seq, p5_deadline, min_remaining=20.0)
        if sc_da < best_score:
            best_seq = a_da.copy()
            best_score = sc_da
            print(f"  NEW BEST: {best_score:.10f} [dual] t={time.monotonic() - t_start:.1f}s")

    # === Phase 6: Multi-n exploration (~100s) ===
    print(f"\n[P6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [n, 12288, 24576, 8192]
    if n not in n_candidates:
        n_candidates.insert(1, n)
    n_candidates = list(dict.fromkeys(n_candidates))  # deduplicate preserving order

    for n_try in n_candidates:
        if time.monotonic() > deadline - 30:
            break
        if n_try == n:
            continue
        if n_try > 32768 or n_try < 4096:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init={sc_init:.10f}")
        if time.monotonic() < deadline - 20:
            a_opt, sc_opt = _lbfgs_anneal(a_init, deadline, min_remaining=15.0, n_temps=6)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                n = n_try
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic() - t_start:.1f}s")

    # === Phase 7: Final ultra-cold L-BFGS ===
    print(f"\n[P7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        n_fin = len(best_seq)
        c_fin = _full_convolve(best_seq, best_seq)
        mb = float(c_fin.max())
        c_s = np.sort(c_fin)[::-1]
        gap_fin = mb - c_s[min(10, len(c_s) - 1)]
        beta_fin = max(gap_fin * 0.1, mb * 0.00001)

        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()

        class _FinObj:
            def __call__(self, z):
                return _z_objective(z, beta_fin, n_fin)

            def grad(self, z):
                return _z_gradient(z, beta_fin, n_fin)

        prob = _FinObj()
        max_iter = min(30000, max(100, int(remaining / 0.005)))
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxcor': 100, 'maxls': 80})
            a_fin = _exp_normalize(result.x)
            sc_fin = _score(a_fin)
            if sc_fin < best_score:
                best_seq = a_fin.copy()
                best_score = sc_fin
                print(f"  NEW BEST: {best_score:.10f} [final] t={time.monotonic() - t_start:.1f}s")
        except Exception:
            pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_initial(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_initial(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
