def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a, b):

    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

def _nesterov_active_set(a_init, deadline, min_remaining=30.0, K=20):
    """Nesterov-accelerated gradient on the active set."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    y = a.copy()  # Nesterov momentum variable
    best_a = a.copy()

    t = 1.0
    lr = 0.02
    
    while time.monotonic() + min_remaining < deadline:
        # Nesterov lookahead
        c = _full_convolve(y, y)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
        
        top_k_idx = np.argsort(c)[::-1][:K]
        w_S = np.zeros(2 * n - 1)
        w_S[top_k_idx] = 1.0 / K
        
        y_rev = y[::-1]
        h = _full_convolve(w_S, y_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        
        if gnorm < 1e-16:
            break
        
        # Step
        a_new = y - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Update momentum
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        beta_mom = (t - 1) / t_new
        y = a_new + beta_mom * (a_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        a = a_new
        t = t_new
        
        # Slowly decrease lr
        lr = max(0.001, lr * 0.999)
    
    return best_a, best_sc

def _resample(a, new_n):

    x_old = np.linspace(0, 1, old_n)

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:

    return a_new

                g_prev = None
                restarts += 1
                if restarts > 12:
                    break
    return best_a, best_sc


def _nesterov_active(a_init, deadline, min_remaining=20.0, K=30):
    """Nesterov-accelerated gradient on active set."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    y = a.copy()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    t = 1.0
    lr = 0.012
    for _ in range(200000):
        if time.monotonic() + min_remaining > deadline:
            break
        c = _full_convolve(y, y)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
        top_k_idx = np.argsort(c)[::-1][:K]
        w_S = np.zeros(2 * n - 1)
        w_S[top_k_idx] = 1.0 / K
        h = _full_convolve(w_S, y[::-1])[n - 1:2 * n - 1]
        grad = 2.0 * h
        m = np.mean(grad)
        grad = grad - m
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        a_new = y - lr * grad
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        bm = (t - 1.0) / t_new
        y = a_new + bm * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        a = a_new
        t = t_new
        lr = max(0.0005, lr * 0.9997)
    return best_a, best_sc


def _micro_polish(a_init, deadline, min_remaining=5.0):
    """Ultra-fine active-set polish with very small K and careful line search."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    for K in [2, 3, 5, 8, 12, 16, 20, 30, 40]:
        if time.monotonic() + min_remaining > deadline:
            break
        for _ in range(200):
            if time.monotonic() + min_remaining > deadline:
                break
            c = _full_convolve(a, a)
            c_max = float(c.max())
            current_sc = 2.0 * n * c_max
            if current_sc < best_sc - 1e-14:
                best_sc = current_sc
                best_a = a.copy()
            top_k_idx = np.argsort(c)[::-1][:K]
            w_S = np.zeros(2 * n - 1)
            w_S[top_k_idx] = 1.0 / K
            h = _full_convolve(w_S, a[::-1])[n - 1:2 * n - 1]
            grad = 2.0 * h
            grad = grad - np.mean(grad)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-18:
                break
            alpha = 0.01 / (gnorm + 1e-15)
            improved = False
            for _ls in range(25):
                a_try = a - alpha * grad
                a_try = np.maximum(a_try, 0.0)
                st = a_try.sum()
                if st < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / st
                sc_try = 2.0 * n * float(_full_convolve(a_try, a_try).max())
                if sc_try < current_sc - 1e-14:
                    a = a_try
                    improved = True
                    break
                alpha *= 0.5
            if not improved:
                break
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.65) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-14.0 * (t - 0.3) ** 2) + 0.85 * np.exp(-14.0 * (t - 0.7) ** 2) + 0.08
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
