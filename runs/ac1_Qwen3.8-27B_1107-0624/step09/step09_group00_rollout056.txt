**New approach: Iterative Peak Reduction (IPR)**

7. Project: a = max(a, 0)
8. Repeat

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Multiplicative KKT fixed-point iteration with Nesterov momentum
2. Smoothed L-BFGS with aggressive beta annealing
3. Stochastic ILS with peak-targeting and spectral perturbations
4. Multi-n exploration

Key innovation over previous: natural multiplicative update derived from KKT
conditions, which respects non-negativity and simplex constraints intrinsically.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _blame_vector(a, w):
    """Compute blame[j] = sum_k w[k]*(a[k-j] + a[j-k]) via FFT convolutions."""
    n = len(a)
    w_rev = w[::-1]
    conv_aw = _full_convolve(a, w)
    conv_awr = _full_convolve(a, w_rev)
    part2 = conv_aw[:n]
    part1 = conv_awr[n - 1:2 * n - 1][::-1]
    return part1 + part2


def _kkt_blame(a, c, tol_frac):
    """Compute KKT blame vector with active-set weighting."""
    n = len(a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros(n)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if not active.any():
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    eps = 1e-12 * c_max
    w = np.zeros_like(c)
    w[active] = 1.0 / (c_max - c[active] + eps)
    w = w / w.sum()
    blame = _blame_vector(a, w)
    blame = blame - np.mean(blame)
    return blame


def _multiplicative_kkt_optimize(a_init, deadline, tol_start=0.05, tol_end=0.0001,
                                  eta_init=0.3, min_time=20.0):
    """
    Multiplicative KKT fixed-point iteration with Nesterov momentum.
    
    Update rule: a_j <- a_j * exp(eta * (mu - blame_j))
    where mu = mean(blame on support), ensuring sum-preserving direction.
    """
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    tol_frac = tol_start
    eta = eta_init
    no_improve = 0
    max_stall = 100
    v = np.zeros(n)  # momentum
    momentum = 0.7

    while time.monotonic() < deadline - min_time:
        c = _full_convolve(a, a)
        cur_sc = 2.0 * n * float(c.max())

        if cur_sc < best_sc - 1e-14:
            best_sc = cur_sc
            best_a = a.copy()
            no_improve = 0

        blame = _kkt_blame(a, c, tol_frac)
        bnorm = np.linalg.norm(blame)
        if bnorm < 1e-18:
            break

        # Nesterov momentum on the "log-update" direction
        v = momentum * v - eta * blame
        # Adaptive step: scale by inverse norm
        scale = 0.5 / (np.max(np.abs(v)) + 1e-30)
        update = v * scale
        # Clamp to prevent extreme values
        update = np.clip(update, -5.0, 5.0)

        # Multiplicative update
        a_new = a * np.exp(update)
        a_new = _normalize(a_new)

        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < cur_sc - 1e-15:
            a = a_new
            if tol_frac > tol_end:
                tol_frac = max(tol_end, tol_frac * 0.97)
            eta = min(eta_init, eta * 1.01)
        else:
            # Backtrack
            a_new = a * np.exp(update * 0.5)
            a_new = _normalize(a_new)
            c_new = _full_convolve(a_new, a_new)
            sc_new = 2.0 * n * float(c_new.max())
            if sc_new < cur_sc - 1e-15:
                a = a_new
            else:
                v = np.zeros(n)
                no_improve += 1
                if no_improve % 20 == 0:
                    eta *= 0.7
                    tol_frac = max(tol_end, tol_frac * 0.8)
                if no_improve >= max_stall:
                    a = best_a.copy()
                    v = np.zeros(n)
                    eta = eta_init * 0.5
                    tol_frac = max(tol_end, tol_frac * 0.3)
                    no_improve = 0

    return best_a, _score(best_a)


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    blame = _blame_vector(a, w)
    W = float(np.dot(w, c))
    return 4.0 * n * a * (blame - 2.0 * W)


class _LbfgsProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def f(self, z):
        return _smoobj(z, self.beta, self.n)

    def g(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with aggressive beta annealing schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008,
        0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(50, int(remaining / (est_time * 20))))

        prob = _LbfgsProblem(beta, n)
        try:
            result = minimize(
                prob.f, z, jac=prob.g, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 120, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.10f}")
        except Exception:
            pass

    return current_a


def _quick_optimize(a_init, deadline, min_time_remaining=20.0):
    """Combined multiplicative KKT + L-BFGS quick optimization."""
    a = _multiplicative_kkt_optimize(a_init, deadline, min_time=10.0)
    if time.monotonic() < deadline - min_time_remaining:
        a2 = _run_lbfgs(a, deadline, min_time_remaining=min_time_remaining + 5)
        if _score(a2) < _score(a):
            a = a2
    return a


def _spectral_perturb(a, rng, sigma):
    """Perturb in frequency domain: modify the phase of the DTFT."""
    n = len(a)
    N = _next_pow2(2 * n)
    A = np.fft.rfft(a, N)
    # Add small random phase perturbation (keeping magnitude roughly the same)
    phases = np.angle(A)
    mags = np.abs(A)
    new_phases = phases + sigma * rng.standard_normal(len(phases))
    new_phases[0] = 0  # Keep DC component real and positive
    A_new = mags * np.exp(1j * new_phases)
    a_new = np.fft.irfft(A_new, N)[:n]
    return _normalize(a_new)


def _peak_target_perturb(a, c, rng, n_push):
    """Perturb specifically to reduce the current peak."""
    n = len(a)
    c_max = float(c.max())
    tol = 0.01 * c_max
    active = c >= (c_max - tol)
    active_idx = np.where(active)[0]

    a_new = a.copy()
    # For each active peak position k, reduce a at positions that contribute most
    for k in active_idx[:10]:  # Limit to top 10 peaks
        # blame at this single peak: a[k-j] + a[j-k]
        blame = np.zeros(n)
        for j in range(n):
            kj = k - j
            jk = j - k
            if 0 <= kj < n:
                blame[j] += a[kj]
            if 0 <= jk < n:
                blame[j] += a[jk]
        # Reduce a proportionally to blame
        bmax = blame.max()
        if bmax > 1e-15:
            reduction = 0.02 * blame / bmax
            a_new *= (1.0 - reduction)

    # Add small noise to avoid getting stuck
    a_new += 0.001 * rng.standard_normal(n)
    return _normalize(a_new)


def _block_swap_perturb(a, rng, n_blocks=4):
    """Randomly swap blocks of the sequence."""
    n = len(a)
    a_new = a.copy()
    block_size = max(10, n // (n_blocks * 4))
    for _ in range(n_blocks):
        s1 = int(rng.integers(0, n - block_size))
        s2 = int(rng.integers(0, n - block_size))
        if abs(s1 - s2) < block_size:
            continue
        tmp = a_new[s1:s1 + block_size].copy()
        a_new[s1:s1 + block_size] = a_new[s2:s2 + block_size]
        a_new[s2:s2 + block_size] = tmp
    return _normalize(a_new)


def _stochastic_ils(a_init, deadline, rng, min_time_remaining=60.0):

    n = len(a_init)
    current_a = _normalize(a_init.copy())
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc

    attempt = 0
    while time.monotonic() + min_time_remaining < deadline:
        strategy = attempt % 6

        if strategy == 0:
            sigma = 0.01 + 0.02 * (attempt // 6) % 3
            a_pert = _spectral_perturb(current_a, rng, sigma)
        elif strategy == 1:
            c = _full_convolve(current_a, current_a)
            a_pert = _peak_target_perturb(current_a, c, rng, 5)
        elif strategy == 2:
            a_pert = _block_swap_perturb(current_a, rng, n_blocks=3)
        elif strategy == 3:
            # Global log-space perturbation
            z = np.log(np.maximum(current_a, 1e-30))
            sigma = 0.005 + 0.01 * (attempt // 6) % 4
            z += sigma * rng.standard_normal(n)
            z -= z.mean()
            a_pert = _softmax(z)
        elif strategy == 4:
            # Sparse perturbation
            n_pert = max(20, n // 10)
            idx = rng.choice(n, size=n_pert, replace=False)
            a_pert = current_a.copy()
            factors = np.exp(0.05 * rng.standard_normal(n_pert))
            a_pert[idx] *= factors
            a_pert = _normalize(a_pert)
        else:

            seg_len = max(20, n // 15)
            start = int(rng.integers(0, n - seg_len))
            a_pert = current_a.copy()
            a_pert[start:start + seg_len] = a_pert[start:start + seg_len][::-1]
            z = np.log(np.maximum(a_pert, 1e-30))
            z += 0.003 * rng.standard_normal(n)
            z -= z.mean()
            a_pert = _softmax(z)

        opt_deadline = min(deadline - min_time_remaining, time.monotonic() + 10)
        a_opt = _quick_optimize(a_pert, opt_deadline, min_time_remaining=5)
        sc_opt = _score(a_opt)

        if sc_opt < best_sc - 1e-13:
            best_sc = sc_opt
            best_a = a_opt.copy()
            current_a = a_opt.copy()
            print(f"    ILS NEW BEST: {best_sc:.10f} (strat={strategy}, att={attempt}) t={time.monotonic()-t_start:.0f}s")
        elif sc_opt < current_sc - 1e-13:
            current_a = a_opt.copy()

        attempt += 1

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric_linear":
        t = np.linspace(0, 1, n)
        a = 1.0 + t  # Linearly increasing
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric_exp":
        t = np.linspace(0, 1, n)
        a = np.exp(2.0 * t)  # Exponentially increasing
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric_quadratic":
        t = np.linspace(0, 1, n)
        a = (1.0 + 3.0 * t) ** 2  # Quadratically increasing
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-method optimizer for minimum peak autocorrelation sequences.
    Combines multiplicative KKT iteration, smoothed L-BFGS, and stochastic ILS.
    """
    global t_start
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 24576
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    n_main = n_parent if n_parent > 0 else 24576

    # === Phase 1: Multiplicative KKT on parent (200s) ===
    print(f"\n[P1] Multiplicative KKT (t={time.monotonic() - t_start:.1f}s)")
    kkt_deadline = min(deadline - 1000, t_start + 200)
    a_kkt, sc_kkt = _multiplicative_kkt_optimize(
        best_seq, kkt_deadline, tol_start=0.1, tol_end=0.001,
        eta_init=0.5, min_time=30.0)
    if sc_kkt < best_score:
        best_seq = a_kkt.copy()
        best_score = sc_kkt
        print(f"  NEW BEST: {best_score:.10f} [kkt1] t={time.monotonic() - t_start:.1f}s")

    # === Phase 2: Multi-start with diverse profiles (200s) ===
    print(f"\n[P2] Multi-start diverse (t={time.monotonic() - t_start:.1f}s)")
    ms_deadline = min(deadline - 800, time.monotonic() + 200)
    profiles = ['triangle', 'hann', 'gaussian', 'asymmetric_linear',
                'asymmetric_exp', 'asymmetric_quadratic']
    for kind in profiles:
        if time.monotonic() > ms_deadline - 30:
            break
        a_init = _make_profile(n_main, kind)
        sc_init = _score(a_init)
        a_opt = _quick_optimize(a_init, ms_deadline, min_time_remaining=40)
        sc_opt = _score(a_opt)
        print(f"  {kind}: init={sc_init:.10f}, opt={sc_opt:.10f}")
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [ms/{kind}]")

    # === Phase 3: Intensive L-BFGS on best (300s) ===
    print(f"\n[P3] Intensive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        lbfgs_deadline = min(deadline - 500, time.monotonic() + 300)
        a_main = _run_lbfgs(best_seq, lbfgs_deadline, min_time_remaining=60)
        sc = _score(a_main)
        if sc < best_score:
            best_seq = a_main.copy()
            best_score = sc
            print(f"  NEW BEST: {best_score:.10f} [lbfgs] t={time.monotonic() - t_start:.1f}s")

    # === Phase 4: Second KKT pass with tighter tolerance (150s) ===
    print(f"\n[P4] KKT pass 2 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        kkt2_deadline = min(deadline - 350, time.monotonic() + 150)
        a_kkt2, sc_kkt2 = _multiplicative_kkt_optimize(
            best_seq, kkt2_deadline, tol_start=0.02, tol_end=0.0001,

        if sc_kkt2 < best_score:
            best_seq = a_kkt2.copy()
            best_score = sc_kkt2
            print(f"  NEW BEST: {best_score:.10f} [kkt2] t={time.monotonic() - t_start:.1f}s")

    # === Phase 5: Stochastic ILS (400s) ===
    print(f"\n[P5] Stochastic ILS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        ils_deadline = min(deadline - 200, time.monotonic() + 400)
        a_ils, sc_ils = _stochastic_ils(best_seq, ils_deadline, rng, min_time_remaining=60)
        if sc_ils < best_score:
            best_seq = a_ils.copy()
            best_score = sc_ils
            print(f"  NEW BEST: {best_score:.10f} [ils] t={time.monotonic() - t_start:.1f}s")

    # === Phase 6: Multi-n exploration (200s) ===
    print(f"\n[P6] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [n_main, int(n_main * 1.25), int(n_main * 0.8),
                    32768, 16384, 40960, 20480]
    seen = set()
    n_unique = []
    for nn in n_candidates:
        if nn not in seen and nn >= 2048 and nn != len(best_seq):
            seen.add(nn)
            n_unique.append(nn)

    for n_try in n_unique:
        if time.monotonic() > deadline - 100:
            break
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init={sc_init:.10f}")
        a_opt = _quick_optimize(a_init, deadline, min_time_remaining=30)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic() - t_start:.1f}s")

    # === Phase 7: Final ILS round (100s) ===
    print(f"\n[P7] ILS round 2 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        ils2_deadline = min(deadline - 80, time.monotonic() + 100)
        rng2 = np.random.default_rng(int(seed) + 7777)
        a_ils2, sc_ils2 = _stochastic_ils(best_seq, ils2_deadline, rng2, min_time_remaining=30)
        if sc_ils2 < best_score:
            best_seq = a_ils2.copy()
            best_score = sc_ils2
            print(f"  NEW BEST: {best_score:.10f} [ils2] t={time.monotonic() - t_start:.1f}s")

    # === Phase 8: Ultra-fine final polish (80s) ===
    print(f"\n[P8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        # Ultra-fine KKT
        fin_deadline = min(deadline - 5, time.monotonic() + 40)
        a_fin, sc_fin = _multiplicative_kkt_optimize(
            best_seq, fin_deadline, tol_start=0.005, tol_end=0.00001,
            eta_init=0.1, min_time=10.0)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_kkt] t={time.monotonic() - t_start:.1f}s")

        # Ultra-low beta L-BFGS
        remaining = deadline - 5 - time.monotonic()
        if remaining > 15:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 5e-6
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _LbfgsProblem(beta_fin, n)
            max_iter = min(5000, max(100, int(remaining / 0.02)))
            try:
                result = minimize(prob.f, z, jac=prob.g, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-20,
                                           'gtol': 1e-16, 'maxcor': 120})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2

            except Exception:
                pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
