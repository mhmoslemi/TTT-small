def _next_pow2(n):
    k = 1
    while k < n:

    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

        else:
            e_full = np.zeros(n_2nm1)
            e_full[top_idx] = e
            w = e_full / s
            obj = 2.0 * self.n * (c_max + self.beta * np.log(s))
        
        # Add variance penalty
        if self.var_weight > 0:
            c_mean = c.mean()
            var_term = self.var_weight * np.sum((c - c_mean) ** 2)
            obj += var_term
        
        ga = _grad_w(a, w)
        W = float(np.dot(a, ga))
        grad_z = a * (ga - W)
        return obj, grad_z


def _run_lbfgs(a_init, deadline, min_remaining=30.0, schedule=None):
    """L-BFGS with top-K smoothing and adaptive schedule."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.01)
    
    if schedule is None:
        schedule = [
            (1.0, 0.08, 0.0), (0.6, 0.06, 0.0), (0.4, 0.04, 0.0),
            (0.2, 0.03, 0.0), (0.1, 0.02, 0.0), (0.05, 0.015, 0.0),
            (0.02, 0.01, 0.0), (0.01, 0.008, 0.0),
            (0.005, 0.005, 0.0), (0.002, 0.003, 0.0),
            (0.001, 0.002, 0.0), (0.0005, 0.001, 0.0),
        ]
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for beta_mult, K_frac, var_w in schedule:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break
        
        beta = base_beta * beta_mult
        K = max(5, int((2 * n - 1) * K_frac))
        
        est_time = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 25))))
        
        prob = _SmoProb(beta, K, n, var_w)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 50}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _fine_polish(a_init, deadline, min_remaining=10.0):
    """Very fine active-set polish targeting the exact top peaks."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.003
    no_improve = 0
    
    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        # Active set: peaks within 0.1% of max
        active = c >= c_max * 0.999
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active
        
        g = _grad_w(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-16:
            break
        
        alpha = min(lr, 0.3 / (gnorm + 1e-15))
        improved = False
        for ls in range(40):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                best_sc = sc_try
                best_a = a_try.copy()
                no_improve = 0
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.7
            if no_improve > 500:
                break
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "optimal_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2) + 0.03
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_ramp":
        t = np.linspace(0, 1, n)
        a = 0.1 + 0.9 * t
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _spectral_shape_optimize(a_init, deadline, min_remaining=20.0):
    """Optimize in the log-frequency domain to encourage flat convolution.
    Uses the fact that conv(a,a) = IDFT(DFT(a)^2) for circular conv,
    and approximately for linear conv with zero-padding."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    N = _next_pow2(2 * n - 1)
    
    # Work with the real and imaginary parts of log(DFT(a))
    A = np.fft.rfft(a, N)
    # Avoid log(0)
    A_safe = np.where(np.abs(A) > 1e-30, A, 1e-30 + 0j)
    log_A = np.log(A_safe)
    
    # Parameters to optimize: Re(log_A) and Im(log_A)
    # But we need a to be real and non-negative, which constrains the DFT.
    # Instead, let's work with a parametrization that guarantees non-negativity.
    
    # Alternative: optimize the "phase" and "magnitude" of DFT(a)
    # with the constraint that IDFT gives a non-negative, real, length-n sequence.
    # This is complex. Let's just do a direct L-BFGS with a variance-aware objective.
    
    # Actually, let's try a different approach: iterate on the target convolution profile.
    # Given a target c_target, solve for a such that conv(a,a) ≈ c_target.
    # This is a deconvolution problem: a ≈ sqrt(c_target) in the frequency domain.
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    # Target: make c as flat as possible
    # c_target[k] = 1/(2n-1) for all k (uniform target)
    # But we can only achieve this if it's consistent with a being non-negative.
    
    # Iterative approach:
    # 1. Compute current c
    # 2. Set target c_target = c / (c / c_mean)^(alpha) where alpha < 1 compresses the dynamic range
    # 3. Find a_new such that conv(a_new, a_new) ≈ c_target (via sqrt in frequency domain)
    # 4. Project a_new to be non-negative and sum=1
    # 5. Repeat
    
    alpha = 0.5  # Compression parameter
    for iteration in range(200):
        if time.monotonic() + min_remaining > deadline:
            break
        
        c = _full_convolve(a, a)
        c_mean = c.mean()
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
        
        # Compress the convolution profile
        c_ratio = c / (c_mean + 1e-30)
        c_target = c_mean * c_ratio ** alpha
        c_target = c_target / c_target.sum() * (a.sum() ** 2)
        
        # Deconvolve: find a_new such that conv(a_new, a_new) ≈ c_target
        # In frequency domain: A_new^2 ≈ C_target => A_new ≈ sqrt(C_target)
        C_target = np.fft.rfft(c_target, N)
        A_new = np.sqrt(C_target + 1e-30 + 0j)
        
        # Take inverse FFT
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        # Blend with current to maintain stability
        a = a_new  # Full update
        
        if iteration % 50 == 0:
            print(f"    [spectral] iter={iteration}, sc={best_sc:.10f}, "
                  f"c_max/c_mean={c_max/c_mean:.4f}")
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-method optimizer combining subgradient, peak equalization,
    spectral shape optimization, L-BFGS, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, tapered n={n_parent}, score={best_score:.10f}")
    
    # === Phase 1: Quick L-BFGS polish (200s) ===
    print(f"\n[P1] L-BFGS polish (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 600, t_start + 200)
    a1 = _run_lbfgs(best_seq, p1_deadline, min_remaining=30)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs1] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 2: Adaptive subgradient (250s) ===
    print(f"\n[P2] Adaptive subgradient (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 450:
        p2_deadline = min(deadline - 400, time.monotonic() + 250)
        a2, sc2 = _adaptive_subgrad(best_seq, p2_deadline, min_remaining=30)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [subgrad] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 3: Peak equalization (200s) ===
    print(f"\n[P3] Peak equalization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        p3_deadline = min(deadline - 250, time.monotonic() + 200)
        a3, sc3 = _peak_equalization(best_seq, p3_deadline, min_remaining=20)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [peak_eq] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 4: Spectral shape optimization (150s) ===
    print(f"\n[P4] Spectral shape (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        p4_deadline = min(deadline - 150, time.monotonic() + 150)
        a4, sc4 = _spectral_shape_optimize(best_seq, p4_deadline, min_remaining=20)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [spectral] t={time.monotonic()-t_start:.1f}s")
        # L-BFGS polish after spectral
        if time.monotonic() < deadline - 150:
            a4b = _run_lbfgs(a4, min(deadline - 130, time.monotonic() + 80), min_remaining=15)
            sc4b = _score(a4b)
            if sc4b < best_score:
                best_seq = a4b.copy()
                best_score = sc4b
                print(f"  NEW BEST: {best_score:.10f} [spectral+lbfgs] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 5: Multi-n exploration (250s) ===
    print(f"\n[P5] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_main = len(best_seq)
    n_options = []
    for n_try in [8192, 12288, 14336, 18432, 20480, 24576, 32768]:
        if n_try != n_main:
            n_options.append(n_try)
    
    if time.monotonic() < deadline - 150 and n_options:
        n_options = n_options[:4]  # Limit to 4 different n values
        time_per_n = min(60, (deadline - 150 - time.monotonic()) / len(n_options))
        
        for n_try in n_options:
            if time.monotonic() > deadline - 80:
                break
            print(f"    Trying n={n_try} (budget={time_per_n:.0f}s)")
            a_res = _resample(best_seq, n_try)
            
            # Quick L-BFGS
            n_deadline = min(deadline - 70, time.monotonic() + time_per_n * 0.6)
            a_opt = _run_lbfgs(a_res, n_deadline, min_remaining=10)
            sc_opt = _score(a_opt)
            
            # Quick subgradient
            if time.monotonic() < min(deadline - 60, time.monotonic() + time_per_n * 0.4):
                sg_deadline = min(deadline - 55, time.monotonic() + time_per_n * 0.4)
                a_sg, sc_sg = _adaptive_subgrad(a_opt, sg_deadline, min_remaining=10)
                if sc_sg < sc_opt:
                    a_opt = a_sg
                    sc_opt = sc_sg
            
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [n={n_try}] t={time.monotonic()-t_start:.1f}s")
            else:
                print(f"    n={n_try}: score={sc_opt:.10f} (no improvement)")
    
    # === Phase 6: L-BFGS round 2 (100s) ===
    print(f"\n[P6] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        p6_deadline = min(deadline - 60, time.monotonic() + 100)
        a6 = _run_lbfgs(best_seq, p6_deadline, min_remaining=20)
        sc6 = _score(a6)
        if sc6 < best_score:
            best_seq = a6.copy()
            best_score = sc6
            print(f"  NEW BEST: {best_score:.10f} [lbfgs2] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 7: Perturbation + re-optimize (80s) ===
    print(f"\n[P7] Perturbation (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        n_perturb = max(2, int((deadline - 40 - time.monotonic()) / 25))
        for i in range(n_perturb):
            if time.monotonic() > deadline - 30:
                break
            sigma = 0.02 + 0.01 * i
            a_pert = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs(a_pert, min(deadline - 25, time.monotonic() + 20), min_remaining=5)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.10f} [perturb_{i}] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 8: Final fine polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _fine_polish(best_seq, deadline - 3, min_remaining=3)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [polish] t={time.monotonic()-t_start:.1f}s")
        
        # Tiny final L-BFGS
        remaining = deadline - 3 - time.monotonic()
        if remaining > 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = max(mb * 0.0003, 1e-9)
            K_fin = max(5, int(n * 0.001))
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _SmoProb(beta_fin, K_fin, n, 0.0)
            max_iter = min(2000, max(20, int(remaining / 0.02)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-14, 'maxcor': 100})
                a_fin2 = np.exp(result.x - result.x.max())
                a_fin2 = a_fin2 / a_fin2.sum()
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass
    
    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        # Ensure non-negative
        final = np.maximum(final, 0.0)
        s = float(final.sum())
        if s > 0:
            final = final / s  # Normalize (score is scale-invariant)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
