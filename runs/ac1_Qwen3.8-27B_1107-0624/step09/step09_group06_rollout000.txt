Equivalently: find the smallest M such that there exists a with sum(a)=1, a>=0, and c[k] <= M for all k.

This is a feasibility problem for a given M. We can solve it using:
- Interior point methods (but we'd need to implement)
- Iterative methods: start with M and find a that satisfies c[k] <= M for all k

The constraint c[k] <= M is a quadratic constraint (not linear), so this is a quadratically constrained optimization.

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous approach:
1. More aggressive L-BFGS beta schedule (much smaller betas, more iterations)
2. Active-set projected gradient with backtracking line search
3. Multi-n exploration with adaptive selection
4. Final ultra-fine polish with very tight convergence
5. Structured perturbations to escape local minima
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a)=1, so just 2*n*max(c)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _conv_grad_at_k(a, k):
    """Gradient of c[k] w.r.t. a, where c = full_convolve(a,a).
    dc[k]/da[i] = 2*a[k-i] (with a[x]=0 for x out of [0,n-1])"""
    n = len(a)
    grad = np.zeros(n)
    # a[k-i] for i in [0,n-1], valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
    i_lo = max(0, k - n + 1)
    i_hi = min(n - 1, k)
    if i_lo <= i_hi:
        grad[i_lo:i_hi + 1] = 2.0 * a[k - np.arange(i_lo, i_hi + 1)]
    return grad


def _conv_grad_active(a, active_k):
    """Sum of gradients over active convolution indices."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_k:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += 2.0 * a[k - np.arange(i_lo, i_hi + 1)]
    return grad


def _smo_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smo_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # Gradient of sum_k w[k]*c[k] w.r.t. a
    # = sum_k w[k] * 2*a_rev_shifted
    # c[k] = sum_j a[j]*a[k-j], dc[k]/da[i] = 2*a[k-i]
    # sum_k w[k]*2*a[k-i] for each i
    # This is 2 * convolve(w, a_rev)[n-1:2n-1] where a_rev = a[::-1]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _SmoProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smo_objective(z, self.beta, self.n)

    def grad(self, z):
        return _smo_gradient(z, self.beta, self.n)


def _run_lbfgs_stage(a_init, deadline, betas, min_time_remaining=30.0, verbose=True):
    """Run L-BFGS with a sequence of decreasing betas."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score_normalized(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(10000, max(100, int(remaining / (est_time_per_iter * 15))))

        prob = _SmoProblem(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 100,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score_normalized(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"      L-BFGS beta={beta:.2e}: score={2*n*current_sc/(1.0):.10f}")
        except Exception:
            pass

    return current_a, current_sc


def _active_set_optimize(a_init, deadline, min_time_remaining=30.0, K=20, verbose=True):
    """Active-set projected gradient with backtracking line search.
    Minimizes max of top-K convolution values."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = 0.01
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            lr *= 0.5
            if lr < 1e-10:
                break
            no_improve = 0

        # Find active set: top K convolution values within 1% of max
        c_sorted = np.sort(c)[::-1]
        threshold = c_max * 0.999
        active_mask = c >= threshold
        active_k = np.where(active_mask)[0]
        if len(active_k) > K:
            top_vals = c[active_k]
            cutoff = np.sort(top_vals)[::-1][K]
            active_k = active_k[top_vals >= cutoff]
            active_k = np.sort(active_k)

        # Compute gradient of sum of active c[k]
        grad = _conv_grad_active(a, active_k)
        # Project to tangent space (sum = 0)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break

        # Backtracking line search
        step_dir = -grad_t / gnorm
        alpha = lr
        for _ in range(20):
            a_trial = a + alpha * step_dir
            if np.any(a_trial < -1e-15):
                alpha *= 0.5
                continue
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                alpha *= 0.5
                continue
            a_trial = a_trial / s_trial
            c_trial = _full_convolve(a_trial, a_trial)
            sc_trial = 2.0 * n * float(c_trial.max())
            if sc_trial < current_sc:
                a = a_trial
                break
            alpha *= 0.5

        if alpha < 1e-15:
            lr *= 0.5
            if lr < 1e-10:
                break

        # Slowly increase lr
        lr = min(0.05, lr * 1.01)

        if verbose and no_improve % 1000 == 0:
            print(f"      Active-set: best={best_sc:.10f}, cur={current_sc:.10f}, K={len(active_k)}, lr={lr:.6f}")

    return best_a, best_sc


def _nesterov_refined(a_init, deadline, min_time_remaining=30.0, K_init=10, K_max=200, seed=0, verbose=True):
    """Refined Nesterov with adaptive K and periodic restarts."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    momentum = 0.9
    base_lr = 0.003
    K = K_init
    t = 0
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive K
        if t % 100 == 0:
            c_sorted = np.sort(c)[::-1]
            gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
            if gap < c_max * 0.003 and K < K_max:
                K = min(K_max, int(K * 1.3) + 1)

        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]
        grad = _conv_grad_active(a, active)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-18:
            break

        lr = base_lr / (t + 10) ** 0.5
        v = momentum * v - lr * grad_t / (gnorm + 1e-18)
        a_cand = a + v
        a_proj = _project_simplex(a_cand)

        c_new = _full_convolve(a_proj, a_proj)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < current_sc * 1.001:
            a = a_proj
        else:
            v *= 0.3

        if no_improve > 300:
            perturb = 0.002 * rng.standard_normal(n)
            a_pert = _project_simplex(a + perturb)
            c_pert = _full_convolve(a_pert, a_pert)
            sc_pert = 2.0 * n * float(c_pert.max())
            if sc_pert < best_sc:
                a = a_pert
                best_a = a.copy()
                best_sc = sc_pert
            v = np.zeros(n)
            no_improve = 0

        if verbose and t % 2000 == 0:
            print(f"      Nesterov t={t}: best={best_sc:.10f}, K={K}")

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.1 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _shrinkage_optimize(a, deadline):
    """Try zeroing out small elements to reduce effective support."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score_normalized(a)

    thresholds = [0.001, 0.005, 0.01, 0.02, 0.05]
    for thresh in thresholds:
        if time.monotonic() > deadline:
            break
        a_try = a.copy()
        mask = a_try < thresh * a_try.max()
        a_try[mask] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score_normalized(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    # Try fractional zeroing
    a_sorted = np.sort(best_a)[::-1]
    for frac in [0.01, 0.02, 0.03, 0.05]:
        if time.monotonic() > deadline:
            break
        n_zero = int(n * frac)
        a_try = best_a.copy()
        # Zero out the smallest n_zero elements
        small_idx = np.argsort(best_a)[:n_zero]
        a_try[small_idx] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score_normalized(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc

        c_cur = _full_convolve(work_a, work_a)
        max_c = float(c_cur.max())
        base_beta = max_c * 0.0001
        betas_fin = base_beta * np.array([1.0, 0.5, 0.25, 0.125])
        a_fin, _ = _run_lbfgs_stage(work_a, deadline - 5, betas_fin,
                                    min_time_remaining=5, verbose=False)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_lbfgs]")

        # Active-set final
        if time.monotonic() < deadline - 5:
            a_fin2, _ = _active_set_optimize(best_seq / best_seq.sum(),
                                            deadline - 2, min_time_remaining=2,
                                            K=10, verbose=False)
            sc_fin2 = _score(a_fin2)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_active]")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None and len(best_seq) > 0:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
