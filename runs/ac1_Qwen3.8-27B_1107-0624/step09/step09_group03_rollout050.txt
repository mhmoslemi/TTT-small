"""

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0

        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec, rhs, max_cg=300, tol=1e-13):
    n = len(rhs)
    x = np.zeros(n)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x += alpha * p
        r -= alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

def _newton_step(a, c, c_max, tol_frac):
    n = len(a)
    active = c >= (c_max - tol_frac * c_max)
    if not active.any():
        active[np.argmax(c)] = True
    m = int(active.sum())
    if m < 2 or m > 15 * n:
        return None, None
    A_vec = active.astype(np.float64) / m
    g = 2.0 * _corr_a_A(a, A_vec, n)
    g_t = g - np.mean(g)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-14:
        return None, None
    mu = 1e-6 / m + 1e-9
    A_vec_copy = A_vec
    n_copy = n
    mu_copy = mu

    def _mv(v):
        return 2.0 * _corr_a_A(v, A_vec_copy, n_copy) + mu_copy * v

    u = _cg_solve(_mv, g_t, max_cg=300)
    w = _cg_solve(_mv, np.ones(n), max_cg=300)
    w_sum = float(w.sum())
    if abs(w_sum) < 1e-15:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w

            tol_schedule.append(0.001)
        elif frac < 0.85:
            tol_schedule.append(0.0005)
        elif frac < 0.95:
            tol_schedule.append(0.0001)
        else:
            tol_schedule.append(0.00005)

    for outer in range(max_outer):
        if time.monotonic() > deadline - 2:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol_frac = tol_schedule[outer]

        dx, A_vec = _newton_step(a, c, c_max, tol_frac)
        if dx is None:
            # Try a looser tolerance
            dx, A_vec = _newton_step(a, c, c_max, tol_frac * 5)
            if dx is None:
                continue

        # Line search
        t_ls = 1.0
        found = False
        for ls in range(60):
            a_try = a + t_ls * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t_ls *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < c_max * (1 - 1e-12):
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                a = a_try
                found = True
                break
            t_ls *= 0.5

        if not found:
            # Gradient step fallback
            g_all = 2.0 * _corr_a_A(a, np.ones(2 * n - 1) / (2 * n - 1), n)
            g_t = g_all - np.mean(g_all)
            t_ls = 0.01
            for ls in range(30):
                a_try = a - t_ls * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_sc - 1e-15:
                        best_sc = sc_try
                        best_a = a_try.copy()
                        a = a_try
                        found = True
                        break
                t_ls *= 0.5

        if not found and outer > max_outer // 2:
            break

        # Convergence check
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _deflation_optimize(a_init, deadline, n_attempts=100):
    """Multi-peak deflation with momentum."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    momentum = np.zeros(n)

    for attempt in range(n_attempts):
        if time.monotonic() > deadline - 1:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find peaks within 99.5% of max
        threshold = c_max * 0.995
        peak_mask = c >= threshold
        if not peak_mask.any():
            peak_mask[np.argmax(c)] = True
        n_peaks = int(peak_mask.sum())

        w = np.zeros_like(c)
        w[peak_mask] = 1.0 / n_peaks

        grad = 2.0 * _corr_a_A(a, w, n)
        g_t = grad - np.mean(grad)

        # Momentum update
        momentum = 0.9 * momentum + 0.1 * g_t

        applied = False
        for strength in [0.0002, 0.0005, 0.001, 0.002, 0.005, 0.01]:
            a_try = a - strength * momentum
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                applied = True
                break

        if not applied:
            # Try random perturbation
            rng_local = np.random.default_rng(attempt * 7 + 3)
            for _ in range(3):
                sigma = 0.005 * np.exp(-attempt / 50)
                z = np.log(np.maximum(a, 1e-30))
                z += sigma * rng_local.standard_normal(n)
                z -= z.mean()
                a_try = _softmax(z)
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    applied = True
                    break
            if not applied and attempt > 30:
                break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n, endpoint=False)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n, endpoint=False)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    a_new = np.interp(x_new, x_old, a, period=1)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _refinement_loop(a, deadline, max_rounds=15):
    """Iterate between Newton and deflation."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline - 5:
            break

        # Active set
        as_dl = min(deadline - 10, time.monotonic() + 40)
        a_as, sc_as = _active_set_optimize(a, as_dl, max_outer=40)
        if sc_as < best_sc:
            best_sc = sc_as
            best_a = a_as.copy()
            a = a_as

        # Deflation
        def_dl = min(deadline - 10, time.monotonic() + 25)
        a_df, sc_df = _deflation_optimize(a, def_dl, n_attempts=30)
        if sc_df < best_sc:
            best_sc = sc_df
            best_a = a_df.copy()
            a = a_df

        sc_cur = _score(a)
        if abs(sc_cur - best_sc) < 1e-15:
            break

    return best_a, best_sc


def _adam_aggressive(z_init, n, deadline, min_time=30.0):
    """Aggressive Adam continuation with very small beta values."""
    a_cur = _softmax(z_init)
    c_cur = _full_convolve(a_cur, a_cur)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_cur

    # Determine initial beta from the gap
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 100:
        gap = c_sorted[0] - c_sorted[100]
    elif len(c_sorted) > 10:
        gap = c_sorted[0] - c_sorted[10]
    else:
        gap = c_sorted[0] - c_sorted[-1]
    base_beta = max(gap * 0.5, max_c * 0.01)

    # Very aggressive: go from base_beta down to max_c * 5e-10
    n_betas = 35
    log_start = np.log(base_beta)
    log_end = np.log(max(max_c * 5e-10, 1e-30))
    log_betas = np.linspace(log_start, log_end, n_betas)
    betas = np.exp(log_betas)

    best_a = a_cur.copy()
    best_sc = _score(best_a)
    z = z_init.copy()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 2:
            break

        a_res = _adam_optimize(z, [beta], n, deadline,
                               lr_base=0.005 * (1.0 + i * 0.1),
                               min_time=min_time)
        sc_res = _score(a_res)
        if sc_res < best_sc:
            best_sc = sc_res
            best_a = a_res.copy()
            if (i % 5 == 0) or (i == len(betas) - 1):
                z = np.log(np.maximum(best_a, 1e-30))
                z -= z.mean()

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    - Adam with aggressive log-sum-exp continuation
    - Multi-peak active-set Newton
    - Deflation with momentum
    - Multi-scale (n=16384, n=32768)
    - Iterative refinement
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 40.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.12f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={len(parent)}, score={sc_parent:.12f}")
        _update(parent, sc_parent, "parent")
    else:
        parent = _make_profile(16384, "uniform")
        _update(parent, _score(parent), "init")

    n_parent = len(best_seq) if best_seq is not None else 16384

    # === Phase 1: Adam aggressive continuation ===
    print(f"\n[Phase 1] Adam aggressive (t={time.monotonic()-t_start:.1f}s)")
    phase1_dl = min(deadline - 600, time.monotonic() + 400)
    a_best = best_seq.copy()
    s_best = a_best.sum()
    a_best = a_best / s_best
    z0 = np.log(np.maximum(a_best, 1e-30))
    z0 -= z0.mean()
    a_p1 = _adam_aggressive(z0, n_parent, phase1_dl, min_time=600.0)
    _update(a_p1, _score(a_p1), "adam_aggressive")

    # === Phase 2: Active-set Newton ===
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        a2 = best_seq / best_seq.sum()
        phase2_dl = min(deadline - 400, time.monotonic() + 200)
        a_as, sc_as = _active_set_optimize(a2, phase2_dl, max_outer=150)
        _update(a_as, sc_as, "newton")

    # === Phase 3: Deflation ===
    print(f"\n[Phase 3] Deflation (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        a3 = best_seq / best_seq.sum()
        phase3_dl = min(deadline - 300, time.monotonic() + 100)
        a_df, sc_df = _deflation_optimize(a3, phase3_dl, n_attempts=80)
        _update(a_df, sc_df, "deflation")

    # === Phase 4: Refinement loop ===
    print(f"\n[Phase 4] Refinement (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        a4 = best_seq / best_seq.sum()
        phase4_dl = min(deadline - 300, time.monotonic() + 150)
        a_ref, sc_ref = _refinement_loop(a4, phase4_dl, max_rounds=10)
        _update(a_ref, sc_ref, "refinement")

    # === Phase 5: Second Adam pass on best ===
    print(f"\n[Phase 5] Adam second pass (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        a5 = best_seq / best_seq.sum()
        z5 = np.log(np.maximum(a5, 1e-30))
        z5 -= z5.mean()
        phase5_dl = min(deadline - 250, time.monotonic() + 120)
        a_p5 = _adam_aggressive(z5, len(a5), phase5_dl, min_time=250.0)
        _update(a_p5, _score(a_p5), "adam_pass2")

    # === Phase 6: Try n=32768 ===
    print(f"\n[Phase 6] n=32768 (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        n_large = 32768
        a_large = _interpolate_seq(best_seq, n_large)
        sc_large_init = _score(a_large)
        print(f"  n={n_large} init score: {sc_large_init:.12f}")

        # Adam on large
        z_large = np.log(np.maximum(a_large, 1e-30))
        z_large -= z_large.mean()
        phase6a_dl = min(deadline - 150, time.monotonic() + 100)
        a_large_opt = _adam_aggressive(z_large, n_large, phase6a_dl, min_time=150.0)
        _update(a_large_opt, _score(a_large_opt), "n32768_adam")

        # Newton on large
        if time.monotonic() < deadline - 100:
            a_large2 = a_large_opt / a_large_opt.sum()
            phase6b_dl = min(deadline - 80, time.monotonic() + 60)
            a_large_n, sc_ln = _active_set_optimize(a_large2, phase6b_dl, max_outer=50)
            _update(a_large_n, sc_ln, "n32768_newton")

    # === Phase 7: Diversified restarts ===
    print(f"\n[Phase 7] Restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        for sigma in [0.01, 0.03, 0.05, 0.02, 0.015]:
            if time.monotonic() > deadline - 40:
                break
            a_r = best_seq / best_seq.sum()
            z_r = np.log(np.maximum(a_r, 1e-30))
            z_r += sigma * rng.standard_normal(len(a_r))
            z_r -= z_r.mean()
            phase7_dl = min(deadline - 30, time.monotonic() + 60)
            a_res = _adam_aggressive(z_r, len(a_r), phase7_dl, min_time=30.0)
            _update(a_res, _score(a_res), f"restart_s{sigma}")

    # === Phase 8: Final micro-newton polish ===
    print(f"\n[Phase 8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_fin = best_seq / best_seq.sum()
        phase8_dl = deadline - 5
        a_fin_n, sc_fin = _active_set_optimize(a_fin, phase8_dl, max_outer=50)
        _update(a_fin_n, sc_fin, "final_newton")

        # Tiny beta Adam
        if time.monotonic() < deadline - 5:
            z_fin = np.log(np.maximum(a_fin_n, 1e-30))
            z_fin -= z_fin.mean()
            n_fin = len(a_fin_n)
            c_fin = _full_convolve(a_fin_n, a_fin_n)
            mb = float(c_fin.max())
            beta_tiny = mb * 1e-9
            remaining = deadline - 5 - time.monotonic()
            if remaining > 3:
                a_tiny = _adam_optimize(z_fin, [beta_tiny], n_fin, deadline,
                                        lr_base=0.001, min_time=3.0)
                _update(a_tiny, _score(a_tiny), "tiny_beta")

    # Final
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
