def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)

def _softmax(z):

                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_p = c_local ** p
                s_c = c_p.sum()
                if s_c < 1e-300:
                    w = np.zeros_like(c_local)
                    w[int(np.argmax(c_local))] = 1.0
                else:
                    pm = s_c ** (1.0/p)
                    w = (c_local ** (p-1)) / (pm ** (p-1))
                    w = w / w.sum()
                grad_a = _hvp_fast(a_local, w, n)
                return a_local * (grad_a - np.dot(a_local, grad_a))
            
            result = minimize(
                obj_pm, z,
                jac=grad_pm,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 40, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return best_a

def _run_adaptive_active_set(a_init, deadline, min_time_remaining=30.0):
    """Adaptive active-set method with progressive tightening."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    tol_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                    0.00005, 0.00002, 0.00001, 0.000005, 0.000002, 0.000001]
    
    for tol in tol_schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        active = c >= c_max * (1 - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / active.sum()
        
        # Projected gradient with backtracking
        for inner in range(100):
            if time.monotonic() + min_time_remaining > deadline:
                break
            grad = _hvp_fast(a, p, n)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 1.0 / (gnorm * gnorm + 1e-10)
            improved = False
            for _ in range(20):
                a_new = _proj_simplex(a - step * grad)
                c_new = _full_convolve(a_new, a_new)
                if float(c_new.max()) < c_max:
                    a = a_new
                    c_max = float(c_new.max())
                    sc = _score(a)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()
                    improved = True
                    break
                step *= 0.5
            if not improved and step < 1e-14:
                break
    
    return best_a

def _run_primal_dual(a_init, deadline, min_time_remaining=30.0):
    """Primal-dual with efficient HVP."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    beta = float(_full_convolve(a, a).max()) * 0.01
    
    for outer in range(200):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = _full_convolve(a, a)
        beta = max(float(c.max()) * 0.01 * (0.85 ** min(outer, 30)), 1e-10)
        p = _softmax(c / beta)
        
        # Primal step: projected gradient
        for inner in range(30):
            if time.monotonic() + min_time_remaining > deadline:
                break
            grad = _hvp_fast(a, p, n)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 0.5 / (gnorm + 1e-10)
            improved = False
            for _ in range(15):
                a_new = _proj_simplex(a - step * grad)
                c_new = _full_convolve(a_new, a_new)
                if float(c_new.max()) < float(c.max()) * 1.001:
                    a = a_new
                    sc = _score(a)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()
                    improved = True
                    break
                step *= 0.5
            if not improved:
                break
    
    return best_a
