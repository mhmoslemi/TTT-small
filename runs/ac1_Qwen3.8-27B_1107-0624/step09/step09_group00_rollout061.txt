"""
Multi-stage optimizer for minimizing
    2 * n * max(conv(a, a)) / sum(a)^2
over non-negative sequences.

The method combines:
- fast FFT-based convolution scoring/grads,
- a log-softmax parameterization with adapted log-sum-exp smoothing,
- coarse-grid global searches followed by fine-grid polishing,
- stochastic iterated local search,
- final low-temperature L-BFGS annealing.

The parent sequence (height_sequence_1, when available) is used as the
primary starting point, with diverse coarse starts used to escape basins.
"""

import numpy as np
import time
import scipy.optimize


_MAIN_BETA_FRACS = (
    0.10, 0.05, 0.025, 0.0125, 0.00625, 0.003125,
    0.0015625, 0.00078125, 0.000390625, 0.0001953125,
    0.00009765625, 0.000048828125, 0.0000244140625,
    0.00001220703125, 0.000006103515625
)

_QUICK_BETA_FRACS = (0.05, 0.02, 0.01, 0.005, 0.002)

_FINE_BETA_FRACS = (
    1e-4, 6e-5, 4e-5, 3e-5, 2e-5, 1.5e-5,
    1e-5, 8e-6, 6e-6, 5e-6, 4e-6, 3e-6, 2e-6, 1e-6
)


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _square_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    out_n = 2 * n - 1
    if out_n <= 0:
        return np.zeros(0, dtype=np.float64)
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    r = np.fft.irfft(fa * fa, N)
    if r.size > out_n:
        r = r[:out_n]
    return r


def _full_conv(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n, m = a.size, b.size
    out_n = n + m - 1
    if out_n <= 0:
        return np.zeros(0, dtype=np.float64)
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    if r.size > out_n:
        r = r[:out_n]
    return r


def _normalize_prob(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return np.ones_like(a, dtype=np.float64) / max(1, a.size)
    return a / s


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01 or not np.isfinite(s):
        return np.inf
    c = _square_conv(a)
    if c.size == 0:
        return np.inf
    cm = float(c.max())
    if not np.isfinite(cm):
        return np.inf
    return float(2.0 * a.size * cm / (s * s))


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - float(z.max())
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones_like(z, dtype=np.float64) / z.size
    return e / s


def _log_from_prob(a):
    a = _normalize_prob(a).copy()
    if a.size == 0:
        return np.zeros(0, dtype=np.float64)
    tiny = np.finfo(np.float64).tiny
    if np.any(a <= 0.0):
        pos = a > 0.0
        if pos.any():
            tiny_val = max(min(np.min(a[pos]) * 1e-12, 1e-20), tiny)
        else:
            tiny_val = 1e-300
        a = np.where(pos, a, tiny_val)
    z = np.log(a)
    z = z - float(z.mean())
    return z


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64)
    old = a.size
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    if old == n:
        return _normalize_prob(a)
    if old == 1:
        return np.ones(n, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old)
    x_new = np.linspace(0.0, 1.0, n)
    b = np.interp(x_new, x_old, a)
    return _normalize_prob(b)


def _make_profile(n, kind="uniform", rng=None):
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    if kind == "uniform":
        a = np.ones(n, dtype=np.float64)
    elif kind == "triangle":
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-12)
    elif kind == "hann":
        t = np.linspace(0.0, 1.0, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-12)
    elif kind == "gaussian":
        t = np.linspace(0.0, 1.0, n)
        a = np.exp(-5.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-12)
    elif kind == "ramp":
        a = np.linspace(1e-6, 1.0, n, dtype=np.float64)
    elif kind == "inverse_ramp":
        a = np.linspace(1.0, 1e-6, n, dtype=np.float64)
    elif kind == "random":
        if rng is None:
            rng = np.random.default_rng(12345)
        a = rng.random(n, dtype=np.float64)
        a = np.maximum(a, 1e-10)
    else:
        a = np.ones(n, dtype=np.float64)
    return _normalize_prob(a)


def _estimate_times(a):
    a = _normalize_prob(a)
    if a.size == 0:
        return 0.05, 0.005
    z = _log_from_prob(a)
    c = _square_conv(a)
    cm = float(c.max()) if c.size else 1e-8
    beta = max(cm * 0.01, 1e-12)
    n = a.size
    t0 = time.monotonic()
    _z_obj(z, beta, n, 0.0)
    t1 = time.monotonic()
    _z_grad(z, beta, n, 0.0)
    t2 = time.monotonic()
    per_eval = max(1e-5, (t2 - t0) / 2.0)
    per_iter = max(1e-4, 8.0 * per_eval)
    return per_iter, per_eval


def _z_obj(z, beta, n, reg):
    a = _softmax(z)
    n = a.size
    c = _square_conv(a)
    if c.size == 0:
        return np.inf
    cm = float(c.max())
    if not np.isfinite(cm):
        return np.inf
    if beta <= 1e-15:
        val = cm
    else:
        shift = (c - cm) / beta
        shift = np.clip(shift, -700.0, 0.0)
        s = float(np.exp(shift).sum())
        if s < 1e-300:
            val = cm
        else:
            val = cm + beta * np.log(s)
    score = 2.0 * n * val
    if reg > 0.0:
        score += 0.5 * float(reg) * float(np.dot(z, z))
    if not np.isfinite(score):
        return np.inf
    return float(score)


def _z_grad(z, beta, n, reg):
    a = _softmax(z)
    n = a.size
    c = _square_conv(a)
    if c.size == 0:
        return np.zeros_like(a, dtype=np.float64)
    cm = float(c.max())
    if not np.isfinite(cm):
        return np.zeros_like(a, dtype=np.float64)
    if beta <= 1e-15:
        w = np.zeros(c.size, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        shift = (c - cm) / beta
        shift = np.clip(shift, -700.0, 0.0)
        e = np.exp(shift)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros(c.size, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
    ar = a[::-1]
    wc = _full_conv(w, ar)
    h = wc[n - 1:2 * n - 1]
    raw = 4.0 * n * h
    grad = a * (raw - float(np.dot(raw, a)))
    if reg > 0.0:
        grad = grad + reg * z
    if not np.all(np.isfinite(grad)):
        grad = np.zeros_like(a, dtype=np.float64)
    return grad


def _run_lbfgs_z(a_init, deadline, min_rem, beta_fracs=None,
                 target_time=30.0, per_eval=0.005, per_iter=0.04,
                 no_improve_limit=6, reg=1e-12):
    a = _normalize_prob(a_init)
    if a.size == 0:
        return a, np.inf
    n = a.size
    best = a.copy()
    best_sc = _score(a)
    z = _log_from_prob(a)
    if beta_fracs is None:
        beta_fracs = _MAIN_BETA_FRACS

    no_improve = 0
    for frac in beta_fracs:
        if time.monotonic() + min_rem >= deadline:
            break
        c = _square_conv(a)
        if c.size == 0:
            break
        cm = float(c.max())
        if cm <= 1e-18:
            break
        beta = max(cm * float(frac), 1e-12)
        rem = deadline - min_rem - time.monotonic()
        if rem < 2.0:
            break
        tt = min(float(target_time), max(1.5, 0.85 * rem))
        if tt < 1.0:
            break
        maxfun = int(min(50000, max(220, tt / max(1e-5, per_eval))))
        maxiter = int(min(5000, max(60, tt / max(1e-4, per_iter))))
        try:
            res = scipy.optimize.minimize(
                _z_obj,
                z,
                jac=_z_grad,
                args=(beta, n, reg),
                method="L-BFGS-B",
                options={
                    "maxiter": maxiter,
                    "maxfun": maxfun,
                    "ftol": 1e-13,
                    "gtol": 1e-10,
                    "maxcor": 30,
                    "maxls": 40
                }
            )
        except Exception:
            break

        if not hasattr(res, "x"):
            break
        x = np.asarray(res.x, dtype=np.float64)
        if x.size != n or not np.all(np.isfinite(x)):
            break

        a_new = _softmax(x)
        sc_new = _score(a_new)
        if not np.isfinite(sc_new):
            break

        # Move current point unless it made a large local regression.
        if sc_new <= best_sc + max(1e-10, 1e-5 * best_sc):
            a = a_new
            z = x - float(x.mean())

        if sc_new < best_sc - 1e-12:
            best = a_new.copy()
            best_sc = sc_new
            no_improve = 0
            print(f"  [lbfgs beta={beta:.3e}] new best {best_sc:.10f} t={time.monotonic():.1f}")
        else:
            no_improve += 1
            if no_improve >= no_improve_limit and frac < 1e-3:
                break

    return best, best_sc


def _coarse_search(best_a, main_n, deadline, per_eval, per_iter, rng):
    best = best_a
    best_sc = _score(best)
    coarse_n = 1024

    inits = []
    inits.append(("parent", _resample(best, coarse_n)))
    inits.append(("uniform", _make_profile(coarse_n, "uniform")))
    inits.append(("random", _make_profile(coarse_n, "random", rng)))
    inits.append(("ramp", _make_profile(coarse_n, "ramp")))

    coarse_fracs = (0.2, 0.1, 0.05, 0.025, 0.0125, 0.00625, 0.003125, 0.0015625)
    time_per_init = max(4.0, max(0.0, deadline - time.monotonic()) / max(1, len(inits)))

    for name, a_c in inits:
        if time.monotonic() + 25.0 >= deadline:
            break
        end = min(deadline - 40.0, time.monotonic() + max(6.0, time_per_init * 0.85))
        if end <= time.monotonic() + 2.0:
            break
        try:
            a_c_opt, _ = _run_lbfgs_z(
                a_c,
                end,
                min_rem=4.0,
                beta_fracs=coarse_fracs,
                target_time=max(2.0, (end - time.monotonic()) / max(1, len(coarse_fracs))),
                per_eval=max(1e-6, per_eval * 0.03),
                per_iter=max(1e-5, per_iter * 0.03),
                no_improve_limit=4,
                reg=1e-12
            )
            a_main = _resample(a_c_opt, main_n)
            sc_main = _score(a_main)
            if sc_main < best_sc - 1e-12:
                best = a_main
                best_sc = sc_main
                print(f"  [coarse {name}] new best {best_sc:.10f} t={time.monotonic():.1f}")

            # Short local main-size polish of the coarsely optimized shape.
            if time.monotonic() + 12.0 < deadline - 80.0:
                endq = min(deadline - 80.0, time.monotonic() + 12.0)
                a_q, sc_q = _run_lbfgs_z(
                    a_main,
                    endq,
                    min_rem=3.0,
                    beta_fracs=_QUICK_BETA_FRACS,
                    target_time=3.0,
                    per_eval=per_eval,
                    per_iter=per_iter,
                    no_improve_limit=4,
                    reg=1e-12
                )
                if sc_q < best_sc - 1e-12:
                    best = a_q
                    best_sc = sc_q
                    print(f"  [coarse+polish {name}] new best {best_sc:.10f} t={time.monotonic():.1f}")
        except Exception:
            continue

    return best, best_sc


def _perturrent(a, rng, sigma):
    z = _log_from_prob(a)
    z = z + float(sigma) * rng.standard_normal(z.size)
    z = z - float(z.mean())
    return _softmax(z)


def _ils(a_init, deadline, rng, min_rem, per_eval, per_iter, quick_time=10.0):
    best = _normalize_prob(a_init)
    best_sc = _score(best)
    current = best.copy()
    current_sc = best_sc
    if current_sc != best_sc:
        best_sc = current_sc

    sigmas = (0.003, 0.008, 0.015, 0.03, 0.06, 0.12, 0.25)
    attempt = 0
    quick_time = max(3.0, min(18.0, float(quick_time)))

    while time.monotonic() + min_rem + quick_time < deadline:
        sigma = sigmas[attempt % len(sigmas)]
        a_pert = _perturrent(current, rng, sigma)
        end = min(deadline - min_rem, time.monotonic() + quick_time)
        if end <= time.monotonic() + 1.0:
            break
        try:
            a_opt, sc_opt = _run_lbfgs_z(
                a_pert,
                end,
                min_rem=2.0,
                beta_fracs=_QUICK_BETA_FRACS,
                target_time=max(1.0, quick_time / max(1, len(_QUICK_BETA_FRACS)) - 0.2),
                per_eval=per_eval,
                per_iter=per_iter,
                no_improve_limit=4,
                reg=1e-12
            )
        except Exception:
            attempt += 1
            continue

        if sc_opt < best_sc - 1e-12:
            best = a_opt
            best_sc = sc_opt
            current = a_opt
            current_sc = best_sc
            print(f"  [ils attempt {attempt}] new best {best_sc:.10f} sigma={sigma:.3f} t={time.monotonic():.1f}")
        elif sc_opt < current_sc - 1e-12:
            current = a_opt
            current_sc = sc_opt

        attempt += 1

    return best, best_sc


def _final_polish(a_init, deadline, per_eval, per_iter):
    if time.monotonic() + 20.0 >= deadline:
        return a_init, _score(a_init)
    best, best_sc = _run_lbfgs_z(
        a_init,
        deadline,
        min_rem=2.0,
        beta_fracs=_FINE_BETA_FRACS,
        target_time=8.0,
        per_eval=per_eval,
        per_iter=per_iter,
        no_improve_limit=5,
        reg=1e-13
    )
    return best, best_sc


def propose_candidate(seed=42, budget_s=1800):
    best_a = None
    best_sc = np.inf

    try:
        t0 = time.monotonic()
        budget = float(budget_s)
        deadline = t0 + max(1.0, budget - 30.0)
        rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

        # Load parent if available.
        parent = None
        try:
            parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if parent_raw.size > 0:
                parent_raw = np.clip(parent_raw, 0.0, 1000.0)
                if float(parent_raw.sum()) > 0.01:
                    parent = _normalize_prob(parent_raw)
        except Exception:
            parent = None

        if parent is None:
            parent = _make_profile(24576, "uniform")
            print("No parent available; using uniform fallback.")

        # Choose a safe main length. The parent length is usually best.
        n_main = int(parent.size)
        if n_main < 8192 or n_main > 32768:
            n_main = 24576
        a = _resample(parent, n_main)
        best_a = a
        best_sc = _score(a)
        print(f"Initial: n={n_main}, score={best_sc:.10f}")

        per_iter, per_eval = _estimate_times(a)
        print(f"Timing estimate: per_eval={per_eval:.6f}s, per_iter={per_iter:.6f}s")

        # Allocate time phases.
        total = max(30.0, deadline - t0)
        coarse_time = min(120.0, 0.10 * total)
        main_time = 0.55 * total
        ils_time = 0.25 * total
        final_time = max(10.0, total - coarse_time - main_time - ils_time)

        t_cur = t0
        coarse_end = t_cur + coarse_time if coarse_time > 25.0 else None
        main_end = min(deadline - final_time - 20.0, t_cur + coarse_time + main_time)
        ils_end = min(deadline - final_time - 10.0, main_end + ils_time)
        final_end = min(deadline - 3.0, ils_end + final_time * 0.85)

        # Coarse global search.
        if coarse_end is not None and time.monotonic() + 30.0 < coarse_end:
            print(f"\n[P1] Coarse search until t={time.monotonic()-t0:.1f}s")
            try:
                a2, sc2 = _coarse_search(a, n_main, coerce_end if False else coarse_end,
                                         per_eval, per_iter, rng)
                if sc2 < best_sc - 1e-12:
                    best_a, best_sc = a2, sc2
                    a = a2
            except Exception:
                pass

        # Main fine-grid L-BFGS annealing.
        if time.monotonic() + 60.0 < main_end:
            print(f"\n[P2] Main L-BFGS annealing until t={time.monotonic()-t0:.1f}s")
            try:
                a2, sc2 = _run_lbfgs_z(
                    a,
                    main_end,
                    min_rem=30.0,
                    beta_fracs=_MAIN_BETA_FRACS,
                    target_time=30.0,
                    per_eval=per_eval,
                    per_iter=per_iter,
                    no_improve_limit=6,
                    reg=1e-12
                )
                if sc2 < best_sc - 1e-12:
                    best_a, best_sc = a2, sc2
                    a = a2
            except Exception:
                pass

        # Stochastic iterated local search.
        ils_quick = max(4.0, min(14.0, ils_time / 35.0))
        if time.monotonic() + 45.0 < ils_end:
            print(f"\n[P3] Iterated local search until t={time.monotonic()-t0:.1f}s")
            try:
                a2, sc2 = _ils(
                    a,
                    ils_end,
                    rng,
                    min_rem=30.0,
                    per_eval=per_eval,
                    per_iter=per_iter,
                    quick_time=ils_quick
                )
                if sc2 < best_sc - 1e-12:
                    best_a, best_sc = a2, sc2
                    a = a2
            except Exception:
                pass

        # Final polish.
        if time.monotonic() + 25.0 < final_end:
            print(f"\n[P4] Final polish until t={time.monotonic()-t0:.1f}s")
            try:
                a2, sc2 = _final_polish(a, final_end, per_eval, per_iter)
                if sc2 < best_sc - 1e-12:
                    best_a, best_sc = a2, sc2
            except Exception:
                pass

        # If there is still a chunk of time, do one more ILS burst.
        if time.monotonic() + 90.0 < deadline - 10.0:
            burst_end = min(deadline - 10.0, time.monotonic() + (deadline - 10.0 - time.monotonic()) * 0.5)
            print(f"\n[P5] Extra ILS burst until t={time.monotonic()-t0:.1f}s")
            try:
                a2, sc2 = _ils(
                    a,
                    burst_end,
                    rng,
                    min_rem=15.0,
                    per_eval=per_eval,
                    per_iter=per_iter,
                    quick_time=8.0
                )
                if sc2 < best_sc - 1e-12:
                    best_a, best_sc = a2, sc2
            except Exception:
                pass

        # Final short polish if time remains.
        if time.monotonic() + 15.0 < deadline - 2.0:
            try:
                a2, sc2 = _run_lbfgs_z(
                    a,
                    deadline - 1.0,
                    min_rem=0.0,
                    beta_fracs=(1e-5, 7e-6, 5e-6, 3e-6, 2e-6),
                    target_time=4.0,
                    per_eval=per_eval,
                    per_iter=per_iter,
                    no_improve_limit=4,
                    reg=1e-13
                )
                if sc2 < best_sc - 1e-12:
                    best_a, best_sc = a2, sc2
            except Exception:
                pass

    except Exception:
        print("Exception in propose_candidate; returning best so far or fallback.")

    if best_a is None or not np.isfinite(best_sc):
        best_a = _make_profile(256, "uniform")
        best_sc = _score(best_a)

    final = np.asarray(best_a, dtype=np.float64)
    final = np.clip(final, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01 or not np.isfinite(s):
        final = _make_profile(256, "uniform")

    # Print our internal FFT-based score. The evaluator may differ by roundoff.
    try:
        print(f"\nFINAL internal score: {best_sc:.10f}, n={final.size}, wall={time.monotonic():.1f}")
    except Exception:
        pass

    return [float(x) for x in final]
