Combines power-mean smoothing, top-K log-sum-exp, active-set subgradient,

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _normalize(a):

    if s < 1e-15:

    return a / s

    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]

            e_full[top_idx] = e
            w = e_full / s
            obj = 2.0 * n * (c.max() + beta * np.log(s))
    else:
        c_pow = c ** p
        s = c_pow.sum()
        if s < 1e-300:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
            obj = 2.0 * n * c.max()
        else:
            w = c_pow / s
            M = (s / len(c)) ** (1.0 / p)
            obj = 2.0 * n * M
    
    ga = _grad_weighted(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


class _PowerMeanProb:
    def __init__(self, p, n):
        self.p = p
        self.n = n

    def __call__(self, z):
        return _power_mean_obj_grad(z, self.p, self.n)[0]

    def grad(self, z):
        return _power_mean_obj_grad(z, self.p, self.n)[1]


def _topk_smoobj_grad(z, beta, K, n):
    """Top-K log-sum-exp smoothed objective and gradient in log-domain."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c
    
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    
    ga = _grad_weighted(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


class _TopkSmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[1]


def _run_lbfgs_powermean(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with power-mean smoothing, adaptive p schedule."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # Power p schedule: start moderate, increase for sharpness
    p_schedule = [20, 40, 80, 120, 200, 400, 800, 1600]
    
    for p in p_schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 15:
            break
        
        est_time = 0.004 * max(1.0, n / 8192.0)
        max_iter = min(4000, max(30, int(remaining / (est_time * 15))))
        
        prob = _PowerMeanProb(p, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 40}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _run_lbfgs_topk(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with top-K log-sum-exp, adaptive K/beta schedule."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    schedule = [
        (1.0, 0.08), (0.6, 0.06), (0.4, 0.04), (0.25, 0.03),
        (0.15, 0.02), (0.1, 0.015), (0.05, 0.01), (0.02, 0.008),
        (0.01, 0.005), (0.005, 0.003), (0.002, 0.002),
    ]
    
    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        beta = base_beta * beta_mult
        K = max(5, int((2 * n - 1) * K_frac))
        
        est_time = 0.004 * max(1.0, n / 8192.0)
        max_iter = min(4000, max(30, int(remaining / (est_time * 15))))
        
        prob = _TopkSmoProb(beta, K, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 40}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _active_set_subgradient(a_init, deadline, min_time_remaining=30.0):
    """Active-set subgradient with adaptive steps and momentum."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    n_2nm1 = 2 * n - 1
    active_frac = 0.005
    momentum = 0.6
    lr = 0.01
    lr_min = 1e-5
    lr_max = 0.05
    stall = 0
    max_stall = 60
    velocity = np.zeros(n)
    prev_a = None
    prev_g = None
    iteration = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        n_active = max(3, min(n_2nm1, int(n_2nm1 * active_frac)))
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_active]
        
        active_vals = c[active_idx]
        w = np.zeros(n_2nm1)
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w[active_idx] = (active_vals - active_vals.min()) / spread + 0.05
        else:
            w[active_idx] = 1.0 / n_active
        w = w / w.sum()
        
        g = _grad_weighted(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break
        
        if prev_a is not None and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g_centered - prev_g
            s_norm = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm > 1e-18 and abs(sy) > 1e-18:
                lr_bb = s_norm / abs(sy)
                lr = np.clip(lr_bb, lr_min, lr_max)
        
        velocity = momentum * velocity - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        
        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            for ls in range(20):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-15:
                    improved = True
                    break
                velocity = momentum * velocity * 0.5 - lr * 0.5 * g_centered
                a_try = a + velocity
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    velocity = np.zeros(n)
                    a_try = a
                    break
                a_try = a_try / s_try
                lr *= 0.7
        
        if improved:
            prev_a = a.copy()
            prev_g = g_centered.copy()
            a = a_try
            lr = min(lr_max, lr * 1.02)
            active_frac = max(0.001, active_frac * 0.97)
        else:
            velocity = np.zeros(n)
            lr = max(lr_min, lr * 0.6)
            active_frac = min(0.05, active_frac * 1.4)
        
        if stall > max_stall:
            active_frac = min(0.05, active_frac * 2.0)
            lr = max(lr_min, lr * 0.4)
            momentum = min(0.85, momentum + 0.05)
            stall = 0
    
    return best_a, best_sc


def _multiplicative_weights(a_init, deadline, min_time_remaining=20.0):
    """Exponentiated gradient / multiplicative weights on the simplex."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.05
    stall = 0
    max_stall = 50
    n_2nm1 = 2 * n - 1
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                break
        
        n_active = max(3, int(n_2nm1 * 0.003))
        top_idx = np.argpartition(c, -n_active)[-n_active:]
        w = np.zeros(n_2nm1)
        w[top_idx] = 1.0 / n_active
        
        g = _grad_weighted(a, w)
        g_centered = g - g.mean()
        
        # Multiplicative update
        a_try = a * np.exp(-lr * g_centered)
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-10:
            a_try = a + (-lr * 0.1) * g_centered
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-10:
                break
        a_try = a_try / s
        
        c_try = _full_convolve(a_try, a_try)
        sc_try = 2.0 * n * float(c_try.max())
        
        if sc_try < current_sc - 1e-15:
            a = a_try
            lr = min(0.1, lr * 1.01)
            stall = 0
        else:
            lr = max(0.005, lr * 0.8)
            stall += 1
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered2":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "arcsine":
        t = np.linspace(0.001, 0.999, n)
        a = 1.0 / (np.pi * np.sqrt(t * (1 - t)))
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2) + 0.03
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.where(np.abs(t - 0.5) < 0.3, 1.0, 0.1)
        a = a + 0.05 * np.sin(20 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss_narrow":
        t = np.linspace(0, 1, n)
        a = np.exp(-5.0 * (t - 0.5) ** 2) + 0.02
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb_log(a, sigma, rng):
    """Perturb in log-space."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _perturb_structured(a, rng):
    """Structured perturbation: add smooth noise."""
    n = len(a)
    # Add low-frequency noise
    n_freq = min(5, n // 4)
    freqs = rng.integers(1, max(2, n_freq + 1), size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    amps = rng.uniform(0.02, 0.08, size=3)
    noise = np.zeros(n)
    for i in range(3):
        noise += amps[i] * np.sin(2 * np.pi * freqs[i] * np.arange(n) / n + phases[i])
    a_new = a * (1.0 + noise)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return _perturb_log(a, 0.05, rng)
    return a_new / s


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    """Very fine active-set polish."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.003
    tol = 0.003
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active
        
        g = _grad_weighted(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-16:
            break
        
        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        improved = False
        for ls in range(30):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                tol = max(0.0002, tol * 0.7)
                lr *= 0.7
            if no_improve > 400:
                a = best_a.copy()
                tol = max(0.0002, tol * 0.5)
                lr = 0.002
                no_improve = 0
        else:
            if tol > 0.0002:
                tol = max(0.0002, tol * 0.96)
            lr = min(0.008, lr * 1.01)
    
    return best_a, best_sc


def _basin_hopping(a_best, deadline, min_time_remaining, rng):
    """Structured basin hopping: perturb and re-optimize."""
    best_a = a_best.copy()
    best_sc = _score(a_best)
    
    n = len(a_best)
    
    # Different perturbation strategies
    strategies = [
        (0.02, "log"), (0.04, "log"), (0.07, "log"), (0.1, "log"),
        (0.0, "struct"), (0.0, "struct"), (0.0, "struct"),
        (0.015, "log"), (0.03, "log"), (0.05, "log"),
    ]
    
    for i, (sigma, kind) in enumerate(strategies):
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 25:
            break
        
        if kind == "log":
            a_pert = _perturb_log(a_best, sigma, rng)
        else:
            a_pert = _perturb_structured(a_best, rng)
        
        # Quick optimization: L-BFGS with top-K
        sub_deadline = time.monotonic() + min(20, remaining * 0.3)
        a_opt = _run_lbfgs_topk(a_pert, sub_deadline, min_time_remaining=2)
        sc_opt = _score(a_opt)
        
        if sc_opt < best_sc:
            best_a = a_opt.copy()
            best_sc = sc_opt
            a_best = a_opt.copy()
            print(f"    Basin hop {i}: NEW BEST {best_sc:.10f}")
    
    return best_a, best_sc

            print(f"  NEW BEST: {best_score:.10f} [P4 multweights] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 5: L-BFGS round 2 (both methods) ===
    print(f"\n[P5] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        p5a_deadline = min(deadline - 250, time.monotonic() + 150)
        a5a = _run_lbfgs_powermean(best_seq, p5a_deadline, min_time_remaining=20)
        sc5a = _score(a5a)
        if sc5a < best_score:
            best_seq = a5a.copy()
            best_score = sc5a
            print(f"  NEW BEST: {best_score:.10f} [P5a powermean] t={time.monotonic()-t_start:.1f}s")
    
    if time.monotonic() < deadline - 200:
        p5b_deadline = min(deadline - 160, time.monotonic() + 100)
        a5b = _run_lbfgs_topk(best_seq, p5b_deadline, min_time_remaining=15)
        sc5b = _score(a5b)
        if sc5b < best_score:
            best_seq = a5b.copy()
            best_score = sc5b
            print(f"  NEW BEST: {best_score:.10f} [P5b topk] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 6: Basin hopping ===
    print(f"\n[P6] Basin hopping (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        p6_deadline = min(deadline - 100, time.monotonic() + 200)
        a6, sc6 = _basin_hopping(best_seq, p6_deadline, min_time_remaining=20, rng=rng)
        if sc6 < best_score:
            best_seq = a6.copy()
            best_score = sc6
            print(f"  NEW BEST: {best_score:.10f} [P6 bashing] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 7: Fine polish ===
    print(f"\n[P7] Fine polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        p7_deadline = min(deadline - 30, time.monotonic() + 80)
        a7, sc7 = _fine_polish(best_seq, p7_deadline, min_time_remaining=5)
        if sc7 < best_score:
            best_seq = a7.copy()
            best_score = sc7
            print(f"  NEW BEST: {best_score:.10f} [P7 polish] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 8: Multi-n exploration ===
    print(f"\n[P8] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 50:
        n_main = len(best_seq)
        n_options = [16384, 24576, 12288, 32768]
        n_options = [x for x in n_options if x != n_main]
        
        for n_try in n_options[:2]:
            remaining = deadline - 25 - time.monotonic()
            if remaining < 25:
                break
            a_res = _resample(best_seq, n_try)
            sub_deadline = time.monotonic() + min(20, remaining * 0.5)
            a_opt = _run_lbfgs_topk(a_res, sub_deadline, min_time_remaining=3)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [P8 n={n_try}] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 9: Final optimization ===
    print(f"\n[P9] Final optimization (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Quick L-BFGS with very tight parameters
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        beta_fin = max(mb * 0.0003, 1e-9)
        K_fin = max(5, int(n * 0.001))
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        prob = _TopkSmoProb(beta_fin, K_fin, n)
        max_iter = min(2000, max(10, int(remaining * 0.5 / 0.02)))
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-15, 'maxcor': 60})
            a_fin = np.exp(result.x - result.x.max())
            a_fin = a_fin / a_fin.sum()
            sc_fin = _score(a_fin)
            if sc_fin < best_score:
                best_seq = a_fin.copy()
                best_score = sc_fin
                print(f"  NEW BEST: {best_score:.10f} [P9 final_lbfgs] t={time.monotonic()-t_start:.1f}s")
        except Exception:
            pass
    
    # Final fine polish
    remaining = deadline - 3 - time.monotonic()
    if remaining > 5:
        a_fp, sc_fp = _fine_polish(best_seq, time.monotonic() + remaining, min_time_remaining=1)
        if sc_fp < best_score:
            best_seq = a_fp.copy()
            best_score = sc_fp
            print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")
    
    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
