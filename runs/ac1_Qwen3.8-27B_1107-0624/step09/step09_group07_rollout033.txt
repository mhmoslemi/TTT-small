        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-17:
            break
        alpha = min(lr, 0.4 / (gnorm + 1e-15))
        improved = False
        for ls in range(40):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-16:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                tol = max(0.00005, tol * 0.7)
                lr *= 0.75
            if no_improve > 400:
                a = best_a.copy()
                tol = max(0.00005, tol * 0.5)
                lr = 0.002
                no_improve = 0
        else:
            if tol > 0.00005:
                tol = max(0.00005, tol * 0.97)
            lr = min(0.008, lr * 1.005)
    return best_a, best_sc
