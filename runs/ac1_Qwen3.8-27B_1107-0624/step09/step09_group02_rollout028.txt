def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

"""
Improved optimizer for minimum peak autocorrelation sequences.
Key fix: corrected gradient computation (both terms of cross-correlation).
Combines subgradient method, L-BFGS polish in log-space, projected gradient,
and multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _compute_subgrad(a, w, n):
    """Compute the full subgradient of weighted sum of autocorrelation lags.
    g[m] = sum_k w[k] * (a[k-m] + a[m-k])
    = convolve(w, a_rev)[n-1+m] + convolve(a, w)[m]
    """
    a_rev = a[::-1]
    t1 = _full_convolve(w, a_rev)[n - 1:2 * n - 1]
    t2 = _full_convolve(a, w)[:n]
    return t1 + t2


def _lbfgs_grad_z(z_local, beta, n):
    """Gradient of smoothed objective w.r.t. log-space params z."""
    a_local = _softmax(z_local)
    c_local = _full_convolve(a_local, a_local)
    c_max_local = float(c_local.max())
    shifted = (c_local - c_max_local) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s_e = e.sum()
    if s_e < 1e-300:
        w = np.zeros_like(c_local)
        w[int(np.argmax(c_local))] = 1.0
    else:
        w = e / s_e
    # Full subgradient (both terms!)
    a_rev = a_local[::-1]
    h1 = _full_convolve(w, a_rev)[n - 1:2 * n - 1]
    h2 = _full_convolve(a_local, w)[:n]
    H = h1 + h2
    W_val = float(np.dot(w, c_local))
    return (4.0 * n / beta) * a_local * (H - W_val)


def _lbfgs_obj_z(z_local, beta, n):
    """Smoothed objective in log-space."""
    a_local = _softmax(z_local)
    c_local = _full_convolve(a_local, a_local)
    c_max_local = float(c_local.max())
    shifted = (c_local - c_max_local) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s_e = e.sum()
    if s_e < 1e-300:
        return 2.0 * n * c_max_local
    return 2.0 * n * (c_max_local + beta * np.log(s_e))

    s = a.sum()
    if s < 1e-15:
        return a, np.inf

    best_a = a.copy()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())

                break
            step *= 0.5
            a_try = _proj_simplex(a - step * momentum)
            if step < 1e-14:
                a = a_try
                break

        if it % 50 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
                best_ever = sc
                stall = 0
            else:
                stall += 1
                if stall > 20:
                    a = best_a.copy()
                    momentum = np.zeros(n)
                    stall = 0
                    beta_m *= 0.9

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * t
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.3) ** 2) + 0.2
        return a / a.sum()
    elif kind == "bump2":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.7) ** 2) + 0.2
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer with corrected gradient for minimum peak autocorrelation.
    Uses subgradient, L-BFGS in log-space, projected gradient, and multi-n search.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score_val:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Subgradient with corrected gradient (300s) ===
    print(f"\n[Phase 1] Subgradient corrected (t={time.monotonic() - t_start:.1f}s)")
    phase1_deadline = t_start + 300
    a_sg, sc_sg = _run_subgradient(best_seq, phase1_deadline, min_time_remaining=30.0)
    update_best(a_sg, _score(a_sg), "subgrad_1")

    # === Phase 2: L-BFGS polish with corrected gradient (400s) ===
    print(f"\n[Phase 2] L-BFGS polish (t={time.monotonic() - t_start:.1f}s)")
    phase2_deadline = min(deadline - 700, phase1_deadline + 400)
    if best_seq is not None and time.monotonic() < phase2_deadline:
        a_lb, sc_lb = _run_lbfgs_polish(best_seq, phase2_deadline, min_time_remaining=30.0)
        update_best(a_lb, sc_lb, "lbfgs_1")

    # === Phase 3: Projected gradient with momentum (200s) ===
    print(f"\n[Phase 3] Projected gradient (t={time.monotonic() - t_start:.1f}s)")
    phase3_deadline = min(deadline - 500, phase2_deadline + 200)
    if best_seq is not None and time.monotonic() < phase3_deadline:
        a_pg, sc_pg = _run_proj_grad_corrected(best_seq, phase3_deadline, min_time_remaining=20.0)
        update_best(a_pg, sc_pg, "projgrad_1")

    # === Phase 4: Second subgradient round from improved point (200s) ===
    print(f"\n[Phase 4] Subgradient round 2 (t={time.monotonic() - t_start:.1f}s)")
    phase4_deadline = min(deadline - 300, phase3_deadline + 200)
    if best_seq is not None and time.monotonic() < phase4_deadline:
        a_sg2, sc_sg2 = _run_subgradient(best_seq, phase4_deadline, min_time_remaining=20.0)
        update_best(a_sg2, sc_sg2, "subgrad_2")

    # === Phase 5: L-BFGS with even tighter beta (200s) ===
    print(f"\n[Phase 5] Tight L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    phase5_deadline = min(deadline - 200, phase4_deadline + 200)
    if best_seq is not None and time.monotonic() < phase5_deadline:
        a_lb2, sc_lb2 = _run_lbfgs_polish(best_seq, phase5_deadline, min_time_remaining=20.0)
        update_best(a_lb2, sc_lb2, "lbfgs_tight")

    # === Phase 6: Multi-n exploration (150s) ===
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    phase6_deadline = min(deadline - 100, phase5_deadline + 150)
    n_tries = [65536, 49152, 24576]
    for n_try in n_tries:
        if time.monotonic() > phase6_deadline - 20:
            break
        if n_try == len(best_seq) if best_seq is not None else False:
            continue
        if best_seq is not None:
            a_init_n = _resample(best_seq, n_try)
        else:
            a_init_n = _make_profile(n_try, "tapered")
        a_n, sc_n = _run_lbfgs_polish(a_init_n, phase6_deadline, min_time_remaining=15.0)
        sc_n_val = _score(a_n)
        update_best(a_n, sc_n_val, f"n_{n_try}")

    # === Phase 7: Perturbation + re-optimize (80s) ===
    print(f"\n[Phase 7] Perturbation (t={time.monotonic() - t_start:.1f}s)")
    phase7_deadline = min(deadline - 60, phase6_deadline + 80)
    if best_seq is not None and time.monotonic() < phase7_deadline:
        for sigma in [0.01, 0.02]:
            if time.monotonic() > phase7_deadline - 15:
                break
            a_pert = _perturb(best_seq, sigma, rng)
            a_po, sc_po = _run_lbfgs_polish(a_pert, phase7_deadline, min_time_remaining=10.0)
            update_best(a_po, sc_po, f"perturb_{sigma}")

    # === Phase 8: Final ultra-tight polish (60s) ===
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        beta_fin = mb * 1e-6
        remaining = deadline - 10 - time.monotonic()
        if remaining > 3:
            max_iter = min(500, max(30, int(remaining / 0.02)))

            def final_obj(zl):
                return _lbfgs_obj_z(zl, beta_fin, n)

            def final_grad(zl):
                return _lbfgs_grad_z(zl, beta_fin, n)

            try:
                result = minimize(
                    final_obj, z,
                    jac=final_grad,
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                             'maxcor': 40, 'maxls': 50}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_polish")
            except Exception:
                pass

        # Final projected gradient
        if time.monotonic() < deadline - 10:
            a_fg, sc_fg = _run_proj_grad_corrected(best_seq, deadline - 10,
                                                   min_time_remaining=3.0)
            update_best(a_fg, sc_fg, "final_pg")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(8192, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
