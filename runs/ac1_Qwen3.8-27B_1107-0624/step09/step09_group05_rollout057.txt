from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a = a / s
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()

        return np.ones_like(z) / len(z)
    return e / s

    wc = _full_convolve(w, a_rev)

    residual = w * (c - t)

        if s_new < 1e-10:
            return a, _score(a)

        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:

        if sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_a = a_new.copy()
        
        a = a_new

    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    M = _next_pow2(2 * n - 1)

    # Target C: DFT of ideal flat convolution (rect window of length 2n-1, value 1/(2n-1))
    rect = np.zeros(M // 2 + 1)
    # Compute target in full complex domain
    rect_full = np.zeros(M)
    rect_full[:2 * n - 1] = 1.0 / (2 * n - 1)
    C_target = np.fft.rfft(rect_full)

    # Precompute sqrt of C_target (complex)
    r_t = np.abs(C_target)
    th_t = np.angle(C_target)
    A_target = np.sqrt(r_t) * np.exp(0.5j * th_t)

    alphas = np.geomspace(0.5, 0.01, n_alpha_steps)

    for alpha in alphas:
        if time.monotonic() + min_time > deadline:
            break
        n_inner = 50
        for it in range(n_inner):
            if time.monotonic() + min_time > deadline:
                break
            A = np.fft.rfft(a, M)
            A_new = (1 - alpha) * A + alpha * A_target
            a_new = np.fft.irfft(A_new, M)[:n]
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-12:
                best_sc = sc_new
                best_a = a_new.copy()
            a = a_new

        if verbose:
            elapsed = time.monotonic() - t_start_global
            print(f"    spectral alpha={alpha:.3f}: score={best_sc:.10f}, t={elapsed:.0f}s")

    return best_a, best_sc


def _multiplicative_weight(a_init, deadline, min_time=10.0, verbose=True):
    """Mirror descent / multiplicative weight updates.
    a_new[j] proportional to a[j] * exp(-eta * g[j]) where g is active gradient.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    eta = 0.5
    tol_scale = 0.01
    stuck = 0

    it = 0
    while time.monotonic() + min_time < deadline:
        it += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 2000:
                eta *= 0.9
                tol_scale = max(0.001, tol_scale * 0.8)
                stuck = 0
                if eta < 1e-4:
                    break

        # Active set
        tol = tol_scale * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        # Average gradient over active set
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        grad = _grad_conv_weighted(a, w)

        # Multiplicative update
        # Project gradient to tangent
        grad_c = grad - np.mean(grad)

        a_log = np.log(np.maximum(a, 1e-30))
        a_log -= eta * grad_c
        a_new = np.exp(a_log - np.max(a_log))
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < current_sc - 1e-14:
            a = a_new
        else:
            # Halve eta and retry
            eta *= 0.7
            a_log = np.log(np.maximum(a, 1e-30))
            a_log -= eta * grad_c
            a_new = np.exp(a_log - np.max(a_log))
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            a = a_new

        if verbose and it % 20000 == 0:
            elapsed = time.monotonic() - t_start_global
            print(f"    mult_weight it={it}: score={best_sc:.10f}, eta={eta:.4f}, t={elapsed:.0f}s")

    return best_a, best_sc


def _active_set_kkt(a_init, deadline, min_time=10.0, verbose=True):
    """Active-set KKT refinement with adaptive tolerance.
    At optimum: gradient of active constraints should be constant on support of a.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_frac = 0.01
    min_tol_frac = 0.0001
    eta = 0.1
    stuck = 0

    it = 0
    while time.monotonic() + min_time < deadline:
        it += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 1000:
            tol_frac = max(min_tol_frac, tol_frac * 0.7)
            eta *= 0.8
            stuck = 0
            if tol_frac < min_tol_frac * 1.2:
                break

        # Active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        # Average gradient
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        grad_avg = _grad_conv_weighted(a, w)

        # Support of a
        support = a > 1e-10 * a.max()
        n_sup = int(support.sum())
        if n_sup < 2:
            break

        # KKT: grad_avg should be constant on support
        lam = float(np.mean(grad_avg[support]))
        residual = grad_avg[support] - lam

        # Update
        a_new = a.copy()
        a_new[support] -= eta * residual
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Line search
        sc_new = _score(a_new)
        if sc_new >= current_sc:
            for trial in [0.5, 0.25, 0.1, 0.05]:
                a_try = a.copy()
                a_try[support] -= trial * eta * residual
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-14:
                    a_new = a_try
                    sc_new = sc_try
                    eta = trial * eta
                    break
            else:
                eta *= 0.7
                continue

        a = a_new

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t_start_global
            print(f"    active_set it={it}: score={best_sc:.10f}, n_active={n_active}, tol={tol_frac:.5f}, t={elapsed:.0f}s")

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time=10.0, verbose=True, max_beta_levels=15):
    """L-BFGS with log-sum-exp soft-max and adaptive beta schedule."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    # Beta schedule
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < c0_max * 0.005:
        gap = c0_max * 0.005

    multipliers = np.array([
        1.0, 0.4, 0.2, 0.1, 0.05, 0.025,
        0.0125, 0.00625, 0.003, 0.0015,
        0.0008, 0.0004, 0.0002, 0.0001, 0.00005
    ])
    betas = gap * multipliers[:max_beta_levels]

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 3:
            break

        max_iter = min(5000, max(50, int(remaining / 0.01)))

        def obj(z_):
            a_ = _softmax(z_)
            c_ = _full_convolve(a_, a_)
            return 2.0 * n * _soft_max_value(c_, beta)

        def grad(z_):
            a_ = _softmax(z_)
            c_ = _full_convolve(a_, a_)
            w_ = _soft_max_weights(c_, beta)
            g_a = _grad_conv_weighted(a_, w_)
            W = float(np.dot(g_a, a_))
            return a_ * (g_a - W)

        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    elapsed = time.monotonic() - t_start_global
                    print(f"    lbfgs beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}, t={elapsed:.0f}s")
        except Exception as e:
            if verbose:
                print(f"    lbfgs beta={beta:.2e}: error {e}")

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


t_start_global = 0.0


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    P1: Convolution matching (fast iterative, main workhorse)
    P2: L-BFGS refinement
    P3: Active-set KKT
    P4: Spectral shaping (frequency domain)
    P5: Multiplicative weight updates
    P6: Multi-n exploration
    P7: Random perturbation + re-optimize
    P8: Final polish
    """
    global t_start_global
    t_start_global = time.monotonic()
    deadline = t_start_global + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(16384, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n=16384: score={best_score:.10f}")

    n_main = len(best_seq)

    # === Phase 1: Convolution matching ===
    p1_time = min(400, (budget_s - 10) * 0.25)
    print(f"\n[P1] Convolution matching (n={n_main}, budget={p1_time:.0f}s, t={time.monotonic()-t_start_global:.0f}s)")
    p1_end = t_start_global + p1_time
    a1, sc1 = _conv_matching(best_seq, p1_end, min_time=200, verbose=True)
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  P1 result: {best_score:.10f}")

    # === Phase 2: L-BFGS ===
    print(f"\n[P2] L-BFGS (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p2_end = min(deadline - 300, time.monotonic() + 350)
        a2, sc2 = _run_lbfgs(best_seq, p2_end, min_time=100, verbose=True, max_beta_levels=12)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(f"  P2 result: {best_score:.10f}")

    # === Phase 3: Active-set KKT ===
    print(f"\n[P3] Active-set KKT (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 40 < deadline:
        p3_end = min(deadline - 200, time.monotonic() + 200)
        a3, sc3 = _active_set_kkt(best_seq, p3_end, min_time=60, verbose=True)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3 result: {best_score:.10f}")

    # === Phase 4: Spectral shaping ===
    print(f"\n[P4] Spectral shaping (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 40 < deadline:
        p4_end = min(deadline - 150, time.monotonic() + 150)
        a4, sc4 = _spectral_shaping(best_seq, p4_end, min_time=50, n_alpha_steps=15, verbose=True)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4 result: {best_score:.10f}")

    # === Phase 5: Multiplicative weight ===
    print(f"\n[P5] Multiplicative weight (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 40 < deadline:
        p5_end = min(deadline - 100, time.monotonic() + 120)
        a5, sc5 = _multiplicative_weight(best_seq, p5_end, min_time=40, verbose=True)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  P5 result: {best_score:.10f}")

    # === Phase 6: Convolution matching round 2 (after other methods) ===
    print(f"\n[P6] Convolution matching round 2 (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p6_end = min(deadline - 80, time.monotonic() + 120)
        a6, sc6 = _conv_matching(best_seq, p6_end, min_time=30, verbose=True)
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  P6 result: {best_score:.10f}")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n exploration (t={time.monotonic()-t_start_global:.0f}s)")
    n_options = [12288, 16384, 20480, 24576]
    n_options = [x for x in n_options if x != n_main]
    for n_try in n_options:
        if time.monotonic() + 60 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if time.monotonic() + 40 > deadline:
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
            continue
        p7_end = min(deadline - 40, time.monotonic() + 80)
        a7, sc7 = _conv_matching(a_r, p7_end, min_time=20, verbose=False)
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()
            print(f"  n={n_try}: {best_score:.10f}")

    # === Phase 8: Random perturbation + re-optimize ===
    print(f"\n[P8] Random perturbation (t={time.monotonic()-t_start_global:.0f}s)")
    rng = np.random.default_rng(seed)
    n_pert = len(best_seq)
    for trial in range(5):
        if time.monotonic() + 50 > deadline:
            break
        # Perturb
        a_p = best_seq.copy()
        noise = rng.normal(0, 0.05, n_pert)
        a_p = a_p + noise * a_p
        a_p = np.maximum(a_p, 0.0)
        s_p = a_p.sum()
        if s_p < 1e-10:
            continue
        a_p = a_p / s_p
        sc_p = _score(a_p)

        # Quick optimization
        if time.monotonic() + 30 > deadline:
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_p.copy()
            continue
        p8_end = min(deadline - 20, time.monotonic() + 50)
        a8, sc8 = _conv_matching(a_p, p8_end, min_time=15, verbose=False)
        if sc8 < best_score:
            best_score = sc8
            best_seq = a8.copy()
            print(f"  pert trial {trial}: {best_score:.10f}")

    # === Phase 9: Final polish ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 15 < deadline:
        p9_end = min(deadline - 5, time.monotonic() + 60)
        a9, sc9 = _run_lbfgs(best_seq, p9_end, min_time=5, verbose=False, max_beta_levels=5)
        if sc9 < best_score:
            best_score = sc9
            best_seq = a9.copy()

        if time.monotonic() + 10 < deadline:
            p9b_end = min(deadline - 3, time.monotonic() + 30)
            a9b, sc9b = _active_set_kkt(best_seq, p9b_end, min_time=3, verbose=False)
            if sc9b < best_score:
                best_score = sc9b
                best_seq = a9b.copy()

    # Final check and return
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start_global:.1f}s")
    return [float(x) for x in final]
