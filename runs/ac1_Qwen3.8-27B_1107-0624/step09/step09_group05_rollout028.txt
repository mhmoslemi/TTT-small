    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad_a = 2.0 * wc[n - 1:2 * n - 1]
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)


class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, verbose=False):
    """L-BFGS with adaptive beta schedule for soft-max smoothed objective."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c0_max * 0.003:
        gap = c0_max * 0.003

    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
        0.00003125, 0.000015625, 0.0000078, 0.000004,
        0.000002, 0.000001, 0.0000005
    ])
    betas = gap * multipliers

    t_start = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        size_factor = max(1.0, n / 8192.0)
        max_iter = min(6000, max(50, int(remaining / (0.003 * size_factor * 15))))
        if beta < gap * 0.001:
            max_iter = min(max_iter, 1500)

        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    elapsed = time.monotonic() - t_start
                    print(f"      beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}, t={elapsed:.0f}s")
        except Exception as e:
            if verbose:
                print(f"      beta={beta:.2e}: error {e}")

    return best_a, best_sc


def _newton_polish(a_init, deadline, min_time_remaining=15.0, verbose=False):
    """Newton active-set method using second-order structure.
    
    Key insight: J*J^T = 4*Toeplitz(autocorrelation) for the active set,
    making each Newton step O(|S|^3 + n*log(n)).
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    a_rev = a[::-1]
    tol_frac = 0.003
    max_active = 150
    reg_scale = 1e-6
    no_improve = 0
    max_no_improve = 50
    iteration = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac *= 0.5
            reg_scale *= 5.0
            if tol_frac < 1e-7:
                break
            no_improve = 0
            max_no_improve = max(20, max_no_improve * 0.8)

        # Identify active set
        threshold = c_max * (1.0 - tol_frac)
        active = c >= threshold
        n_active = int(active.sum())

        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros(len(c), dtype=bool)
            active[idx] = True
            n_active = 1

        # Cap active set
        if n_active > max_active:
            c_masked = np.where(active, c, -1e100)
            top_idx = np.argsort(c_masked)[::-1][:max_active]
            active = np.zeros(len(c), dtype=bool)
            active[top_idx] = True
            n_active = max_active

        S = np.where(active)[0].astype(int)

        # Form Gram matrix: G[i,j] = 4 * c[|S[j]-S[i]|]
        G = np.zeros((n_active, n_active))
        for i in range(n_active):
            si = S[i]
            for j in range(i, n_active):
                lag = abs(int(S[j]) - si)
                if lag < len(c):
                    val = 4.0 * c[lag]
                else:
                    val = 0.0
                G[i, j] = val
                G[j, i] = val

        # Regularization
        tr_G = np.trace(G) / max(n_active, 1)
        lam = reg_scale * tr_G
        G += lam * np.eye(n_active)

        # RHS
        r = c_max - c[S]

        # Solve G w = r
        try:
            cond = np.linalg.cond(G)
            if cond > 1e12:
                w = np.linalg.lstsq(G, r, rcond=1e-10)[0]
            else:
                w = np.linalg.solve(G, r)
        except np.linalg.LinAlgError:
            w = np.linalg.lstsq(G, r, rcond=1e-10)[0]

        # Compute delta_a = 2 * (a_rev * w_ext)[n-1:2n-1]
        w_ext = np.zeros(len(c))
        w_ext[S] = w

        wc = _full_convolve(a_rev, w_ext)
        delta_a = 2.0 * wc[n - 1:2 * n - 1]

        # Project to zero-sum (maintain normalization)
        delta_a -= np.mean(delta_a)

        # Enforce non-negativity constraint on step
        floor = -0.95 * a
        delta_a = np.maximum(delta_a, floor)

        # Line search (backtracking)
        alpha = 1.0
        improved = False
        base_sc = current_sc
        for _ in range(30):
            a_new = a + alpha * delta_a
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < base_sc - 1e-15:
                a = a_new
                a_rev = a[::-1]
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # Try a small gradient step as fallback
            tol_g = 0.002 * c_max
            active_g = c >= (c_max - tol_g)
            if active_g.sum() == 0:
                active_g = np.zeros(len(c), dtype=bool)
                active_g[int(np.argmax(c))] = True
            n_ag = int(active_g.sum())
            w_g = np.zeros(len(c))
            w_g[active_g] = 1.0 / n_ag
            wc_g = _full_convolve(a_rev, w_g)
            grad = 2.0 * wc_g[n - 1:2 * n - 1]
            grad -= np.mean(grad)
            g_norm = np.linalg.norm(grad)
            if g_norm < 1e-20:
                break
            step = 0.01 / g_norm
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-10:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < base_sc - 1e-14:
                    a = a_try
                    a_rev = a[::-1]

        iteration += 1
        if verbose and iteration % 50 == 0:
            print(f"      Newton iter {iteration}: score={best_sc:.10f}, "
                  f"|S|={n_active}, tol={tol_frac:.2e}")

        # Tighten as we converge
        if best_sc < 2.003:
            tol_frac = max(1e-6, tol_frac * 0.98)

    return best_a, best_sc


def _peak_shave(a_init, deadline, min_time_remaining=10.0):
    """Iterative peak shaving: reduce elements contributing most to top peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    a_rev = a[::-1]

    no_improve = 0
    max_no_improve = 100

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        # Find top peaks
        top_k = min(20, len(c))
        top_idx = np.argsort(c)[::-1][:top_k]
        top_val = c[top_idx[0]]

        # Compute blame: for each a[j], how much it contributes to top peaks
        # blame[j] = sum_{k in top_idx} a[k-j] (for valid k-j)
        w = np.zeros(len(c))
        w[top_idx] = 1.0 / top_k
        wc = _full_convolve(a_rev, w)
        blame = wc[n - 1:2 * n - 1]

        # Reduce high-blame elements, increase low-blame elements
        blame_mean = np.mean(blame)
        direction = blame - blame_mean
        # We want to reduce blame: move in -direction
        # But also maintain sum=1
        direction -= np.mean(direction)

        # Adaptive step size
        d_norm = np.linalg.norm(direction)
        if d_norm < 1e-20:
            break
        step = 0.05 / d_norm

        a_new = a - step * direction
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)

        if sc_new < current_sc - 1e-14:
            a = a_new
            a_rev = a[::-1]

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n, dtype=np.float64)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 13.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 7.0 * t / n)
             + 0.08 * np.sin(2.0 * np.pi * 31.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp":
        t = np.arange(n, dtype=np.float64)
        # Use magnitude of chirp-like structure
        phase = np.pi * t * t / n
        a = 0.5 + 0.5 * np.cos(phase)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-60.0 * (t - 0.25) ** 2) + np.exp(-60.0 * (t - 0.75) ** 2) + 0.005
        return a / a.sum()
    elif kind == "random_nonneg":
        rng = np.random.default_rng(123)
        a = np.abs(rng.normal(0, 1, n)) + 0.01
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.5 * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _optimize_sequence(a_init, deadline, min_time_remaining=30.0, verbose=False):
    """Full optimization pipeline: L-BFGS -> Newton -> L-BFGS -> Newton."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(best_a)

    # Step 1: L-BFGS coarse
    t1 = min(deadline - min_time_remaining - 60, time.monotonic() + (deadline - time.monotonic()) * 0.35)
    if time.monotonic() < t1:
        a1, sc1 = _run_lbfgs(best_a, t1, min_time_remaining=min_time_remaining + 60, verbose=verbose)
        if sc1 < best_sc:
            best_sc = sc1
            best_a = a1
            if verbose:
                print(f"      After L-BFGS 1: {best_sc:.10f}")

    # Step 2: Newton polish
    t2 = min(deadline - min_time_remaining - 30, time.monotonic() + (deadline - time.monotonic()) * 0.3)
    if time.monotonic() < t2:
        a2, sc2 = _newton_polish(best_a, t2, min_time_remaining=min_time_remaining + 30, verbose=verbose)
        if sc2 < best_sc:
            best_sc = sc2
            best_a = a2
            if verbose:
                print(f"      After Newton 1: {best_sc:.10f}")

    # Step 3: L-BFGS fine (smaller beta)
    t3 = min(deadline - min_time_remaining - 15, time.monotonic() + (deadline - time.monotonic()) * 0.25)
    if time.monotonic() < t3:
        a3, sc3 = _run_lbfgs(best_a, t3, min_time_remaining=min_time_remaining + 15, verbose=False)
        if sc3 < best_sc:
            best_sc = sc3
            best_a = a3
            if verbose:
                print(f"      After L-BFGS 2: {best_sc:.10f}")

    # Step 4: Final Newton polish
    t4 = deadline - min_time_remaining
    if time.monotonic() < t4:
        a4, sc4 = _newton_polish(best_a, t4, min_time_remaining=min_time_remaining, verbose=False)
        if sc4 < best_sc:
            best_sc = sc4
            best_a = a4
            if verbose:
                print(f"      After Newton 2: {best_sc:.10f}")

    # Step 5: Peak shave
    t5 = deadline - min_time_remaining
    if time.monotonic() < t5:
        a5, sc5 = _peak_shave(best_a, t5, min_time_remaining=min_time_remaining)
        if sc5 < best_sc:
            best_sc = sc5
            best_a = a5

    # Final Newton
    t6 = deadline - min_time_remaining
    if time.monotonic() < t6:
        a6, sc6 = _newton_polish(best_a, t6, min_time_remaining=min_time_remaining, verbose=False)
        if sc6 < best_sc:
            best_sc = sc6
            best_a = a6

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining L-BFGS, Newton active-set, and peak shaving.
    Explores multiple initializations and sequence lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = np.inf

    # Try to load parent sequence
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    n_main = 16384

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
        n_main = len(parent)
    else:
        best_seq = _make_profile(n_main, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n={n_main}: score={best_score:.10f}")

    # === Strategy 1: Optimize parent/main sequence ===
    print(f"\n[1] Optimizing main sequence (n={n_main}, score={best_score:.10f})")
    t1_end = min(deadline - 60, t_start + (budget_s - 5) * 0.40)
    a1, sc1 = _optimize_sequence(best_seq, t1_end, min_time_remaining=60, verbose=True)
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  Result: {best_score:.10f}")
    else:
        print(f"  No improvement (still {best_score:.10f})")

    # === Strategy 2: Different initializations at same n ===
    print(f"\n[2] Alternative initializations (n={n_main})")
    profiles = ["chirp", "periodic", "random_nonneg", "quadratic", "bimodal"]
    remaining_budget = deadline - time.monotonic()
    time_per_profile = max(30, remaining_budget * 0.8 / len(profiles))

    for prof in profiles:
        if time.monotonic() > deadline - 30:
            break
        a_p = _make_profile(n_main, prof)
        sc_p = _score(a_p)
        print(f"  {prof}: initial score={sc_p:.10f}")

        if sc_p < best_score + 0.3:  # Only optimize if potentially competitive
            t_p_end = min(deadline - 30, time.monotonic() + time_per_profile)
            a_p_opt, sc_p_opt = _optimize_sequence(a_p, t_p_end, min_time_remaining=30, verbose=False)
            if sc_p_opt < best_score:
                best_score = sc_p_opt
                best_seq = a_p_opt.copy()
                print(f"    -> improved to {best_score:.10f}!")
            else:
                print(f"    -> best was {sc_p_opt:.10f} (no improvement)")

    # === Strategy 3: Different n ===
    print(f"\n[3] Different sequence lengths")
    n_options = [8192, 12288, 20480, 24576]
    n_options = [x for x in n_options if x != n_main]
    for n_try in n_options:
        if time.monotonic() > deadline - 40:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        print(f"  n={n_try}: initial score={sc_r:.10f}")

        t_r_end = min(deadline - 25, time.monotonic() + 80)
        if time.monotonic() < t_r_end:
            a_r_opt, sc_r_opt = _optimize_sequence(a_r, t_r_end, min_time_remaining=25, verbose=False)
            if sc_r_opt < best_score:
                best_score = sc_r_opt
                best_seq = a_r_opt.copy()
                print(f"    -> improved to {best_score:.10f}!")

    # === Strategy 4: Final intensive Newton polish on best ===
    print(f"\n[4] Final intensive polish")
    if time.monotonic() < deadline - 5:
        a_f, sc_f = _newton_polish(best_seq, deadline - 2, min_time_remaining=2, verbose=True)
        if sc_f < best_score:
            best_score = sc_f
            best_seq = a_f.copy()

        # One more L-BFGS round
        if time.monotonic() < deadline - 3:
            a_f2, sc_f2 = _run_lbfgs(best_seq, deadline - 1, min_time_remaining=1, verbose=False)
            if sc_f2 < best_score:
                best_score = sc_f2
                best_seq = a_f2.copy()

    # Final validation
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"\n{'='*60}")
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    return [float(x) for x in final]
