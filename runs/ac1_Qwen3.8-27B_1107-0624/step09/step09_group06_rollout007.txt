```python
def _spectral_init(n, seed=0):
    rng = np.random.default_rng(seed)
    a = np.abs(rng.standard_normal(n))
    a = a / a.sum()
    
    for iter in range(100):
        A = np.fft.fft(a)
        # Flatten the magnitude (except DC)
        mag = np.abs(A)
        phase = np.angle(A)
        # Set all non-DC magnitudes to the average
        avg_mag = (mag.sum() - mag[0]) / (n - 1)
        new_mag = np.full(n, avg_mag)
        new_mag[0] = 1.0  # Keep DC
        A_new = new_mag * np.exp(1j * phase)
        a = np.real(np.fft.ifft(A_new))
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        else:
            break
    
    return a
```
