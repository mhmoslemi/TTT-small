            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            lr = min(0.02, lr * 1.05)
        else:
            no_improve += 1
            if no_improve > 50:
                lr *= 0.9
            if no_improve > 200:
                a = best_a.copy()
                lr = 0.01
                K = max(3, K // 2)
                no_improve = 0
                if no_improve > 500:
                    break
        
        # Get top K peaks
        top_k = np.argsort(c)[::-1][:K]
        
        # Compute subgradient: average of gradients of top-K peaks
        w_S = np.zeros(2 * n - 1)
        w_S[top_k] = 2.0 / K
        a_rev = a[::-1]

        grad = grad - np.mean(grad)  # stay on simplex

            break

        alpha = lr / (gnorm + 1e-15)
        improved = False

            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:

                continue

                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            if no_improve % 30 == 0:
                K = min(K + 1, 30)
    
    return best_a, best_sc

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * t
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.25) ** 2) + 0.6 * np.exp(-15.0 * (t - 0.75) ** 2) + 0.02
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.2 * np.sin(2.0 * np.pi * 5.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 2.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian_asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.4) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _asymmetric_perturb(a, rng, strength=0.05):
    n = len(a)

    perturbation = perturbation - perturbation.mean()
    z = np.log(np.maximum(a, 1e-30))
    z = z + strength * perturbation
    z = z - z.mean()
    return _softmax(z)

"""
Optimized solver for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses multi-scale smoothed L-BFGS with aggressive final polish,
exact subgradient peak cycling for equalization, and diverse
profile exploration to escape local minima. Allocated more time
to the final refinement phase where small improvements come from.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _SmoothedProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def f(self, z):
        return _smoobj(z, self.beta, self.n)

    def g(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0, n_beta_levels=None):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005) * beta_scale

    all_betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625,
        0.03, 0.015, 0.0075, 0.004,
        0.002, 0.001, 0.0005, 0.0002, 0.0001,
        0.00005, 0.00002, 0.00001, 0.000005, 0.000002
    ])
    if n_beta_levels is not None:
        betas = all_betas[:n_beta_levels]
    else:
        betas = all_betas

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        beta_frac = beta / base_beta
        if beta_frac < 0.005:
            est_time_per_iter = 0.004 * max(1.0, n / 2048.0)
            max_iter = min(12000, max(200, int(remaining / (est_time_per_iter * 20))))
        elif beta_frac < 0.05:
            est_time_per_iter = 0.0035 * max(1.0, n / 2048.0)
            max_iter = min(8000, max(150, int(remaining / (est_time_per_iter * 15))))
        else:
            est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
            max_iter = min(6000, max(100, int(remaining / (est_time_per_iter * 12))))

        prob = _SmoothedProblem(beta, n)
        try:
            result = minimize(
                prob.f, z, jac=prob.g, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 150,
                    'maxls': 80,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _peak_cycling(a_init, deadline, min_time_remaining=10.0):
    """Cycling subgradient method: targets peaks in round-robin fashion."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    no_improve = 0
    base_lr = 0.005
    K = 8
    cycle_offset = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                a = best_a.copy()
                no_improve = 0
                K = max(3, K // 2)
                if no_improve > 300:
                    break

        # Get top K peaks, rotate which subset we target
        top_k = np.argsort(c)[::-1][:K]
        if K > 2:
            target = top_k[cycle_offset % K: (cycle_offset + max(2, K // 2)) % K]
            if len(target) == 0:
                target = top_k[:2]
        else:
            target = top_k
        cycle_offset += 1

        # Compute gradient for targeted peaks
        w_S = np.zeros(2 * n - 1)
        w_S[target] = 2.0 / len(target)
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Backtracking line search
        lr = base_lr / (1.0 + 0.1 * no_improve)
        alpha = lr / (gnorm + 1e-15)
        improved = False
        for _ in range(30):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            base_lr *= 0.97

    return best_a, best_sc


def _proj_simplex(v):
    """Euclidean projection onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "asym_ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.6 * t
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-25.0 * (t - 0.3) ** 2) + 0.7 * np.exp(-20.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "gauss_asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.35) ** 2) + 0.03
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.25 * np.sin(2.0 * np.pi * 3.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, rng, strength=0.05):
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n
    ramp = 0.4 * (t - 0.5)
    noise = 0.6 * rng.standard_normal(n)
    pert = ramp + noise
    pert = pert - pert.mean()
    z = np.log(np.maximum(a, 1e-30))
    z = z + strength * pert
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase optimizer optimized for final polish quality."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "asym_ramp")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, asym_ramp n={n_parent}, score={best_score:.10f}")

    n = len(best_seq)

    # === Phase 1: Quick L-BFGS warmup ===
    print(f"\n[P1] L-BFGS warmup (t={time.monotonic()-t_start:.1f}s)")
    a_p1 = _run_lbfgs(best_seq, deadline, min_time_remaining=800.0, beta_scale=1.0,
                      n_beta_levels=8)
    sc_p1 = _score(a_p1)
    if sc_p1 < best_score:
        best_seq = a_p1.copy()
        best_score = sc_p1
        print(f"  NEW BEST: {best_score:.10f} [warmup]")
    else:
        print(f"  Warmup: {sc_p1:.10f} (no improvement)")

    # === Phase 2: Diverse profile exploration ===
    print(f"\n[P2] Profile exploration (t={time.monotonic()-t_start:.1f}s)")
    profiles = ["asym_ramp", "bimodal", "gauss_asym", "tapered", "sine_mod"]
    for trial, prof in enumerate(profiles[:4]):
        if time.monotonic() > deadline - 500:
            break
        a_init = _make_profile(n, prof)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=500.0, beta_scale=0.7,
                           n_beta_levels=10)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [profile={prof}]")
        else:
            print(f"  Profile {prof}: {sc_opt:.10f}")

    # === Phase 3: Perturbation exploration ===
    print(f"\n[P3] Perturbation exploration (t={time.monotonic()-t_start:.1f}s)")
    for trial in range(3):
        if time.monotonic() > deadline - 400:
            break
        strength = 0.04 * (trial + 1)
        a_pert = _perturb(best_seq, rng, strength=strength)
        a_opt = _run_lbfgs(a_pert, deadline, min_time_remaining=400.0, beta_scale=0.6,
                           n_beta_levels=10)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [perturb s={strength:.3f}]")
        else:
            print(f"  Perturb {trial}: {sc_opt:.10f}")

    # === Phase 4: Peak cycling equalization ===
    print(f"\n[P4] Peak cycling (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        pc_deadline = min(deadline - 300, time.monotonic() + 180)
        a_pc, sc_pc = _peak_cycling(best_seq, pc_deadline, min_time_remaining=30.0)
        if sc_pc < best_score:
            best_seq = a_pc.copy()
            best_score = sc_pc
            print(f"  NEW BEST: {best_score:.10f} [peak_cycling]")
        else:
            print(f"  Peak cycling: {sc_pc:.10f}")

    # === Phase 5: Full L-BFGS with many beta levels ===
    print(f"\n[P5] Full L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a_full = _run_lbfgs(best_seq, deadline, min_time_remaining=200.0, beta_scale=0.5,
                            n_beta_levels=19)
        sc_full = _score(a_full)
        if sc_full < best_score:
            best_seq = a_full.copy()
            best_score = sc_full
            print(f"  NEW BEST: {best_score:.10f} [full_lbfgs]")
        else:
            print(f"  Full L-BFGS: {sc_full:.10f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([n, 32768, 24576]), reverse=True)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 120:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init={sc_init:.10f}")
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=120.0, beta_scale=0.5,
                           n_beta_levels=14)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try})")
        else:
            print(f"  n={n_try}: {sc_opt:.10f}")

    # === Phase 7: Final peak cycling ===
    print(f"\n[P7] Final peak cycling (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        pc2_deadline = min(deadline - 80, time.monotonic() + 60)
        a_pc2, sc_pc2 = _peak_cycling(best_seq, pc2_deadline, min_time_remaining=10.0)
        if sc_pc2 < best_score:
            best_seq = a_pc2.copy()
            best_score = sc_pc2
            print(f"  NEW BEST: {best_score:.10f} [peak_cycling2]")

    # === Phase 8: Ultra-fine L-BFGS (the critical final polish) ===
    print(f"\n[P8] Ultra-fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 15 - time.monotonic()
    if remaining > 20:
        a_fin = _run_lbfgs(best_seq, deadline, min_time_remaining=5.0, beta_scale=0.1,
                           n_beta_levels=19)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [ultra_fine]")

    # === Phase 9: Final tiny peak cycling ===
    if time.monotonic() < deadline - 5:
        a_fin2, sc_fin2 = _peak_cycling(best_seq, deadline - 3, min_time_remaining=2.0)
        if sc_fin2 < best_score:
            best_seq = a_fin2.copy()
            best_score = sc_fin2
            print(f"  NEW BEST: {best_score:.10f} [final_pc]")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
