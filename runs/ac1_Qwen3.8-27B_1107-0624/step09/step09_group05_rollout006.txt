def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    return e / s

        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _top_k_value(c, k):
    """Average of top k values in c."""
    if k >= len(c):
        return float(c.mean())
    # Use partition for efficiency
    if k <= 20:
        idx = np.argpartition(-c, k)[:k]
        return float(c[idx].mean())
    else:
        idx = np.argpartition(-c, k - 1)[:k]
        return float(c[idx].mean())


def _subgrad_at_max(c, a, n, k=1):
    """Subgradient of top-k average of conv(a,a) w.r.t. a.
    
    Returns gradient of (1/k)*sum_{i in top-k} conv[k_i]
    """
    if k == 1:
        peak_idx = int(np.argmax(c))
        w = np.zeros(len(c))
        w[peak_idx] = 1.0
    else:
        idx = np.argpartition(-c, min(k, len(c) - 1))[:k]
        w = np.zeros(len(c))
        w[idx] = 1.0 / k

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_soft_structured(a, beta, n):
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


class _SoftMaxProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        return 2.0 * self.n * _soft_max_value(c, self.beta)

    def grad(self, z):
        a = _softmax(z)
        grad_a = _grad_soft_structured(a, self.beta, self.n)
        W = float(np.dot(grad_a, a))
        return a * (grad_a - W)


class _TopKProb:
    def __init__(self, k, n):
        self.k = k
        self.n = n

    def __call__(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        return 2.0 * self.n * _top_k_value(c, self.k)

    def grad(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad_a = _subgrad_at_max(c, a, self.n, self.k)
        W = float(np.dot(grad_a, a))
        return a * (grad_a - W)


def _run_lbfgs_smooth(a_init, deadline, min_time_remaining=30.0, verbose=False):
    """L-BFGS with aggressive adaptive beta schedule."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < c0_max * 0.002:
        gap = c0_max * 0.002

    # Very aggressive beta schedule with more fine-grained steps
    multipliers = np.array([
        1.0, 0.7, 0.5, 0.35, 0.25, 0.175, 0.125,
        0.0875, 0.0625, 0.04375, 0.03125, 0.021875,
        0.015625, 0.0109, 0.0078, 0.0055,
        0.0039, 0.0027, 0.0019, 0.0013,
        0.0009, 0.0006, 0.0004, 0.0003,
        0.0002, 0.0001, 0.00007, 0.00005,
        0.00003, 0.00002, 0.00001
    ])
    betas = gap * multipliers

    size_factor = max(1.0, n / 8192.0)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break

        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_iter_time = 0.001 * size_factor
        # More iterations for smaller beta (where we need to converge)
        if beta < gap * 0.01:
            max_iter = min(20000, max(500, int(remaining / (est_iter_time * 10))))
        else:
            max_iter = min(10000, max(200, int(remaining / (est_iter_time * 15))))

        prob = _SoftMaxProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 200,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"    beta={beta:.2e} ({i+1}/{len(betas)}): score={best_sc:.10f}, nit={result.nit}")
        except Exception as e:
            if verbose:
                print(f"    beta={beta:.2e}: error {e}")

    return best_a, best_sc


def _run_lbfgs_topk(a_init, deadline, min_time_remaining=20.0, k=3):
    """L-BFGS with top-k average objective."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    prob = _TopKProb(k, n)
    size_factor = max(1.0, n / 8192.0)

    # Multiple restarts with different initializations
    for attempt in range(3):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        max_iter = min(20000, max(200, int(remaining / (0.001 * size_factor * 10))))
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 200,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return best_a, best_sc


def _peak_shaving(a_init, deadline, min_time_remaining=20.0, verbose=False):
    """Iterative subgradient peak shaving with Polyak step size."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lb = 2.0 * n / (2.0 * n - 1.0)
    no_improve = 0
    max_no_improve = 500
    step_decay = 0.995

    for iteration in range(100000):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        # Use top-3 subgradient for better signal
        k = min(3, len(c))
        grad = _subgrad_at_max(c, a, n, k)

        # Center the gradient (for simplex projection)
        grad_c = grad - np.mean(grad)
        g_norm_sq = float(np.dot(grad_c, grad_c))
        if g_norm_sq < 1e-25:
            break

        # Polyak step
        gap_sc = current_sc - lb
        if gap_sc < 1e-12:
            break
        step = min(0.5 * gap_sc / g_norm_sq, 0.05)

        a_new = a - step * grad_c
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Accept or reject
        sc_new = _score(a_new)
        if sc_new < current_sc:
            a = a_new
        else:
            # Try smaller step
            for trial_mult in [0.5, 0.25, 0.1]:
                a_try = a - step * trial_mult * grad_c
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc:
                    a = a_try
                    break
            else:
                # No improvement, decay step
                step *= 0.9
                if step < 1e-15:
                    break

        # Periodic decay
        if iteration % 200 == 0:
            step *= step_decay

    return best_a, best_sc


def _active_set_balance(a_init, deadline, min_time_remaining=20.0):
    """Balance active peaks using averaged subgradient with Newton-like steps."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    no_improve = 0
    max_no_improve = 200

    for iteration in range(20000):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        # Identify active set: peaks within 2% of max
        tol = 0.02 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        # Cap active set size
        if n_active > 50:
            idx = np.argpartition(-c, 49)[:50]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 50

        # Compute average gradient over active set
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]

        # Center for simplex
        grad_c = grad_avg - np.mean(grad_avg)
        g_norm_sq = float(np.dot(grad_c, grad_c))
        if g_norm_sq < 1e-25:
            break

        # Step with backtracking
        step = 0.02
        improved = False
        for _ in range(10):
            a_new = a - step * grad_c
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                step *= 0.5
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                a = a_new
                improved = True
                break
            step *= 0.5

        if not improved:
            # Try with slightly different tolerance
            tol *= 0.5
            if tol < 1e-6 * c_max:
                break

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n, dtype=np.float64)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 7.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n)
             + 0.05 * np.cos(2.0 * np.pi * 53.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-50.0 * (t - 0.25) ** 2) + np.exp(-50.0 * (t - 0.75) ** 2) + 0.01
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.5 * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) / 0.15) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma=0.01, seed=None):
    """Add random perturbation and renormalize."""
    rng = np.random.default_rng(seed)
    noise = rng.normal(0, sigma, len(a))
    a_p = a + noise
    a_p = np.maximum(a_p, 0.0)
    s = a_p.sum()
    if s < 1e-10:
        return a.copy()
    return a_p / s


t_start_global = 0.0


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining smooth-max L-BFGS, top-k L-BFGS,
    peak shaving, active set balancing, and perturbation restarts.
    """
    global t_start_global
    t_start_global = time.monotonic()
    deadline = t_start_global + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = _score(best_seq)
        print(f"Parent: n={len(best_seq)}, score={best_score:.10f}")
    else:
        best_seq = _make_profile(16384, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n=16384: score={best_score:.10f}")

    n_main = len(best_seq)

    # === Phase 1: Main L-BFGS with smooth max ===
    print(f"\n[P1] Smooth-max L-BFGS (n={n_main}, t={time.monotonic()-t_start_global:.0f}s)")
    p1_end = min(deadline - 200, t_start_global + 500)
    a1, sc1 = _run_lbfgs_smooth(best_seq, p1_end, min_time_remaining=100, verbose=True)
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  P1: {best_score:.10f}")

    # === Phase 2: Top-k L-BFGS ===
    print(f"\n[P2] Top-k L-BFGS (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p2_end = min(deadline - 150, time.monotonic() + 250)
        for k in [2, 3, 5]:
            if time.monotonic() + 30 > deadline:
                break
            a2, sc2 = _run_lbfgs_topk(best_seq, p2_end, min_time_remaining=50, k=k)
            if sc2 < best_score:
                best_score = sc2
                best_seq = a2.copy()
                print(f"  Top-k={k}: {best_score:.10f}")

    # === Phase 3: Peak shaving ===
    print(f"\n[P3] Peak shaving (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 20 < deadline:
        p3_end = min(deadline - 100, time.monotonic() + 200)
        a3, sc3 = _peak_shaving(best_seq, p3_end, min_time_remaining=50)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3: {best_score:.10f}")

    # === Phase 4: Active set balancing ===
    print(f"\n[P4] Active set balance (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 20 < deadline:
        p4_end = min(deadline - 80, time.monotonic() + 150)
        a4, sc4 = _active_set_balance(best_seq, p4_end, min_time_remaining=30)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4: {best_score:.10f}")

    # === Phase 5: L-BFGS round 2 (after peak shaving) ===
    print(f"\n[P5] L-BFGS round 2 (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 20 < deadline:
        p5_end = min(deadline - 60, time.monotonic() + 200)
        a5, sc5 = _run_lbfgs_smooth(best_seq, p5_end, min_time_remaining=30, verbose=True)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  P5: {best_score:.10f}")

    # === Phase 6: Perturbation restarts ===
    print(f"\n[P6] Perturbation restarts (t={time.monotonic()-t_start_global:.0f}s)")
    rng = np.random.default_rng(seed)
    n_restarts = 5
    for i in range(n_restarts):
        if time.monotonic() + 80 > deadline:
            break
        sigma = 0.005 * (0.5 ** i)
        a_p = _perturb(best_seq, sigma, seed=seed + i * 100 + int(rng.integers(0, 10000)))
        p6_end = min(deadline - 40, time.monotonic() + 60)
        a6, sc6 = _run_lbfgs_smooth(a_p, p6_end, min_time_remaining=20, verbose=False)
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  Restart {i} (sigma={sigma:.4f}): {best_score:.10f}")

    # === Phase 7: n exploration ===
    print(f"\n[P7] Multi-n exploration (t={time.monotonic()-t_start_global:.0f}s)")
    n_options = [8192, 12288, 16384, 20480, 24576, 32768]
    n_options = [x for x in n_options if x != n_main]
    for n_try in n_options:
        if time.monotonic() + 70 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if time.monotonic() + 50 > deadline:
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  n={n_try}: {best_score:.10f} (resample only)")
            continue
        p7_end = min(deadline - 30, time.monotonic() + 60)
        a7, sc7 = _run_lbfgs_smooth(a_r, p7_end, min_time_remaining=15, verbose=False)
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()
            print(f"  n={n_try}: {best_score:.10f}")

    # === Phase 8: Final polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 15 < deadline:
        p8_end = min(deadline - 3, time.monotonic() + 80)
        a8, sc8 = _run_lbfgs_smooth(best_seq, p8_end, min_time_remaining=3, verbose=False)
        if sc8 < best_score:
            best_score = sc8
            best_seq = a8.copy()

        if time.monotonic() + 10 < deadline:
            p8b_end = min(deadline - 2, time.monotonic() + 30)
            a8b, sc8b = _peak_shaving(best_seq, p8b_end, min_time_remaining=2)
            if sc8b < best_score:
                best_score = sc8b
                best_seq = a8b.copy()

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start_global:.1f}s")
    return [float(x) for x in final]
