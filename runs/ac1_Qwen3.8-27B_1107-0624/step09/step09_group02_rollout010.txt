    n = len(a)
    N = next_pow2(2*n - 1)
    for i in range(n_iters):
        a_padded = np.zeros(N)
        a_padded[:n] = a
        F = np.fft.rfft(a_padded)
        if target is None:
            target_mag = np.ones(len(F)) * np.sqrt(n / (2*n - 1))  # normalize
        else:
            target_mag = target
        # Keep phase, set magnitude
        phase = np.angle(F)
        F_new = target_mag * np.exp(1j * phase)
        # Make sure F_new[0] is real and positive (DC component)
        F_new[0] = np.abs(F_new[0])
        a_new = np.fft.irfft(F_new, N)[:n]
        # Project to non-negative
        a = np.maximum(a_new, 0)
        # Normalize
        s = a.sum()
        if s > 0:
            a = a / s
    return a

Actually, let me reconsider the problem from scratch.

The key mathematical structure:
- We want to minimize max_k conv(a,a)[k] subject to a >= 0, sum(a) = 1.
- This is a convex optimization problem! (The max of linear functions is convex, and the constraint set is convex.)

Wait, is it convex? max_k conv[a,a][k] = max_k sum_j a[j]*a[k-j]. Each term sum_j a[j]*a[k-j] is a quadratic form in a. The max of quadratic forms is not generally convex. So it's NOT a convex problem.

However, we can use the epigraph formulation:
minimize t
subject to sum_j a[j]*a[k-j] <= t for all k
         a >= 0
         sum(a) = 1

This is a quadratically constrained program (QCP), which is non-convex in general. But for our specific structure, it might be tractable.

Actually, each constraint sum_j a[j]*a[k-j] <= t can be written as a^T Q_k a <= t, where Q_k is a symmetric matrix. The matrix Q_k has:
- Q_k[j, k-j] = 1/2 and Q_k[k-j, j] = 1/2 for j != k-j
- Q_k[j, j] = 1/2 if 2j = k (diagonal)

Wait, let me be more careful. conv(a,a)[k] = sum_j a[j] * a[k-j]. If k-j = j (i.e., j = k/2), the term is a[j]^2. Otherwise, the term a[j]*a[k-j] appears twice (once with j and once with k-j).

So conv(a,a)[k] = (1/2) * a^T Q_k a where Q_k is defined appropriately.

This is a max of quadratic forms, which is a non-convex optimization problem. However, we can use SDP relaxations or other techniques.

"""
Multi-strategy optimizer for minimum peak autocorrelation sequences.
Combines spectral balancing (Gerchberg-Saxton), L-BFGS with log-sum-exp,
simulated annealing, and multi-n exploration.
Key new idea: spectral balancing directly targets flat autocorrelation spectrum,
complementing the previous optimization approaches.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * t
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t) + 0.1
        return a / a.sum()
    elif kind == "asym2":
        t = np.linspace(0, 1, n)
        a = np.exp(3.0 * (t - 1)) + 0.1
        return a / a.sum()
    elif kind == "w_shape":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(4 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(42)
        a = rng.random(n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _spectral_balance(a_init, deadline, min_time=30, alpha_schedule=None):
    """Spectral balancing: alternate between spectral flattening and non-negativity."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    N = _next_pow2(2 * n - 1)
    if alpha_schedule is None:
        # Start aggressive (very flat target), gradually reduce
        alpha_schedule = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]
    
    it = 0
    for alpha in alpha_schedule:
        while time.monotonic() + min_time < deadline:
            it += 1
            # Compute DFT
            a_pad = np.zeros(N)
            a_pad[:n] = a
            F = np.fft.rfft(a_pad)
            mag = np.abs(F)
            dc_val = mag[0]
            
            # Target: compressed magnitude
            if mag.mean() < 1e-30:
                break
            mag_ratio = mag / mag.mean()
            new_mag = mag.mean() * np.power(mag_ratio, alpha)
            # Preserve DC
            new_mag[0] = dc_val
            
            phase = np.angle(F)
            F_new = new_mag * np.exp(1j * phase)
            F_new[0] = dc_val  # keep DC real and positive
            
            a_new = np.fft.irfft(F_new, N)[:n]
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            
            sc = _score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
            
            a = a_new
            
            # Check convergence
            if it % 100 == 0:
                c = _full_convolve(a, a)
                spread = c.max() - c.min()
                if spread < 1e-12:
                    break
    
    return best_a, best_sc

                return 2.0 * n * (c_max_local + beta * np.log(s_e))

            def grad_func(z_local):

                c_local = _full_convolve(a_local, a_local)

                h = wc[n - 1:2 * n - 1]
                W_val = float(np.dot(w, c_local))
                return 4.0 * n * a_local * (h - W_val)

            result = minimize(

        z = z + sigma * rng.standard_normal(n)

        if sc_new < current_sc:

def _hvp_pv(v, p, n):
    """Hessian-vector product for the smoothed problem."""
    b = v[::-1]
    conv_len = len(b) + len(p) - 1
    N = _next_pow2(conv_len)
    Fb = np.fft.rfft(b, N)
    Fp = np.fft.rfft(p, N)
    conv = np.fft.irfft(Fb * Fp, N)[:conv_len]
    result = 2.0 * conv[n - 1:2 * n - 1].copy()
    for k in range(0, 2 * n - 1, 2):
        m = k // 2
        if m < n and k < len(p):
            result[m] -= v[m] * p[k]
    return result


def _grad_smoothed_fast(a, p, n):
    return _hvp_pv(a, p, n)

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

            best_sc = sc
            best_a = a_new.copy()

        a = a_new

        if it % 200 == 199:
            alpha = max(alpha * 0.97, 0.01)

    return best_a, best_sc


def _spectral_balance_shift(a_init, deadline, min_time):
    """Spectral balancing with shift-to-nonnegative instead of hard projection."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    N = _next_pow2(2 * n - 1)
    alpha = 0.05

    for it in range(50000):
        if time.monotonic() + min_time > deadline:
            break

        a_pad = np.zeros(N)
        a_pad[:n] = a
        F = np.fft.rfft(a_pad)
        mag = np.abs(F)
        dc_val = mag[0]

        if mag.mean() < 1e-30:
            break

        log_mag = np.log(mag + 1e-30)
        target_log = np.mean(log_mag)
        new_log_mag = (1.0 - alpha) * log_mag + alpha * target_log
        new_mag = np.exp(new_log_mag)
        new_mag[0] = dc_val

        phase = np.angle(F)
        F_new = new_mag * np.exp(1j * phase)
        F_new[0] = dc_val

        a_new = np.fft.irfft(F_new, N)[:n]

        min_val = a_new.min()
        if min_val < 0:
            a_new = a_new - min_val + 1e-10
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

        if it % 300 == 299:
            alpha = max(alpha * 0.98, 0.005)

    return best_a, best_sc


def _lbfgs_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s_e = e.sum()
    if s_e < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s_e))


def _lbfgs_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s_e = e.sum()
    if s_e < 1e-300:
        w = np.zeros_like(c)

        w = e / s_e
