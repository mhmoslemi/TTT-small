import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    fb = np.fft.rfft(b, N)

    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_gradient(a, p, n):
    """
    Compute the gradient g[i] = sum_k p[k] * 2*a[k-i] - a[i]*p[2i]
    Using FFT for the convolution part.
    """

    a_rev = a[::-1]
    ha = np.fft.rfft(a_rev, 2*n-1)
    hp = np.fft.rfft(p, 2*n-1)
    h = np.fft.irfft(ha * hp, 2*n-1)[:2*n-1]
    h = h[n-1:2*n-1]

    diag = np.zeros(n)
    valid = 2*np.arange(n) < 2*n-1
    diag[valid] = a[valid] * p[2*np.arange(n)[valid]]
    
    g = 2.0 * h - diag
    return g

    s = a.sum()

    v = np.zeros(n)

        v = beta2 * v + (1 - beta2) * grad_z**2

        z = z - z.mean()

                best_a = a_check.copy()
                no_improve = 0
                if verbose and t % 500 == 0:

            if no_improve > 100:
                tau = max(tau_min, tau * 0.9)

                z = z - z.mean()

    sc_final = _score(a_final)

        best_a = a_final.copy()

def _kkt_newton(a_init, deadline, max_rounds=200, k_schedule=None):
    """
    KKT-based Newton method with active set.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0
    
    if k_schedule is None:
        k_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]
    
    N = _next_pow2(2 * n - 1)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        K = min(k_schedule[rnd % len(k_schedule)], 2 * n - 1)
        idx = np.argpartition(c, -K)[-K:]
        W = np.zeros(2 * n - 1)
        W[idx] = 1.0 / K
        
        # Gradient: g[i] = sum_k W[k] * (2*a[k-i] - a[i]*delta_{k,2i})
        a_rev = a[::-1]
        fa = np.fft.rfft(a_rev, N)
        fw = np.fft.rfft(W, N)
        h_full = np.fft.irfft(fa * fw, N)[:2 * n - 1]
        h = h_full[n - 1:2 * n - 1]
        
        diag_idx = 2 * np.arange(n)
        valid = diag_idx < (2 * n - 1)
        diag = np.zeros(n)
        diag[valid] = a[valid] * W[diag_idx[valid]]
        
        g = 2.0 * h - diag
        
        # Project gradient (for simplex)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break
        
        # Newton step via CG on the Hessian
        # Hessian of the smooth approximation at the active set:
        # H[i,j] = sum_k W[k] * (2*delta_{i+j,k} - delta_{i,j}*delta_{k,2i})
        # For the off-diagonal: H[i,j] = 2*W[i+j] (if i+j < 2n-1)
        # This is a Hankel-like matrix
        
        # Use CG to solve H @ dx = g_t
        x = np.zeros(n)
        r = g_t.copy()  # We want H @ x = -g_t, so r = -g_t initially... 
        # Actually we want dx = -H^{-1} g_t, so solve H @ x = g_t, then dx = -x
        
        # Let me use a simpler approach: preconditioned gradient step
        # The Hessian is approximately 2 * H where H[i,j] = W[i+j]
        # A good preconditioner is the inverse of this
        
        # For simplicity, use subgradient step with backtracking
        alpha = 1.0
        step_done = False
        for ls in range(40):
            a_try = a - alpha * g_t / gn  # normalized gradient step
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max - 1e-14:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-15:
                break
        
        if not step_done:
            # Try smaller steps
            for trial_alpha in [0.01, 0.005, 0.001, 0.0005]:
                a_try = a - trial_alpha * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-14:
                    a = a_try
                    step_done = True
                    break
        
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not step_done and no_improve > 10:
            break
    
    return best_a, best_sc


def _spectral_flatten(a_init, n_iter=100, strength=0.1):
    """
    Spectral flattening: push the DFT towards constant magnitude.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    
    N = _next_pow2(2 * n - 1)
    
    for i in range(n_iter):
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: make mag as flat as possible, but A[0] = sum(a) = 1 is fixed
        # Blend mag towards a flat target
        avg_mag = mag[1:].mean()
        target_mag = np.full_like(mag, avg_mag)
        target_mag[0] = mag[0]  # keep DC component
        
        # Blend
        new_mag = (1 - strength) * mag + strength * target_mag
        new_mag[0] = mag[0]
        
        # Reconstruct
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project onto non-negative
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
    
    return a


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.5) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "rim":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * (np.sin(np.pi * t) ** 2 + 0.5 * (1 - 2*t)**4)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))
