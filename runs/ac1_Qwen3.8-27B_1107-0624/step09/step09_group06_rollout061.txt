Continuation FISTA + L-BFGS optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses proper FISTA with backtracking on the simplex, followed by aggressive
L-BFGS continuation in log-space with geometrically decreasing smoothing.
Multiple structured initializations and n values are explored.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _smoothed_obj(a, beta, n):
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smoothed_grad_a(a, beta, n):
    """Compute gradient of smoothed objective w.r.t. a."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        # Degenerate: only max matters
        idx = np.argmax(c)
        return _conv_grad_single(a, idx) * 2.0 * n  # approximate
    w = e / s
    # grad = (4n/beta) * cross_corr(w, a)[:n]
    N = _next_pow2(3 * n)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    xcorr = np.fft.irfft(W * np.conj(A), N)
    result = xcorr[:n]
    return (4.0 * n / beta) * result


def _conv_grad_single(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a: 2*a[k-j] for valid j."""
    n = len(a)
    grad = np.zeros(n)
    # grad[i] = 2*a[k-i] if 0 <= k-i < n
    i_lo = max(0, k - n + 1)
    i_hi = min(n - 1, k)
    if i_lo <= i_hi:
        grad[i_lo:i_hi+1] = 2.0 * a[k - np.arange(i_lo, i_hi + 1)]
    return grad


def _fista_smoothed(a_init, beta, n, max_iter=50000, tol=1e-12, verbose=False):
    """FISTA for the smoothed problem on the simplex."""
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    
    x = a.copy()
    x_prev = a.copy()
    t = 1.0
    L = None  # Will be estimated via backtracking
    best_a = a.copy()
    best_f = _smoothed_obj(a, beta, n)
    
    for it in range(max_iter):
        # FISTA extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        if it > 0:
            y = x + (t - 1.0) / t_new * (x - x_prev)
            y = np.maximum(y, 0.0)
        else:
            y = x.copy()
        
        # Backtracking line search
        if L is None:
            L = 1.0
        grad_y = _smoothed_grad_a(y, beta, n)
        
        # Projected gradient step with backtracking
        step = 1.0 / L
        z = y - step * grad_y
        x_new = _project_simplex(z)
        
        # Check decrease (Armijo-like)
        f_y = _smoothed_obj(y, beta, n)
        f_x = _smoothed_obj(x_new, beta, n)
        
        # Update L based on gradient change
        if it > 0:
            grad_prev = _smoothed_grad_a(x_prev, beta, n)
            dx = np.linalg.norm(x - x_prev)
            dg = np.linalg.norm(grad_y - grad_prev)
            if dx > 1e-15:
                L_candidate = 1.5 * dg / dx
                L = max(L, L_candidate)
                step = 1.0 / L
                z = y - step * grad_y
                x_new = _project_simplex(z)
                f_x = _smoothed_obj(x_new, beta, n)
        
        x_prev = x
        x = x_new
        t = t_new
        
        # Track best
        f = _smoothed_obj(x, beta, n)
        if f < best_f - 1e-15:
            best_f = f
            best_a = x.copy()
        
        if verbose and it % 10000 == 0:
            print(f"      FISTA it={it}: f={f:.10f}, L={L:.4f}")
        
        # Convergence check
        if np.linalg.norm(x - x_prev) < tol * max(1.0, np.linalg.norm(x)):
            break
    
    return best_a, best_f


def _lbfgs_logspace(a_init, beta, n, max_iter=5000):
    """L-BFGS in log-space for the smoothed problem."""
    # Parameterize: a = softmax(z)
    z0 = np.log(np.maximum(a_init, 1e-30))
    z0 = z0 - z0.mean()  # Center for numerical stability
    
    def objective(z):
        a = _softmax(z)
        return _smoothed_obj(a, beta, n)
    
    def gradient(z):
        a = _softmax(z)
        grad_a = _smoothed_grad_a(a, beta, n)
        # dF/dz = dF/da * da/dz
        # da/dz is the softmax Jacobian, but for the chain rule:
        # dF/dz[i] = sum_j dF/da[j] * da[j]/dz[i]
        # da[j]/dz[i] = a[j]*(delta_{ji} - a[i])
        # dF/dz = a * (grad_a - sum(grad_a * a))
        return a * (grad_a - np.dot(grad_a, a))
    
    try:
        result = minimize(
            objective, z0, jac=gradient, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                     'maxcor': 100, 'maxls': 50}
        )
        a_new = _softmax(result.x)
        return a_new
    except Exception as e:
        print(f"    L-BFGS error: {e}")
        return a_init

def _resample(a, new_n):

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

1. y_k = x_k + (t_{k-1}-1)/t_k * (x_k - x_{k-1})

        t_prev = t
        t = (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev)) / 2.0

        if it > 0:
            y = x + (t_prev - 1.0) / t * (x - x_prev)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
        else:
            y = x.copy()

        grad_y = _smoothed_grad_a(y, beta, n)

        # Backtracking
        step = 1.0 / L
        for _bt in range(20):
            z = y - step * grad_y
            x_new = _project_simplex(z)
            f_new = _smoothed_obj(x_new, beta, n)
            f_y = _smoothed_obj(y, beta, n)
            # Check: f(x_new) <= f(y) + <grad, x_new - y> + L/2 * ||x_new - y||^2
            rhs = f_y + np.dot(grad_y, x_new - y) + 0.5 * L * np.dot(x_new - y, x_new - y)
            if f_new <= rhs + 1e-15:
                break
            L *= 2.0
            step = 1.0 / L

        # Update empirical L
        if it > 0:
            dx = np.linalg.norm(x - x_prev)
            if dx > 1e-10:
                grad_prev = _smoothed_grad_a(x_prev, beta, n)
                dg = np.linalg.norm(grad_y - grad_prev)
                L_emp = 1.2 * dg / dx
                L = max(L * 0.95, L_emp)

        f_x = _smoothed_obj(x_new, beta, n)
        x_prev = x
        x = x_new

        if f_x < best_f - 1e-14:
            best_f = f_x
            best_a = x.copy()

        if verbose and it % 5000 == 0:
            print(f"      FISTA it={it}: f={f_x:.10f}, L={L:.2f}")

        if it > 100 and np.linalg.norm(x - x_prev) < 1e-12 * max(1.0, np.linalg.norm(x)):
            break

    return best_a, best_f


def _lbfgs_logspace(a_init, beta, n, max_iter=5000):
    """L-BFGS in log-space for the smoothed problem."""
    z0 = np.log(np.maximum(a_init, 1e-30))
    z0 = z0 - z0.mean()

    def objective(z):
        a = _softmax(z)
        return _smoothed_obj(a, beta, n)

    def gradient(z):
        a = _softmax(z)
        grad_a = _smoothed_grad_a(a, beta, n)
        return a * (grad_a - np.dot(grad_a, a))

    try:
        result = minimize(
            objective, z0, jac=gradient, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-22, 'gtol': 1e-18,
                     'maxcor': 120, 'maxls': 60}
        )
        a_new = _softmax(result.x)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        return a_new
    except Exception:
        return a_init


def _make_profile(n, kind="uniform", seed=0):
    rng = np.random.default_rng(seed)
    j = np.arange(n)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos3":
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * j / n)
        return np.maximum(a, 1e-10) / np.maximum(a, 1e-10).sum()
    elif kind == "cos7":
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * 7.0 * j / n)
        return np.maximum(a, 1e-10) / np.maximum(a, 1e-10).sum()
    elif kind == "sin5":
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 5.0 * j / n)
        return np.maximum(a, 1e-10) / np.maximum(a, 1e-10).sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        return a / a.sum()
    elif kind == "random_lf":
        freqs = rng.integers(1, 12, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        amps = rng.uniform(0.15, 0.45, size=4)
        a = np.ones(n)
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.cos(2 * np.pi * f * j / n + p)
        return np.maximum(a, 1e-10) / np.maximum(a, 1e-10).sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-25.0 * (t - 0.3) ** 2) + np.exp(-25.0 * (t - 0.7) ** 2) + 0.08
        return a / a.sum()
    elif kind == "flat_taper":
        a = np.ones(n)
        taper = max(1, n // 40)
        for i in range(taper):
            a[i] = (i + 1) / taper
            a[n - 1 - i] = (i + 1) / taper
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _optimize_pipeline(a_init, deadline, min_time_remaining=30.0, label=""):
    """Full optimization pipeline: FISTA warm-start + L-BFGS continuation."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_score = _score(a)

    # Phase A: FISTA warm-start with a few beta levels
    if time.monotonic() < deadline - min_time_remaining - 60:
        fista_betas = [0.5, 0.25, 0.1]
        for beta in fista_betas:
            if time.monotonic() + min_time_remaining > deadline:
                break
            remaining = deadline - min_time_remaining - time.monotonic()
            max_iter = min(15000, max(2000, int(remaining / 3 / 0.002)))
            a_new, _ = _fista_smoothed(a, beta, n, max_iter=max_iter)
            sc = _score(a_new)
            a = a_new
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()

    # Phase B: L-BFGS continuation (the main workhorse)
    betas = [0.05, 0.025, 0.01, 0.005, 0.0025, 0.00125, 0.0006,
             0.0003, 0.00015, 0.00008, 0.00004, 0.00002,
             0.00001, 0.000005, 0.000002, 0.000001]

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 20:
            break
        # Scale iterations: more for smaller beta
        frac = remaining / max(1, len(betas) - i)
        if i < len(betas) // 3:
            max_iter = min(4000, max(500, int(frac * 0.02)))
        elif i < 2 * len(betas) // 3:
            max_iter = min(3000, max(400, int(frac * 0.015)))
        else:
            max_iter = min(5000, max(200, int(frac * 0.012)))

        a_new = _lbfgs_logspace(a, beta, n, max_iter=max_iter)
        sc = _score(a_new)
        a = a_new
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            if label:
                print(f"    [{label}] beta={beta:.8f}: score={best_score:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """Main search: multiple initializations x L-BFGS continuation x multi-n."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent. Uniform n={n_parent}, score={best_score:.10f}")

    # === Main optimization on parent ===
    print(f"\n[Main] Optimizing parent (n={n_parent})")
    main_deadline = deadline - 400.0  # Reserve 400s for exploration
    if time.monotonic() < main_deadline:
        a_main, sc_main = _optimize_pipeline(parent, main_deadline,
                                             min_time_remaining=30.0, label="main")
        if sc_main < best_score:
            best_score = sc_main
            best_seq = a_main.copy()
            print(f"  Main result: {best_score:.10f}")

    # === Alternative initializations ===
    alt_kinds = [("cos3", 0), ("cos7", 1), ("random_lf", 2), ("tapered", 3),
                 ("sin5", 4), ("bimodal", 5)]
    alt_deadline = deadline - 200.0  # Reserve 200s for n-exploration

    for kind, ks in alt_kinds:
        if time.monotonic() > alt_deadline:
            break
        remaining = alt_deadline - time.monotonic()
        if remaining < 80:
            break
        a_init = _make_profile(n_parent, kind, seed=seed + ks)
        sc_init = _score(a_init)
        a_alt, sc_alt = _optimize_pipeline(a_init, alt_deadline,
                                           min_time_remaining=30.0,
                                           label=f"alt_{kind}")
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = a_alt.copy()
            print(f"  Alt {kind}: NEW BEST {best_score:.10f}")

    # === Multi-n exploration ===
    n_candidates = [16384, 32768, 8192, 24576, 49152]
    if n_parent in n_candidates:
        n_candidates.remove(n_parent)
    n_candidates.insert(0, n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - 30 - time.monotonic()
        if remaining < 60:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"\n[Multi-n] n={n_try}: init score={sc_init:.10f}")

        a_n, sc_n = _optimize_pipeline(a_init, deadline - 10,
                                       min_time_remaining=10.0,
                                       label=f"n{n_try}")
        if sc_n < best_score:
            best_score = sc_n
            best_seq = a_n.copy()
            print(f"  Multi-n: NEW BEST {best_score:.10f} (n={n_try})")

    # === Final ultra-fine L-BFGS polish ===
    if time.monotonic() < deadline - 5:
        n = len(best_seq)
        remaining = deadline - 5 - time.monotonic()
        if remaining > 15:
            print(f"\n[Final Polish] remaining={remaining:.1f}s")
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            c_sorted = np.sort(c_best)[::-1]
            gap = mb - c_sorted[min(20, len(c_sorted) - 1)]
            base_beta = max(gap, mb * 0.001)

            final_betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                                 0.03, 0.015, 0.008, 0.004, 0.002])
            for beta in final_betas:
                if time.monotonic() > deadline - 5:
                    break
                max_iter = min(3000, max(100, int(remaining / 0.08)))
                a_fin = _lbfgs_logspace(best_seq, beta, n, max_iter=max_iter)
                sc_fin = _score(a_fin)
                if sc_fin < best_score:
                    best_score = sc_fin
                    best_seq = a_fin
                    print(f"  Polish beta={beta:.6f}: {best_score:.10f}")

    # === Final validation ===
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
        print("WARNING: sum too small, using fallback")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    print(f"Target: 1.5027, Previous best: 1.505661")
    print(f"Improvement: {1.505661 - sc_final:.10f}")

    return [float(x) for x in final]
