def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _ls_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _ls_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _diag_precond_cg(matvec, rhs, diag_h, max_cg=500, tol=1e-14):
    """CG with diagonal preconditioning."""
    n = len(rhs)
    x = np.zeros(n)
    r = rhs - matvec(x)
    z = r / np.maximum(diag_h, 1e-20)
    p = z.copy()
    rz = np.dot(r, z)
    rhs_norm = np.linalg.norm(rhs)
    rhs_norm = max(rhs_norm, 1e-20)
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rz / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        r_norm = np.sqrt(np.dot(r, r))
        if r_norm < tol * rhs_norm:
            break
        z_new = r / np.maximum(diag_h, 1e-20)
        rz_new = np.dot(r, z_new)
        if rz < 1e-30:
            break
        beta_cg = rz_new / rz
        p = z_new + beta_cg * p
        rz = rz_new
    return x


def _hessian_diag(a, A_vec, n):
    """Compute diagonal of the Hessian for active-set problem."""
    # H*v has components: 2 * sum_k w_k * (contribution to v[i])
    # Diagonal: 2 * sum_k w_k * (1 if there's a valid j such that the term a[i]*a[j] contributes to (a*a)[k])
    # For (a*a)[k] = sum_i a[i]*a[k-i], the diagonal element for variable i in the
    # quadratic form sum_k w_k * (a*a)[k] is:
    # 2 * sum_k w_k * [1 if k-i is valid] (counting the cross term)
    # Actually the Hessian of (a*a)[k] w.r.t. a is a matrix M_k where (M_k)_{i,j} = 1
    # if i+j=k (approximately, with boundary effects). The diagonal is (M_k)_{i,i} = 1 if 2i=k.
    # But the full quadratic form sum_k w_k * (a*a)[k] has Hessian:
    # H_{i,j} = 2 * sum_k w_k * [valid(i,j,k)]
    # where valid means i and k-i are in range (for the i-th variable) AND j and k-j are in range
    # Actually it's simpler: H_{i,j} = 2 * (sum_k w_k * delta_{i+j,k}) for the "off-diagonal" case
    # and H_{i,i} = 2 * (sum_k w_k * delta_{2i,k}) + (sum_k w_k * [2i=k valid])
    # For simplicity, approximate: diag[i] = 2 * (number of k in active set such that a valid j exists)
    # A simpler approximation: diag[i] ≈ 2 * min(i+1, n-i-1+1, len(A_vec_active)) * (2/m)
    # Let's just compute it directly for small active sets or use a bound.
    m = len(A_vec)
    # Each element a[i] contributes to (a*a)[k] for k in [i, i+min(n-1-i, n-1)] intersected with active set
    # Number of active k that a[i] contributes to:
    # (a*a)[k] involves a[i] if 0 <= k-i < n, i.e., i <= k < i+n
    # So a[i] contributes to (a*a)[k] for k in [i, min(i+n-1, 2n-2)]
    # The Hessian diagonal for variable i is 2 * sum_{k in active, i<=k<=i+n-1, 0<=k-i<n} w_k
    # = 2 * sum of A_vec[k] for k in active range that overlap with [i, i+n-1]
    active_mask = A_vec > 0
    if not active_mask.any():
        return np.ones(n) * 0.01
    active_indices = np.where(active_mask)[0]
    weights = A_vec[active_indices]
    # For each i, count weights where active_k is in [i, i+n-1]
    # This is a range sum query - use cumulative sum
    cum_w = np.zeros(2 * n - 1)
    cum_w[active_indices] = weights
    cum_w = np.cumsum(cum_w)
    # For position i: sum of weights in [i, min(i+n-1, 2n-2)]
    lo = np.arange(n)
    hi = np.minimum(lo + n - 1, 2 * n - 2)
    diag = np.zeros(n)
    for i in range(n):
        total = cum_w[hi[i]]
        if lo[i] > 0:
            total -= cum_w[lo[i] - 1]
        diag[i] = 2.0 * total
    return np.maximum(diag, 1e-10)


def _newton_step_prec(a, c, c_max, active_mask, tol=1e-14):
    """Preconditioned Newton step with given active set."""
    n = len(a)
    m = int(active_mask.sum())
    if m < 1:
        return None
    A_vec = active_mask.astype(np.float64) / m

    # Compute gradient
    g = 2.0 * _corr_a_A(a, A_vec, n)
    g_t = g - np.mean(g)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-15:
        return np.zeros(n)

    # Compute diagonal of Hessian for preconditioning
    diag_h = _hessian_diag(a, A_vec, n)
    mu_reg = 1e-8 / m + 1e-12
    diag_h[diag_h < mu_reg] = mu_reg

    A_vec_copy = A_vec
    n_copy = n
    mu_copy = mu_reg
    diag_h_copy = diag_h

    def _hv(v):
        return 2.0 * _corr_a_A(v, A_vec_copy, n_copy) + mu_copy * v

    # Solve H*u = g_t using preconditioned CG
    u = _diag_precond_cg(_hv, g_t, diag_h_copy, max_cg=500, tol=tol)
    # Solve H*w = 1
    w = _diag_precond_cg(_hv, np.ones(n), diag_h_copy, max_cg=300, tol=tol)
    w_sum = float(w.sum())

    if abs(w_sum) < 1e-15:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w

    dx = dx - np.mean(dx)
    return dx


def _active_set_newton(a_init, deadline, max_outer=200):
    """Active-set Newton with adaptive active set based on KKT conditions."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Start with a loose active set
    frac = 0.05
    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set: include peaks within (1 - eps) of max
        # Gradually tighten eps
        if outer < 30:
            eps = 0.05 - 0.001 * outer
        elif outer < 80:
            eps = 0.02 - 0.0002 * (outer - 30)
        else:
            eps = 0.01 - 0.0001 * (outer - 80)
        eps = max(eps, 0.001)

        threshold = c_max * (1.0 - eps)
        active_mask = c >= threshold
        if not active_mask.any():
            active_mask[np.argmax(c)] = True

        # Cap active set size
        m = int(active_mask.sum())
        if m > 50 * n:
            # Keep only top 50*n peaks
            top_k = np.argsort(c)[::-1][:50 * n]
            active_mask = np.zeros_like(c, dtype=bool)
            active_mask[top_k] = True

        # Newton step
        dx = _newton_step_prec(a, c, c_max, active_mask)
        if dx is None:
            break

        # Line search with quadratic interpolation
        f_cur = c_max
        best_t = None
        best_f = f_cur
        t_prev = 0.0
        f_prev = f_cur + 1.0  # force first step
        t = 1.0

        for ls in range(60):
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())

            if f_try < best_f:
                best_f = f_try
                best_t = t
                if f_try < f_cur - 1e-13:
                    break
            if f_try < f_cur * (1 - 1e-4 * t):
                break
            t *= 0.5

        if best_t is None or best_t < 1e-15:
            # Try a smaller fixed step
            for t_fix in [0.01, 0.005, 0.001, 0.0001]:
                a_try = a + t_fix * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_cur - 1e-15:
                    best_t = t_fix
                    best_f = f_try
                    break

        if best_t is None:
            break

        a = a + best_t * dx
        a = np.maximum(a, 0)
        s = a.sum()
        a = a / s

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Check convergence
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _peak_shaving(a, deadline, n_rounds=50):
    """Direct peak shaving: identify elements contributing most to peaks and reduce them."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        c_max = float(c.max())

        # Find the top peaks
        n_peaks = min(10, len(c))
        peak_indices = np.argsort(c)[::-1][:n_peaks]

        # For each peak at position k, find which elements contribute most
        # (a*a)[k] = sum_i a[i]*a[k-i], so element a[i] contributes a[i]*a[k-i]
        total_reduction = np.zeros(n)
        for k in peak_indices:
            # Contributions to (a*a)[k] from element i: a[i]*a[k-i] (if valid)
            i_vals = np.arange(n)
            j_vals = k - i_vals
            valid = (j_vals >= 0) & (j_vals < n)
            if not valid.any():
                continue
            contrib = np.zeros(n)
            contrib[i_vals[valid]] = best_a[i_vals[valid]] * best_a[j_vals[valid]]
            # Also the "other" contribution: element j contributes a[j]*a[k-j] = a[j]*a[i]
            # This is already counted when we swap i and j
            total_reduction += contrib

        # Normalize
        tot_sum = total_reduction.sum()
        if tot_sum < 1e-20:
            break
        total_reduction = total_reduction / tot_sum

        # Reduce elements proportional to their contribution to peaks
        # But don't reduce below zero
        for strength in [0.001, 0.005, 0.01, 0.02, 0.05]:
            a_try = best_a * (1.0 - strength * total_reduction / max(total_reduction.max(), 1e-20))
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                break
        else:
            # No improvement, try additive reduction
            a_try = best_a - 0.001 * total_reduction * best_a.max()
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
            else:
                break

    return best_a, best_sc


def _lbfgs_continuation_v2(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with log-sum-exp continuation and adaptive restart."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)

    # Determine initial beta
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 100:
        gap = c_sorted[0] - c_sorted[100]
    elif len(c_sorted) > 10:
        gap = c_sorted[0] - c_sorted[10]
    else:
        gap = c_sorted[0] - c_sorted[-1]
    base_beta = max(gap * 0.5, max_c * 0.01)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Adaptive continuation: halve beta each stage, but restart if no progress
    beta = base_beta
    min_beta = max_c * 1e-9
    stall_count = 0
    max_stall = 3

    while beta > min_beta and time.monotonic() + min_time_remaining < deadline:
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(3000, max(50, int(remaining / (est_time_per_iter * 15))))

        try:
            def _obj(z_val):
                return _ls_obj(z_val, beta, n)

            def _grad(z_val):
                return _ls_grad(z_val, beta, n)

            result = minimize(
                _obj, z, jac=_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 500, 'iprint': -1}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)

            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                stall_count = 0
            else:
                stall_count += 1
                if stall_count >= max_stall:
                    # Restart with a slightly larger beta
                    beta *= 2
                    stall_count = 0
                    continue

            print(f"    LS beta={beta:.2e}: score={current_sc:.12f}")
        except Exception as ex:
            print(f"    LS beta={beta:.2e}: exception {ex}")

        beta *= 0.5

    return current_a, current_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "block2":
        a = np.zeros(n)
        block = n // 3
        a[:block] = 1.0
        a[2 * block:] = 1.0
        s = a.sum()
        return a / s if s > 0 else np.ones(n) / n
    elif kind == "block3":
        a = np.zeros(n)
        block = n // 4
        a[:block] = 1.0
        a[block * 2:block * 3] = 1.0
        a[3 * block:] = 1.0
        s = a.sum()
        return a / s if s > 0 else np.ones(n) / n
    else:
        return np.ones(n) / n


def _symmetrize(a):
    """Force symmetry: a[i] = a[n-1-i]."""
    n = len(a)
    half = (n + 1) // 2
    a_sym = np.zeros(n)
    for i in range(half):
        a_sym[i] = (a[i] + a[n - 1 - i]) / 2
        a_sym[n - 1 - i] = a_sym[i]
    s = a_sym.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_sym / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _full_optimize(a_init, deadline, min_remaining=30.0, label=""):
    """Full optimization pipeline: L-BFGS -> Newton -> Peak shaving -> Newton again."""
    t0 = time.monotonic()

    # Step 1: L-BFGS continuation
    lbfgs_dl = min(deadline - min_remaining, t0 + 0.35 * (deadline - t0))
    a1, sc1 = _lbfgs_continuation_v2(a_init, lbfgs_dl, min_time_remaining=min_remaining)

    # Step 2: Active-set Newton
    as_dl = min(deadline - min_remaining, time.monotonic() + 0.3 * (deadline - t0))
    a2, sc2 = _active_set_newton(a1, as_dl, max_outer=150)
    if sc2 > sc1:
        a2 = a1
        sc2 = sc1

    # Step 3: Peak shaving
    ps_dl = min(deadline - min_remaining, time.monotonic() + 0.15 * (deadline - t0))
    a3, sc3 = _peak_shaving(a2, ps_dl, n_rounds=30)
    if sc3 > sc2:
        a3 = a2
        sc3 = sc2

    # Step 4: Another Newton round
    if time.monotonic() + min_remaining < deadline:
        as2_dl = min(deadline - min_remaining, time.monotonic() + 0.1 * (deadline - t0))
        a4, sc4 = _active_set_newton(a3, as2_dl, max_outer=100)
        if sc4 < sc3:
            a3 = a4
            sc3 = sc4

    return a3, sc3


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining preconditioned Newton, L-BFGS continuation,
    peak shaving, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 40.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.12f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        n_main = len(parent)
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_main}, score={sc_parent:.12f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_main = 16384
        parent = _make_profile(n_main, "uniform")
        update_best(parent, _score(parent), "init_uniform")

    n = n_main

    # === Phase 1: Main optimization from parent ===
    print(f"\n[Phase 1] Main optimization (t={time.monotonic() - t_start:.1f}s)")
    p1_dl = min(deadline - 400, time.monotonic() + 400)
    a1, sc1 = _full_optimize(parent, p1_dl, min_remaining=400, label="main")
    update_best(a1, sc1, "main_opt")

    # === Phase 2: Symmetric variant ===
    print(f"\n[Phase 2] Symmetric variant (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a_sym_init = _symmetrize(best_seq if best_seq is not None else parent)
        p2_dl = min(deadline - 250, time.monotonic() + 200)
        a2, sc2 = _full_optimize(a_sym_init, p2_dl, min_remaining=250, label="sym")
        update_best(a2, sc2, "symmetric")

    # === Phase 3: Different starting profiles ===
    print(f"\n[Phase 3] Diverse starts (t={time.monotonic() - t_start:.1f}s)")
    profiles_to_try = [
        ("tapered", 0.3),
        ("gaussian", 0.25),
        ("raised_cosine", 0.2),
        ("block2", 0.15),
    ]
    for kind, frac_time in profiles_to_try:
        if time.monotonic() > deadline - 150:
            break
        if best_seq is None:
            continue
        # Interpolate best to this profile's space or create fresh
        a_prof = _make_profile(n, kind)
        # Mix with current best
        mix = 0.5 * best_seq + 0.5 * a_prof
        mix = np.maximum(mix, 0)
        mix = mix / mix.sum()

        rem = deadline - 150 - time.monotonic()
        if rem < 30:
            break
        p_dl = time.monotonic() + min(rem * 0.7, 120)
        a_p, sc_p = _full_optimize(mix, p_dl, min_remaining=150, label=kind)
        update_best(a_p, sc_p, f"start_{kind}")

    # === Phase 4: Perturbed restarts from best ===
    print(f"\n[Phase 4] Perturbed restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        for sigma in [0.003, 0.01, 0.03, 0.08]:
            if time.monotonic() > deadline - 80:
                break
            p_start = _perturb(best_seq, sigma, rng)
            rem = deadline - 80 - time.monotonic()
            if rem < 20:
                break
            p_dl = time.monotonic() + min(rem * 0.8, 80)
            a_r, sc_r = _full_optimize(p_start, p_dl, min_remaining=80, label=f"pert_{sigma}")
            update_best(a_r, sc_r, f"perturb_{sigma}")

    # === Phase 5: Try larger n ===
    print(f"\n[Phase 5] Larger n (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 120 and best_seq is not None:
        n_large = 32768
        a_large = _interpolate_seq(best_seq, n_large)
        sc_large_init = _score(a_large)
        print(f"  n={n_large} init score: {sc_large_init:.12f}")

        # Quick optimization
        rem = deadline - 120 - time.monotonic()
        if rem > 60:
            p5_dl = time.monotonic() + min(rem * 0.7, 180)
            a_l, sc_l = _full_optimize(a_large, p5_dl, min_remaining=120, label="large")
            update_best(a_l, sc_l, f"n_{n_large}")

    # === Phase 6: Final intensive polish ===
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        # Multiple Newton rounds with tightening active set
        remaining = deadline - 10 - time.monotonic()
        if remaining > 20:
            a_pol = best_seq.copy()
            sc_pol = best_score
            for polish_round in range(5):
                if time.monotonic() > deadline - 10:
                    break
                round_dl = min(deadline - 10, time.monotonic() + remaining / (5 - polish_round))
                a_pol, sc_pol = _active_set_newton(a_pol, round_dl, max_outer=50)
                if sc_pol < best_score:
                    best_seq = a_pol.copy()
                    best_score = sc_pol
                    print(f"  Polish round {polish_round}: {best_score:.12f}")
                # Peak shave between Newton rounds
                if time.monotonic() < deadline - 15:
                    ps_dl = min(deadline - 15, time.monotonic() + 5)
                    a_ps, sc_ps = _peak_shaving(a_pol, ps_dl, n_rounds=10)
                    if sc_ps < sc_pol:
                        a_pol = a_ps
                        sc_pol = sc_ps

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
