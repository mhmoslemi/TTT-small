def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

                break
            lr = 0.1 / (gnorm * (1.0 + 0.05 * inner) + 1e-10)
            a = _proj_simplex(a - lr * grad)

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
        elif rnd > 10:
            a = best_a.copy()

        if (rnd + 1) % 8 == 0:
            tol_idx += 1

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Ramp-based optimizer. Key insight: a monotonically increasing sequence
    (ramp) achieves score ~4/3 for large n, far better than symmetric profiles.
    Refines via L-BFGS with log-sum-exp, active-set QP, and peak reduction.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Check for parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                par_sc = _score(parent_raw)
                parent = parent_raw
                n_parent = len(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    # === Phase 1: Try multiple ramp-like initializations ===
    print(f"\n[Phase 1] Ramp initializations (t={time.monotonic()-t_start:.1f}s)")

    # Different n values
    n_candidates = [32768, 16384, 65536]
    n_use = 32768  # primary

    # Try ramps with different powers
    for power in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]:
        a_ramp = _make_ramp(n_use, power)
        sc = _score(a_ramp)
        print(f"  Ramp p={power}: score={sc:.10f}")
        update_best(a_ramp, sc, f"ramp_p{power}")

    # Try exponential
    for rate in [0.5, 1.0, 2.0, 3.0]:
        a_exp = _make_exponential(n_use, rate)
        sc = _score(a_exp)
        print(f"  Exp rate={rate}: score={sc:.10f}")
        update_best(a_exp, sc, f"exp_r{rate}")

    if best_seq is None:
        best_seq = _make_ramp(n_use, 1.0)
        best_score = _score(best_seq)
        print(f"  Fallback ramp: {best_score:.10f}")
        return [float(x) for x in best_seq]

    # === Phase 2: L-BFGS refinement of best ramp ===
    print(f"\n[Phase 2] L-BFGS refinement (t={time.monotonic()-t_start:.1f}s)")
    phase2_deadline = min(deadline - 600, t_start + 400)
    a_lb, sc_lb = _run_lbfgs(best_seq, phase2_deadline, min_time=30.0)
    update_best(a_lb, sc_lb, "lbfgs_r1")

    # === Phase 3: Peak reduction ===
    print(f"\n[Phase 3] Peak reduction (t={time.monotonic()-t_start:.1f}s)")
    phase3_deadline = min(deadline - 400, phase2_deadline + 300)
    a_pr, sc_pr = _run_peak_reduction(best_seq, phase3_deadline, min_time=20.0)
    update_best(a_pr, sc_pr, "peak_red_1")

    # === Phase 4: Active-set QP ===
    print(f"\n[Phase 4] Active-set QP (t={time.monotonic()-t_start:.1f}s)")
    phase4_deadline = min(deadline - 300, phase3_deadline + 300)
    a_qp, sc_qp = _run_active_qp(best_seq, phase4_deadline, min_time=20.0, max_rounds=100)
    update_best(a_qp, sc_qp, "active_qp_1")

    # === Phase 5: Second L-BFGS round with tighter params ===
    print(f"\n[Phase 5] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    phase5_deadline = min(deadline - 200, phase4_deadline + 250)
    a_lb2, sc_lb2 = _run_lbfgs(best_seq, phase5_deadline, min_time=20.0)
    update_best(a_lb2, sc_lb2, "lbfgs_r2")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[Phase 6] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    phase6_deadline = min(deadline - 100, phase5_deadline + 200)
    for n_try in [65536, 16384]:
        if time.monotonic() > phase6_deadline - 30:
            break
        if n_try == len(best_seq):
            continue
        # Resample best to new n
        x_old = np.linspace(0, 1, len(best_seq))
        x_new = np.linspace(0, 1, n_try)
        a_resamp = np.interp(x_new, x_old, best_seq)
        a_resamp = np.maximum(a_resamp, 0)
        s = a_resamp.sum()
        if s > 1e-15:
            a_resamp = a_resamp / s
        # Also try fresh ramp at this n
        a_fresh = _make_ramp(n_try, 1.0)
        sc_fresh = _score(a_fresh)
        sc_resamp = _score(a_resamp)

        a_use = a_fresh if sc_fresh < sc_resamp else a_resamp
        sc_use = min(sc_fresh, sc_resamp)

        a_opt, sc_opt = _run_lbfgs(a_use, phase6_deadline, min_time=15.0)
        update_best(a_opt, sc_opt, f"n_{n_try}")

    # === Phase 7: Final polish ===
    print(f"\n[Phase 7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_fin, sc_fin = _run_active_qp(best_seq, deadline - 10, min_time=5.0, max_rounds=50)
        update_best(a_fin, sc_fin, "final_qp")

        if time.monotonic() < deadline - 10:
            a_pr2, sc_pr2 = _run_peak_reduction(best_seq, deadline - 5, min_time=3.0)
            update_best(a_pr2, sc_pr2, "final_pr")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_ramp(16384, 1.0)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_ramp(16384, 1.0)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
