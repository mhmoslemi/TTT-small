def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _logsumexp_obj(z, beta, n):
    """Smoothed max objective using log-sum-exp."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _logsumexp_grad(z, beta, n):
    """Gradient of the smoothed max objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    e = np.exp(np.clip(shifted, -700, 0))

        w = np.zeros_like(c)

        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)

"""
Peak-reduction optimizer for non-negative auto-convolution.
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using multi-temperature L-BFGS with
analytic gradients, Nesterov-momentum peak equalization, frequency-domain
structured initialization, and adaptive sequence length search.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _logsumexp_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return (4.0 * n / beta) * a * (h - W)


def _power_smooth_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_pos = np.maximum(c, 1e-30)
    val = np.sum(c_pos ** p) ** (1.0 / p)
    return 2.0 * n * val


def _power_smooth_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_pos = np.maximum(c, 1e-30)
    w = c_pos ** (p - 1)
    Ws = w.sum()
    if Ws < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        Ws = 1.0
    else:
        w = w / Ws
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    val = float(np.sum(c_pos ** p) ** (1.0 / p))
    return (4.0 * n / p * (val ** (1 - p))) * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n, mode="lse"):
        self.beta = beta
        self.n = n
        self.mode = mode

    def __call__(self, z):
        if self.mode == "lse":
            return _logsumexp_obj(z, self.beta, self.n)
        else:
            return _power_smooth_obj(z, self.beta, self.n)

    def grad(self, z):
        if self.mode == "lse":
            return _logsumexp_grad(z, self.beta, self.n)
        else:
            return _power_smooth_grad(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=50.0, beta_scale=1.0, mode="lse"):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.01) * beta_scale

    if mode == "lse":
        betas = base_beta * np.array([
            1.0, 0.5, 0.25, 0.125, 0.0625,
            0.03, 0.015, 0.007, 0.003,
            0.0015, 0.0007, 0.0003, 0.0001,
            0.00005, 0.00002, 0.00001
        ])
    else:
        betas = np.array([12.0, 16.0, 24.0, 36.0, 54.0, 84.0, 144.0, 288.0, 576.0])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 4096.0)
        if mode == "lse" and beta < base_beta * 0.01:
            max_iter = min(5000, max(100, int(remaining / (est_time_per_iter * 20))))
        else:
            max_iter = min(3000, max(50, int(remaining / (est_time_per_iter * 15))))

        prob = _LbfgsProb(beta, n, mode=mode)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 40,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            elif i >= len(betas) - 3:
                # Near the end, accept even tiny improvements
                if sc_new < current_sc:
                    current_a = a_new
                    current_sc = sc_new
                    z = np.log(np.maximum(a_new, 1e-30))
                    z = z - z.mean()
        except Exception:
            pass

    return current_a


def _peak_equalize_momentum(a_init, deadline, min_time_remaining=20.0):
    """Nesterov-momentum subgradient method for peak equalization."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    v = np.zeros(n)  # momentum
    mu = 0.9  # momentum coefficient
    K = 5
    no_improve = 0
    lr = 0.01
    t = 0

    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            lr = min(0.03, lr * 1.02)
            if K < 50:
                K = min(50, K + 1)
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.8
                K = max(3, K // 2)
            if no_improve > 500:
                a = best_a.copy()
                v = np.zeros(n)
                lr = 0.01
                K = 5
                no_improve = 0
                mu = min(0.95, mu * 1.01)
                if no_improve > 1000:
                    break

        # Active set: top K peaks
        top_k_idx = np.argsort(c)[::-1][:K]
        w_S = np.zeros(2 * n - 1)
        w_S[top_k_idx] = 1.0 / K
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)[n - 1:2 * n - 1]
        W = float(np.dot(w_S, c))
        grad = 2.0 * a * (h - W)

        # Center the gradient (maintain sum)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Nesterov momentum
        v = mu * v - lr * grad / (gnorm + 1e-15)
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        # Check improvement every few steps
        if t % 10 == 0:
            c_new = _full_convolve(a_new, a_new)
            sc_new = 2.0 * n * float(c_new.max())
            if sc_new < current_sc:
                a = a_new
            else:
                a = a - 0.5 * v  # undo half the step
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                v = 0.5 * v

    return best_a, best_sc


def _peak_equalize_line_search(a_init, deadline, min_time_remaining=20.0):
    """Peak equalization with careful line search."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    K = 3
    no_improve = 0
    lr = 0.03

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            if K < 40:
                K = min(40, K + 1)
        else:
            no_improve += 1
            lr *= 0.9
            if no_improve % 50 == 0:
                K = max(2, K // 2)
            if no_improve > 300:
                a = best_a.copy()
                lr = 0.03
                K = 3
                no_improve = 0
                if no_improve > 600:
                    break

        # Compute gradient direction (average of top-K peak gradients)
        top_k_idx = np.argsort(c)[::-1][:K]
        a_rev = a[::-1]
        grad = np.zeros(n)
        for k in top_k_idx:
            # d c_k / d a_j = 2 * a_{k-j} (if valid)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += 2.0 * a[k - j]
        grad /= K
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Line search
        alpha = lr / (gnorm + 1e-15)
        improved = False
        for _ in range(40):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_freq_init(n, seed):
    """Create initialization with flat power spectrum structure."""
    rng = np.random.default_rng(seed)
    # Create a random phase in frequency domain, take magnitude
    fd = np.ones(n)
    # Add a "chirp" structure: linear frequency sweep
    t = np.arange(n)
    freq = rng.uniform(0.1, 2.0)
    phase = 2.0 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi)
    a = 1.0 + 0.4 * np.cos(phase) + 0.2 * np.cos(2.0 * phase + 0.5)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _make_periodic_init(n, seed):
    """Create a periodic-like initialization."""
    rng = np.random.default_rng(seed)
    # Use multiple incommensurate frequencies
    a = np.ones(n)
    for _ in range(3):
        f = rng.uniform(1, n / 4)
        phase = rng.uniform(0, 2 * np.pi)
        a += 0.3 * np.cos(2 * np.pi * f * np.arange(n) / n + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _asymmetric_perturb(a, rng, strength=0.03):
    n = len(a)
    t = np.arange(n) / n
    ramp = 0.4 * (t - 0.5)
    noise = 0.6 * rng.standard_normal(n)
    perturbation = ramp + noise
    perturbation = perturbation - perturbation.mean()
    z = np.log(np.maximum(a, 1e-30))
    z = z + strength * perturbation
    return _softmax(z)
