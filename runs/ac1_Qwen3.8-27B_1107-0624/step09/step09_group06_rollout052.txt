def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -800, 800))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _log_softmax(z):
    z_max = np.max(z)
    return z - z_max - np.log(np.sum(np.exp(z - z_max)))


def _smooth_max_obj(z, tau, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_shifted = c / tau
    c_shifted = c_shifted - np.max(c_shifted)
    log_s = np.log(np.sum(np.exp(np.clip(c_shifted, -800, 0))))
    return 2.0 * n * (tau * (np.max(c_shifted) + log_s))


def _smooth_max_grad(z, tau, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_shifted = c / tau
    c_shifted = c_shifted - np.max(c_shifted)
    w = np.exp(np.clip(c_shifted, -800, 0))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 4.0 * n * a * h

def _conv_value_at_k(a, k):
    """Compute conv(a,a)[k] directly."""
    n = len(a)
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    if j_lo > j_hi:
        return 0.0
    return float(np.dot(a[j_lo:j_hi+1], a[k - j_hi:k - j_lo + 1][::-1]))


def _conv_grad_at_k(a, k):
    """Compute gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    if j_lo <= j_hi:
        idx = k - np.arange(j_lo, j_hi + 1)
        grad[j_lo:j_hi + 1] += a[idx]
    return grad

                    x1 = x2

                    f2 = score_of_x(x2)
                if time.monotonic() + min_remaining > deadline + 60:

            x_opt = (x_lo + x_hi) / 2.0
            f_opt = min(f1, f2)
            
            if f_opt < score_of_x(x_old) - 1e-14:
                a[j] = x_opt
                s = a.sum()
                # Renormalize

                sc_cur = _score(a)
                if sc_cur < best_sc:
                    best_sc = sc_cur

        if sweep % 1 == 0:

def _subspace_optimize(a_init, deadline, min_remaining=30.0, n_active=50):

    if rest_sum < 1e-15 or len(a_active) < 2:
        return a_init, _score(a_init)
    
    # Optimize a_active with constraint sum(a_active) = s - rest_sum = S_active
    S_active = s - rest_sum
    
    # Use L-BFGS on the active sub-vector (parametrize as softmax for non-negativity)
    z = np.log(np.maximum(a_active, 1e-30))
    z = z - z.mean()
    
    tau = c_max * 0.01
    n_active_len = len(a_active)
    
    def sub_obj(z_sub):
        a_sub = _softmax(z_sub) * S_active
        a_full = np.zeros(n)
        a_full[active_idx] = a_sub
        # Add back the rest
        a_full[~np.isin(np.arange(n), active_idx)] = a[~np.isin(np.arange(n), active_idx)]
        return _score(a_full)
    
    def sub_grad(z_sub):
        a_sub = _softmax(z_sub) * S_active
        a_full = np.zeros(n)
        rest_mask = ~np.isin(np.arange(n), active_idx)
        a_full[active_idx] = a_sub
        a_full[rest_mask] = a[rest_mask]
        
        c_full = _full_convolve(a_full, a_full)
        c_shifted = c_full / tau
        c_shifted = c_shifted - np.max(c_shifted)
        w = np.exp(np.clip(c_shifted, -800, 0))
        sw = w.sum()
        if sw < 1e-300:
            w = np.zeros_like(c_full)
            w[int(np.argmax(c_full))] = 1.0
        else:
            w = w / sw
        
        # Gradient w.r.t. a_full
        a_rev = a_full[::-1]
        wc = _full_convolve(w, a_rev)
        h_full = wc[n - 1:2 * n - 1]
        grad_full = 4.0 * n * a_full * h_full / (a_full.sum()**2)
        
        # Project to active coordinates
        grad_sub = grad_full[active_idx]
        # Account for softmax parametrization
        grad_z = grad_sub * a_sub - np.dot(grad_sub, a_sub) * a_sub
        return grad_z
    
    try:
        result = minimize(sub_obj, z, jac=sub_grad, method='L-BFGS-B',
                          options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-14,
                                   'maxcor': 50, 'iprint': -1})
        a_sub_new = _softmax(result.x) * S_active
        a_new = a.copy()
        a_new[active_idx] = a_sub_new
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        sc_new = _score(a_new)
        sc_old = _score(a_init)
        if sc_new < sc_old:
            return a_new, sc_new
    except Exception:
        pass
    
    return a_init, _score(a_init)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * t / 2.0)**2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sin_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)**0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * t**2) + np.exp(-20 * (1-t)**2) + 0.01
        return a / a.sum()
    elif kind == "edge_conc":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        rng = np.random.default_rng(99)
        a = np.abs(rng.standard_normal(n))
        # Smooth it
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _nesterov_steps(a_init, n_steps, deadline, min_remaining=30.0, seed=0):
    """Quick Nesterov subgradient steps for initial progress."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    momentum = 0.9
    lr = 0.003
    K = 20
    
    for t in range(n_steps):
        if time.monotonic() + min_remaining > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc_cur = 2.0 * n * c_max
        
        if sc_cur < best_sc - 1e-14:
            best_sc = sc_cur
            best_a = a.copy()
        
        # Adaptive K
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(K, len(c_sorted)-1)]
        if gap < c_max * 0.01 and K < 200:
            K = min(200, int(K * 1.2) + 1)
        
        active = np.argsort(c)[::-1][:K]
        grad = np.zeros(n)
        for k in active:
            grad += _conv_grad_at_k(a, k)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break
        
        lr_t = lr / (gnorm + 1e-18) * 0.1
        v = momentum * v - lr_t * grad_t
        a_new = a + v
        # Project to simplex
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        
        sc_new = 2.0 * n * float(_full_convolve(a_new, a_new).max())
        if sc_new < sc_cur:
            a = a_new
        else:
            v *= 0.5
            a = a_new  # still move forward on simplex
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")
    
    # === Phase 1: Brief Nesterov to make initial progress ===
    print(f"\n[P1] Nesterov warmup (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        p1_deadline = min(deadline - 600, t_start + 60)
        a_nes, sc_nes = _nesterov_steps(best_seq, 500, p1_deadline,
                                         min_remaining=30, seed=seed)
        if sc_nes < best_score:
            best_seq = a_nes.copy()
            best_score = sc_nes
            print(f"  NEW BEST: {best_score:.10f} [nesterov] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 2: Adaptive smooth L-BFGS (main workhorse) ===
    print(f"\n[P2] Adaptive smooth L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        p2_deadline = min(deadline - 400, t_start + 600)
        a_lb, sc_lb = _adaptive_smooth_lbfgs(best_seq, p2_deadline,
                                              min_remaining=60,
                                              max_beta_ratio=1.0,
                                              min_beta_ratio=0.002,
                                              seed=seed)
        if sc_lb < best_score:
            best_seq = a_lb.copy()
            best_score = sc_lb
            print(f"  NEW BEST: {best_score:.10f} [lbfgs] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 3: Subspace optimization ===
    print(f"\n[P3] Subspace optimization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        p3_deadline = min(deadline - 300, t_start + 700)
        a_sub, sc_sub = _subspace_optimize(best_seq, p3_deadline,
                                            min_remaining=60, n_active=100)
        if sc_sub < best_score:
            best_seq = a_sub.copy()
            best_score = sc_sub
            print(f"  NEW BEST: {best_score:.10f} [subspace] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 4: Coordinate descent polish ===
    print(f"\n[P4] Coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        p4_deadline = min(deadline - 200, t_start + 900)
        a_cd, sc_cd = _coord_descent_polish(best_seq, p4_deadline,
                                             min_remaining=60, n_sweeps=2)
        if sc_cd < best_score:
            best_seq = a_cd.copy()
            best_score = sc_cd
            print(f"  NEW BEST: {best_score:.10f} [coord_desc] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 5: Second round of adaptive L-BFGS (finer) ===
    print(f"\n[P5] Fine L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        p5_deadline = min(deadline - 200, t_start + 1000)
        a_lb2, sc_lb2 = _adaptive_smooth_lbfgs(best_seq, p5_deadline,
                                                min_remaining=60,
                                                max_beta_ratio=0.1,
                                                min_beta_ratio=0.001,
                                                seed=seed + 1)
        if sc_lb2 < best_score:
            best_seq = a_lb2.copy()
            best_score = sc_lb2
            print(f"  NEW BEST: {best_score:.10f} [lbfgs2] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 6: Multiple restarts with perturbation ===
    print(f"\n[P6] Restart exploration (t={time.monotonic()-t_start:.1f}s)")
    n_restarts = 5
    try:
        for r in range(n_restarts):
            if time.monotonic() + 120 > deadline:
                break
            
            # Create perturbation
            n = len(best_seq)
            eps = 0.005 * (r + 1)
            perturb = eps * rng.standard_normal(n)
            a_pert = best_seq + perturb
            a_pert = np.maximum(a_pert, 0.0)
            s_pert = a_pert.sum()
            if s_pert < 1e-10:
                continue
            a_pert = a_pert / s_pert
            
            sc_pert = _score(a_pert)
            if sc_pert > best_score * 1.05:
                continue  # Perturbation made it too bad
            
            # Quick L-BFGS from perturbed point
            p6_deadline = deadline - 60 - r * 30
            if time.monotonic() + 60 > p6_deadline:
                break
            a_r, sc_r = _adaptive_smooth_lbfgs(a_pert, p6_deadline,
                                                min_remaining=30,
                                                max_beta_ratio=0.5,
                                                min_beta_ratio=0.005,
                                                seed=seed + 100 + r)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  Restart {r+1}: NEW BEST {best_score:.10f} [t={time.monotonic()-t_start:.1f}s]")
    except Exception as e:
        print(f"  Restart error: {e}")
    
    # === Phase 7: Try different n values ===
    print(f"\n[P7] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = []
    base_n = len(best_seq)
    for factor in [0.5, 0.75, 1.5, 2.0]:
        n_try = int(base_n * factor)
        n_try = max(1024, min(32768, n_try))
        # Round to power of 2 for efficiency
        n_try = _next_pow2(n_try)
        if n_try not in n_candidates:
            n_candidates.append(n_try)
    
    for n_try in sorted(n_candidates):
        if time.monotonic() + 60 > deadline:
            break
        if n_try == len(best_seq):
            continue
        
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")
        
        if time.monotonic() + 40 > deadline:
            break
        if sc_init < best_score:
            best_seq = a_init.copy()
            best_score = sc_init
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [resampled]")
        
        # Quick optimization
        if time.monotonic() + 30 < deadline:
            a_opt, sc_opt = _adaptive_smooth_lbfgs(a_init, deadline,
                                                    min_remaining=20,
                                                    max_beta_ratio=0.5,
                                                    min_beta_ratio=0.01,
                                                    seed=seed + n_try)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 8: Final coordinate descent + L-BFGS ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        a_cd2, sc_cd2 = _coord_descent_polish(best_seq, deadline,
                                               min_remaining=20, n_sweeps=1)
        if sc_cd2 < best_score:
            best_seq = a_cd2.copy()
            best_score = sc_cd2
            print(f"  NEW BEST: {best_score:.10f} [final_cd] t={time.monotonic()-t_start:.1f}s")
    
    # Final: ultra-fine L-BFGS
    remaining = deadline - 5 - time.monotonic()
    if remaining > 15:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        c_sorted = np.sort(c_best)[::-1]
        gap = mb - c_sorted[min(5, len(c_sorted)-1)]
        if gap < mb * 0.0001:
            gap = mb * 0.0001
        tau_fin = max(gap * 0.1, mb * 0.0005)
        
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        
        def fin_obj(z_in):
            return _smooth_max_obj(z_in, tau_fin, n)
        
        def fin_grad(z_in):
            return _smooth_max_grad(z_in, tau_fin, n)
        
        max_iter = min(3000, max(30, int(remaining / 0.03)))
        try:
            result = minimize(fin_obj, z, jac=fin_grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxcor': 100, 'iprint': -1})
            a_fin = _softmax(result.x)
            sc_fin = _score(a_fin)
            if sc_fin < best_score:
                best_seq = a_fin.copy()
                best_score = sc_fin
                print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
        except Exception:
            pass
    
    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
