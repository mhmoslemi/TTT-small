"""
Multi-phase optimizer: frequency-domain spectral flattening + Adam in log-space + L-BFGS polish.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _spectral_flatten(a, time_budget):
    """
    Frequency-domain iterative method to flatten the autocorrelation.
    Works by iteratively reducing peaks in the power spectrum.
    """
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    
    N = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = _score(a)
    
    deadline = time.monotonic() + time_budget
    alpha = 0.1  # step size for spectral modification
    
    for iteration in range(10000):
        if time.monotonic() > deadline:
            break
        
        # Compute FFT
        A = np.fft.rfft(a, N)
        P = np.abs(A)**2  # power spectrum (= circular autocorrelation in freq domain)
        
        # The autocorrelation in time domain
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
        
        # Strategy: reduce peaks in the power spectrum
        # The peaks in c correspond to non-uniformity in P
        # We want P to be as uniform as possible
        
        # Find the "target" level: median of P (excluding DC)
        P_non_dc = P[1:]  # exclude DC component
        target = np.median(P_non_dc)
        
        # Reduce peaks: for bins where P > target * factor, reduce them
        factor = 2.0
        mask = P > target * factor
        if not np.any(mask):
            # Already fairly flat, use smaller factor
            factor = 1.2
            mask = P > target * factor
            if not np.any(mask):
                break
        
        # Modify: reduce the magnitude of A in the peak bins
        # |A_new| = |A| * (target / P)^(alpha/2) for peak bins
        reduction = np.ones_like(np.abs(A))
        peak_vals = np.abs(A)[mask]
        reduction[mask] = np.sqrt(target / P[mask]) ** alpha
        
        A_new = A * reduction
        
        # Transform back
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Check if it improved
        sc_new = _score(a_new)
        if sc_new < current_sc + 1e-10:  # allow small degradation for exploration
            a = a_new
        else:
            # Try smaller step
            alpha *= 0.95
            if alpha < 0.001:
                break
        
    return best_a, best_sc

(3) L-BFGS polish, (4) multi-n exploration.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho_idx = np.where(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    r = rho_idx[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _compute_grad_smoothed(a, w, beta, n):
    """
    Compute gradient of 2n * log(sum exp(c_k/beta)) w.r.t. a.
    where c = conv(a, a) and w = softmax(c/beta).
    
    Returns gradient in a-space (length n).
    """
    # grad_a[j] = (2n/beta) * sum_k w[k] * dc_k/da_j
    # dc_k/da_j = 2*a[k-j] (with a out of range = 0)
    # So grad_a[j] = (4n/beta) * sum_k w[k] * a[k-j]
    # = (4n/beta) * sum_m a[m] * w[m+j]  (substituting m = k-j)
    
    N2 = _next_pow2(3 * n)
    A_fft = np.fft.fft(a, N2)
    W_fft = np.fft.fft(w, N2)
    g = np.fft.ifft(np.conj(A_fft) * W_fft).real[:n]
    grad_a = (4.0 * n / beta) * g
    return grad_a


def _smooth_obj_and_grad(a, beta, n):
    """
    Compute smooth objective value and gradient.
    L(a) = 2n * log(sum_k exp(c_k / beta))
    """
    c = _full_convolve(a, a)
    c_s = c / beta
    c_max = c_s.max()
    e = np.exp(c_s - c_max)
    s = e.sum()
    if s < 1e-300:
        val = 2.0 * n * (c_max * beta)  # = 2n * c.max()
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        val = 2.0 * n * (c_max * beta + beta * np.log(s))
        w = e / s
    
    grad_a = _compute_grad_smoothed(a, w, beta, n)
    return val, grad_a, w, c

    beta1 = 0.9
    beta2 = 0.9999
    eps = 1e-8

    t = 0
    no_improve = 0

    c_init = _full_convolve(a, a)
    c_max = float(c_init.max())
    c_sorted = np.sort(c_init)[::-1]
    K = min(50, len(c_sorted) - 1)
    gap = c_max - c_sorted[K]
    if gap < c_max * 1e-4:
        gap = c_max * 1e-4
    beta_start = gap * 10
    beta_end = c_max * 1e-5
    
    print(f"    Adam: n={n}, initial_score={best_sc:.10f}, beta_start={beta_start:.6f}")
    
    while time.monotonic() < deadline:
        t += 1
        
        # Adaptive beta: cosine schedule
        # But also adapt based on progress

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    t_start = time.monotonic()
    deadline = t_start + time_budget

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Estimate iterations based on time
        n_scale = max(1.0, n / 8192.0)
        time_per_iter = 0.003 * n_scale
        max_iter = min(5000, max(30, int(remaining / (time_per_iter * 8))))

        prob = _LbfgsSmoothProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 100, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"      L-BFGS beta[{i}]={beta:.8f}: score={current_sc:.10f}")
            else:
                # Try with perturbed initial point
                z_pert = z + 0.01 * np.random.default_rng(42 + i).standard_normal(n)
                z_pert = z_pert - z_pert.mean()
                result2 = minimize(
                    prob, z_pert, jac=prob.grad, method='L-BFGS-B',
                    options={'maxiter': max_iter // 2, 'ftol': 1e-18,
                             'gtol': 1e-14, 'maxcor': 50, 'maxls': 30}
                )
                a_new2 = _softmax(result2.x)
                sc_new2 = _score(a_new2)
                if sc_new2 < current_sc - 1e-15:
                    current_a = a_new2
                    current_sc = sc_new2
                    z = np.log(np.maximum(a_new2, 1e-30))
                    z = z - z.mean()
                    print(f"      L-BFGS beta[{i}] (pert): score={current_sc:.10f}")
        except Exception as ex:
            print(f"      L-BFGS beta[{i}] error: {ex}")

    return current_a


def _subgradient_descent(a_init, time_budget, seed=0):
    """Projected subgradient with top-K active set and adaptive step."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s

    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(seed)

    v = np.zeros(n)
    momentum = 0.9
    t = 0
    no_improve = 0
    K = 20
    K_max = min(500, n // 4)
    base_lr = 0.003

    t_start = time.monotonic()
    deadline = t_start + time_budget

    while time.monotonic() < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive K
        c_sorted = np.sort(c)[::-1]
        K_use = min(K, len(c_sorted) - 1)
        gap = c_max - c_sorted[K_use]
        if gap < c_max * 0.002 and K < K_max:
            K = min(K_max, int(K * 1.5) + 1)

        # Active peaks
        active = np.argsort(c)[::-1][:K]

        # Subgradient: sum of gradients of active peaks
        a_flipped = a[::-1]
        active_w = np.zeros(2 * n - 1)
        for k in active:
            active_w[k] = 1.0
        conv_wa = _full_convolve(active_w, a_flipped)
        grad = 2.0 * conv_wa[n - 1:2 * n - 1]

        # Project to tangent space
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break

        # Adaptive learning rate
        lr = base_lr / (np.sqrt(t) + 1.0)
        step = lr / (gnorm + 1e-18) * 0.1

        # Momentum step
        v = momentum * v - step * grad_t
        a_cand = _project_simplex(a + v)

        # Check improvement
        c_new = _full_convolve(a_cand, a_cand)
        sc_new = 2.0 * n * float(c_new.max())
        if sc_new <= current_sc + 1e-12:
            a = a_cand
        else:
            v *= 0.5
            a = a_cand  # still stay on simplex

        # Periodic perturbation
        if no_improve > 300 and no_improve % 300 == 0:
            perturb = 0.005 * rng.standard_normal(n)
            a_pert = _project_simplex(a + perturb)
            c_pert = _full_convolve(a_pert, a_pert)
            sc_pert = 2.0 * n * float(c_pert.max())
            if sc_pert < best_sc:
                a = a_pert
                best_a = a.copy()
                best_sc = sc_pert
            v = np.zeros(n)
            no_improve = 0

        if t % 2000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"      SubGrad t={t}: best={best_sc:.10f}, K={K}, time={elapsed:.0f}s")

        if no_improve > 5000:
            break

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine_window":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t - np.pi)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":
        a = np.ones(n)
        taper = max(1, n // 100)
        for i in range(taper):
            a[i] = (i + 1) / taper
            a[n - 1 - i] = (i + 1) / taper
        return a / a.sum()
    else:
        return np.ones(n) / n


def _spectral_init(n, iterations=200):
    """Initialize using frequency-domain spectral flattening."""
    a = _make_profile(n, "flat_taper")
    N = _next_pow2(2 * n - 1)
    alpha = 0.3

    for _ in range(iterations):
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        P = mag ** 2

        # Reduce peaks in power spectrum
        P_non_dc = P[1:]
        target = np.median(P_non_dc)
        reduction = np.ones_like(mag)
        mask = P > target * 1.5
        if np.any(mask):
            reduction[mask] = np.sqrt(target / P[mask]) ** alpha
        A_new = mag * reduction * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        if np.allclose(a_new, a, atol=1e-12):
            break
        a = a_new

    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining subgradient, Adam, L-BFGS, and multi-n search.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    n = len(best_seq)

    # === Phase 1: Subgradient descent (escape local minima) ===
    print(f"\n[P1] Subgradient (t={time.monotonic()-t_start:.1f}s)")
    p1_time = min(200.0, (deadline - t_start) * 0.15)
    if time.monotonic() < deadline - 600:
        a_sg, sc_sg = _subgradient_descent(best_seq, p1_time, seed=seed)
        if sc_sg < best_score:
            best_seq = a_sg.copy()
            best_score = sc_sg
            print(f"  NEW BEST: {best_score:.10f} [subgrad] t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Adam optimizer (main workhorse) ===
    print(f"\n[P2] Adam (t={time.monotonic()-t_start:.1f}s)")
    p2_time = min(350.0, (deadline - t_start) * 0.25)
    if time.monotonic() < deadline - 400:
        a_adam, sc_adam = _adam_optimize(best_seq, p2_time, seed=seed, lr_scale=1.0)
        if sc_adam < best_score:
            best_seq = a_adam.copy()
            best_score = sc_adam
            print(f"  NEW BEST: {best_score:.10f} [adam1] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: L-BFGS polish ===
    print(f"\n[P3] L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        p3_time = min(300.0, (deadline - t_start) * 0.2)
        a_lb = _lbfgs_optimize(best_seq, p3_time, min_time_remaining=60.0)
        sc_lb = _score(a_lb)
        if sc_lb < best_score:
            best_seq = a_lb.copy()
            best_score = sc_lb
            print(f"  NEW BEST: {best_score:.10f} [lbfgs1] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Second Adam pass (finer, different LR) ===
    print(f"\n[P4] Adam pass 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        p4_time = min(200.0, (deadline - t_start) * 0.15)
        a_adam2, sc_adam2 = _adam_optimize(best_seq, p4_time, seed=seed + 1, lr_scale=0.3)
        if sc_adam2 < best_score:
            best_seq = a_adam2.copy()
            best_score = sc_adam2
            print(f"  NEW BEST: {best_score:.10f} [adam2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Another L-BFGS ===
    print(f"\n[P5] L-BFGS pass 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        p5_time = min(200.0, (deadline - t_start) * 0.15)
        a_lb2 = _lbfgs_optimize(best_seq, p5_time, min_time_remaining=40.0)
        sc_lb2 = _score(a_lb2)
        if sc_lb2 < best_score:
            best_seq = a_lb2.copy()
            best_score = sc_lb2
            print(f"  NEW BEST: {best_score:.10f} [lbfgs2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [n]
    for n_try in [8192, 16384, 24576, 32768]:
        if n_try not in n_candidates:
            n_candidates.append(n_try)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 120:
            break
        if n_try == len(best_seq):
            continue

        remaining_for_this = deadline - time.monotonic() - 100
        if remaining_for_this < 60:
            continue

        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}, remaining={remaining_for_this:.0f}s")

        # Quick Adam
        if remaining_for_this > 120:
            adam_time = min(100.0, remaining_for_this * 0.4)
            a_adv, sc_adv = _adam_optimize(a_init, adam_time, seed=seed + n_try, lr_scale=0.5)
            if sc_adv < sc_init:
                a_init = a_adv
                sc_init = sc_adv
            remaining_for_this = deadline - time.monotonic() - 60

        # L-BFGS
        if remaining_for_this > 50:
            lb_time = min(80.0, remaining_for_this * 0.6)
            a_opt = _lbfgs_optimize(a_init, lb_time, min_time_remaining=20.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Perturbation and re-optimize ===
    print(f"\n[P7] Perturbation (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        n_pert = len(best_seq)
        for attempt in range(3):
            if time.monotonic() > deadline - 60:
                break
            noise_scale = 0.002 * (attempt + 1)
            a_pert = best_seq + noise_scale * rng.standard_normal(n_pert)
            a_pert = np.maximum(a_pert, 0.0)
            s_pert = a_pert.sum()
            if s_pert < 1e-10:
                continue
            a_pert = a_pert / s_pert
            sc_pert = _score(a_pert)
            if sc_pert < best_score:
                # Refine this perturbation
                a_adv, sc_adv = _adam_optimize(a_pert, 60.0, seed=seed + 100 + attempt, lr_scale=0.2)
                if sc_adv < best_score:
                    best_seq = a_adv.copy()
                    best_score = sc_adv
                    print(f"  NEW BEST: {best_score:.10f} [perturb {attempt}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final L-BFGS polish with very small beta ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 5 - time.monotonic()
    if remaining > 20:
        n_fin = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        c_sorted = np.sort(c_best)[::-1]
        gap = mb - c_sorted[min(20, len(c_sorted) - 1)]
        beta_final = max(gap * 0.1, mb * 1e-5)

        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        prob = _LbfgsSmoothProb(beta_final, n_fin)
        max_iter = min(3000, max(50, int(remaining / 0.05)))
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-15, 'maxcor': 100, 'maxls': 50}
            )
            a_fin = _softmax(result.x)
            sc_fin = _score(a_fin)
            if sc_fin < best_score:
                best_seq = a_fin.copy()
                best_score = sc_fin
                print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")
        except Exception:
            pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
