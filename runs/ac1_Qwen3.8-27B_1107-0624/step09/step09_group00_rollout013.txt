"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses variance minimization, spectral error reduction, targeted peak gradients,
and stochastic ILS with peak-targeting perturbations.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.all(v <= 0):
        return np.ones(n) / n
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def _gradient_L2sq(a, c):
    """Gradient of sum(c[m]^2) = ||conv(a,a)||_2^2 with respect to a.
    grad[i] = 4 * sum_k c[k+i] * a[k] = 4 * conv(c, reverse(a))[n-1+i]"""
    n = len(a)
    a_rev = a[::-1]
    ca = _full_convolve(c, a_rev)
    # c has length 2n-1, a_rev has length n, so ca has length 3n-2
    # We need ca[n-1 : 2n-1] for i = 0, ..., n-1
    grad = 4.0 * ca[n-1:2*n-1]
    return grad

def _logsumexp_obj(z, beta, n):
    """Log-sum-exp smoothed objective on simplex parameterized by log-z."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    return 2.0 * n * (c_max + beta * np.log(s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _logsumexp_grad(z, beta, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())

        gap = c_max * 0.001

    log_betas = np.linspace(np.log(max(beta_start, 1e-10)), np.log(max(beta_end, 1e-12)), n_betas)

    z = z - z.mean()

        if time.monotonic() > deadline - min_time:
            break
        beta = float(np.exp(log_beta))
        remaining = deadline - min_time - time.monotonic()
        if remaining < 5:
            break
        
        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(8000, max(50, int(remaining / (est_time * 20))))
        
        def f(z):
            return _logsumexp_obj(z, beta, n)
        def g(z):
            return _logsumexp_grad(z, beta, n)

            a_new = _softmax(result.x)

                a = a_new
                sc = sc_new
                z = np.log(np.maximum(a, 1e-30))

    """Spectral error reduction: alternate between time domain projection and frequency domain flattening."""
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_sc = _score(a)
    N = _next_pow2(2 * n - 1)
    
    relaxation = 0.5
    no_improve = 0
    
    while time.monotonic() < deadline - min_time:
        # Time domain: compute convolution
        c = _full_convolve(a, a)
        sc = float(2.0 * n * c.max())
        
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                relaxation = max(0.1, relaxation * 0.95)
            if no_improve > 500:
                a = best_a.copy()
                relaxation = 0.5
                no_improve = 0
                if relaxation < 0.11:
                    break
        
        # Frequency domain: compute spectrum
        A = np.fft.rfft(a, N)
        C = A * A  # This is the spectrum of the convolution
        
        # Target: flat convolution means C should be the DFT of a flat window
        # DFT of [1/(2n-1)] * ones(2n-1) zero-padded to N
        c_target = np.zeros(N)
        c_target[:2*n-1] = 1.0 / (2*n-1)
        C_target = np.fft.rfft(c_target)
        
        # Error reduction: adjust A to move C toward C_target
        # Newton step: A_new = A - 0.5 * (C - C_target) / A
        # But handle A near zero
        eps = 1e-30
        ratio = np.ones_like(A, dtype=complex)
        mask = np.abs(A) > eps * np.max(np.abs(A))
        ratio[mask] = 1.0 - 0.5 * relaxation * (C[mask] - C_target[mask]) / A[mask]
        ratio[~mask] = 1.0  # No update where A is too small
        
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project onto simplex
        a_new = _project_simplex(a_new)
        a = a_new
    
    return best_a, _score(best_a)

def _targeted_peak_reduction(a, deadline, min_time=30.0, n_peaks=100):
    """Reduce the top peaks of the convolution using targeted gradient."""
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_sc = _score(a)
    v = np.zeros(n)  # momentum
    momentum = 0.8
    lr = 0.01
    
    no_improve = 0
    max_stall = 80
    
    while time.monotonic() < deadline - min_time:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.8
                momentum = min(0.95, momentum + 0.01)
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                lr = 0.01
                momentum = 0.8
                no_improve = 0
        
        # Find top n_peaks positions
        top_k = min(n_peaks, len(c))
        idx = np.argsort(c)[-top_k:]
        threshold = c[idx[0]]
        active = c >= threshold
        
        # Weight: inverse gap to c_max
        eps = 1e-12 * c_max
        w = np.zeros_like(c)
        w[active] = 1.0 / (c_max - c[active] + eps)
        w = w / w.sum()
        
        # Gradient: conv(w, a_rev)
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        grad = 2.0 * n * 2.0 * h  # factor of 2 for symmetry
        
        # Project to tangent space
        grad = grad - np.mean(grad)
        
        # Momentum update
        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        
        # Line search
        improved = False
        alpha = 1.0
        for _ in range(30):
            c_try = _full_convolve(a_new, a_new)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < sc - 1e-14:
                a = a_new
                improved = True
                break
            alpha *= 0.5
            a_new = _project_simplex(a + alpha * v)
        
        if not improved:
            v = 0.5 * v
    
    return best_a, _score(best_a)

def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = t ** 2 + 0.01
        return a / a.sum()
    elif kind == "asymmetric_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.4) ** 2) + np.exp(-6.0 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        if rng is None:
            rng = np.random.default_rng(42)
        a = rng.random(n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _quick_optimize(a_init, deadline, min_time=10.0):
    """Quick optimization: spectral flatten + L-BFGS + targeted peaks."""
    n = len(a_init)
    a = a_init.copy()
    
    # Spectral flattening
    if time.monotonic() < deadline - min_time - 20:
        a_spec, sc_spec = _spectral_flatten(a, min(deadline - min_time - 20, time.monotonic() + 30), min_time=5)
        if sc_spec < _score(a):
            a = a_spec
    
    # L-BFGS
    if time.monotonic() < deadline - min_time - 10:
        beta_start = float(_full_convolve(a, a).max()) * 0.01
        a_lbfgs = _lbfgs_smooth(a, min(deadline - min_time - 10, time.monotonic() + 40), 
                                beta_start, beta_start * 0.001, min_time=5)
        if _score(a_lbfgs) < _score(a):
            a = a_lbfgs
    
    # Targeted peak reduction
    if time.monotonic() < deadline - min_time:
        a_peak, sc_peak = _targeted_peak_reduction(a, min(deadline - min_time, time.monotonic() + 20), min_time=3)
        if sc_peak < _score(a):
            a = a_peak
    
    return a

def _ils(a_init, deadline, rng, min_time=50.0):
    """Iterated Local Search with peak-targeting perturbations."""
    n = len(a_init)
    current_a = _normalize(a_init.copy())
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc
    
    t_start_ils = time.monotonic()
    attempt = 0
    
    while time.monotonic() < deadline - min_time:
        attempt += 1
        strategy = attempt % 6
        
        if strategy == 0:
            # Global small noise
            z = np.log(np.maximum(current_a, 1e-30))
            z += 0.01 * rng.standard_normal(n)
            a_pert = _softmax(z - z.mean())
        elif strategy == 1:
            # Peak-targeting: increase values that contribute to peaks, decrease others
            c = _full_convolve(current_a, current_a)
            c_max = c.max()
            active = c > 0.9 * c_max
            # Find which a values contribute to active peaks
            contrib = np.zeros(n)
            for m in np.where(active)[0]:
                for i in range(n):
                    if 0 <= m - i < n:
                        contrib[i] += current_a[m - i]
            # Scale down contributing values, scale up others
            if contrib.max() > 0:
                a_pert = current_a.copy()
                contrib_norm = contrib / contrib.max()
                a_pert[contrib_norm > 0.5] *= (1 - 0.1 * contrib_norm[contrib_norm > 0.5])
                a_pert[contrib_norm <= 0.5] *= (1 + 0.1)
                a_pert = _normalize(a_pert)
            else:
                a_pert = current_a.copy()
        elif strategy == 2:
            # Block perturbation
            block = max(20, n // 10)
            start = int(rng.integers(0, n - block))
            a_pert = current_a.copy()
            z = np.log(np.maximum(a_pert[start:start+block], 1e-30))
            z += 0.05 * rng.standard_normal(block)
            a_pert[start:start+block] = np.exp(z - z.max())
            a_pert = _normalize(a_pert)
        elif strategy == 3:
            # Mirror + noise
            seg = max(10, n // 15)
            start = int(rng.integers(0, n - seg))
            a_pert = current_a.copy()
            a_pert[start:start+seg] = a_pert[start+seg-1:start-1:-1]
            z = np.log(np.maximum(a_pert, 1e-30))
            z += 0.005 * rng.standard_normal(n)
            a_pert = _softmax(z - z.mean())
        elif strategy == 4:
            # Sparse large perturbation
            n_pert = max(5, n // 20)
            idx = rng.choice(n, size=n_pert, replace=False)
            a_pert = current_a.copy()
            a_pert[idx] *= np.exp(0.3 * rng.standard_normal(n_pert))
            a_pert = _normalize(a_pert)
        else:
            # Random restart blending
            alpha = rng.uniform(0.1, 0.4)
            a_random = _make_profile(n, "random", rng)
            a_pert = _normalize((1-alpha) * current_a + alpha * a_random)
        
        # Optimize the perturbation
        opt_deadline = min(deadline - min_time, time.monotonic() + 10)
        a_opt = _quick_optimize(a_pert, opt_deadline, min_time=3)
        sc_opt = _score(a_opt)
        
        if sc_opt < best_sc - 1e-13:
            best_sc = sc_opt
            best_a = a_opt.copy()
            current_a = a_opt.copy()
            elapsed = time.monotonic() - t_start_ils
            print(f"    ILS NEW BEST: {best_sc:.10f} (strat={strategy}) t={elapsed:.0f}s")
        elif sc_opt < current_sc - 1e-13:
            current_a = a_opt.copy()
            current_sc = sc_opt
    
    return best_a, best_sc

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return _normalize(a_new)
