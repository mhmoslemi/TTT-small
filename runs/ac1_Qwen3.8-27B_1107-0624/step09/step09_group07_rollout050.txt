from scipy.optimize import minimize, linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_normalized(a):
    """Score assuming sum(a) = 1, so score = 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _normalize(a):
    s = a.sum()
    if s < 1e-15:

    return a / s

    Returns b or None if infeasible/too slow.

    # A[k, j] = coefficient of b[j] in conv(a,b)[k]
    # conv(a,b)[k] = sum_j a[k-j] * b[j] (where 0 <= k-j < n and 0 <= j < n)
    # So A[k, j] = a[k-j] if 0 <= k-j < n, else 0

    # That's too large for dense. Need to use sparse or exploit structure.

    # A[k, j] = a[k-j] if 0 <= k-j < n
    A = np.zeros((K, n))
    for i, k in enumerate(top_idx):

            A[i, j] = a[k - j]
    
    # LP: max sum(b) s.t. A @ b <= M, b >= 0
    # scipy linprog: min c^T x s.t. A_ub @ x <= b_ub
    c_obj = -np.ones(n)  # minimize -sum(b)
    b_ub = np.full(K, M)
    
    bounds = [(0, None)] * n
    
    try:

Focused optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements:
- Symmetric optimization (projection to halve effective search space)
- Polyak subgradient step with adaptive cap
- LP-based direction finding (state-of-the-art approach) using HiGHS
- Top-K smoothed max with very low temperature for near-exact gradients
- Hysteresis-based active set for stable subgradients
- Conjugate gradient escape from L-BFGS stagnation
- Multi-resolution and multi-restart strategy
- Very precise final polish phase
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm1(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _symmetrize(a):
    return (a + a[::-1]) / 2.0


def _grad_conv(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a). w has length 2n-1."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "optimal_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


class _SmooProb:
    def __init__(self, beta, K, n, symmetric=True):
        self.beta = float(beta)
        self.K = int(K)
        self.n = int(n)
        self.symmetric = symmetric

    def _compute(self, z):
        n = self.n
        z_max = z.max()
        a = np.exp(z - z_max)
        a = a / a.sum()
        if self.symmetric:
            a = _symmetrize(a)
            a = _normalize(a)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if self.K < len(c):
            top_idx = np.argpartition(c, -self.K)[-self.K:]
            c_top = c[top_idx]
        else:
            top_idx = np.arange(len(c))
            c_top = c
        shifted = (c_top - c_max) / self.beta
        shifted = np.clip(shifted, -700.0, 0.0)
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
            obj = 2.0 * n * c_max
        else:
            e_full = np.zeros(2 * n - 1)
            e_full[top_idx] = e
            w = e_full / s
            obj = 2.0 * n * (c_max + self.beta * np.log(s))
        ga = _grad_conv(a, w)
        W = float(np.dot(a, ga))
        grad_z = a * (ga - W)
        if self.symmetric:
            grad_z = _symmetrize(grad_z)
        return obj, grad_z

    def __call__(self, z):
        return self._compute(z)[0]

    def grad(self, z):
        return self._compute(z)[1]


def _run_lbfgs_sym(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with top-K smoothing and symmetric projection."""
    n = len(a_init)
    a = _normalize(_symmetrize(a_init.copy()))
    current_sc = _score(a)
    current_a = a.copy()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    schedule = [
        (1.0, 0.08), (0.7, 0.05), (0.5, 0.04), (0.3, 0.03),
        (0.2, 0.02), (0.1, 0.015), (0.05, 0.01), (0.03, 0.008),
        (0.01, 0.005), (0.005, 0.003), (0.002, 0.002), (0.001, 0.001),
    ]

    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        beta = base_beta * beta_mult
        K = max(5, int((2 * n - 1) * K_frac))

        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(50, int(remaining / (est_time * 15))))

        prob = _SmooProb(beta, K, n, symmetric=True)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxcor': 100, 'maxls': 50}
            )
            z_new = result.x
            a_new = np.exp(z_new - z_new.max())
            a_new = a_new / a_new.sum()
            a_new = _symmetrize(a_new)
            a_new = _normalize(a_new)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a


def _subgradient_polyak(a_init, deadline, min_time_remaining=60.0, use_sym=True,
                        lp_every=300, momentum_decay=0.85):
    """Subgradient with Polyak step, momentum, and periodic LP direction."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    if use_sym:
        a = _symmetrize(a)
        a = _normalize(a)
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    n_c = 2 * n - 1
    lb_score = 2.0 * n / (2 * n - 1)  # theoretical lower bound (very loose)

    active_frac = 0.02
    tol = 0.02
    lr = 0.01
    lr_max = 0.05
    lr_min = 1e-6
    momentum = 0.7
    velocity = np.zeros(n)
    stall = 0
    max_stall = 100
    iteration = 0
    last_lp = 0

    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Active set with hysteresis
        threshold = c_max * (1.0 - tol)
        active_mask = c >= threshold
        n_active = int(active_mask.sum())
        if n_active < 3:
            idx = np.argsort(c)[::-1][:5]
            active_mask = np.zeros(n_c, dtype=bool)
            active_mask[idx] = True
            n_active = 5

        # Gradient: uniform weight on active set
        w = np.zeros(n_c)
        w[active_mask] = 1.0 / n_active
        g = _grad_conv(a, w)
        gCENTERED = g - g.mean()
        gnorm = np.linalg.norm(gCENTERED)
        if gnorm < 1e-18:
            break

        # Polyak step size
        polyak_alpha = (current_sc - lb_score) / (gnorm * gnorm + 1e-30)
        alpha = min(polyak_alpha, lr_max)
        alpha = max(alpha, lr_min)

        # Momentum step
        velocity = momentum * velocity - alpha * gCENTERED
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        if use_sym:
            a_try = _symmetrize(a_try)
        s_try = a_try.sum()

        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            # Line search
            for ls in range(20):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-14:
                    improved = True
                    break
                velocity = 0.5 * velocity
                a_try = a + velocity
                a_try = np.maximum(a_try, 0.0)
                if use_sym:
                    a_try = _symmetrize(a_try)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    velocity = np.zeros(n)
                    break
                a_try = a_try / s_try
                alpha *= 0.5
                if alpha < lr_min:
                    break

        if improved:
            a = a_try
            lr = min(lr_max, lr * 1.02)
            tol = max(0.001, tol * 0.97)
        else:
            velocity = np.zeros(n)
            lr = max(lr_min, lr * 0.6)
            tol = min(0.05, tol * 1.3)

        # Adapt on stall
        if stall > max_stall:
            tol = min(0.08, tol * 2.0)
            momentum = min(0.9, momentum + 0.05)
            lr = max(lr_min, lr * 0.4)
            stall = 0

        # Periodic LP direction
        if iteration - last_lp >= lp_every and time.monotonic() + min_time_remaining + 30 < deadline:
            last_lp = iteration
            b = _lp_direction(a, c_max, n_top=80)
            if b is not None and b.sum() > 1e-15:
                b = _normalize(b)
                if use_sym:
                    b = _symmetrize(b)
                    b = _normalize(b)
                # Blend with gradient direction
                t_blend = 0.3
                a_lp = (1 - t_blend) * a + t_blend * b
                a_lp = np.maximum(a_lp, 0.0)
                a_lp = _normalize(a_lp)
                sc_lp = _score(a_lp)
                if sc_lp < current_sc - 1e-14:
                    a = a_lp
                    velocity = np.zeros(n)
                    if sc_lp < best_sc:
                        best_sc = sc_lp
                        best_a = a.copy()

    return best_a, best_sc


def _lp_direction(a, M, n_top=80):
    """LP: max sum(b) s.t. conv(a,b)[k] <= M for top-K constraints, b >= 0."""
    n = len(a)
    c = _full_convolve(a, a)
    K = min(n_top, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:K]

    # Build constraint matrix
    A = np.zeros((K, n))
    for i, k in enumerate(top_idx):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[i, j] = a[k - j]

    # LP: minimize -sum(b) s.t. A @ b <= M, b >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'time_limit': 30})
        if result.status == 0 and result.fun < -1e-10:
            return result.x
    except Exception:
        pass
    return None


def _conjugate_gradient(a_init, deadline, min_time_remaining=30.0):
    """Conjugate gradient on smoothed objective for escaping local minima."""
    n = len(a_init)
    a = _normalize(_symmetrize(a_init.copy()))
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    max_c = float(c.max())
    beta = max(max_c * 0.002, 1e-6)
    K = max(10, n // 100)
    prob = _SmooProb(beta, K, n, symmetric=True)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    f = prob(z)
    g = prob.grad(z)
    d = -g
    zg = z.copy()
    it = 0
    max_it = 5000

    while it < max_it and time.monotonic() + min_time_remaining < deadline:
        it += 1
        # Line search
        alpha = 1.0
        improved = False
        for ls in range(30):
            z_try = zg + alpha * d
            a_try = np.exp(z_try - z_try.max())
            a_try = a_try.sum()
            f_try = prob(z_try)
            if f_try < f - 1e-15:
                improved = True
                break
            alpha *= 0.5

        if not improved:
            break

        zg = zg + alpha * d
        g_new = prob.grad(zg)
        f = f_try

        a_new = np.exp(zg - zg.max())
        a_new = a_new / a_new.sum()
        a_new = _symmetrize(a_new)
        a_new = _normalize(a_new)
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()

        gy = float(np.dot(g_new, g_new))
        if gy < 1e-20:
            break
        beta_cg = gy / max(float(np.dot(g_new, g)), 1e-20)
        beta_cg = min(beta_cg, 0.9)
        d = -g_new + beta_cg * d
        g = g_new

    return best_a, best_sc


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    """Very fine active-set subgradient with line search."""
    n = len(a_init)
    a = _normalize(_symmetrize(a_init.copy()))
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.003
    tol = 0.003
    no_improve = 0
    max_no_improve = 500

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active
        g = _grad_conv(a, w)
        gC = g - g.mean()
        gnorm = np.linalg.norm(gC)
        if gnorm < 1e-18:
            break

        alpha = min(lr, 1.0 / (gnorm * gnorm + 1e-15))
        improved = False
        for ls in range(50):
            a_try = a - alpha * gC
            a_try = np.maximum(a_try, 0.0)
            a_try = _symmetrize(a_try)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 50 == 0:
                tol = max(0.0001, tol * 0.7)
                lr = max(1e-6, lr * 0.7)
            if no_improve > max_no_improve:
                a = best_a.copy()
                tol = max(0.0001, tol * 0.5)
                lr = 0.002
                no_improve = 0
        else:
            tol = max(0.0001, tol * 0.98)
            lr = min(0.005, lr * 1.005)

    return best_a, best_sc


def _direct_peak_shave(a_init, deadline, min_time_remaining=30.0):
    """Directly shave the top peaks by redistributing mass."""
    n = len(a_init)
    a = _normalize(_symmetrize(a_init.copy()))
    best_a = a.copy()
    best_sc = _score(a)
    patience = 0
    max_patience = 300

    while time.monotonic() + min_time_remaining < deadline and patience < max_patience:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        # Get top peaks
        top_k = max(3, min(10, n // 1000))
        peak_idx = np.argsort(c)[::-1][:top_k]

        # For each peak, compute sensitivity
        delta = np.zeros(n)
        for k in peak_idx:
            k = int(k)
            i_lo = max(0, k - (n - 1))
            i_hi = min(n - 1, k)
            contrib = np.zeros(n)
            for i in range(i_lo, i_hi + 1):
                j = k - i
                contrib[i] += a[j]
            # Move mass from high-contribution to low-contribution
            n_move = max(2, n // 2000)
            top_contrib = np.argsort(contrib)[::-1][:n_move]
            low_contrib = np.argsort(contrib)[:n_move]
            amount = c[k] * 0.002
            for idx in top_contrib:
                if contrib[idx] > 1e-15:
                    delta[idx] -= amount / (2.0 * contrib[idx] + 1e-30)
            for idx in low_contrib:
                delta[idx] += amount * 0.5 / (n_move * (a[idx] + 1e-10) + 1e-30)

        delta = _symmetrize(delta)
        delta -= delta.mean()

        # Line search
        alpha = 1.0
        improved = False
        for ls in range(30):
            a_try = a + alpha * delta
            a_try = np.maximum(a_try, 0.0)
            a_try = _symmetrize(a_try)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            patience += 5

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer with symmetric projection, Polyak subgradient,
    LP direction finding, conjugate gradient escape, and precise polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                # Symmetrize
                parent = _symmetrize(parent)
                parent = _normalize(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent (symmetrized): n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _symmetrize(_make_profile(n_parent, "cos_taper"))
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, cos_taper n={n_parent}, score={best_score:.10f}")

    n = len(best_seq)

    # === Phase 1: L-BFGS warm-up (300s) ===
    t_phase1 = min(300, deadline - t_start - 800)
    print(f"\n[P1] L-BFGS symmetric (budget={t_phase1:.0f}s) t={time.monotonic()-t_start:.1f}s")
    if t_phase1 > 30:
        a1 = _run_lbfgs_sym(best_seq, t_start + t_phase1, min_time_remaining=600)
        sc1 = _score(a1)
        if sc1 < best_score:
            best_seq = a1.copy()
            best_score = sc1
            print(f"  NEW BEST: {best_score:.10f} [P1] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Subgradient + LP (500s) ===
    t_phase2 = min(500, deadline - t_start - 500)
    print(f"\n[P2] Subgradient+LP (budget={t_phase2:.0f}s) t={time.monotonic()-t_start:.1f}s")
    if t_phase2 > 60:
        a2, sc2 = _subgradient_polyak(best_seq, t_start + t_phase2, min_time_remaining=400,
                                      use_sym=True, lp_every=200)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [P2] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Fine polish (250s) ===
    t_phase3 = min(250, deadline - t_start - 250)
    print(f"\n[P3] Fine polish (budget={t_phase3:.0f}s) t={time.monotonic()-t_start:.1f}s")
    if t_phase3 > 30:
        a3, sc3 = _fine_polish(best_seq, t_start + t_phase3, min_time_remaining=100)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [P3] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: CG escape + second subgradient (200s) ===
    t_phase4 = min(200, deadline - t_start - 150)
    print(f"\n[P4] CG + Subgradient (budget={t_phase4:.0f}s) t={time.monotonic()-t_start:.1f}s")
    if t_phase4 > 40:
        # CG escape
        t_cg = max(30, t_phase4 * 0.4)
        a4a = _conjugate_gradient(best_seq, t_start + t_cg, min_time_remaining=100)
        sc4a = _score(a4a)
        if sc4a < best_score:
            best_seq = a4a.copy()
            best_score = sc4a
            print(f"  NEW BEST: {best_score:.10f} [P4-CG] t={time.monotonic()-t_start:.1f}s")

        # Continue subgradient
        t_sub = deadline - t_start - 150 - (time.monotonic() - t_start)
        if t_sub > 30:
            a4b, sc4b = _subgradient_polyak(best_seq, t_start + t_sub + 150,
                                            min_time_remaining=120, use_sym=True,
                                            lp_every=400)
            if sc4b < best_score:
                best_seq = a4b.copy()
                best_score = sc4b
                print(f"  NEW BEST: {best_score:.10f} [P4-SUB] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Peak shaving (150s) ===
    t_phase5 = min(150, deadline - t_start - 100)
    print(f"\n[P5] Peak shave (budget={t_phase5:.0f}s) t={time.monotonic()-t_start:.1f}s")
    if t_phase5 > 30:
        a5, sc5 = _direct_peak_shave(best_seq, t_start + t_phase5, min_time_remaining=60)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [P5] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-restart (150s) ===
    t_phase6 = min(150, deadline - t_start - 60)
    print(f"\n[P6] Multi-restart (budget={t_phase6:.0f}s) t={time.monotonic()-t_start:.1f}s")
    if t_phase6 > 40:
        sigmas = [0.02, 0.04, 0.06, 0.03, 0.08, 0.05]
        n_restarts = min(len(sigmas), max(2, int(t_phase6 / 25)))
        for i in range(n_restarts):
            if time.monotonic() > deadline - 50:
                break
            a_pert = _perturb(best_seq, sigmas[i], rng)
            a_pert = _symmetrize(a_pert)
            a_r = _run_lbfgs_sym(a_pert, min(deadline - 40, time.monotonic() + 20),
                                 min_time_remaining=5)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.10f} [P6-r{i}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Try different n (60s) ===
    t_phase7 = min(60, deadline - t_start - 30)
    print(f"\n[P7] Multi-n (budget={t_phase7:.0f}s) t={time.monotonic()-t_start:.1f}s")
    if t_phase7 > 20:
        n_main = len(best_seq)
        for n_try in [n_main * 2, n_main * 3 // 2]:
            if time.monotonic() > deadline - 20:
                break
            if n_try > 65536 or n_try < 1024:
                continue
            a_res = _resample(best_seq, n_try)
            a_res = _symmetrize(a_res)
            a_opt = _run_lbfgs_sym(a_res, min(deadline - 10, time.monotonic() + 25),
                                   min_time_remaining=5)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [P7 n={n_try}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final fine polish (remaining) ===
    print(f"\n[P8] Final polish t={time.monotonic()-t_start:.1f}s")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _fine_polish(best_seq, deadline - 3, min_time_remaining=3)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [P8] t={time.monotonic()-t_start:.1f}s")

        # Tiny L-BFGS at the end
        remaining = deadline - 3 - time.monotonic()
        if remaining > 8:
            n_fin = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = max(mb * 0.0001, 1e-10)
            K_fin = max(3, n_fin // 50000)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _SmooProb(beta_fin, K_fin, n_fin, symmetric=True)
            max_iter = min(2000, max(20, int(remaining / 0.02)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-20,
                                           'gtol': 1e-16, 'maxcor': 100})
                a_fin2 = np.exp(result.x - result.x.max())
                a_fin2 = a_fin2 / a_fin2.sum()
                a_fin2 = _symmetrize(a_fin2)
                a_fin2 = _normalize(a_fin2)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [P8-lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
