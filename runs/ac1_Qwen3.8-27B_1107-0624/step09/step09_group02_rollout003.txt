import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _compute_gradient_weights(a, beta, n):
    """Compute w[k] = softmax(c/beta) where c = conv(a,a)."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s_e = e.sum()
    if s_e < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s_e
    return w, c, c_max


def _grad_smoothed(z, beta, n):
    """Gradient of the smoothed objective f_beta(z) = 2n*(c_max + beta*log(sum exp((c-c_max)/beta)))."""
    a = _softmax(z)
    w, c, c_max = _compute_gradient_weights(a, beta, n)
    # G[m] = sum_k w[k] * dc[k]/da[m] = 2 * sum_j w[m+j] * a[j] (for 0<=j<n, 0<=m+j<2n-1)
    # This is cross-correlation of w and a
    # G[m] = 2 * conv(w_rev, a)[m + 2n - 2]
    w_rev = w[::-1]
    conv_len = len(w_rev) + len(a) - 1
    N = _next_pow2(conv_len)
    Fw = np.fft.rfft(w_rev, N)
    Fa = np.fft.rfft(a, N)
    conv_full = np.fft.irfft(Fw * Fa, N)[:conv_len]
    G = 2.0 * conv_full[2 * n - 2:3 * n - 2]
    # Gradient: df/dz_j = 2n * a[j] * (G[j] - dot(a, G))
    W_val = float(np.dot(a, G))
    return 4.0 * n * a * (G - W_val), a, c, c_max


def _smoothed_objective(z, beta, n):
    a = _softmax(z)
    w, c, c_max = _compute_gradient_weights(a, beta, n)
    return 2.0 * n * (c_max + beta * np.log(w.sum() * np.exp((c_max - c_max) / beta)))
    # Actually: 2n * (c_max + beta * log(sum exp((c-c_max)/beta)))
    # = 2n * (c_max + beta * log(s_e)) where s_e = sum exp((c-c_max)/beta)

def _run_pair_swap(a_init, deadline, min_remaining=30.0):
    """Pair-swap refinement: try moving mass between pairs of coordinates."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    a = a / a.sum()

    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())

    n_iter = 0

    if max_time < 5:
        return best_a, best_sc

    # Estimate time per iteration (one convolution)
    test_start = time.monotonic()
    _ = _full_convolve(a, a)
    conv_time = time.monotonic() - test_start
    max_iter = int(max_time / (conv_time * 2))
    max_iter = min(max_iter, 50000)

    for it in range(max_iter):
        if time.monotonic() + min_remaining > deadline:
            break
        # Pick two random indices
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Try moving epsilon from i to j
        eps = a[i] * 0.05
        if a[i] < eps:
            eps = a[i] * 0.5
        if eps < 1e-15:
            continue

        a_try = a.copy()
        a_try[i] -= eps
        a_try[j] += eps

        c_try = _full_convolve(a_try, a_try)
        if c_try.max() < c_max - 1e-15:
            a = a_try
            c = c_try
            c_max = float(c_try.max())
            n_iter += 1
            if n_iter % 100 == 0:
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
        else:
            # Try reverse: move from j to i
            eps2 = a[j] * 0.05
            if a[j] >= eps2 and eps2 > 1e-15:
                a_try2 = a.copy()
                a_try2[j] -= eps2
                a_try2[i] += eps2
                c_try2 = _full_convolve(a_try2, a_try2)
                if c_try2.max() < c_max - 1e-15:
                    a = a_try2
                    c = c_try2
                    c_max = float(c_try2.max())
                    n_iter += 1

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _run_active_set_qp(a_init, deadline, min_remaining=30.0, max_rounds=100):

    n = len(a_init)

                    0.0001, 0.00005, 0.00002, 0.00001, 0.000005,
                    0.000002, 0.000001, 0.0000005, 0.0000002,
                    0.0000001, 0.00000005, 0.00000002, 0.00000001]
    tol_idx = 0

    for rnd in range(max_rounds):
        if time.monotonic() + min_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / active.sum()

        # Solve: min sum_k p[k] c[k] s.t. a in simplex
        # Using projected gradient with backtracking
        z = np.log(np.maximum(a, 1e-30))
        z = z - z.mean()

        def qp_obj(z_local):
            a_local = _softmax(z_local)
            c_local = _full_convolve(a_local, a_local)
            return float(2.0 * n * np.dot(p, c_local))

            a_local = _softmax(z_local)
            p_rev = p[::-1]
            conv_len = len(p_rev) + len(a_local) - 1
            N = _next_pow2(conv_len)
            Fp = np.fft.rfft(p_rev, N)
            Fa = np.fft.rfft(a_local, N)

        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        est_time = 0.005 * max(1.0, n / 32768.0)
        max_iter = min(2000, max(30, int(remaining / est_time / 20)))

        try:
            result = minimize(
                qp_obj, z,

            a_new = _softmax(result.x)

                best_a = a_new.copy()
                a = a_new
            else:
                a = best_a.copy()
        except Exception:
            break

        if (rnd + 1) % 3 == 0:
            tol_idx += 1

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * t
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        taper = max(1, n // 20)
        for i in range(taper):
            frac = (i + 1) / taper
            a[i] = frac
            a[n - 1 - i] = frac
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):

    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()

        print(f"No parent, using tapered n={n_parent}", flush=True)
        update_best(parent, _score(parent), "init")

    n_main = n_parent

    # === Phase 1: Multi-start L-BFGS with continuation (350s) ===
    print(f"\n[Phase 1] Multi-start L-BFGS continuation (t={time.monotonic() - t_start:.1f}s)", flush=True)
    phase1_deadline = t_start + 350
    candidates = []

    # Start from parent
    a1 = _run_lbfgs_continuation(parent, phase1_deadline, min_remaining=30.0)
    update_best(a1, _score(a1), "lbfgs_parent")
    candidates.append(a1)

    # Try other profiles at same n
    profiles_to_try = ["flat_top", "bimodal", "cos", "bump"]
    for prof in profiles_to_try:
        if time.monotonic() > phase1_deadline - 60:
            break
        a_init = _make_profile(n_main, prof)
        # Perturb slightly to avoid exact symmetry
        a_init = _perturb(a_init, 0.01, rng)
        a_p = _run_lbfgs_continuation(a_init, phase1_deadline, min_remaining=20.0)
        sc_p = _score(a_p)
        if sc_p < best_score + 0.05:
            update_best(a_p, sc_p, f"lbfgs_{prof}")
            candidates.append(a_p)

    # Perturbation of best
    if time.monotonic() < phase1_deadline - 30 and best_seq is not None:
        for sigma in [0.02, 0.05]:
            if time.monotonic() > phase1_deadline - 20:
                break
            a_pert = _perturb(best_seq, sigma, rng)
            a_p = _run_lbfgs_continuation(a_pert, phase1_deadline, min_remaining=10.0)
            update_best(a_p, _score(a_p), f"lbfgs_pert_{sigma}")

    # === Phase 2: Adversarial primal-dual (400s) ===
    print(f"\n[Phase 2] Adversarial primal-dual (t={time.monotonic() - t_start:.1f}s)", flush=True)
    phase2_deadline = min(deadline - 800, phase1_deadline + 400)
    if best_seq is not None and time.monotonic() < phase2_deadline:
        a_adv, sc_adv = _run_adversarial(best_seq, phase2_deadline, min_remaining=30.0)
        update_best(a_adv, sc_adv, "adversarial_1")

    # === Phase 3: Active-set QP (300s) ===
    print(f"\n[Phase 3] Active-set QP (t={time.monotonic() - t_start:.1f}s)", flush=True)
    phase3_deadline = min(deadline - 500, phase2_deadline + 300)
    if best_seq is not None and time.monotonic() < phase3_deadline:
        a_qp, sc_qp = _run_active_set_qp(best_seq, phase3_deadline,
                                          min_remaining=30.0, max_rounds=80)
        update_best(a_qp, sc_qp, "active_qp_1")

    # === Phase 4: Pair-swap refinement (250s) ===
    print(f"\n[Phase 4] Pair-swap refinement (t={time.monotonic() - t_start:.1f}s)", flush=True)
    phase4_deadline = min(deadline - 250, phase3_deadline + 250)
    if best_seq is not None and time.monotonic() < phase4_deadline:
        a_ps, sc_ps = _run_pair_swap(best_seq, phase4_deadline, min_remaining=20.0)
        update_best(a_ps, sc_ps, "pair_swap_1")

    # === Phase 5: Second round of combined optimization (200s) ===
    print(f"\n[Phase 5] Combined round 2 (t={time.monotonic() - t_start:.1f}s)", flush=True)
    phase5_deadline = min(deadline - 150, phase4_deadline + 200)
    if best_seq is not None and time.monotonic() < phase5_deadline:
        a_lb2 = _run_lbfgs_continuation(best_seq, phase5_deadline, min_remaining=30.0)
        update_best(a_lb2, _score(a_lb2), "lbfgs_r2")

        if time.monotonic() < phase5_deadline - 30:
            a_qp2, sc_qp2 = _run_active_set_qp(best_seq, phase5_deadline,
                                                min_remaining=20.0, max_rounds=60)
            update_best(a_qp2, sc_qp2, "active_qp_2")

        if time.monotonic() < phase5_deadline - 20:
            a_ps2, sc_ps2 = _run_pair_swap(best_seq, phase5_deadline, min_remaining=10.0)
            update_best(a_ps2, sc_ps2, "pair_swap_2")

    # === Phase 6: Multi-n exploration (150s) ===
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)", flush=True)
    phase6_deadline = min(deadline - 100, phase5_deadline + 150)
    n_tries = [int(n_main * 1.5), int(n_main * 0.75), int(n_main * 2)]
    # Round to powers of 2 for FFT efficiency
    n_tries = sorted(set([max(1024, min(65536, _next_pow2(nt))) for nt in n_tries]))
    for n_try in n_tries[:3]:
        if time.monotonic() > phase6_deadline - 30:
            break
        if n_try == n_main:
            continue
        if best_seq is not None:
            a_init_n = _resample(best_seq, n_try)
        else:
            a_init_n = _make_profile(n_try, "tapered")
        a_n = _run_lbfgs_continuation(a_init_n, phase6_deadline, min_remaining=20.0)
        sc_n = _score(a_n)
        update_best(a_n, sc_n, f"n_{n_try}")

    # === Phase 7: Final ultra-tight polish (100s) ===
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 15:
        # Very tight active set
        a_fin, sc_fin = _run_active_set_qp(best_seq, deadline - 20,
                                            min_remaining=5.0, max_rounds=50)
        update_best(a_fin, sc_fin, "final_qp")

        if time.monotonic() < deadline - 15:
            a_ps3, sc_ps3 = _run_pair_swap(best_seq, deadline - 15, min_remaining=5.0)
            update_best(a_ps3, sc_ps3, "final_ps")

        # Final L-BFGS with extremely tight beta
        if time.monotonic() < deadline - 10:
            a_fin2 = _run_lbfgs_continuation(best_seq, deadline - 10, min_remaining=5.0)
            update_best(a_fin2, _score(a_fin2), "final_lbfgs")

    # Final validation
    print(f"\n{'=' * 60}", flush=True)
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s", flush=True)
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
