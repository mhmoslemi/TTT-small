**Idea: Use the structure of the Hessian for a Newton-like step**

The Hessian of the power-sum objective is:
$H_{ij} = 2p \sum_k (p-1) c_k^{p-2} \frac{\partial c_k}{\partial a_i} \frac{\partial c_k}{\partial a_j} + 2p \sum_k c_k^{p-1} \frac{\partial^2 c_k}{\partial a_i \partial a_j}$

The second term involves $\frac{\partial^2 c_k}{\partial a_i \partial a_j} = \delta_{i, k-j} + \delta_{j, k-i}$ (Kronecker deltas), which is sparse.

The first term is $\sim p^2 \sum_k c_k^{p-2} \cdot 4 a_{k-i} a_{k-j}$, which is a "weighted correlation" and can be computed via FFT.

For large $p$, the Hessian is dominated by the peaks (where $c_k$ is large). So the Hessian is approximately:
$H \approx 4p^2 c_{k^*}^{p-2} \cdot \text{corr}(a_{k^*-i}, a_{k^*-j})$

where $k^*$ is the peak position. This is a rank-1 matrix (approximately), so a Newton step would be very specific.

Actually, this suggests that for large $p$, the optimization is essentially trying to reduce the single peak at $k^*$, and the optimal direction is determined by the gradient at that peak. This is exactly what the active-set subgradient method does.

"""
Optimized multi-phase approach for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous approach:
- Power-sum objective (sharper than log-sum-exp for high p)
- Projected gradient phase in a-space (avoids log-domain numerical issues)
- More aggressive active-set refinement
- Better restart strategy with structured perturbations
- Faster inner loops (reduced line search)

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_conv(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _project_simplex(v):
    """Project v onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


class _PowerSumProb:
    """Power-sum smoothed objective in log domain."""
    def __init__(self, p, n):
        self.p = p
        self.n = n
    
    def __call__(self, z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        c = _full_convolve(a, a)
        c_safe = np.maximum(c, 1e-300)
        return float(np.sum(c_safe ** self.p))
    
    def grad(self, z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        c = _full_convolve(a, a)
        c_safe = np.maximum(c, 1e-300)
        cp1 = c_safe ** (self.p - 1)
        a_rev = a[::-1]
        g_a = 2.0 * self.p * _full_convolve(cp1, a_rev)[self.n - 1:2 * self.n - 1]
        # Chain rule through softmax
        W = float(np.dot(a, g_a))
        grad_z = a * (g_a - W)
        return grad_z


class _TopkLogSumExpProb:
    """Top-K log-sum-exp smoothed objective in log domain."""
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n
    
    def __call__(self, z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if self.K < len(c):
            top_idx = np.argpartition(c, -self.K)[-self.K:]
            c_top = c[top_idx]
        else:
            top_idx = np.arange(len(c))
            c_top = c
        shifted = (c_top - c_max) / self.beta
        shifted = np.clip(shifted, -700, 0)
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300:
            obj = 2.0 * self.n * c_max
        else:
            obj = 2.0 * self.n * (c_max + self.beta * np.log(s))
        return obj
    
    def grad(self, z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if self.K < len(c):
            top_idx = np.argpartition(c, -self.K)[-self.K:]
            c_top = c[top_idx]
        else:
            top_idx = np.arange(len(c))
            c_top = c
        shifted = (c_top - c_max) / self.beta
        shifted = np.clip(shifted, -700, 0)
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300:
            w = np.zeros(2 * self.n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            e_full = np.zeros(2 * self.n - 1)
            e_full[top_idx] = e
            w = e_full / s
        ga = _grad_conv(a, w)
        W = float(np.dot(a, ga))
        grad_z = a * (ga - W)
        return grad_z

    """L-BFGS in log domain."""

    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a
    
    if method == 'power_sum':
        prob = _PowerSumProb(p, n)
    else:
