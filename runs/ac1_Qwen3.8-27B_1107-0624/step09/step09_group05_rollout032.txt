"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations:
- Fast variance minimization pre-step (projected gradient on sum(conv^2))
- Aggressive soft-max L-BFGS with structured FFT gradient
- KKT active-set refinement with adaptive tolerance
- Multi-profile and multi-n exploration
- Direct peak-reduction heuristic
- Careful time management to maximize iterations at small beta
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_soft_structured(a, beta, n):
    """Gradient of soft_max(conv(a,a)) using FFT structure."""
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _grad_variance(a, n):
    """Gradient of sum(conv(a,a)^2) = 4 * conv(C, a_rev)[n-1:2n-1]."""
    c = _full_convolve(a, a)
    a_rev = a[::-1]
    gc = _full_convolve(c, a_rev)
    return 4.0 * gc[n - 1:2 * n - 1]


def _variance_minimize(a_init, max_iter=500, deadline=None, min_time_remaining=10.0):
    """Projected gradient on variance of convolution."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.1
    decay = 0.999
    momentum = 0.9
    v = np.zeros(n)

    for it in range(max_iter):
        if deadline is not None and time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        grad = _grad_variance(a, n)
        # Center the gradient (remove mean to maintain sum constraint approximately)
        grad_cent = grad - np.mean(grad)

        v = momentum * v + (1 - momentum) * grad_cent
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new
        lr *= decay

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft_structured(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)


class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0, verbose=True, max_beta_factor=0.01):

    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    # Compute initial gap for beta scheduling
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(200, len(c_sorted) - 1)]
    if gap < c0_max * 0.005:
        gap = c0_max * 0.005

    # More aggressive beta levels
    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
        0.00003125, 0.000015625, 0.0000078, 0.0000039,
        0.000002, 0.000001, 0.0000005, 0.0000002
    ])
    betas = gap * multipliers

    t_start = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Estimate iteration time and set max iterations

        est_iter_time = 0.003 * size_factor
        max_iter = min(10000, max(200, int(remaining / (est_iter_time * 30))))

        else:
            max_iter = min(max_iter, 8000)

        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 200,
                    'maxls': 100,
                }

            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:

        except Exception as e:

        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.5)
            eta = max(0.01, eta * 0.7)
            if tol_frac <= min_tol * 1.05:
                break
            no_improve = 0

        # Identify active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # If too many active, tighten
        if n_active > n * 0.3:
            tol *= 0.1
            active = c >= (c_max - tol)
            n_active = int(active.sum())
            if n_active == 0:
                active[int(np.argmax(c))] = True
                n_active = 1

        # Compute average gradient from active set
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]

        # Support of a
        support = a > 1e-10 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        # KKT: grad should be constant on support
        lam = float(np.mean(grad_avg[support]))
        residual = grad_avg[support] - lam

        # Update
        a_new = a.copy()
        a_new[support] -= eta * residual
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Line search
        sc_new = _score(a_new)
        if sc_new >= current_sc:
            found = False
            for trial_eta in [eta * 0.5, eta * 0.25, eta * 0.1, 0.005]:
                a_try = a.copy()
                a_try[support] -= trial_eta * residual
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-14:
                    a_new = a_try
                    sc_new = sc_try
                    found = True
                    break
            if not found:
                tol_frac = max(min_tol, tol_frac * 0.85)
                eta = max(0.005, eta * 0.8)
                continue

        a = a_new

        if no_improve == 0:
            tol_frac = max(min_tol, tol_frac * 0.98)
            eta = min(0.3, eta * 1.05)

    return best_a, best_sc


def _peak_reduction(a_init, deadline, min_time_remaining=20.0):
    """Direct peak reduction: identify the top peak and reduce its contributors."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_peak = int(np.argmax(c))

        # Identify which a[j] contribute most to c[k_peak]
        # c[k_peak] = sum_j a[j] * a[k_peak - j]
        # Contribution of a[j] to c[k_peak] is a[j]*a[k_peak-j] (and a[k_peak-j]*a[j])
        j_lo = max(0, k_peak - (n - 1))
        j_hi = min(n - 1, k_peak)
        contrib = np.zeros(n)
        for j in range(j_lo, j_hi + 1):
            contrib[j] = 2.0 * a[j] * a[k_peak - j]

        # Reduce elements with highest contribution
        threshold = 0.3 * contrib.max()
        to_reduce = contrib > threshold
        n_reduce = int(to_reduce.sum())
        if n_reduce == 0:
            break

        # Reduce proportionally
        reduction = 0.05
        a_new = a.copy()
        a_new[to_reduce] *= (1.0 - reduction)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller reduction
            reduction2 = 0.02
            a_try = a.copy()
            a_try[to_reduce] *= (1.0 - reduction2)
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                break
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
            else:
                break

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-50.0 * (t - 0.2) ** 2) + np.exp(-50.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.4 * np.sin(2.0 * np.pi * 13.0 * t / n)
             + 0.2 * np.cos(2.0 * np.pi * 7.0 * t / n)
             + 0.15 * np.sin(2.0 * np.pi * 19.0 * t / n)
             + 0.1 * np.cos(2.0 * np.pi * 3.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "one_sided":
        t = np.linspace(0, 1, n)
        a = np.exp(-5.0 * t) + 0.01
        return a / a.sum()
    elif kind == "asymmetric_peak":
        t = np.linspace(0, 1, n)
        # Asymmetric: peak slightly off-center
        a = np.exp(-30.0 * (t - 0.35) ** 2) + 0.02 * np.exp(-30.0 * (t - 0.7) ** 2) + 0.005
        return a / a.sum()
    elif kind == "comb":
        t = np.arange(n)
        period = max(3, n // 100)
        a = np.ones(n)
        a[::period] += 1.0
        a[1::period] += 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sawtooth":
        t = np.linspace(0, 1, n)
        a = np.abs(t - np.floor(t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "noise":
        rng = np.random.default_rng(12345)
        a = np.abs(rng.normal(0, 1, n)) + 0.1
        return a / a.sum()
    elif kind == "quadratic_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.8 * (2 * t - 0.5) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


_t_start_global = 0.0


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining variance minimization, L-BFGS,
    active-set refinement, and peak reduction.
    """
    global _t_start_global
    _t_start_global = time.monotonic()
    deadline = _t_start_global + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(16384, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n=16384: score={best_score:.10f}")

    n_main = len(best_seq)

    # === Phase 1: Variance minimization (fast pre-step) ===
    print(f"\n[P1] Variance minimize (n={n_main}, t={time.monotonic()-_t_start_global:.0f}s)")
    p1_end = min(deadline - 600, _t_start_global + 60)
    a1, sc1 = _variance_minimize(best_seq, max_iter=1000, deadline=p1_end, min_time_remaining=10)
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  P1 result: {best_score:.10f}")

    # === Phase 2: Main L-BFGS ===
    print(f"\n[P2] L-BFGS main (n={n_main}, t={time.monotonic()-_t_start_global:.0f}s)")
    p2_end = min(deadline - 400, _t_start_global + 500)
    a2, sc2 = _run_lbfgs(best_seq, p2_end, min_time_remaining=300, verbose=True)
    if sc2 < best_score:
        best_score = sc2
        best_seq = a2.copy()
        print(f"  P2 result: {best_score:.10f}")

    # === Phase 3: Active-set refinement ===
    print(f"\n[P3] Active-set refine (t={time.monotonic()-_t_start_global:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p3_end = min(deadline - 300, time.monotonic() + 200)
        a3, sc3 = _active_set_refine(best_seq, p3_end, min_time_remaining=50)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3 result: {best_score:.10f}")

    # === Phase 4: L-BFGS round 2 (tighter beta) ===
    print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-_t_start_global:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p4_end = min(deadline - 250, time.monotonic() + 300)
        a4, sc4 = _run_lbfgs(best_seq, p4_end, min_time_remaining=150, verbose=True)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4 result: {best_score:.10f}")

    # === Phase 5: Peak reduction ===
    print(f"\n[P5] Peak reduction (t={time.monotonic()-_t_start_global:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p5_end = min(deadline - 200, time.monotonic() + 80)
        a5, sc5 = _peak_reduction(best_seq, p5_end, min_time_remaining=30)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  P5 result: {best_score:.10f}")

    # === Phase 6: Active-set + L-BFGS cycle ===
    print(f"\n[P6] Refine cycle (t={time.monotonic()-_t_start_global:.0f}s)")
    if time.monotonic() + 40 < deadline:
        p6a_end = min(deadline - 150, time.monotonic() + 100)
        a6a, sc6a = _active_set_refine(best_seq, p6a_end, min_time_remaining=30)
        if sc6a < best_score:
            best_score = sc6a
            best_seq = a6a.copy()

        if time.monotonic() + 20 < deadline:
            p6b_end = min(deadline - 80, time.monotonic() + 80)
            a6b, sc6b = _run_lbfgs(best_seq, p6b_end, min_time_remaining=50, verbose=False)
            if sc6b < best_score:
                best_score = sc6b
                best_seq = a6b.copy()
                print(f"  P6b: {best_score:.10f}")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n (t={time.monotonic()-_t_start_global:.0f}s)")
    n_options = [12288, 14336, 16384, 18432, 20480]
    n_options = [x for x in n_options if abs(x - n_main) > 500]
    for n_try in n_options[:3]:
        if time.monotonic() + 80 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        print(f"  n={n_try} resampled: {sc_r:.10f}")
        if time.monotonic() + 60 > deadline:
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
            continue
        p7_end = min(deadline - 60, time.monotonic() + 80)
        a7, sc7 = _run_lbfgs(a_r, p7_end, min_time_remaining=30, verbose=False)
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()
            print(f"  n={n_try} optimized: {best_score:.10f}")

    # === Phase 8: Diverse profile search ===
    print(f"\n[P8] Diverse profiles (t={time.monotonic()-_t_start_global:.0f}s)")
    profiles = ["asymmetric_peak", "periodic", "comb", "one_sided"]
    n_for_profiles = len(best_seq)
    for prof in profiles:
        if time.monotonic() + 50 > deadline:
            break
        a_p = _make_profile(n_for_profiles, prof)
        sc_p = _score(a_p)
        if sc_p < best_score + 0.3:
            p8_end = min(deadline - 30, time.monotonic() + 60)
            a_p_opt, sc_p_opt = _run_lbfgs(a_p, p8_end, min_time_remaining=20, verbose=False)
            if sc_p_opt < best_score:
                best_score = sc_p_opt
                best_seq = a_p_opt.copy()
                print(f"  {prof}: {best_score:.10f}")

    # === Phase 9: Final polish ===
    print(f"\n[P9] Final polish (t={time.monotonic()-_t_start_global:.0f}s)")
    if time.monotonic() + 20 < deadline:
        p9_end = min(deadline - 5, time.monotonic() + 120)
        a9, sc9 = _run_lbfgs(best_seq, p9_end, min_time_remaining=5, verbose=False)
        if sc9 < best_score:
            best_score = sc9
            best_seq = a9.copy()

        if time.monotonic() + 10 < deadline:
            p9b_end = min(deadline - 3, time.monotonic() + 30)
            a9b, sc9b = _active_set_refine(best_seq, p9b_end, min_time_remaining=3)
            if sc9b < best_score:
                best_score = sc9b
                best_seq = a9b.copy()

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    # Scale to ensure sum is reasonable
    if s != 0:
        final = final * (1.0 / s)  # normalize to sum=1
    sc_final = _score(final)
    # Verify with evaluate_sequence
    try:
        sc_eval = evaluate_sequence([float(x) for x in final])
        print(f"Score (internal): {sc_final:.10f}, Score (evaluate_sequence): {sc_eval:.10f}")
        if sc_eval < best_score:
            best_score = sc_eval
    except Exception:
        pass
    print(f"FINAL: score={best_score:.10f}, n={len(final)}, time={time.monotonic()-_t_start_global:.1f}s")
    return [float(x) for x in final]
