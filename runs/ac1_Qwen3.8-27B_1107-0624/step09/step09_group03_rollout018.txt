def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _lsobj_factory(z, beta, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad_factory(z, beta, n):

def _pmobj_factory(z, p, n):

def _cg_solve(matvec, rhs, max_cg=200, tol=1e-13):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _newton_step(a, c, c_max, tol_frac):
    n = len(a)
    active = c >= (c_max - tol_frac * c_max)
    if not active.any():
        active[np.argmax(c)] = True
    m = int(active.sum())
    if m < 2 or m > 10 * n:
        return None, None

    A_vec = active.astype(np.float64) / m
    g = 2.0 * _corr_a_A(a, A_vec, n)
    g_t = g - np.mean(g)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-14:
        return None, None

    mu = 1e-7 / m + 1e-10
    A_vec_copy = A_vec
    n_copy = n
    mu_copy = mu

    def _hess_vp(v):
        return 2.0 * _corr_a_A(v, A_vec_copy, n_copy) + mu_copy * v

    u = _cg_solve(_hess_vp, g_t, max_cg=200)
    w = _cg_solve(_hess_vp, np.ones(n), max_cg=200)
    w_sum = float(w.sum())

    if abs(w_sum) < 1e-15:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w

    dx = dx - np.mean(dx)
    return dx, A_vec

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block2":
        a = np.zeros(n)
        block = n // 3
        a[:block] = 1.0
        a[2 * block:] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        return a
    elif kind == "multi_block":
        a = np.zeros(n)
        n_blocks = min(n // 4, 8)
        block_size = max(1, n // (2 * n_blocks))
        for i in range(n_blocks):
            start = i * (n // n_blocks)
            end = min(start + block_size, n)
            a[start:end] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        return a
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.25) ** 2) + np.exp(-20 * (t - 0.75) ** 2)
        return a / a.sum()
    elif kind == "comb":
        a = np.zeros(n)
        period = max(4, n // 16)
        a[::period] = 1.0
        # Smooth slightly
        kernel = np.array([0.25, 0.5, 0.25])
        a = np.convolve(a, kernel, mode='same')
        return a / a.sum()
    else:
        return np.ones(n) / n

def _interpolate_seq(a, n_new):
    n_old = len(a)

    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.12f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent score: {sc_parent:.12f}")
        update_best(parent, sc_parent, "parent")
    else:
        parent = _make_profile(16384, "uniform")
        update_best(parent, _score(parent), "init_uniform")

    # === Phase 1: Quick exploration with diverse inits (200s) ===
    print(f"\n[Phase 1] Multi-init exploration (t={time.monotonic() - t_start:.1f}s)")
    n_work = len(parent)
    p1_deadline = min(deadline - 800, t_start + 200)

    inits = [
        ("parent", parent.copy()),
        ("uniform", _make_profile(n_work, "uniform")),
        ("raised_cos", _make_profile(n_work, "raised_cosine")),
        ("tapered", _make_profile(n_work, "tapered")),
        ("bimodal", _make_profile(n_work, "bimodal")),
        ("multi_block", _make_profile(n_work, "multi_block")),
        ("comb", _make_profile(n_work, "comb")),
    ]

    time_per_init = (p1_deadline - time.monotonic()) / len(inits)
    for name, init_a in inits:
        if time.monotonic() > p1_deadline:
            break
        init_dl = min(p1_deadline, time.monotonic() + max(15, time_per_init))
        print(f"  Init: {name}")
        # Quick L-BFGS
        a_opt = _lbfgs_continuation(init_a, init_dl, min_time_remaining=5, n_betas=20)
        update_best(a_opt, _score(a_opt), f"p1_{name}")

    # === Phase 2: Aggressive L-BFGS continuation from best (500s) ===
    print(f"\n[Phase 2] Aggressive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    p2_deadline = min(deadline - 500, t_start + 700)
    if best_seq is not None:
        a_agg = _lbfgs_continuation(best_seq, p2_deadline, min_time_remaining=500, n_betas=50)
        update_best(a_agg, _score(a_agg), "agg_lbfgs")

    # === Phase 3: Power mean continuation (150s) ===
    print(f"\n[Phase 3] Power mean continuation (t={time.monotonic() - t_start:.1f}s)")
    p3_deadline = min(deadline - 300, t_start + 850)
    if best_seq is not None:
        a_pm = _lbfgs_power_mean(best_seq, p3_deadline, min_time_remaining=300)
        update_best(a_pm, _score(a_pm), "power_mean")

    # === Phase 4: Active-set Newton (150s) ===
    print(f"\n[Phase 4] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    p4_deadline = min(deadline - 250, t_start + 1000)
    if best_seq is not None:
        a_as, sc_as = _active_set_newton(best_seq, p4_deadline, max_outer=100)
        update_best(a_as, sc_as, "active_set")

    # === Phase 5: Subgradient with momentum (100s) ===
    print(f"\n[Phase 5] Subgradient momentum (t={time.monotonic() - t_start:.1f}s)")
    p5_deadline = min(deadline - 200, t_start + 1100)
    if best_seq is not None:
        # Patch the global time reference for subgrad
        global t_sub_start
        t_sub_start = time.monotonic()
        a_sg, sc_sg = _subgrad_momentum(best_seq, p5_deadline, max_iter=3000)
        update_best(a_sg, sc_sg, "subgrad")

    # === Phase 6: Deflation (80s) ===
    print(f"\n[Phase 6] Deflation (t={time.monotonic() - t_start:.1f}s)")
    p6_deadline = min(deadline - 150, t_start + 1180)
    if best_seq is not None:
        a_def, sc_def = _deflation_step(best_seq, p6_deadline, n_attempts=40)
        update_best(a_def, sc_def, "deflation")

    # === Phase 7: Multi-scale - try larger n (150s) ===
    print(f"\n[Phase 7] Multi-scale (t={time.monotonic() - t_start:.1f}s)")
    p7_deadline = min(deadline - 100, t_start + 1330)
    if best_seq is not None and time.monotonic() < p7_deadline:
        for n_large in [32768, 65536]:
            if time.monotonic() > p7_deadline - 50:
                break
            a_large = _interpolate_seq(best_seq, n_large)
            sc_init = _score(a_large)
            print(f"  n={n_large} init: {sc_init:.12f}")
            dl = min(p7_deadline, time.monotonic() + 60)
            a_large_opt = _lbfgs_continuation(a_large, dl, min_time_remaining=30, n_betas=15)
            update_best(a_large_opt, _score(a_large_opt), f"n_{n_large}")

    # === Phase 8: Iterative refinement loop (150s) ===
    print(f"\n[Phase 8] Refinement loop (t={time.monotonic() - t_start:.1f}s)")
    p8_deadline = min(deadline - 50, t_start + 1480)
    if best_seq is not None:
        for rnd in range(6):
            if time.monotonic() > p8_deadline:
                break
            # Newton
            dl1 = min(p8_deadline - 20, time.monotonic() + 25)
            a_n, sc_n = _active_set_newton(best_seq, dl1, max_outer=30)
            if sc_n < best_score:
                update_best(a_n, sc_n, f"ref_newton_{rnd}")
            # Deflation
            dl2 = min(p8_deadline - 10, time.monotonic() + 15)
            a_d, sc_d = _deflation_step(best_seq, dl2, n_attempts=15)
            if sc_d < best_score:
                update_best(a_d, sc_d, f"ref_def_{rnd}")
            # Quick L-BFGS
            dl3 = min(p8_deadline, time.monotonic() + 25)
            a_l = _lbfgs_continuation(best_seq, dl3, min_time_remaining=10, n_betas=10)
            update_best(a_l, _score(a_l), f"ref_lbfgs_{rnd}")

    # === Phase 9: Final restarts (100s) ===
    print(f"\n[Phase 9] Final restarts (t={time.monotonic() - t_start:.1f}s)")
    p9_deadline = deadline - 15
    if best_seq is not None:
        for sigma in [0.01, 0.03, 0.08]:
            if time.monotonic() > p9_deadline:
                break
            p_start = _perturb(best_seq, sigma, rng)
            dl = min(p9_deadline, time.monotonic() + 30)
            a_r = _lbfgs_continuation(p_start, dl, min_time_remaining=10, n_betas=15)
            update_best(a_r, _score(a_r), f"restart_s{sigma}")

    # === Phase 10: Final tiny-beta L-BFGS polish ===
    print(f"\n[Phase 10] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 1e-10
        remaining = deadline - 10 - time.monotonic()
        if remaining > 5:
            max_iter_fin = min(3000, max(100, int(remaining / 0.05)))
            try:
                result = minimize(
                    lambda z_val: _lsobj_factory(z_val, beta_final, n),
                    z,
                    jac=lambda z_val: _lsgrad_factory(z_val, beta_final, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter_fin, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 200}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_tiny_beta")
            except Exception:
                pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
