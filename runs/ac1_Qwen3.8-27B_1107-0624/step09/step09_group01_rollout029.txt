"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive smooth minimization, KKT-Newton with pressure-weighted active sets,
and coordinate descent polish. Explores multiple sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _hankel_matvec(v, W, n):
    """Compute the Hankel matrix-vector product: H*v where H is defined by W."""
    # H_{i,j} = W_{i+j} (for the appropriate range)
    # The gradient of sum_k W_k * conv(a,a)_k with respect to a_i is:
    # 2 * sum_k W_k * a_{k-i}
    # Let j = k-i, so sum_j W_{j+i} * a_j
    # This is a Hankel matrix times a vector.
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]

def _cg_solve_hankel(W, n, rhs, max_iter=200, tol=1e-12):
    """Solve H*x = rhs using CG, where H is the Hankel operator defined by W."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x += alpha * p
        r -= alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            x -= alpha * p
            r += alpha * Hp
            break
        beta = rs_new / rs
        p = r + beta * p
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

def _smooth_minimize(a_init, deadline, beta_initial=None, min_remaining=60.0):
    """Minimize using smooth log-sum-exp approximation with adaptive temperature."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    
    if beta_initial is None:
        c_sorted = np.sort(c_cur)[::-1]
        gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
        beta_initial = max(gap * 0.5, max_c * 0.001)
    
    beta = beta_initial
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    
    def obj_fn(z_vec):
        a_local = _softmax(z_vec)
        c_local = _full_convolve(a_local, a_local)
        c_max_local = float(c_local.max())
        shifted = (c_local - c_max_local) / beta
        e = np.exp(np.clip(shifted, -50, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n * c_max_local
        return 2.0 * n * (c_max_local + beta * np.log(s))
    
    def grad_fn(z_vec):
        a_local = _softmax(z_vec)
        c_local = _full_convolve(a_local, a_local)
        c_max_local = float(c_local.max())
        shifted = (c_local - c_max_local) / beta
        e = np.exp(np.clip(shifted, -50, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c_local)
            w[int(np.argmax(c_local))] = 1.0
        else:
            w = e / s
        a_rev = a_local[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c_local))
        return 4.0 * n * a_local * (h - W_val)
    
    min_beta = max_c * 1e-8
    
    while beta > min_beta:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Estimate time per iteration
        scale = max(1.0, n / 16384.0)
        est_time_per_iter = 0.003 * scale
        max_iter = min(3000, max(20, int(remaining / (est_time_per_iter * 20))))
        
        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 80, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
        
        # Update beta
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(10, len(c_sorted_new) - 1)]
        
        if new_gap < 1e-8 * c_max_new:
            beta *= 0.05
        elif new_gap < 1e-6 * c_max_new:
            beta *= 0.1
        elif new_gap < 1e-4 * c_max_new:
            beta *= 0.2
        else:
            beta *= 0.4
        
        beta = min(beta, c_max_new * 1e-4)
    
    return current_a

def _coordinate_polish(a_init, deadline, blocks=64):
    """Block coordinate descent polish."""
    n = len(a_init)
    a = a_init.copy()
    best_sc = _score(a)
    best_a = a.copy()
    
    for sweep in range(100):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for start in range(0, n, blocks):
            if time.monotonic() > deadline:
                break
            end = min(start + blocks, n)
            
            # Current convolution
            c = _full_convolve(a, a)
            c_max = float(c.max())
            
            # Try to improve this block
            # Use gradient for the block
            a_rev = a[::-1]
            wc = _full_convolve(np.ones(2*n-1), a_rev)  # This is just a simple operation
            # Actually, let's use the gradient approach
            # Gradient at current point for the block
            c_max_idx = np.argmax(c)
            W = np.zeros(2*n-1)
            # Use top few active positions
            K = min(10, 2*n-1)
            idx = np.argpartition(c, -K)[-K:]
            W[idx] = 1.0 / K
            
            wc2 = _full_convolve(W, a_rev)
            h = wc2[n-1:2*n-1]
            W_val = float(np.dot(W, c))
            g = 2.0 * a * (h - W_val)
            
            # Update the block
            for trial in [0.1, 0.05, 0.01]:
                a_try = a.copy()
                a_try[start:end] -= trial * g[start:end]
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-15:
                    a = a_try
                    improved = True
                    break
        
        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
        
        if not improved:
            break
    
    return best_a, best_sc

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

"""
Optimized step function coefficient search combining pressure-weighted KKT-Newton,
adaptive smooth minimization, and multi-length exploration to minimize
2*n*max(conv(a,a))/sum(a)^2. Key innovations: pressure-weighted active sets,
aggressive beta cooling, block Newton refinement, and systematic n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve_hankel(W, n, rhs, max_iter=150, tol=1e-12):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp) or pHp <= 0:
            break
        alpha = rs / pHp
        x += alpha * p
        r -= alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new):
            x -= alpha * p
            r += alpha * Hp
            break
        if rs_new > rs * 1.05:
            x -= alpha * p
            r += alpha * Hp
            break
        beta_cg = rs_new / rs
        p = r + beta_cg * p
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _compute_grad(a, W, n):
    a_rev = a[::-1]
    wc = _full_convolve(W, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(W, _full_convolve(a, a)))
    g = 2.0 * a * (h - W_val)
    return g


def _kkt_newton(a_init, deadline, max_rounds=300, k_schedule=None):

    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    if k_schedule is None:

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        K = k_schedule[rnd % len(k_schedule)]

            if w_sum > 1e-30:

            else:

        else:

        gn = float(np.linalg.norm(g_c))
        if gn < 1e-11 * n:
            break

        Hg = _cg_solve_hankel(W, n, g_c, max_iter=120)
        H1 = _cg_solve_hankel(W, n, np.ones(n), max_iter=120)
        H1_denom = float(np.ones(n) @ H1)

            if s_try < 1e-15:

                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)

                step_done = True
                break

            no_improve = 0
        else:

    return best_a, best_sc


def _smooth_minimize(a_init, deadline, min_remaining=60.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())

    c_sorted = np.sort(c_cur)[::-1]

    beta = max(gap * 0.4, max_c * 0.002)

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    current_a = a_init.copy()
    current_sc = _score(current_a)
    min_beta = max_c * 5e-9

    while beta > min_beta:
        if time.monotonic() + min_remaining > deadline:

        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 16384.0)
        est_iter_time = 0.004 * scale
        max_iter = min(4000, max(30, int(remaining / (est_iter_time * 15))))

        def obj_fn(z_vec, beta=beta, n=n):
            a_local = _softmax(z_vec)
            c_local = _full_convolve(a_local, a_local)
            c_max_local = float(c_local.max())
            shifted = (c_local - c_max_local) / beta
            e = np.exp(np.clip(shifted, -50, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                return 2.0 * n * c_max_local
            return 2.0 * n * (c_max_local + beta * np.log(s))

        def grad_fn(z_vec, beta=beta, n=n):
            a_local = _softmax(z_vec)
            c_local = _full_convolve(a_local, a_local)
            c_max_local = float(c_local.max())
            shifted = (c_local - c_max_local) / beta
            e = np.exp(np.clip(shifted, -50, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c_local)
                w[int(np.argmax(c_local))] = 1.0
            else:
                w = e / s
            a_rev = a_local[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W_val = float(np.dot(w, c_local))
            return 4.0 * n * a_local * (h - W_val)

        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(15, len(c_sorted_new) - 1)]

        if new_gap < 1e-8 * c_max_new:
            beta *= 0.05
        elif new_gap < 1e-6 * c_max_new:
            beta *= 0.08
        elif new_gap < 1e-4 * c_max_new:
            beta *= 0.15
        else:
            beta *= 0.3

        beta = min(beta, c_max_new * 5e-5)

    return current_a

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.55
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.5 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    elif kind == "power":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

    update_best(a2, _score(a2), "smooth")

    print(f"\n[Phase 3] Subgrad momentum (t={time.monotonic()-t_start:.1f}s)", flush=True)
    t_p3 = min(deadline - 400, t_start + 750)
    a3, sc3 = _subgrad_momentum(best_seq, t_p3, max_steps=6000)
    update_best(a3, sc3, "subgrad")

    # Phase 4: KKT-Newton refined
    print(f"\n[Phase 4] KKT-Newton refined (t={time.monotonic()-t_start:.1f}s)", flush=True)
    t_p4 = min(deadline - 250, t_start + 950)
    k_sch4 = [5, 10, 20, 50, 100, 300, 1000, 3000]
    a4, sc4 = _kkt_newton(best_seq, t_p4, max_rounds=400, k_schedule=k_sch4)
    update_best(a4, sc4, "kkt_p4")

    # Phase 5: Multi-n exploration
    print(f"\n[Phase 5] Multi-n (t={time.monotonic()-t_start:.1f}s)", flush=True)
    n_candidates = sorted(set([
        n0, n0 // 2, n0 * 2,
        24576, 28672, 32768, 36864, 40960, 49152
    ]))
    n_candidates = [n for n in n_candidates if 4096 <= n <= 65536]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        a_res = _resample(best_seq, n_try)
        sc_res = _score(a_res)

        # Quick KKT on resampled
        if time.monotonic() < deadline - 40 and n_try <= 49152:
            t_end = min(deadline - 30, time.monotonic() + 20)
            a_opt, sc_opt = _kkt_newton(a_res, t_end, max_rounds=80,
                                        k_schedule=[5, 10, 50, 200, 500, 1000])
            update_best(a_opt, sc_opt, f"kkt_n{n_try}")
        else:
            update_best(a_res, sc_res, f"res_n{n_try}")

    # Phase 6: Perturbation restarts
    print(f"\n[Phase 6] Restarts (t={time.monotonic()-t_start:.1f}s)", flush=True)
    for rst in range(3):
        if time.monotonic() > deadline - 20:
            break
        sigma = 0.015 * (rst + 1)
        a_p = _perturb(best_seq, sigma, rng)
        t_end = min(deadline - 15, time.monotonic() + 15)
        a_r, sc_r = _kkt_newton(a_p, t_end, max_rounds=60,
                                k_schedule=[5, 20, 100, 500])
        update_best(a_r, sc_r, f"rst_{rst}")

    # Phase 7: Final ultra-fine polish (tiny active sets)
    print(f"\n[Phase 7] Final polish (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 2:
        k_final = [1, 1, 2, 2, 3, 3, 5, 5, 10, 20, 50]
        a_fin, sc_fin = _kkt_newton(best_seq, deadline - 1, max_rounds=300,
                                    k_schedule=k_final)
        update_best(a_fin, sc_fin, "final_polish")

    # Final output
    print(f"\n{'=' * 60}", flush=True)
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s", flush=True)
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
