"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines analytical gradient descent with frequency-domain equalization
(POCS-like) and adaptive active-set management. Exploits the special
structure of the self-convolution gradient (shifted copies of a) for
efficient computation.

import numpy as np
import time
import math

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _compute_gradient(a, active_indices):

    Uses the analytical formula: grad_j[k] = 2*a[k-j] for valid k-j."""
    n = len(a)
    grad = np.zeros(n)
    # For each active peak k, the gradient contribution is 2*a[k-j] for j in [0, n-1]
    # efficient way: for each k, grad += 2 * flip and shift of a
    # grad[k-j] += 2*a[j] for j in [0,n-1] where k-j in [0,n-1]
    # equivalently: grad[j] += 2*a[k-j] for j where 0 <= k-j <= n-1
    # i.e., j in [max(0, k-n+1), min(k, n-1)]
    
    # More efficient: use convolution
    # grad = sum_{k in active} 2 * (a reversed, shifted by k)
    # (a reversed)[j] = a[n-1-j]

    # = sum_{k in active} a[k-j]

    # = convolve(w, a_flip)[n-1+j] (by the formula above with m = n-1+j)
    #   = sum_i w[i] * a_flip[n-1-(n-1+j)+i] = sum_i w[i] * a_flip[i-j] = sum_i w[i] * a[n-1-i+j]

def _gradient_descent_step(a, active_indices, lr):
    """Take one gradient descent step to reduce the active peaks."""
    n = len(a)
    grad = _compute_gradient(a, active_indices)
    # Project onto tangent space of simplex (preserve sum)
    grad = grad - grad.mean()
    # Step
    a_new = a - lr * grad
    # Project onto non-negative
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    return a_new

def _find_active_set(c, tolerance=1e-6):

    c_max = c.max()

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines analytical gradient descent with adaptive active-set management,
frequency-domain equalization (POCS-like), and multi-scale exploration.
Key improvements: direct analytical gradient (shifted copies of a), 
backtracking line search, tempered frequency equalization, and larger n.
"""
import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_grad(a, active):
    """Compute gradient of average over active peaks of conv(a,a).
    grad[j] = 2/|S| * sum_{k in S} a[k-j] for valid k-j in [0,n-1]."""
    n = len(a)
    # Create weight vector w of length 2n-1
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / len(active)
    # Cross-correlation: grad[j]/2 = sum_k w[k] * a[k-j]
    # = conv(w, a_flipped)[j + n - 1]
    a_flip = a[::-1]
    xcorr = _fft_convolve(w, a_flip)
    # xcorr has length (2n-1)+(n-1) = 3n-2
    # We need indices n-1 to 2n-2
    grad = 2.0 * xcorr[n - 1:2 * n - 1]
    return grad


def _grad_step(a, active, lr):
    """One gradient descent step with simplex projection."""
    n = len(a)
    grad = _compute_grad(a, active)
    # Project to tangent space (preserve sum)
    grad = grad - grad.mean()
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s


def _get_active(c, tol_rel=1e-5, min_count=1, max_count=None):
    """Get active set: peaks within relative tolerance of max."""
    c_max = c.max()
    if max_count is None:
        max_count = len(c)
    threshold = c_max * (1.0 - tol_rel)
    active = np.where(c >= threshold)[0]
    if len(active) < min_count:
        active = np.array([np.argmax(c)])
    if len(active) > max_count:
        # Take top max_count
        idx = np.argsort(c)[::-1][:max_count]
        active = idx
    return active


def _freq_equalize(a, n_iters=200, beta=0.5):
    """Tempered frequency-domain equalization (POCS-like).
    Alternates between flat magnitude in freq domain and non-negativity in time domain."""
    n = len(a)
    L = _next_pow2(2 * n - 1)
    a_cur = a.copy()
    
    for it in range(n_iters):
        # Zero-pad and FFT
        A = np.fft.rfft(a_cur, L)
        mag = np.abs(A)
        # Tempered equalization: A_new = A * (C/mag)^beta
        C = mag[0]  # preserve DC (sum)
        if C < 1e-15:
            break
        ratio = C / (mag + 1e-300)
        A_new = A * (ratio ** beta)
        # IFFT and project
        a_raw = np.fft.irfft(A_new, L)[:n]
        a_new = np.maximum(a_raw, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a_cur = a_new
    
    return a_cur


def _grad_descent_loop(a, deadline, min_time=30.0, tol_rel=1e-5, lr_init=0.05, max_active=200):
    """Main gradient descent loop with adaptive step size and active set."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init
    no_improve = 0
    step_count = 0
    
    while time.monotonic() + min_time < deadline:
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        
        active = _get_active(c, tol_rel=tol_rel, max_count=max_active)
        a_new = _grad_step(a, active, lr)
        
        # Check improvement (compute score of new point)
        c_new = _fft_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())
        
        if sc_new < sc - 1e-14:
            a = a_new
            step_count += 1
            # Slowly increase lr if making steady progress
            if step_count % 100 == 0:
                lr = min(lr * 1.02, 0.2)
        else:
            no_improve += 1
            lr *= 0.9
            if no_improve > 50:
                # Adaptive: try smaller tolerance (fewer active peaks)
                if tol_rel > 1e-8:
                    tol_rel *= 0.5
                    no_improve = 0
                    lr = lr_init * 0.5
                elif no_improve > 200:
                    # Try perturbation to escape
                    break
            if lr < 1e-8:
                break
    
    return best_a, best_sc


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "alternating":
        a = np.zeros(n)
        a[0::2] = 1.0
        a[1::2] = 0.3
        return a / a.sum()
    elif kind == "random":
        if rng is None:
            rng = np.random.default_rng(0)
        a = rng.random(n)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
