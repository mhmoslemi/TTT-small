```python
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _hankel_matvec(v, W, n):
    """Compute H @ v where H[i,j] = W[i+j]"""
    v_rev = v[::-1]
    conv = _fft_convolve(W, v_rev)
    return conv[n-1:2*n-1]

def _solve_hankel(W, n, rhs, max_iter=500, tol=1e-14):
    """Solve H @ x = rhs using CG, where H[i,j] = W[i+j]"""
    x = np.zeros(n)
    if np.allclose(rhs, 0):
        return x
    r = rhs - _hankel_matvec(x, W, n)  # = rhs since x=0
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-300:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if not np.isfinite(pHp) or abs(pHp) < 1e-300:
            break
        if pHp < 0:
            # H is not SPD, use damped step
            alpha = rs / (pHp + 1e-10)
            alpha = min(alpha, 1.0 / np.max(np.abs(p)) * 10)
        else:
            alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x
```

```python
"""
KKT-based optimizer with direct Hankel system solve, adaptive active sets,
smooth L-BFGS polish, and multi-scale exploration.
Minimizes 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _hankel_matvec(v, W, n):
    v_rev = v[::-1]
    conv = _fft_convolve(W, v_rev)
    return conv[n - 1:2 * n - 1]


def _cg_solve_hankel(W, n, rhs, max_iter=500, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-300:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if not np.isfinite(pHp) or abs(pHp) < 1e-300:
            break
        if pHp < 0:
            alpha = rs / (-pHp + 1e-12)
            alpha = -alpha
        else:
            alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.05:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _lsqr_solve_hankel(W, n, rhs, max_iter=200, tol=1e-12):
    x = np.zeros(n)
    u = rhs.copy()
    v = _hankel_matvec(u, W, n)
    beta = float(np.linalg.norm(v))
    if beta < 1e-30:
        return rhs / max(float(np.linalg.norm(rhs)), 1e-30)
    v = v / beta
    alpha = float(np.linalg.norm(u))
    if alpha < 1e-30:
        return x
    p = alpha * v
    rnorm = alpha
    rnrm1sq = 0.0
    for i in range(max_iter):
        w = _hankel_matvec(v, W, n)
        w = w - beta * p
        beta_new = float(np.linalg.norm(w))
        if beta_new < 1e-30:
            break
        w = w / beta_new
        v = _hankel_matvec(w, W, n) - alpha * p
        alpha_new = float(np.linalg.norm(v))
        if alpha_new < 1e-30:
            break
        c = alpha_new / np.sqrt(alpha_new * alpha_new + beta_new * beta_new)
        s = beta_new / np.sqrt(alpha_new * alpha_new + beta_new * beta_new)
        phi_prime = c * rnorm
        rnorm = -s * rnorm
        x = x + phi_prime * p
        p = w - s * v
        v = v / alpha_new
        u = u - alpha * w
        u = u - beta * p
        alpha = alpha_new
        beta = beta_new
        rnrm1sq = rnrm1sq * s * s
        if rnorm * rnorm + rnrm1sq < tol * tol * (alpha * alpha):
            break
    return x


def _kkt_direct_solve(a_init, deadline, max_rounds=300, min_active=5, max_active=None):
    n = len(a_init)
    if max_active is None:
        max_active = 2 * n - 1
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        c_max = float(c.max())

        tol = 1e-12 * c_max
        active = c >= (c_max - tol)
        if active.sum() < min_active:
            K = min(min_active, len(c))
            idx = np.argpartition(c, -K)[-K:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
        active_count = int(active.sum())
        if active_count > max_active:
            idx = np.argpartition(c, -max_active)[-max_active:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            active_count = max_active

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / active_count

        rhs = np.ones(n)
        a_new = _cg_solve_hankel(W, n, rhs, max_iter=400, tol=1e-14)

        min_val = float(a_new.min())
        if min_val < -1e-6:
            a_new = _lsqr_solve_hankel(W, n, rhs, max_iter=100)
            min_val = float(a_new.min())
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-15:
            no_improve += 2
            if no_improve > 10:
                break
            continue
        a_new = a_new / s_new
        sc_new = _score(a_new)

        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = 0.5 * a + 0.5 * a_new
            a = a / a.sum()
            no_improve += 1
            if no_improve > 15:
                break

        c_check = _fft_convolve(a, a)
        if float(c_check.max()) - c_max < 1e-14 * c_max and no_improve == 0:
            if rnd > 5:
                break

    return best_a, best_sc


def _kkt_with_schedule(a_init, deadline, max_rounds=500):

def _smooth_lbfgs(a_init, deadline, beta_start=None, beta_end=None):

    c_init = _fft_convolve(a_init, a_init)
    cmax_init = float(c_init.max())

            c_val = _fft_convolve(a_val, a_val)
            c_max = float(c_max_of_c(c_val))
            shifted = (c_val - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s_val = e.sum()
            if s_val < 1e-300 or not np.isfinite(s_val):
                return 2.0 * n * c_max
            return 2.0 * n * (c_max + beta * np.log(s_val))

        def grad_fn(z):
            a_val = _softmax(z)
            c_val = _fft_convolve(a_val, a_val)
            c_max = float(c_max_of_c(c_val))
            shifted = (c_val - c_max) / beta

            s_val = e.sum()
            if s_val < 1e-300 or not np.isfinite(s_val):
                w = np.zeros_like(c_val)
                w[int(np.argmax(c_val))] = 1.0
            else:
                w = e / s_val
            a_rev = a_val[::-1]
            wc = _fft_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W_val = float(np.dot(w, c_val))
            return 4.0 * n * a_val * (h - W_val)

        remaining = deadline - time.monotonic() - 30
        if remaining < 5:
            break

        max_iter = min(3000, max(50, int(remaining / (scale * 20))))

        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 50, 'maxls': 100}
            )

                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))

        except Exception:
            pass

        c_new = _fft_convolve(current_a, current_a)
        cmax_new = float(c_new.max())
        beta = max(beta * 0.25, cmax_new * 1e-6)

    return current_a, current_sc


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _c_max_of_c(c):
    return c.max()


def _subgrad_step(a, n, tol_frac=0.001):
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() < 3:
        idx = np.argpartition(c, -3)[-3:]
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
    W = np.zeros_like(c)
    W[active] = 1.0 / active.sum()

    a_rev = a[::-1]
    wc = _fft_convolve(W, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(W, c))
    grad = 2.0 * a * (h - W_val)
    grad = grad - np.mean(grad)
    return grad, c_max


def _subgrad_momentum(a_init, deadline, max_steps=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))

    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.015
    momentum = 0.9
    v = np.zeros(n)

    for i in range(max_steps):

            break
        a = _softmax(z)
        grad, c_max = _subgrad_step(a, n)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break
        v = momentum * v - lr * grad

                    z = np.log(np.maximum(best_a, 1e-30))

                    v = np.zeros(n)
                    patience = 0
                    lr = 0.01

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.02
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = (1.0 + np.cos(2 * np.pi * t)) ** 0.8 + 1e-10
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)

        update_best(a_fin, sc_fin, "final_polish")

    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
