```python
import cvxpy as cp

def _solve_peak_equalization_lp(a, k_indices, K):
    """Find the optimal direction d to reduce the top K peaks.
    Returns d (direction) and t (reduction amount)."""
    n = len(a)
    c = _full_convolve(a, a)
    
    # Build the constraint matrix
    # A[i, j] = 2 * a[k_i - j] if 0 <= k_i - j < n, else 0
    A = np.zeros((K, n))
    for i, k in enumerate(k_indices[:K]):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[i, j] = 2.0 * a[idx]
    
    c_vals = c[k_indices[:K]]
    
    # LP: min t
    # s.t. A @ d - t <= -c_vals  (K constraints)
    #      sum(d) = 0              (1 constraint)
    #      d >= -a                  (n constraints)
    
    d_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        A @ d_var - t_var <= -c_vals,
        cp.sum(d_var) == 0,
        d_var >= -a
    ]
    
    objective = cp.Minimize(t_var)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.GLOP)  # or ECOS, or CLARABEL
        if d_var.value is not None and t_var.value is not None:
            return d_var.value, t_var.value
    except:
        pass
    
    return None, None
```

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
