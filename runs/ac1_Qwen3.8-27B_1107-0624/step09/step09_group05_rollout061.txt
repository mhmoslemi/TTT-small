        # Center the gradient (project to sum=0)
        d = grad - grad.mean()

        # Nesterov momentum update
        v = gamma * v + d
        v_centered = v - v.mean()

        # Adaptive step size
        gap = sc - 2.0 * n / (2.0 * n - 1)
        gap = max(gap, 0.001)
        v_norm_sq = float(np.dot(v_centered, v_centered))
        if v_norm_sq < 1e-20:
            base_step *= 0.9
            continue
        eta = base_step * gap / np.sqrt(v_norm_sq)
        eta = min(eta, 0.3)
        eta = max(eta, 0.001)

        # Update with backtracking
        a_new = None
        for trial_ratio in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a_try = a + trial_ratio * eta * v_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < sc:
                a_new = a_try
                break

        if a_new is None:
            # Try even smaller steps
            for trial_eta in [0.001, 0.0005, 0.0001]:
                a_try = a + trial_eta * v_centered
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < sc - 1e-14:
                    a_new = a_try
                    break

        if a_new is None:
            base_step *= 0.95
            if base_step < 0.001:
                break
            continue

        a = a_new
        t_iter += 1

        # Decay momentum
        gamma = gamma_schedule_start + (gamma_schedule_end - gamma_schedule_start) * progress

        if t_iter % 500 == 0 and t_iter < 5000:
            print(f"    [MomentumSubgrad] iter={t_iter}, sc={best_sc:.10f}, eta={eta:.6f}, n_active={n_active}")

    return best_a, best_sc


def _run_lbfgs_extended(a_init, deadline, min_remaining=30.0, verbose=False):
    """L-BFGS with aggressive adaptive beta and restart strategy."""
    global _t_start
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    # Compute gap for beta
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < c0_max * 0.002:
        gap = c0_max * 0.002

    # More aggressive beta schedule
    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.004, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
        0.000031, 0.000015, 0.000008, 0.000004,
        0.000002, 0.000001, 0.0000005
    ])
    betas = gap * multipliers

    stagnation_count = 0
    max_stagnation = 3

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        size_factor = max(1.0, n / 8192.0)
        est_time = 0.003 * size_factor
        max_iter = min(6000, max(200, int(remaining / (est_time * 15))))

        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 200,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)

            if sc_new < best_sc - 1e-13:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                stagnation_count = 0
                if verbose:
                    elapsed = time.monotonic() - _t_start
                    print(f"    [L-BFGS] beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}, t={elapsed:.0f}s")
            else:
                stagnation_count += 1
                if stagnation_count >= max_stagnation:
                    # Restart with perturbation
                    rng = np.random.default_rng(42 + i)
                    z += rng.normal(0, 0.01, size=n)
                    z -= z.mean()
                    stagnation_count = 0
                    if verbose:
                        print(f"    [L-BFGS] beta={beta:.2e}: restart (perturbed)")
                    max_stagnation += 1
        except Exception as e:
            if verbose:
                print(f"    [L-BFGS] beta={beta:.2e}: error {e}")

    return best_a, best_sc


def _run_kkt_refine(a_init, deadline, min_remaining=20.0):
    """Improved KKT active-set refinement with adaptive tolerance."""
    global _t_start
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_frac = 0.002
    min_tol_frac = 0.00005
    no_improve = 0
    max_no_improve = 500
    eta_base = 0.08

    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(min_tol_frac, tol_frac * 0.5)
            if tol_frac <= min_tol_frac * 1.2:
                break
            no_improve = 0
            max_no_improve = int(max_no_improve * 0.8)
            if max_no_improve < 50:
                break

        # Active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_act = int(active.sum())
        if n_act == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_act = 1

        # Compute average gradient from active set
        grad_avg = _compute_active_gradient(a, active, n)

        # Identify support
        support = a > 1e-9 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        # KKT: grad_avg should be constant on support
        mu = float(np.mean(grad_avg[support]))
        residual = grad_avg[support] - mu

        # Update
        a_new = None
        for eta in [eta_base, eta_base * 0.5, eta_base * 0.25, eta_base * 0.1]:
            a_try = a.copy()
            a_try[support] -= eta * residual
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < sc - 1e-14:
                a_new = a_try
                break

        if a_new is None:
            # Try expanding support slightly
            support2 = a > 1e-11 * a.max()
            if support2.sum() > support.sum():
                mu2 = float(np.mean(grad_avg[support2]))
                resid2 = grad_avg[support2] - mu2
                for eta in [0.02, 0.01, 0.005]:
                    a_try = a.copy()
                    a_try[support2] -= eta * resid2
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < sc - 1e-14:
                        a_new = a_try
                        support = support2
                        break

        if a_new is None:
            # Decrease eta
            eta_base *= 0.7
            if eta_base < 0.001:
                break
            continue

        a = a_new

        # Tighten tolerance on improvement
        if no_improve == 0:
            tol_frac = max(min_tol_frac, tol_frac * 0.95)

    return best_a, best_sc


def _run_peak_directed(a_init, deadline, min_remaining=20.0):
    """
    Novel peak-directed method: directly reduce the highest peaks by
    finding the most effective direction in the simplex.
    """
    global _t_start
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    it = 0
    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # Find the single worst lag
        k_star = int(np.argmax(c))

        # Gradient of c[k_star] w.r.t. a
        # c[k] = sum_i a[i]*a[k-i], grad[j] = 2*a[k-j] (for valid k-j, j != k-j)
        # or a[j]^2 contribution when j = k-j (i.e., k = 2j)
        grad = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                if k_j == j:
                    grad[j] = a[j]  # derivative of a[j]^2
                else:
                    grad[j] = 2.0 * a[k_j]

        # Project to sum(d) = 0 and d only affects positive coordinates
        d = grad - grad.mean()
        # Zero out where a is already 0 (can't decrease further)
        zero_mask = a <= 1e-12
        d[zero_mask] = 0.0
        d -= d.sum() / n  # re-center approximately

        # Line search
        best_eta = 0
        best_sc_try = sc
        for eta in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]:
            a_try = a - eta * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_eta = eta

        if best_eta == 0:
            # Try a different approach: redistribute from high-gradient to low-gradient
            support = a > 1e-10
            if support.sum() < 2:
                break
            # Move mass from high grad to low grad
            g_sup = grad[support]
            g_mean = g_sup.mean()
            d_redist = np.zeros(n)
            d_redist[support] = g_sup - g_mean
            for eta in [0.001, 0.0005, 0.0001]:
                a_try = a - eta * d_redist
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < sc:
                    a = a_try
                    best_eta = eta
                    break

        if best_eta == 0:
            break

        a = a - best_eta * d
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        it += 1

    return best_a, best_sc


def _run_cyclic_refine(a_init, deadline, min_remaining=20.0):
    """
    Cyclic refinement: iterate over lags in round-robin, reducing each one.
    Uses the insight that at optimum, all active lags should have equal value.
    """
    global _t_start
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Start with just the top lag, gradually add more
    n_active = 3
    it = 0
    stagnant = 0

    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 3000:
                # Increase active set size
                n_active = min(n, n_active * 2)
                stagnant = 0
                if n_active > n:
                    break

        # Get top n_active lags
        top_idx = np.argsort(c)[-n_active:]
        target = c[top_idx].mean()

        # For each active lag, try to reduce it toward target
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[top_idx] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]

        # Project gradient
        d = grad_avg - grad_avg.mean()
        support = a > 1e-10

        best_eta = 0
        for eta in [0.02, 0.01, 0.005, 0.002, 0.001]:
            a_try = a - eta * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < sc - 1e-15:
                best_eta = eta
                a = a_try
                break

        if best_eta == 0:
            # Try with only the single worst lag
            k_worst = top_idx[-1]
            grad_single = np.zeros(n)
            for j in range(n):
                k_j = k_worst - j
                if 0 <= k_j < n:
                    if k_j == j:
                        grad_single[j] = a[j]
                    else:
                        grad_single[j] = 2.0 * a[k_j]
            d_s = grad_single - grad_single.mean()
            for eta in [0.005, 0.002, 0.001, 0.0005]:
                a_try = a - eta * d_s
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < sc - 1e-15:
                    a = a_try
                    best_eta = eta
                    break

        if best_eta == 0:
            if stagnant > 100:
                break
        it += 1

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "one_sided":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t) + 0.005
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.6 * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    elif kind == "asymmetric_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5 * np.exp(-0.3 * (t - 0.4))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _short_lbfgs(a_init, max_secs=60):
    """Quick L-BFGS with just a few beta levels."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c0 = _full_convolve(a, a)
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < c0_max := float(c0.max()):
        gap = c0_max * 0.01
    gap = max(gap, c0_max * 0.005)

    betas = gap * np.array([0.1, 0.03, 0.01, 0.003])
    end_time = time.monotonic() + max_secs

    for beta in betas:
        if time.monotonic() > end_time:
            break
        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': 1000, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining:
    P1: Extended L-BFGS with restarts (main workhorse)
    P2: Momentum subgradient with adaptive active set
    P3: KKT active-set refinement
    P4: Peak-directed gradient
    P5: Cyclic lag refinement
    P6: Multi-n exploration
    P7: Final polish (L-BFGS + KKT)
    """
    global _t_start
    _t_start = time.monotonic()
    deadline = _t_start + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(16384, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n=16384: score={best_score:.10f}")

    n_main = len(best_seq)

    # === Phase 1: Main L-BFGS with extended schedule ===
    print(f"\n[P1] Extended L-BFGS (n={n_main}, t={time.monotonic()-_t_start:.0f}s)")
    p1_end = _t_start + min(400, (budget_s - 10) * 0.25)
    a1, sc1 = _run_lbfgs_extended(best_seq, p1_end, min_remaining=200, verbose=True)
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  P1 result: {best_score:.10f}")

    # === Phase 2: Momentum subgradient ===
    print(f"\n[P2] Momentum subgradient (t={time.monotonic()-_t_start:.0f}s)")
    if time.monotonic() + 100 < deadline:
        p2_end = min(deadline - 400, time.monotonic() + 300)
        a2, sc2 = _run_momentum_subgrad(best_seq, p2_end, min_remaining=50)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(f"  P2 result: {best_score:.10f}")

    # === Phase 3: KKT refinement ===
    print(f"\n[P3] KKT refine (t={time.monotonic()-_t_start:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p3_end = min(deadline - 300, time.monotonic() + 200)
        a3, sc3 = _run_kkt_refine(best_seq, p3_end, min_remaining=50)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3 result: {best_score:.10f}")

    # === Phase 4: L-BFGS round 2 (after subgradient moved us) ===
    print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-_t_start:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p4_end = min(deadline - 200, time.monotonic() + 250)
        a4, sc4 = _run_lbfgs_extended(best_seq, p4_end, min_remaining=100, verbose=True)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4 result: {best_score:.10f}")

    # === Phase 5: Peak-directed method ===
    print(f"\n[P5] Peak-directed (t={time.monotonic()-_t_start:.0f}s)")
    if time.monotonic() + 40 < deadline:
        p5_end = min(deadline - 150, time.monotonic() + 150)
        a5, sc5 = _run_peak_directed(best_seq, p5_end, min_remaining=30)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  P5 result: {best_score:.10f}")

    # === Phase 6: Cyclic lag refinement ===
    print(f"\n[P6] Cyclic refine (t={time.monotonic()-_t_start:.0f}s)")
    if time.monotonic() + 40 < deadline:
        p6_end = min(deadline - 100, time.monotonic() + 120)
        a6, sc6 = _run_cyclic_refine(best_seq, p6_end, min_remaining=20)
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  P6 result: {best_score:.10f}")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n (t={time.monotonic()-_t_start:.0f}s)")
    n_options = [12288, 20480, 24576]
    n_options = [x for x in n_options if x != n_main and x <= 24576]
    for n_try in n_options:
        if time.monotonic() + 80 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if sc_r < best_score:
            best_score = sc_r
            best_seq = a_r.copy()
            print(f"  n={n_try} (resample): {best_score:.10f}")
        if time.monotonic() + 60 > deadline:
            continue
        p7_end = min(deadline - 80, time.monotonic() + 80)
        a7, sc7 = _short_lbfgs(a_r, max_secs=60)
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()
            print(f"  n={n_try} (L-BFGS): {best_score:.10f}")

    # === Phase 8: Final polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-_t_start:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p8_end = min(deadline - 10, time.monotonic() + 100)
        a8, sc8 = _run_lbfgs_extended(best_seq, p8_end, min_remaining=5, verbose=False)
        if sc8 < best_score:
            best_score = sc8
            best_seq = a8.copy()

        if time.monotonic() + 15 < deadline:
            p8b_end = min(deadline - 5, time.monotonic() + 60)
            a8b, sc8b = _run_kkt_refine(best_seq, p8b_end, min_remaining=5)
            if sc8b < best_score:
                best_score = sc8b
                best_seq = a8b.copy()

        if time.monotonic() + 10 < deadline:
            p8c_end = min(deadline - 3, time.monotonic() + 30)
            a8c, sc8c = _run_cyclic_refine(best_seq, p8c_end, min_remaining=3)
            if sc8c < best_score:
                best_score = sc8c
                best_seq = a8c.copy()

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    # Cross-check with evaluate_sequence
    try:
        eval_score = evaluate_sequence([float(x) for x in final])
        print(f"FINAL: internal score={sc_final:.10f}, eval score={eval_score:.10f}, n={len(final)}, time={time.monotonic()-_t_start:.1f}s")
    except Exception as e:
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-_t_start:.1f}s (eval error: {e})")

    return [float(x) for x in final]
