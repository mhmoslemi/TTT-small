        g = _grad_from_W(a, W, n)
        g_max = float(np.max(np.abs(g)))
        eta = min(0.1, 0.01 / (g_max + 1e-10))

        a_new = _multiplicative_update(a, W, n, eta)

        # Check improvement
        c_new = _full_convolve(a_new, a_new)
        c_max_new = float(c_new.max())
        if c_max_new < c_max - 1e-15:
            a = a_new
            patience = 0
            sc = 2.0 * n * c_max_new
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
        else:
            patience += 1
            if patience > 50:
                # Try larger step
                eta *= 2.0
                a_new = _multiplicative_update(a, W, n, eta)
                c_new = _full_convolve(a_new, a_new)
                if float(c_new.max()) < c_max:
                    a = a_new
                    patience = 0
                else:
                    # Stuck, break
                    break

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _lbfgs_smooth(a_init, deadline, min_remaining=60.0):
    """L-BFGS with smooth (log-sum-exp) approximation, with beta annealing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    beta = max(gap * 0.2, max_c * 0.001)

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    current_a = a_init.copy()
    current_sc = _score(current_a)

    def obj_fn(z_inner, beta_val, n_val):
        a_inner = _softmax(z_inner)
        c_inner = _full_convolve(a_inner, a_inner)
        c_max_inner = float(c_inner.max())
        shifted = (c_inner - c_max_inner) / beta_val
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n_val * c_max_inner
        return 2.0 * n_val * (c_max_inner + beta_val * np.log(s))

    def grad_fn(z_inner, beta_val, n_val):
        a_inner = _softmax(z_inner)
        c_inner = _full_convolve(a_inner, a_inner)
        c_max_inner = float(c_inner.max())
        shifted = (c_inner - c_max_inner) / beta_val
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c_inner)
            w[int(np.argmax(c_inner))] = 1.0
        else:
            w = e / s
        a_rev = a_inner[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n_val - 1:2 * n_val - 1]
        W_val = float(np.dot(w, c_inner))
        return 4.0 * n_val * a_inner * (h - W_val)

    while beta > max_c * 1e-10:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break

        scale = max(1.0, n / 8192.0)
        max_iter = min(3000, max(10, int(remaining / (0.003 * scale * 50))))

        try:
            result = minimize(
                lambda z: obj_fn(z, beta, n),
                z,
                jac=lambda z: grad_fn(z, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

        if new_gap < 1e-6 * c_max_new:
            beta *= 0.1
        elif new_gap < 1e-4 * c_max_new:
            beta *= 0.2
        else:
            beta *= 0.3
        beta = min(beta, c_max_new * 0.0002)

    return current_a


def _make_profile(n, kind="tapered", **kwargs):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        alpha = kwargs.get("alpha", 0.6)
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** alpha
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        rate = kwargs.get("rate", 3.0)
        t = np.linspace(0, 1, n)
        a = np.exp(-rate * t)
        return a / a.sum()
    elif kind == "skew_beta":
        p = kwargs.get("p", 2.0)
        q = kwargs.get("q", 1.0)
        t = np.linspace(0, 1, n)
        a = t ** p * (1 - t) ** q
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "two_bump":
        c1 = kwargs.get("c1", 0.3)
        c2 = kwargs.get("c2", 0.7)
        w = kwargs.get("width", 0.15)
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - c1) / w) ** 2) + 0.6 * np.exp(-((t - c2) / w) ** 2)
        return a / a.sum()
    elif kind == "asym_power":
        p = kwargs.get("p", 1.5)
        t = np.linspace(0, 1, n)
        a = (1 - t) ** p
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate():
    t_start = time.monotonic()
    deadline = t_start + 1770.0  # Leave 30s buffer for return
    rng = np.random.default_rng(42)

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"imp={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                par_sc = _score(parent)
                print(f"Parent: n={len(parent)}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        parent = _make_profile(16384, "tapered", alpha=0.6)
        print(f"No parent, using tapered n=16384")
        update_best(parent, _score(parent), "init")

    n0 = len(parent)
    print(f"Starting n={n0}, score={best_score:.10f}")

    # === Phase 1: Multi-start with diverse profiles (0-150s) ===
    print(f"\n[Phase 1] Multi-start (t={time.monotonic() - t_start:.1f}s)")
    t_p1_end = min(deadline - 1600, t_start + 150)

    profiles = [
        ("tapered", {"alpha": 0.4}),
        ("tapered", {"alpha": 0.6}),
        ("tapered", {"alpha": 0.8}),
        ("tapered", {"alpha": 1.0}),
        ("exp_decay", {"rate": 2.0}),
        ("exp_decay", {"rate": 4.0}),
        ("skew_beta", {"p": 1.5, "q": 0.5}),
        ("skew_beta", {"p": 0.5, "q": 1.5}),
        ("two_bump", {"c1": 0.3, "c2": 0.7, "width": 0.15}),
        ("asym_power", {"p": 1.0}),
        ("asym_power", {"p": 2.0}),
        ("cos", {}),
    ]

    candidates = []
    for kind, params in profiles:
        if time.monotonic() > t_p1_end:
            break
        a_try = _make_profile(n0, kind, **params)
        # Quick multiplicative optimization
        a_opt, sc_opt = _multiplicative_loop(a_try, min(t_p1_end, time.monotonic() + 8), max_steps=300)
        sc_final = max(sc_opt, _score(a_opt))
        candidates.append((sc_final, a_opt))
        print(f"  {kind} {params}: score={sc_final:.10f}")

    # Also include parent
    candidates.append((best_score, parent))

    # Sort by score and keep top 3
    candidates.sort(key=lambda x: x[0])
    top_candidates = [c[1] for c in candidates[:3]]

    # Update best from candidates
    for i, (sc, a) in enumerate(candidates[:3]):
        update_best(a, sc, f"p1_cand_{i}")

    # === Phase 2: KKT-Newton with adaptive K on top candidate (150-600s) ===
    print(f"\n[Phase 2] KKT-Newton adaptive (t={time.monotonic() - t_start:.1f}s)")
    t_p2_end = min(deadline - 1100, t_start + 600)

    a_kkt, sc_kkt = _kkt_adaptive(top_candidates[0], t_p2_end, max_rounds=400)
    update_best(a_kkt, sc_kkt, "kkt_p2")

    # Also try KKT on second candidate if time permits
    if time.monotonic() < t_p2_end - 60 and len(top_candidates) > 1:
        a_kkt2, sc_kkt2 = _kkt_adaptive(top_candidates[1], t_p2_end, max_rounds=200)
        update_best(a_kkt2, sc_kkt2, "kkt_p2_alt")

    # === Phase 3: Extended multiplicative refinement (600-900s) ===
    print(f"\n[Phase 3] Multiplicative refine (t={time.monotonic() - t_start:.1f}s)")
    t_p3_end = min(deadline - 800, t_start + 900)
    a_mp, sc_mp = _multiplicative_loop(best_seq, t_p3_end, max_steps=20000)
    update_best(a_mp, sc_mp, "mult_p3")

    # === Phase 4: L-BFGS smooth polish (900-1200s) ===
    print(f"\n[Phase 4] L-BFGS smooth (t={time.monotonic() - t_start:.1f}s)")
    t_p4_end = min(deadline - 550, t_start + 1200)
    a_lb = _lbfgs_smooth(best_seq, t_p4_end, min_remaining=500.0)
    update_best(a_lb, _score(a_lb), "lbfgs_p4")

    # === Phase 5: Alternating KKT + Multiplicative cycles (1200-1550s) ===
    print(f"\n[Phase 5] Alternating cycles (t={time.monotonic() - t_start:.1f}s)")
    t_p5_end = min(deadline - 250, t_start + 1550)

    for cyc in range(10):
        if time.monotonic() > t_p5_end:
            break

        # KKT step
        t_sub = min(t_p5_end, time.monotonic() + 40)
        a_c1, sc_c1 = _kkt_adaptive(best_seq, t_sub, max_rounds=80)
        update_best(a_c1, sc_c1, f"alt_kkt_{cyc}")

        # Multiplicative step
        t_sub2 = min(t_p5_end, time.monotonic() + 30)
        a_c2, sc_c2 = _multiplicative_loop(best_seq, t_sub2, max_steps=3000)
        update_best(a_c2, sc_c2, f"alt_mp_{cyc}")

        # Small perturbation + quick optimize
        if cyc % 3 == 2:
            a_pert = _perturb(best_seq, 0.01 * (cyc + 1), rng)
            t_sub3 = min(t_p5_end, time.monotonic() + 20)
            a_c3, sc_c3 = _multiplicative_loop(a_pert, t_sub3, max_steps=1000)
            update_best(a_c3, sc_c3, f"alt_pert_{cyc}")

    # === Phase 6: Multi-n exploration (1550-1700s) ===
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    t_p6_end = min(deadline - 70, t_start + 1700)

    n_options = sorted(set([4096, 8192, 16384, 32768, 65536]))
    for n_try in n_options:
        if time.monotonic() > t_p6_end:
            break
        a_res = _resample(best_seq, n_try)
        sc_res = _score(a_res)
        if sc_res < best_score:
            update_best(a_res, sc_res, f"resample_n{n_try}")
        # Quick optimize at this n
        if time.monotonic() < t_p6_end - 20 and n_try <= 32768:
            a_opt_n, sc_opt_n = _multiplicative_loop(a_res, min(t_p6_end, time.monotonic() + 15), max_steps=2000)
            update_best(a_opt_n, sc_opt_n, f"opt_n{n_try}")

    # === Phase 7: Final ultra-fine polish (1700-1770s) ===
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        # Very fine KKT
        a_fin, sc_fin = _kkt_adaptive(best_seq, deadline - 2, max_rounds=100)
        update_best(a_fin, sc_fin, "final_kkt")

        # Final multiplicative
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _multiplicative_loop(best_seq, deadline - 1, max_steps=5000)
            update_best(a_fin2, sc_fin2, "final_mp")

    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered", alpha=0.6)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
