def _full_convolve(a, b):

    fb = np.fft.rfft(b, N)

def _bb_step(s_vec, y_vec):

    return max(1e-6, min(alpha, 0.5))

    z = z - np.max(z)

    e = np.exp(np.clip(shifted, -700, 0))

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    """Score assuming sum(a)=1. Returns 2n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):

    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:

def _bb_step_size(s_vec, y_vec):

        wc = _full_convolve(w, a_rev)
        grad_a = 2.0 * wc[n - 1:2 * n - 1]
        W = float(np.dot(grad_a, av))
        return av * (grad_a - W)

    remaining = deadline - min_remaining - time.monotonic()
    if remaining < 5:
        return a_init, _score(a_init)

    max_iter = min(8000, max(500, int(remaining / 0.008)))

    try:
        result = minimize(f, z, jac=g, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxcor': 120, 'maxls': 60})
        a_new = _softmax(result.x)
        sc_new = _score(a_new)
        if verbose:
            print(f"    L-BFGS (beta={beta:.2e}): score={sc_new:.10f}, nit={result.nit}")
        return a_new, sc_new
    except Exception as e:
        if verbose:
            print(f"    L-BFGS error: {e}")
        return a_init, _score(a_init)


def _adaptive_lbfgs(a_init, deadline, min_remaining=60.0, verbose=True):
    """L-BFGS with adaptive beta schedule - main polishing workhorse."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Compute initial gap
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c_max * 0.002:
        gap = c_max * 0.002

    # Adaptive beta schedule: start moderate, end very small
    multipliers = [0.5, 0.1, 0.05, 0.01, 0.005, 0.001, 0.0005, 0.0001, 0.00005, 0.00001]

    for i, bm in enumerate(multipliers):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 20:
            break

        beta = gap * bm

        # Allocate time proportionally: more time for smaller beta
        time_alloc = min(remaining * 0.7, max(30, 100 * (0.8 ** i)))
        phase_end = time.monotonic() + time_alloc

        a_new, sc_new = _lbfgs_phase(a, phase_end, min_remaining=5.0, beta=beta, verbose=verbose)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
            # Update gap for next iteration
            c = _full_convolve(a, a)
            c_max = float(c.max())
            c_sorted = np.sort(c)[::-1]
            new_gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
            if new_gap > 0:
                gap = new_gap
        else:
            if verbose and i < 5:
                print(f"    beta={beta:.2e}: no improvement (best={best_sc:.10f})")

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.3) ** 2) + np.exp(-40.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    elif kind == "quartic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2.0
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _peak_targeted_descent(a_init, deadline, min_remaining=30.0):
    """Targeted descent: focus on reducing the top few peaks specifically."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score_norm(a)

    step = 0.02
    patience = 0

    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > 200:
                break
            step *= 0.8

        # Find top 5 peaks
        top_k = min(5, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]

        # Compute combined gradient from top peaks
        a_rev = a[::-1]
        w = np.zeros(len(c))
        # Weight peaks by how much above the second-highest they are
        weights = np.zeros(len(c))
        for k_idx in peak_indices:
            weights[k_idx] = max(0, c[k_idx] - c_max * 0.995)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[peak_indices[0]] = 1.0
        else:
            w = weights / w_sum

        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]
        grad_c = grad - float(np.mean(grad))

        # Step
        a_new = a - step * grad_c
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < sc:
            a = a_new
            c = c_new
            step = min(step * 1.05, 0.1)  # slightly increase if improving
        else:
            step *= 0.7

    return best_a, best_sc
