"""

Key improvements over previous:
- Armijo backtracking line search in active-set refinement (vs fixed step)
- Momentum-based subgradient with adaptive active set tolerance

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_soft_structured(a, beta, n):
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft_structured(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)


class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, verbose=True):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(200, len(c_sorted) - 1)]
    if gap < c0_max * 0.002:
        gap = c0_max * 0.002

    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
        0.00003, 0.000015, 0.000008, 0.000004,
        0.000002, 0.000001, 0.0000005
    ])
    betas = gap * multipliers

    t_start = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        size_factor = max(1.0, n / 8192.0)
        est_iter_time = 0.003 * size_factor
        max_iter = min(12000, max(200, int(remaining / (est_iter_time * 15))))

        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 150,
                    'maxls': 80,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    elapsed = time.monotonic() - t_start
                    print(f"    beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}, t={elapsed:.1f}s")
            else:
                if verbose and i < 5:
                    print(f"    beta={beta:.2e}: no improvement (best={best_sc:.10f})")
        except Exception as e:
            if verbose:
                print(f"    beta={beta:.2e}: error {e}")

    return best_a, best_sc


def _active_set_refine(a_init, deadline, min_time_remaining=20.0, verbose=True):
    """Active-set KKT refinement with Armijo backtracking and adaptive tolerance."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_frac = 0.01
    min_tol = 0.00005
    no_improve = 0
    max_no_improve = 200
    momentum = 0.0
    mom_decay = 0.95
    prev_grad = None

    t_start = time.monotonic()
    iter_count = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.5)
            if tol_frac <= min_tol * 1.2:
                if verbose:
                    print(f"    [AS] Stuck at tol={tol_frac:.2e}, breaking. t={time.monotonic()-t_start:.1f}s")
                break
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        # Subtract mean to stay on simplex (direction is in tangent space)
        grad_c = grad - np.mean(grad)

        # Add momentum
        if prev_grad is not None:
            momentum = mom_decay * momentum + (1 - mom_decay) * grad_c
            direction = -momentum
        else:
            momentum = grad_c.copy()
            direction = -grad_c

        # Armijo backtracking line search
        g_norm = float(np.dot(direction, direction))
        if g_norm < 1e-30:
            if no_improve > 50:
                break
            tol_frac = max(min_tol, tol_frac * 0.9)
            continue

        alpha = 0.1
        improved = False
        for _ in range(30):
            a_try = a + alpha * direction
            a_try = np.maximum(a_try, 0.0)
            s_try = float(a_try.sum())
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5
            if alpha < 1e-10:
                break

        if not improved:
            # Try steepest descent direction (no momentum)
            alpha = 0.01
            for _ in range(20):
                a_try = a - alpha * grad_c
                a_try = np.maximum(a_try, 0.0)
                s_try = float(a_try.sum())
                if s_try < 1e-10:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-14:
                    a = a_try
                    improved = True
                    break
                alpha *= 0.5
                if alpha < 1e-12:
                    break

        if not improved:
            tol_frac = max(min_tol, tol_frac * 0.95)
            momentum *= 0.5

        prev_grad = grad_c
        iter_count += 1

        if iter_count % 500 == 0 and verbose:
            print(f"    [AS] iter={iter_count}, sc={best_sc:.10f}, "
                  f"tol={tol_frac:.2e}, n_active={n_active}, t={time.monotonic()-t_start:.1f}s")

    return best_a, best_sc


def _peak_shaving(a_init, deadline, min_time_remaining=15.0):
    """Targeted peak reduction: identify top peaks and reduce contributing a-values."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    no_improve = 0
    max_no_improve = 100

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        # Find top peaks
        top_k = min(10, len(c))
        peak_indices = np.argsort(c)[-top_k:]
        peak_indices = peak_indices[c[peak_indices] > c_max * 0.95]
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(c)])

        # For each peak, find the a-value that contributes most and reduce it
        improved = False
        for k in peak_indices:
            k = int(k)
            # Contributions: a[j] * a[k-j] for valid j
            j_lo = max(0, k - (n - 1))
            j_hi = min(n - 1, k)
            products = np.array([a[j] * a[k - j] for j in range(j_lo, j_hi + 1)])
            if len(products) == 0:
                continue
            max_idx = int(np.argmax(products))
            j_reduce = j_lo + max_idx

            # Try reducing a[j_reduce]
            for frac in [0.7, 0.8, 0.9]:
                a_try = a.copy()
                a_try[j_reduce] *= frac
                s_try = float(a_try.sum())
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-14:
                    a = a_try
                    current_sc = sc_try
                    improved = True
                    break

        if not improved:
            break

    return best_a, best_sc


def _make_profile(n, kind="tapered", rng=None):

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

    p1_end = t_start_global + p1_time
    print(f"\n[P1] L-BFGS main (n={n_main}, budget={p1_time:.0f}s)")
    a1, sc1 = _run_lbfgs(best_seq, p1_end, min_time_remaining=60, verbose=True)
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  P1 result: {best_score:.10f}")

    # === Phase 2: Active-set refinement ===
    p2_time = (budget_s - 10) * 0.30
    p2_end = min(deadline - (budget_s - 10) * 0.35, time.monotonic() + p2_time)
    print(f"\n[P2] Active-set refine (budget={p2_end - time.monotonic():.0f}s)")
    a2, sc2 = _active_set_refine(best_seq, p2_end, min_time_remaining=30, verbose=True)
    if sc2 < best_score:
        best_score = sc2
        best_seq = a2.copy()
        print(f"  P2 result: {best_score:.10f}")

    # === Phase 3: Peak shaving + L-BFGS ===
    print(f"\n[P3] Peak shaving + L-BFGS (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p3_end = min(deadline - (budget_s - 10) * 0.20, time.monotonic() + (budget_s - 10) * 0.15)
        a3, sc3 = _peak_shaving(best_seq, p3_end, min_time_remaining=30)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3a (peak shave): {best_score:.10f}")
        # Follow with L-BFGS
        if time.monotonic() + 30 < p3_end:
            a3b, sc3b = _run_lbfgs(best_seq, p3_end, min_time_remaining=15, verbose=False)
            if sc3b < best_score:
                best_score = sc3b
                best_seq = a3b.copy()
                print(f"  P3b (L-BFGS): {best_score:.10f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n[P4] Multi-n exploration (t={time.monotonic()-t_start_global:.0f}s)")
    n_options = [12288, 20480, 24576, 32768]
    n_options = [x for x in n_options if x != n_main]
    for n_try in n_options:
        if time.monotonic() + 90 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        print(f"  n={n_try}: initial score={sc_r:.10f}")
        if time.monotonic() + 60 > deadline:
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
            continue
        p4_end = min(deadline - 60, time.monotonic() + 120)
        a4, sc4 = _run_lbfgs(a_r, p4_end, min_time_remaining=20, verbose=False)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  n={n_try}: optimized score={best_score:.10f}")

    # === Phase 5: Final polish ===
    print(f"\n[P5] Final polish (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 20 < deadline:
        p5_end = min(deadline - 5, time.monotonic() + (budget_s - 10) * 0.10)
        # Active set first
        a5, sc5 = _active_set_refine(best_seq, p5_end, min_time_remaining=10, verbose=False)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  P5a (active set): {best_score:.10f}")
        # Then L-BFGS
        if time.monotonic() + 15 < p5_end:
            a5b, sc5b = _run_lbfgs(best_seq, p5_end, min_time_remaining=3, verbose=False)
            if sc5b < best_score:
                best_score = sc5b
                best_seq = a5b.copy()
                print(f"  P5b (L-BFGS): {best_score:.10f}")
        # Final active set
        if time.monotonic() + 5 < deadline - 2:
            a5c, sc5c = _active_set_refine(best_seq, deadline - 2, min_time_remaining=2, verbose=False)
            if sc5c < best_score:
                best_score = sc5c
                best_seq = a5c.copy()
                print(f"  P5c (final AS): {best_score:.10f}")

    # Final output
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start_global:.1f}s")
    return [float(x) for x in final]
