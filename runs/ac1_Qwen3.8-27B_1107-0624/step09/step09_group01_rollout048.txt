"""
Optimized soft-max L-BFGS with adaptive temperature, KKT-Newton refinement,
and multi-resolution exploration for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements: adaptive beta scheduling based on flatness detection,
longer L-BFGS convergence, structured KKT with CG inner solver,
and diverse restart strategies.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _log_sum_exp(shifted, beta):
    """Compute log(sum(exp((c - c_max)/beta))) stably."""
    x = (shifted - shifted.max()) / beta
    x = np.clip(x, -700, 0)
    e = np.exp(x)
    return np.log(e.sum() + 1e-300)


def _soft_objective(z, beta, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    val = _log_sum_exp(c, beta)
    return 2.0 * n * (c_max + beta * val)


def _soft_gradient(z, beta, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    x = (c - c_max) / beta
    x = np.clip(x, -700, 0)
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _fft_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W_val)
    return grad


def _gaussian_process(z, beta, n, maxiter, min_remaining, deadline):
    """Run L-BFGS for the smoothed objective. Returns (z, obj_val)."""
    def obj_fn(z_local):
        return _soft_objective(z_local, beta, n)

    def grad_fn(z_local):
        return _soft_gradient(z_local, beta, n)

    try:
        est_time_per_iter = 0.003 * max(1.0, n / 8192.0)
        max_iter_use = max(10, min(8000, int(min_remaining / est_time_per_iter)))
        max_iter_use = min(max_iter_use, maxiter)

        result = minimize(
            obj_fn, z, jac=grad_fn,
            method='L-BFGS-B',
            options={'maxiter': max_iter_use, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxcor': 50, 'maxls': 100}
        )
        return result.x, obj_fn(result.x)
    except Exception:
        return z, obj_fn(z)


def _compute_subgradient(a, n, K=50):
    """Compute subgradient using top-K active set."""
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    K = min(K, 2 * n - 1)
    idx = np.argpartition(c, -K)[-K:]
    W = np.zeros(2 * n - 1)
    W[idx] = 1.0 / K
    a_rev = a[::-1]
    wc = _fft_convolve(W, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(W, c))
    g = 2.0 * a * (h - W_val)
    g = g - np.mean(g)
    return g, c, c_max, W


def _hankel_matvec(v, W, n):
    """Compute H*v where H is the Hankel matrix defined by W."""
    v_rev = v[::-1]
    conv_len = len(v_rev) + len(W) - 1
    conv = _fft_convolve(v_rev, W)
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_hankel(W, n, rhs, max_iter=300, tol=1e-14):
    """Conjugate gradient to solve H*x = rhs where H is the Hankel operator."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.05:
            break
        r = r_new
        beta_cg = rs_new / rs
        p = r + beta_cg * p
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _kkt_refine(a_init, deadline, max_rounds=300):
    """KKT-based Newton refinement for the exact max problem."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    k_schedule = [1, 2, 5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        c_max = float(c.max())

        K = k_schedule[rnd % len(k_schedule)]
        K = min(K, 2 * n - 1)
        idx = np.argpartition(c, -K)[-K:]
        W = np.zeros(2 * n - 1)
        W[idx] = 1.0 / K

        g, _, _, _ = _compute_subgradient(a, n, K)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-13 * n:
            break

        # CG solve for Newton direction
        Hg = _cg_hankel(W, n, g, max_iter=200)
        H1 = _cg_hankel(W, n, np.ones(n), max_iter=200)
        H1_denom = float(np.dot(np.ones(n), H1))
        if abs(H1_denom) < 1e-25 or not np.isfinite(H1_denom):
            dx = -g_t / max(gn, 1e-30)
        else:
            lam = float(np.dot(np.ones(n), Hg)) / H1_denom
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / max(gn, 1e-30)

        # Backtracking line search
        alpha = 1.0
        step_done = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _fft_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1.0 - 0.005 * alpha) - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-15:
                break

        if not step_done:
            # Fallback: steepest descent in log-space
            for trial in [0.01, 0.005, 0.001, 0.0005]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 1e-15)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _fft_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 10:
            break

    return best_a, best_sc

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.02
        return a / a.sum()
    elif kind == "power":
        t = np.linspace(0, 1, n)
        a = (1 - t) ** 0.8 + t ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sin2":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2 + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

        print(f"No parent, using tapered n=16384, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "init")

    n0 = len(parent)

    # Phase 1: Adaptive soft-max L-BFGS (main workhorse)
    print(f"\n[Phase 1] Adaptive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    t_p1_end = min(deadline - 900.0, t_start + 500.0)
    a_p1, sc_p1 = _adaptive_beta_lbfgs(parent, t_p1_end, min_remaining=100.0)
    update_best(a_p1, sc_p1, "lbfgs_phase1")

    # Phase 2: KKT refinement with stable active set
    print(f"\n[Phase 2] KKT Refine (t={time.monotonic() - t_start:.1f}s)")
    t_p2_end = min(deadline - 500.0, t_start + 800.0)
    a_p2, sc_p2 = _kkt_with_stable_active_set(best_seq, t_p2_end, max_rounds=500)
    update_best(a_p2, sc_p2, "kkt_phase2")

    # Phase 3: More L-BFGS with smaller beta (fine tuning)
    print(f"\n[Phase 3] Fine L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    t_p3_end = min(deadline - 400.0, t_start + 1100.0)
    a_p3, sc_p3 = _adaptive_beta_lbfgs(best_seq, t_p3_end, min_remaining=200.0)
    update_best(a_p3, sc_p3, "lbfgs_phase3")

    # Phase 4: Subgradient momentum for escaping local minima
    print(f"\n[Phase 4] Subgrad+Momentum (t={time.monotonic() - t_start:.1f}s)")
    t_p4_end = min(deadline - 300.0, t_start + 1300.0)
    a_p4, sc_p4 = _subgrad_momentum(best_seq, t_p4_end, max_steps=5000)
    update_best(a_p4, sc_p4, "subgrad_phase4")

    # Phase 5: Multi-n exploration
    print(f"\n[Phase 5] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_options = [n0 // 2, n0, n0 * 2]
    for extra_n in [8192, 16384, 32768, 65536]:
        if extra_n not in n_options and 2048 <= extra_n <= 65536:
            n_options.append(extra_n)
    n_options = sorted(set(n_options))

    for n_try in n_options:
        if time.monotonic() > deadline - 150:
            break
        a_res = _resample(best_seq, n_try)
        sc_res = _score(a_res)
        if sc_res < best_score:
            update_best(a_res, sc_res, f"resample_n{n_try}")

        if time.monotonic() < deadline - 80 and n_try <= 32768:
            t_end = min(deadline - 60, time.monotonic() + 30)
            a_opt, sc_opt = _adaptive_beta_lbfgs(a_res, t_end, min_remaining=40.0)
            update_best(a_opt, sc_opt, f"lbfgs_n{n_try}")

    # Phase 6: Diversified restarts
    print(f"\n[Phase 6] Diversified restarts (t={time.monotonic() - t_start:.1f}s)")
    for restart in range(6):
        if time.monotonic() > deadline - 30:
            break
        sigma = 0.01 * (restart + 1)
        a_pert = _perturb_log(best_seq, sigma, rng)
        t_end = min(deadline - 20, time.monotonic() + 15)
        a_r, sc_r = _kkt_with_stable_active_set(a_pert, t_end, max_rounds=50)
        update_best(a_r, sc_r, f"restart_{restart}")

    # Phase 7: Final ultra-fine KKT polish
    print(f"\n[Phase 7] Final KKT polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 2:
        a_fin, sc_fin = _kkt_with_stable_active_set(best_seq, deadline - 1, max_rounds=500)
        update_best(a_fin, sc_fin, "final_kkt")

    # Final output
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        # Verify with evaluate_sequence
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"FINAL: internal_score={sc_final:.10f}, "
                  f"evaluate_score={ev_score:.10f}, n={len(final)}, "
                  f"time={time.monotonic() - t_start:.1f}s")
        except Exception as e:
            print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
                  f"evaluate_error={e}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(8192, "tapered")
    sc_fb = _score(fallback)
    print(f"FALLBACK: score={sc_fb:.10f}, n={len(fallback)}")
    return [float(x) for x in fallback]
