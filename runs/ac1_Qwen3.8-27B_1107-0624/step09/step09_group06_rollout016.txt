def _lp_direction(f, M, K=50, timeout=None):
    """Find direction g that minimizes max(f*g) using LP with top-K constraints."""
    n = len(f)
    c = np.convolve(f, f)
    # Get top K indices
    top_k = np.argsort(c)[::-1][:K]
    
    # Build constraint matrix: (F g)_k <= M for k in top_k
    # (F g)_k = sum_j f[j] * g[k-j] (with bounds)
    # For each k in top_k, the constraint is:
    # sum_j f[max(0,k-n+1):min(n,k+1)] * g[...] <= M
    
    # Actually, (f*g)[k] = sum_{j=0}^{n-1} f[j] * g[k-j] where g is defined for 0..n-1
    # So (f*g)[k] = sum_{j: 0<=j<n, 0<=k-j<n} f[j] * g[k-j]
    #            = sum_{j: max(0,k-n+1) <= j <= min(n-1,k)} f[j] * g[k-j]
    
    # Let's build the A matrix for linprog
    # A @ g <= b, where b[k] = M
    # A[k, j] = f[j] if 0 <= k-j < n, else 0
    
    # For efficiency, we build A sparsely
    from scipy.sparse import csc_matrix
    
    rows = []
    cols = []
    vals = []
    for ii, k in enumerate(top_k):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(ii)
            cols.append(j)
            vals.append(f[j])
    
    A = csc_matrix((vals, (rows, cols)), shape=(len(top_k), n))
    b = np.full(len(top_k), M)
    
    # Maximize sum(g) subject to A@g <= M, g >= 0
    # linprog minimizes, so minimize -sum(g)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        g = res.x
        s = g.sum()
        if s > 1e-10:
            g = g / s
        return g
    return None

"""
LP-based direction finding + L-BFGS polishing for minimizing
2*n*max(conv(a,a))/sum(a)^2.

Key innovation: Uses linear programming to find optimal update directions.
Given current f with peak M, solves max sum(g) s.t. (f*g)[k] <= M for top-K
peaks, g >= 0. The LP solution provides the best direction to reduce the
autocorrelation peak. Combined with L-BFGS polishing and multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csc_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _lp_direction(f, M, K=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-K peaks, g >= 0."""
    n = len(f)
    c = _full_convolve(f, f)
    top_k = np.argsort(c)[::-1][:K]

    rows = []
    cols = []
    vals = []
    for i, k in enumerate(top_k):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        j = np.arange(j_lo, j_hi + 1)
        if len(j) > 0:
            rows.extend([i] * len(j))
            cols.extend(j.tolist())
            vals.extend(f[j].tolist())

    if len(rows) == 0:
        return None, 0.0

    A = csc_matrix((vals, (rows, cols)), shape=(K, n))
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(K, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 5.0})

    if res.success:
        g = res.x
        s = float(g.sum())
        if s > 1e-15:
            g = g / s
        return g, s
    return None, 0.0


def _lp_optimize(f_init, deadline, min_time_remaining=60.0, max_iters=5000):
    """Main LP-based optimizer with adaptive K and line search."""
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    best_f = f.copy()
    c = _full_convolve(f, f)
    best_sc = 2.0 * n * float(c.max())
    cur_sc = best_sc

    K = 10
    K_max = 100
    no_improve = 0
    iter_count = 0

    while time.monotonic() + min_time_remaining < deadline and iter_count < max_iters:
        iter_count += 1
        c = _full_convolve(f, f)
        c_max = float(c.max())
        cur_sc = 2.0 * n * c_max

        if cur_sc < best_sc - 1e-14:
            best_sc = cur_sc
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            break

        # Adaptive K
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
        if gap < c_max * 0.01 and K < K_max:
            K = min(K_max, int(K * 1.2) + 1)

        g, S = _lp_direction(f, c_max, K=K)
        if g is None:
            break

        if S <= 1.0 + 1e-6:
            # No improvement from LP, try with slightly relaxed M
            g, S = _lp_direction(f, c_max * 1.001, K=K)
            if g is None or S <= 1.0 + 1e-6:
                if no_improve > 200:
                    break
                # Perturb and continue
                rng = np.random.default_rng(iter_count)
                f_pert = _project_simplex(f + 0.001 * rng.standard_normal(n))
                c_pert = _full_convolve(f_pert, f_pert)
                if 2.0 * n * float(c_pert.max()) < best_sc:
                    f = f_pert
                    best_f = f.copy()
                    best_sc = 2.0 * n * float(c_pert.max())
                    no_improve = 0
                else:
                    break
            else:
                pass

        # Line search for step size
        best_alpha = 0.1
        best_new_sc = cur_sc
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            h = (f + alpha * g) / (1 + alpha * S * (f.sum() / (f.sum() + 1e-300)))
            # Simpler: h = (f + alpha*S*g) / (1 + alpha*S)
            h = (f + alpha * S * g) / (1.0 + alpha * S)
            c_h = _full_convolve(h, h)
            sc_h = 2.0 * n * float(c_h.max())
            if sc_h < best_new_sc:
                best_new_sc = sc_h
                best_alpha = alpha

        f = (f + best_alpha * S * g) / (1.0 + best_alpha * S)
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-15:
            f = f / s

        if iter_count % 200 == 0:
            print(f"    LP iter={iter_count}: best={best_sc:.10f}, cur={_score(f):.10f}, K={K}, S={S:.4f}")

    return best_f, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta

"""
LP-based direction finding + L-BFGS polishing for minimizing
2*n*max(conv(a,a))/sum(a)^2.

Key innovation: Uses linear programming to find optimal update directions.
Given current f with peak M, solves max sum(g) s.t. (f*g)[k] <= M for top-K
peaks, g >= 0. The LP solution provides the best direction to reduce the
autocorrelation peak. Combined with L-BFGS polishing and multi-n exploration.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _lp_direction(f, M, K=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-K peaks, g >= 0.
    Returns (g_normalized, optimal_sum) or (None, 0)."""
    n = len(f)
    c = _full_convolve(f, f)
    top_k = np.argsort(c)[::-1][:K]

    rows = []
    cols = []
    vals = []
    for i, k in enumerate(top_k):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        j = np.arange(j_lo, j_hi + 1)
        if len(j) > 0:
            rows.extend([i] * len(j))
            cols.extend(j.tolist())
            vals.extend(f[j].tolist())

    if len(rows) == 0:
        return None, 0.0

    A = csc_matrix((vals, (rows, cols)), shape=(K, n))
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(K, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True})
    except Exception:
        return None, 0.0

    if res.success:
        g = res.x
        s = float(g.sum())
        if s > 1e-15:
            g = g / s
        return g, s
    return None, 0.0

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03125, 0.015, 0.0075, 0.00375,
                                  0.0019, 0.00095, 0.0005, 0.00025,
                                  0.00012, 0.00006, 0.00003, 0.000015])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.004 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 8))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-15, 'maxcor': 80, 'maxls': 50})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _nesterov_subgrad(a_init, deadline, min_time_remaining=30.0, K_init=8, K_max=80, seed=0):
    """Nesterov subgradient as fallback."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    momentum = 0.85
    lr = 0.005
    K = K_init
    t = 0
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 1000:
            break

        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
        if gap < c_max * 0.005 and K < K_max:
            K = min(K_max, int(K * 1.3) + 1)

        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]

        grad = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                idx = k - np.arange(j_lo, j_hi + 1)
                grad[j_lo:j_hi + 1] += a[idx]
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_lo2 <= j_hi2:
                idx = np.arange(j_lo2, j_hi2 + 1) - k
                grad[j_lo2:j_hi2 + 1] += a[idx]

        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break

        effective_lr = lr / (gnorm + 1e-18) * 0.1
        v = momentum * v - effective_lr * grad_t
        a_cand = a + v
        a = _project_simplex(a_cand)

        if t % 2000 == 0:
            print(f"    Nesterov t={t}: best={best_sc:.10f}, K={K}")

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _shrinkage_optimize(a, deadline):
    """Try zeroing out small elements."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.001, 0.002, 0.005, 0.01, 0.02]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """LP-based optimizer with L-BFGS polishing for minimum autocorrelation peak."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: LP-based optimization ===
    print(f"\n[P1] LP-based optimization (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 600, t_start + 500)
    if time.monotonic() < p1_deadline:
        a_lp, sc_lp = _lp_optimize(best_seq, p1_deadline, min_time_remaining=30.0,
                                   max_iters=10000)
        if sc_lp < best_score:
            best_seq = a_lp.copy()
            best_score = sc_lp
            print(f"  NEW BEST: {best_score:.10f} [LP] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  LP: no improvement (best={best_score:.10f})")

    # === Phase 2: L-BFGS polishing ===
    print(f"\n[P2] L-BFGS polish (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 400, t_start + 800)
    if time.monotonic() < p2_deadline:
        a_lb = _run_lbfgs(best_seq, p2_deadline, min_time_remaining=30.0)
        sc_lb = _score(a_lb)
        if sc_lb < best_score:
            best_seq = a_lb.copy()
            best_score = sc_lb
            print(f"  NEW BEST: {best_score:.10f} [L-BFGS] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Second LP pass ===
    print(f"\n[P3] LP round 2 (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 300, t_start + 1000)
    if time.monotonic() < p3_deadline:
        a_lp2, sc_lp2 = _lp_optimize(best_seq, p3_deadline, min_time_remaining=30.0,
                                     max_iters=5000)
        if sc_lp2 < best_score:
            best_seq = a_lp2.copy()
            best_score = sc_lp2
            print(f"  NEW BEST: {best_score:.10f} [LP2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: L-BFGS round 2 ===
    print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 200, t_start + 1200)
    if time.monotonic() < p4_deadline:
        a_lb2 = _run_lbfgs(best_seq, p4_deadline, min_time_remaining=30.0)
        sc_lb2 = _score(a_lb2)
        if sc_lb2 < best_score:
            best_seq = a_lb2.copy()
            best_score = sc_lb2
            print(f"  NEW BEST: {best_score:.10f} [L-BFGS2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Nesterov fallback ===
    print(f"\n[P5] Nesterov (t={time.monotonic()-t_start:.1f}s)")
    p5_deadline = min(deadline - 150, t_start + 1350)
    if time.monotonic() < p5_deadline:
        a_nes, sc_nes = _nesterov_subgrad(best_seq, p5_deadline,
                                          min_time_remaining=20.0, K_init=5, K_max=60,
                                          seed=seed + 7)
        if sc_nes < best_score:
            best_seq = a_nes.copy()
            best_score = sc_nes
            print(f"  NEW BEST: {best_score:.10f} [nesterov] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Shrinkage ===
    print(f"\n[P6] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 120:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 80)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} [shrinkage] t={time.monotonic()-t_start:.1f}s")
            if time.monotonic() < deadline - 60:
                a_shr2 = _run_lbfgs(best_seq, deadline - 40, min_time_remaining=20.0)
                sc_shr2 = _score(a_shr2)
                if sc_shr2 < best_score:
                    best_seq = a_shr2.copy()
                    best_score = sc_shr2
                    print(f"  NEW BEST: {best_score:.10f} [post_shrink] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [16384, 12288, 8192]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try == len(best_seq):
            continue
        if n_try > 32768 or n_try < 1024:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")

        if time.monotonic() < deadline - 50:
            a_opt = _run_lbfgs(a_init, deadline - 30, min_time_remaining=20.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 5 - time.monotonic()
    if remaining > 15:
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        c_sorted = np.sort(c_best)[::-1]
        gap = mb - c_sorted[min(10, len(c_sorted) - 1)]
        beta_fin = max(gap * 0.01, mb * 0.00001)
        prob = _LbfgsProb(beta_fin, n)
        max_iter = min(3000, max(20, int(remaining / 0.05)))
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-15, 'maxcor': 60})
            a_fin2 = _softmax(result.x)
            sc_fin2 = _score(a_fin2)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
        except Exception:
            pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
