        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005) * beta_scale

    # Finer geometric temperature schedule (30 steps instead of 17)
    ratio = 0.82
    betas = [base_beta * (ratio ** i) for i in range(35)]
    # Ensure minimum beta
    min_beta = base_beta * 1e-6
    betas = [max(b, min_beta) for b in betas]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        # Scale iterations based on beta position
        frac = i / len(betas)
        if frac < 0.3:
            max_iter = min(5000, max(50, int(remaining / 0.04)))
        elif frac < 0.7:
            max_iter = min(4000, max(30, int(remaining / 0.035)))
        else:
            max_iter = min(6000, max(50, int(remaining / 0.03)))

        est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(max_iter, int(remaining / (est_time_per_iter * 20) + 10))
        max_iter = max(max_iter, 30)

        class _Prob:
            def __init__(self, beta_val, n_val):
                self.beta = beta_val
                self.n = n_val

            def __call__(self, z_val):
                return _smoobj(z_val, self.beta, self.n)

            def grad(self, z_val):
                return _smograd(z_val, self.beta, self.n)

        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"    beta={beta:.2e}: score={sc_new:.10f} (iter={i})")
        except Exception:
            pass

    return current_a


def _peak_equalize(a_init, deadline, min_time_remaining=30.0):
    """Direct peak equalization using analytical gradient with adaptive step size."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_conv(a, a).max())

    lr = 0.015
    no_improve = 0
    max_no_improve = 300
    soft_width = 0.001  # peaks within this fraction of max are "active"

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_conv(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            lr = min(0.03, lr * 1.02)

        # Soft active set: peaks within soft_width of max
        threshold = c_max * (1.0 - soft_width)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        # Cap active set size for stability
        if len(active) > 200:
            top_vals = c[active]
            top_idx_in_active = np.argsort(top_vals)[::-1][:200]
            active = active[top_idx_in_active]

        K = len(active)
        w_S = np.zeros(2 * n - 1)
        w_S[active] = 1.0 / K
        a_rev = a[::-1]
        h = _full_conv(w_S, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Line search with backtracking
        alpha = lr / (gnorm + 1e-15)
        improved = False
        for _ in range(40):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_conv(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.4

        if not improved:
            no_improve += 1
            lr *= 0.9
            soft_width = min(0.01, soft_width * 1.5)
            if no_improve > max_no_improve:
                a = best_a.copy()
                lr = 0.015
                soft_width = 0.001
                no_improve = 0
                # Try a small random perturbation to escape
                rng_local = np.random.default_rng(int(time.monotonic() * 1e6) % 100000)
                perturb = rng_local.standard_normal(n) * 0.01
                perturb -= perturb.mean()
                a = a + perturb
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                if no_improve > max_no_improve + 200:
                    break
        else:
            soft_width = max(0.0005, soft_width * 0.95)

    return best_a, best_sc


def _subgradient_polish(a_init, deadline, min_time_remaining=10.0):
    """Final polish with diminishing step size subgradient method."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_conv(a, a).max())

    max_iters = 3000
    for it in range(max_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_conv(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()

        # Active: peaks within 0.005% of max
        threshold = c_max * (1 - 0.00005)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        w_S = np.zeros(2 * n - 1)
        w_S[active] = 1.0 / len(active)
        a_rev = a[::-1]
        h = _full_conv(w_S, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        eta = 0.005 / (it + 1.0) ** 0.6
        eta = min(eta, 0.3 / (np.max(np.abs(grad)) + 1e-15))

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        # Accept only if clearly better (check every 3 iterations)
        if (it + 1) % 3 == 0:
            c_new = _full_conv(a_new, a_new)
            sc_new = 2.0 * n * float(c_new.max())
            if sc_new < current_sc - 1e-15:
                a = a_new
            # Else keep old a

    return best_a, best_sc


def _perturb_asymmetric(a, rng, pattern="ramp", strength=0.05):
    """Apply asymmetric perturbation to break symmetry."""
    n = len(a)
    t = np.linspace(0, 1, n)
    if pattern == "ramp":
        pert = (t - 0.5)
    elif pattern == "quadratic":
        pert = (t - 0.3) ** 2 - 0.1
    elif pattern == "sine":
        pert = np.sin(2 * np.pi * t) * 0.5 + (t - 0.5) * 0.3
    elif pattern == "poly":
        pert = (t - 0.4) * (1 - t) - 0.05
    else:
        pert = rng.standard_normal(n)
    pert = pert - pert.mean()
    noise = rng.standard_normal(n) * 0.3
    pert = 0.7 * pert + 0.3 * noise
    pert = pert - pert.mean()

    z = np.log(np.maximum(a, 1e-30))
    z = z + strength * pert
    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.25) ** 2) + 0.8 * np.exp(-15.0 * (t - 0.75) ** 2) + 0.03
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _antisymmetric_refine(a_init, deadline, min_time_remaining=30.0):
    """Optimize specifically the antisymmetric component.
    Write a = s + d where s is symmetric, d is antisymmetric.
    Vary d to reduce the peak."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_conv(a, a).max())

    # Extract symmetric and antisymmetric parts
    s = 0.5 * (a + a[::-1])
    d = 0.5 * (a - a[::-1])

    # Optimize d parameterized by a vector
    # d_j = v_j - v_{n-1-j} (antisymmetric by construction)
    # Actually, just optimize d directly with the constraint d = -d[::-1]
    # Parameterize: d[j] = x[j] for j < n//2, d[n-1-j] = -x[j]
    half = n // 2
    x = np.copy(d[:half])
    if n % 2 == 1:
        x = np.copy(d[:half + 1])  # middle element is 0 for antisymmetric

    lr = 0.01
    no_improve = 0

    for it in range(5000):
        if time.monotonic() + min_time_remaining > deadline:
            break

        # Construct a from s and current x
        d_cur = np.zeros(n)
        d_cur[:len(x)] = x
        d_cur[n - 1 : n - 1 - len(x) : -1] = -x
        a_cur = s + d_cur
        a_cur = np.maximum(a_cur, 0.0)
        sc_sum = a_cur.sum()
        if sc_sum < 1e-15:
            break
        a_cur = a_cur / sc_sum

        c = _full_conv(a_cur, a_cur)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a_cur.copy()
            no_improve = 0
            lr = min(0.02, lr * 1.01)
        else:
            no_improve += 1
            if no_improve > 100:
                break
            lr *= 0.95

        # Gradient with respect to x: project the full gradient onto the antisymmetric subspace
        threshold = c_max * (1 - 0.001)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        w_S = np.zeros(2 * n - 1)
        w_S[active] = 1.0 / len(active)
        a_rev = a_cur[::-1]
        h = _full_conv(w_S, a_rev)[n - 1:2 * n - 1]
        grad_full = 2.0 * h
        grad_full = grad_full - np.mean(grad_full)

        # Project onto antisymmetric subspace
        grad_anti = 0.5 * (grad_full - grad_full[::-1])
        grad_x = grad_anti[:len(x)]
        gnorm = np.linalg.norm(grad_x)
        if gnorm < 1e-14:
            break

        x_new = x - lr * grad_x / (gnorm + 1e-15)
        # Check if this gives a valid a (non-negative)
        d_new = np.zeros(n)
        d_new[:len(x_new)] = x_new
        d_new[n - 1 : n - 1 - len(x_new) : -1] = -x_new
        a_test = s + d_new
        if np.any(a_test < -1e-10):
            # Scale back
            factor = 0.5
            x_new = x - lr * factor * grad_x / (gnorm + 1e-15)
            d_new = np.zeros(n)
            d_new[:len(x_new)] = x_new
            d_new[n - 1 : n - 1 - len(x_new) : -1] = -x_new
            a_test = s + d_new
            if np.any(a_test < -1e-10):
                continue
        x = x_new

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase optimizer with improved temperature schedule,
    antisymmetric refinement, and direct peak equalization."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS warmup with fine temperature schedule ===
    print(f"\n[P1] L-BFGS warmup (t={time.monotonic()-t_start:.1f}s)")
    a_p1 = _run_lbfgs(parent, deadline, min_time_remaining=900.0, beta_scale=1.0, verbose=True)
    sc_p1 = _score(a_p1)
    if sc_p1 < best_score:
        best_seq = a_p1.copy()
        best_score = sc_p1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_warmup]")
    else:
        print(f"  L-BFGS warmup: {sc_p1:.10f} (no improvement over {best_score:.10f})")

    # === Phase 2: Antisymmetric refinement ===
    print(f"\n[P2] Antisymmetric refinement (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        pol_deadline = min(deadline - 400, time.monotonic() + 300)
        a_anti, sc_anti = _antisymmetric_refine(best_seq, pol_deadline, min_time_remaining=50.0)
        if sc_anti < best_score:
            best_seq = a_anti.copy()
            best_score = sc_anti
            print(f"  NEW BEST: {best_score:.10f} [antisym]")
        else:
            print(f"  Antisym: {sc_anti:.10f} (no improvement)")

    # === Phase 3: Peak equalization ===
    print(f"\n[P3] Peak equalization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        pe_deadline = min(deadline - 250, time.monotonic() + 200)
        a_eq, sc_eq = _peak_equalize(best_seq, pe_deadline, min_time_remaining=40.0)
        if sc_eq < best_score:
            best_seq = a_eq.copy()
            best_score = sc_eq
            print(f"  NEW BEST: {best_score:.10f} [peak_eq]")
        else:
            print(f"  Peak equalize: {sc_eq:.10f} (no improvement)")

    # === Phase 4: Asymmetric perturbation + re-optimize ===
    print(f"\n[P4] Asymmetric exploration (t={time.monotonic()-t_start:.1f}s)")
    patterns = ["ramp", "quadratic", "sine", "poly"]
    for trial in range(4):
        if time.monotonic() > deadline - 250:
            break
        strength = 0.02 * (trial + 1)
        pattern = patterns[trial % len(patterns)]
        a_asym = _perturb_asymmetric(best_seq, rng, pattern=pattern, strength=strength)
        a_opt = _run_lbfgs(a_asym, deadline, min_time_remaining=200.0, beta_scale=0.5)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (trial={trial}, pattern={pattern})")
        else:
            print(f"  Asym trial {trial} ({pattern}): {sc_opt:.10f}")

    # === Phase 5: Second peak equalization ===
    print(f"\n[P5] Peak equalization 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        pe2_deadline = min(deadline - 120, time.monotonic() + 100)
        a_eq2, sc_eq2 = _peak_equalize(best_seq, pe2_deadline, min_time_remaining=30.0)
        if sc_eq2 < best_score:
            best_seq = a_eq2.copy()
            best_score = sc_eq2
            print(f"  NEW BEST: {best_score:.10f} [peak_eq2]")

    # === Phase 6: Fine L-BFGS ===
    print(f"\n[P6] Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        a_fine = _run_lbfgs(best_seq, deadline, min_time_remaining=60.0, beta_scale=0.3)
        sc_fine = _score(a_fine)
        if sc_fine < best_score:
            best_seq = a_fine.copy()
            best_score = sc_fine
            print(f"  NEW BEST: {best_score:.10f} [fine_lbfgs]")
        else:
            print(f"  Fine L-BFGS: {sc_fine:.10f}")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([len(best_seq), 16384, 20480, 12288, 24576]), reverse=True)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init={sc_init:.10f}")
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=40.0, beta_scale=0.5)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try})")
        else:
            print(f"  n={n_try}: {sc_opt:.10f} (no improvement)")

    # === Phase 8: Final subgradient polish ===
    print(f"\n[P8] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        sg_deadline = min(deadline - 20, time.monotonic() + 80)
        a_sg, sc_sg = _subgradient_polish(best_seq, sg_deadline, min_time_remaining=15.0)
        if sc_sg < best_score:
            best_seq = a_sg.copy()
            best_score = sc_sg
            print(f"  NEW BEST: {best_score:.10f} [subgrad]")
        else:
            print(f"  Subgrad: {sc_sg:.10f}")

    # === Phase 9: Final ultra-fine L-BFGS + peak equalize ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 5 - time.monotonic()
    if remaining > 15:
        a_final = _run_lbfgs(best_seq, deadline, min_time_remaining=5.0, beta_scale=0.15)
        sc_final = _score(a_final)
        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final
            print(f"  NEW BEST: {best_score:.10f} [final_lbfgs]")

        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _peak_equalize(best_seq, deadline - 2, min_time_remaining=2.0)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_eq]")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
