"""
KKT Active Set Descent for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses principled KKT conditions with QP-based direction finding on adaptive
active sets, combined with L-BFGS smoothing and multi-restart strategy.
Key innovation: exploits the autocorrelation structure to compute the QP
Hessian in O(n log n + m^2) per iteration via FFT.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _autocorr(a):
    """Compute autocorrelation R[l] = sum_u a[u]*a[u+l] for l in [-(n-1), n-1].
    Returns array of length 2n-1 where index n-1+l corresponds to lag l."""
    n = len(a)
    a_rev = a[::-1]
    return _full_convolve(a, a_rev)


def _kkt_direction(a, active_idx, autocorr_arr):
    """Compute KKT descent direction using QP on active set.
    Returns (direction, qp_value, weights)."""
    n = len(a)
    m = len(active_idx)
    
    if m == 1:
        k = active_idx[0]
        # Gradient of c[k]
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        g = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
            g[i] = 2.0 * a[k - i]
        g_centered = g - g.mean()
        gn = np.linalg.norm(g_centered)
        if gn < 1e-15:
            return np.zeros(n), 0.0, np.array([1.0])
        return -g_centered / gn, gn, np.array([1.0])
    
    # Compute means: mp = mean(g_p) for each active peak p
    # g_p[i] = 2*a[p-i] for valid i
    # sum_i g_p[i] = 2 * sum of a[p-i] for valid i
    prefix = np.cumsum(a)
    total = prefix[-1]
    
    means = np.zeros(m)
    for idx, k in enumerate(active_idx):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        s = prefix[i_hi] - (prefix[i_lo - 1] if i_lo > 0 else 0.0)
        means[idx] = 2.0 * s / n
    
    # Compute Hessian H[p,q] = 4*R[q-p] - n*mp*nq
    # autocorr_arr[n-1+l] = R[l] = sum_u a[u]*a[u+l]
    H = np.zeros((m, m))
    for p in range(m):
        for q in range(m):
            lag = int(active_idx[q]) - int(active_idx[p])
            # R[lag] = autocorr_arr[n-1+lag]
            ac_idx = n - 1 + lag
            if 0 <= ac_idx < len(autocorr_arr):
                R_val = autocorr_arr[ac_idx]
            else:
                R_val = 0.0
            H[p, q] = 4.0 * R_val - n * means[p] * means[q]
    
    # Symmetrize
    H = 0.5 * (H + H.T)
    
    # Solve QP: min (1/2) w^T H w s.t. w >= 0, sum(w) = 1
    # Use projected gradient on simplex
    w = np.ones(m) / m
    for it in range(200):
        grad = H @ w
        # Projected gradient step with backtracking
        lr = 2.0 / (np.abs(H).max(axis=1).max() + 1e-30)
        w_new = w - lr * grad
        # Project onto simplex
        w_new = _project_simplex(w_new)
        if np.max(np.abs(w_new - w)) < 1e-12:
            w = w_new
            break
        w = w_new
    
    qp_val = float(w @ H @ w)
    
    # Compute direction: d = -G_c^T w
    G_cw = np.zeros(n)
    for idx, k in enumerate(active_idx):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            G_cw[i] += w[idx] * (2.0 * a[k - i] - means[idx])
    
    gnorm = np.linalg.norm(G_cw)
    if gnorm < 1e-15:
        return np.zeros(n), qp_val, w
    
    return -G_cw, qp_val, w


def _project_simplex(v):
    """Project v onto the probability simplex {w >= 0, sum(w) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _kkt_descent(a_init, deadline, min_time_remaining=30.0):

    n = len(a_init)
    a = _normalize(a_init.astype(np.float64).copy())

    best_sc = _score(a)
    
    print(f"  KKT descent start: n={n}, score={best_sc:.10f}")

    min_tol = 0.0001
    patience = 0
    max_patience = 300
    stall_count = 0
    max_stall = 50

        c_max = float(c.max())

            stall_count += 1

def _topk_smoothed_obj_grad(z, beta, K, n):
    """Top-K log-sum-exp smoothed objective and gradient."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c
    
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    
    # Gradient
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    ga = 2.0 * wc[n - 1:2 * n - 1]
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z

    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    schedule = [
        (1.0, 0.15), (0.7, 0.10), (0.5, 0.07), (0.3, 0.05),
        (0.2, 0.035), (0.1, 0.025), (0.05, 0.015), (0.03, 0.01),

        if time.monotonic() + min_time_remaining > deadline:

        remaining = deadline - min_time_remaining - time.monotonic()

            break

        def obj_func(z2):
            return _topk_smoothed_obj_grad(z2, beta, K, n)[0]
        
        def grad_func(z2):
            return _topk_smoothed_obj_grad(z2, beta, K, n)[1]
        
        try:
            result = minimize(
                obj_func, z, jac=grad_func, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-13,
                         'maxcor': 150, 'maxls': 60}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                base_beta = max_c * 0.003  # Rescale
        except Exception:
            pass
    
    return current_a


def _pairwise_descent(a_init, deadline, min_time_remaining=10.0):
    """Sequentially reduce pairs of the tallest peaks."""
    n = len(a_init)
    a = _normalize(a_init.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    patience = 0
    max_pat = 500
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_pat:
                break
        
        # Find top 2 peaks
        top2 = np.argsort(c)[::-1][:2]
        k1, k2 = int(top2[0]), int(top2[1])
        
        if k1 == k2:
            break
        
        # Compute gradients
        g1 = np.zeros(n)
        g2 = np.zeros(n)
        i_lo1 = max(0, k1 - n + 1)
        i_hi1 = min(n - 1, k1)
        for i in range(i_lo1, i_hi1 + 1):
            g1[i] = 2.0 * a[k1 - i]
        
        i_lo2 = max(0, k2 - n + 1)
        i_hi2 = min(n - 1, k2)
        for i in range(i_lo2, i_hi2 + 1):
            g2[i] = 2.0 * a[k2 - i]
        
        # Center
        g1c = g1 - g1.mean()
        g2c = g2 - g2.mean()
        
        # Find direction that reduces both: minimize max(g1c^T d, g2c^T d)
        # This is: minimize t s.t. g1c^T d <= t, g2c^T d <= t, ||d|| <= 1
        # Approximate: d = -(g1c + g2c) / ||g1c + g2c||
        d = -(g1c + g2c)
        dn = np.linalg.norm(d)
        if dn < 1e-15:
            break
        d = d / dn
        
        # Line search
        alpha = 0.05
        improved = False
        for ls in range(40):
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            # Try individual peak reduction
            for g_peak in [g1c, g2c]:
                d2 = -g_peak
                d2n = np.linalg.norm(d2)
                if d2n < 1e-15:
                    continue
                d2 = d2 / d2n
                alpha2 = 0.02
                for ls in range(30):
                    a_try = a + alpha2 * d2
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-10:
                        alpha2 *= 0.5
                        continue
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    sc_try = 2.0 * n * float(c_try.max())
                    if sc_try < current_sc - 1e-13:
                        a = a_try
                        improved = True
                        break
                    alpha2 *= 0.5
                if improved:
                    break
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2) + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer using KKT active set descent as the primary method,
    with L-BFGS smoothing and pairwise descent as supplements.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent if available
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n0 = 16384
        best_seq = _make_profile(n0, "tapered")
        best_score = _score(best_seq)
        print(f"No parent, using tapered n={n0}, score={best_score:.10f}")
    
    # === Phase 1: Quick L-BFGS warmup ===
    print(f"\n[P1] L-BFGS warmup (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 1400:
        a1 = _run_lbfgs(best_seq, min(deadline - 1300, time.monotonic() + 120),
                        min_time_remaining=30)
        sc1 = _score(a1)
        if sc1 < best_score:
            best_seq = a1.copy()
            best_score = sc1
            print(f"  NEW BEST: {best_score:.10f} [P1 lbfgs] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 2: KKT Active Set Descent (main) ===
    print(f"\n[P2] KKT Active Set Descent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a2, sc2 = _kkt_descent(best_seq, min(deadline - 300, time.monotonic() + 800),
                               min_time_remaining=60)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [P2 kkt] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 3: L-BFGS on improved point ===
    print(f"\n[P3] L-BFGS refinement (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a3 = _run_lbfgs(best_seq, min(deadline - 200, time.monotonic() + 200),
                        min_time_remaining=30)
        sc3 = _score(a3)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [P3 lbfgs] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 4: Second KKT pass ===
    print(f"\n[P4] KKT round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a4, sc4 = _kkt_descent(best_seq, min(deadline - 100, time.monotonic() + 150),
                               min_time_remaining=30)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [P4 kkt2] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 5: Pairwise descent ===
    print(f"\n[P5] Pairwise descent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        a5, sc5 = _pairwise_descent(best_seq, min(deadline - 50, time.monotonic() + 80),
                                    min_time_remaining=10)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [P5 pairwise] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 6: Multi-restart exploration ===
    print(f"\n[P6] Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        n_restarts = 5
        for i in range(n_restarts):
            if time.monotonic() > deadline - 40:
                break
            sigma = [0.02, 0.03, 0.05, 0.01, 0.04][i % 5]
            a_pert = _perturb(best_seq, sigma, rng)
            # Quick KKT from perturbed point
            a_r, sc_r = _kkt_descent(a_pert, min(deadline - 30, time.monotonic() + 20),
                                     min_time_remaining=5)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.10f} [P6 restart_{i}] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 7: Final KKT polish ===
    print(f"\n[P7] Final KKT polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a7, sc7 = _kkt_descent(best_seq, deadline - 2, min_time_remaining=2)
        if sc7 < best_score:
            best_seq = a7.copy()
            best_score = sc7
            print(f"  NEW BEST: {best_score:.10f} [P7 final] t={time.monotonic()-t_start:.1f}s")
    
    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
