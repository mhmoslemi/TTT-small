def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    n = len(a)
    s = float(a.sum())

        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    """Flatten the power spectrum iteratively."""
    n = len(a)

        # A_new = A * (P_mean / P)^alpha (in magnitude)
        ratio = (P_mean / (P + 1e-30)) ** alpha
        A_new = A * ratio
        a_work = np.fft.irfft(A_new, N)[:n]
        a_work = np.maximum(a_work, 0)
        s = a_work.sum()
        if s > 1e-15:

    fw = np.fft.rfft(w, N)
    result = np.fft.irfft(fa * fw, N)[:out_len]
    return result[n-1:2*n-1]

"""
Optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Frequency-domain spectral flattening for diverse initialization
2. L-BFGS with log-sum-exp continuation (improved multi-stage schedule)
3. Adaptive active-set Newton with dynamic active set recomputation
4. Kick-and-refine diversification loop
5. Multi-scale (n=16384 and n=32768)
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _spectral_flatten(a, n_iters=50, alpha=0.5):
    """Flatten the power spectrum to reduce convolution peaks."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    a_work = a.copy().astype(np.float64)
    for it in range(n_iters):
        A = np.fft.rfft(a_work, N)
        P = np.abs(A) ** 2
        P_mean = P.mean() + 1e-30
        # Soft flattening: reduce peaks, boost valleys
        # A_new magnitude = P^(alpha/2) * P_mean^((1-alpha)/2) ... 
        # Simpler: ratio = (P_mean / P)^alpha, A_new = A * ratio
        ratio = np.power(P_mean / (P + 1e-30), alpha)
        A_new = A * ratio
        a_work = np.fft.irfft(A_new, N)[:n]
        a_work = np.maximum(a_work, 0)
        s = a_work.sum()
        if s > 1e-15:
            a_work = a_work / s
    return a_work


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _correlate_a_w(a, w, n):
    """Cross-correlation: result[i] = sum_k w[k] * a[k-i] for valid k-i"""
    flip_a = a[::-1]
    out_len = n + len(w) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fw = np.fft.rfft(w, N)
    result = np.fft.irfft(fa * fw, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    wc = _correlate_a_w(a, w, n)
    W = float(np.dot(w, c))
    return 4.0 * n * a * (wc - W)

    n = len(a_init)
    current_a = a_init.copy()

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    log_betas = np.linspace(np.log(beta_start), np.log(beta_end), n_stages)
    
    for i, log_beta in enumerate(log_betas):
        if time.monotonic() > deadline:

        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        
        beta = np.exp(log_beta)
        
        # Estimate iterations based on remaining time
        est_time_per_iter = 0.005 * max(1.0, n / 8192.0)
        max_iter = min(3000, max(20, int(remaining / (est_time_per_iter * 15))))
        
        try:
            result = minimize(
                _lsobj,
                z,
                args=(beta, n
