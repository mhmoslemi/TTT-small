Let me think about the mirror descent more carefully.

For the mirror descent with multiplicative weights:
- State: $p \in \Delta_{n-1}$ (simplex)
- At each step:
  1. Compute $c = p*p$ (via FFT)
  2. Find $k^* = \arg\max c$
  3. Compute subgradient: $g[j] = 2p[k^*-j]$ (with zero-padding)
  4. Project to tangent space: $g_t = g - \bar{g}$
  5. Polyak step: $\eta = (f(p) - f_{lb}) / \|g_t\|^2$
  6. Multiplicative update: $p \leftarrow p \cdot \exp(-\eta g_t) / Z$

The issue is that with a single subgradient, the method might oscillate (always chasing the same peak). To address this:
- Use **randomized subgradients**: with probability proportional to $c[k]/\sum c$, pick a random $k$ instead of the argmax
- Use **exponential averaging** of past iterates
- Use **adaptive step size** with diminishing schedule

Actually, for the max function, using the argmax subgradient is standard and should work. The issue is convergence speed.

Let me think about the convergence rate. For mirror descent on the simplex with the KL divergence:
$f(\bar{p}_T) - f^* \leq \frac{D}{\eta T} + \frac{\eta}{2} \sum_{t=1}^T \|g_t\|^2$

where $D$ is the diameter of the simplex (in KL divergence) and $\bar{p}_T$ is the average iterate.

With Polyak step, we get:
$f(\bar{p}_T) - f^* \leq \frac{(f_0 - f^*) \|g\|_{max}^2}{2T}$

So after $T$ iterations, the gap is $O(1/T)$. To get a gap of $10^{-4}$ (i.e., score improvement of $10^{-4}$ over the lower bound), we need $T \approx 10^8$ iterations. At 0.1ms per iteration, that's 10000 seconds. Too slow!

With Nesterov acceleration, we can get $O(1/T^2)$, which would give $T \approx 10^4$ iterations. That's more feasible.

But Nesterov acceleration for mirror descent is non-trivial. Let me use a simpler approach:

**Heavy ball + mirror descent**:

- Use $v_t$ instead of $g_t$ in the update

Actually, I just realized something important. The mirror descent step with the subgradient $g[j] = 2p[k^*-j]$ has a very special property:

The update $p_{new}[j] = p[j] \exp(-\eta (g[j] - \bar{g}))$ means:
- For $j$ in the active segment (where $g[j] > 0$): $p$ decreases
- For $j$ outside the active segment: $p$ increases (since $g[j] - \bar{g} < 0$)

The amount of decrease is proportional to $p[j] \cdot p[k^*-j]$, which is exactly the contribution of position $j$ to the peak. So the method is automatically doing "peak shaving" in a principled way.

"""
Optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Mirror descent with multiplicative weights (subgradient method on non-smooth max)
2. LSE-continued L-BFGS for smooth approximation
3. Spectral peak shaving in frequency domain
4. Coordinate descent polish
5. Multiple diverse initializations and restarts
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _subgradient_conv_max(p, k_star):
    """Compute subgradient of max(p*p) w.r.t. p, when k_star is the argmax."""
    n = len(p)
    g = np.zeros(n)
    # g[j] = 2 * p[k_star - j] (with zero-padding)
    # Valid j: max(0, k_star - n + 1) <= j <= min(n-1, k_star)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if hi >= lo:
        # g[lo:hi+1] = 2 * p[k_star-hi : k_star-lo+1][::-1]
        g[lo:hi+1] = 2.0 * p[k_star - hi : k_star - lo + 1][::-1]
    return g

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)

def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _mirror_descent(p_init, deadline, max_iters=2000000):
    """
    Mirror descent (multiplicative weights) with Polyak step size.
    Optimizes the non-smooth max autocorrelation directly.
    """
    n = len(p_init)
    p = p_init.copy()
    p = np.maximum(p, 0)
    s = p.sum()
    p = p / s
    
    f_best = _score(p)
    p_best = p.copy()
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    
    # Running average for FTRL-like approach
    p_avg = p.copy()
    count = 0
    
    no_improve = 0
    best_no_improve_score = f_best
    
    for t in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(p, p)
        k_star = int(np.argmax(c))
        f_val = float(c[k_star])
        score_val = 2.0 * n * f_val
        
        if score_val < f_best - 1e-14:
            f_best = score_val
            p_best = p.copy()
            no_improve = 0
            best_no_improve_score = f_best
        else:
            no_improve += 1
        
        # Update average
        count += 1
        p_avg = ((count - 1) * p_avg + p) / count
        
        # Compute subgradient
        g = _subgradient_conv_max(p, k_star)
        g_t = g - g.mean()
        g_norm_sq = float(np.dot(g_t, g_t))
        
        if g_norm_sq < 1e-30:
            break
        
        # Polyak step size
        gap = max(score_val - f_lb, 1e-10)
        eta = gap / g_norm_sq
        eta = min(eta, 0.5)  # cap for stability
        
        # Multiplicative update
        update = -eta * g_t
        update -= update.max()
        weights = np.exp(update)
        p_new = p * weights
        p_new_sum = p_new.sum()
        if p_new_sum < 1e-30:
            break
        p_new = p_new / p_new_sum
        p = p_new
        
        # Periodically check the average
        if count % 1000 == 0:
            sc_avg = _score(p_avg)
            if sc_avg < f_best - 1e-10:
                f_best = sc_avg
                p_best = p_avg.copy()
    
    return p_best, f_best

def _lbfgs_lse(p_init, deadline, min_time_remaining=30.0):
    """L-BFGS with LSE continuation."""
    n = len(p_init)
    c_cur = _full_convolve(p_init, p_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return p_init
    
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 50:
        gap = c_sorted[0] - c_sorted[50]
    else:
        gap = c_sorted[0] - c_sorted[-1]
    base_beta = max(gap * 0.3, max_c * 0.005)
    
    n_betas = 20
    log_betas = np.linspace(np.log(base_beta), np.max([np.log(max_c * 1e-6), -30]), n_betas)
    betas = np.exp(log_betas)
    
    current_a = p_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        
        est_time = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 10))))
        
        try:
            result = minimize(

                z,
                args=(beta, n),

                    'maxls': 200,

            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:

                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a
