def _next_pow2(n):

    fa = np.fft.rfft(a, N)

        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _power_sum_value(c, p):
    """Smooth max using power sum: (sum c_k^p)^(1/p)"""
    c_pos = np.maximum(c, 1e-30)
    log_val = (np.log(c_pos) * p).max()
    if log_val > 700 / p:
        log_val = 700 / p
    shifted = p * np.log(c_pos) - log_val
    s = np.exp(np.clip(shifted, -700, 0)).sum()
    return float(np.exp(log_val / p + np.log(s) / p))


def _grad_soft_structured(a, beta, n):
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft_structured(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)


class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0, verbose=True):
    """L-BFGS with aggressive adaptive beta schedule."""
    global t_start_global
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c0_max * 0.003:
        gap = c0_max * 0.003

    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
        0.00003125, 0.000015625, 0.0000078, 0.0000039,
        0.000002, 0.000001, 0.0000005, 0.0000002, 0.0000001
    ])
    betas = gap * multipliers

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        size_factor = max(1.0, n / 4096.0)
        est_iter_time = 0.002 * size_factor
        max_iter = min(10000, max(200, int(remaining / (est_iter_time * 15))))

        if beta < gap * 0.0001:
            max_iter = min(max_iter, 3000)

        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 200,
                    'maxls': 80,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    elapsed = time.monotonic() - t_start_global
                    print(f"    beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}, t={elapsed:.0f}s")
        except Exception as e:
            if verbose:
                print(f"    beta={beta:.2e}: error {e}")

    return best_a, best_sc


def _spectral_equalization(a_init, deadline, min_time_remaining=30.0, verbose=True):
    """SA-inspired Successive Overweighting for autocorrelation flattening.
    
    Inspired by PAPR reduction algorithms. At each step:
    1. Find peak lags in autocorrelation
    2. Compute "pressure" on each a[j] from peaks
    3. Apply weighted update to reduce peak contribution
    4. Project to simplex
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    no_improve = 0
    max_no_improve = 500
    tol_start = 0.01
    tol = tol_start
    alpha_scale = 1.0

    for iteration in range(100000):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            if tolerance check:  # tighten and reset
                pass
            tol *= 0.7
            alpha_scale *= 0.8
            no_improve = 0
            if tol < 0.0005 and alpha_scale < 0.01:
                break

        # Identify active (peak) set
        peak_threshold = c_max * (1.0 - tol)
        active = c >= peak_threshold
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros(len(c), dtype=bool)
            active[idx] = True
            n_active = 1

        # Compute pressure: gradient of avg peak w.r.t. a
        a_rev = a[::-1]
        h = np.zeros(len(c))
        h[active] = 1.0 / n_active
        hc = _full_convolve(h, a_rev)
        pressure = 2.0 * hc[n - 1:2 * n - 1]

        # Remove mean to stay on simplex (sum-preserving direction)
        pressure_centered = pressure - np.mean(pressure)

        # Adaptive step size: aim to reduce peak by a target fraction
        # The change in c[k] from delta_a is approximately 2*conv(delta_a, a)
        # We want 2*conv(-alpha*pressure_centered, a)[peak] ≈ -target_frac * c_max
        # Simplified: use BB-like step
        p_norm_sq = float(np.dot(pressure_centered, pressure_centered))
        if p_norm_sq < 1e-30:
            break

        # Estimate step from current gap
        gap_to_bound = c_max - 1.0 / (2 * n - 1)
        if gap_to_bound < 1e-15:
            break
        step = alpha_scale * gap_to_bound / (n * p_norm_sq / (2 * n))
        step = min(step, 0.05 * a.max() / max(np.abs(pressure_centered).max(), 1e-15))
        step = max(step, 1e-10)

        # Apply update
        a_new = a - step * pressure_centered
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Line search: try a few step sizes
        best_step_sc = current_sc
        best_a_try = a
        for trial_frac in [1.0, 0.5, 0.25, 0.1]:
            a_try = a - step * trial_frac * pressure_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_step_sc:
                best_step_sc = sc_try
                best_a_try = a_try.copy()

        if best_step_sc < current_sc - 1e-14:
            a = best_a_try
            # Sometimes increase alpha if making progress
            if iteration % 50 == 0 and no_improve == 0:
                alpha_scale = min(alpha_scale * 1.1, 5.0)
        else:
            # No improvement, reduce step
            alpha_scale *= 0.85
            tol = max(tol * 0.9, 0.0005)

        if verbose and iteration % 200 == 0:
            elapsed = time.monotonic() - t_start_global
            print(f"    SE iter={iteration}: score={best_sc:.10f}, "
                  f"tol={tol:.4f}, alpha={alpha_scale:.3f}, t={elapsed:.0f}s")

    return best_a, best_sc


def _nesterov_projected(a_init, deadline, min_time_remaining=30.0, verbose=True):
    """Nesterov-accelerated projected gradient with Barzilai-Borwein steps."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    y = a.copy()  # momentum variable
    a_prev = a.copy()
    g_prev = None
    alpha_bb = 0.01

    for iteration in range(200000):
        if time.monotonic() + min_time_remaining > deadline:
            break

        # Nesterov extrapolation
        y = a + alpha_bb * (a - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        else:
            y = a.copy()

        # Compute gradient at y
        c = _full_convolve(y, y)
        c_max = float(c.max())
        # Simple: use top-k peaks
        k = min(50, len(c))
        top_k = np.argsort(c)[::-1][:k]
        active = np.zeros(len(c), dtype=bool)
        active[top_k] = True
        n_active = k

        a_rev = y[::-1]
        h = np.zeros(len(c))
        h[active] = 1.0 / n_active
        hc = _full_convolve(h, a_rev)
        g = 2.0 * hc[n - 1:2 * n - 1]
        g = g - np.mean(g)  # project to tangent of simplex

        # BB step size
        if g_prev is not None:
            s_vec = y - a_prev
            y_vec = g - g_prev
            sy = float(np.dot(s_vec, y_vec))
            yy = float(np.dot(y_vec, y_vec))
            if yy > 1e-20 and sy > 1e-20:
                alpha_bb = min(max(sy / yy, 1e-6), 0.1)
            else:
                alpha_bb *= 0.9

        # Gradient step
        a_new = y - alpha_bb * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-13:
            best_sc = sc_new
            best_a = a_new.copy()

        a_prev = a
        a = a_new
        g_prev = g

        if verbose and iteration % 500 == 0:
            elapsed = time.monotonic() - t_start_global
            print(f"    NGD iter={iteration}: best={best_sc:.10f}, "
                  f"alpha={alpha_bb:.6f}, t={elapsed:.0f}s")

    return best_a, best_sc


def _kkt_refine(a_init, deadline, min_time_remaining=30.0, verbose=True):
    """KKT active-set refinement with linear system solving.
    
    At optimum with active set S and support T:
    - sum_{k in S} a[k-j] = const for all j in T
    - This is a linear system in a restricted to T
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    no_improve = 0
    max_no_improve = 200
    tol_frac = 0.003
    min_tol = 0.0001

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.6)
            if tol_frac <= min_tol * 1.1:
                break
            no_improve = 0

        # Active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Compute gradient (average over active set)
        a_rev = a[::-1]
        h = np.zeros(len(c))
        h[active] = 1.0 / n_active
        hc = _full_convolve(h, a_rev)
        grad_avg = 2.0 * hc[n - 1:2 * n - 1]

        # Support
        support = a > 1e-6 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        # KKT: grad_avg should be constant on support
        lam = float(np.mean(grad_avg[support]))
        residual = grad_avg[support] - lam

        # Try to solve: adjust a[support] to make grad_avg[support] = lam
        # This is: sum_{k in active} a[k-j] = R for j in support
        # Linear in a, but only for j in support
        # Use iterative approach: damped update
        eta = 0.03
        a_new = a.copy()
        a_new[support] -= eta * residual
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new >= current_sc:
            found = False
            for trial_eta in [0.01, 0.005, 0.002, 0.001]:
                a_try = a.copy()
                a_try[support] -= trial_eta * residual
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-12:
                    a_new = a_try
                    sc_new = sc_try
                    found = True
                    break
            if not found:
                tol_frac = max(min_tol, tol_frac * 0.8)
                continue

        a = a_new
        if no_improve == 0:
            tol_frac = max(min_tol, tol_frac * 0.97)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.6 * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS, spectral equalization,
    Nesterov gradient, and KKT refinement.
    
    Phases:
    P1: L-BFGS coarse (30% time)
    P2: Spectral equalization (20% time)  
    P3: L-BFGS fine (25% time)
    P4: KKT refinement (15% time)
    P5: Nesterov + final L-BFGS (10% time)
    """
    global t_start_global
    t_start_global = time.monotonic()
    deadline = t_start_global + max(1.0, float(budget_s) - 8.0)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(16384, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n=16384: score={best_score:.10f}")

    n_main = len(best_seq)

    # === Phase 1: L-BFGS coarse ===
    print(f"\n[P1] L-BFGS coarse (n={n_main}, t={time.monotonic()-t_start_global:.0f}s)")
    p1_frac = 0.25
    p1_end = t_start_global + (deadline - t_start_global) * p1_frac
    a1, sc1 = _run_lbfgs(best_seq, p1_end, min_time_remaining=100, verbose=True)
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  P1 result: {best_score:.10f}")

    # === Phase 2: Spectral equalization ===
    print(f"\n[P2] Spectral equalization (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p2_end = min(deadline - 300, t_start_global + (deadline - t_start_global) * 0.50)
        a2, sc2 = _spectral_equalization(best_seq, p2_end, min_time_remaining=60, verbose=True)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(f"  P2 result: {best_score:.10f}")

    # === Phase 3: L-BFGS fine ===
    print(f"\n[P3] L-BFGS fine (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p3_end = min(deadline - 200, t_start_global + (deadline - t_start_global) * 0.75)
        a3, sc3 = _run_lbfgs(best_seq, p3_end, min_time_remaining=80, verbose=True)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3 result: {best_score:.10f}")

    # === Phase 4: KKT refinement ===
    print(f"\n[P4] KKT refine (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p4_end = min(deadline - 80, t_start_global + (deadline - t_start_global) * 0.90)
        a4, sc4 = _kkt_refine(best_seq, p4_end, min_time_remaining=40, verbose=True)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4 result: {best_score:.10f}")

    # === Phase 5: Nesterov + final L-BFGS ===
    print(f"\n[P5] Final polish (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 20 < deadline:
        # Nesterov
        p5a_end = min(deadline - 30, time.monotonic() + 60)
        a5a, sc5a = _nesterov_projected(best_seq, p5a_end, min_time_remaining=20, verbose=True)
        if sc5a < best_score:
            best_score = sc5a
            best_seq = a5a.copy()
            print(f"  P5a (Nesterov): {best_score:.10f}")

        # Final L-BFGS
        if time.monotonic() + 15 < deadline:
            p5b_end = min(deadline - 5, time.monotonic() + 50)
            a5b, sc5b = _run_lbfgs(best_seq, p5b_end, min_time_remaining=5, verbose=False)
            if sc5b < best_score:
                best_score = sc5b
                best_seq = a5b.copy()
                print(f"  P5b (final L-BFGS): {best_score:.10f}")

        # Final KKT
        if time.monotonic() + 10 < deadline:
            p5c_end = min(deadline - 3, time.monotonic() + 25)
            a5c, sc5c = _kkt_refine(best_seq, p5c_end, min_time_remaining=3, verbose=False)
            if sc5c < best_score:
                best_score = sc5c
                best_seq = a5c.copy()
                print(f"  P5c (final KKT): {best_score:.10f}")

    # Final
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start_global:.1f}s")
    return [float(x) for x in final]
