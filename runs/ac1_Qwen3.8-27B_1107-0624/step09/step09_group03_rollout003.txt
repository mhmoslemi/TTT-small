    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec, rhs, max_cg=200, tol=1e-13):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _deflation_step(a, c, c_max, deadline):
    """
    Peak-driven deflation with adaptive step size.
    Targets all peaks above a fraction of c_max.
    Returns updated a and score.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = float(c_max)

    # Find significant peaks: those above 95% of max
    threshold = c_max * 0.95
    peak_mask = c >= threshold
    n_peaks = int(peak_mask.sum())
    if n_peaks == 0:
        peak_mask[np.argmax(c)] = True
        n_peaks = 1

    # Weight peaks by their excess
    excess = np.zeros_like(c)
    excess[peak_mask] = c[peak_mask] - c_max * 0.9
    total_excess = excess.sum()
    if total_excess < 1e-30:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = excess / total_excess

    # Compute gradient direction
    grad = 2.0 * _corr_a_A(a, w, n)
    g_t = grad - np.mean(grad)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-20:
        return best_a, best_sc

    # Adaptive line search: try multiple step sizes
    g_t = g_t / gnorm  # normalize
    for strength in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
        if time.monotonic() > deadline:
            break
        a_try = a - strength * g_t
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            continue
        a_try = a_try / s_try
        c_try = _full_convolve(a_try, a_try)
        sc_try = float(c_try.max())
        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _lbfgs_run(a_init, deadline, min_remaining=20.0, n_betas=20):
    """
    L-BFGS with log-sum-exp continuation.
    Starts from a relatively large beta and decreases.
    """
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    idx = min(100, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[idx]
    base_beta = max(gap * 0.5, max_c * 0.005)

    # Very small final beta
    final_beta = max(max_c * 2e-8, 1e-12)
    log_betas = np.linspace(np.log(base_beta), np.log(final_beta), n_betas)
    betas = np.exp(log_betas)

    current_a = a_init.copy()
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        # Estimate iterations based on remaining time
        est_time_per_iter = 0.002 * max(1.0, n / 8192.0)
        max_iter = max(50, min(5000, int(remaining / (est_time_per_iter * 15))))

        try:
            result = minimize(
                _lsobj,
                z,
                args=(beta, n),
                jac=lambda z_val: _lsgrad(z_val, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            c_new = _full_convolve(a_new, a_new)
            max_new = float(c_new.max())
            if max_new < max_c - 1e-16:
                current_a = a_new
                max_c = max_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            if time.monotonic() - t_start > 10:
                print(f"    L-BFGS beta={beta:.3e}: max_c={max_c:.10f} "
                      f"score={sc_new:.10f} t={time.monotonic()-t_start:.1f}s")
        except Exception as ex:
            print(f"    L-BFGS beta={beta:.3e}: exception {ex}")

    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = t + 0.01
        return a / a.sum()
    elif kind == "ramp2":
        t = np.linspace(0, 1, n)
        a = t**2 + 0.01
        return a / a.sum()
    elif kind == "exponent":
        t = np.linspace(0, 1, n)
        a = np.exp(-t) + 0.01
        return a / a.sum()
    elif kind == "two_bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-50*(t-0.3)**2) + 0.5*np.exp(-50*(t-0.7)**2) + 0.01
        return a / a.sum()
    elif kind == "asym_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-20*(t-0.4)**2) + 0.01
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.4*np.cos(2*np.pi*t) + 0.11
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 1e-15)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _newton_refine(a, deadline, max_iters=50):
    """
    Active-set Newton refinement for precise KKT conditions.
    """
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set: peaks within 0.5% of max
        tol_frac = 0.005
        active = c >= (c_max - tol_frac * c_max)
        if not active.any():
            active[np.argmax(c)] = True
        m = int(active.sum())
        if m < 1:
            break

        A_vec = active.astype(np.float64) / m
        g = 2.0 * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-6 / m + 1e-10
        A_copy = A_vec
        mu_copy = mu
        n_copy = n

        def _hess_vp(v):
            return 2.0 * _corr_a_A(v, A_copy, n_copy) + mu_copy * v

        u = _cg_solve(_hess_vp, g_t, max_cg=150)
        w_vec = _cg_solve(_hess_vp, np.ones(n), max_cg=150)
        w_sum = float(w_vec.sum())

        if abs(w_sum) < 1e-15:
            dx = -u
        else:
            lam = float(u.sum()) / w_sum
            dx = -u + lam * w_vec

        dx = dx - np.mean(dx)

        # Line search
        improved = False
        t = 1.0
        for ls in range(40):
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < c_max - 1e-15:
                a = a_try
                c_max = f_try
                sc_try = _score(a)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a.copy()
                improved = True
                break
            t *= 0.5

        if not improved:
            break

    return best_a


def _refinement_loop(a, deadline, max_rounds=20):
    """
    Alternating deflation + L-BFGS + Newton refinement.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Deflation step
        def_time = min(deadline - 10, time.monotonic() + 10)
        c_cur = _full_convolve(a, a)
        a_def, sc_def = _deflation_step(a, c_cur, float(c_cur.max()), def_time)
        if sc_def < best_sc:
            best_sc = sc_def
            best_a = a_def.copy()
            a = a_def
        else:
            a = best_a

        # Short L-BFGS polish
        if time.monotonic() < deadline - 15:
            lb_time = min(deadline - 10, time.monotonic() + 20)
            a_lb = _lbfgs_run(a, lb_time, min_remaining=10, n_betas=8)
            sc_lb = _score(a_lb)
            if sc_lb < best_sc:
                best_sc = sc_lb
                best_a = a_lb.copy()
                a = a_lb
            else:
                a = best_a

        # Newton refinement
        if time.monotonic() < deadline - 5:
            nw_time = min(deadline - 3, time.monotonic() + 15)
            a_nw = _newton_refine(a, nw_time, max_iters=20)
            sc_nw = _score(a_nw)
            if sc_nw < best_sc:
                best_sc = sc_nw
                best_a = a_nw.copy()
                a = a_nw
            else:
                a = best_a

        print(f"    Round {rnd}: score={best_sc:.12f} t={time.monotonic()-t_start:.1f}s")

    return best_a, best_sc


t_start = 0.0


def propose_candidate(seed=42, budget_s=1800):
    global t_start
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.12f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                sc_parent = _score(parent)
                print(f"Parent loaded: n={n_parent}, score={sc_parent:.12f}")
                update_best(parent, sc_parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")
        parent = None

    # If no parent, create initial
    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "ramp")
        update_best(parent, _score(parent), "init_ramp")

    # === Phase 1: Try multiple starting profiles with L-BFGS ===
    n_opt = n_parent  # primary optimization length
    print(f"\n[Phase 1] Multi-profile L-BFGS (n={n_opt}) t={time.monotonic()-t_start:.1f}s")

    profiles = ["ramp", "ramp2", "exponent", "two_bump", "asym_gauss", "cos_taper"]
    if parent is not None:
        # Also try the parent as starting point
        profile_starts = [parent]
    else:
        profile_starts = []

    # Allocate time: ~200s for phase 1
    p1_dl = min(deadline - 800, t_start + 200)
    time_per_profile = max(20, (p1_dl - t_start) / max(1, len(profiles) + 1))

    # Start with the parent (best known)
    if parent is not None:
        a_p = _lbfgs_run(parent, t_start + time_per_profile, min_remaining=5, n_betas=12)
        update_best(a_p, _score(a_p), "parent_lbfgs")

    for prof in profiles:
        if time.monotonic() > p1_dl:
            break
        a_init = _make_profile(n_opt, prof)
        prof_dl = min(p1_dl, time.monotonic() + time_per_profile)
        if time.monotonic() >= prof_dl:
            break
        a_opt = _lbfgs_run(a_init, prof_dl, min_remaining=5, n_betas=12)
        update_best(a_opt, _score(a_opt), f"profile_{prof}")

    # === Phase 2: Refinement loop on best ===
    print(f"\n[Phase 2] Refinement loop (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        p2_dl = min(deadline - 400, time.monotonic() + 300)
        a_ref, sc_ref = _refinement_loop(best_seq, p2_dl, max_rounds=15)
        update_best(a_ref, sc_ref, "refinement_main")

    # === Phase 3: Deflation-focused refinement ===
    print(f"\n[Phase 3] Deflation refinement (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        p3_dl = min(deadline - 200, time.monotonic() + 200)
        a_cur = best_seq.copy()
        s_cur = a_cur.sum()
        a_cur = a_cur / s_cur
        for it in range(200):
            if time.monotonic() > p3_dl:
                break
            c_cur = _full_convolve(a_cur, a_cur)
            cm = float(c_cur.max())
            a_new, sc_new = _deflation_step(a_cur, c_cur, cm, p3_dl)
            if sc_new < best_score:
                best_score = sc_new
                best_seq = a_new.copy()
                a_cur = a_new
            else:
                # Try smaller perturbation to escape
                a_try = a_cur * (1.0 + 0.005 * rng.standard_normal(n_opt))
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_score:
                        best_score = sc_try
                        best_seq = a_try.copy()
                        a_cur = a_try
                    else:
                        break
        update_best(best_seq, best_score, "deflation_loop")

    # === Phase 4: Newton refinement ===
    print(f"\n[Phase 4] Newton refinement (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        p4_dl = min(deadline - 100, time.monotonic() + 100)
        a_nw = _newton_refine(best_seq, p4_dl, max_iters=80)
        update_best(a_nw, _score(a_nw), "newton")

    # === Phase 5: Try n=32768 ===
    print(f"\n[Phase 5] Try n=32768 (t={time.monotonic()-t_start:.1f}s")
    if time.monotonic() < deadline - 60:
        n_large = 32768
        if best_seq is not None:
            a_large = _interpolate_seq(best_seq, n_large)
        else:
            a_large = _make_profile(n_large, "ramp")
        sc_large = _score(a_large)
        print(f"  n={n_large} interpolated score: {sc_large:.12f}")

        # Quick L-BFGS on large n
        p5_dl = min(deadline - 30, time.monotonic() + 80)
        a_large_opt = _lbfgs_run(a_large, p5_dl, min_remaining=15, n_betas=10)
        sc_large_opt = _score(a_large_opt)
        update_best(a_large_opt, sc_large_opt, "n_32768")

        # Quick refinement on large n
        if time.monotonic() < deadline - 20 and best_seq is not None and len(best_seq) == n_large:
            p5b_dl = min(deadline - 10, time.monotonic() + 40)
            a_5b, sc_5b = _refinement_loop(best_seq, p5b_dl, max_rounds=5)
            update_best(a_5b, sc_5b, "n32768_refine")

    # === Phase 6: Final L-BFGS polish with very small beta ===
    print(f"\n[Phase 6] Final polish (t={time.monotonic()-t_start:.1f}s")
    if best_seq is not None:
        n_fin = len(best_seq)
        c_fin = _full_convolve(best_seq, best_seq)
        mb = float(c_fin.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 1e-8
        remaining = deadline - 5 - time.monotonic()
        if remaining > 3:
            max_iter_fin = min(3000, max(100, int(remaining / 0.05)))
            try:
                result = minimize(
                    _lsobj,
                    z,
                    args=(beta_final, n_fin),
                    jac=lambda z_val: _lsgrad(z_val, beta_final, n_fin),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter_fin, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 100}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_polish")
            except Exception:
                pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
