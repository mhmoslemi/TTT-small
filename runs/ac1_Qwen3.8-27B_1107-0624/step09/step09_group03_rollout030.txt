```python
def hankel_matvec(h, v, n):
    """Compute H @ v where H[j][m] = h[j+m], for j,m in [0,n-1]."""
    v_rev = v[::-1]
    N = next_pow2(2*n)
    H_f = np.fft.rfft(h, N)
    v_f = np.fft.rfft(v_rev, N)
    conv = np.fft.irfft(H_f * v_f, N)
    return conv[n-1 : 2*n-1]

def solve_hankel(h, rhs, n, reg=1e-8):
    """Solve (H + reg*I) @ x = rhs where H is Hankel with symbol h."""
    def matvec(v):
        return hankel_matvec(h, v, n) + reg * v
    # Use CG
    x = cg_solve(matvec, rhs, n)
    return x
```

Given h of length 2n-1 and v of length n:
(H @ v)[j] = Σ_{m=0}^{n-1} h[j+m] * v[m] for j = 0, ..., n-1
(where h[k] = 0 for k >= 2n-1, which is automatic since h has length 2n-1)

To compute this using FFT:
- v_flipped[m] = v[n-1-m] (length n)
- conv(h, v_flipped) has length (2n-1) + n - 1 = 3n-2
- (H @ v)[j] = conv(h, v_flipped)[j + n - 1]

conv[h, v_flipped] has length 5+3-1 = 7:
conv[0] = h0*v2
conv[1] = h0*v1 + h1*v2
conv[2] = h0*v0 + h1*v1 + h2*v2
conv[3] = h1*v0 + h2*v1 + h3*v2
conv[4] = h2*v0 + h3*v1 + h4*v2
conv[5] = h3*v0 + h4*v1
conv[6] = h4*v0

(H@v)[0] = h0*v0 + h1*v1 + h2*v2 = conv[2] ✓ (j=0, conv[0+3-1] = conv[2])
(H@v)[1] = h1*v0 + h2*v1 + h3*v2 = conv[3] ✓ (j=1, conv[1+3-1] = conv[3])
(H@v)[2] = h2*v0 + h3*v1 + h4*v2 = conv[4] ✓ (j=2, conv[2+3-1] = conv[4])

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _hankel_matvec(h, v, n):
    """Compute H @ v where H[j][m] = h[j+m] for j,m in [0,n-1]."""
    v_flipped = v[::-1]
    out_len = len(h) + len(v_flipped) - 1
    N = _next_pow2(out_len)
    hf = np.fft.rfft(h, N)
    vf = np.fft.rfft(v_flipped, N)
    conv = np.fft.irfft(hf * vf, N)
    return conv[n-1 : 2*n-1]

def _cg_hankel(h, n, rhs, reg, max_iter=200, tol=1e-8):
    """Solve (H + reg*I) @ x = rhs where H is Hankel with symbol h."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    
    def mv(v):
        return _hankel_matvec(h, v, n) + reg * v
    
    for _ in range(max_iter):
        Ap = mv(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * rhs_norm:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    
    return x
