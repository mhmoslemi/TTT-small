"""
Multi-scale Adam + L-BFGS continuation optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous: (1) multi-scale warm start from n=64 to target, (2) Adam optimizer
with per-coordinate adaptive learning rates, (3) smarter beta continuation schedule for L-BFGS,
(4) multiple diverse initializations, (5) larger n exploration, (6) palindromic symmetry exploitation.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _compute_conv_and_weights(a, beta):
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return c, c_max, w


def _grad_smooth_obj(a, beta):
    """Gradient of the smooth approximation 2n*logsumexp(conv(a,a)/beta) w.r.t. a."""
    n = len(a)
    c, c_max, w = _compute_conv_and_weights(a, beta)
    # grad[m] = (2/beta) * sum_k w[k] * a[k-m] for valid k-m
    # = (2/beta) * cross_correlation(w, a)[m]
    N = _next_pow2(2 * n - 1)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    cross_corr = np.fft.irfft(W * np.conj(A), N)
    grad = (2.0 / beta) * cross_corr[:n]
    return grad, c_max, w


def _smooth_obj_value(a, beta):
    n = len(a)
    c, c_max, w = _compute_conv_and_weights(a, beta)
    return 2.0 * n * (c_max + beta * np.log(max(w.sum() * np.exp((c_max - c_max) / beta), 1e-300)))


def _softmax_grad(a, grad_a):
    """Chain rule for softmax: dF/dz = a * (grad_a - <a, grad_a>)."""
    return a * (grad_a - float(np.dot(a, grad_a)))


def _adam_optimize(a_init, deadline, beta_init=None, min_time=20.0, seed=42, max_iter=None):
    """Adam optimizer on softmax parameterization."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s

    c0 = _full_convolve(a, a)
    c_max0 = float(c0.max())
    if beta_init is None:
        beta_init = max(c_max0 * 0.05, 1e-8)

    z = np.log(np.maximum(a, 1e-300))
    z = z - z.mean()

    rng = np.random.default_rng(seed)

    # Adam hyperparameters
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_a = a.copy()
    best_sc = 2.0 * n * c_max0
    no_improve = 0
    staleness = 0

    # Adaptive beta: decrease over time
    beta = beta_init
    beta_decay = 0.995  # per step
    beta_min = max(c_max0 * 0.001, 1e-10)

    while time.monotonic() + min_time < deadline:
        if max_iter is not None and t >= max_iter:
            break

        t += 1
        a = _softmax(z)

        # Compute smooth gradient
        grad_a, c_max_cur, w = _grad_smooth_obj(a, beta)
        current_sc = 2.0 * n * c_max_cur

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Check for stagnation
        if no_improve > 5000:
            staleness += 1
            if staleness > 5:
                break
            # Perturb
            z += 0.01 * rng.standard_normal(n)
            z -= z.mean()
            no_improve = 0
            # Reduce beta more aggressively
            beta = max(beta * 0.5, beta_min)
            continue

        # Convert to softmax parameter space
        grad_z = _softmax_grad(a, grad_a)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z * grad_z
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # Adaptive learning rate
        lr_t = lr / np.sqrt(1 + t * 0.0001)
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        z -= z.mean()  # Keep centered

        # Decay beta
        beta = max(beta * beta_decay, beta_min)

        # Periodic checkpoint
        if t % 10000 == 0:
            print(f"      Adam t={t}: best={best_sc:.10f}, cur={current_sc:.10f}, beta={beta:.2e}, lr={lr_t:.6f}")

    return best_a, best_sc


def _lbfgs_objective(z, beta, n):
    a = _softmax(z)
    c, c_max, w = _compute_conv_and_weights(a, beta)
    s = w.sum()
    if s < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lbfgs_gradient(z, beta, n):
    a = _softmax(z)
    c, c_max, w = _compute_conv_and_weights(a, beta)
    N = _next_pow2(2 * n - 1)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    cross_corr = np.fft.irfft(W * np.conj(A), N)
    grad_a = (2.0 / beta) * cross_corr[:n]
    grad_z = _softmax_grad(a, grad_a)
    return grad_z


def _run_lbfgs(a_init, deadline, min_time=20.0, n_betas=12, tag=""):
    """L-BFGS with continuation on beta."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a

    current_sc = 2.0 * n * max_c
    best_a = a.copy()
    best_sc = current_sc

    z = np.log(np.maximum(a, 1e-300))
    z = z - z.mean()

    # Beta schedule: geometric from 0.1*max_c down to 0.001*max_c
    beta_start = 0.1 * max_c
    beta_end = 0.001 * max_c
    betas = beta_start * (beta_end / beta_start) ** (np.arange(n_betas) / (n_betas - 1))

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        remaining = (deadline - min_time) - time.monotonic()
        if remaining < 5:
            break

        # Estimate iterations
        est_time_per_iter = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(3000, max(30, int(remaining / (est_time_per_iter * 5))))

        def obj(z_v):
            return _lbfgs_objective(z_v, beta, n)

        def grad(z_v):
            return _lbfgs_gradient(z_v, beta, n)

        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 80}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-300))
                z = z - z.mean()
                if i % 3 == 0:
                    print(f"      L-BFGS{i}{tag}: beta={beta:.4e}, score={current_sc:.10f}")
        except Exception as e:
            pass

    return best_a


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_tapered(n, power=0.5):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_dirichlet(n, alpha=1.0, seed=0):
    rng = np.random.default_rng(seed)
    a = rng.dirichlet(np.ones(n) * alpha)
    return a


def _make_multimodal(n, n_modes=4, seed=0):
    rng = np.random.default_rng(seed)
    t = np.linspace(0, 1, n)
    a = np.zeros(n)
    centers = rng.uniform(0.1, 0.9, n_modes)
    widths = rng.uniform(0.02, 0.08, n_modes)
    for c, w in zip(centers, widths):
        a += np.exp(-((t - c) ** 2) / (2 * w ** 2))
    a += 0.01
    return a / a.sum()

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 3000:
            break

        # Project gradient to palindromic subspace
        if n % 2 == 0:
            half_g = (grad_a[:half] + grad_a[half-1::-1]) / 2.0
            grad_proj = np.concatenate([half_g, half_g[::-1]])
        else:
            half_g = (grad_a[:half] + grad_a[n-1-half+1::-1]) / 2.0
            grad_proj = np.concatenate([half_g, half_g[-2::-1]])

        grad_z = _softmax_grad(a, grad_proj)

        m_adam = beta1_adam * m_adam + (1 - beta1_adam) * grad_z
        v_adam = beta2_adam * v_adam + (1 - beta2_adam) * grad_z * grad_z
        m_hat = m_adam / (1 - beta1_adam ** t_step)
        v_hat = v_adam / (1 - beta2_adam ** t_step)

        lr_t = lr / np.sqrt(1 + t_step * 0.0001)
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        z -= z.mean()
        beta = max(beta * 0.997, beta_min)

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """Multi-scale Adam + L-BFGS continuation with diverse initializations."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        n_parent = 16384
        parent = _make_tapered(n_parent)

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
        n_primary = len(parent)
    else:
        n_primary = 16384

    # === Phase 0: Multi-scale warm start ===
    print(f"\n[P0] Multi-scale warm start (t={time.monotonic()-t_start:.1f}s)")
    multiscale_deadline = min(deadline - 600, t_start + 120)
    scales = [64, 256, 1024, 4096]
    if n_primary <= 4096:
        scales = [s for s in scales if s < n_primary]
        scales.append(n_primary)
    else:
        scales.append(n_primary)

    a_ms = None
    for ns in scales:
        if time.monotonic() + 30 > multiscale_deadline:
            break
        if a_ms is None:
            a_ms = _make_tapered(ns, power=0.5)
        else:
            a_ms = _resample(a_ms, ns)
        # Quick Adam
        a_ms, sc_ms = _adam_optimize(a_ms, multiscale_deadline, min_time=5.0,
                                     seed=seed + ns, max_iter=min(5000, ns * 5))
        print(f"  n={ns}: score={sc_ms:.10f}")

    if a_ms is not None:
        a_primary = _resample(a_ms, n_primary)
        sc_primary = _score(a_primary)
        print(f"  Resampled to n={n_primary}: score={sc_primary:.10f}")
        if sc_primary < best_score:
            best_seq = a_primary.copy()
            best_score = sc_primary
        multiscale_start = a_primary.copy()
    else:
        multiscale_start = best_seq.copy()

    # === Phase 1: Adam on primary n (parent init) ===
    print(f"\n[P1] Adam primary - parent init (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        p1_deadline = min(deadline - 400, t_start + 350)
        a1, sc1 = _adam_optimize(best_seq, p1_deadline, min_time=30.0, seed=seed)
        if sc1 < best_score:
            best_seq = a1.copy()
            best_score = sc1
            print(f"  NEW BEST: {best_score:.10f} [adam_parent] t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Adam on primary n (multiscale init) ===
    print(f"\n[P2] Adam primary - multiscale init (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        p2_deadline = min(deadline - 300, t_start + 500)
        a2, sc2 = _adam_optimize(multiscale_start, p2_deadline, min_time=30.0, seed=seed + 1)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [adam_ms] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: L-BFGS continuation ===
    print(f"\n[P3] L-BFGS continuation (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        p3_deadline = min(deadline - 200, t_start + 700)
        a3 = _run_lbfgs(best_seq, p3_deadline, min_time=30.0, n_betas=12, tag="_main")
        sc3 = _score(a3)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_main] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Second L-BFGS round (finer) ===
    print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        p4_deadline = min(deadline - 150, t_start + 900)
        a4 = _run_lbfgs(best_seq, p4_deadline, min_time=30.0, n_betas=8, tag="_r2")
        sc4 = _score(a4)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_r2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Adam on diverse inits ===
    print(f"\n[P5] Diverse inits (t={time.monotonic()-t_start:.1f}s)")
    inits = [
        ("tapered1", _make_tapered(n_primary, 0.3)),
        ("tapered2", _make_tapered(n_primary, 1.5)),
        ("random", _make_random_dirichlet(n_primary, alpha=0.5, seed=seed+10)),
        ("multimodal", _make_multimodal(n_primary, n_modes=6, seed=seed+20)),
    ]
    for name, a_init in inits:
        if time.monotonic() > deadline - 100:
            break
        remaining_init = deadline - time.monotonic() - 50
        if remaining_init < 100:
            break
        a_div, sc_div = _adam_optimize(a_init, t_start + time.monotonic() - t_start + remaining_init,
                                       min_time=20.0, seed=seed + hash(name) % 10000)
        # Just check score
        sc_div = _score(a_div)
        if sc_div < best_score:
            best_seq = a_div.copy()
            best_score = sc_div
            print(f"  NEW BEST: {best_score:.10f} [diverse_{name}] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  diverse_{name}: score={sc_div:.10f} (no improvement)")

    # === Phase 6: Palindromic optimization ===
    print(f"\n[P6] Palindromic (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        p6_deadline = min(deadline - 100, t_start + time.monotonic() - t_start + 150)
        a_pal = _palindromic_optimize(best_seq, p6_deadline, min_time=20.0)
        sc_pal = _score(a_pal)
        if sc_pal < best_score:
            best_seq = a_pal.copy()
            best_score = sc_pal
            print(f"  NEW BEST: {best_score:.10f} [palindromic] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  palindromic: score={sc_pal:.10f} (no improvement)")

    # === Phase 7: Larger n exploration ===
    print(f"\n[P7] Larger n (t={time.monotonic()-t_start:.1f}s)")
    larger_ns = [n_primary * 2, n_primary * 3, n_primary * 4]
    larger_ns = [n for n in larger_ns if n <= 131072]
    for n_try in larger_ns:
        if time.monotonic() > deadline - 80:
            break
        a_large = _resample(best_seq, n_try)
        sc_large_init = _score(a_large)
        print(f"  n={n_try}: init score={sc_large_init:.10f}")
        if time.monotonic() < deadline - 80:
            a_large_opt, sc_large_opt = _adam_optimize(a_large, deadline - 40, min_time=30.0,
                                                       seed=seed + n_try, max_iter=3000)
            if sc_large_opt < best_score:
                best_seq = a_large_opt.copy()
                best_score = sc_large_opt
                print(f"  NEW BEST: {best_score:.10f} [n={n_try}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final L-BFGS polish ===
    print(f"\n[P8] Final L-BFGS polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 20:
        a_fin = _run_lbfgs(best_seq, deadline - 5, min_time=5.0, n_betas=10, tag="_fin")
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")

    # === Phase 9: Tiny Adam polish ===
    print(f"\n[P9] Tiny Adam polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        beta_tiny = max(mb * 0.0005, 1e-10)
        a_tiny, sc_tiny = _adam_optimize(best_seq, deadline - 1, min_time=1.0,
                                         beta_init=beta_tiny, seed=seed + 999, max_iter=2000)
        if sc_tiny < best_score:
            best_seq = a_tiny.copy()
            best_score = sc_tiny
            print(f"  NEW BEST: {best_score:.10f} [tiny_adam] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_tapered(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_tapered(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
