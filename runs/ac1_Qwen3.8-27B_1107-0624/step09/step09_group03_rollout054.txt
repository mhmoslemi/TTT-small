"""

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _lsobj_factory(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad_factory(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec, rhs, max_cg=300, tol=1e-14):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _newton_step(a, c, c_max, tol_frac, mu_base=1e-8):
    n = len(a)
    active = c >= (c_max - tol_frac * c_max)
    if not active.any():
        active[np.argmax(c)] = True
    m = int(active.sum())
    if m < 2 or m > 20 * n:
        return None, None

    A_vec = active.astype(np.float64) / m
    g = 2.0 * _corr_a_A(a, A_vec, n)
    g_t = g - np.mean(g)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-14:
        return None, None

    mu = mu_base / m + 1e-10
    A_vec_copy = A_vec
    n_copy = n
    mu_copy = mu

    def _hess_vp(v):
        return 2.0 * _corr_a_A(v, A_vec_copy, n_copy) + mu_copy * v

    u = _cg_solve(_hess_vp, g_t, max_cg=300)
    w = _cg_solve(_hess_vp, np.ones(n), max_cg=300)
    w_sum = float(w.sum())

    if abs(w_sum) < 1e-15:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w

    dx = dx - np.mean(dx)
    return dx, A_vec


def _mirror_deflation(a, deadline, n_attempts=100):
    """
    Mirror-symmetric multi-peak deflation exploiting conv(a,a) symmetry.
    (a*a)[k] = (a*a)[2n-2-k], so we deflate pairs simultaneously.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        total_len = 2 * n - 1

        # Find peaks with mirror symmetry
        threshold = c_max * 0.985
        peak_mask = c >= threshold
        n_peaks = int(peak_mask.sum())
        if n_peaks == 0:
            peak_mask[np.argmax(c)] = True
            n_peaks = 1

        # Weight peaks: give more weight to the highest
        weights = np.zeros_like(c)
        peak_vals = c[peak_mask]
        weights[peak_mask] = (peak_vals - peak_vals.min() + 1e-10) / (n_peaks * peak_vals.mean())

        # Compute mirror-symmetric gradient
        grad = np.zeros(n)
        for k in range(total_len):
            if not peak_mask[k]:
                continue
            w_k = weights[k]
            # contrib to c[k]: sum_i a[i]*a[k-i]
            # grad[i] += w_k * a[k-i] (mirror: also w_k * a[(2n-2-k)-i])
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += w_k * a[k - i]
            # Mirror peak
            k_mirror = total_len - 1 - k
            if k_mirror != k:
                lo_m = max(0, k_mirror - (n - 1))
                hi_m = min(n - 1, k_mirror)
                for i in range(lo_m, hi_m + 1):
                    grad[i] += w_k * a[k_mirror - i]

        # Normalize gradient for stable step
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm

        # Try various step sizes
        applied = False
        for strength in [0.0001, 0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
            a_try = a - strength * grad
            a_try = np.maximum(a_try, 0)
            s_try = float(a_try.sum())
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                applied = True
                break

        if not applied:
            # Random perturbation fallback
            rng_local = np.random.default_rng(attempt + 7919)
            improved = False
            for _ in range(3):
                a_try = a * (1.0 + 0.02 * rng_local.standard_normal(n))
                a_try = np.maximum(a_try, 0)
                s_try = float(a_try.sum())
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_sc - 1e-14:
                        best_sc = sc_try
                        best_a = a_try.copy()
                        improved = True
                        break
            if not improved:
                break

    return best_a, best_sc


def _fft_deflation(a, deadline, n_attempts=80):
    """
    Frequency-domain deflation: modify the DFT to reduce time-domain peaks.
    For peak at k*, the gradient in freq domain is A * exp(-2*pi*i*k*/N) (phase shift).
    """
    n = len(a)
    N = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = _score(a)

    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        # Compute FFT of a
        A = np.fft.rfft(a, N)

        # Gradient of c[k_star] w.r.t. a: g[i] = a[k_star - i] (valid range)
        # In freq domain, this is: exp(-2*pi*1j*k_star*f/N) * conj(A[f])
        # For real a: conj(A[f]) = A[N-f] which in rfft terms...
        # Simpler: just compute the gradient in time domain and use it
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        grad_time = np.zeros(n)
        grad_time[lo:hi + 1] = a[k_star - np.arange(lo, hi + 1)]

        # Also add mirror peak contribution
        k_mirror = 2 * n - 2 - k_star
        if k_mirror != k_star:
            lo_m = max(0, k_mirror - (n - 1))
            hi_m = min(n - 1, k_mirror)
            grad_time[lo_m:hi_m + 1] += a[k_mirror - np.arange(lo_m, hi_m + 1)]

        g_norm = np.linalg.norm(grad_time)
        if g_norm < 1e-15:
            break

        # Try decaying strengths
        applied = False
        for strength in [0.0002, 0.0005, 0.001, 0.002, 0.005, 0.01, 0.03]:
            a_try = a - strength * grad_time
            a_try = np.maximum(a_try, 0)
            s_try = float(a_try.sum())
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                applied = True
                break

        if not applied:
            break

    return best_a, best_sc


def _ifta_init(n, iterations=50, seed=123):
    """
    IFTA (Iterative Fourier Transform Algorithm) to create a non-negative
    sequence whose squared DFT is as flat as possible.
    """
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = a / a.sum()
    N = _next_pow2(2 * n - 1)

    for _ in range(iterations):
        A = np.fft.rfft(a, N)
        # Shape |A|^2 to be flatter
        B = A * np.conj(A)  # power spectrum = |A|^2
        # Apply soft threshold to flatten peaks in |A|
        abs_A = np.sqrt(np.abs(B))
        if abs_A.max() > 0:
            # Reduce peaks in amplitude
            target = abs_A.mean() * 1.5
            mask = abs_A > target
            abs_A[mask] = target + (abs_A[mask] - target) * 0.7
        # Reconstruct A with modified amplitude
        phase = np.angle(A)
        A_new = abs_A * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        # Enforce non-negativity
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a = np.ones(n) / n
            break
        a = a_new / s

    return a


def _lbfgs_adaptive(a_init, deadline, min_time_remaining=30.0):
    """
    L-BFGS with adaptive beta continuation.
    Reduces beta based on observed improvement rate.
    """
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 100:
        gap = c_sorted[0] - c_sorted[100]
    else:
        gap = c_sorted[0] - c_sorted[-1]
    base_beta = max(gap * 0.5, max_c * 0.01)

    if base_beta < 1e-15:
        base_beta = max_c * 0.01

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 1e-9
    stall_count = 0

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.004 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(20, int(remaining / (est_time_per_iter * 15))))

        sc_before = current_sc
        try:
            def _obj(z_val):
                return _lsobj_factory(z_val, beta, n)

            def _grad(z_val):
                return _lsgrad_factory(z_val, beta, n)

            result = minimize(
                _obj, z, jac=_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-19,
                         'gtol': 1e-15, 'maxcor': 80, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-17:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                improvement = sc_before - sc_new
                stall_count = 0
                # Adaptive beta: if improvement is small relative to beta, reduce faster
                if improvement < beta * 0.01:
                    beta *= 0.3
                else:
                    beta *= 0.5
            else:
                stall_count += 1
                if stall_count >= 3:
                    break
                beta *= 0.4
        except Exception:
            beta *= 0.5
            stall_count += 1

    return current_a


def _active_set_polish(a_init, deadline, max_outer=200):
    """
    Two-phase active-set Newton for precise KKT satisfaction.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive tolerance: start loose, tighten over iterations
        if outer < max_outer // 3:
            tol_list = [0.03, 0.015, 0.005, 0.002]
        elif outer < 2 * max_outer // 3:
            tol_list = [0.002, 0.001, 0.0005, 0.0002]
        else:
            tol_list = [0.0002, 0.0001, 0.00005, 0.00002, 0.00001]

        best_improvement = 0
        best_a_new = None

        for tol_frac in tol_list:
            dx, A_vec = _newton_step(a, c, c_max, tol_frac)
            if dx is None:
                continue

            t = 1.0
            for ls in range(60):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = float(a_try.sum())
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < c_max - 1e-15 * t:
                    improvement = c_max - f_try
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_a_new = a_try.copy()
                    break
                t *= 0.5

        if best_a_new is None:
            # Gradient fallback with mirror symmetry
            c = _full_convolve(a, a)
            c_max = float(c.max())
            k_star = int(np.argmax(c))
            total_len = 2 * n - 1
            grad = np.zeros(n)
            # Primary peak
            lo = max(0, k_star - (n - 1))
            hi = min(n - 1, k_star)
            grad[lo:hi + 1] = a[k_star - np.arange(lo, hi + 1)]
            # Mirror peak
            k_m = total_len - 1 - k_star
            if k_m != k_star:
                lo_m = max(0, k_m - (n - 1))
                hi_m = min(n - 1, k_m)
                grad[lo_m:hi_m + 1] += a[k_m - np.arange(lo_m, hi_m + 1)]

            g_norm = np.linalg.norm(grad)
            if g_norm > 1e-15:
                grad = grad / g_norm
                for t in [0.001, 0.002, 0.005, 0.01, 0.02]:
                    a_try = a - t * grad
                    a_try = np.maximum(a_try, 0)
                    s_try = float(a_try.sum())
                    if s_try > 1e-15:
                        a_try = a_try / s_try
                        sc_try = _score(a_try)
                        if sc_try < best_sc - 1e-15:
                            best_a_new = a_try.copy()
                            best_improvement = 1
                            break

        if best_a_new is None:
            break

        a = best_a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if best_improvement < 1e-16:
            break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "skewed":
        t = np.linspace(0, 1, n)
        a = (1 - t) ** 0.5 + 0.1
        return a / a.sum()
    elif kind == "asym_cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 1.5 * t)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "gaussian_offset":
        t = np.linspace(0, 1, n)
        a = np.exp(-4 * (t - 0.35) ** 2) + 0.05
        return a / a.sum()
    elif kind == "two_hump":
        t = np.linspace(0, 1, n)
        a = np.exp(-8 * (t - 0.25) ** 2) + np.exp(-8 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_seq(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _refinement_loop(a, deadline, max_rounds=15):
    """
    Iterate between Newton, mirror deflation, and FFT deflation.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    stagnant = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        round_start_sc = best_sc

        # Newton step
        as_dl = min(deadline - 5, time.monotonic() + 40)
        a_newton, sc_newton = _active_set_polish(a, as_dl, max_outer=40)
        if sc_newton < best_sc - 1e-15:
            best_sc = sc_newton
            best_a = a_newton.copy()
            a = a_newton

        # Mirror deflation
        def_dl = min(deadline - 5, time.monotonic() + 25)
        a_def, sc_def = _mirror_deflation(a, def_dl, n_attempts=30)
        if sc_def < best_sc - 1e-15:
            best_sc = sc_def
            best_a = a_def.copy()
            a = a_def

        # FFT deflation
        fft_dl = min(deadline - 5, time.monotonic() + 20)
        a_fft, sc_fft = _fft_deflation(a, fft_dl, n_attempts=30)
        if sc_fft < best_sc - 1e-15:
            best_sc = sc_fft
            best_a = a_fft.copy()
            a = a_fft

        if best_sc < round_start_sc - 1e-15:
            stagnant = 0
        else:
            stagnant += 1
            if stagnant >= 3:
                break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: adaptive L-BFGS, mirror deflation,
    active-set Newton, IFTA initialization, multi-scale.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 40.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.12f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.12f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Quick L-BFGS to warm up (n=16384) ===
    print(f"\n[Phase 1] Quick L-BFGS warmup (t={time.monotonic() - t_start:.1f}s)")
    n_work = n_parent if n_parent <= 32768 else 16384
    a_work = _interpolate_seq(parent, n_work) if len(parent) != n_work else parent.copy()
    lbfgs_dl = min(deadline - 1400, time.monotonic() + 120)
    a_warm = _lbfgs_adaptive(a_work, lbfgs_dl, min_time_remaining=1400.0)
    update_best(a_warm, _score(a_warm), "warmup_lbfgs")
    if best_seq is not None and len(best_seq) != n_work:
        a_work = _interpolate_seq(best_seq, n_work)

    # === Phase 2: Aggressive adaptive L-BFGS ===
    print(f"\n[Phase 2] Aggressive adaptive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    lbfgs_dl = min(deadline - 1100, time.monotonic() + 350)
    a_aggr = _lbfgs_adaptive(a_work, lbfgs_dl, min_time_remaining=1100.0)
    update_best(a_aggr, _score(a_aggr), "aggr_lbfgs")

    # === Phase 3: Active-set Newton polish ===
    print(f"\n[Phase 3] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        as_dl = min(deadline - 900, time.monotonic() + 150)
        a_as, sc_as = _active_set_polish(best_seq, as_dl, max_outer=150)
        update_best(a_as, sc_as, "newton_polish")

    # === Phase 4: Mirror deflation ===
    print(f"\n[Phase 4] Mirror deflation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        def_dl = min(deadline - 750, time.monotonic() + 100)
        a_def, sc_def = _mirror_deflation(best_seq, def_dl, n_attempts=100)
        update_best(a_def, sc_def, "mirror_def")

    # === Phase 5: Refinement loop (Newton + deflation + FFT) ===
    print(f"\n[Phase 5] Refinement loop (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        ref_dl = min(deadline - 550, time.monotonic() + 200)
        a_ref, sc_ref = _refinement_loop(best_seq, ref_dl, max_rounds=12)
        update_best(a_ref, sc_ref, "refinement")

    # === Phase 6: L-BFGS polish with very small beta ===
    print(f"\n[Phase 6] L-BFGS final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        lbfgs2_dl = min(deadline - 350, time.monotonic() + 150)
        a_pol = _lbfgs_adaptive(best_seq, lbfgs2_dl, min_time_remaining=350.0)
        update_best(a_pol, _score(a_pol), "lbfgs_polish")

    # === Phase 7: Try n=32768 ===
    print(f"\n[Phase 7] Try n=32768 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        n_large = 32768
        a_large = _interpolate_seq(best_seq, n_large) if best_seq is not None else _make_profile(n_large, "uniform")
        sc_large_init = _score(a_large)
        print(f"  n={n_large} init score: {sc_large_init:.12f}")
        update_best(a_large, sc_large_init, "n32768_init")

        # L-BFGS on large n
        lbfgs3_dl = min(deadline - 150, time.monotonic() + 150)
        a_large_opt = _lbfgs_adaptive(a_large, lbfgs3_dl, min_time_remaining=150.0)
        update_best(a_large_opt, _score(a_large_opt), "n32768_opt")

        # Newton on large n
        if best_seq is not None and len(best_seq) == n_large:
            as3_dl = min(deadline - 80, time.monotonic() + 60)
            a_large_as, sc_large_as = _active_set_polish(a_large_opt, as3_dl, max_outer=80)
            update_best(a_large_as, sc_large_as, "n32768_newton")

    # === Phase 8: Diverse restarts from IFTA ===
    print(f"\n[Phase 8] IFTA restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        n_work2 = len(best_seq)
        for ifta_seed in [7, 42, 123, 999]:
            if time.monotonic() > deadline - 80:
                break
            a_ifta = _ifta_init(n_work2, iterations=30, seed=ifta_seed)
            # Blend with best: 50/50 mix
            a_blend = 0.5 * a_ifta + 0.5 * best_seq
            a_blend = np.maximum(a_blend, 0)
            s_blend = a_blend.sum()
            if s_blend > 1e-15:
                a_blend = a_blend / s_blend
            r_dl = min(deadline - 50, time.monotonic() + 30)
            a_r = _lbfgs_adaptive(a_blend, r_dl, min_time_remaining=50.0)
            update_best(a_r, _score(a_r), f"ifta_blend_{ifta_seed}")

    # === Phase 9: Final refinement ===
    print(f"\n[Phase 9] Final refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        ref_final = min(deadline - 10, time.monotonic() + 60)
        a_fin, sc_fin = _refinement_loop(best_seq, ref_final, max_rounds=8)
        update_best(a_fin, sc_fin, "final_ref")

        # Tiny beta L-BFGS
        if time.monotonic() < deadline - 8:
            n_fin = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 2e-9
            remaining = deadline - 8 - time.monotonic()
            if remaining > 3:
                max_iter_fin = min(3000, max(50, int(remaining / 0.08)))
                try:
                    def _obj_final(z_val):
                        return _lsobj_factory(z_val, beta_final, n_fin)

                    def _grad_final(z_val):
                        return _lsgrad_factory(z_val, beta_final, n_fin)

                    result = minimize(
                        _obj_final, z, jac=_grad_final, method='L-BFGS-B',
                        options={'maxiter': max_iter_fin, 'ftol': 1e-19,
                                 'gtol': 1e-16, 'maxcor': 100, 'maxls': 300}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_tiny_beta")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
