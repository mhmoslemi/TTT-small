6. Note that `c = ifft(P)`, so `c[k] = (1/N) * sum_m P[m] * exp(2*pi*i*m*k/N)`.
7. To reduce `c[k*]`, we should decrease `P[m]` where `exp(2*pi*i*m*k*/N)` has positive real part... but this is complicated.

Simpler approach: Use the gradient. The gradient of `c[k]` with respect to `a` is `2 * corr(a, e_k)` (restricted to valid range). So the gradient of `max(c)` is the gradient of `c` at the argmax.

In the frequency domain, the gradient of `c[k]` with respect to `A` is more complex. Let me just work in the time domain for the gradient, but use the frequency domain for a "projection" step.

**Hybrid approach:**
1. Compute gradient in time domain: `g = 2 * corr(a, e_{k*})` (restricted).
2. Make a step: `a_new = a - alpha * g_t` (where `g_t` is traceless).
3. In frequency domain: `A = fft(pad(a_new))`.
4. Modify `P = |A|^2` to be smoother: `P = P^0.5` (square root compresses dynamic range).
5. `A_mod = sqrt(P_new) * exp(j*angle(A))`.
6. `a_mod = ifft(A_mod)[:n]`.
7. Project: `a_mod = max(a_mod, 0)`, normalize.

This combines time-domain gradient descent with frequency-domain smoothing.

"""
Optimizer using frequency-domain smoothing, low-dimensional Fourier basis,
and multi-start L-BFGS to minimize 2*n*max(conv(a,a))/sum(a)^2.

Key innovations over previous approach:
1. Frequency-domain IAA (Iterative Amplitude Adjustment) for spectrum flattening
2. Low-dimensional cosine-basis parameterization for efficient global exploration
3. Adaptive multi-start strategy with diverse structured initializations

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    r = irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def _make_uniform(n):
    return np.ones(n) / n


def _make_cosine_basis(n, K):
    """Generate first K cosine basis functions for a sequence of length n."""
    i = np.arange(n)
    basis = np.zeros((K, n))
    for j in range(K):
        basis[j] = np.cos(2.0 * np.pi * (j + 1) * i / n)
    return basis


def _freq_smooth_iterate(a, deadline, n_iter_max=2000, t_initial=0.3, beta_initial=0.1):
    """
    Frequency-domain Iterative Amplitude Adjustment.
    Alternates between time-domain peak deflation and frequency-domain smoothing.
    """
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_sc = _score(a)
    t = t_initial
    beta = beta_initial
    
    N_conv = _next_pow2(2 * n - 1)
    
    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution
        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))
        
        # Compute time-domain gradient at peak
        # dc[k_star]/da[j] = a[k_star - j] + a[j - k_star] (for valid indices)
        grad = np.zeros(n)
        for j in range(n):
            val = 0.0
            idx1 = k_star - j
            idx2 = j - k_star
            if 0 <= idx1 < n:
                val += a[idx1]
            if 0 <= idx2 < n:
                val += a[idx2]
            grad[j] = 2.0 * val
        
        # Make traceless
        grad = grad - np.mean(grad)
        
        # Time-domain step
        a_td = a - t * grad
        a_td = _normalize(a_td)
        
        # Frequency-domain smoothing
        A = rfft(np.concatenate([a_td, np.zeros(N_conv - n)]), N_conv)
        P = np.abs(A) ** 2
        P_max = float(P.max())
        
        if P_max > 1e-30:
            # Soft flatten: compress dynamic range
            P_smooth = P / (P + beta * P_max)
            # Normalize to same mean
            P_smooth = P_smooth * (float(P.mean()) / float(P_smooth.mean()))
        else:
            P_smooth = P
        
        A_smooth = np.sqrt(P_smooth) * (A / np.where(np.abs(A) > 1e-30, np.abs(A), 1.0))
        
        a_fd = irfft(A_smooth, N_conv)[:n]
        a_new = _normalize(a_fd)
        
        # Check improvement
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            best_a = a_new.copy()
            best_sc = sc_new
            a = a_new
        else:
            # Reduce step size
            t *= 0.9
            beta = max(0.01, beta * 0.95)
            # Try just the time-domain step
            sc_td = _score(a_td)
            if sc_td < best_sc - 1e-15:
                best_a = a_td.copy()
                best_sc = sc_td
                a = a_td
            else:
                # Very small step
                a_small = _normalize(a - 0.01 * t * grad)
                sc_small = _score(a_small)
                if sc_small < best_sc - 1e-15:
                    best_a = a_small.copy()
                    best_sc = sc_small
                    a = a_small
                else:
                    break
        
        if it % 100 == 0:
            print(f"    FREQ iter {it}: score={best_sc:.12f}, t={t:.4f}, beta={beta:.4f}")
    
    return best_a, best_sc


def _fourier_basis_optimize(a_init, deadline, K=20):
    """
    Optimize using low-dimensional cosine basis parameterization.
    a[i] = max(0, 1/n + sum_j c_j * cos(2*pi*(j+1)*i/n + phi_j))
    We optimize over amplitudes c_j (phases fixed for simplicity).
    """
    n = len(a_init)
    sc = _score(a_init)
    if not np.isfinite(sc):
        return a_init, sc
    
    # Fit cosine coefficients to initial sequence
    i = np.arange(n)
    basis = np.zeros((K, n))
    for j in range(K):
        basis[j] = np.cos(2.0 * np.pi * (j + 1) * i / n)
    
    # Least squares fit: a_init ≈ mean + basis @ c
    a_centered = a_init - a_init.mean()
    c, _, _, _ = np.linalg.lstsq(basis, a_centered, rcond=None)
    
    def _obj_sc(params):
        c_val = params[:K]
        a_val = a_init.mean() + basis @ c_val
        a_val = np.maximum(a_val, 0)
        s = a_val.sum()
        if s < 1e-15:
            return 1e10
        a_val = a_val / s
        return _score(a_val)
    
    # Use L-BFGS on the K-dimensional problem
    x0 = c.copy()
    
    def _objective(x):
        return _obj_sc(x)
    
    # Numerical gradient
    def _numerical_grad(x, eps=1e-5):
        g = np.zeros(K)
        f0 = _obj_sc(x)
        for j in range(K):
            x_plus = x.copy()
            x_plus[j] += eps
            g[j] = (_obj_sc(x_plus) - f0) / eps
        return g
    
    remaining = deadline - time.monotonic()
    if remaining < 5:
        return a_init, sc
    
    max_iter = min(500, int(remaining / 0.5))  # Each eval takes ~0.1s for n=16384
    try:
        result = minimize(
            _objective,
            x0,
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50}
        )
        c_best = result.x
        a_best = a_init.mean() + basis @ c_best
        a_best = _normalize(a_best)
        sc_best = _score(a_best)
        return a_best, min(sc, sc_best)
    except Exception:
        return a_init, sc


def _lbfgs_lse(a_init, deadline, min_time=20.0):
    """
    L-BFGS with log-sum-exp smoothing for the max function.
    """
    n = len(a_init)
    a = a_init.copy()
    a = _normalize(a)
    current_sc = _score(a)
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a
    
    # Beta schedule
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted)-1)]
    base_beta = max(gap * 0.3, max_c * 0.002)
    
    n_betas = 20
    log_betas = np.linspace(np.log(base_beta), np.max([np.log(max_c * 1e-7), -30]), n_betas)
    betas = np.exp(log_betas)
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    def _lsobj(z_val, beta):
        a_val = _softmax(z_val)
        c = _full_convolve(a_val, a_val)
        c_max = float(c.max())
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n * c_max
        return 2.0 * n * (c_max + beta * np.log(s))
    
    def _lsgrad(z_val, beta):
        a_val = _softmax(z_val)
        c = _full_convolve(a_val, a_val)
        c_max = float(c.max())
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a_val[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        return 4.0 * n * a_val * (h - W)
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 3:
            break
        
        try:
            def _obj(z_val):
                return _lsobj(z_val, beta)
            
            def _grad(z_val):
                return _lsgrad(z_val, beta)
            
            est_time = 0.003 * max(1.0, n / 8192.0)
            max_iter = min(3000, max(30, int(remaining / (est_time * 10))))
            
            result = minimize(
                _obj,
                z,
                jac=_grad,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 300,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                a = a_new
            print(f"    LSE beta={beta:.2e}: score={current_sc:.12f}")
        except Exception as ex:
            print(f"    LSE beta={beta:.2e}: exception {ex}")
    
    return a

def _make_structured_init(n, kind, param):
    """Create structured initializations."""
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "bump":
        i = np.arange(n) / n
        a = np.exp(-param * (i - 0.5)**2)
        return a / a.sum()
    elif kind == "cosine":
        i = np.arange(n) / n
        a = 0.5 * (1 + np.cos(2 * np.pi * i * param))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block":
        n_blocks = max(2, int(param))
        a = np.zeros(n)
        block_size = n // (2 * n_blocks)
        for b in range(n_blocks):
            start = b * 2 * block_size
            a[start:start+block_size] = 1.0
        return a / a.sum()
    elif kind == "gaussian":
        sigma = n / param
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - n/2) / sigma)**2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, seed):
    rng = np.random.default_rng(seed)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    return _normalize(a_new)

"""
Optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Frequency-domain IAA (Iterative Amplitude Adjustment) for spectrum flattening
2. Low-dimensional cosine-basis parameterization for efficient global search
3. L-BFGS with log-sum-exp continuation (aggressive beta schedule)
4. Active-set Newton deflation for refinement
5. Multi-start with diverse structured initializations
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft

# Global state for optimization callbacks (to avoid closures)
_g_beta = 1.0
_g_n = 0
_g_basis = None
_g_a_mean = 0.0


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    r = irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def _gradient_at_peak(a, k):
    """Compute gradient of c[k] w.r.t. a in O(n)."""
    n = len(a)
    g = np.zeros(n)
    # a[k-j] for valid j: k-n+1 <= j <= k
    lo1 = max(0, k - n + 1)
    hi1 = min(n - 1, k)
    if hi1 >= lo1:
        g[lo1:hi1 + 1] = a[hi1 - lo1:k - lo1 + 1][::-1]
    # a[j-k] for valid j: k <= j <= k+n-1
    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if hi2 >= lo2:
        g[lo2:hi2 + 1] += a[lo2 - k:hi2 - k + 1]
    g *= 2.0
    return g


def _lse_obj_impl(z_val, beta, n):
    a_val = _softmax(z_val)
    c = _full_convolve(a_val, a_val)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad_impl(z_val, beta, n):
    a_val = _softmax(z_val)
    c = _full_convolve(a_val, a_val)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a_val[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a_val * (h - W)


def _lse_objective(z_val):
    return _lse_obj_impl(z_val, _g_beta, _g_n)


def _lse_gradient(z_val):
    return _lse_grad_impl(z_val, _g_beta, _g_n)


def _fb_objective(x):

    a_val = _g_a_mean + _g_basis @ c_val
    a_val = np.maximum(a_val, 0.0)
    s = a_val.sum()
    if s < 1e-15:
        return 1e10
    a_val = a_val / s
    return _score(a_val)


def _lbfgs_lse(a_init, deadline, min_time=20.0):
    """L-BFGS with log-sum-exp smoothing."""
    global _g_beta, _g_n
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)

    n_betas = 20
    log_betas = np.linspace(np.log(base_beta), np.max([np.log(max_c * 1e-7), -30]), n_betas)
    betas = np.exp(log_betas)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 3:
            break

        try:
            _g_beta = beta
            _g_n = n

            result = minimize(
                _lse_objective,
                z,
                jac=_lse_gradient,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,

                    'maxcor': 100,
                    'maxls': 300,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                a = a_new
            print(f"    LSE beta={beta:.2e}: score={current_sc:.12f}")
        except Exception as ex:
            print(f"    LSE beta={beta:.2e}: exception {ex}")

    return a


def _freq_smooth_iterate(a_init, deadline, n_iter_max=3000, t_initial=0.1, beta_initial=0.05):
    """Frequency-domain IAA: alternates time-domain deflation with frequency smoothing."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    t = t_initial
    beta = beta_initial

    N_conv = _next_pow2(2 * n - 1)
    pad = np.zeros(N_conv)
    pad[:n] = 1.0  # template

    stall_count = 0

    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        grad = _gradient_at_peak(a, k_star)
        grad = grad - np.mean(grad)

        # Time-domain step
        a_td = _normalize(a - t * grad)

        # Frequency-domain smoothing
        pad[:] = 0.0
        pad[:n] = a_td
        A = rfft(pad, N_conv)
        P = np.abs(A) ** 2
        P_max = float(P.max())

        if P_max > 1e-30:
            P_smooth = P / (P + beta * P_max)
            P_mean_orig = float(P.mean())
            P_mean_smooth = float(P_smooth.mean())
            if P_mean_smooth > 1e-30:
                P_smooth = P_smooth * (P_mean_orig / P_mean_smooth)
            A_smooth = np.sqrt(P_smooth) * (A / np.where(np.abs(A) > 1e-30, np.abs(A), 1.0))
        else:
            A_smooth = A

        a_fd = irfft(A_smooth, N_conv)[:n]
        a_new = _normalize(a_fd)

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            best_a = a_new.copy()
            best_sc = sc_new
            a = a_new
            stall_count = 0
        else:
            stall_count += 1
            # Try just time-domain step
            sc_td = _score(a_td)
            if sc_td < best_sc - 1e-15:
                best_a = a_td.copy()
                best_sc = sc_td
                a = a_td
                stall_count = 0
            else:
                t *= 0.85
                beta = max(0.005, beta * 0.97)
                if stall_count > 50:
                    # Try small random perturbation to escape
                    rng_local = np.random.default_rng(it + 7919)
                    pert = rng_local.standard_normal(n) * 0.005
                    a_try = _normalize(a + pert)
                    sc_try = _score(a_try)
                    if sc_try < best_sc:
                        best_a = a_try.copy()
                        best_sc = sc_try
                        a = a_try
                        stall_count = 0
                        t = t_initial * 0.5
                    else:
                        break
            if stall_count > 200:
                break

        if it % 200 == 0:
            print(f"    FREQ iter {it}: score={best_sc:.12f}, t={t:.5f}, beta={beta:.4f}")

    return best_a, best_sc


def _fourier_basis_optimize(a_init, deadline, K=25):
    """Optimize using low-dimensional cosine basis."""
    global _g_basis, _g_a_mean
    n = len(a_init)
    sc = _score(a_init)
    if not np.isfinite(sc):
        return a_init, sc

    i = np.arange(n)
    basis = np.zeros((K, n))
    for j in range(K):
        basis[j] = np.cos(2.0 * np.pi * (j + 1) * i / n)
        basis[j] = basis[j] - basis[j].mean()

    a_mean = float(a_init.mean())
    a_centered = a_init - a_mean
    c, _, _, _ = np.linalg.lstsq(basis, a_centered, rcond=None)

    _g_basis = basis
    _g_a_mean = a_mean

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return a_init, sc

    # Each eval is O(n) for the basis product + O(n log n) for score
    est_per_eval = 0.05 * max(1.0, n / 8192.0)
    max_iter = min(1000, max(20, int(remaining / (est_per_eval * 8))))

    try:
        result = minimize(
            _fb_objective,
            c.copy(),
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50}
        )
        c_best = result.x
        a_best = _g_a_mean + _g_basis @ c_best
        a_best = _normalize(a_best)
        sc_best = _score(a_best)
        if sc_best < sc:
            return a_best, sc_best
        return a_init, sc
    except Exception:
        return a_init, sc


def _newton_deflation(a_init, deadline, max_iter=100):
    """Active-set Newton/deflation refinement."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find active peaks (within 0.5% of max)
        threshold = c_max * 0.995
        active_indices = np.where(c >= threshold)[0]
        if len(active_indices) == 0:
            active_indices = np.array([int(np.argmax(c))])

        # Compute composite gradient
        g = np.zeros(n)
        for k in active_indices:
            g += _gradient_at_peak(a, k)
        g /= len(active_indices)
        g = g - np.mean(g)
        gnorm = np.linalg.norm(g)

        if gnorm < 1e-15:
            break

        improved = False
        for t in [0.01 / len(active_indices), 0.005 / len(active_indices),
                  0.001 / len(active_indices), 0.0005 / len(active_indices),
                  0.0001 / len(active_indices)]:
            a_try = _normalize(a - t * g)
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-15:
                best_a = a_try.copy()
                best_sc = sc_try
                a = a_try
                improved = True
                break

        if not improved:
            # Try wider tolerance for active set
            threshold2 = c_max * 0.99
            active2 = np.where(c >= threshold2)[0]
            if len(active2) > len(active_indices):
                g2 = np.zeros(n)
                for k in active2:
                    g2 += _gradient_at_peak(a, k)
                g2 /= len(active2)
                g2 = g2 - np.mean(g2)
                for t in [0.005 / len(active2), 0.001 / len(active2), 1e-4 / len(active2)]:
                    a_try = _normalize(a - t * g2)
                    sc_try = _score(a_try)
                    if sc_try < best_sc - 1e-15:
                        best_a = a_try.copy()
                        best_sc = sc_try
                        a = a_try
                        improved = True
                        break

            if not improved:
                # Random escape
                rng_local = np.random.default_rng(it * 31 + 17)
                for _ in range(5):
                    sigma = 0.005 * (1 + it * 0.001)
                    z = np.log(np.maximum(a, 1e-30))
                    z += sigma * rng_local.standard_normal(n)
                    z -= z.mean()
                    a_try = _softmax(z)
                    sc_try = _score(a_try)
                    if sc_try < best_sc - 1e-15:
                        best_a = a_try.copy()
                        best_sc = sc_try
                        a = a_try
                        improved = True
                        break
                if not improved:
                    break

    return best_a, best_sc


def _make_structured_init(n, kind, param):
    """Create structured initializations."""
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "bump":
        i = np.arange(n) / n
        a = np.exp(-param * (i - 0.5) ** 2)
        return a / a.sum()
    elif kind == "cosine":
        i = np.arange(n) / n
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * i * param))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block":
        n_blocks = max(2, int(param))
        a = np.zeros(n)
        total_active = n * n_blocks // (2 * n_blocks)
        block_on = max(1, n * n_blocks // (2 * n_blocks * n_blocks))
        block_off = max(1, n // (2 * n_blocks) - block_on)
        for b in range(n_blocks):
            start = b * (block_on + block_off)
            end = min(start + block_on, n)
            a[start:end] = 1.0
        s = a.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return a / s
    elif kind == "alternating":
        a = np.zeros(n)
        period = max(4, int(param))
        on = period // 2
        for i in range(n):
            if (i % period) < on:
                a[i] = 1.0
        return a / a.sum()
    elif kind == "gaussian":
        sigma = n / max(1.0, param)
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - n / 2.0) / sigma) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, seed):
    rng = np.random.default_rng(seed)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    return _normalize(a_new)


def _refinement_loop(a, deadline, max_rounds=8):
    """Alternate between Newton deflation and L-BFGS."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Newton deflation
        nd_dl = min(deadline - 10, time.monotonic() + 30)
        a_nd, sc_nd = _newton_deflation(a, nd_dl, max_iter=40)
        if sc_nd < best_sc:
            best_sc = sc_nd
            best_a = a_nd.copy()
            a = a_nd

        # L-BFGS polish
        lb_dl = min(deadline - 5, time.monotonic() + 40)
        a_lb = _lbfgs_lse(a, lb_dl, min_time=5.0)
        sc_lb = _score(a_lb)
        if sc_lb < best_sc:
            best_sc = sc_lb
            best_a = a_lb.copy()
            a = a_lb

        if abs(_score(a) - best_sc) < 1e-15:
            break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining frequency-domain IAA, Fourier basis
    optimization, L-BFGS with log-sum-exp, and active-set Newton deflation.
    Uses multi-start with diverse initializations to escape local minima.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.12f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.12f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = np.ones(n_parent) / n_parent
        update_best(parent, _score(parent), "init")

    n_work = n_parent if n_parent > 0 else 16384

    # === Phase 1: L-BFGS from parent (proven effective) ===
    print(f"\n[Phase 1] L-BFGS LSE from parent (t={time.monotonic() - t_start:.1f}s)")
    lb1_dl = min(deadline - 800, time.monotonic() + 300)
    a_lse1 = _lbfgs_lse(parent, lb1_dl, min_time=800.0)
    update_best(a_lse1, _score(a_lse1), "lbfgs_parent")

    # === Phase 2: Frequency-domain IAA ===
    print(f"\n[Phase 2] Freq-domain IAA (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        f_dl = min(deadline - 600, time.monotonic() + 150)
        a_freq, sc_freq = _freq_smooth_iterate(best_seq, f_dl,
                                               n_iter_max=3000,
                                               t_initial=0.08,
                                               beta_initial=0.04)
        update_best(a_freq, sc_freq, "freq_iaa")

    # === Phase 3: Fourier basis optimization ===
    print(f"\n[Phase 3] Fourier basis opt (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        fb_dl = min(deadline - 500, time.monotonic() + 80)
        a_fb, sc_fb = _fourier_basis_optimize(best_seq, fb_dl, K=30)
        update_best(a_fb, sc_fb, "fourier_basis")

    # === Phase 4: Newton deflation ===
    print(f"\n[Phase 4] Newton deflation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        nd_dl = min(deadline - 400, time.monotonic() + 80)
        a_nd, sc_nd = _newton_deflation(best_seq, nd_dl, max_iter=80)
        update_best(a_nd, sc_nd, "newton_def")

    # === Phase 5: Multi-start with diverse initializations ===
    print(f"\n[Phase 5] Multi-start (t={time.monotonic() - t_start:.1f}s)")
    candidates = []
    if best_seq is not None:
        candidates.append(("pert_small", _perturb(best_seq, 0.02, 100)))
        candidates.append(("pert_med", _perturb(best_seq, 0.05, 200)))
        candidates.append(("pert_large", _perturb(best_seq, 0.1, 300)))

    candidates.append(("uniform", np.ones(n_work) / n_work))
    candidates.append(("bump5", _make_structured_init(n_work, "bump", 5.0)))
    candidates.append(("bump15", _make_structured_init(n_work, "bump", 15.0)))
    candidates.append(("cos2", _make_structured_init(n_work, "cosine", 2.0)))
    candidates.append(("cos3", _make_structured_init(n_work, "cosine", 3.0)))
    candidates.append(("block4", _make_structured_init(n_work, "block", 4.0)))
    candidates.append(("block6", _make_structured_init(n_work, "block", 6.0)))
    candidates.append(("alt8", _make_structured_init(n_work, "alternating", 8)))
    candidates.append(("alt16", _make_structured_init(n_work, "alternating", 16)))

    for label, init_a in candidates:
        if time.monotonic() > deadline - 300:
            break
        sc_init = _score(init_a)
        # Only pursue if not too far from best
        if sc_init < best_score * 1.15 or sc_init < 2.5:
            ms_dl = min(deadline - 250, time.monotonic() + 60)
            a_ms = _lbfgs_lse(init_a, ms_dl, min_time=250.0)
            update_best(a_ms, _score(a_ms), f"ms_{label}")
            # Quick freq IAA on the result
            if time.monotonic() < deadline - 250:
                f2_dl = min(deadline - 240, time.monotonic() + 20)
                a_f2, sc_f2 = _freq_smooth_iterate(a_ms, f2_dl,
                                                   n_iter_max=500,
                                                   t_initial=0.03,
                                                   beta_initial=0.02)
                update_best(a_f2, sc_f2, f"ms_{label}_freq")

    # === Phase 6: Refinement loop on best ===
    print(f"\n[Phase 6] Refinement loop (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        ref_dl = min(deadline - 100, time.monotonic() + 120)
        a_ref, sc_ref = _refinement_loop(best_seq, ref_dl, max_rounds=6)
        update_best(a_ref, sc_ref, "refinement")

    # === Phase 7: Try larger n ===
    print(f"\n[Phase 7] Larger n (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_large = 32768
        a_large = _interpolate_seq(best_seq, n_large)
        sc_large = _score(a_large)
        print(f"  n={n_large} interpolated score: {sc_large:.12f}")
        if sc_large < best_score * 1.01:
            lb3_dl = min(deadline - 60, time.monotonic() + 50)
            a_large_opt = _lbfgs_lse(a_large, lb3_dl, min_time=60.0)
            update_best(a_large_opt, _score(a_large_opt), "n32768")

    # === Phase 8: Final polish ===
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        fin_dl = min(deadline - 5, time.monotonic() + 60)
        a_fin = _lbfgs_lse(best_seq, fin_dl, min_time=5.0)
        update_best(a_fin, _score(a_fin), "final_lbfgs")

        if time.monotonic() < deadline - 3:
            a_np, sc_np = _newton_deflation(a_fin, deadline, max_iter=20)
            update_best(a_np, sc_np, "final_newton")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
