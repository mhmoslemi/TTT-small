import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

Optimizer for 2*n*max(conv(a,a))/sum(a)^2 using:

2. L-BFGS with log-sum-exp smoothing (fine polish)
3. Multi-n exploration with resampling
4. Structured random restarts
5. Top-K peak sum objective (alternative smoothing)

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)

This is: minimize `sum_{k in active} w[k] * sum_j a[j] * a[k-j]` subject to `a >= 0, sum a = 1`.

The Hessian of the objective is: `H[j,l] = sum_{k in active} w[k] * (1[k-j=l] + 1[k-l=j])`

This is a sparse matrix! For each `k`, the Hessian has non-zeros at positions `(j, k-j)` and `(k-j, j)`. So the Hessian has at most `2 * |active| * n` non-zeros (actually less due to overlap).

For `|active| = 10` and `n = 16384`, that's about `327680` non-zeros. This is a very sparse matrix!

I can use `cvxpy` with a sparse QP. But the QP is `n = 16384`, which might be too large for most solvers.

Alternative: Use **scipy.optimize.minimize** with the method 'trust-constr' or 'SLSQP', providing the sparse Hessian.

Actually, for a QP with `n = 16384` variables, even sparse, most solvers will struggle. Let me use a custom solver.

**Custom QP solver: Projected gradient with FFT-based Hessian-vector product**

The Hessian-vector product: `(H*v)[j] = sum_l H[j,l] * v[l] = sum_{k in active} w[k] * (v[k-j] + v[j-k])`

(with boundary handling). This can be computed in `O(|active| * n)` directly, or in `O(n log n)` using FFT (by treating it as a sum of shifted correlations).

For `|active| = 10` and `n = 16384`, direct computation is `10 * 16384 = 163840` operations per Hessian-vector product. That's very fast!

So I'll use **projected gradient** for the QP solve:
- `a_{new} = project_simplex(a - lr * H * a)`
- where `H * a` is computed directly (since `|active|` is small)

"""
Optimizer for 2*n*max(conv(a,a))/sum(a)^2 using:
1. QP-based peak shaving via projected gradient with FFT/Hankel structure
2. L-BFGS with log-sum-exp smoothing (coarse-to-fine beta schedule)
3. Adaptive active-set selection
4. Multi-n exploration with resampling
5. Structured restarts and symmetry exploitation

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _hess_vec(active, w, v):
    """Compute Q_w * v where Q_w[j,l] = sum_k w[k]*delta_{j+l,k}.
    Direct computation, O(K*n)."""
    n = len(v)
    result = np.zeros(n)
    for k, wk in zip(active, w):
        if wk < 1e-15:
            continue
        # (contribution from peak k): sum_l v[l] * [j+l == k] means l = k-j
        # So (H*v)[j] gets wk * v[k-j] if 0 <= k-j < n
        # And also wk * v[j-k] if 0 <= j-k < n (for the symmetric part)
        # Actually: (H*v)[j] = sum_k w[k] * (v[k-j] + v[j-k]) with bounds
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            idx = k - np.arange(j_lo, j_hi + 1)
            result[j_lo:j_hi + 1] += wk * v[idx]
        j_lo2 = max(0, k)
        j_hi2 = min(n - 1, k + n - 1)
        if j_lo2 <= j_hi2:
            idx = np.arange(j_lo2, j_hi2 + 1) - k
            result[j_lo2:j_hi2 + 1] += wk * v[idx]
    return result


def _qp_projected_grad(a_init, active, w, max_iter=500, lr_init=0.1):
    """Solve min a^T Q_w a s.t. a>=0, sum(a)=1 via projected gradient."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_f = float(a @ _hess_vec(active, w, a))
    lr = lr_init
    no_improve = 0

    for it in range(max_iter):
        grad = _hess_vec(active, w, a)
        # Project gradient to tangent space
        grad_t = grad - np.mean(grad)
        gn = np.linalg.norm(grad_t)
        if gn < 1e-18:
            break
        # Adaptive step
        step = lr / gn
        a_cand = a - step * grad_t
        a_proj = _project_simplex(a_cand)
        f_new = float(a_proj @ _hess_vec(active, w, a_proj))
        if f_new < best_f - 1e-16:
            best_f = f_new
            best_a = a_proj.copy()
            a = a_proj
            no_improve = 0
            lr = min(lr * 1.02, 0.5)
        else:
            no_improve += 1
            lr *= 0.7
            if no_improve > 50:
                break

    return best_a, best_f


def _qp_peak_shave(a_init, deadline, min_remain=20.0):
    """Iteratively solve QP subproblems with adaptive active sets."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    K = 5
    K_max = 200
    patience = 0
    max_patience = 300

    while time.monotonic() + min_remain < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            break

        # Adaptive K
        c_sorted = np.sort(c)[::-1]
        if len(c_sorted) > K:
            gap = c_sorted[0] - c_sorted[K]
            if gap < c_max * 0.01 and K < K_max:
                K = min(K_max, int(K * 1.5) + 1)
        else:
            K = min(K_max, K + 1)

        # Select active peaks
        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]

        # Weights: softmax over active peaks
        active_vals = c[active]
        active_shifted = active_vals - active_vals.max()
        w = np.exp(active_shifted / (c_max * 0.01 + 1e-30))
        w = w / w.sum()

        # Solve QP
        a_new, _ = _qp_projected_grad(a, active, w, max_iter=300, lr_init=0.05)
        sc_new = _score(a_new)
        if sc_new < sc:
            a = a_new
        else:
            # Try smaller step
            a_mix = 0.5 * a + 0.5 * a_new
            a_mix = _project_simplex(a_mix)
            sc_mix = _score(a_mix)
            if sc_mix < sc:
                a = a_mix
            else:
                if patience > 50:
                    # Random perturbation
                    rng = np.random.default_rng(patience)
                    pert = 0.005 * rng.standard_normal(n)
                    a_pert = _project_simplex(a + pert)
                    sc_pert = _score(a_pert)
                    if sc_pert < sc:
                        a = a_pert
                    else:
                        break

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_remain=15.0, n_beta_levels=12):
    """L-BFGS with geometric beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap * 0.5, max_c * 0.005)

    betas = base_beta * (0.6 ** np.arange(n_beta_levels))

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remain > deadline:
            break
        remaining = deadline - min_remain - time.monotonic()
        if remaining < 3:
            break

        # Scale iterations based on remaining time
        est_time = 0.003 * max(1.0, n / 16384.0)
        max_iter = min(8000, max(50, int(remaining / (est_time * 15))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 3 == 0:
                    print(f"    LBFGS beta_idx={i}/{n_beta_levels}: score={current_sc:.10f}")
        except Exception:
            pass

    return current_a


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _nesterov_subgrad(a_init, deadline, min_remain=20.0, K_init=5, K_max=150, seed=0):
    """Nesterov-accelerated subgradient with adaptive K."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    momentum = 0.8
    lr = 0.003
    K = K_init
    t = 0
    no_improve = 0

    while time.monotonic() + min_remain < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive K
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
        if gap < c_max * 0.003 and K < K_max:
            K = min(K_max, int(K * 1.4) + 1)

        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]

        # Subgradient
        grad = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                idx = k - np.arange(j_lo, j_hi + 1)
                grad[j_lo:j_hi + 1] += a[idx]
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_lo2 <= j_hi2:
                idx = np.arange(j_lo2, j_hi2 + 1) - k
                grad[j_lo2:j_hi2 + 1] += a[idx]

        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break

        effective_lr = lr / (gnorm + 1e-18) * 0.15
        v = momentum * v - effective_lr * grad_t
        a_cand = a + v
        a_proj = _project_simplex(a_cand)

        c_new = _full_convolve(a_proj, a_proj)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < current_sc:
            a = a_proj
            lr = min(lr * 1.01, 0.01)
        else:
            v = 0.5 * v
            lr *= 0.85

        if no_improve > 150:
            if no_improve > 600:
                break
            # Perturbation
            perturb = 0.002 * rng.standard_normal(n)
            a_pert = _project_simplex(a + perturb)
            sc_pert = 2.0 * n * float(_full_convolve(a_pert, a_pert).max())
            if sc_pert < best_sc:
                a = a_pert
                best_a = a.copy()
                best_sc = sc_pert
            v = np.zeros(n)
            lr = 0.003
            no_improve = 0

    return best_a, best_sc

    p1_deadline = min(deadline - 900, t_start + 350)
    if time.monotonic() < p1_deadline:
        a_qp1, sc_qp1 = _qp_peak_shave(best_seq, p1_deadline, min_remain=30.0)
        if sc_qp1 < best_score:
            best_seq = a_qp1.copy()
            best_score = sc_qp1
            print(f"  NEW BEST: {best_score:.10f} [QP1] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  QP1: no improvement (best={best_score:.10f})")

    # === Phase 2: L-BFGS coarse-to-fine ===
    print(f"\n[P2] LBFGS coarse-to-fine (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 600, t_start + 600)
    if time.monotonic() < p2_deadline:
        a_lb1 = _run_lbfgs(best_seq, p2_deadline, min_remain=30.0, n_beta_levels=14)
        sc_lb1 = _score(a_lb1)
        if sc_lb1 < best_score:
            best_seq = a_lb1.copy()
            best_score = sc_lb1
            print(f"  NEW BEST: {best_score:.10f} [LBFGS1] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Nesterov subgradient ===
    print(f"\n[P3] Nesterov subgradient (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 400, t_start + 800)
    if time.monotonic() < p3_deadline:
        a_nes1, sc_nes1 = _nesterov_subgrad(best_seq, p3_deadline, min_remain=30.0,
                                            K_init=5, K_max=100, seed=seed)
        if sc_nes1 < best_score:
            best_seq = a_nes1.copy()
            best_score = sc_nes1
            print(f"  NEW BEST: {best_score:.10f} [Nes1] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Second QP pass (finer) ===
    print(f"\n[P4] QP Peak Shaving round 2 (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 250, t_start + 1000)
    if time.monotonic() < p4_deadline:
        a_qp2, sc_qp2 = _qp_peak_shave(best_seq, p4_deadline, min_remain=20.0)
        if sc_qp2 < best_score:
            best_seq = a_qp2.copy()
            best_score = sc_qp2
            print(f"  NEW BEST: {best_score:.10f} [QP2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Fine L-BFGS ===
    print(f"\n[P5] Fine LBFGS (t={time.monotonic()-t_start:.1f}s)")
    p5_deadline = min(deadline - 150, t_start + 1200)
    if time.monotonic() < p5_deadline:
        a_lb2 = _run_lbfgs(best_seq, p5_deadline, min_remain=20.0, n_beta_levels=16)
        sc_lb2 = _score(a_lb2)
        if sc_lb2 < best_score:
            best_seq = a_lb2.copy()
            best_score = sc_lb2
            print(f"  NEW BEST: {best_score:.10f} [LBFGS2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([n_parent, 8192, 12288, 16384, 24576, 32768]))
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        if sc_init >= best_score * 1.05:
            continue  # skip if too far from best
        print(f"  n={n_try}: init score={sc_init:.10f}")
        # Quick L-BFGS
        if time.monotonic() < deadline - 40:
            a_opt = _run_lbfgs(a_init, deadline - 20, min_remain=10.0, n_beta_levels=8)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final Nesterov + QP ===
    print(f"\n[P7] Final optimization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        a_nes_f, sc_nes_f = _nesterov_subgrad(best_seq, deadline - 20, min_remain=10.0,
                                              K_init=3, K_max=80, seed=seed + 99)
        if sc_nes_f < best_score:
            best_seq = a_nes_f.copy()
            best_score = sc_nes_f
            print(f"  NEW BEST: {best_score:.10f} [NesF] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final L-BFGS polish ===
    if time.monotonic() < deadline - 5:
        a_fin = _run_lbfgs(best_seq, deadline - 2, min_remain=2.0, n_beta_levels=10)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [FinLBFGS] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
