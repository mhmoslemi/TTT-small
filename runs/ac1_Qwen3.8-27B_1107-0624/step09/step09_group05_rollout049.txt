Improves on previous by: gap-adaptive active sets, structured perturbations,

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)

def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

def _grad_soft_structured(a, beta, n):
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)

def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft_structured(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)

class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)
    def grad(self, z):
        return _smograd(z, self.beta, self.n)

This is a linear system in the unknowns $a$ and $p$ and $\lambda$. If we fix the support of $a$ (i.e., assume $a_j > 0$ for all $j$), then we have $n$ equations in $n + m + 1$ unknowns. We can solve this by:
- Setting $p_i = 1/m$ (uniform weights) and solving for $a$ and $\lambda$.
- Or, using a more sophisticated approach.

Actually, for the case where the active set is just the single point $k^* = \arg\max (a*a)$, the KKT condition is:
$$2 a[k^* - j] = \lambda \quad \forall j$$
which means $a[k^* - j]$ is constant for all $j$, i.e., $a$ is constant. This gives the uniform solution with score 2.

For a larger active set, the condition is more interesting. Let's say the active set is $\{k_1, ..., k_m\}$. Then:
$$\sum_{i=1}^m p_i \cdot a[k_i - j] = \lambda/2 \quad \forall j$$

This means that the "blurred" version of $a$ (blurred by the distribution $p$ on the active set) is constant. If $p$ is uniform on $S$, this means the average of $a$ over the shifted active set is constant.

This is a linear system that can be solved using FFT-based methods.

gap-adaptive active-set KKT refinement, structured perturbation exploration,
and multi-length resampling. Key differences from previous:
- Gap-adaptive active set (spectral gap based tolerance)
- Structured sinusoidal perturbations for exploration
- More aggressive beta schedule with warm-starting
- Direct KKT linear-system solver for active set verification

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)

def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

def _grad_soft_structured(a, beta, n):
    """Gradient of soft_max(conv(a,a)) using convolution structure."""
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)

def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft_structured(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)

class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)
    def grad(self, z):
        return _smograd(z, self.beta, self.n)

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, verbose=True, max_beta=None):
    """L-BFGS with aggressive adaptive beta schedule and warm-starting."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = float(2.0 * n * c0_max)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    # Compute gap for beta scheduling
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(200, len(c_sorted) - 1)]
    if gap < c0_max * 0.002:
        gap = c0_max * 0.002

    if max_beta is not None:
        multipliers = [m for m in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
                                   0.015625, 0.0078125, 0.00390625, 0.002,
                                   0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
                                   0.00003125, 0.000015625, 0.0000078,
                                   0.0000039, 0.000002, 0.000001, 0.0000005,
                                   0.0000002, 0.0000001]
                      if gap * m <= max_beta + 1e-30]
        if not multipliers:
            multipliers = [max_beta / gap]
    else:
        multipliers = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
                       0.015625, 0.0078125, 0.00390625, 0.002,
                       0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
                       0.00003125, 0.000015625, 0.0000078,
                       0.0000039, 0.000002, 0.000001, 0.0000005,
                       0.0000002, 0.0000001]
    betas = gap * multipliers

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        size_factor = max(1.0, n / 4096.0)
        est_iter_time = 0.0015 * size_factor
        max_iter = min(12000, max(200, int(remaining / (est_iter_time * 25))))

        # For very small beta, use more iterations
        if beta < gap * 0.001:
            max_iter = min(max_iter, int(remaining / (est_iter_time * 10)))

        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    elapsed = time.monotonic() - _t_start
                    print(f"    beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}, t={elapsed:.0f}s")
            else:
                if verbose and i < 5:
                    print(f"    beta={beta:.2e}: no improvement (best={best_sc:.10f})")
        except Exception as e:
            if verbose:
                print(f"    beta={beta:.2e}: error {e}")

    return best_a, best_sc

def _active_set_refine_v2(a_init, deadline, min_time_remaining=30.0):
    """Improved active-set refinement with gap-adaptive tolerance and
    direct KKT linear system solving."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_frac = 0.003
    min_tol = 5e-5
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            tol_frac = max(min_tol, tol_frac * 0.5)
            if tol_frac <= min_tol * 1.05:
                break
            no_improve = 0

        # Gap-adaptive active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros(len(c), dtype=bool)
            active[idx] = True
            n_active = 1

        # Compute gradient from active set
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]

        # Support
        support = a > 1e-10 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        # KKT: grad_avg should be constant on support
        lam = float(np.mean(grad_avg[support]))
        residual = grad_avg[support] - lam
        res_norm = float(np.sqrt(np.mean(residual**2)))

        # If KKT is nearly satisfied, tighten
        if res_norm < 1e-10 * c_max:
            tol_frac = max(min_tol, tol_frac * 0.7)
            continue

        # Projected gradient step
        for eta in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]:
            a_new = a.copy()
            a_new[support] -= eta * residual
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                a = a_new
                break
        else:
            # No improvement, tighten tolerance
            tol_frac = max(min_tol, tol_frac * 0.85)
            continue

        # Tighten when improving
        if no_improve == 0:
            tol_frac = max(min_tol, tol_frac * 0.96)

    return best_a, best_sc

def _polyak_subgradient(a_init, deadline, min_time_remaining=30.0):
    """Polyak-step-size subgradient method with structured step decay."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lb = 2.0 * n / (2.0 * n - 1.0)
    step_scale = 1.0
    t = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()

        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros(len(c), dtype=bool)
            active[idx] = True
            n_active = 1

        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        gap = current_sc - lb
        if gap < 1e-12:
            break
        grad_c = grad - np.mean(grad)
        g_norm_sq = float(np.dot(grad_c, grad_c))
        if g_norm_sq < 1e-20:
            break

        step = step_scale * gap / g_norm_sq
        step = min(step, 0.05)

        a_new = a - step * grad_c
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a = a_new / s_new
        t += 1

        if t % 200 == 0:
            step_scale *= 0.93

    return best_a, best_sc

def _perturb_and_optimize(a_base, deadline, min_time_remaining=60.0, n_perturb=5, rng=None):
    """Add structured perturbations and re-optimize."""
    if rng is None:
        rng = np.random.default_rng(123)
    n = len(a_base)
    best_a = a_base.copy()
    best_sc = _score(a_base)

    for ip in range(n_perturb):
        if time.monotonic() + min_time_remaining > deadline:
            break

        # Choose perturbation type
        pert_type = ip % 3
        if pert_type == 0:
            # Gaussian noise
            d = rng.standard_normal(n)
        elif pert_type == 1:
            # Low-frequency sinusoid
            freq = rng.uniform(2, 20)
            t = np.arange(n)
            phase = rng.uniform(0, 2 * np.pi)
            d = np.sin(2 * np.pi * freq * t / n + phase)
        else:
            # Tapered noise (more at edges)
            d = rng.standard_normal(n)
            edge_weight = 1.0 + 3.0 * (2.0 * np.abs(np.arange(n) / n - 0.5)) ** 2
            d = d * edge_weight

        d = d - np.mean(d)  # Zero mean to preserve sum approximately
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-10:
            continue
        d = d / d_norm

        # Try different perturbation sizes
        for eps in [0.05, 0.02, 0.01, 0.005]:
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_pert = a_base + eps * d
            a_pert = np.maximum(a_pert, 0.0)
            s_p = a_pert.sum()
            if s_p < 1e-10:
                continue
            a_pert = a_pert / s_p

            # Quick L-BFGS optimization
            p_end = min(deadline - 10, time.monotonic() + min_time_remaining * 0.6)
            a_opt, sc_opt = _run_lbfgs(a_pert, p_end, min_time_remaining=min_time_remaining * 0.5,
                                       verbose=False, max_beta=_get_max_beta(a_pert))
            if sc_opt < best_sc - 1e-14:
                best_sc = sc_opt
                best_a = a_opt.copy()
                print(f"    Pert {ip} (eps={eps}): {best_sc:.10f}")

def _get_max_beta(a):

    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]

def _kkt_direct_solve(a_init, deadline, min_time_remaining=30.0):
    """Direct KKT solve: given active set, solve the linear system.
    
    The KKT conditions for the active set S with uniform weights p:
    (1/n_active) * sum_{k in S} 2*a[k-j] = lambda for all j in support
    
    This means: for all j in support, the average of a over positions
    {k-j : k in S, k-j in [0,n-1]} equals lambda.
    
    We can set this up as a linear system and solve iteratively.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for iteration in range(200):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()

        # Active set with tight tolerance
        tol = 1e-4 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active < 1:
            break

        # Build the linear system: for each j in support,
        # (2/n_active) * sum_{k in active, 0<=k-j<n} a[k-j] = lambda
        # Let b[j] = (2/n_active) * sum_{k in active} a[max(0,min(n-1,k-j))]
        # We want b[j] to be constant (= lambda) for j in support.

        # Compute b via convolution
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 2.0 / n_active
        wc = _full_convolve(w, a_rev)
        b = wc[n - 1:2 * n - 1]

        support = a > 1e-10 * a.max()
        lam = float(np.mean(b[support]))
        residual = b[support] - lam

        # Update: a[j] -= eta * residual[j] for j in support
        # This is equivalent to the gradient step, but with exact active set
        eta = 0.05
        a_new = a.copy()
        a_new[support] -= eta * residual
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a = a_new / s_new

        # Check convergence
        res_norm = float(np.sqrt(np.mean(residual**2)))
        if res_norm < 1e-12 * c_max:
            break

    return best_a, best_sc

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _make_profile(n, kind="tapered"):
    if kind == "uniform":

    elif kind == "tapered":

        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)

    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.6 * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

    p1_end = _t_start + min(400, (budget_s - 6) * 0.25)
    a1, sc1 = _run_lbfgs(best_seq, p1_end, min_time_remaining=200, verbose=True)
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  P1 result: {best_score:.10f}")

    # === Phase 2: KKT direct solve ===
    print(f"\n[P2] KKT direct solve (t={time.monotonic()-_t_start:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p2_end = min(deadline - 300, time.monotonic() + 200)
        a2, sc2 = _kkt_direct_solve(best_seq, p2_end, min_time_remaining=50)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(f"  P2 result: {best_score:.10f}")

    # === Phase 3: Active set refinement ===
    print(f"\n[P3] Active-set refine (t={time.monotonic()-_t_start:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p3_end = min(deadline - 200, time.monotonic() + 250)
        a3, sc3 = _active_set_refine_v2(best_seq, p3_end, min_time_remaining=50)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3 result: {best_score:.10f}")

    # === Phase 4: More L-BFGS (tighter) ===
    print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-_t_start:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p4_end = min(deadline - 200, time.monotonic() + 250)
        a4, sc4 = _run_lbfgs(best_seq, p4_end, min_time_remaining=100, verbose=True,
                             max_beta=_get_max_beta(best_seq))
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4 result: {best_score:.10f}")

    # === Phase 5: Structured perturbation exploration ===
    print(f"\n[P5] Perturbation exploration (t={time.monotonic()-_t_start:.0f}s)")
    if time.monotonic() + 120 < deadline:
        p5_end = min(deadline - 150, time.monotonic() + 300)
        a5, sc5 = _perturb_and_optimize(best_seq, p5_end,
                                         min_time_remaining=80, n_perturb=8, rng=rng)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  P5 result: {best_score:.10f}")

    # === Phase 6: Multi-length search ===
    print(f"\n[P6] Multi-length (t={time.monotonic()-_t_start:.0f}s)")
    n_options = sorted(set([n_main // 2, n_main * 3 // 4, n_main,
                            n_main * 5 // 4, n_main * 2]) - {0})
    n_options = [x for x in n_options if 2048 <= x <= 32768 and x != n_main]
    for n_try in n_options:
        if time.monotonic() + 80 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        print(f"  n={n_try}: initial score={sc_r:.10f}")
        if time.monotonic() + 60 > deadline:
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
            continue
        p6_end = min(deadline - 80, time.monotonic() + 120)
        a6, sc6 = _run_lbfgs(a_r, p6_end, min_time_remaining=30, verbose=False,
                             max_beta=_get_max_beta(a_r))
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  n={n_try}: {best_score:.10f}")

    # === Phase 7: Subgradient + final polish ===
    print(f"\n[P7] Subgradient + final (t={time.monotonic()-_t_start:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p7a_end = min(deadline - 60, time.monotonic() + 100)
        a7a, sc7a = _polyak_subgradient(best_seq, p7a_end, min_time_remaining=40)
        if sc7a < best_score:
            best_score = sc7a
            best_seq = a7a.copy()
            print(f"  P7a subgrad: {best_score:.10f}")

    # Final L-BFGS polish
    if time.monotonic() + 20 < deadline:
        p7b_end = min(deadline - 5, time.monotonic() + 80)
        a7b, sc7b = _run_lbfgs(best_seq, p7b_end, min_time_remaining=5, verbose=False,
                               max_beta=_get_max_beta(best_seq))
        if sc7b < best_score:
            best_score = sc7b
            best_seq = a7b.copy()
            print(f"  P7b final L-BFGS: {best_score:.10f}")

    # Very final active set polish
    if time.monotonic() + 10 < deadline:
        p7c_end = min(deadline - 3, time.monotonic() + 30)
        a7c, sc7c = _active_set_refine_v2(best_seq, p7c_end, min_time_remaining=3)
        if sc7c < best_score:
            best_score = sc7c
            best_seq = a7c.copy()
            print(f"  P7c final active set: {best_score:.10f}")

    # Final evaluation
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-_t_start:.1f}s")
    return [float(x) for x in final]
