Actually, one more important consideration: the log-sum-exp approximation. When beta is very small, the log-sum-exp is a very accurate approximation of max, but the gradient becomes concentrated on a single lag (the maximum). For k=5: average of top-5 (even smoother)

"""
Improved optimizer for minimum peak linear autocorrelation sequences.
Key improvements: correct FFT-based gradient, adaptive beta continuation,
top-k objective for polishing, multi-scale exploration, and random restarts.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _sliding_dot(a, w, n):
    """Compute first_term[i] = sum_{j=0}^{n-1} a[j]*w[i+j] for i=0..n-1.
    Uses FFT for O(n log n) computation.

    N = _next_pow2(3 * n)
    fa = np.fft.rfft(a, N)
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(fa * fw, N)[:3 * n - 2]
    return conv[n - 1:2 * n - 1][::-1]
