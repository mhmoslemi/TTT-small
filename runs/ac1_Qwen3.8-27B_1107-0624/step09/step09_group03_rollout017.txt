```python
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)

def _successive_lp(a_init, deadline, max_iter=200):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Identify active set
        # Use a tolerance relative to c_max
        eps = 0.001
        active = c >= c_max * (1 - eps)
        if not active.any():
            active[np.argmax(c)] = True
        K = np.where(active)[0]
        m = len(K)
        
        # Compute gradients for active lags
        # grad[k, j] = 2 * a[K[k] - j] for valid j
        G = np.zeros((m, n))
        for i, k in enumerate(K):
            # grad at a_j = 2 * a[k-j] for max(0,k-n+1) <= j <= min(k, n-1)
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            if j_lo <= j_hi:
                G[i, j_lo:j_hi+1] = 2 * a[k - np.arange(j_lo, j_hi+1)]
        
        # LP: min t s.t. G @ a <= t * ones + c_active, sum(a) = 1, a >= 0
        # Rewrite: G @ a - t <= c_active
        # Variables: [a_0, ..., a_{n-1}, t]
        
        # c_active = c[K]
        c_act = c[K]
        
        # A_ub: [G, -1] @ [a; t] <= c_act
        A_ub = np.hstack([G, -np.ones((m, 1))])
        b_ub = c_act
        
        # A_eq: [ones, 0] @ [a; t] = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        # Objective: minimize t = e_{n+1}^T [a; t]
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        # Bounds: a_j >= 0, t unbounded below (but practically t >= 0)
        bounds = [(0, None)] * n + [(0, None)]
        
        # Solve LP
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        
        if not result.success:
            break
        
        a_new = result.x[:n]
        t_new = result.x[n]
        
        # Check if the new point satisfies the actual constraints
        c_new = _full_convolve(a_new, a_new)
        if float(c_new.max()) <= t_new + 1e-10:
            # Success! Update
            a = a_new
        else:
            # Infeasible, take a fraction of the step
            progress = 0.5
            for _ in range(10):
                a_try = a + progress * (a_new - a)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    break
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max * (1 + 1e-10):
                    a = a_try
                    break
                progress *= 0.5
            else:
                break
        
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Check convergence
        if abs(sc - best_sc) < 1e-14:
            break
    
    return best_a, best_sc
```

Let me also consider: what if I use a second-order method directly? The KKT conditions for the problem (with active set $K$) are:

$$\min_a t \quad \text{s.t.} \quad (a*a)_k \leq t \; \forall k \in K, \quad \sum a = 1, \quad a \geq 0$$

At the optimum, some of the constraints are active. If we know which ones, we can solve the KKT system. The KKT system is:

For each $a_j > 0$:
$$\sum_{k \in K_{\text{active}}} \lambda_k \cdot 2a_{k-j} + \nu - \mu_j = 0$$
$$\mu_j \geq 0$$

For each active constraint $k \in K_{\text{active}}$:
$$(a*a)_k = t$$

Plus $\sum a = 1$ and $a_j \geq 0$.

This is a nonlinear system (because the Hessian of $(a*a)_k$ depends on $a$). We can solve it with Newton's method.

The Newton system is:
$$\begin{pmatrix} H & G^T & \mathbf{1} \\ G & 0 & 0 \\ \mathbf{1}^T & 0 & 0 \end{pmatrix} \begin{pmatrix} \Delta a \\ \Delta \lambda \\ \Delta \nu \end{pmatrix} = -\begin{pmatrix} \nabla_a L \\ (a*a)_K - t \\ \sum a - 1 \end{pmatrix}$$

where $H_{ij} = \sum_k \lambda_k \frac{\partial^2 (a*a)_k}{\partial a_i \partial a_j} = 2\sum_k \lambda_k \delta_{i+j,k}$

This Hessian has a special structure! It's a weighted sum of "anti-diagonal" matrices. Specifically, if we define $w_k = \lambda_k$ for $k \in K$ and $w_k = 0$ otherwise, then:
$$H_{ij} = 2w_{i+j}$$ (for valid $i+j$)

So $H$ is determined by the sequence $w = (\lambda_0, \lambda_1, \ldots, \lambda_{2n-2})$ restricted to the active set.

This means $H$ is a "Hankel-like" matrix (constant on anti-diagonals). Such matrices can be multiplied by a vector in $O(n \log n)$ using FFT!

$(Hv)_i = 2\sum_j w_{i+j} v_j = 2(h * v)_i$ where $h = w$ (reversed and appropriately padded) and $*$ is convolution.

More precisely: $(Hv)_i = 2\sum_j w_{i+j} v_j$. If we define $w'$ as the sequence $w$ reversed, then $(Hv)_i = 2(\text{conv}(w', v))_i$ (with appropriate indexing).
