Another alternative: use the **entropic regularization** approach where we solve:
`min_a sum_k c[k] * p[k] + (1/tau) * sum_i a_i * log(a_i / u_i)`
where `u` is a reference distribution. This has a closed-form solution (water-filling).

Actually, let me think about this more carefully. The mirror descent with negative entropy mirror on the simplex gives:
`a_i^{new} ∝ a_i^{old} * exp(-eta * grad_i)`

This is equivalent to multiplicative updates. For our problem:
`a_i^{new} ∝ a_i^{old} * exp(-eta * 2 * sum_k p[k] * a_{k-i})`

This is a clean algorithm! Let me implement it.

**Mirror Descent / Multiplicative Weights Algorithm:**

```
Initialize a (e.g., uniform)
For t = 1, 2, ..., T:
    Compute c = conv(a, a)
    Compute p[k] = c[k] / sum(c)  (or top-K version)
    Compute grad[i] = 2 * sum_k p[k] * a[k-i]  (valid indices only)
    Update: a_new[i] = a[i] * exp(-eta_t * grad[i])
    Normalize: a_new = a_new / sum(a_new)
    a = a_new
```

The step size `eta_t` can be set to `1/sqrt(t)` or use a backtracking line search.

This is clean and has theoretical guarantees for the matrix game formulation.

Let me also add: **Nesterov acceleration** for the mirror descent.

And: **Adaptive top-K** where K increases over time (start with large K for smoothness, decrease for focus).

Let me now write the full code:

"""
Multi-phase optimizer using dual-ascent (matrix game), mirror descent,
and L-BFGS polishing to minimize 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_smooth(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _topk_smooth_obj_grad(z, beta, K, n):
    z_max = z.max()
    a = np.exp(z - z_max)
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        c_top = c
    
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    
    ga = _grad_smooth(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


class _SmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _topk_smooth_obj_grad(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _topk_smooth_obj_grad(z, self.beta, self.K, self.n)[1]


def _mirror_descent(a_init, deadline, min_time_remaining=60.0):
    """Mirror descent (multiplicative weights) on the simplex."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    n_2nm1 = 2 * n - 1
    T = max(100, int((deadline - min_time_remaining - time.monotonic()) * 50))
    
    # Save iterates for averaging
    a_avg = np.zeros(n)
    count = 0
    
    # Parameters
    K_frac = 0.05  # Start with 5% of peaks
    min_K = 5
    
    for t in range(T):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        # Compute convolution
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        # Track and average
        a_avg += a
        count += 1
        a_avg_norm = a_avg / count
        sc_avg = 2.0 * n * float(_full_convolve(a_avg_norm, a_avg_norm).max())
        if sc_avg < best_sc:
            best_sc = sc_avg
            best_a = a_avg_norm.copy()
        
        # Select active peaks (top K)
        K = max(min_K, int(n_2nm1 * K_frac))
        top_idx = np.argpartition(c, -K)[-K:]
        
        # Dual update: p proportional to c on active set
        p = np.zeros(n_2nm1)
        p[top_idx] = c[top_idx]
        p_sum = p.sum()
        if p_sum < 1e-30:
            p[int(np.argmax(c))] = 1.0
        else:
            p /= p_sum
        
        # Primal update: multiplicative weights
        # grad[i] = 2 * sum_k p[k] * a[k-i]
        grad = _grad_smooth(a, p)
        
        # Step size: eta = 1 / (t+1) * scale
        eta = 2.0 / (np.sqrt(t + 1) + 1.0)
        
        # Multiplicative update
        log_update = -eta * grad
        log_update -= log_update.max()  # For numerical stability
        a_new = a * np.exp(log_update)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a
        else:
            a_new /= s_new
        
        a = a_new
        
        # Animate K: start large, decrease
        if t < T // 2:
            K_frac = max(0.005, 0.05 * (1.0 - t / (T * 0.5)))
    
    # Final comparison
    c = _full_convolve(a, a)
    sc_final = 2.0 * n * float(c.max())
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    
    return best_a, best_sc


def _mirror_descent_nesterov(a_init, deadline, min_time_remaining=60.0):
    """Mirror descent with Nesterov acceleration."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    y = a.copy()  # Nesterov momentum point
    best_a = a.copy()
    best_sc = _score(a)
    
    n_2nm1 = 2 * n - 1
    
    # Step size schedule
    beta = 2.0  # Nesterov parameter
    
    t = 0
    max_iter = max(100, int((deadline - min_time_remaining - time.monotonic()) * 80))
    
    for t in range(max_iter):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc:
            best_sc = current_sc
            best_a = y.copy()
        
        # Select active peaks
        K = max(5, int(n_2nm1 * 0.02))
        top_idx = np.argpartition(c_y, -K)[-K:]
        p = np.zeros(n_2nm1)
        p[top_idx] = c_y[top_idx]
        p_sum = p.sum()
        if p_sum > 1e-30:
            p /= p_sum
        else:
            p[int(np.argmax(c_y))] = 1.0
        
        # Gradient at y
        grad_y = _grad_smooth(y, p)
        
        # Step size
        eta = 2.0 / (beta + t)  # Standard Nesterov step size
        
        # Gradient step (in log space for multiplicative)
        log_step = -eta * grad_y
        log_step -= log_step.max()
        a_new = y * np.exp(log_step)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new /= s_new
        
        # Nesterov update
        theta = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y_new = (theta * a_new + (beta - 1.0) * a / (beta + t)) 
        # Actually, let me use the standard Nesterov acceleration:
        # t_{k+1} = (1 + sqrt(1 + 4 t_k^2)) / 2
        # x_{k+1} = (t_k - 1)/t_{k+1} * x_k + t_{k+1}/(t_k - 1 + t_{k+1}) * (x_k - eta * grad)
        # This is getting complicated. Let me use a simpler momentum approach.
        
        # Simple momentum:
        # y = a - m * (a - a_prev)  (in log space)
        if t == 0:
            a_prev = a
            y = a_new
            a = a_new
        else:
            momentum = min(0.9, t / (10.0 + t))
            # Update in log space
            log_a = np.log(np.maximum(a, 1e-30))
            log_a_new = np.log(np.maximum(a_new, 1e-30))
            log_y = momentum * (log_a_new - log_a) + log_a
            y = np.exp(log_y - log_y.max())
            y /= y.sum()
            a = a_new
    
    return best_a, best_sc


def _lbtfgs_optimize(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with top-K smoothing and annealing."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.008)
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # More aggressive schedule
    schedule = [
        (1.0, 0.12), (0.7, 0.10), (0.5, 0.07), (0.3, 0.05),
        (0.2, 0.04), (0.1, 0.03), (0.05, 0.02), (0.03, 0.015),
        (0.01, 0.01), (0.005, 0.007), (0.002, 0.005), (0.001, 0.003),
        (0.0005, 0.002), (0.0002, 0.001),
    ]
    
    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        beta = base_beta * beta_mult
        K = max(3, int((2 * n - 1) * K_frac))
        
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(20, int(remaining / (est_time * 15))))
        
        prob = _SmoProb(beta, K, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 150, 'maxls': 100}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _adam_smooth(a_init, deadline, min_time_remaining=30.0):
    """Adam optimizer with top-K smoothing."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    base_beta = max(max_c * 0.01, 1e-6)
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # Adam parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    # K annealing
    K_schedule = [0.1, 0.08, 0.06, 0.04, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
    K_idx = 0
    
    max_iter = int((deadline - min_time_remaining - time.monotonic()) * 100)
    max_iter = min(max_iter, 50000)
    
    for t in range(max_iter):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        # Update K every 5000 iterations
        K_frac = K_schedule[min(K_idx, len(K_schedule) - 1)]
        K = max(3, int((2 * n - 1) * K_frac))
        if t > 0 and t % 5000 == 0:
            K_idx = min(K_idx + 1, len(K_schedule) - 1)
        
        beta = base_beta * max(0.001, 1.0 - t / max_iter)
        
        a_curr = np.exp(z - z.max())
        a_curr = a_curr / a_curr.sum()
        
        obj, grad_z = _topk_smooth_obj_grad(z, beta, K, n)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** (t + 1))
        v_hat = v / (1 - beta2 ** (t + 1))
        
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Check improvement every 100 steps
        if t % 100 == 0:
            a_check = np.exp(z - z.max())
            a_check = a_check / a_check.sum()
            sc_check = _score(a_check)
            if sc_check < current_sc - 1e-15:
                current_sc = sc_check
                current_a = a_check.copy()
    
    return current_a


def _subgradient_accelerated(a_init, deadline, min_time_remaining=30.0):
    """Accelerated subgradient with adaptive active set."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    y = a.copy()  # Nesterov point
    best_a = a.copy()
    best_sc = _score(a)
    
    n_2nm1 = 2 * n - 1
    momentum = 0.0
    lr = 0.05
    stall = 0
    
    max_iter = int((deadline - min_time_remaining - time.monotonic()) * 200)
    max_iter = min(max_iter, 100000)
    
    for t in range(max_iter):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        c = _full_convolve(y, y)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Top-K active set
        K = max(3, int(n_2nm1 * 0.01))
        top_idx = np.argpartition(c, -K)[-K:]
        w = np.zeros(n_2nm1)
        w[top_idx] = 1.0 / K
        
        # Gradient
        g = _grad_smooth(y, w)
        g_centered = g - g.mean()
        
        # Step
        a_new = y - lr * g_centered
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        
        # Nesterov momentum
        momentum = min(0.9, momentum + 0.001)
        y = (1 + momentum) * a_new - momentum * a
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        else:
            y = a_new
        
        a = a_new
        
        # Adaptive step size
        if stall > 50:
            lr = max(0.001, lr * 0.9)
            if stall > 200:
                a = best_a.copy()
                y = a.copy()
                lr = 0.02
                stall = 0
        elif stall == 0:
            lr = min(0.1, lr * 1.01)
    
    # Final check
    c = _full_convolve(best_a, best_a)
    sc = 2.0 * n * float(c.max())
    if sc < best_sc:
        best_sc = sc
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.4 + 0.6 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "optimal_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.5 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2.0
        a
