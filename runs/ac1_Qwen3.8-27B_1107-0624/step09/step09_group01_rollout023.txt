def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _log_softmax(z):
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return z - z_max - np.log(e.sum())


def _make_asymmetric_profile(n, kind="ramp", param=0.3):
    t = np.linspace(0, 1, n)
    if kind == "linear":
        a = 1.0 + param * t
    elif kind == "exp":
        a = np.exp(param * t)
    elif kind == "power":
        a = (t + 0.01) ** param
    elif kind == "bias":
        a = 1.0 + param * (t - 0.5) * 2.0
    elif kind == "exp_ramp":
        a = np.exp(param * t) * (1.0 - 0.3 * t)
    elif kind == "gauss_asym":
        center = 0.3 + 0.4 * param
        a = np.exp(-2.0 * ((t - center) / (0.2 + 0.1 * param)) ** 2) + 0.05
    elif kind == "tapered_asym":
        a = np.sin(np.pi * t ** (0.5 + 0.5 * param)) ** 0.8
    elif kind == "uniform_tilt":
        a = 1.0 + param * np.sin(np.pi * t) * (2.0 * t - 1.0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_symmetric_profile(n, kind="tapered"):
    t = np.linspace(0, 1, n)
    if kind == "tapered":
        a = np.sin(np.pi * t) ** 0.6
    elif kind == "cos":

    elif kind == "uniform":
        a = np.ones(n)
    else:

    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _smoothed_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smoothed_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)

            beta *= 0.2
        else:
            beta *= 0.35

def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s
