Let me also add a new technique: **spectral shaping**. In the frequency domain, we want the periodogram to be flat. We can iteratively:
1. Compute the periodogram P(f) = |DFT(a)|^2
2. Take the "sqrt" to get |A(f)|
3. Modify |A(f)| to be flatter (e.g., clip the peaks in the periodogram)
4. Take the inverse DFT to get a modified sequence
5. Project to non-negativity (clip negatives)
6. Renormalize
7. Repeat

This is similar to the iteratively reweighted least squares (IRLS) or the "projection onto convex sets" (POCS) approach.

Actually, this is very similar to the algorithm for designing "flat-top" windows! The key difference is that we require non-negativity.

```python
def spectral_flatten(a, n_iters=100, clip_factor=2.0):
    """Flatten the periodogram to reduce autocorrelation peak."""
    a = a.copy()
    n = len(a)
    N = _next_pow2(2*n)
    
    for i in range(n_iters):
        # Compute DFT
        A = np.fft.rfft(a, N)
        P = np.abs(A)**2
        
        # Flatten: where P is above threshold, reduce it
        P_target = P.mean() * clip_factor
        P_clipped = np.minimum(P, P_target)
        
        # Update amplitude (keep phase)
        A_new = A * np.sqrt(P_clipped / P)
        
        # Inverse DFT
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project to non-negativity
        a_new = np.maximum(a_new, 0)
        
        # Renormalize
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        a = a_new
    
    return a
```

"""
Multi-resolution optimizer with adaptive L-BFGS, spectral flattening,
Nesterov active-set polish, and frequency-domain ILS for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Key improvements over previous:
- Multi-resolution warm start (coarse-to-fine)
- Adaptive beta annealing with convergence detection
- Spectral flattening as an alternative optimization pathway
- Frequency-domain perturbations in ILS for better diversity
- Better step size control in gradient methods
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_unnorm(a):
    """Score assuming sum(a)=1 (faster, no division)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos2":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "kaiser":
        t = np.linspace(-1, 1, n)
        beta_k = 4.0
        from scipy.special import i0
        a = i0(beta_k * np.sqrt(np.maximum(1 - t**2, 0))) / i0(beta_k)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _downsample(a, factor):
    n = len(a)
    new_n = n // factor
    if new_n < 4:
        return a.copy()
    a_padded = np.zeros(n + (factor - n % factor) % factor)
    a_padded[:n] = a
    a_blocks = a_padded[:new_n * factor].reshape(new_n, factor)
    return a_blocks.mean(axis=1)


def _upsample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _active_set_gradient(a, c, tol_frac):
    n = len(a)
    c_max = float(c.max())
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    n_active = int(active.sum())
    if n_active == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        n_active = 1
    eps = 1e-14 * c_max
    w = np.zeros_like(c)
    w[active] = 1.0 / (c_max - c[active] + eps)
    w = w / w.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad = 2.0 * h
    grad = grad - np.mean(grad)
    return grad


def _nesterov_polish(a_init, deadline, tol_start=0.008, tol_end=0.00002,
                     lr_init=0.04, momentum=0.85):
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score_unnorm(a)
    v = np.zeros(n)
    tol_frac = tol_start
    lr = lr_init
    no_improve = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        grad = _active_set_gradient(a, c, tol_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        v = momentum * v - lr * grad
        alpha = 1.0
        improved = False
        for _ in range(40):
            a_try = _normalize(a + alpha * v)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            v = np.zeros(n)
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_end, tol_frac * 0.65)
                lr *= 0.7
            if no_improve >= 200:
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(tol_end, tol_frac * 0.3)
                lr = lr_init * 0.5
                no_improve = 0
        else:
            if tol_frac > tol_end:
                tol_frac = max(tol_end, tol_frac * 0.98)
            lr = min(lr_init * 1.5, lr * 1.005)

    return best_a, _score(best_a)


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self._beta = beta
        self._n = n

    def f(self, z):
        return _smoobj(z, self._beta, self._n)

    def g(self, z):
        return _smograd(z, self._beta, self._n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.0005:
        gap = max_c * 0.0005
    base_beta = max(gap, max_c * 0.003)

    betas = base_beta * np.array([1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1,
                                  0.06, 0.04, 0.025, 0.015, 0.008,
                                  0.005, 0.003, 0.0015, 0.0008,
                                  0.0004, 0.0002, 0.0001, 0.00005])

    current_a = a_init.copy()
    current_sc = _score_unnorm(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(30, int(remaining / (est_time * 20))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob.f, z, jac=prob.g, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-16,
                         'maxcor': 150, 'maxls': 60}
            )
            a_new = _softmax(result.x)
            sc_new = _score_unnorm(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _spectral_flatten(a, n_iters=50, clip_ratio=1.5, deadline=None):
    """Iteratively flatten the periodogram to reduce autocorrelation peak."""
    n = len(a)
    a = a.copy()
    N = _next_pow2(2 * n)

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        threshold = P_mean * clip_ratio

        mask = P > threshold
        if not np.any(mask):
            break

        P_new = P.copy()
        P_new[mask] = threshold
        ratio = np.sqrt(P_new / (P + 1e-30))
        A_new = A * ratio

        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new

    return a


def _quick_optimize(a_init, deadline, min_time_remaining=20.0):
    a = _run_lbfgs(a_init, deadline, min_time_remaining=min_time_remaining + 10)
    if time.monotonic() < deadline - min_time_remaining:
        pol_deadline = min(deadline - min_time_remaining, time.monotonic() + 12)
        a2, sc2 = _nesterov_polish(a, pol_deadline,
                                   tol_start=0.004, tol_end=0.0003,
                                   lr_init=0.015, momentum=0.8)
        if sc2 < _score(a):
            a = a2
    return a


def _freq_perturb(a, rng, strength=0.1):
    """Perturb in frequency domain for diverse exploration."""
    n = len(a)
    N = _next_pow2(2 * n)
    A = np.fft.rfft(a, N)
    # Add random phase and amplitude perturbation
    phase_pert = strength * rng.standard_normal(len(A))
    amp_pert = 1.0 + strength * 0.5 * rng.standard_normal(len(A))
    A = A * amp_pert * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _stochastic_ils(a_init, deadline, rng, min_time_remaining=60.0):
    n = len(a_init)
    current_a = _normalize(a_init.copy())
    current_sc = _score_unnorm(current_a)
    best_a = current_a.copy()
    best_sc = current_sc

    strategies = [
        ('global', 0.004),
        ('global', 0.015),
        ('global', 0.04),
        ('block', 0.05),
        ('sparse', 0.1),
        ('mirror', 0.02),
        ('freq', 0.08),
        ('freq', 0.15),
        ('global', 0.002),
        ('block', 0.08),
        ('freq', 0.05),
        ('sparse', 0.05),
    ]

    attempt = 0
    while time.monotonic() + min_time_remaining < deadline:
        kind, sigma = strategies[attempt % len(strategies)]

        if kind in ('global',):
            z = np.log(np.maximum(current_a, 1e-30))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_perturbed = _softmax(z)
        elif kind == 'block':
            block_size = max(30, n // 6)
            start = int(rng.integers(0, max(1, n - block_size)))
            a_perturbed = current_a.copy()
            z = np.log(np.maximum(a_perturbed[start:start + block_size], 1e-30))
            z = z + sigma * rng.standard_normal(block_size)
            a_perturbed[start:start + block_size] = np.exp(z - np.max(z))
            a_perturbed = _normalize(a_perturbed)
        elif kind == 'sparse':
            n_perturb = max(20, n // 12)
            idx = rng.choice(n, size=n_perturb, replace=False)
            a_perturbed = current_a.copy()
            factors = np.exp(sigma * rng.standard_normal(n_perturb))
            a_perturbed[idx] *= factors
            a_perturbed = _normalize(a_perturbed)
        elif kind == 'mirror':
            seg_len = max(20, n // 15)
            start = int(rng.integers(0, max(1, n - seg_len)))
            a_perturbed = current_a.copy()
            seg = a_perturbed[start:start + seg_len].copy()
            a_perturbed[start:start + seg_len] = seg[::-1]
            z = np.log(np.maximum(a_perturbed, 1e-30))
            z = z + 0.003 * rng.standard_normal(n)
            z = z - z.mean()
            a_perturbed = _softmax(z)
        elif kind == 'freq':
            a_perturbed = _freq_perturb(current_a, rng, strength=sigma)
        else:
            a_perturbed = current_a.copy()

        opt_deadline = min(deadline - min_time_remaining, time.monotonic() + 10)
        a_opt = _quick_optimize(a_perturbed, opt_deadline, min_time_remaining=3)
        sc_opt = _score_unnorm(a_opt)

        if sc_opt < best_sc - 1e-13:
            best_sc = sc_opt
            best_a = a_opt.copy()
            current_a = a_opt.copy()
            print(f"      ILS BEST: {2*n*best_sc/best_sc:.6f} (sc={best_sc:.10f}) [{kind},{sigma:.3f}]")
        elif sc_opt < current_sc - 1e-13:
            current_a = a_opt.copy()

        attempt += 1

    return best_a, _score(best_a)

            best_score = sc_after
            best_seq = a_opt.copy()

    # === Phase 2: Spectral flattening + polish (150s) ===
    print(f"\n[P2] Spectral flattening (t={time.monotonic()-t_start:.1f}s)")
    sf_deadline = min(t_start + 400, deadline - 800)
    if time.monotonic() < sf_deadline:
        a_sf = _spectral_flatten(best_seq, n_iters=80, clip_ratio=1.3,
                                 deadline=sf_deadline - 30)
        sc_sf = _score(a_sf)
        print(f"  Spectral: {sc_sf:.10f}")
        if sc_sf < best_score:
            best_seq = a_sf.copy()
            best_score = sc_sf
        # Polish the spectral result
        if time.monotonic() < sf_deadline - 5:
            a_sf_pol, sc_sf_pol = _nesterov_polish(a_sf, sf_deadline - 5,
                                                    tol_start=0.003, tol_end=0.0001,
                                                    lr_init=0.01, momentum=0.8)
            if sc_sf_pol < best_score:
                best_seq = a_sf_pol.copy()
                best_score = sc_sf_pol
                print(f"  Spectral+polish: {sc_sf_pol:.10f}")

    # === Phase 3: Intensive L-BFGS (300s) ===
    print(f"\n[P3] Intensive L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    lbfgs_deadline = min(t_start + 700, deadline - 500)
    if time.monotonic() < lbfgs_deadline:
        a_main = _run_lbfgs(best_seq, lbfgs_deadline, min_time_remaining=60)
        sc = _score(a_main)
        if sc < best_score:
            best_seq = a_main.copy()
            best_score = sc
            print(f"  L-BFGS: {sc:.10f}")

    # === Phase 4: Nesterov polish (120s) ===
    print(f"\n[P4] Nesterov polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        pol_deadline = min(deadline - 350, time.monotonic() + 120)
        a_pol, sc_pol = _nesterov_polish(best_seq, pol_deadline,
                                          tol_start=0.002, tol_end=0.00001,
                                          lr_init=0.015, momentum=0.85)
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  Polish: {sc_pol:.10f}")

    # === Phase 5: ILS with spectral perturbations (350s) ===
    print(f"\n[P5] ILS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        ils_deadline = min(deadline - 250, time.monotonic() + 350)
        a_ils, sc_ils = _stochastic_ils(best_seq, ils_deadline, rng,
                                        min_time_remaining=80)
        if sc_ils < best_score:
            best_seq = a_ils.copy()
            best_score = sc_ils
            print(f"  ILS: {sc_ils:.10f}")

    # === Phase 6: Multi-n exploration (200s) ===
    print(f"\n[P6] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([8192, 16384, 24576, 32768, 40960, 49152]),
                          key=lambda x: abs(x - n_main))
    if time.monotonic() < deadline - 130:
        for n_try in n_candidates[:4]:
            if time.monotonic() > deadline - 130:
                break
            a_try = _upsample(best_seq, n_try)
            sc_try_init = _score(a_try)
            opt_dl = min(deadline - 130, time.monotonic() + 45)
            a_opt = _quick_optimize(a_try, opt_dl, min_time_remaining=5)
            sc_opt = _score(a_opt)
            print(f"  n={n_try}: {sc_try_init:.8f} -> {sc_opt:.8f}")
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {sc_opt:.10f} (n={n_try})")

    # === Phase 7: Second ILS round (100s) ===
    print(f"\n[P7] ILS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        rng2 = np.random.default_rng(int(seed) + 7777)
        ils2_dl = min(deadline - 80, time.monotonic() + 100)
        a_ils2, sc_ils2 = _stochastic_ils(best_seq, ils2_dl, rng2,
                                          min_time_remaining=40)
        if sc_ils2 < best_score:
            best_seq = a_ils2.copy()
            best_score = sc_ils2
            print(f"  ILS2: {sc_ils2:.10f}")

    # === Phase 8: Final ultra-fine polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _nesterov_polish(best_seq, deadline - 3,
                                          tol_start=0.0005, tol_end=0.000002,
                                          lr_init=0.004, momentum=0.75)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  Final polish: {sc_fin:.10f}")

        # Final L-BFGS with tiny beta
        remaining = deadline - 3 - time.monotonic()
        if remaining > 8:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.00002
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _LbfgsProb(beta_fin, n)
            max_iter = min(5000, max(30, int(remaining / 0.02)))
            try:
                result = minimize(prob.f, z, jac=prob.g, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-16, 'maxcor': 200})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  Final L-BFGS: {sc_fin2:.10f}")
            except Exception:
                pass

    # Output
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
