def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Buitinolph's algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Multi-scale adaptive smoothing optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:
- Multi-scale coarse-to-fine initialization
- FISTA (accelerated gradient) with backtracking line search in a-space
- Adaptive beta/K scheduling based on peak gap dynamics
- Simplex projection for constraint handling
- Direct peak shaving phase with gradient projection
- Diversified multi-restart with spectral awareness

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _smooth_obj_grad_a(a, beta, K):
    """Smoothed objective and gradient in a-space (a on simplex)."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c

    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()

    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))

    ga = _grad_maxconv(a, w)
    ga_centered = ga - ga.mean()
    return obj, ga_centered


def _fista_smooth(a_init, beta, K, deadline, min_time_remaining=30.0, verbose=False):
    """FISTA with backtracking for smoothed objective on simplex."""
    n = len(a_init)
    a = a_init.copy()
    y = a.copy()
    t_k = 1.0
    lr = 1.0 / (2.0 * n * 0.01)
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0
    max_no_improve = 200

    for it in range(100000):
        if time.monotonic() + min_time_remaining > deadline:
            break

        obj_y, grad_y = _smooth_obj_grad_a(y, beta, K)
        a_new = _proj_simplex(y - lr * grad_y)

        # Backtracking line search
        max_bt = 20
        for bt in range(max_bt):
            obj_new, _ = _smooth_obj_grad_a(a_new, beta, K)
            if obj_new <= obj_y + 1e-12 * np.linalg.norm(a_new - y):
                break
            lr *= 0.5
            a_new = _proj_simplex(y - lr * grad_y)
        else:
            lr *= 0.5
            a_new = _proj_simplex(y - lr * grad_y)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = _proj_simplex(y)
        a = a_new
        t_k = t_new

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

    return best_a


def _lbfgs_smooth(a_init, beta, K, n, deadline, min_time_remaining=10.0):
    """L-BFGS on log-domain for smoothed objective."""
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    def fun(z_in):
        a_in = np.exp(z_in - z_in.max())
        a_in = a_in / a_in.sum()
        return _smooth_obj_grad_a(a_in, beta, K)[0]

    def grad(z_in):
        a_in = np.exp(z_in - z_in.max())
        a_in = a_in / a_in.sum()
        _, ga = _smooth_obj_grad_a(a_in, beta, K)
        W = float(np.dot(a_in, ga))
        return a_in * (ga - W)

    try:
        result = minimize(fun, z, jac=grad, method='L-BFGS-B',
                          options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-14,
                                   'maxcor': 50, 'maxls': 30})
        a_new = np.exp(result.x - result.x.max())
        a_new = a_new / a_new.sum()
        return a_new
    except Exception:
        return a


def _adapt_schedule(a, n):
    """Adaptively choose beta and K based on peak structure."""
    c = _full_convolve(a, a)
    c_sorted = np.sort(c)[::-1]
    c_max = c_sorted[0]
    if c_max < 1e-15:
        return 0.01, 10

    # Gap between top peaks
    gap_10 = c_max - c_sorted[min(10, len(c_sorted) - 1)]
    gap_50 = c_max - c_sorted[min(50, len(c_sorted) - 1)]

    # If peaks are very close, use larger K and smaller beta
    if gap_10 < c_max * 0.005:
        beta = c_max * 0.002
        K = min(50, n)
    elif gap_10 < c_max * 0.02:
        beta = c_max * 0.005
        K = min(20, n)
    elif gap_50 < c_max * 0.05:
        beta = c_max * 0.01
        K = min(10, n)
    else:
        beta = c_max * 0.02
        K = min(5, n)

    return max(beta, 1e-10), K


def _run_optimization_phase(a_init, n, deadline, min_time_remaining=60.0):
    """Run adaptive optimization with decreasing beta."""
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    # Determine initial beta and K
    c = _full_convolve(a, a)
    c_max = float(c.max())
    beta_start = max(c_max * 0.05, 1e-6)

    # Schedule of beta multipliers (geometric decrease)
    beta_mults = [1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]

    for bm in beta_mults:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        beta = beta_start * bm
        # Adaptive K
        _, K = _adapt_schedule(a, n)
        K = max(3, K)

        # FISTA phase
        a_f = _fista_smooth(a, beta, K, deadline, min_time_remaining=min_time_remaining - 5)
        sc_f = _score(a_f)
        if sc_f < best_sc - 1e-14:
            best_sc = sc_f
            best_a = a_f.copy()
            a = a_f.copy()

        # L-BFGS phase (shorter)
        if time.monotonic() + min_time_remaining + 10 < deadline:
            a_l = _lbfgs_smooth(a, beta, K, n,
                                min(deadline - min_time_remaining - 5, time.monotonic() + 30),
                                min_time_remaining=5)
            sc_l = _score(a_l)
            if sc_l < best_sc - 1e-14:
                best_sc = sc_l
                best_a = a_l.copy()
                a = a_l.copy()

    return best_a


def _peak_shave(a_init, deadline, min_time_remaining=30.0):
    """Directly shave the top peaks using gradient projection."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.01
    no_improve = 0

    for it in range(50000):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break

        # Active set: peaks within 2% of max
        tol = c_max * 0.02
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        # Weighted gradient: higher weight for taller peaks
        w = np.zeros(2 * n - 1)
        active_vals = c[active]
        w[active] = 1.0 + (active_vals - active_vals.min()) / max(active_vals.max() - active_vals.min(), 1e-15)
        w = w / w.sum()

        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-16:
            break

        # Step with projection
        alpha = min(lr, 1.0 / (gnorm + 1e-10))
        improved = False
        for ls in range(30):
            a_try = _proj_simplex(a - alpha * g_centered)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            lr *= 0.7
            tol = max(c_max * 0.005, tol * 0.8)
        else:
            lr = min(0.05, lr * 1.02)
            tol = max(c_max * 0.005, tol * 0.95)

    return best_a


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-25.0 * (t - 0.3) ** 2) + np.exp(-25.0 * (t - 0.7) ** 2) + 0.03
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_cos":
        t = np.linspace(0, 1, n)
        a = 0.8 + 0.2 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

        if n_active == 0:

        g = _grad_maxconv(a, w)

            break

    if time.monotonic() < deadline - 50:
        n_options = [n_main * 2, n_main // 2]
        n_options = [x for x in n_options if x >= 4096 and x <= 65536]
        for n_try in n_options[:1]:
            if time.monotonic() > deadline - 20:
                break
            a_res = _resample(best_seq, n_try)
            a_opt = _fista_smooth(a_res, best_score / (2 * n_try) * 0.01, 10,
                                  min(deadline - 10, time.monotonic() + 25),
                                  min_time_remaining=3)
            a_opt = _peak_shave(a_opt, min(deadline - 5, time.monotonic() + 10),
                                min_time_remaining=2)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [n={n_try}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3:
        a_fin = _fine_polish(best_seq, deadline - 1, min_time_remaining=1)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
