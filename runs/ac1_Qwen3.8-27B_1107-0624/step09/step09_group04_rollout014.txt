def _kkt_weights(a, S):
    """Compute KKT-optimal lambda weights for active set S."""
    n = len(a)
    K = len(S)
    if K <= 1:
        return np.ones(max(K,1)) / max(K,1)
    
    # We need: sum_k lambda_k * a[k-j] = mu for all j = 0,...,n-1
    # and sum_k lambda_k = 1
    # Matrix M[j, k_idx] = a[S[k_idx] - j] if 0 <= S[k_idx]-j < n, else 0
    
    # Use FFT-based approach for efficiency
    # M @ lambda = sum_k lambda_k * a[S[k_idx] - j] for each j
    # This is: for each k in S, the vector [a[k], a[k-1], ..., a[k-n+1]] (padded with 0s)
    # which is a rotated and truncated version of a
    
    # Build M efficiently using the structure
    # For k < n: row j of column k_idx is a[k-j] for j=0..k, else 0
    # For k >= n: row j of column k_idx is a[k-j] for j=k-n+1..n-1, else 0
    
    M = np.zeros((n, K))
    for idx, k in enumerate(S):
        if 0 <= k < 2*n - 1:
            # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                a_vals = a[k - j_end:k - j_start + 1][::-1]
                M[j_start:j_end+1, idx] = a_vals
    
    # Solve: [M | -1] @ [lambda; mu] = 0, [1 ... 1 | 0] @ [lambda; mu] = 1
    A_mat = np.column_stack([M, -np.ones(n)])
    b_vec = np.zeros(n)
    # Add normalization constraint
    A_full = np.vstack([A_mat, np.append(np.ones(K), 0.0)])
    b_full = np.append(b_vec, 1.0)
    
    sol, residuals, rank, sv = np.linalg.lstsq(A_full, b_full, rcond=None)
    lam = np.maximum(sol[:K], 0.0)
    s = lam.sum()
    if s < 1e-15:
        return np.ones(K) / K
    return lam / s

"""
Optimized step function search using KKT-based active-set equalization,
adaptive log-sum-exp smoothing with L-BFGS, and multi-scale refinement.
Key innovation: computes optimal Lagrange multiplier weights for the active set
via the KKT linear system, yielding better descent directions than uniform weights.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _kkt_weights(a, S):
    """Compute KKT-optimal lambda weights for active set S.
    Solves: sum_k lambda_k * a[S[k]-j] = mu for all j, sum(lambda)=1.
    """
    n = len(a)
    K = len(S)
    if K <= 1:
        return np.ones(max(K, 1)) / max(K, 1)

    # Build M matrix: M[j, idx] = a[S[idx] - j] if 0 <= S[idx]-j < n
    S_arr = np.asarray(S, dtype=int)
    M = np.zeros((n, K))
    for idx in range(K):
        k = S_arr[idx]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # a[k-j] for j = j_start..j_end
            # = a[k-j_end : k-j_start+1][::-1]
            M[j_start:j_end + 1, idx] = a[k - j_end:k - j_start + 1][::-1]

    # Solve augmented system: [M | -1; 1 1 ... 1 | 0] [lambda; mu] = [0; 1]
    A_mat = np.column_stack([M, -np.ones(n)])
    b_vec = np.zeros(n)
    A_full = np.vstack([A_mat, np.append(np.ones(K), 0.0)])
    b_full = np.append(b_vec, 1.0)

    sol, _, _, _ = np.linalg.lstsq(A_full, b_full, rcond=None)
    lam = np.maximum(sol[:K], 0.0)
    s = lam.sum()
    if s < 1e-15:
        return np.ones(K) / K
    return lam / s


def _active_set_step(a, deadline, min_time_remaining=30.0, K_max=60, tol_frac=1e-4):
    """Active-set equalization with KKT-optimal weights and backtracking line search."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    lr = 0.03
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 300:
                break

        # Active set: peaks within tol_frac of max
        threshold = c_max * (1.0 - tol_frac)
        active = np.where(c >= threshold)[0]
        # Cap the active set size
        if len(active) > K_max:
            # Keep the top K_max
            active = active[np.argsort(c[active])[::-1][:K_max]]
        K = len(active)

        if K == 0:
            break

        # Compute KKT weights
        lam = _kkt_weights(a, active)

        # Compute weighted gradient: grad[j] = 2 * sum_k lam[k] * a[active[k]-j]
        w_sig = np.zeros(2 * n - 1)
        w_sig[active] = lam
        a_rev = a[::-1]
        h = _full_convolve(w_sig, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        mu = float(np.dot(grad, a))
        grad = grad - mu  # Project onto sum-zero direction

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Backtracking line search in z-space
        z = np.log(np.maximum(a, 1e-30))
        z = z - z.mean()
        grad_z = a * (grad - mu)  # Gradient in z-space (already sum-zero direction)

        improved = False
        alpha = lr / (np.linalg.norm(grad_z) + 1e-15)
        for _ in range(60):
            z_try = z - alpha * grad_z
            a_try = _softmax(z_try)
            a_try = np.maximum(a_try, 1e-30)
            a_try = a_try / a_try.sum()
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                z = np.log(np.maximum(a, 1e-30))
                z = z - z.mean()
                improved = True
                break
            alpha *= 0.5

        if not improved:
            lr *= 0.9
            if no_improve > 100:
                a = best_a.copy()
                lr = 0.03
                no_improve = 0
                if no_improve > 250:
                    break
        else:
            lr = min(0.06, lr * 1.02)

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=50.0, beta_scale=1.0):
    """L-BFGS optimization in log-space with geometric beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003) * beta_scale

    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.12, 0.06,
        0.03, 0.015, 0.008, 0.004,
        0.002, 0.001, 0.0005, 0.0002, 0.0001,
        0.00005, 0.00002, 0.00001
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 8:
            break

        # Adaptive iteration count
        if beta < base_beta * 0.01:
            est_time = 0.005 * max(1.0, n / 16384.0)
            max_iter = min(8000, max(100, int(remaining / (est_time * 20))))
        else:
            est_time = 0.004 * max(1.0, n / 16384.0)
            max_iter = min(5000, max(50, int(remaining / (est_time * 15))))

        def obj_fn(zv):
            return _smoobj(zv, beta, n)

        def grad_fn(zv):
            return _smograd(zv, beta, n)

        try:
            result = minimize(
                obj_fn, z, jac=grad_fn, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgradient_polish(a_init, deadline, min_time_remaining=15.0):
    """Final subgradient polish with diminishing step sizes."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    eta = 0.02
    it = 0
    max_iters = 8000

    for it in range(max_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()

        # Active set
        threshold = c_max * (1.0 - 1e-5)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        if len(active) > 30:
            active = active[np.argsort(c[active])[::-1][:30]]

        # KKT weights
        lam = _kkt_weights(a, active)
        w_sig = np.zeros(2 * n - 1)
        w_sig[active] = lam
        a_rev = a[::-1]
        h = _full_convolve(w_sig, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h - float(np.dot(2.0 * h, a))

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Diminishing step size
        eta = min(0.02, 0.5 / (gnorm + 1e-15) * 0.1)
        eta = 0.02 / (1.0 + it * 0.002)

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        # Check every 3 iterations
        if (it + 1) % 3 == 0:
            c_new = _full_convolve(a_new, a_new)
            sc_new = 2.0 * n * float(c_new.max())
            if sc_new < current_sc:
                a = a_new
            # else keep old a (eta will shrink naturally)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * t
        return a / a.sum()
    elif kind == "exp_asym":
        t = np.linspace(0, 1, n)
        a = np.exp(1.5 * (t - 0.3))
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.6) ** 2) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _asymmetric_perturb(a, rng, strength=0.05):
    n = len(a)
    t = np.arange(n) / n
    ramp = 0.4 * (t - 0.5)
    noise = 0.6 * rng.standard_normal(n)
    perturbation = ramp + noise
    perturbation = perturbation - perturbation.mean()
    z = np.log(np.maximum(a, 1e-30))
    z = z + strength * perturbation
    z = z - z.mean()
    return _softmax(z)

            if time.monotonic() < deadline - 60:
                a_eqn, sc_eqn = _active_set_step(a_opt, deadline - 50, min_time_remaining=30.0, K_max=30)
                if sc_eqn < best_score:
                    best_seq = a_eqn.copy()
                    best_score = sc_eqn
                    print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_kkt]")

    # === Phase 8: Final KKT + subgradient polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        # One more KKT round
        pol_dl3 = min(deadline - 25, time.monotonic() + 60)
        a_fin, sc_fin = _active_set_step(best_seq, pol_dl3, min_time_remaining=20.0, K_max=50)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_kkt]")

    if time.monotonic() < deadline - 15:
        a_sg, sc_sg = _subgradient_polish(best_seq, deadline - 10, min_time_remaining=10.0)
        if sc_sg < best_score:
            best_seq = a_sg.copy()
            best_score = sc_sg
            print(f"  NEW BEST: {best_score:.10f} [subgrad]")

    # === Final micro-L-BFGS ===
    remaining = deadline - 5 - time.monotonic()
    if remaining > 15:
        a_last = _run_lbfgs(best_seq, deadline - 3, min_time_remaining=3.0, beta_scale=0.1)
        sc_last = _score(a_last)
        if sc_last < best_score:
            best_seq = a_last.copy()
            best_score = sc_last
            print(f"  NEW BEST: {best_score:.10f} [micro_lbfgs]")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
