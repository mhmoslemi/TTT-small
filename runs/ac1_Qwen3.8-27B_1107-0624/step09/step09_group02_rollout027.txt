"""
Focused continuation + active-set optimizer for minimum peak linear autocorrelation.
Uses aggressive L-BFGS log-sum-exp continuation with periodic random restarts,
adaptive active-set QP with FFT-based gradient, and multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def get_next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def do_full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = get_next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def do_score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = do_full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def do_softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def do_proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def do_gradient(a, p, n):
    """Compute grad of sum_k p[k]*R(k) w.r.t. a, where R(k) = (a*a)[k] linear conv.
    
    Key formula: grad[m] = 2 * sum_j a[j] * p[j+m]
    Computed via: grad = 2 * conv(a_rev, p)[n-1 : 2n-1]
    """
    a_rev = a[::-1]
    conv_len = len(a_rev) + len(p) - 1
    N = get_next_pow2(conv_len)
    Fa_rev = np.fft.rfft(a_rev, N)
    Fp = np.fft.rfft(p, N)
    conv = np.fft.irfft(Fa_rev * Fp, N)[:conv_len]
    h = conv[n - 1:2 * n - 1]
    return 2.0 * h


def run_lbfgs_cont(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with log-sum-exp continuation."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = do_score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_cur = do_full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    base_beta = max(gap * 0.03, max_c * 0.003)

    ratios = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
              0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001,
              0.000005, 0.000002, 0.000001, 0.0000005, 0.0000002, 0.0000001]

    for i, ratio in enumerate(ratios):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        beta = base_beta * ratio
        est_time = 0.005 * max(1.0, n / 8192.0)
        max_iter = min(8000, max(50, int(remaining / (est_time * 3))))

        try:
            result = minimize(
                _lbfgs_obj_wrapper, z,
                args=(n, beta),
                jac=_lbfgs_grad_wrapper,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxcor': 60, 'maxls': 100}
            )
            a_new = do_softmax(result.x)
            sc_new = do_score(a_new)
            if sc_new < best_sc - 1e-16:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            # Update beta reference if we made progress
            c_new = do_full_convolve(a_new, a_new)
            mc_new = float(c_new.max())
            if mc_new < max_c:
                max_c = mc_new
                base_beta = max(base_beta * 0.9, mc_new * 0.003)
        except Exception:
            pass

    return best_a


def _lbfgs_obj_wrapper(z_local, n, beta):
    a_local = do_softmax(z_local)
    c_local = do_full_convolve(a_local, a_local)
    c_max_local = float(c_local.max())
    shifted = (c_local - c_max_local) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s_e = e.sum()
    if s_e < 1e-300:
        return 2.0 * n * c_max_local
    return 2.0 * n * (c_max_local + beta * np.log(s_e))


def _lbfgs_grad_wrapper(z_local, n, beta):
    a_local = do_softmax(z_local)
    c_local = do_full_convolve(a_local, a_local)
    c_max_local = float(c_local.max())
    shifted = (c_local - c_max_local) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s_e = e.sum()
    if s_e < 1e-300:
        w = np.zeros_like(c_local)
        w[int(np.argmax(c_local))] = 1.0
    else:
        w = e / s_e
    # w = softmax(c_local / beta)
    # dF/da = 2n * 2 * (a_rev * w)[n-1:2n-1]
    a_rev = a_local[::-1]
    wc = do_full_convolve(a_rev, w)
    h = wc[n - 1:2 * n - 1]
    return 4.0 * n * h


def run_active_set_qp(a_init, deadline, min_time_remaining=30.0, max_rounds=200):
    """Active-set QP: minimize weighted sum of top-k autocorrelation lags."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = do_score(a)

    tol_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                    0.00005, 0.00002, 0.00001, 0.000005, 0.000002]
    tol_idx = 0
    rounds_per_tol = 8
    top_k_schedule = [20, 15, 10, 8, 6, 5, 4, 3, 3, 2, 2, 1]

    for rnd in range(max_rounds):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = do_full_convolve(a, a)
        c_max = float(c.max())
        top_k = top_k_schedule[min(tol_idx, len(top_k_schedule) - 1)]
        top_lags = np.argsort(c)[::-1][:top_k]
        # Adaptive weights: softmax with small temperature
        c_top = c[top_lags]
        temp = (c_top[0] - c_top[-1]) * 0.1 + 1e-15
        w_top = do_softmax((c_top - c_top[0]) / (-temp))
        p = np.zeros(2 * n - 1)
        p[top_lags] = w_top

        # Projected gradient with backtracking
        for inner in range(30):
            grad = do_gradient(a, p, n)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            lr = 0.5 / (gnorm * gnorm + 1e-10)
            for ls in range(20):
                a_new = do_proj_simplex(a - lr * grad)
                obj_new = float(np.dot(p, do_full_convolve(a_new, a_new)))
                obj_cur = float(np.dot(p, do_full_convolve(a, a_new)))  # linear approx
                if obj_new <= obj_cur + 1e-15 * gnorm:
                    a = a_new
                    break
                lr *= 0.5

        sc = do_score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
        elif rnd > 10:
            a = best_a.copy()

        if (rnd + 1) % rounds_per_tol == 0:
            tol_idx += 1

    return best_a, best_sc


def run_proj_grad_momentum(a_init, deadline, min_time_remaining=30.0):
    """Projected gradient with Nesterov momentum in log-space."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = do_score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    momentum = np.zeros(n)
    beta_m = 0.9
    patience = 0
    max_patience = 40

    for step in range(10000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        a_curr = do_softmax(z)
        c = do_full_convolve(a_curr, a_curr)
        c_max = float(c.max())
        k_active = min(8, len(c))
        top_k = np.argsort(c)[::-1][:k_active]
        p = np.zeros(2 * n - 1)
        c_top = c[top_k]
        temp = (c_top[0] - c_top[-1]) * 0.2 + 1e-15
        w_top = do_softmax((c_top - c_top[0]) / (-temp))
        p[top_k] = w_top

        grad = do_gradient(a_curr, p, n)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        momentum = beta_m * momentum + (1 - beta_m) * grad
        lr = 0.03 / (np.linalg.norm(momentum) / grad_norm + 1e-10)
        z_new = z - lr * momentum
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()
        z = z_new

        if step % 15 == 0:
            a_check = do_softmax(z)
            sc = do_score(a_check)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros(n)
                    patience = 0
                    beta_m = max(0.5, beta_m * 0.9)

    a_final = do_softmax(z)
    sc_final = do_score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def run_direct_flatten(a_init, deadline, min_time_remaining=30.0):
    """Direct flattening: iteratively push down peak autocorrelation."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = do_score(a)

    for it in range(500):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = do_full_convolve(a, a)
        c_max = float(c.max())
        # Focus on lags within 1% of max
        peaks = c >= c_max * 0.998
        if peaks.sum() == 0:
            peaks[int(np.argmax(c))] = True
        if peaks.sum() > 15:
            top_lags = np.argsort(c)[::-1][:15]
            peaks = np.zeros_like(c)
            peaks[top_lags] = True
        p = np.zeros(2 * n - 1)
        p[peaks] = 1.0 / peaks.sum()

        grad = do_gradient(a, p, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 0.1 / (gnorm + 1e-10)
        for ls in range(25):
            a_new = do_proj_simplex(a - step * grad)
            c_new = do_full_convolve(a_new, a_new)
            if float(c_new.max()) < c_max - 1e-16:
                a = a_new
                sc = do_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                break
            step *= 0.5
            if step < 1e-14:
                break

    return best_a, best_sc


def make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "skew_left":
        t = np.linspace(0, 1, n)
        a = (1 - t) ** 0.5 + 0.1
        return a / a.sum()
    elif kind == "skew_right":
        t = np.linspace(0, 1, n)
        a = t ** 0.5 + 0.1
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.35) ** 2) + 0.05
        return a / a.sum()
    elif kind == "bump2":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.65) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def do_resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def do_perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return do_softmax(z)

                update_best(a_p, sc_p, f"lbfgs_{prof}")

    # === Phase 2: Active-set QP (350s) ===
    print(f"\n[Phase 2] Active-Set QP (t={time.monotonic() - t_start:.1f}s)")
    phase2_deadline = min(deadline - 700, phase1_deadline + 350)
    if best_seq is not None and time.monotonic() < phase2_deadline:
        a_qp, sc_qp = run_active_set_qp(best_seq, phase2_deadline,
                                         min_time_remaining=20.0, max_rounds=150)
        update_best(a_qp, sc_qp, "active_qp_1")

    # === Phase 3: Projected gradient with momentum (250s) ===
    print(f"\n[Phase 3] Projected Gradient (t={time.monotonic() - t_start:.1f}s)")
    phase3_deadline = min(deadline - 500, phase2_deadline + 250)
    if best_seq is not None and time.monotonic() < phase3_deadline:
        a_pg, sc_pg = run_proj_grad_momentum(best_seq, phase3_deadline,
                                              min_time_remaining=15.0)
        update_best(a_pg, sc_pg, "proj_grad_1")

    # === Phase 4: Direct flattening (200s) ===
    print(f"\n[Phase 4] Direct Flatten (t={time.monotonic() - t_start:.1f}s)")
    phase4_deadline = min(deadline - 350, phase3_deadline + 200)
    if best_seq is not None and time.monotonic() < phase4_deadline:
        a_fl, sc_fl = run_direct_flatten(best_seq, phase4_deadline,
                                          min_time_remaining=15.0)
        update_best(a_fl, sc_fl, "flatten_1")

    # === Phase 5: Random restarts + re-optimization (300s) ===
    print(f"\n[Phase 5] Random Restarts (t={time.monotonic() - t_start:.1f}s)")
    phase5_deadline = min(deadline - 200, phase4_deadline + 300)
    restarts = 8
    restart_time = max(20.0, (phase5_deadline - time.monotonic()) / restarts - 10)
    for ri in range(restarts):
        if time.monotonic() > phase5_deadline - 20:
            break
        sigma = 0.02 * (1 + 0.5 * ri)
        a_pert = do_perturb(best_seq, sigma, rng)
        # Quick L-BFGS restart
        a_r = run_lbfgs_cont(a_pert, min(phase5_deadline - 10, time.monotonic() + restart_time),
                             min_time_remaining=5.0)
        sc_r = do_score(a_r)
        if sc_r < best_score:
            update_best(a_r, sc_r, f"restart_{ri}")
        if time.monotonic() > phase5_deadline - 30:
            break

    # === Phase 6: Second round of active-set + L-BFGS (200s) ===
    print(f"\n[Phase 6] Round 2 Polish (t={time.monotonic() - t_start:.1f}s)")
    phase6_deadline = min(deadline - 100, phase5_deadline + 200)
    if best_seq is not None and time.monotonic() < phase6_deadline:
        a_qp2, sc_qp2 = run_active_set_qp(best_seq, min(phase6_deadline, time.monotonic() + 100),
                                           min_time_remaining=10.0, max_rounds=80)
        update_best(a_qp2, sc_qp2, "active_qp_2")

        if time.monotonic() < phase6_deadline - 30:
            a_lb2 = run_lbfgs_cont(best_seq, phase6_deadline, min_time_remaining=10.0)
            update_best(a_lb2, do_score(a_lb2), "lbfgs_r2")

    # === Phase 7: Multi-n exploration (100s) ===
    print(f"\n[Phase 7] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    phase7_deadline = min(deadline - 50, phase6_deadline + 100)
    n_tries = [65536, 49152, 16384]
    for n_try in n_tries:
        if time.monotonic() > phase7_deadline - 30:
            break
        if n_try == n_parent:
            continue
        a_init_n = do_resample(best_seq, n_try)
        a_n = run_lbfgs_cont(a_init_n, min(phase7_deadline - 20, time.monotonic() + 50),
                             min_time_remaining=10.0)
        sc_n = do_score(a_n)
        update_best(a_n, sc_n, f"n_{n_try}")

    # === Phase 8: Final ultra-tight polish (50s) ===
    print(f"\n[Phase 8] Final Polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = run_active_set_qp(best_seq, deadline - 5,
                                           min_time_remaining=3.0, max_rounds=50)
        update_best(a_fin, sc_fin, "final_qp")

        if time.monotonic() < deadline - 8:
            a_fl2, sc_fl2 = run_direct_flatten(best_seq, deadline - 8,
                                                min_time_remaining=3.0)
            update_best(a_fl2, sc_fl2, "final_flatten")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = make_profile(4096, "tapered")
        sc_final = do_score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        # Verify with evaluate_sequence
        try:
            ev = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence confirms: {ev:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = make_profile(16384, "tapered")
    print(f"FALLBACK: score={do_score(fallback):.10f}")
    return [float(x) for x in fallback]
