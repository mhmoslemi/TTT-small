def _lp_step(a, n_active_max=100, epsilon=0.01):
    """One step of LP-based active set refinement.
    Returns (new_a, step_size, improved).
    """
    n = len(a)
    s = a.sum()
    a = a / s  # normalize to sum 1
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    current_score = 2.0 * n * c_max
    
    # Determine active set
    threshold = c_max * (1 - epsilon)
    active_indices = np.where(c >= threshold)[0]
    
    # Also always include the argmax
    argmax_idx = int(np.argmax(c))
    if argmax_idx not in active_indices:
        active_indices = np.append(active_indices, argmax_idx)
    
    # Cap the active set size
    if len(active_indices) > n_active_max:
        # Keep the top n_active_max
        active_c = c[active_indices]
        top_order = np.argsort(active_c)[::-1][:n_active_max]
        active_indices = active_indices[top_order]
    
    # Build LP
    # Variables: [da_0, ..., da_{n-1}, M]  (n+1 variables)
    n_vars = n + 1
    M_idx = n  # index of M variable
    
    # Objective: minimize M
    c_obj = np.zeros(n_vars)
    c_obj[M_idx] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # 1. For each k in active set: 2*sum_j G[k,j]*da[j] - M <= -c[k]
    # 2. For each j: -da[j] <= a[j]  (i.e., da[j] >= -a[j])
    
    n_cons = len(active_indices) + n
    
    # Build sparse A_ub
    rows = []
    cols = []
    vals = []
    
    # Active set constraints
    for i, k in enumerate(active_indices):
        k = int(k)
        # G[k,j] = a[k-j] if 0 <= k-j < n, else 0
        #         + a[j-k] if 0 <= j-k < n, else 0
        # For row i:
        #   Columns 0..n-1: 2*G[k,j]
        #   Column n (M): -1
        
        # 2*a[k-j] for j in [max(0, k-n+1), min(k, n-1)]
        j_lo1 = max(0, k - n + 1)
        j_hi1 = min(k, n - 1)
        for j in range(j_lo1, j_hi1 + 1):
            rows.append(i)
            cols.append(j)
            vals.append(2.0 * a[k - j])
        
        # 2*a[j-k] for j in [max(k, 0), min(k+n-1, n-1)]
        j_lo2 = max(k, 0)
        j_hi2 = min(k + n - 1, n - 1)
        for j in range(j_lo2, j_hi2 + 1):
            rows.append(i)
            cols.append(j)
            vals.append(2.0 * a[j - k])
        
        # M column
        rows.append(i)
        cols.append(M_idx)
        vals.append(-1.0)
    
    # Non-negativity constraints: -da[j] <= a[j]
    for j in range(n):
        row_idx = len(active_indices) + j
        rows.append(row_idx)
        cols.append(j)
        vals.append(-1.0)
    
    b_ub = np.concatenate([
        -c[active_indices],
        a
    ])
    
    A_ub = coo_matrix((vals, (rows, cols)), shape=(n_cons, n_vars))
    
    # Equality constraint: sum da[j] = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = [(-a[j], 1.0) for j in range(n)] + [(0.0, None)]
    
    # Solve
    result = linprog(c_obj, A_ub=csr_matrix(A_ub), b_ub=b_ub,
                     A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if not result.success:
        return a.copy(), 0.0, False
    
    da = result.x[:n]
    M_opt = result.x[n]
    
    # Line search
    best_alpha = 0.0
    best_sc = current_score
    for alpha in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
        a_try = a + alpha * da
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc:
            best_sc = sc_try
            best_alpha = alpha
    
    if best_alpha == 0.0:
        return a.copy(), 0.0, False
    
    a_new = a + best_alpha * da
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    a_new = a_new / s_new
    
    return a_new, best_alpha, best_sc < current_score

LP-based active set refinement combined with L-BFGS soft-max optimization.
Key innovation: Uses linear programming to find optimal first-order direction
at each step, rather than simple gradient descent.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix, coo_matrix, hstack, vstack, eye as sp_eye

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)

def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _grad_soft_structured(a, beta, n):
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)

def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft_structured(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)

        self.n = n
    def __call__(self, z):

    def grad(self, z):
        return _smograd(z, self.beta, self.n)

    k_arr = active_indices.astype(np.int64)

    mask1 = (diff1 >= 0) & (diff1 < n)
    idx1 = np.where(mask1, diff1, 0)
    part1 = a[idx1] * mask1
    
    diff2 = j_arr[None, :] - k_arr[:, None]
    mask2 = (diff2 >= 0) & (diff2 < n)
    idx2 = np.where(mask2, diff2, 0)
    part2 = a[idx2] * mask2
    
    G = part1 + part2  # (|S|, n)
    n_active = len(active_indices)
    
    # Build A_ub (sparse): (n_active + n, n + 1)
    # First n_active rows: [2*G[i,:], -1]
    # Next n rows: [-e_j, 0]
    
    # Use COO format for efficiency
    rows = np.zeros(n_active * n + n_active + n, dtype=np.int32)
    cols = np.zeros(n_active * n + n_active + n, dtype=np.int32)
    vals = np.zeros(n_active * n + n_active + n, dtype=np.float64)
    
    # Hmm, the active set rows each have at most n non-zeros (since G is dense).
    # But many entries of G might be zero. Let me check:
    # G[k,j] = a[k-j] + a[j-k]. If a is positive everywhere, G is dense.
    # If a has zeros, G has corresponding zeros.
    
    # For simplicity, let me just use dense G and convert to sparse.
    G_sparse = csr_matrix(2.0 * G)
    M_col = csr_matrix(-np.ones((n_active, 1)))
    A_active = hstack([G_sparse, M_col], format='csr')
    
    I_nn = csr_matrix(-np.eye(n))
    zero_col = csr_matrix(np.zeros((n, 1)))
    A_nn = hstack([I_nn, zero_col], format='csr')
    
    A_ub = vstack([A_active, A_nn], format='csr')
    b_ub = np.concatenate([
        -c[active_indices],
        a
    ])
    
    # A_eq: sum da = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    lower = np.concatenate([-a, [0.0]])
    upper = np.concatenate([np.full(n, 1.0), [np.inf]])
    bounds = list(zip(lower, upper))
    
    # Objective: minimize M
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    result = linprog(
        c_obj,
        A_ub=A_ub,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method='highs'
    )
    
    if not result.success:
        return None, 0.0, False
    
    da = result.x[:n]
    
    # Line search
    current_sc = 2.0 * n * float(c.max())
    best_alpha = 0.0
    best_sc_ls = current_sc
    
    for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
        a_try = a + alpha * da
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        c_try = _full_convolve(a_try, a_try)
        sc_try = 2.0 * n * float(c_try.max())
        if sc_try < best_sc_ls:
            best_sc_ls = sc_try
            best_alpha = alpha
    
    if best_alpha == 0.0:
        return None, 0.0, False
    
    a_new = a + best_alpha * da
    a_new = np.maximum(a_new, 0.0)
    a_new = a_new / a_new.sum()
    
    return a_new, best_alpha, best_sc_ls < current_sc

"""
LP-based active set refinement for step function optimization.
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using linear programming
to find optimal first-order directions at each step.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix, hstack, vstack

t_start_global = 0.0

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
