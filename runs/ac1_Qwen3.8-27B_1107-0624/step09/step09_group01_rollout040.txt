"""
Optimized hybrid solver for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses aggressive L-BFGS log-sum-exp annealing + KKT-Newton with adaptive active sets.
Explores multiple dimensions and starting points.

import numpy as np
import time
from scipy.optimize import minimize

# --- Helper functions ---

def _next_pow2(n):
    """Next power of 2 >= n."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    """Compute the objective value."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_grad_smoothed(a, w, n):
    """
    Compute grad_a of beta*log(sum exp(c_k/beta)) where c = conv(a,a).
    w = softmax(c/beta) of length 2n-1.
    Returns gradient of length n.
    """
    N = _next_pow2(3 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    
    # term2 = conv(w, a)[:n]
    term2 = np.fft.irfft(fw * fa, N)[:n]
    
    # term1 = conv(w_flip, a)[2n-2 : 2n-2+n] (i.e., positions 2n-2-j for j=0..n-1)
    w_flip = w[::-1]
    fw_flip = np.fft.rfft(w_flip, N)

    term1 = full[2*n-2 : n-1 : -1]
    
    return term1 + term2

            best_a = a.copy()

def _hankel_matvec(v, W, n):
    """Compute H*v where H is the Hessian of sum_k W[k]*c_k."""
    # H[v]_j = 2 * sum_k W[k] * (v[k-j] + v[j-k])
    # = 2 * (conv(W, v)[j] + conv(W_flip, v)[2n-2-j])
    # Actually same formula as gradient but with v instead of a, and factor of 2
    N = _next_pow2(3*n)
    fv = np.fft.rfft(v, N)
    fW = np.fft.rfft(W, N)
    
    term2 = np.fft.irfft(fW * fv, N)[:n]
    W_flip = W[::-1]
    fW_flip = np.fft.rfft(W_flip, N)
    full = np.fft.irfft(fW_flip * fv, N)
    term1 = full[2*n-2 : n-1 : -1]
    
    return 2.0 * (term1 + term2)

def _cg_solve(W, n, rhs, max_iter=200):
    """Solve H*x = rhs using CG, where H is the Hankel-type matrix."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < 1e-24 * rs0:
            break
    return x

def _newton_step(a, W, g, n, g_t):
    """Compute Newton step for KKT system."""
    # Solve: H * dx = -g + lambda * 1, with sum(dx) = 0
    # H = Hessian of sum W[k]*c_k
    
    # First, solve H * x = -g (unconstrained)
    x = _cg_solve(W, n, -g, max_iter=200)
    
    # Then solve H * y = 1 (to compute the correction)
    y = _cg_solve(W, n, np.ones(n), max_iter=200)
    
    # Apply constraint sum(dx) = 0:
    # dx = x + lambda * y where lambda = -sum(x) / sum(y)
    sy = float(np.sum(y))
    if abs(sy) < 1e-25 or not np.isfinite(sy):
        # Fallback
        return None
    
    lam = -float(np.sum(x)) / sy
    dx = x + lam * y
    
    # Check that dx is reasonable
    if not np.all(np.isfinite(dx)) or np.linalg.norm(dx) > 10.0:
        return None
    
    return dx

def _make_profile(n, kind="tapered"):
    """Create initial profile."""
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.6
    elif kind == "cos":
        a = 0.5 + 0.5 * np.cos(np.pi * t)
    elif kind == "bell":
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.01
    elif kind == "flat_top":
        a = np.ones(n)
        edge = max(1, n // 10)
        a[:edge] = np.linspace(0.1, 1, edge)
        a[-edge:] = np.linspace(1, 0.1, edge)
    else:
        a = np.sin(np.pi * t) ** 0.6
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _resample(a, n_new):
    """Resample sequence to new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _perturb(a, sigma, rng):
    """Perturb sequence in log space."""
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)
