import numpy as np

from scipy.optimize import minimize, minimize_scalar


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-300:
        return a
    return a / s


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _convolve_weighted(w, a):
    """Compute h[j] = sum_k w[k] * a[k-j] (the gradient building block)."""
    n = len(a)
    a_rev = a[::-1]
    h = _full_convolve(w, a_rev)
    return h[n - 1:2 * n - 1]


def _gradient_smoothed(a, w):
    """Gradient of the weighted sum of peaks: 2 * sum_k w[k] * a[k-j]."""
    return 2.0 * _convolve_weighted(w, a)

    no_improve = 0
    max_no_improve = 300

        current_sc = 2.0 * n * c_max

            no_improve = 0

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _spectral_init(n, tilt=0.0):
    """Initialize using a spectral approach: create a flat-ish power spectrum
    and invert to get a smooth non-negative sequence."""
    N = _next_pow2(2 * n - 1)
    # Target: flat power spectrum with smooth taper at edges
    freqs = np.fft.rfftfreq(N) * 2 * N  # in cycles
    # Taper: smooth window in frequency
    taper = np.ones(N // 2 + 1)
    edge = max(1, N // 20)
    taper[:edge] = np.linspace(0.3, 1.0, edge)
    taper[-edge:] = np.linspace(1.0, 0.3, edge)
    
    # Create a "phase" that gives a real, non-negative sequence
    # Use a linear phase (shift) + small random phase
    phase = np.zeros(N // 2 + 1)
    if tilt != 0.0:
        phase += tilt * np.arange(N // 2 + 1) / (N // 2) * np.pi
    
    # A(f) = sqrt(taper) * exp(i * phase)
    A = np.sqrt(taper) * np.exp(1j * phase)
    # Make it conjugate symmetric for real output
    a_full = np.fft.irfft(A * np.sqrt(N), N)
    # Extract the middle n samples (or shift to center)
    center = N // 2
    a_extracted = a_full[center - n // 2: center - n // 2 + n]
    a_extracted = np.maximum(a_extracted, 0.0)
    s = a_extracted.sum()
    if s > 1e-15:
        a_extracted = a_extracted / s
    else:
        a_extracted = np.ones(n) / n
    return a_extracted


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _multi_peak_equalize(a_init, deadline, min_time_remaining=30.0):
    """More sophisticated peak equalization:
    - Dynamically grow/shrink the active set
    - Use weighted peaks (weight inversely proportional to distance from max)
    - Exact line search per step
    """
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    K_start = 3
    K_max = min(100, n // 4)
    K = K_start
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            K = min(K_max, K + 1)
        
        # Active set: peaks within 0.1% of max, plus top K
        threshold = c_max * (1 - 0.001)
        active_cand = np.where(c >= threshold)[0]
        top_K = np.argsort(c)[::-1][:K]
        active = np.unique(np.concatenate([active_cand, top_K]))
        active = active[:min(len(active), K_max)]
        
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        K_actual = len(active)
        
        # Weights: emphasize the highest peaks
        active_vals = c[active]
        # Weight = exp(-gamma * (c_max - c[active]) / c_max)
        gamma = 5.0
        weights = np.exp(-gamma * (c_max - active_vals) / max(c_max, 1e-30))
        weights = weights / weights.sum()
        
        w_S = np.zeros(2 * n - 1)
        w_S[active] = weights
        
        # Gradient
        grad = _gradient_smoothed(a, w_S)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        d = -grad / gnorm
        
        # Find max step
        d_neg = d < 0
        if d_neg.any():
            alpha_max = np.min(-a[d_neg] / d[d_neg])
        else:
            alpha_max = 1.0
        alpha_max = min(alpha_max, 1.5)
        
        if alpha_max < 1e-12:
            no_improve += 1
            if no_improve > 100:
                K = max(K_start, K // 2)
                no_improve = 0
            if no_improve > 500:
                break
            continue
        
        # Line search
        def f_1d(alpha):
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                return np.inf
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            return float(c_try.max())
        
        try:
            res = minimize_scalar(f_1d, bounds=(0.0, alpha_max), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 80})
            alpha_opt = res.x
        except Exception:
            alpha_opt = alpha_max * 0.5
        
        a_new = a + alpha_opt * d
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())
        
        if sc_new < current_sc - 1e-14:
            a = a_new
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                K = max(K_start, K // 2)
            if no_improve > 500:
                break
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase optimizer exploiting asymmetry:
    1. Load parent, try multiple n values
    2. For each n: L-BFGS warmup -> tilt optimization -> peak equalization -> fine L-BFGS
    3. Pick best and do final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, init n={n_parent}, score={best_score:.10f}")
    
    # Determine n values to try
    n_values = sorted(set([
        n_parent,
        max(4096, n_parent // 2),
        n_parent * 2,
        12288,
        20480,
        24576,
    ]), key=lambda x: -abs(x - n_parent))
    # Cap n to avoid too-large sequences
    n_values = [n for n in n_values if 4096 <= n <= 49152]
    # Ensure n_parent is first
    n_values = [n_parent] + [n for n in n_values if n != n_parent]
    
    print(f"\nTrying n values: {n_values}")
    
    for n_try in n_values:
        if time.monotonic() > deadline - 120:
            break
        
        print(f"\n{'='*50}")
        print(f"Optimizing n={n_try} (time={time.monotonic()-t_start:.1f}s)")
        
        # Get initial sequence for this n
        if n_try == len(best_seq):
            a_init = best_seq.copy()
        else:
            a_init = _resample(best_seq, n_try)
        
        sc_init = _score(a_init)
        print(f"  Init score: {sc_init:.10f}")
        
        # Also try spectral init
        a_spectral = _spectral_init(n_try, tilt=0.2)
        sc_spectral = _score(a_spectral)
        if sc_spectral < sc_init:
            a_init = a_spectral
            sc_init = sc_spectral
            print(f"  Spectral init better: {sc_init:.10f}")
        
        # Phase 1: L-BFGS warmup (moderate beta)
        a_p1 = _run_lbfgs_smoothed(a_init, deadline, min_time_remaining=200.0,
                                    beta_scale=1.0, max_beta_iters=2500)
        sc_p1 = _score(a_p1)
        print(f"  After L-BFGS warmup: {sc_p1:.10f}")
        if sc_p1 < best_score:
            best_score = sc_p1
            best_seq = a_p1.copy()
        
        # Phase 2: Tilt optimization
        if time.monotonic() < deadline - 150:
            a_tilt, sc_tilt = _tilt_optimization(a_p1, deadline, min_time_remaining=100.0)
            print(f"  After tilt opt: {sc_tilt:.10f}")
            if sc_tilt < best_score:
                best_score = sc_tilt
                best_seq = a_tilt.copy()
            a_current = a_tilt
        else:
            a_current = a_p1
        
        # Phase 3: Multi-peak equalization
        if time.monotonic() < deadline - 100:
            pe_deadline = min(deadline - 80, time.monotonic() + 120)
            a_eq, sc_eq = _multi_peak_equalize(a_current, pe_deadline, min_time_remaining=60.0)
            print(f"  After peak equalize: {sc_eq:.10f}")
            if sc_eq < best_score:
                best_score = sc_eq
                best_seq = a_eq.copy()
            a_current = a_eq
        
        # Phase 4: Fine L-BFGS (small beta)
        if time.monotonic() < deadline - 60:
            a_fine = _run_lbfgs_smoothed(a_current, deadline, min_time_remaining=40.0,
                                          beta_scale=0.3, max_beta_iters=2000)
            sc_fine = _score(a_fine)
            print(f"  After fine L-BFGS: {sc_fine:.10f}")
            if sc_fine < best_score:
                best_score = sc_fine
                best_seq = a_fine.copy()
            a_current = a_fine
        
        # Phase 5: Final peak equalization
        if time.monotonic() < deadline - 30:
            eq_deadline = min(deadline - 20, time.monotonic() + 50)
            a_eq2, sc_eq2 = _multi_peak_equalize(a_current, eq_deadline, min_time_remaining=15.0)
            print(f"  After final equalize: {sc_eq2:.10f}")
            if sc_eq2 < best_score:
                best_score = sc_eq2
                best_seq = a_eq2.copy()
    
    # Final polish on the best sequence found
    print(f"\n{'='*50}")
    print(f"Final polish on best (n={len(best_seq)}, score={best_score:.10f})")
    
    if time.monotonic() < deadline - 10:
        # One more round of peak equalization with more patience
        a_final, sc_final = _multi_peak_equalize(best_seq, deadline - 5, min_time_remaining=5.0)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final
            print(f"  Polish improved: {best_score:.10f}")
    
    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    
    sc_verify = _score(final)
    print(f"FINAL: score={sc_verify:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    
    try:
        ev_score = evaluate_sequence([float(x) for x in final])
        print(f"evaluate_sequence: {ev_score:.10f}")
    except Exception as e:
        print(f"evaluate_sequence error: {e}")
    
    return [float(x) for x in final]
