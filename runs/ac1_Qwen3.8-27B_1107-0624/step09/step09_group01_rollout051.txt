import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize
from scipy.special import softmax

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _solve_cvxpy(n, time_limit=120):
    """Solve the convex problem exactly using CVXPY."""
    a = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [cp.sum(a) == 1, a >= 0]
    
    # For each k = 0, ..., n-1, add constraint (a*a)[k] <= t
    # (a*a)[k] = sum_{i=0}^{k} a[i] * a[k-i]
    # Matrix form: a[:k+1]^T @ anti_id(k+1) @ a[:k+1]
    for k in range(n):
        # Build the anti-identity matrix of size (k+1)
        # Actually, we can just write it as a sum of products
        # For efficiency, let's use the quad_form approach
        size = k + 1
        H_k = np.zeros((size, size))
        H_k[np.arange(size), np.arange(size-1, -1, -1)] = 1.0
        constraints.append(cp.quad_form(a[:size], H_k) <= t)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    
    # Try different solvers
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, time_limit=time_limit)
            break
        except Exception as e:
            continue
    
    if a.value is not None and np.isfinite(a.value).all():
        a_val = np.maximum(a.value, 0)
        s = a_val.sum()
        if s < 1e-15:
            return None
        a_val = a_val / s
        sc = _score(a_val)
        return a_val, float(sc)
    return None
