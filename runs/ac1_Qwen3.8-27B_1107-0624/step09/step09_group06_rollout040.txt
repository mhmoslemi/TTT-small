"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using power-smoothed L-BFGS,
Barzilai-Borwein subgradient, and peak-directed local search.
Explores multiple sequence lengths and structured initializations.

import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    """Compute full (linear) convolution using FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u > (cssv - 1) / idx
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def _conv_grad(a, k):
    """Gradient of (a*a)[k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    # c[k] = sum_j a[j] * a[k-j]
    # dc[k]/da[j] = a[k-j] + a[j-k] (when valid)
    for j in range(n):
        val = 0.0
        if 0 <= k - j < n:
            val += a[k - j]
        if 0 <= j - k < n:
            val += a[j - k]
        g[j] = val
    return g


def _conv_grad_vec(a, indices):
    """Gradient of sum_{k in indices} (a*a)[k] w.r.t. a, vectorized."""
    n = len(a)
    g = np.zeros(n)
    for k in indices:
        # Part 1: a[k-j] for valid j
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            g[j_lo:j_hi+1] += a[k - np.arange(j_lo, j_hi + 1)]
        # Part 2: a[j-k] for valid j
        j_lo2 = max(0, k)
        j_hi2 = min(n - 1, k + n - 1)
        if j_lo2 <= j_hi2:
            g[j_lo2:j_hi2+1] += a[np.arange(j_lo2, j_hi2 + 1) - k]
    return g


def _pow_smooth_obj(z, p, n):
    """Objective: 2*n*(sum c^p)^(1/p) with a = softmax(z). S=1 so score = 2*n*F_p."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    cp = np.power(np.maximum(c, 0), p)
    return 2.0 * n * np.power(cp.sum(), 1.0 / p)


def _pow_smooth_grad(z, p, n):
    """Gradient of power-smoothed objective w.r.t. z."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    cp = np.power(np.maximum(c, 0), p)
    sp = cp.sum()
    Fp = np.power(sp, 1.0 / p)
    # dFp/dc[k] = (1/p) * sp^(1/p - 1) * p * c[k]^(p-1) = sp^(1/p-1) * c[k]^(p-1)
    dFp_dc = np.power(sp, 1.0 / p - 1.0) * np.power(np.maximum(c, 0), p - 1)
    # dc[dFp_dc] w.r.t. a
    # dc[k]/da[j] = a[k-j] + a[j-k]
    # dFp/da[j] = sum_k dFp_dc[k] * (a[k-j] + a[j-k])
    # This is like a correlation
    w = dFp_dc  # weight for each c[k]
    # dFp/da[j] = sum_k w[k] * a[k-j] + sum_k w[k] * a[j-k]
    # = (w_rev * a)[j] + (w * a)[j]  where w_rev[k] = w[2n-2-k]
    # Actually, let's use the convolution approach:
    # sum_k w[k] * a[k-j] for k-j in [0,n-1] => this is (w * a_pad)[something]
    # Let's just use the correlation via FFT
    N = _next_pow2(2 * n)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    # sum_k w[k] * a[k-j] = correlation of w and a
    # In frequency: IFFT( W * conj(A) )... let me think
    # sum_k w[k] * a[k-j]: let m = k-j, so k = m+j
    # = sum_m w[m+j] * a[m]
    # This is the cross-correlation (w ⋆ a)[j]
    # In FFT: IFFT( conj(W) * A ) ... hmm, let me be careful
    # Actually: sum_k w[k] a[k-j] = sum_k w[k] a[k-j]
    # Let's define: (w * a_flipped)[x] where a_flipped[m] = a[n-1-m]
    # This is getting complicated. Let me just do it directly for small n or use a simpler approach.
    
    # Simpler: use the fact that dFp/da = conv(w, a) + conv(w_rev, a) restricted to [0, n-1]
    # where conv is linear convolution and w_rev is w reversed
    # Actually: sum_k w[k] * a[k-j] for j in [0, n-1]
    # Let's compute h1[j] = sum_k w[k] * a[k-j] for j = 0,...,n-1
    # h1[j] = sum_{k=j}^{j+n-1} w[k] * a[k-j] (approximately, with bounds)
    # This is the cross-correlation of w and a at lag j
    
    # Using FFT: cross-correlation can be computed as IFFT(FFT(w) * conj(FFT(a)))
    # But we need to be careful with the one-sided vs two-sided
    # Let's just pad and compute
    wf = np.fft.rfft(w, N)
    af = np.fft.rfft(a, N)
    # h1[j] = sum_k w[k] a[k-j] = (w ⋆ a)[j]
    # In freq domain: W * conj(A) gives the spectrum of the cross-correlation
    h1 = np.fft.irfft(wf * np.conj(af), N)[:n]
    h2 = np.fft.irfft(np.conj(wf) * af, N)[:n]
    # dFp/da = h1 + h2
    dFp_da = h1 + h2
    
    # Now chain rule: dFp/dz = diag(a) * (dFp_da - sum(dFp_da * a))
    # because d(softmax(z))/dz = diag(a) - a a^T
    s = np.dot(dFp_da, a)
    grad_z = 2.0 * n * a * (dFp_da - s)
    return grad_z

    """L-BFGS on power-smoothed objective with softmax parameterization."""
    n = len(a_init)
    if p_schedule is None:
        p_schedule = [10, 50, 100, 200, 500, 1000, 2000, 5000]
    
    current_a = a_init.copy()
    current_a = current_a / current_a.sum()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.sum() / len(z)
    
    for p in p_schedule:
        if time.monotonic() > deadline:
            break
        
        remaining = deadline - time.monotonic()
        est_per_iter = 0.005 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(20, int(remaining / (est_per_iter * 15))))
        
        def obj(z_):
            return _pow_smooth_obj(z_, p, n)
        
        def grad(z_):
            return _pow_smooth_grad(z_, p, n)

                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))

def _bb_subgradient(a_init, deadline, K=20, max_iter=5000):
    """Barzilai-Borwein subgradient method."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    g_prev = None
    x_prev = None
    alpha = 0.01
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max  # since sum(a) = 1
        
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
        
        # Select top K peaks
        K_eff = min(K, len(c))
        top_idx = np.argsort(c)[::-1][:K_eff]
        
        # Compute subgradient
        g = _conv_grad_vec(a, top_idx)
        g = g - g.mean()  # project to tangent space
        
        # BB step size
        if g_prev is not None and x_prev is not None:
            s = a - x_prev
            y = g - g_prev
            sy = np.dot(s, y)
            ss = np.dot(s, s)
            ys = np.dot(y, s)
            if ss > 1e-18:
                if abs(sy) > 1e-18:
                    alpha_bb = ss / abs(sy)
                    alpha = 0.5 * alpha + 0.5 * min(max(alpha_bb, 1e-6), 1.0)
                else:
                    alpha *= 0.9
        
        # Step
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-18:
            break
        step = alpha * g / gnorm
        a_new = _project_simplex(a - step)
        
        x_prev = a.copy()
        g_prev = g.copy()
        a = a_new
    
    return best_a, best_sc


def _peak_directed(a_init, deadline, max_iter=10000):
    """Peak-directed optimization: systematically reduce the top peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.005
    no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                lr *= 0.7
                no_improve = 0
                if lr < 1e-6:
                    break
        
        # Find top peaks
        K = max(3, int(n * 0.001))
        top_idx = np.argsort(c)[::-1][:K]
        
        # Compute gradient for active peaks
        g = _conv_grad_vec(a, top_idx)
        g_tangent = g - g.mean()
        
        # Move in the direction that reduces active peaks
        gnorm = np.linalg.norm(g_tangent)
        if gnorm < 1e-18:
            break
        
        a_new = a - lr * g_tangent / gnorm
        a_new = _project_simplex(a_new)
        a = a_new
    
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.arange(n, dtype=np.float64)
        a = np.exp(-0.5 * t / n)
        return a / a.sum()
    elif kind == "front_loaded":
        t = np.arange(n, dtype=np.float64)
        a = (1.0 - t / (n - 1)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _smoobj_lse(z, beta, n):
    """Log-sum-exp smoothed max objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd_lse(z, beta, n):
    """Gradient of log-sum-exp smoothed max."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # dw/dc[k] = w[k]
    # dLSE/da[j] = sum_k w[k] * (a[k-j] + a[j-k])
    N = _next_pow2(2 * n)
    Wf = np.fft.rfft(w, N)
    Af = np.fft.rfft(a, N)
    h1 = np.fft.irfft(Wf * np.conj(Af), N)[:n]
    h2 = np.fft.irfft(np.conj(Wf) * Af, N)[:n]
    dLSE_da = h1 + h2
    s_dot = np.dot(dLSE_da, a)
    grad_z = 2.0 * n * a * (dLSE_da - s_dot)
    return grad_z

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LSE-smoothed L-BFGS, Barzilai-Borwein subgradient, peak-directed search,
multi-n exploration, and structured random restarts to escape local minima.
Key innovations over previous: adaptive K for active peaks, power-smoothing
as alternative landscape, sinusoidal/cyclic restart patterns, and finer
beta schedules with more L-BFGS iterations.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u > (cssv - 1) / idx
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def _conv_grad_vec(a, indices):
    n = len(a)
    g = np.zeros(n)
    for k in indices:
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            g[j_lo:j_hi + 1] += a[k - np.arange(j_lo, j_hi + 1)]
        j_lo2 = max(0, k)
        j_hi2 = min(n - 1, k + n - 1)
        if j_lo2 <= j_hi2:
            g[j_lo2:j_hi2 + 1] += a[np.arange(j_lo2, j_hi2 + 1) - k]
    return g


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    N = _next_pow2(2 * n)
    Wf = np.fft.rfft(w, N)
    Af = np.fft.rfft(a, N)
    h1 = np.fft.irfft(np.conj(Wf) * Af, N)[:n]
    h2 = np.fft.irfft(Wf * np.conj(Af), N)[:n]
    dLSE_da = h1 + h2
    s_dot = np.dot(dLSE_da, a)
    grad_z = 2.0 * n * a * (dLSE_da - s_dot)
    return grad_z


def _pow_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c = np.maximum(c, 0.0)
    c_max = float(c.max())
    if c_max < 1e-300:
        return 0.0
    r = c / c_max
    rp_sum = np.sum(np.power(r, p))
    if rp_sum < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * c_max * np.power(rp_sum, 1.0 / p)


def _pow_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c = np.maximum(c, 0.0)
    c_max = float(c.max())
    if c_max < 1e-300:
        return np.zeros_like(z)
    r = c / c_max
    rp_sum = np.sum(np.power(r, p))
    if rp_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        # dF/dc[k] proportional to r[k]^(p-1)
        w = np.power(r, p - 1)
        w = w / w.sum()
    N = _next_pow2(2 * n)
    Wf = np.fft.rfft(w, N)
    Af = np.fft.rfft(a, N)
    h1 = np.fft.irfft(np.conj(Wf) * Af, N)[:n]
    h2 = np.fft.irfft(Wf * np.conj(Af), N)[:n]
    dF_da = h1 + h2
    s_dot = np.dot(dF_da, a)
    grad_z = 2.0 * n * a * (dF_da - s_dot)
    return grad_z


def _run_lse_lbfgs(a_init, deadline, beta_schedule=None, n_betas=None):
    n = len(a_init)
    current_a = a_init.copy()
    current_a = current_a / current_a.sum()
    current_sc = _score(current_a)

    if beta_schedule is None:
        c = _full_convolve(current_a, current_a)
        c_max = float(c.max())
        if c_max < 1e-15:
            return current_a, current_sc
        c_sorted = np.sort(c)[::-1]
        K_gap = min(50, len(c_sorted) - 1)
        gap = c_max - c_sorted[K_gap]
        if gap < c_max * 0.0005:
            gap = c_max * 0.0005
        base_beta = max(gap * 0.5, c_max * 0.002)
        if n_betas is None:
            n_betas = 18
        ratios = np.array([1.0, 0.55, 0.35, 0.22, 0.14, 0.09, 0.055, 0.035,
                           0.022, 0.014, 0.009, 0.006, 0.0038, 0.0024,
                           0.0015, 0.0009, 0.0006, 0.0003])
        beta_schedule = base_beta * ratios[:n_betas]

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.sum() / len(z)

    for i, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        scale = max(1.0, n / 16384.0)
        est_per_iter = 0.006 * scale
        max_iter = min(8000, max(30, int(remaining / (est_per_iter * 25))))

        def obj(z_):
            return _lse_obj(z_, beta, n)

        def grad_fn(z_):
            return _lse_grad(z_, beta, n)

        try:
            result = minimize(obj_fn_wrapper(obj, n), z, jac=obj_grad_wrapper(grad_fn, n),
                              method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-16, 'maxcor': 100, 'maxls': 50})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.sum() / len(z)
                if i % 3 == 0 or i == len(beta_schedule) - 1:
                    print(f"    LSE-LBFGS beta[{i}]={beta:.2e}: score={current_sc:.10f}")
        except Exception:
            pass

    return current_a, current_sc


def obj_fn_wrapper(obj, n):
    return _ObjWrapper(obj, n)


def obj_grad_wrapper(grad_fn, n):
    return _GradWrapper(grad_fn, n)


class _ObjWrapper:
    def __init__(self, obj, n):
        self.obj = obj
        self.n = n

    def __call__(self, z):
        return self.obj(z)


class _GradWrapper:
    def __init__(self, grad_fn, n):
        self.grad_fn = grad_fn
        self.n = n

    def __call__(self, z):
        return self.grad_fn(z)


def _run_pow_lbfgs(a_init, deadline, p_schedule=None):
    n = len(a_init)
    current_a = a_init.copy()
    current_a = current_a / current_a.sum()
    current_sc = _score(current_a)

    if p_schedule is None:
        p_schedule = [10, 30, 60, 120, 250, 500, 1000, 2000]

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.sum() / len(z)

    for p in p_schedule:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        scale = max(1.0, n / 16384.0)
        est_per_iter = 0.006 * scale
        max_iter = min(6000, max(20, int(remaining / (est_per_iter * 25))))

        def obj_p(z_):
            return _pow_obj(z_, p, n)

        def grad_p(z_):
            return _pow_grad(z_, p, n)

        try:
            result = minimize(obj_p, z, jac=grad_p, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-16, 'maxcor': 80, 'maxls': 40})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.sum() / len(z)
                print(f"    Pow-LBFGS p={p}: score={current_sc:.10f}")
        except Exception:
            pass

    return current_a, current_sc


def _bb_subgradient(a_init, deadline, K_init=10, max_iter=5000):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    g_prev = None
    x_prev = None
    alpha = 0.01
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 1000:
            break

        c_sorted = np.sort(c)[::-1]
        if no_improve > 200:
            K = min(int(K_init * (1 + no_improve / 200.0)), 200)
        else:
            K = K_init

        K = max(3, min(K, len(c) // 2))
        top_idx = np.argsort(c)[::-1][:K]

        g = _conv_grad_vec(a, top_idx)
        g = g - g.mean()

        if g_prev is not None and x_prev is not None:
            s_vec = a - x_prev
            y_vec = g - g_prev
            sy = np.dot(s_vec, y_vec)
            ss = np.dot(s_vec, s_vec)
            if ss > 1e-18:
                if abs(sy) > 1e-18:
                    alpha_bb = ss / abs(sy)
                    alpha = 0.3 * alpha + 0.7 * min(max(alpha_bb * 0.01, 1e-6), 0.1)
                else:
                    alpha *= 0.95

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-18:
            break

        step_size = alpha / gnorm
        a_new = _project_simplex(a - step_size * g)

        x_prev = a.copy()
        g_prev = g.copy()
        a = a_new

    return best_a, best_sc


def _peak_directed(a_init, deadline, max_iter=20000):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.003
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                lr *= 0.8
                no_improve = 0
                if lr < 1e-7:
                    break

        c_sorted = np.sort(c)[::-1]
        threshold = c_max - c_max * 0.005
        K = max(3, int(np.sum(c_sorted >= threshold)))
        K = min(K, max(5, int(n * 0.005)))

        top_idx = np.argsort(c)[::-1][:K]
        g = _conv_grad_vec(a, top_idx)
        g_t = g - g.mean()

        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-18:
            break

        a_new = _project_simplex(a - lr * g_t / gnorm)
        a = a_new

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.arange(n, dtype=np.float64)
        a = np.exp(-0.3 * t / n)
        return a / a.sum()
    elif kind == "front_loaded":
        t = np.arange(n, dtype=np.float64)
        a = (1.0 - 0.5 * t / (n - 1)) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-25.0 * (t - 0.3) ** 2) + np.exp(-25.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "flat_taper":
        a = np.ones(n)
        taper = max(1, n // 40)
        for i in range(taper):
            a[i] = (i + 1) / taper
            a[n - 1 - i] = (i + 1) / taper
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _shrinkage_optimize(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.001, 0.003, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05, 0.08]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)
            print(f"    Shrinkage frac={frac}: score={best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining LSE-LBFGS, power-smoothed L-BFGS,
    BB-subgradient, peak-directed search, multi-n exploration, shrinkage,
    and structured random restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: LSE-LBFGS (main workhorse, fine beta schedule) ===
    print(f"\n[P1] LSE-LBFGS fine (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 700:
        p1_dl = min(deadline - 700, t_start + 350)
        a1, sc1 = _run_lse_lbfgs(best_seq, p1_dl, n_betas=18)
        if sc1 < best_score:
            best_seq = a1.copy()
            best_score = sc1
            print(f"  IMPROVED: {best_score:.10f} [LSE-1] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  LSE-1: no improve (best={best_score:.10f})")

    # === Phase 2: Power-smoothed L-BFGS (different landscape) ===
    print(f"\n[P2] Power-LBFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 550:
        p2_dl = min(deadline - 550, t_start + 500)
        a2, sc2 = _run_pow_lbfgs(best_seq, p2_dl,
                                  p_schedule=[10, 30, 80, 200, 500, 1200])
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  IMPROVED: {best_score:.10f} [Pow-LBFGS] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Pow-LBFGS: no improve (best={best_score:.10f})")

    # === Phase 3: BB Subgradient (escape local minima) ===
    print(f"\n[P3] BB Subgradient (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 450:
        p3_dl = min(deadline - 450, t_start + 620)
        a3, sc3 = _bb_subgradient(best_seq, p3_dl, K_init=15, max_iter=4000)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  IMPROVED: {best_score:.10f} [BB] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  BB: no improve (best={best_score:.10f})")

    # === Phase 4: LSE-LBFGS round 2 (refined) ===
    print(f"\n[P4] LSE-LBFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        p4_dl = min(deadline - 350, t_start + 720)
        a4, sc4 = _run_lse_lbfgs(best_seq, p4_dl, n_betas=15)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  IMPROVED: {best_score:.10f} [LSE-2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Peak-directed refinement ===
    print(f"\n[P5] Peak-directed (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        p5_dl = min(deadline - 250, t_start + 850)
        a5, sc5 = _peak_directed(best_seq, p5_dl, max_iter=15000)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  IMPROVED: {best_score:.10f} [peak-dir] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Peak-dir: no improve (best={best_score:.10f})")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [32768, 16384, 24576]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 120:
            break
        if n_try == len(best_seq):
            continue

        # Try resampling current best
        if time.monotonic() < deadline - 80:
            a_res = _resample(best_seq, n_try)
            sc_res = _score(a_res)
            print(f"  n={n_try} (resampled): init score={sc_res:.10f}")
            if time.monotonic() < deadline - 60:
                time_budget = min(deadline - 50, t_start + (n_try // 512) * 2 + 100)
                a_opt, sc_opt = _run_lse_lbfgs(a_res, time_budget, n_betas=10)
                if sc_opt < best_score:
                    best_seq = a_opt.copy()
                    best_score = sc_opt
                    print(f"  IMPROVED: {best_score:.10f} [resample n={n_try}] t={time.monotonic()-t_start:.1f}s")

        # Try fresh structured initialization
        if time.monotonic() < deadline - 60:
            for kind in ["tapered", "cos_mod"]:
                if time.monotonic() > deadline - 50:
                    break
                a_init = _make_profile(n_try, kind)
                sc_init = _score(a_init)
                print(f"  n={n_try}, init={kind}: score={sc_init:.10f}")
                if sc_init < best_score + 0.3:
                    time_budget = min(deadline - 40, t_start + (n_try // 512) * 2 + 80)
                    a_opt2, sc_opt2 = _run_lse_lbfgs(a_init, time_budget, n_betas=8)
                    if sc_opt2 < best_score:
                        best_seq = a_opt2.copy()
                        best_score = sc_opt2
                        print(f"  IMPROVED: {best_score:.10f} [n={n_try},{kind}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Shrinkage ===
    print(f"\n[P7] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 60)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  IMPROVED: {best_score:.10f} [shrinkage] t={time.monotonic()-t_start:.1f}s")
            # Re-optimize after shrinkage
            if time.monotonic() < deadline - 50:
                a_shr2, sc_shr2 = _run_lse_lbfgs(best_seq, deadline - 30, n_betas=8)
                if sc_shr2 < best_score:
                    best_seq = a_shr2.copy()
                    best_score = sc_shr2
                    print(f"  IMPROVED: {best_score:.10f} [post-shrink] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Structured random restarts ===
    print(f"\n[P8] Random restarts (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_seq)
    for restart in range(30):
        if time.monotonic() > deadline - 40:
            break

        pert_type = restart % 5
        if pert_type == 0:
            noise = 0.005 * rng.standard_normal(n_best)
            a_try = _project_simplex(best_seq + noise)
        elif pert_type == 1:
            freq = rng.integers(1, 15)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.02, 0.08)
            t = np.arange(n_best)
            mod = 1.0 + amp * np.cos(2 * np.pi * freq * t / n_best + phase)
            a_try = _project_simplex(best_seq * mod)
        elif pert_type == 2:
            shift = rng.integers(1, max(2, n_best // 4))
            a_try = np.roll(best_seq, shift)
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s > 1e-10:
                a_try = a_try / s
            else:
                continue
        elif pert_type == 3:
            # Two-sinusoid modulation
            f1, f2 = rng.integers(1, 20, size=2)
            p1, p2 = rng.uniform(0, 2 * np.pi, size=2)
            a1_amp, a2_amp = rng.uniform(0.02, 0.06, size=2)
            t = np.arange(n_best)
            mod = 1.0 + a1_amp * np.cos(2 * np.pi * f1 * t / n_best + p1) + \
                  a2_amp * np.cos(2 * np.pi * f2 * t / n_best + p2)
            a_try = _project_simplex(best_seq * mod)
        else:
            # Block swap
            block_size = rng.integers(n_best // 10, n_best // 3)
            start = rng.integers(0, n_best - block_size)
            a_try = best_seq.copy()
            tmp = a_try[start:start + block_size].copy()
            a_try[start:start + block_size] = a_try[n_best - block_size:n_best - block_size + start - start]
            a_try[n_best - block_size:] = tmp[:n_best - start if n_best - start < block_size else block_size]
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s > 1e-10:
                a_try = a_try / s
            else:
                continue

        sc_try = _score(a_try)
        if sc_try < best_score + 0.1:
            if time.monotonic() < deadline - 25:
                a_opt, sc_opt = _run_lse_lbfgs(a_try, min(deadline - 15, t_start + (n_best // 1024) * 3 + 60),
                                               n_betas=6)
                if sc_opt < best_score:
                    best_seq = a_opt.copy()
                    best_score = sc_opt
                    print(f"  IMPROVED restart {restart} (type {pert_type}): {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 9: Final LSE polish ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_fin, sc_fin = _run_lse_lbfgs(best_seq, deadline - 3, n_betas=6)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  FINAL: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'=' * 60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")

    eval_score = evaluate_sequence([float(x) for x in final])
    print(f"Eval: {eval_score:.10f}")

    if not np.isfinite(eval_score):
        print("Falling back to uniform 256")
        fallback = _make_profile(256, "uniform")
        return [float(x) for x in fallback]

    return [float(x) for x in final]
