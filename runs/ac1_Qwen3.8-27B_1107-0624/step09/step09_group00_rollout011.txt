"""
Optimized step function search using:
1. Asymmetric profile initialization (ramps, skewed Gaussians)
2. Nesterov accelerated subgradient method with simplex projection

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _project_simplex(v, target_sum=1.0):
    """Project vector v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    # If all values are already non-negative and sum is correct
    if np.all(v >= 0) and abs(v.sum() - target_sum) < 1e-15:
        return v.copy()
    
    # Standard simplex projection algorithm
    u = v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    
    rho = np.sort(u)[::-1]
    cssv = np.cumsum(rho)
    rho_ind = np.arange(1, n + 1)
    cond = rho - (cssv - target_sum) / rho_ind > 0
    k = cond[0]
    theta = (cssv[k-1] - target_sum) / k
    
    w = np.maximum(u - theta, 0)
    return w

def _autocorr_gradient_at_lag(a, k):
    """Compute gradient of c[k] = sum_j a[j]*a[k-j] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    # c[k] = sum_j a[j] * a[k-j] for valid j, k-j
    # d c[k] / d a[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
    # Term 1: a[k-i] valid when 0 <= k-i < n, i.e., max(0, k-n+1) <= i <= min(n-1, k)
    lo1 = max(0, k - n + 1)
    hi1 = min(n - 1, k)
    if lo1 <= hi1:
        grad[lo1:hi1+1] += a[k - np.arange(lo1, hi1+1)]
    
    # Term 2: a[i-k] valid when 0 <= i-k < n, i.e., k <= i <= k+n-1, intersect [0, n-1]
    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if lo2 <= hi2:
        grad[lo2:hi2+1] += a[np.arange(lo2, hi2+1) - k]
    
    return grad

def _subgradient_of_max(a, c, top_k=5, tol=0.01):
    """Compute a subgradient of max(c) with respect to a.
    Uses a weighted combination of the top peaks."""
    n = len(a)
    c_max = float(c.max())
    
    # Find peaks within tol of max
    threshold = c_max * (1 - tol)
    active = np.where(c >= threshold)[0]
    
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    
    # Weight by how close to max (exponential weighting)
    weights = np.exp(-(c_max - c[active]) / (c_max * 0.001 + 1e-15))
    weights = weights / weights.sum()
    
    # Combine gradients
    grad = np.zeros(n)
    for w, k in zip(weights, active):
        grad += w * _autocorr_gradient_at_lag(a, k)
    
    return grad, c_max

def _nesterov_subgradient(a_init, deadline, max_iter=100000):
    """Nesterov accelerated subgradient method for minimizing max(c) on the simplex."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    y = a.copy()  # Nesterov's lookahead point
    a_prev = a.copy()
    best_a = a.copy()
    best_obj = float('inf')
    
    t = 1.0
    t_prev = 1.0
    L_bound = 2.0 * n  # Lower bound on Lipschitz constant
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        obj = 2.0 * n * c_max
        s = float(a.sum())
        obj_norm = 2.0 * n * c_max / (s*s)  # Actually we work with sum=1
        
        if obj_norm < best_obj:
            best_obj = obj_norm
            best_a = a.copy()
        
        # Compute subgradient at y
        grad, _ = _subgradient_of_max(y, c_y, tol=0.05)
        
        # Step size (Nesterov: step = 1 / (t * L))
        step = 1.0 / (t * L_bound * 0.1)  # Conservative
        # Actually, for the constrained problem, we need to be more careful
        # Use backtracking
        step_size = 0.1 / t
        
        # Update
        a_new = _project_simplex(y - step_size * grad)
        
        # Check improvement
        c_new = _full_convolve(a_new, a_new)
        obj_new = 2.0 * n * float(c_new.max())
        if obj_new < obj:
            a = a_new
        else:
            # Try smaller step
            step_size *= 0.5
            a = _project_simplex(y - step_size * grad)
        
        # Nesterov update
        t = (1 + np.sqrt(1 + 4 * t_prev**2)) / 2
        momentum = (t_prev - 1) / t
        y = a + momentum * (a - a_prev)
        y = _project_simplex(y)  # Keep y in the feasible set
        a_prev = a
        t_prev = t
        
        if it % 1000 == 0:
            print(f"    Nesterov it={it}, obj={obj_norm:.10f}, best={best_obj:.10f}")
    
    return best_a, best_obj

def _pnorm_objective(a, p):
    """Lp norm of autocorrelation (smooth approximation to L-infinity)."""
    c = _full_convolve(a, a)
    c_pos = np.maximum(c, 1e-30)
    return float((np.sum(c_pos**p))**(1.0/p))

def _pnorm_gradient(a, p):
    """Gradient of Lp norm of autocorrelation."""
    n = len(a)
    c = _full_convolve(a, a)
    c_pos = np.maximum(c, 1e-30)
    c_p = c_pos**p
    sum_cp = float(np.sum(c_p))
    norm = sum_cp**(1.0/p)
    
    # d/d a of (sum c[k]^p)^(1/p) = (1/p) * (sum c[k]^p)^(1/p - 1) * sum p*c[k]^(p-1) * dc[k]/da
    # = (sum c[k]^p)^(1/p - 1) * sum c[k]^(p-1) * dc[k]/da
    coeff = sum_cp**(1.0/p - 1)
    
    # Weighted combination of autocorrelation gradients
    weights = c_pos**(p-1)
    grad = np.zeros(n)
    # Use FFT for efficiency
    # grad[j] = sum_k weights[k] * dc[k]/da[j]
    # dc[k]/da[j] = a[k-j] + a[j-k] (with valid index handling)
    # This is essentially 2 * convolve(weights, a_rev)[j+n-1] approximately
    
    a_rev = a[::-1]
    wc = _full_convolve(weights, a_rev)
    grad = 2.0 * wc[n-1:2*n-1] * coeff
    
    # Project to tangent space (sum = 0)
    grad = grad - np.mean(grad)
    return grad * 2.0 * n  # Scale for objective 2n * Lp_norm

def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        a = np.arange(1, n + 1, dtype=np.float64)
        return a / a.sum()
    elif kind == "ramp2":
        a = np.arange(1, n + 1, dtype=np.float64) ** 2
        return a / a.sum()
    elif kind == "skew_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.3) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "pow_half":
        t = np.linspace(0, 1, n)
        a = t + 0.1
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_asym":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * 0.5 * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20*(t-0.2)**2) + np.exp(-20*(t-0.8)**2) + 0.1
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        if rng is None:
            rng = np.random.default_rng(42)
        a = rng.random(n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _lbfgs_pnorm(a_init, deadline, p_start=8, p_schedule=None, min_time=30):
    """L-BFGS optimization with p-norm smoothing and adaptive p."""
    n = len(a_init)
    current = _normalize(a_init.copy())
    current_sc = _score(current)
    
    if p_schedule is None:
        p_schedule = [p_start, p_start*2, p_start*4, p_start*8, p_start*16, 
                     p_start*32, p_start*64, p_start*128, p_start*256, p_start*512]
    
    for p in p_schedule:
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 5:
            break
        
        z = np.log(np.maximum(current, 1e-30))
        z = z - z.mean()
        
        def obj_fn(z_vec):
            a_vec = _softmax(z_vec)
            c = _full_convolve(a_vec, a_vec)
            c_pos = np.maximum(c, 1e-30)
            val = (np.sum(c_pos**p))**(1.0/p)
            return 2.0 * n * val
        
        def grad_fn(z_vec):
            a_vec = _softmax(z_vec)
            c = _full_convolve(a_vec, a_vec)
            c_pos = np.maximum(c, 1e-30)
            c_p = c_pos**(p-1)
            sum_cp = float(np.sum(c_pos**p))
            
            # Gradient wrt a: weighted combination of autocorr gradients
            a_rev = a_vec[::-1]
            wc = _full_convolve(c_p, a_rev)
            grad_a = 2.0 * wc[n-1:2*n-1]
            
            # Chain rule: dz -> da via softmax
            # da_i/dz_j = a_i*(delta_ij - a_j)
            # grad_z = (diag(a) - outer(a,a)) @ grad_a
            grad_z = a_vec * (grad_a - np.dot(a_vec, grad_a))
            return 2.0 * n * grad_z
        
        try:
            max_iter = min(3000, max(20, int(remaining / 0.05)))
            result = minimize(obj_fn, z, jac=grad_fn, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12,
                                      'maxcor': 50})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc:
                current = a_new
                current_sc = sc_new
        except Exception as e:
            pass
    
    return current, current_sc

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _targeted_ils(a_init, deadline, rng, min_time=60):
    """ILS with perturbations targeted at reducing peak structure."""
    n = len(a_init)
    current = _normalize(a_init.copy())
    current_sc = _score(current)
    best = current.copy()
    best_sc = current_sc
    
    attempt = 0
    while time.monotonic() + min_time < deadline:
        # Analyze current peak structure
        c = _full_convolve(current, current)
        c_max = float(c.max())
        peaks = np.where(c >= c_max * 0.95)[0]
        
        if len(peaks) > 0:
            peak_center = float(np.mean(peaks))
            n_peaks = len(peaks)
        else:
            peak_center = n - 1
            n_peaks = 1
        
        # Choose perturbation based on peak structure
        strategy = attempt % 6
        
        a_pert = current.copy()
        if strategy == 0:
            # Reduce energy at peak-contributing positions
            # The peak at lag k gets contribution from positions that overlap at that lag
            # For peak at k=n-1, all positions contribute
            # For off-center peaks, specific positions matter more
            sigma = 0.02
            z = np.log(np.maximum(a_pert, 1e-30))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif strategy == 1:
            # Targeted: reduce values at positions that contribute most to the peak
            c = _full_convolve(current, current)
            k_peak = int(np.argmax(c))
            # Positions that contribute to c[k_peak]: j and k_peak-j
            # Increase "spreading" by adding anti-correlation
            sigma = 0.03
            z = np.log(np.maximum(a_pert, 1e-30))
            # Modulate with a sinusoid to create anti-correlation
            freq = rng.uniform(1, 5)
            phase = rng.uniform(0, 2*np.pi)
            modulation = 1.0 + sigma * np.sin(2*np.pi*freq*np.arange(n)/n + phase)
            z += np.log(np.maximum(modulation, 1e-10))
            z = z - z.mean()
            a_pert = _softmax(z)
        elif strategy == 2:
            # Block perturbation
            block = max(50, n // 10)
            start = int(rng.integers(0, n - block))
            sigma = 0.05
            z = np.log(np.maximum(a_pert, 1e-30))
            z[start:start+block] += sigma * rng.standard_normal(block)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif strategy == 3:
            # Mirror + noise
            seg = max(20, n // 15)
            start = int(rng.integers(0, n - seg))
            a_pert[start:start+seg] = current[start:start+seg][::-1]
            z = np.log(np.maximum(a_pert, 1e-30))
            z += 0.01 * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif strategy == 4:
            # Sparse large perturbation
            n_sparse = max(20, n // 20)
            idx = rng.choice(n, size=n_sparse, replace=False)
            a_pert[idx] *= np.exp(0.1 * rng.standard_normal(n_sparse))
            a_pert = _normalize(a_pert)
        else:
            # Frequency domain perturbation
            A = np.fft.rfft(current)
            # Add small noise to magnitude (not DC)
            mag = np.abs(A).copy()
            phase = np.angle(A)
            mag[1:] *= np.exp(0.01 * rng.standard_normal(len(mag)-1))
            A_new = mag * np.exp(1j * phase)
            a_pert = np.fft.irfft(A_new, n)
            a_pert = _normalize(a_pert)
        
        # Quick optimization
        opt_dl = min(deadline - min_time, time.monotonic() + 10)
        a_opt, sc_opt = _logsumexp_lbfgs(a_pert, opt_dl, min_time=2)
        
        if sc_opt < best_sc - 1e-12:
            best_sc = sc_opt
            best = a_opt.copy()
            current = a_opt.copy()
            print(f"    ILS NEW BEST: {best_sc:.10f} (strat={strategy}) t={time.monotonic()-t_start:.0f}s")
        elif sc_opt < current_sc - 1e-14:
            current = a_opt.copy()
            current_sc = sc_opt
        
        attempt += 1
    
    return best, best_sc

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
