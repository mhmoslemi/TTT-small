"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using a hybrid approach:
1. Multi-scale L-BFGS with log-sum-exp smoothing (coarse-to-fine beta annealing)
2. Active-set peak equalization with adaptive line search

4. Symmetry exploitation and diverse multi-start
Key innovations: adaptive beta schedule based on flatness, wider active sets,
Newton-inspired steps using the convolution Hessian structure, and better time allocation.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _compute_gradient_active(a, S):
    """Compute average gradient of active peaks S."""
    n = len(a)
    K = len(S)
    w_S = np.zeros(2 * n - 1)
    w_S[S] = 1.0 / K
    a_rev = a[::-1]
    h = _full_convolve(w_S, a_rev)
    g = h[n - 1:2 * n - 1]
    return 2.0 * g


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0, beta_scale=1.0, max_total_iter=20000):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.008) * beta_scale

    # Adaptive beta schedule: start coarse, go fine
    betas = base_beta * np.array([
        2.0, 1.0, 0.5, 0.25, 0.125,

        0.004, 0.002, 0.001, 0.0005,

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    total_iters = 0
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:

                current_a = a_new

def _newton_active_step(a, S, n):
    """Try a Newton-inspired step using the Hessian structure of the convolution.
    The Hessian of c[k] has entries H[j,l] = 2 if j+l=k, else 0.
    For the active set average, H_avg = (2/|S|) * sum_{k in S} Q_k.
    We solve H_avg * d = -g + lambda*1 for the Newton direction."""
    K = len(S)
    # Gradient
    g = _compute_gradient_active(a, S)
    g = g - np.mean(g)

    # Build and solve the banded system
    # H[j,l] = (2/K) if j+l in S, else 0
    # H*d = b where b[j] = -(2/K)*sum_{k in S} 2*a[k-j] + lambda
    # With constraint sum(d) = 0

    # For efficiency, use iterative approach:
    # The matrix H is symmetric with nonzeros on anti-diagonals j+l in S
    # We can compute H*v in O(n*K) time

    def matvec(v):
        # (H*v)[j] = (2/K) * sum_{k in S} v[k-j] for valid k-j
        result = np.zeros(n)
        for k in S:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            # v[k-j] for j in [j_min, j_max]
            # k-j ranges from k-j_max to k-j_min
            idx_start = k - j_max
            idx_end = k - j_min + 1
            result[j_min:j_max+1] += v[idx_start:idx_end]
        result *= 2.0 / K
        return result

    # Solve using a few CG-like iterations (H may not be SPD, so use GMRES-like)
    # For small K, the matrix is nearly diagonal in the anti-diagonal basis
    # Use simple iterative method
    b = -g  # ignoring lambda for now, will enforce sum(d)=0
    # Add lambda to make sum(d)=0: solve [H 1; 1' 0][d; lambda] = [b; 0]
    # Approximate: d = H^{-1} b, then d -= mean(d)*1

    # Use a few power iterations / direct approach for small bandwidth
    # For practical purposes, just use the gradient direction with a
    # curvature estimate
    Hg = matvec(g)
    curvature = float(np.dot(g, Hg))
    if abs(curvature) < 1e-15:
        return -g / (np.linalg.norm(g) + 1e-15)

    # Newton step size: t = g^T g / (g^T H g)
    # But H may not be PSD, so use damped version
    grad_norm_sq = float(np.dot(g, g))
    if curvature > 0:
        t = grad_norm_sq / (curvature + 1e-15)
        t = min(t, 2.0 / (np.max(np.abs(g)) + 1e-15))
    else:
        # H not positive in this direction, use steepest descent
        t = 1.0 / (np.max(np.abs(g)) + 1e-15) * 0.5

    d = -t * g
    d = d - np.mean(d)  # enforce sum = 0
    return d


def _make_profile(n, kind="uniform", rng=None):

    elif kind == "random_smooth":
        if rng is None:
            rng = np.random.default_rng(123)
        a = rng.standard_normal(n)
        # Smooth with a simple filter
        kernel = np.ones(5) / 5.0
        a = np.convolve(a, kernel, mode='same')
        a = np.abs(a) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _asymmetric_perturb(a, rng, strength=0.05):
    n = len(a)
    t = np.arange(n) / n
    ramp = 0.4 * (t - 0.5)
    noise = 0.6 * rng.standard_normal(n)
    perturbation = ramp + noise
    perturbation = perturbation - perturbation.mean()
    z = np.log(np.maximum(a, 1e-30))
    z = z + strength * perturbation
    z = z - z.mean()
    return _softmax(z)


def _symmetrize(a):
    """Make sequence symmetric."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent available. Uniform n={n_parent}, score={best_score:.10f}")

    n = len(best_seq)

    # === Phase 1: L-BFGS warmup on parent ===
    print(f"\n[P1] L-BFGS warmup (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        a_p1 = _run_lbfgs(parent, deadline, min_time_remaining=600.0, beta_scale=1.0)
        sc_p1 = _score(a_p1)
        if sc_p1 < best_score:
            best_seq = a_p1.copy()
            best_score = sc_p1
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_warmup]")
        else:
            print(f"  L-BFGS warmup: {sc_p1:.10f} (best={best_score:.10f})")

    # === Phase 2: Active set equalization ===
    print(f"\n[P2] Active set equalization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        pol_deadline = min(deadline - 400, time.monotonic() + 200)
        a_eq, sc_eq = _active_set_equalize(best_seq, pol_deadline, min_time_remaining=50.0)
        if sc_eq < best_score:
            best_seq = a_eq.copy()
            best_score = sc_eq
            print(f"  NEW BEST: {best_score:.10f} [active_set_1]")
        else:
            print(f"  Active set: {sc_eq:.10f} (best={best_score:.10f})")

    # === Phase 3: L-BFGS refinement after active set ===
    print(f"\n[P3] L-BFGS refinement (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a_p3 = _run_lbfgs(best_seq, deadline, min_time_remaining=300.0, beta_scale=0.3)
        sc_p3 = _score(a_p3)
        if sc_p3 < best_score:
            best_seq = a_p3.copy()
            best_score = sc_p3
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_refine]")

    # === Phase 4: Asymmetric perturbation + re-optimize ===
    print(f"\n[P4] Asymmetric exploration (t={time.monotonic()-t_start:.1f}s)")
    for trial in range(3):
        if time.monotonic() > deadline - 250:
            break
        strength = 0.04 * (trial + 1)
        a_asym = _asymmetric_perturb(best_seq, rng, strength=strength)
        if time.monotonic() < deadline - 100:
            a_opt = _run_lbfgs(a_asym, deadline, min_time_remaining=100.0, beta_scale=0.5)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [asym trial {trial}]")
            else:
                print(f"  Asym trial {trial}: {sc_opt:.10f} (no improvement)")

    # === Phase 5: Symmetry exploitation ===
    print(f"\n[P5] Symmetry exploitation (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a_sym = _symmetrize(best_seq)
        sc_sym = _score(a_sym)
        print(f"  Symmetrized score: {sc_sym:.10f}")
        if sc_sym < best_score:
            # Optimize the symmetric version
            a_segym = _run_lbfgs(a_sym, deadline, min_time_remaining=100.0, beta_scale=0.3)
            sc_segym = _score(a_segym)
            if sc_segym < best_score:
                best_seq = a_segym.copy()
                best_score = sc_segym
                print(f"  NEW BEST: {best_score:.10f} [symmetric]")

    # === Phase 6: Active set round 2 ===
    print(f"\n[P6] Active set round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        pol_deadline2 = min(deadline - 150, time.monotonic() + 120)
        a_eq2, sc_eq2 = _active_set_equalize(best_seq, pol_deadline2, min_time_remaining=50.0)
        if sc_eq2 < best_score:
            best_seq = a_eq2.copy()
            best_score = sc_eq2
            print(f"  NEW BEST: {best_score:.10f} [active_set_2]")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([n, 16384, 24576, 12288, 8192, 32768]), reverse=True)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")
        if time.monotonic() < deadline - 30:
            a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=0.5)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n]")

    # === Phase 8: Final L-BFGS polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 5 - time.monotonic()
    if remaining > 20:
        a_final = _run_lbfgs(best_seq, deadline, min_time_remaining=5.0, beta_scale=0.1)
        sc_final = _score(a_final)
        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final
            print(f"  NEW BEST: {best_score:.10f} [final_lbfgs]")

    # === Phase 9: Final active set micro-polish ===
    if time.monotonic() < deadline - 2:
        a_fin, sc_fin = _active_set_equalize(best_seq, deadline - 1, min_time_remaining=1.0)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_active]")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc = _score(final)
        print(f"FINAL: score={sc:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
