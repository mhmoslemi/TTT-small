def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_fast_len(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lp_direction(f, M, active_indices, timeout=10.0):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for k in active_indices.
    Returns the optimal g (or None if infeasible/timeout)."""
    n = len(f)
    K = len(active_indices)
    
    # Build constraint matrix A (K x n)
    # A[i, l] = f[k_i - l] if 0 <= k_i - l <= n-1
    A = np.zeros((K, n))
    for i in range(K):
        k = active_indices[i]
        l_lo = max(0, k - n + 1)
        l_hi = min(n - 1, k)
        if l_lo <= l_hi:
            indices = np.arange(l_lo, l_hi + 1)
            A[i, indices] = f[k - indices]
    
    # LP: min -sum(g) s.t. A@g <= M, g >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(K, M + 1e-10)  # Small slack for numerical feasibility
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.status == 0:  # Optimal
            g = res.x
            if g.sum() > 1e-10:
                return g
    except Exception as e:
        print(f"    LP error: {e}")
    
    return None


def _lp_iteration(a, K=100, timeout=5.0):
    """One LP-based iteration. Returns improved sequence or None."""
    n = len(a)
    c = _full_convolve(a, a)
    M = float(c.max())
    
    # Select active peaks
    c_idx = np.argsort(c)[::-1]
    active = c_idx[:min(K, 2*n-1)]
    
    # Solve LP
    g = _lp_direction(a, M, active, timeout=timeout)
    if g is None:
        return None
    
    # Rescale g to have same sum as a
    s_a = a.sum()
    s_g = g.sum()
    if s_g < 1e-15:
        return None
    g = g * (s_a / s_g)
    
    # Line search: find best t in [0, 1]
    best_t = 0
    best_sc = _score(a)
    for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try = a_try / s
        sc = _score(a_try)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    
    if best_t > 0:
        a_new = (1 - best_t) * a + best_t * g
        a_new = np.maximum(a_new, 0)
        return a_new / a_new.sum()
    
    return None

def _lbfgs_smooth(a_init, deadline, min_time_remaining=30.0, top_k=None):
    """L-BFGS optimization using smooth approximation in log-space."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted)-1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)
    
    # Beta schedule: geometric
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01,
                                   0.005, 0.002, 0.001, 0.0005, 0.0002,
                                   0.0001, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        def obj(z):
            val, _ = _smooth_obj_grad(z, beta, n, top_k)
            return val
        
        def grad(z):
            _, g = _smooth_obj_grad(z, beta, n, top_k)
            return g
        
        # Estimate max iterations
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 15))))
        
        try:
            result = minimize(obj, z, jac=grad, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'ftol': 1e-16,
                                      'gtol': 1e-14, 'maxcor': 50, 'maxls': 40})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-12:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _nesterov_phase(a_init, deadline, min_time_remaining=30.0, K=50, seed=0):
    """Nesterov-accelerated subgradient method."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    momentum = 0.9
    lr = 0.01
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
        
        # Active set
        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]
        
        # Subgradient
        grad = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                idx = k - np.arange(j_lo, j_hi + 1)
                grad[j_lo:j_hi+1] += a[idx]
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_lo2 <= j_hi2:
                idx2 = np.arange(j_lo2, j_hi2 + 1) - k
                grad[j_lo2:j_hi2+1] += a[idx2]
        
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break
        
        v = momentum * v - (lr / (gnorm + 1e-18)) * grad_t * 0.1
        a_cand = a + v
        a_proj = _project_simplex(a_cand)
        
        c_new = _full_convolve(a_proj, a_proj)
        sc_new = 2.0 * n * float(c_new.max())
        
        if sc_new < current_sc:
            a = a_proj
            lr = min(lr * 1.01, 0.05)
        else:
            v = 0.5 * v
            lr *= 0.95
            a = a_proj
        
        if no_improve > 200 and no_improve % 100 == 0:
            perturb = 0.005 * rng.standard_normal(n)
            a_pert = _project_simplex(a + perturb)
            sc_pert = 2.0 * n * float(_full_convolve(a_pert, a_pert).max())
            if sc_pert < best_sc:
                a = a_pert
                best_a = a.copy()
                best_sc = sc_pert
                no_improve = 0
            v = np.zeros(n)
    
    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)**0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n
