import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_normalized(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score(a):

def _compute_gradient(a, active_indices, n):

    w = np.zeros(2 * n - 1)
    w[active_indices] = 1.0 / len(active_indices)
    a_rev = a[::-1]
    h = _full_convolve(w, a_rev)
    grad = 2.0 * h[n - 1:2 * n - 1]
    return grad


def _project_simplex(v):
    """Project v onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

        else:
            no_improve += 1

def _optimize_active_set_qp(a_init, deadline, min_time_remaining=30.0):

    Exploits the fact that H_k has bounded eigenvalues for efficient line search."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()

    best_a = a.copy()

    lr = 0.15
    no_improve = 0

        it += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-16:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            a = best_a.copy()
            K = max(3, K // 2)
            lr = min(0.3, lr * 1.5)
            no_improve = 0
            if no_improve > 300:
                break
        
        # Adaptive K: tighten as we converge
        if it > 200:
            K = max(3, min(K, 10))
        
        top_k = np.argsort(c)[::-1][:K]
        
        # Compute average gradient over active set
        grad = _compute_gradient(a, top_k, n)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Line search with quadratic model
        # The objective along the path a - t*grad is:
        # f(t) = max_k c_k(a - t*grad) ≈ c_k(a) - 2t*(grad c_k · grad) + 2t^2*(grad^T H_k grad)/2
        # For the active set: f(t) ≈ c_max - 2t*G + 2t^2*H where G = avg(grad_k · grad), H = avg(grad^T H_k grad)/2
        # Optimal t ≈ G / H (capped by feasibility)
        
        # Simple backtracking line search
        t = lr / (gnorm ** 0.5 + 1e-10)
        improved = False
        for _ in range(30):

            a_try = np.maximum(a_try, 0.0)

            if s_try < 1e-15:

                continue
            a_try = a_try / s_try

            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-16:
                a = a_try
                improved = True
                break
            t *= 0.6
        
        if not improved:
            lr *= 0.8
        else:
            lr = min(0.3, lr * 1.02)
    
    return best_a, best_sc

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

            print(f"  Fine SG: no improvement (best={best_score:.10f})")
    
    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([n, 16384, 12288, 24576, 8192, 32768]), reverse=True)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 20:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")
        # Quick optimization at this n
        if time.monotonic() < deadline - 15:
            a_opt, sc_opt = _optimize_momentum(a_init, deadline - 15, min_time_remaining=10.0)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n]")
    
    # === Phase 7: Final ultra-fine ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 5 - time.monotonic()
    if remaining > 5:
        a_fin, sc_fin = _optimize_active_set_qp(best_seq, deadline - 3, min_time_remaining=3.0)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_qp]")
    
    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]
    
    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
