import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))

    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))

def _banded_hmv(active_set, delta, n):
    """Compute H * delta where H = sum_{k in active_set} Hessian(q_k).
    Hessian(q_k) for k>0 has 1s at (i,i+k) and (i+k,i).
    Hessian(q_0) = 2*I.
    We use the average: (1/|S|) * sum."""
    result = np.zeros(n)
    inv_K = 1.0 / len(active_set)
    for k in active_set:

        else:
            if k < n:
                result[:n-k] += inv_K * delta[k:]
                result[k:] += inv_K * delta[:n-k]
    return result

        # Line search: try Newton step first, fall back to gradient step
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.01, 0.005]:
            a_try = a + alpha * d_newton
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break

        if not improved:
            # Fall back to gradient step
            for alpha in [0.1 / (gnorm + 1e-15), 0.05 / (gnorm + 1e-15),
                          0.01 / (gnorm + 1e-15), 0.005 / (gnorm + 1e-15)]:
                a_try = a - alpha * grad
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-13:
                    a = a_try
                    improved = True
                    break

        if not improved:
            if no_improve > 50:
                break

    return best_a, best_sc


def _spectral_perturb(a, rng, strength=0.05):
    """Perturb in the frequency domain to explore different spectral shapes."""
    n = len(a)
    N = _next_pow2(2 * n)
    A = np.fft.rfft(a, N)
    # Add noise to the amplitude (keep phase mostly)
    amp = np.abs(A)
    phase = np.angle(A)
    amp_noisy = amp * (1.0 + strength * rng.standard_normal(len(amp)))
    amp_noisy = np.maximum(amp_noisy, 0.0)
    A_noisy = amp_noisy * np.exp(1j * phase)
    a_new = np.fft.irfft(A_noisy, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _spatial_perturb(a, rng, strength=0.03):
    """Spatial perturbation with structured (low-frequency) component."""
    n = len(a)
    t = np.arange(n) / n
    # Low-frequency structured perturbation
    k1 = rng.integers(1, 5)
    k2 = rng.integers(5, 15)
    structured = (0.5 * np.sin(2 * np.pi * k1 * t + rng.uniform(0, 2*np.pi)) +
                  0.3 * np.cos(2 * np.pi * k2 * t + rng.uniform(0, 2*np.pi)))
    noise = 0.2 * rng.standard_normal(n)
    perturbation = structured + noise
    perturbation = perturbation - perturbation.mean()
    z = np.log(np.maximum(a, 1e-30))
    z = z + strength * perturbation
    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine_sq":
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * (t - 0.5)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flattop_like":
        # Flat-top window approximation
        t = np.linspace(0, 1, n)
        a = 1.0 - 1.93 * np.cos(2*np.pi*t) + 1.29 * np.cos(4*np.pi*t) - 0.39 * np.cos(6*np.pi*t) + 0.09 * np.cos(8*np.pi*t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "broad_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0, max_beta_stages=12):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003) * beta_scale

    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015,
        0.0075, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001
    ])[:max_beta_stages]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # Scale max iterations based on remaining time and n
        est_time_per_iter = 0.003 * max(1.0, n / 16384.0)
        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 14))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _run_lbfgs_multi_start(a_init, deadline, min_time_remaining=30.0, n_starts=3, rng=None):
    """Run L-BFGS from multiple slightly perturbed starting points."""
    if rng is None:
        rng = np.random.default_rng(0)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    for _ in range(n_starts):
        if time.monotonic() + min_time_remaining > deadline:
            break
        a_pert = _spatial_perturb(a_init, rng, strength=0.01)
        a_opt = _run_lbfgs(a_pert, deadline, min_time_remaining=min_time_remaining,
                           beta_scale=0.5, max_beta_stages=6)
        sc = _score(a_opt)
        if sc < best_sc:
            best_a = a_opt
            best_sc = sc
    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining:
    - L-BFGS with adaptive log-sum-exp surrogate
    - Banded CG Newton steps for active-set equalization
    - Spectral and spatial perturbations for exploration
    - Multi-resolution search
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}", flush=True)
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "sine")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, sine n={n_parent}, score={best_score:.10f}", flush=True)

    # === Phase 1: Quick L-BFGS refinement ===
    t_phase = time.monotonic()
    print(f"\n[P1] L-BFGS refinement (t={t_phase - t_start:.1f}s)", flush=True)
    a_p1 = _run_lbfgs(best_seq, deadline, min_time_remaining=600.0, beta_scale=1.0, max_beta_stages=10)
    sc_p1 = _score(a_p1)
    if sc_p1 < best_score:
        best_seq = a_p1.copy()
        best_score = sc_p1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_r1]", flush=True)
    else:
        print(f"  No improvement: {sc_p1:.10f} vs {best_score:.10f}", flush=True)

    # === Phase 2: Banded CG active set equalization ===
    t_phase = time.monotonic()
    print(f"\n[P2] Banded CG equalization (t={t_phase - t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 400:
        cg_deadline = min(deadline - 400, time.monotonic() + 200)
        a_cg, sc_cg = _active_set_equalize_cg(best_seq, cg_deadline, min_time_remaining=30.0)
        if sc_cg < best_score:
            best_seq = a_cg.copy()
            best_score = sc_cg
            print(f"  NEW BEST: {best_score:.10f} [banded_cg]", flush=True)
        else:
            print(f"  No improvement: {sc_cg:.10f}", flush=True)

    # === Phase 3: Spectral perturbation exploration ===
    t_phase = time.monotonic()
    print(f"\n[P3] Spectral exploration (t={t_phase - t_start:.1f}s)", flush=True)
    for trial in range(5):
        if time.monotonic() > deadline - 300:
            break
        strength = 0.02 + 0.01 * trial
        a_spec = _spectral_perturb(best_seq, rng, strength=strength)
        a_opt = _run_lbfgs(a_spec, deadline, min_time_remaining=250.0,
                           beta_scale=0.7, max_beta_stages=8)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [spec_trial={trial}, s={strength:.3f}]", flush=True)
        else:
            print(f"  Spec trial {trial}: {sc_opt:.10f} (no improvement)", flush=True)

    # === Phase 4: Spatial perturbation exploration ===
    t_phase = time.monotonic()
    print(f"\n[P4] Spatial exploration (t={t_phase - t_start:.1f}s)", flush=True)
    for trial in range(4):
        if time.monotonic() > deadline - 200:
            break
        strength = 0.02 + 0.015 * trial
        a_spat = _spatial_perturb(best_seq, rng, strength=strength)
        a_opt = _run_lbfgs(a_spat, deadline, min_time_remaining=180.0,
                           beta_scale=0.5, max_beta_stages=7)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [spat_trial={trial}, s={strength:.3f}]", flush=True)
        else:
            print(f"  Spat trial {trial}: {sc_opt:.10f} (no improvement)", flush=True)

    # === Phase 5: Another banded CG round ===
    t_phase = time.monotonic()
    print(f"\n[P5] Banded CG round 2 (t={t_phase - t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 150:
        cg2_deadline = min(deadline - 150, time.monotonic() + 100)
        a_cg2, sc_cg2 = _active_set_equalize_cg(best_seq, cg2_deadline, min_time_remaining=30.0)
        if sc_cg2 < best_score:
            best_seq = a_cg2.copy()
            best_score = sc_cg2
            print(f"  NEW BEST: {best_score:.10f} [banded_cg2]", flush=True)

    # === Phase 6: Multi-n exploration ===
    t_phase = time.monotonic()
    print(f"\n[P6] Multi-n search (t={t_phase - t_start:.1f}s)", flush=True)
    n_current = len(best_seq)
    n_candidates = sorted(set([n_current, 16384, 24576, 32768, 12288, 8192]), reverse=True)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_current:
            continue
        # Skip if too large (time concern)
        if n_try > 32768:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}", flush=True)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=60.0,
                           beta_scale=0.8, max_beta_stages=6)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n]", flush=True)
        else:
            print(f"  n={n_try}: {sc_opt:.10f} (no improvement)", flush=True)

    # === Phase 7: Try different initial profiles at current best n ===
    t_phase = time.monotonic()
    print(f"\n[P7] Profile exploration (t={t_phase - t_start:.1f}s)", flush=True)
    n_best = len(best_seq)
    profiles = ["flat_top_like", "broad_taper", "cosine_sq"]
    for prof_name in profiles:
        if time.monotonic() > deadline - 50:
            break
        a_prof = _make_profile(n_best, prof_name)
        # Blend with current best to stay in neighborhood
        a_blend = 0.5 * best_seq + 0.5 * a_prof
        a_blend = np.maximum(a_blend, 0)
        a_blend = a_blend / a_blend.sum()
        a_opt = _run_lbfgs(a_blend, deadline, min_time_remaining=40.0,
                           beta_scale=0.5, max_beta_stages=5)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [profile={prof_name}]", flush=True)
        else:
            print(f"  Profile {prof_name}: {sc_opt:.10f} (no improvement)", flush=True)

    # === Phase 8: Final L-BFGS + CG polish ===
    t_phase = time.monotonic()
    print(f"\n[P8] Final polish (t={t_phase - t_start:.1f}s)", flush=True)
    remaining = deadline - 10 - time.monotonic()
    if remaining > 20:
        a_final = _run_lbfgs(best_seq, deadline, min_time_remaining=5.0,
                             beta_scale=0.2, max_beta_stages=5)
        sc_final = _score(a_final)
        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final
            print(f"  NEW BEST: {best_score:.10f} [final_lbfgs]", flush=True)

    if time.monotonic() < deadline - 5:
        a_fin2, sc_fin2 = _active_set_equalize_cg(best_seq, deadline - 3, min_time_remaining=3.0)
        if sc_fin2 < best_score:
            best_seq = a_fin2.copy()
            best_score = sc_fin2
            print(f"  NEW BEST: {best_score:.10f} [final_cg]", flush=True)

    # Final validation
    print(f"\n{'=' * 60}", flush=True)
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s", flush=True)
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}", flush=True)
        except Exception as e:
            print(f"evaluate_sequence error: {e}", flush=True)
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
