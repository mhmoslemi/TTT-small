```python
    n = len(a)
    # Find active peaks (those within tolerance of max)
    c_max = c.max()
    tol = c_max * 0.01
    active = c >= (c_max - tol)
    
    # Gradient: for each active peak k, g[i] += a[k-i] (for valid k-i)
    g = np.zeros(n)
    for k in np.where(active)[0]:
        for i in range(max(0, k-n+1), min(n, k+1)):
            j = k - i
            g[i] += a[j]
    g *= 2.0  # factor of 2 from the concatenation
    
    # Center the gradient (since sum(a) is fixed)
    g -= g.mean()
    
    # Mirror descent update
    log_a = np.log(np.maximum(a, 1e-30))
    log_a -= eta * g
    a_new = np.exp(log_a - log_a.max())
    a_new /= a_new.sum()
    return a_new
```

```python
def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
```

```python
"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
- Adam optimizer on log-scale parameters with adaptive learning rates
- Mirror descent (multiplicative weight updates) for natural simplex handling

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1: returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c=conv(a,a). w has length 2n-1."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _grad_single_peak(a, k_peak):
    """Gradient of c[k_peak] = conv(a,a)[k_peak] w.r.t. a. Factor of 2 included."""
    n = len(a)
    w = np.zeros(2 * n - 1)
    w[k_peak] = 1.0
    return _grad_maxconv(a, w)


def _adaptive_K(c, n):
    """Choose K based on how flat the peak region is."""
    c_max = c.max()
    if c_max < 1e-30:

    # Count peaks within 2% of max

    elif n_close <= 10:

    elif n_close_5 <= 50:

    else:

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=False):
    """L-BFGS with adaptive top-K smoothing."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score_norm(a)
    current_a = a.copy()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a

    # Initial beta from gap
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    if aggressive:
        schedule = [
            (0.5, 5), (0.3, 3), (0.1, 3), (0.05, 2), (0.02, 2),
            (0.01, 1), (0.005, 1), (0.002, 1),
        ]
    else:
        schedule = [
            (1.0, 20), (0.7, 15), (0.5, 10), (0.3, 8),
            (0.2, 5), (0.1, 4), (0.05, 3), (0.02, 2),
            (0.01, 2), (0.005, 1), (0.002, 1),
        ]

    for beta_mult, K in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        beta = base_beta * beta_mult
        K = max(1, min(K, 2 * n - 1))

        # Define objective and gradient

            a_in = a_in / a_in.sum()
            c_in = _full_convolve(a_in, a_in)
            c_max_in = float(c_in.max())
            if K < len(c_in):
                top_idx = np.argpartition(c_in, -K)[-K:]
                c_top = c_in[top_idx]
            else:
                top_idx = np.arange(len(c_in))
                c_top = c_in
            shifted = (c_top - c_max_in) / beta
            shifted = np.clip(shifted, -700, 0)
            e = np.exp(shifted)
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                return 2.0 * n * c_max_in
            return 2.0 * n * (c_max_in + beta * np.log(s))

        def grad(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            c_in = _full_convolve(a_in, a_in)
            c_max_in = float(c_in.max())

                c_top = c_in
            shifted = (c_top - c_max_in) / beta
            shifted = np.clip(shifted, -700, 0)
            e = np.exp(shifted)
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros(2 * n - 1)
                w[int(np.argmax(c_in))] = 1.0
            else:
                e_full = np.zeros(2 * n - 1)
                e_full[top_idx] = e
                w = e_full / s
            ga = _grad_maxconv(a_in, w)
            W = float(np.dot(a_in, ga))
            grad_z = a_in * (ga - W)
            return grad_z

        try:

            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:

def _adam_optimize(a_init, deadline, min_time_remaining=30.0):
    """Adam optimizer on log-scale parameters, targeting the max peak."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score_norm(a)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.max()
    z = z - z.mean()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.005
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    K = 5  # Top-K for smoothing

    stall = 0
    max_stall = 500

    while time.monotonic() + min_time_remaining < deadline:
        a_cur = np.exp(z - z.max())
        a_cur = a_cur / a_cur.sum()
        c = _full_convolve(a_cur, a_cur)
        c_max = float(c.max())
        sc_cur = 2.0 * n * c_max

        if sc_cur < best_sc - 1e-15:
            best_sc = sc_cur
            best_a = a_cur.copy()
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                # Adjust hyperparameters
                lr *= 0.5
                if lr < 1e-6:
                    break
                K = max(1, K // 2)
                stall = 0

        # Compute gradient
        if K < len(c):
            top_idx = np.argpartition(c, -K)[-K:]
        else:
            top_idx = np.arange(len(c))

        w = np.zeros(2 * n - 1)
        c_top = c[top_idx]
        c_top_max = c_top.max()
        beta_local = max(c_top_max * 0.01, 1e-12)
        weights = np.exp((c_top - c_top_max) / beta_local)
        weights = weights / weights.sum()
        w[top_idx] = weights

        g_a = _grad_maxconv(a_cur, w)
        W = float(np.dot(a_cur, g_a))
        g_z = a_cur * (g_a - W)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * g_z
        v = beta2 * v + (1 - beta2) * g_z ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Orthogonalize (keep sum of exp(z) normalization stable)
        if t % 100 == 0:
            z = z - z.mean()

        # Background check
        if t % 50 == 0:
            a_check = np.exp(z - z.max())
            a_check = a_check / a_check.sum()
            sc_check = _score_norm(a_check)
            if sc_check < best_sc - 1e-14:
                best_sc = sc_check
                best_a = a_check.copy()

    return best_a, best_sc

    best_sc = _score_norm(a)

    eta = 0.02
    eta_min = 1e-6
    eta_max = 0.5
    patience = 0
    max_patience = 500

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc_cur = 2.0 * n * c_max

        if sc_cur < best_sc - 1e-15:
            best_sc = sc_cur
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break

        # Gradient: target the top few peaks
        k_adapt = _adaptive_K(c, n)
        K = min(k_adapt, 20)
        top_idx = np.argpartition(c, -K)[-K:]

        # Weight: emphasize the worst peak
        c_top = c[top_idx]
        c_top_max = c_top.max()
        beta_w = max(c_top_max * 0.005, 1e-12)
        weights = np.exp((c_top - c_top_max) / beta_w)
        weights = weights / weights.sum()

        w = np.zeros(2 * n - 1)
        w[top_idx] = weights
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()

        # Mirror descent step: a_new[i] = a[i] * exp(-eta * g[i]) / Z
        log_scale = -eta * g_centered
        log_scale = np.clip(log_scale, -50, 50)  # Prevent overflow
        a_try = a * np.exp(log_scale)
        s_try = a_try.sum()
        if s_try < 1e-10:
            eta *= 0.5
            if eta < eta_min:
                break
            continue
        a_try = a_try / s_try

        # Line search
        improved = False
        for ls in range(15):
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < sc_cur - 1e-16:
                a = a_try
                improved = True
                break
            eta *= 0.7
            if eta < eta_min:
                break
            log_scale = -eta * g_centered
            log_scale = np.clip(log_scale, -50, 50)
            a_try = a_init_orig = a.copy()
            a_try = a * np.exp(log_scale)
            s_try = a_try.sum()
            if s_try < 1e-10:
                break
            a_try = a_try / s_try

        if improved:
            eta = min(eta_max, eta * 1.05)
        else:
            eta = max(eta_min, eta * 1.5)
            patience += 5

    return best_a, best_sc


def _peak_surgery(a_init, deadline, min_time_remaining=30.0):
    """Directly manipulate the top contributing pairs to reduce the worst peak."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score_norm(a)
    patience = 0
    max_patience = 300

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc_cur = 2.0 * n * c_max
        k_peak = int(np.argmax(c))

        if sc_cur < best_sc - 1e-15:
            best_sc = sc_cur
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break

        # Compute contribution of each pair to the peak
        # c[k_peak] = sum_i a[i]*a[k_peak-i]
        i_lo = max(0, k_peak - (n - 1))
        i_hi = min(n - 1, k_peak)
        pairs = []
        for i in range(i_lo, i_hi + 1):
            j = k_peak - i
            if j != i:
                pairs.append((i, j, a[i] * a[j]))
            else:
                pairs.append((i, j, a[i] * a[j]))

        pairs.sort(key=lambda x: -x[2])

        # Reduce top contributors, add to bottom contributors
        n_reduce = max(2, min(len(pairs) // 10, 20))
        n_add = n_reduce

        reduction = c_max * 0.003

        delta = np.zeros(n)
        # Reduce top pairs
        for idx in range(n_reduce):
            i, j, val = pairs[idx]
            if val < 1e-20:
                continue
            amt = reduction / (2 * n_reduce)
            # Reduce proportionally
            frac_i = a[i] / (a[i] + a[j] + 1e-30)
            frac_j = 1 - frac_i
            delta[i] -= amt * frac_i / (a[i] + 1e-30) * a[i]
            delta[j] -= amt * frac_j / (a[j] + 1e-30) * a[j]

        # Add to low contributors (pairs that don't hit the peak)
        # Find indices that minimize contribution to the peak
        contrib_to_peak = np.zeros(n)
        for i in range(n):
            j = k_peak - i
            if 0 <= j < n:
                contrib_to_peak[i] = a[j]

        low_idx = np.argsort(contrib_to_peak)[:n_add]
        add_total = -delta.sum()
        if add_total > 0:
            for idx in low_idx:
                delta[idx] += add_total / n_add

        # Line search
        alpha = 1.0
        improved = False
        for ls in range(25):
            a_try = a + alpha * delta
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < sc_cur - 1e-16:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            patience += 10

    return best_a, best_sc


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    """Very fine subgradient polish with adaptive active set."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score_norm(a)

    lr = 0.003
    no_improve = 0
    tol = 0.003

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc_cur = 2.0 * n * c_max

        if sc_cur < best_sc - 1e-16:
            best_sc = sc_cur
            best_a = a.copy()
            no_improve = 0

        # Active set: peaks within tol of max
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            k_best = int(np.argmax(c))
            active = np.zeros_like(c)
            active[k_best] = 1
            n_active = 1

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break

        alpha = min(lr, 0.3 / (gnorm + 1e-15))
        improved = False
        for ls in range(40):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < sc_cur - 1e-16:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                tol = max(0.00005, tol * 0.7)
                lr *= 0.8
            if no_improve > 500:
                a = best_a.copy()
                tol = max(0.00005, tol * 0.5)
                lr = 0.002
                no_improve = 0
        else:
            tol = max(0.00005, tol * 0.98)
            lr = min(0.008, lr * 1.01)

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flattop":
        # Harris flat-top window coefficients
        a_k = [0.3635811, 0.4891721, 0.1365985, 0.0116814, 0.0003744]
        t = np.linspace(0, 1, n)
        a = np.zeros(n)
        for k, ak in enumerate(a_k):
            a += ak * np.cos(2 * np.pi * k * t - np.pi * k / 2)
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * (t - 0.5)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS, Adam, mirror descent,
    peak surgery, and fine polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score_norm(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        best_seq = parent.copy()
        best_score = _score_norm(parent)
        print(f"No parent, tapered n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Aggressive L-BFGS ===
    print(f"\n[P1] L-BFGS aggressive (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 600, t_start + 350)
    a1 = _run_lbfgs(best_seq, p1_deadline, min_time_remaining=600, aggressive=True)
    sc1 = _score_norm(a1)
    print(f"  L-BFGS1: {sc1:.10f} (time={time.monotonic()-t_start:.1f}s)")
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f}")

    # === Phase 2: Adam optimizer ===
    print(f"\n[P2] Adam (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 500, t_start + 600)
    a2, sc2 = _adam_optimize(best_seq, p2_deadline, min_time_remaining=500)
    print(f"  Adam: {sc2:.10f} (time={time.monotonic()-t_start:.1f}s)")
    if sc2 < best_score:
        best_seq = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f}")

    # === Phase 3: Mirror descent ===
    print(f"\n[P3] Mirror descent (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 400, t_start + 850)
    a3, sc3 = _mirror_descent(best_seq, p3_deadline, min_time_remaining=400)
    print(f"  Mirror: {sc3:.10f} (time={time.monotonic()-t_start:.1f}s)")
    if sc3 < best_score:
        best_seq = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f}")

    # === Phase 4: Peak surgery ===
    print(f"\n[P4] Peak surgery (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 300, t_start + 1050)
    a4, sc4 = _peak_surgery(best_seq, p4_deadline, min_time_remaining=300)
    print(f"  Surgery: {sc4:.10f} (time={time.monotonic()-t_start:.1f}s)")
    if sc4 < best_score:
        best_seq = a4.copy()
        best_score = sc4
        print(f"  NEW BEST: {best_score:.10f}")

    # === Phase 5: L-BFGS round 2 (from improved point) ===
    print(f"\n[P5] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    p5_deadline = min(deadline - 200, t_start + 1300)
    a5 = _run_lbfgs(best_seq, p5_deadline, min_time_remaining=200, aggressive=True)
    sc5 = _score_norm(a5)
    print(f"  L-BFGS2: {sc5:.10f} (time={time.monotonic()-t_start:.1f}s)")
    if sc5 < best_score:
        best_seq = a5.copy()
        best_score = sc5
        print(f"  NEW BEST: {best_score:.10f}")

    # === Phase 6: Multi-restart ===
    print(f"\n[P6] Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    p6_deadline = min(deadline - 100, t_start + 1550)
    n_restarts = 5
    for i in range(n_restarts):
        if time.monotonic() > p6_deadline:
            break
        sigma = [0.02, 0.04, 0.01, 0.06, 0.03][i]
        a_pert = _perturb(best_seq, sigma, rng)
        sub_deadline = min(p6_deadline, time.monotonic() + 40)
        a_r = _run_lbfgs(a_pert, sub_deadline, min_time_remaining=5, aggressive=True)
        sc_r = _score_norm(a_r)
        if sc_r < best_score:
            best_seq = a_r.copy()
            best_score = sc_r
            print(f"  NEW BEST: {best_score:.10f} [restart_{i}] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  restart_{i}: {sc_r:.10f} (no improvement) t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final Adam + polish ===
    print(f"\n[P7] Final optimization (t={time.monotonic()-t_start:.1f}s)")
    p7_deadline = min(deadline - 20, t_start + 1700)
    a7, sc7 = _adam_optimize(best_seq, p7_deadline, min_time_remaining=80)
    if sc7 < best_score:
        best_seq = a7.copy()
        best_score = sc7
        print(f"  Final Adam: {best_score:.10f}")

    # === Phase 8: Fine polish ===
    print(f"\n[P8] Fine polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a8, sc8 = _fine_polish(best_seq, deadline - 2, min_time_remaining=3)
        if sc8 < best_score:
            best_seq = a8.copy()
            best_score = sc8
            print(f"  Final polish: {best_score:.10f}")

    # === Final: Tiny L-BFGS ===
    remaining = deadline - 2 - time.monotonic()
    if remaining > 10:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        beta_fin = max(mb * 0.0001, 1e-10)
        K_fin = max(1, int(n * 0.0005))

        def obj_final(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            c_in = _full_convolve(a_in, a_in)
            c_max_in = float(c_in.max())
            if K_fin < len(c_in):
                top_idx = np.argpartition(c_in, -K_fin)[-K_fin:]
                c_top = c_in[top_idx]
            else:
                c_top = c_in
            shifted = (c_top - c_max_in) / beta_fin
            shifted = np.clip(shifted, -700, 0)
            e = np.exp(shifted)
            s = e.sum()
            if s < 1e-300:
                return 2.0 * n * c_max_in
            return 2.0 * n * (c_max_in + beta_fin * np.log(s))

        def grad_final(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            c_in = _full_convolve(a_in, a_in)
            c_max_in = float(c_in.max())
            if K_fin < len(c_in):
                top_idx = np.argpartition(c_in, -K_fin)[-K_fin:]
                c_top = c_in[top_idx]
            else:
                top_idx = np.arange(len(c_in))
                c_top = c_in
            shifted = (c_top - c_max_in) / beta_fin
            shifted = np.clip(shifted, -700, 0)
            e = np.exp(shifted)
            s = e.sum()
            if s < 1e-300:
                w = np.zeros(2 * n - 1)
                w[int(np.argmax(c_in))] = 1.0
            else:
                e_full = np.zeros(2 * n - 1)
                e_full[top_idx] = e
                w = e_full / s
            ga = _grad_maxconv(a_in, w)
            W = float(np.dot(a_in, ga))
            return a_in * (ga - W)

        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        max_iter = min(5000, max(20, int(remaining / 0.02)))
        try:
            result = minimize(obj_final, z, jac=grad_final, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxcor': 100})
            a_fin = np.exp(result.x - result.x.max())
            a_fin = a_fin / a_fin.sum()
            sc_fin = _score_norm(a_fin)
            if sc_fin < best_score:
                best_seq = a_fin.copy()
                best_score = sc_fin
                print(f"  Final L-BFGS: {best_score:.10f}")
        except Exception:
            pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score_norm(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score_norm(fallback):.10f}")
    return [float(x) for x in fallback]
```
