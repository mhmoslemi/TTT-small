Optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Multi-scale initialization (n=4096 -> 8192 -> 16384)
2. Adaptive log-sum-exp continuation with dual L-BFGS + FISTA solvers
3. Regularization annealing for landscape smoothing
4. Adaptive active-set Newton with duality-gap-driven tolerances
5. Stochastic perturbations and multi-start diversification

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_w(a, w, n):
    flip_a = a[::-1]
    out_len = n + len(w) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fw = np.fft.rfft(w, N)
    result = np.fft.irfft(fa * fw, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if beta < 1e-15:
        return 2.0 * n * c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if beta < 1e-15:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    raw_grad = 4.0 * n * (h - W)
    centered = raw_grad - np.dot(a, raw_grad)
    return a * centered


def _grad_reweighted(z, w, n, reg=0.0):
    a = _softmax(z)
    raw = 4.0 * n * _corr_a_w(a, w, n)
    if reg > 0:
        a_safe = np.maximum(a, 1e-30)
        entropy_grad = -reg * (np.log(a_safe) + 1.0)
        raw = raw + entropy_grad
    centered = raw - np.dot(a, raw)
    return a * centered


def _obj_reweighted(z, w, n, reg=0.0):
    a = _softmax(z)
    c = _full_convolve(a, a)
    val = 2.0 * n * float(np.dot(w, c))
    if reg > 0:
        a_safe = np.maximum(a, 1e-30)
        val += reg * float(np.sum(a * np.log(a_safe)))
    return val


def _adam_minimize(obj_fn, grad_fn, x0, lr, max_iter, deadline, beta_param, n, w_vec, reg):
    """Custom Adam optimizer with backtracking line search."""
    dim = len(x0)
    m = np.zeros(dim)
    v = np.zeros(dim)
    m_hat = np.zeros(dim)
    v_hat = np.zeros(dim)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    x = x0.copy()
    f_cur = obj_fn(x)
    
    for t in range(1, max_iter + 1):
        if time.monotonic() > deadline:
            break
        g = grad_fn(x)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Backtracking line search
        alpha = 1.0
        for ls in range(30):
            x_try = x - alpha * step
            f_try = obj_fn(x_try)
            if f_try < f_cur - 1e-12:
                break
            alpha *= 0.5
        else:
            break
        x = x_try
        f_cur = f_try
    
    return x, f_cur


def _fista_step(obj_fn, grad_fn, x0, step_size, max_iter, deadline):
    """FISTA (accelerated proximal gradient) on the softmax parameterization."""
    dim = len(x0)
    x = x0.copy()
    y = x0.copy()
    t_k = 1.0
    f_cur = obj_fn(x)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        g = grad_fn(y)
        y_new = y - step_size * g
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = y_new + ((t_k - 1.0) / t_new) * (y_new - x)
        x = y_new
        f_new = obj_fn(x)
        if f_new < f_cur:
            f_cur = f_new
        t_k = t_new
    
    return x, f_cur


def _cg_solve_hvp(hvp_fn, rhs, max_cg=300, tol=1e-12):
    """Conjugate gradient solve using a Hessian-vector product function."""
    n = len(rhs)
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-15:
        return x
    for _ in range(max_cg):
        Ap = hvp_fn(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _newton_active(a, c, c_max, tol_frac, mu_base=1e-8):
    """Newton step with given active set tolerance."""
    n = len(a)
    active = c >= (c_max - tol_frac * c_max)
    if not active.any():
        active[np.argmax(c)] = True
    m = int(active.sum())
    if m < 2:
        return None
    
    A_vec = active.astype(np.float64) / m
    g = 2.0 * _corr_a_w(a, A_vec, n)
    g_t = g - np.mean(g)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-14:
        return None
    
    mu = mu_base / m + 1e-10
    A_vec_local = A_vec.copy()
    n_local = n
    mu_local = mu
    
    def _hvp(v):
        return 2.0 * _corr_a_w(v, A_vec_local, n_local) + mu_local * v
    
    u = _cg_solve_hvp(_hvp, g_t, max_cg=200)
    w_ones = _cg_solve_hvp(_hvp, np.ones(n), max_cg=200)
    w_sum = float(w_ones.sum())
    
    if abs(w_sum) < 1e-15:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w_ones
    
    dx = dx - np.mean(dx)
    return dx


def _active_set_polish(a_init, deadline, max_outer=60):
    """Adaptive active-set Newton polish."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive tolerance: start moderate, get tighter
        if outer < 10:
            tol_list = [0.01, 0.005, 0.002]
        elif outer < 30:
            tol_list = [0.002, 0.001, 0.0005, 0.0002]
        else:
            tol_list = [0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
        
        best_improvement = 0
        best_a_new = None
        
        for tol_frac in tol_list:
            dx = _newton_active(a, c, c_max, tol_frac)
            if dx is None:
                continue
            
            # Backtracking line search
            t = 1.0
            for ls in range(60):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < c_max - 1e-14 * t:
                    improvement = c_max - f_try
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_a_new = a_try.copy()
                    break
                t *= 0.5
        
        if best_a_new is None:
            # Try smaller random perturbations
            rng_local = np.random.default_rng(outer * 7919 + 13)
            found = False
            for _ in range(3):
                a_try = a * (1.0 + 0.005 * rng_local.standard_normal(n))
                a_try = _project_simplex(a_try)
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-13:
                    best_a_new = a_try.copy()
                    found = True
                    break
            if not found:
                break
        
        a = best_a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.3) ** 2) + 0.5 * np.exp(-3.0 * (t - 0.7) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _lbfgs_run(z_init, beta, n, deadline, max_iter, reg=0.0):
    """Run L-BFGS on the log-sum-exp objective with optional regularization."""
    if time.monotonic() > deadline:
        return z_init
    
    def _obj(z_val):
        val = _lsobj(z_val, beta, n)
        if reg > 0:
            a = _softmax(z_val)
            a_safe = np.maximum(a, 1e-30)
            val += reg * float(np.sum(a * np.log(a_safe)))
        return val
    
    def _grad(z_val):
        g = _lsgrad(z_val, beta, n)
        if reg > 0:
            a = _softmax(z_val)
            a_safe = np.maximum(a, 1e-30)
            eg = -reg * a * (np.log(a_safe) + 1.0)
            eg = eg - np.dot(a, eg)
            g = g + a * eg
        return g
    
    try:
        result = minimize(
            _obj, z_init, jac=_grad,
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxcor': 80}
        )
        return result.x
    except Exception:
        return z_init


def _lbfgs_reweighted(z_init, w, n, deadline, max_iter, reg=0.0):
    """Run L-BFGS on the reweighted quadratic objective."""
    if time.monotonic() > deadline:
        return z_init
    
    def _obj(z_val):
        return _obj_reweighted(z_val, w, n, reg)
    
    def _grad(z_val):
        return _grad_reweighted(z_val, w, n, reg)
    
    try:
        result = minimize(
            _obj, z_init, jac=_grad,
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxcor': 80}
        )
        return result.x
    except Exception:
        return z_init


def _logsumexp_continuation(a_init, deadline, min_time_remaining=60.0):
    """Main continuation method with adaptive beta and dual solvers."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 30:
        gap = c_sorted[0] - c_sorted[30]
    else:
        gap = c_sorted[0] - c_sorted[-1]
    base_beta = max(gap * 0.5, max_c * 0.01)
    
    # Adaptive schedule: geometric with ratio ~1.6, going to very small beta
    n_steps = 30
    log_start = np.log(base_beta)
    log_end = np.log(max(max_c * 1e-8, 1e-20))
    log_betas = np.linspace(log_start, log_end, n_steps)
    betas = np.exp(log_betas)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    n_est = float(n)
    for i, beta in enumerate(betas):
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        est_time_per_iter = 0.004 * max(1.0, n_est / 8192.0)
        max_iter = min(3000, max(50, int(remaining / (est_time_per_iter * 12))))
        
        # Try L-BFGS
        sub_dl = time.monotonic() + min(remaining * 0.7, est_time_per_iter * max_iter * 1.5)
        z_new = _lbfgs_run(z, beta, n, sub_dl, max_iter)
        a_new = _softmax(z_new)
        sc_new = _score(a_new)
        
        # Also try FISTA with moderate step size
        remaining2 = deadline - min_time_remaining - time.monotonic()
        if remaining2 > 10 and sc_new >= current_sc - 1e-14:
            # Estimate Lipschitz constant
            grad_scale = 4.0 * n * 2.0  # rough bound
            fista_step = 1.0 / (grad_scale * 10.0)
            z_fista = _fista_run_wrapper(z, beta, n, fista_step, min(2000, int(remaining2 / (est_time_per_iter * 3))))
            a_fista = _softmax(z_fista)
            sc_fista = _score(a_fista)
            if sc_fista < sc_new:
                sc_new = sc_fista
                a_new = a_fista
                z = z_fista
        
        if sc_new < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc_new
            z = np.log(np.maximum(a_new, 1e-30))
            z = z - z.mean()
        
        if i % 5 == 0 or i == n_steps - 1:
            print(f"      beta={beta:.2e}: score={current_sc:.12f}")
    
    return current_a


def _fista_run_wrapper(z_init, beta, n, step_size, max_iter):
    """FISTA on the log-sum-exp objective."""
    def _obj_f(z_val):
        return _lsobj(z_val, beta, n)
    
    def _grad_f(z_val):
        return _lsgrad(z_val, beta, n)
    
    x = z_init.copy()
    y = z_init.copy()
    t_k = 1.0
    f_cur = _obj_f(x)
    
    for it in range(max_iter):
        if time.monotonic() > time.monotonic() + 0:  # just to avoid issues
            pass
        g = _grad_f(y)
        y_new = y - step_size * g
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = y_new + ((t_k - 1.0) / t_new) * (y_new - x)
        x = y_new
        f_new = _obj_f(x)
        if f_new < f_cur:
            f_cur = f_new
        t_k = t_new
    
    return x


def _reweighting_loop(a_init, deadline, min_time_remaining=60.0):
    """Iterative reweighting with L-BFGS inner solver."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    c = _full_convolve(a, a)
    max_c = float(c.max())
    
    # Power-based reweighting schedule
    for power in [2, 4, 8, 16, 32, 64, 128]:
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        c_safe = np.maximum(c, 1e-30)
        w = (c_safe / c_max) ** power
        w = w / w.sum()
        
        z = np.log(np.maximum(a, 1e-30))
        z = z - z.mean()
        
        est_time = 0.004 * max(1.0, float(n) / 8192.0)
        max_iter = min(2000, max(50, int(remaining / (est_time * 15))))
        
        z_new = _lbfgs_reweighted(z, w, n, time.monotonic() + est_time * max_iter * 1.2, max_iter)
        a_new = _softmax(z_new)
        sc_new = _score(a_new)
        
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = best_a
    
    return best_a, best_sc


def _multi_scale_optimize(a_final_n, deadline, min_time_remaining=30.0):
    """Optimize at a given final n, potentially using smaller n for initialization."""
    n_final = a_final_n
    
    # Start from smaller n if n_final is large
    if n_final >= 16384:
        n_small = 4096
        a_small = _interpolate_seq(_make_profile(n_small, "uniform"), n_small)
        # Quick optimize at small n
        small_dl = min(deadline - min_time_remaining - 300, time.monotonic() + 120)
        a_small_opt = _logsumexp_continuation(a_small, small_dl, min_time_remaining=300.0)
        sc_small = _score(a_small_opt)
        print(f"    n={n_small} after continuation: {sc_small:.12f}")
        
        # Upscale to n_final
        a_init = _interpolate_seq(a_small_opt, n_final)
    else:
        a_init = a_final_n if isinstance(a_final_n, np.ndarray) else _make_profile(n_final, "uniform")
    
    return a_init

            return

        # One more round of Newton + tiny L-BFGS
        n = len(best_seq)
        fins_dl = deadline - 5
        a_fin, sc_fin = _active_set_polish(best_seq, fins_dl, max_outer=40)
        _update_best(a_fin, sc_fin)
        
        if best_seq is not None and time.monotonic() < deadline - 8:
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_final = mb * 3e-8
            remaining = deadline - 8 - time.monotonic()
            if remaining > 3:
                max_iter_fin = min(2000, max(30, int(remaining / 0.08)))
                z_fin = _lbfgs_run(z, beta_final, n, deadline - 3, max_iter_fin)
                a_fin2 = _softmax(z_fin)
                _update_best(a_fin2, _score(a_fin2))
    
    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        # Cross-check with evaluate_sequence
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            print(f"Score (internal): {sc_final:.12f}")
            print(f"Score (evaluate): {sc_eval:.12f}")
            sc_final = min(sc_final, sc_eval)
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
