Key improvements: corrected gradient computation, adaptive active-set KKT-Newton,

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    """Correct gradient of sum_k W_k * conv(a,a)_k w.r.t. a."""
    W_flip = W[::-1]

    return 2.0 * v[n - 1:2 * n - 1][::-1]


def _hankel_matvec(v, W, n):
    """Compute H*v where H[i,j] = 2*W[i+j]."""
    v_flip = v[::-1]
    conv = _full_convolve(v_flip, W)
    return 2.0 * conv[n - 1:2 * n - 1][::-1]


def _cg_solve_hankel(W, n, rhs, max_iter=300, tol=1e-14):
    """Solve H*x = rhs using CG, where H[i,j] = 2*W[i+j]."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-300:

    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)

            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new):
            break
        if rs < tol * tol * rs0:
            break
        p = (rs_new / rs) * p + r
        rs = rs_new
    return x


def _newton_kkt(a_init, deadline, max_rounds=300, k_schedule=None):
    """KKT Newton method with corrected gradient and adaptive K."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    if k_schedule is None:
        k_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive K: use more active lags if the top is flat
        c_sorted = np.sort(c)[::-1]
        K = k_schedule[rnd % len(k_schedule)]
        K = min(K, 2 * n - 1)

        # Adaptive: if top is very flat, increase K
        if K < len(c_sorted) - 1:
            gap = c_sorted[0] - c_sorted[K]
            if gap < 1e-6 * c_max and K < 2 * n - 1:
                K = min(K * 2, 2 * n - 1)

        idx = np.argpartition(c, -K)[-K:]
        W = np.zeros(2 * n - 1)
        W[idx] = 1.0 / K

        g = _grad_correct(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        # Newton step with Hessian H[i,j] = 2*W[i+j]
        Hg = _cg_solve_hankel(W, n, g_t, max_iter=200)
        H1 = _cg_solve_hankel(W, n, np.ones(n), max_iter=200)
        H1_denom = float(np.ones(n) @ H1)

        if abs(H1_denom) < 1e-25 or not np.isfinite(H1_denom):
            dx = g_t / max(gn, 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / H1_denom
            dx = Hg - lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = g_t / max(gn, 1e-30)

        # Line search
        alpha = 1.0
        step_done = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1.0 - 0.005 * alpha) - 1e-18:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not step_done:
            # Fallback: subgradient step (note: g_t is already the descent direction
            # for maximizing, so we subtract it for minimization of max)
            for trial in [0.05, 0.02, 0.01, 0.005, 0.001, 0.0005]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-18:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if not step_done and no_improve > 10:
            break

    return best_a, best_sc


def _fista_subgrad(a_init, deadline, max_steps=5000):
    """FISTA-accelerated subgradient method with adaptive active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.01
    t_ista = 1.0
    y = z.copy()
    v = np.zeros(n)

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(y)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set
        tol_frac = 0.001
        active = c >= (c_max - tol_frac * c_max)
        if active.sum() < 5:
            K = min(20, len(c))
            idx = np.argpartition(c, -K)[-K:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True

        W = np.zeros_like(c)
        W[active] = 1.0 / active.sum()

        g = _grad_correct(a, W, n)
        g_t = g - np.mean(g)
        grad_norm = np.linalg.norm(g_t)
        if grad_norm < 1e-15:
            break

        # FISTA update
        z_new = y - lr * g_t / max(grad_norm, 1e-30)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_ista * t_ista))
        y = z_new + ((t_ista - 1.0) / t_new) * (z_new - z)
        z = z_new
        t_ista = t_new

        if i % 100 == 99:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.1, 0.05)
            else:
                patience += 1
                lr *= 0.85
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    y = z.copy()
                    t_ista = 1.0
                    patience = 0
                    lr = 0.01

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _smooth_max_lbfgs(a_init, deadline, min_remaining=60.0):
    """L-BFGS on smooth max approximation with adaptive temperature."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    beta = max(gap * 0.1, max_c * 0.0005)

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    current_a = a_init.copy()
    current_sc = _score(current_a)

    while beta > max_c * 1e-12:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break

        scale = max(1.0, n / 8192.0)
        est_iter_time = 0.005 * scale
        max_iter = min(3000, max(20, int(remaining / (est_iter_time * 50))))

        try:
            def obj_fn(z_vec, beta_val=beta, n_val=n):
                a_vec = _softmax(z_vec)
                c_vec = _full_convolve(a_vec, a_vec)
                c_max_vec = float(c_vec.max())
                shifted = (c_vec - c_max_vec) / beta_val
                e = np.exp(np.clip(shifted, -700, 0))
                s = e.sum()
                if s < 1e-300 or not np.isfinite(s):
                    return 2.0 * n_val * c_max_vec
                return 2.0 * n_val * (c_max_vec + beta_val * np.log(s))

            def grad_fn(z_vec, beta_val=beta, n_val=n):
                a_vec = _softmax(z_vec)
                c_vec = _full_convolve(a_vec, a_vec)
                c_max_vec = float(c_vec.max())
                shifted = (c_vec - c_max_vec) / beta_val
                e = np.exp(np.clip(shifted, -700, 0))
                s = e.sum()
                if s < 1e-300 or not np.isfinite(s):
                    w = np.zeros_like(c_vec)
                    w[int(np.argmax(c_vec))] = 1.0
                else:
                    w = e / s
                g = _grad_correct(a_vec, w, n_val)
                return 2.0 * n_val * g

            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 80, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

        if new_gap < 1e-8 * c_max_new:
            beta *= 0.1
        elif new_gap < 1e-5 * c_max_new:
            beta *= 0.2
        else:
            beta *= 0.4
        beta = min(beta, c_max_new * 0.0001)

    return current_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        taper = int(n * 0.1)
        if taper > 0:
            edge = np.linspace(0, 1, taper) ** 2
            a[:taper] = edge
            a[-taper:] = edge[::-1]
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate():
    t_start = time.monotonic()
    deadline = t_start + 1770.0
    rng = np.random.default_rng(42)

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"imp={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                par_sc = _score(parent)
                print(f"Parent: n={len(parent)}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        parent = _make_profile(16384, "tapered")
        print(f"No parent, using tapered n=16384")
        update_best(parent, _score(parent), "init")

    n0 = len(parent)

    # Phase 1: KKT-Newton
    print(f"\n[Phase 1] KKT-Newton (t={time.monotonic() - t_start:.1f}s)")
    t_p1_end = min(deadline - 900, t_start + 250)
    k_sch = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]
    a_kkt, sc_kkt = _newton_kkt(parent, t_p1_end, max_rounds=400, k_schedule=k_sch)
    update_best(a_kkt, sc_kkt, "kkt_1")

    # Phase 2: FISTA subgradient
    print(f"\n[Phase 2] FISTA subgrad (t={time.monotonic() - t_start:.1f}s)")
    t_p2_end = min(deadline - 700, t_start + 450)
    a_sg, sc_sg = _fista_subgrad(best_seq, t_p2_end, max_steps=10000)
    update_best(a_sg, sc_sg, "fista")

    # Phase 3: L-BFGS smooth max
    print(f"\n[Phase 3] L-BFGS smooth max (t={time.monotonic() - t_start:.1f}s)")
    t_p3_end = min(deadline - 500, t_start + 750)
    a_lb = _smooth_max_lbfgs(best_seq, t_p3_end, min_remaining=400.0)
    update_best(a_lb, _score(a_lb), "lbfgs_1")

    # Phase 4: Cycled Newton + FISTA
    print(f"\n[Phase 4] Cycle (t={time.monotonic() - t_start:.1f}s)")
    cycle_end = min(deadline - 300, t_start + 1250)
    for cyc in range(10):
        if time.monotonic() > cycle_end:
            break
        t_sub = min(cycle_end, time.monotonic() + 50)
        a_c1, sc_c1 = _fista_subgrad(best_seq, t_sub, max_steps=3000)
        update_best(a_c1, sc_c1, f"cyc_fista_{cyc}")

        t_new = min(cycle_end, time.monotonic() + 60)
        k_sch2 = [3, 5, 10, 20, 50, 100, 300, 1000, 3000]
        a_c2, sc_c2 = _newton_kkt(best_seq, t_new, max_rounds=150, k_schedule=k_sch2)
        update_best(a_c2, sc_c2, f"cyc_np_{cyc}")

        t_lb = min(cycle_end, time.monotonic() + 40)
        if t_lb - time.monotonic() > 10:
            a_c3 = _smooth_max_lbfgs(best_seq, t_lb, min_remaining=5.0)
            update_best(a_c3, _score(a_c3), f"cyc_lbfgs_{cyc}")

    # Phase 5: Multi-n exploration
    print(f"\n[Phase 5] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_options = []
    for n_try in [n0 // 4, n0 // 2, n0, n0 * 2, 8192, 16384, 32768, 65536]:
        if n_try not in n_options and 2048 <= n_try <= 65536:
            n_options.append(n_try)

    for n_try in sorted(n_options):
        if time.monotonic() > deadline - 150:
            break
        a_res = _resample(best_seq, n_try)
        sc_res = _score(a_res)
        if sc_res < best_score:
            update_best(a_res, sc_res, f"resample_n{n_try}")

        if time.monotonic() < deadline - 60 and n_try <= 32768:
            t_end = min(deadline - 40, time.monotonic() + 30)
            a_opt = _smooth_max_lbfgs(a_res, t_end, min_remaining=5.0)
            update_best(a_opt, _score(a_opt), f"opt_n{n_try}")

    # Phase 6: Restarts with perturbation
    print(f"\n[Phase 6] Restarts (t={time.monotonic() - t_start:.1f}s)")
    for restart in range(6):
        if time.monotonic() > deadline - 30:

        sigma = 0.01 * (restart + 1)

                                 k_schedule=[5, 10, 50, 200, 1000])
        update_best(a_r, sc_r, f"restart_{restart}")

    # Phase 7: Final ultra-fine polish
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        k_final = [1, 2, 3, 5, 10, 20, 50, 100, 200]
        a_fin, sc_fin = _newton_kkt(best_seq, deadline - 2, max_rounds=200,
                                     k_schedule=k_final)
        update_best(a_fin, sc_fin, "final_np")

    if time.monotonic() < deadline - 3:
        a_fin2 = _smooth_max_lbfgs(best_seq, deadline - 1, min_remaining=2.0)
        update_best(a_fin2, _score(a_fin2), "final_lbfgs")

    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
