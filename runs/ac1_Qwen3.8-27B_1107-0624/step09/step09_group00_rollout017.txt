def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    """Full convolution via FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _auto_conv(a):
    """Autocorrelation (full convolution of a with itself)."""
    return _full_convolve(a, a)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _auto_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _project_simplex(v):
    """Project v onto {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0.0)

    if kind == "uniform":

    elif kind == "triangle":

        t_k = 1.0

        v = a.copy()

            v_new = _project_simplex(y - step * grad_y)

            # FISTA update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = v_new + ((t_k - 1.0) / t_new) * (v_new - v)
            y = np.maximum(y, 0.0)
            # Renormalize y to stay close to simplex
            sy = y.sum()
            if sy > 1e-10:
                y = y / sy * (1.0)  # keep sum close to 1, will project properly
            v = v_new
            t_k = t_new

            # Check convergence
            if it > 50:
                diff = np.linalg.norm(v_new - v)
                if diff < 1e-12:
                    break

        a = _project_simplex(v)

        # Evaluate
        new_sc = _score(a)
        if new_sc >= best_sc * 0.999:
            stall_count += 1
        else:
            stall_count = 0

        if stall_count > 10:
            tol_frac *= 0.6
            a = best_a.copy()
            stall_count = 0
        else:
            tol_frac *= 0.92

        if tol_frac > tol_end:
            tol_frac = max(tol_end, tol_frac * 0.95)

    return best_a, best_sc


def _peak_equalization(a_init, deadline, min_remaining=30.0):
    """Directly equalize the top peaks of the autocorrelation."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.01
    momentum = 0.8
    v = np.zeros(n)
    no_improve = 0

    while time.monotonic() + min_remaining < deadline:
        c = _auto_conv(a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Find top peaks
        n_top = min(10, len(c))
        top_idx = np.argsort(c)[::-1][:n_top]
        top_c = c[top_idx]
        c_top_max = top_c[0]

        # Weight: emphasize peaks that are significantly above the mean of top peaks
        c_mean_top = float(top_c.mean())
        weights = np.zeros(len(c))
        for i, idx in enumerate(top_idx):
            if top_c[i] > c_mean_top:
                weights[idx] = (top_c[i] - c_mean_top) / (c_top_max - c_mean_top + 1e-15)
        if weights.sum() < 1e-15:
            weights[top_idx[0]] = 1.0
        else:
            weights /= weights.sum()

        # Gradient step
        a_rev = a[::-1]
        wc = _full_convolve(weights, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        # Momentum + projected step
        v = momentum * v - lr * grad
        a_try = _project_simplex(a + v)

        # Backtracking
        sc_try = _score(a_try)
        alpha = 1.0
        improved = False
        for _ in range(30):
            a_try = _project_simplex(a + alpha * v)
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            v = np.zeros(n)
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.7
            if no_improve >= 200:
                break
        else:
            lr = min(0.05, lr * 1.01)

    return best_a, best_sc


def _coordinate_descent(a_init, deadline, min_remaining=30.0):
    """Coordinate descent on individual elements."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(777)

    no_improve = 0
    while time.monotonic() + min_remaining < deadline and no_improve < 500:
        # Random coordinate
        i = int(rng.integers(0, n))
        a_old = a[i]

        # Try different values for a[i]
        c_base = _auto_conv(a)
        best_val = a_old
        best_sc_local = _score(a)

        for frac in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 0.1]:
            a_try = a.copy()
            a_try[i] = a_old * frac
            a_try = _project_simplex(a_try)
            sc_try = _score(a_try)
            if sc_try < best_sc_local - 1e-14:
                best_sc_local = sc_try
                best_val = a_try[i]

        if best_sc_local < best_sc - 1e-14:
            a[i] = best_val
            a = _project_simplex(a)
            best_sc = best_sc_local
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_a, best_sc


def _ils_refine(a_init, deadline, min_remaining=60.0, n_iter=50):
    """Iterated Local Search with diverse perturbations."""
    n = len(a_init)
    current_a = _normalize(a_init.copy())
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc
    rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)

    for it in range(n_iter):
        if time.monotonic() + min_remaining > deadline:
            break

        # Perturb
        strategy = it % 4
        sigma = 0.01 * (1 + 0.1 * it)

        if strategy == 0:
            # Global Gaussian
            z = np.log(np.maximum(current_a, 1e-30))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif strategy == 1:
            # Block perturbation
            block = max(20, n // 10)
            start = int(rng.integers(0, n - block))
            a_pert = current_a.copy()
            z = np.log(np.maximum(a_pert[start:start+block], 1e-30))
            z = z + sigma * 2 * rng.standard_normal(block)
            a_pert[start:start+block] = np.exp(z - np.max(z))
            a_pert = _normalize(a_pert)
        elif strategy == 2:
            # Sparse perturbation
            n_pert = max(10, n // 20)
            idx = rng.choice(n, size=n_pert, replace=False)
            a_pert = current_a.copy()
            factors = np.exp(sigma * 3 * rng.standard_normal(n_pert))
            a_pert[idx] *= factors
            a_pert = _normalize(a_pert)
        else:
            # Antisymmetric perturbation (breaks symmetry)
            a_pert = current_a.copy()
            half = n // 2
            for i in range(min(half, 100)):
                delta = sigma * rng.standard_normal()
                a_pert[i] = max(0, a_pert[i] + delta)
                a_pert[n-1-i] = max(0, a_pert[n-1-i] - delta)
            a_pert = _normalize(a_pert)

        # Quick optimization
        opt_deadline = min(deadline - min_remaining, time.monotonic() + 10)
        a_opt, sc_opt = _peak_equalization(a_pert, opt_deadline, min_remaining=2)

        if sc_opt < best_sc - 1e-13:
            best_a = a_opt.copy()
            best_sc = sc_opt
            current_a = a_opt.copy()
            print(f"    ILS new best: {best_sc:.10f} (strat={strategy}, it={it})")
        elif sc_opt < current_sc - 1e-13:
            current_a = a_opt.copy()
            current_sc = sc_opt

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _quick_optimize(a_init, deadline, min_remaining=20.0):
    """Combined optimization: L-BFGS + peak equalization."""
    a, sc = _run_lbfgs_adaptive(a_init, deadline, min_remaining=min_remaining + 10)
    if time.monotonic() < deadline - min_remaining:
        a2, sc2 = _peak_equalization(
            a, min(deadline - min_remaining, time.monotonic() + 20), min_remaining=3)
        if sc2 < sc:
            a = a2
            sc = sc2
    return a, sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining adaptive L-BFGS, FISTA active-set,
    peak equalization, coordinate descent, and ILS.
    """
    global t_start
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    n_main = n_parent if n_parent > 0 else 16384

    # === Phase 1: Multi-start L-BFGS (0-350s) ===
    print(f"\n[P1] Multi-start L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = t_start + 350

    starts = [('parent', parent.copy())]
    for kind in ['triangle', 'hann', 'gaussian', 'power_law', 'bilinear', 'random']:
        starts.append((kind, _make_profile(n_main, kind, rng)))

    for name, a_init in starts:
        if time.monotonic() > p1_deadline - 50:
            break
        sc_init = _score(a_init)
        a_opt, sc_opt = _quick_optimize(a_init, p1_deadline, min_remaining=50)
        print(f"  {name}: init={sc_init:.8f} -> {sc_opt:.8f}")
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [{name}]")

    # === Phase 2: FISTA active-set refinement (350-700s) ===
    print(f"\n[P2] FISTA active-set (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(t_start + 700, deadline - 600)
    if time.monotonic() < p2_deadline - 30:
        a_fista, sc_fista = _fista_active_set(
            best_seq, p2_deadline, min_remaining=30,
            tol_start=0.003, tol_end=0.00005)
        if sc_fista < best_score:
            best_seq = a_fista.copy()
            best_score = sc_fista
            print(f"  NEW BEST: {best_score:.10f} [fista]")

    # === Phase 3: Peak equalization (700-1000s) ===
    print(f"\n[P3] Peak equalization (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(t_start + 1000, deadline - 400)
    if time.monotonic() < p3_deadline - 30:
        a_pe, sc_pe = _peak_equalization(
            best_seq, p3_deadline, min_remaining=30)
        if sc_pe < best_score:
            best_seq = a_pe.copy()
            best_score = sc_pe
            print(f"  NEW BEST: {best_score:.10f} [peak_eq]")

    # === Phase 4: Second L-BFGS with very small beta (1000-1300s) ===
    print(f"\n[P4] Deep L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(t_start + 1300, deadline - 300)
    if time.monotonic() < p4_deadline - 30:
        a_deep, sc_deep = _run_lbfgs_adaptive(
            best_seq, p4_deadline, min_remaining=30)
        if sc_deep < best_score:
            best_seq = a_deep.copy()
            best_score = sc_deep
            print(f"  NEW BEST: {best_score:.10f} [deep_lbfgs]")

    # === Phase 5: ILS (1300-1550s) ===
    print(f"\n[P5] ILS (t={time.monotonic()-t_start:.1f}s)")
    p5_deadline = min(t_start + 1550, deadline - 200)
    if time.monotonic() < p5_deadline - 60:
        a_ils, sc_ils = _ils_refine(
            best_seq, p5_deadline, min_remaining=60, n_iter=80)
        if sc_ils < best_score:
            best_seq = a_ils.copy()
            best_score = sc_ils
            print(f"  NEW BEST: {best_score:.10f} [ils]")

    # === Phase 6: Multi-n exploration (1550-1700s) ===
    print(f"\n[P6] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    p6_deadline = min(t_start + 1700, deadline - 88)
    n_candidates = [n_main, n_main + 2048, n_main - 2048,
                    int(n_main * 1.25), int(n_main * 0.75)]
    seen = set()
    n_unique = []
    for nn in n_candidates:
        nn = max(1024, nn)
        if nn not in seen and nn != len(best_seq):
            seen.add(nn)
            n_unique.append(nn)

    for n_try in n_unique:
        if time.monotonic() > p6_deadline - 30:
            break
        a_try = _resample(best_seq, n_try)
        a_opt, sc_opt = _quick_optimize(a_try, p6_deadline, min_remaining=30)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try})")

    # === Phase 7: Final intensive polish (1700-1785s) ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        # Peak equalization
        a_fin, sc_fin = _peak_equalization(best_seq, deadline - 3, min_remaining=3)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_pe]")

        # L-BFGS with tiny beta
        remaining = deadline - 5 - time.monotonic()
        if remaining > 15:
            n = len(best_seq)
            c_best = _auto_conv(best_seq)
            mb = float(c_best.max())
            beta_tiny = mb * 0.000005
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            max_iter = min(5000, max(50, int(remaining / 0.02)))
            try:
                result = minimize(
                    lambda zz: _lse_objective(zz, beta_tiny, n),
                    z,
                    jac=lambda zz: _lse_gradient(zz, beta_tiny, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-20,
                             'gtol': 1e-16, 'maxcor': 200})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs]")
            except Exception:
                pass

        # Coordinate descent on top elements
        remaining = deadline - 5 - time.monotonic()
        if remaining > 10:
            a_cd, sc_cd = _coordinate_descent(best_seq, deadline - 3, min_remaining=3)
            if sc_cd < best_score:
                best_seq = a_cd.copy()
                best_score = sc_cd
                print(f"  NEW BEST: {best_score:.10f} [coord_desc]")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
