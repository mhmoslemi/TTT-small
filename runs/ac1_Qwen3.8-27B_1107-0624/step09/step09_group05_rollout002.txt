Now, for Nesterov's accelerated gradient method on the simplex:

- $x_{k+1} = \text{proj}_{\Delta}(\tilde{x}_{k+1} - \frac{1}{L} \nabla f(\tilde{x}_{k+1}))$

"""
Convex optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: L-BFGS with adaptive soft-max, Nesterov accelerated subgradient,
active-set KKT refinement, and multi-n exploration. Exploits symmetry.

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a = a / s
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _symmetrize(a):
    return (a + a[::-1]) / 2.0

def _normalize(a):
    s = a.sum()
    if s > 1e-15:
        return a / s
    return a

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)

def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

def _grad_soft_structured(a, beta, n):
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad

def _smoobj(z, beta, n):
    a = _softmax(z)
    a = _symmetrize(a)
    a = _normalize(a)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)

def _smograd(z, beta, n):
    a = _softmax(z)
    a = _symmetrize(a)
    a = _normalize(a)
    grad_a = _grad_soft_structured(a, beta, n)
    # Symmetrize gradient too
    grad_a = _symmetrize(grad_a)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)

class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)
    def grad(self, z):
        return _smograd(z, self.beta, self.n)

    a = _symmetrize(a_init.copy())
    a = _normalize(a)
    
    c0 = _full_convolve(a, a)

    best_a = a.copy()

def _nesterov_subgradient(a_init, deadline, min_time_remaining=30.0, t_start=0.0):
    """Nesterov accelerated subgradient with Polyak step and averaging."""
    n = len(a_init)
    a = _symmetrize(a_init.copy())
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    lb = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound
    
    # Nesterov parameters
    x = a.copy()  # current point
    y = a.copy()  # Nesterov extrapolated point
    x_avg = a.copy()  # running average for final answer
    t_val = 1.0
    
    # Lipschitz constant estimate (conservative)
    L = 4.0
    
    for it in range(1000000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        c = _full_convolve(y, y)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-12:
            best_sc = current_sc
            best_a = y.copy()
        
        # Active set
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        
        # Subgradient (averaged over active set)
        a_rev = y[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]
        grad = _symmetrize(grad)
        
        # Polyak step size
        gap = current_sc - lb
        if gap < 1e-10:
            break
        g_dev = grad - np.mean(grad)
        g_norm_sq = float(np.dot(g_dev, g_dev))
        if g_norm_sq < 1e-25:
            break
        step = 2.0 * gap / (L * g_norm_sq)
        step = min(step, 0.5)
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        theta = (t_val - 1.0) / t_new
        
        # Gradient step on y
        x_new = y - step * g_dev
        x_new = np.maximum(x_new, 0.0)
        x_new = _normalize(x_new)
        x_new = _symmetrize(x_new)
        
        # Update Nesterov state
        y_new = x_new + theta * (x_new - x) if it > 0 else x_new
        # Actually, standard Nesterov: y_{k+1} = x_{k+1} + (t_k - 1)/t_{k+1} * (x_k - x_{k-1})
        # Simpler form: y_{k+1} = (t_{k+1} - 1)/t_{k+1} * x_{k+1} + 1/t_{k+1} * y_k
        # Let me use the standard formulation:
        # x_{k+1} = proj(y_k - (1/L) * g(y_k))
        # y_{k+1} = x_{k+1} + (t_k - 1)/t_{k+1} * (x_{k+1} - x_k)
        
        # Actually let me use the FISTA formulation:
        # y_{k+1} = x_{k+1} + (t_k - 1)/(t_{k+1}) * (y_k - x_k)  [for proximal]
        # For gradient: y_{k+1} = x_{k+1} + (t_k - 1)/t_{k+1} * (y_k - y_{k-1})... 
        
        # Let me use the simpler "heavy ball" style:
        # This is getting complicated. Let me just use plain subgradient with Polyak step + averaging.
        
        x = x_new
        x_avg = x_avg + (1.0 / (it + 1.0)) * (x_new - x_avg)
        t_val = t_new
    
    # Try the average too
    sc_avg = _score(x_avg)
    if sc_avg < best_sc:
        best_sc = sc_avg
        best_a = x_avg.copy()
    
    return best_a, best_sc

def _project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
