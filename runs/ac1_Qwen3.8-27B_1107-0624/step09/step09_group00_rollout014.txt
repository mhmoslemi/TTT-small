import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    c = _full_convolve(a_norm, a_norm)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos2":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.2) ** 2) + np.exp(-30.0 * (t - 0.8) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edge_conc":
        t = np.linspace(0, 1, n)
        a = (t * (1 - t)) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(12345)
        a = rng.random(n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _grad_exact_max(a, c, top_k=1):
    """Compute gradient using exact top-k peaks."""
    n = len(a)
    c_max = float(c.max())
    # Get top-k peaks
    if top_k >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argsort(c)[::-1][:top_k]
    
    # Weight: uniform over top-k, or weighted by height
    heights = c[top_idx]
    w = np.zeros_like(c)
    # Use softmax weighting over the top-k
    w_shifted = heights - heights[0]
    w_vals = np.exp(w_shifted)
    w_vals = w_vals / w_vals.sum()
    for idx, wv in zip(top_idx, w_vals):
        w[idx] = wv
    
    # Gradient: g_i = 2 * sum_k w_k * a_{k-i} for valid k-i
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad = 2.0 * h
    
    # Project onto tangent space of sum(a)=1
    grad = grad - np.mean(grad)
    return grad


def _active_set_polish(a_init, deadline, tol_start=0.01, tol_end=0.00003,
                       lr_init=0.03, momentum=0.85, top_k_max=50):
    """Active-set gradient with momentum and adaptive step size.
    Combines tolerance-based and top-k active set strategies."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    v = np.zeros(n)
    tol_frac = tol_start
    lr = lr_init
    no_improve = 0
    max_stall = 200
    top_k = min(5, top_k_max)
    
    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        # Adaptive: switch between tolerance-based and top-k
        tol = tol_frac * c_max
        n_tol = int((c >= (c_max - tol)).sum())
        use_top_k = min(top_k, max(1, n_tol))
        
        if use_top_k <= 5:
            grad = _grad_exact_max(a, c, top_k=use_top_k)
        else:
            # Tolerance-based with inverse-gap weighting
            active = c >= (c_max - tol)
            eps = 1e-14 * c_max
            w = np.zeros_like(c)
            w[active] = 1.0 / (c_max - c[active] + eps)
            w = w / w.sum()
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            grad = 2.0 * h
            grad = grad - np.mean(grad)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Momentum with restart
        v_new = momentum * v - lr * grad
        # Check for restart: if v and -grad point in opposite directions
        if np.dot(v_new, -grad) < 0 and np.dot(v, -grad) > 0:
            v = np.zeros(n)
            v_new = -lr * grad
        
        alpha = 1.0
        improved = False
        for _ in range(40):
            a_try = _normalize(a + alpha * v_new)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                v = v_new
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            v = np.zeros(n)
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_end, tol_frac * 0.6)
                lr = max(0.001, lr * 0.7)
                top_k = max(1, top_k - 1)
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(tol_end, tol_frac * 0.3)
                lr = lr_init * 0.3
                top_k = max(1, top_k // 2)
                no_improve = 0
        else:
            if tol_frac > tol_end:
                tol_frac = max(tol_end, tol_frac * 0.98)
            lr = min(lr_init, lr * 1.005)
            top_k = min(top_k_max, top_k + 1)
    
    return best_a, best_sc


def _mult_weight_polish(a_init, deadline, eta_init=0.01, eta_min=0.0001):
    """Multiplicative weight updates with exact max.
    p_i <- p_i * (1 - eta * g_i), clip, normalize."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    eta = eta_init
    
    for iteration in range(100000):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
        
        # Find peak(s)
        peak_idx = int(np.argmax(c))
        
        # Gradient: g_i = 2 * a_{peak-i} for valid peak-i
        g = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                g[i] = 2.0 * a[j]
        
        # Project: g = g - mean(g) (tangent to simplex)
        g = g - np.mean(g)
        
        # Multiplicative update
        a_try = a * np.exp(-eta * g)
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-30:
            break
        a_try = a_try / s
        
        sc_try = 2.0 * n * float(_full_convolve(a_try, a_try).max())
        
        if sc_try < current_sc - 1e-15:
            a = a_try
            eta = min(eta_init, eta * 1.01)
        else:
            eta *= 0.8
            if eta < eta_min:
                break
    
    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def f(self, z):
        return _smoobj(z, self.beta, self.n)

    def g(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0):
    """L-BFGS on log-softmax parameterization with adaptive beta annealing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003) * beta_scale

    # More aggressive annealing schedule
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.008, 0.004, 0.002,
                                  0.001, 0.0005, 0.0002, 0.0001,
                                  0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(50, int(remaining / (est_time * 15))))

        prob = _LbfgsProblem(beta, n)
        try:
            result = minimize(
                prob.f, z, jac=prob.g, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 150, 'maxls': 60}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _quick_optimize(a_init, deadline, min_time_remaining=10.0):
    """Combined optimization: L-BFGS + active-set polish."""
    remaining = deadline - min_time_remaining - time.monotonic()
    if remaining < 5:
        return a_init
    
    # Split time between L-BFGS and polish
    lbfgs_frac = 0.6
    lbfgs_deadline = time.monotonic() + remaining * lbfgs_frac
    a = _run_lbfgs(a_init, lbfgs_deadline, min_time_remaining=max(5, remaining * 0.1))
    
    if time.monotonic() < deadline - min_time_remaining:
        pol_deadline = deadline - min_time_remaining
        a2, sc2 = _active_set_polish(a, pol_deadline,
                                       tol_start=0.003, tol_end=0.00002,
                                       lr_init=0.015, momentum=0.8)
        if sc2 < _score(a):
            a = a2
    return a


def _peak_focused_perturb(a, rng, strategy="peak_shave"):
    """Perturbation that specifically targets the peak positions."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if strategy == "peak_shave":
        # Reduce values that contribute to the top peaks
        top_k = min(10, n // 4)
        peak_indices = np.argsort(c)[::-1][:top_k]
        
        a_new = a.copy()
        # For each peak, reduce the contributing a values
        reduction = np.zeros(n)
        for k in peak_indices:
            for i in range(n):
                j = int(k) - i
                if 0 <= j < n:
                    reduction[i] += a[j]
        
        # Scale reduction to be a fraction of the total
        total_reduction = reduction.sum()
        if total_reduction > 0:
            scale = 0.05 * a.sum() / total_reduction
            a_new = a - scale * reduction
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
            else:
                a_new = a.copy()
        
        # Add small noise
        z = np.log(np.maximum(a_new, 1e-30))
        z = z + 0.002 * rng.standard_normal(n)
        z = z - z.mean()
        a_new = _softmax(z)
        
    elif strategy == "anti_peak":
        # Increase values that DON'T contribute to peaks
        peak_idx = int(np.argmax(c))
        contrib = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contrib[i] = a[j]
        
        # Inverse contribution - boost low-contribution positions
        a_new = a * (1 + 0.1 * (1 - contrib / max(contrib.max(), 1e-15)))
        a_new = _normalize(a_new)
        
    elif strategy == "phase_shift":
        # Circular shift by a small amount (interpolation)
        shift = rng.uniform(-0.01, 0.01) * n
        x = np.arange(n)
        x_new = (x - shift) % n
        a_new = np.interp(x_new, np.arange(n), a, period=n)
        a_new = _normalize(a_new)
        
    else:
        a_new = a.copy()
    
    return a_new


def _stochastic_ils(a_init, deadline, rng, min_time_remaining=30.0):
    """ILS with diverse strategies including peak-focused perturbations."""
    n = len(a_init)
    current_a = _normalize(a_init.copy())
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc
    
    strategies = [
        ('global_small', 0.003, 0),
        ('global_med', 0.015, 0),
        ('peak_shave', 0, 1),
        ('anti_peak', 0, 2),
        ('phase_shift', 0, 3),
        ('block', 0.04, 0),
        ('sparse', 0.06, 0),
        ('global_tiny', 0.001, 0),
    ]
    
    attempt = 0
    while time.monotonic() + min_time_remaining < deadline:
        kind, sigma, strat_id = strategies[attempt % len(strategies)]
        
        if strat_id == 1:  # peak_shave
            a_perturbed = _peak_focused_perturb(current_a, rng, "peak_shave")
        elif strat_id == 2:  # anti_peak
            a_perturbed = _peak_focused_perturb(current_a, rng, "anti_peak")
        elif strat_id == 3:  # phase_shift
            a_perturbed = _peak_focused_perturb(current_a, rng, "phase_shift")
        elif kind == 'block':
            block_size = max(20, n // 10)
            start = int(rng.integers(0, n - block_size))
            a_perturbed = current_a.copy()
            z = np.log(np.maximum(a_perturbed[start:start + block_size], 1e-30))
            z = z + sigma * rng.standard_normal(block_size)
            a_perturbed[start:start + block_size] = np.exp(z - np.max(z))
            a_perturbed = _normalize(a_perturbed)
        elif kind == 'sparse':
            n_perturb = max(10, n // 20)
            idx = rng.choice(n, size=n_perturb, replace=False)
            a_perturbed = current_a.copy()
            factors = np.exp(sigma * rng.standard_normal(n_perturb))
            a_perturbed[idx] *= factors
            a_perturbed = _normalize(a_perturbed)
        else:
            z = np.log(np.maximum(current_a, 1e-30))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_perturbed = _softmax(z)
        
        # Quick optimization of perturbed point
        opt_deadline = min(deadline - min_time_remaining, time.monotonic() + 6)
        a_opt = _quick_optimize(a_perturbed, opt_deadline, min_time_remaining=2)
        sc_opt = _score(a_opt)
        
        if sc_opt < best_sc - 1e-13:
            best_sc = sc_opt
            best_a = a_opt.copy()
            current_a = a_opt.copy()
            print(f"    ILS NEW BEST: {best_sc:.10f} ({kind}) t={time.monotonic()-t_start:.0f}s")
        elif sc_opt < current_sc - 1e-13:
            current_a = a_opt.copy()
        
        attempt += 1
    
    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _symmetric_optimize(a_init, deadline):
    """Optimize a symmetric sequence by only varying the first half."""
    n = len(a_init)
    if n % 2 != 0:
        return a_init, _score(a_init)
    
    half = n // 2
    # Extract first half
    x = a_init[:half].copy()
    
    def _expand(x):
        a = np.zeros(n)
        a[:half] = x
        a[half:] = x[::-1]
        return a
    
    def _obj(x):
        a = _expand(x)
        a = _normalize(a)
        return _score(a)
    
    def _grad(x):
        a = _expand(x)
        a = _normalize(a)
        c = _full_convolve(a, a)
        # Use log-sum-exp with small beta
        beta = float(c.max()) * 0.001
        c_max = float(c.max())
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        w = e / e.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[half:2 * half]
        g = 2.0 * h[:half]
        return g
    
    best_x = x.copy()
    best_sc = _obj(best_x)
    
    try:
        result = minimize(_obj, best_x, jac=_grad, method='L-BFGS-B',
                         options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-14})
        if _obj(result.x) < best_sc:
            best_x = result.x
            best_sc = _obj(best_x)
    except Exception:
        pass
    
    return _expand(_normalize(best_x)), best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    
    Phases:
    1. Multi-start L-BFGS with diverse profiles
    2. Intensive L-BFGS annealing
    3. Active-set polish with momentum
    4. Peak-focused ILS
    5. Multi-n exploration
    6. Second ILS round
    7. Final ultra-fine polish with multiplicative weights
    """
    global t_start
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")
    
    n_main = n_parent if n_parent > 0 else 16384
    
    # === Phase 1: Multi-start (200s) ===
    print(f"\n[P1] Multi-start (t={time.monotonic() - t_start:.1f}s)")
    ms_deadline = t_start + 200
    
    starts = [
        ('parent', parent.copy()),
        ('triangle', _make_profile(n_main, "triangle")),
        ('hann', _make_profile(n_main, "hann")),
        ('gaussian', _make_profile(n_main, "gaussian")),
        ('raised_cos2', _make_profile(n_main, "raised_cos2")),
        ('bimodal', _make_profile(n_main, "bimodal")),
        ('edge_conc', _make_profile(n_main, "edge_conc")),
    ]
    
    for name, a_init in starts:
        if time.monotonic() > ms_deadline - 30:
            break
        sc_init = _score(a_init)
        a_opt = _quick_optimize(a_init, ms_deadline, min_time_remaining=40)
        sc_opt = _score(a_opt)
        print(f"  {name}: init={sc_init:.10f}, opt={sc_opt:.10f}")
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [ms/{name}] t={time.monotonic() - t_start:.1f}s")
    
    # === Phase 2: Intensive L-BFGS (150s) ===
    print(f"\n[P2] Intensive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 450:
        lbfgs_deadline = min(deadline - 450, time.monotonic() + 150)
        a_main = _run_lbfgs(best_seq, lbfgs_deadline, min_time_remaining=50, beta_scale=0.5)
        sc = _score(a_main)
        if sc < best_score:
            best_seq = a_main.copy()
            best_score = sc
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_int] t={time.monotonic() - t_start:.1f}s")
    
    # === Phase 3: Active-set polish (150s) ===
    print(f"\n[P3] Active-set polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        pol_deadline = min(deadline - 300, time.monotonic() + 150)
        a_pol, sc_pol = _active_set_polish(best_seq, pol_deadline,
                                            tol_start=0.002, tol_end=0.00001,
                                            lr_init=0.01, momentum=0.8, top_k_max=20)
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  NEW BEST: {best_score:.10f} [active_set] t={time.monotonic() - t_start:.1f}s")
    
    # === Phase 4: Peak-focused ILS (350s) ===
    print(f"\n[P4] Peak-focused ILS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        ils_deadline = min(deadline - 300, time.monotonic() + 350)
        a_ils, sc_ils = _stochastic_ils(best_seq, ils_deadline, rng, min_time_remaining=50)
        if sc_ils < best_score:
            best_seq = a_ils.copy()
            best_score = sc_ils
            print(f"  NEW BEST: {best_score:.10f} [ils] t={time.monotonic() - t_start:.1f}s")
    
    # === Phase 5: Multi-n exploration (300s) ===
    print(f"\n[P5] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [n_main, int(n_main * 1.5), int(n_main * 0.75),
                    int(n_main * 1.25), int(n_main / 1.5),
                    20000, 30000, 25000, 35000, 40000]
    seen = set()
    n_unique = []
    for nn in n_candidates:
        nn = max(1024, nn)
        if nn not in seen and nn != len(best_seq):
            seen.add(nn)
            n_unique.append(nn)
    
    for n_try in n_unique:
        if time.monotonic() > deadline - 180:
            break
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init={sc_init:.10f}")
        
        a_opt = _quick_optimize(a_init, deadline - 100, min_time_remaining=30)
