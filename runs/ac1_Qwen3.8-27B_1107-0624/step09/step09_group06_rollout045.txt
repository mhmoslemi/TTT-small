import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _conv_max(a):
    return float(_full_convolve(a, a).max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _lp_direction(a, M, active_k, n):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in active_k, b >= 0.
    Returns the optimal b (normalized to sum 1) or None if infeasible."""
    K = len(active_k)
    # Build constraint matrix: K x n
    rows = []
    cols = []
    vals = []
    for idx_k, k in enumerate(active_k):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            i = k - j  # index in b
            if 0 <= i < n:
                rows.append(idx_k)
                cols.append(i)
                vals.append(a[j])
    
    if len(rows) == 0:
        return None
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(K, n))
    c = -np.ones(n)  # maximize sum(b) = minimize -sum(b)
    b_ub = np.full(K, M)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            b_opt = res.x
            s = b_opt.sum()
            if s > 1e-15:
                return b_opt / s
    except Exception:
        pass
    return None

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)

def _peak_equalize(a_init, deadline, min_time_remaining=30.0):
    """Directly equalize top peaks using gradient of peak variance."""
    n = len(a_init)

    s = a.sum()

        else:

            if stall > 15:

        top_vals = c[top_idx]
        mean_top = top_vals.mean()
        
        # Gradient of sum of (c[k_i] - mean)^2 w.r.t. a
        # d/d a_m of c[k] = 2 * a[k-m] (if in range)
        grad = np.zeros(n)

            diff = c[k] - mean_top
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                for m in range(j_lo, j_hi + 1):
                    grad[m] += 4.0 * diff * a[k - m]
        
        # Project to tangent space
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        step = lr * grad / gnorm
        a_cand = a - step
        a_cand = np.maximum(a_cand, 0)

        if s_new < 1e-10:

        else:
            a = a_cand  # stay on simplex anyway

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

    elif kind == "tapered":

    deadline = t_start + max(1.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")
    
    # === Phase 1: LP-based iterative method ===
    print(f"\n[P1] LP-iterative (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 800, t_start + 350)
    if time.monotonic() < p1_deadline:
        a_lp, sc_lp = _lp_iterative_method(best_seq, p1_deadline,
                                           min_time_remaining=60.0, K_init=20, K_max=200)
        if sc_lp < best_score:
            best_seq = a_lp.copy()
            best_score = sc_lp
            print(f"  NEW BEST: {best_score:.10f} [lp] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  LP: no improvement (best={best_score:.10f})")
    
    # === Phase 2: Aggressive L-BFGS ===
    print(f"\n[P2] Aggressive L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 500, t_start + 600)
    if time.monotonic() < p2_deadline:
        a_lb = _run_lbfgs_aggressive(best_seq, p2_deadline, min_time_remaining=60.0)
        sc_lb = _score(a_lb)
        if sc_lb < best_score:
            best_seq = a_lb.copy()
            best_score = sc_lb
            print(f"  NEW BEST: {best_score:.10f} [lbfgs] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 3: Peak equalization ===
    print(f"\n[P3] Peak equalization (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 400, t_start + 750)
    if time.monotonic() < p3_deadline:
        a_eq, sc_eq = _peak_equalize(best_seq, p3_deadline, min_time_remaining=60.0)
        if sc_eq < best_score:
            best_seq = a_eq.copy()
            best_score = sc_eq
            print(f"  NEW BEST: {best_score:.10f} [equalize] t={time.monotonic()-t_start:.1f}s")
            # Follow up with L-BFGS
            if time.monotonic() < p3_deadline + 50 and time.monotonic() < deadline - 350:
                a_eq2 = _run_lbfgs_aggressive(best_seq, min(deadline - 350, t_start + 800),
                                               min_time_remaining=60.0)
                sc_eq2 = _score(a_eq2)
                if sc_eq2 < best_score:
                    best_seq = a_eq2.copy()
                    best_score = sc_eq2
                    print(f"  NEW BEST: {best_score:.10f} [eq+lbfgs] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 4: Nesterov + LP (different starts) ===
    print(f"\n[P4] Nesterov+LP round 2 (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 300, t_start + 950)
    if time.monotonic() < p4_deadline:
        a_nes, sc_nes = _nesterov_subgrad(best_seq, p4_deadline,
                                          min_time_remaining=60.0, K_init=5, K_max=100, seed=seed+1)
        if sc_nes < best_score:
            best_seq = a_nes.copy()
            best_score = sc_nes
            print(f"  NEW BEST: {best_score:.10f} [nesterov2] t={time.monotonic()-t_start:.1f}s")
        
        if time.monotonic() < p4_deadline + 50 and time.monotonic() < deadline - 200:
            a_lp2, sc_lp2 = _lp_iterative_method(best_seq, min(deadline - 200, t_start + 1000),
                                                  min_time_remaining=60.0, K_init=10, K_max=150)
            if sc_lp2 < best_score:
                best_seq = a_lp2.copy()
                best_score = sc_lp2
                print(f"  NEW BEST: {best_score:.10f} [lp2] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 5: Multi-n exploration ===
    print(f"\nP5] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [n_parent]
    for nn in [16384, 12288, 24576, 8192, 32768]:
        if nn not in n_candidates:
            n_candidates.append(nn)
    
    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        if n_try == len(best_seq):
            continue
        if n_try > 65536:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}", end=" ")
        
        # Quick L-BFGS
        if time.monotonic() < deadline - 100:
            a_opt = _run_lbfgs_aggressive(a_init, deadline - 50, min_time_remaining=30.0)
            sc_opt = _score(a_opt)
            print(f"-> {sc_opt:.10f}", end=" ")
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f" NEW BEST! [multi_n {n_try}]")
            else:
                print()
    
    # === Phase 6: Final L-BFGS polish ===
    print(f"\n[P6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 10:
        a_fin = _run_lbfgs_aggressive(best_seq, deadline - 5, min_time_remaining=5.0)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 7: LP polish ===
    if time.monotonic() < deadline - 10:
        a_lp3, sc_lp3 = _lp_iterative_method(best_seq, deadline - 5,
                                              min_time_remaining=5.0, K_init=10, K_max=80)
        if sc_lp3 < best_score:
            best_seq = a_lp3.copy()
            best_score = sc_lp3
            print(f"  NEW BEST: {best_score:.10f} [lp_final] t={time.monotonic()-t_start:.1f}s")
    
    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
