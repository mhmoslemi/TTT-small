Key innovations over previous approach:
1. Top-K weighted smooth max (more focused on important peaks)
2. Momentum-based active-set equalization with adaptive K cycling
3. Aggressive multi-beta L-BFGS with exponential decay
4. Periodic random "kicks" to escape local minima

6. Final ultra-fine polish with many small-beta iterations

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _topk_smooth_max(c, beta, K):
    """Smooth max using top-K elements with log-sum-exp."""
    n_c = len(c)
    K = min(K, n_c)
    # Get top-K values
    if K == n_c:
        idx = np.arange(n_c)
    else:
        idx = np.argpartition(c, -K)[-K:]
    top_k = c[idx]
    c_max = float(c.max())
    shifted = (top_k - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return c_max
    return c_max + beta * np.log(s)


def _topk_smooth_max_grad(c, a, n, beta, K):
    """Gradient of 2n * topk_smooth_max(a*a) with respect to a."""
    n_c = len(c)
    K = min(K, n_c)
    if K == n_c:
        idx = np.arange(n_c)
    else:
        idx = np.argpartition(c, -K)[-K:]
    top_k = c[idx]
    c_max = float(c.max())
    shifted = (top_k - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        # Degenerate: gradient of hard max at argmax
        k_star = int(np.argmax(c))
        a_rev = a[::-1]
        h = _full_convolve(np.eye(2 * n - 1)[k_star], a_rev)[n - 1:2 * n - 1]
        return 4.0 * n * a * h
    else:
        w = e / s
        w_full = np.zeros(n_c)
        w_full[idx] = w
        a_rev = a[::-1]
        wc = _full_convolve(w_full, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w_full, c))
        return 4.0 * n * a * (h - W)


def _run_lbfgs_smooth(a_init, deadline, min_time_remaining, beta_scale=1.0, K_init=5):
    """Run L-BFGS with top-K smooth max and adaptive K."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    current_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    # Adaptive beta and K schedule
    c_cur = _full_convolve(a, a)
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < c_sorted[0] * 0.001:
        gap = c_sorted[0] * 0.001
    base_beta = max(gap, c_sorted[0] * 0.01) * beta_scale

    # Schedule: (beta_multiplier, K)
    schedule = [
        (1.0, K_init),
        (0.5, K_init),
        (0.25, K_init + 2),
        (0.125, K_init + 5),
        (0.0625, K_init + 10),
        (0.03, K_init + 15),
        (0.015, K_init + 25),
        (0.0075, K_init + 40),
        (0.004, K_init + 60),
        (0.002, K_init + 80),
        (0.001, K_init + 100),
        (0.0005, K_init + 150),
        (0.0002, K_init + 200),
        (0.0001, K_init + 300),
    ]

    for beta_mult, K in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        beta = base_beta * beta_mult
        K_eff = min(K, 2 * n - 1)

        # Define objective and gradient
        def obj(z_in):
            a_in = _softmax(z_in)
            c_val = _full_convolve(a_in, a_in)
            return 2.0 * n * _topk_smooth_max(c_val, beta, K_eff)

        def grad(z_in):
            a_in = _softmax(z_in)
            c_val = _full_convolve(a_in, a_in)
            g = _topk_smooth_max_grad(c_val, a_in, n, beta, K_eff)
            # Project gradient to simplex tangent (remove mean)
            g = g - np.mean(g)
            return g

        est_time_per_iter = 0.005 * max(1.0, n / 2048.0)
        max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 10))))

        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 100, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return a, current_sc


def _active_set_momentum(a_init, deadline, min_time_remaining=30.0):
    """Active-set equalization with Nesterov momentum and adaptive K."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    # Momentum state
    v = np.zeros(n)
    beta_mom = 0.9

    K = 5
    K_max = min(200, n)
    lr = 0.01
    no_improve = 0
    total_iters = 0

    while time.monotonic() + min_time_remaining < deadline:
        total_iters += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive K: cycle through different K values
        K = min(K_max, 5 + (total_iters // 20) * 3)
        if no_improve > 50:
            K = 5  # Reset K to focus on main peak

        top_k_idx = np.argsort(c)[::-1][:K]

        # Compute weighted gradient (top-K peaks, weighted by excess)
        top_k_vals = c[top_k_idx]
        weights = np.maximum(top_k_vals - c_max * 0.999, 0)
        if weights.sum() < 1e-30:
            weights = np.ones(K) / K
        else:
            weights = weights / weights.sum()

        w_S = np.zeros(2 * n - 1)
        for i, k in enumerate(top_k_idx):
            w_S[k] = weights[i]
        a_rev = a[::-1]
        h = _full_convolve(w_S, a_rev)[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)  # Tangent to simplex
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Nesterov momentum
        v = beta_mom * v - lr * grad / (gnorm + 1e-15)

        # Apply step with line search
        improved = False
        for trial in range(40):
            a_try = a + v
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                v *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            v *= 0.7

        if not improved:
            v = np.zeros(n)
            lr *= 0.9
            if no_improve > 100:
                a = best_a.copy()
                v = np.zeros(n)
                lr = 0.01
                no_improve = 0
                beta_mom = min(0.95, beta_mom + 0.01)
                if no_improve > 300:
                    break
        else:
            lr = min(0.03, lr * 1.005)

    return best_a, best_sc

def _asymmetric_perturb(a, rng, strength=0.05):
    n = len(a)
    t = np.arange(n) / n
    ramp = 0.3 * (t - 0.5)
    noise = 0.7 * rng.standard_normal(n)
    perturbation = ramp + noise
    perturbation = perturbation - perturbation.mean()
    z = np.log(np.maximum(a, 1e-30))
    z = z + strength * perturbation
    z = z - z.mean()
    return _softmax(z)


def _random_kick(a, rng, strength=0.02):

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edges":
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * t) ** 2 + 0.01
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.2) ** 2) + np.exp(-20.0 * (t - 0.8) ** 2) + 0.05
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 5.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

    if time.monotonic() < deadline - 400:

        a_eq, sc_eq = _active_set_momentum(best_seq, pol_deadline, min_time_remaining=30.0)
        if sc_eq < best_score:
            best_seq = a_eq.copy()
            best_score = sc_eq
            print(f"  NEW BEST: {best_score:.10f} [momentum_AS]")
        else:
            print(f"  Momentum AS: no improvement (best={best_score:.10f})")

    # === Phase 3: Cyclical active-set ===
    print(f"\n[P3] Cyclical active-set (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        pol_deadline2 = min(deadline - 300, time.monotonic() + 200)
        a_cyc, sc_cyc = _active_set_cyclical(best_seq, pol_deadline2, min_time_remaining=30.0)
        if sc_cyc < best_score:
            best_seq = a_cyc.copy()
            best_score = sc_cyc
            print(f"  NEW BEST: {best_score:.10f} [cyclical_AS]")
        else:
            print(f"  Cyclical AS: no improvement (best={best_score:.10f})")

    # === Phase 4: Asymmetric perturbation + re-optimize ===
    print(f"\n[P4] Asymmetric exploration (t={time.monotonic()-t_start:.1f}s)")
    for trial in range(3):
        if time.monotonic() > deadline - 250:
            break
        strength = 0.02 * (trial + 1)
        a_asym = _asymmetric_perturb(best_seq, rng, strength=strength)
        a_opt, sc_opt = _run_lbfgs_smooth(a_asym, deadline, min_time_remaining=250.0,
                                           beta_scale=0.5, K_init=10)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [asym_trial_{trial}]")
        else:
            print(f"  Asym trial {trial}: {sc_opt:.10f} (no improvement)")

    # === Phase 5: Fine L-BFGS with small beta ===
    print(f"\n[P5] Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        a_fine, sc_fine = _run_lbfgs_smooth(best_seq, deadline, min_time_remaining=150.0,
                                             beta_scale=0.3, K_init=20)
        if sc_fine < best_score:
            best_seq = a_fine.copy()
            best_score = sc_fine
            print(f"  NEW BEST: {best_score:.10f} [fine_lbfgs]")

    # === Phase 6: More momentum active-set ===
    print(f"\n[P6] Momentum AS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        pol_dl3 = min(deadline - 100, time.monotonic() + 80)
        a_eq2, sc_eq2 = _active_set_momentum(best_seq, pol_dl3, min_time_remaining=20.0)
        if sc_eq2 < best_score:
            best_seq = a_eq2.copy()
            best_score = sc_eq2
            print(f"  NEW BEST: {best_score:.10f} [momentum_AS2]")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([n, int(n * 1.1), int(n * 0.9),
                               int(n * 1.2), int(n * 0.8)]), reverse=True)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try == n or n_try < 1000:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        a_opt, sc_opt = _run_lbfgs_smooth(a_init, deadline, min_time_remaining=60.0,
                                           beta_scale=0.5, K_init=10)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            n = n_try
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n]")
        else:
            print(f"  n={n_try}: {sc_opt:.10f} (init={sc_init:.10f}, no improvement)")

    # === Phase 8: Random kicks + re-optimize ===
    print(f"\n[P8] Random kicks (t={time.monotonic()-t_start:.1f}s)")
    for kick in range(5):
        if time.monotonic() > deadline - 40:
            break
        a_kick = _random_kick(best_seq, rng, strength=0.01 * (kick + 1))
        a_opt, sc_opt = _run_lbfgs_smooth(a_kick, deadline, min_time_remaining=40.0,
                                           beta_scale=0.3, K_init=15)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [kick_{kick}]")

    # === Phase 9: Subgradient polish ===
    print(f"\n[P9] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 20:
        sg_dl = min(deadline - 15, time.monotonic() + 60)
        a_sg, sc_sg = _subgradient_polish(best_seq, sg_dl, min_time_remaining=15.0)
        if sc_sg < best_score:
            best_seq = a_sg.copy()
            best_score = sc_sg
            print(f"  NEW BEST: {best_score:.10f} [subgradient]")

    # === Phase 10: Final ultra-fine L-BFGS ===
    print(f"\n[P10] Final ultra-fine (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 5 - time.monotonic()
    if remaining > 10:
        a_final, sc_final = _run_lbfgs_smooth(best_seq, deadline, min_time_remaining=5.0,
                                               beta_scale=0.1, K_init=50)
        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final
            print(f"  NEW BEST: {best_score:.10f} [final_ultra]")

        # Tiny final active-set
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _active_set_momentum(best_seq, deadline - 2,
                                                   min_time_remaining=2.0)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_AS]")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None and len(best_seq) > 0:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
