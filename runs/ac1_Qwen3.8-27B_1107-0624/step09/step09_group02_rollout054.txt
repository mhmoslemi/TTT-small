import numpy as np

from scipy.optimize import minimize


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def compute_score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * t
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.3) ** 2) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def spectral_init(n, rng):
    K = next_pow2(2 * n - 1)
    phases = rng.uniform(0, 2 * np.pi, K // 2 + 1)
    A = np.exp(1j * phases)
    # Ensure Hermitian symmetry for real output
    A_full = np.zeros(K, dtype=complex)
    A_full[:K // 2 + 1] = A
    for k in range(2, K // 2 + 1):
        A_full[K - k] = np.conj(A[k])
    a = np.fft.ifft(A_full).real[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return make_profile(n, "tapered")
    return a / s


def resample_fn(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))

    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val), c_max


def run_lbfgs_betas(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = compute_score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_cur = full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap * 0.05, max_c * 0.002)

    ratios = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
              0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001,
              0.000005, 0.000002, 0.000001]

    for i, ratio in enumerate(ratios):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        beta = base_beta * ratio
        n_scale = max(1.0, n / 16384.0)
        est_time = 0.005 * n_scale
        max_iter = min(5000, max(50, int(remaining / (est_time * 3))))

        def obj_func(z_local):
            a_local = softmax_fn(z_local)
            c_local = full_convolve(a_local, a_local)
            c_max_local = float(c_local.max())
            shifted_local = (c_local - c_max_local) / beta
            e_local = np.exp(np.clip(shifted_local, -700, 0))
            s_e_local = e_local.sum()
            if s_e_local < 1e-300:
                return 2.0 * n * c_max_local
            return 2.0 * n * (c_max_local + beta * np.log(s_e_local))

        def grad_func(z_local):
            a_local = softmax_fn(z_local)
            c_local = full_convolve(a_local, a_local)
            grad_local, _ = compute_gradients(a_local, n, beta, c_local)
            return grad_local

        try:
            result = minimize(
                obj_func, z,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxcor': 60, 'maxls': 80}
            )
            a_new = softmax_fn(result.x)
            sc_new = compute_score(a_new)
            if sc_new < best_sc - 1e-16:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS ratio={ratio}: score={best_sc:.10f}, "
                      f"iter={result.nit}, t={time.monotonic() - t_start_ref:.1f}s")
        except Exception:
            pass

    return best_a


def run_active_set_pg(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    tol_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                    0.00005, 0.00002, 0.00001, 0.000005, 0.000002]
    tol_idx = 0
    no_improve = 0

    while time.monotonic() + min_remaining < deadline:
        c = full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        k_act = int(active.sum())
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / k_act

        lr = 0.05
        for inner in range(100):
            if time.monotonic() + min_remaining > deadline:
                break
            a_rev = a[::-1]
            wc = full_convolve(p, a_rev)
            h = wc[n - 1:2 * n - 1]
            W_val = float(np.dot(p, c))
            grad = 2.0 * a * (h - W_val)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-16:
                break
            lr = 0.1 / (gnorm * (1.0 + 0.005 * inner))
            a_try = proj_simplex(a - lr * grad)
            a = a_try

        sc = compute_score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
            if (tol_idx + 1) % 3 == 0:
                print(f"    ActiveSet PG: tol_idx={tol_idx}, K={k_act}, "
                      f"score={best_sc:.10f}, t={time.monotonic() - t_start_ref:.1f}s")
        else:
            no_improve += 1
            if no_improve > 5:
                a = best_a.copy()
                tol_idx = min(tol_idx + 1, len(tol_schedule) - 1)
                no_improve = 0

        if (tol_idx + 1) % 4 == 0 and no_improve < 3:
            tol_idx = min(tol_idx + 1, len(tol_schedule) - 1)

    return best_a, best_sc

                      f"score={best_sc:.10f}, t={time.monotonic() - t_start_ref:.1f}s")

            pass

    return best_a, best_sc


def perturb_fn(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return softmax_fn(z)

def propose_candidate(seed=42, budget_s=1800):
    global t_start_ref
    t_start_ref = time.monotonic()
    t_start = t_start_ref
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        p = p / s
        if sc is None:
            sc = compute_score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Try to load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = compute_score(parent)
                print(f"Parent loaded: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 32768
        parent = make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, compute_score(parent), "init")

    # === Phase 1: Quick multi-start (250s) ===
    print(f"\n[Phase 1] Multi-start (deadline={250}s)")
    phase1_deadline = t_start + 250.0

    # Spectral initialization at parent size
    a_spec = spectral_init(n_parent, rng)
    print(f"  Spectral init score: {compute_score(a_spec):.10f}")
    update_best(a_spec, compute_score(a_spec), "spectral_init")

    # L-BFGS from best available
    best_init = best_seq.copy() if best_seq is not None else parent
    a1 = run_lbfgs_betas(best_init, phase1_deadline, min_remaining=20.0)
    update_best(a1, compute_score(a1), "lbfgs_phase1")

    # Try spectral + L-BFGS if time remains
    if time.monotonic() < phase1_deadline - 40:
        a_spec2 = spectral_init(n_parent, rng)
        a1b = run_lbfgs_betas(a_spec2, phase1_deadline, min_remaining=15.0)
        update_best(a1b, compute_score(a1b), "lbfgs_spectral")

    # === Phase 2: Try n=65536 (200s) ===
    print(f"\n[Phase 2] Try n=65536 (200s)")
    phase2_deadline = phase1_deadline + 200.0
    n_large = 65536
    if time.monotonic() < phase2_deadline - 30:
        # Resample best to larger n
        a_large_init = resample_fn(best_seq, n_large)
        a_large = run_lbfgs_betas(a_large_init, phase2_deadline, min_remaining=20.0)
        sc_large = compute_score(a_large)
        print(f"  n=65536 L-BFGS score: {sc_large:.10f}")
        if sc_large < best_score:
            update_best(a_large, sc_large, "n65536_lbfgs")
        # Only keep if competitive
        if sc_large > best_score + 0.001:
            print(f"  n=65536 not competitive, discarding")
            a_large = None
    else:
        a_large = None

    # Try n=49152
    if time.monotonic() < phase2_deadline - 30:
        n_mid = 49152
        a_mid_init = resample_fn(best_seq, n_mid)
        a_mid = run_lbfgs_betas(a_mid_init, phase2_deadline, min_remaining=10.0)
        sc_mid = compute_score(a_mid)
        print(f"  n=49152 L-BFGS score: {sc_mid:.10f}")
        if sc_mid < best_score:
            update_best(a_mid, sc_mid, "n49152_lbfgs")

    # === Phase 3: Long polish on best (800s) ===
    print(f"\n[Phase 3] Long polish (800s)")
    phase3_deadline = min(deadline - 350, phase2_deadline + 800.0)
    if best_seq is not None and time.monotonic() < phase3_deadline:
        # Sharpened L-BFGS (longer running with decreasing beta)
        a_pol, sc_pol = run_sharpened_lbfgs(best_seq, phase3_deadline, min_remaining=60.0)
        update_best(a_pol, sc_pol, "sharpened_phase3")

        # Active set PG
        if time.monotonic() < phase3_deadline - 30:
            a_as, sc_as = run_active_set_pg(best_seq, phase3_deadline, min_remaining=30.0)
            update_best(a_as, sc_as, "active_set_phase3")

    # === Phase 4: Reheating and re-optimization (300s) ===
    print(f"\n[Phase 4] Reheating (300s)")
    phase4_deadline = min(deadline - 200, phase3_deadline + 300.0)
    if best_seq is not None and time.monotonic() < phase4_deadline:
        # Try several perturbations with decreasing sigma
        sigmas = [0.02, 0.01, 0.005, 0.002, 0.001]
        for sig in sigmas:
            if time.monotonic() > phase4_deadline - 30:
                break
            a_pert = perturb_fn(best_seq, sig, rng)
            # Quick L-BFGS from perturbed point
            n = len(a_pert)
            z = np.log(np.maximum(a_pert, 1e-30))
            z = z - z.mean()
            c_cur = full_convolve(a_pert, a_pert)
            max_c = float(c_cur.max())
            c_sorted = np.sort(c_cur)[::-1]
            gap = max_c - c_sorted[min(10, len(c_sorted) - 1)]
            base_beta = max(gap * 0.02, max_c * 0.001)
            beta = base_beta

            remaining = phase4_deadline - 30 - time.monotonic()
            if remaining < 5:
                break
            n_scale = max(1.0, n / 16384.0)
            max_iter = min(1000, max(20, int(remaining / (0.005 * n_scale * 3))))

            def obj_reheat(z_local):
                a_local = softmax_fn(z_local)
                c_local = full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted_local = (c_local - c_max_local) / beta
                e_local = np.exp(np.clip(shifted_local, -700, 0))
                s_e_local = e_local.sum()
                if s_e_local < 1e-300:
                    return 2.0 * n * c_max_local
                return 2.0 * n * (c_max_local + beta * np.log(s_e_local))

            def grad_reheat(z_local):
                a_local = softmax_fn(z_local)
                c_local = full_convolve(a_local, a_local)
                grad_local, _ = compute_gradients(a_local, n, beta, c_local)
                return grad_local

            try:
                result = minimize(
                    obj_reheat, z,
                    jac=grad_reheat,
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                             'maxcor': 40}
                )
                a_new = softmax_fn(result.x)
                sc_new = compute_score(a_new)
                if sc_new < best_score:
                    update_best(a_new, sc_new, f"reheat_sig{sig}")
            except Exception:
                pass

    # === Phase 5: Final ultra-tight polish (200s) ===
    print(f"\n[Phase 5] Final polish (200s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        n = len(best_seq)
        # Very tight active set with projected gradient
        a_final = best_seq.copy()
        a_final = a_final / a_final.sum()

        # Active set with K=1 (just the peak)
        for k_top in [1, 2, 3, 5, 10]:
            if time.monotonic() > deadline - 15:
                break
            c = full_convolve(a_final, a_final)
            c_max = float(c.max())
            top_k = np.argsort(c)[::-1][:k_top]
            p = np.zeros(2 * n - 1)
            p[top_k] = 1.0 / k_top

            # Many iterations of projected gradient
            for it in range(500):
                if time.monotonic() > deadline - 10:
                    break
                a_rev = a_final[::-1]
                wc = full_convolve(p, a_rev)
                h = wc[n - 1:2 * n - 1]
                W_val = float(np.dot(p, c))
                grad = 2.0 * a_final * (h - W_val)
                gnorm = np.linalg.norm(grad)
                if gnorm < 1e-18:
                    break
                # Backtracking line search
                lr = 0.5 / (gnorm + 1e-10)
                improved = False
                for ls in range(20):
                    a_try = proj_simplex(a_final - lr * grad)
                    c_try = full_convolve(a_try, a_try)
                    if float(c_try.max()) < c_max - 1e-16:
                        a_final = a_try
                        c = c_try
                        c_max = float(c.max())
                        improved = True
                        break
                    lr *= 0.5
                if not improved:
                    # Try smaller step from best
                    p_top = np.zeros(2 * n - 1)
                    p_top[int(np.argmax(c))] = 1.0
                    p = p_top
                    break

            sc_final = compute_score(a_final)
            if sc_final < best_score:
                update_best(a_final, sc_final, f"final_K={k_top}")
                a_final = a_final / a_final.sum()

        # Final L-BFGS with extremely tight beta
        if time.monotonic() < deadline - 10:
            z = np.log(np.maximum(a_final, 1e-30))
            z = z - z.mean()
            c_b = full_convolve(a_final, a_final)
            mb = float(c_b.max())
            c_sort = np.sort(c_b)[::-1]
            gap_final = mb - c_sort[min(5, len(c_sort) - 1)]
            beta_fin = max(gap_final * 0.005, mb * 1e-7)

            remaining = deadline - 10 - time.monotonic()
            if remaining > 3:
                n_scale = max(1.0, n / 16384.0)
                max_iter = min(2000, max(50, int(remaining / (0.005 * n_scale * 3))))

                def obj_final(z_local):
                    a_local = softmax_fn(z_local)
                    c_local = full_convolve(a_local, a_local)
                    c_max_local = float(c_local.max())
                    shifted_local = (c_local - c_max_local) / beta_fin
                    e_local = np.exp(np.clip(shifted_local, -700, 0))
                    s_e_local = e_local.sum()
                    if s_e_local < 1e-300:
                        return 2.0 * n * c_max_local
                    return 2.0 * n * (c_max_local + beta_fin * np.log(s_e_local))

                def grad_final(z_local):
                    a_local = softmax_fn(z_local)
                    c_local = full_convolve(a_local, a_local)
                    grad_local, _ = compute_gradients(a_local, n, beta_fin, c_local)
                    return grad_local

                try:
                    result = minimize(
                        obj_final, z,
                        jac=grad_final,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                                 'maxcor': 50}
                    )
                    a_flb = softmax_fn(result.x)
                    update_best(a_flb, compute_score(a_flb), "final_lbfgs_ultra")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = make_profile(256, "uniform")
        sc_final_val = compute_score(final)
        print(f"FINAL: score={sc_final_val:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = make_profile(8192, "tapered")
    print(f"FALLBACK: score={compute_score(fallback):.10f}")
    return [float(x) for x in fallback]
