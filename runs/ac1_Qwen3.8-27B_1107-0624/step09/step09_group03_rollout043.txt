"""

1. Structured initialization with asymmetric tilt search

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    """Compute the valid part of corr: sum over k of A_vec[k] * a[k-j] for each j."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec, rhs, max_cg=300, tol=1e-14):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = float(np.dot(r, r))
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = float(np.dot(p, Ap))
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = float(np.dot(r, r))
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _newton_step_active(a, c, c_max, tol_frac):
    """Compute Newton step for the active set optimization."""
    n = len(a)
    active = c >= (c_max - tol_frac * c_max)
    if not active.any():
        active[int(np.argmax(c))] = True
    m = int(active.sum())
    if m < 2 or m > 20 * n:
        return None
    
    A_vec = active.astype(np.float64) / m
    g = 2.0 * _corr_a_A(a, A_vec, n)
    g_t = g - np.mean(g)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-15:
        return None
    
    mu = 1e-8 / m + 1e-10
    
    A_vec_copy = A_vec.copy()
    mu_copy = mu
    n_copy = n
    
    def _hess_vp(v):
        return 2.0 * _corr_a_A(v, A_vec_copy, n_copy) + mu_copy * v
    
    u = _cg_solve(_hess_vp, g_t, max_cg=300)
    w = _cg_solve(_hess_vp, np.ones(n), max_cg=300)
    w_sum = float(w.sum())
    
    if abs(w_sum) < 1e-16:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w
    
    dx = dx - np.mean(dx)
    return dx


def _newton_polish(a_init, deadline, max_outer=200, min_time_remaining=30.0):
    """Active-set Newton polish with adaptive tolerance."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for outer in range(max_outer):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive tolerance: start loose, tighten over iterations
        if outer < 20:
            tol_list = [0.05, 0.02, 0.01, 0.005]
        elif outer < 50:
            tol_list = [0.005, 0.002, 0.001, 0.0005]
        else:
            tol_list = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
        
        best_improvement = 0
        best_a_new = None
        
        for tol_frac in tol_list:
            dx = _newton_step_active(a, c, c_max, tol_frac)
            if dx is None:
                continue
            
            # Backtracking line search
            t = 1.0
            for ls in range(60):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_max_try = float(_full_convolve(a_try, a_try).max())
                if c_max_try < c_max - 1e-14 * t:
                    improvement = c_max - c_max_try
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_a_new = a_try.copy()
                    break
                t *= 0.5
        
        if best_a_new is None:
            # Try gradient descent fallback
            g_all = 2.0 * _corr_a_A(a, np.ones(2 * n - 1) / (2 * n - 1), n)
            g_t = g_all - np.mean(g_all)
            t = 0.01
            for ls in range(30):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_sc - 1e-15:
                        best_a_new = a_try.copy()
                        best_improvement = 1
                        break
                t *= 0.5
        
        if best_a_new is None:
            break
        
        a = best_a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Convergence check
        if best_improvement < 1e-16:
            break
    
    return best_a, best_sc


def _make_window(n, kind, param=0.0):
    """Generate a window function with optional asymmetric tilt."""
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        w = np.ones(n)
    elif kind == "raised_cosine":
        w = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif kind == "hann":
        w = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif kind == "hamming":
        w = 0.54 + 0.46 * np.cos(2 * np.pi * t)
    elif kind == "blackman":
        w = 0.42 + 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    elif kind == "gaussian":
        w = np.exp(-param * (t - 0.5) ** 2)
    elif kind == "triangle":
        w = 1 - 2 * np.abs(t - 0.5)
    elif kind == "exp_ramp":
        w = np.exp(param * t)
    elif kind == "linear_ramp":
        w = 1 + param * (t - 0.5)
    else:
        w = np.ones(n)
    
    # Apply asymmetric tilt
    if param != 0 and kind not in ("exp_ramp", "linear_ramp"):
        w = w * np.exp(param * (t - 0.5))
    
    w = np.maximum(w, 1e-10)
    s = w.sum()
    return w / s

def _lbfgs_continuation(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with log-sum-exp continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)
    
    # Compute initial beta from the gap
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 100:
        gap = c_sorted[0] - c_sorted[100]
    else:
        gap = c_sorted[0] - c_sorted[-1]
    base_beta = max(gap * 0.5, max_c * 0.01)
    
    # Aggressive geometric schedule
    n_betas = 30
    min_beta = max_c * 1e-8
    log_betas = np.linspace(np.log(base_beta), np.log(min_beta), n_betas)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(log_betas):
        beta = float(np.exp(beta))
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        est_time_per_iter = 0.004 * max(1.0, n / 16384.0)
        max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 15))))
        
        try:
            result = minimize(
                _lsobj,
                z,
                args=(beta, n),
                jac=_lsgrad,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 500,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-18:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            if i % 5 == 0:
                print(f"    L-BFGS beta={beta:.2e} (iter {i+1}/{n_betas}): score={current_sc:.12f}")
        except Exception as ex:
            print(f"    L-BFGS beta={beta:.2e}: exception {ex}")
    
    return current_a, current_sc

Optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 using:

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec, rhs, max_cg, tol):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = float(np.dot(r, r))
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = float(np.dot(p, Ap))
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = float(np.dot(r, r))
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            break
        if rs < 1e-30:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


class _HessCtx:
    def __init__(self, A_vec, n, mu):
        self.A_vec = A_vec
        self.n = n
        self.mu = mu


def _hess_matvec(ctx, v):
    return 2.0 * _corr_a_A(v, ctx.A_vec, ctx.n) + ctx.mu * v


def _newton_step_active(a, c, c_max, tol_frac):
    n = len(a)
    active = c >= (c_max - tol_frac * c_max)
    if not active.any():
        active[int(np.argmax(c))] = True
    m = int(active.sum())
    if m < 2 or m > 30 * n:
        return None

    A_vec = active.astype(np.float64) / m
    g = 2.0 * _corr_a_A(a, A_vec, n)
    g_t = g - np.mean(g)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-15:
        return None

    mu = 1e-9 / m + 1e-12
    ctx = _HessCtx(A_vec, n, mu)

    u = _cg_solve(ctx, g_t, max_cg=400, tol=1e-14)
    w = _cg_solve(ctx, np.ones(n), max_cg=400, tol=1e-14)
    w_sum = float(w.sum())

    if abs(w_sum) < 1e-16:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w

    dx = dx - np.mean(dx)
    if not np.all(np.isfinite(dx)):
        return None
    return dx


def _newton_polish(a_init, deadline, max_outer, min_time_remaining):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    c_best = float(_full_convolve(a, a).max())

    for outer in range(max_outer):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        if outer < 10:
            tol_list = [0.05, 0.02, 0.01, 0.005, 0.002]
        elif outer < 40:
            tol_list = [0.005, 0.002, 0.001, 0.0005, 0.0002]
        else:
            tol_list = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]

        best_improvement = 0.0
        best_a_new = None

        for tol_frac in tol_list:
            dx = _newton_step_active(a, c, c_max, tol_frac)
            if dx is None:
                continue

            t = 1.0
            for ls in range(80):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                c_max_try = float(c_try.max())
                if c_max_try < c_max - 1e-13 * t:
                    improvement = c_max - c_max_try
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_a_new = a_try.copy()
                    break
                t *= 0.5

        if best_a_new is None:
            # Gradient fallback with multiple step sizes
            w_flat = np.ones(2 * n - 1) / (2 * n - 1)
            g_all = 2.0 * _corr_a_A(a, w_flat, n)
            g_t = g_all - np.mean(g_all)
            gnorm = np.linalg.norm(g_t)
            if gnorm > 1e-15:
                g_t = g_t / gnorm
                for t in [0.001, 0.003, 0.01, 0.03, 0.1]:
                    a_try = a - t * g_t
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try > 1e-15:
                        a_try = a_try / s_try
                        sc_try = _score(a_try)
                        if sc_try < best_sc - 1e-14:
                            best_a_new = a_try.copy()
                            best_improvement = 1.0
                            break

        if best_a_new is None:
            break

        a = best_a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            c_best = sc / (2.0 * n)

        if best_improvement < 1e-17:
            break

        if outer % 20 == 0:
            print(f"    Newton iter {outer}: sc={best_sc:.12f}, improve={best_improvement:.2e}")

    return best_a, best_sc


def _make_window(n, kind, param):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        w = np.ones(n)
    elif kind == "hann":
        w = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    elif kind == "hamming":
        w = 0.54 + 0.46 * np.cos(2.0 * np.pi * t)
    elif kind == "blackman":
        w = 0.42 + 0.5 * np.cos(2.0 * np.pi * t) + 0.08 * np.cos(4.0 * np.pi * t)
    elif kind == "gaussian":
        w = np.exp(-param * (t - 0.5) ** 2)
    elif kind == "triangle":
        w = 1.0 - 2.0 * np.abs(t - 0.5)
    elif kind == "exp":
        w = np.exp(param * t)
    elif kind == "power":
        w = t ** param
    elif kind == "power_rev":
        w = (1.0 - t) ** param
    else:
        w = np.ones(n)

    if kind in ("hann", "hamming", "blackman", "gaussian", "uniform", "triangle") and param != 0:
        w = w * np.exp(param * (t - 0.5))

    w = np.maximum(w, 1e-10)
    s = w.sum()
    return w / s


def _structured_search(n, deadline):
    results = []
    candidates = [
        ("uniform", 0.0),
        ("hann", 0.0),
        ("hamming", 0.0),
        ("blackman", 0.0),
        ("gaussian", 3.0),
        ("gaussian", 5.0),
        ("gaussian", 8.0),
        ("gaussian", 12.0),
        ("gaussian", 20.0),
        ("triangle", 0.0),
        ("exp", 1.0),
        ("exp", 2.0),
        ("exp", 3.0),
        ("exp", -1.0),
        ("exp", -2.0),
        ("exp", -3.0),
        ("power", 0.5),
        ("power", 1.0),
        ("power", 2.0),
        ("power", 3.0),
        ("power_rev", 0.5),
        ("power_rev", 1.0),
        ("power_rev", 2.0),
        ("power_rev", 3.0),
    ]

    for kind in ["hann", "hamming", "blackman", "gaussian"]:
        for tilt in [-4.0, -3.0, -2.0, -1.5, -1.0, -0.5, 0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
            p = 5.0 if kind == "gaussian" else 0.0
            candidates.append((kind, tilt))
            candidates.append((kind, p + tilt * 0.1))

    best_a = None
    best_sc = np.inf

    for idx, (kind, param) in enumerate(candidates):
        if time.monotonic() > deadline:
            break
        a = _make_window(n, kind, param)
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if idx < 30 or sc < 1.7:
                print(f"  Structured ({kind}, {param:.2f}): score={sc:.10f}")

    return best_a, best_sc


def _tilt_optimize(a, deadline):
    """Optimize a 1D exponential tilt parameter."""
    n = len(a)
    best_sc = _score(a)
    best_a = a.copy()
    t = np.linspace(0, 1, n)

    for eps in np.concatenate([
        np.linspace(-0.1, 0.1, 41),
        np.linspace(-0.5, -0.1, 20),
        np.linspace(0.1, 0.5, 20),
        np.linspace(-2.0, -0.5, 15),
        np.linspace(0.5, 2.0, 15),
    ]):
        if time.monotonic() > deadline:
            break
        a_try = a * np.exp(eps * (t - 0.5))
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        sc = _score(a_try)
        if sc < best_sc:
            best_sc = sc
            best_a = a_try.copy()

    return best_a, best_sc


def _lbfgs_continuation(a_init, deadline, min_time_remaining):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 200:
        gap = c_sorted[0] - c_sorted[200]
    else:
        gap = max(c_sorted[0] - c_sorted[-1], max_c * 0.001)
    base_beta = max(gap * 0.5, max_c * 0.005)

    n_betas = 35
    min_beta = max_c * 5e-9
    if min_beta < 1e-15:
        min_beta = 1e-15
    log_betas = np.linspace(np.log(base_beta), np.log(min_beta), n_betas)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i in range(n_betas):
        beta = float(np.exp(log_betas[i]))
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.003 * max(1.0, n / 16384.0)
        max_iter = min(6000, max(50, int(remaining / (est_time * 20))))

        try:
            result = minimize(
                _lsobj, z, args=(beta, n),
                jac=_lsgrad,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 100,
                    'maxls': 500,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-18:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            if i % 5 == 0 or i == n_betas - 1:
                print(f"    L-BFGS beta={beta:.2e} ({i+1}/{n_betas}): score={current_sc:.12f}")
        except Exception as ex:
            print(f"    L-BFGS beta={beta:.2e}: exception {ex}")

    return current_a, current_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _deflation_step(a, deadline, n_attempts):
    """Multi-peak deflation: reduce all significant peaks simultaneously."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for attempt in range(n_attempts):
        if time.monotonic() + 2 > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        threshold = c_max * 0.98
        peak_mask = c >= threshold
        n_peaks = int(peak_mask.sum())
        if n_peaks == 0:
            peak_mask[int(np.argmax(c))] = True
            n_peaks = 1

        w = np.zeros_like(c)
        w[peak_mask] = 1.0 / n_peaks

        grad = 2.0 * _corr_a_A(a, w, n)
        g_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-15:
            break
        g_t = g_t / gnorm

        applied = False
        for strength in [0.0002, 0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
            a_try = a - strength * g_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-15:
                best_sc = sc_try
                best_a = a_try.copy()
                applied = True
                break

        if not applied:
            break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 45.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf
    best_info = []

    def update_best(p, sc, label):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None or not np.isfinite(sc):
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.12f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    n = 16384

    if parent is not None:
        if len(parent) != n:
            parent = _interpolate_seq(parent, n)
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n}, score={sc_parent:.12f}")
        update_best(parent, sc_parent, "parent")
    else:
        parent = np.ones(n) / n
        update_best(parent, _score(parent), "uniform_init")

    # === Phase 1: Structured search ===
    print(f"\n[Phase 1] Structured search (t={time.monotonic() - t_start:.1f}s)")
    struct_dl = min(deadline - 1100, time.monotonic() + 90)
    a_struct, sc_struct = _structured_search(n, struct_dl)
    if a_struct is not None:
        update_best(a_struct, sc_struct, "structured")

    # Tilt optimization on best so far
    if best_seq is not None:
        tilt_dl = min(deadline - 1080, time.monotonic() + 30)
        a_tilt, sc_tilt = _tilt_optimize(best_seq, tilt_dl)
        update_best(a_tilt, sc_tilt, "tilt")

    # === Phase 2: Main L-BFGS continuation ===
    print(f"\n[Phase 2] L-BFGS continuation (t={time.monotonic() - t_start:.1f}s)")
    lbfgs_dl = min(deadline - 500, time.monotonic() + 650)
    a_lbfgs, sc_lbfgs = _lbfgs_continuation(best_seq, lbfgs_dl, min_time_remaining=500.0)
    update_best(a_lbfgs, sc_lbfgs, "lbfgs_main")

    # === Phase 3: Newton polish ===
    print(f"\n[Phase 3] Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        newton_dl = min(deadline - 350, time.monotonic() + 250)
        a_newton, sc_newton = _newton_polish(best_seq, newton_dl, max_outer=300, min_time_remaining=350.0)
        update_best(a_newton, sc_newton, "newton_main")

    # === Phase 4: Deflation ===
    print(f"\n[Phase 4] Deflation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        def_dl = min(deadline - 300, time.monotonic() + 60)
        a_def, sc_def = _deflation_step(best_seq, def_dl, n_attempts=40)
        update_best(a_def, sc_def, "deflation")

    # === Phase 5: Iterative refinement loop ===
    print(f"\n[Phase 5] Refinement loop (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        for rnd in range(6):
            if time.monotonic() + 80 > deadline:
                break
            # Tilt
            if time.monotonic() + 40 > deadline:
                break
            td = min(deadline - 40, time.monotonic() + 20)
            a_t, sc_t = _tilt_optimize(best_seq, td)
            if sc_t < best_score:
                update_best(a_t, sc_t, f"refine_tilt_{rnd}")

            # L-BFGS small
            if time.monotonic() + 60 > deadline:
                break
            ld = min(deadline - 60, time.monotonic() + 60)
            a_l, sc_l = _lbfgs_continuation(best_seq, ld, min_time_remaining=60.0)
            if sc_l < best_score:
                update_best(a_l, sc_l, f"refine_lbfgs_{rnd}")

            # Newton
            if time.monotonic() + 50 > deadline:
                break
            nd = min(deadline - 50, time.monotonic() + 50)
            a_n, sc_n = _newton_polish(best_seq, nd, max_outer=100, min_time_remaining=50.0)
            if sc_n < best_score:
                update_best(a_n, sc_n, f"refine_newton_{rnd}")

    # === Phase 6: Multi-start restarts ===
    print(f"\n[Phase 6] Multi-start (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        for sigma in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
            if time.monotonic() + 40 > deadline:
                break
            p_start = _perturb(best_seq, sigma, rng)
            rd = min(deadline - 40, time.monotonic() + 50)
            a_r, sc_r = _lbfgs_continuation(p_start, rd, min_time_remaining=40.0)
            update_best(a_r, sc_r, f"restart_s{sigma}")

    # === Phase 7: Try n=32768 ===
    print(f"\n[Phase 7] n=32768 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 100 and best_seq is not None:
        n_large = 32768
        a_large = _interpolate_seq(best_seq, n_large)
        sc_large_init = _score(a_large)
        print(f"  n={n_large} init: {sc_large_init:.12f}")
        ld = min(deadline - 60, time.monotonic() + 80)
        a_lo, sc_lo = _lbfgs_continuation(a_large, ld, min_time_remaining=60.0)
        update_best(a_lo, sc_lo, "n32768_lbfgs")
        if time.monotonic() < deadline - 30:
            nd = min(deadline - 30, time.monotonic() + 40)
            a_ln, sc_ln = _newton_polish(a_lo, nd, max_outer=50, min_time_remaining=30.0)
            update_best(a_ln, sc_ln, "n32768_newton")

    # === Phase 8: Final ultra-precise polish ===
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        n_best = len(best_seq)
        # Ultra-small beta L-BFGS
        remaining = deadline - 8 - time.monotonic()
        if remaining > 10:
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 1e-9
            max_iter_fin = min(5000, max(100, int(remaining / 0.04)))
            try:
                result = minimize(
                    _lsobj, z, args=(beta_final, n_best),
                    jac=_lsgrad, method='L-BFGS-B',
                    options={'maxiter': max_iter_fin, 'ftol': 1e-20, 'gtol': 1e-16, 'maxcor': 100}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_ultra_beta")
            except Exception:
                pass

        # Final Newton
        if time.monotonic() < deadline - 5:
            a_fn, sc_fn = _newton_polish(best_seq, deadline, max_outer=100, min_time_remaining=5.0)
            update_best(a_fn, sc_fn, "final_newton")

        # Final tilt
        if time.monotonic() < deadline - 3:
            a_ft, sc_ft = _tilt_optimize(best_seq, deadline)
            update_best(a_ft, sc_ft, "final_tilt")

    # Final output
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
