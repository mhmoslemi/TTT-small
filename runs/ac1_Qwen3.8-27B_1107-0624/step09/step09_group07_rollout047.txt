import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _maxconv_score(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _pnorm_obj_grad(a, p, n_active=None):
    """Compute p-norm of c = conv(a,a) and its gradient.
    If n_active is given, only use top n_active values."""
    c = _full_convolve(a, a)
    if n_active is not None and n_active < len(c):
        idx = np.argpartition(c, -n_active)[-n_active:]
        c_use = c[idx]
    else:
        c_use = c
        idx = np.arange(len(c))
    
    c_max = float(c_use.max())
    if c_max < 1e-30:
        return 0.0, np.zeros(len(a))
    
    # p-norm of c_use
    cp = c_use ** p
    sum_cp = float(cp.sum())
    obj = sum_cp ** (1.0 / p)
    
    if sum_cp < 1e-300:
        return obj, np.zeros(len(a))
    
    # Gradient: d(obj)/da = obj^{-1} * sum_i c_i^{p-1} * dc_i/da
    weights = c_use ** (p - 1)
    w_full = np.zeros(2 * len(a) - 1)
    w_full[idx] = weights
    w_full = w_full / w_full.sum()  # normalize for stability
    
    grad = _grad_maxconv(a, w_full)
    grad = grad / obj  # chain rule
    
    return obj, grad


class _PnormProb:
    def __init__(self, p, n, n_active=None):
        self.p = p
        self.n = n
        self.n_active = n_active
    
    def __call__(self, z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        return _pnorm_obj_grad(a, self.p, self.n_active)[0]
    
    def grad(self, z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        _, g_a = _pnorm_obj_grad(a, self.p, self.n_active)
        # Chain rule: z -> a (softmax-like)
        W = float(np.dot(a, g_a))
        return a * (g_a - W)


def _topk_smoobj_grad(z, beta, K, n):

    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)

        w[int(np.argmax(c))] = 1.0

def _block_update(a, block_size, deadline, min_time_remaining=10.0):

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses top-K log-sum-exp smoothing with aggressive K annealing,
active-set subgradient with peak balancing, and multi-restart strategy.

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    c = _full_convolve(a, a)
    return float(c.max())


def _obj_normalized(a):
    """Objective assuming sum(a)=1: 2*n*max(c)."""
    n = len(a)
    return 2.0 * n * _maxconv(a)


def _grad_maxconv(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c=conv(a,a)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _topk_smoothing(z, beta, K, n):
    """Top-K log-sum-exp: returns (obj, grad_z) in log-space."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())

    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c

    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s_e = e.sum()

    if s_e < 1e-300 or not np.isfinite(s_e):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s_e
        obj = 2.0 * n * (c_max + beta * np.log(s_e))

    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


class _TopkProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _topk_smoothing(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _topk_smoothing(z, self.beta, self.K, self.n)[1]


def _lp_peak_reduction(a, top_k=5, max_iters=200, deadline=None, min_time=10.0):
    """
    Linearized peak reduction: at each step, linearize the top peaks
    and solve a small QP/LP to reduce the max peak.
    Uses scipy minimize on a piecewise-linear approximation.
    """
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _obj_normalized(a)

    lr = 0.3
    stall = 0

    for it in range(max_iters):
        if deadline is not None and time.monotonic() + min_time > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        cur_sc = 2.0 * n * c_max

        if cur_sc < best_sc - 1e-14:
            best_sc = cur_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break

        # Get top peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]

        # If peaks are already well-balanced, try更小 top_k
        if top_vals.max() - top_vals.min() < c_max * 0.001:
            top_k = max(1, top_k // 2)

        # Compute gradient for each active peak
        grads = []
        for k in top_idx:
            w = np.zeros(2 * n - 1)
            w[k] = 1.0
            grads.append(_grad_maxconv(a, w))
        grads = np.array(grads)  # shape: (top_k, n)

        # For each peak, the linearized constraint is:
        # c[k] + grads[i] @ d <= t  where d = a_new - a (with sum(d)=0)
        # We want to minimize t.
        # This is equivalent to: find d with sum(d)=0, a+d>=0,
        #   grads[i] @ d <= t - c[k] for all i
        # Minimize t.

        # Simpler approach: use the average gradient weighted by peak values
        w_avg = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_avg[k] = 1.0 / top_k
        g_avg = _grad_maxconv(a, w_avg)
        g_centered = g_avg - g_avg.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break

        # Line search along -g_centered
        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        improved = False
        for ls in range(40):
            d = -alpha * g_centered
            a_try = a + d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < cur_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # Try individual peak reduction
            for i, k in enumerate(top_idx[:3]):
                w = np.zeros(2 * n - 1)
                w[k] = 1.0
                g_i = _grad_maxconv(a, w)
                g_i_c = g_i - g_i.mean()
                gnorm_i = np.linalg.norm(g_i_c)
                if gnorm_i < 1e-18:
                    continue
                alpha_i = min(0.1, 0.3 / (gnorm_i + 1e-15))
                for ls in range(20):
                    a_try = a - alpha_i * g_i_c
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-10:
                        alpha_i *= 0.5
                        continue
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    sc_try = 2.0 * n * float(c_try.max())
                    if sc_try < cur_sc - 1e-14:
                        a = a_try
                        improved = True
                        break
                    alpha_i *= 0.5
                if improved:
                    break

        if not improved:
            lr = max(0.01, lr * 0.7)

    return best_a, best_sc


def _run_lbfgs_topk(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with top-K log-sum-exp and aggressive annealing."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    current_sc = _obj_normalized(a)
    current_a = a.copy()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(200, len(c_sorted) - 1)]
    if gap < max_c * 0.0005:
        gap = max_c * 0.0005
    base_beta = max(gap, max_c * 0.003)

    # Aggressive schedule: start smooth, end very sharp
    schedule = [
        (1.0, 0.10), (0.7, 0.08), (0.5, 0.06), (0.35, 0.05),
        (0.25, 0.04), (0.18, 0.03), (0.12, 0.025), (0.08, 0.02),
        (0.05, 0.015), (0.03, 0.012), (0.02, 0.01), (0.01, 0.008),
        (0.006, 0.006), (0.003, 0.004), (0.0015, 0.003), (0.0008, 0.002),
        (0.0004, 0.0015), (0.0002, 0.001), (0.0001, 0.0008),
    ]

    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 15:
            break

        beta = base_beta * beta_mult
        K = max(2, int((2 * n - 1) * K_frac))

        est_per_iter = 0.003 * max(1.0, n / 16384.0)
        max_iter = min(20000, max(50, int(remaining / (est_per_iter * 20))))

        prob = _TopkProb(beta, K, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-22, 'gtol': 1e-18,
                         'maxcor': 200, 'maxls': 100}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _obj_normalized(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgrad_peak_balance(a_init, deadline, min_time_remaining=30.0):
    """
    Subgradient method with peak balancing:
    - Focus on the top peaks
    - Use adaptive step sizes with BB rule
    - Add a "balancing" term that tries to equalize top peaks
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _obj_normalized(a)

    n_2nm1 = 2 * n - 1
    momentum = 0.4
    lr = 0.02
    lr_min = 5e-7
    lr_max = 0.1
    stall = 0
    max_stall = 150
    velocity = np.zeros(n)
    prev_a = None
    prev_g = None
    top_k = 10

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        cur_sc = 2.0 * n * c_max

        if cur_sc < best_sc - 1e-14:
            best_sc = cur_sc
            best_a = a.copy()
            stall = 0
            # Gradually reduce top_k as we converge
            if best_sc < 2.0 * n * max_c * 0.75:
                top_k = max(3, top_k - 1)
        else:
            stall += 1

        # Get top peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]

        # Weighted gradient: weight peaks by how much they exceed the mean
        w = np.zeros(n_2nm1)
        w_mean = top_vals.mean()
        w_range = top_vals.max() - w_mean
        for i, k in enumerate(top_idx):
            if w_range > 1e-18:
                w[k] = (top_vals[i] - w_mean) / w_range + 0.3
            else:
                w[k] = 1.0 / top_k
        w = w / w.sum()

        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break

        # BB step
        if prev_a is not None and prev_g is not None:
            sv = a - prev_a
            yv = g_centered - prev_g
            sn = float(np.dot(sv, sv))
            sy = float(np.dot(sv, yv))
            if sn > 1e-18 and abs(sy) > 1e-18:
                lr_bb = sn / abs(sy)
                lr = float(np.clip(lr_bb, lr_min, lr_max))

        velocity = momentum * velocity - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()

        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            for ls in range(35):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < cur_sc - 1e-14:
                    improved = True
                    break
                velocity = (velocity + lr * g_centered) * 0.5 - lr * 0.5 * g_centered
                a_try = a + velocity
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    velocity = np.zeros(n)
                    break
                a_try = a_try / s_try
                lr *= 0.65

        if improved:
            prev_a = a.copy()
            prev_g = g_centered.copy()
            a = a_try
            lr = min(lr_max, lr * 1.03)
        else:
            velocity = np.zeros(n)
            lr = max(lr_min, lr * 0.5)

        if stall > max_stall:
            momentum = min(0.8, momentum + 0.1)
            top_k = min(20, top_k + 5)
            lr = max(lr_min, lr * 0.4)
            stall = 0

        if stall > max_stall * 3:
            break

    return best_a, best_sc


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    """Very fine polish: tiny steps on the active set."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _obj_normalized(a)

    lr = 0.002
    tol = 0.002
    momentum = 0.2
    velocity = np.zeros(n)
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        cur_sc = 2.0 * n * c_max

        if cur_sc < best_sc - 1e-15:
            best_sc = cur_sc
            best_a = a.copy()
            no_improve = 0

        # Active set: peaks within tol of max
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active < 1:
            active[int(np.argmax(c))] = True
            n_active = 1

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active

        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-16:
            break

        alpha = min(lr, 0.2 / (gnorm + 1e-15))
        velocity = momentum * velocity - alpha * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()

        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            for ls in range(50):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < cur_sc - 1e-15:
                    a = a_try
                    improved = True
                    break
                alpha *= 0.5
                velocity = momentum * velocity - alpha * g_centered
                a_try = a + velocity
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    velocity = np.zeros(n)
                    a_try = a
                    break
                a_try = a_try / s_try

        if not improved:
            no_improve += 1
            if no_improve % 40 == 0:
                tol = max(0.00005, tol * 0.6)
                lr = max(0.0001, lr * 0.6)
                momentum = max(0.05, momentum * 0.7)
                velocity = np.zeros(n)
            if no_improve > 500:
                # Restart from best
                a = best_a.copy()
                tol = max(0.00005, tol * 0.5)
                lr = 0.001
                no_improve = 0
                if tol <= 0.00006:
                    break
        else:
            if tol > 0.00005:
                tol = max(0.00005, tol * 0.97)
            lr = min(0.005, lr * 1.01)

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS with top-K smoothing, peak-balancing
    subgradient, LP peak reduction, fine polish, and multi-restart.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_init = 16384
        t = np.linspace(0, 1, n_init)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        parent = a / a.sum()
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, tapered n={n_init}, score={best_score:.10f}")

    a_norm = best_seq / best_seq.sum()
    n_main = len(best_seq)

    # === Phase 1: L-BFGS top-K (main optimization, 500s) ===
    print(f"\n[P1] L-BFGS topK main (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 500, t_start + 500)
    a1 = _run_lbfgs_topk(a_norm, p1_deadline, min_time_remaining=100)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [P1] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Peak-balancing subgradient (300s) ===
    print(f"\n[P2] Peak-balance subgrad (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        p2_deadline = min(deadline - 300, time.monotonic() + 350)
        a2, sc2_norm = _subgrad_peak_balance(best_seq / best_seq.sum(), p2_deadline, min_time_remaining=60)
        sc2 = _score(a2)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [P2] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: LP peak reduction (200s) ===
    print(f"\n[P3] LP peak reduction (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        p3_deadline = min(deadline - 200, time.monotonic() + 250)
        a3, sc3_norm = _lp_peak_reduction(best_seq / best_seq.sum(), top_k=8,
                                          max_iters=5000, deadline=p3_deadline, min_time=30)
        sc3 = _score(a3)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [P3] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: L-BFGS round 2 (200s) ===
    print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        p4_deadline = min(deadline - 150, time.monotonic() + 200)
        a4 = _run_lbfgs_topk(best_seq / best_seq.sum(), p4_deadline, min_time_remaining=50)
        sc4 = _score(a4)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [P4] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Fine polish (150s) ===
    print(f"\n[P5] Fine polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        p5_deadline = min(deadline - 60, time.monotonic() + 150)
        a5, sc5_norm = _fine_polish(best_seq / best_seq.sum(), p5_deadline, min_time_remaining=30)
        sc5 = _score(a5)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [P5] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-restart (200s) ===
    print(f"\n[P6] Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        remain = deadline - 60 - time.monotonic()
        n_restart = max(3, int(min(remain * 0.5, 200) / 30))
        sigmas = [0.015, 0.03, 0.05, 0.02, 0.07, 0.01, 0.04, 0.08, 0.025, 0.06]
        for i in range(n_restart):
            if time.monotonic() > deadline - 50:
                break
            sigma = sigmas[i % len(sigmas)]
            a_pert = _perturb(best_seq / best_seq.sum(), sigma, rng)
            r_deadline = min(deadline - 40, time.monotonic() + 28)
            a_r = _run_lbfgs_topk(a_pert, r_deadline, min_time_remaining=5)
            # Quick polish
            sc_r_pre = _score(a_r)
            if sc_r_pre < best_score + 0.002:  # Close enough to try polishing
                p_pol_deadline = min(deadline - 35, time.monotonic() + 10)
                a_pol, _ = _fine_polish(a_r / a_r.sum(), p_pol_deadline, min_time_remaining=2)
                sc_r = _score(a_pol)
                if sc_r < sc_r_pre:
                    a_r = a_pol
            else:
                sc_r = sc_r_pre
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.10f} [P6_r{i}_s{sigma}] t={time.monotonic()-t_start:.1f}s")
        print(f"  After restarts: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Different n (100s) ===
    print(f"\n[P7] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        n_options = [16384, 15360, 17408, 20480, 12288, 18432]
        n_options = [x for x in n_options if x != n_main]
        for n_try in n_options[:3]:
            if time.monotonic() > deadline - 20:
                break
            a_res = _resample(best_seq / best_seq.sum(), n_try)
            r_deadline = min(deadline - 15, time.monotonic() + 30)
            a_opt = _run_lbfgs_topk(a_res, r_deadline, min_time_remaining=5)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [P7_n{n_try}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final fine polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 2:
        a_fin, _ = _fine_polish(best_seq / best_seq.sum(), deadline - 1, min_time_remaining=1)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [P8] t={time.monotonic()-t_start:.1f}s")

        # Tiny L-BFGS at the very end
        remaining = deadline - 1.5 - time.monotonic()
        if remaining > 8:
            n = len(best_seq)
            c_best = _full_convolve(best_seq / best_seq.sum(), best_seq / best_seq.sum())
            mb = float(c_best.max())
            beta_fin = max(mb * 0.0001, 1e-10)
            K_fin = max(2, int(n * 0.0005))
            a_n = best_seq / best_seq.sum()
            z = np.log(np.maximum(a_n, 1e-30))
            z = z - z.mean()
            prob = _TopkProb(beta_fin, K_fin, n)
            max_iter = min(5000, max(30, int(remaining / 0.03)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-22,
                                           'gtol': 1e-18, 'maxcor': 200})
                a_fin2 = np.exp(result.x - result.x.max())
                a_fin2 = a_fin2 / a_fin2.sum()
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [P8_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        t = np.linspace(0, 1, 256)
        final = np.sin(np.pi * t) ** 0.6
        final = np.maximum(final, 1e-10)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
