def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):

        return np.inf
    c = _full_convolve(a, a)

def _normalize(a):

def _grad_maxconv(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a).
    w has length 2n-1. Returns array of length n."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _topk_smoobj_grad(z, beta, K, n):

    Only the top K values of c contribute to the smoothing."""

class _TopkSmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[1]

def _subgradient_bb(a_init, deadline, min_time_remaining=60.0):

    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    c = _full_convolve(a, a)
    n_2nm1 = 2 * n - 1
    
    active_frac = 0.01
    momentum = 0.7
    lr = 0.02
    lr_min = 2e-5
    lr_max = 0.1
    stall = 0
    max_stall = 80
    velocity = np.zeros(n)
    
    prev_a = None
    prev_g = None
    
    iteration = 0
    t_start = time.monotonic()
    last_lbfgs_time = t_start
    
    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Select active peaks
        n_active = max(3, min(n_2nm1, int(n_2nm1 * active_frac)))
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_active]
        
        # Weights: proportional to how much above the median of active peaks
        active_vals = c[active_idx]
        w = np.zeros(n_2nm1)
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w[active_idx] = (active_vals - active_vals.min()) / spread + 0.1
        else:
            w[active_idx] = 1.0 / n_active
        w = w / w.sum()
        
        # Gradient
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break
        
        # BB step size
        if prev_a is not None and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g_centered - prev_g
            s_norm = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm > 1e-18 and abs(sy) > 1e-18:
                lr_bb = s_norm / abs(sy)
                lr = np.clip(lr_bb, lr_min, lr_max)
        
        # Momentum step
        velocity = momentum * velocity - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        
        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try

            for ls in range(25):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-15:
                    improved = True
                    break
                # Halve
                velocity = momentum * velocity * 0.5 - lr * 0.5 * g_centered
                a_try = a + velocity
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    velocity = np.zeros(n)
                    a_try = a
                    break
                a_try = a_try / s_try
                lr *= 0.7
        
        if improved:
            prev_a = a.copy()
            prev_g = g_centered.copy()
            a = a_try
            lr = min(lr_max, lr * 1.03)
            # Shrink active set
            active_frac = max(0.002, active_frac * 0.98)
        else:
            velocity = np.zeros(n)
            lr = max(lr_min, lr * 0.6)
            active_frac = min(0.08, active_frac * 1.3)
        
        # Adapt on stall
        if stall > max_stall:
            active_frac = min(0.08, active_frac * 2.0)
            lr = max(lr_min, lr * 0.3)
            momentum = min(0.9, momentum + 0.05)
            stall = 0
        
        # Periodic L-BFGS restart
        if iteration % 500 == 0 and time.monotonic() - last_lbfgs_time > 120:

                lbfgs_deadline = time.monotonic() + 80

                sc_tmp = _score(a_tmp)
                if sc_tmp < best_sc:
                    best_a = a_tmp.copy()
                    best_sc = sc_tmp
                    a = a_tmp.copy()
                    velocity = np.zeros(n)
                    prev_a = None
                    prev_g = None
                    last_lbfgs_time = time.monotonic()
    
    return best_a, best_sc


def _direct_peak_reduction(a_init, deadline, min_time_remaining=30.0):
    """Directly reduce the top few peaks by adjusting contributing elements."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    t_start = time.monotonic()
    patience = 0
    max_patience = 200
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        
        # Find top peak
        k_peak = int(np.argmax(c))
        
        # Compute contributions to this peak
        # c[k_peak] = sum_i a[i]*a[k_peak-i]
        i_lo = max(0, k_peak - (n - 1))
        i_hi = min(n - 1, k_peak)
        contrib = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
            j = k_peak - i
            contrib[i] += a[j]
            if i != j:
                contrib[j] += a[i]
        
        # Reduce top contributors
        n_reduce = max(1, int(n * 0.002))
        top_contrib = np.argsort(contrib)[::-1][:n_reduce]
        
        reduction_amount = c_max * 0.005
        delta = np.zeros(n)
        for idx in top_contrib:
            if contrib[idx] > 1e-15:
                delta[idx] = -reduction_amount / (2.0 * contrib[idx] + 1e-30)
        
        # Redistribute to low contributors
        low_idx = np.argsort(contrib)[:n_reduce]
        for idx in low_idx:
            delta[idx] = reduction_amount * 0.5 / (n_reduce * (a[idx] + 1e-10) + 1e-30)
        
        delta -= delta.mean()  # Preserve sum
        
        # Line search
        alpha = 1.0
        improved = False
        for ls in range(30):
            a_try = a + alpha * delta
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            patience += 10
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.4 + 0.6 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t) + 0.1
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2) + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "optimal_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.5 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _fine_polish(a_init, deadline, min_time_remaining=10.0):

        c_max = float(c.max())

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        # Active set: peaks within tol of max
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)

            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:

                lr *= 0.7
            if no_improve > 300:
                a = best_a.copy()
