```python
"""
KKT active-set optimizer with FFT-based linear solver for minimum peak autocorrelation.
Combines direct KKT solving, L-BFGS with power-mean objective, and multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize

t_start = 0.0

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _hankel_mv(a, lam):
    """Compute G @ a where G[i,j] = 2*lam[i+j] for i,j in [0,n-1]."""
    n = len(a)
    N = _next_pow2(3 * n)
    A = np.fft.rfft(a, N)
    L = np.fft.rfft(lam, N)
    # (f ⋆ g)(i) = sum_k f[k] g[i+k], DFT = conj(F) * G
    R = np.fft.irfft(np.conj(A) * L, N)[:n]
    return 2.0 * R

def _solve_kkt(n, lam, max_cg_iter=200):
    """Solve the KKT linear system using CG with FFT mat-vec."""
    # System: G @ a - mu * 1 = 0, 1^T @ a = 1
    # Combined: [G  -1; 1^T  0] [a; mu] = [0; 1]
    # 
    # We'll solve using a block approach:
    # From the first eq: a = G^{-1} @ mu * 1 = mu * G^{-1} @ 1
    # Substituting into second: mu * 1^T @ G^{-1} @ 1 = 1, so mu = 1 / (1^T @ G^{-1} @ 1)
    # Then a = G^{-1} @ 1 / (1^T @ G^{-1} @ 1)
    #
    # So we need to solve G @ x = 1 (for x = G^{-1} @ 1)
    # Then mu = 1 / sum(x), a = x / sum(x)
    #
    # Use CG to solve G @ x = 1.
    
    ones = np.ones(n)
    b = ones  # RHS
    
    # CG for G @ x = b
    x = np.ones(n) / n  # initial guess
    r = b - _hankel_mv(x, lam)
    p = r.copy()
    rs = float(r @ r)
    lam_sum = float(lam.sum())
    
    for it in range(max_cg_iter):
        Ap = _hankel_mv(p, lam)
        pAp = float(p @ Ap)
        if abs(pAp) < 1e-30:
            break
        alpha = rs / pAp
        x += alpha * p
        r_new = r - alpha * Ap
        rs_new = float(r_new @ r_new)
        if np.sqrt(rs_new) < 1e-12 * np.sqrt(rs + 1e-30):
            r = r_new
            break
        p_new = r_new + (rs_new / rs) * p
        p = p_new
        r = r_new
        rs = rs_new
    
    x_sum = float(x.sum())
    if x_sum < 1e-15:
        return np.ones(n) / n
    a_sol = x / x_sum
    a_sol = np.maximum(a_sol, 0.0)
    s = a_sol.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_sol / s
```

Actually, you know what, let me try a completely different approach that avoids the linear system entirely.

**Direct optimization using the dual variables:**

Instead of solving for a given lambda, optimize over both a and lambda simultaneously.

The objective is: min_t t s.t. R(k) <= t for all k, sum a = 1, a >= 0.

 equivalently: min_{a, t} t s.t. sum a = 1, a >= 0, and max_k R(k) <= t.

We can parameterize as: minimize f(a) = max_k R(k) subject to sum a = 1, a >= 0.

The subdifferential of f at a is: conv{grad R(k) : R(k) = max R}.

At the optimum, 0 is in the subdifferential (projected onto the tangent space).

So: 0 = proj_{T} (sum_{k in A} lambda_k grad R(k)) for some lambda (probability vector on A).

Where T is the tangent space of the simplex: {v : sum v_i = 0}.

This is the same condition as before. But instead of solving the linear system, we can use an **alternating direction method**:

1. Given current a, compute active set A.
2. Find lambda that minimizes ||sum_{k in A} lambda_k grad R(k) - mean(sum_{k in A} lambda_k grad R(k))|| (i.e., find the lambda that makes the projected gradient closest to zero).
3. Use this lambda to compute the search direction.
4. Line search along the direction.
5. Update a.

Step 2 is a quadratic program in lambda, which is small (|A| variables). This is tractable!

def _find_optimal_lambda(a, active_idx):
    """Find lambda on active set that minimizes the projected gradient norm."""
    n = len(a)
    m = len(active_idx)
    
    # grad R(k)_i = 2 * a_{k-i} (if valid)
    # We need: for each k in active, the gradient vector grad_k in R^n
    # Then find lambda minimizing ||sum_k lambda_k * (grad_k - mean(grad_k))||^2
    
    # This is: min_lambda ||G_lambda * lambda - (1/n) * sum_i (sum_k lambda_k grad_k_i) * 1||^2
    # where G_lambda is the matrix with columns (grad_k - mean(grad_k))
    
    # Simpler: min_lambda lambda^T H lambda - 2 lambda^T b + c
    # where H[k,l] = (grad_k - mean_k) . (grad_l - mean_l)
    # and b, c are constants

    # f(lambda) = ||sum_k lambda_k * grad_k||^2 - (1/n) * (sum_k lambda_k * sum_i grad_k_i)^2

    # - s_k for each k in active set

    # Build H matrix:

    # => lambda = (2H - mu*1*1^T H)^{-1} ...

    try:

        lambda_eq = H_inv_1 / float(np.sum(H_inv_1))
        if np.all(lambda_eq >= -1e-10):

    except:
        pass
    
    # Fallback: solve QP with inequality constraints
    def qp_obj(lam_vec):
        return float(lam_vec @ H @ lam_vec)
    
    def qp_grad(lam_vec):
        return 2.0 * (H @ lam_vec)

    bounds = [(0.0, 1.0)] * m
    x0 = np.ones(m) / m
    res = minimize_qp(qp_obj, x0, jac=qp_grad, method='SLSQP',
                      constraints=cons, bounds=bounds,
                      options={'maxiter': 100, 'ftol': 1e-15})

def _full_convolve(a, b):

    fa = np.fft.rfft(a, N)

def _compute_grad_lag(a, k, n):
    """Compute gradient of R(k) with respect to a: grad_i = 2*a_{k-i} if valid."""
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if hi >= lo:

        a_idx_hi = k - lo
        a_idx_lo = k - hi
        grad[lo:hi+1] = 2.0 * a[a_idx_lo:a_idx_hi+1][::-1]
    return grad

def _kkt_residual(a, c, active_idx, n):
    """Compute the KKT residual using the optimal dual variables.
    Returns (residual_norm, optimal_lambda, search_direction)."""
    m = len(active_idx)
    if m == 0:
        return 0.0, np.array([]), np.zeros(n)
    if m == 1:
        grad = _compute_grad_lag(a, active_idx[0], n)
        proj = grad - np.mean(grad)
        return float(np.linalg.norm(proj)), np.array([1.0]), -proj
    
    # Build H matrix: H[ii,jj] = g_ii^T g_jj - (1/n) * s_ii * s_jj
    # where g_ii^T g_jj = 4 * c[|active_ii - active_jj|] (autocorrelation of gradients)
    # and s_ii = sum of gradient components = 2 * (sum of a over valid window for lag k)
    
    # Compute s values using prefix sums
    a_prefix = np.zeros(n + 1)
    a_prefix[1:] = np.cumsum(a)
    
    s_vals = np.zeros(m)
    for ii in range(m):
        k = int(active_idx[ii])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            s_vals[ii] = 2.0 * (a_prefix[hi+1] - a_prefix[lo])
    
    # Build H
    H = np.zeros((m, m))
    for ii in range(m):
        k_i = int(active_idx[ii])
        for jj in range(ii, m):
            k_j = int(active_idx[jj])
            d = abs(k_i - k_j)
            # g_ii^T g_jj = 4 * R(d) where R is full autocorrelation
            # R(d) = c[d] for d in [0, 2n-2]
            val = 4.0 * float(c[d]) - (1.0 / n) * s_vals[ii] * s_vals[jj]
            H[ii, jj] = val
            H[jj, ii] = val
    
    # Solve: min lambda^T H lambda s.t. sum lambda = 1, lambda >= 0
    # First try equality constraint only: lambda = H^{-1} 1 / (1^T H^{-1} 1)
    try:
        eye_reg = 1e-14 * np.eye(m)
        H_inv_1 = np.linalg.solve(H + eye_reg, np.ones(m))
        lam_sum = float(H_inv_1.sum())
        if abs(lam_sum) > 1e-15:
            lambda_eq = H_inv_1 / lam_sum
            if np.all(lambda_eq >= -1e-8):
                lambda_opt = np.maximum(lambda_eq, 0.0)
                lambda_opt = lambda_opt / lambda_opt.sum()
            else:
                # Need inequality constraints - use SLSQP
                def qp_obj(lv):
                    return float(lv @ H @ lv)
                def qp_grad(lv):
                    return 2.0 * (H @ lv)
                cons = [{'type': 'eq', 'fun': lambda x: x.sum() - 1.0, 'jac': lambda x: np.ones(m)}]
                bounds = [(0.0, 1.0)] * m
                x0 = np.ones(m) / m
                if m <= 50:
                    res = minimize(qp_obj, x0, jac=qp_grad, method='SLSQP',
                                   constraints=cons, bounds=bounds,
                                   options={'maxiter': 200, 'ftol': 1e-16})
                    lambda_opt = res.x
                else:
                    lambda_opt = np.ones(m) / m
        else:
            lambda_opt = np.ones(m) / m
    except:
        lambda_opt = np.ones(m) / m
    
    # Compute residual: proj_T(sum_k lambda_k grad_k)
    # = sum_k lambda_k grad_k - mean(sum_k lambda_k grad_k)
    combined_grad = np.zeros(n)
    for ii in range(m):
        combined_grad += lambda_opt[ii] * _compute_grad_lag(a, active_idx[ii], n)
    residual = combined_grad - np.mean(combined_grad)
    res_norm = float(np.linalg.norm(residual))
    
    return res_norm, lambda_opt, -residual
