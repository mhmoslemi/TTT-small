"""
LP-based active set optimizer with L-BFGS hybrid for minimizing
2*n*max(conv(a,a))/sum(a)^2.
Uses constraint generation (cutting plane) with LP subproblems,
combined with smooth optimization for fast convergence.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf, 0.0
    c = _full_convolve(a, a)
    mb = float(c.max())
    return float(2.0 * n * mb / (s * s)), mb


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _topk_smoobj_grad(z, beta, K, n):
    z_max = z.max()
    a = np.exp(z - z_max)
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c
    
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    
    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


class _TopkSmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[1]


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc, _ = _score(a)
    current_a = a.copy()
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.008)
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    schedule = [
        (1.0, 0.1), (0.7, 0.08), (0.5, 0.06), (0.3, 0.04),
        (0.2, 0.03), (0.1, 0.02), (0.05, 0.015), (0.03, 0.01),
        (0.01, 0.008), (0.005, 0.005), (0.002, 0.003), (0.001, 0.002),
    ]
    
    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        beta = base_beta * beta_mult
        K = max(5, int((2 * n - 1) * K_frac))
        
        est_time = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(6000, max(50, int(remaining / (est_time * 20))))
        
        prob = _TopkSmoProb(beta, K, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 50}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new, _ = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

    c[n] = 1.0
    
    # Bounds: g >= 0, t >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.success:
            g = result.x[:n]
            t = result.x[n]
            g = np.maximum(g, 0)
            g = g / g.sum()
            return g, t
    except Exception:
        pass
    
    return f.copy(), float(_full_convolve(f, f)[active_indices].max())


def _lp_refine(a_init, deadline, min_time_remaining=60.0):
    """
    LP-based active set refinement using constraint generation.
    """
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc, best_mb = _score(a)
    
    K_initial = 50
    K_max = 300
    blend_rate = 0.3
    stall = 0
    max_stall = 20
    
    t_start = time.monotonic()
    
    while time.monotonic() + min_time_remaining < deadline and stall < max_stall:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Select active set: top K peaks
        K = min(K_max, K_initial + stall * 10)
        active_idx = np.argpartition(c, -K)[-K:]
        active_idx = np.sort(active_idx)
        
        # Solve LP
        g, t_opt = _lp_active_set(a, active_idx, deadline, min_time_remaining)
        
        if g is not None and t_opt < c_max:
            # Blend
            a_new = (1 - blend_rate) * a + blend_rate * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            
            # Check all constraints (not just active)
            c_new = _full_convolve(a_new, a_new)
            sc_new = 2.0 * n * float(c_new.max())
            
            if sc_new < current_sc * (1 + 1e-10):
                a = a_new
                # Gradually increase K
                K_initial = min(K_max, K_initial + 5)
                blend_rate = max(0.1, blend_rate * 0.95)
            else:
                # Constraint violation, add violated constraints
                # This is handled by increasing K next iteration
                K_initial = min(K_max, K + 20)
        else:
            # LP didn't help, try smaller blend
            blend_rate = max(0.05, blend_rate * 0.5)
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "optimal_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "butterfly":
        # Two peaks
        t = np.linspace(0, 1, n)
        a = np.exp(-30*(t-0.25)**2) + np.exp(-30*(t-0.75)**2) + 0.02
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "uniform_taper":
        # Mostly uniform with strong tapering at edges
        t = np.linspace(0, 1, n)
        a = np.minimum(1.0, np.sin(np.pi * t) ** 0.3)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    """Very fine active-set smooth polish."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)[0]
    
    lr = 0.003
    tol = 0.003
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active
        
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-16:
            break
        
        alpha = min(lr, 0.3 / (gnorm + 1e-15))
        improved = False
        for ls in range(30):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve % 15 == 0:
                tol = max(0.0001, tol * 0.7)
                lr *= 0.7
            if no_improve > 200:
                a = best_a.copy()
                tol = max(0.0001, tol * 0.5)
                lr = 0.002
                no_improve = 0
        else:
            if tol > 0.0001:
                tol = max(0.0001, tol * 0.97)
            lr = min(0.008, lr * 1.01)
    
    return best_a, best_sc
