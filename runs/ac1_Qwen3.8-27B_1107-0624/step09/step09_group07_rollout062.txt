"""
KKT Active-Set Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: Solves the exact KKT linear system for a given active set,
providing a much faster convergence than gradient-based methods.
Combined with L-BFGS polishing and multi-restart for robustness.
"""
import numpy as np
import time
from scipy import sparse
from scipy.sparse.linalg import spsolve
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _solve_kkt(active_peaks, weights, n):
    """Solve the KKT linear system for given active peaks and weights.
    
    The system is:
      W * a[j] + sum_r w_r * a[k_r - j] = lambda, for j = 0,...,n-1
      sum_j a[j] = 1
    
    where W = sum_r w_r, k_r are active peak positions.
    
    Returns (a, lambda) or None if the system is singular.
    """
    m = len(active_peaks)
    if m == 0:
        return None
    W = float(np.sum(weights))
    if W < 1e-15:
        return None
    
    # Count non-zeros
    nz_count = n  # diagonal
    for r in range(m):
        k = int(active_peaks[r])
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        nz_count += (j_hi - j_lo + 1)
    nz_count += n + 1  # lambda column + last row
    
    rows = np.empty(nz_count, dtype=np.int32)
    cols = np.empty(nz_count, dtype=np.int32)
    vals = np.empty(nz_count, dtype=np.float64)
    idx = 0
    
    # Diagonal: W
    rows[idx:idx+n] = np.arange(n, dtype=np.int32)
    cols[idx:idx+n] = np.arange(n, dtype=np.int32)
    vals[idx:idx+n] = W
    idx += n
    
    # Shift terms
    for r in range(m):
        k = int(active_peaks[r])
        w = float(weights[r])
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        nj = j_hi - j_lo + 1
        j_range = np.arange(j_lo, j_hi + 1, dtype=np.int32)
        col_range = (k - j_range).astype(np.int32)
        rows[idx:idx+nj] = j_range
        cols[idx:idx+nj] = col_range
        vals[idx:idx+nj] = w
        idx += nj
    
    # Lambda column (first n rows)
    rows[idx:idx+n] = np.arange(n, dtype=np.int32)
    cols[idx:idx+n] = n
    vals[idx:idx+n] = -1.0
    idx += n
    
    # Last row: sum(a) = 1
    rows[idx:idx+n] = np.arange(n, dtype=np.int32)
    cols[idx:idx+n] = np.arange(n, dtype=np.int32)
    vals[idx:idx+n] = 1.0
    idx += n
    rows[idx] = n
    cols[idx] = n
    vals[idx] = 0.0
    idx += 1
    
    A = sparse.coo_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(n+1, n+1)).tocsr()
    b = np.zeros(n+1)
    b[n] = 1.0
    
    try:
        x = spsolve(A, b)
    except Exception:
        return None
    
    if not np.all(np.isfinite(x)):
        return None
    
    a = x[:n]
    lam = x[n]
    return a, lam


def _kkt_iteration(a, deadline, tol=0.001, max_iter=5000, min_time_remaining=60):
    """KKT active-set iteration.
    
    Iteratively:
    1. Find active peaks (within tol of max)
    2. Solve KKT linear system
    3. Update a (with projection if needed)
    4. Repeat
    """
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    stall = 0
    max_stall = 100
    damping = 0.5  # step size towards KKT solution
    
    for it in range(max_iter):
        if time.monotonic() + min_time_remaining >= deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                # Reset to best and try different tolerance
                a = best_a.copy()
                tol = max(0.0001, tol * 0.5)
                damping = min(1.0, damping * 1.5)
                stall = 0
                if tol < 0.0001:
                    break
        
        # Find active peaks
        active_thresh = c_max * (1.0 - tol)
        active_mask = c >= active_thresh
        n_active = int(active_mask.sum())
        
        if n_active == 0:
            active_idx = [int(np.argmax(c))]
        elif n_active > 200:
            # Too many active peaks; use top 200
            top_idx = np.argpartition(c, -200)[-200:]
            active_idx = top_idx
        else:
            active_idx = np.where(active_mask)[0]
        
        # Weights: uniform for active peaks
        weights = np.ones(len(active_idx)) / len(active_idx)
        
        # Solve KKT system
        result = _solve_kkt(active_idx, weights, n)
        if result is None:
            # Fallback: use gradient step
            w_full = np.zeros(2*n-1)
            w_full[active_idx] = weights
            g = _grad_maxconv(a, w_full)
            g_centered = g - g.mean()
            a_new = a - 0.01 * g_centered / (np.linalg.norm(g_centered) + 1e-15)
            a_new = np.maximum(a_new, 0)
            a = _normalize(a_new)
            continue
        
        a_kkt, lam = result
        
        # Check non-negativity
        if np.all(a_kkt >= -1e-10):
            a_kkt = np.maximum(a_kkt, 0)
            a_kkt = _normalize(a_kkt)
            # Use full step
            a_new = a_kkt
        else:
            # Damped step towards KKT solution
            a_new = (1 - damping) * a + damping * a_kkt
            a_new = np.maximum(a_new, 0)
            a_new = _normalize(a_new)
        
        # Verify improvement
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())
        
        if sc_new < current_sc + 1e-12:
            a = a_new
            # Increase damping if we made progress
            if sc_new < current_sc - 1e-14:
                damping = min(1.0, damping * 1.1)
        else:
            # Step was too large; reduce damping
            damping = max(0.05, damping * 0.5)
            a_new2 = (1 - damping) * a + damping * a_kkt
            a_new2 = np.maximum(a_new2, 0)
            a_new2 = _normalize(a_new2)
            c_new2 = _full_convolve(a_new2, a_new2)
            if 2.0 * n * float(c_new2.max()) < current_sc:
                a = a_new2
    
    return best_a, best_sc

"""
KKT Active-Set Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: Solves the exact KKT linear system for a given active set of peaks,
providing Newton-like convergence much faster than gradient methods.
Combined with L-BFGS polishing, adaptive active set management, and multi-restart.
"""
import numpy as np
import time
from scipy import sparse
from scipy.sparse.linalg import spsolve
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _solve_kkt(active_peaks, weights, n):
    m = len(active_peaks)
    if m == 0:
        return None
    W = float(np.sum(weights))
    if W < 1e-15:
        return None

    nz_count = n
    for r in range(m):
        k = int(active_peaks[r])
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        nz_count += (j_hi - j_lo + 1)
    nz_count += n + 1

    rows = np.empty(nz_count, dtype=np.int32)
    cols = np.empty(nz_count, dtype=np.int32)
    vals = np.empty(nz_count, dtype=np.float64)
    idx = 0

    rows[idx:idx+n] = np.arange(n, dtype=np.int32)
    cols[idx:idx+n] = np.arange(n, dtype=np.int32)
    vals[idx:idx+n] = W
    idx += n

    for r in range(m):
        k = int(active_peaks[r])
        w = float(weights[r])
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        nj = j_hi - j_lo + 1
        j_range = np.arange(j_lo, j_hi + 1, dtype=np.int32)
        col_range = (k - j_range).astype(np.int32)
        rows[idx:idx+nj] = j_range
        cols[idx:idx+nj] = col_range
        vals[idx:idx+nj] = w
        idx += nj

    rows[idx:idx+n] = np.arange(n, dtype=np.int32)
    cols[idx:idx+n] = n
    vals[idx:idx+n] = -1.0
    idx += n

    rows[idx:idx+n] = np.arange(n, dtype=np.int32)
    cols[idx:idx+n] = np.arange(n, dtype=np.int32)
    vals[idx:idx+n] = 1.0
    idx += n
    rows[idx] = n
    cols[idx] = n
    vals[idx] = 0.0
    idx += 1

    A = sparse.coo_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(n+1, n+1)).tocsr()
    b = np.zeros(n+1)
    b[n] = 1.0

    try:
        x = spsolve(A, b)
    except Exception:
        return None

    if not np.all(np.isfinite(x)):
        return None

    return x[:n], x[n]


def _kkt_optimize(a_init, deadline, min_time_remaining=60, tol_init=0.002, max_iter=8000):
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    tol = tol_init
    damping = 0.7
    stall = 0
    max_stall = 60

    for it in range(max_iter):
        if time.monotonic() + min_time_remaining >= deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                a = best_a.copy()
                tol = max(0.00005, tol * 0.6)
                damping = min(1.0, damping * 1.4)
                stall = 0
                if tol < 0.00005:
                    break

        active_thresh = c_max * (1.0 - tol)
        active_mask = c >= active_thresh
        n_active = int(active_mask.sum())

        if n_active == 0:
            active_idx = np.array([int(np.argmax(c))])
        elif n_active > 300:
            active_idx = np.argpartition(c, -300)[-300:]
        else:
            active_idx = np.where(active_mask)[0]

        n_act = len(active_idx)
        weights = np.ones(n_act) / n_act

        result = _solve_kkt(active_idx, weights, n)
        if result is None:
            w_full = np.zeros(2*n-1)
            w_full[active_idx] = weights
            g = _grad_maxconv(a, w_full)
            g_centered = g - g.mean()
            gn = np.linalg.norm(g_centered)
            if gn < 1e-18:
                break
            a_new = a - 0.01 * g_centered / gn
            a_new = np.maximum(a_new, 0)
            a = _normalize(a_new)
            continue

        a_kkt, lam = result

        if np.all(a_kkt >= -1e-8):
            a_kkt = np.maximum(a_kkt, 0.0)
            a_kkt = _normalize(a_kkt)
            a_new = a_kkt
            c_new = _full_convolve(a_new, a_new)
            sc_new = 2.0 * n * float(c_new.max())
            if sc_new < current_sc + 1e-12:
                a = a_new
                if sc_new < current_sc - 1e-14:
                    damping = min(1.0, damping * 1.05)
            else:
                damping = max(0.1, damping * 0.5)
                a_blend = (1.0 - damping) * a + damping * a_kkt
                a_blend = np.maximum(a_blend, 0.0)
                a_blend = _normalize(a_blend)
                c_blend = _full_convolve(a_blend, a_blend)
                if 2.0 * n * float(c_blend.max()) < current_sc:
                    a = a_blend
        else:
            a_new = (1.0 - damping) * a + damping * a_kkt
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new)
            c_new = _full_convolve(a_new, a_new)
            sc_new = 2.0 * n * float(c_new.max())
            if sc_new < current_sc:
                a = a_new
                damping = min(1.0, damping * 1.1)
            else:
                damping = max(0.05, damping * 0.5)
                a_new2 = (1.0 - damping) * a + damping * a_kkt
                a_new2 = np.maximum(a_new2, 0.0)
                a_new2 = _normalize(a_new2)
                c_new2 = _full_convolve(a_new2, a_new2)
                if 2.0 * n * float(c_new2.max()) < current_sc:
                    a = a_new2

        if it % 500 == 0 and it > 0:
            sc_check = _score(a)
            if sc_check < best_sc:
                best_sc = sc_check
                best_a = a.copy()

    return best_a, best_sc

def _topk_smoobj_grad(z, beta, K, n):
    z_max = z.max()
    a = np.exp(z - z_max)
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())

    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c

    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()

    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))

    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


class _TopkSmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[1]


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    schedule = [
        (1.0, 0.08), (0.6, 0.06), (0.4, 0.04), (0.2, 0.03),
        (0.1, 0.02), (0.05, 0.015), (0.02, 0.01), (0.01, 0.007),
        (0.005, 0.005), (0.002, 0.003), (0.001, 0.002), (0.0005, 0.001),
    ]

    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        beta = base_beta * beta_mult
        K = max(5, int((2 * n - 1) * K_frac))

        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(50, int(remaining / (est_time * 15))))

        prob = _TopkSmoProb(beta, K, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 150, 'maxls': 80}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgradient_bb(a_init, deadline, min_time_remaining=60.0):
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    n_2nm1 = 2 * n - 1
    active_frac = 0.01
    momentum = 0.7
    lr = 0.015
    lr_min = 1e-5
    lr_max = 0.08
    stall = 0
    max_stall = 100
    velocity = np.zeros(n)
    prev_a = None
    prev_g = None
    last_lbfgs_time = time.monotonic()

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        n_active = max(3, min(n_2nm1, int(n_2nm1 * active_frac)))
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_active]

        active_vals = c[active_idx]
        w = np.zeros(n_2nm1)
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w[active_idx] = (active_vals - active_vals.min()) / spread + 0.05
        else:
            w[active_idx] = 1.0 / n_active
        w = w / w.sum()

        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break

        if prev_a is not None and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g_centered - prev_g
            s_norm = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm > 1e-18 and abs(sy) > 1e-18:
                lr_bb = s_norm / abs(sy)
                lr = np.clip(lr_bb, lr_min, lr_max)

        velocity = momentum * velocity - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()

        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            for ls in range(20):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-15:
                    improved = True
                    break
                velocity = momentum * velocity * 0.5 - lr * 0.5 * g_centered
                a_try = a + velocity
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    velocity = np.zeros(n)
                    a_try = a
                    break
                a_try = a_try / s_try
                lr *= 0.7

        if improved:
            prev_a = a.copy()
            prev_g = g_centered.copy()
            a = a_try
            lr = min(lr_max, lr * 1.02)
            active_frac = max(0.002, active_frac * 0.97)
        else:
            velocity = np.zeros(n)
            lr = max(lr_min, lr * 0.6)
            active_frac = min(0.06, active_frac * 1.4)

        if stall > max_stall:
            active_frac = min(0.06, active_frac * 2.0)
            lr = max(lr_min, lr * 0.3)
            momentum = min(0.9, momentum + 0.05)
            stall = 0

        if ((stall + it) % 400 == 0) and time.monotonic() - last_lbfgs_time > 100:
            if time.monotonic() + min_time_remaining + 80 < deadline:
                lbfgs_dl = time.monotonic() + 70
                a_tmp = _run_lbfgs(a, lbfgs_dl, min_time_remaining=5)
                sc_tmp = _score(a_tmp)
                if sc_tmp < best_sc:
                    best_a = a_tmp.copy()
                    best_sc = sc_tmp
                    a = a_tmp.copy()
                    velocity = np.zeros(n)
                    prev_a = None
                    prev_g = None
                    last_lbfgs_time = time.monotonic()

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.4 + 0.6 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2) + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "optimal_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.5 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wide_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-1.5 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.003
    tol = 0.003
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active

        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-16:
            break

        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        improved = False
        for ls in range(40):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                tol = max(0.0001, tol * 0.6)
                lr *= 0.6
            if no_improve > 400:
                a = best_a.copy()
                tol = max(0.0001, tol * 0.4)
                lr = 0.002
                no_improve = 0
        else:
            if tol > 0.0001:
                tol = max(0.0001, tol * 0.96)
            lr = min(0.008, lr * 1.02)

    return best_a, best_sc
