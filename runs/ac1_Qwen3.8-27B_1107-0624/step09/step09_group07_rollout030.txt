Specifically, c[k] contains the term 2*a[m]*a[k-m] (or a[m]^2 if k=2m). So changing a[m] affects c[k] for all k = m + j where j is a valid index of a.

The coordinate descent step for a[m] would be:
- Fix all other a[j] for j != m
- c[k] as a function of a[m]: c[k] = c[k]_fixed + 2*a[m]*a[k-m] + (a[m]^2 if k=2m)
  where c[k]_fixed = c[k] - 2*a[m]*a[k-m] - (a[m]^2 if k=2m) is the part not involving a[m].
- We want to minimize max_k (c[k]_fixed + 2*a[m]*a[k-m] + a[m]^2*delta_{k,2m})
- This is a 1D minimization problem (quadratic in a[m] for each k, take the max)

This could be solved by bisection or by finding the "crossing points" of the parabolas.

"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses exact active-set subgradient with Nesterov acceleration, adaptive active set,
coordinate descent polishing, zero-entry exploration, and multi-restart strategy.
Key improvements: proper Lipschitz-based step sizes, dynamic active set that shrinks

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming a is already normalized to sum=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_weighted_conv(a, w):
    """Gradient of sum_k w[k]*c[k] where c = conv(a,a). w has length 2n-1."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _project_simplex(x):
    """Project x onto the simplex {a >= 0, sum(a) = 1}."""
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)

            last_print = time.monotonic()

def _lbfgs_polish(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with very fine top-K smoothing for final polishing."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.0005:
        gap = max_c * 0.0005
    base_beta = max(gap * 0.5, max_c * 0.002)
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # Fine-tuning schedule
    schedule = [
        (0.5, 0.01), (0.3, 0.008), (0.15, 0.005), (0.08, 0.004),
        (0.04, 0.003), (0.02, 0.002), (0.01, 0.001), (0.005, 0.0008),
    ]
    
    for beta_mult, K_frac in schedule:
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        beta = base_beta * beta_mult
        K = max(3, int((2 * n - 1) * K_frac))
        
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(4000, max(30, int(remaining / (est_time * 25))))

            def __init__(self, beta, K, n):
                self.beta = beta
                self.K = K
                self.n = n
            
            def __call__(self, z):
                return _topk_smoobj(z, self.beta, self.K, self.n)[0]
            
            def grad(self, z):
                return _topk_smoobj(z, self.beta, self.K, self.n)[1]
        
        prob = _Prob(beta, K, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',

def _topk_smoobj(z, beta, K, n):
    """Top-K log-sum-exp smoothed objective (normalized a)."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        c_top = c
    
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    
    ga = _grad_weighted_conv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


def _coordinate_polish(a_init, deadline, min_time_remaining=10.0):
    """Local coordinate descent: try small transfers between pairs of entries."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score_normalized(a)
    rng = np.random.default_rng(12345)
    
    # Precompute the convolution
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    no_improve = 0
    max_no_improve = 500
    t_start = time.monotonic()
    
    while time.monotonic() + min_time_remaining < deadline and no_improve < max_no_improve:
        # Find the max peak
        k_peak = int(np.argmax(c))
        
        # Compute contributions to this peak
        i_lo = max(0, k_peak - (n - 1))
        i_hi = min(n - 1, k_peak)
        contrib = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
            j = k_peak - i
            if i == j:
                contrib[i] += 2.0 * a[i]
            else:
                contrib[i] += a[j]
        
        # Try transferring mass from high-contributing to low-contributing entries
        n_try = max(3, int(n * 0.001))
        
        # Sort by contribution
        idx_sorted = np.argsort(contrib)
        
        found_improvement = False
        for trial in range(n_try):
            # Pick a source (high contrib) and target (low contrib)
            src = idx_sorted[-(trial % max(1, int(n*0.002)))]
            tgt = idx_sorted[trial % max(1, int(n*0.002))]
            
            if src == tgt or a[src] < 1e-12:
                continue
            
            # Try several transfer amounts
            for frac in [0.001, 0.005, 0.01, 0.003]:
                delta = a[src] * frac
                a_try = a.copy()
                a_try[src] -= delta
                a_try[tgt] += delta
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                
                if sc_try < best_sc - 1e-16:
                    a = a_try
                    best_a = a.copy()
                    best_sc = sc_try
                    c = c_try
                    no_improve = 0
                    found_improvement = True
                    break
            
            if found_improvement:
                break
        
        if not found_improvement:
            no_improve += 1
    
    return best_a, best_sc


def _perturb(a, sigma, rng=None):
    n = len(a)
    if rng is None:
        rng = np.random.default_rng()
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def rng_choice(arr, k, rng):
    k = min(k, len(arr))
    return rng.choice(arr, size=k, replace=False)


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edge_heavy":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * (np.cos(np.pi * t) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t) + 0.2 * np.exp(-3.0 * (t - 1.0) ** 2) + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        # Create a plateau in the left half
        mask = t < 0.4
        a[mask] = 2.0
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

            print(f"  NEW BEST: {best_score:.10f} [LBFGS2] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 6: Multi-restart with perturbations ===
    print(f"\n[P6] Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        sigmas = [0.01, 0.02, 0.03, 0.015, 0.025, 0.008, 0.04, 0.012]
        n_restarts = min(8, int((deadline - 100 - time.monotonic()) / 30))
        for i in range(n_restarts):
            if time.monotonic() > deadline - 60:
                break
            sigma = sigmas[i % len(sigmas)]
            a_pert = _perturb(best_seq, sigma, rng)
            # Quick active-set descent
            remaining_re = deadline - 50 - time.monotonic()
            if remaining_re < 15:
                break
            a_r, sc_r = _active_set_descent(a_pert, time.monotonic() + remaining_re * 0.6, min_time_remaining=10)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.10f} [restart_{i}] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 7: Try different n values ===
    print(f"\n[P7] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        n_options = [n_main + 1024, n_main - 1024, 20480, 14336]
        n_options = [x for x in n_options if x != n_main and x > 1000]
        for n_try in n_options[:2]:
            if time.monotonic() > deadline - 30:
                break
            a_res = _resample(best_seq, n_try)
            remaining_re = deadline - 20 - time.monotonic()
            if remaining_re < 15:
                break
            a_r, sc_r = _active_set_descent(a_res, time.monotonic() + remaining_re * 0.7, min_time_remaining=10)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.10f} [n={n_try}] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 8: Final coordinate + L-BFGS polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        # Coordinate descent
        remaining = deadline - 3 - time.monotonic()
        if remaining > 10:
            a_c, sc_c = _coordinate_polish(best_seq, time.monotonic() + remaining * 0.5, min_time_remaining=3)
            if sc_c < best_score:
                best_seq = a_c.copy()
                best_score = sc_c
                print(f"  NEW BEST: {best_score:.10f} [final_coord] t={time.monotonic()-t_start:.1f}s")
        
        # Tiny L-BFGS
        remaining = deadline - 2 - time.monotonic()
        if remaining > 8:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = max(mb * 0.0001, 1e-9)
            K_fin = max(3, int(n * 0.001))
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            
            class _FinProb:
                def __init__(self, beta, K, n):
                    self.beta = beta
                    self.K = K
                    self.n = n
                def __call__(self, z):
                    return _topk_smoobj(z, self.beta, self.K, self.n)[0]
                def grad(self, z):
                    return _topk_smoobj(z, self.beta, self.K, self.n)[1]
            
            prob = _FinProb(beta_fin, K_fin, n)
            max_iter = min(2000, max(10, int(remaining / 0.05)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-20,
                                           'gtol': 1e-17, 'maxcor': 30})
                a_fin = np.exp(result.x - result.x.max())
                a_fin = a_fin / a_fin.sum()
                sc_fin = _score(a_fin)
                if sc_fin < best_score:
                    best_seq = a_fin.copy()
                    best_score = sc_fin
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass
    
    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
