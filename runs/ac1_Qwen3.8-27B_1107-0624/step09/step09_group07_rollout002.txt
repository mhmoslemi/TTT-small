        c_max = float(c.max())
        
        if self.K < len(c):
            top_idx = np.argpartition(c, -self.K)[-self.K:]
            c_top = c[top_idx]
        else:
            top_idx = np.arange(len(c))
            c_top = c
        
        beta = self.beta
        shifted = (c_top - c_max) / beta
        shifted = np.clip(shifted, -700, 0)
        e = np.exp(shifted)
        s = e.sum()
        
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(2*self.n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            e_full = np.zeros(2*self.n - 1)
            e_full[top_idx] = e
            w = e_full / s
        
        # Gradient of the main term
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        ga_main = 2.0 * wc[self.n - 1:2*self.n - 1]
        
        # Gradient of variance penalty
        if self.variance_weight > 0:
            c_mean = float(c.mean())
            # d(var(c))/d(a) - this is complex, approximate
            # var(c) = mean(c^2) - mean(c)^2
            # d(c_k)/d(a_j) = a_{k-j} + a_{k-j'} (for full conv)
            # This is expensive, so we use a simpler approximation
            ga_var = self.variance_weight * 2.0 * (c - c_mean) @ np.eye(2*self.n-1)[:self.n, :self.n]
            # Actually this is too expensive, skip variance gradient
            ga_var = np.zeros(self.n)
        
        ga = ga_main + ga_var
        
        # Chain rule through softmax
        W = float(np.dot(a, ga))
        grad_z = a * (ga - W)
        return grad_z


def _grad_maxconv(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a"""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n-1:2*n-1]


def _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive smoothing."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    current_sc = _normalized_score(a)
    current_a = a.copy()
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted)-1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap * 0.5, max_c * 0.005)
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.max()
    
    # Adaptive schedule: start broad, narrow down
    # (K_frac, variance_weight)
    schedule = [
        (0.15, 0.0), (0.10, 0.0), (0.08, 0.0), (0.06, 0.0),
        (0.04, 0.0), (0.03, 0.0), (0.02, 0.0), (0.015, 0.0),
        (0.01, 0.0), (0.008, 0.0), (0.005, 0.0), (0.003, 0.0),
    ]
    
    for i, (K_frac, var_w) in enumerate(schedule):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        K = max(5, int((2*n-1) * K_frac))
        
        est_time_per_iter = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 25))))
        
        prob = _AdaptiveSmoProb(base_beta, K_frac, n, var_w)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 120, 'maxls': 50}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _normalized_score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.max()
        except Exception:
            pass
    
    return current_a


def _subgradient_adaptive(a_init, deadline, min_time_remaining=60.0):
    """Adaptive active-set subgradient with BB steps."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _normalized_score(a)
    
    c = _full_convolve(a, a)
    n_2nm1 = 2 * n - 1
    
    active_frac = 0.02
    momentum = 0.7
    lr = 0.015
    lr_min = 1e-5
    lr_max = 0.08
    stall = 0
    max_stall = 100
    velocity = np.zeros(n)
    
    prev_a = None
    prev_g = None
    
    iteration = 0
    last_lbfgs_time = time.monotonic()
    
    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive active set
        n_active = max(3, min(n_2nm1, int(n_2nm1 * active_frac)))
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_active]
        
        # Softmax weights over active peaks
        active_vals = c[active_idx]
        # Use temperature-based weights
        temp = (active_vals.max() - active_vals.min()) / max(1, n_active) + 1e-30
        w_vals = np.exp((active_vals - active_vals.max()) / temp)
        w_vals = w_vals / w_vals.sum()
        
        w = np.zeros(n_2nm1)
        w[active_idx] = w_vals
        
        # Gradient
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break
        
        # BB step size
        if prev_a is not None and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g_centered - prev_g
            s_norm = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm > 1e-18 and abs(sy) > 1e-18:
                lr_bb = s_norm / abs(sy)
                lr = np.clip(lr_bb, lr_min, lr_max)
        
        # Step with momentum
        velocity = momentum * velocity - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        
        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            for ls in range(20):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-15:
                    improved = True
                    break
                velocity = momentum * velocity * 0.5 - lr * 0.5 * g_centered
                a_try = a + velocity
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    velocity = np.zeros(n)
                    a_try = a
                    break
                a_try = a_try / s_try
                lr *= 0.7
        
        if improved:
            prev_a = a.copy()
            prev_g = g_centered.copy()
            a = a_try
            lr = min(lr_max, lr * 1.02)
            active_frac = max(0.003, active_frac * 0.99)
        else:
            velocity = np.zeros(n)
            lr = max(lr_min, lr * 0.6)
            active_frac = min(0.1, active_frac * 1.2)
        
        # Adapt on stall
        if stall > max_stall:
            active_frac = min(0.1, active_frac * 1.5)
            lr = max(lr_min, lr * 0.4)
            momentum = min(0.85, momentum + 0.03)
            stall = 0
        
        # Periodic L-BFGS
        if iteration % 400 == 0 and time.monotonic() - last_lbfgs_time > 100:
            if time.monotonic() + min_time_remaining + 80 < deadline:
                lbfgs_deadline = time.monotonic() + 70
                a_tmp = _run_adaptive_lbfgs(a, lbfgs_deadline, min_time_remaining=5)
                sc_tmp = _normalized_score(a_tmp)
                if sc_tmp < best_sc:
                    best_a = a_tmp.copy()
                    best_sc = sc_tmp
                    a = a_tmp.copy()
                    velocity = np.zeros(n)
                    prev_a = None
                    prev_g = None
                    last_lbfgs_time = time.monotonic()
    
    return best_a, best_sc


def _peak_equalization(a_init, deadline, min_time_remaining=30.0):
    """Try to equalize the peaks of the autocorrelation."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _normalized_score(a)
    
    patience = 0
    max_patience = 500
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        
        # Find peaks that are significantly above the mean
        c_mean = float(c.mean())
        threshold = c_mean + 0.5 * (c_max - c_mean)
        
        # Get all peaks above threshold
        above = np.where(c > threshold)[0]
        if len(above) == 0:
            above = np.array([int(np.argmax(c))])
        
        # Weight by excess
        excess = c[above] - c_mean
        w = np.zeros(2*n-1)
        w[above] = excess / excess.sum()
        
        # Gradient
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break
        
        # Line search
        alpha = min(0.02, 0.5 / (gnorm + 1e-15))
        improved = False
        for ls in range(30):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            patience += 5
            # Try a different threshold
            threshold = c_mean + 0.3 * (c_max - c_mean)
    
    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _perturb_structural(a, rng):
    """Structure-aware perturbation: modify chunks of the sequence."""
    n = len(a)
    n_chunks = rng.integers(3, 8)
    chunk_size = n // n_chunks
    a_new = a.copy()
    for _ in range(n_chunks):
        start = rng.integers(0, max(1, n - chunk_size))
        end = min(n, start + chunk_size)
        # Scale this chunk
        scale = rng.uniform(0.5, 1.5)
        a_new[start:end] *= scale
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "blackman":
        t = np.linspace(0, 1, n)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "optimal_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "super_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 4)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.3 * np.abs(2*t - 1) ** 8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n
