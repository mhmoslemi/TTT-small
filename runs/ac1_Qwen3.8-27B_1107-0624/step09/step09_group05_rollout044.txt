"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:
- More aggressive beta schedule with finer granularity at small beta
- Direct simplex optimization via L-BFGS-B (eliminating one variable)

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)

def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

def _grad_soft_structured(a, beta, n):
    """Efficient gradient of soft_max(conv(a,a)) using structure."""
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)

def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft_structured(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)

class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)

def _run_lbfgs_r développeur(a_init, deadline, min_time_remaining=30.0, verbose=True, extra_betas=None):
    """L-BFGS with aggressive adaptive beta schedule."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(200, len(c_sorted) - 1)]
    if gap < c0_max * 0.002:
        gap = c0_max * 0.002

    t_start = time.monotonic()
    
    # More fine-grained beta levels, especially at small values
    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
        0.00003125, 0.000015625, 0.0000078, 0.0000039,
        0.000002, 0.000001, 0.0000005, 0.0000002, 0.0000001
    ])
    betas = gap * multipliers
    
    if extra_betas is not None:
        betas = np.concatenate([betas, extra_betas])
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        size_factor = max(1.0, n / 4096.0)
        est_iter_time = 0.002 * size_factor
        max_iter = min(10000, max(200, int(remaining / (est_iter_time * 15))))

        if beta < gap * 0.001:
            max_iter = min(max_iter, 3000)

        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 150,
                    'maxls': 80,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    elapsed = time.monotonic() - t_start
                    print(f"    beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}, t={elapsed:.0f}s")
            else:
                if verbose and (i < 5 or beta < gap * 0.01):
                    print(f"    beta={beta:.2e}: no improvement (best={best_sc:.10f})")
        except Exception as e:
            if verbose:
                print(f"    beta={beta:.2e}: error {e}")

    return best_a, best_sc

def _exact_subgradient_polish(a_init, deadline, min_time_remaining=10.0):
    """Polish using exact subgradient with adaptive step sizes."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    no_improve = 0
    max_no_improve = 500
    eta = 0.1
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            eta *= 0.95
            if no_improve > max_no_improve:
                break

        # Subgradient: average over all "near-max" convolution coefficients
        # Use a tight threshold
        tol = 1e-6 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Compute subgradient
        # grad[j] = (1/n_active) * sum_{k in active} 2*a[k-j] (for valid k-j)
        # = (2/n_active) * conv(w_active, a_rev)[n-1+j]
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        # Normalize gradient (remove mean to stay on simplex)
        grad_c = grad - np.mean(grad)
        g_norm = float(np.sqrt(np.dot(grad_c, grad_c)))
        if g_norm < 1e-15:
            break

        # Step
        step = eta / g_norm
        a_new = a - step * grad_c
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new

    return best_a, best_sc

def _small_perturbation(a_init, magnitude=0.01, seed=None):
    """Add small random perturbation and re-optimize."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    noise = rng.normal(0, magnitude, n)
    a = a_init + noise
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a_init.copy()
    a = a / s
    return a

def _symmetrize(a):
    """Symmetrize: a -> (a + rev(a)) / 2"""
    a_sym = (a + a[::-1]) / 2.0
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    else:
        return np.ones(n) / n
