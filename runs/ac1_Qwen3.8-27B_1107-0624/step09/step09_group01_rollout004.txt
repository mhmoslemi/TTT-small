# --- Utility functions ---

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fftc(a, b):
    """Full convolution using FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    """Compute the objective value."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fftc(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - z.max()
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _run_lbfgs_adaptive(a_init, deadline, min_remaining=30.0):
    """Run L-BFGS with adaptive beta annealing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    current_a = a_init.copy()
    current_sc = _score(current_a)
    
    # Start with a reasonable beta
    c_init = _fftc(current_a, current_a)
    c_max_init = float(c_init.max())
    # Estimate the gap between max and the rest
    c_sorted = np.sort(c_init)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted)-1)]
    beta = max(gap * 0.5, c_max_init * 0.001)
    
    min_beta = c_max_init * 1e-12
    
    beta_history = []
    while beta > min_beta:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        
        # Estimate iterations: each iteration is roughly one FFT convolution
        # For n elements, FFT convolution takes ~O(n log n)
        scale = max(1.0, n / 8192.0)
        est_time_per_iter = 0.003 * scale
        max_iter = min(3000, max(20, int(remaining / (est_time_per_iter * 40))))
        
        def obj_fn(z):
            return _smooth_obj(z, beta, n)
        
        def grad_fn(z):
            return _smooth_grad(z, beta, n)
        
        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 300, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS: beta={beta:.6e}, score={current_sc:.10f}, iter={result.nit}")
            else:
                print(f"    L-BFGS: beta={beta:.6e}, no improvement (got {sc_new:.10f})")
        except Exception as e:
            print(f"    L-BFGS error at beta={beta:.6e}: {e}")
        
        beta_history.append(beta)
        
        # Adaptive beta reduction
        c_new = _fftc(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new)-1)]
        
        if new_gap < 1e-8 * c_max_new:
            beta *= 0.1
        elif new_gap < 1e-6 * c_max_new:
            beta *= 0.15
        elif new_gap < 1e-4 * c_max_new:
            beta *= 0.2
        elif new_gap < 1e-3 * c_max_new:
            beta *= 0.3
        else:
            beta *= 0.4
        
        beta = max(beta * 0.8, min_beta)
        # Don't reduce beta too fast
        if len(beta_history) > 1 and beta < beta_history[-1] * 0.05:
            beta = beta_history[-1] * 0.05
    
    return current_a, current_sc
