"""
Optimized step function search using adaptive L-BFGS with log-sum-exp smoothing,
gradient-informed iterated local search, and multi-resolution exploration.
Key improvements: tighter beta annealing, gradient-directed perturbations,
population-based candidate management, and adaptive time allocation.
"""
import numpy as np
import time
from scipy.optimize import minimize
import sys


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _conv_peak_info(a):
    c = _full_convolve(a, a)
    c_max = float(c.max())
    sorted_c = np.sort(c)[::-1]
    n = len(c)
    gap3 = sorted_c[0] - sorted_c[min(3, n - 1)]
    gap10 = sorted_c[0] - sorted_c[min(10, n - 1)]
    gap50 = sorted_c[0] - sorted_c[min(50, n - 1)]
    return c, c_max, gap3, gap10, gap50


def _active_gradient(a, c, tol_frac):
    """Compute active-set gradient for L-infinity peak minimization."""
    n = len(a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros(n)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True

    eps = 1e-10 * c_max
    w = np.zeros_like(c)
    w[active] = 1.0 / (c_max - c[active] + eps)
    w = w / w.sum()

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad = 2.0 * h
    grad = grad - np.mean(grad)
    return grad


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_sorted = np.sort(c)[::-1]
    c_max = float(c_sorted[0])
    # Use log-sum-exp over top peaks for better conditioning
    top_k = min(100, len(c))
    top = c_sorted[:top_k]
    shifted = (top - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_sorted = np.sort(c)[::-1]
    c_max = float(c_sorted[0])
    top_k = min(100, len(c))
    # Find indices of top elements
    top_idx = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_idx]
    shifted = (top_vals - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = np.zeros_like(c)
        w[top_idx] = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_run(a_init, deadline, min_remaining=20.0, verbose=False):
    """L-BFGS with adaptive beta annealing."""
    n = len(a_init)
    c_cur, c_max, gap3, gap10, gap50 = _conv_peak_info(a_init)
    if c_max < 1e-15:
        return a_init, _score(a_init)

    base_beta = max(gap10, c_max * 0.003, 1e-8)
    betas = base_beta * np.array([1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1,
                                  0.06, 0.04, 0.025, 0.015, 0.01,
                                  0.006, 0.004, 0.002, 0.001, 0.0005,
                                  0.0003, 0.00015, 0.00008, 0.00004])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(8000, max(60, int(remaining / (est_time * 12))))

        def f_obj(z_val):
            return _smoobj(z_val, beta, n)

        def g_obj(z_val):
            return _smograd(z_val, beta, n)

        try:
            result = minimize(
                f_obj, z, jac=g_obj, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 150, 'maxls': 60}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"    L-BFGS beta={beta:.2e}: {sc_new:.10f}")
        except Exception:
            pass

    return current_a, _score(current_a)


def _nesterov_polish(a_init, deadline, tol_start=0.005, tol_end=0.00005,
                     lr_init=0.03, momentum=0.85):
    """Active-set gradient with momentum and adaptive step."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    v = np.zeros(n)
    tol_frac = tol_start
    lr = lr_init
    no_improve = 0
    max_stall = 200

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        grad = _active_gradient(a, c, tol_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        v = momentum * v - lr * grad
        alpha = 1.0
        improved = False
        for _ in range(40):
            a_try = _normalize(a + alpha * v)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            v = np.zeros(n)
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_end, tol_frac * 0.65)
                lr *= 0.7
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(tol_end, tol_frac * 0.3)
                lr = lr_init * 0.4
                no_improve = 0
        else:
            if tol_frac > tol_end:
                tol_frac = max(tol_end, tol_frac * 0.97)
            lr = min(lr_init, lr * 1.01)

    return best_a, _score(best_a)


def _gradient_ilss(a_init, deadline, rng, min_remaining=60.0):
    """Gradient-informed Iterated Local Search."""
    n = len(a_init)
    current_a = _normalize(a_init.copy())
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc

    # Compute gradient for perturbation direction
    c = _full_convolve(current_a, current_a)
    grad = _active_gradient(current_a, c, 0.01)
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-18:
        grad = np.zeros(n)
    else:
        grad = grad / grad_norm

    attempt = 0
    while time.monotonic() + min_remaining < deadline:
        # Choose perturbation strategy
        r = rng.random()
        if r < 0.3:
            # Gradient-aligned perturbation
            sigma = 0.01 * (1.0 + 0.5 * (attempt % 5))
            z = np.log(np.maximum(current_a, 1e-30))
            z = z + sigma * (0.7 * grad + 0.3 * rng.standard_normal(n))
            z = z - z.mean()
            a_perturbed = _softmax(z)
        elif r < 0.55:
            # Global random perturbation
            sigma = 0.005 + 0.01 * rng.random()
            z = np.log(np.maximum(current_a, 1e-30))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_perturbed = _softmax(z)
        elif r < 0.75:
            # Block perturbation
            block_size = max(10, n // 10)
            start = int(rng.integers(0, max(1, n - block_size)))
            a_perturbed = current_a.copy()
            z_block = np.log(np.maximum(a_perturbed[start:start + block_size], 1e-30))
            z_block = z_block + 0.03 * rng.standard_normal(block_size)
            a_perturbed[start:start + block_size] = np.exp(z_block - np.max(z_block))
            a_perturbed = _normalize(a_perturbed)
        else:
            # Sparse large perturbation
            n_pert = max(5, n // 20)
            idx = rng.choice(n, size=n_pert, replace=False)
            a_perturbed = current_a.copy()
            factors = np.exp(0.1 * rng.standard_normal(n_pert))
            a_perturbed[idx] *= factors
            a_perturbed = _normalize(a_perturbed)

        # Quick optimization of perturbed candidate
        opt_deadline = min(deadline - min_remaining, time.monotonic() + 6)
        a_opt, sc_opt = _lbfgs_run(a_perturbed, opt_deadline, min_remaining=2, verbose=False)

        # Accept if better
        if sc_opt < best_sc - 1e-12:
            best_sc = sc_opt
            best_a = a_opt.copy()
            current_a = a_opt.copy()
            # Update gradient
            c = _full_convolve(current_a, current_a)
            grad = _active_gradient(current_a, c, 0.01)
            gn = np.linalg.norm(grad)
            if gn > 1e-18:
                grad = grad / gn
            print(f"    ILS NEW BEST: {best_sc:.10f} (attempt {attempt}) t={time.monotonic()-t_start:.0f}s")
        elif sc_opt < current_sc - 1e-12:
            current_a = a_opt.copy()

        attempt += 1

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-5.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(42)
        a = rng.random(n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    return np.ones(n) / n
