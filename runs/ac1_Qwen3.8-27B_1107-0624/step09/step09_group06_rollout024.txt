import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _logsumexp_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _logsumexp_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _topk_obj(z, K, n):

    c = _full_convolve(a, a)
    idx = np.argpartition(c, -K)[-K:]

def _topk_grad(z, K, n):

    h = wc[n - 1:2 * n - 1]

        grad = np.zeros(n)
        a_rev = a[::-1]
        w = np.zeros(2*n-1)
        w[active] = 1.0 / K
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        W = float(np.dot(w, c))
        grad = 4.0 * a * (h - W)
        
        # Project to tangent space
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        
        if gnorm < 1e-18:
            break
        
        # Nesterov step
        step_size = lr / (gnorm + 1e-15)
        v = momentum * v - step_size * grad_t
        a_cand = a + v
        a_proj = _project_simplex(a_cand)
        
        c_new = _full_convolve(a_proj, a_proj)
        sc_new = 2.0 * n * float(c_new.max())
        
        if sc_new <= current_sc:
            a = a_proj
        else:
            v = 0.5 * v
            lr *= 0.95
            a = a_proj
        
        if t % 500 == 0:
            print(f"    Subgrad t={t}: best={best_sc:.10f}, cur={sc_new:.10f}, K={K}")
    
    return best_a, best_sc

        return a / a.sum()
