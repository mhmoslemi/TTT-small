```python
import numpy as np
import time

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, 2*N)  # Use 2N to avoid circular convolution
    fb = np.fft.rfft(b, 2*N)
    r = np.fft.irfft(fa * fb, 2*N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))
```
