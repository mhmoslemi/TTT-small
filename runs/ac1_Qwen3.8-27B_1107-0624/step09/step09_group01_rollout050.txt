import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * t / 2) ** 2 + 0.01
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n

def _mirror_descent(a_init, deadline, max_iters=200000, verbose=True):
    """Main optimization loop using multiplicative weights (mirror descent)."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    if verbose:
        print(f"  MD start: n={n}, score={best_sc:.10f}")
    
    lr = 1.0 / np.sqrt(n)  # initial learning rate
    temp_frac = 0.05  # temperature as fraction of c_max
    patience = 0
    max_patience = 200
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        temp = c_max * temp_frac
        
        # Dual weights
        z = c / temp - c.max() / temp
        e = np.exp(np.clip(z, -700, 0))
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        
        # Pressure
        wc = _full_convolve(w, a[::-1])
        h = wc[n-1:2*n-1]
        
        # Multiplicative weights update
        h_centered = h - h.mean()
        a = a * np.exp(-lr * h_centered)
        a = a / a.sum()
        
        # Periodic check
        if (it + 1) % 50 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_a = a.copy()
                patience = 0
                lr = min(lr * 1.02, 2.0 / np.sqrt(n))
                temp_frac = max(temp_frac * 0.98, 0.005)
            else:
                patience += 1
                lr *= 0.95
                if patience > max_patience:
                    break

    # Final
    sc = _score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    
    if verbose:
        print(f"  MD end: score={best_sc:.10f}, iters={it+1}")
    
    return best_a, best_sc

def _lbfgs_polish(a_init, deadline, min_remaining=30.0):
    """Polish using L-BFGS on smoothed objective."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    sc = _score(a)
    
    # Adaptive temperature for smoothing
    c = _full_convolve(a, a)
    c_max = float(c.max())
    beta = c_max * 0.01
    
    current_a = a.copy()
    current_sc = sc
    
    for round in range(10):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        
        def obj(z):
            aa = _softmax(z)
            cc = _full_convolve(aa, aa)
            cm = float(cc.max())
            shifted = (cc - cm) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                return 2.0 * n * cm
            return 2.0 * n * (cm + beta * np.log(s))
        
        def grad(z):
            aa = _softmax(z)
            cc = _full_convolve(aa, aa)
            cm = float(cc.max())
            shifted = (cc - cm) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                w = np.zeros_like(cc)
                w[int(np.argmax(cc))] = 1.0
            else:
                w = e / s
            wc = _full_convolve(w, aa[::-1])
            h = wc[n-1:2*n-1]
            Wc = float(np.dot(w, cc))
            return 4.0 * n * aa * (h - Wc)
        
        try:
            max_iter = min(3000, max(20, int(remaining / 0.05)))
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxcor': 50, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS round {round}: score={current_sc:.10f}")
        except Exception as ex:
            print(f"    L-BFGS round {round} failed: {ex}")
            break
        
        # Update beta
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        beta = c_max_new * 0.005
        beta = max(beta, c_max_new * 1e-6)
    
    return current_a

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)
