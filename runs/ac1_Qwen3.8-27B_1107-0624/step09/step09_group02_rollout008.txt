Multi-resolution L-BFGS optimizer with log-sum-exp smoothing for
minimizing peak linear self-convolution of non-negative sequences.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))

def _run_lbfgs_with_restarts(a_init, deadline, min_time_remaining=30.0, n_restarts=3):
    """L-BFGS with random restarts to escape local minima."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(best_a)
    
    rng = np.random.default_rng(12345)
    
    time_per_restart = (deadline - min_time_remaining - time.monotonic()) / max(1, n_restarts)
    
    for r in range(n_restarts):
        t_rem = time.monotonic() + time_per_restart > deadline - min_time_remaining
        if not t_rem:
            break
        
        if r == 0:
            a_try = best_a.copy()
        else:
            # Perturb in log space
            z = np.log(np.maximum(best_a, 1e-30))
            sigma = 0.05 * (1.0 + 0.5 * r)
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_try = _softmax(z)
        
        phase_deadline = min(deadline - min_time_remaining, time.monotonic() + time_per_restart)
        a_out, sc_out = _run_lbfgs_refined(a_try, phase_deadline, min_time_remaining=min_time_remaining)
        if sc_out < best_sc:
            best_sc = sc_out
            best_a = a_out.copy()
    
    return best_a, best_sc


def _direct_grad_descent(a_init, deadline, min_time_remaining=30.0):
    """Direct projected gradient descent with adaptive step size."""
    n = len(a_init)
