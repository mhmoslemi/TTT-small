    a = a / a.sum()

    best_score = 2.0 * n * float(_full_convolve(a, a).max())
    
    eps = 1e-10  # tolerance for "active" peaks
    max_no_improve = 500
    no_improve = 0
    step_fraction = 0.9
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_score = 2.0 * n * c_max
        
        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        
        # Active set: peaks within relative tolerance of max
        tolerance = c_max * 1e-6
        active = np.where(c >= c_max - tolerance)[0]
        
        # Compute average subgradient
        g = np.zeros(n)
        for k in active:
            # g_k[j] = a[k-j] if 0 <= k-j < n
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo <= hi:

        g /= len(active)

        g -= np.mean(g)

        neg_mask = g < -1e-15
        if np.any(neg_mask):
            alpha_max = np.min(a[neg_mask] / (-g[neg_mask]))
        else:
            alpha_max = np.inf

            alpha = 1e-10
        else:
            alpha = step_fraction * alpha_max
        
        a_new = a - alpha * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Verify improvement (every few iterations)
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_gradient(a, active_indices):
    """Compute the average subgradient for the given active set."""
    n = len(a)
    w = np.zeros(2 * n - 1)
    w[active_indices] = 1.0 / len(active_indices)
    a_rev = a[::-1]
    h = _full_convolve(w, a_rev)
    return 2.0 * h[n - 1:2 * n - 1]

                    tol=1e-8, max_iters=None):

    n = len(a_init)

    a = a / a.sum()
    best_a = a.copy()

    best_score = 2.0 * n * float(c.max())

    K_max = min(200, n)

    for it in range(max_iters):
        if time.monotonic() + min_time_remaining >= deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_score = 2.0 * n * c_max
        
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:

            a = best_a.copy()
            # Small random perturbation

            pert = rng.standard_normal(n) * 0.01
            a = a + pert
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
            K = max(3, K // 2)

        neg_mask = g < -1e-15
        if np.any(neg_mask):

        alpha = min(1.0, 0.5 * alpha_ub)

            a_try = np.maximum(a_try, 0.0)

            if s < 1e-15:

        # Adapt K
        if improved and no_improve == 0:
            # Making progress, grow K slowly
            if it % 100 == 0:
                K = min(K_max, K + 1)
    
    return best_a, best_score

Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _subgradient(a, active_indices):
    """Compute average subgradient of max over active set.
    grad[j] = 2 * sum_{k in S} a[k-j] / |S| (with a[m]=0 if m out of range)"""
    n = len(a)
    w = np.zeros(2 * n - 1)
    w[active_indices] = 1.0 / len(active_indices)
    a_rev = a[::-1]
    h = _full_convolve(w, a_rev)
    return 2.0 * h[n - 1:2 * n - 1]

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs_v2(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0):

    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003) * beta_scale

    # Extended beta schedule with more fine-grained values
    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015, 0.008, 0.004, 0.002, 0.001,
        0.0005, 0.00025, 0.0001, 0.00005, 0.00002,
        0.00001, 0.000005, 0.000002, 0.000001
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining >= deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        if beta < base_beta * 0.01:
            est_time = 0.005 * max(1.0, n / 2048.0)
            max_iter = min(8000, max(100, int(remaining / (est_time * 10))))
        elif beta < base_beta * 0.1:
            est_time = 0.004 * max(1.0, n / 2048.0)
            max_iter = min(6000, max(50, int(remaining / (est_time * 12))))
        else:
            est_time = 0.003 * max(1.0, n / 2048.0)
            max_iter = min(4000, max(30, int(remaining / (est_time * 15))))

        def _obj(z_cur):
            return _smoobj(z_cur, beta, n)

        def _grad(z_cur):
            return _smograd(z_cur, beta, n)

                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        a = np.arange(1, n + 1, dtype=np.float64)
        return a / a.sum()
    elif kind == "ramp_soft":
        a = np.arange(1, n + 1, dtype=np.float64) ** 0.7
        return a / a.sum()
    elif kind == "ramp_quad":
        a = np.arange(1, n + 1, dtype=np.float64) ** 0.5
        return a / a.sum()
    elif kind == "asym_bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.35) ** 2) + 0.3 * np.exp(-20.0 * (t - 0.75) ** 2) + 0.01
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.arange(n)
        a = 1.0 + 0.4 * np.sin(2.0 * np.pi * 3.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
