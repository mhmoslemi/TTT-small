"""
Multi-phase optimizer for minimum peak linear autocorrelation of non-negative sequences.
Uses adaptive log-sum-exp L-BFGS, active-set QP with CG inner solver,
periodic diversification, and multi-n exploration.
Key improvements over previous: active-set QP with explicit free-set Newton step,
adaptive beta with gap-based scheduling, and periodic ensemble mixing.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _hvp(a, p, n):
    """Compute H_p * a where H_p = 2 * sum_k p_k H_k.
    p has length 2n-1, a has length n.
    """
    a_rev = a[::-1]
    conv_len = n + len(p) - 1
    N = _next_pow2(conv_len)
    Fa = np.fft.rfft(a_rev, N)
    Fp = np.fft.rfft(p, N)
    conv = np.fft.irfft(Fa * Fp, N)[:conv_len]
    result = 2.0 * conv[n - 1:2 * n - 1].copy()
    # Diagonal correction
    for k in range(0, 2 * n - 1, 2):
        m = k // 2
        if m < n and k < len(p):
            result[m] -= a[m] * p[k]
    return result


def _qpv(a, p, n):
    """Compute Q_p * a where Q_p = (1/|K|) sum_{k in K} H_k (the QP Hessian).
    This is _hvp(a, p, n) / 2 since _hvp uses 2*sum p_k H_k.
    """
    return _hvp(a, p, n) / 2.0


def _cg_solve(A_func, b_vec, n, max_iter=500, tol=1e-10):
    """Conjugate gradient for solving A_func(x) = b_vec.
    A_func should be symmetric positive semi-definite.
    Returns the solution x.
    """
    x = np.zeros(n)
    r = b_vec - A_func(x)
    p_vec = r.copy()
    rs_old = float(r @ r)
    for i in range(max_iter):
        if np.sqrt(rs_old) < tol * (np.linalg.norm(b_vec) + 1e-30):
            break
        Ap = A_func(p_vec)
        pAp = float(p_vec @ Ap)
        if abs(pAp) < 1e-30:
            # A is singular in this direction; use steepest descent step
            alpha = rs_old / max(float(r @ r), 1e-30)
        else:
            alpha = rs_old / pAp
        x = x + alpha * p_vec
        r_new = r - alpha * Ap
        rs_new = float(r_new @ r_new)
        if rs_new > rs_old * 1.1:
            # Reset
            r = b_vec - A_func(x)
            p_vec = r.copy()
            rs_old = float(r @ r)
            break
        beta = rs_new / max(rs_old, 1e-30)
        p_vec = r_new + beta * p_vec
        r = r_new
        rs_old = rs_new
    return x

            a = a_new

                    continue
            # Check if any free coordinate should leave
            leaving = free_idx[a_new[free_idx] < tol]
            if len(leaving) > 0:
                free[leaving] = False
                a[leaving] = 0
                continue
            # Converged
            break
        else:
            # Some free coordinates went negative; move them to active set
            neg_idx = free_idx[a_new[free_idx] < 0]
            free[neg_idx] = False
            a[neg_idx] = 0
            continue
    
    return a


def _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with adaptive beta based on autocorrelation gap, and restarts."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    current_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    best_a = a.copy()
    best_sc = current_sc
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap * 0.05, max_c * 0.0001)
    
    # Adaptive ratio schedule with restarts
    ratios = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
              0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001,
              0.000005, 0.000002, 0.000001, 0.0000005, 0.0000002, 0.0000001,
              0.00000005, 0.00000002, 0.00000001]
    
    no_improve_count = 0
    max_no_improve = 8
    
    for i, ratio in enumerate(ratios):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        beta = base_beta * ratio
        
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 5))))
        
        prev_sc = best_sc
        try:
            def obj_func(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    return 2.0 * n * c_max_local
                return 2.0 * n * (c_max_local + beta * np.log(s_e))

            def grad_func(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    w = np.zeros_like(c_local)
                    w[int(np.argmax(c_local))] = 1.0
                else:
                    w = e / s_e
                a_rev = a_local[::-1]
                wc = _full_convolve(w, a_rev)
                h = wc[n - 1:2 * n - 1]
                W_val = float(np.dot(w, c_local))
                return 4.0 * n * a_local * (h - W_val)

            result = minimize(
                obj_func, z,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                no_improve_count = 0
            else:
                no_improve_count += 1
                # Try to restart with a perturbation if stuck
                if no_improve_count > max_no_improve and i < len(ratios) - 3:
                    z = z + 0.02 * np.random.standard_normal(n)
                    z = z - z.mean()
                    no_improve_count = 0
        except Exception:
            pass
    
    return best_a


def _run_proj_grad_adaptive(a_init, deadline, min_time_remaining=60.0):
    """Projected gradient with adaptive top-K and momentum."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    momentum = np.zeros(n)
    beta_m = 0.9
    patience = 0
    max_patience = 100
    k_schedule = [1, 2, 3, 5, 8, 10, 15, 20, 30]
    k_idx = 0
    
    for step in range(10000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        a_curr = _softmax(z)
        c = _full_convolve(a_curr, a_curr)
        c_max = float(c.max())
        
        k_act = k_schedule[min(k_idx, len(k_schedule) - 1)]
        k_act = min(k_act, 2 * n - 1)
        top_k = np.argsort(c)[::-1][:k_act]
        p = np.zeros(2 * n - 1)
        p[top_k] = 1.0 / k_act
        
        grad = _hvp(a_curr, p, n)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        momentum = beta_m * momentum + (1.0 - beta_m) * grad
        grad_signal = np.linalg.norm(momentum) / max(grad_norm, 1e-15)
        lr = 0.02 / max(grad_signal, 1.0)
        lr = min(lr, 0.1 / max(grad_signal, 1.0))
        
        z_new = z - lr * momentum
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new = z_new - z_new.mean()
        z = z_new
        
        if step % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                # Increase k_idx occasionally for more smoothing
                if patience == 0 and step > 100:
                    k_idx = min(k_idx + 1, len(k_schedule) - 1)
            else:
                patience += 1
                if patience > max_patience:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros(n)
                    patience = 0
                    beta_m = max(0.5, beta_m * 0.9)
                    k_idx = min(k_idx - 1, len(k_schedule) - 1)
                    k_idx = max(k_idx, 0)
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _flatten_direct(a_init, deadline, min_time_remaining=60.0):
    """Direct flattening with backtracking line search."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(500):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Use top peaks with soft threshold
        threshold = c_max * 0.99
        peaks = c >= threshold
        if peaks.sum() == 0:
            peaks[int(np.argmax(c))] = True
        p = np.zeros(2 * n - 1)
        p[peaks] = 1.0 / peaks.sum()
        
        grad = _hvp(a, p, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        step = 0.1 / (gnorm + 1e-10)
        improved = False
        for ls in range(40):
            a_new = _proj_simplex(a - step * grad)
            c_new = _full_convolve(a_new, a_new)
            if float(c_new.max()) < c_max * (1 - 1e-12):
                a = a_new
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                improved = True
                break
            step *= 0.5
            if step < 1e-15:
                break
        if not improved and it > 20:
            # Try a different peak threshold
            threshold = c_max * (0.99 + 0.01 * (it % 10) / 10)
            peaks = c >= threshold
            if peaks.sum() == 0:
                peaks[int(np.argmax(c))] = True
            p = np.zeros(2 * n - 1)
            p[peaks] = 1.0 / peaks.sum()
            grad = _hvp(a, p, n)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            step = 0.1 / (gnorm + 1e-10)
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * t
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.3) ** 2) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _spectral_flatten(a_init, deadline, min_time_remaining=60.0):
    """Spectral flattening: iteratively flatten the power spectrum."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    N = _next_pow2(2 * n - 1)
    
    for it in range(200):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        # Compute power spectrum
        A = np.fft.rfft(a, N)
        power = np.abs(A) ** 2
        
        # Target: flat spectrum (or slightly peaked at DC)
        # Use sqrt of average power as target
        avg_power = power.mean()
        target = np.sqrt(avg_power)
        
        # Adjust magnitude: compress the spectrum
        alpha = 0.3  # flattening strength
        new_mag = np.abs(A)
        new_mag = (1 - alpha) * new_mag + alpha * target
        # Keep phase
        new_A = new_mag * np.exp(1j * np.angle(A))
        
        # Inverse FFT
        a_new = np.fft.irfft(new_A, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce flattening strength
            alpha *= 0.7
            if alpha < 0.01:
                break
    
    return best_a, best_sc


def _ensemble_mix(a1, a2, rng, weight=0.5):
    """Mix two sequences in log-space."""
    n = min(len(a1), len(a2))
    a1 = a1[:n]
    a2 = a2[:n]
    z1 = np.log(np.maximum(a1, 1e-30))
    z2 = np.log(np.maximum(a2, 1e-30))
    z = weight * z1 + (1 - weight) * z2
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: spectral init, adaptive L-BFGS, active-set QP,
    projected gradient, direct flattening, ensemble mixing, multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf
    ensemble = []  # Keep multiple good candidates

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")
            return True
        return False

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
                ensemble.append(parent.copy())
    except Exception:
        pass

    if parent is None:
        n_parent = 32768
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")
        ensemble.append(parent.copy())

    # === Phase 1: Spectral flattening (120s) ===
    print(f"\n[Phase 1] Spectral Flattening (t={time.monotonic() - t_start:.1f}s)")
    phase1_deadline = t_start + 120
    a_spec, sc_spec = _spectral_flatten(parent, phase1_deadline, min_time_remaining=30.0)
    if sc_spec < best_score + 0.01:
        update_best(a_spec, sc_spec, "spectral")
        ensemble.append(a_spec.copy())

    # === Phase 2: Adaptive L-BFGS (500s) ===
    print(f"\n[Phase 2] Adaptive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    phase2_deadline = min(deadline - 700, phase1_deadline + 500)
    
    # Start from best
    a_lb = _run_lbfgs_adaptive(best_seq, phase2_deadline, min_time_remaining=30.0)
    improved = update_best(a_lb, _score(a_lb), "lbfgs_main")
    if improved:
        ensemble.append(a_lb.copy())
    
    # Try from spectral
    if a_spec is not None and time.monotonic() < phase2_deadline - 60:
        a_lb2 = _run_lbfgs_adaptive(a_spec, phase2_deadline, min_time_remaining=20.0)
        if update_best(a_lb2, _score(a_lb2), "lbfgs_spectral"):
            ensemble.append(a_lb2.copy())
    
    # Try perturbation of parent
    if time.monotonic() < phase2_deadline - 30:
        a_pert = parent + 0.01 * rng.standard_normal(n_parent)
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        a_lb3 = _run_lbfgs_adaptive(a_pert, phase2_deadline, min_time_remaining=15.0)
        if update_best(a_lb3, _score(a_lb3), "lbfgs_perturb"):

    # === Phase 3: Active-set QP (300s) ===
    print(f"\n[Phase 3] Active-set QP (t={time.monotonic() - t_start:.1f}s)")
    phase3_deadline = min(deadline - 500, phase2_deadline + 300)
    if best_seq is not None and time.monotonic() < phase3_deadline:
        n = len(best_seq)
        c = _full_convolve(best_seq, best_seq)
        c_max = float(c.max())

        if update_best(a_fl, sc_fl, "flatten"):
            ensemble.append(a_fl.copy())

    # === Phase 6: Second round of combined optimization (200s) ===
    print(f"\n[Phase 6] Combined Round 2 (t={time.monotonic() - t_start:.1f}s)")
    phase6_deadline = min(deadline - 150, phase5_deadline + 200)
    if best_seq is not None and time.monotonic() < phase6_deadline:
        a_lb4 = _run_lbfgs_adaptive(best_seq, phase6_deadline, min_time_remaining=30.0)
        if update_best(a_lb4, _score(a_lb4), "lbfgs_r2"):
            ensemble.append(a_lb4.copy())
        
        if time.monotonic() < phase6_deadline - 30:
            n = len(best_seq)
            c = _full_convolve(best_seq, best_seq)

            p_act[top_K] = 1.0 / 15
            a_qp3 = _active_set_qp(best_seq, p_act, n, max_rounds=80, tol=1e-11)
            if update_best(a_qp3, _score(a_qp3), "active_qp_r2"):
                ensemble.append(a_qp3.copy())

    # === Phase 7: Ensemble mixing and re-optimization (100s) ===
    print(f"\n[Phase 7] Ensemble Mixing (t={time.monotonic() - t_start:.1f}s)")
    phase7_deadline = min(deadline - 100, phase6_deadline + 100)
    if len(ensemble) >= 2 and time.monotonic() < phase7_deadline:
        for i in range(min(len(ensemble), 5)):
            if time.monotonic() > phase7_deadline - 20:
                break

    phase8_deadline = min(deadline - 60, phase7_deadline + 100)
