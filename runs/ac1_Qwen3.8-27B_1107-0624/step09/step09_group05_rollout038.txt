Key innovations:

- Multiplicative update algorithm (NMF-style) for the max objective

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)

def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

# Gradient of ||conv(a,a)||^2 w.r.t. a (for sum-normalized a)
def _grad_l2conv(a):
    """Gradient of ||conv(a,a)||_2^2 w.r.t. a."""
    n = len(a)
    c = _full_convolve(a, a)
    # d||c||^2/da[j] = 4 * sum_k c[k]*a[k-j] for valid k-j
    # = 4 * (cross_correlation of c and a) at appropriate lags
    # cross_correlation[c, a](j) = sum_k c[k+j]*a[k] (for linear, non-circular)
    # But we need: sum_k c[k]*a[k-j] = cross_correlation[a, c](j) = (a * rev(c))(n-1+j) ... 
    # Let me just compute it directly using FFT
    # sum_k c[k]*a[k-j] for k in valid range
    # This is the cross-correlation of a and c, which equals conv(a, rev(c))
    c_rev = c[::-1]
    ac = _full_convolve(a, c_rev)
    # ac[m] = sum_i a[i]*c_rev[m-i] = sum_i a[i]*c[len(c)-1-m+i]
    # We want sum_k c[k]*a[k-j] = sum_k c[k]*a[k-j]
    # Let m = k-j, then k = m+j: sum_m c[m+j]*a[m]
    # = sum_m a[m]*c[m+j]
    # This is the cross-correlation of a and c at lag j
    # = (a * rev_c)[n-1+j] where rev_c[i] = c[len(c)-1-i]
    # Wait, let me just use the direct formula:
    # grad[j] = 4 * sum_{k: 0<='k-j'<n} c[k] * a[k-j]
    # Let l = k-j, then k = l+j, and we need 0 <= l < n and 0 <= l+j < len(c) = 2n-1
    # So l in [max(0, -(j)), min(n-1, 2n-2-j)] = [0, min(n-1, 2n-2-j)] since j>=0
    # = sum_l c[l+j]*a[l] for l in [0, min(n-1, 2n-2-j)]
    # This is the cross-correlation R_ac(j) = sum_l a[l]*c[l+j]
    # Which equals the coefficient of z^j in A(z)*C(z^{-1}) ... 
    # Or: R_ac(j) = (a * c_rev_flipped)[n-1+j] where c_rev_flipped[i] = c[2n-2-i]
    # Actually c_rev = c[::-1] has length 2n-1, c_rev[i] = c[2n-2-i]
    # (a * c_rev)[m] = sum_i a[i]*c_rev[m-i] = sum_i a[i]*c[2n-2-(m-i)] = sum_i a[i]*c[2n-2-m+i]
    # For m = n-1+j: sum_i a[i]*c[2n-2-(n-1+j)+i] = sum_i a[i]*c[n-1-j+i]
    # We want sum_l a[l]*c[l+j]. Setting i=l: sum_l a[l]*c[n-1-j+l]
    # We want c[l+j], not c[n-1-j+l]. These differ by (l+j) - (n-1-j+l) = 2j-(n-1).
    # So (a * c_rev)[n-1+j] = sum_l a[l]*c[n-1-j+l] which is NOT what we want.
    
    # Let me just compute it differently.
    # grad[j] = 4 * sum_{l=0}^{min(n-1, 2n-2-j)} a[l] * c[l+j]
    # This is the "cross-correlation" of a and c at lag j.
    # In FFT terms: cross_corr(a, c, j) = IFFT(FFT(a) * conj(FFT(c))) ... 
    # Actually, the linear cross-correlation R(j) = sum_l a[l]*c[l+j] can be computed as:
    # R = IFFT(FFT(a_padded) * FFT(c_padded)[::-1]) ... this is getting complicated.
    
    # Let me just use the property: grad[j] = 4 * (a ⋆ c)[j] where ⋆ denotes cross-correlation
    # and compute it via: IFFT(conj(FFT(a)) * FFT(c)) gives the cross-correlation
    # But we need to be careful about the linear vs circular distinction.
    
    # Simplest approach: just compute the convolution
    # grad[j] = 4 * sum_l a[l] * c[l+j] 
    # = 4 * (a * c_flipped_shifted)[...]
    # Let me define: d[m] = c[2n-2-m] (flip c), length 2n-1
    # (a * d)[n-1+j] = sum_i a[i]*d[n-1+j-i] = sum_i a[i]*c[2n-2-(n-1+j-i)] = sum_i a[i]*c[n-1-j+i]
    # Still not right.
    
    # OK let me just do it the brute force way using the fact that it's a convolution:
    # grad[j] = 4 * sum_{l} a[l] * c[l+j]
    # Define e[k] = c[k] for k in [0, 2n-2], 0 otherwise (length 2n-1)
    # Then sum_l a[l]*c[l+j] = (a * e_rev_shifted)[...]
    
    # Actually the cleanest way: 
    # sum_l a[l]*c[l+j] = sum_l a[l] * c[l+j]
    # This is the same as: if we reverse a, get a_rev[i] = a[n-1-i], then:
    # sum_l a[l]*c[l+j] = sum_i a_rev[n-1-i]*c[n-1-i+j] = conv(a_rev, c)[n-1+j]
    # Let's verify: conv(a_rev, c)[m] = sum_i a_rev[i]*c[m-i] = sum_i a[n-1-i]*c[m-i]
    # For m = n-1+j: sum_i a[n-1-i]*c[n-1+j-i] = sum_i a[n-1-i]*c[n-1-i+j]
    # Let l = n-1-i, i = n-1-l: sum_l a[l]*c[l+j] ✓
    
    a_rev = a[::-1]
    ac = _full_convolve(a_rev, c)
    # ac has length n + (2n-1) - 1 = 3n-2
    # We want ac[n-1 : 2n-1] (n elements)
    grad = 4.0 * ac[n-1:2*n-1]
    return grad

def _proj_grad_l2conv(a, deadline, steps=None):
    """Projected gradient method to minimize ||conv(a,a)||^2 subject to sum(a)=1, a>=0."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_val = float(np.sum(_full_convolve(a, a)**2))
    
    # Backtracking line search
    step = 1.0
    no_improve = 0
    
    while time.monotonic() < deadline:
        if steps is not None:
            steps -= 1
            if steps <= 0:
                break
        
        c = _full_convolve(a, a)
        val = float(np.sum(c**2))
        if val < best_val - 1e-15:
            best_val = val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break
        
        grad = _grad_l2conv(a)
        # Center the gradient (since we're on the hyperplane sum=1)
        grad_c = grad - np.mean(grad)
        
        # Backtracking line search
        t = step
        improved = False
        for _ in range(20):
            a_new = a - t * grad_c
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                t *= 0.5
                continue
            a_new = a_new / s_new
            c_new = _full_convolve(a_new, a_new)
            val_new = float(np.sum(c_new**2))
            if val_new < val - 1e-15:
                a = a_new
                improved = True
                break
            t *= 0.5
        
        if not improved:
            step *= 0.5
            if step < 1e-10:
                break
        
    return best_a, _score(best_a)

def _multiplicative_update(a_init, deadline, min_time_remaining=30.0):
    """NMF-style multiplicative update for minimizing max(conv(a,a)).
    
    We use the soft-max approximation and multiplicative updates:
    a[j] <- a[j] * (target / (grad[j] + eps))^eta
    
    where target is chosen to maintain sum(a)=1.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    c0 = _full_convolve(a, a)
    gap = float(c0.max() - c0.mean())
    if gap < 1e-10:
        return best_a, best_sc
    
    beta = gap * 0.1
    eta = 0.5
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        w = _soft_max_weights(c, beta)
        
        # Gradient of soft_max(conv) w.r.t. a
        # = sum_k w[k] * 2*a[k-j] for valid k-j
        # = 2 * conv(w, a_rev)[n-1 : 2n-1]
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n-1:2*n-1]
        
        # Center gradient
        grad_c = grad - np.mean(grad)
        
        # Multiplicative update with centering
        # a[j] <- a[j] * exp(-eta * grad_c[j]) (approximately)
        # But we need to maintain sum(a)=1
        # Use: a_new[j] = a[j] * exp(-eta * grad_c[j]) / (1 + eta * |grad_c[j]|)
        # Or simpler: a_new[j] = a[j] * (1 - eta * grad_c[j]) for small eta
        
        update = np.exp(-eta * np.clip(grad_c, -50, 50))
        a_new = a * update
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-13:
            best_sc = sc_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Decay parameters
        if no_improve % 50 == 0 and no_improve > 0:
            eta *= 0.9
            beta *= 0.8
            if beta < gap * 1e-6:
                break
        
    return best_a, best_sc

def _active_set_qp(a_init, deadline, min_time_remaining=30.0):
    """Active-set refinement using direct QP-like updates.
    
    At the optimum, for the active set S of peaks:
    The subgradient condition says: for j in support(a),
    (1/|S|) * sum_{k in S} 2*a[k-j] = lambda (constant)
    
    We iterate:
    1. Find active set S
    2. Compute target gradient g[j] = (1/|S|) * sum_{k in S} 2*a[k-j]
    3. Update a to equalize g on the support
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    tol_frac = 0.01
    min_tol = 0.0005
    no_improve = 0
    max_no_improve = 500
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.7)
            if tol_frac <= min_tol * 1.2:
                break
            no_improve = 0
        
        # Active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        
        # Compute average gradient from active set
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n-1:2*n-1]
        
        # Support of a
        support = a > 1e-10 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break
        
        # Target: make grad_avg[j] = lambda for j in support
        lam = float(np.mean(grad_avg[support]))
        residual = grad_avg[support] - lam
        
        # Newton-like step:
        # We want to change a so that the new grad_avg is closer to constant
        # da affects grad_avg through: dgrad_avg[j] = 2*(1/n_active)*sum_{k in active} da[k-j]
        # This is a linear operation. For a small step, we solve:
        # B * da = -residual (on support)
        # where B[j,j'] = 2*(1/n_active)*sum_{k in active} delta(k-j-j') = 2*(1/n_active)*1[j'-j in active_shifts]
        # This is like a banded matrix. For simplicity, use gradient step.
        
        eta = 0.1
        for trial_eta in [eta, eta*0.5, eta*0.25, eta*0.1]:
            a_try = a.copy()
            a_try[support] -= trial_eta * residual
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-14:
                a = a_try
                if trial_eta < eta:
                    eta = trial_eta
                break
        
        # Gradually tighten
        if no_improve == 0:
            tol_frac = max(min_tol, tol_frac * 0.98)
    
    return best_a, best_sc

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "one_sided":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t) + 0.005
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.6 * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    elif kind == "super_tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2.0
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_tops":
        # Multiple flat regions with smooth transitions
        t = np.linspace(0, 1, n)
        a = 1.0
        for center in [0.15, 0.5, 0.85]:
            a += 0.5 * np.exp(-50*(t-center)**2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _run_lbfgs(a_init, deadline, min_time_remaining=60.0, verbose=True):
    """L-BFGS with aggressive adaptive beta schedule."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    
    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    # Compute initial gap for beta scheduling
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c0_max * 0.003:
        gap = c0_max * 0.003
    
    # Many fine-grain beta levels (more aggressive schedule)
    multipliers = np.array([
        2.0, 1.0, 0.5, 0.25, 0.125, 0.0625,
        0.03125, 0.015625, 0.0078125, 0.00390625,
        0.002, 0.001, 0.0005, 0.00025, 0.000125,
        0.0000625, 0.00003125, 0.000015625, 0.0000078,
        0.0000039, 0.000002, 0.000001, 0.0000005
    ])
    betas = gap * multipliers
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        size_factor = max(1.0, n / 4096.0)
        est_iter_time = 0.002 * size_factor
        max_iter = min(10000, max(200, int(remaining / (est_iter_time * 15))))
        
        if beta < gap * 0.001:
            max_iter = min(max_iter, 5000)
        
        # Objective
        def obj(z):
            a_z = _softmax(z)
            c = _full_convolve(a_z, a_z)
            return 2.0 * n * _soft_max_value(c, beta)
        
        # Gradient
        def grad(z):
            a_z = _softmax(z)
            c = _full_convolve(a_z, a_z)
            w = _soft_max_weights(c, beta)
            a_rev = a_z[::-1]
            wc = _full_convolve(w, a_rev)
            grad_a = 2.0 * wc[n-1:2*n-1]
            W = float(np.dot(grad_a, a_z))
            return a_z * (grad_a - W)
        
        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"    beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}")
        except Exception as e:
            if verbose:
                print(f"    beta={beta:.2e}: error {e}")
    
    return best_a, best_sc

def _perturb(a, sigma, rng):
    """Add multiplicative noise and renormalize."""
    noise = np.exp(sigma * rng.standard_normal(len(a)))
    a_new = a * noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def _annealing_search(a_init, deadline, min_time_remaining=60.0, rng=None):
    """Simulated annealing on top of local optimization."""
    if rng is None:
        rng = np.random.default_rng(12345)
    
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    T = 0.5  # initial temperature
    
    while time.monotonic() + min_time_remaining < deadline:
        # Perturb
        sigma = T * 0.1
        a_try = _perturb(best_a, sigma, rng)
        
        # Quick local optimization (short L-BFGS)
        local_deadline = time.monotonic() + min(30, (deadline - time.monotonic()) * 0.3)
        a_opt, sc_opt = _run_lbfgs(a_try, local_deadline, min_time_remaining=min_time_remaining, verbose=False)
        
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            T *= 0.95  # cool down on success
        else:
            # Accept with probability exp(-(sc_opt - best_sc)/T)
            if rng.random() < np.exp(-min((sc_opt - best_sc) / max(T, 1e-10), 100)):
                best_a = a_opt.copy()
            T *= 0.97  # cool down
        
        if T < 0.01:
            break
    
    return best_a, best_sc

def propose_candidate(budget_s=1800):
    """
    Multi-strategy optimizer combining:
    1. Convex warm-start (minimize ||conv||^2 via projected gradient)
    2. L-BFGS with soft-max (main workhorse)
    3. Active-set KKT refinement
    4. Multiplicative updates
    5. Simulated annealing for escaping local minima
    6. Multi-n exploration
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(42)
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(16384, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n=16384: score={best_score:.10f}")
    
    n_main = len(best_seq)
    
    # === Phase 0: Convex warm-start (projected gradient on ||conv||^2) ===
    print(f"\n[P0] Convex warm-start (t={time.monotonic()-t_start:.0f}s)")
    p0_deadline = t_start + min(120, (deadline - t_start) * 0.08)
    a0, sc0 = _proj_grad_l2conv(best_seq, p0_deadline)
    if sc0 < best_score:
        best_score = sc0
        best_seq = a0.copy()
        print(f"  P0 result: {best_score:.10f}")

"""

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


class _SoftMaxObj:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        return 2.0 * self.n * _soft_max_value(c, self.beta)

    def grad(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        w = _soft_max_weights(c, self.beta)
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad_a = 2.0 * wc[self.n - 1:2 * self.n - 1]
        W = float(np.dot(grad_a, a))
        return a * (grad_a - W)


def _grad_l2conv(a):
    """Gradient of ||conv(a,a)||_2^2 w.r.t. a."""
    n = len(a)
    c = _full_convolve(a, a)
    a_rev = a[::-1]
    ac = _full_convolve(a_rev, c)
    grad = 4.0 * ac[n - 1:2 * n - 1]
    return grad

def _active_set_refine(a_init, deadline, min_time_remaining=30.0):
    """KKT-inspired active-set refinement with tighter convergence."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_frac = 0.005
    min_tol = 0.0002
    no_improve = 0
    max_no_improve = 400
    eta_base = 0.08

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.6)
            if tol_frac <= min_tol * 1.1:
                break
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]

        support = a > 1e-9 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        lam = float(np.mean(grad_avg[support]))
        residual = grad_avg[support] - lam

        found = False
        for trial_eta in [eta_base, eta_base * 0.5, eta_base * 0.25,
                          eta_base * 0.125, eta_base * 0.0625]:
            a_try = a.copy()
            a_try[support] -= trial_eta * residual
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-14:
                a = a_try
                eta_base = trial_eta
                found = True
                break

        if not found:
            tol_frac = max(min_tol, tol_frac * 0.85)
            continue

        if no_improve == 0:
            tol_frac = max(min_tol, tol_frac * 0.97)

    return best_a, best_sc


def _multiplicative_update(a_init, deadline, min_time_remaining=30.0):

    n = len(a_init)

    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c0 = _full_convolve(a, a)

    if gap < 1e-10:
        return best_a, best_sc

    eta = 0.6
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        w = _soft_max_weights(c, beta)
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]
        grad_c = grad - np.mean(grad)

        log_update = -eta * np.clip(grad_c, -50, 50)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-10:

        a_new = a_new / s_new

            best_a = a_new.copy()

        else:

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "super_tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2.0
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.5 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quartic":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 3.0
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb_and_optimize(a_init, sigma, deadline, min_time_remaining, rng):
    """Perturb with multiplicative noise, then brief local optimization."""
    a_try = a_init * np.exp(sigma * rng.standard_normal(len(a_init)))
    a_try = np.maximum(a_try, 0.0)
    s = a_try.sum()
    if s < 1e-10:
        return a_init, _score(a_init)
    a_try = a_try / s
    local_deadline = min(deadline - min_time_remaining, time.monotonic() + min(40, (deadline - time.monotonic()) * 0.2))
    return _run_lbfgs(a_try, local_deadline, min_time_remaining=min_time_remaining, verbose=False)
