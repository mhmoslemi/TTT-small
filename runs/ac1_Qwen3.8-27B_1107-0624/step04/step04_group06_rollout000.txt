def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())

def _exact_subgrad_update(a, c, peak_frac=0.1):
    """Compute exact subgradient of max(c) w.r.t. a.
    The subgradient is uniform over argmax set of c.
    Returns gradient vector of shape (n,)."""
    n = len(a)
    M = float(c.max())
    # Find peaks within peak_frac of max
    threshold = M * (1.0 - peak_frac)
    peak_mask = c >= threshold
    n_peaks = int(peak_mask.sum())
    if n_peaks == 0:
        peak_mask[int(np.argmax(c))] = True
        n_peaks = 1

    # Gradient of c_k w.r.t. a_i is a_{k-i} (for valid i)
    # Subgradient of max = average over peaks
    grad = np.zeros(n)
    peak_indices = np.where(peak_mask)[0]
    for k in peak_indices:

        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            grad[i] += a[k - i]
    grad /= n_peaks
    return grad

        a_new = a_new / a_new.sum()

        c = _full_convolve(a_new)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.001, lr * 0.9)

        a = a_new

        if t_adam % 50 == 0:
            beta = max(1e-6, beta * beta_decay)
            # Adapt based on peak spread
            top3 = np.sort(c)[::-1][:3]
            if len(top3) >= 2 and top3[0] - top3[1] < beta * 0.01:
                beta = max(1e-6, beta * 0.5)

        limit = max(80, no_imp_limit)
        if no_imp > limit:
            # Restart with perturbation
            scale = 0.05 + 0.15 * (1.0 - best_score / 2.0)
            pert = a + scale * rng.normal(size=n)
            a = np.maximum(pert, 1e-12)
            a = a / a.sum()
            c = _full_convolve(a)
            maxc = float(c.max())
            beta = max(1e-4, beta0 * maxc)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = 0.04
            no_imp = 0

    return best_a, best_score


def _lp_solve_fast(a, M):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M, b >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_aggressive(a, deadline, max_solves=80, m_shrink=0.993, t_init=0.3):
    """Aggressive LP refinement: many small steps with decreasing M."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    t = t_init
    n_solves = 0
    stall = 0

    while time.monotonic() < deadline and n_solves < max_solves:
        b = _lp_solve_fast(a, M)
        if b is None:
            # Try slightly relaxed
            b = _lp_solve_fast(a, M * 1.02)
            if b is None:
                break
            M = float(c.max()) * 1.02

        n_solves += 1
        s_b = float(b.sum())
        if s_b < 1e-10:
            break

        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)

        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break

        a = a_new
        c = _full_convolve(a)
        M_new = float(c.max())
        M = max(M_new * m_shrink, best_score / (2.0 * n) * 1.001)
        t = max(0.08, t * 0.93)

    return best_a, best_score


def _peak_shave(a, deadline, n_iters=2000):
    """Direct peak shaving: move mass away from top contributors to peaks."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    rng = np.random.default_rng(777)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        M = float(c.max())
        peak_k = int(np.argmax(c))
        score = 2.0 * n * M

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        # Find top contributors to the peak
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        partners = peak_k - np.arange(i_start, i_end + 1)
        contrib = a[i_start:i_end + 1] * a[partners]
        top_idx = np.argsort(contrib)[::-1]

        # Move mass from top contributor to a "cold" position
        src_i = i_start + top_idx[0]
        # Find cold positions (low contribution to any peak)
        cold_score = a.copy()
        # Reduce score for positions that contribute to top peaks
        top_peaks = np.argsort(c)[::-1][:5]
        for pk in top_peaks:
            ps = max(0, pk - n + 1)
            pe = min(n - 1, pk)
            for i in range(ps, pe + 1):
                cold_score[i] -= a[pk - i] * 0.5
                if pk - i < n:
                    cold_score[pk - i] -= a[i] * 0.5

        # Pick target from bottom half of cold_score
        bottom_half = np.argsort(cold_score)[:n // 2]
        target = bottom_half[rng.integers(0, len(bottom_half))]

        if target == src_i:
            continue

        # Small transfer
        frac = 0.005 + 0.01 * rng.random()
        amount = a[src_i] * frac
        a[src_i] -= amount
        a[target] += amount
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        c = _full_convolve(a)

        # Check all peaks, not just the first
        if it % 50 == 0:
            top_c = np.sort(c)[::-1][:3]
            if len(top_c) >= 3 and top_c[0] - top_c[2] < 1e-8:
                # All peaks nearly equal - we're in a good region
                pass

    return best_a, best_score


def _mirror_sym_optimize(a, deadline, rng):
    """Optimize under mirror symmetry constraint (a_i = a_{n-1-i})."""
    n = len(a)
    if n % 2 != 0:
        return a, _score_norm(a)

    a = np.asarray(a, dtype=np.float64).ravel().copy()
    # Enforce symmetry: average with reversed
    a_sym = 0.5 * (a + a[::-1])
    a_sym = a_sym / a_sym.sum()

    best_a = a_sym.copy()
    best_score = _score_norm(a_sym)

    # Optimize only first half, then mirror
    half_n = n // 2
    # Use Adam on the half
    c = _full_convolve(a_sym)
    maxc = float(c.max())
    beta = 0.05 * maxc

    m = np.zeros(half_n)
    v = np.zeros(half_n)
    t_adam = 0
    lr = 0.03
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_imp = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a_sym)
        M = float(c.max())
        x = (c - M) / beta
        ex = np.exp(np.clip(x, -50, 0))
        s = float(ex.sum())
        if s < 1e-30:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s

        # Compute gradient on full sequence
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a_sym[::-1], N)
        h = np.fft.irfft(fw * fa_rev, N)
        grad_full = h[n - 1:2 * n - 1]

        # Average gradient with its mirror (for symmetry)
        grad_half = 0.5 * (grad_full[:half_n] + grad_full[half_n:][::-1])
        # Also include the middle pair if n is even
        dot = float(np.dot(a_sym[:half_n], grad_half))
        gz = a_sym[:half_n] * (grad_half - dot)

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)

        new_half = a_sym[:half_n] * (1.0 - lr * mh / (np.sqrt(vh) + eps))
        new_half = np.maximum(new_half, 1e-15)

        # Mirror
        a_new = np.zeros(n)
        a_new[:half_n] = new_half
        a_new[half_n:] = new_half[::-1]
        a_new = a_new / a_new.sum()

        sc_new = _score_norm(a_new)
        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()
            a_sym = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 200:
                break

        a_sym = a_new
        beta = max(1e-5, beta * 0.97)

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def _make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / sigma_frac) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bilateral(n, alpha=0.4):
    """Bilateral (high at edges, low in middle)."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_plateau(n, w=0.3, h=2.0):
    """Plateau in the middle."""
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    mask = np.abs(t - 0.5) < w / 2
    a[mask] = h
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc > 0:
            sc = _score_norm(p)
            if sc < best_score:
                best_seq = np.asarray(p, dtype=np.float64).copy()
                best_score = float(sc)
                print(f"  NEW BEST: {best_score:.10f} n={p.size} [{label}] t={time.monotonic()-t_start:.0f}s")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        pn = parent.size
        sc_parent = _score_norm(parent)
        print(f"Parent: n={pn}, score={sc_parent:.10f}")
        _update(parent, sc_parent, 'parent')
    else:
        parent = _make_bilateral(1024, 0.5)
        print("No parent, using bilateral init n=1024")
        _update(parent, _score_norm(parent), 'init_bilateral')

    # === Phase 1: Quick Adam on parent (0-80s) ===
    p1_end = t_start + 80.0
    print(f"Phase 1: Adam polish (80s)")
    p1, sc1 = _adam_improved(parent, p1_end, rng, lr=0.04, beta0=0.06,
                             beta_decay=0.94, no_imp_limit=200)
    _update(p1, sc1, 'adam_p1')

    # === Phase 2: Aggressive LP (80-450s) ===
    p2_end = t_start + 450.0
    print(f"Phase 2: Aggressive LP ({best_score:.10f})")
    if best_seq is not None and best_seq.size <= 2048:
        p2, sc2 = _lp_aggressive(best_seq, p2_end, max_solves=60, m_shrink=0.993, t_init=0.25)
        _update(p2, sc2, 'lp_aggr')

    # Adam after LP
    if time.monotonic() < p2_end and best_seq is not None:
        p2a, sc2a = _adam_improved(best_seq, min(p2_end, time.monotonic() + 40), rng,
                                   lr=0.02, beta0=0.03, beta_decay=0.96, no_imp_limit=150)
        _update(p2a, sc2a, 'post_lp_adam')

    # === Phase 3: Peak shaving (450-650s) ===
    p3_end = t_start + 650.0
    print(f"Phase 3: Peak shaving ({best_score:.10f})")
    if best_seq is not None:
        p3, sc3 = _peak_shave(best_seq, p3_end, n_iters=3000)
        _update(p3, sc3, 'peak_shave')

    # More LP after shaving
    if time.monotonic() < p3_end and best_seq is not None and best_seq.size <= 2048:
        p3l, sc3l = _lp_aggressive(best_seq, p3_end, max_solves=20, m_shrink=0.995, t_init=0.15)
        _update(p3l, sc3l, 'lp_after_shave')

    # === Phase 4: Subgradient with backtracking (650-800s) ===
    p4_end = t_start + 800.0
    print(f"Phase 4: Exact subgradient ({best_score:.10f})")
    if best_seq is not None:
        p4, sc4 = _subgrad_backtrack(best_seq, p4_end, max_iter=3000, peak_frac=0.02)
        _update(p4, sc4, 'subgrad_bt')

    # === Phase 5: Try n=2048 (800-1200s) ===
    p5_end = t_start + 1200.0
    print(f"Phase 5: Multi-scale n=2048 ({best_score:.10f})")
    if time.monotonic() < p5_end and best_seq is not None:
        n_new = 2048
        p_int = _interp_profile(best_seq, n_new)
        if p_int is not None:
            sc_int = _score_norm(p_int)
            print(f"  Interpolated to n={n_new}: {sc_int:.10f}")
            _update(p_int, sc_int, f'interp_n{n_new}')

            # Adam on new scale
            per = min(100.0, p5_end - time.monotonic() - 30)
            if per > 30:
                p5a, sc5a = _adam_improved(p_int, time.monotonic() + per, rng,
                                           lr=0.03, beta0=0.05, beta_decay=0.94, no_imp_limit=200)
                _update(p5a, sc5a, f'adam_n{n_new}')

            # LP on new scale
            if time.monotonic() < p5_end - 20 and best_seq.size <= n_new:
                p5l, sc5l = _lp_aggressive(best_seq, min(p5_end - 10, time.monotonic() + 120),
                                           max_solves=25, m_shrink=0.994, t_init=0.2)
                _update(p5l, sc5l, f'lp_n{best_seq.size}')

    # === Phase 6: Mirror symmetry exploration (1200-1400s) ===
    p6_end = t_start + 1400.0
    print(f"Phase 6: Mirror symmetry ({best_score:.10f})")
    if time.monotonic() < p6_end and best_seq is not None:
        # Make current best symmetric
        a_sym = 0.5 * (best_seq + best_seq[::-1])
        a_sym = a_sym / a_sym.sum()
        sc_sym = _score_norm(a_sym)
        _update(a_sym, sc_sym, 'symmetrized')

        p6, sc6 = _mirror_sym_optimize(a_sym, p6_end, rng)
        _update(p6, sc6, 'mirror_opt')

    # === Phase 7: Alternative initializations (1400-1600s) ===
    p7_end = t_start + 1600.0
    print(f"Phase 7: Alt inits ({best_score:.10f})")
    if time.monotonic() < p7_end:
        n_try = 1024
        alts = [
            ('gauss_25', _make_gaussian(n_try, 0.25)),
            ('gauss_35', _make_gaussian(n_try, 0.35)),
            ('bilateral_6', _make_bilateral(n_try, 0.6)),
            ('plateau', _make_plateau(n_try, 0.4, 1.5)),
            ('plateau_wide', _make_plateau(n_try, 0.6, 1.3)),
        ]
        per_alt = max(20.0, (p7_end - time.monotonic()) / len(alts))
        for label, pa in alts:
            if time.monotonic() > p7_end - 10:
                break
            per = min(per_alt, p7_end - time.monotonic() - 5)
            if per < 15:
                break
            p_alt, sc_alt = _adam_improved(pa, time.monotonic() + per, rng,
                                           lr=0.05, beta0=0.08, beta_decay=0.93, no_imp_limit=100)
            _update(p_alt, sc_alt, f'alt_{label}')

    # === Phase 8: Final intensive alternation (1600-1780s) ===
    p8_end = deadline
    print(f"Phase 8: Final alternation ({best_score:.10f})")
    if time.monotonic() < p8_end and best_seq is not None:
        # Cycle through methods
        cycle = [
            ('adam', lambda a, dl: _adam_improved(a, dl, rng, lr=0.015, beta0=0.02,
                                                   beta_decay=0.97, no_imp_limit=80)),
            ('lp', lambda a, dl: _lp_aggressive(a, dl, max_solves=8, m_shrink=0.998, t_init=0.1)),
            ('shave', lambda a, dl: _peak_shave(a, dl, n_iters=500)),
        ]

        ci = 0
        while time.monotonic() < p8_end - 10:
            label, fn = cycle[ci % len(cycle)]
            remaining = p8_end - time.monotonic() - 10
            per = min(60.0, remaining)
            if per < 10:
                break
            try:
                p8, sc8 = fn(best_seq, time.monotonic() + per)
                _update(p8, sc8, f'final_{label}')
            except Exception as e:
                print(f"  Error in {label}: {e}")
            ci += 1

    # === Final validation ===
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_bilateral(1024, 0.5)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_bilateral(1024, 0.5)
    print(f"FALLBACK n=1024, score={_score_norm(fb):.10f}")
    return [float(x) for x in fb]
