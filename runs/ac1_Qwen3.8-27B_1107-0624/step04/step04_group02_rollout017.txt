"""
LP-based optimizer with gradient polishing for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Primary method: iterative LP to find optimal update directions.
Secondary method: L-BFGS with logsumexp continuation for polishing.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix, coo_matrix
import time

def _fn_next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fn_full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _fn_next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _fn_score_norm(a):
    n = len(a)
    c = _fn_full_convolve(a, a)
    return float(2.0 * n * c.max())

def _fn_score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fn_full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _fn_softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _fn_build_conv_matrix(f):
    """Build sparse matrix A where (A @ g)[k] = (f * g)[k] for k=0..2n-2."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def _fn_lp_update(f, M):
    """
    Solve: maximize sum(g) subject to (f*g)[k] <= M for all k, g >= 0.
    Returns (g, sum_g) where g is the optimal solution.
    """
    n = len(f)
    A = _fn_build_conv_matrix(f)
    # c = -ones (we minimize -sum(g))
    c_obj = -np.ones(n)
    # A_ub @ g <= b_ub
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None
    
    g = res.x
    sum_g = float(g.sum())
    return g, sum_g

def _fn_lp_iterate(f, deadline, max_iters=100, blend_t=0.1):
    """
    Iterative LP-based optimization.
    f is normalized (sum=1, non-negative).
    """
    n = len(f)
    f = f.copy()
    best_f = f.copy()
    best_score = _fn_score_norm(f)
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 10:
            break
        
        M = float(_fn_full_convolve(f, f).max())
        
        g, sum_g = _fn_lp_update(f, M)
        if g is None:
            break
        
        if sum_g <= 1.0 + 1e-10:
            # No improvement possible
            break
        
        # Rescale g to sum=1
        g_norm = g / sum_g
        
        # Blend
        t = min(blend_t, 0.5)
        f_new = (1 - t) * f + t * g_norm
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        
        sc = _fn_score_norm(f_new)
        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            f = f_new.copy()
        else:

            if t < 0.001:

            f_new = (1 - t) * f + t * g_norm
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            sc = _fn_score_norm(f_new)
            if sc < best_score:

                best_f = f_new.copy()
                f = f_new.copy()
    
    return best_f, best_score

def _fn_local_search(a_init, deadline, rng, n_trials=5000):
    """Local search with random block perturbations."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _fn_score_norm(a)
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break

        start = rng.integers(0, n - block_size)
        
        a_trial = a.copy()
        perturbation = rng.uniform(0.85, 1.15, size=block_size)
        a_trial[start:start + block_size] *= perturbation
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial /= a_trial.sum()
        
        sc = _fn_score_norm(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
    
    return best_a, best_score

def _fn_interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    if len(p0) == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, len(p0))
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

def _fn_random_spectrum_init(n, rng):
    """Generate a sequence with nearly flat spectrum (random phases)."""
    # Random phases in frequency domain
    k = np.arange(n // 2 + 1)
    phases = rng.uniform(0, 2*np.pi, size=len(k))
    # Constant magnitude
    spec = np.exp(1j * phases)
    # Handle conjugate symmetry for real signal
    # For even n: spec[0] and spec[n/2] must be real
    spec[0] = 1.0
    if n % 2 == 0:
        spec[-1] = 1.0
    
    a = np.fft.irfft(spec, n)
    # Project to non-negative
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
