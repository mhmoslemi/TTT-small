"""
Optimizer for minimizing 2*n*max(convolve(a,a))/(sum(a))^2 over non-negative sequences.
Uses smooth-max (log-sum-exp) approximation with multiple optimization strategies:
L-BFGS-B, Adam with decreasing temperature, and active-set subgradient methods.
Key insight: for non-negative sequences, the autocorrelation peak is at lag 0,
so the objective simplifies to 2*n*sum(a_i^2)/(sum(a_i))^2, minimized by uniform.
However, we still run optimization in case the evaluator has nuances we miss.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _gradient_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _objective_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_obj(z, beta, n):
    return _objective_smooth(z, beta)


def _lbfgs_grad(z, beta, n):

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _run_lbfgs(a_init, deadline, max_wall_time=None):

    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, beta_frac * max_c)

    while time.monotonic() < deadline:
        t += 1

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a = _softmax_fn(z)

        if t % 100 == 0:
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
            if t % 500 == 0:
                beta = max(1e-12, beta * 0.7)
                c_new = _full_convolve(a, a)
                max_c = float(c_new.max())
                beta = max(1e-12, 0.005 * max_c)

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _nesterov_optimize(a_init, deadline, beta_fracs, lr_init=0.02, mom_init=0.9):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())

    lr = lr_init
    mom = mom_init
    t = 0

    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-12, beta_frac * max_c)

        while time.monotonic() < deadline:
            t += 1
            grad_y = _gradient_smooth(y, beta)
            if not np.all(np.isfinite(grad_y)):
                grad_y = _gradient_smooth(z, beta)

            z_new = z - lr * grad_y
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -50, 50)

            y_new = z_new + mom * (z_new - z)
            y_new = y_new - y_new.mean()

            z = z_new
            y = y_new

            if t % 50 == 0:
                a_curr = _softmax_fn(z)
                sc = _score_norm_fn(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()

            if t % 200 == 0:
                mom = min(0.999, mom + 0.0003)

        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        max_c = float(c.max())
        lr = max(0.002, lr * 0.75)

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _active_set_polish(a_init, deadline, n_iter=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    lr = 0.003
    momentum = np.zeros(n)
    mom_coeff = 0.85

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        active = c >= cm * 0.998
        if not np.any(active):
            active[int(np.argmax(c))] = True

        w = np.zeros_like(c)
        w[active] = 1.0 / np.sum(active)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        momentum = mom_coeff * momentum - lr * grad
        z = z + momentum
        z = z - z.mean()
        z = np.clip(z, -50, 50)

        if it % 100 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            else:
                lr = lr * 0.85

    return best_a, best_score


def _direct_subgradient(a_init, deadline, n_iter=3000):
    """Direct subgradient on the objective 2*n*max(conv)/(sum)^2 using normalized a."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    lr = 1e-4
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        # Subgradient of max at the maximizing lag
        idx = int(np.argmax(c))
        # d/d a_i of conv(idx) = a[(idx - i) mod ... ] -- use FFT approach
        a_rev = a[::-1]
        one_hot = np.zeros(2 * n - 1)
        one_hot[idx] = 1.0
        oc = _full_convolve(one_hot, a_rev)
        g = oc[n - 1:2 * n - 1]

        # Normalize: we're optimizing in the space where sum(a)=1
        # Project gradient onto sum=0 hyperplane
        g = g - np.mean(g)
        
        a = a - lr * g
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if it % 200 == 0:
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
            else:
                lr *= 0.8

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: evaluates parametric families, then applies
    L-BFGS, Adam, Nesterov, and active-set methods with decreasing temperature.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # Get parent sequence if available
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    # Determine target length
    if parent is not None:
        n_target = len(parent)
    else:
        n_target = 4096

    print(f"Target n={n_target}")

    # Phase 0: Evaluate diverse initial candidates (30s)
    phase0_end = min(deadline, t_start + 30.0)
    candidates = []

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        candidates.append((sc_parent, parent.copy(), "parent"))
        update_best(parent, sc_parent, "parent")
        print(f"  Parent score: {sc_parent:.10f}")

    # Parametric families at target length
    families = [
        ("uniform", _make_uniform(n_target)),
        ("cosine", _make_cosine_window(n_target)),
        ("triangle", _make_triangle(n_target)),
        ("gauss_0.2", _make_gaussian_window(n_target, 0.2)),
        ("gauss_0.4", _make_gaussian_window(n_target, 0.4)),
        ("gauss_0.7", _make_gaussian_window(n_target, 0.7)),
        ("ramp_1", _make_ramp_window(n_target, 1.0)),
        ("ramp_2", _make_ramp_window(n_target, 2.0)),
        ("linear_pos", _make_linear_profile(n_target, 1.0)),
        ("linear_neg", _make_linear_profile(n_target, -1.0)),
        ("linear_2", _make_linear_profile(n_target, 2.0)),
        ("linear_2n", _make_linear_profile(n_target, -2.0)),
    ]

    # Also try interpolated parent at different lengths
    if parent is not None and parent.size != n_target:
        p_interp = _interp_profile(parent, n_target)
        candidates.append((_score_norm_fn(p_interp), p_interp, "parent_interp"))

    for name, p in families:
        sc = _score_norm_fn(p)
        candidates.append((sc, p, name))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0: top candidates:")
    for sc, p, name in candidates[:5]:
        print(f"    {name}: {sc:.10f}")

    # Keep the best as our working candidate
    if candidates:
        sc_best0, p_best0, name_best0 = candidates[0]
        if np.isfinite(sc_best0) and sc_best0 < best_score:
            best_seq = p_best0.copy()
            best_score = sc_best0
            print(f"  Best after Phase 0: {name_best0} with {best_score:.10f}")

    if best_seq is None:
        best_seq = _make_uniform(n_target)
        best_score = _score_norm_fn(best_seq)

    # Phase 1: L-BFGS continuation (300s)
    phase1_end = min(deadline, t_start + 330.0)
    if time.monotonic() < phase1_end:
        per_time = min(280.0, phase1_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _run_lbfgs(best_seq, min(phase1_end, time.monotonic() + per_time),
                                         max_wall_time=per_time)
            update_best(p_best, sc_best, "lbfgs_p1")

    # Phase 2: Adam with moderate temperature (300s)
    phase2_end = min(deadline, t_start + 630.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        per_time = min(280.0, phase2_end - time.monotonic())
        if per_time > 30:
            p_best, sc_best = _adam_optimize(best_seq, min(phase2_end, time.monotonic() + per_time),
                                             lr=0.04, beta_frac=0.005)
            update_best(p_best, sc_best, "adam_p2")

    # Phase 3: Nesterov with aggressive beta schedule (300s)
    phase3_end = min(deadline, t_start + 930.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        per_time = min(280.0, phase3_end - time.monotonic())
        if per_time > 30:
            beta_fracs = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6]
            p_best, sc_best = _nesterov_optimize(best_seq, min(phase3_end, time.monotonic() + per_time),
                                                  beta_fracs, lr_init=0.015, mom_init=0.9)
            update_best(p_best, sc_best, "nesterov_p3")

    # Phase 4: Active set polishing (200s)
    phase4_end = min(deadline, t_start + 1130.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        per_time = min(180.0, phase4_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _active_set_polish(best_seq, min(phase4_end, time.monotonic() + per_time))
            update_best(p_best, sc_best, "active_p4")

    # Phase 5: Fine Adam with very low temperature (200s)
    phase5_end = min(deadline, t_start + 1330.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        per_time = min(180.0, phase5_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _adam_optimize(best_seq, min(phase5_end, time.monotonic() + per_time),
                                             lr=0.01, beta_frac=0.001)
            update_best(p_best, sc_best, "adam_fine_p5")

    # Phase 6: Restart with noise and re-optimize (200s)
    phase6_end = min(deadline, t_start + 1530.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        n_best = len(best_seq)
        n_restarts = 3
        per_restart = min(60.0, (phase6_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 10:
                break
            noise_level = 0.03 + 0.02 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_level * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            per_dl = min(phase6_end - 5, time.monotonic() + per_restart)
            beta_fracs_short = [0.003, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]
            p_best, sc_best = _nesterov_optimize(p_start, per_dl, beta_fracs_short,
                                                  lr_init=0.01, mom_init=0.85)
            update_best(p_best, sc_best, f"restart{i}_p6")

    # Phase 7: Final L-BFGS polish (150s)
    phase7_end = min(deadline, t_start + 1680.0)
    if best_seq is not None and time.monotonic() < phase7_end:
        per_time = min(130.0, phase7_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _run_lbfgs(best_seq, min(phase7_end, time.monotonic() + per_time),
                                         max_wall_time=per_time)
            update_best(p_best, sc_best, "lbfgs_final_p7")

    # Phase 8: Direct subgradient (remaining time)
    if best_seq is not None and time.monotonic() < deadline - 10:
        per_time = min(deadline - time.monotonic() - 10, 100.0)
        if per_time > 10:
            p_best, sc_best = _direct_subgradient(best_seq, min(deadline - 5, time.monotonic() + per_time))
            update_best(p_best, sc_best, "subgrad_p8")

    print(f"\nFinal best score (norm): {best_score:.10f}")
    print(f"Total elapsed: {time.monotonic() - t_start:.1f}s")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_uniform(n_target)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        # Validate with evaluate_sequence
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned non-finite, using internal: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback: uniform sequence
    fallback = _make_uniform(4096)
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n=4096, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
