import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm_fn(a):
    """Score assuming sum(a) = 1: returns 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _conv_max_and_grad(a, k_top=1):
    """Compute conv(a,a), its max, and subgradient (averaged over top-k)."""
    n = len(a)
    c = _full_convolve(a, a)
    k_actual = min(k_top, len(c))
    # Get top-k indices
    if k_actual == len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argpartition(c, -k_actual)[-k_actual:]
    
    # Gradient: for each active k, grad[i] += 2*a[k-i] (if 0<=k-i<n)
    grad = np.zeros(n)
    for k_val in top_idx:

            idx = k_val - np.arange(lo, hi + 1)
            grad[lo:hi+1] += 2.0 * a[idx]
    
    grad /= k_actual
    return c, float(c.max()), grad

def _next_pow2(n):

def _full_convolve(a, b):

    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

def _softmax_fn(z):

    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()

        return np.ones_like(z) / len(z)
    return e / s

    c = _full_convolve(a, a)

def _frank_wolfe(a_init, deadline, rng, n_iters=50000):
    """Frank-Wolfe (conditional gradient) for the non-convex problem."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        if time.monotonic() > deadline - 1.0:
            break
        if it % 500 == 0 and time.monotonic() > deadline - 30.0:
            break

        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))
        max_c = float(c[k_star])

        # Subgradient at a
        grad = np.zeros(n)
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        if lo <= hi:
            idx = k_star - np.arange(lo, hi + 1)
            grad[lo:hi + 1] = 2.0 * a[idx]
        grad *= 2.0 * n

        # Frank-Wolfe: s = e_{argmin grad}
        j_min = int(np.argmin(grad))
        s = np.zeros(n)
        s[j_min] = 1.0

        # Polyak step size
        gap = 2.0 * n * max_c - lower_bound
        if gap > 1e-15:
            fw_gap = float(np.dot(grad, a - s))
            if fw_gap > 1e-15:
                t_step = gap / fw_gap
                t_step = min(1.0, max(0.0, t_step))
            else:
                t_step = 2.0 / (it + 2.0)
        else:
            break

        a_new = (1.0 - t_step) * a + t_step * s
        a_new = np.maximum(a_new, 0.0)
        s_sum = a_new.sum()
        if s_sum < 1e-15:
            break
        a_new /= s_sum

        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score

            res = minimize(block_obj, block, method='Nelder-Mead',
                           options={'maxiter': 100, 'xatol': 1e-10, 'fatol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_trial = a.copy()
                a_trial[start:end] = np.maximum(res.x, 0.0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial /= ss
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
        except Exception:
            pass

    return best_a, best_score


def _adam_subgrad(a_init, deadline, rng, lr=0.05, momentum=0.9, n_iters=30000):
    """Momentum-based subgradient method with Polyak step size."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)

    m = np.zeros(n)

    for it in range(n_iters):
        if time.monotonic() > deadline - 1.0:
            break

        c = _full_convolve(a, a)
        max_c = float(c.max())

        # Use top-3 for smoothing
        k_top = min(3, len(c))
        top_idx = np.argpartition(c, -k_top)[-k_top:]
        grad = _subgradient(a, top_idx) * 2.0 * n

        gap = 2.0 * n * max_c - lower_bound
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break

        # Polyak-like step size
        t = 0.5 / (it + 2.0)
        # Clip
        t = min(t, lr / (grad_norm + 1e-10))

        m = momentum * m + (1.0 - momentum) * grad
        a_new = a - t * m
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def _freq_guided_perturbation(a_init, deadline, rng, n_trials=2000):
    """Perturb using sinusoidal patterns in the spatial domain."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    t = np.arange(n, dtype=np.float64)

    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break

        # Random frequency and phase
        freq = rng.uniform(0.5, n * 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.01, 0.1)

        perturbation = amp * np.cos(2.0 * np.pi * freq * t / n + phase)
        a_trial = a * (1.0 + perturbation)
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial = a_trial / a_trial.sum()

        sc = _score_norm_fn(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()

    return best_a, best_score


def _make_uniform(n):
    return np.ones(n) / n


def _make_hann(n):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / (n - 1)))
    return a / a.sum()


def _make_hamming(n):
    t = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * t / (n - 1))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_blackman(n):
    t = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * t / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * t / (n - 1))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_kaiser(n, alpha=8.0):
    t = np.arange(n, dtype=np.float64)
    u = 2.0 * t / (n - 1) - 1.0
    a = bessel_i0(alpha * np.sqrt(np.maximum(1.0 - u * u, 0.0)))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian(n, width=0.3):
    t = np.arange(n, dtype=np.float64)
    u = t / n - 0.5
    a = np.exp(-u * u / (2.0 * width * width))
    return a / a.sum()


def _make_linear(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power(n, p):
    t = np.linspace(0.0, 1.0, n)
    u = np.abs(2.0 * t - 1.0)
    a = (1.0 - u) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_bump(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = np.cos(alpha * np.pi * u / 2.0)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_two_hex(n, frac=0.3):
    """Two concentrated blocks."""
    a = np.ones(n) / n
    k = int(n * frac / 2)
    a[:k] = 0.0
    a[n-k:] = 0.0
    middle = a[k:n-k]
    if middle.sum() > 0:
        middle = middle / middle.sum()
        a[k:n-k] = middle
    else:
        a = np.ones(n) / n
    return a


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _topk_adam(a_init, deadline, rng, lr=0.02, k_top=3, decay=0.9999):
    """Adam with top-K smoothed gradient in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8

    for it in range(100000):
        if time.monotonic() > deadline - 1.0:
            break

        t += 1
        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        max_c = float(c.max())

        # Top-K gradient in z-space
        k_actual = min(k_top, len(c))
        top_vals = np.sort(c)[-k_actual:]
        # Smooth weight: use softmax over top-K
        top_idx = np.argsort(c)[-k_actual:]
        w_top = np.zeros(len(c))
        shift = top_vals[-1]
        e_top = np.exp((top_vals - shift) / 0.01)
        w_top[top_idx] = e_top / e_top.sum()

        a_rev = a_curr[::-1]
        h_full = _full_convolve(w_top, a_rev)
        h = h_full[n - 1:n - 1 + n]
        W = float(np.dot(w_top, c))
        grad_z = 4.0 * n * a_curr * (h - W)

        if not np.all(np.isfinite(grad_z)):
            break

        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if t % 20 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()

        if t % 2000 == 0:
            lr = max(0.001, lr * decay)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining L-BFGS continuation, Frank-Wolfe,
    block coordinate descent, frequency-guided perturbations, and Adam.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 4096
        parent = _make_hann(n_parent)
        print(f"No parent, using Hann profile n={n_parent}")

    n = n_parent

    # --- Phase 0: Diverse initialization (20s) ---
    phase0_end = min(deadline, t_start + 20.0)
    candidates = [(best_score, best_seq.copy(), "parent")]

    profiles = [
        _make_hann(n),
        _make_hamming(n),
        _make_blackman(n),
        _make_kaiser(n, 6.0),
        _make_kaiser(n, 10.0),
        _make_gaussian(n, 0.25),
        _make_gaussian(n, 0.35),
        _make_linear(n, 1.0),
        _make_linear(n, 2.0),
        _make_quadratic(n, 0.5, 0.3),
        _make_quadratic(n, 1.0, -0.2),
        _make_power(n, 1.5),
        _make_power(n, 2.0),
        _make_power(n, 3.0),
        _make_cosine_bump(n, 1.0),
        _make_cosine_bump(n, 1.5),
    ]

    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p, f"prof{idx}"))
        update_best(p, sc, f"prof{idx}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}, {len(candidates)} candidates")

    # --- Phase 1: L-BFGS on top-3 candidates (300s) ---
    phase1_end = min(deadline, t_start + 320.0)
    n_to_try = 3
    per_candidate = max(60.0, (phase1_end - time.monotonic()) / max(1, n_to_try))
    count = 0

    for sc, p, label in candidates:
        if time.monotonic() > phase1_end - 5.0 or count >= n_to_try:
            break
        if sc > best_score * 1.01:
            continue
        dl = min(phase1_end - 5.0, time.monotonic() + per_candidate)
        p_best, sc_best = _run_lbfgs_continuation(p, dl, rng, max_wall_time=per_candidate)
        update_best(p_best, sc_best, f"lbfgs_{label}")
        count += 1

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Adam top-K (200s) ---
    phase2_end = min(deadline, t_start + 520.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        dl = phase2_end - 10.0
        p_best, sc_best = _topk_adam(best_seq, dl, rng, lr=0.02, k_top=3)
        update_best(p_best, sc_best, "adam_k3")

        if time.monotonic() < phase2_end - 30:
            dl2 = phase2_end - 10.0
            p_best2, sc_best2 = _topk_adam(best_seq, dl2, rng, lr=0.01, k_top=5)
            update_best(p_best2, sc_best2, "adam_k5")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Frank-Wolfe (100s) ---
    phase3_end = min(deadline, t_start + 620.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        dl = phase3_end - 5.0
        p_best, sc_best = _frank_wolfe(best_seq, dl, rng, n_iters=30000)
        update_best(p_best, sc_best, "frank_wolfe")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Subgradient with momentum (100s) ---
    phase4_end = min(deadline, t_start + 720.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        dl = phase4_end - 5.0
        p_best, sc_best = _adam_subgrad(best_seq, dl, rng, lr=0.1, momentum=0.95, n_iters=20000)
        update_best(p_best, sc_best, "subgrad_mom")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Fine L-BFGS on best (200s) ---
    phase5_end = min(deadline, t_start + 920.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        dl = phase5_end - 10.0
        p_best, sc_best = _run_lbfgs_continuation(best_seq, dl, rng, max_wall_time=200.0, beta_scale=0.5)
        update_best(p_best, sc_best, "lbfgs_fine")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Frequency-guided perturbation (100s) ---
    phase6_end = min(deadline, t_start + 1020.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        dl = phase6_end - 5.0
        p_best, sc_best = _freq_guided_perturbation(best_seq, dl, rng, n_trials=3000)
        update_best(p_best, sc_best, "freq_perturb")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Multi-restart with noise (200s) ---
    phase7_end = min(deadline, t_start + 1220.0)
    if best_seq is not None and time.monotonic() < phase7_end:
        n_restarts = 4
        per_restart = max(30.0, (phase7_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase7_end - 10:
                break
            noise = 0.02 + 0.02 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=len(best_seq))
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            dl = min(phase7_end - 5, time.monotonic() + per_restart)
            p_best, sc_best = _run_lbfgs_continuation(p_start, dl, rng, max_wall_time=per_restart)
            update_best(p_best, sc_best, f"restart{i}")

    print(f"Phase 7 done: best={best_score:.10f}")

    # --- Phase 8: Block coordinate descent (200s) ---
    phase8_end = min(deadline, t_start + 1420.0)
    if best_seq is not None and time.monotonic() < phase8_end:
        dl = phase8_end - 10.0
        p_best, sc_best = _block_coordinate_descent(best_seq, dl, rng, block_size=6, n_trials=5000)
        update_best(p_best, sc_best, "block_cd")

    print(f"Phase 8 done: best={best_score:.10f}")

    # --- Phase 9: Final intensive polishing (remaining) ---
    if best_seq is not None:
        remaining = max(20.0, deadline - time.monotonic())

        # 9a: Fine L-BFGS
        dl = min(deadline - 30, time.monotonic() + remaining * 0.4)
        p_best, sc_best = _run_lbfgs_continuation(best_seq, dl, rng, max_wall_time=remaining * 0.4, beta_scale=0.3)
        update_best(p_best, sc_best, "final_lbfgs")

        # 9b: Adam fine
        if time.monotonic() < deadline - 30:
            dl = min(deadline - 30, time.monotonic() + remaining * 0.3)
            p_best, sc_best = _topk_adam(best_seq, dl, rng, lr=0.005, k_top=2)
            update_best(p_best, sc_best, "final_adam")

        # 9c: Fine local search
        if time.monotonic() < deadline - 15:
            dl = deadline - 10.0
            # Quick coordinate perturbations
            n = len(best_seq)
            a = best_seq.copy()
            best_score_local = best_score
            for trial in range(5000):
                if time.monotonic() > dl:
                    break
                idx = int(rng.integers(0, n))
                a_trial = a.copy()
                a_trial[idx] *= rng.uniform(0.9, 1.1)
                a_trial = np.maximum(a_trial, 1e-15)
                a_trial = a_trial / a_trial.sum()
                sc = _score_norm_fn(a_trial)
                if sc < best_score_local:
                    best_score_local = sc
                    a = a_trial.copy()
                    update_best(a, sc, "local_fine")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_hann(4096)
            s = float(final_seq.sum())
        final_seq = final_seq / s

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_hann(4096)
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n=4096, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
