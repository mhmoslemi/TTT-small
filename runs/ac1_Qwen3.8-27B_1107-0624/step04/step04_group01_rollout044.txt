Wait, there's an issue with building A. Let me double-check the convolution definition.

`np.convolve(a, b)[k] = sum_j a[j] * b[k-j]`

So for the constraint `np.convolve(a, b)[k] <= M`:
`sum_j a[j] * b[k-j] <= M`

The coefficient of `b[j]` in this sum is `a[k-j]` (when `0 <= k-j < n`).

So `A[k, j] = a[k-j]` if `0 <= k-j < n`, else 0.

For a fixed row k:
- j ranges from `max(0, k-(n-1))` to `min(k, n-1)`
- `A[k, j] = a[k-j]`

Let me verify: k=n-1 (center):
- j ranges from `max(0, n-1-(n-1))=0` to `min(n-1, n-1)=n-1`
- `A[n-1, j] = a[n-1-j]` for j = 0, ..., n-1
- So the row is `[a[n-1], a[n-2], ..., a[0]]`
- `conv(a,b)[n-1] = sum_j a[j] * b[n-1-j] = sum_j a[n-1-j] * b[j]` ✓

OK so my formula is correct.

import numpy as np
from scipy.optimize import linprog
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes a is normalized to sum=1, returns 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(c.max())

def _build_conv_rows(a, ks, n):
    """Build rows of convolution matrix for given ks."""
    A = np.zeros((len(ks), n))
    for i, k in enumerate(ks):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            A[i, j_min:j_max+1] = a[k - j_max : k - j_min + 1]
    return A

def _lp_step(a, M, n, max_inners=30, max_constraints=200):
    """
    Solve: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0
    Using constraint generation (separation oracle).
    """
    active_ks = [n - 1]
    b = np.ones(n) / n
    
    for _ in range(max_inners):
        A = _build_conv_rows(a, active_ks, n)
        b_ub = np.full(len(active_ks), M)
        
        try:
            res = linprog(-np.ones(n), A_ub=A, b_ub=b_ub,
                         bounds=[(0, None)]*n, method='highs')
        except Exception:
            return b
        
        if not res.success:
            return b
        
        b = res.x
        b_sum = float(b.sum())
        if b_sum < 1e-15:
            return b
        
        # Separation: check if any constraint is violated
        conv_ab = _full_convolve(a, b)
        max_idx = int(np.argmax(conv_ab))
        if conv_ab[max_idx] <= M * (1.0 + 1e-9):
            return b
        
        if max_idx not in active_ks:
            active_ks.append(max_idx)
            if len(active_ks) > max_constraints:
                # Prune: keep only most violated
                vals = np.array([conv_ab[k] for k in active_ks])
                keep = np.argsort(vals)[-150:]
                active_ks = [active_ks[i] for i in keep]
    
    return b

def _lp_optimize(a_init, deadline, max_iters=200):

    best_m = _score_norm(a)

        b = _lp_step(a, M, n)
        b_sum = float(b.sum())
        
        if b_sum <= 1.0 + 1e-8:
            # No improvement possible
            break
        
        b_norm = b / b_sum

        for t in [0.2, 0.3, 0.5, 0.7, 1.0]:

                best_m = sc

            else:
                break
    
    return best_a, best_m

def _ adam_polish(a_init, deadline, max_iter=100000, lr=0.005):
    """Adam optimization in log-space for final polishing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    def softmax(z):
        z = z - np.max(z)
        e = np.exp(z)
        return e / e.sum()
    
    best_a = a_init.copy()
    best_m = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    beta_start = _score_norm(a_init) * 0.01
    beta_end = _score_norm(a_init) * 0.0001
    n_phases = 5
    betas = np.logspace(np.log10(beta_start), np.log10(beta_end), n_phases)
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        for _ in range(max_iter // n_phases):
            if time.monotonic() > deadline:
                break
            t_step += 1
            
            a = softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            
            # Gradient: 4*n*a*(h-W) where h[m]=sum_i w[i]*a[i-m], W=dot(w,c)
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n-1:2*n-1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1-b1) * grad
            v_buf = b2 * v_buf + (1-b2) * (grad * grad)
            mh = m_buf / (1 - b1**t_step)
            vh = v_buf / (1 - b2**t_step)
            
            z = z - lr * mh / (np.sqrt(vh) + eps)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            
            if t_step % 200 == 0:
                sc = _score_norm(a)
                if sc < best_m:
                    best_m = sc
                    best_a = a.copy()
    
    # Final check
    a_final = softmax(z)
    sc = _score_norm(a_final)
    if sc < best_m:
        best_m = sc
        best_a = a_final.copy()
    
    return best_a, best_m

    elif kind == "sinusoidal":

        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)

    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
