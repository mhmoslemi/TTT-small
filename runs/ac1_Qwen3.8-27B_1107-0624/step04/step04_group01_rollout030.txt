import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    r = irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    """Returns 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _lp_direction(a, M):
    """Solve LP to find direction that maximizes sum(b) subject to (a*b)_k <= M, b >= 0."""
    n = len(a)
    num_constraints = 2 * n - 1
    
    # Build constraint matrix: (a * b)_k <= M for k = 0, ..., 2n-2
    # (a * b)_k = sum_i a_i * b_{k-i}
    # For variable b_j, it appears in constraints k = j, j+1, ..., j+n-1
    # with coefficient a_{k-j}
    
    # Using sparse matrix for efficiency
    from scipy.sparse import csr_matrix
    
    row_indices = []
    col_indices = []
    values = []
    
    for j in range(n):
        for i in range(n):
            k = i + j  # constraint index
            # (a*b)_k has term a_i * b_j when i+j = k
            row_indices.append(k)
            col_indices.append(j)
            values.append(a[i])
    
    A = csr_matrix((values, (row_indices, col_indices)), shape=(num_constraints, n))
    
    # Maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A, b_ub=np.full(num_constraints, M),
                  bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return res.x
    return None

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)

        a = np.sin(np.pi * t) ** 0.5

        return a / a.sum()

Actually, one more important consideration: the LP direction finding gives us a direction $b$ that minimizes $\max_k (a*b)_k$. But the actual objective we care about is $\max_k (b*b)_k$ (after the update). The cross-convolution $a*b$ is related but not the same as the self-convolution $b*b$.

The connection: $(a_{new} * a_{new})_k = (1-t)^2 (a*a)_k + 2t(1-t)(a*b)_k + t^2 (b*b)_k$

So minimizing $(a*b)_k$ helps reduce the cross-term, but we also need $(b*b)_k$ to be small. The LP only controls the cross-term directly.

However, if $b$ is chosen well (minimizing $\max_k (a*b)_k$), and we take a small step $t$, then the self-convolution of $a_{new}$ will be close to that of $a$, with the cross-term being the main change. For small $t$, the Object improvement comes primarily from the cross-term.

For larger $t$, the $t^2 (b*b)_k$ term becomes important. So we should be conservative with the step size, or we should also consider $(b*b)_k$ in the LP.

An alternative: instead of minimizing $\max_k (a*b)_k$, we could minimize $\max_k ((a*b)_k + (b*b)_k/2)$ or some combination. But this makes the LP more complex.

I think the simple approach (minimize $\max_k (a*b)_k$) with a conservative step size should work well, especially for polishing.

    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":

        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

    a1, s1 = _adam_optimize(parent, phase1_end, lr=0.02, max_iter=100000,
                            beta=beta_start, verbose=True)
    update_best(a1, _score(a1), "adam_phase1")

    # ===== PHASE 2: LP-based optimization on n_main (200s - 700s) =====
    print(f"\n=== Phase 2: LP optimization (n={n_main}) ===")
    phase2_end = min(deadline, t_start + 700.0)

    if best_seq is not None:
        a_lp_start = best_seq.copy()
        a_lp_start = a_lp_start / a_lp_start.sum()
        a2, s2 = _lp_optimize(a_lp_start, phase2_end, max_iter=400, verbose=True)
        update_best(a2, _score(a2), "lp_phase2")

        # Continue LP from the result
        if time.monotonic() < phase2_end - 10:
            a2b, s2b = _lp_optimize(a2, phase2_end, max_iter=200, verbose=True)
            update_best(a2b, _score(a2b), "lp_phase2b")

    # ===== PHASE 3: Adam polishing with exact subgradient (700s - 1000s) =====
    print(f"\n=== Phase 3: Adam polish with exact subgradient ===")
    phase3_end = min(deadline, t_start + 1000.0)

    if best_seq is not None and time.monotonic() < phase3_end:
        a3, s3 = _adam_optimize(best_seq, phase3_end, lr=0.005, max_iter=200000,
                                beta=None, verbose=True)
        update_best(a3, _score(a3), "adam_exact_phase3")

    # ===== PHASE 4: Try n=4096 (1000s - 1400s) =====
    print(f"\n=== Phase 4: Try n=4096 ===")
    phase4_end = min(deadline, t_start + 1400.0)

    if best_seq is not None and len(best_seq) < 4096 and time.monotonic() < phase4_end - 30:
        target_n = 4096
        a_interp = _interp_profile(best_seq, target_n)
        sc_interp = _score(a_interp)
        print(f"  Interpolated to n={target_n}: score={sc_interp:.10f}")
        update_best(a_interp, sc_interp, f"interp_n{target_n}")

        # Adam on the interpolated
        if time.monotonic() < phase4_end - 60:
            a4a, s4a = _adam_optimize(a_interp, min(phase4_end - 60, time.monotonic() + 120),
                                      lr=0.01, max_iter=100000, beta=None, verbose=True)
            update_best(a4a, _score(a4a), "adam_n4096")

        # LP on the result
        if time.monotonic() < phase4_end - 20 and best_seq is not None and len(best_seq) == 4096:
            a4b, s4b = _lp_optimize(best_seq, phase4_end, max_iter=50, verbose=True)
            update_best(a4b, _score(a4b), "lp_n4096")

    # ===== PHASE 5: Multi-restart with perturbations (1400s - 1650s) =====
    print(f"\n=== Phase 5: Multi-restart ===")
    phase5_end = min(deadline, t_start + 1650.0)

    if best_seq is not None and time.monotonic() < phase5_end:
        n_restarts = 4
        per_restart = max(5.0, (phase5_end - time.monotonic() - 10) / n_restarts)

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 5:
                break

            sigma = 0.02 + 0.01 * i
            a_pert = _perturb(best_seq, sigma, rng)

            if i % 2 == 0:
                # Adam
                a_r, s_r = _adam_optimize(a_pert, time.monotonic() + per_restart,
                                          lr=0.008, max_iter=50000, beta=None)
            else:
                # LP (fewer iterations)
                a_r, s_r = _lp_optimize(a_pert, time.monotonic() + per_restart,
                                        max_iter=30, verbose=False)
            update_best(a_r, _score(a_r), f"restart{i}")

    # ===== PHASE 6: Final fine polish (1650s - deadline) =====
    print(f"\n=== Phase 6: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # LP polish with very conservative steps
        remaining = deadline - time.monotonic() - 3
        if remaining > 10 and len(best_seq) <= 4096:
            a_fin, s_fin = _lp_optimize(best_seq, time.monotonic() + remaining,
                                        max_iter=100, verbose=True)
            update_best(a_fin, _score(a_fin), "final_lp")
        elif remaining > 5:
            a_fin, s_fin = _adam_optimize(best_seq, time.monotonic() + remaining,
                                          lr=0.002, max_iter=50000, beta=None)
            update_best(a_fin, _score(a_fin), "final_adam")

    # ===== Final validation =====
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
