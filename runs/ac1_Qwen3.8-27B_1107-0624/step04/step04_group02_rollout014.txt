"""

Uses L-BFGS with adaptive logsumexp continuation, active-set balancing,
top-K gradient descent, spectral analysis, multi-restart with structural

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _topk_mean_gradient(a, K=3):
    """Compute gradient of top-K mean of convolution coefficients."""
    n = len(a)
    c = _full_convolve(a, a)
    k_actual = min(K, len(c))
    top_idx = np.argpartition(c, -k_actual)[-k_actual:]
    top_val = c[top_idx].mean()
    grad = np.zeros(n)
    for k_val in top_idx:
        k_val = int(k_val)
        lo = max(0, k_val - (n - 1))
        hi = min(n - 1, k_val)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k_val - hi:k_val - lo + 1][::-1]
    grad /= k_actual
    return grad, float(top_val), c


def _run_lbfgs_smooth(a_init, deadline, rng, beta_list=None, max_wall_time=None):
    """L-BFGS with logsumexp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if beta_list is None:
        beta_list = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max_c0 for b in beta_list]

    if max_wall_time is not None:
        per_level = max(3.0, max_wall_time / len(betas))
    else:
        per_level = 30.0

    for beta in betas:
        if time.monotonic() > deadline - 3.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        def obj(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            log_sum = float(np.log(s_))
            return 2.0 * n * (cm_ + beta * log_sum)

        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            W = float(np.dot(w, c_))
            w_conv = _full_convolve(w, a_[::-1])
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g

        try:
            result = minimize(
                obj, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 2000, 'ftol': 1e-20, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 10000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _active_set_balance(a_init, deadline, rng, K=5, n_iters=500, lr_schedule='exp'):
    """Balance the top-K convolution values using projected gradient."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    for it in range(n_iters):
        if time.monotonic() > deadline - 2.0:
            break

        c = _full_convolve(a, a)
        k_actual = min(K, len(c))
        top_idx = np.argpartition(c, -k_actual)[-k_actual:]
        top_vals = c[top_idx]
        top_mean = top_vals.mean()
        top_std = top_vals.std()

        # Compute gradient of top-K mean
        grad = np.zeros(n)
        for k_val in top_idx:
            k_val = int(k_val)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * a[k_val - hi:k_val - lo + 1][::-1]
        grad /= k_actual

        # Learning rate
        if lr_schedule == 'exp':
            lr = 0.01 * (0.995 ** it)
        else:
            lr = 0.01 / (1 + 0.01 * it)

        # Update in log-space
        z = z - lr * grad / np.maximum(a, 1e-15)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a_new = _softmax_fn(z)

        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()

        # Early stop if balanced
        if top_std < 1e-10 * top_mean:
            break

    return best_a, best_score


def _adam_topk(a_init, deadline, rng, lr=0.01, k_top=3, decay=0.999):
    """Adam optimization with top-K smoothed gradient in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8

    for it in range(100000):
        if time.monotonic() > deadline - 1.0:
            break

        t += 1
        a_curr = _softmax_fn(z)
        grad, _, _ = _topk_mean_gradient(a_curr, k_top)
        grad *= 2.0 * n

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if t % 20 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()

        if t % 500 == 0:
            lr = max(0.0005, lr * decay)

    return best_a, best_score


def _block_local_search(a_init, deadline, rng, n_trials=5000):
    """Local search: perturb blocks and keep improvements."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break

        block_size = int(rng.integers(2, min(30, max(3, n // 32))))
        start = int(rng.integers(0, n - block_size))

        a_trial = a.copy()
        noise_scale = 0.05 + 0.05 * (it / max(1, n_trials))
        perturbation = 1.0 + noise_scale * rng.normal(size=block_size)
        a_trial[start:start + block_size] *= perturbation
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial /= a_trial.sum()

        sc = _score_norm_fn(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
        elif sc < best_score * 1.005 and rng.random() < 0.05:
            a = a_trial.copy()

    return best_a, best_score


def _spectral_refine(a_init, deadline, rng, n_iters=200):
    """Use frequency domain analysis to reduce convolution peaks."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    for it in range(n_iters):
        if time.monotonic() > deadline - 2.0:
            break

        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        cm = float(c.max())

        # Find the argmax lag
        k_max = int(np.argmax(c))

        # Gradient direction to reduce (a*a)[k_max]
        # d(a*a)[k_max]/da[j] = 2*a[k_max-j] for valid j
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        grad = np.zeros(n)
        if lo <= hi:
            grad[lo:hi + 1] = 2.0 * a_curr[k_max - hi:k_max - lo + 1][::-1]

        # Also add a small regularization toward uniform
        grad += 0.001 * (a_curr - 1.0 / n)

        lr = 0.005 * (0.99 ** it)
        z = z - lr * grad / np.maximum(a_curr, 1e-15)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        a_new = _softmax_fn(z)
        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc

    return best_a, best_score


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_profile(n, width):
    t = np.linspace(0.0, 1.0, n)
    u = t - 0.5
    a = np.exp(-u ** 2 / (2.0 * width ** 2))
    return a / a.sum()


def _make_raised_cosine(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5) * alpha))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power_profile(n, p):
    t = np.linspace(0.0, 1.0, n)
    u = np.abs(2.0 * t - 1.0)
    a = (1.0 - u) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal_profile(n, center_sep):
    """Bimodal: two Gaussian bumps."""
    t = np.linspace(0.0, 1.0, n)
    c1 = 0.5 - center_sep / 2.0
    c2 = 0.5 + center_sep / 2.0
    a = np.exp(-((t - c1) ** 2) / 0.02) + np.exp(-((t - c2) ** 2) / 0.02)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_asymmetric_profile(n, asym):
    """Asymmetric profile with linear + nonlinear terms."""
    t = np.linspace(0.0, 1.0, n)
    u = t - 0.5
    a = 1.0 + asym * u + 0.5 * asym ** 2 * (u ** 2 - 1/12)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _blend_profiles(a1, a2, t):
    a = (1 - t) * a1 + t * a2
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization with active-set balancing, spectral refinement,
    adaptive continuation, multi-restart with structural perturbations,
    and multi-length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 4096
        parent = _make_linear_profile(n_parent, 1.0)
        sc_parent = _score_norm_fn(parent)
        print(f"No parent, using linear profile n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "default")

    # === Phase 0: Diverse initializations (40s) ===
    phase0_end = min(deadline, t_start + 40.0)
    candidates = []
    candidates.append((best_score, best_seq.copy(), "parent"))

    n_try = n_parent
    profiles = [
        _make_asymmetric_profile(n_try, 0.5),
        _make_asymmetric_profile(n_try, 1.0),
        _make_asymmetric_profile(n_try, 1.5),
        _make_asymmetric_profile(n_try, -0.5),
        _make_asymmetric_profile(n_try, -1.0),
        _make_bimodal_profile(n_try, 0.3),
        _make_bimodal_profile(n_try, 0.5),
        _make_quadratic_profile(n_try, 0.5, 0.3),
        _make_quadratic_profile(n_try, 1.0, -0.2),
        _make_gaussian_profile(n_try, 0.3),
        _make_raised_cosine(n_try, 1.5),
        _make_raised_cosine(n_try, 2.0),
        _make_power_profile(n_try, 2.0),
        _make_power_profile(n_try, 4.0),
    ]

    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p, f"prof{idx}"))
        update_best(p, sc, f"prof{idx}")

    # Blend parent with some profiles
    for idx, p in enumerate(profiles[:6]):
        if time.monotonic() > phase0_end:
            break
        for t_blend in [0.3, 0.5, 0.7]:
            blended = _blend_profiles(parent, p, t_blend)
            sc = _score_norm_fn(blended)
            candidates.append((sc, blended, f"blend{idx}_{t_blend}"))
            update_best(blended, sc, f"blend{idx}_{t_blend}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}, {len(candidates)} candidates")

    # === Phase 1: L-BFGS continuation on top candidates (500s) ===
    phase1_end = min(deadline, t_start + 540.0)
    n_to_opt = 4
    per_cand = max(60.0, (phase1_end - time.monotonic()) / max(1, n_to_opt))
    count = 0
    seen_labels = set()

    for sc, p, label in candidates:
        if time.monotonic() > phase1_end - 10.0:
            break
        if count >= n_to_opt:
            break
        if sc > best_score * 1.03:
            break
        if label in seen_labels:
            continue
        seen_labels.add(label)

        dl = min(phase1_end - 10.0, time.monotonic() + per_cand)
        betas = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        p_best, sc_best = _run_lbfgs_smooth(p, dl, rng, beta_list=betas,
                                            max_wall_time=per_cand)
        update_best(p_best, sc_best, f"lbfgs_{label}")
        count += 1

    print(f"Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Active set balancing (150s) ===
    phase2_end = min(deadline, t_start + 690.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        dl = phase2_end - 3.0
        for K in [3, 5, 10]:
            if time.monotonic() > phase2_end - 5:
                break
            p_best, sc_best = _active_set_balance(best_seq, time.monotonic() + 40, rng, K=K, n_iters=300)
            update_best(p_best, sc_best, f"balance_K{K}")

    print(f"Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Fine L-BFGS polishing (200s) ===
    phase3_end = min(deadline, t_start + 890.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        remaining = phase3_end - time.monotonic() - 5
        dl = min(phase3_end - 5, t_start + 890.0)
        p_best, sc_best = _run_lbfgs_smooth(
            best_seq, dl, rng,
            beta_list=[0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005],
            max_wall_time=remaining
        )
        update_best(p_best, sc_best, "lbfgs_fine")

        if time.monotonic() < phase3_end - 10:
            p_best2, sc_best2 = _adam_topk(best_seq, phase3_end - 5, rng, lr=0.005, k_top=2)
            update_best(p_best2, sc_best2, "adam_fine")

    print(f"Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Spectral refinement (100s) ===
    phase4_end = min(deadline, t_start + 990.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        p_best, sc_best = _spectral_refine(best_seq, phase4_end - 3, rng, n_iters=500)
        update_best(p_best, sc_best, "spectral")

    print(f"Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Multi-restart with structural perturbations (400s) ===
    phase5_end = min(deadline, t_start + 1390.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        n_restarts = 6
        per_restart = max(30.0, (phase5_end - time.monotonic()) / max(1, n_restarts))

        perturbation_types = [
            'noise', 'structural', 'bimodal', 'asym', 'block', 'freq'
        ]

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 15:
                break

            ptype = perturbation_types[i % len(perturbation_types)]
            n_curr = len(best_seq)

            if ptype == 'noise':
                noise = 0.02 + 0.02 * i
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z - z.mean() + noise * rng.normal(size=n_curr)
                z = np.clip(z, -60, 60) - np.log(np.maximum(best_seq, 1e-15)).mean()
                p_start = _softmax_fn(z - z.mean())
            elif ptype == 'structural':
                # Multiply by a smooth structural perturbation
                t_arr = np.linspace(0, 1, n_curr)
                freq = rng.integers(1, 5)
                phase = rng.uniform(0, 2 * np.pi)
                struct = 1.0 + 0.1 * np.sin(2 * np.pi * freq * t_arr + phase)
                p_start = np.maximum(best_seq * struct, 1e-15)
                p_start /= p_start.sum()
            elif ptype == 'bimodal':
                bim = _make_bimodal_profile(n_curr, rng.uniform(0.2, 0.6))
                p_start = _blend_profiles(best_seq, bim, 0.3)
            elif ptype == 'asym':
                asym = _make_asymmetric_profile(n_curr, rng.uniform(-1, 1))
                p_start = _blend_profiles(best_seq, asym, 0.3)
            elif ptype == 'block':
                p_start = best_seq.copy()
                n_blocks = 4
                block_size = n_curr // n_blocks
                for b in range(n_blocks):
                    scale = rng.uniform(0.7, 1.3)
                    p_start[b * block_size:(b + 1) * block_size] *= scale
                p_start = np.maximum(p_start, 1e-15)
                p_start /= p_start.sum()
            else:  # freq
                # Perturb in frequency domain
                N = _next_pow2(2 * n_curr - 1)
                fa = np.fft.rfft(best_seq, N)
                # Randomly adjust a few frequency bins
                n_freq = len(fa)
                for _ in range(3):
                    fi = int(rng.integers(1, n_freq - 1))
                    fa[fi] *= rng.uniform(0.9, 1.1)
                    if n_freq - 1 - fi < n_freq:
                        fa[n_freq - 1 - fi] = fa[fi].conjugate()
                p_start = np.fft.irfft(fa, N)[:n_curr]
                p_start = np.maximum(p_start, 1e-15)
                p_start /= p_start.sum()

            dl = min(phase5_end - 10, time.monotonic() + per_restart)
            p_best, sc_best = _run_lbfgs_smooth(
                p_start, dl, rng,
                beta_list=[0.05, 0.02, 0.01, 0.005, 0.002, 0.001],
                max_wall_time=per_restart
            )
            update_best(p_best, sc_best, f"restart_{ptype}_{i}")

    print(f"Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Try different n values (300s) ===
    phase6_end = min(deadline, t_start + 1690.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        n_curr = len(best_seq)

        # Try larger n
        for target_n in [n_curr * 2, n_curr * 4]:
            if target_n > 16384 or time.monotonic() > phase6_end - 30:
                break
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{target_n}")

            if time.monotonic() < phase6_end - 40:
                dl = min(phase6_end - 30, time.monotonic() + 120)
                p_best, sc_best = _run_lbfgs_smooth(
                    p_interp, dl, rng,
                    beta_list=[0.05, 0.02, 0.01, 0.005, 0.002],
                    max_wall_time=100
                )
                update_best(p_best, sc_best, f"opt_n{target_n}")

        # Try smaller n (sometimes smaller n is better)
        if n_curr > 2048:
            for target_n in [n_curr // 2, 2048]:
                if time.monotonic() > phase6_end - 20:
                    break
                p_interp = _interp_profile(best_seq, target_n)
                sc_interp = _score_norm_fn(p_interp)
                update_best(p_interp, sc_interp, f"interp_n{target_n}_small")

                if time.monotonic() < phase6_end - 20:
                    dl = min(phase6_end - 15, time.monotonic() + 60)
                    p_best, sc_best = _run_lbfgs_smooth(
                        p_interp, dl, rng,
                        beta_list=[0.05, 0.02, 0.01, 0.005],
                        max_wall_time=50
                    )
                    update_best(p_best, sc_best, f"opt_n{target_n}_small")

    print(f"Phase 6 done: best={best_score:.10f}")

    # === Phase 7: Final intensive polishing (remaining time) ===
    if best_seq is not None:
        remaining = max(10.0, deadline - time.monotonic())
        n_polish = max(1, int(remaining // 30))

        for i in range(n_polish):
            if time.monotonic() > deadline - 5:
                break
            per = remaining / max(1, n_polish - i)

            if i % 3 == 0:
                dl = min(deadline - 5, time.monotonic() + per)
                p_best, sc_best = _run_lbfgs_smooth(
                    best_seq, dl, rng,
                    beta_list=[0.001, 0.0005, 0.0002, 0.0001],
                    max_wall_time=per
                )
                update_best(p_best, sc_best, f"polish_lbfgs_{i}")
            elif i % 3 == 1:
                dl = min(deadline - 5, time.monotonic() + per)
                p_best, sc_best = _active_set_balance(
                    best_seq, dl, rng, K=3, n_iters=int(per * 50), lr_schedule='exp'
                )
                update_best(p_best, sc_best, f"polish_balance_{i}")
            else:
                dl = min(deadline - 5, time.monotonic() + per)
                p_best, sc_best = _block_local_search(
                    best_seq, dl, rng, n_trials=int(per * 30)
                )
                update_best(p_best, sc_best, f"polish_local_{i}")

    # === Final validation and output ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_linear_profile(4096, 1.0)
            s = float(final_seq.sum())
        final_seq = final_seq / s

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        if len(final_seq) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                else:
                    print(f"Evaluator returned inf, using fallback")
                    final_seq = _make_linear_profile(4096, 1.0)
                    sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                    print(f"Fallback evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_linear_profile(4096, 1.0)
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n=4096, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
