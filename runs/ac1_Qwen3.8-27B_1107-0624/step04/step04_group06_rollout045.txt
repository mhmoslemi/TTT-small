    while k < n:
        k <<= 1
    return max(2, k)

    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]

    return c

    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())

    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s

def _lp_solve(a, M):
    """Solve max sum(b) s.t. b>=0, a*b <= M."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

    n = len(a)

    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

        return np.maximum(res.x, 0.0)
    return None


def _lp_solve_sparse(a, M, n_constraints, rng):
    """Stochastic LP: only enforce a subset of constraints."""
    n = len(a)
    all_k = list(range(2 * n - 1))
    # Always include the peak constraint
    c = _full_convolve(a)
    peak_k = int(np.argmax(c))
    selected = [peak_k]
    remaining = [k for k in all_k if k != peak_k]
    n_to_pick = min(n_constraints - 1, len(remaining))
    if n_to_pick > 0:
        selected.extend(rng.choice(remaining, size=n_to_pick, replace=False).tolist())
    # Also include top-k peak constraints
    top_k_list = np.argsort(c)[::-1][:min(20, 2 * n - 1)]
    for tk in top_k_list:
        if int(tk) not in selected:
            selected.append(int(tk))
    selected = list(set(selected))
    n_sel = len(selected)
    rows = []
    cols = []
    vals = []
    for k in selected:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(n_sel, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(n_sel)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_optimize(a, deadline, max_solves=2000, m_shrink=0.9995, t_mix=0.4,
                 use_sparse=False, n_sparse_constraints=300, rng=None):
    """Iterative LP optimization with adaptive M."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0
    patience = 0
    m_history = []

    while time.monotonic() < deadline and n_solves < max_solves:
        t0 = time.monotonic()
        n_solves += 1

        if use_sparse and rng is not None:
            b = _lp_solve_sparse(a, M, n_sparse_constraints, rng)
        else:
            b = _lp_solve_full(a, M)

        if b is None:
            M *= 1.02
            patience += 1
            if patience > 50:
                break
            continue

        s_b = float(b.sum())
        if s_b < 1e-10:
            break

        b_norm = b / s_b
        a_new = (1.0 - t_mix) * a + t_mix * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = _full_convolve(a_new)
        max_new = float(c_new.max())
        sc_new = float(2.0 * n * max_new)

        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
            if patience > 200:
                # Reset: go back to best and retry with slightly different M
                a = best_a.copy()
                c = _full_convolve(a)
                M = float(c.max()) * 0.999
                patience = 0

        a = a_new
        c = c_new
        m_history.append(max_new)

        # Adapt M: shrink if making progress, expand if stuck
        if len(m_history) >= 20:
            recent = m_history[-20:]
            trend = recent[-1] - recent[0]
            if trend < -1e-10:
                M = max_new * m_shrink
            elif trend > 1e-8:
                M = max_new * 0.999
            else:
                M = max_new * 0.9995
        else:
            M = max_new * m_shrink

        # Gradually decrease t_mix for finer adjustments
        t_mix = max(0.1, 0.4 * (0.98 ** n_solves))

        if n_solves % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP[{n_solves}]: score={best_score:.10f}, M={M:.8f},"
                  f" time/iter={elapsed:.3f}s, patience={patience}")

        if time.monotonic() > deadline:
            break

    return best_a, best_score


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _adam_optimize(p0, deadline, rng, lr=0.05, beta_frac=0.08, beta_decay=0.94,
                   no_imp_limit=250):
    """Adam optimization on logit parameterization."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)
        g = _conv_cross_corr(w, p)

        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 300 == 0:
                    lr = max(0.001, lr * 0.9)

            if t_adam % 50 == 0:
                beta = max(1e-5, beta * beta_decay)
                if c.size >= 2:
                    sidx = np.argsort(c)[::-1]
                    gap = c[sidx[0]] - c[sidx[1]]
                    if gap < beta * 0.03:
                        beta = max(1e-5, beta * 0.5)

            limit = max(100, no_imp_limit)
            if no_imp > limit:
                if best_score < 1.51:
                    ns = 0.04
                elif best_score < 1.53:
                    ns = 0.08
                else:
                    ns = 0.15
                z = best_z + ns * rng.normal(size=n)
                z = np.clip(z, -50.0, 50.0)
                z -= float(np.mean(z))
                p = _softmax_from_z(z)
                c = _full_convolve(p)
                maxc = float(c.max())
                beta = max(1e-3, beta_frac * maxc)
                m.fill(0.0)
                v.fill(0.0)
                t_adam = 0
                lr = base_lr
                no_imp = 0
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0

    return best_p, best_score


def _iterate_average(iterates, weights=None):
    """Weighted average of iterates, normalized."""
    if len(iterates) == 0:
        return None
    iters = np.array(iterates)
    if weights is not None:
        weights = np.asarray(weights, dtype=np.float64)
        weights = weights / weights.sum()
        avg = np.sum(iters * weights[:, None], axis=0)
    else:
        avg = np.mean(iters, axis=0)
    avg = np.maximum(avg, 0.0)
    s = avg.sum()
    if s < 1e-30:
        return None
    return avg / s


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x1, x0, p0))


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _peak_flood(a, deadline, eta=0.03, n_iter_max=3000):
    """Multiplicative peak-flooding."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)

    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break

        M = float(c.max())
        score = float(2.0 * n * M)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        threshold = M * 0.98
        hot_mask = c >= threshold
        if not hot_mask.any():
            hot_mask[int(np.argmax(c))] = True

        heat = np.zeros(n)
        hot_indices = np.where(hot_mask)[0]
        for k in hot_indices:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                partner = k - np.arange(i_start, i_end + 1)
                heat[i_start:i_end + 1] += a[partner]

        heat_max = float(heat.max())
        if heat_max < 1e-15:
            break
        heat = heat / heat_max

        factor = 1.0 - eta * heat
        factor = np.maximum(factor, 0.01)
        a = a * factor
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        c = _full_convolve(a)

        if it % 200 == 0 and it > 0:
            if score >= best_score:
                eta = max(0.005, eta * 0.85)
            else:
                eta = min(0.12, eta * 1.03)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            p = np.maximum(p, 0.0)
            s = float(p.sum())
            if s > 0.01:
                p = p / s
            sc = _score_norm(p)
            if np.isfinite(sc) and sc < best_score:
                best_seq = p.copy()
                best_score = float(sc)
                print(f"  BEST: {best_score:.10f} n={p.size} [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    if parent is not None:
        sc_parent = _score_norm(parent)
        print(f"Parent score: {sc_parent:.10f}")
        _update(parent, sc_parent, 'parent')
    else:
        p_init = _make_profile_linear(1024, 2.0)
        _update(p_init, _score_norm(p_init), 'linear_init')
        parent = best_seq

    # --- Phase 1: Quick Adam polish (100s) ---
    phase1_end = t_start + 100.0
    print("Phase 1: Adam polish...")
    p1, sc1 = _adam_optimize(parent, phase1_end, rng, lr=0.04, beta_frac=0.06,
                             beta_decay=0.95, no_imp_limit=200)
    _update(p1, _score_norm(p1), 'adam1')

    # --- Phase 2: Aggressive LP optimization (500s) ---
    phase2_end = t_start + 600.0
    print(f"Phase 2: LP optimization, score={best_score:.10f}...")
    if best_seq is not None and best_seq.size <= 2048:
        # Keep track of iterates for averaging
        iterate_log = [best_seq.copy()]
        a_cur = best_seq.copy()

        for lp_round in range(3):
            per_round = min(150.0, phase2_end - time.monotonic() - 5)
            if per_round < 20:
                break
            a_k, sc_k = _lp_optimize(a_cur, time.monotonic() + per_round,
                                     max_solves=5000, m_shrink=0.9995,
                                     t_mix=0.3, use_sparse=False, rng=rng)
            _update(a_k, _score_norm(a_k), f'lp_round{lp_round}')
            if a_k is not None:
                iterate_log.append(a_k.copy())
                a_cur = a_k

        # Iterate averaging
        if len(iterate_log) >= 2:
            avg = _iterate_average(iterate_log)
            if avg is not None:
                _update(avg, _score_norm(avg), 'lp_avg')

        # Peak flood after LP
        if time.monotonic() < phase2_end - 30 and best_seq is not None:
            per_flood = min(80.0, phase2_end - time.monotonic() - 10)
            if per_flood > 20:
                p_f, sc_f = _peak_flood(best_seq, time.monotonic() + per_flood, eta=0.02)
                _update(p_f, sc_f, 'flood_post_lp')

    # --- Phase 3: Adam smoothing (100s) ---
    phase3_end = t_start + 700.0
    print(f"Phase 3: Adam smoothing, score={best_score:.10f}...")
    if time.monotonic() < phase3_end and best_seq is not None:
        per3 = phase3_end - time.monotonic() - 5
        if per3 > 20:
            p3, sc3 = _adam_optimize(best_seq, time.monotonic() + per3, rng,
                                     lr=0.02, beta_frac=0.04,
                                     beta_decay=0.96, no_imp_limit=150)
            _update(p3, _score_norm(p3), 'adam_smooth')

    # --- Phase 4: Scale up to n=2048 (400s) ---
    phase4_end = t_start + 1100.0
    print(f"Phase 4: Scale to n=2048, score={best_score:.10f}...")
    if time.monotonic() < phase4_end and best_seq is not None:
        n_new = 2048
        if best_seq.size < n_new:
            p_int = _interp_profile(best_seq, n_new)
            if p_int is not None:
                _update(p_int, _score_norm(p_int), f'interp_n{n_new}')

                if time.monotonic() < phase4_end - 30:
                    # LP at n=2048 (use sparse for speed)
                    per_lp = min(200.0, phase4_end - time.monotonic() - 30)
                    if per_lp > 30:
                        p4l, sc4l = _lp_optimize(p_int, time.monotonic() + per_lp,
                                                 max_solves=5000, m_shrink=0.999,
                                                 t_mix=0.3, use_sparse=True,
                                                 n_sparse_constraints=400, rng=rng)
                        _update(p4l, _score_norm(p4l), 'lp_n2048')

                if time.monotonic() < phase4_end - 20 and best_seq is not None and best_seq.size == n_new:
                    per_a = min(100.0, phase4_end - time.monotonic() - 10)
                    if per_a > 20:
                        p4a, sc4a = _adam_optimize(best_seq, time.monotonic() + per_a, rng,
                                                   lr=0.03, beta_frac=0.05,
                                                   beta_decay=0.95, no_imp_limit=150)
                        _update(p4a, _score_norm(p4a), 'adam_n2048')

    # --- Phase 5: Try n=4096 (200s) ---
    phase5_end = t_start + 1300.0
    print(f"Phase 5: Scale to n=4096, score={best_score:.10f}...")
    if time.monotonic() < phase5_end and best_seq is not None:
        n_new2 = 4096
        if best_seq.size < n_new2:
            p_int2 = _interp_profile(best_seq, n_new2)
            if p_int2 is not None:
                sc_int2 = _score_norm(p_int2)
                print(f"  Interp to n={n_new2}: {sc_int2:.10f}")
                _update(p_int2, sc_int2, f'interp_n{n_new2}')

                if time.monotonic() < phase5_end - 20 and best_seq.size == n_new2:
                    per = min(100.0, phase5_end - time.monotonic() - 10)
                    if per > 20:
                        p5, sc5 = _adam_optimize(best_seq, time.monotonic() + per, rng,
                                                 lr=0.03, beta_frac=0.05,
                                                 beta_decay=0.95, no_imp_limit=150)
                        _update(p5, _score_norm(p5), 'adam_n4096')

    # --- Phase 6: Final intensive polish (remaining time) ---
    phase6_end = deadline
    print(f"Phase 6: Final polish, score={best_score:.10f}...")
    if time.monotonic() < phase6_end and best_seq is not None:
        remaining = phase6_end - time.monotonic()

        # Best scale LP
        if remaining > 100 and best_seq.size <= 2048:
            per_lp = remaining * 0.4
            p6l, sc6l = _lp_optimize(best_seq, time.monotonic() + per_lp,
                                     max_solves=5000, m_shrink=0.9998,
                                     t_mix=0.2, use_sparse=False, rng=rng)
            _update(p6l, _score_norm(p6l), 'final_lp')

        # Adam
        if time.monotonic() < phase6_end - 20:
            per_a = min(80.0, phase6_end - time.monotonic() - 10)
            if per_a > 15:
                p6a, sc6a = _adam_optimize(best_seq, time.monotonic() + per_a, rng,
                                           lr=0.01, beta_frac=0.03,
                                           beta_decay=0.97, no_imp_limit=80)
                _update(p6a, _score_norm(p6a), 'final_adam')

        # Peak flood
        if time.monotonic() < phase6_end - 15:
            per_f = min(40.0, phase6_end - time.monotonic() - 5)
            if per_f > 10:
                p6f, sc6f = _peak_flood(best_seq, time.monotonic() + per_f, eta=0.01)
                _update(p6f, sc6f, 'final_flood')

        # One more LP if time
        if time.monotonic() < phase6_end - 30 and best_seq.size <= 2048:
            perLP = phase6_end - time.monotonic() - 10
            if perLP > 30:
                p6l2, sc6l2 = _lp_optimize(best_seq, time.monotonic() + perLP,
                                           max_solves=5000, m_shrink=0.9999,
                                           t_mix=0.15, use_sparse=False, rng=rng)
                _update(p6l2, _score_norm(p6l2), 'final_lp2')

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(1024, 2.0)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile_linear(1024, 2.0)
    print(f"FALLBACK n=1024")
    return [float(x) for x in fb]
