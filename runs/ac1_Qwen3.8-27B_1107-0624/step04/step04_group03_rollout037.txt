    s = e.sum()

    # Column j has non-zeros at rows j to j+n-1 (clipped to 2n-2)
    # with values f[0], f[1], ..., f[n-1] (clipped)
    all_rows = np.empty(n * n, dtype=np.int32)
    all_cols = np.empty(n * n, dtype=np.int32)
    all_data = np.empty(n * n, dtype=np.float64)

        all_cols[idx:idx+length] = j
        all_data[idx:idx+length] = f[:length]
        idx += length

**Key structural insight**: The objective `2n * max_k c[k] / S^2` where `c = a * a` (full convolution).

So `max_k c[k]` is the peak of the inverse Fourier transform of `|A(ω)|^2`.

**Minimizing the peak of the inverse FT**: This is related to the "minimum peak" problem in signal processing. The optimal solution (without the non-negativity constraint on `a`) would make `c[k]` as flat as possible, which means `|A(ω)|^2` should be as "concentrated" as possible in frequency (since a flat time-domain autocorrelation corresponds to a concentrated frequency-domain energy).

        w, cm = _compute_smooth_weights(c, beta)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            continue
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40, 40)
        z = z - z.mean()
        if t % 200 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
                print(f"    Adam t={t}: score={best_score:.10f} "
                      f"(beta={beta:.2e}, total={time.monotonic()-t_start:.0f}s)")
            beta = max(1e-10, beta * 0.85)
            c_new = _full_convolve(a_curr, a_curr)
            max_c = float(c_new.max())
    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _active_set_polish(a_init, deadline):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    lr = 0.003
    momentum = np.zeros(n)
    mom_coeff = 0.85
    t_start = time.monotonic()
    print(f"  Active-set start: score={best_score:.10f}, n={n}")

    for it in range(100000):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        active = c >= cm * 0.998
        if not np.any(active):
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / np.sum(active)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        momentum = mom_coeff * momentum - lr * grad
        z = z + momentum
        z = z - z.mean()
        z = np.clip(z, -40, 40)
        if it % 200 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
                print(f"    AS it={it}: score={best_score:.10f} "
                      f"(total={time.monotonic()-t_start:.0f}s)")
            else:
                lr *= 0.9
    return best_a, best_score


def _make_profile(n, prof_type="cos"):
    t = np.linspace(0.0, 1.0, n)
    if prof_type == "cos":
        a = np.cos(np.pi * t / 2.0) ** 2
    elif prof_type == "lin":
        a = 1.0 + 0.8 * (t - 0.5)
    elif prof_type == "quad":
        u = 2.0 * t - 1.0
        a = 1.0 + 0.5 * u + 0.2 * u ** 2
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
