import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm_fn(a):
    """Score assuming sum(a) = 1, so = 2n * max(conv)"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _exact_grad_max_conv(a, z, top_k=1):
    """
    Compute exact subgradient of 2n * max(conv(a,a)) with respect to z (log-parameters).
    Uses top_k peaks for subgradient averaging.
    Returns gradient with respect to z.
    """
    n = len(a)
    c = _full_convolve(a, a)
    
    # Get top K peak positions
    c_sorted = np.sort(c)[::-1]
    # Find positions of top peaks
    threshold = c_sorted[min(top_k, len(c)-1)] * 0.99  # 99% of the k-th largest
    peak_positions = np.where(c >= threshold)[0]
    
    # Subgradient of max(conv) w.r.t. a[i]:
    # ∂max/∂a[i] = (1/|P|) * sum_{k in P} a[k-i] 
    # where P is the set of peak positions (with valid k-i)
    
    # For each peak position k, contribution to grad_a[i] is a[k-i] (if 0 <= k-i < n)
    grad_a = np.zeros(n, dtype=np.float64)
    for k in peak_positions:
        valid_i = np.arange(max(0, k - n + 1), min(n - 1, k) + 1)
        grad_a[valid_i] += a[k - valid_i]
    grad_a /= len(peak_positions)
    
    # Chain rule: a = softmax(z), so da/dz[i] = a[i] * (delta_ij - a[j])
    # grad_z = a * (grad_a - sum(grad_a * a))
    # Actually: d(softmax(z))_i / d(z)_j = a_i * (delta_ij - a_j)
    # So grad_z[j] = sum_i grad_a[i] * da_i/dz_j = sum_i grad_a[i] * a_i * (delta_ij - a_j)
    # = a_j * (grad_a[j] - sum_i grad_a[i] * a_i)
    
    dot_ga_a = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - dot_ga_a)
    
    # Scale: f = 2n * max(conv) / (sum)^2
    # For sum=1 (our parameterization), f = 2n * max(conv)
    # grad_z of f = 2n * grad_z of max(conv)
    grad_z *= 2.0 * n
    
    return grad_z, c, peak_positions

    """Smooth objective: 2n * logsumexp(conv/beta) * beta (approximately 2n*max for small beta)"""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())

    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    lse = cm + beta * float(np.log(e.sum()))
    return 2.0 * n * lse

    c = _full_convolve(a, a)
    cm = float(c.max())

    # grad_z = a * (dlse_da - sum(dlse_da * a)) * 2n
    dot = float(np.dot(dlse_da, a))
    grad_z = a * (dlse_da - dot) * 2.0 * n
    
    if not np.all(np.isfinite(grad_z)):
        return np.zeros_like(z)
    return grad_z

def _line_search(a, z, grad_z, lr, c_current, max_conv_current):
    """Backtracking line search for mirror descent step."""
    n = len(a)
    best_z = z
    best_c = c_current
    best_score = max_conv_current
    
    for backtrack in range(30):
        trial_z = z - lr * grad_z / (a + 1e-15)  # mirror descent step in a-space
        # Actually, mirror descent with entropic regularization:
        # z_new = z - lr * grad_z (in z-space, this is just gradient descent)
        # But for mirror descent: a_new = a * exp(-lr * grad_a)
        
        # Let's just do gradient descent in z-space with line search
        trial_z = z - lr * grad_z
        trial_a = _softmax_fn(trial_z)
        trial_c = _full_convolve(trial_a, trial_a)
        trial_max = float(trial_c.max())
        
        if trial_max < best_score:
            best_z = trial_z
            best_a = trial_a
            best_c = trial_c
            best_score = trial_max
        
        lr *= 0.5
        
        if lr < 1e-10:
            break
    
    return best_z, best_score

def _mirror_descent_optimize(a_init, deadline, max_iter=100000, lr_init=0.1):

def _adam_smooth(a_init, deadline, beta, lr=0.1, max_iter=200000):
    """Adam optimizer with smooth approximation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _full_convolve(a_init, a_init).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t += 1
        grad = _grad_smooth(z, beta, n)
        if not np.all(np.isfinite(grad)):
            break
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        
        a = _softmax_fn(z)
        sc = _full_convolve(a, a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Decay LR
        if t % 5000 == 0:
            lr *= 0.9
            if lr < 0.001:
                break
    
    return best_a, best_score

def _diverse_init(n, rng, k=20):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Linear ramps (various slopes)
    for slope in [0.5, 1.0, 2.0, 3.0, -0.5, -1.0, -2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    # Parabolic
    for p in [1.0, 2.0, 3.0, 5.0]:
        a = np.maximum(1.0 - (2*t - 1)**p, 1e-10)
        inits.append(a / a.sum())
    
    # Two bumps
    for dist in [0.2, 0.3, 0.4]:
        a = np.exp(-((t - 0.5 + dist)**2) / 0.01 + np.exp(-((t - 0.5 - dist)**2) / 0.01)
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    # Cosine-based
    for freq in [1, 2, 3, 4]:
        a = np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * freq * t), 1e-10)
        inits.append(a / a.sum())
    
    # Random smooth (Gaussian process-like)
    for _ in range(5):
        # Random Fourier series with limited bandwidth
        coeffs = rng.normal(0, 1, min(n, 50))
        a = np.zeros(n)
        for k, c in enumerate(coeffs):
            a += c * np.cos(2 * np.pi * (k+1) * t)
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    return inits

def _multi_scale_optimize(n_fine, a_fine_init, deadline, rng):

    n_coarse = max(64, n_fine // 16)
    t_coarse = np.linspace(0, 1, n_coarse)
    t_fine = np.linspace(0, 1, n_fine)

    a_coarse = np.maximum(a_coarse, 1e-15)

    return a_interp, _full_convolve(a_interp, a_interp).max()

def _peak_targeting(a, deadline, rng, max_rounds=100):
    """
    Directly target the top peaks by finding the optimal local modification.
    Uses the insight that we should reduce contributions to peaks while
    increasing contributions to valleys.
    """
    n = len(a)
    best_a = a.copy()
    best_score = _full_convolve(a, a).max()
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    for round_idx in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        c_max = float(c.max())
        
        # Find top peaks
        top_peaks = np.argsort(c)[::-1][:5]
        
        # Find valleys (minimum values in conv, excluding boundaries)
        min_val = c.min()
        valleys = np.where(c <= min_val + 0.01 * c_max)[0]
        
        improved = False
        
        # For each peak, try to reduce it by modifying a
        for pk in top_peaks[:3]:
            # Contributions to c[pk]: a[i] * a[pk-i] for valid i
            valid_i = np.arange(max(0, pk - n + 1), min(n - 1, pk) + 1)
            if len(valid_i) == 0:
                continue
            
            contrib = a_cur[valid_i] * a_cur[pk - valid_i]
            
            # Find the top contributors
            top_contrib_idx = valid_i[np.argsort(contrib)[::-1][:min(5, len(valid_i))]]
            
            # Try reducing each top contributor
            for idx in top_contrib_idx:
                for delta in [0.005, 0.01, 0.02, 0.05]:
                    z_trial = z.copy()
                    z_trial[idx] -= delta
                    # Compensate by increasing a valley contributor
                    a_trial = _softmax_fn(z_trial)
                    sc_trial = _full_convolve(a_trial, a_trial).max()
                    if sc_trial < best_score:
                        best_score = sc_trial
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if improved:
                break
        
        if not improved:
            # Try random perturbations
            for _ in range(20):
                idx = rng.integers(0, n)
                delta = rng.uniform(-0.05, 0.05)
                z_trial = z.copy()
                z_trial[idx] += delta
                a_trial = _softmax_fn(z_trial)
                sc_trial = _full_convolve(a_trial, a_trial).max()
                if sc_trial < best_score:
                    best_score = sc_trial
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
            
            if not improved:
                # Try a bigger random perturbation
                z_trial = z + 0.02 * rng.normal(size=n)
                z_trial = z_trial - z_trial.mean()
                a_trial = _softmax_fn(z_trial)
                sc_trial = _full_convolve(a_trial, a_trial).max()
                if sc_trial < best_score:
                    best_score = sc_trial
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                else:
                    break  # Stuck
    
    return best_a, best_score

"""

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _norm_score(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _conv_max(a):
    c = _full_convolve(a, a)
    return float(c.max())


def _exact_subgrad_z(a, z, top_k=1):
    """Exact subgradient of 2n*max(conv(a,a)) w.r.t. z (log-params).
    Averages over top_k peak positions for subgradient selection."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros(n), c, 0

    # Determine which peaks to average over
    c_sorted = np.sort(c)[::-1]
    if top_k == 1:
        # Use just the argmax, but check if 2nd is very close
        if len(c_sorted) > 1 and c_sorted[1] > 0.999 * c_sorted[0]:
            top_k = 3
        elif len(c_sorted) > 2 and c_sorted[2] > 0.98 * c_sorted[0]:
            top_k = 5
    # Find positions with value >= 99.5% of max
    threshold = c_sorted[min(top_k - 1, len(c_sorted) - 1)] * 0.995
    peak_pos = np.where(c >= threshold)[0]
    if len(peak_pos) == 0:
        peak_pos = np.array([int(np.argmax(c))])
    n_peaks = len(peak_pos)

    # Subgradient of max(conv) w.r.t. a[i] = (1/n_peaks) * sum_{k in peaks} a[k-i]
    grad_a = np.zeros(n, dtype=np.float64)
    for k in peak_pos:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            valid_i = np.arange(lo, hi + 1)
            grad_a[valid_i] += a[k - valid_i]
    grad_a /= n_peaks

    # Chain rule through softmax: grad_z[j] = a[j] * (grad_a[j] - dot(grad_a, a))
    dot_val = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - dot_val) * (2.0 * n)

    if not np.all(np.isfinite(grad_z)):
        return np.zeros(n), c, c_max
    return grad_z, c, c_max


def _smooth_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    return 2.0 * n * (cm + beta * float(np.log(e.sum())))


def _smooth_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    shifted = (c - cm) / beta
    w = np.exp(shifted)
    w /= w.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    dlse_da = w_conv[n - 1:2 * n - 1]
    dot_val = float(np.dot(dlse_da, a))
    grad_z = a * (dlse_da - dot_val) * (2.0 * n)
    if not np.all(np.isfinite(grad_z)):
        return np.zeros(n)
    return grad_z


def _optimize_exact_grad(a_init, deadline, lr_init=0.05, momentum_init=0.9):

    n = len(a_init)

        z_trial = np.clip(z_trial, -50, 50)

            alpha *= 0.5

            z_trial = z_trial - z_trial.mean()

def _optimize_adam(a_init, deadline, beta, lr=0.05, max_iter=300000):
    """Adam optimizer with smooth max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _conv_max(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        grad = _smooth_grad(z, beta, n)
        if not np.all(np.isfinite(grad)):
            break
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()

        a = _softmax_fn(z)
        sc = _conv_max(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 10000 == 0:
            lr *= 0.85
            if lr < 0.001:
                break

    return best_a, best_score


def _peak_local_search(a, deadline, rng, max_rounds=500):
    """Targeted local search: perturb elements contributing to top peaks."""
    n = len(a)
    best_a = a.copy()
    best_score = _conv_max(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        c_max = float(c.max())

        # Top 3 peak positions
        top_peaks = np.argsort(c)[::-1][:3]

        improved = False
        for pk in top_peaks:

            hi = min(n - 1, pk)
            if lo > hi:
                continue
            valid_i = np.arange(lo, hi + 1)
            contrib = a_cur[valid_i] * a_cur[pk - valid_i]
            # Top 8 contributors
            order = np.argsort(contrib)[::-1][:8]
            for oi in order:
                idx = valid_i[oi]
                for delta in [0.003, 0.006, 0.012, 0.025]:
                    z_t = z.copy()
                    z_t[idx] -= delta
                    a_t = _softmax_fn(z_t)
                    sc_t = _conv_max(a_t)
                    if sc_t < best_score:
                        best_score = sc_t
                        best_a = a_t.copy()
                        z = z_t.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try increasing small elements (redistribute)
            small_idx = np.argsort(a_cur)[:6]
            for si in small_idx:
                for delta in [0.003, 0.006, 0.012]:

                    z_t[si] += delta
                    a_t = _softmax_fn(z_t)
                    sc_t = _conv_max(a_t)

def _generate_inits(n, rng):
    """Generate diverse initial sequences."""
    t = np.linspace(0, 1, n)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Linear
    for s in [0.3, 0.7, 1.0, 1.5, 2.0, -0.3, -0.7, -1.0]:
        a = 1.0 + s * (t - 0.5)
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())

    # Power/cosine
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * (2 * t - 1) / 2), 1e-10) ** p
        inits.append(a / a.sum())

    # Two bumps
    for d in [0.1, 0.15, 0.2, 0.25, 0.3]:
        a = np.exp(-((t - 0.5 - d) ** 2) / 0.005) + np.exp(-((t - 0.5 + d) ** 2) / 0.005)
        a = np.maximum(a, 1e-12)
        inits.append(a / a.sum())

    # Modulated
    for freq in [2, 3, 4, 6]:
        a = np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 1e-10)
        inits.append(a / a.sum())

    # Random smooth
    for _ in range(4):
        bw = rng.integers(3, 15)
        coeffs = rng.normal(0, 1, bw)
        a = np.zeros(n)
        for k, cv in enumerate(coeffs):
            a += cv * np.cos(2 * np.pi * (k + 1) * t + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())

    return inits


def _multi_scale_refine(n_fine, a_fine, deadline, rng):

    n_coarse = max(128, n_fine // 8)
    tc = np.linspace(0, 1, n_coarse)
    tf = np.linspace(0, 1, n_fine)

    a_coarse = np.interp(tc, tf, a_fine)
    a_coarse = np.maximum(a_coarse, 1e-12)
    a_coarse = a_coarse / a_coarse.sum()

    coarse_dl = time.monotonic() + (deadline - time.monotonic()) * 0.25
    a_c, sc_c = _optimize_lbfgs_smooth(a_coarse, coarse_dl, rng)

    a_interp = np.interp(tf, tc, a_c)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp, _conv_max(a_interp)

    scored = sorted(inits, key=lambda x: _conv_max(x))

    n_restarts = min(5, len(scored))
    per_r = (p1_end - time.monotonic()) / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > p1_end - 20:
            break
        a_s = scored[r].copy()
        dl = min(p1_end - 5, time.monotonic() + per_r)
        pb, sb = _optimize_lbfgs_smooth(a_s, dl, rng)
        update_best(pb, sb, f"lbfgs_{r}")

    print(f"  P1 done: best={best_score:.10f}")

    # === Phase 2: Coarse-to-fine (200s) ===
    p2_end = min(deadline, t_start + 700.0)
    print("Phase 2: Coarse-to-fine")

    if time.monotonic() < p2_end and best_seq is not None:
        dl = min(p2_end - 10, time.monotonic() + 190.0)
        a_cf, sc_cf = _multi_scale_refine(len(best_seq), best_seq, dl, rng)
        update_best(a_cf, sc_cf, "coarse_fine")

    print(f"  P2 done: best={best_score:.10f}")

    # === Phase 3: Adam smooth (250s) ===
    p3_end = min(deadline, t_start + 950.0)
    print("Phase 3: Adam smooth")

    if time.monotonic() < p3_end and best_seq is not None:
        c_b = _full_convolve(best_seq, best_seq)
        mc = float(c_b.max())
        betas = [0.005 * mc, 0.002 * mc, 0.0008 * mc]
        for bi, bt in enumerate(betas):
            if time.monotonic() > p3_end - 20:
                break
            dl = min(p3_end - 5, time.monotonic() + (p3_end - time.monotonic()) / (len(betas) - bi))
            pb, sb = _optimize_adam(best_seq, dl, bt, lr=0.04)
            update_best(pb, sb, f"adam_{bi}")

    print(f"  P3 done: best={best_score:.10f}")

    # === Phase 4: Exact gradient mirror descent (350s) ===
    p4_end = min(deadline, t_start + 1300.0)
    print("Phase 4: Exact gradient mirror descent")

    if time.monotonic() < p4_end and best_seq is not None:
        dl = min(p4_end - 10, time.monotonic() + 200.0)
        pb, sb = _optimize_exact_grad(best_seq, dl, lr_init=0.03)
        update_best(pb, sb, "exact_md")

        if time.monotonic() < p4_end - 30:
            dl2 = min(p4_end - 10, time.monotonic() + 140.0)
            pb2, sb2 = _optimize_exact_grad(best_seq, dl2, lr_init=0.01)
            update_best(pb2, sb2, "exact_md_slow")

    print(f"  P4 done: best={best_score:.10f}")

    # === Phase 5: Peak local search (200s) ===
    p5_end = min(deadline, t_start + 1500.0)
    print("Phase 5: Peak local search")

    if time.monotonic() < p5_end and best_seq is not None:
        dl = min(p5_end - 10, time.monotonic() + 190.0)
        pb, sb = _peak_local_search(best_seq, dl, rng)
        update_best(pb, sb, "peak_local")

    print(f"  P5 done: best={best_score:.10f}")

    # === Phase 6: Final L-BFGS tiny beta (remaining) ===
    p6_end = deadline
    print("Phase 6: Final L-BFGS tiny beta")

    if time.monotonic() < p6_end and best_seq is not None:
        c_f = _full_convolve(best_seq, best_seq)
        mc_f = float(c_f.max())
        btiny = 1e-4 * mc_f
        dl = min(p6_end - 15, time.monotonic() + (p6_end - time.monotonic()) * 0.6)
        bl = [btiny, btiny * 0.5, btiny * 0.2]
        pb, sb = _optimize_lbfgs_smooth(best_seq, dl, rng, beta_list=bl)
        update_best(pb, sb, "final_tiny")

        if time.monotonic() < p6_end - 15:
            dl2 = min(p6_end - 10, time.monotonic() + 60.0)
            pb2, sb2 = _optimize_exact_grad(best_seq, dl2, lr_init=0.01)
            update_best(pb2, sb2, "final_exact")

    print(f"  P6 done: best={best_score:.10f}")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = np.ones(256) / 256

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 5000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
            except Exception as e:
                print(f"Eval failed: {e}")

        return [float(x) for x in final_seq]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
