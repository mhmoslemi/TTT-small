from scipy import sparse

def full_convolve(a, b=None):

def build_conv_matrix_rows_cols(n):
    """Precompute row/col indices for the convolution matrix structure."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def make_conv_matrix(a, rows, cols):
    """Create sparse convolution matrix A where (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    vals = a[rows - cols]
    return sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def score_of(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def lp_step(a, precomp_rows, precomp_cols):
    """
    One LP-based optimization step.
    Returns (b_opt, sum_b, success).
    """
    n = len(a)
    c = np.convolve(a, a)
    M = float(c.max())
    
    A = make_conv_matrix(a, precomp_rows, precomp_cols)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

        return None, 0.0, False
    return res.x, float(res.x.sum()), True

def gradient_step(a, lr=0.01):
    """One subgradient step to reduce max(conv(a,a))."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    # Subgradient: g[i] = 2 * a[k_star - i] if valid
    g = np.zeros(n)
    for i in range(n):

        if 0 <= j < n:

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:

    n = len(a)
    z = np.log(np.maximum(a, 1e-18))
    n_freqs = 3
    t = np.arange(n)
    for _ in range(n_freqs):
        freq = rng.integers(1, n // 3 + 1)
        amp = rng.uniform(0.01, 0.08)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    z = np.clip(z, -25.0, 25.0)
    e = np.exp(z)
    return e / e.sum()


def run_lp_optimization(a_init, time_limit, rows, cols, max_iters=5000, base_step=0.15, label=""):
    """Run LP-based optimization from a_init until time_limit."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    n = len(a)

    local_best = a.copy()
    local_best_score = 2.0 * n * np.convolve(a, a).max()
    M_prev = local_best_score / (2.0 * n)
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() > time_limit:
            break

        b_opt, sum_b, M, success = solve_lp_step(a, rows, cols)
        if not success:
            break

        if sum_b <= 1.0 + 1e-8:
            break

        # Adaptive step: larger when sum_b is large
        t_step = min(0.5, max(0.03, base_step * (sum_b - 1.0 + 0.5)))

        b_norm = b_opt / sum_b
        a_new = (1.0 - t_step) * a + t_step * b_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        c_new = np.convolve(a_new, a_new)
        M_new = float(c_new.max())
        sc_new = 2.0 * n * M_new

        if sc_new < local_best_score:
            local_best_score = sc_new
            local_best = a_new.copy()
            stagnant = 0
        else:
            stagnant += 1

        a = a_new

        if it % 100 == 0:
            print(f"    {label} iter={it}, sum_b={sum_b:.6f}, M={M_new:.8f}, score={sc_new:.8f}, "
                  f"stag={stagnant}, t={time.monotonic()-t_start:.1f}s", flush=True)

        if stagnant > 200:
            break

    return local_best, local_best_score


def propose_candidate(seed=42, budget_s=1800):

    LP-based iterative optimization with multi-restart for minimizing
    2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

    - Phase 1: Main LP optimization on parent
    - Phase 2: Multi-restart with perturbations + LP
    - Phase 3: Smoothed gradient polishing
    - Phase 4: Final LP + gradient hybrid
    """
    global t_start
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_a = None
    best_score = np.inf

    def update_best(p, sc=None):
        nonlocal best_a, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = score_of(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_a = p.copy()
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s ***", flush=True)

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        a0 = parent.copy()
        sc0 = score_of(parent)
        print(f"Parent loaded: n={n}, score={sc0:.10f}", flush=True)
        update_best(parent, sc0)
    else:
        n = 2048
        a0 = np.ones(n) / n
        print(f"No parent available, using uniform n={n}", flush=True)
        update_best(a0, 2.0)

    # Precompute matrix structure
    rows, cols = build_matrix_structure(n)
    nnz = len(rows)
    print(f"Matrix structure: shape=({2*n-1},{n}), nnz={nnz}", flush=True)

    # --- Phase 1: Main LP optimization (0 - 300s) ---
    phase1_end = min(deadline, t_start + 300.0)
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: Main LP optimization (target: {phase1_end - t_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    result_a, result_sc = run_lp_optimization(
        a0, phase1_end, rows, cols,
        max_iters=10000, base_step=0.15, label="main"
    )
    update_best(result_a, result_sc)
    print(f"Phase 1 done: best={best_score:.10f}", flush=True)

    # --- Phase 2: Multi-restart with LP (300s - 900s) ---
    phase2_end = min(deadline, t_start + 900.0)
    if time.monotonic() < phase2_end and best_a is not None:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: Multi-restart (target: {phase2_end - t_start:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)

        n_restarts = 8
        per_restart = max(30.0, (phase2_end - time.monotonic() - 30) / n_restarts)

        for i in range(n_restarts):
            if time.monotonic() > phase2_end - 20:
                break

            # Various perturbation strategies
            if i == 0:
                p_start = perturb(best_a, 0.04, rng)
            elif i == 1:
                p_start = perturb_fourier(best_a, rng)
            elif i == 2:
                p_start = perturb(best_a, 0.08, rng)
            elif i == 3:
                p_start = perturb_fourier(best_a, rng)
            elif i == 4:
                # Shift perturbation
                p_start = perturb(best_a, 0.03, rng)
                shift = rng.integers(1, n // 32 + 1)
                p_start = np.roll(p_start, int(shift))
                p_start = p_start / p_start.sum()
            elif i == 5:
                # Mirror + perturb
                p_start = (best_a + best_a[::-1]) / 2.0
                p_start = p_start / p_start.sum()
                p_start = perturb(p_start, 0.05, rng)
            elif i == 6:
                p_start = perturb(best_a, 0.02, rng)
            else:
                p_start = perturb_fourier(best_a, rng)

            sc_init = score_of(p_start)
            print(f"  Restart {i}: initial score={sc_init:.10f}", flush=True)
