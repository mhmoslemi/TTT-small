This is a linear program with 2n-1 constraints (one for each lag k) and n variables (b_0, ..., b_{n-1}).

For n = 4096, this is an LP with 4096 variables and 8191 constraints. This should be solvable with scipy's linprog or cvxpy.

```python
def lap_based_optimization(a_init, deadline, rng, n_outer=100, step_size=0.1):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()  # Normalize
    
    best_a = a.copy()
    best_score = 2.0 * n * _full_convolve(a, a).max()
    
    for outer in range(n_outer):
        if time.monotonic() > deadline - 5:
            break
        
        c = _full_convolve(a, a)
        M = c.max()
        
        # Solve LP: max sum(b) s.t. (a*b)(k) <= M for all k, b >= 0
        # The constraint (a*b)(k) = sum_i a[i]*b[k-i] <= M
        # For each k, this gives a linear constraint on b.
        
        # The constraint matrix A has rows corresponding to k (0 to 2n-2)
        # and columns corresponding to b_j (j = 0 to n-1).
        # A[k, j] = a[i] where i = k-j, if 0 <= k-j < n, else 0.
        
        # A is (2n-1) x n
        # A[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
        # = a[k-j] if max(0, k-n+1) <= j <= min(k, n-1), else 0
        
        # For n=4096, this matrix is 8191 x 4096, which is sparse.
        # Each row has at most n non-zero entries (actually at most min(k+1, n, 2n-1-k) entries).
        
        # Use scipy.optimize.linprog
        from scipy.optimize import linprog
        
        # c (objective) = -1 (we minimize -sum(b))
        c_obj = -np.ones(n)
        
        # A_ub * x <= b_ub
        # A_ub has shape (2n-1, n)
        A_ub = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            lo = max(0, k-n+1)
            hi = min(n-1, k)
            for j in range(lo, hi+1):
                A_ub[k, j] = a[k-j]
        
        b_ub = np.full(2*n-1, M)
        
        # Bounds: b >= 0
        bounds = [(0, None)] * n
        
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if result.success:
            b = result.x
            b_sum = b.sum()
            if b_sum > 0:
                g = b / b_sum  # Normalize g to sum 1
                # Update: a = (1-t)*a + t*g
                t = step_size
                a = (1-t)*a + t*g
                a = np.maximum(a, 0)
                a = a / a.sum()
                
                sc = 2.0 * n * _full_convolve(a, a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"  LP iter {outer}: score={sc:.10f}")
    
    return best_a, best_score
```
