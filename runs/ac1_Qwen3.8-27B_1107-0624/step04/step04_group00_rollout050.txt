import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k *= 2
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _log_param(a):
    return np.log(np.maximum(a, 1e-18))


def _compute_peaks_weights(c, beta):
    cm = float(c.max())
    if cm < 1e-30:
        return np.ones(len(c)) / len(c), cm
    c_shift = (c - cm) / beta
    e = np.exp(c_shift)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _adaptive_beta(c, n_top=5):
    cm = float(c.max())
    if cm < 1e-30:
        return 1e-12
    sorted_c = np.sort(c)[::-1]
    k = min(n_top, len(sorted_c) - 1)
    top_avg = float(np.mean(sorted_c[:k]))
    gap = cm - top_avg
    ratio = gap / cm if cm > 0 else 1.0
    if ratio < 0.02:
        return max(1e-12, 0.05 * cm)
    elif ratio < 0.1:
        return max(1e-12, 0.1 * cm)
    else:
        return max(1e-12, 0.3 * cm)


def _grad_smoothed(a, beta, n):
    c = _full_convolve(a, a)
    w, cm = _compute_peaks_weights(c, beta)
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n)
    return grad


def _obj_smoothed(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    w, cm = _compute_peaks_weights(c, beta)
    return 2.0 * n * (cm + beta * float(np.log(np.sum(np.exp((c - cm) / beta)))))


def _grad_obj(z, beta, n):
    a = _softmax_fn(z)
    return _grad_smoothed(a, beta, n)


def _spectral_shaping(a, n_iter=50, beta_ss=0.3, n_peaks=3):
    """Reduce autocorrelation peaks via frequency-domain modifications."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = _score_norm(a)

    for it in range(n_iter):
        A = np.fft.fft(a, N)
        c = np.fft.ifft(np.abs(A) ** 2, N).real[:2 * n - 1]
        cm = float(c.max())
        if cm < best_sc:
            best_sc = cm
            best_a = a.copy()

        # Find top peaks
        peak_idx = np.argsort(c)[::-1][:n_peaks]
        peak_vals = c[peak_idx]
        total_peak = float(peak_vals.sum())
        if total_peak < 1e-15:
            break

        # Build reduction mask: reduce spectrum where it contributes to peaks
        mask = np.ones(N)
        for pk in peak_idx:
            weight = float(c[pk]) / total_peak
            cos_term = np.cos(2.0 * np.pi * np.arange(N) * pk / N)
            # Reduce where cos is positive (constructive interference at peak)
            mask -= beta_ss * weight * np.maximum(cos_term, 0.0)

        mask = np.maximum(mask, 0.05)

        A_new = A * mask
        a_new = np.fft.ifft(A_new, N).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        sc_new = _score_norm(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller beta
            beta_ss *= 0.8

    return best_a, best_sc


def _extragradient_step(a, beta, n, lr):
    """Extragradient step for the smoothed min-max problem."""
    # First step
    g1 = _grad_smoothed(a, beta, n)
    y = a - lr * g1
    y = np.maximum(y, 1e-18)
    y = y / y.sum()

    # Second step using gradient at y
    g2 = _grad_smoothed(y, beta, n)
    a_new = a - lr * g2
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s


def _subgradient_multi_peak(a, n_active=5, n_iter=2000, lr_init=0.005):
    """Subgradient method handling multiple active peaks."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score_norm(a)
    lr = lr_init

    for it in range(n_iter):
        c = _full_convolve(a, a)
        cm = float(c.max())
        if cm < best_sc:
            best_sc = cm
            best_a = a.copy()

        # Find active peaks (within 5% of max)
        threshold = cm * 0.95
        active = np.where(c >= threshold)[0]
        active = active[active < 2 * n - 1]

        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Weighted subgradient
        active_vals = c[active]
        weights = active_vals / active_vals.sum()

        g = np.zeros(n)
        for w_k, k in zip(weights, active):
            idx = k - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            g[valid] += w_k * a[idx[valid]]

        # Step
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a = a_new / s

        # Decay learning rate
        if it % 100 == 0:
            lr *= 0.98

    return best_a, best_sc


def _coordinate_descent(a, n_rounds=200, n_top=10):
    """Direct coordinate descent on the peak objective."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score_norm(a)
    c = _full_convolve(a, a)

    for rnd in range(n_rounds):
        improved = False
        # Focus on coordinates that contribute to top peaks
        peak_positions = np.argsort(c)[::-1][:3]

        for pk in peak_positions:
            # Find contributing indices
            valid_i = np.arange(max(0, pk - n + 1), min(n - 1, pk) + 1)
            if len(valid_i) == 0:
                continue
            contribs = a[valid_i] * a[pk - valid_i]
            top_idx = valid_i[np.argsort(contribs)[::-1][:n_top]]

            for idx in top_idx:
                # Try reducing this coordinate
                for frac in [0.99, 0.98, 0.97, 0.95, 0.93]:
                    a_trial = a.copy()
                    a_trial[idx] *= frac
                    s = a_trial.sum()
                    if s < 1e-10:
                        break
                    a_trial = a_trial / s
                    sc = _score_norm(a_trial)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_trial.copy()
                        a = a_trial
                        improved = True
                        break

        c = _full_convolve(a, a)
        if not improved and rnd > 20:
            break

    return best_a, best_sc


def _polynomial_profile(n, degree, coeffs):
    """Evaluate polynomial profile."""
    x = np.linspace(-1, 1, n)
    a = np.zeros(n)
    for d, c in enumerate(coeffs):
        a += c * x ** d
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    return a


def _optimize_polynomial(n, degree, deadline, rng):

    best_a = None
    best_sc = np.inf

            coeffs = np.zeros(degree + 1)
            coeffs[0] = 1.0
        elif i == 1:
            coeffs = np.zeros(degree + 1)
            coeffs[0] = 1.0
            coeffs[1] = 0.5
        elif i == 2:
            coeffs = np.zeros(degree + 1)
            coeffs[0] = 1.0
            coeffs[1] = -0.5
        else:
            coeffs = rng.normal(0, 0.3, degree + 1)
            coeffs[0] = 1.0 + coeffs[0] * 0.5

        def poly_obj(c):
            a = _polynomial_profile(n, degree, c)
            return _score_norm(a)

        try:

def _lbfgs_continuation(a_init, deadline, n_top=5):
    """L-BFGS with adaptive continuation schedule."""
    n = len(a_init)
    z = _log_param(a_init)
    z -= z.mean()

    best_a = a_init.copy()
    best_sc = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())

    # Continuation schedule: many small steps
    alphas = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.06, 0.045,
              0.035, 0.028, 0.022, 0.018, 0.014, 0.011, 0.009,
              0.007, 0.006, 0.005, 0.004, 0.003, 0.0025, 0.002,
              0.0015, 0.0012, 0.001, 0.0008, 0.0006, 0.0005]
    betas = [alpha * cm0 for alpha in alphas]

    n_betas = len(betas)
    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break

        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 80.0)
        level_dl = time.monotonic() + time_per

        try:
            res = minimize(_obj_smoothed, z, args=(beta, n),
                           method='L-BFGS-B', jac=_grad_obj,
                           options={'maxiter': 500, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxls': 40,
                                    'maxfun': 3000})
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_new = _softmax_fn(res.x)
                sc_new = _score_norm(a_new)
                if np.isfinite(sc_new) and sc_new < best_sc:
                    best_sc = sc_new
                    best_a = a_new.copy()
                    z = res.x.copy()
                    z -= z.mean()
        except Exception:
            pass

    return best_a, best_sc


def _make_diverse_starts(n, rng):
    """Generate diverse initial profiles."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    starts = []

    # Linear profiles
    for slope in [0.3, 0.5, 0.7, 1.0, 1.5, -0.3, -0.5, -0.7, -1.0]:
        a = 1.0 + slope * u
        a = np.maximum(a, 1e-12)
        starts.append(a / a.sum())

    # Quadratic
    for a1, a2 in [(0.5, 0.3), (0.8, -0.2), (1.0, 0.5), (-0.5, 0.3)]:
        a = 1.0 + a1 * u + a2 * u ** 2
        a = np.maximum(a, 1e-12)
        starts.append(a / a.sum())

    # Exponential
    for rate in [1.0, 2.0, 3.0, -1.0, -2.0]:
        a = np.exp(rate * (t - 0.5))
        starts.append(a / a.sum())

    # Cosine power (asymmetric)
    for p, shift in [(1.0, 0.1), (1.5, 0.2), (2.0, -0.1), (1.0, -0.15)]:
        x = u + shift
        x = np.clip(x, -0.99, 0.99)
        a = np.maximum(np.cos(np.pi * x / 2), 1e-12) ** p
        starts.append(a / a.sum())

    # Random smooth
    for _ in range(3):
        a = 1.0 + 0.3 * np.sin(2 * np.pi * rng.uniform(0.5, 3.0) * t + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 1e-12)
        starts.append(a / a.sum())

    return starts


def _interp_profile(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = len(p)
    if n0 == n_new:
        return p / p.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _finalize_sequence(a):
    """Ensure the sequence is valid for the evaluator."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        # Fallback: uniform
        n = len(a)
        a = np.ones(n) / n
    return a.tolist()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining spectral shaping, adaptive gradient descent,
    extragradient, L-BFGS continuation, and coordinate descent.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        parent = None
        print("No parent available, starting fresh")

    # --- Phase 1: Quick initialization (60s) ---
    phase1_end = min(deadline, t_start + 60.0)
    print("Phase 1: Quick initialization")

    # Choose primary n
    n_primary = 4096
    if parent is not None and len(parent) != 4096:
        n_primary = len(parent)

    # Polynomial optimization
    a_poly, sc_poly = _optimize_polynomial(n_primary, 3, phase1_end, rng)
    if a_poly is not None:
        update_best(a_poly, sc_poly, f"poly3_n{n_primary}")

    # Try a few diverse starts
    starts = _make_diverse_starts(n_primary, rng)
    for i, st in enumerate(starts[:6]):
        if time.monotonic() > phase1_end:
            break
        sc_st = _score_norm(st)
        if sc_st < best_score:
            update_best(st, sc_st, f"start{i}_n{n_primary}")

    # Spectral shaping on best
    if best_seq is not None and time.monotonic() < phase1_end:
        a_ss, sc_ss = _spectral_shaping(best_seq, n_iter=20, beta_ss=0.5)
        update_best(a_ss, sc_ss, f"spectral_shaping_n{n_primary}")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Main gradient descent (500s) ---
    phase2_end = min(deadline, t_start + 560.0)
    print("Phase 2: Main gradient descent with continuation")

    if best_seq is not None:
        n_opt = len(best_seq)
        a_init = best_seq.copy()
        c_init = _full_convolve(a_init, a_init)
        cm_init = float(c_init.max())
        beta_start = 0.3 * cm_init
        beta_end = 0.001 * cm_init

        a_best, sc_best = _gradient_descent_extended(a_init, phase2_end,
                                                     beta_start, beta_end,
                                                     n_restarts=3)
        update_best(a_best, sc_best, f"grad_descent_n{n_opt}")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: L-BFGS fine-tuning (400s) ---
    phase3_end = min(deadline, t_start + 960.0)
    print("Phase 3: L-BFGS fine-tuning")

    if best_seq is not None and time.monotonic() < phase3_end:
        n_opt = len(best_seq)
        a_init = best_seq.copy()

        # L-BFGS continuation
        a_lbfgs, sc_lbfgs = _lbfgs_continuation(a_init, phase3_end, n_top=5)
        update_best(a_lbfgs, sc_lbfgs, f"lbfgs_n{n_opt}")

        # Spectral shaping after L-BFGS
        if time.monotonic() < phase3_end:
            a_ss, sc_ss = _spectral_shaping(best_seq, n_iter=40, beta_ss=0.4)
            update_best(a_ss, sc_ss, f"post_lbfgs_spectral_n{n_opt}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Subgradient + coordinate descent (200s) ---
    phase4_end = min(deadline, t_start + 1160.0)
    print("Phase 4: Subgradient + coordinate descent")

    if best_seq is not None and time.monotonic() < phase4_end:
        n_opt = len(best_seq)

        # Subgradient
        t_sub = min(phase4_end - 50, time.monotonic() + 80)
        a_sub, sc_sub = _subgradient_multi_peak(best_seq, n_active=5,
                                                n_iter=3000, lr_init=0.003)
        update_best(a_sub, sc_sub, f"subgrad_n{n_opt}")

        # Coordinate descent
        if time.monotonic() < phase4_end - 20:
            t_cd = min(phase4_end - 5, time.monotonic() + 60)
            # Coordinate descent is expensive, limit rounds
            a_cd, sc_cd = _coordinate_descent(best_seq, n_rounds=50, n_top=5)
            update_best(a_cd, sc_cd, f"coord_descent_n{n_opt}")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Try n=8192 (200s) ---
    phase5_end = min(deadline, t_start + 1360.0)
    print("Phase 5: Try n=8192")

    if best_seq is not None and time.monotonic() < phase5_end:
        n_curr = len(best_seq)
        if n_curr <= 4096:
            n_new = 8192
            p_interp = _interp_profile(best_seq, n_new)
            sc_interp = _score_norm(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{n_new}")

            if time.monotonic() < phase5_end - 30:
                # Quick gradient descent on the interpolated profile
                a_init = p_interp
                c_init = _full_convolve(a_init, a_init)
                cm_init = float(c_init.max())
                beta_s = 0.2 * cm_init
                beta_e = 0.005 * cm_init
                a_bd, sc_bd = _gradient_descent_extended(a_init, phase5_end,
                                                         beta_s, beta_e, n_restarts=1)
                update_best(a_bd, sc_bd, f"grad_n{n_new}")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Final polish (remaining time) ---
    phase6_end = deadline
    print("Phase 6: Final polish")

    if best_seq is not None and time.monotonic() < phase6_end:
        n_final = len(best_seq)

        # Very fine L-BFGS
        if time.monotonic() < phase6_end - 20:
            c_final = _full_convolve(best_seq, best_seq)
            cm_final = float(c_final.max())
            beta_final = 0.0005 * cm_final
            a_fine, sc_fine = _lbfgs_continuation(best_seq, phase6_end - 10, n_top=3)
            update_best(a_fine, sc_fine, f"final_lbfgs_n{n_final}")

        # Final spectral shaping
        if time.monotonic() < phase6_end - 5:
            a_fs, sc_fs = _spectral_shaping(best_seq, n_iter=50, beta_ss=0.8)
            update_best(a_fs, sc_fs, f"final_spectral_n{n_final}")

        # Final coordinate descent
        if time.monotonic() < phase6_end - 3:
            a_fc, sc_fc = _coordinate_descent(best_seq, n_rounds=30, n_top=8)
            update_best(a_fc, sc_fc, f"final_coord_n{n_final}")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            # Keep as-is (score is scale-invariant)
            pass
        else:
            final_seq = np.ones(len(final_seq)) / len(final_seq)

        sc_final = _score_full(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        # Validate with evaluator if feasible
        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirms: {sc_eval:.10f}")
            except Exception as ex:
                print(f"Evaluator check: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    print("FALLBACK: using uniform sequence")
    fallback_n = 256
    fallback = np.ones(fallback_n) / fallback_n
    print(f"Fallback score: {_score_full(fallback):.10f}")
    return [float(x) for x in fallback]
