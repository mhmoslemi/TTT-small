        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)

"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/sum(a)^2 for non-negative a.
Uses binomial-greedy initialization, L-BFGS log-sum-exp continuation,
LP-based peak balancing, active-set refinement, and Adam polish.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _greedy_init(n):
    """Binomial sequence: b[j]=C(2j,j)/4^j reversed to make a increasing."""
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    a = a / a.sum()
    return a


def _parametric_init(n, etype='asym'):
    t = np.linspace(0, 1, n)
    if etype == 'asym_exp':
        a = np.exp(3.0 * (t - 0.5))
    elif etype == 'asym_pow':
        a = np.maximum(0.05 + 0.95 * t, 0.01) ** 2
    elif etype == 'linear':
        a = 0.1 + 0.9 * t
    elif etype == 'sqrt':
        a = np.sqrt(0.01 + 0.99 * t)
    elif etype == 'exp5':
        a = np.exp(5.0 * (t - 0.5))
    elif etype == 'cubic':
        a = 0.05 + 0.5 * t + 0.4 * t**3
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _softmin_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return float(2.0 * n * lse)


def _softmin_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z)
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs(a_init, deadline, beta_list=None, maxiter_per_beta=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        fracs = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005,
                 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001,
                 0.00005, 0.00003, 0.00001]
        beta_list = [b * m0 for b in fracs]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _softmin_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_softmin_grad,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxls': 100, 'maxfun': 20000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score


def _build_lp_matrix(a, peak_indices, n):
    """Build constraint matrix A where A[k,j] = a[k-j] if valid."""
    K = len(peak_indices)
    A_mat = np.zeros((K, n))
    for row, k in enumerate(peak_indices):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            A_mat[row, i_lo:i_hi + 1] = a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
    return A_mat


def _lp_phase(a_init, deadline, K=30, max_rounds=10):
    """LP-based peak balancing: iteratively solve LP to maximize sum(b)
    subject to (a*b)[k] <= M for active peaks, then blend."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        S = float(best_a.sum())

        # Get active peaks (top K or those within 90% of max)
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K]
        # Also include any peaks within 95% of max
        threshold = 0.95 * M
        additional = np.where(c >= threshold)[0]
        active = np.unique(np.concatenate([active, additional]))
        if len(active) > 200:
            active = active[:200]

        A_mat = _build_lp_matrix(best_a, active, n)
        b_vec = np.full(len(active), M * 1.001)  # small slack

        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 30})
            if res.status != 0:
                break
            b_new = res.x
            sum_b = b_new.sum()
            if sum_b < 1e-15:
                break
            b_scaled = b_new * (S / sum_b)
        except Exception:
            break

        # Try blending
        improved = False
        current_score = 2.0 * n * M / (S * S)
        for t in [1.0, 0.7, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1]:
            a_trial = (1.0 - t) * best_a + t * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            c_trial = _full_convolve(a_trial, a_trial)
            M_trial = float(c_trial.max())
            sc_trial = 2.0 * n * M_trial / (s_trial * s_trial)
            if sc_trial < current_score - 1e-12:
                best_a = a_trial.copy()
                best_score = _score_norm_fn(a_trial)
                current_score = sc_trial
                improved = True
                break

        if not improved:
            break

    return best_a, best_score


def _active_set_phase(a_init, deadline, K=4):
    """Minimize max of top K peaks directly via L-BFGS on smooth surrogate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    for outer in range(100):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        peak_indices = np.argsort(c)[::-1][:K]

        def ao(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            peaks = c_in[peak_indices]
            pm = float(peaks.max())
            if pm < 1e-300:
                return 0.0
            beta = 1e-7 * pm
            sh = (peaks - pm) / beta
            lse = pm + beta * float(np.log(np.sum(np.exp(sh - sh.max()))))
            return float(2.0 * n * lse)

        def ag(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            peaks = c_in[peak_indices]
            pm = float(peaks.max())
            if pm < 1e-300:
                return np.zeros_like(z_in)
            beta = 1e-7 * pm
            sh = (peaks - pm) / beta
            e = np.exp(sh - sh.max())
            w = e / e.sum()

            grad_a = np.zeros(n)
            for idx_k in range(len(peak_indices)):
                k = peak_indices[idx_k]
                wi = w[idx_k]
                i_lo = max(0, k - n + 1)
                i_hi = min(n - 1, k)
                if i_lo <= i_hi:
                    grad_a[i_lo:i_hi + 1] += wi * 2.0 * a_in[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]

            Wa = float(np.dot(grad_a, a_in))
            grad_z = 2.0 * n * a_in * (grad_a - Wa)
            if not np.all(np.isfinite(grad_z)):
                return np.zeros_like(z_in)
            return grad_z

        try:
            result = minimize(ao, z, method='L-BFGS-B', jac=ag,
                              options={'maxiter': 3000, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxls': 50, 'maxfun': 10000})
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
                else:
                    break
        except Exception:
            break

    return best_a, best_score


def _adam_polish(a_init, deadline, lr=0.03, max_iter=300000):
    """Adam optimizer with adaptive beta log-sum-exp surrogate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta = 0.01 * cm0

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0
    check_interval = 50

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-12 * cm, cm0 * 0.01 * (0.97 ** (t // 50)))

        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100, 100)
        z = z - z.mean()

        if t % check_interval == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm_fn(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 30:
                z = z + 0.08 * np.random.randn(n)
                z = z - z.mean()
                no_improve = 0

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: greedy init, L-BFGS continuation, LP peak
    balancing, active-set refinement, and Adam polish. Targets score < 1.5027.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        p = np.asarray(p, dtype=np.float64)
        sc = float(sc)
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = sc
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n}, norm_score={sc_parent:.10f}, actual={_score_fn(parent):.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n = 4096
        parent = _greedy_init(n)
        update_best(parent, _score_norm_fn(parent), "greedy_default")

    print(f"Starting n={n}, initial best={best_score:.10f}")

    # === Phase 1: L-BFGS with aggressive continuation (500s) ===
    print("Phase 1: L-BFGS continuation")
    p1_end = min(deadline, t_start + 500.0)

    a_s = best_seq.copy()
    a_b1, sc1 = _run_lbfgs(a_s, p1_end)
    update_best(a_b1, sc1, "lbfgs_main")

    # Also try from greedy
    if time.monotonic() < p1_end - 60:
        a_g = _greedy_init(n)
        sc_g = _score_norm_fn(a_g)
        update_best(a_g, sc_g, "greedy")
        a_gb, sc_gb = _run_lbfgs(a_g, min(p1_end - 10, time.monotonic() + 100))
        update_best(a_gb, sc_gb, "lbfgs_greedy")

    # Try a few parametric inits
    if time.monotonic() < p1_end - 40:
        for et in ['asym_exp', 'sqrt', 'exp5']:
            a_p = _parametric_init(n, et)
            sc_p = _score_norm_fn(a_p)
            if sc_p < best_score * 1.03:
                a_pb, sc_pb = _run_lbfgs(a_p, min(p1_end - 10, time.monotonic() + 60))
                update_best(a_pb, sc_pb, f"lbfgs_{et}")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: LP-based peak balancing (400s) ===
    print("Phase 2: LP peak balancing")
    p2_end = min(deadline, t_start + 900.0)

    if time.monotonic() < p2_end and best_seq is not None:
        a_lp, sc_lp = _lp_phase(best_seq, p2_end, K=30, max_rounds=15)
        update_best(a_lp, sc_lp, "lp_phase1")

        # Second LP round with refreshed peaks
        if time.monotonic() < p2_end - 60:
            a_lp2, sc_lp2 = _lp_phase(best_seq, p2_end, K=50, max_rounds=10)
            update_best(a_lp2, sc_lp2, "lp_phase2")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Active set refinement (250s) ===
    print("Phase 3: Active set refinement")
    p3_end = min(deadline, t_start + 1150.0)

    if time.monotonic() < p3_end and best_seq is not None:
        for K in [3, 5]:
            if time.monotonic() > p3_end - 30:
                break
            a_act, sc_act = _active_set_phase(best_seq, p3_end, K=K)
            update_best(a_act, sc_act, f"active_K{K}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Final L-BFGS tiny beta (200s) ===
    print("Phase 4: Final L-BFGS tiny beta")
    p4_end = min(deadline, t_start + 1350.0)

    if time.monotonic() < p4_end and best_seq is not None:
        c_b = _full_convolve(best_seq, best_seq)
        m_b = float(c_b.max())
        tiny_betas = [m_b * b for b in [0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]]
        a_f, sc_f = _run_lbfgs(best_seq, p4_end, beta_list=tiny_betas, maxiter_per_beta=8000)
        update_best(a_f, sc_f, "final_tiny_beta")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Adam polish (remaining ~420s) ===
    print("Phase 5: Adam polish")
    if time.monotonic() < deadline - 10 and best_seq is not None:
        a_ad, sc_ad = _adam_polish(best_seq, deadline - 60, lr=0.02)
        update_best(a_ad, sc_ad, "adam_l_r02")

        if time.monotonic() < deadline - 30:
            a_ad2, sc_ad2 = _adam_polish(best_seq, deadline - 10, lr=0.008)
            update_best(a_ad2, sc_ad2, "adam_slow")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Try n*2 if time allows ===
    if time.monotonic() < deadline - 100 and best_seq is not None:
        n_new = min(2 * len(best_seq), 8192)
        if n_new > len(best_seq):
            print(f"Phase 6: Interpolating to n={n_new}")
            a_interp = _interp_profile(best_seq, n_new)
            sc_in = _score_norm_fn(a_interp)
            update_best(a_interp, sc_in, f"interp_n{n_new}")
            if time.monotonic() < deadline - 30:
                a_in_b, sc_in_b = _run_lbfgs(a_interp, deadline - 10, maxiter_per_beta=5000)
                update_best(a_in_b, sc_in_b, f"lbfgs_n{n_new}")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(256)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _greedy_init(512)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(256)
    print(f"FALLBACK: n=256, score={_score_fn(fb):.10f}")
    return [float(x) for x in fb]
