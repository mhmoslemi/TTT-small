Let me now think about the coordinate descent approach more carefully.

If I fix all `a[j]` for `j ≠ i` and vary `a[i]`, the constraint `sum(a) = 1` means `a[i] = 1 - sum_{j≠i} a[j] = const`. So I can't vary `a[i]` independently.

Instead, I can vary `a[i]` and `a[j]` simultaneously (a "2-coordinate" descent):
- `a[i]' = a[i] + δ`
- `a[j]' = a[j] - δ`
- This maintains `sum(a) = 1`.
- We need `a[i] + δ >= 0` and `a[j] - δ >= 0`.

The objective as a function of `δ`:
`f(δ) = max_k conv(a(δ), a(δ))[k]`

This is a 1D non-smooth optimization. We can use:
- Evaluate at several points.
- Use ternary search (if we smooth it).
- Or: find the peak index, then minimize the quadratic in `δ` for that peak.

For a fixed peak index `k*`:
`c[k*](δ) = c[k*] + δ * (d(c[k*])/d(a[i]) - d(c[k*])/d(a[j]))`

where `d(c[k*])/d(a[i]) = a[k*-i]` (if valid) and similarly for j.

So `c[k*](δ)` is linear in `δ` (for `k* ≠ 2i` and `k* ≠ 2j`) or quadratic (if `k* = 2i` or `k* = 2j`).

The max of several linear/quadratic functions is a convex function of `δ` (if all the individual functions are convex). But they're not all convex (some are linear, some are concave quadratic).

So the 2-coordinate problem is non-convex in general. We can still use a simplex method (grid search) over the allowable range of `δ`.

**Practical approach for coordinate descent:**
1. For each pair (i, j), find the best `δ` by evaluating `f(δ)` at a grid of points.
2. Apply the best move.
3. Repeat.

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Adam with backtracking line search, adaptive beta-continuation,
multi-restart diversification, peak-shaving polish, and dual-length search.
Key improvements over previous: backtracking for stability, aggressive peak
shaving in final phase, better diversification, and n=4096 exploration.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z_smooth(a, c, n, beta):
    """Gradient of 2*n*beta*log(sum(exp(c/beta))) w.r.t. z via chain rule.
    
    dF/dz[i] = (dF/da) . (da/dz)
    da/dz[i] = a[i]*(delta_ij - a[j])
    dF/da[i] = 2*n * sum_k w_k * dc_k/da[i]
    dc_k/da[i] = a[k-i] (if valid)
    """
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s

    # dF/da[i] = 2*n * sum_k w[k] * (a[k-i] if 0<=k-i<n else 0)
    # This is a cross-correlation: h[i] = sum_k w[k]*a[k-i]
    # h[i] = (w * a_rev)[i + n - 1] where a_rev = a[::-1]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]  # h[i] = sum_k w[k] * a[k-i] for valid k
    
    # dF/da = 2*n*h
    dF_da = 2.0 * n * h

    # da/dz[i] = a[i]*(delta_ij - a[j])
    # dF/dz[i] = a[i] * (dF_da[i] - sum_j a[j]*dF_da[j])
    W = float(np.dot(a, dF_da))
    grad_z = a * (dF_da - W)
    
    return grad_z


def _score_and_grad_z(z, beta):
    """Compute score (smooth) and gradient w.r.t. z."""
    a = _softmax(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    smooth_score = 2.0 * n * (cm + beta * np.log(np.exp((c - cm) / beta).sum()))
    grad_z = _compute_grad_z_smooth(a, c, n, beta)
    return smooth_score, grad_z, a, c

def _peak_shaving(a, max_iters, deadline):
    """Aggressively reduce the peak of the autocorrelation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_sc = _score_norm(a)

    lr = 0.005
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        cm = float(c.max())
        sc = 2.0 * n * cm
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top peaks
        k_max = int(np.argmax(c))
        
        # Compute gradient at the peak
        # dc[k_max]/da[i] = a[k_max - i] if 0 <= k_max-i < n
        k_valid = np.arange(max(0, k_max - n + 1), min(k_max + 1, n))
        idx_a = k_max - k_valid
        
        # Gradient of c[k_max] w.r.t. a
        grad = np.zeros(n)
        for idx in range(len(k_valid)):
            j = k_valid[idx]
            k = idx_a[idx]
            # c[k_max] includes a[j]*a[k]
            # dc/da[j] includes a[k], dc/da[k] includes a[j]
            if j != k:
                grad[j] += a[k]
                grad[k] += a[j]
            else:
                grad[j] += 2.0 * a[j]
        
        # Project onto tangent space of simplex
        grad = grad - np.dot(a, grad)
        
        # Normalize
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn
        
        # Line search
        improved = False
        for trial_lr in [lr, lr*0.5, lr*0.25, lr*0.125]:
            a_trial = a - trial_lr * grad
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            c_trial = _full_convolve(a_trial, a_trial)
            sc_trial = 2.0 * n * float(c_trial.max())
            if sc_trial < sc:
                a = a_trial
                improved = True
                break
        
        if not improved:
            lr *= 0.7
            if lr < 1e-6:
                break
        else:
            lr = min(0.02, lr * 1.05)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * t
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng):
    """Add Fourier perturbation."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    n_fourier = rng.integers(1, max(2, n // 8))
    for _ in range(n_fourier):
        freq = rng.integers(1, n // 4)
        amp = 0.02 * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        t = np.arange(n)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_shift(a, rng):
    """Circulant shift perturbation."""
    n = len(a)
    shift = rng.integers(1, max(2, n // 20))
    a_new = np.roll(a, shift)
    # Smooth the boundary
    w = len(a) // 50
    if w > 1:
        for i in range(w):
            frac = i / w
            a_new[i] = (1 - frac) * a_new[i] + frac * a[(i - shift) % n]
            a_new[-1 - i] = (1 - frac) * a_new[-1 - i] + frac * a[(-1 - i - shift) % n]
    return a_new / a_new.sum()


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

            0.0005 * max_cb, 0.0002 * max_cb, 0.0001 * max_cb,
            0.00005 * max_cb, 0.00002 * max_cb
        ]
        z_init = np.log(np.maximum(best_seq, 1e-15))
        z_init = z_init - z_init.mean()
        remaining = phase6_end - time.monotonic() - 5
        p_best, sc_best, _ = _optimize_adam_bt(
            z_init, time.monotonic() + remaining, beta_fine,
            max_iter=200000, lr=0.005
        )
        update_best(p_best, _score(p_best))

        # Peak shave again
        if time.monotonic() < phase6_end - 5:
            p_sh, sc_sh = _peak_shaving(best_seq, 20000, time.monotonic() + (phase6_end - time.monotonic() - 3))
            update_best(p_sh, _score(p_sh))

    print(f"Phase 6: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase 7: Last-ditch diversification + final polish (1750s-deadline)
    if best_seq is not None and time.monotonic() < deadline - 10:
        # Try a few more perturbations
        for _ in range(3):
            if time.monotonic() > deadline - 10:
                break
            p_try = _perturb(best_seq, 0.01, rng)
            # Quick peak shave
            p_try, _ = _peak_shaving(p_try, 5000, time.monotonic() + 20)
            update_best(p_try, _score(p_try))

        # Final peak shave
        if time.monotonic() < deadline - 5:
            p_final, _ = _peak_shaving(best_seq, 10000, deadline - 3)
            update_best(p_final, _score(p_final))

    print(f"Phase 7: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
