"""
LP fixed-point with golden section line search, dual-guided perturbation,
palindromic symmetry, and multi-scale escalation for minimizing
2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _cross_correlate(w, a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]


def _build_conv_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_primal(a, M, n):
    A = _build_conv_matrix(n, a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 90})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_dual(a, n):
    A = _build_conv_matrix(n, a)
    c_obj = np.ones(2 * n - 1)
    A_ub = -A.T
    b_ub = -np.ones(n)
    bounds = [(0, None)] * (2 * n - 1)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 120})
        if res.success:
            y_star = np.maximum(res.x, 0.0)
            dual_obj = float(y_star.sum())
            return dual_obj, y_star
    except Exception:
        pass
    return None, None


def _golden_search(a, b_dir, deadline, max_evals=25):
    phi = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    a_pt = hi - phi * (hi - lo)
    b_pt = lo + phi * (hi - lo)

    def eval_t(t):
        x = (1.0 - t) * a + t * b_dir
        x = np.maximum(x, 1e-15)
        x = x / x.sum()
        return _score_norm(x)

    fa = eval_t(a_pt)
    fb = eval_t(b_pt)
    n_evals = 2

    while (hi - lo) > 0.005 and n_evals < max_evals:
        if time.monotonic() > deadline:
            break
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - phi * (hi - lo)
            fa = eval_t(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + phi * (hi - lo)
            fb = eval_t(b_pt)
        n_evals += 1

    best_t = (a_pt + b_pt) / 2.0
    return best_t, min(fa, fb)


def _lp_fixed_point(a, deadline, max_iters=200, use_golden=True, min_step=0.05):

            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b = b / s_b

        if use_golden:

        else:
            t_best = 0.5
            x = np.maximum((1 - t_best) * a + t_best * b, 1e-15)
            x = x / x.sum()
            sc_est = _score_norm(x)

        x = (1.0 - t_best) * a + t_best * b
        x = np.maximum(x, 1e-15)
        x = x / x.sum()
        sc = _score_norm(x)

        if sc < best_score - 1e-10:

def _adam_smooth_max(a, deadline, rng, lr=0.05, beta_init=None,
                     beta_decay=0.95, no_imp_limit=150):

    a = a / a.sum()

    z = np.log(a)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = _full_convolve(a)
    maxc = float(c.max())
    beta = beta_init if beta_init is not None else 0.1 * maxc

    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0

    while time.monotonic() < deadline:
        # Smooth max weights
        x = (c - maxc) / beta
        ex = np.exp(x)
        s = float(ex.sum())
        if not np.isfinite(s) or s <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
            smooth_val = maxc
        else:
            w = ex / s
            smooth_val = maxc + beta * float(np.log(s))

        g = _cross_correlate(w, a)
        gz = a * (g - float(np.dot(a, g)))
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()

        a = np.exp(z - z.max())
        a = a / a.sum()
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        c = _full_convolve(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.003, lr * 0.9)

        if t_adam % 40 == 0:
            beta = max(1e-5, beta * beta_decay)

        if no_imp > no_imp_limit:
            ns = 0.1
            z = best_z + ns * rng.normal(size=n)

            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a)
            beta = 0.1 * c.max()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = 0.05
            no_imp = 0

    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()

def _make_profile(n, ptype, p1=1.0, p2=0.0, p3=0.0):
    if ptype == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (t - 0.5)
    elif ptype == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (2 * t - 1) ** 2
    elif ptype == 'cubic':
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + p1 * u + p2 * u ** 3
    elif ptype == 'exp':
        t = np.arange(n)
        a = np.exp(-p1 * t / n)
    elif ptype == 'cosine':
        t = np.arange(n)
        a = 1.0 + p1 * np.cos(2 * np.pi * p2 * t / n)
    elif ptype == 'sine':
        t = np.arange(n)
        a = 1.0 + p1 * np.sin(2 * np.pi * p2 * t / n)
    elif ptype == 'parabola':
        t = np.linspace(0, 1, n)
        a = (t - 0.5) ** 2 + p1
    elif ptype == 'gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / p1) ** 2)
    elif ptype == 'U_shape':
        t = np.linspace(0, 1, n)
        a = (2 * t - 1) ** 2 + p1
    elif ptype == 'edge_heavy':
        t = np.arange(n) / max(1, n - 1)
        a = 1.0 / (0.01 + abs(t - 0.5) * 2.0) ** p1
    elif ptype == 'bell':
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * (t - 0.5)) ** 2 + p1
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new


def _make_palindromic(a):
    """Make sequence palindromic by averaging with its reverse."""
    n = len(a)
    a_pal = 0.5 * (a + a[::-1])
    a_pal = np.maximum(a_pal, 1e-15)
    return a_pal / a_pal.sum()


def _peak_shave(a, deadline, max_iters=200):

        # dc[peak]/da[m] = a[peak-m] + a[m] (if peak-m != m and valid)

        grad = np.zeros(n)
        for m in range(n):

            km = peak_idx - m

            if km != m and 0 <= m < n:
                val += a[m]
            grad[m] = val

        # But maintain non-negativity and sum=1
        step_lr = 0.01

        a_new = np.maximum(a_new, 1e-15)

        else:
            step_lr = max(0.0005, step_lr * 0.7)
            if step_lr < 0.0005:

    return best_a, best_score

    A_active = np.zeros((len(active_lags), n_actual))

    try:
        ATA = A_active @ A_active.T
        ATA_inv = np.linalg.pinv(ATA)
        proj = A_active.T @ (ATA_inv @ (A_active @ grad_peak))
        d = grad_peak - proj
        d = d - d.mean()  # maintain sum=0
    except Exception:
        d = grad_peak - grad_peak.mean()

    d_norm = np.linalg.norm(d)
    if d_norm < 1e-10:
        return a, _score_norm(a)
    d = d / d_norm

    # Try small steps in direction -d (to reduce peak)
    best_a = a_work.copy()
    best_score = _score_norm(a_work)

    for step in [0.001, 0.005, 0.01, 0.02]:
        a_try = a_work - step * d
        a_try = np.maximum(a_try, 1e-15)
        a_try = a_try / a_try.sum()
        sc = _score_norm(a_try)
        if sc < best_score:
            best_score = sc
            best_a = a_try

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP fixed-point with golden section search,
    Adam smooth-max, peak shaving, dual-guided perturbation, and multi-scale.
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    # === Phase 1: Generate initial candidates ===
    print("Phase 1: Generating candidates...")
    candidates = []

    n_sizes = [256, 512, 1024]
    profiles = [
        ('lin1.0', 'linear', 1.0, 0, 0),
        ('lin1.5', 'linear', 1.5, 0, 0),
        ('lin2.0', 'linear', 2.0, 0, 0),
        ('lin2.5', 'linear', 2.5, 0, 0),
        ('lin3.0', 'linear', 3.0, 0, 0),
        ('quad1', 'quadratic', 1.0, 0, 0),
        ('quad2', 'quadratic', 2.0, 0, 0),
        ('cubic1', 'cubic', 1.5, -1.0, 0),
        ('cubic2', 'cubic', 2.0, -1.5, 0),
        ('exp0.5', 'exp', 0.5, 0, 0),
        ('exp1.0', 'exp', 1.0, 0, 0),
        ('exp1.5', 'exp', 1.5, 0, 0),
        ('exp2.0', 'exp', 2.0, 0, 0),
        ('cos1_08', 'cosine', 0.8, 1, 0),
        ('cos2_06', 'cosine', 0.6, 2, 0),
        ('para05', 'parabola', 0.5, 0, 0),
        ('para10', 'parabola', 1.0, 0, 0),
        ('g03', 'gauss', 0.3, 0, 0),
        ('U03', 'U_shape', 0.3, 0, 0),
        ('U05', 'U_shape', 0.5, 0, 0),
        ('edge1', 'edge_heavy', 1.0, 0, 0),
        ('edge15', 'edge_heavy', 1.5, 0, 0),
        ('bell05', 'bell', 0.5, 0, 0),
        ('bell10', 'bell', 1.0, 0, 0),
    ]

    for n in n_sizes:
        for (label, ptype, p1, p2, p3) in profiles:
            p = _make_profile(n, ptype, p1, p2, p3)
            # Also try palindromic version
            p_pal = _make_palindromic(p)
            if np.sum(np.abs(p_pal - p)) > 1e-6:
                sc_pal = _score_norm(p_pal)
                candidates.append((sc_pal, p_pal, n, label + '_pal'))
            sc = _score_norm(p)
            candidates.append((sc, p, n, label))

    if parent is not None:
        for n in [512, 1024, 2048]:
            pi = _interp(parent, n)
            sc = _score_norm(pi)
            candidates.append((sc, pi, n, 'parent'))
        if 0 < pn <= 2048:
            sc = _score_norm(parent)
            candidates.append((sc, parent.copy(), pn, 'parent_exact'))
            p_pal = _make_palindromic(parent)
            sc_pal = _score_norm(p_pal)
            candidates.append((sc_pal, p_pal, pn, 'parent_pal'))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, best: {candidates[0][0]:.8f} n={candidates[0][2]} [{candidates[0][3]}]")

    if candidates[0][1] is not None and candidates[0][0] < best_score:
        best_seq = np.asarray(candidates[0][1], dtype=np.float64).copy()
        best_score = float(candidates[0][0])
        print(f"  BEST: {best_score:.10f} n={best_seq.size} [init]")

    # === Phase 2: LP fixed-point on top candidates ===
    print("Phase 2: LP fixed-point optimization...")
    phase2_end = t_start + 600.0
    polished = []

    # Sort by (n, score) - prefer smaller n for LP speed
    candidates_lp = sorted(candidates, key=lambda x: (x[2], x[0]))

    count = 0
    for (sc, p, n, label) in candidates_lp:
        if time.monotonic() > phase2_end - 30 or count >= 8:
            break
        if n > 1024:
            continue

        time_alloc = min(120.0, phase2_end - time.monotonic() - 10)
        if time_alloc < 30:
            break
        per_end = time.monotonic() + time_alloc

        a_lp, sc_lp = _lp_fixed_point(p, per_end, max_iters=200, use_golden=True)
        if sc_lp < best_score:
            best_seq = np.asarray(a_lp, dtype=np.float64).copy()
            best_score = float(sc_lp)
            print(f"  BEST: {best_score:.10f} n={a_lp.size} [LP_{label}_n{n}]")
        polished.append((sc_lp, a_lp, n, label + '_lp'))
        count += 1

    polished.sort(key=lambda x: x[0])
    print(f"  After Phase 2: best={best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    # === Phase 3: Adam refinement on top LP results ===
    print("Phase 3: Adam smooth-max refinement...")
    phase3_end = t_start + 950.0

    for (sc, p, n, label) in polished[:3]:
        if time.monotonic() > phase3_end - 20:
            break
        per = min(80.0, phase3_end - time.monotonic() - 10)
        if per < 20:
            break
        a_ad, sc_ad = _adam_smooth_max(p, time.monotonic() + per, rng,
                                       lr=0.04, beta_decay=0.95, no_imp_limit=200)
        if sc_ad < best_score:
            best_seq = np.asarray(a_ad, dtype=np.float64).copy()
            best_score = float(sc_ad)
            print(f"  BEST: {best_score:.10f} n={a_ad.size} [Adam_{label}]")

        # LP after Adam
        if n <= 1024 and time.monotonic() < phase3_end - 20:
            per2 = min(60.0, phase3_end - time.monotonic() - 10)
            if per2 > 20:
                a_lp2, sc_lp2 = _lp_fixed_point(a_ad, time.monotonic() + per2,
                                                max_iters=100, use_golden=True)
                if sc_lp2 < best_score:
                    best_seq = np.asarray(a_lp2, dtype=np.float64).copy()
                    best_score = float(sc_lp2)
                    print(f"  BEST: {best_score:.10f} n={a_lp2.size} [LP_Adam_{label}]")

    # === Phase 4: Peak shaving ===
    print("Phase 4: Peak shaving...")
    phase4_end = t_start + 1150.0

    if best_seq is not None and time.monotonic() < phase4_end:
        per = phase4_end - time.monotonic() - 10
        if per > 15:
            a_ps, sc_ps = _peak_shave(best_seq, time.monotonic() + per, max_iters=500)
            if sc_ps < best_score:
                best_seq = a_ps.copy()
                best_score = sc_ps
                print(f"  BEST: {best_score:.10f} n={a_ps.size} [peak_shave]")

            # LP after peak shaving
            if best_seq.size <= 1024 and time.monotonic() < phase4_end - 15:
                per2 = phase4_end - time.monotonic() - 10
                if per2 > 20:
                    a_ps2, sc_ps2 = _lp_fixed_point(a_ps, time.monotonic() + per2,
                                                    max_iters=100, use_golden=True)
                    if sc_ps2 < best_score:
                        best_seq = a_ps2.copy()
                        best_score = sc_ps2
                        print(f"  BEST: {best_score:.10f} n={a_ps2.size} [LP_peak_shave]")

    # === Phase 5: Dual-guided perturbation ===
    print("Phase 5: Dual-guided perturbation...")
    phase5_end = t_start + 1350.0

    if best_seq is not None and time.monotonic() < phase5_end:
        per = phase5_end - time.monotonic() - 10
        if per > 30:
            a_dg, sc_dg = _dual_guided_perturb(best_seq, time.monotonic() + per)
            if sc_dg < best_score:
                best_seq = a_dg.copy()
                best_score = sc_dg
                print(f"  BEST: {best_score:.10f} n={a_dg.size} [dual_guided]")

            # LP after dual perturbation
            if best_seq.size <= 1024 and time.monotonic() < phase5_end - 15:
                per2 = phase5_end - time.monotonic() - 10
                if per2 > 20:
                    a_dg2, sc_dg2 = _lp_fixed_point(a_dg, time.monotonic() + per2,
                                                    max_iters=100, use_golden=True)
                    if sc_dg2 < best_score:
                        best_seq = a_dg2.copy()
                        best_score = sc_dg2
                        print(f"  BEST: {best_score:.10f} n={a_dg2.size} [LP_dual_guided]")

    # === Phase 6: Multi-scale escalation ===
    print("Phase 6: Multi-scale...")
    phase6_end = t_start + 1550.0

    if best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024]
        elif n_cur <= 1024:
            target_ns = [1536, 2048]
        else:
            target_ns = []

        for tn in target_ns:
            if time.monotonic() > phase6_end - 60:
                break
            p_int = _interp(best_seq, tn)
            sc_int = _score_norm(p_int)
            print(f"  Interp to n={tn}: {sc_int:.8f}")
            if sc_int < best_score:
                best_seq = p_int.copy()
                best_score = sc_int
                print(f"  BEST: {best_score:.10f} n={tn} [interp]")

            if time.monotonic() < phase6_end - 40:
                per = min(60.0, phase6_end - time.monotonic() - 20)
                if per > 20:
                    a_e, sc_e = _adam_smooth_max(p_int, time.monotonic() + per, rng,
                                                 lr=0.03, beta_decay=0.96, no_imp_limit=200)
                    if sc_e < best_score:
                        best_seq = a_e.copy()
                        best_score = sc_e
                        print(f"  BEST: {best_score:.10f} n={tn} [Adam_esc]")

            if tn <= 1024 and time.monotonic() < phase6_end - 30:
                per2 = min(40.0, phase6_end - time.monotonic() - 15)
                if per2 > 15:
                    a_le, sc_le = _lp_fixed_point(p_int, time.monotonic() + per2,
                                                  max_iters=50, use_golden=True)
                    if sc_le < best_score:
                        best_seq = a_le.copy()
                        best_score = sc_le
                        print(f"  BEST: {best_score:.10f} n={tn} [LP_esc]")

    # === Phase 7: Random restarts ===
    print("Phase 7: Random restarts...")
    phase7_end = t_start + 1700.0

    if best_seq is not None and time.monotonic() < phase7_end:
        n_best = best_seq.size
        n_restarts = 4
        per_r = max(10.0, (phase7_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase7_end - 5:
                break
            per = min(per_r, phase7_end - time.monotonic() - 5)
            if per < 10:
                break

            # Perturb in different ways
            if i % 2 == 0:
                z = np.log(np.maximum(best_seq, 1e-12))
                z -= z.mean()
                z = z + 0.08 * rng.normal(size=n_best)
            else:
                # Frequency domain perturbation
                N = _next_pow2(n_best)
                A = np.fft.rfft(best_seq, N)
                A = A * (1 + 0.05 * rng.normal(size=N // 2 + 1))
                a_pert = np.fft.irfft(A, N)[:n_best]
                a_pert = np.maximum(a_pert, 1e-12)
                a_pert = a_pert / a_pert.sum()
                z = np.log(a_pert)
                z -= z.mean()

            z = np.clip(z, -50, 50)
            z -= z.mean()
            a_s = np.exp(z - z.max())
            a_s = a_s / a_s.sum()

            a_r, sc_r = _adam_smooth_max(a_s, time.monotonic() + min(40, per), rng,
                                         lr=0.04, beta_decay=0.94, no_imp_limit=100)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  BEST: {best_score:.10f} n={n_best} [restart{i}]")

            if n_best <= 1024 and time.monotonic() < phase7_end - 10:
                per2 = min(20.0, phase7_end - time.monotonic() - 5)
                if per2 > 10:
                    a_rl, sc_rl = _lp_fixed_point(a_r, time.monotonic() + per2,
                                                  max_iters=30, use_golden=True)
                    if sc_rl < best_score:
                        best_seq = a_rl.copy()
                        best_score = sc_rl
                        print(f"  BEST: {best_score:.10f} n={n_best} [LP_restart{i}]")

    # === Phase 8: Final extended LP ===
    print("Phase 8: Final LP polish...")
    if best_seq is not None and time.monotonic() < deadline - 15:
        per = deadline - time.monotonic() - 15
        if per > 30:
            n_final = min(best_seq.size, 1024)
            if best_seq.size != n_final:
                a_f = _interp(best_seq, n_final)
            else:
                a_f = best_seq.copy()

            if n_final <= 1024:
                a_fp, sc_fp = _lp_fixed_point(a_f, time.monotonic() + per,
                                              max_iters=500, use_golden=True)
                if sc_fp < best_score:
                    best_seq = a_fp.copy()
                    best_score = sc_fp
                    print(f"  BEST: {best_score:.10f} n={a_fp.size} [final_LP]")
            else:
                a_fp, sc_fp = _adam_smooth_max(a_f, time.monotonic() + per, rng,
                                               lr=0.02, beta_decay=0.97, no_imp_limit=200)
                if sc_fp < best_score:
                    best_seq = a_fp.copy()
                    best_score = sc_fp
                    print(f"  BEST: {best_score:.10f} n={a_fp.size} [final_Adam]")

    # Final validation
    print(f"\nFinal score (norm): {best_score:.10f}")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile(512, 'linear', 1.5, 0, 0)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile(512, 'linear', 1.5, 0, 0)
    print("FALLBACK n=512")
    return [float(x) for x in fb]
