"""
Optimized step function search using LP-based alternating minimization with
dual-guided perturbations, smooth-max Adam polishing, and multi-scale escalation.
Key innovation: uses min-peak LP (min M s.t. a*b <= M, sum(b)=1) instead of
max-sum LP, enabling more direct optimization of the self-convolution peak.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, hstack
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        c = np.convolve(a, a)
    else:
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    mval = float(c.max())
    if beta < 1e-13:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-9
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 0.0))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-14):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -80.0, 80.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-15, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _build_conv_matrix_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_min_peak(a):
    """Solve: min M s.t. (a*b)(k) <= M for all k, sum(b)=1, b>=0.
    Returns (M_opt, b_opt) or (None, None)."""
    n = len(a)
    A_conv = _build_conv_matrix_sparse(a)
    # A_ub = [A_conv | -1] of shape (2n-1, n+1)
    neg_col = csr_matrix(np.ones((2 * n - 1, 1))) * (-1.0)
    A_ub = hstack([A_conv, neg_col]).tocsr()
    b_ub = np.zeros(2 * n - 1)
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    # Objective: min M
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        M_opt = float(res.x[n])
        b_opt = np.maximum(res.x[:n], 0.0)
        s = float(b_opt.sum())
        if s > 1e-10:
            b_opt = b_opt / s
        return M_opt, b_opt
    return None, None


def _lp_max_sum(a, M):
    """Solve: max sum(b) s.t. (a*b)(k) <= M, b>=0. Returns b_opt or None."""
    n = len(a)
    A_conv = _build_conv_matrix_sparse(a)
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_conv, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_alternate(a, b, deadline, max_iters=200, verbose=True):
    """Alternating LP: minimize max(a*b) by alternating LP solves.
    Returns best (a, b, max_ab_score)."""
    n = len(a)
    best_a = a.copy()
    best_b = b.copy()
    best_score = _score_norm(a)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        # Solve LP for b given a
        M_b, b_new = _lp_min_peak(a)
        if b_new is None:
            break
        # Update a using b_new
        a_try = 0.5 * a + 0.5 * b_new
        a_try = np.maximum(a_try, 1e-15)
        a_try = a_try / a_try.sum()
        # Solve LP for a' given b_new
        M_a, a_new = _lp_min_peak(b_new)
        if a_new is None:
            a_new = a_try
        # Mix
        a_cand = 0.5 * a_try + 0.5 * a_new
        a_cand = np.maximum(a_cand, 1e-15)
        a_cand = a_cand / a_cand.sum()
        sc = _score_norm(a_cand)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_cand.copy()
            best_b = b_new.copy()
            stall = 0
            if verbose and it % 20 == 0:
                print(f"    LP-alt iter {it}: score={sc:.10f}")
        else:
            stall += 1
            if stall > 15:
                break
        a = a_cand
        b = b_new
    return best_a, best_b, best_score


def _lp_iterative(a, deadline, max_iters=300, verbose=True):
    """Iterative LP: repeatedly solve LP for b given a, update a."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        M_lp, b = _lp_min_peak(a)
        if b is None:
            break
        # Try different mixing ratios
        best_local_score = best_score
        best_local_a = best_a.copy()
        for t in [0.3, 0.5, 0.7, 0.85, 1.0]:
            a_try = (1.0 - t) * a + t * b
            a_try = np.maximum(a_try, 1e-15)
            a_try = a_try / a_try.sum()
            sc = _score_norm(a_try)
            if sc < best_local_score:

                best_local_a = a_try
        if best_local_score < best_score - 1e-14:

        else:

        a = best_local_a
    return best_a, best_score


def _adam_logit(p0, deadline, rng, lr=0.05, beta_frac=0.06, beta_decay=0.94,
                no_imp_limit=400):
    """Adam optimizer in logit space with smooth max."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    z = np.log(p0)
    z -= float(np.mean(z))
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-4, beta_frac * maxc)
    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_and_weights(c, beta)
        g = _conv_cross_corr(w, p)
        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        z = np.clip(z, -80.0, 80.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 300 == 0:
                    lr = max(0.003, lr * 0.9)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        if t_adam % 20 == 0:
            beta = max(1e-5, beta * beta_decay)
        if no_imp > no_imp_limit:
            ns = 0.04 + 0.02 * min(no_imp // no_imp_limit, 5)
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -80, 80)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-4, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0
    return best_p, best_score


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_quadratic(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + a1 * t + a2 * t * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):

    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)

def _make_profile_two_bump(n, c1, c2, s):

def _make_profile_sine_sq(n, freq):

def _make_profile_w_mold(n, depth):

def _make_profile_mold(n, depth):

    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + depth * np.cos(2.0 * np.pi * t)

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))

        if n > 2048:
            continue
        per_end = min(phase2_end, time.monotonic() + 80.0)
        p_lp, sc_lp = _lp_iterative(p, per_end, max_iters=100)
        _update(p_lp, sc_lp, f'LP_{label}_n{n}')
        polished.append((sc_lp, p_lp, n, label))
        count += 1

    polished.sort(key=lambda x: x[0])
    print(f"Phase 2 done. Best: {best_score:.10f}")

    # === Phase 3: LP alternating minimization (600s - 1000s) ===
    phase3_end = t_start + 1000.0
    if best_seq is not None and best_seq.size <= 2048:
        n_cur = best_seq.size
        print(f"  Phase 3: LP-alternating n={n_cur}")
        # Initialize b as the current best
        b_init = best_seq.copy()
        per_end = min(phase3_end - 30, time.monotonic() + 300)
        p_alt, b_alt, sc_alt = _lp_alternate(best_seq, b_init, per_end, max_iters=150)
        _update(p_alt, sc_alt, 'LP_alt')
        # Also try with b as the LP solution
        M_lp, b_lp = _lp_min_peak(best_seq)
        if b_lp is not None and time.monotonic() < phase3_end - 60:
            per_end2 = min(phase3_end - 30, time.monotonic() + 200)
            p_alt2, b_alt2, sc_alt2 = _lp_alternate(best_seq, b_lp, per_end2, max_iters=100)
            _update(p_alt2, sc_alt2, 'LP_alt2')

    # === Phase 4: Adam polish (1000s - 1300s) ===
    phase4_end = t_start + 1300.0
    if time.monotonic() < phase4_end and best_seq is not None:
        per = min(200.0, phase4_end - time.monotonic() - 10)
        if per > 30:
            p_adam, sc_adam = _adam_logit(best_seq, time.monotonic() + per, rng,
                                           lr=0.04, beta_frac=0.05,
                                           beta_decay=0.95, no_imp_limit=500)
            _update(p_adam, sc_adam, 'adam_polish')
            # LP after Adam
            if time.monotonic() < phase4_end - 10 and best_seq.size <= 2048:
                per2 = min(60.0, phase4_end - time.monotonic() - 5)
                if per2 > 15:
                    p_lp2, sc_lp2 = _lp_iterative(best_seq, time.monotonic() + per2, max_iters=50)
                    _update(p_lp2, sc_lp2, 'LP_post_adam')

    # === Phase 5: Multi-scale escalation (1300s - 1550s) ===
    phase5_end = t_start + 1550.0
    if time.monotonic() < phase5_end and best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [768, 1024]
        elif n_cur <= 768:
            target_ns = [1024, 1536]
        elif n_cur <= 1024:
            target_ns = [1536, 2048]
        elif n_cur <= 1536:
            target_ns = [2048]

        for tn in target_ns:
            if time.monotonic() > phase5_end - 60:
                break
            p_int = _interp_profile(best_seq, tn)
            if p_int is None:
                continue
            sc_int = _score_norm(p_int)
            print(f"  Escalate to n={tn}: {sc_int:.10f}")
            _update(p_int, sc_int, f'interp_n{tn}')

            # Quick LP refine at new scale
            if tn <= 2048 and time.monotonic() < phase5_end - 30:
                per3 = min(30.0, phase5_end - time.monotonic() - 15)
                if per3 > 10:
                    p_esc, sc_esc = _lp_iterative(p_int, time.monotonic() + per3, max_iters=30)
                    _update(p_esc, sc_esc, f'LP_esc_n{tn}')

            # Quick Adam at new scale
            if time.monotonic() < phase5_end - 20:
                per4 = min(20.0, phase5_end - time.monotonic() - 10)
                if per4 > 8:
                    p_ad4, sc_ad4 = _adam_logit(p_int, time.monotonic() + per4, rng,
                                                 lr=0.03, beta_frac=0.05,
                                                 beta_decay=0.95, no_imp_limit=200)
                    _update(p_ad4, sc_ad4, f'adam_esc_n{tn}')

    # === Phase 6: Multi-restart (1550s - 1750s) ===
    phase6_end = deadline - 30.0
    if time.monotonic() < phase6_end and best_seq is not None:
        n_best = best_seq.size
        n_restarts = 4
        per_r = max(20.0, (phase6_end - time.monotonic()) / max(1, n_restarts + 1))

        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 15:
                break
            # Perturb in log-space
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            ns = 0.02 + 0.015 * i
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -80, 80)
            z -= z.mean()
            p_s = _softmax_from_z(z)

            per_adam = min(per_r * 0.4, phase6_end - time.monotonic() - 25)
            if per_adam < 8:
                break
            p_r, sc_r = _adam_logit(p_s, time.monotonic() + per_adam, rng,
                                     lr=0.04, beta_frac=0.05, beta_decay=0.95,
                                     no_imp_limit=int(0.2 * n_best))
            _update(p_r, sc_r, f'restart{i}')

            # LP polish after restart
            if best_seq.size <= 2048 and time.monotonic() < phase6_end - 20:
                per_lp = min(25.0, phase6_end - time.monotonic() - 15)
                if per_lp > 10:
                    p_lp3, sc_lp3 = _lp_iterative(best_seq, time.monotonic() + per_lp,
                                                   max_iters=20)
                    _update(p_lp3, sc_lp3, f'LP_restart{i}')

    # === Phase 7: Final intensive LP + Adam (1750s - deadline) ===
    if time.monotonic() < deadline - 5 and best_seq is not None:
        # Final LP
        if best_seq.size <= 2048:
            per = min(40.0, deadline - time.monotonic() - 15)
            if per > 10:
                p_f, sc_f = _lp_iterative(best_seq, time.monotonic() + per, max_iters=30)
                _update(p_f, sc_f, 'final_LP')

        # Final Adam
        if time.monotonic() < deadline - 5:
            per2 = deadline - time.monotonic() - 3
            if per2 > 8:
                p_f2, sc_f2 = _adam_logit(best_seq, time.monotonic() + per2, rng,
                                           lr=0.01, beta_frac=0.03,
                                           beta_decay=0.97, no_imp_limit=100)
                _update(p_f2, sc_f2, 'final_adam')

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(1024, 1.5)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")
        return [float(x) for x in final]

    fb = _make_profile_linear(1024, 1.5)
    print(f"FALLBACK n=1024")
    return [float(x) for x in fb]
