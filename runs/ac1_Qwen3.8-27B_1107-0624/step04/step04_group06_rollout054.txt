def _adam_lp(a_init, deadline, p_schedule, rng, lr=0.05):
    """Adam optimization of Lp norm of convolution, in logit space."""
    a = _normalize(a_init.copy())
    n = len(a)
    z = np.log(a + 1e-30)
    z -= np.mean(z)
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    
    current_p = p_schedule[0]
    p_idx = 0
    p_sum = 0  # total iterations at current p
    
    while time.monotonic() < deadline:
        # Compute gradient
        g_a = _lp_grad(a, current_p)
        # Convert to logit space: dz = a * (g_a - sum(a*g_a))
        s = np.sum(a * g_a)
        gz = a * (g_a - s)
        
        if not np.all(np.isfinite(gz)):
            a = best_a.copy()
            z = np.log(a + 1e-30)
            m[:] = 0
            v[:] = 0
            t = 0
            continue
        
        # Adam update
        t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz**2
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Convert back

        a = exp_z / exp_z.sum()

        sc = _score(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Schedule
        p_sum += 1

2. Computing h = c^{p-1}
3. Computing two FFTs and two IFFTs for the gradient: O(N' log N') where N' = next_pow2(3n-2)
4. Computing the score: already have c from step 1

So each iteration is about 3 FFTs of size O(n log n). For n=1024, this is very fast (microseconds).

Wait, but there's also the score computation which requires finding the max of c. That's O(n) which is negligible.

Let me now think about whether the Lp approach will actually be better than the smooth-max approach.

The smooth-max approach effectively minimizes:
max(c) ≈ (1/beta) * log(sum(exp(beta * c_k)))

For large beta, this approaches max(c). The gradient is:
g_k = c_k * (d/dc of smooth_max) = c_k * softmax(beta * c)

Wait no, the gradient of log-sum-exp with respect to a is:
d/da [log sum_k exp(beta * c_k)] = sum_k exp(beta * c_k) / sum_j exp(beta * c_j) * dc_k/da
= sum_k w_k * dc_k/da where w_k = exp(beta * c_k) / sum_j exp(beta * c_j)

So the smooth-max gradient is a weighted sum of dc_k/da with weights concentrated on the peak.

The Lp gradient is:
d/da [sum_k c_k^p] = p * sum_k c_k^{p-1} * dc_k/da

This is also a weighted sum of dc_k/da, but with weights c_k^{p-1} which are more spread out (for moderate p).

The key difference:
- Smooth-max: weights are exponential in c_k, so very concentrated on the peak
- Lp: weights are polynomial in c_k, so more spread out

The Lp approach might be better for:
- Getting out of local minima (more spread gradient)
- Finding the overall shape (not just the peak)

The smooth-max might be better for:
- Final polishing of the peak

So a good strategy is: use Lp for the bulk of the optimization, then switch to smooth-max for the final polish.

Let me also add one more technique: **direct peak reduction via quadratic programming**.

Given the current a and the peak at position k*, we can solve:
minimize a[k*-i] * a[i] for the specific peak position
subject to a >= 0, sum(a) = 1

This is a quadratic program (QP) that can be solved efficiently. The solution tells us how to redistribute mass to specifically reduce the peak at k*.

Actually, this is more complex because changing a[i] affects all convolution values, not just the peak. But as a first approximation, we can focus on the peak.

Let me think about a simpler version: for the peak at position k*, the peak value is:
c[k*] = sum_i a[i] * a[k*-i]

To reduce c[k*], we want to reduce the products a[i] * a[k*-i] for the terms that contribute most. The most effective way is to reduce the larger of a[i] and a[k*-i] (since the product is sensitive to the larger factor).

"""
Multi-moment gradient optimization for minimum peak auto-convolution.
Combines Lp-norm L^p (p=2,4,8,16) gradient descent with FFT-based gradients,
Adam optimizer in logit-space, spectral flattening, periodic LP refinement,
and smooth-max peak targeting for final polish.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _to_logits(a):
    a = np.maximum(a, 1e-30)
    z = np.log(a)
    z -= np.mean(z)
    return z


def _from_logits(z):
    z = np.clip(z, -50.0, 50.0)
    z = z - np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-30:
        return np.ones_like(e) / len(e)
    return e / s


def _lp_grad(a, p_power):
    """Gradient of sum(c_k^p_power) wrt a, via FFT cross-correlations."""
    n = len(a)
    c = _full_convolve(a)
    h = np.maximum(c, 1e-30) ** (p_power - 1)

    a_r = a[::-1].copy()
    N = _next_pow2(n + len(h) - 1)
    fa = np.fft.rfft(a, N)
    fa_r = np.fft.rfft(a_r, N)
    fh = np.fft.rfft(h, N)

    conv_ar_h = np.fft.irfft(fa_r * fh, N)
    conv_a_h = np.fft.irfft(fa * fh, N)

    term1 = conv_ar_h[n - 1:2 * n - 1]
    term2 = conv_a_h[0:n]

    return p_power * (term1 + term2)


def _smooth_max_grad(a, c, beta):
    """Gradient of smooth-max(beta * max(c)) wrt a."""
    n = len(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    # grad[m] = sum_k w_k * dc_k/da_m = sum_k w_k * (a_{k-m} + a_{m-k})
    a_r = a[::-1].copy()
    N = _next_pow2(n + len(w) - 1)
    fa = np.fft.rfft(a, N)
    fa_r = np.fft.rfft(a_r, N)
    fw = np.fft.rfft(w, N)

    conv_ar_w = np.fft.irfft(fa_r * fw, N)
    conv_a_w = np.fft.irfft(fa * fw, N)

    term1 = conv_ar_w[n - 1:2 * n - 1]
    term2 = conv_a_w[0:n]

    return term1 + term2


def _adam_lp(a_init, deadline, p_power, rng, lr=0.05, no_imp_limit=300, restart_noise=0.05):
    """Adam optimization of Lp norm in logit space."""
    a = _normalize(a_init.copy())
    n = len(a)
    z = _to_logits(a)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a)
        g_a = _lp_grad(a, p_power)
        s_ag = float(np.sum(a * g_a))
        gz = a * (g_a - s_ag)

        if not np.all(np.isfinite(gz)):
            z = _to_logits(best_a)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            continue

        t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if not np.all(np.isfinite(z)):
            z = _to_logits(best_a)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            continue

        a = _from_logits(z)
        c = _full_convolve(a)
        sc = float(2.0 * n * c.max())

        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > no_imp_limit:
            z = _to_logits(best_a) + restart_noise * rng.normal(size=n)
            z -= np.mean(z)
            a = _from_logits(z)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            no_imp = 0

    return best_a, best_score


def _adam_smoothmax(a_init, deadline, rng, lr=0.04, beta_init=0.05, beta_decay=0.95,
                    no_imp_limit=250, restart_noise=0.04):
    """Adam optimization of smooth-max in logit space."""
    a = _normalize(a_init.copy())
    n = len(a)
    z = _to_logits(a)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    beta = beta_init
    c = _full_convolve(a)

    while time.monotonic() < deadline:
        g_a = _smooth_max_grad(a, c, beta)
        s_ag = float(np.sum(a * g_a))
        gz = a * (g_a - s_ag)

        if not np.all(np.isfinite(gz)):
            z = _to_logits(best_a)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            continue

        t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if not np.all(np.isfinite(z)):
            z = _to_logits(best_a)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            continue

        a = _from_logits(z)
        c = _full_convolve(a)
        sc = float(2.0 * n * c.max())

        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if t % 20 == 0:
            beta = max(1e-5, beta * beta_decay)

        if no_imp > no_imp_limit:
            z = _to_logits(best_a) + restart_noise * rng.normal(size=n)
            z -= np.mean(z)
            a = _from_logits(z)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            no_imp = 0
            beta = max(beta_init * 0.5, 0.02)

    return best_a, best_score


def _spectral_flatten(a, alpha=0.7, n_iter=100):
    """Spectral flattening: iterate between time and frequency domains."""
    a = _normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)

    for _ in range(n_iter):
        A = np.fft.rfft(a, 2 * n)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-30)
        # Flatten the non-DC magnitudes
        mag[1:] = mag[1:] ** alpha
        # Keep DC
        A_new = mag * phase
        a_new = np.fft.irfft(A_new, 2 * n)[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new)

        sc = _score(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a_init, deadline, t=0.3, m_shrink=0.998, max_solves=30):
    """LP-based refinement: find optimal b given peak constraint M."""
    a = _normalize(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(a, M)
        if b is None:
            b = _lp_solve(a, M * 1.02)
            if b is None:
                break

        s_b = float(b.sum())
        if s_b < 1e-10:
            break

        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = _normalize(a_new)
        sc = _score(a_new)

        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()

        a = a_new
        c = _full_convolve(a)
        M_new = float(c.max())
        M = max(M_new * m_shrink, best_score / (2.0 * n) * 1.001)
        t = max(0.15, t * 0.93)

    return best_a, best_score


def _peak_flood(a_init, deadline, eta=0.03, n_iter=2000):
    """Multiplicative peak flooding."""
    a = _normalize(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    c = _full_convolve(a)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        M = float(c.max())
        sc = float(2.0 * n * M)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()

        threshold = M * 0.92
        hot_mask = c >= threshold
        if not hot_mask.any():
            hot_mask[int(np.argmax(c))] = True

        heat = np.zeros(n)
        hot_indices = np.where(hot_mask)[0]
        for k in hot_indices:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                partner = k - np.arange(i_start, i_end + 1)
                heat[i_start:i_end + 1] += a[partner]

        heat_max = float(heat.max())
        if heat_max < 1e-15:
            break
        heat /= heat_max

        factor = np.maximum(1.0 - eta * heat, 0.01)
        a = a * factor
        a = _normalize(a)
        c = _full_convolve(a)

        if it % 200 == 0 and it > 0:
            if sc >= best_score:
                eta = max(0.005, eta * 0.85)

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x1, x0, p0))


def _make_random_init(n, rng, sparsity=0.3):
    """Create a random initialization with some structure."""
    a = np.ones(n)
    # Add harmonic structure
    freq = rng.integers(1, max(2, n // 4))
    amp = rng.uniform(0.3, 1.0)
    t = np.arange(n)
    a += amp * np.cos(2.0 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi))
    # Add random noise
    a += 0.2 * rng.standard_normal(n)
    # Sparsify: zero out some fraction
    if sparsity > 0:
        mask = rng.random(n) > sparsity
        a[~mask] *= 0.1
    a = np.maximum(a, 0.01)
    return _normalize(a)

                               rng=rng, lr=0.06, no_imp_limit=150)
        _update(p6r2, "rand_2048")

    # === Phase 7: Final intensive polish ===
    t7 = deadline
    print(f"\nPhase 7: Final polish (score={best_score:.10f})")
    if time.monotonic() < t7 - 30 and best_seq is not None:
        # Determine best base
        base = best_seq.copy()

        # L8 with small lr
        rem = t7 - time.monotonic() - 30
        per1 = rem * 0.25
        if per1 > 20:
            p7a, sc7a = _adam_lp(base, time.monotonic() + per1, p_power=8,
                                  rng=rng, lr=0.01, no_imp_limit=150, restart_noise=0.02)
            _update(p7a, "final_L8")

        # Smooth-max very fine
        if time.monotonic() < t7 - 20:
            per2 = min(rem * 0.3, 120.0)
            if per2 > 15:
                p7b, sc7b = _adam_smoothmax(best_seq, time.monotonic() + per2, rng=rng,
                                            lr=0.008, beta_init=0.01, beta_decay=0.98,
                                            no_imp_limit=100, restart_noise=0.01)
                _update(p7b, "final_sm")

        # LP final
        if time.monotonic() < t7 - 15 and best_seq.size <= 2048:
            per3 = min(45.0, t7 - time.monotonic() - 10)
            if per3 > 10:
                p7c, sc7c = _lp_refine(best_seq, time.monotonic() + per3, t=0.1,
                                       m_shrink=0.9998, max_solves=8)
                _update(p7c, "final_lp")

        # L2 final (smoothing)
        if time.monotonic() < t7 - 10:
            per4 = t7 - time.monotonic() - 5
            if per4 > 10:
                p7d, sc7d = _adam_lp(best_seq, time.monotonic() + per4, p_power=2,
                                      rng=rng, lr=0.005, no_imp_limit=80, restart_noise=0.01)
                _update(p7d, "final_L2")

    # === Final validation ===
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = np.ones(1024) / 1024

        exact_sc = _score(final)
        print(f"\nFINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"EVAL error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = np.ones(1024) / 1024
    print(f"FALLBACK n=1024, score={_score(fb):.6f}")
    return [float(x) for x in fb]
