    """

    # --- Security Checks ---

    # Verify that the input is a list

        if np.isnan(x) or np.isinf(x):
            return np.inf

    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

def _next_pow2(n):

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1

    fa = np.fft.rfft(a, N)

def _make_profile(n, kind="uniform"):

    elif kind == "bimodal":

        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)

def _perturb_structured(a, rng, mode="shift"):
    """Apply structured perturbation."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    if mode == "shift":
        shift = rng.integers(1, max(2, n // 10))
        z = np.roll(z, shift)

        z = z * scale
    elif mode == "fourier":
        freq = rng.integers(1, n // 4)
        amp = 0.05 * rng.uniform(0.5, 2.0)
        t = np.arange(n)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n)
    z = z - z.mean()
    return _softmax(z)


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)

    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
