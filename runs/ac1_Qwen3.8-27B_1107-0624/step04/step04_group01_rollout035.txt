```python
    n = len(a)
    # Build constraint matrix A (2n-1 x n) where A[k,i] = a[i] if k-i in [0,n-1]
    # Actually, we only need the active constraints
    c = np.convolve(a, a)
    # Find active lags
    if max_constraints is not None:
        active = np.where(c >= M * 0.95)[0]  # Lags where conv is close to max
    else:
        active = np.where(c >= M - 1e-10)[0]
    
    # Build A matrix for active constraints
    A = np.zeros((len(active), n))
    for idx, k in enumerate(active):
        for i in range(max(0, k-n+1), min(n, k+1)):
            A[idx, i] = a[i]
    
    # LP: maximize sum(b) subject to A @ b <= M, b >= 0
    # In scipy linprog form: minimize -sum(b) subject to A @ b <= M, b >= 0
    c_vec = -np.ones(n)
    res = scipy.optimize.linprog(c_vec, A_ub=A, b_ub=np.full(len(active), M), bounds=(0, None), method='highs')
    if res.success:
        return res.x
    return None
```

```python
"""
L-BFGS-B + LP active-set hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp smoothing, FFT-based gradients, scipy L-BFGS-B for smooth optimization,
and LP-based active set refinement. Includes multi-restart, asymmetric initialization,
and size sweep strategies.
"""
import numpy as np
import time
from scipy import optimize as sp_opt


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lbgbs_smooth_max_grad(z, beta, n):
    """Compute gradient of 2n * logsumexp(conv(a,a)/beta) w.r.t. z (softmax params)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # grad_a[i] = 2 * sum_j w[j] * a[j-i] = 2 * conv(w, a_rev)[n-1+i]
    a_rev = a[::-1]
    N = _next_pow2(3 * n)
    W = np.fft.rfft(w, N)
    AR = np.fft.rfft(a_rev, N)
    conv_w_ar = np.fft.irfft(W * AR, N)[:3 * n - 1]
    grad_a = 2.0 * conv_w_ar[n - 1:2 * n - 1]
    # Softmax chain rule: d_obj/dz[i] = 2n * a[i] * (grad_a[i] - dot(a, grad_a))
    W_dot = float(np.dot(a, grad_a))
    grad_z = 2.0 * n * a * (grad_a - W_dot)
    if not np.all(np.isfinite(grad_z)):
        grad_z = np.zeros_like(grad_z)
    return grad_z


def _lbgbs_objective(z, beta, n):
    """Objective: 2n * beta * logsumexp(conv(a,a)/beta) (log-space smooth max)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return cm
    return beta * np.log(s) + cm


def _lbgbs_optimize(a_init, deadline, beta_schedule, max_iter=5000):
    """L-BFGS-B optimization with beta continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score(a_init)

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break

        def obj(zf):
            return _lbgbs_objective(zf, beta, n)

        def grad(zf):
            return _lbgbs_smooth_max_grad(zf, beta, n)

        try:
            res = sp_opt.minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8,
                         'maxls': 50}
            )
            z = res.x
        except Exception:
            break

        a_cur = _softmax(z)
        sc = _score(a_cur)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_cur.copy()

        z = z - z.mean()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _lp_step(a, M, n_active=100, tol=0.9):
    """LP-based active set step. Returns new b (normalized to sum=1) or None."""
    n = len(a)
    c = _full_convolve(a, a)
    threshold = M * tol
    active = np.where(c >= threshold - 1e-12)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    active = active[:n_active]

    # Build constraint matrix: A[k,i] = a[i] if 0 <= k-i < n
    A = np.zeros((len(active), n))
    for idx, k in enumerate(active):
        i_lo = max(0, k - n + 1)
        i_hi = min(n, k + 1)
        A[idx, i_lo:i_hi] = a[i_lo:i_hi]

    # LP: maximize sum(b) s.t. A @ b <= M, b >= 0
    c_vec = -np.ones(n)
    b_ub = np.full(len(active), M + 1e-12)
    try:
        res = sp_opt.linprog(c_vec, A_ub=A, b_ub=b_ub,
                             bounds=[(0, None)] * n, method='highs')
        if res.success and res.x is not None and res.x.sum() > 1e-10:
            b = res.x / res.x.sum()
            return b
    except Exception:
        pass
    return None


def _lp_refine(a, deadline, max_iters=200, tol_ratio=0.95, lr=0.1):
    """Iterative LP-based refinement."""
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        M = float(c.max())
        b = _lp_step(a, M, n_active=200, tol=tol_ratio)
        if b is None:
            break
        a_new = (1.0 - lr) * a + lr * b
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        sc = _score(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        # Tighten tolerance as we refine
        tol_ratio = max(0.99, tol_ratio * 0.999)

    return best_a, best_score


def _adam_optimize(a_init, deadline, beta_schedule, max_iter=300000, lr=0.012):
    """Adam optimizer in log-space (fallback/backup method)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_phase = 0
        while it_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_phase += 1
            t += 1
            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_s = (c - cm) / beta
            e = np.exp(c_s)
            s_e = float(e.sum())
            if s_e < 1e-300 or not np.isfinite(s_e):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_e
            # Gradient via FFT
            a_rev = a[::-1]
            N = _next_pow2(3 * n)
            W = np.fft.rfft(w, N)
            AR = np.fft.rfft(a_rev, N)
            conv_w_ar = np.fft.irfft(W * AR, N)[:3 * n - 1]
            grad_a = 2.0 * conv_w_ar[n - 1:2 * n - 1]
            Wd = float(np.dot(a, grad_a))
            grad = 2.0 * n * a * (grad_a - Wd)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1 - b1) * grad
            v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
            mh = m_buf / (1 - b1 ** t)
            vh = v_buf / (1 - b2 ** t)
            update = lr * mh / (np.sqrt(vh) + eps)
            un = np.linalg.norm(update)
            if un > 0.5:
                update *= 0.5 / un
            z = z - update
            z = np.clip(z, -50, 50)
            z = z - z.mean()

            if t % 200 == 0:
                sc = _score(_softmax(z))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = _softmax(z).copy()

    a_f = _softmax(z)
    sc_f = _score(a_f)
    if np.isfinite(sc_f) and sc_f < best_score:
        best_score = sc_f
        best_a = a_f.copy()
    return best_a, best_score


def _make_profile(n, kind):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp_up":
        t = np.arange(n, dtype=np.float64)
        a = t + 1.0
        return a / a.sum()
    elif kind == "ramp_down":
        t = np.arange(n, dtype=np.float64)
        a = (n - t)
        return a / a.sum()
    elif kind == "exponential":
        t = np.arange(n, dtype=np.float64)
        a = np.exp(-t / (n * 0.3))
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.3) ** 2) + np.exp(-20 * (t - 0.7) ** 2)
        return a / a.sum()
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric_sin":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * t) * (0.5 + t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _zero_small(a, frac=0.1):
    """Zero out the smallest frac fraction of entries and renormalize."""
    n = len(a)
    k = max(1, int(n * frac))
    idx = np.argpartition(a, k - 1)[:k - 1] if k > 1 else np.array([])
    a_new = a.copy()
    a_new[idx] = 0.0
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid L-BFGS-B + LP active-set optimizer with multi-restart and size sweep.
    Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        update_best(parent, _score(parent), "parent")
        print(f"Parent: n={n_main}, score={best_score:.10f}")
    else:
        n_main = 2048
        parent = _make_profile(n_main, "uniform")
        update_best(parent, _score(parent), "uniform_init")

    # === Phase 1: L-BFGS-B main optimization (0 - 400s) ===
    phase1_end = min(deadline, t_start + 400.0)
    print(f"\nPhase 1: L-BFGS-B optimization (target: {phase1_end - t_start:.0f}s)")

    if best_seq is not None:
        c_init = _full_convolve(best_seq, best_seq)
        max_c = float(c_init.max())
        beta_schedule = [
            0.02 * max_c, 0.01 * max_c, 0.005 * max_c,
            0.002 * max_c, 0.001 * max_c, 0.0005 * max_c,
            0.0002 * max_c, 0.0001 * max_c
        ]
        p_best, sc_best = _lbgbs_optimize(
            best_seq, time.monotonic() + (phase1_end - time.monotonic() - 10),
            beta_schedule, max_iter=3000
        )
        update_best(p_best, _score(p_best), "lbgfs_main")

    # === Phase 2: LP refinement (400s - 700s) ===
    phase2_end = min(deadline, t_start + 700.0)
    print(f"\nPhase 2: LP active-set refinement")

    if best_seq is not None and time.monotonic() < phase2_end - 5:
        p_lpf, sc_lpf = _lp_refine(
            best_seq, time.monotonic() + (phase2_end - time.monotonic() - 5),
            max_iters=300, tol_ratio=0.92, lr=0.08
        )
        update_best(p_lpf, _score(p_lpf), "lp_refine")

        # Re-optimize with L-BFGS-B after LP
        if time.monotonic() < phase2_end - 30:
            c2 = _full_convolve(best_seq, best_seq)
            mc2 = float(c2.max())
            bs2 = [0.003 * mc2, 0.001 * mc2, 0.0003 * mc2, 0.0001 * mc2]
            p2, sc2 = _lbgbs_optimize(
                best_seq, time.monotonic() + (phase2_end - time.monotonic() - 5),
                bs2, max_iter=2000
            )
            update_best(p2, _score(p2), "lbgfs_post_lp")

    # === Phase 3: Multi-restart with diverse initializations (700s - 1200s) ===
    phase3_end = min(deadline, t_start + 1200.0)
    print(f"\nPhase 3: Multi-restart (target: {phase3_end - t_start:.0f}s)")

    if best_seq is not None and time.monotonic() < phase3_end - 30:
        n_restarts = 6
        per_restart = (phase3_end - time.monotonic() - 20) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 15:
                break
            # Different perturbation strategies
            if i == 0:
                p_start = _perturb(best_seq, 0.04, rng)
            elif i == 1:
                p_start = _perturb(best_seq, 0.1, rng)
            elif i == 2:
                # Asymmetric perturbation
                z = np.log(np.maximum(best_seq, 1e-15))
                t = np.arange(len(best_seq), dtype=np.float64) / len(best_seq)
                z = z + 0.1 * np.sin(4 * np.pi * t)
                z = z - z.mean()
                p_start = _softmax(z)
            elif i == 3:
                # Zero out some entries
                p_start = _zero_small(best_seq, frac=0.05)
            elif i == 4:
                # Fourier perturbation
                z = np.log(np.maximum(best_seq, 1e-15))
                freq = rng.integers(2, max(3, n_main // 8))
                amp = 0.05 * rng.uniform(0.5, 2.0)
                t = np.arange(n_main)
                z = z + amp * np.cos(2 * np.pi * freq * t / n_main)
                z = z - z.mean()
                p_start = _softmax(z)
            else:
                p_start = _perturb(best_seq, 0.06, rng)

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = [0.01 * max_cs, 0.005 * max_cs, 0.002 * max_cs, 0.0005 * max_cs]
            per_time = min(per_restart, phase3_end - time.monotonic() - 10)
            if per_time < 10:
                break

            p_best, sc_best = _lbgbs_optimize(
                p_start, time.monotonic() + per_time,
                bs, max_iter=2000
            )
            update_best(p_best, _score(p_best), f"restart{i}")

            # Quick LP polish on the best of this restart
            if time.monotonic() < phase3_end - 20:
                p_lpf, sc_lpf = _lp_refine(
                    best_seq, time.monotonic() + min(20, phase3_end - time.monotonic() - 5),
                    max_iters=50, tol_ratio=0.95, lr=0.05
                )
                update_best(p_lpf, _score(p_lpf), f"lp_r{i}")

    # === Phase 4: Try different n (1200s - 1500s) ===
    phase4_end = min(deadline, t_start + 1500.0)
    print(f"\nPhase 4: Size exploration")

    if best_seq is not None and time.monotonic() < phase4_end - 30:
        # Try larger n by interpolation
        for target_n in [4096, 3072]:
            if time.monotonic() > phase4_end - 60:
                break
            if len(best_seq) >= target_n:
                continue
            x0 = np.linspace(0, 1, len(best_seq))
            x1 = np.linspace(0, 1, target_n)
            p_interp = np.interp(x1, x0, best_seq)
            p_interp = np.maximum(p_interp, 1e-12)
            p_interp = p_interp / p_interp.sum()
            sc_interp = _score(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{target_n}")

            # Optimize at this size
            if time.monotonic() < phase4_end - 40:
                c_int = _full_convolve(p_interp, p_interp)
                max_ci = float(c_int.max())
                bs_int = [0.005 * max_ci, 0.002 * max_ci, 0.0005 * max_ci, 0.0001 * max_ci]
                rem = min(60, phase4_end - time.monotonic() - 10)
                p_best, sc_best = _lbgbs_optimize(
                    p_interp, time.monotonic() + rem,
                    bs_int, max_iter=1500
                )
                update_best(p_best, _score(p_best), f"opt_n{target_n}")

        # Also try smaller n (subsample)
        for target_n in [1024, 1536]:
            if time.monotonic() > phase4_end - 30:
                break
            if len(best_seq) <= target_n:
                continue
            idx = np.linspace(0, len(best_seq) - 1, target_n).astype(int)
            p_sub = best_seq[idx]
            p_sub = p_sub / p_sub.sum()
            sc_sub = _score(p_sub)
            if sc_sub < best_score:
                update_best(p_sub, sc_sub, f"sub_n{target_n}")

    # === Phase 5: Fine polish with very low beta (1500s - 1700s) ===
    phase5_end = min(deadline, t_start + 1700.0)
    print(f"\nPhase 5: Fine polish")

    if best_seq is not None and time.monotonic() < phase5_end - 10:
        c_best = _full_convolve(best_seq, best_seq)
        max_cb = float(c_best.max())
        beta_fine = [0.0001 * max_cb, 0.00005 * max_cb, 0.00002 * max_cb, 0.00001 * max_cb]
        rem = phase5_end - time.monotonic() - 5
        if rem > 10:
            p_best, sc_best = _lbgbs_optimize(
                best_seq, time.monotonic() + rem,
                beta_fine, max_iter=3000
            )
            update_best(p_best, _score(p_best), "fine_polish")

        # Adam fine polish as backup
        if time.monotonic() < phase5_end - 10:
            beta_adam = [0.00005 * max_cb, 0.00002 * max_cb, 0.00001 * max_cb]
            rem2 = phase5_end - time.monotonic() - 5
            if rem2 > 10:
                p_best2, sc_best2 = _adam_optimize(
                    best_seq, time.monotonic() + rem2,
                    beta_adam, max_iter=200000, lr=0.003
                )
                update_best(p_best2, _score(p_best2), "adam_fine")

    # === Phase 6: Final LP + L-BFGS-B cycle (1700s - deadline) ===
    print(f"\nPhase 6: Final cycle")
    if best_seq is not None and time.monotonic() < deadline - 15:
        # LP step
        p_lpf, sc_lpf = _lp_refine(
            best_seq, time.monotonic() + min(30, deadline - time.monotonic() - 20),
            max_iters=100, tol_ratio=0.97, lr=0.03
        )
        update_best(p_lpf, _score(p_lpf), "final_lp")

        # L-BFGS-B final
        if time.monotonic() < deadline - 10:
            c_fin = _full_convolve(best_seq, best_seq)
            mf = float(c_fin.max())
            bs_fin = [0.00005 * mf, 0.00002 * mf, 0.00001 * mf]
            rem = deadline - time.monotonic() - 5
            if rem > 5:
                p_best, sc_best = _lbgbs_optimize(
                    best_seq, time.monotonic() + rem,
                    bs_fin, max_iter=2000
                )
                update_best(p_best, _score(p_best), "final_lbgfs")

    # === Final validation ===
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
