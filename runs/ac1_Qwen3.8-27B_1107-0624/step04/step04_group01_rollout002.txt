```python
"""
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses a linear program to find the best direction at each step, combined with
gradient-based polishing.
"""
import numpy as np
import time
import cvxpy as cp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _solve_lp_direction(f, n, solver=None):
    """Solve LP: min M s.t. (f*g)_k <= M for all k, sum(g)=1, g>=0"""
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    # Build constraint matrix A where (A@g)[j] = (f*g)[j]
    A = np.zeros((2*n-1, n))
    for i in range(n):
        A[i:i+n, i] = f
    
    constraints = [cp.sum(g) == 1, A @ g <= M]
    objective = cp.Minimize(M)
    prob = cp.Problem(objective, constraints)
    
    if solver is None:
        solver = cp.CLARABEL
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                return None
    
    if g.value is None or not np.all(np.isfinite(g.value)):
        return None
    
    g_val = np.maximum(g.value, 0)
    g_sum = g_val.sum()
    if g_sum < 1e-15:
        return None
    return g_val / g_sum


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z(a, c, w, n):
    """Compute 4*n*a*(h-W) where h[m]=sum_i w[i]*a[i-m], W=dot(w,c)."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    best_a = a.copy()
    best_score = _score(a)

    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    beta = 0.001 * cm

        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        e = np.exp(c_shifted)
        s = e.sum()
        if s < 1e-300:

            w[np.argmax(c)] = 1.0
```

```python
def _solve_lp_direction(f, n):
    """Solve LP: min M s.t. (f*g)_k <= M for all k, sum(g)=1, g>=0"""
    # Variables: [g_0, ..., g_{n-1}, M]
    # Objective: minimize M -> c = [0, ..., 0, 1]

    b_ub = np.zeros(2*n-1)

    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]
```

```python
"""
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal descent directions, combined with
gradient-based polishing. The LP approach: min M s.t. (f*g)_k <= M, sum(g)=1, g>=0.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _build_conv_matrix(f, n):
    """Build matrix A where (A@g)[j] = (f*g)[j] for full convolution."""
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = f
    return A


def _solve_lp_direction(f, n, method='highs'):
    """Solve LP: min M s.t. (f*g)_k <= M for all k, sum(g)=1, g>=0.
    Returns optimal g or None on failure."""
    vars_n = n + 1  # g_0..g_{n-1}, M
    c_obj = np.zeros(vars_n)
    c_obj[-1] = 1.0  # minimize M

    A_conv = _build_conv_matrix(f, n)

    # Inequality: (f*g)_k - M <= 0
    A_ub = np.hstack([A_conv, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, vars_n))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(-np.inf, None)]

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method=method,
                         options={'presolve': True})
    except Exception:
        return None

    if not result.success or result.x is None:
        return None

    g = np.maximum(result.x[:n], 0.0)
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None
    return g / g_sum


def _lp_iterate(f, deadline, max_iter=5000, min_improve=1e-10):
    """LP-based iterative optimization loop. Returns (best_f, best_score)."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = f / f.sum()

    best_f = f.copy()
    best_score_norm = _score_norm(f)
    n = len(f)

    no_improve_count = 0

    for it in range(max_iter):

        g = _solve_lp_direction(f, n)
        lp_time = time.monotonic() - t0

            g = _solve_lp_direction(f, n, method='highs-ds')
            if g is None:

        # Line search
        best_t = 0.0
        best_new_score = current_score
        for t in [0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02]:
            new_f = (1.0 - t) * f + t * g
            new_f = np.maximum(new_f, 0.0)

            if sf < 1e-15:

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z(a, c, w, n):
    """Gradient of smooth-max objective w.r.t. log-parameters."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_polish(a_init, deadline, max_iter=100000, lr=0.003):
    """Adam gradient polishing in log-space with beta-continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c = _full_convolve(a_init, a_init)
    cm = float(c.max())
    beta = 0.005 * cm

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for _ in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = e / s

        grad = _compute_grad_z(a, c, w, n)
        if not np.all(np.isfinite(grad)):
            beta *= 0.5
            continue

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * grad * grad
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if t % 200 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                beta *= 0.85

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_a = a_final
        best_score = sc_final

    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)

                alloc = min(50.0, (phase5_end - time.monotonic()) - 10)
                f_try, _ = _lp_iterate(p_interp, time.monotonic() + alloc, max_iter=2000)
                sc_try = _score(f_try)
                if sc_try < best_score:
                    best_seq = f_try.copy()
                    best_score = sc_try
                    print(f"  NEW BEST from LP n={target_n}: {best_score:.10f}")

    # === Phase 6: Final gradient polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        f_final, _ = _gradient_polish(best_seq, deadline - 3, max_iter=50000, lr=0.001)
        sc_final = _score(f_final)
        if sc_final < best_score:
            best_seq = f_final.copy()
            best_score = sc_final

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
