```python
def _project_simplex(x):
    """Project x onto the probability simplex {p: sum(p)=1, p>=0}."""
    n = len(x)
    if n == 0:
        return x
    # Algorithm from Duchi et al. (2011)
    if np.all(x <= 0):
        return np.ones(n) / n
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_hat = rho[-1]
    theta = (cssv[rho_hat] - 1) / (rho_hat + 1)
    return np.maximum(x - theta, 0)
```

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Strategy: Fine L-BFGS continuation with log-sum-exp smoothing, followed by
direct subgradient polishing with simplex projection, and adaptive multi-peak
balancing. Uses both n=4096 and n=8192 to find the best solution.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(x):
    """Project x onto the probability simplex {p: sum(p)=1, p>=0}."""
    n = len(x)
    if n == 0:
        return x.copy()
    if n == 1:
        return np.ones(1)
    # Duchi et al. algorithm
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.where(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = int(rho_idx[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(x - theta, 0.0)


def _compute_smoothed_obj_grad(z, beta, n):
    """Compute log-sum-exp smoothed objective and gradient in z-space."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    obj = 2.0 * n * (cm + beta * float(np.log(s)))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        grad = np.zeros_like(z)
    return obj, grad


def _run_lbfgs_single_beta(z_init, beta, n, maxiter=5000):
    """Run L-BFGS-B at a single beta value."""
    try:
        result = minimize(
            _compute_smoothed_obj_grad, z_init, args=(beta, n),
            method='L-BFGS-B', jac=True,
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxls': 50, 'maxfun': 20000}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            return result.x
    except Exception:
        pass
    return z_init


def _run_lbfgs_continuation(a_init, deadline_time, n_restarts=3):
    """L-BFGS with fine beta continuation and multiple restarts."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    beta_schedule = [0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006,
                     0.005, 0.004, 0.003, 0.0025, 0.002, 0.0015,
                     0.001, 0.0008, 0.0006, 0.0005, 0.0004,
                     0.0003, 0.00025, 0.0002, 0.00015, 0.0001,
                     0.00008, 0.00006, 0.00004, 0.00003, 0.00002,
                     0.000015, 0.00001]

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    rng = np.random.default_rng(12345)

    for restart in range(n_restarts):
        if time.monotonic() > deadline_time:
            break

        if restart == 0:
            z_cur = z.copy()
        else:
            z_cur = np.log(np.maximum(best_a, 1e-15))
            z_cur = z_cur - z_cur.mean()
            noise = 0.03 * (restart + 1)
            z_cur = z_cur + noise * rng.normal(size=n)
            z_cur = np.clip(z_cur, -50, 50)
            z_cur = z_cur - z_cur.mean()

        for beta_frac in beta_schedule:
            if time.monotonic() > deadline_time - 5:
                break
            beta = beta_frac * max_c0
            if beta < 1e-12:
                beta = 1e-12
            z_cur = _run_lbfgs_single_beta(z_cur, beta, n, maxiter=3000)
            a_cur = _softmax_fn(z_cur)
            sc = _score_norm_fn(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                z = z_cur.copy()

    return best_a, best_score


def _run_adaptive_gd(a_init, deadline_time, lr_init=0.1):
    """Adam gradient descent with adaptive beta in z-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init

    rng = np.random.default_rng(999)

    max_iter = 50000
    for it in range(max_iter):
        if time.monotonic() > deadline_time - 2:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Adaptive beta
        sorted_c = np.sort(c)[::-1]
        k = min(20, len(sorted_c) - 1)
        gap = sorted_c[0] - sorted_c[k]
        if gap < 0.05 * cm:
            beta = max(1e-10, 0.1 * cm)
        elif gap > 0.5 * cm:
            beta = max(1e-10, 0.01 * cm)
        else:
            beta = max(1e-10, 0.5 * gap)

        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 2000 == 0:
            lr = max(0.01, lr * 0.9)

        if t % 10000 == 0:
            z = z + 0.03 * rng.normal(size=n)
            z = z - z.mean()

    return best_a, best_score


def _run_subgradient_polish(a_init, deadline_time):
    """Direct subgradient steps on the unsmoothed max with simplex projection."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    a = a_init.copy()

    # Compute initial max for scaling
    c_init = _full_convolve(a_init, a_init)
    max_c_init = float(c_init.max())

    # Start with larger eps, decrease over time
    total_time = max(1.0, deadline_time - time.monotonic())
    it = 0
    max_iters = 200000

    for it in range(max_iters):
        if time.monotonic() > deadline_time - 2:
            break

        # Adaptive eps: start wide, narrow down
        progress = min(1.0, it / max_iters)
        eps = (0.05 * (1 - progress) + 1e-6) * max_c_init

        step_size = 0.02 * (1 - 0.5 * progress)

        c = _full_convolve(a, a)
        C = float(c.max())

        active = c >= (C - eps)
        n_active = int(active.sum())
        if n_active == 0:
            active = np.zeros_like(c)
            active[int(np.argmax(c))] = 1.0
            n_active = 1

        w = active.astype(np.float64) / n_active
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]

        # Subgradient of 2n*max(c_k) at a
        g = 4.0 * n * h

        # Normalize and take step
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        direction = g / g_norm

        # Try multiple step sizes (line search)
        best_trial_score = _score_norm_fn(a)
        best_trial_a = a.copy()
        for trial_scale in [step_size, step_size * 0.5, step_size * 0.1,
                            step_size * 0.05, step_size * 0.01]:
            a_trial = a - trial_scale * direction
            a_trial = _project_simplex(a_trial)
            sc_trial = _score_norm_fn(a_trial)
            if sc_trial < best_trial_score:
                best_trial_score = sc_trial
                best_trial_a = a_trial.copy()

        if best_trial_score < best_score:
            best_score = best_trial_score
            best_a = best_trial_a.copy()
            a = best_trial_a.copy()
        else:
            # No improvement: try smaller eps or add noise
            a = a.copy()
            if it % 500 == 0:
                # Small random perturbation
                noise = 0.001 * step_size
                a_pert = a + noise * np.random.default_rng(it).normal(size=n)
                a_pert = _project_simplex(a_pert)
                sc_pert = _score_norm_fn(a_pert)
                if sc_pert < best_score:
                    best_score = sc_pert
                    best_a = a_pert.copy()
                    a = a_pert.copy()

        # Check convergence
        if it > 10000 and it % 5000 == 0:
            improvement = best_score - _score_norm_fn(best_a)
            if improvement < 1e-12:

    return best_a, best_score


def _run_multi_peak_balance(a_init, deadline_time):
    """Iteratively balance peaks by redistributing mass from high to low convolution lags."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    a = a_init.copy()

    c_init = _full_convolve(a_init, a_init)
    max_c_init = float(c_init.max())

    it = 0
    max_iters = 100000
    for it in range(max_iters):
        if time.monotonic() > deadline_time - 2:
            break

        c = _full_convolve(a, a)
        C = float(c.max())
        k_star = int(np.argmax(c))

        # Compute contributions to the peak
        valid_start = max(0, k_star - n + 1)
        valid_end = min(n - 1, k_star)
        if valid_end < valid_start:
            break
        valid_idx = np.arange(valid_start, valid_end + 1)
        contributions = a[valid_idx] * a[k_star - valid_idx]
        total_contrib = float(contributions.sum())
        if total_contrib < 1e-15:
            break

        # Find the most imbalanced contributing pair
        # We want to reduce a_i * a_{k-i} for the pair with max product
        # Strategy: reduce the larger of the pair, add to the smallest a value
        # that doesn't contribute to the peak

        top_j = int(np.argmax(contributions))
        i_hi = int(valid_idx[top_j])
        i_lo = k_star - i_hi

        # Find a position to add mass (one that minimally increases the peak)
        # Use positions far from the peak contributors
        a_sorted_idx = np.argsort(a)
        found = False
        for candidate_idx in a_sorted_idx[:20]:
            ci = int(candidate_idx)
            # Check if adding to ci would increase c_{k_star}
            delta_k = abs(ci - k_star)
            if 0 <= ci < n and not (valid_start <= ci <= valid_end):
                # This position doesn't directly contribute to lag k_star
                # Check effect on nearby lags
                a_trial = a.copy()
                # Move small amount from max contributor to candidate
                amt = min(a[i_hi], a[i_lo]) * 0.01
                a_trial[i_hi] -= amt
                a_trial[ci] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                c_trial = _full_convolve(a_trial, a_trial)
                sc_trial = float(2.0 * n * c_trial.max())
                if sc_trial < best_score:
                    best_score = sc_trial
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    found = True
                    break

        if not found:
            # Try a different strategy: reduce from both i_hi and i_lo
            a_trial = a.copy()
            amt = min(a[i_hi], a[i_lo]) * 0.005
            a_trial[i_hi] -= amt
            a_trial[i_lo] -= amt
            # Redistribute to positions with smallest values
            a_sorted_idx2 = np.argsort(a_trial)
            for candidate_idx in a_sorted_idx2[:5]:
                ci = int(candidate_idx)
                if ci != i_hi and ci != i_lo:
                    a_trial[ci] += amt
                    break
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial = a_trial / s
                c_trial = _full_convolve(a_trial, a_trial)
                sc_trial = float(2.0 * n * c_trial.max())
                if sc_trial < best_score:
                    best_score = sc_trial
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    found = True

        if not found and it > 2000:
            # Stuck, try z-space perturbation
            z = np.log(np.maximum(a, 1e-15))
            z = z - z.mean()
            z += 0.01 * np.random.default_rng(it + 7).normal(size=n)
            z = np.clip(z, -50, 50)
            a_new = _softmax_fn(z)
            sc_new = _score_norm_fn(a_new)
            if sc_new < best_score:

def _make_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []

    for slope in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    for a1, a2 in [(1.0, 0.5), (2.0, -0.3), (1.5, 0.2), (-1.0, 0.5)]:
        a = 1.0 + a1 * u + a2 * u ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    for rate in [1.0, 2.0, 3.0, -1.0, -2.0, -3.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())

    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())

    return profiles


def _interp_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):

    def update_best(p, sc, label=""):

            best_seq = np.asarray(p, dtype=np.float64).copy()

            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    try:

        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
