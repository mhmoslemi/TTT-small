LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses the linear programming approach: at each step, solve an LP to find the
best direction of improvement, then mix with the current sequence.

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _build_conv_matrix(f):
    """Build sparse matrix A (2n-1 x n) where A @ b = f * b."""
    n = len(f)
    diagonals = [np.full(n, f[d]) for d in range(n)]
    offsets = list(range(0, -n, -1))
    A = diags(diagonals, offsets, shape=(2*n-1, n), format='csr')
    return A


def _solve_lp(f, M, solver='highs'):
    """Solve: max sum(b) s.t. A@b <= M, b >= 0."""
    n = len(f)
    A = _build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method=solver)
    if result.success:
        return result.x, -result.fun
    return None, None

"""
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal improvement directions, combined with
adaptive mixing, multi-restart diversification, subgradient polishing, and
multi-scale exploration. Key improvement over gradient-based approaches:
each step solves an exact convex subproblem (LP) for the best direction.
"""
import numpy as np
from scipy.sparse import diags
from scipy.optimize import linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _peak(a):
    c = _full_convolve(a, a)
    return float(c.max())


def _build_conv_matrix(f):
    n = len(f)
    diagonals = [np.full(n, f[d]) for d in range(n)]
    offsets = list(range(0, -n, -1))
    A = diags(diagonals, offsets, shape=(2 * n - 1, n), format='csr')
    return A


def _solve_lp(f, M):
    n = len(f)
    A = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        b = np.maximum(result.x, 0.0)
        return b, float(-result.fun)
    return None, None


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        a[:n // 10] = 0.3
        a[-n // 10:] = 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    e = np.exp(z)
    return e / e.sum()


def _perturb_fourier(a, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    freq = rng.integers(1, max(2, n // 4))
    amp = rng.uniform(0.03, 0.15)
    phase = rng.uniform(0, 2 * np.pi)
    t = np.arange(n)
    z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    e = np.exp(z)
    return e / e.sum()


def _perturb_shift(a, rng):
    n = len(a)
    shift = rng.integers(1, max(2, n // 8))
    return np.roll(a, shift)


def _lp_iterate(f, deadline, max_iters=500, label=""):
    n = len(f)
    f = f.copy() / f.sum()
    best_f = f.copy()
    best_peak = _peak(f)

    iteration = 0
    no_improve = 0
    mix_ratio = 0.4

    while time.monotonic() < deadline - 5 and iteration < max_iters:

        c = _full_convolve(f, f)
        M = float(c.max())

            best_f = f.copy()
            best_score = float(2.0 * n * M)
            if iteration % 50 == 0:

        if b is None:
            break

        b_sum = float(b.sum())
        if b_sum < 1e-10:
            break
        b = b / b_sum

        # Check if b is significantly different from f
        dist = float(np.linalg.norm(b - f))
        if dist < 1e-8:
            no_improve += 1
            if no_improve > 10:
                break
            f = f + 0.01 * (rng.standard_normal(n) - f.mean())
            f = np.maximum(f, 0)
            f = f / f.sum()
            iteration += 1
            continue

        # Try mixing ratios

def _subgradient_polish(f, deadline, max_iters=50000):
    n = len(f)
    f = f.copy() / f.sum()
    best_f = f.copy()
    best_peak = _peak(f)
    best_score = float(2.0 * n * best_peak)

    alpha = 0.01
    it = 0
    prev_score = best_score

    while time.monotonic() < deadline - 5 and it < max_iters:
        c = _full_convolve(f, f)
        M = float(c.max())
        k_star = int(np.argmax(c))
        S = float(f.sum())

        # Subgradient: sg[j] = 2n * (f[k*-j] * S - 2*M) / S^3
        sg = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                sg[j] = 2.0 * n * (f[idx] * S - 2.0 * M) / (S ** 3)

        f_new = f - alpha * sg
        f_new = np.maximum(f_new, 0.0)
        f_sum = float(f_new.sum())
        if f_sum < 1e-10:
            break
        f_new = f_new / f_sum

        new_peak = _peak(f_new)
        if new_peak < best_peak * (1 - 1e-10):
            best_peak = new_peak
            best_f = f_new.copy()
            best_score = float(2.0 * n * new_peak)

        f = f_new
        it += 1

        # Adaptive step size
        if it % 100 == 0:
            if best_score < prev_score * 0.999:
                alpha = min(0.05, alpha * 1.05)
            else:
                alpha = max(0.001, alpha * 0.95)
            prev_score = best_score

    return best_f, best_score

            update_best(p_best, _score(p_best), f"P3_{label}")

    print(f"  Phase 3 best: {best_score:.10f}")

    # Phase 4: Continued LP on best (1100s - 1450s)
    phase4_end = min(deadline, t_start + 1450.0)
    print(f"\nPhase 4: Continued LP (budget={phase4_end - t_start:.0f}s)")
    if best_seq is not None and time.monotonic() < phase4_end - 10:
        f_best, sc_best = _lp_iterate(best_seq, phase4_end, max_iters=500, label="P4")
        update_best(f_best, _score(f_best), "P4_lp")

    # Phase 5: Try n=4096 (1450s - 1700s)
    phase5_end = min(deadline, t_start + 1700.0)
    if best_seq is not None and len(best_seq) < 4096 and time.monotonic() < phase5_end - 60:
        print(f"\nPhase 5: Try n=4096 (budget={phase5_end - t_start:.0f}s)")
        target_n = 4096
        p_interp = _interp_profile(best_seq, target_n)
        sc_interp = _score(p_interp)
        update_best(p_interp, sc_interp, "P5_interp4096")

        if time.monotonic() < phase5_end - 40:
            p_best, sc_best = _lp_iterate(p_interp, phase5_end, max_iters=80, label="P5")
            update_best(p_best, _score(p_best), "P5_lp4096")

    # Phase 6: LP on n=3072 if time permits (1450s-1700s alternative)
    if best_seq is not None and len(best_seq) < 3072 and time.monotonic() < deadline - 100:
        print(f"\nPhase 6: Try n=3072")
        target_n = 3072
        p_interp = _interp_profile(best_seq, target_n)
        sc_interp = _score(p_interp)
        update_best(p_interp, sc_interp, "P6_interp3072")

        if time.monotonic() < deadline - 60:
            p_best, sc_best = _lp_iterate(p_interp, min(deadline - 50, time.monotonic() + 120),
                                          max_iters=60, label="P6")
            update_best(p_best, _score(p_best), "P6_lp3072")

    # Phase 7: Final polish (last 80s)
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\nPhase 7: Final polish")
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            f_best, sc_best = _lp_iterate(best_seq, deadline, max_iters=200, label="P7")
            update_best(f_best, _score(f_best), "P7_final")
        else:
            f_sp, sc_sp = _subgradient_polish(best_seq, deadline, max_iters=20000)
            update_best(f_sp, _score(f_sp), "P7_sg")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
