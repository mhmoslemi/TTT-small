"""
LP-based iterative optimizer with active set method, combined with
Adam/gradient polishing for minimizing 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix, diags
from scipy import sparse
import time
import cvxpy as cp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    """Assumes sum(a) = 1, returns 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}"""
    n = len(v)
    # Solve for threshold
    # The projection onto the simplex {x >= 0, sum(x) = s} is:
    # x_i = max(v_i - t, 0) where t is chosen so sum = s
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _build_active_constraint_matrix(a, active_lags):
    """Build constraint matrix for active lags only.
    Returns (A, b) where A @ g <= b are the active constraints.
    A has shape (|active_lags|, n)
    """
    n = len(a)
    num_active = len(active_lags)
    rows_data = []
    cols_data = []
    vals_data = []
    
    for idx, k in enumerate(active_lags):
        k = int(k)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            # A[idx, j] = a[k-j] for j in [lo, hi]
            js = np.arange(lo, hi + 1)
            rows_data.append(np.full(hi - lo + 1, idx))
            cols_data.append(js)
            vals_data.append(a[k - hi:k - lo + 1][::-1])
    
    if not rows_data:
        return None, None
    
    rows_arr = np.concatenate(rows_data)
    cols_arr = np.concatenate(cols_data)
    vals_arr = np.concatenate(vals_data)
    
    A = csr_matrix((vals_arr, (rows_arr, cols_arr)), shape=(num_active, n))
    return A, None  # b will be M * ones


def _solve_lp_active(a, M, active_lags, tolerance=0.01, max_iters=20):
    """
    Solve the LP using active set method.
    max sum(g) s.t. conv(a,g)[k] <= M for k in active_lags (initially), g >= 0
    
    Returns the optimal g (or None if infeasible).
    """
    n = len(a)
    
    # Start with given active lags, expand as needed
    current_active = list(active_lags)
    
    for outer_iter in range(max_iters):
        A, _ = _build_active_constraint_matrix(a, current_active)
        if A is None:
            return None
        
        b = M * np.ones(len(current_active))
        
        g = cp.Variable(n)
        objective = cp.Maximize(cp.sum(g))
        constraints = [A @ g <= b, g >= 0]
        
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except Exception:
                try:
                    prob.solve(solver=cp.CLARABEL, verbose=False)
                except Exception:
                    return None
        
        if g.value is None:
            return None
        
        g_val = np.asarray(g.value, dtype=np.float64)
        g_val = np.maximum(g_val, 0)
        
        # Check if any inactive constraint is violated
        c_ag = _full_convolve(a, g_val)
        all_lags = np.arange(2 * n - 1)
        inactive = np.array([k for k in all_lags if k not in set(current_active)])
        
        if len(inactive) > 0:
            violations = c_ag[inactive] - M
            if violations.max() > tolerance:
                # Add the most violated constraint
                worst_k = inactive[np.argmax(violations)]
                current_active.append(int(worst_k))
                continue
        
        return g_val
    
    # If we didn't converge, return the last g
    A, _ = _build_active_constraint_matrix(a, current_active)
    if A is not None:
        b = M * np.ones(len(current_active))
        g = cp.Variable(n)
        objective = cp.Maximize(cp.sum(g))
        constraints = [A @ g <= b, g >= 0]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if g.value is not None:
                return np.maximum(np.asarray(g.value, dtype=np.float64), 0)
        except Exception:
            pass
    return None

def _lp_iteration(a, tolerance=0.01, blend=0.3):
    """
    One LP-based iteration:
    1. Compute M = max(conv(a,a))
    2. Find active lags
    3. Solve LP
    4. Blend
    """
    n = len(a)
    c = _full_convolve(a, a)
    M = float(c.max())
    
    # Active lags: those within tolerance of M
    threshold = M * (1 - tolerance)
    active_lags = np.where(c >= threshold)[0].tolist()
    
    # Solve LP
    g = _solve_lp_active(a, M, active_lags, tolerance=0.01)
    if g is None:
        return a.copy()
    
    # Rescale g to match sum of a
    sum_g = float(g.sum())
    if sum_g < 1e-15:
        return a.copy()
    g = g * (float(a.sum()) / sum_g)
    
    # Blend
    a_new = (1 - blend) * a + blend * g
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    
    return a_new


def _run_lbfgs_continuation(a_init, deadline, rng, max_wall_time=100.0):
    """L-BFGS with logsumexp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    betas = [b * max_c0 for b in beta_starts]

    per_level = max(3.0, max_wall_time / len(betas))

    for beta in betas:
        if time.monotonic() > deadline - 2.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        def obj(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            log_sum = float(np.log(s_))
            return 2.0 * n * (cm_ + beta * log_sum)

        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            W = float(np.dot(w, c_))
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g

        try:
            result = minimize(
                obj, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 3000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_profile(n, width):
    t = np.linspace(0.0, 1.0, n)
    u = t - 0.5
    a = np.exp(-u ** 2 / (2.0 * width ** 2))
    return a / a.sum()


def _make_raised_cosine(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5) * alpha))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _local_search(a_init, deadline, rng, n_trials=5000):
    """Local search with multiplicative perturbations."""
    n = len(a_init)

        if sc < best_score:

            best_a = a_trial.copy()
            a = a_trial.copy()
