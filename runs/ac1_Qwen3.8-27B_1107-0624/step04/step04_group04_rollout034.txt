    k = 1
    while k < n:

    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

"""
Optimized search for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses FFT-based gradient computation, adaptive log-sum-exp smoothing with continuation,
top-k active-set focusing, periodic spectral diversity injection, and multi-scale n search.
Key improvements: verified gradient via FFT, adaptive beta tied to max value,
spectral diversity restarts, and n=8192 exploration.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _compute_weights(c, beta, top_k):
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_norm = (c_top - c_top.max()) / beta
        e = np.exp(c_norm)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[idx[0]] = 1.0
        else:
            w = np.zeros_like(c)
            w[idx] = e / s
    else:
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            idx = int(np.argmax(c))
            w = np.zeros_like(c)
            w[idx] = 1.0
        else:
            w = e / s
    return w


def _grad_obj(a, beta, n, top_k=0):
    """Compute gradient of 2n * smoothed_max(conv(a,a)) w.r.t. a using FFT."""
    c = _full_convolve(a, a)
    w = _compute_weights(c, beta, top_k)
    W = float(np.dot(w, c))

    # h[i] = sum_k w[k] * (a[k-i] + a[i-k]) for valid indices
    # h[i] = (a_rev * w)[i+n-1] + (a * w)[i]
    a_rev = a[::-1]
    N = _next_pow2(3 * n - 2)
    a_rev_ft = np.fft.rfft(a_rev, N)
    w_ft = np.fft.rfft(w, N)
    conv1 = np.fft.irfft(a_rev_ft * w_ft, N)
    h1 = conv1[n - 1:2 * n - 1]

    a_ft = np.fft.rfft(a, N)
    conv2 = np.fft.irfft(a_ft * w_ft, N)
    h2 = conv2[0:n]

    h = h1 + h2
    grad = 2.0 * n * h
    return grad, c, W


def _z_obj_grad(z, beta, n, top_k=0):
    """Objective and gradient in logit space."""
    a = _softmax_fn(z)
    grad_a, c, W = _grad_obj(a, beta, n, top_k)
    dot = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - dot)

    # Compute smoothed max objective value
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        if beta < 1e-12:
            sm = float(c_top.max())
        else:
            cm = float(c_top.max())
            sm = cm + beta * float(np.log(np.sum(np.exp((c_top - cm) / beta))))
    else:
        if beta < 1e-12:
            sm = float(c.max())
        else:
            cm = float(c.max())
            sm = cm + beta * float(np.log(np.sum(np.exp((c - cm) / beta))))
    obj = 2.0 * n * sm
    return obj, grad_z


def _adam_optimize(a_init, deadline, lr=0.05, top_k_init=3, top_k_max=50,
                   beta_frac=0.01, n_max_iter=100000, noise_every=2000, noise_level=0.02):
    """Adam optimizer with adaptive beta and periodic noise injection."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    beta = max(1e-14, beta_frac * max_c0)
    cur_k = top_k_init

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_fn(z)
        grad = _z_obj_grad(z, beta, n, cur_k)[1]

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Adaptive beta: decrease every 500 iterations
        if t % 500 == 0:
            c_curr = _full_convolve(a, a)
            max_c_curr = float(c_curr.max())
            beta = max(1e-16, beta * 0.85)
            beta = max(beta, 0.001 * max_c_curr)
            if t % 500 == 0:
                cur_k = min(cur_k + 1, top_k_max)

        # Periodic noise injection for exploration
        if noise_level > 0 and t % noise_every == 0 and t > 0:
            rng = np.random.default_rng(t)
            noise = noise_level * rng.normal(size=n)
            z = z + noise
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

        # Adaptive learning rate decay
        if t % 5000 == 0:
            lr = max(0.001, lr * 0.95)

    return best_a, best_score


def _lbfgs_optimize(a_init, deadline, top_k=0, n_max_iter_per_beta=200):
    """L-BFGS with beta continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_schedule = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max(max_c0, 1e-10) for b in beta_schedule]

    remaining = max(1.0, deadline - time.monotonic())
    per_level = remaining / len(betas)

    for beta in betas:
        if time.monotonic() > deadline:
            break

        def obj_fn(z_vec):
            return _z_obj_grad(z_vec, beta, n, top_k)[0]

        def grad_fn(z_vec):
            return _z_obj_grad(z_vec, beta, n, top_k)[1]

        level_start = time.monotonic()
        level_deadline = min(deadline, level_start + per_level)

        try:
            def maxiter_fn():
                # Estimate max iterations based on remaining time
                elapsed = time.monotonic() - level_start
                time_per_iter = max(0.001, elapsed / max(1, n_max_iter_per_beta * 0.1))
                remaining_time = level_deadline - time.monotonic()
                return max(10, int(remaining_time / time_per_iter))

            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': n_max_iter_per_beta, 'ftol': 1e-18,
                         'gtol': 1e-12, 'maxls': 20, 'maxfun': n_max_iter_per_beta * 3}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                if np.all(np.isfinite(a_new)) and a_new.sum() > 1e-15:
                    a_new = a_new / a_new.sum()
                    sc = _score_norm_fn(a_new)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _spectral_diversity(a, rng, strength=0.3):
    """Apply a random spectral transformation to inject diversity, then project back to non-negative."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.fft(a, N)
    # Random smooth phase perturbation
    phase_noise = rng.normal(size=N) * strength
    # Make it smooth by keeping only low frequencies
    smooth_mask = np.zeros(N)
    bw = max(1, N // 100)
    smooth_mask[:bw] = 1.0
    smooth_mask[N - bw:] = 1.0
    smooth_mask[bw:N - bw] = 0.1
    phase_noise *= smooth_mask
    A_new = A * np.exp(1j * phase_noise)
    a_new = np.fft.ifft(A_new).real[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def _make_exp_profile(n, alpha):
    r = 1.0 - alpha
    a = r ** np.arange(n)
    return a / a.sum()


def _make_cosine_profile(n, f, amp):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_random_profile(n, seed):
    rng = np.random.default_rng(seed)
    a = rng.exponential(size=n)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

        dl = min(phase2_end, time.monotonic() + remaining * 0.7)
        print(f"  Adam main (n={len(best_seq)}), budget={remaining*0.7:.1f}s")
        p_best, sc_best = _adam_optimize(best_seq, dl, lr=0.04, top_k_init=2,
                                         top_k_max=40, beta_frac=0.01,
                                         n_max_iter=200000, noise_every=3000, noise_level=0.03)
        update_best(p_best, sc_best, "adam_main")

        if time.monotonic() < phase2_end - 30:
            dl2 = min(phase2_end - 10, time.monotonic() + (phase2_end - time.monotonic()) * 0.6)
            p_best2, sc_best2 = _adam_optimize(best_seq, dl2, lr=0.02, top_k_init=5,
                                                top_k_max=60, beta_frac=0.005,
                                                n_max_iter=100000, noise_every=5000, noise_level=0.015)
            update_best(p_best2, sc_best2, "adam_fine")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Spectral diversity + re-optimize ---
    print("=== Phase 3: Spectral diversity restarts ===")
    phase3_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 5
        per_restart = min(60.0, (phase3_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 15:
                break
            sub_rng = np.random.default_rng(int(seed) + 100 + i)

            # Apply spectral diversity
            p_div = _spectral_diversity(best_seq, sub_rng, strength=0.2 + 0.1 * i)
            sc_div = _score_norm_fn(p_div)
            update_best(p_div, sc_div, f"spectral_{i}")

            # Quick Adam on the diversified version
            dl = min(phase3_end - 10, time.monotonic() + per_restart * 0.7)
            p_opt, sc_opt = _adam_optimize(p_div, dl, lr=0.03, top_k_init=3,
                                            top_k_max=30, beta_frac=0.005,
                                            n_max_iter=30000, noise_every=0, noise_level=0)
            update_best(p_opt, sc_opt, f"spec_adam_{i}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Try alternate n ---
    print("=== Phase 4: Alternate length ===")
    phase4_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        n_curr = len(best_seq)
        target_n = n_alt if n_curr != n_alt else n_base

        p_interp = _interp_profile(best_seq, target_n)
        sc_interp = _score_norm_fn(p_interp)
        update_best(p_interp, sc_interp, f"interp_n{target_n}")

        if time.monotonic() < phase4_end - 60:
            remaining = phase4_end - time.monotonic()
            dl = min(phase4_end - 20, time.monotonic() + min(180.0, remaining * 0.7))
            p_best, sc_best = _adam_optimize(p_interp, dl, lr=0.03, top_k_init=3,
                                              top_k_max=40, beta_frac=0.008,
                                              n_max_iter=80000, noise_every=4000, noise_level=0.02)
            update_best(p_best, sc_best, f"adam_n{target_n}")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Fine polish ---
    print("=== Phase 5: Final polish ===")
    if time.monotonic() < deadline and best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            dl = min(deadline - 10, time.monotonic() + min(120.0, remaining * 0.6))
            p_best, sc_best = _adam_optimize(best_seq, dl, lr=0.01, top_k_init=10,
                                              top_k_max=80, beta_frac=0.002,
                                              n_max_iter=60000, noise_every=0, noise_level=0)
            update_best(p_best, sc_best, "final_adam")

        if time.monotonic() < deadline - 20:
            dl2 = min(deadline - 5, time.monotonic() + min(80.0, deadline - time.monotonic() - 5))
            p_best2, sc_best2 = _lbfgs_optimize(best_seq, dl2, top_k=5, n_max_iter_per_beta=100)
            update_best(p_best2, sc_best2, "final_lbfgs")

    # --- Final validation ---
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = _make_exp_profile(4096, 0.003)
            final_score = _score_fn(final_seq)

        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")

        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 16384:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  evaluate_sequence failed: {e}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_exp_profile(4096, 0.003)
    print(f"  FALLBACK: n=4096, score={_score_norm_fn(fallback):.10f}")
    return [float(x) for x in fallback]
