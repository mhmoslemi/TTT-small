def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    n = len(a)
    s = float(a.sum())

        return np.inf

    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)

def _precompute_structure(n):
    rows = []
    cols = []
    for j in range(n):
        length = min(n, 2 * n - 1 - j)
        rows.extend(range(j, j + length))
        cols.extend([j] * length)
    return np.array(rows, dtype=np.int64), np.array(cols, dtype=np.int64)


def _build_A_dense(a, rows, cols):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    A[rows, cols] = a[rows - cols]
    return A


def _lp_solve_minmax(A, n):
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.hstack([np.ones((1, n)), np.zeros(1)])
    b_eq = np.array([1.0])

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return float(res.x[-1]), np.maximum(res.x[:n], 0.0)
    return None, 1.0

def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)

        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w


def _cross_correlate(w, a):
    n = len(a)

    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]

def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()

                a_fp, sc_fp = _lp_fixed_point(best_seq, time.monotonic() + per, max_iters=30)
                if sc_fp < best_score:
                    best_seq = a_fp.copy()
                    best_score = sc_fp
                    print(f"  BEST: {best_score:.10f} n={best_seq.size} [final_LP]")
            else:
                a_fp, sc_fp = _adam_smooth_max(best_seq, time.monotonic() + per, rng,
                                               lr=0.02, beta_decay=0.97, no_imp_limit=100)
                if sc_fp < best_score:
                    best_seq = a_fp.copy()
                    best_score = sc_fp
                    print(f"  BEST: {best_score:.10f} n={best_seq.size} [final_Adam]")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile(512, 'linear', 2.0, 0, 0)

        exact_sc = _score(final)
        print(f"\nFINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
            else:
                print("EVAL returned inf, using fallback")
                fb = _make_profile(512, 'linear', 2.0, 0, 0)
                return [float(x) for x in fb]
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    fb = _make_profile(512, 'linear', 2.0, 0, 0)
    print("FALLBACK n=512")
    return [float(x) for x in fb]
