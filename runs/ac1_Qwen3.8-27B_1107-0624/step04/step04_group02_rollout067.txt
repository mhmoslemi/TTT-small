import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v, target=1.0):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    if not np.any(cond):
        return np.full(n, target / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _ls_value_grad(a, beta):
    n = len(a)
    c = _full_convolve(a, a)

    if s < 1e-300:
        val = cm
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0

    Af = np.fft.fft(a, Nf)
    grad = 2.0 * np.fft.ifft(Wf * np.conj(Af)).real[:n]
    return val, grad, w, cm

        x_init = a_init[:m].copy()
        x_init = np.maximum(x_init, 1e-15)
        z = np.log(x_init / x_init.sum() * 2.0)
        z = z - z.mean()
    else:
        z = np.log(np.maximum(a_init, 1e-15))
        z = z - z.mean()
        m = n

    def _z_to_a(zv):
        xv = _softmax_fn(zv)
        if use_symmetry and len(a_init) % 2 == 0:
            av = np.zeros(len(a_init))
            av[:m] = xv
            av[m:] = xv[::-1]
            av = av / av.sum()
        else:
            av = xv
        return av

    per_beta_time = max(5.0, (deadline - time.monotonic()) / len(beta_schedule))

    for beta in beta_schedule:
        if time.monotonic() > deadline - 3.0:
            break
        bdl = min(deadline - 3.0, time.monotonic() + per_beta_time)

        def obj(zv):
            av = _z_to_a(zv)
            c = _full_convolve(av, av)
            cm = float(c.max())
            b = min(beta, 500.0 / max(cm, 1e-15))
            cs = (c - cm) / b
            ev = np.exp(cs)
            s = ev.sum()
            if s < 1e-300:
                return 2.0 * n * cm
            return 2.0 * n * (cm + b * float(np.log(s)))

        def grad_fn(zv):
            av = _z_to_a(zv)
            c = _full_convolve(av, av)
            cm = float(c.max())
            b = min(beta, 500.0 / max(cm, 1e-15))
            cs = (c - cm) / b
            ev = np.exp(cs)
            s = ev.sum()
            if s < 1e-300:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = ev / s
            Nf = _next_pow2(3 * n)
            Wf = np.fft.fft(w, Nf)
            Af = np.fft.fft(av, Nf)
            g_a = 2.0 * np.fft.ifft(Wf * np.conj(Af)).real[:n]
            if use_symmetry and n % 2 == 0:
                g_x = g_a[:m] + g_a[m:][::-1]
                xv = _softmax_fn(zv)
                grad_z = xv * (g_x - float(np.dot(xv, g_x)))
            else:
                grad_z = av * (g_a - float(np.dot(av, g_a)))
            grad_z = 2.0 * n * grad_z
            if not np.all(np.isfinite(grad_z)):
                return np.zeros_like(zv)
            return grad_z

        try:
            res = minimize(obj, z, jac=grad_fn, method='L-BFGS-B',
                           options={'maxiter': 3000, 'ftol': 1e-22, 'gtol': 1e-14,
                                    'maxls': 100, 'maxfun': 20000})
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_new = _z_to_a(res.x)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = res.x.copy()
        except Exception:
            pass

    return best_a, best_score

def _fista_simplex(a_init, deadline, beta_schedule):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    for beta in beta_schedule:
        if time.monotonic() > deadline - 2.0:
            break
        a = a_init.copy()
        a_prev = a.copy()
        t_k = 1.0
        c0 = _full_convolve(a, a)
        cm0 = float(c0.max())
        step = beta / (4.0 * n * max(a.max(), 1e-10))
        step = min(step, 1.0)

        max_iter = 3000
        for it in range(max_iter):
            if time.monotonic() > deadline - 2.0:
                break
            momentum = (t_k - 1.0) / (t_k + 1.0)
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy

            val, grad, w, cm = _ls_value_grad(y, beta)
            a_new = y - step * grad
            a_new = _project_simplex(a_new, 1.0)

            if it % 50 == 0:
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                a_init = a_new.copy()

            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            a_prev = a.copy()
            a = a_new.copy()
            t_k = t_new

            if it % 500 == 0 and it > 0:
                step *= 0.95

        a_init = best_a.copy()

    return best_a, best_score


def _local_search(a_init, deadline, rng, n_trials=10000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break
        bs = int(rng.integers(2, min(40, n // 2)))
        start = int(rng.integers(0, n - bs))
        a_trial = a.copy()
        if rng.random() < 0.5:
            pert = rng.uniform(0.92, 1.08, size=bs)
            a_trial[start:start + bs] *= pert
        else:
            delta = rng.normal(0, 0.02, size=bs)
            a_trial[start:start + bs] += delta * a_trial[start:start + bs].mean()
        a_trial = np.maximum(a_trial, 1e-15)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial /= s

        sc = _score_norm(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
        elif rng.random() < 0.03 and sc < best_score * 1.0005:
            a = a_trial.copy()

    return best_a, best_score


def _symmetric_local_search(a_init, deadline, rng, n_trials=8000):
    n = len(a_init)
    if n % 2 != 0:
        return _local_search(a_init, deadline, rng, n_trials)
    m = n // 2
    x = a_init[:m].copy()
    best_x = x.copy()
    a_full = np.zeros(n)
    a_full[:m] = x
    a_full[m:] = x[::-1]
    best_score = _score_norm(a_full)

    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break
        bs = int(rng.integers(2, min(30, m // 2)))
        start = int(rng.integers(0, m - bs))
        x_trial = x.copy()
        pert = rng.uniform(0.93, 1.07, size=bs)
        x_trial[start:start + bs] *= pert
        x_trial = np.maximum(x_trial, 1e-15)
        s = x_trial.sum()
        if s < 1e-15:
            continue
        x_trial = x_trial / (2.0 * x_trial.sum())
        a_trial = np.zeros(n)
        a_trial[:m] = x_trial
        a_trial[m:] = x_trial[::-1]

        sc = _score_norm(a_trial)
        if sc < best_score:
            best_score = sc
            best_x = x_trial.copy()
            x = x_trial.copy()
            a_full = a_trial.copy()
        elif rng.random() < 0.03 and sc < best_score * 1.0005:
            x = x_trial.copy()

    a_out = np.zeros(n)
    a_out[:m] = best_x
    a_out[m:] = best_x[::-1]
    a_out /= a_out.sum()
    return a_out, _score_norm(a_out)


def _make_bathtub(n, depth):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - depth * np.cos(np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosh(n, k):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = np.cosh(k * u)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear(n, slope):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + slope * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quad(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian(n, width):
    t = np.linspace(0.0, 1.0, n)
    u = t - 0.5
    a = np.exp(-u * u / (2.0 * width * width))
    return a / a.sum()


def _make_raised_cos(n, alpha):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5) * alpha))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power(n, p):
    t = np.linspace(0.0, 1.0, n)
    u = np.abs(2.0 * t - 1.0)
    a = (1.0 - u) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _enforce_symmetry(a):
    n = len(a)
    if n % 2 == 0:
        m = n // 2
        x = (a[:m] + a[m:][::-1]) / 2.0
        out = np.zeros(n)
        out[:m] = x
        out[m:] = x[::-1]
    else:
        m = n // 2
        x = (a[:m] + a[m+1:][::-1]) / 2.0
        out = np.zeros(n)
        out[:m] = x
        out[m] = a[m]
        out[m+1:] = x[::-1]
    out = np.maximum(out, 1e-15)
    out /= out.sum()
    return out

    best_seq = None

        if parent_raw.size > 0:

            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        parent = None

    if parent is not None:
        sc_p = _score_norm(parent)
        print(f"Parent: n={n_parent}, score={sc_p:.10f}")
        best_seq = parent.copy()
        best_score = sc_p
    else:
        parent = _make_cosh(n_parent, 2.0)
        best_seq = parent.copy()
        best_score = _score_norm(parent)
        print(f"No parent, using cosh profile n={n_parent}, score={best_score:.10f}")

    # --- Phase 1: Diverse initialization (40s) ---
    phase1_end = min(deadline, t_start + 40.0)
    candidates = [(best_score, best_seq.copy(), n_parent, "parent")]

    profile_specs = [
        ('cosh', {'k': 1.5}), ('cosh', {'k': 2.5}), ('cosh', {'k': 3.5}),

        ('gaussian', {'width': 0.2}), ('gaussian', {'width': 0.35}),
        ('raised_cos', {'alpha': 1.0}), ('raised_cos', {'alpha': 2.0}),
        ('power', {'p': 1.5}), ('power', {'p': 3.0}), ('power', {'p': 5.0}),

            break
        kind = kws[0]
        params = kws[1]

            p = _make_cosh(n_parent, params['k'])
        elif kind == 'bathtub':
            p = _make_bathtub(n_parent, params['depth'])
        elif kind == 'linear':
            p = _make_linear(n_parent, params['slope'])
        elif kind == 'quad':
            p = _make_quad(n_parent, params['a1'], params['a2'])
        elif kind == 'gaussian':
            p = _make_gaussian(n_parent, params['width'])
        elif kind == 'raised_cos':
            p = _make_raised_cos(n_parent, params['alpha'])
        elif kind == 'power':
            p = _make_power(n_parent, params['p'])
        else:
            continue
        sc = _score_norm(p)
        candidates.append((sc, p, n_parent, f"{kind}_{kws[1]}"))
        if sc < best_score:
            best_score = sc
            best_seq = p.copy()
