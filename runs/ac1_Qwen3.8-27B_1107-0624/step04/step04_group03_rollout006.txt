    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _objective_smooth(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _gradient_smooth(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _hann_window(n):
    t = np.arange(n) / (n - 1)
    return 0.5 * (1 - np.cos(2 * np.pi * t)) + 1e-12


def _hamming_window(n):
    t = np.arange(n) / (n - 1)
    return 0.54 - 0.46 * np.cos(2 * np.pi * t) + 1e-12


def _blackman_window(n):
    t = np.arange(n) / (n - 1)
    return 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t) + 1e-12


def _gaussian_window(n, sigma_frac=0.3):
    t = np.linspace(0, 1, n)
    center = 0.5
    sigma = sigma_frac
    return np.exp(-0.5 * ((t - center) / sigma) ** 2) + 1e-12


def _tukey_window(n, alpha=0.5):
    t = np.arange(n) / (n - 1)
    a = np.ones(n)
    left = t < (1 - alpha) / 2
    right = t > (1 + alpha) / 2
    a[left] = 0.5 * (1 + np.cos(np.pi * (2 * t[left] / (1 - alpha) - 1)))
    a[right] = 0.5 * (1 + np.cos(np.pi * (2 * t[right] / (1 - alpha) - 1)))
    return a + 1e-12


def _bimodal_window(n, gap_frac=0.1):
    t = np.linspace(0, 1, n)
    a = np.exp(-10 * (t - 0.25) ** 2 / (gap_frac * gap_frac)) + np.exp(-10 * (t - 0.75) ** 2 / (gap_frac * gap_frac))
    return a + 1e-12


def _plateau_window(n, width=0.6):
    t = np.linspace(0, 1, n)
    a = np.ones(n)
    edge = (1 - width) / 2
    trans = 0.1
    fade_in = t < edge
    fade_out = t > 1 - edge
    mid_start = edge - trans
    mid_end = 1 - edge + trans
    a[fade_in] = 0.5 * (1 + np.cos(np.pi * (t[fade_in] - mid_start) / trans))
    a[fade_out] = 0.5 * (1 + np.cos(np.pi * (t[fade_out] - mid_end) / trans))
    return a + 1e-12


def _linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    return np.maximum(a, 1e-12)


def _quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    return np.maximum(a, 1e-12)


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _adam_optimize(a_init, deadline, lr=0.05, beta_frac=0.01, max_iter=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-15, beta_frac * max_c)

    patience = 0
    best_since = best_score

    while time.monotonic() < deadline:
        t += 1
        if max_iter is not None and t > max_iter:
            break

        a = _softmax_fn(z)
        gr = _gradient_smooth(z, beta, n)
        if not np.all(np.isfinite(gr)):
            continue

        m = b1 * m + (1 - b1) * gr
        v = b2 * v + (1 - b2) * (gr * gr)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        if t % 200 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
                best_since = sc
                patience = 0
            else:
                patience += 1
                if patience > 20:
                    beta = max(1e-15, beta * 0.7)
                    c_new = _full_convolve(a_curr, a_curr)
                    max_c = float(c_new.max())
                    beta = max(beta, 1e-15 * max_c)
                    patience = 0

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _nesterov_optimize(a_init, deadline, beta_fracs, lr_init=0.02, mom_init=0.9):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())

    lr = lr_init
    mom = mom_init
    t = 0

    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-15, beta_frac * max_c)

        while time.monotonic() < deadline:
            t += 1
            grad_y = _gradient_smooth(y, beta, n)
            if not np.all(np.isfinite(grad_y)):
                grad_y = _gradient_smooth(z, beta, n)

            z_new = z - lr * grad_y
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -60, 60)

            y_new = z_new + mom * (z_new - z)
            y_new = y_new - y_new.mean()

            z = z_new
            y = y_new

            if t % 50 == 0:
                a_curr = _softmax_fn(z)
                sc = _score_norm_fn(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()

            if t % 200 == 0:
                mom = min(0.999, mom + 0.0003)

        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        max_c = float(c.max())
        lr = max(0.002, lr * 0.6)

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _lbfgs_run(z_init, beta, n, max_iter=500):
    result = minimize(
        _objective_smooth, z_init.copy(), args=(beta, n),
        method='L-BFGS-B', jac=_gradient_smooth,
        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12,
                 'maxls': 50, 'maxfun': 3000}
    )
    if result.x is not None and np.all(np.isfinite(result.x)):
        return result.x
    return z_init


def _run_lbfgs_continuation(a_init, deadline):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    betas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
             5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]
    betas = [b * max_c0 for b in betas]

    n_levels = len(betas)
    per_level = max(2.0, (deadline - time.monotonic()) / n_levels)

    for i, beta in enumerate(betas):
        if time.monotonic() >= deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level)
        try:
            z = _lbfgs_run(z, beta, n, max_iter=200)
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _multires_optimize(a_init, deadline, n_target):
    """Optimize at lower resolution, interpolate up, then optimize at target."""
    n_curr = len(a_init)
    if n_curr == n_target:
        return a_init

    # Downsample
    n_low = max(64, n_target // 4)
    a_low = _interp_profile(a_init, n_low)

    # Optimize at low resolution
    low_dl = min(deadline - 60, time.monotonic() + 30)
    if low_dl > time.monotonic() + 5:
        a_low, _ = _lbfgs_continuation(a_low, low_dl)

    # Interpolate up
    a_mid = _interp_profile(a_low, n_target)
    return a_mid


def _jittered_adam(a_init, deadline, lr=0.03, beta_frac=0.005, jitter_interval=500, jitter_scale=0.02):
    """Adam with periodic random jitter to escape plateaus."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    rng = np.random.default_rng(12345)

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-15, beta_frac * max_c)

    while time.monotonic() < deadline:
        t += 1

        # Periodic jitter
        if t % jitter_interval == 0 and t > 0:
            z = z + jitter_scale * rng.normal(size=n)
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            jitter_scale *= 0.8

        a = _softmax_fn(z)
        gr = _gradient_smooth(z, beta, n)
        if not np.all(np.isfinite(gr)):
            continue

        m = b1 * m + (1 - b1) * gr
        v = b2 * v + (1 - b2) * (gr * gr)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        if t % 300 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            # Adapt beta
            c_new = _full_convolve(a_curr, a_curr)
            max_c = float(c_new.max())
            beta = max(1e-15, beta * 0.85)

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _active_set_subgradient(a_init, deadline, lr_init=0.003):
    """Subgradient method focusing on active (near-max) lags with averaging."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    lr = lr_init
    momentum = np.zeros(n)
    mom_coeff = 0.85
    avg_grad = np.zeros(n)
    avg_count = 0.0

    for it in range(100000):
        if time.monotonic() > deadline:
            break

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Active set: lags within 0.5% of max
        threshold = cm * 0.9995
        active = c >= threshold
        n_active = np.sum(active)
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        # Polyak-style averaging of gradients
        avg_count += 1.0
        alpha = 1.0 / avg_count
        avg_grad = (1 - alpha) * avg_grad + alpha * grad

        # Use averaged gradient with momentum
        lr_t = lr / (1 + it * 0.001)
        momentum = mom_coeff * momentum - lr_t * avg_grad
        z = z + momentum
        z = z - z.mean()
        z = np.clip(z, -60, 60)

        if it % 200 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: diverse initialization -> L-BFGS continuation ->
    Nesterov/Adam fine polishing -> jittered restarts -> multi-length -> final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        print(f"Parent: n={n_parent}, score={_score_fn(parent):.10f}")
    else:
        n_parent = 4096
        parent = _hann_window(n_parent)
        parent = parent / parent.sum()
        print(f"No parent, using Hann window n={n_parent}")

    # --- Phase 0: Diverse initialization (60s) ---
    phase0_end = min(deadline, t_start + 60.0)
    n_base = n_parent

    candidates = []
    sc = _score_norm_fn(parent)
    candidates.append((sc, parent.copy(), "parent"))
    update_best(parent, sc, "parent")

    diverse_starts = []
    diverse_starts.append(("hann", _hann_window(n_base)))
    diverse_starts.append(("hamming", _hamming_window(n_base)))
    diverse_starts.append(("blackman", _blackman_window(n_base)))
    diverse_starts.append(("gauss", _gaussian_window(n_base, 0.25)))
    diverse_starts.append(("gauss_wide", _gaussian_window(n_base, 0.4)))
    diverse_starts.append(("tukey", _tukey_window(n_base, 0.5)))
    diverse_starts.append(("tukey_narrow", _tukey_window(n_base, 0.3)))
    diverse_starts.append(("bimodal", _bimodal_window(n_base, 0.15)))
    diverse_starts.append(("plateau", _plateau_window(n_base, 0.7)))
    diverse_starts.append(("linear1", _linear_profile(n_base, 1.0)))
    diverse_starts.append(("linear2", _linear_profile(n_base, 2.0)))
    diverse_starts.append(("quad1", _quadratic_profile(n_base, 0.5, 0.3)))
    diverse_starts.append(("quad2", _quadratic_profile(n_base, -0.5, 0.2)))

    # Smooth random
    for r_i in range(3):
        freq = np.array([1, 2, 3, 5])
        coeffs = rng.normal(size=len(freq))
        t = np.linspace(0, 1, n_base)
        a_r = 1.0
        for f, c in zip(freq, coeffs):
            a_r += 0.3 * c * np.cos(2 * np.pi * f * t)
        a_r = np.maximum(a_r, 1e-10)
        diverse_starts.append((f"rand_smooth{r_i}", a_r))

    for name, prof in diverse_starts:
        if time.monotonic() > phase0_end - 5:
            break
        prof = prof / prof.sum()
        sc = _score_norm_fn(prof)
        candidates.append((sc, prof.copy(), name))
        update_best(prof, sc, f"init_{name}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f} (n={len(best_seq)})")

    # --- Phase 1: L-BFGS continuation on best candidate (350s) ---
    phase1_end = min(deadline, t_start + 410.0)
    if best_seq is not None and time.monotonic() < phase1_end:
        p_best, sc_best = _run_lbfgs_continuation(best_seq, phase1_end)
        update_best(p_best, sc_best, "lbfgs_cont")

    # Also try L-BFGS on second-best if different and time allows
    if time.monotonic() < phase1_end - 60:
        for sc_c, p_c, name_c in candidates[1:3]:
            if time.monotonic() > phase1_end - 30:
                break
            if len(p_c) != len(best_seq):
                continue
            per_time = min(60.0, phase1_end - time.monotonic() - 10)
            if per_time > 20:
                z_tmp = np.log(np.maximum(p_c, 1e-15))
                z_tmp = z_tmp - z_tmp.mean()
                c_tmp = _full_convolve(p_c, p_c)
                max_c_tmp = float(c_tmp.max())
                betas_short = [b * max_c_tmp for b in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]]
                per_level = per_time / len(betas_short)
                for beta in betas_short:
                    if time.monotonic() >= phase1_end - 10:
                        break
                    level_dl = min(phase1_end - 10, time.monotonic() + per_level)
                    try:
                        z_tmp = _lbfgs_run(z_tmp, beta, len(p_c), max_iter=100)
                    except Exception:
                        break
                a_tmp = _softmax_fn(z_tmp)
                sc_tmp = _score_norm_fn(a_tmp)
                update_best(a_tmp, sc_tmp, f"lbfgs_{name_c}")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Nesterov with long beta schedule (300s) ---
    phase2_end = min(deadline, t_start + 710.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        beta_fracs = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                      5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7]
        p_best, sc_best = _nesterov_optimize(best_seq, phase2_end, beta_fracs, lr_init=0.012, mom_init=0.9)
        update_best(p_best, sc_best, "nesterov_long")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Jittered Adam (300s) ---
    phase3_end = min(deadline, t_start + 1010.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        per_time = min(250.0, phase3_end - time.monotonic() - 10)
        if per_time > 30:
            p_best, sc_best = _jittered_adam(best_seq, min(phase3_end - 10, time.monotonic() + per_time),
                                             lr=0.025, beta_frac=0.003, jitter_interval=400, jitter_scale=0.03)
            update_best(p_best, sc_best, "jittered_adam")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Diverse restarts with optimization (300s) ---
    phase4_end = min(deadline, t_start + 1310.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        n_best = len(best_seq)
        n_restarts = 6
        per_restart = max(20.0, (phase4_end - time.monotonic()) / max(1, n_restarts))

        restart_configs = [
            (0.03, 0.9), (0.05, 0.85), (0.1, 0.8),
            (0.02, 0.95), (0.07, 0.85), (0.15, 0.75),
        ]

        for i in range(min(n_restarts, len(restart_configs))):
            if time.monotonic() > phase4_end - 15:
                break
            noise_level, mom_l = restart_configs[i]
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_level * rng.normal(size=n_best)
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            per_dl = min(phase4_end - 10, time.monotonic() + per_restart)
            beta_fracs_short = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]
            p_best, sc_best = _nesterov_optimize(p_start, per_dl, beta_fracs_short,
                                                  lr_init=0.01, mom_init=mom_l)
            update_best(p_best, sc_best, f"restart{i}_nl{noise_level}")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multi-length exploration (200s) ---
    phase5_end = min(deadline, t_start + 1510.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        n_curr = len(best_seq)

        # Try larger n
        if n_curr <= 4096:
            target_n = min(8192, n_curr * 2)
            target_n = _next_pow2(target_n)
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{target_n}")

            if time.monotonic() < phase5_end - 60:
                per_time = min(100.0, phase5_end - time.monotonic() - 30)
                if per_time > 30:
                    p_best, sc_best = _run_lbfgs_continuation(p_interp, min(phase5_end - 20, time.monotonic() + per_time))
                    update_best(p_best, sc_best, f"lbfgs_n{target_n}")

        # Try smaller n
        if n_curr >= 4096:
            target_n = max(1024, n_curr // 2)
            target_n = _next_pow2(target_n)
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{target_n}_sm")

            if time.monotonic() < phase5_end - 40:
                per_time = min(60.0, phase5_end - time.monotonic() - 20)
                if per_time > 20:
                    p_best, sc_best = _run_lbfgs_continuation(p_interp, min(phase5_end - 10, time.monotonic() + per_time))
                    update_best(p_best, sc_best, f"lbfgs_n{target_n}_sm")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Active set subgradient (150s) ---
    phase6_end = min(deadline, t_start + 1660.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        per_time = min(120.0, phase6_end - time.monotonic() - 5)
        if per_time > 20:
            p_best, sc_best = _active_set_subgradient(best_seq, min(phase6_end - 5, time.monotonic() + per_time),
                                                       lr_init=0.002)
            update_best(p_best, sc_best, "active_set")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final fine polishing (remaining time) ---
    phase7_end = deadline
    if best_seq is not None and time.monotonic() < phase7_end:
        remaining = phase7_end - time.monotonic()

        if remaining > 40:
            per_time = min(60.0, remaining - 15)
            beta_fracs_final = [1e-6, 5e-7, 2e-7, 1e-7, 5e-8]
            p_best, sc_best = _nesterov_optimize(best_seq, min(phase7_end - 15, time.monotonic() + per_time),
                                                  beta_fracs_final, lr_init=0.002, mom_init=0.97)
            update_best(p_best, sc_best, "final_nesterov")

        if time.monotonic() < phase7_end - 15:
            per_time = min(40.0, phase7_end - time.monotonic() - 5)
            if per_time > 10:
                p_best, sc_best = _adam_optimize(best_seq, min(phase7_end - 5, time.monotonic() + per_time),
                                                  lr=0.003, beta_frac=0.0005)
                update_best(p_best, sc_best, "final_adam")

        if time.monotonic() < phase7_end - 10:
            per_time = min(30.0, phase7_end - time.monotonic() - 3)
            if per_time > 10:
                p_best, sc_best = _jittered_adam(best_seq, min(phase7_end - 3, time.monotonic() + per_time),
                                                  lr=0.005, beta_frac=0.001, jitter_interval=200, jitter_scale=0.01)
                update_best(p_best, sc_best, "final_jitter")

    print(f"Phase 7 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _hann_window(4096)
            final_seq = final_seq / final_seq.sum()

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned non-finite, using internal: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _hann_window(4096)
    fallback = fallback / fallback.sum()
    print(f"FALLBACK: n=4096, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
