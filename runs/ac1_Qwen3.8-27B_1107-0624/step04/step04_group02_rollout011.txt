Multi-strategy optimizer combining L-BFGS logsumexp continuation with LP-based 
peak-reduction iterations for minimizing 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
from scipy.optimize import minimize
from scipy import sparse
from scipy.optimize import linprog
import time
import math


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1 (normalized score)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _run_lbfgs_continuation(a_init, deadline, rng, beta_scale=1.0, max_wall_time=None):
    """L-BFGS with logsumexp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max_c0 * beta_scale for b in beta_starts]

    if max_wall_time is not None:
        per_level = max(3.0, max_wall_time / len(betas))
    else:
        per_level = 30.0

    for beta in betas:
        if time.monotonic() > deadline - 2.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        def obj(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            log_sum = float(np.log(s_))
            return 2.0 * n * (cm_ + beta * log_sum)

        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            W = float(np.dot(w, c_))
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g

        try:
            result = minimize(
                obj, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 2000, 'ftol': 1e-20, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 8000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _build_conv_constraint_matrix(f, n):
    """Build sparse matrix A where A[k,j] = f[j] if 0 <= k-j <= n-1."""
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(f[j])
    A = sparse.csc_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_peak_reduction(f, M, max_iter_lp=1):
    """
    Solve LP: maximize sum(g) subject to (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g (or None if infeasible).
    """
    n = len(f)
    # Build constraint matrix
    A = _build_conv_constraint_matrix(f, n)
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: A @ g <= M * ones(2n-1)
    b_ub = np.full(2 * n - 1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.status == 0:
            g = result.x
            if g is not None and np.all(g >= -1e-12) and np.sum(g) > 1e-12:
                g = np.maximum(g, 0)
                return g
    except Exception as e:
        print(f"  LP failed: {e}")
    
    return None


def _lp_iteration(f, t_blend=0.3, max_lp_iterations=5):
    """
    Perform LP-based peak reduction iterations.
    Returns improved sequence and score.
    """
    n = len(f)
    f_curr = f.copy()
    f_curr = f_curr / f_curr.sum()
    best_f = f_curr.copy()
    best_score = _score_norm_fn(f_curr)
    
    for it in range(max_lp_iterations):
        c_curr = _full_convolve(f_curr, f_curr)
        M = float(c_curr.max())
        
        g = _lp_peak_reduction(f_curr, M)
        if g is None:
            break
        
        g_norm = g / g.sum()
        
        # Blend
        f_new = (1 - t_blend) * f_curr + t_blend * g_norm
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-12:
            break
        f_new = f_new / s
        
        sc_new = _score_norm_fn(f_new)
        if sc_new < best_score:
            best_score = sc_new
            best_f = f_new.copy()
            f_curr = f_new.copy()
        else:
            # Try smaller blend
            f_new2 = (1 - t_blend/2) * f_curr + (t_blend/2) * g_norm
            f_new2 = np.maximum(f_new2, 0)
            f_new2 = f_new2 / f_new2.sum()
            sc_new2 = _score_norm_fn(f_new2)
            if sc_new2 < best_score:
                best_score = sc_new2
                best_f = f_new2.copy()
                f_curr = f_new2.copy()
            else:
                break
    
    return best_f, best_score


def _make_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    
    profiles = []
    names = []
    
    # Linear profiles
    for slope in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0]:
        a = 1.0 + slope * u
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
        names.append(f"lin_{slope}")
    
    # Quadratic
    for a2 in [0.3, 0.5, 1.0, -0.3]:
        a = 1.0 + 0.8 * u + a2 * u**2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
        names.append(f"quad_{a2}")
    
    # Gaussian
    for w in [0.15, 0.2, 0.25, 0.3, 0.4]:
        a = np.exp(-u**2 / (2 * w**2))
        profiles.append(a / a.sum())
        names.append(f"gauss_{w}")
    
    # Raised cosine
    for alpha in [0.5, 0.7, 1.0, 1.5]:
        a = 0.5 * (1 + np.cos(np.pi * u * alpha))
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
        names.append(f"rccos_{alpha}")
    
    # Power / Super-Gaussian
    for p in [1.5, 2.0, 3.0, 4.0, 6.0]:
        a = (1.0 - np.abs(u)) ** p
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
        names.append(f"pow_{p}")
    
    # Two-hump
    for sep in [0.2, 0.3, 0.4]:
        a = np.exp(-((t - (0.5 - sep))**2) / (2*0.1**2)) + np.exp(-((t - (0.5 + sep))**2) / (2*0.1**2))
        profiles.append(a / a.sum())
        names.append(f"twohump_{sep}")
    
    # Modulated
    for freq in [5, 10, 20]:
        a = (1 + 0.3 * np.cos(2 * np.pi * freq * t)) * np.exp(-u**2 / (2 * 0.3**2))
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
        names.append(f"mod_{freq}")
    
    return list(zip(names, profiles))


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

    momentum = np.zeros(n)
    beta_mom = 0.9
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break
        
        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))
        
        # Subgradient of max at k_star: d(a*a)[k_star]/da_j = a[k_star - j] (reversed)
        # (a*a)[k] = sum_j a[j] a[k-j], d/da_j = a[k-j] + a[j-k] (but for our case just a[k-j] if in range)
        grad = np.zeros(n)
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        for j in range(lo, hi + 1):
            grad[j] = a[k_star - j]
        
        # Projected subgradient with momentum
        momentum = beta_mom * momentum + (1 - beta_mom) * grad
        a = a - lr * momentum
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        if it % 5 == 0:
            sc = _score_norm_fn(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        lr = lr * decay
        
        # Adaptive: if stuck, try a different peak lag
        if it % 100 == 50:
            top3 = np.argsort(c)[-3:]
            k_choice = top3[rng_local.integers(3)]
            grad2 = np.zeros(n)
            lo2 = max(0, k_choice - (n - 1))
            hi2 = min(n - 1, k_choice)
            for j in range(lo2, hi2 + 1):
                grad2[j] = a[k_choice - j]
            momentum = 0.5 * momentum + 0.5 * grad2
    
    return best_a, best_score

import numpy as np
from scipy.optimize import minimize
from scipy import sparse
from scipy.optimize import linprog
import time

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lbfgs_obj_grad_factory(n, beta):
    """Create objective and gradient closures for L-BFGS at given beta."""

    def obj(z_):
        a_ = _softmax_fn(z_)
        c_ = _full_convolve(a_, a_)
        cm_ = float(c_.max())
        c_shifted = (c_ - cm_) / beta
        e_ = np.exp(c_shifted)
        s_ = e_.sum()
        if s_ < 1e-300:
            return 2.0 * n * cm_
        log_sum = float(np.log(s_))
        return 2.0 * n * (cm_ + beta * log_sum)

    def grad_fn(z_):
        a_ = _softmax_fn(z_)
        c_ = _full_convolve(a_, a_)
        cm_ = float(c_.max())
        c_shifted = (c_ - cm_) / beta
        e_ = np.exp(c_shifted)
        s_ = e_.sum()
        if s_ < 1e-300:
            w = np.zeros_like(c_)
            w[int(np.argmax(c_))] = 1.0
        else:
            w = e_ / s_
        W = float(np.dot(w, c_))
        a_rev = a_[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        g = 4.0 * n * a_ * (h - W)
        if not np.all(np.isfinite(g)):
            return np.zeros_like(z_)
        return g

    return obj, grad_fn


def _run_lbfgs_continuation(a_init, deadline, beta_scale=1.0, max_wall_time=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max_c0 * beta_scale for b in beta_starts]

    if max_wall_time is not None:
        per_level = max(3.0, max_wall_time / len(betas))
    else:
        per_level = 30.0

    for beta in betas:
        if time.monotonic() > deadline - 2.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)
        obj, grad_fn = _lbfgs_obj_grad_factory(n, beta)
        try:
            result = minimize(
                obj, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 3000, 'ftol': 1e-20, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 10000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _build_conv_matrix(f, n):
    """Build sparse A where A[k,m] = f[k-m] for valid indices."""
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        m_start = max(0, k - (n - 1))
        m_end = min(n - 1, k)
        if m_start <= m_end:
            ms = np.arange(m_start, m_end + 1)
            rows.extend([k] * len(ms))
            cols.extend(ms.tolist())
            data.extend(f[k - ms].tolist())
    A = sparse.csc_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_peak_reduction(f, M, timeout=60.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    A = _build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': timeout, 'presolve': True})
        if result.status == 0 and result.x is not None:
            g = np.maximum(result.x, 0.0)
            if g.sum() > 1e-12:
                return g
    except Exception as e:
        print(f"  LP error: {e}")
    return None


def _lp_iteration(f, t_blend=0.3, max_iters=3, timeout_per_lp=45.0):
    """LP-based peak reduction iterations."""
    n = len(f)
    f_curr = f.copy()
    f_curr = f_curr / f_curr.sum()
    best_f = f_curr.copy()
    best_score = _score_norm_fn(f_curr)

    for it in range(max_iters):
        c_curr = _full_convolve(f_curr, f_curr)
        M = float(c_curr.max())

        g = _lp_peak_reduction(f_curr, M, timeout=timeout_per_lp)
        if g is None:
            break
        g_norm = g / g.sum()

        for t in [t_blend, t_blend * 0.5, t_blend * 0.25]:
            f_new = (1 - t) * f_curr + t * g_norm
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            if s < 1e-12:
                break
            f_new = f_new / s
            sc_new = _score_norm_fn(f_new)
            if sc_new < best_score:
                best_score = sc_new
                best_f = f_new.copy()
                f_curr = f_new.copy()
                break
        else:
            break

    return best_f, best_score


def _make_profiles(n):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    profiles = []
    names = []
    for slope in [0.5, 1.0, 1.5, 2.0, 2.5, 3.5]:
        a = np.maximum(1.0 + slope * u, 1e-12)
        profiles.append(a / a.sum())
        names.append(f"lin_{slope}")
    for a2 in [0.3, 0.7, 1.2, -0.5]:
        a = np.maximum(1.0 + 0.6 * u + a2 * u ** 2, 1e-12)
        profiles.append(a / a.sum())
        names.append(f"q_{a2}")
    for w in [0.15, 0.2, 0.25, 0.3, 0.35, 0.45]:
        a = np.exp(-u ** 2 / (2 * w ** 2))
        profiles.append(a / a.sum())
        names.append(f"g_{w}")
    for alpha in [0.5, 0.8, 1.2, 1.8]:
        a = np.maximum(0.5 * (1.0 + np.cos(np.pi * u * alpha)), 1e-12)
        profiles.append(a / a.sum())
        names.append(f"rc_{alpha}")
    for p in [1.5, 2.0, 3.0, 5.0, 8.0]:
        a = np.maximum((1.0 - np.abs(u)) ** p, 1e-12)
        profiles.append(a / a.sum())
        names.append(f"pw_{p}")
    for sep in [0.15, 0.25, 0.35]:
        a = (np.exp(-((t - 0.5 + sep) ** 2) / (2 * 0.08 ** 2)) +
             np.exp(-((t - 0.5 - sep) ** 2) / (2 * 0.08 ** 2)))
        profiles.append(a / a.sum())
        names.append(f"2h_{sep}")
    for freq in [3, 7, 15, 30]:
        a = np.maximum((1 + 0.4 * np.cos(2 * np.pi * freq * t)) * np.exp(-u ** 2 / 0.18), 1e-12)
        profiles.append(a / a.sum())
        names.append(f"m{freq}")
    # Palindromic profiles
    for w in [0.2, 0.3, 0.35]:
        a = np.exp(-((t - 0.5) ** 2) / (2 * w ** 2))
        a = 0.5 * (a + a[::-1])
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
        names.append(f"pal_g_{w}")
    return list(zip(names, profiles))


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _subgradient_steps(a_init, deadline, lr_init=0.005, decay=0.9995, rng=None):
    """Subgradient with random peak selection and momentum."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    lr = lr_init
    momentum = np.zeros(n)
    beta_mom = 0.85

    while time.monotonic() < deadline - 1.0:
        c = _full_convolve(a, a)
        # Pick from top few peaks for diversity
        top_k = min(5, len(c))
        top_idx = np.argsort(c)[-top_k:]
        k_star = top_idx[rng.integers(top_k)]

        grad = np.zeros(n)
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        for j in range(lo, hi + 1):
            grad[j] = 2.0 * a[k_star - j]

        momentum = beta_mom * momentum + (1 - beta_mom) * grad
        a = a - lr * momentum
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        lr *= decay

        if lr < 1e-8:
            break

    sc_final = _score_norm_fn(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score


def _local_search(a_init, deadline, rng, n_trials=8000):
    """Adaptive local search with block and point mutations."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    temp = 0.01

    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break

        mode = rng.random()
        a_trial = a.copy()

        if mode < 0.3:
            # Point mutation
            idx = int(rng.integers(0, n))
            a_trial[idx] *= rng.uniform(0.85, 1.15)
        elif mode < 0.7:
            # Block mutation
            bs = int(rng.integers(3, min(30, n // 10)))
            start = int(rng.integers(0, n - bs))
            a_trial[start:start + bs] *= rng.uniform(0.9, 1.1, size=bs)
        else:
            # Smooth perturbation
            center = int(rng.integers(0, n))
            radius = int(rng.integers(5, min(50, n // 4)))
            lo = max(0, center - radius)
            hi = min(n, center + radius)
            window = np.exp(-((np.arange(lo, hi) - center) ** 2) / (2 * (radius / 2) ** 2))
            a_trial[lo:hi] *= (1 + 0.1 * window * rng.choice([-1, 1]))

        a_trial = np.maximum(a_trial, 1e-15)
        a_trial = a_trial / a_trial.sum()
        sc = _score_norm_fn(a_trial)

        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
        elif rng.random() < temp and sc < best_score * 1.01:
            a = a_trial.copy()
        temp *= 0.999

    return best_a, best_score

        p_b, sc_b = _run_lbfgs_continuation(best_seq, dl, beta_scale=0.3,
                                            max_wall_time=150.0)
        update_best(p_b, sc_b, "refine_A")
        if time.monotonic() < phase4_end - 30:
            dl2 = min(phase4_end - 10, time.monotonic() + 50.0)
            p_b2, sc_b2 = _run_lbfgs_continuation(best_seq, dl2, beta_scale=0.1,
                                                  max_wall_time=50.0)
            update_best(p_b2, sc_b2, "refine_B")
    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multi-restart (250s) ---
    phase5_end = min(deadline, t_start + 1045.0)
    print(f"\n=== Phase 5: Multi-Restart ===")
    if best_seq is not None and time.monotonic() < phase5_end:
        n_restarts = 4
        per_r = max(35.0, (phase5_end - time.monotonic()) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 15:
                break
            if i == 0:
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z - z.mean() + 0.02 * rng.normal(size=len(best_seq))
                z = np.clip(z, -60, 60)
                p_start = _softmax_fn(z - z.mean())
            elif i == 1:
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z - z.mean() + 0.06 * rng.normal(size=len(best_seq))
                z = np.clip(z, -60, 60)
                p_start = _softmax_fn(z - z.mean())
            elif i == 2:
                # Palindromic projection
                n_p = len(best_seq)
                p_pal = 0.5 * (best_seq + best_seq[::-1])
                p_pal = np.maximum(p_pal, 1e-15)
                p_start = p_pal / p_pal.sum()
            else:
                # Blend with random profile
                _, profs = _make_profiles(len(best_seq))
                p_rand = profs[rng.integers(len(profs))]
                p_start = 0.6 * best_seq + 0.4 * p_rand
                p_start = p_start / p_start.sum()

            dl = min(phase5_end - 10, time.monotonic() + per_r)
            p_b, sc_b = _run_lbfgs_continuation(p_start, dl, max_wall_time=per_r)
            update_best(p_b, sc_b, f"restart_{i}")
    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Second LP round (120s) ---
    phase6_end = min(deadline, t_start + 1165.0)
    print(f"\n=== Phase 6: LP Round 2 ===")
    if best_seq is not None and time.monotonic() < phase6_end:
        n_cur = len(best_seq)
        if n_cur <= 4096:
            t0 = time.monotonic()
            p_lp2, sc_lp2 = _lp_iteration(best_seq, t_blend=0.25,
                                          max_iters=2, timeout_per_lp=40.0)
            update_best(p_lp2, sc_lp2, f"LP2_n{n_cur}")
            print(f"  LP2 done in {time.monotonic()-t0:.1f}s")
            if time.monotonic() < phase6_end - 20:
                dl = min(phase6_end - 10, time.monotonic() + 50.0)
                p_b, sc_b = _run_lbfgs_continuation(p_lp2, dl, beta_scale=0.2,
                                                    max_wall_time=50.0)
                update_best(p_b, sc_b, "post_LP2_lbfgs")
    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Local search (150s) ---
    phase7_end = min(deadline, t_start + 1315.0)
    print(f"\n=== Phase 7: Local Search ===")
    if best_seq is not None and time.monotonic() < phase7_end:
        dl = min(phase7_end - 5, time.monotonic() + 140.0)
        p_ls, sc_ls = _local_search(best_seq, dl, rng, n_trials=15000)
        update_best(p_ls, sc_ls, "local_search")
    print(f"Phase 7 done: best={best_score:.10f}")

    # --- Phase 8: Length adaptation (130s) ---
    phase8_end = min(deadline, t_start + 1445.0)
    print(f"\n=== Phase 8: Length Adaptation ===")
    if best_seq is not None and time.monotonic() < phase8_end:
        n_curr = len(best_seq)
        # Try 2x
        target = min(8192, n_curr * 2)
        if target > n_curr and time.monotonic() < phase8_end - 25:
            p2x = _interp_profile(best_seq, target)
            sc2x = _score_norm_fn(p2x)
            update_best(p2x, sc2x, f"n{target}")
            if time.monotonic() < phase8_end - 15:
                dl = min(phase8_end - 10, time.monotonic() + 50.0)
                p_b, sc_b = _run_lbfgs_continuation(p2x, dl, max_wall_time=50.0)
                update_best(p_b, sc_b, f"lbfgs_n{target}")
        # Try 0.5x
        if n_curr > 2048 and time.monotonic() < phase8_end - 20:
            target2 = max(1024, n_curr // 2)
            p_half = _interp_profile(best_seq, target2)
            sc_half = _score_norm_fn(p_half)
            update_best(p_half, sc_half, f"n{target2}_half")
    print(f"Phase 8 done: best={best_score:.10f}")

    # --- Phase 9: Final polish (remaining) ---
    print(f"\n=== Phase 9: Final Polish ===")
    if best_seq is not None:
        remaining = max(30.0, deadline - time.monotonic())
        n_polish = max(2, int(remaining / 70))
        per_p = remaining / n_polish

        for i in range(n_polish):
            if time.monotonic() > deadline - 10:
                break
            if i % 3 == 0:
                dl = min(deadline - 10, time.monotonic() + per_p)
                p_b, sc_b = _run_lbfgs_continuation(best_seq, dl,
                                                    beta_scale=0.1, max_wall_time=per_p)
                update_best(p_b, sc_b, f"polish_lbfgs_{i}")
            elif i % 3 == 1:
                dl = min(deadline - 10, time.monotonic() + per_p)
                p_b, sc_b = _subgradient_steps(best_seq, dl, lr_init=0.002,
                                               decay=0.9997, rng=rng)
                update_best(p_b, sc_b, f"polish_sg_{i}")
            else:
                dl = min(deadline - 10, time.monotonic() + per_p)
                p_b, sc_b = _local_search(best_seq, dl, rng, n_trials=5000)
                update_best(p_b, sc_b, f"polish_ls_{i}")

    # --- Final output ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            t = np.linspace(0, 1, 4096)
            final_seq = np.maximum(1.0 + 1.5 * (2 * t - 1), 1e-12)
            final_seq = final_seq / final_seq.sum()
            s = float(final_seq.sum())
        final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator: {sc_eval:.10f}")
                else:
                    print(f"WARNING: evaluator returned {sc_eval}")
            except Exception as ex:
                print(f"eval error: {ex}")
        return [float(x) for x in final_seq]

    # Fallback
    t = np.linspace(0, 1, 4096)
    fb = np.maximum(1.0 + 1.5 * (2 * t - 1), 1e-12)
    fb = fb / fb.sum()
    print(f"FALLBACK: score={_score_fn(fb):.10f}")
    return [float(x) for x in fb]
