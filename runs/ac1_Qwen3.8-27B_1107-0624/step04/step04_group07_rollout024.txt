    c = _full_convolve(a)

    active_lags = np.where(c > n_active_thresh * M)[0]
    
    a_new = a.copy()
    for block_start in range(0, n, block_size):
        block_end = min(block_start + block_size, n)
        B = block_end - block_start
        I = set(range(block_start, block_end))

        # For each active lag k:
        # sum_{i in I, i+k not in I, i+k valid} a[i+k] * x[i-block_start] 
        # + sum_{i not in I, i+k in I, i valid} a[i] * x[i+k-block_start]

        constraints_A = []
        constraints_b = []

                    const += a[i] * a[j]
                elif i in I and j not in I:
                    row[i - block_start] += a[j]
                elif i not in O and j in I:
                    row[j - block_start] += a[i]

            if rhs < 0:

                continue

"""
Optimized step function search combining Nesterov-accelerated LP fixed-point,
fast coordinate descent with exact 1D optimization, active-set LP polishing,
and multi-scale escalation to minimize 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    if n <= 1:
        return 1
    k = 1
    while k < n:
        k <<= 1
    return k


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def _cross_correlate(w, a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w


def _build_conv_matrix_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_min_peak(a, n, top_k=None):
    """Solve: min M s.t. conv(a,b)[m] <= M, sum(b)=1, b>=0.
    If top_k is given, only use the top_k most active constraints."""
    c = _full_convolve(a)
    M_cur = float(c.max())
    
    if top_k is not None and top_k < 2 * n - 1:
        # Only include the most active constraints
        thresh = np.sort(c)[::-1][min(top_k, len(c)-1)]
        active = c >= thresh
        active_idx = np.where(active)[0]
    else:
        active_idx = np.arange(2 * n - 1)
    
    # Build constraint matrix for active constraints only
    A_full = _build_conv_matrix_sparse(n, a)
    A_sub = A_full[active_idx, :]
    n_active = len(active_idx)
    
    # LP: min M
    # s.t. A_sub @ b - M <= 0  (n_active inequalities)
    #      sum(b) = 1          (1 equality)
    #      b >= 0, M >= 0
    # Variables: x = [M, b_0, ..., b_{n-1}], total 1+n variables
    
    # Inequality: A_sub @ b - M <= 0 => [-1, A_sub] @ x <= 0
    # Build sparse matrix for inequalities
    # Each row: -1 in col 0, then A_sub row
    ii = list(range(n_active))
    jj = [0] * n_active
    vv = [-1.0] * n_active
    # Add A_sub entries
    A_sub_coo = A_sub.tocoo()
    ii.extend(A_sub_coo.row.tolist())
    jj.extend([c + 1 for c in A_sub_coo.col])
    vv.extend(A_sub_coo.data.tolist())
    
    A_ineq = csr_matrix((vv, (ii, jj)), shape=(n_active, 1 + n))
    b_ineq = np.zeros(n_active)
    
    # Equality: sum(b) = 1 => [0, 1, 1, ..., 1] @ x = 1
    A_eq = np.zeros((1, 1 + n))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0])
    
    c_obj = np.zeros(1 + n)
    c_obj[0] = 1.0  # minimize M
    
    bounds = [(0, None)] * (1 + n)
    
    res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        M_opt = float(res.x[0])
        b_opt = np.maximum(res.x[1:], 0.0)
        return M_opt, b_opt
    return M_cur, None


def _lp_fixed_point_nesterov(a, deadline, max_iters=500, top_k=None):
    """LP fixed-point with Nesterov acceleration."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    a_prev = a.copy()
    a_curr = a.copy()
    t_nest = 1.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Nesterov extrapolation
        t_nest = 1.0 + (t_nest - 1.0) * 0.99  # slowly increase momentum
        y = a_curr + (t_nest - 1.0) / t_nest * (a_curr - a_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Solve LP with y as the fixed sequence
        M_opt, b = _lp_min_peak(y, n, top_k=top_k)
        if b is None:
            break
        s_b = b.sum()
        if s_b < 1e-10:
            break
        b = b / s_b
        
        # Update
        a_prev = a_curr.copy()
        a_curr = b
        
        sc = _score_norm(a_curr)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()
        
        # Convergence check
        if it > 20 and np.max(np.abs(a_curr - a_prev)) < 1e-10:
            break
    
    return best_a, best_score

        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        
        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.002, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            vel.fill(0)
            t_adam = 0
            continue
        
        if t_adam % 30 == 0:
            beta = max(1e-5, beta * beta_decay)
        
        if no_imp > no_imp_limit:
            ns = 0.06 if best_score < 1.51 else 0.12
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a)
            beta = 0.05 * c.max()
            m.fill(0)
            v.fill(0)
            vel.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0
    
    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score


def _score_single_coord(a, c, j, old, x, sum_a, n):
    """Fast score evaluation for single coordinate change a[j]: old -> x."""
    if abs(x - old) < 1e-18:
        return 2.0 * n * c.max() / (sum_a * sum_a)
    
    delta = x - old
    # Change in convolution:
    # For m in [j, j+n-1] (clamped to [0, 2n-2]): delta_c[m] = 2*delta*a[m-j]
    # For m = 2j (if in range): delta_c[2j] = x^2 - old^2 (override)
    
    idx = np.arange(2 * n - 1)
    other = idx - j
    valid = (other >= 0) & (other < n)
    
    change = np.where(valid, 2.0 * delta * a[other], 0.0)
    if 0 <= 2 * j < 2 * n - 1:
        change[2 * j] = x * x - old * old
    
    new_max = np.maximum(c + change, 0.0).max()
    new_sum = sum_a - old + x
    if new_sum < 1e-15:
        return 1e10
    return 2.0 * n * new_max / (new_sum * new_sum)


def _coord_descent_fast(a, deadline, max_sweeps=100):
    """Coordinate descent with exact 1D golden-section optimization."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    c = _full_convolve(a)
    sum_a = float(a.sum())
    
    best_a = a.copy()
    best_score = float(2.0 * n * c.max() / (sum_a * sum_a))
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        
        for j in range(n):
            if time.monotonic() > deadline:
                break
            old = a[j]
            if old < 1e-14:
                continue
            
            # Golden section search for optimal x in [0, 4*old]
            lo, hi = 0.0, 4.0 * old
            gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate
            
            m1 = hi - gr * (hi - lo)
            m2 = lo + gr * (hi - lo)
            f1 = _score_single_coord(a, c, j, old, m1, sum_a, n)
            f2 = _score_single_coord(a, c, j, old, m2, sum_a, n)
            
            for _ in range(40):
                if f1 < f2:
                    hi = m2
                    m2 = m1
                    f2 = f1
                    m1 = hi - gr * (hi - lo)
                    f1 = _score_single_coord(a, c, j, old, m1, sum_a, n)
                else:
                    lo = m1
                    m1 = m2
                    f1 = f2
                    m2 = lo + gr * (hi - lo)
                    f2 = _score_single_coord(a, c, j, old, m2, sum_a, n)
            
            best_x = (lo + hi) / 2.0
            best_fx = _score_single_coord(a, c, j, old, best_x, sum_a, n)
            
            if best_fx < best_score - 1e-14:
                # Update a[j] and c incrementally
                delta = best_x - old
                a[j] = best_x
                
                # Update c
                idx = np.arange(2 * n - 1)
                other = idx - j
                valid = (other >= 0) & (other < n)
                c[valid] += 2.0 * delta * a[other]
                if 0 <= 2 * j < 2 * n - 1:
                    c[2 * j] = c[2 * j] - 2.0 * delta * a[j] + (best_x * best_x - old * old)
                
                sum_a += delta
                best_score = best_fx
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_score


def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new


def _make_profile(n, ptype, p1=1.0, p2=0.0, p3=0.0):
    if ptype == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (t - 0.5)
    elif ptype == 'ramp':
        t = np.linspace(0, 1, n)
        a = t ** p1
    elif ptype == 'inv_ramp':
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** p1
    elif ptype == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (2 * t - 1) ** 2
    elif ptype == 'cubic':
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + p1 * u + p2 * u ** 3
    elif ptype == 'exp':
        t = np.arange(n, dtype=np.float64)
        a = np.exp(-p1 * t / n)
    elif ptype == 'exp_inv':
        t = np.arange(n, dtype=np.float64)
        a = np.exp(-p1 * (n - 1 - t) / n)
    elif ptype == 'cosine':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + p1 * np.cos(2 * np.pi * p2 * t / n)
    elif ptype == 'sine':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + p1 * np.sin(2 * np.pi * p2 * t / n)
    elif ptype == 'skew_gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - p1) / p2) ** 2)
    elif ptype == 'sigmoid':
        t = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + np.exp(-p1 * (t - p2)))
    elif ptype == 'two_ramp':
        t = np.linspace(0, 1, n)
        a = np.minimum(t / p1, (1.0 - t) / p1) + 0.1
    elif ptype == 'edge_heavy':
        t = np.arange(n) / max(1, n - 1)
        a = 1.0 / (0.01 + abs(t - 0.5) * 2.0) ** p1
    elif ptype == 'chirp_real':
        t = np.arange(n, dtype=np.float64)
        a = np.abs(np.exp(1j * np.pi * p1 * t * t / n)) + 0.1
        # Real chirp: use cosine modulation
        a = np.abs(np.cos(np.pi * p1 * t * t / n)) + 0.1
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):

    Adam smooth-max, fast coordinate descent, active-set LP, multi-scale.
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")
    
    # === Phase 1: Diverse initialization ===
    print("Phase 1: Initialization...")
    candidates = []
    
    n_sizes = [256, 512, 1024]
    profile_defs = [
        ('lin2', 'linear', 2.0, 0, 0),

        ('ramp2', 'ramp', 2.0, 0, 0),
        ('ramp3', 'ramp', 3.0, 0, 0),
        ('inv_ramp2', 'inv_ramp', 2.0, 0, 0),
        ('exp1', 'exp', 1.0, 0, 0),
        ('exp2', 'exp', 2.0, 0, 0),
        ('exp_inv1', 'exp_inv', 1.0, 0, 0),
        ('exp_inv2', 'exp_inv', 2.0, 0, 0),

        ('skg_07_15', 'skew_gauss', 0.7, 0.15, 0),
        ('skg_02_10', 'skew_gauss', 0.2, 0.1, 0),
