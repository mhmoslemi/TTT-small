"""
Step-function optimizer minimizing 2*n*max(a*a)/(sum(a))^2.

Approach: intensive LP-based alternating best-response with adaptive constraint
tightening (M = factor * current_peak), a fine line search over the blend step,
and a smooth-max (log-sum-exp) Adam polish in logit space. Multiple phases:
diverse short initialization, top-seed LP refinement, length escalation via
interpolation (n -> 2048 / 4096), perturbed multi-restarts, and a final polish.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 512:
        c = np.convolve(a, a)
    else:
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start - 0, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_solve(a, M):
    n = len(a)
    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None
    if res is not None and res.success and res.x is not None:
        return np.maximum(np.asarray(res.x, dtype=np.float64), 0.0)
    return None


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    m = float(c.max())
    if beta < 1e-13:
        w = np.zeros_like(c)
        mask = c >= m - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return w
    x = (c - m) / beta
    ex = np.exp(np.minimum(x, 0.0))
    s = float(ex.sum())
    if (not np.isfinite(s)) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax(z, floor=1e-14):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(z.max())
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if (not np.isfinite(s)) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 1e-15)
    s = float(a.sum())
    if (not np.isfinite(s)) or s < 1e-30:
        return None
    return a / s


def _blend_line_search(p, b_norm, base_score):
    best_t = 0.0
    best_sc = base_score
    best_p = None
    for t in (0.05, 0.08, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6):
        p_try = (1.0 - t) * p + t * b_norm
        p_try = np.maximum(p_try, 1e-15)
        p_try = p_try / p_try.sum()
        sc = _score_norm(p_try)
        if sc < best_sc:
            best_sc = sc
            best_t = t
            best_p = p_try
    return best_p, best_sc, best_t


def _lp_refine_intensive(p, deadline, m_factors, max_iters):
    p = _normalize(p)
    if p is None:
        return p, np.inf
    best_p = p.copy()
    best_score = _score_norm(p)
    c = _full_convolve(p)
    M_cur = float(c.max())
    stall = 0
    n = p.size
    # fewer, cheaper line-search steps for large n
    n_tsteps = 8 if n <= 2048 else 5

    for _ in range(max_iters):
        if time.monotonic() > deadline:
            break
        improved = False

            s_b = float(b.sum())
            if s_b < 1e-10:

                    break

                    best_score = sc
                    best_p = p.copy()
                    c = _full_convolve(p)
                    M_cur = float(c.max())
                    stall = 0
                    improved = True
                    break
        if not improved:
            stall += 1
            if stall > 6:
                break
        if best_score <= 1.5027:
            break
    return best_p, best_score


def _adam_logit(p0, deadline, rng, lr, beta_frac, beta_decay, no_imp_limit):
    p0 = np.maximum(np.asarray(p0, dtype=np.float64).ravel(), 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    z = np.log(p0)
    z -= float(z.mean())

    v = np.zeros(n)

        g = _cross_corr(w, p)
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(gz.mean())
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_step = 0
            continue

        v = b2 * v + (1.0 - b2) * (gz * gz)

        vh = v / (1.0 - b2 ** t_step)

        maxc = float(c.max())

        if np.isfinite(score) and score < best_score - 1e-14:

            best_z = z.copy()
            no_imp = 0

def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_exp(n, lam):
    t = np.arange(n)
    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x1, x0, p0))
