    # = sum_{k: 0<=k<2n-1, 0<=k-j<n} w[k] * a[k-j]
    # Let m = k-j, then k = m+j, so m in [0,n-1] and j in [0,n-1], k = m+j in [0, 2n-2]
    # h[j] = sum_{m=0}^{n-1} w[m+j] * a[m] (where w[m+j] = 0 if m+j >= 2n-1)
    # This is (w_shifted * a)[j] where w_shifted is w padded
    # Equivalently: h = ifft(fft(w, N) * conj(fft(a, N)))... no
    # Let's use: h[j] = sum_m a[m] * w[m+j] = (a cross-correlate w)[j]
    # = ifft(conj(fft(a)) * fft(w))[j + (2n-2)] ... this is getting complicated
    # Simplest: use the fact that h = 2 * (w_pad * a_rev)[n-1 : 2n-1] where * is convolution
    a_rev = a[::-1]
    w_conv_a = np.fft.irfft(np.fft.rfft(w, N) * np.fft.rfft(a_rev, N), N)
    h = 2.0 * w_conv_a[n - 1:2 * n - 1]
    # Objective value (smooth approximation)
    val = 2.0 * n * (cm + beta * float(np.log(s + 1e-30)))
    # Gradient: d/dz_j of val
    # val = 2n * (cm + beta*log(sum exp(c/beta)))
    # d/dz_j = 2n * sum_k w[k] * dc[k]/dz_j = 2n * sum_k w[k] * a_j * dc[k]/da_j
    #         = 2n * a_j * h[j]  (where h[j] = sum_k w[k] * dc[k]/da[j])
    # But we defined h above as the inner product, so:
    grad = 2.0 * n * a * h
    # Actually wait - the gradient of the log-sum-exp is sum_k w[k] * dc[k]/da[j]
    # and dc[k]/da[j] is what we computed as the building block of h
    # Let me recompute: h[j] = sum_k w[k] * (dc[k]/da[j])
    # dc[k]/da[j] = 2*a[k-j] (if 0<=k-j<n) + extra if k-j==j
    # So h[j] = 2*sum_k w[k]*a[k-j] = 2*(w*a_cross)[j]
    # Then grad[j] = 2n * a[j] * h[j] ... no wait
    # val = 2n * logsumexp(c/beta) approx
    # d(val)/d(z_j) = 2n * sum_k w[k] * dc[k]/da_j * da_j/dz_j
    # da_j/dz_j = a[j] (for softmax)
    # So d(val)/d(z_j) = 2n * a[j] * sum_k w[k] * dc[k]/da_j
    # = 2n * a[j] * h[j]
    # where h[j] = sum_k w[k] * dc[k]/da_j (NOT multiplied by 2 already)
    # Let me recompute h without the factor of 2:
    w_conv_a2 = np.fft.irfft(np.fft.rfft(w, N) * np.fft.rfft(a_rev, N), N)
    h_no2 = w_conv_a2[n - 1:2 * n - 1]
    grad = 2.0 * n * a * h_no2
    if not np.all(np.isfinite(grad)):
        grad = np.zeros_like(z)
    return val, grad


def _spectral_flatten(a, deadline, rng=None):
    """
    Spectral domain iterative method to flatten the autocorrelation.
    Uses magnitude modification in frequency domain with projection to a>=0.
    """
    n = len(a)
    N = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _score_norm(a)

    # Strategy 1: Power-law flattening with various exponents
    alphas = [0.95, 0.9, 0.85, 0.8, 0.75, 0.7, 0.65, 0.6, 0.55, 0.5]
    for alpha in alphas:
        if time.monotonic() > deadline:
            break
        a_cur = best_a.copy()
        for it in range(50000):
            if time.monotonic() > deadline:
                break
            A = np.fft.rfft(a_cur, N)
            mag = np.abs(A)
            phase = A / (mag + 1e-30)
            # Compress magnitude: new_mag = mag^alpha (normalized)
            mag_max = mag.max()
            if mag_max < 1e-20:
                break
            mag_norm = mag / mag_max
            mag_new = mag_max * (mag_norm ** alpha + (1 - alpha) * 1e-10)
            # Actually, let's blend: new = (1-t)*old + t*flat
            # Flat would be constant magnitude = sqrt(mean(|A|^2))
            t = 0.3  # blending factor
            flat_mag = np.sqrt(np.mean(mag ** 2))
            mag_blend = (1 - t) * mag + t * flat_mag
            A_new = mag_blend * phase
            a_new = np.fft.irfft(A_new, N)[:n]
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new /= s
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            a_cur = a_new

    # Strategy 2: Peak-targeted spectral modification
    for outer in range(5):
        if time.monotonic() > deadline:
            break
        a_cur = best_a.copy()
        A = np.fft.rfft(a_cur, N)
        for it in range(20000):
            if time.monotonic() > deadline:
                break
            c = np.fft.irfft(np.fft.rfft(a_cur, N) * np.fft.rfft(a_cur, N), N)[:2 * n - 1]
            c_max = c.max()
            c_min = c.min()
            if c_max < 1e-20:
                break
            # Identify peak frequency region
            # The peak in c corresponds to... actually c is already in time domain
            # Let's modify the spectrum to reduce the peak
            A = np.fft.rfft(a_cur, N)
            mag = np.abs(A)
            phase = A / (mag + 1e-30)
            # Reduce the most oscillatory components (high frequency)
            freq = np.arange(len(mag))
            # Apply a notch at frequencies that contribute most to the peak
            # Simple approach: compress the dynamic range of |A|
            mag_min = mag.min()
            mag_range = mag_max_v = mag.max()
            if mag_range - mag_min < 1e-20:
                break
            # Soft compression
            gamma = 0.85
            mag_compressed = mag_min + (mag - mag_min) ** gamma
            # Rescale to maintain energy
            current_energy = np.sum(mag ** 2)
            new_energy = np.sum(mag_compressed ** 2)
            if new_energy > 0:
                mag_compressed = mag_compressed * np.sqrt(current_energy / new_energy)
            A_new = mag_compressed * phase
            a_new = np.fft.irfft(A_new, N)[:n]
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new /= s
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            a_cur = a_new

    return best_a, best_score


def _spectral_flatten_v2(a, deadline, rng=None):
    """
    Improved spectral flattening: iteratively modify |A(w)| to flatten c[k].
    Uses the relationship c = IDFT(|A|^2) directly.
    """
    n = len(a)
    N = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _score_norm(a)

    # Work directly with the power spectrum P = |A|^2
    # c[k] = IDFT(P)[k]
    # We want to modify P to reduce max(c) while keeping a = IDFT(sqrt(P)*phase) >= 0

    a_cur = best_a.copy()
    A = np.fft.rfft(a_cur, N)
    phase = np.exp(1j * np.angle(A))
    P = np.abs(A) ** 2

    gamma = 0.5  # interpolation parameter
    for it in range(100000):
        if time.monotonic() > deadline:
            break
        c = np.fft.irfft(P, N)[:2 * n - 1]
        c_max = c.max()
        c_mean = c[0:2 * n - 1].mean()
        if c_max < 1e-20:
            break
        # Modify P to reduce the peak in c
        # c = IDFT(P), so we want P to be flatter
        # Strategy: P_new = (1-gamma)*P + gamma*mean(P)
        P_new = (1 - gamma) * P + gamma * c_mean
        # Ensure P >= 0 (it should be since P >= 0 and c_mean > 0)
        P_new = np.maximum(P_new, 0)
        # New magnitude
        mag_new = np.sqrt(P_new)
        # Reconstruct
        A_new = mag_new * phase
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        a_cur = a_new
        # Update phase
        A = np.fft.rfft(a_cur, N)
        phase = A / (np.abs(A) + 1e-30)
        P = np.abs(A) ** 2
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

    return best_a, best_score


def _adam_gd(a_init, deadline, rng, lr=0.1, max_iter=500000):
    """
    Fast Adam gradient descent on the smooth approximation of the objective.
    Uses log-parameterization (softmax).
    """
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    # Adaptive beta
    c0 = _convolve_self(a_init, N)
    cm0 = float(c0.max())
    beta = 0.05 * cm0
    beta_min = 1e-6 * cm0

    no_improve = 0
    improve_threshold = best_score * 1e-10

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        val, grad = _grad_and_value(z, beta, n, N)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            if best_score - sc > improve_threshold:
                no_improve = 0
            else:
                no_improve += 1
            best_score = sc
            best_a = a.copy()
        else:
            no_improve += 1

        # Adaptive beta: decrease when stuck
        if no_improve > 1000:
            beta = max(beta_min, beta * 0.8)
            no_improve = 0
            # Small perturbation
            z = z + 0.02 * rng.normal(size=n)
            z = z - z.mean()

        # Learning rate decay
        if t % 50000 == 0:
            lr = max(0.01, lr * 0.9)

    return best_a, best_score


def _adam_gd_multibeta(a_init, deadline, rng):
    """Adam GD with a schedule of beta values (continuation method)."""
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    c0 = _convolve_self(a_init, N)
    cm0 = float(c0.max())

    beta_schedule = [0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001, 0.0005]
    beta_schedule = [b * cm0 for b in beta_schedule]

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    n_betas = len(beta_schedule)
    time_per = (deadline - time.monotonic()) / n_betas

    for idx, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline:
            break
        level_deadline = min(deadline, time.monotonic() + time_per)
        a_cur, sc_cur = _adam_gd_single_beta(best_a, level_deadline, rng, beta, n, N, lr=0.1)
        if sc_cur < best_score:
            best_score = sc_cur
            best_a = a_cur.copy()

    return best_a, best_score


def _adam_gd_single_beta(a_init, deadline, rng, beta, n, N, lr=0.1, max_iter=200000):
    """Adam GD at a fixed beta."""
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        val, grad = _grad_and_value(z, beta, n, N)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:

    return best_a, best_score


def _peak_surgery(a, deadline, rng):
    """Targeted coordinate adjustments to reduce the peak."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = _score_norm(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    for outer in range(200):
        if time.monotonic() > deadline:
            break
        a_cur = _softmax_fn(z)
        c = _convolve_self(a_cur, N)
        cm = float(c.max())
        k_star = int(np.argmax(c))

        # Find top contributors to c[k_star]
        valid_i = np.arange(max(0, k_star - n + 1), min(n - 1, k_star) + 1)
        if len(valid_i) == 0:
            break
        contributions = a_cur[valid_i] * a_cur[k_star - valid_i]
        order = np.argsort(contributions)[::-1]
        top_idx = valid_i[order[:15]]

        improved = False
        for idx in top_idx:
            for delta in [0.005, 0.01, 0.02, 0.05, 0.1]:
                z_trial = z.copy()
                z_trial[idx] -= delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm(a_trial)
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try increasing small elements (redistribute mass)
            small_idx = np.argsort(a_cur)[:10]
            for idx in small_idx:
                for delta in [0.005, 0.01, 0.02, 0.05]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm(a_trial)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Try pair swaps
            large_idx = np.argsort(a_cur)[::-1][:5]
            small_idx2 = np.argsort(a_cur)[:5]
            found = False
            for li in large_idx:
                for si in small_idx2:
                    if li == si:
                        continue
                    for frac in [0.01, 0.02, 0.05]:
                        z_trial = z.copy()
                        z_trial[li] -= frac
                        z_trial[si] += frac
                        a_trial = _softmax_fn(z_trial)
                        sc = _score_norm(a_trial)
                        if sc < best_score:
                            best_score = sc
                            best_a = a_trial.copy()
                            z = z_trial.copy()
                            found = True
                            break
                    if found:
                        break
                if found:
                    break
            if not found:
                break

    return best_a, best_score


def _make_profiles(n):
    """Generate diverse initial profiles, all normalized to sum=1."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []

    # Linear ramps
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    # Quadratic
    for a1, a2 in [(1.0, 0.5), (2.0, -0.3), (1.5, 0.2), (3.0, -0.5), (-2.0, 0.3)]:
        a = 1.0 + a1 * u + a2 * u ** 2
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    # Exponential
    for rate in [1.0, 2.0, 3.0, 5.0, -1.0, -2.0, -3.0, -5.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())

    # Cosine power
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-10) ** p
        profiles.append(a / a.sum())

    # Asymmetric cosine
    for p in [0.5, 1.0, 2.0]:
        for shift in [-0.3, -0.15, 0.15, 0.3]:
            a = np.maximum(np.cos(np.pi * (u + shift) / 2), 1e-10) ** p
            profiles.append(a / a.sum())

    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _lbfgs_refine(a_init, deadline, n, N):
    """L-BFGS refinement with very small beta (near-exact max)."""
    c0 = _convolve_self(a_init, N)
    cm0 = float(c0.max())
    beta = 1e-4 * cm0

    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    try:
        result = minimize(
            lambda zz: _grad_and_value(zz, beta, n, N)[0],
            z,
            method='L-BFGS-B',
            jac=lambda zz: _grad_and_value(zz, beta, n, N)[1],
            options={'maxiter': 2000, 'ftol': 1e-20, 'gtol': 1e-14,
                     'maxls': 50, 'maxfun': 10000}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            a_new = _softmax_fn(result.x)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 0.01:
                a_new /= s
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
    except Exception:
        pass

    return best_a, best_score

            # Use exponential
            a_cur = _make_profiles(n_best)[7].copy()

        per_dl = min(phase2_end - 20, time.monotonic() + per_restart_time)
        label = f"adam_r{r}"
        p_best, sc_best = _adam_gd_multibeta(a_cur, per_dl, rng)
        update_best(p_best, sc_best, label)

    print(f"  Phase 2 done: best={best_score:.8f}")

    # === Phase 3: More Adam GD with different LRs (400s) ===
    phase3_end = min(deadline, t_start + 1450.0)
    print("Phase 3: Adam GD fine-tuning")

    if time.monotonic() < phase3_end and best_seq is not None:
        n_cur = len(best_seq)
        N_cur = _next_pow2(2 * n_cur - 1)

        # Slow Adam for fine-tuning
        per_dl = min(phase3_end - 10, time.monotonic() + 200)
        p_b, sc_b = _adam_gd(best_seq, per_dl, rng, lr=0.02)
        update_best(p_b, sc_b, "adam_slow")

        # Even slower
        if time.monotonic() < phase3_end - 50:
            per_dl2 = min(phase3_end - 10, time.monotonic() + 150)
            p_b2, sc_b2 = _adam_gd(best_seq, per_dl2, rng, lr=0.005)
            update_best(p_b2, sc_b2, "adam_vslow")

    print(f"  Phase 3 done: best={best_score:.8f}")

    # === Phase 4: Peak surgery (200s) ===
    phase4_end = min(deadline, t_start + 1650.0)
    print("Phase 4: Peak surgery")

    if time.monotonic() < phase4_end and best_seq is not None:
        per_dl = min(phase4_end - 10, time.monotonic() + 180)
        p_b, sc_b = _peak_surgery(best_seq, per_dl, rng)
        update_best(p_b, sc_b, "peak_surgery")

    print(f"  Phase 4 done: best={best_score:.8f}")

    # === Phase 5: Try different n (100s) ===
    phase5_end = min(deadline, t_start + 1750.0)
    print("Phase 5: Try other n values")

    if time.monotonic() < phase5_end and best_seq is not None:
        n_cur = len(best_seq)
        for n_try in [2048, 8192]:
            if time.monotonic() > phase5_end - 30:
                break
            if n_try == n_cur:
                continue
            p_interp = _interp_profile(best_seq, n_try)
            sc_interp = _score_norm(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{n_try}")

            if time.monotonic() < phase5_end - 40:
                per_dl = min(phase5_end - 20, time.monotonic() + 40)
                p_b, sc_b = _adam_gd(p_interp, per_dl, rng, lr=0.05, max_iter=50000)
                update_best(p_b, sc_b, f"adam_n{n_try}")

    print(f"  Phase 5 done: best={best_score:.8f}")

    # === Phase 6: Final polish (remaining) ===
    phase6_end = deadline
    print("Phase 6: Final polish")

    if time.monotonic() < phase6_end and best_seq is not None:
        n_fin = len(best_seq)
        N_fin = _next_pow2(2 * n_fin - 1)

        # Very slow Adam
        per_dl = min(phase6_end - 10, time.monotonic() + (phase6_end - time.monotonic()) * 0.5)
        p_b, sc_b = _adam_gd(best_seq, per_dl, rng, lr=0.003, max_iter=100000)
        update_best(p_b, sc_b, "final_adam")

        # Final peak surgery
        if time.monotonic() < phase6_end - 10:
            per_dl2 = min(phase6_end - 5, time.monotonic() + 60)
            p_b2, sc_b2 = _peak_surgery(best_seq, per_dl2, rng)
            update_best(p_b2, sc_b2, "final_surgery")

    print(f"  Phase 6 done: best={best_score:.8f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profiles(256)[2]

        sc_final = _score_full(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, norm_score={_score_norm(final_seq):.10f}")

        # Validate with evaluator
        if len(final_seq) <= 5000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                else:
                    print(f"Evaluator returned inf, adjusting...")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")
        else:
            # For large sequences, just verify manually
            print(f"Sequence too large for evaluator check (n={len(final_seq)}), trusting manual score")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profiles(256)[2]
    print(f"FALLBACK: n=256, score={_score_full(fallback):.10f}")
    return [float(x) for x in fallback]
