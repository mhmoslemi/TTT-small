def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)

**Let me think about the theoretical optimum.**

For a sequence of length n with sum 1, the minimum possible value of max(conv(a,a)) is related to the problem of finding a sequence with minimal peak autocorrelation. This is related to:
- Barker sequences (binary)
- Zadoff-Chu sequences (complex)
- Low autocorrelation sequences

For real non-negative sequences, the problem is different. The theoretical lower bound for `2n * max(conv(a,a)) / sum(a)^2` when sum(a)=1 is `2n * min max(conv(a,a))`.

For a uniform sequence `a[i] = 1/n`, we get:
`conv(a,a)[k] = (n-|k-(n-1)|) / n^2` for the triangular shape
`max = 1/n` (at k=n-1)
`score = 2n * (1/n) / 1 = 2`

So the uniform sequence gives score 2. Our target of 1.5027 is much better.

The theoretical minimum is at least 1 (achieved when conv(a,a) is as flat as possible, i.e., a "perfect" sequence where all convolution values are equal to 1/n). But for non-negative sequences, this might not be achievable.

Actually, `sum_k conv(a,a)[k] = (sum a)^2 = 1` (for sum(a)=1). And there are 2n-1 values. So the average is `1/(2n-1)`. The minimum possible max is at least the average, `1/(2n-1)`. This gives a score of `2n/(2n-1) ≈ 1` for large n.

But can we achieve this? We'd need `conv(a,a)[k] = 1/(2n-1)` for all k, which means the convolution is perfectly flat. This is related to the existence of "perfect" sequences, which for real non-negative sequences is very restrictive.

Actually, for non-negative sequences, the convolution at lag 0 is `a[0]^2` and at lag 2n-2 is `a[n-1]^2`. For these to equal `1/(2n-1)`, we need `a[0] = a[n-1] = 1/sqrt(2n-1)`. Similarly, `conv(a,a)[1] = a[0]a[1] + a[1]a[0] = 2a[0]a[1] = 1/(2n-1)`, so `a[1] = 1/(2a[0]*(2n-1)) = 1/(2*sqrt(2n-1)*(2n-1))`... This gets complicated and probably leads to contradictions for large n.

So the practical minimum is above 1, and we're at 1.506 trying to get to 1.5027.

import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def score_norm(a):
    """Score assuming sum(a) = 1"""
    n = len(a)
    c = full_convolve(a, a)
    return 2.0 * n * c.max()

def softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def build_spasm_matrix(n, f):
    """Build the sparse convolution constraint matrix."""
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows.append(np.full(len(j_range), k))
        cols.append(j_range)
        vals.append(f[k - j_range])
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    vals = np.concatenate(vals)
    return sp.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def lp_solve(f, M, max_iter=None):
    """Solve: max sum(g) s.t. conv(f,g) <= M, g >= 0"""
    n = len(f)
    A = build_spasm_matrix(n, f)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x

"""
LP-based iterative optimization with gradient polishing for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses linear programming to make principled
large-step updates, followed by Adam-based fine-tuning.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp

# ============ Core functions ============

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score for normalized sequence (sum=1)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

# ============ LP-based optimization ============

def _build_lp_structure(n):
    """Precompute row/col structure for the convolution constraint matrix."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    return rows, cols

def _solve_lp(f, M, rows, cols):
    """Solve: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns g or None."""
    n = len(f)
    # A[k, j] = f[k-j] where k is row, j is col
    # rows = k values, cols = j values
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.status != 0:
            return None
        g = np.maximum(res.x, 0)
        return g
    except Exception:
        return None

def _lp_iteration(f, rows, cols, max_blends=10):
    """One LP iteration: solve LP, find best blend, return updated f."""
    n = len(f)
    c = _full_convolve(f, f)
    M = float(c.max())
    
    g_0 = _solve_lp(f, M, rows, cols)
    if g_0 is None:
        return f, 0.0
    
    s_g = float(g_0.sum())
    if s_g < 1e-15:
        return f, 0.0
    g = g_0 / s_g  # normalize to sum=1
    
    # Find best blend parameter t
    best_t = 0.0
    best_sc = _score_norm(f)
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        f_new = (1 - t) * f + t * g
        sc = _score_norm(f_new)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    
    if best_t == 0.0:
        # Try smaller values
        for t in [0.01, 0.02, 0.03]:
            f_new = (1 - t) * f + t * g
            sc = _score_norm(f_new)
            if sc < best_sc:
                best_sc = sc
                best_t = t
    
    if best_t == 0.0:
        return f, 0.0
    
    f_new = (1 - best_t) * f + best_t * g
    return f_new, M

# ============ Gradient-based polishing ============

def _compute_grad(a, w, n):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a."""
    a_rev = a[::-1]

    conv_w_a = _full_convolve(w, a)
    grad1 = conv_w_ar[n-1:2*n-1]
    grad2 = conv_w_a[:n]

"""
LP + Mirror Descent hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming for coarse optimization (principled large jumps) 
combined with multiplicative weight updates (mirror descent) for fine-tuning.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _build_lp_indices(n):
    """Precompute row/col indices for the sparse convolution matrix."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)

def _solve_lp(f, M, rows, cols):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 60})
        if res.status != 0:
            return None
        return np.maximum(res.x, 0.0)
    except Exception:
        return None

def _lp_step(f, rows, cols):
    """One LP iteration with adaptive blending."""
    c = _full_convolve(f, f)
    M = float(c.max())
    g_0 = _solve_lp(f, M, rows, cols)
    if g_0 is None or g_0.sum() < 1e-15:
        return f, 0.0
    g = g_0 / g_0.sum()
    # Find best blend
    best_t = 0.0
    best_sc = _score_norm(f)
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        f_cand = (1-t) * f + t * g
        sc = _score_norm(f_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t == 0.0:
        for t in [0.01, 0.02]:
            f_cand = (1-t) * f + t * g
            sc = _score_norm(f_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
    if best_t > 0:
        f = (1-best_t) * f + best_t * g
    return f, M

def _mwu_step(a, beta, eta):
    """One step of mirror descent (multiplicative weight update)."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = c.max()
    w = np.exp((c - c_max) / beta)
    w /= w.sum()
    # Gradient
    w_rev = w[::-1]
    conv_aw = _full_convolve(a, w_rev)
    grad = 2.0 * conv_aw[n-1:2*n-1][::-1]
    # Multiplicative update
    z = np.log(np.maximum(a, 1e-300)) - eta * grad
    z -= z.mean()
    a_new = _softmax(z)
    return a_new
