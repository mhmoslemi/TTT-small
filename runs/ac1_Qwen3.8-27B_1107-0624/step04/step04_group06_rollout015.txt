def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())

def _subgradient_multipeak(a, c, beta=0.01):
    """Compute weighted subgradient over all near-peak positions."""
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)
    
    # Softmax weights over peaks
    x = (c - c_max) / beta
    x = np.clip(x, -50, 0)  # Avoid overflow
    w = np.exp(x)
    w = w / w.sum()
    
    # Only keep significant weights
    w_thresh = 1e-6
    active = w > w_thresh
    if not active.any():
        active[int(np.argmax(c))] = True
        w[~active] = 0
        w = w / w.sum()
    
    # Compute subgradient: g[j] = 2 * sum_k w[k] * a[k-j] (if valid)
    g = np.zeros(n)
    active_ks = np.where(active)[0]
    for k in active_ks:
        # a[k-j] is valid when 0 <= k-j < n, i.e., max(0,k-n+1) <= j <= min(n-1, k)
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)  # exclusive
        if j_start < j_end:
            indices = np.arange(j_start, j_end)
            partner = k - indices
            g[j_start:j_end] += 2.0 * w[k] * a[partner]
    
    return g

def _mirror_descent(a, deadline, initial_eta=0.05, beta=0.005, 
                     no_imp_limit=200, eta_decay=0.95, eta_min=0.001):
    """Mirror descent with multi-peak subgradient."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size
    
    best_a = a.copy()
    best_score = _score_norm(a)
    eta = initial_eta
    no_imp = 0
    it = 0
    
    while time.monotonic() < deadline:
        c = _full_convolve(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > no_imp_limit:
                eta = max(eta_min, eta * eta_decay)
                no_imp = 0
        
        g = _subgradient_multipeak(a, c, beta=beta)
        
        # Mirror descent update
        log_a = np.log(a)
        log_a -= eta * g
        log_a = np.clip(log_a, -50, 50)
        log_a -= log_a.max()
        a = np.exp(log_a)
        s = a.sum()
        if s < 1e-30 or not np.isfinite(s):
            a = best_a.copy()
            eta = max(eta_min, eta * 0.5)
            continue
        a = a / s
        
        it += 1
    
    return best_a, best_score

"""
Optimization of non-negative sequences to minimize 2n*max(conv(a,a))/(sum(a))^2
using mirror descent with multi-peak subgradients, adaptive step sizes,
LP-based refinement, and multi-start exploration.
Key innovation: direct subgradient computation on the actual max (not smooth approximation),
with softmax-weighted multi-peak handling and exponential-tilt (mirror descent) updates
on the simplex.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 1e-15)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / a.size
    return a / s


def _multi_peak_subgradient(a, c, beta):
    """Compute softmax-weighted subgradient over near-peak positions."""
    n = len(a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros(n)

    x = (c - c_max) / beta
    x = np.clip(x, -60.0, 0.0)
    w = np.exp(x)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / ws

    # Prune negligible weights
    active = w > 1e-8
    if not active.any():
        active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0

    g = np.zeros(n)
    active_ks = np.where(active)[0]
    for k in active_ks:
        wk = w[k]
        if wk < 1e-10:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        if j_start < j_end:
            idx = np.arange(j_start, j_end)
            partner = k - idx
            g[j_start:j_end] += 2.0 * wk * a[partner]

    return g


def _mirror_descent(a, deadline, initial_eta=0.8, beta_init=0.003,
                    beta_decay=0.995, beta_min=1e-5,
                    no_imp_limit=300, eta_decay=0.9, eta_min=0.01,
                    restart_noise=0.05, max_iter=None):
    """Mirror descent with multi-peak subgradient on the simplex."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size

    best_a = a.copy()
    best_score = _score_norm(a)
    eta = initial_eta
    beta = beta_init
    no_imp = 0
    it = 0
    base_beta = beta_init
    max_iter = max_iter if max_iter is not None else 10 ** 9

    while time.monotonic() < deadline and it < max_iter:
        it += 1
        c = _full_convolve(a)
        c_max = float(c.max())
        score = 2.0 * n * c_max

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Adaptive beta: tighten when peaks are close together, loosen when far
        if it % 20 == 0:
            sidx = np.argsort(c)[::-1]
            top_gap = c[sidx[0]] - c[sidx[1]] if c.size >= 2 else c_max
            if top_gap < beta * 0.1:
                beta = max(beta_min, beta * 0.5)
            elif top_gap > beta * 5.0:
                beta = min(base_beta, beta * 1.5)
            else:
                beta = max(beta_min, beta * beta_decay)

        # Compute subgradient
        g = _multi_peak_subgradient(a, c, beta)
        g_max = float(g.max())
        if g_max < 1e-30:
            break

        # Adaptive eta based on gradient magnitude
        if it % 50 == 0:
            # Scale eta so that eta * g_max ~ O(1) for meaningful updates
            target = 0.5
            eta = max(eta_min, min(2.0, target / g_max))

        # Mirror descent update: a <- a * exp(-eta * g) / Z
        log_a = np.log(a + 1e-30) - eta * g
        log_a = np.clip(log_a, -80.0, 80.0)
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        s = a_new.sum()
        if s < 1e-30 or not np.isfinite(s):
            a = best_a.copy()
            eta = max(eta_min, eta * 0.5)
            continue
        a = a_new / s
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        # Restart if stuck
        if no_imp > no_imp_limit:
            # Perturb from best
            z = np.log(best_a + 1e-30) + restart_noise * np.random.standard_normal(n)
            z = np.clip(z, -50.0, 50.0)
            z -= z.max()
            a = np.exp(z)
            a = a / a.sum()
            best_score = _score_norm(a)
            beta = base_beta
            eta = initial_eta
            no_imp = 0
            restart_noise = max(0.01, restart_noise * 0.7)

        # Gradually decrease eta to converge
        if it % 100 == 0:
            eta = max(eta_min, eta * 0.99)

    return best_a, best_score


def _lp_refine(a, deadline, t_init=0.3, max_solves=30, m_shrink=0.995):
    """LP-based refinement: find b maximizing sum(b) s.t. conv(a,b) <= M, b>=0."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    t = t_init
    n_solves = 0

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1

        # Build constraint matrix: A_ub @ b <= M * ones
        # A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            if j_start < j_end:
                idx = np.arange(j_start, j_end)
                partner = k - idx
                rows.extend([k] * len(idx))
                cols.extend(idx.tolist())
                vals.extend(a[partner].tolist())

        A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b_ub_arr = M * np.ones(2 * n - 1)
        bounds = [(0, None)] * n

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
        if not res.success:
            # Try with slightly relaxed M
            b_ub_arr = (M * 1.05) * np.ones(2 * n - 1)
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
            if not res.success:
                break

        b = np.maximum(res.x, 0.0)
        s_b = float(b.sum())
        if s_b < 1e-10:
            break

        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)

        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
        M = max(M * m_shrink, best_score / (2.0 * n) * 1.001)
        t = max(0.1, t * 0.93)

    return best_a, best_score


def _peak_shave(a, deadline, max_iter=2000):
    """Directly shave the peak by moving mass from contributing positions."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = _normalize(a)
    n = a.size
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(999)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a)
        c_max = float(c.max())
        score = 2.0 * n * c_max
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()

        # Find peak position
        k_star = int(np.argmax(c))

        # Find top contributors to the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(n, k_star + 1)
        idx = np.arange(j_start, j_end)
        partner = k_star - idx
        products = a[idx] * a[partner]

        if products.max() < 1e-15:
            break

        # Move mass from the largest contributor
        top_local = int(np.argmax(products))
        i_src = idx[top_local]
        i_partner = partner[top_local]

        # Choose destination: a position that minimizes contribution to all peaks
        # Simple heuristic: pick a random position far from the peak's partner set
        all_ks = np.where(c >= 0.95 * c_max)[0]
        forbidden = set()
        for k_val in all_ks:
            fs = max(0, k_val - n + 1)
            fe = min(n, k_val + 1)
            for i_val in range(fs, fe):
                forbidden.add(k_val - i_val)
        forbidden.add(i_src)

        valid_dest = [j for j in range(n) if j not in forbidden]
        if not valid_dest:
            valid_dest = [j for j in range(n) if j != i_src]
        if not valid_dest:
            break

        # Pick destination that currently has small value
        dest_vals = a[valid_dest]
        min_idx_in_valid = int(np.argmin(dest_vals))
        i_dst = valid_dest[min_idx_in_valid]

        # Move a fraction of mass
        frac = 0.05 + 0.1 * (a[i_src] * n)
        frac = min(frac, 0.3)
        amount = a[i_src] * frac
        a[i_src] -= amount
        a[i_dst] += amount
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

    return best_a, best_score


def _adaptive_mirror(a, deadline, phase_tag=""):

        {"initial_eta": 2.0, "beta_init": 0.001, "no_imp_limit": 150},
        {"initial_eta": 0.3, "beta_init": 0.01, "no_imp_limit": 400},
    ]

    remaining_deadline = deadline
    n_configs = len(configs)
    time_per_config = (remaining_deadline - time.monotonic()) / n_configs - 5.0

    for ci, cfg in enumerate(configs):
        if time.monotonic() > deadline - 10:
            break
        cfg_end = time.monotonic() + max(30.0, time_per_config)

    return best_a, best_score


def _try_different_n(a_best, n_new, deadline):
    """Try interpolating to a different length and optimizing."""
    a_best = np.asarray(a_best, dtype=np.float64).ravel()
    n0 = a_best.size
    if n0 == n_new:
        return a_best, _score_norm(a_best)

    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x1, x0, a_best)
    a_new = _normalize(a_new)
    sc = _score_norm(a_new)
    print(f"  Interpolated to n={n_new}: {sc:.10f}")

    if time.monotonic() < deadline - 30:
        a_new, sc = _mirror_descent(a_new, min(deadline, time.monotonic() + 60),
                                     initial_eta=0.8, beta_init=0.003)
    return a_new, sc
