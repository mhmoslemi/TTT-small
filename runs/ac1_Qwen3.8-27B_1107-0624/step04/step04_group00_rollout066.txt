"""
LP-based iterative optimization minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find globally optimal descent directions
with constraint generation for speed, combined with gradient polishing.
"""
import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import diags
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _final_score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return _score_norm_fn(a) / (s * s)


def _build_conv_matrix(a):
    n = len(a)
    diagonals = [a[d] * np.ones(n) for d in range(n)]
    offsets = list(range(0, -n, -1))
    A = diags(diagonals, offsets, shape=(2 * n - 1, n), format='csr')
    return A


def _build_conv_matrix_subset(a, row_indices):
    n = len(a)
    rows = len(row_indices)
    if rows == 0:
        return diags([], [], shape=(0, n), format='csr')
    diag_data = []
    diag_off = []
    for d in range(n):
        vals = np.zeros(rows)
        for r, k in enumerate(row_indices):
            j = k - d
            if 0 <= j <= n - 1:
                vals[r] = a[d]
        if np.any(vals != 0.0):
            diag_data.append(vals)
            diag_off.append(-d)
    if not diag_data:
        return diags([], [], shape=(rows, n), format='csr')
    return diags(diag_data, diag_off, shape=(rows, n), format='csr')


def _solve_full_lp(a, M):
    n = len(a)
    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if np.all(np.isfinite(g)) and g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def _solve_subset_lp(a, M, row_indices, max_rounds=10):
    n = len(a)
    active = list(set(int(x) for x in row_indices))
    g = None
    for rnd in range(max_rounds):
        A_sub = _build_conv_matrix_subset(a, active)
        c_obj = -np.ones(n)
        b_ub = np.full(len(active), M)
        bounds = [(0, None)] * n
        try:
            res = linprog(c_obj, A_ub=A_sub, b_ub=b_ub, bounds=bounds, method='highs')
            if not res.success or res.x is None:
                break
            g = np.maximum(res.x, 0.0)
            if g.sum() < 1e-15:
                break
        except Exception:
            break
        ag = _full_convolve(a, g)
        viols = np.where(ag > M * (1.0 + 1e-10))[0]
        new_v = [int(v) for v in viols if int(v) not in set(active)]
        if not new_v:
            return g
        new_v.sort(key=lambda v: -ag[v])
        active.extend(new_v[:20])
    if g is not None and g.sum() > 1e-15:
        return g
    return None


def _line_search(a, direction, base_score):
    ts = [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
    best_t = 0.0
    best_sc = base_score
    best_a = a
    for t in ts:
        a_new = (1.0 - t) * a + t * direction
        if np.any(a_new < -1e-12):
            continue
        a_new = np.maximum(a_new, 0.0)
        sc = _score_norm_fn(a_new)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_t = t
            best_a = a_new
    if best_t > 0.0:
        return best_a, best_t, best_sc
    return a, 0.0, base_score


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _adaptive_beta(c):
    cm = float(c.max())
    if cm < 1e-15:
        return 1e-10
    sc = np.sort(c)[::-1]
    k = min(20, len(sc) - 1)
    gap = sc[0] - sc[k]
    if gap < 0.05 * cm:
        return max(1e-10, 0.1 * cm)
    elif gap > 0.5 * cm:
        return max(1e-10, 0.01 * cm)
    else:
        return max(1e-10, 0.5 * gap)


def _smooth_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    cs = (c - cm) / beta
    e = np.exp(cs)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return cm
    return cm + beta * np.log(s)


def _smooth_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    cs = (c - cm) / beta
    e = np.exp(cs)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _gradient_polish(a, deadline):

    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    best_a = a.copy()

    mc0 = float(c0.max())
    betas = [mc0 * f for f in [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]]
    for i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = len(betas) - i
        tp = (deadline - time.monotonic()) / max(1, remaining)
        tp = min(tp, 60.0)
        dl = time.monotonic() + tp
        try:
            res = minimize(
                _smooth_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_smooth_grad,
                options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 8000}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_new = _softmax_fn(res.x)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = a_new.copy()
                    z = res.x.copy()
        except Exception:
            pass
    return best_a, best_sc


def _adam_polish(a, deadline, lr_init=0.08, max_iter=50000):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

def _make_diverse_inits(n, rng):
    inits = []
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    for slope in [1.0, 2.0, 3.0, -1.0, -2.0]:

        a = np.maximum(a, 1e-12)
        inits.append(a / a.sum())
    for rate in [1.5, 3.0, -1.5, -3.0]:
        a = np.exp(rate * (t - 0.5))
        inits.append(a / a.sum())
    for p in [0.5, 1.5, 3.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        inits.append(a / a.sum())
    a = np.ones(n)
    mid = n // 2
    a[mid - n // 4:mid + n // 4] = 0.5
    inits.append(a / a.sum())
    a = np.ones(n)
    a[n // 4:3 * n // 4] = 0.3
    inits.append(a / a.sum())
    a = 1.0 + 0.5 * np.sin(2 * np.pi * t) + 0.3 * np.cos(4 * np.pi * t)
    a = np.maximum(a, 1e-12)
    inits.append(a / a.sum())
    return inits


def _lp_search(a_init, deadline, rng, max_iters=500, use_subset=True):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    no_imp = 0
    max_no_imp = 15

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        if no_imp >= max_no_imp:
            break

        c = _full_convolve(a, a)
        M = float(c.max())

        if use_subset:
            pk = np.argsort(c)[::-1][:20]
            pk = list(sorted(set(int(x) for x in pk)))
            g = _solve_subset_lp(a, M, pk, max_rounds=8)
        else:
            g = _solve_full_lp(a, M)

        if g is None or g.sum() < 1e-12:
            break

        g = g / g.sum()
        a_new, t_used, new_sc = _line_search(a, g, best_sc)

        if t_used > 0.0:
            a = a_new
            if new_sc < best_sc:
                best_sc = new_sc
                best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp >= 5:
                z = np.log(np.maximum(a, 1e-15))
                z = z + 0.1 * rng.normal(size=n)
                z = z - z.mean()
                a = _softmax_fn(z)
                sc = _score_norm_fn(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                no_imp = 0

    return best_a, best_sc


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
