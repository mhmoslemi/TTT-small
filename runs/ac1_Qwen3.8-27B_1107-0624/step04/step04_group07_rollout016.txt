Actually P(m) = |A(m)|² where A(m) = DFT of a. P(0) = |sum a|² = 1 (for sum=1). If P is flat, P(m) = N * (sum a_j²) / N = sum a_j² for all m. Then r(0) = sum a_j² and r(k) = 0 for k ≠ 0.

So the max of the periodic autocorrelation would be sum a_j², and the score would be 2n * sum a_j². By Cauchy-Schwarz, sum a_j² >= 1/n, so score >= 2.

But this is the PERIODIC autocorrelation. The aperiodic one is different. For the aperiodic autocorrelation, we need to zero-pad a to length N >= 2n-1.

If we zero-pad and the DFT is flat, then the aperiodic autocorrelation is a delta at 0, meaning conv(a,a) = [sum a_j², 0, 0, ..., 0]. The max is sum a_j² and score = 2n * sum a_j².

But can we have a flat DFT with a non-negative sequence? The DFT of a non-negative sequence has specific properties. A flat DFT would mean a is a "sequence with flat spectrum", which for real non-negative a means a is constant (since the inverse DFT of a flat spectrum is a delta, which isn't non-negative and non-zero everywhere... wait).

Actually, if N = 2n-1 and we zero-pad a to length N, the DFT is:
A(m) = sum_{j=0}^{n-1} a_j * exp(-2πimj/N)

For this to be flat (|A(m)| constant for all m), we'd need a very specific structure. For n = N (no zero-padding), a flat DFT means a is a delta (concentrated at one point). With zero-padding, it's more complex.

In any case, perfectly flat isn't achievable for non-negative a. But we can get close.

"""
Multi-strategy optimization minimizing 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Combines spectral shaping, LP fixed-point, Adam smooth-max in logit space,
and multi-scale escalation. Key improvements: spectral pre-conditioning,
adaptive beta scheduling, structured restarts, and larger n exploration.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w


def _cross_correlate(w, a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]


def _build_conv_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_primal(a, M, n):
    A = _build_conv_matrix(n, a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _spectral_shape(a, n_iter=5, gamma=0.8):
    """Iteratively flatten the power spectrum to reduce peak autocorrelation."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    N = _next_pow2(2 * n - 1)
    for _ in range(n_iter):
        fa = np.fft.rfft(a, N)
        mag = np.abs(fa)
        phase = np.angle(fa)
        # Compress the magnitude spectrum
        mag_max = mag.max()
        if mag_max < 1e-15:
            break
        mag_norm = mag / mag_max
        mag_new = mag_max * (mag_norm ** gamma)
        fa_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(fa_new, N)[:n]
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        if s < 1e-10:
            break
        a = a_new / s
    return a


def _lp_fixed_point(a, deadline, max_iters=50, use_line_search=True):
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a)
        M = float(c.max())
        b = _lp_primal(a, M, n)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b = b / s_b

        if use_line_search:
            best_t = 0.5
            best_sc = _score_norm(a)
            for t_candidate in [0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
                a_t = (1.0 - t_candidate) * a + t_candidate * b
                a_t = np.maximum(a_t, 1e-15)
                a_t = a_t / a_t.sum()
                sc_t = _score_norm(a_t)
                if sc_t < best_sc:
                    best_sc = sc_t
                    best_t = t_candidate
            a = (1.0 - best_t) * a + best_t * b
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
        else:
            a = b

        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _adam_smooth_max(a, deadline, rng, lr=0.05, beta_init=None,
                     beta_decay=0.97, no_imp_limit=300, grad_clip=10.0):
    n = len(a)
    a = np.maximum(a, 1e-12).copy()
    a = a / a.sum()
    z = np.log(a)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = _full_convolve(a)
    maxc = float(c.max())
    beta = beta_init if beta_init is not None else 0.08 * maxc

    best_z = z.copy()
    best_a = a.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    best_no_imp_decay = 0

    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)
        g = _cross_correlate(w, a)

        dot = float(np.dot(a, g))
        gz = a * (g - dot)
        gz -= gz.mean()

        # Gradient clipping for stability
        g_norm = np.linalg.norm(gz)
        if g_norm > grad_clip:
            gz = gz * (grad_clip / g_norm)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            a = np.exp(z - z.max())
            a = a / a.sum()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40, 40)
        z -= z.mean()

        a = np.exp(z - z.max())
        a = a / a.sum()
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        c = _full_convolve(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_z = z.copy()
                best_a = a.copy()
                no_imp = 0
                # Slightly increase lr when making good progress
                if best_no_imp_decay > 0:
                    lr = min(base_lr * 1.2, lr * 1.05)
                    best_no_imp_decay = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.002, lr * 0.95)
                    best_no_imp_decay += 1
        else:
            z = best_z.copy()
            a = np.exp(z - z.max())
            a = a / a.sum()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        # Adaptive beta: decrease slowly, increase if gap is large
        if t_adam % 30 == 0:
            beta = max(1e-5, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.05:
                    beta = max(1e-5, beta * 0.6)
                elif gap > beta * 5.0:
                    beta = min(0.5 * maxc, beta * 1.5)

        if no_imp > no_imp_limit:
            # Structured restart: combine noise with spectral perturbation
            ns = 0.06 + 0.03 * min(no_imp // no_imp_limit, 5)
            z_new = best_z + ns * rng.normal(size=n)
            # Also try a spectral perturbation
            if rng.random() < 0.3:
                n_pert = n
                N_pert = _next_pow2(2 * n_pert - 1)
                fa_pert = np.fft.rfft(np.exp(best_z - best_z.max()), N_pert)
                fa_pert = fa_pert * (1 + 0.1 * rng.normal(size=len(fa_pert)))
                a_pert = np.fft.irfft(fa_pert, N_pert)[:n_pert]
                a_pert = np.maximum(a_pert, 1e-15)
                a_pert = a_pert / a_pert.sum()
                z_spectral = np.log(a_pert) + best_z.max()
                z_new = 0.5 * z_new + 0.5 * z_spectral

            z = np.clip(z_new, -40, 40)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a)
            beta = 0.08 * c.max()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_a, best_score


def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new


def _make_profile(n, profile_type, param1=1.0, param2=0.0, param3=0.0):
    if profile_type == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + param1 * (t - 0.5)
    elif profile_type == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 + param1 * (2 * t - 1) ** 2
    elif profile_type == 'cubic':
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + param1 * u + param2 * u ** 3
    elif profile_type == 'exp':
        t = np.arange(n)
        a = np.exp(-param1 * t / n)
    elif profile_type == 'cosine':
        t = np.arange(n)
        a = 1.0 + param1 * np.cos(2 * np.pi * param2 * t / n)
    elif profile_type == 'sine':
        t = np.arange(n)
        a = 1.0 + param1 * np.sin(2 * np.pi * param2 * t / n)
    elif profile_type == 'parabola':
        t = np.linspace(0, 1, n)
        a = (t - 0.5) ** 2 + param1
    elif profile_type == 'two_gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - param1) / param3) ** 2) + np.exp(-0.5 * ((t - param2) / param3) ** 2)
    elif profile_type == 'gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / param1) ** 2)
    elif profile_type == 'U_shape':
        t = np.linspace(0, 1, n)
        a = (2 * t - 1) ** 2 + param1
    elif profile_type == 'edge_heavy':
        t = np.arange(n) / max(1, n - 1)
        a = 1.0 / (0.01 + abs(t - 0.5) * 2.0) ** param1
    elif profile_type == 'W_shape':
        t = np.linspace(0, 1, n)
        a = (4 * t * (1 - t)) ** param1 + 0.1
    elif profile_type == 'flat_cos':
        t = np.linspace(0, 1, n)
        a = 1.0 + param1 * np.cos(np.pi * t) ** 2
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _score_key(x):
    return x[0]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization: spectral shaping + LP fixed-point + Adam smooth-max
    + multi-scale escalation to minimize 2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    # === Phase 1: Generate diverse initial candidates with spectral shaping ===
    print("Phase 1: Generating initial candidates + spectral shaping...")
    candidates = []

    n_sizes = [256, 512, 1024, 2048]
    profile_defs = [
        ('lin1.0', 'linear', 1.0, 0, 0),
        ('lin1.5', 'linear', 1.5, 0, 0),
        ('lin2.0', 'linear', 2.0, 0, 0),
        ('lin2.5', 'linear', 2.5, 0, 0),
        ('lin3.0', 'linear', 3.0, 0, 0),
        ('lin4.0', 'linear', 4.0, 0, 0),
        ('quad1', 'quadratic', 1.0, 0, 0),
        ('quad2', 'quadratic', 2.0, 0, 0),
        ('quad3', 'quadratic', 3.0, 0, 0),
        ('quad4', 'quadratic', 4.0, 0, 0),
        ('cubic1', 'cubic', 1.5, -1.0, 0),
        ('cubic2', 'cubic', 2.0, -1.5, 0),
        ('cubic3', 'cubic', 2.5, -2.0, 0),
        ('cubic4', 'cubic', 3.0, -2.5, 0),
        ('exp0.5', 'exp', 0.5, 0, 0),
        ('exp1.0', 'exp', 1.0, 0, 0),
        ('exp2.0', 'exp', 2.0, 0, 0),
        ('exp3.0', 'exp', 3.0, 0, 0),
        ('cos1_08', 'cosine', 0.8, 1, 0),
        ('cos2_06', 'cosine', 0.6, 2, 0),
        ('cos3_05', 'cosine', 0.5, 3, 0),
        ('sin2_05', 'sine', 0.5, 2, 0),
        ('sin3_07', 'sine', 0.7, 3, 0),
        ('para05', 'parabola', 0.5, 0, 0),
        ('para10', 'parabola', 1.0, 0, 0),
        ('g03', 'gauss', 0.3, 0, 0),
        ('g05', 'gauss', 0.5, 0, 0),
        ('tg30', 'two_gauss', 0.3, 0.7, 0.15),
        ('tg25', 'two_gauss', 0.25, 0.75, 0.2),
        ('U03', 'U_shape', 0.3, 0, 0),
        ('U05', 'U_shape', 0.5, 0, 0),
        ('edge1', 'edge_heavy', 1.0, 0, 0),
        ('edge15', 'edge_heavy', 1.5, 0, 0),
        ('edge2', 'edge_heavy', 2.0, 0, 0),
        ('W1', 'W_shape', 1.0, 0, 0),
        ('W2', 'W_shape', 2.0, 0, 0),
        ('fc05', 'flat_cos', 0.5, 0, 0),
        ('fc10', 'flat_cos', 1.0, 0, 0),
    ]

    for n in n_sizes:
        for (label, ptype, p1, p2, p3) in profile_defs:
            p = _make_profile(n, ptype, p1, p2, p3)
            sc = _score_norm(p)
            candidates.append((sc, p, n, label))

    # Add parent-based candidates
    if parent is not None:
        for n in [256, 512, 1024, 2048, 4096]:
            pi = _interp(parent, n)
            sc = _score_norm(pi)
            candidates.append((sc, pi, n, 'parent'))
        if 0 < pn <= 8192:
            sc = _score_norm(parent)
            candidates.append((sc, parent.copy(), pn, 'parent_exact'))

    # Sort by score
    candidates = sorted(candidates, key=_score_key)
    print(f"  {len(candidates)} candidates, best init: {candidates[0][0]:.8f} n={candidates[0][2]} [{candidates[0][3]}]")

    if candidates[0][1] is not None and candidates[0][0] < best_score:
        best_seq = np.asarray(candidates[0][1], dtype=np.float64).copy()
        best_score = float(candidates[0][0])
        print(f"  BEST: {best_score:.10f} n={best_seq.size} [init]")

    # Apply spectral shaping to top candidates
    phase1_end = t_start + 120.0
    spectral_candidates = []
    top_k = min(15, len(candidates))
    for i in range(top_k):
        if time.monotonic() > phase1_end - 5:
            break
        sc, p, n, label = candidates[i]
        # Try different gamma values
        for gamma in [0.7, 0.8, 0.9]:
            if time.monotonic() > phase1_end - 2:
                break
            p_shaped = _spectral_shape(p, n_iter=8, gamma=gamma)
            sc_shaped = _score_norm(p_shaped)
            spectral_candidates.append((sc_shaped, p_shaped, n, f'{label}_sp{gamma}'))

    spectral_candidates = sorted(spectral_candidates, key=_score_key)
    if spectral_candidates and spectral_candidates[0][0] < best_score:
        best_seq = np.asarray(spectral_candidates[0][1], dtype=np.float64).copy()
        best_score = float(spectral_candidates[0][0])
        print(f"  BEST after spectral: {best_score:.10f} n={best_seq.size} [{spectral_candidates[0][3]}]")

    # Combine all candidates
    all_candidates = candidates + spectral_candidates
    all_candidates = sorted(all_candidates, key=_score_key)

    print(f"Phase 1 done. Best: {best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    # === Phase 2: LP fixed-point on top candidates ===
    print("Phase 2: LP fixed-point optimization...")
    phase2_end = t_start + 500.0
    polished = []

    # Prefer smaller n for LP speed
    candidates_lp = sorted([c for c in all_candidates if c[2] <= 2048],
                           key=lambda x: (x[0], x[2]))

    count = 0
    for (sc, p, n, label) in candidates_lp:
        if time.monotonic() > phase2_end or count >= 12:
            break
        time_alloc = max(20.0, min(60.0, 380.0 * (256.0 / float(n))))
        per_end = min(phase2_end, time.monotonic() + time_alloc)

        a_lp, sc_lp = _lp_fixed_point(p, per_end, max_iters=30, use_line_search=True)
        if sc_lp < best_score:
            best_seq = np.asarray(a_lp, dtype=np.float64).copy()
            best_score = float(sc_lp)
            print(f"  BEST: {best_score:.10f} n={a_lp.size} [LP_{label}]")
        polished.append((sc_lp, a_lp, n, label + '_lp'))
        count += 1

    polished = sorted(polished, key=_score_key)
    print(f"Phase 2 done. Best: {best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    # === Phase 3: Adam refinement on best LP results ===
    print("Phase 3: Adam smooth-max refinement...")
    phase3_end = t_start + 1000.0

    for (sc, p, n, label) in polished[:5]:
        if time.monotonic() > phase3_end - 40:
            break
        per = min(120.0, phase3_end - time.monotonic() - 15)
        if per < 25:
            break
        a_ad, sc_ad = _adam_smooth_max(p, time.monotonic() + per, rng,
                                       lr=0.05, beta_decay=0.97, no_imp_limit=250)
        if sc_ad < best_score:
            best_seq = np.asarray(a_ad, dtype=np.float64).copy()
            best_score = float(sc_ad)
            print(f"  BEST: {best_score:.10f} n={a_ad.size} [Adam_{label}]")

        # LP polish after Adam
        if n <= 1024 and time.monotonic() < phase3_end - 30:
            per2 = min(40.0, phase3_end - time.monotonic() - 15)
            if per2 > 15:
                a_lp2, sc_lp2 = _lp_fixed_point(a_ad, time.monotonic() + per2,
                                                max_iters=12, use_line_search=True)
                if sc_lp2 < best_score:
                    best_seq = np.asarray(a_lp2, dtype=np.float64).copy()
                    best_score = float(sc_lp2)
                    print(f"  BEST: {best_score:.10f} n={a_lp2.size} [LPpostAdam_{label}]")

        # Also try spectral shaping on Adam result
        if time.monotonic() < phase3_end - 20:
            a_sp = _spectral_shape(a_ad, n_iter=5, gamma=0.85)
            sc_sp = _score_norm(a_sp)
            if sc_sp < best_score:
                best_seq = a_sp.copy()
                best_score = sc_sp
                print(f"  BEST: {best_score:.10f} n={a_sp.size} [Spectral_{label}]")

    print(f"Phase 3 done. Best: {best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    # === Phase 4: Multi-scale escalation ===
    print("Phase 4: Multi-scale escalation...")
    phase4_end = t_start + 1400.0

    if best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048, 4096]
        elif n_cur <= 1024:
            target_ns = [2048, 4096]
        elif n_cur <= 2048:
            target_ns = [4096]

        for tn in target_ns:
            if time.monotonic() > phase4_end - 60:
                break
            p_int = _interp(best_seq, tn)
            # Also try spectral shaping on the interpolated version
            p_int_sp = _spectral_shape(p_int, n_iter=5, gamma=0.85)

            sc_int = _score_norm(p_int)
            sc_int_sp = _score_norm(p_int_sp)
            print(f"  n={tn}: interp={sc_int:.8f}, spectral={sc_int_sp:.8f}")

            use_seq = p_int if sc_int < sc_int_sp else p_int_sp
            sc_use = min(sc_int, sc_int_sp)

            if sc_use < best_score:
                best_seq = use_seq.copy()
                best_score = sc_use
                print(f"  BEST: {best_score:.10f} n={tn} [interp/spectral]")

            # Adam on the best interpolated
            if time.monotonic() < phase4_end - 50:
                per = min(80.0, phase4_end - time.monotonic() - 30)
                if per > 20:
                    a_e, sc_e = _adam_smooth_max(use_seq, time.monotonic() + per, rng,
                                                 lr=0.03, beta_decay=0.97, no_imp_limit=200)
                    if sc_e < best_score:
                        best_seq = a_e.copy()
                        best_score = sc_e
                        print(f"  BEST: {best_score:.10f} n={tn} [Adam_esc]")

                    # Spectral polish
                    if time.monotonic() < phase4_end - 20:
                        a_esc_sp = _spectral_shape(a_e, n_iter=5, gamma=0.85)
                        sc_esc_sp = _score_norm(a_esc_sp)
                        if sc_esc_sp < best_score:
                            best_seq = a_esc_sp.copy()
                            best_score = sc_esc_sp
                            print(f"  BEST: {best_score:.10f} n={tn} [Spectral_esc]")

    print(f"Phase 4 done. Best: {best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    # === Phase 5: Random restarts with structured perturbations ===
    print("Phase 5: Random restarts...")
    phase5_end = t_start + 1650.0

    if best_seq is not None and time.monotonic() < phase5_end:
        n_best = best_seq.size
        n_restarts = 6
        per_r = max(15.0, (phase5_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase5_end:
                break
            per = min(per_r, phase5_end - time.monotonic() - 5)
            if per < 10:
                break

            if i == 0:
                # Reverse
                a_s = best_seq[::-1].copy()
            elif i == 1:
                # Shift (circular)
                shift = n_best // 4
                a_s = np.roll(best_seq, shift).copy()
            elif i == 2:
                # Noise
                ns = 0.08
                z = np.log(np.maximum(best_seq, 1e-12))
                z -= z.mean()
                z = z + ns * rng.normal(size=n_best)
                z = np.clip(z, -40, 40)
                z -= z.mean()
                a_s = np.exp(z - z.max())
                a_s = a_s / a_s.sum()
            elif i == 3:
                # Spectral perturbation
                N_p = _next_pow2(2 * n_best - 1)
                fa_p = np.fft.rfft(best_seq, N_p)
                fa_p = fa_p * (1 + 0.08 * rng.normal(size=len(fa_p)))
                a_s = np.fft.irfft(fa_p, N_p)[:n_best]
                a_s = np.maximum(a_s, 1e-12)
                a_s = a_s / a_s.sum()
            elif i == 4:
                # Blend with a different profile
                profile = _make_profile(n_best, 'quadratic', 2.0)
                blend = rng.uniform(0.3, 0.7)
                a_s = blend * best_seq + (1 - blend) * profile
                a_s = np.maximum(a_s, 1e-12)
                a_s = a_s / a_s.sum()
            else:
                # Spectral shaping with different gamma
                a_s = _spectral_shape(best_seq, n_iter=10, gamma=0.6)

            a_r, sc_r = _adam_smooth_max(a_s, time.monotonic() + per, rng,
                                         lr=0.04, beta_decay=0.96, no_imp_limit=150)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  BEST: {best_score:.10f} n={n_best} [restart{i}]")

            # Quick LP polish
            if n_best <= 1024 and time.monotonic() < phase5_end - 10:
                per2 = min(20.0, phase5_end - time.monotonic() - 5)
                if per2 > 10:
                    a_rl, sc_rl = _lp_fixed_point(a_r, time.monotonic() + per2,
                                                  max_iters=8, use_line_search=True)
                    if sc_rl < best_score:
                        best_seq = a_rl.copy()
                        best_score = sc_rl
                        print(f"  BEST: {best_score:.10f} n={n_best} [LP_restart{i}]")

    # === Phase 6: Final polish ===
    print("Phase 6: Final polish...")
    if time.monotonic() < deadline - 10 and best_seq is not None:
        per = deadline - time.monotonic() - 10
        if per > 15:
            if best_seq.size <= 1024:
                a_fp, sc_fp = _lp_fixed_point(best_seq, time.monotonic() + per,
                                              max_iters=20, use_line_search=True)
                if sc_fp < best_score:
                    best_seq = a_fp.copy()
                    best_score = sc_fp
                    print(f"  BEST: {best_score:.10f} n={best_seq.size} [final_LP]")
            elif best_seq.size <= 2048:
                # For larger n, use Adam with very small lr
                a_fp, sc_fp = _adam_smooth_max(best_seq, time.monotonic() + per, rng,
                                               lr=0.02, beta_decay=0.98, no_imp_limit=80)
                if sc_fp < best_score:
                    best_seq = a_fp.copy()
                    best_score = sc_fp
                    print(f"  BEST: {best_score:.10f} n={best_seq.size} [final_Adam]")

            # Final spectral touch
            if time.monotonic() < deadline - 5:
                a_final_sp = _spectral_shape(best_seq, n_iter=3, gamma=0.9)
                sc_final_sp = _score_norm(a_final_sp)
                if sc_final_sp < best_score:
                    best_seq = a_final_sp.copy()
                    best_score = sc_final_sp
                    print(f"  BEST: {best_score:.10f} n={best_seq.size} [final_spectral]")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile(512, 'linear', 1.5, 0, 0)

        exact_sc = _score_norm(final)
        print(f"\nFINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile(512, 'linear', 1.5, 0, 0)
    print("FALLBACK n=512")
    return [float(x) for x in fb]
