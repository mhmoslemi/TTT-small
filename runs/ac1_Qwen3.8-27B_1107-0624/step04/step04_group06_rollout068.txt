    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s < 1e-30:
        return 1e100, np.zeros(n)
    p = x / s
    c = _safe_conv(p)
    exact_m = float(c.max())
    exact_score = 2.0 * n * exact_m
    if exact_score < best_state[1]:
        best_state[0] = p.copy()
        best_state[1] = exact_score
    phi, w = _smooth_phi(c, beta)
    smooth_score = 2.0 * n * phi
    penalty = scale_eps * (s - 1.0) * (s - 1.0)

def _safe_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:

    a = np.where(np.isfinite(a), a, 0.0)

"""
Search for non-negative step-function heights minimizing
    2 * n * max(conv(a,a)) / (sum(a)^2).

Strategy:
- Warm-start from the provided global height_sequence_1 when available.
- Optimize mainly with L-BFGS-B on a scale-invariant log-sum-exp approximation
  of the max-convolution objective.
- Use a small scale penalty to keep the unnormalized parameter vector sane.
- Alternately apply an LP "cross-peak" refinement:
      for fixed current a, minimize t subject to sum(b)=1, b>=0, (a*b)_k <= t.
- Use multiplicative mirror-gradient polishing and micro zeroing of tiny
  coordinates at the end.
- Try one upsampling stage at a larger length if time remains.
"""

import time
import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _safe_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return float("inf")
    c = _safe_conv(a)
    if c.size == 0:
        return float("inf")
    m = float(np.max(c))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m / (s * s))


def _score_direct(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return float("inf")
    c = np.convolve(a, a, mode="full")
    m = float(np.max(c))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m / (s * s))


def _normalize_array(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def _project_simplex(v, target_sum=1.0):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return None
    v = np.where(np.isfinite(v), v, 0.0)
    if n == 1:
        w = np.array([target_sum if v[0] > 0 else 0.0], dtype=np.float64)
        sw = w.sum()
        if sw <= 0:
            w[0] = target_sum
        else:
            w[0] = target_sum
        return w

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # Projection onto {x >= 0, sum(x)=1}.
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        w = np.maximum(v, 0.0)
        sw = float(w.sum())
        if sw <= 0:
            return np.full(n, target_sum / n, dtype=np.float64)
        return w / sw * target_sum

    rho = int(np.where(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w * target_sum


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    n = z.size
    if n == 0:
        return np.array([], dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return e / s


def _smooth_phi(c, beta):
    """
    phi(c) = logsumexp(c / beta), stable via subtracting max.
    Returns (phi, weights) where weights are the gradient of phi wrt c.
    """
    c = np.asarray(c, dtype=np.float64).ravel()
    mlen = c.size
    if mlen == 0:
        return 0.0, np.array([], dtype=np.float64)
    m = float(np.max(c))
    idx = int(np.argmax(c))
    w = np.zeros(mlen, dtype=np.float64)

    if not np.isfinite(m) or m <= 0.0:
        w[idx] = 1.0
        return 0.0, w

    if not np.isfinite(beta) or beta <= 1e-300:
        tol = max(1e-15, 1e-10 * abs(m))
        mask = c >= (m - tol)
        cnt = int(mask.sum())
        if cnt > 0:
            w[mask] = 1.0 / cnt
        else:
            w[idx] = 1.0
        return m, w

    x = (c - m) / beta
    x = np.clip(x, -700.0, 0.0)
    e = np.exp(x)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        w[idx] = 1.0
        return m, w
    w = e / s
    phi = m + beta * float(np.log(s))
    if not np.isfinite(phi):
        w.fill(0.0)
        w[idx] = 1.0
        return m, w
    return float(phi), w


def _cross_corr_ra(r, p):
    """
    cross[i] = sum_k r[k] * p[k-i], valid p index, r out of [0, 2n-2] is zero.
    r is expected length 2*n-1, p length n.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    r = np.asarray(r, dtype=np.float64).ravel()
    n = p.size
    L = r.size
    if n == 0 or L == 0:
        return np.array([], dtype=np.float64)
    need = L + n - 1
    N = _next_pow2(max(1, need))
    fr = np.fft.rfft(r, N)
    fp = np.fft.rfft(p[::-1], N)
    cc = np.fft.irfft(fr * fp, N)
    start = n - 1
    out = cc[start:start + n]
    return np.maximum(out, 0.0)


def _make_linear_profile(n, eta):
    n = max(2, int(n))
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    if s <= 0:
        return np.full(n, 1.0 / n)
    return a / s


def _make_cosine_profile(n, freq, amp):
    n = max(2, int(n))
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / max(1, n))
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    if s <= 0:
        return np.full(n, 1.0 / n)
    return a / s


def _interpolate_seq(a, n_new):
    a = _normalize_array(a)
    if a is None:
        return None
    n_old = a.size
    n_new = max(2, int(n_new))
    if n_old == n_new:
        return a
    if n_old == 1:
        return np.full(n_new, 1.0 / n_new)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    b = np.interp(x_new, x_old, a)
    return _normalize_array(b)


def _lbfgs_objective_a(x, n, beta, best_state, scale_eps):
    """
    Objective/gradient for L-BFGS-B over unnormalized x >= 0.
    Internally optimizes p = x / sum(x).
    """
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size != n:
        return 1e100, np.zeros(n, dtype=np.float64)

    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if not np.isfinite(s) or s < 1e-300:
        return 1e100, np.zeros(n, dtype=np.float64)

    p = x / s
    c = _safe_conv(p)
    if c.size == 0:
        return 1e100, np.zeros(n, dtype=np.float64)

    m = float(np.max(c))
    exact = float(2.0 * n * m)
    if np.isfinite(exact) and exact < best_state[1]:
        best_state[0] = p.copy()
        best_state[1] = exact

    phi, w = _smooth_phi(c, beta)
    if not np.isfinite(phi):
        phi = m if np.isfinite(m) else 0.0
        if w.size == c.size:
            w.fill(0.0)
            w[int(np.argmax(c))] = 1.0
        else:
            w = np.zeros(c.size, dtype=np.float64)
            w[0] = 1.0

    smooth = 2.0 * n * phi
    penalty = scale_eps * (s - 1.0) * (s - 1.0)
    f = smooth + penalty
    if not np.isfinite(f):
        f = 1e100

    # dF/dp = 2n * dphi/dp = 2n * 2 * cross_corr(w,p)
    g_p = 4.0 * n * _cross_corr_ra(w, p)
    if not np.all(np.isfinite(g_p)):
        g_p = np.zeros(n, dtype=np.float64)

    dot = float(np.dot(p, g_p))
    if not np.isfinite(dot):
        dot = 0.0

    grad = (g_p - p * dot) / s
    grad = grad + 2.0 * scale_eps * (s - 1.0)
    grad = np.where(np.isfinite(grad), grad, 0.0)

    return float(f), grad


def _benchmark_eval_time(n, p, beta, trials=3):
    p = _normalize_array(p)
    if p is None:
        return 1e-4
    dummy = [p.copy(), _score_fft(p)]
    t0 = time.monotonic()
    for _ in range(max(1, int(trials))):
        _lbfgs_objective_a(p, n, beta, dummy, 1e-8)
    dt = (time.monotonic() - t0) / float(max(1, int(trials)))
    if not np.isfinite(dt) or dt < 1e-6:
        dt = 1e-5
    return float(dt)


def _lbfgs_run(p0, beta, time_budget, max_maxfun=4000, m=20, ftol=1e-12, gtol=1e-8):
    p0 = _normalize_array(p0)
    if p0 is None:
        return None, float("inf")
    n = p0.size
    if n < 2:
        return p0, _score_fft(p0)
    if n > 4096:
        return p0, _score_fft(p0)
    if not np.isfinite(time_budget) or time_budget < 5.0:
        return p0, _score_fft(p0)

    best_state = [p0.copy(), float(_score_fft(p0))]

    # Estimate evaluation time and choose a safe maxfun.
    est = _benchmark_eval_time(n, p0, beta, trials=2)
    max_fun_cap = 5500 if n <= 1536 else 2600
    maxfun = int(min(max_maxfun, max_fun_cap, max(30, (time_budget * 0.55) / max(est, 1e-9))))
    maxiter = int(max(100, maxfun))

    options = {
        "maxiter": maxiter,
        "maxfun": maxfun,
        "ftol": ftol,
        "gtol": gtol,
        "maxls": 10,
        "m": m,
    }

    try:
        res = minimize(
            _lbfgs_objective_a,
            p0.copy(),
            jac=True,
            method="L-BFGS-B",
            args=(n, beta, best_state, 1e-8),
            bounds=(0.0, 10.0),
            options=options,
        )
        if hasattr(res, "x") and res.x is not None:
            xp = _normalize_array(res.x)
            if xp is not None:
                sc = _score_fft(xp)
                if np.isfinite(sc) and sc < best_state[1]:
                    best_state[0] = xp
                    best_state[1] = float(sc)
    except Exception:
        pass

    return best_state[0], float(best_state[1])


def _lp_min_t(a, max_n=1200):
    """
    For fixed a (nonnegative, normalized), solve:
        minimize  t
        subject to sum(b)=1, b>=0, (a*b)_k <= t for all k.
    Returns (b, t) or (None, None).
    """
    a = _normalize_array(a)
    if a is None:
        return None, None
    n = a.size
    if n < 2 or n > max_n:
        return None, None
    if n > 4096:
        return None, None

    n2 = 2 * n - 1
    total = n * n + n2
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    vals = np.empty(total, dtype=np.float64)

    idx = 0
    for k in range(n2):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idxs = np.arange(i_start, i_end + 1, dtype=np.int64)
            cnt = int(idxs.size)
            rows[idx:idx + cnt] = k
            cols[idx:idx + cnt] = idxs.astype(np.int32)
            vals[idx:idx + cnt] = a[k - idxs]
            idx += cnt
        rows[idx] = k
        cols[idx] = n
        vals[idx] = -1.0
        idx += 1

    A_ub = csr_matrix((vals, rows, cols), shape=(n2, n + 1))
    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0
    b_ub = np.zeros(n2, dtype=np.float64)

    A_eq = np.ones((1, n + 1), dtype=np.float64)
    A_eq[0, n] = 0.0
    b_eq = np.array([1.0], dtype=np.float64)

    bounds = [(0.0, None)] * n + [(0.0, None)]

    res = linprog(
        c_obj,
        A_ub=A_ub,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method="highs",
    )

    if res is None or not getattr(res, "success", False):
        return None, None

    if res.x is None or res.x.size != n + 1:
        return None, None

    b = np.maximum(res.x[:n], 0.0)
    t = float(res.x[n])
    if not np.isfinite(t):
        return None, None
    b = np.where(np.isfinite(b), b, 0.0)
    s = float(b.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None, None
    b = b / s
    return b, max(0.0, t)


def _lp_mix_best(a, b):
    a = _normalize_array(a)
    b = _normalize_array(b)
    if a is None or b is None:
        return a, float("inf")

    best = a.copy()
    best_sc = float(_score_fft(a))

    sc_b = float(_score_fft(b))
    if sc_b < best_sc:
        best = b.copy()
        best_sc = sc_b

    for lam in (1.0, 0.85, 0.7, 0.55, 0.4, 0.25):
        mix = (1.0 - lam) * a + lam * b
        mix = _normalize_array(mix)
        if mix is None:
            continue
        sc = float(_score_fft(mix))
        if sc < best_sc:
            best = mix.copy()
            best_sc = sc

    return best, float(best_sc)


def _mirror_descent(a0, deadline, beta_frac=0.08, eta=0.10,
                    beta_decay=0.995, no_imp_limit=250, seed=42):
    a = _normalize_array(a0)
    if a is None:
        return None, float("inf")
    n = a.size
    if n < 2:
        return a, float(_score_fft(a))

    best_a = a.copy()
    best_sc = float(_score_fft(a))
    rng = np.random.default_rng(int(seed) % (2**31 - 1))

    c = _safe_conv(a)
    m0 = float(c.max()) if c.size else 1.0
    beta = max(1e-9, beta_frac * max(m0, 1e-12))
    base_eta = float(eta)
    no_imp = 0
    it = 0

    while time.monotonic() < deadline:
        c = _safe_conv(a)
        phi, w = _smooth_phi(c, beta)
        if w.size != c.size or not np.all(np.isfinite(w)):
            break

        # dF/dp = 4n cross(w,p)
        g_p = 4.0 * n * _cross_corr_ra(w, a)
        if not np.all(np.isfinite(g_p)):
            break

        dot = float(np.dot(a, g_p))
        if not np.isfinite(dot):
            break

        g_t = g_p - a * dot
        gnorm = float(np.linalg.norm(g_t))
        if not np.isfinite(gnorm) or gnorm < 1e-20:
            break

        logfac = -base_eta * (g_t / gnorm)
        logfac = np.clip(logfac, -6.0, 6.0)
        a2 = a * np.exp(logfac)
        a2 = _normalize_array(a2)
        a2 = _project_simplex(a2) if a2 is not None else None
        if a2 is None:
            break

        sc2 = float(_score_fft(a2))
        if np.isfinite(sc2) and sc2 < best_sc - 1e-15:
            best_a = a2.copy()
            best_sc = sc2
            no_imp = 0
        else:
            no_imp += 1

        a = a2
        it += 1
        if it % 50 == 0:
            beta = max(1e-11, beta * beta_decay)

        if no_imp > no_imp_limit:
            z = np.log(np.maximum(best_a, 1e-12))
            z = z - float(np.mean(z))
            noise = 0.08 if best_sc < 1.52 else 0.16
            z = z + noise * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            a = _softmax(z)
            c = _safe_conv(a)
            beta = max(1e-9, 0.08 * max(float(c.max()) if c.size else 1e-12, 1e-12))
            no_imp = 0

    return best_a, float(best_sc)


def _perturb_simplex(p, noise, seed=1):
    p = _normalize_array(p)
    if p is None:
        return None
    z = np.log(np.maximum(p, 1e-12))
    z = z - float(np.mean(z))
    rng = np.random.default_rng(int(seed) % (2**31 - 1))
    z = z + float(noise) * rng.normal(size=p.size)
    z = np.clip(z, -50.0, 50.0)
    return _softmax(z)


def _micro_zeroing(a, max_trials=1000):
    a = _normalize_array(a)
    if a is None:
        return None, float("inf")
    sc = float(_score_fft(a))

    # Zero all tiny entries by increasing thresholds.
    for thresh in (1e-7, 1e-6, 5e-6, 1e-5, 5e-5, 1e-4):
        a2 = a.copy()
        a2[a2 < thresh] = 0.0
        a2 = _normalize_array(a2)
        if a2 is None:
            continue
        sc2 = float(_score_fft(a2))
        if sc2 < sc - 1e-15:
            a = a2
            sc = sc2

    # Greedy single-coordinate zeroing among smallest entries.
    order = np.argsort(a)
    tried = 0
    for idx in order:
        if tried >= max_trials:
            break
        if a[idx] > 5e-5:
            break
        a2 = a.copy()
        a2[idx] = 0.0
        a2 = _normalize_array(a2)
        if a2 is None:
            break
        sc2 = float(_score_fft(a2))
        tried += 1
        if sc2 < sc - 1e-15:
            a = a2
            sc = sc2

    return a, float(sc)


def _safe_final_score(a):
    a = _normalize_array(a)
    if a is None:
        return float("inf")
    if a.size <= 4096:
        sc = _score_direct(a)
        if np.isfinite(sc):
            return sc
    return _score_fft(a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1800.0
    if not np.isfinite(budget) or budget <= 0:
        budget = 1800.0
    budget = max(30.0, budget)

    t_start = time.monotonic()
    deadline = t_start + max(5.0, budget - 45.0)

    seed_int = int(seed) % (2**31 - 1)

    # Warm start from provided construction if present.
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = None

    best = None
    best_score = float("inf")
    global_best = None
    global_score = float("inf")

    if parent is not None:
        pn = _normalize_array(parent)
        if pn is not None:
            if pn.size > 4096:
                gb = _interpolate_seq(pn, 2048)
            else:
                gb = pn.copy()
            if gb is not None:
                global_best = gb.copy()
                global_score = float(_safe_final_score(gb))
                if pn.size < 64:
                    best = _interpolate_seq(pn, 1024)
                else:
                    best = pn.copy()
    if best is None:
        n0 = 1024
        cands = [
            _make_linear_profile(n0, 1.5),
            _make_linear_profile(n0, 2.0),
            _make_linear_profile(n0, 2.6),
            _make_cosine_profile(n0, 1, 0.65),
            _make_cosine_profile(n0, 2, 0.35),
        ]
        for cc in cands:
            sc = float(_score_fft(cc))
            if sc < best_score:
                best = cc.copy()
                best_score = sc
        global_best = best.copy()
        global_score = best_score

    print(f"initial n={best.size}, score={best_score:.10f}")

    # --- Phase 1: L-BFGS-B at native length, decreasing smoothing ---
    ph1_end = min(deadline, t_start + 600.0)
    betas1 = [0.35, 0.18, 0.09, 0.045, 0.022, 0.011]
    for frac in betas1:
        rem = ph1_end - time.monotonic()
        if rem < 25.0:
            break
        c = _safe_conv(best)
        m0 = float(c.max()) if c.size else 1e-12
        beta = max(1e-9, frac * max(m0, 1e-12))
        new, sc = _lbfgs_run(best, beta, min(rem, 170.0), max_maxfun=5000, m=20,
                             ftol=1e-12, gtol=1e-8)
        if new is not None and np.isfinite(sc) and sc < best_score - 1e-14:
            best = new.copy()
            best_score = float(sc)
            print(f"phase1 lbfgs beta_frac={frac:.3f} n={best.size} score={best_score:.10f}")

    # --- Phase 2: Mirror-gradient polishing ---
    ph2_end = min(deadline, t_start + 760.0)
    if time.monotonic() < ph2_end - 25.0:
        new, sc = _mirror_descent(best, ph2_end, beta_frac=0.07, eta=0.08,
                                  beta_decay=0.995, no_imp_limit=220,
                                  seed=seed_int + 1)
        if new is not None and np.isfinite(sc) and sc < best_score - 1e-14:
            best = new.copy()
            best_score = float(sc)
            print(f"phase2 mirror n={best.size} score={best_score:.10f}")

    # --- Phase 3: LP cross-peak refinement ---
    ph3_end = min(deadline, t_start + 1080.0)
    fail_streak = 0
    solved = 0
    while time.monotonic() < ph3_end - 10.0:
        if best.size > 1200:
            break
        b, tv = _lp_min_t(best, max_n=1200)
        if b is None:
            break
        solved += 1
        mix, sc_mix = _lp_mix_best(best, b)
        if mix is not None and np.isfinite(sc_mix) and sc_mix < best_score - 1e-14:
            best = mix.copy()
            best_score = float(sc_mix)
            fail_streak = 0
            print(f"phase3 lp solve#{solved} t={float(tv):.8g} n={best.size} score={best_score:.10f}")
        else:
            fail_streak += 1
            if fail_streak >= 3:
                break

        if solved >= 12:
            break

        rem = ph3_end - time.monotonic()
        if rem > 35.0 and best.size <= 2500:
            c = _safe_conv(best)
            m0 = float(c.max()) if c.size else 1e-12
            beta = max(1e-9, 0.012 * max(m0, 1e-12))
            new, sc = _lbfgs_run(best, beta, min(35.0, rem - 5.0), max_maxfun=1800, m=16,
                                 ftol=1e-12, gtol=1e-8)
            if new is not None and np.isfinite(sc) and sc < best_score - 1e-14:
                best = new.copy()
                best_score = float(sc)
                print(f"phase3 post-lp lbfgs n={best.size} score={best_score:.10f}")

    # --- Phase 4: Upsampled optimization if time remains ---
    ph4_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < ph4_end - 70.0 and best.size <= 1600:
        n2 = min(2 * best.size, 2048)
        work2 = _interpolate_seq(best, n2)
        if work2 is not None:
            work_score = float(_score_fft(work2))
            if work_score < best_score:
                best = work2.copy()
                best_score = work_score
                print(f"phase4 upsample n={best.size} score={best_score:.10f}")

            betas2 = [0.25, 0.12, 0.06, 0.03]
            for frac in betas2:
                rem = ph4_end - time.monotonic()
                if rem < 40.0:
                    break
                c = _safe_conv(work2)
                m0 = float(c.max()) if c.size else 1e-12
                beta = max(1e-9, frac * max(m0, 1e-12))
                new, sc = _lbfgs_run(work2, beta, min(rem, 180.0),
                                     max_maxfun=3200 if n2 <= 1536 else 2200,
                                     m=18, ftol=1e-12, gtol=1e-8)
                if new is not None:
                    work2 = new.copy()
                    work_score = float(sc)
                    if work_score < best_score:
                        best = work2.copy()
                        best_score = work_score
                        print(f"phase4 lbfgs ups n={best.size} beta_frac={frac:.3f} score={best_score:.10f}")

    # --- Phase 5: Final polishing and perturbations ---
    while time.monotonic() < deadline - 35.0:
        rem = deadline - time.monotonic()
        if rem < 35.0:
            break
        c = _safe_conv(best)
        m0 = float(c.max()) if c.size else 1e-12
        beta = max(1e-10, 0.004 * max(m0, 1e-12))

        improved_this_loop = False
        if best.size <= 2500:
            new, sc = _lbfgs_run(best, beta, min(120.0, rem - 15.0),
                                 max_maxfun=2600 if best.size <= 1536 else 1800,
                                 m=18, ftol=1e-12, gtol=1e-8)
            if new is not None and np.isfinite(sc) and sc < best_score - 1e-14:
                best = new.copy()
                best_score = float(sc)
                improved_this_loop = True
                print(f"final lbfgs n={best.size} score={best_score:.10f}")

        if not improved_this_loop:
            for noise in (0.03, 0.08, 0.16):
                if time.monotonic() > deadline - 40.0:
                    break
                p2 = _perturb_simplex(best, noise, seed=seed_int + int(1000 * noise))
                if p2 is None:
                    continue
                rem2 = deadline - time.monotonic() - 15.0
                if rem2 < 25.0:
                    break
                c2 = _safe_conv(p2)
                m1 = float(c2.max()) if c2.size else 1e-12
                beta2 = max(1e-10, 0.012 * max(m1, 1e-12))
                new2, sc2 = _lbfgs_run(p2, beta2, rem2,
                                       max_maxfun=1400 if p2.size <= 1536 else 900,
                                       m=16, ftol=1e-12, gtol=1e-8)
                if new2 is not None and np.isfinite(sc2) and sc2 < best_score - 1e-14:
                    best = new2.copy()
                    best_score = float(sc2)
                    improved_this_loop = True
                    print(f"final perturb noise={noise:.2f} n={best.size} score={best_score:.10f}")
                    break

        if time.monotonic() > deadline - 20.0:
            break

    # --- Final micro cleanup ---
    best, best_score = _micro_zeroing(best, max_trials=700)
    if best is not None and np.isfinite(best_score):
        final_exact = _safe_final_score(best)
        if np.isfinite(final_exact):
            best_score = float(final_exact)
        print(f"after micro n={best.size} score={best_score:.10f}")

    # Compare with original/global candidate if shorter and safe.
    final = best
    final_score = best_score
    if global_best is not None and global_best.size <= 4096:
        gscore = _safe_final_score(global_best)
        if np.isfinite(gscore) and gscore < final_score:
            final = global_best.copy()
            final_score = float(gscore)

    final = _normalize_array(final)
    if final is None:
        final = _make_linear_profile(1024, 2.0)
        final_score = _safe_final_score(final)

    # Final validation against the evaluator, if available.
    try:
        ev = float(evaluate_sequence([float(x) for x in final]))
        if not np.isfinite(ev):
            raise ValueError("evaluator non-finite")
        # The evaluator is authoritative for returned correctness.
        # If it somehow reports a worse value, still return a finite valid sequence.
        print(f"evaluator score={ev:.10f}, internal={final_score:.10f}, n={final.size}")
    except Exception:
        final = _normalize_array(_make_linear_profile(1024, 2.0))
        if final is None:
            final = np.full(1024, 1.0 / 1024.0)
        print(f"fallback n={final.size} score={_safe_final_score(final):.10f}")

    final = np.asarray(final, dtype=np.float64).ravel()
    final = np.clip(final, 0.0, 1000.0)
    s = float(final.sum())
    if not np.isfinite(s) or s <= 0.01:
        final = np.full(1024, 1.0 / 1024.0)
    else:
        # Keep exactly normalized to avoid evaluator edge cases.
        final = final / s

    return [float(x) for x in final]
