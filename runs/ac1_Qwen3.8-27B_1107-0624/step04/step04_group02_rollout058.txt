import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    """Score with sum(a)=1 assumed: 2n*max(conv)."""
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())


def _score_full(a):
    """Full score: 2n*max(conv)/sum^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _weighted_subgrad(a, temperature, K_top):
    """Compute soft-weighted subgradient using top-K + temperature weighting.
    Returns (gradient_vector, max_conv_value)."""
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    k_actual = min(K_top, len(c))
    top_idx = np.argsort(c)[-k_actual:]
    top_vals = c[top_idx]
    # Soft weighting over top-K
    w = np.exp(top_vals / max(temperature, 1e-10))
    w = w / w.sum()
    grad = np.zeros(n)
    for i, k_val in enumerate(top_idx):
        k_val = int(k_val)
        lo = max(0, k_val - (n - 1))
        hi = min(n - 1, k_val)
        if lo <= hi:
            grad[lo:hi + 1] += w[i] * 2.0 * a[k_val - hi:k_val - lo + 1][::-1]
    return grad, cm


def _mirror_descent(a_init, deadline, rng, eta0=0.05, K_top=10, temp0=0.01,

def _lbfgs_smoothing(a_init, deadline, rng, max_wall_time=None):
    """L-BFGS with log-sum-exp continuation for precise local optimization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max_c0 for b in beta_starts]

    if max_wall_time is not None:
        per_level = max(3.0, max_wall_time / len(betas))
    else:
        per_level = 30.0

    for beta in betas:
        if time.monotonic() > deadline - 2.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        def obj_lse(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            cs = (c_ - cm_) / beta
            e_ = np.exp(cs)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            return 2.0 * n * (cm_ + beta * np.log(s_))

        def grad_lse(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            cs = (c_ - cm_) / beta
            e_ = np.exp(cs)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            W = float(np.dot(w, c_))
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g

        try:
            result = minimize(
                obj_lse, z, jac=grad_lse,
                method='L-BFGS-B',
                options={'maxiter': 2000, 'ftol': 1e-20, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 10000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z = result.x
                a_new = _softmax_fn(z)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _block_local_search(a_init, deadline, rng, n_trials=20000):
    """Block-level local search with simulated annealing acceptance."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    T = 0.001
    T_min = 0.00001

    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break
        # Random block perturbation
        block_len = int(rng.integers(5, max(6, n // 16)))
        start = int(rng.integers(0, n - block_len))
        a_trial = a.copy()
        # Two types of perturbation
        if rng.random() < 0.5:
            # Multiplicative
            pert = rng.uniform(0.9, 1.1, size=block_len)
            a_trial[start:start + block_len] *= pert
        else:
            # Additive redistribution
            delta = rng.normal(0, 0.02, size=block_len)
            a_trial[start:start + block_len] += delta
        a_trial = np.maximum(a_trial, 1e-10)
        a_trial /= a_trial.sum()
        sc = _score_norm(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
        elif rng.random() < np.exp(-(sc - best_score) / T):
            a = a_trial.copy()
        T = max(T_min, T * 0.9995)

    return best_a, best_score


def _make_profile_mono(n, power, direction=1):
    """Monotone profile: a_j proportional to (j+1)^power or (n-j)^power."""
    j = np.arange(n, dtype=np.float64)
    if direction == 1:
        a = (j + 1.0) ** power
    else:
        a = (n - j) ** power
    return a / a.sum()


def _make_profile_exp(n, rate, direction=1):
    """Exponential profile."""
    j = np.arange(n, dtype=np.float64)
    if direction == 1:
        a = np.exp(rate * j / n)
    else:
        a = np.exp(-rate * j / n)
    return a / a.sum()


def _make_profile_poly(n, coeffs):
    """Polynomial profile: a_j = sum c_k * (j/(n-1))^k."""
    t = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = np.zeros(n)
    for c in coeffs:
        a += c
    # Use numpy.polyval style
    a = np.zeros(n)
    degree = len(coeffs) - 1
    for k, c in enumerate(coeffs):
        a += c * t ** k
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_bimodal(n, peak_pos, width):
    """Bimodal: two Gaussians."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.exp(-((t - peak_pos) ** 2) / (2 * width ** 2)) + \
        np.exp(-((t - (1 - peak_pos)) ** 2) / (2 * width ** 2))
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _mutation(a, rng, strength=0.05):
    """Create a mutated variant of a."""
    n = len(a)
    a_mut = a.copy()
    # Add multiplicative noise
    noise = rng.uniform(1 - strength, 1 + strength, size=n)
    a_mut *= noise
    # Maybe swap a block
    if rng.random() < 0.3:
        blen = int(rng.integers(10, max(11, n // 8)))
        s1 = int(rng.integers(0, n - blen))
        s2 = int(rng.integers(0, n - blen))
        if s1 != s2:
            tmp = a_mut[s1:s1 + blen].copy()
            a_mut[s1:s1 + blen] = a_mut[s2:s2 + blen]
            a_mut[s2:s2 + blen] = tmp
    a_mut = np.maximum(a_mut, 1e-15)
    return a_mut / a_mut.sum()

            p_best2, sc_best2 = _lbfgs_smoothing(best_seq, dl2, rng, max_wall_time=wall2)
            _update(p_best2, sc_best2, "lbfgs_polish2")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Block local search (200s) ---
    phase3_end = min(deadline, t_start + 1060.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        dl = phase3_end - 5.0
        p_best, sc_best = _block_local_search(best_seq, dl, rng, n_trials=30000)
        _update(p_best, sc_best, "block_search")

        # L-BFGS after local search
        if time.monotonic() < phase3_end - 20:
            dl2 = phase3_end - 5.0
            wall2 = max(10.0, dl2 - time.monotonic())
            p_best2, sc_best2 = _lbfgs_smoothing(best_seq, dl2, rng, max_wall_time=wall2)
            _update(p_best2, sc_best2, "lbfgs_after_bs")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Mutation restarts (300s) ---
    phase4_end = min(deadline, t_start + 1360.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        n_restarts = 4
        per_restart = max(20.0, (phase4_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 10:
                break
            strength = 0.02 + 0.02 * i
            p_mut = _mutation(best_seq, rng, strength=strength)
            sc_mut = _score_norm(p_mut)

            # Quick mirror descent on mutant
            dl = min(phase4_end - 5, time.monotonic() + per_restart * 0.6)
            p_best, sc_best = _mirror_descent(p_mut, dl, rng, eta0=0.02, K_top=10,
                                             temp0=0.003, temp_decay=0.9995)
            _update(p_best, sc_best, f"mut_md_{i}")

            # L-BFGS on best of mutant
            if time.monotonic() < phase4_end - 5:
                dl2 = min(phase4_end - 5, time.monotonic() + per_restart * 0.4)
                wall2 = max(5.0, dl2 - time.monotonic())
                p_best2, sc_best2 = _lbfgs_smoothing(p_best, dl2, rng, max_wall_time=wall2)
                _update(p_best2, sc_best2, f"mut_lbfgs_{i}")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Try different n (200s) ---
    phase5_end = min(deadline, t_start + 1560.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        # Try scaling up
        for target_n in [8192]:
            if time.monotonic() > phase5_end - 30:
                break
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm(p_interp)
            _update(p_interp, sc_interp, f"interp_n{target_n}")

            if time.monotonic() < phase5_end - 20:
                dl = min(phase5_end - 10, time.monotonic() + 120.0)
                p_best, sc_best = _lbfgs_smoothing(p_interp, dl, rng, max_wall_time=100.0)
                _update(p_best, sc_best, f"lbfgs_n{target_n}")

        # Try scaling down
        if len(best_seq) > 2048:
            p_small = _interp_profile(best_seq, 2048)
            sc_small = _score_norm(p_small)
            _update(p_small, sc_small, "interp_n2048")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Final intensive polishing (remaining time) ---
    if best_seq is not None:
        remaining = max(10.0, deadline - time.monotonic())
        # Multiple polishing rounds
        for round_idx in range(3):
            if time.monotonic() > deadline - 5:
                break
            per_round = remaining / (3 - round_idx)
            if round_idx == 0:
                # L-BFGS with very fine continuation
                dl = min(deadline - 5, time.monotonic() + per_round)
                wall = max(10.0, dl - time.monotonic())
                p_best, sc_best = _lbfgs_smoothing(best_seq, dl, rng, max_wall_time=wall)
                _update(p_best, sc_best, f"final_lbfgs_{round_idx}")
            elif round_idx == 1:
                # Very careful mirror descent
                dl = min(deadline - 5, time.monotonic() + per_round)
                p_best, sc_best = _mirror_descent(best_seq, dl, rng, eta0=0.01,
                                                 K_top=5, temp0=0.002,
                                                 temp_decay=0.9999)
                _update(p_best, sc_best, f"final_md_{round_idx}")
            else:
                # Final block search with very low T
                dl = min(deadline - 5, time.monotonic() + per_round)
                p_best, sc_best = _block_local_search(best_seq, dl, rng, n_trials=20000)
                _update(p_best, sc_best, f"final_block_{round_idx}")
                # Quick L-BFGS after
                if time.monotonic() < deadline - 10:
                    dl2 = min(deadline - 5, time.monotonic() + 30.0)
                    wall2 = max(5.0, dl2 - time.monotonic())
                    p_best2, sc_best2 = _lbfgs_smoothing(best_seq, dl2, rng, max_wall_time=wall2)
                    _update(p_best2, sc_best2, f"final_lbfgs_{round_idx}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_profile_mono(4096, 0.5, 1)
            s = float(final_seq.sum())
        # Keep sum as-is (scale invariant objective)
        # But ensure it's valid
        final_seq = final_seq  # already normalized to sum=1 from optimization

        sc_final = _score_full(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        # Validate with evaluator if feasible
        if len(final_seq) <= 8192:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                else:
                    print(f"Evaluator returned non-finite: {sc_eval}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profile_mono(4096, 0.5, 1)
    sc_fb = _score_full(fallback)
    print(f"FALLBACK: n=4096, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
