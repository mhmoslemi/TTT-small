    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    """Compute 2*n*max(conv(a,a)) for normalized sequence (sum=1)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_full(a):
    """Compute the full evaluation score (scale-invariant)."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _proj_simplex(v):
    """Project v onto the standard simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Binary search for tau
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - 1))[0][-1]
    tau = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - tau, 0)

def _make_decreasing_profile(n, p):
    """Power-law decreasing: a[j] ∝ (1 - j/(n-1))^p + eps"""
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 - t) ** p + 1e-10
    return a / a.sum()

def _make_exponential_profile(n, rate):
    """Exponential decay: a[j] ∝ exp(-rate * j/(n-1))"""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-rate * t)
    return a / a.sum()

def _make_geometric_profile(n, ratio):
    """Geometric: a[j] ∝ ratio^j"""
    j = np.arange(n)
    a = ratio ** j
    return a / a.sum()

def _make_bimodal_profile(n, pos, width):
    """Two peaks"""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.2) ** 2) / (2 * width ** 2)) + np.exp(-((t - pos) ** 2) / (2 * width ** 2))
    return a / a.sum()

def _make_spectral_profile(n, freq):
    """Based on spectral seeding: a[j] ∝ |sum_k exp(2πi freq k^2/n)|^2 type construction"""
    # Use a chirp-like profile
    t = np.linspace(0.0, 1.0, n)
    a = np.abs(np.sin(freq * t + 0.5 * t ** 2)) ** 2 + 1e-10
    return a / a.sum()

def _active_lag_subgradient(a, lr, deadline, n_iters=50000):
    """
    Active-lag subgradient method with projection onto simplex.
    a is normalized (sum=1).
    Returns best (a, score) found.
    """
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    
    cur_a = a.copy()
    cur_score = best_score
    
    lr_cur = lr
    patience = 0
    no_improve = 0
    best_patience_score = best_score
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(cur_a, cur_a)
        k_star = int(np.argmax(c))
        C = c[k_star]
        
        # Gradient of conv[k_star] w.r.t. a
        # d conv[k]/da[m] = 2 * a[k-m] (with 0-padding)
        # The relevant slice: m ranges such that 0 <= k-m < n
        # i.e., k-(n-1) <= m <= k
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        
        grad = np.zeros(n)
        # a[k_star - m] for m in [lo, hi]
        # When m = lo, k_star - m = k_star - lo (which is in [0, n-1])
        # When m = hi, k_star - m = k_star - hi (which is in [0, n-1])
        indices = k_star - np.arange(lo, hi + 1)
        grad[lo:hi+1] = 2.0 * cur_a[indices]
        
        # Subtract mean for simplex compatibility (the constraint sum=1 means
        # only the "direction" matters, and adding a constant doesn't change
        # the projected gradient direction)
        grad_mean = grad.mean()
        grad = grad - grad_mean
        
        # Adaptive step size
        cur_a_new = cur_a - lr_cur * grad
        cur_a_new = _proj_simplex(cur_a_new)
        
        sc = _score_norm(cur_a_new)
        if np.isfinite(sc):
            if sc < cur_score * (1 - 1e-10):
                cur_a = cur_a_new
                cur_score = sc
                no_improve = 0
                if sc < best_score:
                    best_score = sc
                    best_a = cur_a.copy()
                    best_patience_score = sc
            else:
                no_improve += 1
                if no_improve > 500:
                    lr_cur *= 0.5
                    no_improve = 0
                elif no_improve > 100:
                    lr_cur *= 0.9
        else:
            lr_cur *= 0.5
        
        if it % 5000 == 0:
            # Occasionally try a slightly larger step
            if lr_cur < 0.001:
                lr_cur = 0.01
    
    return best_a, best_score

def _logsumexp_optimize(a_init, deadline, n_continuations=8):
    """
    L-BFGS optimization with log-sum-exp continuation.
    a_init is normalized (sum=1).
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    # Continuation schedule: geometric
    alphas = max_c0 * np.array([0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005])
    
    time_per = max(5.0, (deadline - time.monotonic()) / len(alphas))
    
    for alpha in alphas:
        if time.monotonic() > deadline - 3:
            break
        level_dl = min(deadline - 3, time.monotonic() + time_per)
        
        def obj(z_):
            a_ = _softmax(z_)
            c_ = _full_convolve(a_, a_)
            cm = float(c_.max())
            c_shift = (c_ - cm) / alpha
            c_shift = c_shift - c_shift.max()  # For numerical stability
            e = np.exp(c_shift)
            s = e.sum()
            if s < 1e-300:
                return 2.0 * n * cm
            return 2.0 * n * (cm + alpha * np.log(s))
        
        def grad_fn(z_):
            a_ = _softmax(z_)
            c_ = _full_convolve(a_, a_)
            cm = float(c_.max())
            c_shift = (c_ - cm) / alpha
            c_shift = c_shift - c_shift.max()
            e = np.exp(c_shift)
            s = e.sum()
            if s < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e / s
            # d/dz_ [max(c) + alpha * log(sum(exp((c-max)/alpha)))]
            # = alpha * log(sum(exp(...))) derivative + max derivative
            # The gradient of max + alpha*logsumexp w.r.t. c is: w (the softmax weights)
            # Then chain rule: dc/dz * (z -> a -> c)
            # dc[k]/da[m] = 2*a[k-m] (0-padded)
            # da[m]/dz[l] = a[m](delta_ml - a[l])
            # So dc[k]/dz[l] = sum_m 2*a[k-m] * a[m](delta_ml - a[l])
            # = 2*a[k-l]*a[l] - 2*a[l]*sum_m a[k-m]*a[m]
            # = 2*a[k-l]*a[l] - 2*a[l]*conv[k]  ... wait that's not right.
            
            # Let me redo: dc[k]/dz[l] = sum_m (dc[k]/da[m]) * (da[m]/dz[l])
            # dc[k]/da[m] = 2*a[k-m] (0-padded)
            # da[m]/dz[l] = a[m]*delta[ml] - a[m]*a[l]
            # So dc[k]/dz[l] = 2*a[k-l]*a[l] - a[l]*sum_m 2*a[k-m]*a[m]
            # = 2*a[k-l]*a[l] - 2*a[l]*conv[k]
            
            # The objective gradient: sum_k w[k] * dc[k]/dz[l]
            # = sum_k w[k] * (2*a[k-l]*a[l] - 2*a[l]*conv[k])
            # = 2*a[l] * (sum_k w[k]*a[k-l] - sum_k w[k]*conv[k])
            # = 2*a[l] * (w_conv[l] - W)
            # where w_conv = w * a_reversed (flipped convolution)
            # Actually: sum_k w[k]*a[k-l] = (w * flip(a))[l + n - 1]... 
            # Let me think more carefully.
            # sum_k w[k]*a[k-l]: this is the cross-correlation of w and a at lag l.
            # = sum_k w[k]*a[k-l] = conv(w, flip(a))[l + len(a) - 1]... 
            # Actually it's simpler: let's define h[l] = sum_k w[k]*a[k-l]
            # For fixed l, sum over k such that 0 <= k-l < n, i.e., l <= k <= l+n-1
            # h[l] = sum_{k=l}^{l+n-1} w[k]*a[k-l] (with w[k]=0 for k out of [0, 2n-2])
            # = (w * a_reversed)[l + n - 1] where a_reversed = a[::-1]
            # Actually: (w * a_rev)[m] = sum_k w[k]*a_rev[m-k] = sum_k w[k]*a[n-1-(m-k)]
            # For m = l+n-1: = sum_k w[k]*a[n-1-l-n+1+k] = sum_k w[k]*a[k-l]
            # Yes! So h[l] = (w * a_rev)[l+n-1]
            
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n-1:2*n-1]  # h[l] for l = 0, ..., n-1
            W = float(np.dot(w, c_))
            
            g = 2.0 * a_ * (h - W)
            # Scale by 2n (from the objective)
            g *= 2.0 * n
            
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g
        
        try:
            result = minimize(
                obj, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-8,
                         'maxls': 50, 'maxfun': 3000, 'maxls': 30}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax(z_new)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass
    
    return best_a, best_score

def _lp_direction(a, C, sparsify=False):
    """
    Solve the LP: maximize sum(b) subject to b >= 0, (a*b)[k] <= C for all k.
    Returns the optimal b (or None if infeasible/too slow).
    """
    n = len(a)
    
    # For large n, this LP might be slow. Use a simplified version.
    # Only enforce constraints at the top few lags.
    c = _full_convolve(a, a)
    top_k = min(10, 2 * n - 1)
    top_indices = np.argsort(c)[-top_k:]
    
    # Build constraint matrix A: A[k, j] = a[k-j] if 0 <= k-j < n
    # For each constraint k, the row is a slice of a (reversed)
    A_rows = []
    b_vec = []
    for k in top_indices:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        row = np.zeros(n)
        # (a*b)[k] = sum_j a[j]*b[k-j]
        # For constraint k: sum_j a[j]*b[k-j] <= C
        # Let m = k-j, so j = k-m: sum_m a[k-m]*b[m] <= C
        # So A[k, m] = a[k-m] for valid m
        for m in range(0, n):
            j = k - m
            if 0 <= j < n:
                row[m] = a[j]
        A_rows.append(row)
        b_vec.append(C)
    
    A_arr = np.array(A_rows)
    b_arr = np.array(b_vec)
    
    # Maximize sum(b) = - minimize -sum(b)
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_arr, b_ub=b_arr, bounds=[(0, None)] * n,
                     method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def _coordinate_descent(a, deadline, n_passes=100):
    """
    Coordinate descent: optimize one element at a time.
    a is normalized (sum=1).
    """
    n = len(a)
    cur_a = a.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            
            old_val = cur_a[j]
            if old_val < 1e-15:
                continue
            
            # Try to optimize a[j] while keeping others fixed
            # conv[k] = old_conv[k] + a[j]*(a[k-j] + a[j]... ) 
            # Actually, changing a[j] from old to new:
            # conv[k] changes by: (new-old)*a[k-j] + a[j_old]*(new-old) ... 
            # Wait: conv[k] = sum_i a[i]*a[k-i]
            # If we change a[j] to new_val:
            # conv[k]_new = conv[k]_old + (new_val - old_val) * (a[k-j] + a[j]_old) 
            # No wait: 
            # conv[k] = sum_i a[i]*a[k-i]
            # The terms involving a[j]: 
            #   i=j: a[j]*a[k-j]
            #   k-i=j, i=k-j: a[k-j]*a[j] (same term if 2j=k)
            # So conv[k] = ... + a[j]*a[k-j] + ... (appears once or twice)
            # If j != k-j: conv[k] has terms a[j]*a[k-j] and a[k-j]*a[j] = 2*a[j]*a[k-j]
            # If j = k-j: conv[k] has term a[j]*a[j] = a[j]^2
            # 
            # d conv[k]/d a[j] = 2*a[k-j] (as we derived before)
            # 
            # So if we change a[j] by delta:
            # conv[k]_new ≈ conv[k]_old + 2*delta*a[k-j] (linear approx)
            # 
            # For exact: conv[k]_new = conv[k]_old + delta*(2*a[k-j] - delta*delta[k=2j])
            # (the -delta^2 term only when k=2j, because then a[j]*a[j] changes by 2*delta*a[j]+delta^2)
            
            # The max of conv changes in a complicated way.
            # For coordinate descent, we can try a few candidate values and pick the best.
            
            # Grid search over a[j]
            # The sum constraint: sum(a) = 1, so if we change a[j] by delta, 
            # we need to redistribute. But in coordinate descent on the simplex,
            # we typically fix the sum by adjusting another coordinate or 
            # working in the "free" space.
            # 
            # Alternative: work with the unnormalized objective.
            # Score = 2n*max(conv)/sum^2
            # If we change a[j] to new_val (keeping others fixed):
            # new_sum = old_sum + (new_val - old_val)
            # new_conv[k] = old_conv[k] + (new_val - old_val)*(2*a[k-j] - (new_val-old_val)*delta[k-2j])
            
            # This is getting complicated. Let me use a simpler approach:
            # Just try new_val = old_val * factor for a few factors.
            
            for factor in [0.9, 0.95, 1.05, 1.1]:
                trial = cur_a.copy()
                trial[j] = old_val * factor
                trial = np.maximum(trial, 0)
                s = trial.sum()
                if s < 1e-15:
                    continue
                trial = trial / s  # Renormalize
                sc = _score_norm(trial)
                if sc < best_score * (1 - 1e-12):
                    cur_a = trial
                    best_score = sc
                    best_a = trial.copy()
                    improved = True
                    break
        
        if not improved:
            # Try zeroing out small elements
            small = np.where(cur_a < 0.01 * cur_a.max())[0]
            for j in small[:5]:
                trial = cur_a.copy()
                trial[j] = 0
                s = trial.sum()
                if s < 1e-15:
                    continue
                trial = trial / s
                sc = _score_norm(trial)
                if sc < best_score:
                    cur_a = trial
                    best_score = sc
                    best_a = trial.copy()
                    improved = True
    
    return best_a, best_score

def _momentum_subgradient(a_init, deadline, lr_init=0.05, momentum=0.9):
    """
    Subgradient method with momentum and adaptive step size.
    """
    n = len(a_init)
    cur_a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    velocity = np.zeros(n)
    lr = lr_init
    no_improve = 0
    
    t = 0
    while time.monotonic() < deadline:
        t += 1
        c = _full_convolve(cur_a, cur_a)
        k_star = int(np.argmax(c))
        C = c[k_star]
        
        # Gradient of conv[k_star] w.r.t. a
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        grad = np.zeros(n)
        indices = k_star - np.arange(lo, hi + 1)
        grad[lo:hi+1] = 2.0 * cur_a[indices]
        
        # Center the gradient (for simplex)
        grad = grad - grad.mean()
        
        # Momentum update
        velocity = momentum * velocity - lr * grad
        cur_a_new = cur_a + velocity
        cur_a_new = _proj_simplex(cur_a_new)
        
        sc = _score_norm(cur_a_new)
        if np.isfinite(sc):
            if sc < best_score * (1 - 1e-12):
                best_score = sc
                best_a = cur_a_new.copy()
                no_improve = 0
            elif sc < _score_norm(cur_a) * 1.01:
                no_improve += 1
            else:
                no_improve += 5
            
            cur_a = cur_a_new
        else:
            no_improve += 10
        
        if no_improve > 200:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-6:
                lr = lr_init * 0.1
                velocity *= 0.1
        
        # Decay momentum slightly
        if t % 10000 == 0:
            momentum = max(0.5, momentum - 0.01)
    
    return best_a, best_score
