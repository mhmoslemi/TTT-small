"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c[np.abs(c) < 1e-15] = 0.0
    return c


def _score(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _simplex_project(v):
    """Exact Euclidean projection onto the probability simplex {x>=0, sum=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.where(cond)[0][-1]
    rho_val = idx + 1
    theta = (cssv[idx] - 1.0) / rho_val
    p = np.maximum(v - theta, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return p / s


def _cross_corr_fft(w, a):
    """Cross-correlation: result[i] = sum_k w[k] * a[k-i] (valid range)."""
    n = a.size
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    result = np.fft.irfft(fw * fa, N)
    return result[n - 1:2 * n - 1]


def _smooth_max_weights(c, beta):
    """Compute weights for smooth max approximation."""
    c_max = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= c_max - 1e-12
        w[mask] = 1.0 / mask.sum()
        return c_max, w
    x = (c - c_max) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return c_max, w
    w = ex / s
    val = c_max + beta * float(np.log(s))
    return val, w


def _fista_optimize(a0, deadline, beta_frac=0.03, beta_decay=0.96,
                    beta_min=1e-6, step_scale=0.8, restart_every=80):
    """FISTA with adaptive smooth-max parameter for minimizing max(C) over simplex."""
    n = a0.size
    a0 = np.asarray(a0, dtype=np.float64)
    a0 = np.maximum(a0, 0)
    s = a0.sum()
    if s < 1e-30:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / s

    z = a0.copy()
    z_prev = a0.copy()
    t = 1.0

    c = _full_convolve(z)
    c_max = float(c.max())
    beta = max(beta_frac * c_max, 1e-8)

    best_a = a0.copy()
    best_score = float(2.0 * n * c_max)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        if it > 1:
            y = z + ((t - 1.0) / t_new) * (z - z_prev)
            y = _simplex_project(y)
        else:
            y = z

        # Gradient at y
        c_y = _full_convolve(y)
        c_y_max = float(c_y.max())
        _, w = _smooth_max_weights(c_y, beta)
        g = 2.0 * _cross_corr_fft(w, y)

        # Step size: Lipschitz constant ~ 2 + 4*c_max/beta
        L = 2.0 + 4.0 * c_y_max / beta
        step = step_scale / L

        z_new = _simplex_project(y - step * g)

        z_prev = z.copy()
        z = z_new
        t = t_new

        # Evaluate
        c = _full_convolve(z)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive beta
        if it % 40 == 0:
            beta = max(beta_min, beta * beta_decay)

        # Nesterov restart
        if no_improve >= restart_every:
            t = 1.0
            z_prev = z.copy()
            no_improve = 0

    return best_a, best_score


def _adam_logit(a0, deadline, lr=0.05, beta_frac=0.06, beta_decay=0.94,

        rng = np.random.default_rng(123)

    z = np.log(p)
    z -= float(np.mean(z))

    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-4, beta_frac * maxc)

    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        t_adam += 1
        _, w = _smooth_max_weights(c, beta)
        g = _cross_corr_fft(w, p)

        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        p = e / e.sum()

        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:

            beta = max(1e-5, beta * beta_decay)

def _lp_solve(a, M):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a, deadline, t_init=0.35, t_min=0.08, t_decay=0.97,

    """Intensive LP refinement with adaptive M and decreasing t."""

    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _score(a)
    c = _full_convolve(a)
    M = float(c.max())
    t = t_init
    n_solves = 0
    n_infeasible = 0

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1

        b = _lp_solve(a, M)
        if b is None:
            n_infeasible += 1
            M *= 1.005
            if n_infeasible > 50:
                # Reset: use current peak
                c = _full_convolve(a)
                M = float(c.max()) * m_tol
                n_infeasible = 0
            continue
        n_infeasible = 0

        s_b = float(b.sum())
        if s_b < 1e-10:
            M *= 1.01
            continue

        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm

        if s_new < 1e-30:

        else:
            a_new = a_new / s_new

        sc_new = _score(a_new)

            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        c = _full_convolve(a)
        M_new = float(c.max())

        # Update M: use new peak (tight)
        M = M_new * m_tol
        # Slightly decrease mixing parameter
        t = max(t_min, t * t_decay)

    return best_a, best_score
