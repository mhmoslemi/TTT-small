    # We want (x ⋆ y)[i] = sum_j x[j] * y[i+j] for i = 0, ..., n-1
    # This is the "offset cross-correlation"
    
    # Using convolution: (x ⋆ y)[i] = sum_j x[j] * y[i+j]
    # = sum_j x[j] * y[i+j]
    # Let's define the convolution of x with reversed y:
    # (x * y_rev)[k] = sum_j x[j] * y_rev[k-j] = sum_j x[j] * y[len_y-1-(k-j)]
    # = sum_j x[j] * y[len_y-1-k+j]
    # We want this to equal sum_j x[j] * y[i+j]
    # So: len_y-1-k = i, i.e., k = len_y-1-i = 2n-2-i
    
    # So (x ⋆ y)[i] = (x * y_rev)[2n-2-i]
    
    # Let me just verify with a small example.
    # a = [1, 2], w = [0.1, 0.2, 0.3] (n=2, 2n-1=3)
    # (a ⋆ w)[0] = a[0]*w[0] + a[1]*w[1] = 1*0.1 + 2*0.2 = 0.5
    # (a ⋆ w)[1] = a[0]*w[1] + a[1]*w[2] = 1*0.2 + 2*0.3 = 0.8
    # 
    # x * y_rev where y_rev = [0.3, 0.2, 0.1]:
    # conv[0] = 1*0.3 = 0.3
    # conv[1] = 1*0.2 + 2*0.3 = 0.8
    # conv[2] = 1*0.1 + 2*0.2 = 0.5
    # conv[3] = 2*0.1 = 0.2
    # 
    # (a ⋆ w)[0] should = conv[2n-2-0] = conv[2] = 0.5 ✓
    # (a ⋆ w)[1] should = conv[2n-2-1] = conv[1] = 0.8 ✓
    
    # Great! So:
    # grad_a[i] = 2 * (a ⋆ w)[i] = 2 * conv(a, w_rev)[2n-2-i]
    
    w_rev = w[::-1]
    N2 = _next_pow2(n + len(w_rev) - 1)
    fa2 = np.fft.rfft(a, N2)
    fw2 = np.fft.rfft(w_rev, N2)
    conv_result = np.fft.irfft(fa2 * fw2, N2)[:n + len(w_rev) - 1]
    # conv_result[k] = sum_j a[j] * w_rev[k-j]
    # We want conv_result[2n-2-i] for i=0,...,n-1
    idx = 2*n-2 - np.arange(n)
    # idx goes from 2n-2 down to n-1
    corr = conv_result[idx]
    return 2.0 * corr


def _smooth_max_obj_and_grad(z, beta):
    """
    Objective and gradient for log-parameter z.
    a = softmax(z), objective = 2n * smooth_max(conv(a,a), beta)
    """
    n = len(z)
    a = _softmax(z)
    c = _full_convolve(a)
    sm_val = _smooth_max_val(c, beta)
    w = _smooth_max_grad(c, beta)
    
    # Gradient of smooth_max w.r.t. a
    grad_a = _conv_grad_wrt_a(a, w)
    
    # Gradient of a = softmax(z) w.r.t. z: da_i/dz_j = a_i*(delta_ij - a_j)
    # So grad_z_j = sum_i grad_a_i * da_i/dz_j = sum_i grad_a_i * a_i * (delta_ij - a_j)
    # = a_j * (grad_a_j - sum_i grad_a_i * a_i)
    dot = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - dot)
    
    obj = 2.0 * n * sm_val
    grad_obj = 2.0 * n * grad_z
    
    return obj, grad_obj

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    return e / e.sum()

def _make_initial(n, style='bump'):
    """Create initial sequence."""
    if style == 'bump':
        t = np.linspace(0, 1, n)
        a = np.exp(-8 * (t - 0.5)**2)
    elif style == 'linear':
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * t
    elif style == 'cosine':
        t = np.arange(n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * t / n)
    elif style == 'arch':
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
    elif style == 'flat':
        a = np.ones(n)
    elif style == 'bimodal':

        a = np.exp(-20*(t-0.25)**2) + np.exp(-20*(t-0.75)**2) + 0.1

def _adapt_beta(c, beta, min_beta=1e-6):
    """Adaptively adjust beta based on the spread of c."""
    m = float(c.max())
    sorted_c = np.sort(c)[::-1]
    if len(sorted_c) < 2:
        return beta
    gap = sorted_c[0] - sorted_c[1]
    if gap < beta * 0.1:
        return max(min_beta, beta * 0.5)
    elif gap > beta * 5:
        return min(m * 0.5, beta * 2.0)
    return beta

def _optimize_adam(z0, n, max_time, rng, beta_init=None, lr_init=0.1):
    """Adam optimization on log-parameters with adaptive smooth-max."""
    t0 = time.monotonic()
    deadline = t0 + max_time
    
    z = z0.copy().astype(np.float64)
    z = z - np.mean(z)
    
    if beta_init is None:
        a = _softmax(z)
        c = _full_convolve(a)
        beta = 0.1 * float(c.max())
    else:
        beta = beta_init
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    a = _softmax(z)
    c = _full_convolve(a)
    max_c = float(c.max())
    best_score = 2.0 * n * max_c
    best_z = z.copy()
    best_a = a.copy()
    
    no_improve = 0
    lr = lr_init
    
    while time.monotonic() < deadline:
        t += 1
        
        # Compute smooth max and gradient
        sm_val = _smooth_max_val(c, beta)
        w = _smooth_max_grad(c, beta)
        grad_a = _conv_grad_wrt_a(a, w)
        dot = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - dot)
        
        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m[:] = 0
            v[:] = 0
            t = 0
            continue
        
        # Adam update
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * grad_z**2
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = z - np.mean(z)
        z = np.clip(z, -50, 50)
        
        a = _softmax(z)
        c = _full_convolve(a)
        max_c = float(c.max())
        score = 2.0 * n * max_c
        
        if score < best_score - 1e-13:
            best_score = score
            best_z = z.copy()
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr = max(0.005, lr * 0.9)
        
        # Adapt beta
        if t % 20 == 0:
            beta = _adapt_beta(c, beta)
        
        # Restart if stuck
        if no_improve > 300:
            noise = 0.1 * rng.standard_normal(n)
            z = best_z + noise
            z = z - np.mean(z)
            z = np.clip(z, -50, 50)
            a = _softmax(z)
            c = _full_convolve(a)
            beta = 0.1 * float(c.max())
            m[:] = 0
            v[:] = 0
            t = 0
            lr = lr_init
            no_improve = 0
    
    return best_a, best_score, best_z

def _lp_refine(a, max_time, initial_t=0.3):
    """LP-based refinement."""
    t0 = time.monotonic()
    deadline = t0 + max_time
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = 2.0 * n * float(_full_convolve(a).max())
    t = initial_t
    m_shrink = 0.998
    
    for it in range(100):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a)
        M = float(c.max())
        
        # Solve LP: maximize sum(b) s.t. a*b <= M, b >= 0
        # (a*b)[k] = sum_i a[i]*b[k-i] <= M for all k
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                rows.extend([k] * len(idx))
                cols.extend(idx.tolist())
                vals.extend(a[k - idx].tolist())
        
        A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        res = linprog(-np.ones(n), A_ub=A, b_ub=M*np.ones(2*n-1),
                     bounds=[(0, None)]*n, method='highs',
                     options={'presolve': True, 'time_limit': 30})
        
        if not res.success:
            # Try with slightly relaxed M
            res = linprog(-np.ones(n), A_ub=A * 1.01, b_ub=M*1.01*np.ones(2*n-1),
                         bounds=[(0, None)]*n, method='highs',
                         options={'presolve': True, 'time_limit': 30})
            if not res.success:
                break
        
        b = np.maximum(res.x, 0)
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        
        b = b / s_b
        a_new = (1 - t) * a + t * b
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        sc = 2.0 * n * float(_full_convolve(a_new).max())
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
        t = max(0.1, t * 0.95)
        # Don't shrink M too aggressively
        # M will be recomputed from a_new next iteration
    
    return best_a, best_score

def _multi_scale_refine(a_best, max_time):
