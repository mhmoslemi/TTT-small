"""
Optimized step function search combining smooth L-BFGS continuation,
Nesterov-accelerated gradient descent, direct subgradient polishing,
and structured restarts to minimize 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous: cleaner time management, direct subgradient

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _gradient_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _objective_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_obj(z, beta, n):
    return _objective_smooth(z, beta)


def _lbfgs_grad(z, beta, n):

def _direct_subgradient_polish(a_init, deadline, lr_init=0.01, mom_init=0.85):
    """Subgradient method on the exact max function (no smooth approximation)."""
    n = len(a_init)
    a = a_init.copy()
    S = float(a.sum())
    a = a * (S / float(a.sum()))  # ensure sum is S
    S = float(a.sum())

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    momentum = np.zeros(n)
    mom = mom_init
    lr = lr_init
    t = 0

    while time.monotonic() < deadline:
        t += 1
        c = _full_convolve(a, a)
        cm = float(c.max())
        sc = 2.0 * n * cm

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Compute subgradient
        # Active set: lags within a small tolerance of max
        tol = 1e-8 * cm + 1e-12
        active = c >= cm - tol
        n_active = int(np.sum(active))
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active

        # Subgradient of max_c w.r.t. a[i]:
        # g[i] = sum_k w[k] * (a[i+k] + a[i-k]) / 2 (with boundary handling)
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        g = w_conv[n - 1:2 * n - 1]

        # Project gradient to maintain sum constraint
        g_proj = g - g.mean()

        # Momentum step
        momentum = mom * momentum - lr * g_proj
        a_new = np.maximum(0.0, a + momentum)
        s_new = float(a_new.sum())
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * (S / s_new)
        a = a_new

        # Adaptive parameters
        if t % 50 == 0:
            mom = min(0.99, mom + 0.0002)
        if t % 200 == 0:
            lr = max(0.001, lr * 0.9)

    return best_a, best_score


def _b_subgradient_polish(a_init, deadline, lr_init=0.005):

        a_rev = a[::-1]

        g = w_conv[n - 1:2 * n - 1]
        g_proj = g - g.mean()

        for trial in range(3):

            else:
                a_try = a_try * (1.0 / s_try)

            c_try = _full_convolve(a_try, a_try)

        # Accept the best trial
        a = best_a_try

            lr = best_lr
        else:
            lr = min(lr * 1.05, 0.05)

def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_symmetric_tri(n):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - np.abs(2.0 * t - 1.0)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_profile(n, freq=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.3 * np.cos(np.pi * freq * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _structured_perturb(a, rng, order=3):
    """Perturb using low-order polynomials to explore structured variations."""
    n = len(a)
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    perturb = np.zeros(n)
    for k in range(1, order + 1):
        coeff = rng.normal(0, 0.1 / k)
        if k % 2 == 0:
            perturb += coeff * u ** k
        else:
            perturb += coeff * u ** k
    z = np.log(np.maximum(a, 1e-15))
    z = z + perturb
    z = z - z.mean()
    a_new = _softmax_fn(z)
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS continuation -> Nesterov -> Adam ->
    Direct subgradient polishing -> structured restarts -> final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        print(f"Parent: n={n}, score={_score_fn(parent):.10f}")
    else:
        n = 4096
        parent = _make_linear_profile(n, 1.5)
        print(f"No parent, using linear profile n={n}")

    # --- Phase 0: Quick L-BFGS from multiple starts (60s) ---
    phase0_end = min(deadline, t_start + 60.0)
    candidates = []

    sc_parent = _score_norm_fn(parent)
    candidates.append((sc_parent, parent.copy(), "parent"))
    update_best(parent, sc_parent, "parent")

    # Try a few other profiles at same length
    other_profiles = [
        ("lin1.0", _make_linear_profile(n, 1.0)),
        ("lin2.0", _make_linear_profile(n, 2.0)),
        ("quad", _make_quadratic_profile(n, 1.0, 0.3)),
        ("cos", _make_cosine_profile(n, 1.0)),
        ("tri", _make_symmetric_tri(n)),
    ]
    for label, p in other_profiles:
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p, label))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 candidates: best={candidates[0][0]:.10f} [{candidates[0][2]}]")

    # Quick L-BFGS on top 2 candidates
    for sc, p, label in candidates[:2]:
        if time.monotonic() > phase0_end - 5:
            break
        per_time = min(25.0, phase0_end - time.monotonic() - 3)
        if per_time > 5:
            p_best, sc_best = _run_lbfgs(p, min(phase0_end - 2, time.monotonic() + per_time),
                                         max_wall_time=per_time)
            update_best(p_best, sc_best, f"lbfgs0_{label}")

    print(f"Phase 0 done: best={best_score:.10f}")

    # --- Phase 1: L-BFGS continuation (350s) ---
    phase1_end = min(deadline, t_start + 410.0)
    if best_seq is not None and time.monotonic() < phase1_end:
        per_time = min(340.0, phase1_end - time.monotonic() - 10)
        if per_time > 20:
            p_best, sc_best = _run_lbfgs(best_seq, min(phase1_end - 5, time.monotonic() + per_time),
                                         max_wall_time=per_time)
            update_best(p_best, sc_best, "lbfgs_phase1")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Nesterov (350s) ---
    phase2_end = min(deadline, t_start + 760.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        beta_fracs = [0.01, 0.005, 0.003, 0.0015, 0.0008, 0.0004, 0.0002,
                      0.0001, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6]
        per_time = min(340.0, phase2_end - time.monotonic() - 10)
        if per_time > 20:
            p_best, sc_best = _nesterov_optimize(best_seq,
                                                  min(phase2_end - 5, time.monotonic() + per_time),
                                                  beta_fracs, lr_init=0.015, mom_init=0.9)
            update_best(p_best, sc_best, "nesterov_phase2")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Adam (250s) ---
    phase3_end = min(deadline, t_start + 1010.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        per_time = min(240.0, phase3_end - time.monotonic() - 10)
        if per_time > 20:
            p_best, sc_best = _adam_optimize(best_seq,
                                             min(phase3_end - 5, time.monotonic() + per_time),
                                             lr=0.03, beta_frac=0.005)
            update_best(p_best, sc_best, "adam_phase3")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Direct subgradient polishing (200s) ---
    phase4_end = min(deadline, t_start + 1210.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        per_time = min(100.0, phase4_end - time.monotonic() - 50)
        if per_time > 10:
            p_best, sc_best = _direct_subgradient_polish(
                best_seq, min(phase4_end - 50, time.monotonic() + per_time),
                lr_init=0.008, mom_init=0.8)
            update_best(p_best, sc_best, "subgrad_phase4a")

        # Second round with smaller lr
        if time.monotonic() < phase4_end - 10:
            per_time2 = min(90.0, phase4_end - time.monotonic() - 5)
            if per_time2 > 10:
                p_best, sc_best = _b_subgradient_polish(
                    best_seq, min(phase4_end - 2, time.monotonic() + per_time2),
                    lr_init=0.003)
                update_best(p_best, sc_best, "subgrad_phase4b")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Structured restarts (150s) ---
    phase5_end = min(deadline, t_start + 1360.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        n_restarts = 3
        per_restart = min(45.0, (phase5_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            # Structured perturbation
            p_perturbed = _structured_perturb(best_seq, rng, order=4 + i)

            # Nesterov on perturbed
            per_dl = min(phase5_end - 5, time.monotonic() + per_restart)
            beta_fracs_short = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
            p_best, sc_best = _nesterov_optimize(p_perturbed, per_dl,
                                                  beta_fracs_short, lr_init=0.01, mom_init=0.85)
            update_best(p_best, sc_best, f"restart_{i}")

            # Quick subgradient polish on result
            if time.monotonic() < phase5_end - 5 and per_dl - time.monotonic() > 5:
                p_best2, sc_best2 = _direct_subgradient_polish(
                    p_best, min(phase5_end - 3, time.monotonic() + 15.0),
                    lr_init=0.004, mom_init=0.75)
                update_best(p_best2, sc_best2, f"restart_{i}_polish")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Try different length (100s) ---
    phase6_end = min(deadline, t_start + 1460.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        n_curr = len(best_seq)

        # Try a slightly different length
        for target_n in [n_curr * 2, n_curr // 2]:
            if time.monotonic() > phase6_end - 20:
                break
            if target_n < 128 or target_n > 32768:
                continue
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            print(f"  Interp n={target_n}: score={sc_interp:.10f}")
            if sc_interp < best_score:
                update_best(p_interp, sc_interp, f"interp_n{target_n}")

            if time.monotonic() < phase6_end - 15:
                per_time = min(40.0, phase6_end - time.monotonic() - 10)
                if per_time > 10:
                    p_best, sc_best = _run_lbfgs(p_interp,
                                                  min(phase6_end - 8, time.monotonic() + per_time),
                                                  max_wall_time=per_time)
                    update_best(p_best, sc_best, f"lbfgs_n{target_n}")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final fine polishing (remaining time) ---
    phase7_end = deadline
    if best_seq is not None and time.monotonic() < phase7_end:
        remaining = phase7_end - time.monotonic()

        # Final Nesterov with tiny beta
        if remaining > 50:
            per_time = min(80.0, remaining - 30)
            if per_time > 15:
                beta_fracs_final = [2e-6, 1e-6, 5e-7, 2e-7, 1e-7]
                p_best, sc_best = _nesterov_optimize(
                    best_seq, min(phase7_end - 30, time.monotonic() + per_time),
                    beta_fracs_final, lr_init=0.003, mom_init=0.95)
                update_best(p_best, sc_best, "final_nesterov")

        # Final direct subgradient
        if time.monotonic() < phase7_end - 15:
            per_time = min(40.0, phase7_end - time.monotonic() - 10)
            if per_time > 10:
                p_best, sc_best = _direct_subgradient_polish(
                    best_seq, min(phase7_end - 8, time.monotonic() + per_time),
                    lr_init=0.003, mom_init=0.7)
                update_best(p_best, sc_best, "final_subgrad")

        # Ultra-fine Adam
        if time.monotonic() < phase7_end - 10:
            per_time = min(30.0, phase7_end - time.monotonic() - 5)
            if per_time > 8:
                p_best, sc_best = _adam_optimize(
                    best_seq, min(phase7_end - 3, time.monotonic() + per_time),
                    lr=0.003, beta_frac=0.0005)
                update_best(p_best, sc_best, "final_adam")

    print(f"Phase 7 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_linear_profile(2048, 1.5)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        # Try evaluate_sequence for authoritative score
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned inf/NaN, using internal score: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_linear_profile(4096, 1.5)
    print(f"FALLBACK: n=4096, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
