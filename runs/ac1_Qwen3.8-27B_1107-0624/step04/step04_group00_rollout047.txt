4. Multi-restart strategy

import numpy as np
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    n = len(a)
    out_n = 2 * n - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    r = np.fft.irfft(fa * fa, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64) - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * (np.arange(1, n+1)) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _compute_gradient(a, beta):
    """Compute gradient of 2*n*LSE(c, beta) with respect to a."""
    n = len(a)
    c = _full_convolve(a)
    c_shifted = c / beta
    c_shifted = c_shifted - np.max(c_shifted)
    e = np.exp(c_shifted)
    w = e / e.sum()

    # (w * a_rev)[i + n - 1] for i = 0, ..., n-1

    n = len(a)
    c = _full_convolve(a)
