import numpy as np

from scipy.optimize import minimize as scipy_minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smooth_max(c, beta):
    """Compute log-sum-exp smooth max of c."""
    cm = float(c.max())
    if beta < 1e-12:
        return cm
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return cm
    return cm + beta * np.log(s)

        w[idx] = 1.0

    return e / s

    obj = 2.0 * n * smooth_val

    # Gradient: d/dz [2n * sum_k W_k * c_k]
    # dc_k/da_j = sum_l delta(j+l=k or l+j=k) * a[l]
    # = (cross correlation of a with "indicator of lag k")
    # More efficiently: gradient w.r.t. a of c_k is the "unshifted" a
    # grad_a = 2 * n * (W * a)^T convolved with a... 
    # Actually: d(c_k)/da_j = a[k-j] + a[k-j] (symmetric) = 2*a[k-j] for j != k-j, a[k-j] for j=k-j
    # So d(c_k)/da = 2*a reversed and shifted, with the diagonal halved.
    # More simply: d(c_k)/da_j = a[k-j] (if 0<=k-j<n) + a[k-j] (same term counted once)
    # Wait: c_k = sum_j a[j]*a[k-j], so d(c_k)/da_j = a[k-j] + a[k-j] = 2*a[k-j] if j != k-j
    # and = a[k-j] if j = k-j (i.e., k=2j).
    
    # Actually the cleaner way: the gradient of c_k w.r.t. a is the "cross-correlation" 
    # of a with the unit vector at position... 
    # Let me use the chain rule properly.
    # c_k = sum_j a[j]*a[k-j]
    # d c_k / d a[m] = sum_j (delta_{j,m} * a[k-j] + a[j] * delta_{k-j,m})
    # = a[k-m] (if 0 <= k-m < n) + a[k-m] (if 0 <= k-m < n)
    # = 2 * a[k-m] (if 0 <= k-m < n)
    # But wait, when j = m and k-j = m (i.e., k = 2m), we get a[m] + a[m] = 2*a[m], which is correct.
    # When j = m but k-j != m: contribution is a[k-m]
    # When k-j = m but j != m: contribution is a[j] = a[k-m]
    # So total: 2 * a[k-m] if 0 <= k-m < n, else 0.
    
    # So grad_a[m] = 2 * sum_k W_k * a[k-m] (where a[k-m] is 0 if k-m not in [0,n-1])
    # = 2 * (sum_k W_k * a_shifted)[m]
    # where a_shifted[k] = a[k-m]... this is just 2 * convolve(W, a)[m] flipped
    
    # More precisely: grad_a[m] = 2 * sum_k W[k] * a[k-m] for valid k-m
    # This is 2 * (W * a_cross)[m] where a_cross is the "reflected" a
    # = 2 * np.convolve(W, a[::-1])[m + n - 1] ... let me verify with a small example
    
    # Actually let's just compute it directly using the cross-correlation:
    # grad_a[m] = 2 * sum_{k: 0<=k<2n-1} W[k] * (a[k-m] if 0<=k-m<n else 0)
    # = 2 * (W cross-correlated with a)[m] where the cross-correlation is defined appropriately
    
    # The cleanest: create a kernel K where K[k] = W[k], and compute 
    # grad_a = 2 * ifft(fft(K) * conj(fft(a_padded)))
    # But let me just use the convolution approach:
    
    # grad_a[m] = 2 * sum_k W[k] * a[(k-m) mod n]... no, it's not circular.
    # grad_a[m] = 2 * sum_{k=m}^{m+n-1} W[k] * a[k-m] (where W[k]=0 for k>=2n-1)
    # This is 2 * (W * a)[m] where * denotes "cross-correlation" in the linear sense:
    # (W cross a)[m] = sum_k W[k] * a[k-m]
    # = sum_j W[m+j] * a[j]  (substituting k = m+j)
    # = (reverse_convolve(W, a))[2n-2-m] ... 
    
    # Let me just compute it as: grad_a = 2 * _full_convolve(W, a)[n-1::-1][:n] ... 
    # Hmm, let me think more carefully.
    
    # _full_convolve(W, a)[k] = sum_j W[j]*a[k-j] for valid j, k-j
    # We want: grad_a[m] = 2 * sum_k W[k]*a[k-m]
    # Let's substitute: this equals 2 * sum_{k>=m} W[k]*a[k-m] = 2 * sum_{j>=0} W[m+j]*a[j]
    # = 2 * (W_reversed_conv_with_a)[n-1-m] where W_reversed_conv... 
    
    # OK simplest approach: grad_a[m] = 2 * sum_j W_{m+j} * a_j for j=0..n-1 (with W=0 outside [0,2n-2])
    # This is the cross-correlation of W and a at lag m.
    # In numpy: this is np.correlate(W, a, mode='full')[2*(n-1)-m] ... let me not overthink this.
    
    # Let's use: grad_a = 2 * _full_convolve(W[::-1], a)[n-1:n-1+n][::-1]
    # Actually the simplest correct formula:
    # grad_a[m] = 2 * sum_{k=0}^{2n-2} W[k] * a[k-m] * I(0<=k-m<n)
    # = 2 * (W convolved with reversed a, then appropriate slice)
    
    # Let me just verify: _full_convolve(W, a)[k] = sum_j W[j]*a[k-j]
    # We want sum_k W[k]*a[k-m]. Let j = k-m, k = j+m:
    # = sum_j W[j+m]*a[j]
    # This is the cross-correlation at lag m.
    # = _full_convolve(W, a)[m + (n-1) - (n-1)] ... 
    
    # I'll use a different approach: 
    # sum_j W[j+m]*a[j] = dot(W[m:m+n], a) (with W assumed to be zero-padded to length >= 2n-1)
    # This is the m-th element of the "cross-correlation" which can be computed as:
    # corr[m] = dot(W[m:m+n], a) for m = 0, ..., 2n-2
    # In terms of convolution: corr = _full_convolve(W, a)[n-1::-1]... 
    
    # You know what, let me just use the FFT approach directly:
    # grad_a[m] = 2 * sum_k W[k] * a[k-m] (with a[x]=0 for x outside [0,n-1])
    # = 2 * (W ⋆ a)[m] where ⋆ denotes cross-correlation
    # In frequency domain: FFT(grad_a / 2) = FFT(W) * conj(FFT(a))
    
    N = _next_pow2(2 * n - 1)
    FW = np.fft.rfft(W, N)
    Fa = np.fft.rfft(a, N)
    grad_a_fft = FW * np.conj(Fa)
    grad_a = 2.0 * np.fft.irfft(grad_a_fft, N)[:n]
    
    # Now chain rule: dJ/dz[m] = sum_i dJ/da[i] * da[i]/dz[m]
    # da[i]/dz[m] = a[i]*(delta_{i,m} - a[m]) = a[i] if i==m, -a[i]*a[m] otherwise
    # = a[m] * (1/a[m] - a[m]) if i==m ... 
    # sum_i grad_a[i] * a[i]*(delta_{i,m} - a[m])
    # = grad_a[m]*a[m] - a[m]*sum_i grad_a[i]*a[i]
    # = a[m] * (grad_a[m] - dot(grad_a, a))
    
    W_dot_a = float(np.dot(W, c))  # This is sum_k W_k * c_k = the smooth max value (approx)
    # Actually W_dot_a = sum_k W[k] * c[k] which is the "weighted average" of c
    
    # Let me recompute: sum_i grad_a[i]*a[i]
    ga_dot_a = float(np.dot(grad_a, a))
    
    grad_z = a * (grad_a - ga_dot_a)
    
    return obj, grad_z


def _subgradient_step(a, n, step_size):
    """Apply a subgradient step in the original space (simplex projected)."""
    c = _full_convolve(a, a)
    max_val = float(c.max())
    max_indices = np.where(c >= max_val - 1e-12)[0]
    
    # Subgradient of max at the argmax positions
    # d/dc_k [(a*a)[k]] = 2*a shifted, so we average over all max positions
    grad_a = np.zeros(n)
    for k in max_indices:
        for m in range(n):
            km = k - m
            if 0 <= km < n:
                grad_a[m] += 2.0 * a[km]
    grad_a /= len(max_indices)
    
    # Projected subgradient step on simplex
    a_new = a - step_size * grad_a
    # Project onto simplex
    a_new = _project_simplex(a_new)
    return a_new


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    sv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu - (sv - 1) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_star = np.max(rho[cond])
    theta = (sv[rho_star - 1] - 1) / rho_star
    return np.maximum(v - theta, 0)


def _adam_optimize(z, beta_schedule, deadline, max_iter=200000, lr=0.01, n=None):
    """Adam optimizer in log-space with beta-continuation."""
    if n is None:
        n = len(z)
    
    best_a = _softmax(z)
    c = _full_convolve(best_a, best_a)
    best_score = float(2.0 * n * c.max())
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            
            a = _softmax(z)
            c = _full_convolve(a, a)
            W = _softmax_weights(c, beta)
            
            # Compute gradient
            N = _next_pow2(2 * n - 1)
            FW = np.fft.rfft(W, N)
            Fa = np.fft.rfft(a, N)
            grad_a = 2.0 * np.fft.irfft(FW * np.conj(Fa), N)[:n]
            ga_dot_a = float(np.dot(grad_a, a))
            grad_z = a * (grad_a - ga_dot_a)
            
            if not np.all(np.isfinite(grad_z)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            max_update = 0.3
            update_norm = np.linalg.norm(update)
            if update_norm > max_update:
                update = update * (max_update / update_norm)
            
            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()
            
            if t % 200 == 0:
                a = _softmax(z)
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()
            
            if it_in_phase % 3000 == 0:
                current_lr = max(0.001, current_lr * 0.97)
    
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _lbfgs_optimize(a_init, beta, deadline, max_iter=5000, maxfun=2000):
    """L-BFGS optimization in log-space for a fixed beta."""
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    # Time check helper
    state = {'done': False}
    
    def obj_and_grad(z):
        if time.monotonic() > deadline or state['done']:
            state['done'] = True
            a = _softmax(z)
            c = _full_convolve(a, a)
            W = _softmax_weights(c, beta)
            N = _next_pow2(2 * n - 1)
            FW = np.fft.rfft(W, N)
            Fa = np.fft.rfft(a, N)
            grad_a = 2.0 * np.fft.irfft(FW * np.conj(Fa), N)[:n]
            ga_dot_a = float(np.dot(grad_a, a))
            grad_z = a * (grad_a - ga_dot_a)
            if not np.all(np.isfinite(grad_z)):
                grad_z = np.zeros(n)
            return 0.0, grad_z
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        W = _softmax_weights(c, beta)
        smooth_val = _smooth_max(c, beta)
        obj = 2.0 * n * smooth_val
        
        N = _next_pow2(2 * n - 1)
        FW = np.fft.rfft(W, N)
        Fa = np.fft.rfft(a, N)
        grad_a = 2.0 * np.fft.irfft(FW * np.conj(Fa), N)[:n]
        ga_dot_a = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - ga_dot_a)
        
        if not np.all(np.isfinite(grad_z)):
            grad_z = np.zeros(n)
        
        return obj, grad_z
    
    try:
        result = scipy_minimize(
            obj_and_grad, z0,
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': max_iter, 'maxfun': maxfun, 'maxls': 20,
                     'gtol': 1e-6, 'ftol': 1e-12}
        )
        z_opt = result.x
        z_opt = z_opt - z_opt.mean()
        a_opt = _softmax(z_opt)
    except Exception as e:
        a_opt = a_init.copy()
    
    return a_opt


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "centered":
        t = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + 50.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_harmonics=5):
    """Apply random Fourier perturbation."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    for _ in range(n_harmonics):
        freq = rng.integers(1, n // 3)
        amp = 0.03 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        t = np.arange(n)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block(a, rng):
    """Randomly modify a random block of the sequence."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    block_size = rng.integers(n // 10, n // 3)
    start = rng.integers(0, n - block_size)
    z[start:start + block_size] += 0.1 * rng.standard_normal(block_size)
    z = z - z.mean()
    return _softmax(z)


def _palindromic_project(a):
    """Project a to its palindromic version (a[i] = a[n-1-i])."""
    n = len(a)
    half = n // 2
    result = np.zeros(n)
    if n % 2 == 0:
        result[:half] = (a[:half] + a[n-1:half-1:-1]) / 2
        result[half:] = result[half-1::-1]
    else:
        result[:half] = (a[:half] + a[n-1:half:-1]) / 2
        result[half] = a[half]
        result[half+1:] = result[half-1::-1]
    return np.maximum(result, 0) / np.maximum(result.sum(), 1e-15)


def _palindromic_optimize(a_init, beta, deadline, max_iter=3000):
    """Optimize with palindromic constraint using L-BFGS on half the variables."""
    n = len(a_init)
    a_pal = _palindromic_project(a_init)
    half = (n + 1) // 2
    
    def make_full(z_half):
        """Expand half vector to full palindromic sequence."""
        z_full = np.zeros(n)
        if n % 2 == 0:
            z_full[:half] = z_half
            z_full[(half-1):0:-1] = z_half[half-1::-1]
        else:
            z_full[:half] = z_half
            z_full[(half-1):0:-1] = z_half[half-1::-1]
        # Now softmax in full space
        a_full = _softmax(z_full)
        return a_full
    
    z_half = np.log(np.maximum(a_pal[:half], 1e-15))
    
    def obj_and_grad(z_h):
        a_full = make_full(z_h)
        c = _full_convolve(a_full, a_full)
        W = _softmax_weights(c, beta)
        smooth_val = _smooth_max(c, beta)
        obj = 2.0 * n * smooth_val
        
        N = _next_pow2(2 * n - 1)
        FW = np.fft.rfft(W, N)
        Fa = np.fft.rfft(a_full, N)
        grad_a = 2.0 * np.fft.irfft(FW * np.conj(Fa), N)[:n]
        
        # Project gradient to palindromic constraint
        grad_pal = _palindromic_project(grad_a)
        
        # Chain rule: d(a_full[i])/d(z_h[j])
        # a_full[i] = exp(z_full[i]) / sum(exp(z_full))
        # This is complex; let's just use the direct gradient on the first half
        grad_z_h = grad_pal[:half] * a_pal[:half]
        
        if not np.all(np.isfinite(grad_z_h)):
            grad_z_h = np.zeros(half)
        
        return obj, grad_z_h
    
    try:
        result = scipy_minimize(
            obj_and_grad, z_half,
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': max_iter, 'maxfun': max_iter * 2, 'maxls': 15,
                     'gtol': 1e-5}
        )
        z_half_opt = result.x
        a_opt = make_full(z_half_opt)
    except Exception:
        a_opt = a_pal
    
    return a_opt


def _subgradient_polish(a, deadline, n_iter=5000):
    """Polish using projected subgradient steps with decreasing step size."""
    n = len(a)
    c = _full_convolve(a, a)
    max_c = float(c.max())
    best_a = a.copy()
    best_score = 2.0 * n * max_c
    
    step_base = 0.01 / max(1.0, max_c)
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        step = step_base / (1.0 + 0.01 * it)
        a_new = _subgradient_step(a, n, step)
        
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

    # --- Phase 3: Multi-restart with L-BFGS (350s - 900s) ---
    phase3_end = min(deadline, t_start + 900.0)
    if time.monotonic() < phase3_end - 30:
        n_restarts = 12
        per_restart = (phase3_end - time.monotonic() - 20) / n_restarts
        
        print(f"Phase 3: {n_restarts} restarts, {per_restart:.1f}s each")
        
        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 15:
                break
            
            # Different perturbation strategies
            if i % 4 == 0:
                p_start = _perturb(current, 0.03 + 0.015 * (i % 4), rng)
            elif i % 4 == 1:
                p_start = _perturb_fourier(current, rng, n_harmonics=3 + (i % 3))
            elif i % 4 == 2:
                p_start = _perturb_block(current, rng)
            else:
                # Mix of current best and a different profile
                profile = _make_profile(n_main, ["tapered", "bimodal", "centered"][i % 3])
                p_start = 0.7 * current + 0.3 * profile
                p_start = p_start / p_start.sum()
            
            c_s = _full_convolve(p_start, p_start)
            beta_s = 0.01 * float(c_s.max())
            
            # L-BFGS from this start
            remaining = min(per_restart, phase3_end - time.monotonic() - 5)
            if remaining > 5:
                a_opt = _lbfgs_optimize(p_start, beta_s, t_start + remaining, max_iter=1500)
                sc = _score(a_opt)
                if np.isfinite(sc) and sc < _score(current):
                    current = a_opt
                    update_best(current, f"restart_lbfgs_{i}")
            
            # Quick Adam polish if time allows
            if time.monotonic() < phase3_end - 10 and remaining > 15:
                beta_a = [0.005 * float(c_s.max()), 0.002 * float(c_s.max()), 0.0005 * float(c_s.max())]
                z_s = np.log(np.maximum(p_start, 1e-15))
                z_s = z_s - z_s.mean()
                t_sub = min(10.0, phase3_end - time.monotonic() - 5)
                a_opt2, _ = _adam_optimize(z_s, beta_a, t_start + t_sub, max_iter=50000, lr=0.008, n=n_main)
                sc2 = _score(a_opt2)
                if np.isfinite(sc2) and sc2 < _score(current):
                    current = a_opt2
                    update_best(current, f"restart_adam_{i}")
    
    print(f"Phase 3 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # --- Phase 4: Palindromic optimization (900s - 1100s) ---
    phase4_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase4_end - 20:
        print("Phase 4: Palindromic optimization")
        a_pal = _palindromic_project(current)
        sc_pal = _score(a_pal)
        if sc_pal < best_score:
            update_best(a_pal, "palindromic_proj")
        
        c_pal = _full_convolve(a_pal, a_pal)
        beta_pal = 0.005 * float(c_pal.max())
        remaining = phase4_end - time.monotonic() - 10
        if remaining > 10:
            a_opt_pal = _palindromic_optimize(a_pal, beta_pal, t_start + remaining, max_iter=2000)
            sc_opt = _score(a_opt_pal)
            if np.isfinite(sc_opt) and sc_opt < best_score:
                current = a_opt_pal
                update_best(current, "palindromic_opt")
    
    print(f"Phase 4 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # --- Phase 5: Subgradient polish (1100s - 1350s) ---
    phase5_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < phase5_end - 10:
        print("Phase 5: Subgradient polishing")
        a_sg, _ = _subgradient_polish(current, t_start + (phase5_end - time.monotonic() - 5), n_iter=10000)
        sc_sg = _score(a_sg)
        if np.isfinite(sc_sg) and sc_sg < best_score:
            current = a_sg
            update_best(current, "subgradient")
    
    print(f"Phase 5 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # --- Phase 6: Fine L-BFGS with very small beta (1350s - 1550s) ---
    phase6_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase6_end - 15:
        print("Phase 6: Fine L-BFGS")
        c_fine = _full_convolve(current, current)
        max_fine = float(c_fine.max())
        
        for beta_f in [0.001 * max_fine, 0.0005 * max_fine]:
            if time.monotonic() > phase6_end - 10:
                break
            remaining = phase6_end - time.monotonic() - 5
            if remaining < 10:
                break
            a_opt = _lbfgs_optimize(current, beta_f, t_start + remaining, max_iter=3000)
            sc = _score(a_opt)
            if np.isfinite(sc) and sc < _score(current):
                current = a_opt
                update_best(current, f"fine_lbfgs_{beta_f:.6f}")
    
    print(f"Phase 6 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # --- Phase 7: Try different n (1550s - 1700s) ---
    phase7_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase7_end - 20:
        # Try interpolating to a different n
        for target_n in [n_main * 2 if n_main <= 2048 else n_main // 2]:
            if time.monotonic() > phase7_end - 15:
                break
            if target_n < 64 or target_n > 8192:
                continue
            p_interp = _interp_profile(current, target_n)
            sc_int = _score(p_interp)
            update_best(p_interp, f"interp_n{target_n}")
            
            if sc_int < best_score + 0.01:
                c_int = _full_convolve(p_interp, p_interp)
                beta_int = 0.01 * float(c_int.max())
                remaining = phase7_end - time.monotonic() - 10
                if remaining > 10:
                    a_opt = _lbfgs_optimize(p_interp, beta_int, t_start + remaining, max_iter=1000)
                    sc_opt = _score(a_opt)
                    if np.isfinite(sc_opt) and sc_opt < best_score:
                        current = a_opt
                        update_best(current, f"opt_n{target_n}")
    
    print(f"Phase 7 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # --- Phase 8: Final subgradient + L-BFGS polish (1700s - deadline) ---
    if time.monotonic() < deadline - 10:
        # Subgradient
        remaining_sg = min(30.0, deadline - time.monotonic() - 10)
        if remaining_sg > 5:
            a_sg, _ = _subgradient_polish(current, t_start + remaining_sg, n_iter=5000)
            sc_sg = _score(a_sg)
            if np.isfinite(sc_sg) and sc_sg < _score(current):
                current = a_sg
                update_best(current, "final_sg")
        
        # L-BFGS
        remaining_lb = deadline - time.monotonic() - 5
        if remaining_lb > 10:
            c_fin = _full_convolve(current, current)
            beta_fin = 0.0005 * float(c_fin.max())
            a_opt = _lbfgs_optimize(current, beta_fin, t_start + remaining_lb, max_iter=2000)
            sc_opt = _score(a_opt)
            if np.isfinite(sc_opt) and sc_opt < best_score:
                current = a_opt
                update_best(current, "final_lbfgs")
    
    print(f"Phase 8 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        final = final / final.sum()  # normalize
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        
        return [float(x) for x in final]
    
    # Fallback
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
