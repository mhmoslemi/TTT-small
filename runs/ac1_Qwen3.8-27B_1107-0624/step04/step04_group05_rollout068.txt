"""
Step-function search minimizing  2*n*max(conv(a,a)) / (sum a)^2  (scale invariant).
With sum(a)=1 this is 2*n*max(c) where sum(c)=1, so we want the self-convolution
as flat as possible (theoretical floor 2n/(2n-1) ~ 1). Strategy: build a diverse
set of starts over several lengths, then polish each with (a) spectral-energy
concentration toward DC (flattens c), (b) smooth-max Adam in logit space, and
(c) LP best-response refinement for small n. Best sequence across all trials is
returned, normalized to unit sum (>= 0.01).
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _smooth_max_and_grad(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    mval = float(c.max())
    x = np.minimum((c - mval) / beta, 0.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _score(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-30:
        return None
    return a / s


def _make_uniform(n):
    return np.ones(n) / n


def _make_cosine(n):
    t = np.arange(n)
    a = 0.5 * (1.0 + np.cos(np.pi * t / max(1, n - 1)))
    return a / a.sum()


def _make_gaussian(n, s):
    t = np.arange(n)
    c = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - c) / (s * n)) ** 2)
    return a / a.sum()


def _make_two_gaussian(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    return a / a.sum()


def _make_windowed_chirp(n):
    t = np.arange(n)
    phase = np.pi * t * t / n
    a = np.abs(np.exp(1j * phase)) * (0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1))))
    return a / a.sum()


def _make_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = np.maximum(1.0 + eta * (t - 0.5), 1e-3)
    return a / a.sum()


def _spectral_shape(a, power, blend):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    P = np.abs(A) ** 2
    P_new = P ** power
    phase = np.angle(A)
    P_new = np.maximum(P_new, 0.0)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = float(a_new.sum())
    if s < 1e-30:
        return a.copy()
    a_new = a_new / s
    out = (1.0 - blend) * a + blend * a_new
    out = _normalize(out)
    return out if out is not None else a.copy()


def _spectral_iter(a0, deadline, power, blend, every=1):
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    it = 0
    while time.monotonic() < deadline:
        a = _spectral_shape(a, power, blend)
        sc = float(_score(a))
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
        it += 1
    return best_a, best_sc


def _lp_best_response(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j0 = max(0, k - n + 1)
        j1 = min(n - 1, k)
        if j0 <= j1:
            idx = np.arange(j0, j1 + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a0, deadline, max_iters=25):
    n = len(a0)
    if n > 1024:
        return a0.copy(), float(_score(a0))
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = float(_score(a))
    stall = 0
    it = 0
    while it < max_iters and time.monotonic() < deadline:
        c = _full_convolve(a)
        M = float(c.max())
        b = _lp_best_response(a, M)
        if b is None:
            break
        sb = float(b.sum())
        if sb < 1e-10:
            break
        b = b / sb
        best_t = 0.2
        best_try = best_sc
        for tc in (0.05, 0.1, 0.2, 0.3, 0.5, 0.7):
            a_try = _normalize((1.0 - tc) * a + tc * b)
            if a_try is None:
                continue
            sc_try = float(_score(a_try))
            if sc_try < best_try:
                best_try = sc_try
                best_t = tc
        a_new = _normalize((1.0 - best_t) * a + best_t * b)
        if a_new is None:
            break
        sc = float(_score(a_new))
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 6:
                break
        a = a_new
        it += 1
    return best_a, best_sc

    b1, b2, eps = 0.9, 0.999, 1e-8

    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_sc = float(_score(a))

        _, w = _smooth_max_and_grad(c, beta)
        g_c = 2.0 * _conv_cross_corr(w, a)
        dot = float(np.dot(a, g_c))
        gz = a * (g_c - dot)
        gz = gz - gz.mean()
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            it += 1
            continue
        t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        es = np.exp(z - z.max())
        a = es / es.sum()
        if use_spectral and (it % sev == 0):
            a = _spectral_shape(a, spow, sblend)
        sc = float(_score(a))
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
        if it % 25 == 0:
            beta = max(1e-6, beta * beta_decay)
        if no_imp > patience:
            z = best_z + 0.03 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            es = np.exp(z - z.max())
            a = es / es.sum()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_imp = 0
            beta = max(1e-6, 0.05 * float(_full_convolve(a).max()))
            lr = max(0.002, lr * 0.85)
        it += 1
    return best_a, best_sc
