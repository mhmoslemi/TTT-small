Optimization using aggressive LP-based refinement with adaptive M,
line search for step size, and spectral gradient polishing.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _cross_convolve(a, b):
    """Compute a * b (cross-correlation/convolution)."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())

def _make_conv_matrix(a):
    """Build the convolution constraint matrix A where (A*b)[k] = (a*b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def _lp_solve(a, M, obj=None):
    """Solve: maximize obj @ b subject to (a*b)_k <= M, b >= 0.

    n = len(a)
    A_ub = _make_conv_matrix(a)

    return None

"""
Aggressive LP-based refinement with adaptive M, line search for step size,
dual-guided gradient polishing, and multi-scale optimization (n=1024, n=2048).

(2) adaptive M schedule that tightens based on progress, (3) gradient steps

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _make_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_solve_max_sum(a, M):
    n = len(a)
    A_ub = _make_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _grad_from_weights(w, p, n):
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                partner = k - np.arange(i_start, i_end + 1)
                heat[i_start:i_end + 1] += a[partner]

        heat_max = float(heat.max())
        if heat_max < 1e-15:
            break
        heat = heat / heat_max

        factor = 1.0 - eta * heat
        factor = np.maximum(factor, 0.01)
        a = a * factor
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        c = _full_convolve(a)

        if it % 200 == 0 and it > 0:
            if score >= best_score:
                eta = max(0.005, eta * 0.9)
            else:
                eta = min(0.12, eta * 1.05)

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 1e-15)
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p0)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_bell(n, width):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, label):
        nonlocal best_seq, best_score
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            p = np.clip(p, 0.0, 1000.0)
            s = float(p.sum())
            if s > 0.01:
                p = p / s
            sc = _score_norm(p)
            if sc < best_score:
                best_seq = p.copy()
                best_score = float(sc)
                print(f"  [{label}] score={best_score:.10f} n={p.size} "
                      f"time={time.monotonic()-t_start:.0f}s")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    if parent is not None:
        sc_parent = _score_norm(parent)
        print(f"Parent score: {sc_parent:.10f}")
        _update(parent, 'parent')
    else:
        parent = _make_profile_cosine(1024, 1, 0.5)
        _update(parent, 'init_cos')

    # === Phase 1: Adam polish (120s) ===
    phase1_end = t_start + 120.0
    print(f"Phase 1: Adam polish...")
    p1, sc1 = _adam_logit(parent, phase1_end, rng, lr=0.04, beta_frac=0.06,
                          beta_decay=0.94, no_imp_limit=150)
    _update(p1, 'adam1')

    # === Phase 2: Aggressive LP refinement (500s) ===
    phase2_end = t_start + 620.0
    print(f"Phase 2: LP aggressive, score={best_score:.10f}...")
    if best_seq is not None and best_seq.size <= 2048:
        p2, sc2 = _lp_refine_aggressive(best_seq, phase2_end,
                                         max_solves=300,
                                         m_shrink_init=0.998,
                                         m_shrink_min=0.97)
        _update(p2, 'lp_aggr')

    # === Phase 3: Adam polish after LP (100s) ===
    phase3_end = t_start + 720.0
    print(f"Phase 3: Post-LP Adam, score={best_score:.10f}...")
    p3, sc3 = _adam_logit(best_seq, phase3_end, rng, lr=0.02, beta_frac=0.04,
                          beta_decay=0.96, no_imp_limit=150)
    _update(p3, 'adam_post_lp')

    # === Phase 4: More LP (300s) ===
    phase4_end = t_start + 1020.0
    print(f"Phase 4: LP round 2, score={best_score:.10f}...")
    if best_seq is not None and best_seq.size <= 2048:
        p4, sc4 = _lp_refine_aggressive(best_seq, phase4_end,
                                         max_solves=200,
                                         m_shrink_init=0.997,
                                         m_shrink_min=0.96)
        _update(p4, 'lp2')

    # === Phase 5: Peak flood (100s) ===
    phase5_end = t_start + 1120.0
    print(f"Phase 5: Peak flood, score={best_score:.10f}...")
    if best_seq is not None and best_seq.size <= 2048:
        p5, sc5 = _peak_flood(best_seq, phase5_end, eta=0.02)
        _update(p5, 'flood')

    # === Phase 6: Try n=2048 (400s) ===
    phase6_end = t_start + 1520.0
    print(f"Phase 6: Multi-scale n=2048, score={best_score:.10f}...")
    if time.monotonic() < phase6_end and best_seq is not None and best_seq.size <= 2048:
        n_new = 2048
        p_int = _interp_profile(best_seq, n_new)
        sc_int = _score_norm(p_int)
        print(f"  Interpolated n={n_new}: {sc_int:.10f}")
        _update(p_int, f'interp_{n_new}')

        # Adam at new scale
        per_adam = min(120.0, phase6_end - time.monotonic() - 100)
        if per_adam > 30:
            p6a, sc6a = _adam_logit(p_int, time.monotonic() + per_adam, rng,
                                     lr=0.03, beta_frac=0.05,
                                     beta_decay=0.95, no_imp_limit=200)
            _update(p6a, f'adam_{n_new}')

        # LP at new scale
        if time.monotonic() < phase6_end - 30:
            per_lp = phase6_end - time.monotonic() - 30
            p6l, sc6l = _lp_refine_aggressive(best_seq, time.monotonic() + per_lp,
                                               max_solves=100,
                                               m_shrink_init=0.998,
                                               m_shrink_min=0.97)
            _update(p6l, f'lp_{n_new}')

    # === Phase 7: Final intensive polish ===
    phase7_end = deadline
    print(f"Phase 7: Final polish, score={best_score:.10f}...")
    if time.monotonic() < phase7_end and best_seq is not None:
        # Determine which n to polish
        work_seq = best_seq
        n_work = work_seq.size

        # LP round 3
        if n_work <= 2048:
            per = min(150.0, phase7_end - time.monotonic() - 60)
            if per > 30:
                p7l, sc7l = _lp_refine_aggressive(work_seq, time.monotonic() + per,
                                                   max_solves=80,
                                                   m_shrink_init=0.996,
                                                   m_shrink_min=0.95)
                _update(p7l, 'lp_final')

        # Final Adam
        per = phase7_end - time.monotonic() - 10
        if per > 20:
            p7a, sc7a = _adam_logit(best_seq, time.monotonic() + per, rng,
                                     lr=0.01, beta_frac=0.03,
                                     beta_decay=0.97, no_imp_limit=100)
            _update(p7a, 'adam_final')

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_cosine(1024, 1, 0.5)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile_cosine(1024, 1, 0.5)
    print(f"FALLBACK n=1024")
    return [float(x) for x in fb]
