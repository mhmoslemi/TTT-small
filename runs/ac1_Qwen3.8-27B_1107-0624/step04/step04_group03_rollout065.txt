def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):

    return e / s


def _score_fn(a):

def _make_profile(n, profile_type='exp_decay', param=1.0):
    t = np.linspace(0.0, 1.0, n)
    if profile_type == 'exp_decay':
        a = np.exp(-param * t)
    elif profile_type == 'exp_growth':
        a = np.exp(param * t)
    elif profile_type == 'linear':
        a = 1.0 + param * t
    elif profile_type == 'tilted_hann':
        hann = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = hann * np.exp(param * (t - 0.5))
    elif profile_type == 'tilted_blackman':
        x = 2.0 * t - 1.0
        blackman = 0.42 + 0.5 * np.cos(np.pi * x) + 0.08 * np.cos(2 * np.pi * x)
        a = np.maximum(blackman, 0.01) * np.exp(param * (t - 0.5))
    elif profile_type == 'power':
        a = (t + 0.05) ** param
    elif profile_type == 'inv_power':
        a = (1.05 - t) ** param
    elif profile_type == 'gaussian_tilt':
        a = np.exp(-4.0 * (t - 0.5) ** 2) * np.exp(param * (t - 0.5))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

        fallback = _make_profile(4096, 'exp_decay', 1.0)
        result = [float(x) for x in fallback]
        print(f"FALLBACK: n=4096, score={_score_fn(fallback):.10f}")

    return result
