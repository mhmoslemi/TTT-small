from scipy.optimize import linprog
import time

def _lp_step(f, M, n):
    """
    Solve: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns optimal g.
    """
    # c: objective (we want to maximize, so minimize -sum)
    c = -np.ones(n)
    
    # A_ub: each row is a constraint (f*g)[k] <= M
    # (f*g)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} f[k-i] * g[i]

def _compute_score(a):

    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    cols = []
    vals = []

def _score_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = np.convolve(a, a)
    return float(2.0 * n * c.max())

def _score(a):
    """Compute full score."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _build_A(f, n):
    """Build sparse convolution matrix for f."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        cnt = i_hi - i_lo + 1
        rows.extend([k] * cnt)
        cols.extend(range(i_lo, i_hi + 1))
        # f[k-i] for i in [i_lo, i_hi] = f[k-i_hi : k-i_lo+1] reversed
        vals.extend(f[k-i_hi:k-i_lo+1][::-1])
    return coo_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def _lp_solve(f, M, n, time_limit=30):
    """Solve: max sum(g) s.t. g>=0, A@b_ub <= M."""
    A = _build_A(f, n)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return res.x
    return None

def _fft_convolve(a, b):
