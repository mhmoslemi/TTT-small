```python
def solve_lp_direction(a, M, n, max_cuts=50):
    """Solve LP using cutting plane method."""
    # Active constraints: lag indices
    active_lags = [n-1]  # Start with the center lag (largest for uniform-ish)
    
    for _ in range(max_cuts):
        # Build constraint matrix A_eq for active lags
        # convolve(a, b)[k] = sum_i a[i] * b[i+k-n+1] (need to be careful with indexing)
        # Actually, convolve(a, b)[k] = sum_{i} a[i] * b[k-i] where both go from 0 to n-1
        # = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * b[k-i]
        
        # For constraint at lag k: sum_i a[i] * b[k-i] <= M
        # This is a linear form in b: sum_j b[j] * (coefficient depending on k and j)
        # coefficient of b[j] in constraint k is a[k-j] if 0<=k-j<=n-1, else 0
        
        # Build A matrix (sparse would be better but let's use dense for now)
        # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
        
        if len(active_lags) > 50:
            # Use only top constraints
            active_lags = active_lags[:50]
        
        A = np.zeros((len(active_lags), n))
        for idx, k in enumerate(active_lags):
            for j in range(max(0, k-n+1), min(n, k+1)):
                A[idx, j] = a[k-j]
        
        # Solve: max sum(b) s.t. A @ b <= M, b >= 0
        c_obj = -np.ones(n)  # Minimize -sum(b)
        res = linprog(c_obj, A_ub=A, b_ub=np.full(len(active_lags), M), 
                      bounds=[(0, None)]*n, method='highs')
        
        if res.status != 0:
            return None
        
        b = res.x
        
        # Check for violated constraints
        c_full = np.convolve(a, b)
        violations = c_full > M * (1 + 1e-8)
        
        if not violations.any():
            return b
        
        # Add most violated constraint
        viol_idx = np.where(violations)[0]
        new_lag = viol_idx[np.argmax(c_full[viol_idx])]
        active_lags.append(new_lag)
        
        # Sort by violation amount, keep top
        c_at_lags = c_full[active_lags]
        active_lags = [active_lags[i] for i in np.argsort(c_at_lags)[::-1]]
    
    return b
```

```python
    top_lags = np.argsort(_full_convolve(a, a))[::-1][:num_active]
    active_lags = list(top_lags[:num_active])

    ub_val = 2.0 * float(np.max(a)) + 1e-6

    for cut_iter in range(max_cuts):
        lags = active_lags[:]
        if len(lags) == 0:
            return None

        # Build constraint matrix A where A[idx, j] = a[k-j] if valid
        A = np.zeros((len(lags), n))
        for idx, k in enumerate(lags):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_min <= j_max:
                for j in range(j_min, j_max + 1):
                    A[idx, j] = a[k - j]

        # Solve LP: maximize sum(b) -> minimize -sum(b)
        c_obj = -np.ones(n)
        b_ub_vec = np.full(len(lags), M)

        try:
            bounds_list = [(0.0, ub_val)] * n
            res = linprog(c_obj, A_ub=A, b_ub=b_ub_vec,
                          bounds=bounds_list, method='highs',
                          options={'time_limit': 30})
        except Exception:
            return None

        if res.status != 0 or res.x is None:
            return None

        b = res.x.copy()
        b = np.maximum(b, 0)

        # Check all constraints for violations
        c_ab = _full_convolve(a, b)
        violations = c_ab > M * (1 + 1e-6)

        if not violations.any():
            return b

        # Add most violated lags
        viol_idx = np.where(violations)[0]
        worst = viol_idx[np.argsort(c_ab[viol_idx])[::-1]][:5]
        for w_idx in worst:
            wi = int(w_idx)
            if wi not in active_lags:
                active_lags.append(wi)

        # Keep only top num_active by violation severity
        if len(active_lags) > num_active:
            lags_arr = np.array(active_lags)
            vals = c_ab[lags_arr]
            sorted_idx = np.argsort(vals)[::-1][:num_active]
            active_lags = lags_arr[sorted_idx].tolist()

    # Return last solution
    return b


def _line_search(a, b, n):
    """Find best mixing parameter t in (0,1]."""
    base_score = _score(a)
    best_t = 0.0
    best_sc = base_score
    trial_ks = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
    for t in trial_ks:
        a_trial = (1.0 - t) * a + t * b
        sc = _score(a_trial)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t == 0.0:
        for t in [0.005, 0.01]:
            a_trial = (1.0 - t) * a + t * b
            sc = _score(a_trial)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_t = t
    return best_t, best_sc


def _smooth_polish(a_init, deadline, beta_init=0.005):
    """Adam-based smooth polishing in log-softmax space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_sc = _score(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    beta = beta_init * max_c0
    lr = 0.02

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0

    while time.monotonic() < deadline:
        t += 1
        grad = _smooth_gradient(z, beta)
        if not np.all(np.isfinite(grad)):
            continue

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50, 50)
        z_new = z_new - z_new.mean()

        a_new = _softmax_fn(z_new)
        sc = _score(a_new)

        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            z = z_new
        else:
            lr = max(0.002, lr * 0.95)
            beta = max(1e-10, beta * 0.9)

        if t % 200 == 0:
            beta = max(1e-10, beta * 0.85)
            c_curr = _full_convolve(_softmax_fn(z), _softmax_fn(z))
            max_c_curr = float(c_curr.max())

    return best_a, best_sc


def _active_set_polish(a_init, deadline):
    """Subgradient method on active (near-max) lags."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.003
    mom = np.zeros(n)
    mom_coeff = 0.9
    stuck = 0

    for it in range(50000):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        active = c >= cm * 0.999
        if not active.any():
            active[int(np.argmax(c))] = True

        w = np.zeros_like(c)
        w[active] = 1.0 / float(active.sum())
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 2.0 * n * (h - W)

        mom = mom_coeff * mom - lr * grad
        a_new = np.maximum(a + mom, 0.0)
        sc_new = _score(a_new)

        if np.isfinite(sc_new) and sc_new < _score(a):
            a = a_new
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a.copy()
                stuck = 0
                if it % 500 == 0:
                    print(f"    Active iter {it}: {best_sc:.10f}")
            else:
                stuck += 1
        else:
            lr *= 0.95
            stuck += 1

        if stuck > 200:
            break

    return best_a, best_sc


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: LP cutting-plane direction finding ->
    smooth Adam polishing -> active-set subgradient -> length adaptation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    def _update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:7.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size == 0 or parent.sum() < 0.01:
            raise ValueError("empty")
    except Exception:
        parent = np.ones(2048)

    n = len(parent)
    sc_parent = _score(parent)
    print(f"Parent: n={n}, score={sc_parent:.10f}")
    best_seq = parent.copy()
    best_score = sc_parent

    # --- Phase 0: Try a few different starting points ---
    phase0_end = min(deadline, t_start + 30.0)
    if time.monotonic() < phase0_end:
        for trial_n in [n, 2048, 8192]:
            t_arr = np.linspace(0, 1, trial_n)
            for slope in [1.0, 2.0, 0.5]:
                prof = 1.0 + slope * (t_arr - 0.5)
                prof = np.maximum(prof, 0.01)
                prof = prof / prof.sum()
                sc = _score(prof)
                if sc < best_score:
                    _update_best(prof, sc, f"init_n{trial_n}_s{slope}")

    # --- Phase 1: LP-based optimization (main phase) ---
    print("Phase 1: LP cutting-plane optimization")
    phase1_end = min(deadline, t_start + 1200.0)
    a = best_seq.copy()
    current_score = _score(a)

    lp_no_improve = 0
    for it in range(2000):
        if time.monotonic() > phase1_end:
            break

        c = _full_convolve(a, a)
        M = float(c.max())

        t0 = time.monotonic()
        b = _lp_direction(a, M, n, max_cuts=20, num_active=10)
        t_lp = time.monotonic() - t0

        if b is None or b.sum() < 1e-10:
            lp_no_improve += 1
            if lp_no_improve > 3:
                print(f"  Phase 1: LP failing after iter {it}, stopping")
                break
            # Try small perturbation
            pert = a + 0.005 * rng.normal(size=n)
            pert = np.maximum(pert, 0)
            sc_pert = _score(pert)
            if sc_pert < current_score:
                a = pert
                current_score = sc_pert
                if current_score < best_score:
                    _update_best(a, current_score, f"perturb_{it}")
            continue

        b = b * (a.sum() / b.sum())

        best_t, best_sc_trial = _line_search(a, b, n)

        if best_t == 0.0:
            lp_no_improve += 1
            if lp_no_improve > 5:
                print(f"  Phase 1: No improvement after iter {it}, stopping")
                break
            # Try smaller steps
            for t_small in [0.001, 0.003, 0.005]:
                a_trial = (1 - t_small) * a + t_small * b
                sc_trial = _score(a_trial)
                if sc_trial < current_score:
                    a = a_trial
                    current_score = sc_trial
                    if current_score < best_score:
                        _update_best(a, current_score, f"small_t_{it}")
                    lp_no_improve = 0
                    break
        else:
            a = (1.0 - best_t) * a + best_t * b
            current_score = _score(a)
            lp_no_improve = 0

            if it % 3 == 0 or current_score < best_score:
                if current_score < best_score:
                    _update_best(a, current_score, f"lp_{it}")
                else:
                    print(f"  Iter {it}: sc={current_score:.10f}, t={best_t:.4f}, lp={t_lp:.1f}s")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Smooth Adam polishing ---
    print("Phase 2: Smooth Adam polishing")
    phase2_end = min(deadline, t_start + 1500.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        polish_time = min(phase2_end - time.monotonic(), 300.0)
        if polish_time > 20:
            a_pol, sc_pol = _smooth_polish(best_seq, time.monotonic() + polish_time, beta_init=0.003)
            if sc_pol < best_score:
                _update_best(a_pol, sc_pol, "adam_polish")
            # Continue with even smaller beta
            if time.monotonic() < phase2_end - 10:
                a_pol2, sc_pol2 = _smooth_polish(best_seq, phase2_end - 10, beta_init=0.001)
                if sc_pol2 < best_score:
                    _update_best(a_pol2, sc_pol2, "adam_fine")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Active-set subgradient ---
    print("Phase 3: Active-set subgradient")
    phase3_end = min(deadline, t_start + 1650.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        a_act, sc_act = _active_set_polish(best_seq, phase3_end)
        if sc_act < best_score:
            _update_best(a_act, sc_act, "active_set")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Try different lengths ---
    print("Phase 4: Length adaptation")
    phase4_end = min(deadline - 20, t_start + 1770.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        n_curr = len(best_seq)
        for target_n in [n_curr * 2, n_curr // 2]:
            if target_n < 256 or target_n > 16384:
                continue
            if time.monotonic() > phase4_end - 30:
                break
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score(p_interp)
            if sc_interp < best_score:
                _update_best(p_interp, sc_interp, f"interp_n{target_n}")
                # Quick polish on new length
                if time.monotonic() < phase4_end - 30:
                    a_i, sc_i = _smooth_polish(p_interp, min(phase4_end - 10, time.monotonic() + 60), beta_init=0.01)
                    if sc_i < best_score:
                        _update_best(a_i, sc_i, f"polish_n{target_n}")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Final micro-polish ---
    phase5_end = deadline - 5
    if best_seq is not None and time.monotonic() < phase5_end:
        a_final_pol, sc_final_pol = _active_set_polish(best_seq, phase5_end)
        if sc_final_pol < best_score:
            _update_best(a_final_pol, sc_final_pol, "final_active")

    # --- Final validation ---
    if best_seq is None:
        best_seq = parent

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s_final = float(final_seq.sum())
    if s_final > 0.01:
        final_seq = final_seq / s_final
    else:
        final_seq = np.ones(2048)

    sc_final = _score(final_seq)
    print(f"FINAL: n={len(final_seq)}, score={sc_final:.10f}")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
        print(f"Evaluator score: {sc_eval:.10f}")
        if not np.isfinite(sc_eval):
            print("WARNING: Evaluator returned non-finite!")
    except Exception as ex:
        print(f"Evaluator error: {ex}")

    return [float(x) for x in final_seq]
```
