Can we do better? We need the center to be < 1/n while keeping the edges from being too large.

The constraint is that sum a[i] = 1 (probability distribution). The center value is sum a[i]*a[n-1-i]. By Cauchy-Schwarz, this is at most (sum a[i]^2)^(1/2) * (sum a[n-1-i]^2)^(1/2) = sum a[i]^2. But that's an upper bound, not helpful.

Actually, for the center: `b[n-1] = sum_i a[i] * a[n-1-i]`. If we let c[i] = a[n-1-i], then `b[n-1] = a · c = a · reverse(a)`. This is the "anti-correlation" of a with itself.

For a uniform distribution, a · reverse(a) = 1/n. Can we make this smaller?

If a is concentrated at one end (say a[0] = 1), then reverse(a) has its mass at the other end, so a · reverse(a) = 0. But then other entries of the convolution are large.

The tradeoff is: making the center small requires spreading the mass to the edges, but that makes the edge entries of the convolution large.

This is a minimax problem with a clear tradeoff. The optimal balance is what we're searching for.

            idx = np.argpartition(c, -top_k)[-top_k:]
            w_hard = np.zeros_like(c)
            w_hard[idx] = 1.0
            w_hard /= w_hard.sum()

            w = 0.5 * w_soft + 0.5 * w_hard

        da = _compute_grad(a, c, w, n)
        grad_z = 2.0 * n * a * (da - float(np.dot(a, da)))
        if not np.all(np.isfinite(grad_z)):
            break

        t += 1
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)

        update = lr * mh / (np.sqrt(vh) + eps)
        max_up = 0.5
        un = np.linalg.norm(update)
        if un > max_up:
            update = update * (max_up / un)

        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if (it + 1) % 100 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

        if (it + 1) % 5000 == 0:
            lr = max(0.001, lr * 0.97)

    a_final = _softmax(z)
    sc = _score_norm(a_final)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a_final.copy()

    return best_a, best_score


def _adam_subgrad(a_init, deadline, lr=0.005, max_iter=200000):
    """Adam with hard subgradient (no smoothing) for final polishing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    # Also track near-max positions for a smoother subgradient
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Use a very narrow window around the max
        threshold = cm - 0.001 * cm
        mask = c >= threshold
        if mask.sum() == 0:
            mask[int(np.argmax(c))] = True
        w = mask.astype(float)
        s_w = w.sum()
        if s_w == 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= s_w

        da = _compute_grad(a, c, w, n)
        grad_z = 2.0 * n * a * (da - float(np.dot(a, da)))
        if not np.all(np.isfinite(grad_z)):
            break

        t += 1
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)

        update = lr * mh / (np.sqrt(vh) + eps)
        max_up = 0.3
        un = np.linalg.norm(update)
        if un > max_up:
            update = update * (max_up / un)

        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if (it + 1) % 100 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

        if (it + 1) % 3000 == 0:
            lr = max(0.0005, lr * 0.95)

    a_final = _softmax(z)
    sc = _score_norm(a_final)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a_final.copy()

    return best_a, best_score


def _block_descent(a_init, deadline, n_blocks=8):
    """Block coordinate descent in z-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    block_size = max(1, n // n_blocks)
    lr = 0.01

    outer = 0
    while time.monotonic() < deadline:
        outer += 1
        for b in range(n_blocks):
            if time.monotonic() > deadline:
                break
            start = b * block_size
            end = start + block_size if b < n_blocks - 1 else n

            # Inner loop: optimize this block
            for inner in range(50):
                a = _softmax(z)
                c = _full_convolve(a, a)
                cm = float(c.max())
                beta_blk = 0.002 * cm
                c_shifted = (c - cm) / beta_blk
                e = np.exp(c_shifted)
                s = float(e.sum())
                if s < 1e-300 or not np.isfinite(s):
                    w = np.zeros_like(c)
                    w[int(np.argmax(c))] = 1.0
                else:
                    w = e / s

                da = _compute_grad(a, c, w, n)
                grad_z = 2.0 * n * a * (da - float(np.dot(a, da)))

                # Only update this block
                z[start:end] = z[start:end] - lr * grad_z[start:end]

            z = z - z.mean()
            a_now = _softmax(z)
            sc = _score_norm(a_now)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_now.copy()

        if outer % 5 == 0:
            lr = max(0.003, lr * 0.97)

    return best_a, best_score


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng):
    """Structured perturbation in Fourier domain."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    freq = rng.integers(1, max(2, n // 4))
    amp = rng.uniform(0.02, 0.1)
    phase = rng.uniform(0, 2 * np.pi)
    t = np.arange(n)
    z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: Nesterov with beta-continuation, Adam with top-k
    active-set, subgradient polishing, block coordinate descent, and progressive
    size scaling. Targets score < 1.5027.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(parent)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        update_best(parent, sc0, "parent")
    else:
        n_main = 2048
        a0 = np.ones(n_main) / n_main
        print(f"No parent, using uniform n={n_main}")
        update_best(a0, _score(a0), "uniform_init")

    # --- Phase 1: Nesterov with beta continuation (0 - 300s) ---
    phase1_end = min(deadline, t_start + 300.0)
    c0 = _full_convolve(a0, a0)
    max_c0 = float(c0.max())

    beta_sched = [
        0.005 * max_c0, 0.002 * max_c0, 0.001 * max_c0,
        0.0005 * max_c0, 0.0002 * max_c0, 0.0001 * max_c0
    ]
    p_best, sc_b = _nesterov_beta_cont(a0, phase1_end, beta_sched, lr=0.02, max_iter_per=100000)
    update_best(p_best, _score(p_best), "nesterov_p1")

    if best_seq is not None:
        a0 = best_seq / best_seq.sum()

    print(f"Phase 1: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 2: Adam with top-k active set (300s - 600s) ---
    phase2_end = min(deadline, t_start + 600.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        a0 = best_seq / best_seq.sum()
        c1 = _full_convolve(a0, a0)
        max_c1 = float(c1.max())
        beta2 = 0.001 * max_c1

        p_best, sc_b = _adam_topk(a0, phase2_end, beta2, top_k=30, lr=0.008, max_iter=300000)
        update_best(p_best, _score(p_best), "adam_topk30")

        # Retry with smaller top-k for more focus
        if time.monotonic() < phase2_end - 60:
            a0 = best_seq / best_seq.sum()
            p_best, sc_b = _adam_topk(a0, phase2_end, beta2, top_k=10, lr=0.006, max_iter=150000)
            update_best(p_best, _score(p_best), "adam_topk10")

    print(f"Phase 2: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 3: Subgradient polishing (600s - 850s) ---
    phase3_end = min(deadline, t_start + 850.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        a0 = best_seq / best_seq.sum()
        p_best, sc_b = _adam_subgrad(a0, phase3_end, lr=0.005, max_iter=200000)
        update_best(p_best, _score(p_best), "subgrad")

    print(f"Phase 3: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 4: Block coordinate descent (850s - 1100s) ---
    phase4_end = min(deadline, t_start + 1100.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        a0 = best_seq / best_seq.sum()
        p_best, sc_b = _block_descent(a0, phase4_end, n_blocks=16)
        update_best(p_best, _score(p_best), "block_desc")

    print(f"Phase 4: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 5: Multi-restart with perturbations (1100s - 1400s) ---
    phase5_end = min(deadline, t_start + 1400.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        n_restarts = 4
        per_restart = (phase5_end - time.monotonic() - 10) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break

            if i % 2 == 0:
                p_start = _perturb(best_seq / best_seq.sum(), 0.03 + 0.01 * i, rng)
            else:
                p_start = _perturb_fourier(best_seq / best_seq.sum(), rng)

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            beta_s = [0.002 * max_cs, 0.001 * max_cs, 0.0005 * max_cs]

            p_best, sc_b = _nesterov_beta_cont(
                p_start, time.monotonic() + per_restart, beta_s, lr=0.01, max_iter_per=50000
            )
            update_best(p_best, _score(p_best), f"restart{i}")

    print(f"Phase 5: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 6: Try larger n (1400s - 1700s) ---
    phase6_end = min(deadline, t_start + 1700.0)
    if best_seq is not None and len(best_seq) <= 2048 and time.monotonic() < phase6_end - 30:
        for target_n in [4096]:
            if time.monotonic() > phase6_end - 30:
                break
            p_interp = _interp_profile(best_seq / best_seq.sum(), target_n)
            sc_interp = _score(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{target_n}")

            remaining = phase6_end - time.monotonic() - 15
            if remaining > 30:
                c_int = _full_convolve(p_interp, p_interp)
                max_ci = float(c_int.max())
                beta_int = [0.002 * max_ci, 0.001 * max_ci, 0.0005 * max_ci]
                p_best, sc_b = _adam_topk(
                    p_interp, time.monotonic() + remaining, beta_int[0],
                    top_k=50, lr=0.006, max_iter=100000
                )
                update_best(p_best, _score(p_best), f"adam_n{target_n}")

    print(f"Phase 6: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 7: Final polish (1700s - deadline) ---
    if best_seq is not None and time.monotonic() < deadline - 5:
        a0 = best_seq / best_seq.sum()
        remaining = deadline - time.monotonic() - 3
        if remaining > 10:
            c_f = _full_convolve(a0, a0)
            max_cf = float(c_f.max())
            beta_f = 0.0005 * max_cf
            p_best, sc_b = _adam_topk(a0, time.monotonic() + remaining, beta_f, top_k=5, lr=0.003, max_iter=100000)
            update_best(p_best, _score(p_best), "final_polish")

    print(f"Phase 7: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
