Mirror descent (multiplicative weights) with time-interpolated step size,
top-k subgradient, multi-phase strategy, and both n=2048 and n=4096 exploration.
Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_subgradient(c, a, top_k=5):
    n = len(a)
    k = min(top_k, len(c))
    if k == 1:
        k_idx = int(np.argmax(c))
        weights = np.zeros(len(c))
        weights[k_idx] = 1.0
    else:
        k_indices = np.argpartition(c, -k)[-k:]
        c_top = c[k_indices]
        spread = c_top.max() - c_top.min()
        if spread < 1e-18:
            w_top = np.ones(k) / k
        else:
            temp = spread * 0.5
            w_top = np.exp((c_top - c_top.max()) / temp)
            w_top /= w_top.sum()
        weights = np.zeros(len(c))
        weights[k_indices] = w_top

    a_rev = a[::-1]
    wc = _full_convolve(weights, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _mirror_step(a, g, eta):
    g_max = float(np.max(np.abs(g)))
    if g_max < 1e-18:
        return a

    a_new = a * ratio
    s = float(a_new.sum())
    if s < 1e-300 or not np.isfinite(s):
        return a
    a_new /= s
    return a_new

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

    e = np.exp(z - z.max())
    return e / e.sum()
