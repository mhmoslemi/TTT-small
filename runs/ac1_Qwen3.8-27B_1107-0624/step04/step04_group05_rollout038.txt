def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 512:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _grad_smooth_max_a(c, a, beta):
    """Exact gradient of smooth_max_beta(c) w.r.t. a, where c = convolve(a,a).
    Uses the identity: d(c[k])/d(a[j]) = 2*a[k-j] if 0<=k-j<n else 0.
    So g_a[j] = 2 * sum_k w[k]*a[k-j] = 2 * cross_corr(w, a)[j].
    """
    n = len(a)
    mval = float(c.max())
    if beta < 1e-13:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
    else:

        s = float(ex.sum())
        if not np.isfinite(s) or s <= 0.0:

            ex[int(np.argmax(c))] = 1.0
            s = 1.0
        w = ex / s

    top_lags = np.argsort(c)[::-1][:K]
    A_ub = _build_conv_matrix(a, active_lags=top_lags.tolist())
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(len(top_lags))
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), top_lags
    except Exception:
        pass
    return None, None


def _full_lp(a, M):
    """Solve max sum(b) s.t. (a*b)[k] <= M for ALL k, b >= 0."""
    n = len(a)
    A_ub = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _adam_logit_fft(p0, deadline, rng, lr=0.03, beta_init=None, beta_decay=0.96,
                    max_no_imp=600, clip_range=45.0):
    """Adam optimizer in logit space with exact FFT gradient."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    if beta_init is None:
        beta = max(1e-4, 0.05 * maxc)
    else:
        beta = beta_init

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        g_a, w = _grad_smooth_max_a(c, p, beta)

        # Transform gradient to logit space
        dot = float(np.dot(p, g_a))
        g_z = p * (g_a - dot)
        g_z -= float(np.mean(g_z))

        if not np.all(np.isfinite(g_z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * g_z
        v = b2 * v + (1.0 - b2) * (g_z * g_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -clip_range, clip_range)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.002, lr * 0.9)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 20 == 0:
            beta = max(1e-5, beta * beta_decay)

        if no_imp > max_no_imp:
            ns = 0.03 + 0.02 * min(5, no_imp // max_no_imp)
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -clip_range, clip_range)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-4, 0.05 * maxc)
            lr = base_lr
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            no_imp = 0

    return best_p, best_score


def _lp_refine_active(a, deadline, K_start=5, K_end=30, max_iters=200,
                      t_init=0.3, t_decay=0.92, min_improve=1e-10):
    """Active-set LP refinement with adaptive K."""
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    t = t_init
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptively increase K
        frac = it / max(1, max_iters - 1)
        K = int(K_start + frac * (K_end - K_start))

        b, _ = _active_set_lp(a, M, K)
        if b is None or float(b.sum()) < 1e-10:
            # Try full LP
            b = _full_lp(a, M)
            if b is None or float(b.sum()) < 1e-10:
                break
        b_norm = b / b.sum()

        # Line search
        best_sc = current_score
        best_t = t
        for tc in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            a_try = (1.0 - tc) * a + tc * b_norm
            a_try = np.maximum(a_try, 1e-15)
            a_try = a_try / a_try.sum()
            sc_try = _score_norm(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_t = tc

        if best_sc < current_score - min_improve:
            a = (1.0 - best_t) * a + best_t * b_norm
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            current_score = _score_norm(a)
            current_best = a.copy()
            c = _full_convolve(a)
            M = float(c.max())
            stall = 0
        else:
            stall += 1
            if stall > 10:
                break

        t = max(0.03, t * t_decay)

    return current_best, current_score


def _final_lp_polish(a, deadline, max_iters=50):
    """Full-constraint LP with very small steps for final polishing."""
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = _full_lp(a, M)
        if b is None or float(b.sum()) < 1e-10:
            break
        b_norm = b / b.sum()

        best_sc = current_score
        best_t = 0.1
        for tc in [0.02, 0.05, 0.08, 0.1, 0.15, 0.2]:
            a_try = (1.0 - tc) * a + tc * b_norm
            a_try = np.maximum(a_try, 1e-15)
            a_try = a_try / a_try.sum()
            sc_try = _score_norm(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_t = tc

        if best_sc < current_score - 1e-12:
            a = (1.0 - best_t) * a + best_t * b_norm
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            current_score = _score_norm(a)
            current_best = a.copy()
            c = _full_convolve(a)
            M = float(c.max())
            stall = 0
        else:
            stall += 1
            if stall > 8:
                break

    return current_best, current_score


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_quadratic(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + a1 * t + a2 * t * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_random(n, rng, smooth=3):
    a = rng.random(n)
    for _ in range(smooth):
        a = np.convolve(a, np.ones(7) / 7, mode='same')
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main search function. Combines active-set LP, FFT-gradient Adam,
    and multi-scale escalation to minimize the step function score."""
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}]")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    print(f"Started. Parent n={parent.size if parent is not None else 0}, budget={budget_s}s")

    # === Phase 1: Initialization (0-40s) ===
    phase1_end = t_start + 40.0
    candidates = []

    if parent is not None:
        pn = parent.size
        sc = _score_norm(parent)
        candidates.append((sc, parent.copy(), pn, 'parent'))
        print(f"  Parent score: {sc:.10f} n={pn}")
        for n_try in [512, 1024, 2048]:
            if pn != n_try:
                pi = _interp_profile(parent, n_try)
                if pi is not None:
                    sc = _score_norm(pi)
                    candidates.append((sc, pi, n_try, f'parent_interp_n{n_try}'))

    n_base = 1024
    profs = [
        ('lin1.5', _make_profile_linear(n_base, 1.5)),
        ('lin2.5', _make_profile_linear(n_base, 2.5)),
        ('cos1_0.5', _make_profile_cosine(n_base, 1, 0.5)),
        ('cos3_0.3', _make_profile_cosine(n_base, 3, 0.3)),
        ('quad1', _make_profile_quadratic(n_base, 1.0, -0.8)),
        ('rand1', _make_profile_random(n_base, rng)),
        ('rand2', _make_profile_random(n_base, rng, smooth=5)),
    ]
    for label, p in profs:
        sc = _score_norm(p)
        candidates.append((sc, p, n_base, label))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f}")
    _update(candidates[0][1], candidates[0][0], 'init')

    # === Phase 2: Active-set LP refinement (40s-700s) ===
    phase2_end = t_start + 700.0
    if best_seq is not None and best_seq.size <= 2048:
        # First round of LP
        per = min(300.0, phase2_end - time.monotonic() - 10)
        if per > 20:
            p_lp, sc_lp = _lp_refine_active(best_seq, time.monotonic() + per,
                                            K_start=5, K_end=30, max_iters=300,
                                            t_init=0.3, t_decay=0.93)
            _update(p_lp, sc_lp, 'LP_round1')

        # Second round with finer parameters
        if time.monotonic() < phase2_end - 30:
            per = min(300.0, phase2_end - time.monotonic() - 10)
            if per > 20:
                p_lp2, sc_lp2 = _lp_refine_active(best_seq, time.monotonic() + per,
                                                  K_start=10, K_end=50, max_iters=200,
                                                  t_init=0.2, t_decay=0.94)
                _update(p_lp2, sc_lp2, 'LP_round2')

    print(f"Phase 2 done. Best: {best_score:.10f}")

    # === Phase 3: Gradient polish (700s-1200s) ===
    phase3_end = t_start + 1200.0
    if best_seq is not None:
        # Adam with decreasing beta
        per = min(400.0, phase3_end - time.monotonic() - 10)
        if per > 20:
            p_ad, sc_ad = _adam_logit_fft(best_seq, time.monotonic() + per, rng,
                                          lr=0.03, beta_decay=0.965, max_no_imp=500)
            _update(p_ad, sc_ad, 'adam_round1')

        # Second Adam round with smaller lr
        if time.monotonic() < phase3_end - 20:
            per = min(200.0, phase3_end - time.monotonic() - 10)
            if per > 15:
                p_ad2, sc_ad2 = _adam_logit_fft(best_seq, time.monotonic() + per, rng,
                                                lr=0.01, beta_decay=0.975, max_no_imp=400)
                _update(p_ad2, sc_ad2, 'adam_round2')

        # LP after Adam
        if time.monotonic() < phase3_end - 10 and best_seq.size <= 2048:
            per = min(100.0, phase3_end - time.monotonic() - 5)
            if per > 15:
                p_lp3, sc_lp3 = _lp_refine_active(best_seq, time.monotonic() + per,
                                                  K_start=15, K_end=60, max_iters=100,
                                                  t_init=0.15, t_decay=0.94)
                _update(p_lp3, sc_lp3, 'post_adam_LP')

    print(f"Phase 3 done. Best: {best_score:.10f}")

    # === Phase 4: Final LP polish (1200s-1450s) ===
    phase4_end = t_start + 1450.0
    if best_seq is not None and best_seq.size <= 2048:
        per = min(250.0, phase4_end - time.monotonic() - 5)
        if per > 20:
            p_fp, sc_fp = _final_lp_polish(best_seq, time.monotonic() + per, max_iters=100)
            _update(p_fp, sc_fp, 'final_LP')

        # Tiny Adam step
        if time.monotonic() < phase4_end - 5:
            per = min(60.0, phase4_end - time.monotonic() - 3)
            if per > 10:
                p_fa, sc_fa = _adam_logit_fft(best_seq, time.monotonic() + per, rng,
                                              lr=0.005, beta_init=1e-4,
                                              beta_decay=0.98, max_no_imp=200)
                _update(p_fa, sc_fa, 'final_adam')

    print(f"Phase 4 done. Best: {best_score:.10f}")

    # === Phase 5: Multi-restart (1450s-1650s) ===
    phase5_end = t_start + 1650.0
    if best_seq is not None and time.monotonic() < phase5_end:
        n_best = best_seq.size
        n_restarts = 4
        remaining = phase5_end - time.monotonic()
        per_r = remaining / (n_restarts + 1)

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break

            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            ns = 0.02 + 0.015 * i
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -45, 45)
            z -= z.mean()
            p_s = _softmax_from_z(z)

            per_ad = min(per_r * 0.5, phase5_end - time.monotonic() - 15)
            if per_ad > 10:
                p_r, sc_r = _adam_logit_fft(p_s, time.monotonic() + per_ad, rng,
                                            lr=0.02, beta_decay=0.96, max_no_imp=300)
                _update(p_r, sc_r, f'restart{i}')

            if best_seq.size <= 2048 and time.monotonic() < phase5_end - 10:
                per_l = min(30.0, phase5_end - time.monotonic() - 8)
                if per_l > 10:
                    p_lr, sc_lr = _lp_refine_active(best_seq, time.monotonic() + per_l,
                                                    K_start=10, K_end=40, max_iters=50,
                                                    t_init=0.1, t_decay=0.93)
                    _update(p_lr, sc_lr, f'LP_restart{i}')

    print(f"Phase 5 done. Best: {best_score:.10f}")

    # === Phase 6: Multi-scale escalation if time remains (1650s-1770s) ===
    phase6_end = deadline - 10.0
    if best_seq is not None and time.monotonic() < phase6_end:
        n_cur = best_seq.size
        if n_cur <= 1024:
            target_n = 2048
            p_int = _interp_profile(best_seq, target_n)
            if p_int is not None:
                sc_int = _score_norm(p_int)
                _update(p_int, sc_int, f'interp_n{target_n}')
                if time.monotonic() < phase6_end - 15:
                    per = min(60.0, phase6_end - time.monotonic() - 10)
                    if per > 10:
                        p_e, sc_e = _adam_logit_fft(p_int, time.monotonic() + per, rng,
                                                    lr=0.03, beta_decay=0.96, max_no_imp=200)
                        _update(p_e, sc_e, f'escalate_n{target_n}')
                        if sc_e < best_score:
                            # Try LP on the escalated version
                            if time.monotonic() < phase6_end - 5 and p_e.size <= 2048:
                                per2 = min(30.0, phase6_end - time.monotonic() - 3)
                                if per2 > 10:
                                    p_el, sc_el = _lp_refine_active(p_e, time.monotonic() + per2,
                                                                    K_start=10, K_end=30,
                                                                    max_iters=30,
                                                                    t_init=0.15, t_decay=0.93)
                                    _update(p_el, sc_el, f'LP_escalate_n{target_n}')

    print(f"Phase 6 done. Best: {best_score:.10f}")

    # === Final: one more polish on absolute best ===
    if time.monotonic() < deadline - 3 and best_seq is not None and best_seq.size <= 2048:
        per = deadline - time.monotonic() - 3
        if per > 8:
            p_fl, sc_fl = _final_lp_polish(best_seq, time.monotonic() + per, max_iters=20)
            _update(p_fl, sc_fl, 'absolute_final')

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(1024, 1.5)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    fb = _make_profile_linear(1024, 1.5)
    print(f"FALLBACK n=1024")
    return [float(x) for x in fb]
