def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_actual(a):
    """Full score: 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _make_profiles(n):
    """Generate diverse initial profiles, all normalized to sum=1."""
    t = np.linspace(0, 1, n)

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-15)
    return p / p.sum()

def _run_lp_iteration(a, M=None, timeout=60):
    """
    LP-based peak flattening.
    Given a (sum=1), find b >= 0 maximizing sum(b)
    subject to conv(a,b)_k <= M for all k.
    Returns the optimal b normalized to sum=1.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    a = a / s

    if M is None:
        c = _full_convolve(a, a)
        M = float(c.max())

    # Build constraint matrix: A[k, i] = a[i] if 0 <= k-i < n
    # Max(2n-1) x n sparse matrix
    # Each column i has non-zeros at rows i, i+1, ..., i+n-1
    # Row k, col i is nonzero if 0 <= k-i <= n-1, i.e., k-n+1 <= i <= k
    # and 0 <= i <= n-1
    # So for row k: i ranges from max(0, k-n+1) to min(n-1, k)
    # A[k, i] = a[i]

    # Use sparse format
    from scipy import sparse

    rows = []
    cols = []
    vals = []

    for i in range(n):
        # a[i] appears in conv(a,b)[k] for k = i, i+1, i+2, ..., i+n-1
        # (since b[k-i] is nonzero when 0 <= k-i <= n-1)
        valid_rows = np.arange(i, min(i + n, 2 * n - 1))
        for r in valid_rows:
            rows.append(r)
            cols.append(i)
            vals.append(a[i])

    A_mat = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    # Maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)

    # Constraints: A_mat @ b <= M, b >= 0
    b_ub = M * np.ones(2 * n - 1)

    t_start = time.monotonic()
    try:
        result = linprog(c_obj, A_ub=A_mat, b_ub=b_ub,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'time_limit': timeout, 'presolve': True})

        if time.monotonic() - t_start > timeout + 5:
            return a  # timeout, return unchanged
    except Exception as e:
        return a

    if result.success and result.x is not None:
        b_opt = np.maximum(result.x, 0)
        s_b = b_opt.sum()
        if s_b > 1e-10:
            return b_opt / s_b
    return a


def _run_coordinate_descent(a_init, deadline, rng, max_sweeps=100):
    """
    Coordinate-wise optimization: for each coordinate i, find the best value
    given all others fixed. Uses ternary search in log-space.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)

    indices = list(range(n))

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        # Random order each sweep
        rng.shuffle(indices)

        improved = 0
        for i in indices:
            if time.monotonic() > deadline:
                break

            # Save current value
            old = a[i]

            # Try log-space perturbations
            best_val = old
            best_sc = best_score  # local best

            # Ternary-ish search: try several multipliers
            multipliers = [0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.4, 1.7, 2.0]
            if old < 1e-10:
                multipliers = [1e-8, 1e-6, 1e-4, 1e-3, 1e-2, old + 0.001]

            for mult in multipliers:
                trial = old * mult
                a[i] = trial
                s = a.sum()
                a_norm = a / s
                sc = _score_norm(a_norm)
                if sc < best_sc:
                    best_sc = sc
                    best_val = trial
                a[i] = old

            if best_val != old:
                a[i] = best_val
                s = a.sum()
                a = a / s
                if best_sc < best_score:
                    best_score = best_sc
                    best_a = a.copy()
                improved += 1

        if improved == 0:
            break

    return best_a, best_score

        c_shifted = (c - cm) / beta
        max_cs = c_shifted.max()
        e = np.exp(c_shifted - max_cs)
        w = e / e.sum()

        # Gradient
        a_rev = a[::-1]
        W = float(np.dot(w, c))
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]

        # d/dz_i [sum_k w_k * sum_j a[j]*a[k-j]]
        # = sum_k w_k * a[k-i] (the "h" vector)
        # But since a = softmax(z), we need the full gradient through softmax
        # grad_z = a * (h - dot(w, a * h) / dot(w, a)) -- actually:
        # The objective in z is: sum_k w_k * sum_j softmax(z)[j] * softmax(z)[k-j]

        grad = 4.0 * n * a * (h - W)

                beta = 0.01 * cm
                c_shifted = (c - cm) / beta
                max_cs = c_shifted.max()
                e = np.exp(c_shifted - max_cs)
                w = e / e.sum()
                a_rev = a[::-1]
                W = float(np.dot(w, c))
                w_conv = _full_convolve(w, a_rev)
                h = w_conv[n - 1:2*n - 1]
                grad = 4.0 * n * a * (h - W)
                if not np.all(np.isfinite(grad)):
                    break
                m = b1 * m + (1 - b1) * grad
                v = b2 * v + (1 - b2) * (grad * grad)
                mh = m / (1 - b1 ** t_count)
                vh = v / (1 - b2 ** t_count)
                z = z - lr * mh / (np.sqrt(vh) + eps)
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                a = _softmax_fn(z)
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_sc_sgd:
                    best_sc_sgd = sc
                    best_a_sgd = a.copy()
                if t_count % 50000 == 0:
                    lr = max(0.005, lr * 0.9)

            update_best(best_a_sgd, best_sc_sgd, "sgd_fine")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Coordinate descent on active set (200s) ---
    phase5_end = min(deadline, t_start + 1460.0)
    print("Phase 5: Coordinate descent (active set)")

    if time.monotonic() < phase5_end and best_seq is not None:
        n = len(best_seq)
        a_cur = best_seq.copy()

        # Only optimize "active" coordinates (top 50% by value)
        threshold = np.median(a_cur) * 0.1
        active = np.where(a_cur > threshold)[0]
        print(f"  Active set size: {len(active)} / {n}")

        if len(active) <= 2000:
            # Run coordinate descent on active set
            indices_list = list(active)
            best_sc_cd = _score_norm(a_cur)

            for sweep in range(50):
                if time.monotonic() > phase5_end - 10:
                    break
                rng.shuffle(indices_list)
                for i in indices_list:
                    if time.monotonic() > phase5_end - 10:
                        break
                    old = a_cur[i]
                    if old < 1e-15:
                        continue
                    best_val = old
                    for mult in [0.8, 0.9, 1.1, 1.2]:
                        a_cur[i] = old * mult
                        a_cur = a_cur / a_cur.sum()
                        sc = _score_norm(a_cur)
                        if sc < best_sc_cd * 0.9999:
                            pass  # will be caught below
                        a_cur[i] = old * mult
                    # Just do a simple line search
                    for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                        a_cur[i] = old * mult
                        a_cur = a_cur / a_cur.sum()
                        sc = _score_norm(a_cur)
                        a_cur[i] = old * mult
                        if sc < _score_norm(a_cur):
                            a_cur[i] = old * mult
                            a_cur = a_cur / a_cur.sum()

                    a_cur[i] = old
                    a_cur = a_cur / a_cur.sum()

            sc_cd = _score_norm(a_cur)
            update_best(a_cur, sc_cd, "coord_descent")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Final polish with very fine optimization (200s) ---
    phase6_end = deadline
    print("Phase 6: Final fine polish")

    if time.monotonic() < phase6_end and best_seq is not None:
        # Very small learning rate mirror descent
        a_start = best_seq.copy()
        dl = min(phase6_end - 20, time.monotonic() + (phase6_end - time.monotonic()) * 0.7)

        # Fine mirror descent
        n = len(a_start)
        z = np.log(np.maximum(a_start, 1e-30))
        z = z - z.mean()
        v = np.zeros(n)
        lr = 0.05
        momentum = 0.95
        best_sc_f = _score_norm(a_start)
        best_a_f = a_start.copy()

        for it in range(1000000):
            if time.monotonic() > dl:
                break
            a = _softmax_fn(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            beta = 0.005 * cm

            c_shifted = (c - cm) / beta
            max_cs = c_shifted.max()
            e = np.exp(c_shifted - max_cs)
            w = e / e.sum()

            a_rev = a[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2*n - 1]

            v = momentum * v - lr * h
            z = z + v
            z = z - z.mean()
            z = np.clip(z, -100, 100)
            a = _softmax_fn(z)

            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_sc_f:
                best_sc_f = sc
                best_a_f = a.copy()

            if it % 200000 == 0:
                lr *= 0.8
                v *= 0.5

        update_best(best_a_f, best_sc_f, "final_polish")

        # Try one more LP at the very end
        if time.monotonic() < phase6_end - 30:
            c = _full_convolve(best_a_f, best_a_f)
            M = float(c.max())
            a_lp = _run_lp_iteration(best_a_f, M=M, timeout=20)
            sc_lp = _score_norm(a_lp)
            if sc_lp < best_sc_f:
                update_best(a_lp, sc_lp, "final_LP")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            # Fallback
            final_seq = np.ones(256) / 256.0

        sc_final = _score_actual(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        # Validate with evaluator if size allows
        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = np.exp(np.linspace(0, 3, 1024))
    fallback = fallback / fallback.sum()
    print(f"FALLBACK: n=1024, score={_score_actual(fallback):.10f}")
    return [float(x) for x in fallback]
