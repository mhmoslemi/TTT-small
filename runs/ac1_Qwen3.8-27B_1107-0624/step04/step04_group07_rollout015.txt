        a_half = np.exp(z - z.max())
        a_half = a_half / a_half.sum()
        dot_h = float(np.dot(a_half, g_half))
        gz_h = a_half * (g_half - dot_h)
        gz_h -= gz_h.mean()

        if not np.all(np.isfinite(gz_h)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t = 0
            continue

        t += 1
        m = b1 * m + (1 - b1) * gz_h
        v = b2 * v + (1 - b2) * (gz_h * gz_h)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40, 40)
        z -= z.mean()

        a = _expand(z)
        a = a / a.sum()
        c = _full_convolve(a)
        score = float(2.0 * n * c.max())

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        if t % 15 == 0:
            sidx = np.argsort(c)[::-1]
            if c.size >= 2:
                gap = c[sidx[0]] - c[sidx[1]]
                if gap > beta * 0.8:
                    beta = max(1e-5, beta * 0.85)
                elif gap < beta * 0.03:
                    beta = min(0.5 * c.max(), beta * 1.3)

        if no_imp > 0 and no_imp % 80 == 0:
            lr = max(0.001, lr * 0.85)

        if no_imp > 250:
            z = best_z + 0.1 * rng.normal(size=len(z))
            z = np.clip(z, -40, 40)
            z -= z.mean()
            a = _expand(z)
            a = a / a.sum()
            c = _full_convolve(a)
            beta = 0.08 * c.max()
            m.fill(0)
            v.fill(0)
            t = 0
            lr = base_lr
            no_imp = 0

    a_best = _expand(best_z)
    a_best = a_best / a_best.sum()
    return a_best, best_score


def _coord_descent(a, deadline, max_sweeps=30):
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            old = a[j]
            found = False
            for factor in [0.6, 0.8, 1.25, 1.5, 2.0]:
                a[j] = old * factor
                sc = _score_norm(a)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    found = True
                    # Try to push further in same direction
                    if factor < 1.0:
                        a[j] = old * 0.7
                        sc2 = _score_norm(a)
                        if sc2 < best_score - 1e-13:
                            best_score = sc2
                            best_a = a.copy()
                    break
            a[j] = old
        if not improved:
            break

    return best_a, best_score


def _peak_targeting(a, deadline, num_peaks=5, iterations=50):
    """Specifically target the top peaks in the convolution."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()

    for it in range(iterations):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a)
        top_idx = np.argsort(c)[::-1][:num_peaks]

        # For each peak, reduce contributing elements
        for k in top_idx:
            # Elements contributing to peak at lag k
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            contrib = np.zeros(n)
            for i in range(i_start, i_end + 1):
                j = k - i
                contrib[i] += a[j]
                if i != j:
                    contrib[j] += a[i]

            # Reduce the top contributors
            top_contrib = np.argsort(contrib)[::-1][:min(10, n // 4)]
            reduction = 0.95
            old_vals = a[top_contrib].copy()
            a[top_contrib] *= reduction
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            sc = _score_norm(a)
            if sc >= best_score:
                a[top_contrib] = old_vals
                a = a / a.sum()

        sc = _score_norm(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()


def _make_profile(n, ptype, p1=1.0, p2=0.0, p3=0.0):
    t = np.linspace(0, 1, n)
    if ptype == 'linear':
        a = 1.0 + p1 * (t - 0.5)
    elif ptype == 'quadratic':
        a = 1.0 + p1 * (2 * t - 1) ** 2
    elif ptype == 'cosine':
        a = 1.0 + p1 * np.cos(2 * np.pi * p2 * t)
    elif ptype == 'sine':
        a = 1.0 + p1 * np.sin(2 * np.pi * p2 * t)
    elif ptype == 'gauss':
        a = np.exp(-0.5 * ((t - 0.5) / p1) ** 2)
    elif ptype == 'edge':
        a = 1.0 / (0.05 + abs(t - 0.5) * 2.0) ** p1
    elif ptype == 'parabola':
        a = (t - 0.5) ** 2 + p1
    elif ptype == 'two_gauss':
        a = np.exp(-0.5 * ((t - p1) / p3) ** 2) + np.exp(-0.5 * ((t - p2) / p3) ** 2)
    elif ptype == 'raised_sine':
        a = 1.0 - p1 * np.sin(np.pi * t)
    elif ptype == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * t)) + 0.01
    elif ptype == 'hamming':
        a = 0.54 + 0.46 * np.cos(2 * np.pi * t) + 0.01
    elif ptype == 'flat':
        a = np.ones(n)
    elif ptype == 'triangular':
        a = 1.0 - np.abs(2 * t - 1)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: diverse initialization -> FFT Adam with adaptive
    smooth-max -> symmetric optimization -> peak targeting -> multi-scale
    escalation -> coordinate descent refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent if available
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    # === Phase 1: Diverse initialization ===
    print("Phase 1: Generating diverse initial candidates...")
    candidates = []

    n_sizes = [128, 256, 512, 1024]
    profile_list = [
        ('lin1.0', 'linear', 1.0, 0, 0),
        ('lin2.0', 'linear', 2.0, 0, 0),
        ('lin3.0', 'linear', 3.0, 0, 0),
        ('quadr1', 'quadratic', 1.0, 0, 0),
        ('quadr2', 'quadratic', 2.5, 0, 0),
        ('cos0.8_1', 'cosine', 0.8, 1, 0),
        ('cos0.6_2', 'cosine', 0.6, 2, 0),
        ('cos0.5_3', 'cosine', 0.5, 3, 0),
        ('sin0.7_1', 'sine', 0.7, 1, 0),
        ('sin0.5_2', 'sine', 0.5, 2, 0),
        ('gauss0.25', 'gauss', 0.25, 0, 0),
        ('gauss0.35', 'gauss', 0.35, 0, 0),
        ('edge1.5', 'edge', 1.5, 0, 0),
        ('edge2.5', 'edge', 2.5, 0, 0),
        ('para0.3', 'parabola', 0.3, 0, 0),
        ('tg_0.3_0.7', 'two_gauss', 0.3, 0.7, 0.15),
        ('rs0.7', 'raised_sine', 0.7, 0, 0),
        ('rs0.85', 'raised_sine', 0.85, 0, 0),
        ('hann', 'hann', 0, 0, 0),
        ('tri', 'triangular', 0, 0, 0),
    ]

    for n in n_sizes:
        for (label, ptype, p1, p2, p3) in profile_list:
            p = _make_profile(n, ptype, p1, p2, p3)
            sc = _score_norm(p)
            candidates.append((sc, p, n, label))

    # Parent-based candidates
    if parent is not None:
        pn = len(parent)
        for n in [256, 512, 1024, 2048]:
            pi = _interp(parent, n)
            sc = _score_norm(pi)
            candidates.append((sc, pi, n, 'parent'))
        if 0 < pn <= 4096:
            sc = _score_norm(parent)
            candidates.append((sc, parent.copy(), pn, 'parent_exact'))

    # Random perturbations of parent
    if parent is not None:
        pn = len(parent)
        for i in range(5):
            z = np.log(np.maximum(parent, 1e-12))
            z -= z.mean()
            z += (0.05 + 0.05 * i) * rng.normal(size=pn)
            z = np.clip(z, -40, 40)
            z -= z.mean()
            a_p = np.exp(z - z.max())
            a_p = a_p / a_p.sum()
            sc = _score_norm(a_p)
            candidates.append((sc, a_p, pn, f'perturb{i}'))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates. Best init: {candidates[0][0]:.8f} n={candidates[0][2]} [{candidates[0][3]}]")

    if candidates[0][1] is not None:
        best_seq = candidates[0][1].copy()
        best_score = candidates[0][0]
        print(f"  BEST: {best_score:.10f} n={best_seq.size} [init]")

    # === Phase 2: Adam optimization on top candidates ===
    print("Phase 2: Adam optimization...")
    phase2_end = t_start + 700.0
    polished = []

    # Sort by (n, score) to try smaller n first for LP-like quality
    top_candidates = [c for c in candidates if c[2] <= 1024][:12]
    top_candidates.sort(key=lambda x: (x[2], x[0]))

    time_per = max(30.0, (phase2_end - time.monotonic()) / max(1, len(top_candidates)))

    for i, (sc, p, n_c, label) in enumerate(top_candidates):
        if time.monotonic() > phase2_end - 20:
            break
        per = min(time_per, phase2_end - time.monotonic() - 10)
        if per < 20:
            break

        print(f"  [{i+1}/{len(top_candidates)}] n={n_c} [{label}] init_score={sc:.8f} alloc={per:.0f}s")
        a_opt, sc_opt = _adam_optimize(p, time.monotonic() + per, rng,
                                        lr=0.04, no_imp_limit=200)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  BEST: {best_score:.10f} n={a_opt.size} [Adam_{label}]")
        polished.append((sc_opt, a_opt, n_c, label))

        # Also try symmetric version
        if per > 40 and time.monotonic() < phase2_end - 30:
            per_sym = min(30.0, phase2_end - time.monotonic() - 10)
            if per_sym > 15:
                a_sym, sc_sym = _adam_optimize_symmetric(p, time.monotonic() + per_sym, rng, lr=0.04)
                if sc_sym < best_score:
                    best_seq = a_sym.copy()
                    best_score = sc_sym
                    print(f"  BEST: {best_score:.10f} n={a_sym.size} [AdamSym_{label}]")
                polished.append((sc_sym, a_sym, n_c, label + '_sym'))

    polished.sort(key=lambda x: x[0])
    print(f"  After Phase 2: best={best_score:.10f}")

    # === Phase 3: Peak targeting on best results ===
    print("Phase 3: Peak targeting...")
    phase3_end = t_start + 1000.0

    for (sc, p, n_c, label) in polished[:3]:
        if time.monotonic() > phase3_end - 15:
            break
        per = min(40.0, phase3_end - time.monotonic() - 10)
        if per < 15:
            break
        a_pt, sc_pt = _peak_targeting(p, time.monotonic() + per)
        if sc_pt < best_score:
            best_seq = a_pt.copy()
            best_score = sc_pt
            print(f"  BEST: {best_score:.10f} n={a_pt.size} [peak_{label}]")
        polished.append((sc_pt, a_pt, n_c, label + '_peak'))

    # === Phase 4: Multi-scale escalation ===
    print("Phase 4: Multi-scale escalation...")
    phase4_end = t_start + 1400.0

    if best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048]
        elif n_cur <= 1024:
            target_ns = [2048]
        elif n_cur <= 2048:
            target_ns = [4096]

        for tn in target_ns:
            if time.monotonic() > phase4_end - 60:
                break
            p_int = _interp(best_seq, tn)
            sc_int = _score_norm(p_int)
            print(f"  Interp to n={tn}: {sc_int:.8f}")
            if sc_int < best_score:
                best_seq = p_int.copy()
                best_score = sc_int
                print(f"  BEST: {best_score:.10f} n={tn} [interp]")

            if time.monotonic() < phase4_end - 40:
                per = min(60.0, phase4_end - time.monotonic() - 20)
                if per > 20:
                    a_e, sc_e = _adam_optimize(p_int, time.monotonic() + per, rng,
                                               lr=0.03, no_imp_limit=200)
                    if sc_e < best_score:
                        best_seq = a_e.copy()
                        best_score = sc_e
                        print(f"  BEST: {best_score:.10f} n={tn} [escalate]")

    # === Phase 5: Coordinate descent refinement ===
    print("Phase 5: Coordinate descent...")
    phase5_end = t_start + 1600.0

    if best_seq is not None:
        per = min(80.0, phase5_end - time.monotonic() - 10)
        if per > 20:
            a_cd, sc_cd = _coord_descent(best_seq, time.monotonic() + per)
            if sc_cd < best_score:
                best_seq = a_cd.copy()
                best_score = sc_cd
                print(f"  BEST: {best_score:.10f} n={best_seq.size} [coord]")

        # Peak targeting after coordinate descent
        if time.monotonic() < phase5_end - 10:
            per2 = min(30.0, phase5_end - time.monotonic() - 5)
            if per2 > 10:
                a_pt2, sc_pt2 = _peak_targeting(best_seq, time.monotonic() + per2,
                                                num_peaks=3, iterations=30)
                if sc_pt2 < best_score:
                    best_seq = a_pt2.copy()
                    best_score = sc_pt2
                    print(f"  BEST: {best_score:.10f} n={best_seq.size} [peak2]")

    # === Phase 6: Final Adam polish ===
    print("Phase 6: Final polish...")
    if best_seq is not None and time.monotonic() < deadline - 20:
        per = min(60.0, deadline - time.monotonic() - 15)
        if per > 20:
            a_final, sc_final = _adam_optimize(best_seq, time.monotonic() + per, rng,
                                              lr=0.01, no_imp_limit=150,
                                              beta_init=0.01 * _full_convolve(best_seq).max())
            if sc_final < best_score:
                best_seq = a_final.copy()
                best_score = sc_final
                print(f"  BEST: {best_score:.10f} n={best_seq.size} [final_polish]")

            # Quick coordinate descent on the polished result
            if time.monotonic() < deadline - 10:
                per_cd = min(20.0, deadline - time.monotonic() - 5)
                if per_cd > 10:
                    a_cdf, sc_cdf = _coord_descent(best_seq, time.monotonic() + per_cd)
                    if sc_cdf < best_score:
                        best_seq = a_cdf.copy()
                        best_score = sc_cdf
                        print(f"  BEST: {best_score:.10f} n={best_seq.size} [final_cd]")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        sc = _score_norm(final)
        print(f"\nFINAL: score={sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
                if eval_sc < best_score:
                    best_score = eval_sc
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile(512, 'raised_sine', 0.8)
    print("FALLBACK n=512 raised_sine")
    return [float(x) for x in fb]
