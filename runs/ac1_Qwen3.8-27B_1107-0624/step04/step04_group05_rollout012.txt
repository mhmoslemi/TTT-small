"""
SLP (Sequential Linear Programming) approach for minimizing self-convolution peak.
Uses linearized quadratic constraints in an LP, with line search and multiple
restart strategies. Combined with LP best-response and smooth-max Adam for polishing.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return None
    return a / s


def _slp_lp(a0, c0):
    """
    Solve the SLP linear program at point a0 with convolution c0 = a0*a0.
    Returns the direction delta (or None if LP fails).
    
    LP: minimize M
        subject to: 2*(delta*a0)[k] <= M - c0[k] for all k
                    delta[i] >= -a0[i] for all i
                    sum(delta) = 0
    """
    n = len(a0)
    # Build the convolution matrix for a0 (maps delta to delta*a0)
    # (delta*a0)[k] = sum_i delta[i] * a0[k-i]
    # Row k has nonzeros at positions i where 0 <= k-i <= n-1, i.e., max(0, k-n+1) <= i <= min(n-1, k)
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            # (delta*a0)[k] = sum_{i in idx} delta[i] * a0[k-i]
            coefs = 2.0 * a0[k - idx]
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(coefs.tolist())
    
    A_conv = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # Full A_ub: [A_conv, -1; -I_n, 0]
    # b_ub: [-c0; a0]
    
    # Build as sparse
    from scipy.sparse import eye as speye, vstack, hstack
    
    # First block: A_conv @ delta - M <= -c0
    # Second block: -delta <= a0 (i.e., delta >= -a0)
    
    A_ub_top = hstack([A_conv, -np.ones((2 * n - 1, 1))]).tocsr()
    A_ub_bot = hstack([-speye(n), np.zeros((n, 1))]).tocsr()
    A_ub = vstack([A_ub_top, A_ub_bot]).tocsr()
    
    b_ub = np.concatenate([-c0, a0])
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Objective: minimize M
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds
    bounds = [(-a0[i], None) for i in range(n)] + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n].copy()
    return None


def _slp_iterate(a0, max_iters=200, deadline=None):
    """
    Run SLP iterations starting from a0.
    Returns the best sequence found.
    """
    n = len(a0)
    a = a0.copy()
    c = _full_convolve(a)
    M = float(c.max())
    best_a = a.copy()
    best_M = M
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        delta = _slp_lp(a, c)
        if delta is None:
            break
        
        # Line search: try t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        best_t = 0.0
        best_sc = float('inf')
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = a + t * delta
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            sc = _score_norm(a_try)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        
        if best_t == 0.0 or best_sc >= M - 1e-12:
            break
        
        a = a + best_t * delta
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        c = _full_convolve(a)
        M = float(c.max())
        
        if M < best_M:
            best_M = M
            best_a = a.copy()
    
    return best_a, best_M

    Returns normalized b or None.
    """

    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)

            coefs = a[k - idx]

    """Compute softmax weights for smooth max."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 0.0))
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    return w
