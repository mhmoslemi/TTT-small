    return max(2, k)

def _full_convolve(a):

    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _next_pow2(n):

def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

        a_new = a_new / a_new.sum()
        sc = _score(a_new)

        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()

        a = a_new
        c = _full_convolve(a)
        M_new = float(c.max())
        M = max(M_new * m_shrink, best_score / (2.0 * n) * 1.003)
        t_blend = max(0.15, t_blend * 0.9)

    return best_a, best_score


def _peak_shave(a, deadline, n_iter=200):
    """Peak shaving: move mass away from peak-contributing positions."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng(999)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a)
        M = float(c.max())
        peak_k = int(np.argmax(c))
        score = float(2.0 * n * M)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        # Find positions contributing to peak
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        partners = np.arange(i_start, i_end + 1)
        contrib = a[partners] * a[peak_k - partners]
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            continue

        # Move mass from highest contributor to a random low position
        top_idx = partners[np.argmax(contrib)]
        # Find a position that doesn't contribute to any of the top 3 peaks
        top3 = np.argsort(c)[::-1][:3]
        exclude = set()
        for kk in top3:
            for ii in range(max(0, kk - n + 1), min(n, kk + 1)):
                exclude.add(ii)
                exclude.add(kk - ii)

        candidates = np.array([x for x in range(n) if x not in exclude], dtype=int)
        if len(candidates) == 0:
            candidates = np.array([x for x in range(n) if x != top_idx], dtype=int)
        if len(candidates) == 0:
            continue

        target = candidates[rng.integers(0, len(candidates))]
        frac = 0.02 + 0.03 * rng.random()
        amount = a[top_idx] * frac
        a[top_idx] -= amount
        a[target] += amount
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 1e-12)
    return p1 / p1.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, label=""):
        nonlocal best_seq, best_score
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            p = np.maximum(p, 1e-12)
            p = p / p.sum()
            sc = _score(p)
            if np.isfinite(sc) and sc < best_score:
                best_seq = p.copy()
                best_score = float(sc)
                print(f"  BEST: {best_score:.10f} n={p.size} [{label}]")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        print(f"Parent: n={parent.size}, score={_score(parent):.10f}")
        p0 = parent
    else:
        n = 1024
        t = np.linspace(0, 1, n)
        p0 = 1.0 + t  # linear
        p0 = p0 / p0.sum()
        print(f"No parent, using linear init n={n}")

    _update(p0, "initial")

    # Phase 1: Aggressive Adam (350s)
    print(f"Phase 1: Adam, score={best_score:.8f}")
    phase1_end = t_start + 350.0
    p1, sc1 = _adam_optimize(best_seq, phase1_end, rng, lr=0.06,
                             beta_init_frac=0.05, beta_decay=0.93,
                             no_imp_limit=200)
    _update(p1, 'adam1')

    # Phase 2: LP refinement (300s)
    print(f"Phase 2: LP refine, score={best_score:.8f}")
    phase2_end = t_start + 650.0
    if best_seq.size <= 2048 and time.monotonic() < phase2_end:
        p2, sc2 = _lp_refine(best_seq, phase2_end, t_blend=0.4,
                             max_solves=35, m_shrink=0.997)
        _update(p2, 'lp1')

    # Phase 3: Adam after LP (250s)
    print(f"Phase 3: Adam post-LP, score={best_score:.8f}")
    phase3_end = t_start + 900.0
    p3, sc3 = _adam_optimize(best_seq, phase3_end, rng, lr=0.04,
                             beta_init_frac=0.04, beta_decay=0.95,
                             no_imp_limit=200)
    _update(p3, 'adam2')

    # Phase 4: Peak shaving + LP (200s)
    print(f"Phase 4: Peak shave + LP, score={best_score:.8f}")
    phase4_end = t_start + 1100.0
    if time.monotonic() < phase4_end:
        p4s, sc4s = _peak_shave(best_seq, min(phase4_end - 60, time.monotonic() + 60),
                                n_iter=100)
        _update(p4s, 'shave1')
        if time.monotonic() < phase4_end - 30 and best_seq.size <= 1536:
            p4l, sc4l = _lp_refine(best_seq, phase4_end, t_blend=0.3,
                                   max_solves=15, m_shrink=0.998)
            _update(p4l, 'lp2')

    # Phase 5: Try n=2048 (250s)
    print(f"Phase 5: Multi-scale n=2048, score={best_score:.8f}")
    phase5_end = t_start + 1350.0
    if time.monotonic() < phase5_end:
        n2 = 2048
        p_int = _interp_profile(best_seq, n2)
        sc_int = _score(p_int)
        print(f"  Interpolated n={n2}: {sc_int:.10f}")
        _update(p_int, f'interp{n2}')

        per = min(120.0, phase5_end - time.monotonic() - 30)
        if per > 40:
            p5, sc5 = _adam_optimize(p_int, time.monotonic() + per, rng,
                                     lr=0.04, beta_init_frac=0.05,
                                     beta_decay=0.94, no_imp_limit=150)
            _update(p5, f'adam_n{n2}')

    # Phase 6: Try alternative initializations at n=1024 (150s)
    print(f"Phase 6: Alt inits, score={best_score:.8f}")
    phase6_end = t_start + 1500.0
    if time.monotonic() < phase6_end:
        n_alt = 1024
        alts = []
        # Cosine-shaped
        t = np.linspace(0, 1, n_alt)
        for amp in [0.3, 0.5, 0.7]:
            pa = 1.0 + amp * np.cos(np.pi * t)
            pa = np.maximum(pa, 1e-12)
            alts.append((f'cos_{amp}', pa / pa.sum()))
        # Quadratic
        for a1 in [1.5, 2.5, 3.5]:
            pa = 1.0 + a1 * t
            alts.append((f'lin_{a1}', pa / pa.sum()))
        pa = 1.0 + 2.0 * t + 1.0 * t**2
        alts.append(('quad2', pa / pa.sum()))
        pa = 1.0 + 3.0 * t - 1.5 * t**2
        alts.append(('quad_mix', np.maximum(pa, 1e-12) / np.maximum(pa, 1e-12).sum()))

        per_alt = max(15.0, (phase6_end - time.monotonic()) / len(alts))
        for label, pa in alts:
            if time.monotonic() > phase6_end - 15:
                break
            per = min(per_alt, phase6_end - time.monotonic() - 5)
            if per < 12:
                break
            p_alt, sc_alt = _adam_optimize(pa, time.monotonic() + per, rng,
                                           lr=0.05, beta_init_frac=0.06,
                                           beta_decay=0.93, no_imp_limit=100)
            _update(p_alt, f'alt_{label}')

    # Phase 7: Final intensive polish (remaining time)
    print(f"Phase 7: Final polish, score={best_score:.8f}")
    remaining = deadline - time.monotonic()
    if remaining > 30 and best_seq is not None:
        # Adam with very fine steps
        per1 = min(remaining * 0.4, 200.0)
        if per1 > 20:
            p7a, sc7a = _adam_optimize(best_seq, time.monotonic() + per1, rng,
                                       lr=0.02, beta_init_frac=0.03,
                                       beta_decay=0.97, no_imp_limit=80)
            _update(p7a, 'final_adam')

        # LP
        if time.monotonic() < deadline - 40 and best_seq.size <= 1536:
            per2 = min(80.0, deadline - time.monotonic() - 20)
            if per2 > 30:
                p7l, sc7l = _lp_refine(best_seq, time.monotonic() + per2,
                                       t_blend=0.25, max_solves=15, m_shrink=0.999)
                _update(p7l, 'final_lp')

        # Final Adam
        if time.monotonic() < deadline - 10:
            per3 = deadline - time.monotonic() - 10
            if per3 > 15:
                p7b, sc7b = _adam_optimize(best_seq, time.monotonic() + per3, rng,
                                           lr=0.01, beta_init_frac=0.02,
                                           beta_decay=0.98, no_imp_limit=50)
                _update(p7b, 'final_adam2')

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = np.ones(1024) / 1024

        exact_sc = _score(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    # Fallback
    fb = np.ones(1024) / 1024
    print("FALLBACK: uniform n=1024")
    return [float(x) for x in fb]
