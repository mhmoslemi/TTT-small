Optimized step function search using constraint-generation LP refinement,
asymmetric exponential ramp initialization, Adam polishing, and multi-scale
escalation. Exploits the key insight that optimal sequences are strongly

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _build_conv_matrix_sparse(a, constraint_indices):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for row_idx, k in enumerate(constraint_indices):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([row_idx] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(len(constraint_indices), n))
    return A


def _lp_solve(a, M, constraint_indices, x0=None):
    n = len(a)
    A = _build_conv_matrix_sparse(a, constraint_indices)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(len(constraint_indices))
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0), res
    return None, None

    best_z = z.copy()

        # Gradient via cross-correlation
        fw = np.fft.rfft(w, N_fft)
        fa_rev = np.fft.rfft(a[::-1], N_fft)
        h = np.fft.irfft(fw * fa_rev, N_fft)
        g = h[n - 1:2 * n - 1]

        # Project to tangent of simplex
        dot = float(np.dot(a, g))
        gz = a * (g - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(z.mean())
        e = np.exp(z)
        a = e / e.sum()

        c = _full_convolve(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-15:
                best_score = score
                best_a = a.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 500 == 0:
                    lr = max(0.005, lr * 0.9)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 50 == 0:
            beta = max(1e-6, beta * beta_decay)

        if no_imp > no_imp_limit:
            ns = restart_ns * (1.0 + 0.5 * float(no_imp // no_imp_limit))
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(z.mean())
            e = np.exp(z)
            a = e / e.sum()
            c = _full_convolve(a)
            maxc = float(c.max())
            beta = max(1e-4, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_a, best_score


def _dual_gap_check(a, M):
    """Compute the LP duality gap to assess convergence."""
    n = len(a)
    n_all = 2 * n - 1
    constraint_indices = list(range(n_all))
    b, res = _lp_solve(a, M, constraint_indices)
    if b is None or res is None:
        return None, None
    primal = float(res.fun)  # This is -sum(b) since c_obj = -1
    primal_obj = -primal
    # Get dual variables from HiGHS
    try:
        duals = res.ineqlin.marginals
        dual_obj = M * float(duals.sum())
        gap = dual_obj - primal_obj
        return primal_obj, gap
    except Exception:
        return primal_obj, None
