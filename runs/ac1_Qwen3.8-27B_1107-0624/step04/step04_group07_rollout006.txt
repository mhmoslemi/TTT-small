import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _score(a):
    n = len(a)
    c = _full_convolve(a)
    s = a.sum()
    return 2.0 * n * c.max() / (s * s)

def _smooth_max(c, beta):
    """Compute smooth max and gradient weights."""
    m = c.max()
    x = (c - m) / beta
    ex = np.exp(x)
    s = ex.sum()
    smax = m + beta * np.log(s)
    w = ex / s
    return smax, w

    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _full_convolve(a)
    return 2.0 * n * c.max() / (s * s)

def _smooth_max_and_weights(c, beta):
    m = c.max()
    x = (c - m) / beta
    ex = np.exp(x)
    s = ex.sum()
    smax = m + beta * np.log(s)
    w = ex / s
    return smax, w

def _grad_convol(w, a):
    """Gradient of sum_k w[k]*C[k] w.r.t. a, where C = full_conv(a)."""
    n = len(a)
    N = _next_pow2(3 * n)  # enough to avoid circular wrap-around
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    g1 = np.fft.irfft(W * np.conj(A), N)  # sum_k w[k]*a[k-i]
    g2 = np.fft.irfft(W * A, N)  # sum_k w[k]*a[i-k]
    return (g1 + g2)[:n]

def _grad_kurtosis(a):
    """Gradient of (1/N)*sum_m |fft(a,m)|^4 w.r.t. a."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.fft(a, N)
    P = np.abs(A) ** 2
    B = P * np.conj(A)
    g = (4.0 / N) * np.real(np.fft.ifft(B))[:n]
    return g

def _adam_logspace(a, deadline, rng, lr=0.01, beta_init=None, alpha=0.5,
                   beta_decay=0.995, no_imp_limit=500, kurt_weight=0.3):
    """Adam optimization in log-softmax space with combined objective."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    
    z = np.log(a)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    c = _full_convolve(a)
    maxc = c.max()
    if beta_init is not None:
        beta = beta_init
    else:
        beta = 0.05 * maxc
    
    # Compute initial score
    score = 2.0 * n * maxc
    
    best_z = z.copy()
    best_score = score
    no_imp = 0
    base_lr = lr
    
    while time.monotonic() < deadline:
        t += 1
        
        # Compute smooth max and weights
        c = _full_convolve(a)
        smax, w = _smooth_max_and_weights(c, beta)
        
        # Gradient of smooth max part
        g_max = _grad_convol(w, a)
        
        # Gradient of kurtosis part (normalized version)
        g_kurt = _grad_kurtosis(a)
        
        # Combine gradients
        # Normalize the kurtosis gradient to be comparable in scale to the max gradient
        g_max_norm = np.max(np.abs(g_max))
        g_kurt_norm = np.max(np.abs(g_kurt))
        if g_max_norm > 1e-15 and g_kurt_norm > 1e-15:
            g_kurt_scaled = g_kurt * (g_max_norm / g_kurt_norm)
        else:
            g_kurt_scaled = g_kurt
        
        g = alpha * g_max + (1 - alpha) * kurt_weight * g_kurt_scaled
        
        # Transform to log-space gradient (softmax chain rule)
        dot = np.dot(a, g)
        gz = a * (g - dot)
        gz -= gz.mean()  # redundancy for numerical stability
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t = 0
            continue
        
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()
        
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        c = _full_convolve(a)
        maxc = c.max()
        score = 2.0 * n * maxc
        
        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Adaptive beta
        if t % 50 == 0:
            beta = max(1e-6, beta * beta_decay)
            # Check gap
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.1:
                    beta = max(1e-6, beta * 0.5)
        
        # Learning rate decay
        if no_imp % 200 == 0:
            lr = max(0.001, lr * 0.95)
        
        # Restart if stuck
        if no_imp > no_imp_limit:
            ns = 0.1 + 0.05 * (no_imp // no_imp_limit)
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a)
            beta = 0.05 * c.max()
            m.fill(0)
            v.fill(0)
            t = 0
            lr = base_lr
            no_imp = 0
    
    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score


def _lp_primal(a, M, n):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0, sum(b) free."""
    # Build constraint matrix: (A_ub @ b) <= M
    # A_ub[k, j] = a[k-j] + a[j-k] (with bounds)
    # Actually: conv(a,b)[k] = sum_j a[j] * b[k-j]
    # So A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[kj])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_fixed_point(a, deadline, max_iters=30):
    """LP-based fixed point iteration."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a)
        M = c.max()
        
        b = _lp_primal(a, M, n)
        if b is None:
            break
        s_b = b.sum()
        if s_b < 1e-10:
            break
        b = b / s_b
        
        # Line search
        best_t = 0.5
        best_sc = _score(a)
        for t_c in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_t = (1.0 - t_c) * a + t_c * b
            a_t = np.maximum(a_t, 1e-15)
            a_t = a_t / a_t.sum()
            sc_t = _score(a_t)
            if sc_t < best_sc:
                best_sc = sc_t
                best_t = t_c
        
        a = (1.0 - best_t) * a + best_t * b
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score
