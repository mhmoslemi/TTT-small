    n = len(f)
    # Variables: g[0..n-1]
    # Objective: max sum(g) => min -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: conv(f, g)[k] <= M for k in 0..2n-2
    n_cons = 2 * n - 1
    A = np.zeros((n_cons, n))
    b = np.full(n_cons, M)
    
    for k in range(n_cons):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, k - j] = f[j]
    
    # Non-negativity: g >= 0 => bounds
    bounds = [(0, None)] * n
    
    from scipy.optimize import linprog
    result = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if result.success:
        g = -result.fun  # This is sum(g) since we minimized -sum(g)
        g_vec = result.x
        return g_vec
    return None

Optimization of 2n*max(conv(a,a))/sum(a)^2 using Lp-norm continuation
and fixed-point iteration. The key insight is that the gradient of the
Lp norm of the autocorrelation reduces to a cross-correlation, enabling
fast O(n log n) iterations.

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _lp_norm_obj_and_grad(a, p):
    """Compute Lp norm of conv(a,a) and its gradient.
    Returns (objective, gradient) where objective = (sum c^p)^(1/p) / (2n)
    and gradient is wrt a.
    """
    n = len(a)
    c = _full_convolve(a, a)
    c_pos = np.maximum(c, 0)  # c should be non-negative for non-negative a
    S = float(np.sum(c_pos ** p))
    if S < 1e-300:
        return 0.0, np.zeros(n)
    obj = S ** (1.0 / p)
    # Gradient: d[i] = conv(a[::-1], c^{p-1})[i+n-1]
    c_pw = c_pos ** (p - 1)
    a_rev = a[::-1]
    w = _full_convolve(a_rev, c_pw)
    d = w[n-1:2*n-1]
    grad = 2.0 * p * S ** (1.0/p - 1.0) * d
    return obj, grad

def _lp_continuation(a_init, p_list, deadline, max_iter_per_p=2000, lr=0.1):
    """Run Lp-norm continuation optimization."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    for p in p_list:
        if time.monotonic() > deadline:
            break
        
        obj_prev = np.inf
        for it in range(max_iter_per_p):
            if time.monotonic() > deadline:
                break
            
            obj, grad = _lp_norm_obj_and_grad(a, p)
            
            # Check convergence
            if abs(obj_prev - obj) < 1e-12 * max(1, obj):
                break
            obj_prev = obj
            
            # Projected gradient step (multiplicative)
            # a_new[i] = a[i] * (mu / grad[i])^lr where mu = weighted mean of grad
            g_pos = np.maximum(grad, 1e-30)
            mu = float(np.dot(a, grad)) / float(np.dot(a, np.ones(n)))  # weighted mean
            mu = max(mu, 1e-30)
            ratio = mu / g_pos
            a = a * ratio ** lr
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-30:
                a = np.ones(n) / n
            else:
                a = a / s
            
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score

def _lp_continuation_additive(a_init, p_list, deadline, max_iter_per_p=5000, lr=0.01):
    """Additive projected gradient for Lp norm."""
    n = len(a_init)

                best_a = a.copy()

            if it % 500 == 499:

        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _compute_d_normalized(a, c, p):
    """Compute d[i] = conv(a[::-1], (c/c_max)^(p-1))[i+n-1]."""
    n = len(a)
    c_max = float(c.max())
    if c_max < 1e-300:
        return np.ones(n)
    c_norm = c / c_max
    if p == 1:
        c_pw = c_norm
    else:
        c_pw = c_norm ** (p - 1)
    a_rev = a[::-1]
    w = _full_convolve(a_rev, c_pw)
    d = w[n - 1:2 * n - 1]
    return d


def _lp_iterate(a, p, lr, n_iter, deadline, verbose=False):
    """Run multiplicative fixed-point iterations for Lp norm."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    a = a.copy()

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        d = _compute_d_normalized(a, c, p)
        d_pos = np.maximum(d, 1e-30)
        mu = float(np.dot(a, d))
        mu = max(mu, 1e-30)
        ratio = mu / d_pos
        a = a * ratio ** lr
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-30:
            a = np.ones(n) / n
        else:
            a = a / s

        if it % 50 == 0 or it == n_iter - 1:
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                if verbose:
                    print(f"    p={p}, it={it}: score={sc:.10f}")

    return best_a, best_score


def _lp_continuation(a_init, p_list, deadline, lr_base=0.3, verbose=False):
    """Run Lp continuation over a list of p values."""
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for p in p_list:
        if time.monotonic() > deadline:
            break
        lr = lr_base / (1.0 + 0.1 * (p - 2))
        lr = max(lr, 0.05)
        n_iter = min(3000, max(200, int(50000 / max(1, p))))
        a, sc = _lp_iterate(a, p, lr, n_iter, deadline, verbose)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if verbose:
            print(f"  After p={p}: score={_score_norm_fn(a):.10f}, best={best_score:.10f}")

    return best_a, best_score


def _smoothed_max_obj(z, beta, n):
    """Objective: smoothed max of conv(softmax(z), softmax(z))."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shift = (c - cm) / beta
    lse = cm + beta * float(np.log(np.sum(np.exp(c_shift))))
    return 2.0 * n * lse


def _smoothed_max_grad(z, beta, n):
    """Gradient of smoothed max objective."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    w = np.exp((c - cm) / beta)
    w = w / w.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _run_lbfgs_smoothed(a_init, deadline, beta_list, maxiter_per_beta=5000):
    """L-BFGS with continuation on smoothed max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    for beta_frac in beta_list:
        if time.monotonic() > deadline:
            break
        beta = beta_frac * max_c0
        if beta < 1e-15:
            beta = 1e-15
        try:
            result = minimize(
                _smoothed_max_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_smoothed_max_grad,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-18,
                         'gtol': 1e-12, 'maxls': 50, 'maxfun': 20000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _peak_targeted_reduction(a, deadline, max_rounds=200, nrng_seed=42):
    """Directly target and reduce the highest peaks."""
    n = len(a)
    rng = np.random.default_rng(nrng_seed)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())
        k_star = int(np.argmax(c))

        # Find top contributors
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        valid_j = np.arange(j_min, j_max + 1)
        contribs = a_cur[valid_j] * a_cur[k_star - valid_j]
        top_k = 5
        top_idx = valid_j[np.argsort(contribs)[::-1][:top_k]]

        improved = False
        for idx in top_idx:
            for delta in [0.002, 0.005, 0.01, 0.02, 0.05]:
                z_trial = z.copy()
                z_trial[idx] -= delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try adding to low elements
            small_idx = np.argsort(a_cur)[:3]
            for idx in small_idx:
                for delta in [0.005, 0.01, 0.02]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Try a random kick
            kick_idx = rng.integers(0, n)
            for delta in [0.01, 0.02, 0.05]:
                z_trial = z.copy()
                z_trial[kick_idx] += delta * rng.choice([-1, 1])
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
            if not improved:
                break

    return best_a, best_score


def _make_asymmetric_profiles(n):
    """Generate diverse asymmetric initial profiles."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []

    # Linear ramps (asymmetric)
    for slope in [0.3, 0.5, 0.8, 1.0, 1.5, 2.0, -0.3, -0.5, -0.8, -1.0, -1.5, -2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Quadratic (asymmetric)
    for a1, a2 in [(0.5, 0.3), (-0.5, 0.3), (1.0, -0.2), (-1.0, -0.2),
                   (0.5, -0.5), (-0.5, -0.5), (1.5, 0.3), (-1.5, 0.3)]:
        a = 1.0 + a1 * u + a2 * u ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Exponential
    for rate in [0.5, 1.0, 1.5, 2.0, -0.5, -1.0, -1.5, -2.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())

    # Cosine power
    for p in [0.3, 0.5, 1.0, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())

    # Bump at edge
    for center in [0.2, 0.3, 0.7, 0.8]:
        for width in [0.1, 0.2, 0.3]:
            a = np.exp(-((t - center) / width) ** 2)
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())

    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _annealed_lp(a_init, deadline, rng, p_list=None, n_restarts=3):
    """Lp continuation with periodic annealing kicks."""
    if p_list is None:
        p_list = [4, 8, 16, 32, 64, 128]

    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for p in p_list:
        if time.monotonic() > deadline:
            break
        lr = 0.3 / (1.0 + 0.05 * p)
        lr = max(lr, 0.05)
        n_iter = min(2000, max(100, 30000 // max(1, p)))

        for it in range(n_iter):
            if time.monotonic() > deadline:
                break

            c = _full_convolve(a, a)
            c_max = float(c.max())
            c_norm = c / max(c_max, 1e-30)
            c_pw = c_norm ** (p - 1)
            a_rev = a[::-1]
            w = _full_convolve(a_rev, c_pw)
            d = w[n - 1:2 * n - 1]
            d_pos = np.maximum(d, 1e-30)
            mu = float(np.dot(a, d))
            mu = max(mu, 1e-30)
            ratio = mu / d_pos
            a = a * ratio ** lr
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-30:
                a = np.ones(n) / n
            else:
                a = a / s

            if it % 100 == 99:
                sc = _score_norm_fn(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

            # Annealing kick
            if it % 500 == 499:
                kick_scale = 0.02 * (1 + n_restarts)
                z = np.log(np.maximum(a, 1e-15))
                z = z - z.mean()
                z = z + kick_scale * rng.normal(size=n)
                z = np.clip(z, -60, 60)
                a = _softmax_fn(z)

    return best_a, best_score


def _fourier_polish(a_init, deadline, n_basis=8):
    """Polish using a polynomial basis parameterization."""
    n = len(a_init)
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    K = n_basis
    basis = np.ones((n, K + 1))
    for k in range(1, K + 1):
        basis[:, k] = u ** k

    # Initialize from current best
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    a0 = _softmax_fn(z0)
    # Fit: minimize ||log(a0) - basis @ alpha||^2
    log_a0 = np.log(np.maximum(a0, 1e-15))
    alpha0, _, _, _ = np.linalg.lstsq(basis, log_a0, rcond=None)

    best_score = _score_norm_fn(a_init)
    best_a = a_init.copy()

    def obj(alpha):
        log_a = basis @ alpha
        log_a = log_a - np.max(log_a)
        a = np.exp(log_a)
        a = a / a.sum()
        return _score_norm_fn(a)

    def grad(alpha):
        log_a = basis @ alpha
        log_a = log_a - np.max(log_a)
        a = np.exp(log_a)
        a = a / a.sum()
        c = _full_convolve(a, a)
        cm = float(c.max())
        k_star = int(np.argmax(c))
        # Subgradient of max
        w = np.zeros(len(c))
        w[k_star] = 1.0
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        dlog_a_da = h * 2 * n
        dlog_a_da = dlog_a_da - a * dlog_a_da.sum()
        grad = basis.T @ (a * dlog_a_da)
        return grad

    try:
        res = minimize(obj, alpha0, method='L-BFGS-B', jac=grad,
                       options={'maxiter': 5000, 'ftol': 1e-16, 'gtol': 1e-10})
        if res.fun < best_score:
            log_a = basis @ res.x
            log_a = log_a - np.max(log_a)
            a_new = np.exp(log_a)
            a_new = a_new / a_new.sum()
            best_score = float(res.fun)
            best_a = a_new.copy()
    except Exception:
        pass

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage optimization combining Lp-norm continuation (main method),
    smoothed-max L-BFGS polishing, peak-targeted reduction, and
    Fourier basis parameterization.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_parent}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 4096
        parent = _make_asymmetric_profiles(n_parent)[5]
        update_best(parent, _score_norm_fn(parent), "default")

    n = n_parent

    # --- Phase 1: Quick Lp continuation (60s) ---
    phase1_end = min(deadline, t_start + 60.0)
    print("Phase 1: Quick Lp continuation")
    p_list_1 = [2, 4, 8, 16]
    a_p1, sc_p1 = _lp_continuation(parent, p_list_1, phase1_end, lr_base=0.4)
    update_best(a_p1, sc_p1, "lp_p1")
    print(f"  Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Main Lp continuation (500s) ---
    phase2_end = min(deadline, t_start + 560.0)
    print("Phase 2: Main Lp continuation")
    p_list_2 = [16, 32, 64, 128, 256]
    a_p2, sc_p2 = _lp_continuation(best_seq, p_list_2, phase2_end, lr_base=0.3)
    update_best(a_p2, sc_p2, "lp_p2")

    # Also try annealed version
    if time.monotonic() < phase2_end - 60:
        per_dl = min(phase2_end - 10, time.monotonic() + 200.0)
        a_anneal, sc_anneal = _annealed_lp(best_seq, per_dl, rng,
                                           p_list=[8, 16, 32, 64], n_restarts=3)
        update_best(a_anneal, sc_anneal, "annealed_lp")
    print(f"  Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Smoothed max L-BFGS polishing (400s) ---
    phase3_end = min(deadline, t_start + 960.0)
    print("Phase 3: Smoothed max L-BFGS")
    beta_list_3 = [0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002,
                   0.001, 0.0005, 0.0003, 0.0002, 0.0001]
    per_dl = min(phase3_end - 10, time.monotonic() + 380.0)
    a_smooth, sc_smooth = _run_lbfgs_smoothed(best_seq, per_dl, beta_list_3, maxiter_per_beta=8000)
    update_best(a_smooth, sc_smooth, "lbfgs_smoothed")

    # Second round with perturbation
    if time.monotonic() < phase3_end - 60:
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        z = z + 0.03 * rng.normal(size=n)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a_pert = _softmax_fn(z)
        per_dl2 = min(phase3_end - 10, time.monotonic() + 180.0)
        a_smooth2, sc_smooth2 = _run_lbfgs_smoothed(a_pert, per_dl2, beta_list_3[-6:], maxiter_per_beta=5000)
        update_best(a_smooth2, sc_smooth2, "lbfgs_perturbed")
    print(f"  Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Peak targeted reduction (150s) ---
    phase4_end = min(deadline, t_start + 1110.0)
    print("Phase 4: Peak targeted reduction")
    if time.monotonic() < phase4_end:
        per_dl = min(phase4_end - 5, time.monotonic() + 140.0)
        a_pk, sc_pk = _peak_targeted_reduction(best_seq, per_dl, max_rounds=500, nrng_seed=123)
        update_best(a_pk, sc_pk, "peak_target")
    print(f"  Phase 4 done: best={best_score:.10f}")

        profiles = _make_asymmetric_profiles(n)

            if time.monotonic() < phase6_end - 15:
                per_dl2 = min(phase6_end - 10, time.monotonic() + per_profile_time * 0.4)
                a_q2, sc_q2 = _run_lbfgs_smoothed(a_q, per_dl2, [0.01, 0.005, 0.001], maxiter_per_beta=2000)
                update_best(a_q2, sc_q2, f"multi_{idx}_smooth")
    print(f"  Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final fine polish (remaining time) ---
    phase7_end = deadline
    print("Phase 7: Final fine polish")
    if time.monotonic() < phase7_end and best_seq is not None:

            update_best(a_fin, sc_fin, "final_lp_high_p")

                                                   [0.0005, 0.0003, 0.0002, 0.0001, 0.00005],
                                                   maxiter_per_beta=10000)
            update_best(a_fin2, sc_fin2, "final_lbfgs_fine")

        # Peak surgery on final

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                else:
                    # Fallback to a simpler sequence
                    print("Evaluator returned inf, using fallback")
                    final_seq = _make_asymmetric_profiles(512)[5]
                    sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                    print(f"Fallback evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")
                final_seq = _make_asymmetric_profiles(512)[5]

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_asymmetric_profiles(512)[5]
    print(f"FALLBACK: n=512, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
