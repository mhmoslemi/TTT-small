def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a):

    n = a.size
    if n == 0:

    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)  # numerical roundoff
    return c

    c = _full_convolve(a)

def _cross_corr(w, p):
    """Compute cross-correlation of w (length 2n-1) and p (length n).
    Returns vector of length n: result[i] = sum_k w[k] * p[k-i] (valid range)."""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    hat = cssv[rho[-1]] / (rho[-1] + 1.0)
    w = np.maximum(v - hat, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

    """Build the n x n lower-triangular Toeplitz matrix for half-convolution.
    Row k: a[k], a[k-1], ..., a[0], 0, ..., 0 (for k = 0, ..., n-1)
    So A[k,i] = a[k-i] if i <= k, else 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(n):
        for i in range(k + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = csr_matrix((vals, (rows, cols)), shape=(n, n))
    return A

    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    hat = cssv[rho[-1]] / (rho[-1] + 1.0)
    w = np.maximum(v - hat, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _subgradient_polish(a, deadline, lr_init=0.005, max_iters=10000):
    n = len(a)

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()

    return p / p.sum()

                if sc_try < best_sc - 1e-14:
                    best_sc = sc_try
                    best_t = t
            if best_t > 0:
                a = (1.0 - best_t) * a + best_t * b
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_score = best_sc
                stall = 0
                print(f"    LP mix it={it} t={best_t:.2f} score={best_score:.10f}")
            else:
                stall += 1
                if stall >= 6:
                    break
    return best_a, best_score


def _smoother_plus_lp(a, deadline):
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    stall = 0
    it = 0
    while time.monotonic() < deadline:
        it += 1
        c = _full_convolve(a)
        M = float(c.max())
        tol = max(1e-12, M * 1e-7)
        active = c >= M - tol
        n_act = max(1, int(active.sum()))
        w = active.astype(np.float64) / n_act
        grad = 2.0 * _cross_corr(w, a)
        grad -= grad.mean()
        gnorm = float(np.sqrt(np.dot(grad, grad)))
        if gnorm < 1e-15:
            break
        grad_normed = grad / gnorm
        best_t = 0.0
        best_sc = best_score
        for t in [0.01, 0.02, 0.04, 0.06, 0.08, 0.1, 0.12, 0.15, 0.2]:
            a_try = a - t * grad_normed
            a_try = _project_simplex(a_try)
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_t = t
        if best_t > 0:
            a = a - best_t * grad_normed
            a = _project_simplex(a)
            best_a = a.copy()
            best_score = best_sc
            stall = 0
            if it % 10 == 0:
                print(f"    smoother it={it} score={best_score:.10f}")
        else:
            stall += 1
            b = _lp_best_response(a, M, max_time=90.0)
            if b is None:
                break
            s_b = float(b.sum())
            if s_b < 1e-10:
                break
            b = b / s_b
            sc_b = _score(b)
            if sc_b < best_score - 1e-12:
                a = b.copy()
                best_a = b.copy()
                best_score = sc_b
                stall = 0
                print(f"    smoother->LP it={it} score={best_score:.10f}")
            else:
                best_t2 = 0.0
                best_sc2 = best_score
                for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                    a_try = (1.0 - t) * a + t * b
                    a_try = np.maximum(a_try, 1e-15)
                    a_try = a_try / a_try.sum()
                    sc_try = _score(a_try)
                    if sc_try < best_sc2 - 1e-14:
                        best_sc2 = sc_try
                        best_t2 = t
                if best_t2 > 0:
                    a = (1.0 - best_t2) * a + best_t2 * b
                    a = np.maximum(a, 1e-15)
                    a = a / a.sum()
                    best_a = a.copy()
                    best_score = best_sc2
                    stall = 0
                    print(f"    smoother mix it={it} t={best_t2:.2f} score={best_score:.10f}")
                else:
                    if stall >= 8:
                        break
    return best_a, best_score
