"""
Optimization of 2*n*max(conv(a,a))/sum(a)^2 using:
- Diverse asymmetric profile initialization
- Mirror descent (natural gradient) with exact subgradient
- LP fixed-point polishing
- Multi-resolution escalation
- Random restarts

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c  # Don't clip to 0, keep exact

def _score(a):
    n = len(a)
    c = _full_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))

def _subgradient(a, k_star):
    """Compute subgradient of max_k conv(a,a)[k] at lag k_star.
    g[j] = 2*a[k_star - j] if 0 <= k_star - j < n, else 0"""
    n = len(a)
    g = np.zeros(n)
    valid = (k_star - np.arange(n) >= 0) & (k_star - np.arange(n) < n)
    idx = np.arange(n)[valid]
    g[idx] = 2.0 * a[k_star - idx]
    return g

def _mirror_descent(a, deadline, eta=0.1, momentum=0.9, polyak=True):
    """Mirror descent on the simplex with exact subgradient.
    Works in log-space (entropic mirror)."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    
    z = np.log(a)
    z -= z.mean()
    
    best_a = a.copy()
    best_score = _score(a)
    
    v = np.zeros(n)  # momentum
    t = 0
    f_star = 2.0 * n / (2 * n - 1)  # lower bound
    
    polyak_a = a.copy()
    polyak_score = _score(a)
    
    no_imp = 0
    
    while time.monotonic() < deadline:
        c = _full_convolve(a)
        k_star = int(np.argmax(c))
        M = c[k_star]
        
        g = _subgradient(a, k_star)
        
        # Momentum
        v = momentum * v + (1 - momentum) * g
        
        # Polyak step size
        current_score = 2.0 * n * M  # since sum(a) = 1
        if polyak and current_score > f_star:
            eta_t = (current_score - f_star) / (np.dot(v, v) + 1e-10)
            eta_t = min(eta_t, eta)  # cap
        else:
            eta_t = eta / (1 + 0.1 * t)
        
        # Update in log-space
        z = z - eta_t * v
        z = np.clip(z, -50, 50)
        
        # Convert back to simplex
        a_new = np.exp(z - z.max())
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        # Check improvement
        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Polyak averaging
        if polyak:
            polyak_a = (t * polyak_a + a_new) / (t + 1)
            polyak_sc = _score(polyak_a)
            if polyak_sc < polyak_score:
                polyak_score = polyak_sc
        
        a = a_new
        t += 1
        
        if no_imp > 500:
            break
    
    # Return better of direct and polyak
    if polyak_score < best_score:
        return polyak_a, polyak_score
    return best_a, best_score

def _adam_smooth_max(a, deadline, rng, lr=0.05, beta_init=None, beta_decay=0.95):
    """Adam optimization on smooth-max approximation in log-space."""
    n = len(a)
    a = np.maximum(a, 1e-12).copy()
    a = a / a.sum()
    
    z = np.log(a)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    c = _full_convolve(a)
    maxc = float(c.max())
    beta = beta_init if beta_init is not None else 0.05 * maxc
    if beta < 1e-6:
        beta = 0.01
    
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    
    while time.monotonic() < deadline:
        # Smooth max weights
        x = (c - maxc) / beta
        ex = np.exp(np.clip(x, -50, 50))
        s = ex.sum()
        if s < 1e-30:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s
        
        # Gradient: cross-correlation of w with a
        # d/d a_j sum_k w[k] * C[k] where C = conv(a,a)
        # = sum_k w[k] * dC[k]/da[j]
        # = sum_k w[k] * 2*a[k-j] (if valid)
        # This is a weighted sum of shifted copies of a
        # = 2 * (w * a_reversed) cross-correlation
        # More efficiently: FFT
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        g = np.fft.irfft(fw * fa_rev, N)[n-1:2*n-1] * 2.0
        
        # Project to tangent space (sum(constraint))
        dot = np.dot(a, g)
        gz = a * (g - dot)
        
        # Adam update
        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()
        
        a = np.exp(z - z.max())
        a = a / a.sum()
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        c = _full_convolve(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.005, lr * 0.95)
        
        # Decay beta
        if t_adam % 30 == 0:
            beta = max(1e-4, beta * beta_decay)
        
        # Restart if stuck
        if no_imp > 300:
            ns = 0.1
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a)
            maxc = float(c.max())
            beta = 0.05 * maxc
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = 0.05
            no_imp = 0
    
    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score

def _build_conv_matrix(n, a):
    """Build sparse matrix A such that (A @ b)[k] = (a * b)[k] for all k."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def _lp_fixed_point(a, deadline, max_iters=30):
    """LP fixed-point: find b that maximizes sum(b) subject to conv(a,b) <= M, b >= 0."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    
    c = _full_convolve(a)
    M = float(c.max())
    
    A = _build_conv_matrix(n, a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return a, _score(a)
    
    b = np.maximum(res.x, 0.0)
    s_b = b.sum()
    if s_b < 1e-10:
        return a, _score(a)
    b = b / s_b
    
    best_a = a.copy()
    best_score = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c_a = _full_convolve(a)
        M = float(c_a.max())
        
        A = _build_conv_matrix(n, a)
        b_ub = M * np.ones(2 * n - 1)
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            break
        
        b = np.maximum(res.x, 0.0)
        s_b = b.sum()
        if s_b < 1e-10:
            break
        b = b / s_b
        
        # Line search
        best_t = 0.5
        best_sc = _score(a)
        for t_c in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_t = (1 - t_c) * a + t_c * b
            a_t = np.maximum(a_t, 1e-15)
            a_t = a_t / a_t.sum()
            sc_t = _score(a_t)
            if sc_t < best_sc:
                best_sc = sc_t
                best_t = t_c
        
        a = (1 - best_t) * a + best_t * b
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score

def _make_profile(n, ptype, p1=1.0, p2=0.0, p3=0.0):
    if ptype == 'linear':
        t = np.linspace(0, 1, n)
        a = 0.1 + p1 * (1 - t)  # decreasing
    elif ptype == 'linear_inc':
        t = np.linspace(0, 1, n)
        a = 0.1 + p1 * t  # increasing
    elif ptype == 'exp_decay':
        t = np.arange(n)
        a = np.exp(-p1 * t / n)
    elif ptype == 'exp_decay_fast':
        t = np.arange(n)
        a = np.exp(-p1 * t / (n * 0.3))
    elif ptype == 'power':
        t = np.linspace(0.01, 1, n)
        a = t ** p1
    elif ptype == 'power_dec':
        t = np.linspace(0.01, 1, n)
        a = (1 - t) ** p1
    elif ptype == 'asym_gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - p1) / p2) ** 2)
    elif ptype == 'bimodal':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - p1) / 0.1) ** 2) + p2 * np.exp(-0.5 * ((t - p3) / 0.1) ** 2)
    elif ptype == 'ramp_plateau':
        t = np.linspace(0, 1, n)
        a = np.concatenate([
            np.linspace(1, 0.1, int(n * 0.7)),
            np.full(n - int(n * 0.7), 0.1)
        ])
    elif ptype == 'triangle':
        t = np.linspace(0, 1, n)
        a = 1 - np.abs(2 * t - p1)  # peak at p1
    elif ptype == 'log_linear':
        t = np.linspace(0, 1, n)
        a = np.exp(p1 * (1 - t) * np.log(n))
    elif ptype == 'cos_asym':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * np.cos(np.pi * t)  # goes from 1+p1 to 1-p1
    elif ptype == 'sigmoid':
        t = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + np.exp(-p1 * (t - 0.5)))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new

def _interp_log(a, n_new):
    """Interpolate in log-space for better preservation of structure."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    log_a = np.log(np.maximum(a, 1e-30))
    a_new = np.exp(np.interp(x1, x0, log_a))
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new
