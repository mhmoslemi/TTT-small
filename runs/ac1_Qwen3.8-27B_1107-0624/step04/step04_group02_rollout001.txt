    fb = rfft(b, N)

"""
LP-based optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses constraint generation LP (dual/Golenberg-Welch style) as the main driver,
combined with diverse initialization, multi-scale search, and local polishing.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assume sum(a) = 1, returns 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind, param1=1.0, param2=0.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if kind == 'linear':
        a = 1.0 + param1 * u
    elif kind == 'quadratic':
        a = 1.0 + param1 * u + param2 * u * u
    elif kind == 'gaussian':
        a = np.exp(-u * u / (2.0 * param1 ** 2))
    elif kind == 'raised_cosine':
        a = 0.5 * (1.0 + np.cos(np.pi * u * param1))
    elif kind == 'power':
        a = (1.0 - np.abs(u)) ** param1
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _convolve_with_f(f, g):
    """Compute conv(f, g) efficiently."""
    return _full_convolve(f, g)


def _solve_lp_with_constraints(f, n, M, constraint_indices):
    """
    Solve max sum(g) s.t. conv(f,g)[k] <= M for k in constraint_indices, g >= 0.
    Returns the optimal g vector.
    """
    K = len(constraint_indices)
    c_obj = -np.ones(n)
    
    # Build constraint matrix
    A = np.zeros((K, n))
    for i, k in enumerate(constraint_indices):
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[i, j_min:j_max+1] = f[k - j_max : k - j_min + 1][::-1]
    
    b_ub = np.full(K, M)
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub,
                     bounds=[(0, None)] * n, method='highs',
                     options={'presolve': True})
    if result.success:
        return result.x
    return None


def _lp_iteration(f, n, M, max_cg_iters=20, initial_constraints=None):
    """
    One LP iteration with constraint generation.
    Returns the optimal g (before rescaling).
    """
    # Start with initial constraints (top K of conv(f,f))
    c_ff = _full_convolve(f, f)
    
    if initial_constraints is None:
        K_init = min(50, 2 * n - 1)
        constraint_indices = np.argsort(c_ff)[-K_init:]
    else:
        constraint_indices = initial_constraints.copy()
    
    g = None
    for cg_iter in range(max_cg_iters):
        g = _solve_lp_with_constraints(f, n, M, constraint_indices)
        if g is None:
            return None
        
        # Check violation of all constraints
        c_fg = _full_convolve(f, g)
        violations = c_fg - M
        max_viol_idx = int(np.argmax(violations))
        max_viol = float(violations[max_viol_idx])
        
        if max_viol < 1e-8 * M:
            return g  # Feasible for all constraints
        
        # Add the most violated constraint
        if max_viol_idx not in constraint_indices:
            constraint_indices = np.append(constraint_indices, max_viol_idx)
        else:
            # Shouldn't happen if LP is solved correctly, but just in case
            break
    
    return g


def _lp_optimize(f_init, deadline, rng, max_iterations=5000):
    """
    Main LP-based optimization loop.
    f_init: initial sequence (will be normalized to sum=1)
    Returns best sequence found.
    """
    n = len(f_init)
    f = f_init.copy()
    f = np.maximum(f, 0.0)
    f = f / f.sum()
    
    best_f = f.copy()
    best_score = _score_norm(f)
    
    t_step = 0.3  # relaxation parameter
    constraint_indices = None  # will be initialized
    
    for it in range(max_iterations):
        if time.monotonic() > deadline - 2.0:
            break
        
        M = float(_full_convolve(f, f).max())
        
        # Solve LP
        g = _lp_iteration(f, n, M, max_cg_iters=20, initial_constraints=constraint_indices)
        if g is None:
            t_step *= 0.5
            continue
        
        # Rescale g to have same sum as f
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            t_step *= 0.5
            continue
        g = g / g_sum  # now sum(g) = 1
        
        # Update
        f_new = (1.0 - t_step) * f + t_step * g
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        
        sc_new = _score_norm(f_new)
        
        if sc_new < best_score:
            best_score = sc_new
            best_f = f_new.copy()
            f = f_new.copy()
            # Update constraint indices for warm start
            # (keep them, they should still be relevant)
        else:
            # Objective increased, reduce step size
            t_step *= 0.7
            if t_step < 0.01:
                t_step = 0.01
            f = f_new.copy()  # Still update to explore
        
        # Periodically reset step size if making progress
        if it % 100 == 0 and sc_new < best_score * 1.001:
            t_step = min(0.3, t_step * 1.2)
    
    return best_f, best_score


def _local_search(a_init, deadline, rng, n_trials=5000):
    """Local search with block perturbations."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break
        
        block_size = int(rng.integers(3, min(30, n // 4)))
        start = int(rng.integers(0, n - block_size))
        
        a_trial = a.copy()
        perturbation = rng.uniform(0.8, 1.2, size=block_size)
        a_trial[start:start + block_size] *= perturbation
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial = a_trial / a_trial.sum()
        
        sc = _score_norm(a_trial)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
        elif rng.random() < 0.02 and sc < best_score * 1.005:
            a = a_trial.copy()
    
    return best_a, best_score


def _lbfgs_polish(a_init, deadline, beta_starts=None):

    from scipy.optimize import minimize
    
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    if beta_starts is None:
        c0 = _full_convolve(a_init, a_init)
        M0 = float(c0.max())
        beta_starts = [b * M0 for b in [0.01, 0.005, 0.002, 0.001, 0.0005]]
    
    per_level = max(5.0, (deadline - time.monotonic()) / len(beta_starts))
    
    for beta in beta_starts:
        if time.monotonic() > deadline - 2.0:
            break
        level_dl = min(deadline - 1.0, time.monotonic() + per_level)
        
        def obj_fn(z_):
            a_ = _softmax(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_sh = (c_ - cm_) / beta
            e_ = np.exp(c_sh)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            return 2.0 * n * (cm_ + beta * np.log(s_))
        
        def grad_fn(z_):
            a_ = _softmax(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_sh = (c_ - cm_) / beta
            e_ = np.exp(c_sh)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            W = float(np.dot(w, c_))
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n-1:2*n-1]
            g = 4.0 * n * a_ * (h - W)

            if result.x is not None and np.all(np.isfinite(result.x)):

                if np.isfinite(sc) and sc < best_score:
