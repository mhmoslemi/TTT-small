```python
"""
L-BFGS based optimization with smooth max continuation for minimizing
2*n*max(convolve(a,a))/sum(a)^2. Uses softmax parameterization,
multi-phase beta continuation, and diverse restarts.
"""
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as _sp_minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    na, nb = len(a), len(b)
    out_n = na + nb - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:out_n]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assume sum(a)=1, return 2n*max(c)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _smooth_max(c, beta):
    """Compute smooth max value and weights."""
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        val = cm
    else:
        w = e / s
        val = cm + beta * float(np.log(s))
    return val, w


def _grad_z(a, c, w, n):
    """Gradient of 2n*smooth_max(conv(a,a)) w.r.t. softmax params z."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbgfs_phase(a_init, beta, max_iter=300):
    """Run one L-BFGS-B phase with given beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    class _FunJac:
        def __init__(self, n, beta):
            self.n = n
            self.beta = beta
            self._cache = None

        def fun(self, z):
            z = z - z.mean()
            a = _softmax(z)
            c = _full_convolve(a, a)
            val, w = _smooth_max(c, self.beta)
            self._cache = (a, c, w)
            return 2.0 * self.n * val

        def jac(self, z):
            if self._cache is not None:
                # Use cached values from fun call
                a, c, w = self._cache
                self._cache = None
            else:
                z = z - z.mean()
                a = _softmax(z)
                c = _full_convolve(a, a)
                _, w = _smooth_max(c, self.beta)
            return _grad_z(a, c, w, self.n)

    fj = _FunJac(n, beta)
    
    result = _sp_minimize(
        fj.fun, z, method='L-BFGS-B',
        jac=fj.jac,
        options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8, 'maxfun': max_iter * 2}
    )
    
    a_final = _softmax(result.x)
    a_final = np.maximum(a_final, 1e-15)
    a_final = a_final / a_final.sum()
    return a_final, _score_norm(a_final)
```

```python
"""
L-BFGS + Nesterov momentum optimization with smooth max continuation for
minimizing 2*n*max(convolve(a,a))/sum(a)^2. Uses softmax parameterization,
multi-phase beta continuation, diverse restarts, and symmetrization heuristics.
"""
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as _sp_minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    na, nb = len(a), len(b)
    out_n = na + nb - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:out_n]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _smooth_max(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        val = cm
    else:
        w = e / s
        val = cm + beta * float(np.log(s))
    return val, w


def _grad_z(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbgfs_phase(a_init, beta, max_iter=300, deadline=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    def fun(z_in):
        if deadline is not None and time.monotonic() > deadline:
            return 2.0 * n * float(_full_convolve(_softmax(z_in), _softmax(z_in)).max())
        z_c = z_in - z_in.mean()
        a = _softmax(z_c)
        c = _full_convolve(a, a)
        val, _ = _smooth_max(c, beta)
        return 2.0 * n * val

    def jac(z_in):
        if deadline is not None and time.monotonic() > deadline:
            return np.zeros(n)
        z_c = z_in - z_in.mean()
        a = _softmax(z_c)
        c = _full_convolve(a, a)
        _, w = _smooth_max(c, beta)
        return _grad_z(a, c, w, n)

    try:
        result = _sp_minimize(
            fun, z.copy(), method='L-BFGS-B',
            jac=jac,
            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': max_iter * 3, 'maxls': 50}
        )
        a_final = _softmax(result.x)
    except Exception:
        a_final = _softmax(z)

    a_final = np.maximum(a_final, 1e-15)
    a_final = a_final / a_final.sum()
    return a_final, _score_norm(a_final)


def _nesterov_phase(a_init, beta, deadline, lr=0.02, max_iter=50000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    z_prev = z.copy()
    best_a = a_init.copy()

    mu = 0.0

        if time.monotonic() > deadline:
            break

        a = _softmax(y_c)
        c = _full_convolve(a, a)
        _, w = _smooth_max(c, beta)
```
