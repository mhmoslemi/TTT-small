```python
"""
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find the best direction at each step, combined
with gradient-based polishing.
"""
import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_unnormalized(a):
    """Score without the sum normalization (assumes sum=1)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _build_conv_matrix(f, n):
    """Build sparse convolution matrix A where A[k,j] = f[k-j] if valid."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(f[idx])
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def _solve_lp(f, M, n):
    """Solve LP: max sum(b) s.t. conv_matrix @ b <= M, b >= 0."""
    A = _build_conv_matrix(f, n)
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    A_ub = A
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None

def _lp_iterate(f, n, max_iters, deadline):
    """Iterative LP-based optimization."""
    f = f / f.sum()  # Normalize to sum = 1
    M = _full_convolve(f, f).max()
    best_f = f.copy()
    best_score = _score_unnormalized(f)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        M = _full_convolve(f, f).max()
        b = _solve_lp(f, M, n)
        if b is None:
            break
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum  # Rescale
        
        # Try t = 1 first
        f_new = b
        score_new = _score_unnormalized(f_new)
        
        if score_new < best_score:
            best_f = f_new.copy()
            best_score = score_new
        
        # If t=1 didn't improve, try intermediate t
        if score_new >= M:  # In units of sum=1
            # Try bisection
            t_lo, t_hi = 0.0, 1.0
            f_curr = f
            for _ in range(10):
                t_mid = (t_lo + t_hi) / 2
                f_test = (1 - t_mid) * f + t_mid * b
                sc_test = _full_convolve(f_test, f_test).max()
                if sc_test <= M:
                    t_lo = t_mid
                else:
                    t_hi = t_mid
            t = t_lo
            f = (1 - t) * f + t * b
        else:
            f = f_new
        
        if it % 10 == 0:
            print(f"  LP iter {it}: M={M:.6f}, best_score={best_score:.6f}, t used")
    
    return best_f, best_score
```
