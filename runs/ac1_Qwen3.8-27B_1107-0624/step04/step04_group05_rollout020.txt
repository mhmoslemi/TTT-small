"""
Optimized step function search using Adam in logit space, multiplicative weights
(mirror descent), LP best-response, and coordinate descent. Targets score < 1.5027
by combining multiple optimization strategies with adaptive scheduling.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 512:
        c = np.convolve(a, a)
    else:
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    mval = float(c.max())
    if beta < 1e-13:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-9
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 0.0))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def _cross_corr_w_a(w, a):
    """Compute h[i] = sum_m w[m+i] * a[m] for i = 0, ..., len(a)-1.
    Uses FFT for efficiency."""
    n = len(a)
    N = _next_pow2(len(w) + n - 1)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    h = np.fft.irfft(W * np.conj(A), N)[:n]
    return h


def _grad_smooth_max(a, w):
    """Gradient of weighted sum of convolution w.r.t. a.
    g[i] = 2 * sum_k w[k] * a[k-i] = 2 * cross_corr(w, a)[i]"""
    n = len(a)
    h = _cross_corr_w_a(w, a)
    return 2.0 * h


def _build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

    if res.success:
        return np.maximum(res.x, 0.0)
    return None
