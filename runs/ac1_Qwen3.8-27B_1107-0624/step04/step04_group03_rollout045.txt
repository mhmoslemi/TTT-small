"""
Optimized step function search using multi-phase gradient methods with
log-sum-exp smoothing, Nesterov acceleration, and structural restarts.
Key improvements over previous version: fixed bugs, better beta schedule,
structural restarts (mirror, harmonic), multi-length exploration, and
randomized smoothing for escape from local minima.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    """Assumes sum(a) = 1, returns 2*n*max(conv)"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _gradient_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _objective_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_obj(z, beta, n):
    return _objective_smooth(z, beta)


def _lbfgs_grad(z, beta, n):
    return _gradient_smooth(z, beta)


def _run_lbfgs(a_init, deadline, max_wall_time=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if max_wall_time is None:
        max_wall_time = min(300.0, max(1.0, deadline - time.monotonic()))

    # More aggressive beta schedule
    betas = [0.1, 0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005,
             0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6]
    betas = [b * max_c0 for b in betas]

    per_level = max_wall_time / len(betas)

    for beta in betas:
        if time.monotonic() >= deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level)
        try:
            result = minimize(
                _lbfgs_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_grad,
                options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-12,

                a_new = _softmax_fn(z_new)

                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new
        except Exception:

def _nesterov_optimize(a_init, deadline, beta_fracs, lr_init=0.02, mom_init=0.9):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())

    lr = lr_init
    mom = mom_init
    t = 0

    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-15, beta_frac * max_c)

        step_count_in_level = 0
        while time.monotonic() < deadline:
            t += 1
            step_count_in_level += 1

            grad_y = _gradient_smooth(y, beta)
            if not np.all(np.isfinite(grad_y)):
                grad_y = _gradient_smooth(z, beta)

            z_new = z - lr * grad_y
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -60, 60)

            y_new = z_new + mom * (z_new - z)
            y_new = y_new - y_new.mean()

            z = z_new
            y = y_new

            # Evaluate periodically
            if t % 30 == 0:
                a_curr = _softmax_fn(z)
                sc = _score_norm_fn(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()

            # Adapt momentum

                mom = min(0.999, mom + 0.0003)

        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        max_c = float(c.max())
        lr = max(0.001, lr * 0.6)

    # Final evaluation
    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

            beta = max(1e-15, beta * 0.9)
            c_new = _full_convolve(a, a)

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_sine_profile(n, freq=1.0, power=2.0):
    """Create a sine-based profile."""
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 + 0.5 * np.sin(2.0 * np.pi * freq * t + np.pi / 2)) ** power
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_harmonic_profile(n, n_harmonics=3, amplitudes=None):
    """Create a profile with multiple harmonic components."""
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    if amplitudes is None:
        amplitudes = [0.3, 0.15, 0.08]
    for i in range(min(n_harmonics, len(amplitudes))):
        a = a + amplitudes[i] * np.sin(2.0 * np.pi * (i + 1) * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_arcosh_profile(n, center=0.5, width=0.3):
    """Create an arcosh-like bump profile."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - center) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _flatten_convolution(a_init, deadline, n_iter=5000):
    """
    Directly try to flatten the convolution by iteratively reducing peaks.
    Uses a multiplicative update rule inspired by ISL (iterative scaling).
    """
    n = len(a_init)
    a = a_init.copy() / a_init.sum()

    best_a = a.copy()
    best_score = _score_norm_fn(a)

    lr = 0.01
    patience = 0
    max_patience = 500

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        c_avg = float(c.mean())

        # Find peaks (lags where c is above average by a threshold)
        threshold = c_avg * 1.05  # 5% above average
        peaks = c > threshold

        if not np.any(peaks):
            # If no clear peaks, use top few lags
            top_k = min(100, len(c) // 10)
            peak_indices = np.argsort(c)[-top_k:]
            peaks = np.zeros_like(c, dtype=bool)
            peaks[peak_indices] = True

        # For each peak lag k, the contribution to c[k] is sum_i a[i]*a[i+k-n+1]
        # To reduce c[k], we should reduce a at positions that contribute most
        # to the peak lags.

        # Compute the "pressure" on each position: sum over peak lags of
        # the contribution of position i to that lag.
        a_rev = a[::-1]

        # Weight peaks by how much they exceed the threshold
        w = np.zeros_like(c)
        w[peaks] = (c[peaks] - threshold) / (c[peaks].sum() - threshold * peaks.sum() + 1e-30)

        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]

        # Multiplicative update: reduce a where pressure is high
        # a_i *= exp(-lr * h_i / max(h))
        h_norm = h / (h.max() + 1e-30)
        a = a * np.exp(-lr * h_norm)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        if it % 200 == 0:
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                patience = 0
                lr = min(0.1, lr * 1.05)  # slightly increase lr
            else:
                patience += 1
                lr = max(0.001, lr * 0.95)
                if patience > max_patience:
                    break

    a_final = a / a.sum()
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _proportional_to_scores(a_init, deadline, n_iter=3000):
    """
    Alternative: proportional-to-scores update.
    At each step, compute c = conv(a,a), then update a_i proportional to
    how much position i contributes to reducing the max.
    """
    n = len(a_init)
    a = a_init.copy() / a_init.sum()

    best_a = a.copy()
    best_score = _score_norm_fn(a)

    alpha = 0.5  # Interpolation parameter
    lr = 0.01

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        # "Ideal" a would have c as flat as possible
        # c_flat = sum(a)^2 / (2n-1) = 1/(2n-1) (since sum(a)=1)
        # We want to move a toward a distribution that would produce flat c

        # The "deficit" at each position: how much should a_i increase/decrease?
        # If position i is at a "peak contributing" location, decrease it
        # If position i is at a "valley contributing" location, increase it

        # Compute contribution of each position to the max lag
        max_lags = np.where(c >= cm * 0.999)[0]
        if len(max_lags) == 0:
            max_lags = np.array([np.argmax(c)])

        a_rev = a[::-1]
        w = np.zeros_like(c)
        w[max_lags] = 1.0 / len(max_lags)
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]

        # Update: move a away from high-contribution positions
        # Use a normalization that preserves sum
        a_new = a * np.exp(-lr * (h - h.mean()))
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        # Interpolate
        a = (1 - alpha) * a + alpha * a_new

        if it % 200 == 0:
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                lr = min(0.1, lr * 1.02)

    return best_a, best_score


def _randomized_nesterov(a_init, deadline, beta_fracs, lr_init=0.015,
                          mom_init=0.9, noise_scale=0.01):
    """Nesterov with random perturbations to escape local minima."""
    n = len(a_init)
    rng = np.random.default_rng(42)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())

    lr = lr_init
    mom = mom_init
    t = 0

    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-15, beta_frac * max_c)

        while time.monotonic() < deadline:
            t += 1

            grad_y = _gradient_smooth(y, beta)
            if not np.all(np.isfinite(grad_y)):
                grad_y = _gradient_smooth(z, beta)

            # Add small random noise
            noise = noise_scale * rng.standard_normal(n)
            noise = noise - noise.mean()

            z_new = z - lr * (grad_y + noise)
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -60, 60)

            y_new = z_new + mom * (z_new - z)
            y_new = y_new - y_new.mean()

            z = z_new
            y = y_new

            if t % 50 == 0:
                a_curr = _softmax_fn(z)
                sc = _score_norm_fn(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()

            # Decrease noise over time
            noise_scale *= 0.9999
            if noise_scale < 1e-6:
                noise_scale = 1e-6

        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        max_c = float(c.max())
        lr = max(0.001, lr * 0.7)

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _multi_length_search(best_seq, deadline, rng):
    """Try different sequence lengths to see if a different n gives better score."""
    n_curr = len(best_seq)
    results = []

    for target_n in [2048, 4096, 8192]:
        if target_n == n_curr:
            continue
        if target_n > 16384:  # Too large
            continue
        p_interp = _interp_profile(best_seq, target_n)
        sc = _score_norm_fn(p_interp)
        results.append((sc, p_interp, target_n))

    results.sort(key=lambda x: x[0])
    return results
