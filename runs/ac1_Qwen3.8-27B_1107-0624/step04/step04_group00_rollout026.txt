    fb = np.fft.rfft(b, N)

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses subgradient descent with Adam, softplus continuation, coordinate descent,
multi-restart with diverse initializations, and multi-scale (n=4096, n=8192) exploration.
Key improvements: efficient vectorized subgradient computation, adaptive restart,
and structured initializations based on spectral considerations.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax(fn):
    z = np.asarray(fn, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    return _score_norm(a_norm)


def _softplus_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    shifted = (c - cm) / beta
    logsum = np.log(np.sum(np.exp(shifted)))
    return 2.0 * n * (cm + beta * logsum)


def _softplus_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = 2.0 * wc[n - 1:2 * n - 1]
    Wh = float(np.dot(w, c))
    grad = 2.0 * n * a * (h - Wh)
    if not np.all(np.isfinite(grad)):
        return np.zeros(n)
    return grad


def _subgrad_descend(a_init, deadline, rng, lr_init=0.08, max_iter=200000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    no_improve = 0
    stall_patience = 300
    best_at_stall = best_score

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))

        h = np.zeros(n, dtype=np.float64)
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        h[j_min:j_max + 1] = 2.0 * a[j_min:j_max + 1][::-1]

        ah = float(np.dot(a, h))
        grad_z = a * (h - ah)

        if not np.all(np.isfinite(grad_z)):
            m, v = np.zeros(n), np.zeros(n)
            t = 0
            continue

        lr = lr_init / (1.0 + 0.00005 * t)
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100, 100)
        z = z - z.mean()

        a_new = _softmax(z)
        sc = _score_norm(a_new)

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
            best_at_stall = best_score
        else:
            no_improve += 1

        if no_improve > stall_patience:
            noise_scale = min(0.05, 0.01 * np.sqrt(no_improve / 100.0))
            z = z + noise_scale * rng.normal(size=n)
            z = z - z.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0

    return best_a, best_score


def _softplus_optimize(a_init, deadline, rng):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())

    beta_schedule = [0.1, 0.05, 0.03, 0.02, 0.015, 0.01, 0.008,
                     0.006, 0.005, 0.004, 0.003, 0.002, 0.0015,
                     0.001, 0.0008, 0.0006, 0.0005, 0.0004]
    beta_schedule = [b * cm0 for b in beta_schedule]

    n_betas = len(beta_schedule)
    for idx, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline:
            break

        remaining = n_betas - idx
        time_left = deadline - time.monotonic()
        time_per = min(120.0, time_left / max(1, remaining))
        level_deadline = time.monotonic() + time_per

        if time_per < 5.0:
            break

        try:
            result = minimize(
                _softplus_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_softplus_grad,
                options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 3000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax(z_new)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

        if time.monotonic() > level_deadline:
            continue

    return best_a, best_score


def _coord_descent(a_init, deadline, rng, max_rounds=500):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax(z)
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())
        k_star = int(np.argmax(c))

        valid_i = np.arange(max(0, k_star - n + 1), min(n - 1, k_star) + 1)
        if len(valid_i) == 0:
            break
        contributions = a_cur[valid_i] * a_cur[k_star - valid_i]
        top_idx = valid_i[np.argsort(contributions)[::-1][:15]]

        improved = False
        for idx in top_idx:
            for delta in [0.005, 0.01, 0.02, 0.05, 0.1]:
                z_trial = z.copy()
                z_trial[idx] -= delta
                a_trial = _softmax(z_trial)
                sc = _score_norm(a_trial)
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            small_idx = np.argsort(a_cur)[:10]
            for idx in small_idx:
                for delta in [0.01, 0.05, 0.1]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = _softmax(z_trial)
                    sc = _score_norm(a_trial)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            n_rnd = min(20, n // 10)
            candidates = rng.choice(n, size=n_rnd, replace=False)
            for idx in candidates:
                for delta in [0.02, 0.05, 0.1]:
                    for sign in [-1, 1]:
                        z_trial = z.copy()
                        z_trial[idx] += sign * delta
                        a_trial = _softmax(z_trial)
                        sc = _score_norm(a_trial)
                        if sc < best_score:
                            best_score = sc
                            best_a = a_trial.copy()
                            z = z_trial.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            break

    return best_a, best_score


def _make_diverse_profiles(n, rng):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []

    for slope in [0.3, 0.7, 1.0, 1.5, 2.0, 3.0, -1.0, -2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    for a1, a2 in [(1.0, 0.5), (2.0, -0.5), (0.5, 1.0), (-1.0, 0.5),
                   (1.5, 1.0), (0.0, 2.0)]:
        a = 1.0 + a1 * u + a2 * u ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    for rate in [1.0, 2.0, 3.0, -1.0, -2.0, -3.0, 0.5, -0.5]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())

    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())

    a = np.exp(-4.0 * (t - 0.5) ** 2)
    profiles.append(a / a.sum())

    a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2)
    profiles.append(a / a.sum())

    a = np.exp(-6.0 * (t - 0.3) ** 2) + 0.5 * np.exp(-6.0 * (t - 0.7) ** 2)
    profiles.append(a / a.sum())

    for _ in range(5):
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * freq * t + phase)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    return profiles


def _interp_to_length(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, n_old)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _multi_restart_optimize(a_best, n, deadline, rng, n_restarts=5):

    best_score = _score_norm(a_best)

    per_restart_time = (deadline - time.monotonic()) / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break

        if r == 0:
            a_start = a_best.copy()
        else:
            z = np.log(np.maximum(a_best, 1e-15))
            z = z - z.mean()
            noise_scale = 0.03 * (r + 1)
            z = z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -80, 80)
            z = z - z.mean()
            a_start = _softmax(z)

            if r >= 3:
                z_alt = np.log(np.maximum(a_best, 1e-15))
                z_alt = z_alt - z_alt.mean()
                z_alt = z_alt + 0.1 * rng.normal(size=n)
                z_alt = np.clip(z_alt, -80, 80)
                z_alt = z_alt - z_alt.mean()
                a_alt = _softmax(z_alt)
                if _score_norm(a_alt) < _score_norm(a_start):
                    a_start = a_alt

        per_deadline = time.monotonic() + per_restart_time * 0.85

        p1, s1 = _subgrad_descend(a_start, per_deadline, rng, lr_init=0.06)
        if s1 < best_score:

            p2, s2 = _softplus_optimize(best_a, deadline, rng)

            p6a, s6a = _subgrad_descend(a_large, min(phase6_end - 120, time.monotonic() + 200), rng, lr_init=0.05)
            update_best(p6a, s6a, "subgrad_8192")

        if time.monotonic() < phase6_end - 30:
            p6b, s6b = _softplus_optimize(best_seq if len(best_seq) == n_large else a_large, phase6_end, rng)
            update_best(p6b, s6b, "softplus_8192")

    # Phase 7: Final polish on best (remaining)
    phase7_end = deadline
    print(f"Phase 7: Final polish (target {phase7_end - t_start:.0f}s)")
    if time.monotonic() < phase7_end and best_seq is not None:
        n_final = len(best_seq)

        if time.monotonic() < phase7_end - 60:
            p7a, s7a = _subgrad_descend(best_seq, time.monotonic() + (phase7_end - time.monotonic()) * 0.5, rng, lr_init=0.02)
            update_best(p7a, s7a, "final_subgrad")

        if time.monotonic() < phase7_end - 20:
            p7b, s7b = _coord_descent(best_seq, phase7_end, rng, max_rounds=200)
            update_best(p7b, s7b, "final_coord")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = np.ones(256, dtype=np.float64) / 256.0

        sc_final = _score(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 20000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    fallback = np.ones(256, dtype=np.float64) / 256.0
    print(f"FALLBACK: n=256, score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
