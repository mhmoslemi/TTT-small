def _next_pow2(n):
    k = 1
    while k < n:

    return max(2, k)


def _full_conv(a):
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    n = a.size
    if n == 0:
        return np.inf
    c = _full_conv(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def _grad_from_weights(a, p, N=None):

        g = _grad_from_weights(a, p, N)

        # Gradient in log-space: grad_z = a * (g - dot(a, g))
        dot_ag = float(np.dot(a, g))
        gz = a * (g - dot_ag)
        gz -= gz.mean()  # additional centering for numerical stability

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        step_count += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40, 40)
        z -= z.mean()

        a = np.exp(z - z.max())
        a = a / a.sum()
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        c = _full_conv(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-10:
                best_score = score
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(lr_min, lr * lr_decay)
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        # Adaptive beta
        if step_count % 20 == 0:
            beta = max(beta_min, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]] if sidx[1] < c.size else 0
                if gap < beta * 0.05:
                    beta = max(beta_min, beta * 0.5)

        # Random restart
        if no_imp > no_imp_limit:
            ns = 0.05 + 0.03 * (no_imp // no_imp_limit)
            ns = min(ns, 0.2)
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -40, 40)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_conv(a)
            beta = 0.05 * c.max()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score


def _lp_polish(a, deadline, max_iters=30):
    """
    LP-based fixed point: solve min t s.t. (b*a)[k] <= t, sum(b)=1, b>=0.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Build constraint matrix: (b*a)[k] <= t for k=0..2n-2
        # (b*a)[k] = sum_j b[j] * a[k-j] (valid where 0<=k-j<n)
        # So the row for constraint k has entries a[k-j] at column j (where valid)
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                rows.extend([k] * len(idx))
                cols.extend(idx.tolist())
                vals.extend(a[k - idx].tolist())

        A_conv = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        # Full constraint: A_conv @ b - t <= 0 => [A_conv, -I] @ [b; t] <= 0
        # Plus: sum(b) = 1, b >= 0, t free
        # Convert: minimize t
        # Variables: b[0..n-1], t
        # Constraints: A_conv @ b - t <= 0 (2n-1 constraints)
        #              sum(b) = 1 (equality)
        #              b >= 0
        # t is free (can be negative, but won't be at optimum)

        A_ub = A_conv
        b_ub = np.zeros(2 * n - 1)  # A_conv @ b <= t => A_conv @ b - 1*t <= 0
        # Actually linprog form: A_ub @ x <= b_ub
        # We want: A_conv @ b <= t, i.e., A_conv @ b - t <= 0
        # Variable vector x = [b; t], so A_ub @ x = [A_conv, -I] @ [b; t] <= 0
        # But this has 2n-1 rows and n+1 columns.
        # Alternatively, for each iteration, substitute t = max(A_conv @ b):
        # Just solve: minimize max_k (A_conv @ b)[k] s.t. sum(b)=1, b>=0
        # This is equivalent to: minimize t s.t. (A_conv@b)[k] - t <= 0, sum(b)=1, b>=0

        # Build the full LP:
        # min [0, 0, ..., 0, 1] @ [b; t]
        # s.t. [A_conv, -I] @ [b; t] <= 0   (2n-1 constraints)
        #      [1, 1, ..., 1, 0] @ [b; t] = 1  (equality)
        #      b >= 0, t free

        n_vars = n + 1
        c_obj = np.zeros(n_vars)
        c_obj[n] = 1.0  # minimize t

        # A_ub for linprog: shape (2n-1, n+1)
        A_full = np.zeros((2 * n - 1, n_vars))
        A_full[:, :n] = A_conv.toarray()
        A_full[:, n] = -1.0
        b_ub_full = np.zeros(2 * n - 1)

        # Equality: sum(b) = 1
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n + [(None, None)]

        res = linprog(c_obj, A_ub=A_full, b_ub=b_ub_full,
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')

        if not res.success:
            break

        b_new = np.maximum(res.x[:n], 0.0)
        s_b = float(b_new.sum())
        if s_b < 1e-10:
            break
        b_new = b_new / s_b

        # Line search
        best_t = 1.0
        best_sc = best_score
        for t_candidate in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_t = (1.0 - t_candidate) * a + t_candidate * b_new
            a_t = np.maximum(a_t, 1e-15)
            a_t = a_t / a_t.sum()
            sc_t = _score(a_t)
            if sc_t < best_sc:
                best_sc = sc_t
                best_t = t_candidate

        a = (1.0 - best_t) * a + best_t * b_new
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _make_profile(n, ptype, p1=1.0, p2=0.0, p3=0.0):
    if ptype == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (t - 0.5)
    elif ptype == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (2 * t - 1) ** 2
    elif ptype == 'cubic':
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + p1 * u + p2 * u ** 3
    elif ptype == 'exp':
        t = np.arange(n, dtype=np.float64)
        a = np.exp(-p1 * t / n)
    elif ptype == 'exp_inv':
        t = np.arange(n, dtype=np.float64)
        a = np.exp(p1 * t / n)
    elif ptype == 'power':
        t = np.linspace(0.01, 1.0, n)
        a = t ** p1
    elif ptype == 'power_inv':
        t = np.linspace(0.01, 1.0, n)
        a = t ** (-p1)
    elif ptype == 'cosine':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + p1 * np.cos(2 * np.pi * p2 * t / n)
    elif ptype == 'sine':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + p1 * np.sin(2 * np.pi * p2 * t / n)
    elif ptype == 'parabola':
        t = np.linspace(0, 1, n)
        a = (t - 0.5) ** 2 + p1
    elif ptype == 'U_shape':
        t = np.linspace(0, 1, n)
        a = (2 * t - 1) ** 2 + p1
    elif ptype == 'edge_heavy':
        t = np.arange(n, dtype=np.float64) / max(1, n - 1)
        a = 1.0 / (0.05 + abs(t - 0.5) * 2.0) ** p1
    elif ptype == 'gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / p1) ** 2)
    elif ptype == 'chirp_envelope':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * t + p2 * t * t
    elif ptype == 'log_linear':
        t = np.linspace(0, 1, n)
        a = np.exp(p1 * t)
    elif ptype == 'two_hump':
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - p1) / 0.1) ** 2) + 0.5 * np.exp(-((t - p2) / 0.15) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new


def _peak_shave(a, deadline, rng, n_shaves=5):
    """
    Targeted peak shaving: identify the argmax of convolution,
    and perturb a to reduce it.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)

    for _ in range(n_shaves):
        if time.monotonic() > deadline:
            break
        c = _full_conv(a)
        k_star = int(np.argmax(c))
        # Perturb: reduce elements that contribute most to c[k_star]
        # c[k_star] = sum_i a[i] * a[k_star - i]
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        # Reduce high-contribution elements slightly
        top_k = max(1, n // 10)
        top_indices = np.argsort(contrib)[::-1][:top_k]
        a_new = a.copy()
        for idx in top_indices:
            a_new[idx] *= 0.9
            a_new[(k_star - idx) % n] *= 1.05  # increase the paired element
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            break

    return best_a, best_score

    n_sizes_phase1 = [256, 512, 1024, 2048]

    for n in n_sizes_phase1:
        for (label, ptype, p1, p2, p3) in profile_defs:
            p = _make_profile(n, ptype, p1, p2, p3)
            sc = _score(p)
            candidates.append((sc, p, n, label))

    if parent is not None:
        for n in [512, 1024, 2048, 4096]:
            pi = _interp(parent, n)

        pn = parent.size

        per = min(60.0, phase3_end - time.monotonic() - 10)
        if per > 15:
            a_lp, sc_lp = _lp_polish(best_seq, time.monotonic() + per, max_iters=25)
            if sc_lp < best_score:
                best_seq = a_lp.copy()
                best_score = sc_lp
                print(f"  BEST: {best_score:.10f} n={best_seq.size} [LP_best]")

    # === Phase 4: Multi-scale escalation ===
    print("Phase 4: Multi-scale escalation...")
    phase4_end = t_start + 1300.0

    if best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048, 4096]
        elif n_cur <= 1024:
            target_ns = [2048, 4096]
        elif n_cur <= 2048:
            target_ns = [4096]

        for tn in target_ns:

                break
            p_int = _interp(best_seq, tn)
            sc_int = _score(p_int)
            print(f"  Interpolated to n={tn}: {sc_int:.8f}")
            if sc_int < best_score:
                best_seq = p_int.copy()
                best_score = sc_int
                print(f"  BEST: {best_score:.10f} n={tn} [interp]")

            # Adam on interpolated
            if time.monotonic() < phase4_end - 40:
                per = min(100.0, phase4_end - time.monotonic() - 20)
                if per > 30:
                    a_e, sc_e = _adam_log_space(p_int, time.monotonic() + per, rng,
                                                 lr=0.02, beta_init=None,
                                                 beta_decay=0.97, no_imp_limit=200)
                    if sc_e < best_score:
                        best_seq = a_e.copy()
                        best_score = sc_e
                        print(f"  BEST: {best_score:.10f} n={tn} [Adam_esc]")

    # === Phase 5: Peak shaving + targeted restarts ===
    print("Phase 5: Peak shaving and targeted restarts...")
    phase5_end = t_start + 1550.0

    if best_seq is not None:
        # Peak shave
        if time.monotonic() < phase5_end - 60:
            per = min(60.0, phase5_end - time.monotonic() - 30)
            a_ps, sc_ps = _peak_shave(best_seq, time.monotonic() + per)
            if sc_ps < best_score:
                best_seq = a_ps.copy()
                best_score = sc_ps
                print(f"  BEST: {best_score:.10f} n={best_seq.size} [peak_shave]")

        # Targeted restarts: perturb around best
        n_best = best_seq.size
        n_restarts = 6
        per_r = max(20.0, (phase5_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            per = min(per_r, phase5_end - time.monotonic() - 5)
            if per < 15:
                break

            # Different perturbation strategies
            if i == 0:
                # Small noise
                ns = 0.03
            elif i == 1:
                # Medium noise
                ns = 0.08
            elif i == 2:
                # Large noise
                ns = 0.15
            elif i == 3:
                # Frequency-domain perturbation
                z = np.log(np.maximum(best_seq, 1e-12))
                z -= z.mean()
                Z = np.fft.rfft(z)
                Z[1:-1] *= (1.0 + 0.1 * rng.standard_normal(len(Z[1:-1])))

                z = np.clip(z, -40, 40)
                z -= z.mean()
                a_s = np.exp(z - z.max())
                a_s = a_s / a_s.sum()
                a_r, sc_r = _adam_log_space(a_s, time.monotonic() + per, rng,
                                             lr=0.03, beta_decay=0.97, no_imp_limit=100)
                if sc_r < best_score:
                    best_seq = a_r.copy()
                    best_score = sc_r
                    print(f"  BEST: {best_score:.10f} n={n_best} [restart_freq{i}]")
                continue
            elif i == 4:
                # Mirror perturbation
                a_mirror = best_seq[::-1].copy()
                a_s = 0.5 * best_seq + 0.5 * a_mirror
                a_s = np.maximum(a_s, 1e-12)
                a_s = a_s / a_s.sum()
                # Add noise
                ns = 0.05
                z = np.log(a_s)
                z -= z.mean()
                z += ns * rng.normal(size=n_best)
                z = np.clip(z, -40, 40)
                z -= z.mean()
                a_s = np.exp(z - z.max())
                a_s = a_s / a_s.sum()
                a_r, sc_r = _adam_log_space(a_s, time.monotonic() + per, rng,
                                             lr=0.03, beta_decay=0.97, no_imp_limit=100)
                if sc_r < best_score:
                    best_seq = a_r.copy()
                    best_score = sc_r
                    print(f"  BEST: {best_score:.10f} n={n_best} [restart_mirror{i}]")
                continue
            else:
                ns = 0.10

            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -40, 40)
            z -= z.mean()
            a_s = np.exp(z - z.max())
            a_s = a_s / a_s.sum()

            a_r, sc_r = _adam_log_space(a_s, time.monotonic() + per, rng,
                                         lr=0.03, beta_decay=0.97, no_imp_limit=100)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  BEST: {best_score:.10f} n={n_best} [restart{i}]")

            # Quick LP polish
            if n_best <= 1024 and time.monotonic() < phase5_end - 10:
                per2 = min(20.0, phase5_end - time.monotonic() - 5)
                if per2 > 10:
                    a_rl, sc_rl = _lp_polish(a_r, time.monotonic() + per2, max_iters=10)
                    if sc_rl < best_score:
                        best_seq = a_rl.copy()
                        best_score = sc_rl
                        print(f"  BEST: {best_score:.10f} n={n_best} [LP_restart{i}]")

    # === Phase 6: Final polish ===
    print("Phase 6: Final polish...")
    if time.monotonic() < deadline - 10 and best_seq is not None:
        per = deadline - time.monotonic() - 10
        if per > 20:
            if best_seq.size <= 1024:
                a_fp, sc_fp = _lp_polish(best_seq, time.monotonic() + per * 0.7, max_iters=30)
                if sc_fp < best_score:
                    best_seq = a_fp.copy()
                    best_score = sc_fp
                    print(f"  BEST: {best_score:.10f} n={best_seq.size} [final_LP]")
            # Adam fine-tune with very low learning rate
            if time.monotonic() < deadline - 10:
                per2 = deadline - time.monotonic() - 10
                if per2 > 15:
                    a_fa, sc_fa = _adam_log_space(best_seq, time.monotonic() + per2, rng,
                                                   lr=0.005, beta_decay=0.99,
                                                   no_imp_limit=100)
                    if sc_fa < best_score:
                        best_seq = a_fa.copy()
                        best_score = sc_fa
                        print(f"  BEST: {best_score:.10f} n={best_seq.size} [final_Adam]")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile(512, 'linear', 2.0, 0, 0)

        exact_sc = _score(final)
        print(f"\nFINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile(512, 'linear', 2.0, 0, 0)
    print("FALLBACK n=512")
    return [float(x) for x in fb]
