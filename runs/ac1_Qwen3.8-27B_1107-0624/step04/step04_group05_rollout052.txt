Optimized step function search using spectral shaping, smooth-max Adam,
and LP refinement. Combines frequency-domain peak reduction with
time-domain gradient descent for minimizing the self-convolution peak.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _smooth_max_and_grad(c, beta):
    """Compute smooth max and its gradient w.r.t. c."""
    c = np.asarray(c, dtype=np.float64).ravel()
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.minimum(x, 0.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s  # weights (gradient of smooth max w.r.t. c)
    return val, w


def _conv_cross_corr(w, p):
    """Compute cross-correlation of w and p (p reversed)."""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _score(p):

    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())

    a = np.maximum(a, 0.0)
    s = float(a.sum())

    return a / s

    t = np.arange(n)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / (sigma_frac * n)) ** 2)
    return a / a.sum()

    w = 0.5 * (1 + np.cos(2 * np.pi * t / (n - 1)))
    a = a * w
    return a / a.sum()


def _spectral_shape(a, power=2.0, blend=0.5):
    """Apply spectral concentration to flatten autocorrelation."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    P = np.abs(A) ** 2
    # Concentrate: raise to power (p > 1 concentrates at peaks)
    P_new = P ** power
    # Reconstruct with original phase
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)

    a_new = a_new / s
    return (1 - blend) * a + blend * a_new


def _lp_best_response(a, M):
    """Find b that maximizes sum(b) subject to conv(a,b) <= M, b >= 0."""
    n = len(a)
    # Build the convolution matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _adam_smooth_max(a0, deadline, rng, lr=0.01, beta_init=None, 
                     beta_decay=0.97, patience=200, use_spectral=True,
                     spectral_power=1.5, spectral_blend=0.3, spectral_every=50):
    """Adam optimization in logit space with smooth-max objective."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    
    # Initialize logit
    z = np.log(a)
    z = z - z.mean()
    
    if beta_init is None:
        c = _full_convolve(a)
        beta_init = 0.05 * float(c.max())
    
    beta = beta_init
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_a = a.copy()
    best_score = _score(a)
    best_z = z.copy()
    no_imp = 0
    
    it = 0
    while time.monotonic() < deadline:
        c = _full_convolve(a)
        val, w = _smooth_max_and_grad(c, beta)
        
        # Gradient of smooth max w.r.t. a
        # d(val)/da = 2 * convolve(w, a) (since val = smooth_max(a*a))
        # Actually: val = smooth_max(c) where c = conv(a,a)
        # d(val)/da_i = sum_k w[k] * dc[k]/da_i = sum_k w[k] * 2 * a[k-i] (with appropriate bounds)
        # This is 2 * convolve_correlation(w, a)
        g_c = 2.0 * _conv_cross_corr(w, a)
        
        # Project gradient to the simplex (sum a = 1)
        # The gradient in z-space: dz = a * (g_c - a dot g_c)
        dot = float(np.dot(a, g_c))
        gz = a * (g_c - dot)
        gz = gz - gz.mean()  # Centering for numerical stability
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t = 0
            continue
        
        t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        
        # Softmax
        z_shift = z - z.max()
        e = np.exp(z_shift)
        a = e / e.sum()
        
        # Spectral shaping every N iterations
        if use_spectral and it % spectral_every == 0:
            a = _spectral_shape(a, power=spectral_power, blend=spectral_blend)
        
        sc = _score(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Decay beta
        if it % 20 == 0:
            beta = max(1e-6, beta * beta_decay)
        
        # Restart if stuck
        if no_imp > patience:
            z = best_z + 0.02 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            e = np.exp(z - z.max())
            a = e / e.sum()
            m.fill(0)
            v.fill(0)
            t = 0
            no_imp = 0
            beta = 0.05 * float(_full_convolve(a).max())
            lr = max(0.001, lr * 0.8)
        
        it += 1
    
    return best_a, best_score


def _lp_refine(a, deadline, max_iters=30):

    current_score = _score(a)
    c = _full_convolve(a)
    M = float(c.max())

        if time.monotonic() > deadline:
            break

            break
        sb = float(b.sum())
        if sb < 1e-10:
            break
