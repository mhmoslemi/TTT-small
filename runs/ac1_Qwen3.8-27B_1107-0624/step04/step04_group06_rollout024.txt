Optimized step function search using:
1. Smooth-max gradient descent (Adam on logits)
2. LP-based active set refinement with decreasing M
3. Multi-resolution (n=1024 -> n=2048)
4. Randomized restarts to escape local minima

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())

def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p

def _adam_logit(p0, deadline, rng, lr=0.05, beta_frac=0.07, beta_decay=0.93,
                no_imp_limit=250):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    
    z = np.log(p0)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)
    
    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    
    while time.monotonic() < deadline:
        _, w = _smooth_max_and_weights(c, beta)
        g = _conv_cross_corr(w, p)
        
        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        
        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        
        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.002, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        
        if t_adam % 30 == 0:
            beta = max(1e-5, beta * beta_decay)
        
        if no_imp > no_imp_limit:
            ns = 0.08 if best_score < 1.51 else 0.15
            z = best_z + ns * rng.normal(size=n)

def _build_conv_matrix(a):
    """Build sparse matrix A where A[k,i] = a[k-i] if 0<=k-i<n else 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    A = _build_conv_matrix(a)
    n_solves = 0
    t = t_start

        n_solves += 1
        
        # Solve LP: minimize M s.t. A@b - M <= 0, sum(b)=1, b>=0
        # Variables: [b_0, ..., b_{n-1}, M]
        # c_obj = [0, ..., 0, 1]
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        # A_ub @ x <= b_ub
        # A @ b - M <= 0  ->  [A, -1] @ [b; M] <= 0
        A_ub_full = A.copy()
        # We need to add the -1 column for M
        import scipy.sparse as sp
        A_m = sp.csc_matrix(A)
        neg_one_col = sp.csc_matrix(-np.ones((2*n-1, 1)))
        A_ub_full = sp.hstack([A_m, neg_one_col])
        b_ub_full = np.zeros(2 * n - 1)
        
        # sum(b) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(0, None)]
        
        res = linprog(c_obj, A_ub=A_ub_full, b_ub=b_ub_full,
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        
        if not res.success:
            # Try a relaxed version
            break
        
        b_new = np.maximum(res.x[:n], 0.0)
        M_new = max(float(res.x[-1]), 0.0)
        s_b = float(b_new.sum())
        if s_b < 1e-10:
            break
        b_new = b_new / s_b
        
        a_new = (1.0 - t) * a + t * b_new
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)
        
        if sc_new < best_score - 1e-14:

            best_a = a_new.copy()
        
        a = a_new
        c = _full_convolve(a)
        M = float(c.max())

        M = max(M * m_factor, best_score / (2.0 * n) * 1.001)
        t = max(0.15, t * 0.93)
        
        # Rebuild matrix with new a
        A = _build_conv_matrix(a)
    
    return best_a, best_score

"""
Step function optimization via:
1. Smooth-max Adam on logits (fast local optimization)
2. LP active-set refinement (principled global improvement)
3. Multi-resolution exploration (n=512, 1024, 2048)
4. Randomized restarts (escape local minima)
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, csc_matrix, hstack
import time

# Helper functions (top-level, no closures)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _score(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())

def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if mval < 1e-15:
        return mval, np.ones_like(c) / c.size
    x = (c - mval) / max(beta, 1e-12)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[np.argmax(c)] = 1.0
        s = 1.0
    w = ex / s
    return mval + beta * float(np.log(s)), w

def _gradient_via_cross(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1: 2 * n - 1]

def _softmax_z(z, floor=1e-12):
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0:
        return np.ones(e.size) / e.size
    return p / s

def _adam_optimize(p0, deadline, rng, lr=0.05, beta_frac=0.07, beta_decay=0.93,
                   no_imp_limit=200):
    # Optimizes using smooth-max gradient + Adam on logits
    p0 = np.maximum(np.asarray(p0, dtype=np.float64).ravel(), 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    z = np.log(p0)
    z -= float(np.mean(z))
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_step = 0
    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)
    best_p, best_z = p0.copy(), z.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    
    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)
        g = _gradient_via_cross(w, p)
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))
        if not np.all(np.isfinite(gz)):
            z = best_z.copy(); m.fill(0); v.fill(0); t_step = 0
            continue
        t_step += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** t_step)
        vh = v / (1 - b2 ** t_step)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        if not np.all(np.isfinite(z)):
            z = best_z.copy(); m.fill(0); v.fill(0); t_step = 0
            continue
        z = np.clip(z, -50, 50)
        z -= float(np.mean(z))
        p = _softmax_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        sc = 2.0 * n * maxc
        if np.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.002, lr * 0.95)
        if t_step % 30 == 0:
            beta = max(1e-5, beta * beta_decay)
        if no_imp > no_imp_limit:
            ns = 0.08
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= float(np.mean(z))
            p = _softmax_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0); v.fill(0); t_step = 0
            lr = 0.05
            no_imp = 0
    return best_p, best_score

def _lp_solve_min_m(a):
    """Solve: min M s.t. conv(a_new, a)_k <= M for all k, sum(a_new)=1, a_new>=0"""
    n = len(a)
    # Build A matrix: A[k,i] = a[k-i] if 0<=k-i<n else 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_mat = csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # Full constraint matrix: [A, -1] @ [b; M] <= 0
    neg_col = csc_matrix(-np.ones((2 * n - 1, 1)))
    A_ub = hstack([A_mat, neg_col], format='csc')
    b_ub = np.zeros(2 * n - 1)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return np.maximum(res.x[:n], 0.0), float(max(res.x[-1], 0))
    return None, None

def _lp_refine(a, deadline, max_solves=200):
    """Iterative LP: find b minimizing max conv(a,b), blend into a."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    t_blend = 0.3
    n_solves = 0
    
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b, M_opt = _lp_solve_min_m(a)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b = b / s_b
        a_new = (1 - t_blend) * a + t_blend * b
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc = _score(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        t_blend = max(0.15, t_blend * 0.95)
    
    return best_a, best_score

def _interp_profile(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        p = np.maximum(p, 1e-15)
        return p / p.sum()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()

def _make_dome(n, sharpness=1.0):
    """Create a dome-shaped profile."""
    t = np.linspace(0, 1, n)
    a = np.cos(np.pi * (t - 0.5)) ** sharpness
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_flat_osc(n, freq=1, amp=0.3):
    """Flat with oscillation."""
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

import numpy as np

from scipy.sparse import csc_matrix, hstack
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if mval < 1e-15:
        return mval, np.ones_like(c) / c.size
    x = (c - mval) / max(beta, 1e-12)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[np.argmax(c)] = 1.0
        s = 1.0
    w = ex / s
    return mval + beta * float(np.log(s)), w


def _gradient_via_cross(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1: 2 * n - 1]


def _softmax_z(z, floor=1e-12):
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0:
        return np.ones(e.size) / e.size
    return p / s


def _adam_optimize(p0, deadline, rng, lr=0.05, beta_frac=0.07, beta_decay=0.93,
                   no_imp_limit=200, noise_scale=0.08):

    p0 = p0 / p0.sum()
    n = p0.size
    z = np.log(p0)
    z -= float(np.mean(z))
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_step = 0
    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())

    best_z = z.copy()

    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:

        g = _gradient_via_cross(w, p)
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_step = 0
            continue
        t_step += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * gz * gz
        mh = m / (1.0 - b1 ** t_step)
        vh = v / (1.0 - b2 ** t_step)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        if not np.all(np.isfinite(z)):

            m.fill(0.0)

def _build_conv_rows(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return rows, cols, vals


def _lp_solve_min_m(a):
    n = len(a)
    rows, cols, vals = _build_conv_rows(a)
    A_mat = csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    neg_col = csc_matrix(-np.ones((2 * n - 1, 1)))
    A_ub = hstack([A_mat, neg_col], format='csc')
    b_ub = np.zeros(2 * n - 1)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return np.maximum(res.x[:n], 0.0), float(max(res.x[-1], 0.0))
    return None, None

def _interp_profile(p, n_new):

    n0 = p.size

        return p / p.sum()
    x0 = np.linspace(0.0, 1.0, n0)

    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()
