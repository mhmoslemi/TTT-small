"""
Optimized step function search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive smooth-max Adam optimization with multi-scale escalation,
symmetry enforcement, profile fitting, and coordinate descent polishing.
Key improvements: adaptive beta scheduling, larger FFT for gradient accuracy,
multi-n exploration, and smarter restart strategy.

import numpy as np
import time


def _next_pow2(n):
    if n <= 1:
        return 1
    k = 1
    while k < n:
        k <<= 1
    return k


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _cross_correlate(w, a):
    """Compute cross-correlation g_i = sum_k w_k * a_{k-i} for i in [0, n-1]."""
    n = len(a)
    len_w = len(w)
    N = _next_pow2(len_w + n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    h = np.fft.irfft(fw * np.conj(fa), N)
    return h[len_w - 1:len_w - 1 + n]


def _score(a):
    """Score = 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

        grad_norm = float(np.sqrt(np.mean(gz**2))) + 1e-10
        adapt_lr = lr * min(2.0, max(0.3, 0.1 / grad_norm))

        z = z - adapt_lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40, 40)
        z -= z.mean()

        if symmetric:
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            a = np.zeros(n)
            a[:m_eff] = a_new
            a[n-1 : n-1-m_eff:-1] = a_new
        else:
            a = np.exp(z - z.max())
            a = a / a.sum()

        a = _normalize(a)
        c = _full_convolve(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score) and score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Curriculum beta
        elapsed = time.monotonic() - t0 + elapsed
        progress = min(1.0, (time.monotonic() - (deadline - total_time)) / total_time)
        beta_target = maxc * (0.3 * (1 - progress) + 0.02 * progress)
        # Also adapt based on peak gap
        if len(c) >= 2:
            top = np.sort(c)[::-1][:3]
            gap = top[0] - top[-1]
            if gap < best_beta * 0.3:
                beta_target = max(1e-4, best_beta * 0.5)
        best_beta = 0.7 * best_beta + 0.3 * beta_target
        best_beta = max(1e-4, min(0.5 * maxc, best_beta))

        # Restart strategy
        if no_imp > 250:
            noise = 0.03 + 0.12 * (no_imp / 250)
            z = z + noise * rng.normal(size=m_eff)
            z = np.clip(z, -40, 40)
            z -= z.mean()
            if symmetric:
                a_new = np.exp(z - z.max())
                a_new = a_new / a_new.sum()
                a = np.zeros(n)
                a[:m_eff] = a_new
                a[n-1 : n-1-m_eff:-1] = a_new
            else:
                a = np.exp(z - z.max())
                a = a / a.sum()
            a = _normalize(a)
            c = _full_convolve(a)
            best_beta = 0.2 * c.max()
            m_arr = np.zeros(m_eff)
            v_arr = np.zeros(m_eff)
            t = 0
            no_imp = 0

    return best_a, best_score


def _coord_descent_fast(a, deadline, max_sweeps=30):
    """Fast coordinate descent with binary search for optimal value."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score(a)
    c_max = float(_full_convolve(a).max())

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        # Randomize sweep order
        order = np.random.permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            old = a[j]
            # Try multiplicative factors
            best_local = old
            best_local_score = best_score
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]:
                a[j] = max(old * factor, 1e-15)
                sc = _score(a)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local = a[j]
            a[j] = best_local
            if best_local != old:
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    improved = True
        if not improved:
            # Try smaller steps
            improved2 = False
            for j in range(0, n, max(1, n // 50)):
                if time.monotonic() > deadline:
                    break
                old = a[j]
                for factor in [0.9, 0.95, 1.05, 1.1]:
                    a[j] = max(old * factor, 1e-15)
                    sc = _score(a)
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        improved2 = True
                    else:
                        a[j] = old
                        break
            if not improved2:
                break

    return best_a, best_score


def _make_profile(n, ptype, p1=1.0, p2=0.0, p3=0.0):
    if ptype == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (t - 0.5)
    elif ptype == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (2 * t - 1) ** 2
    elif ptype == 'cosine':
        t = np.arange(n)
        a = 1.0 + p1 * np.cos(2 * np.pi * p2 * t / n)
    elif ptype == 'sine':
        t = np.arange(n)
        a = 1.0 + p1 * np.sin(2 * np.pi * p2 * t / n)
    elif ptype == 'parabola':
        t = np.linspace(0, 1, n)
        a = (t - 0.5) ** 2 + p1
    elif ptype == 'gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / p1) ** 2)
    elif ptype == 'U_shape':
        t = np.linspace(0, 1, n)
        a = (2 * t - 1) ** 2 + p1
    elif ptype == 'edge_heavy':
        t = np.arange(n) / max(1, n - 1)
        a = 1.0 / (0.01 + abs(t - 0.5) * 2.0) ** p1
    elif ptype == 'wavy':
        t = np.arange(n)
        a = 1.0 + p1 * np.cos(2 * np.pi * p2 * t / n) + p3 * np.cos(2 * np.pi * 2 * p2 * t / n)
    elif ptype == 'plateau':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / p1) ** p2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_scale(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()
