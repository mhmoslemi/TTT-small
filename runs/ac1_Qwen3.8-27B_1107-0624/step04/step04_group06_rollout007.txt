def _lp_refine(a, deadline, t_start=0.3, max_solves=30, m_shrink=0.995):
    """LP refinement with decreasing M."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    c = _full_convolve(a)
    M = float(c.max())

    for solve_idx in range(max_solves):
        if time.monotonic() > deadline:
            break

        # Build LP
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                rows.extend([k] * len(idx))
                cols.extend(idx.tolist())
                vals.extend((a[k - idx]).tolist())
        A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b_ub_arr = M * np.ones(2 * n - 1)
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
        if not res.success:
            M *= 1.02
            continue

        b = np.maximum(res.x, 0.0)
        s_b = float(b.sum())
        if s_b < 1e-10:
            break

        b_norm = b / s_b
        a_new = (1.0 - t_start) * a + t_start * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score(a_new)

        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        c = _full_convolve(a)
        M = max(float(c.max()) * m_shrink, best_score / (2.0 * n) * 1.001)
        t_start = max(0.1, t_start * 0.93)

    return best_a, best_score

def _make_symmetric_profile(n, func):
    """Create a symmetric (palindrome) profile, which minimizes c[n-1]."""
    half = (n + 1) // 2
    t = np.linspace(0.0, 1.0, half)
    a_half = func(t)
    a_half = np.maximum(a_half, 1e-12)
    a = np.concatenate([a_half, a_half[-2::-1]])  # palindrome
    return a / a.sum()

def _spectral_random(n, rng):
    """Generate a random sequence with prescribed spectral shape."""
    # Create a random phase in frequency domain
    N = n
    freq = np.fft.rfftfreq(N)
    # Low-pass filter to get smooth sequences
    mask = 1.0 / (1.0 + 10 * freq**2)
    amp = rng.standard_normal(N // 2 + 1) * mask
    phase = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    spec = amp * np.exp(1j * phase)
    # Make DC positive
    spec[0] = abs(spec[0]) + 1.0
    a = np.fft.irfft(spec * N, N)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _recombine(p1, p2, t, rng):

    p = (1 - t) * p1 + t * p2
    p = np.maximum(p, 1e-12)
    # Add small noise
    p = p + 0.01 * rng.normal(size=p.size)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-15, 1000.0)

    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s

    topk_idx = np.argsort(c)[::-1][:k]
    topk_vals = c[topk_idx]
    x = (topk_vals - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros(n)
        w[topk_idx[0]] = 1.0
        return mval, w
    weights_topk = ex / s
    w = np.zeros(n)
    w[topk_idx] = weights_topk
    val = mval + beta * float(np.log(s))
    return val, w


def _conv_gradient(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _cyclic_shift_best(p, max_shifts=300):
    n = len(p)
    best_p = p.copy()
    best_score = _score(p)
    n_try = min(max_shifts, n)
    step = max(1, n // n_try)
    for s in range(0, n, step):
        if s == 0:
            continue
        p_shift = np.roll(p, s)
        sc = _score(p_shift)
        if sc < best_score - 1e-14:
            best_score = sc
            best_p = p_shift.copy()
    return best_p, best_score


def _adam_optimize(p0, deadline, rng, lr=0.05, beta_frac=0.07, k_top=5,
                   beta_decay=0.94, no_imp_limit=200, k_schedule='linear'):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    z = np.log(np.maximum(p0, 1e-15))
    z -= float(np.mean(z))
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)
    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    max_iters_est = max(100, int((deadline - time.monotonic()) / 0.01))

    while time.monotonic() < deadline:
        if k_schedule == 'linear':
            progress = min(1.0, t_adam / max(1, max_iters_est * 0.7))
            current_k = max(1, k_top - int(progress * (k_top - 1)))
        else:
            current_k = k_top

        _, w = _topk_smooth_max(c, current_k, beta)
        g = _conv_gradient(w, p)
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 80 == 0:
                    lr = max(0.001, lr * 0.96)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 25 == 0:
            beta = max(1e-6, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap > beta * 3:
                    beta = max(1e-6, beta * 0.6)

        if no_imp > no_imp_limit:
            # Spectral perturbation for diversity
            spec = np.fft.rfft(p)
            phase_pert = rng.normal(size=spec.size) * 0.08
            spec_new = spec * np.exp(1j * phase_pert)
            spec_new[0] = abs(spec_new[0]) + 0.5
            p_new = np.fft.irfft(spec_new, 2 * n)[:n]
            p_new = np.maximum(p_new, 1e-12)
            sn = p_new.sum()
            if sn > 0.01:
                p_new = p_new / sn
                z = np.log(np.maximum(p_new, 1e-12))
                z -= float(np.mean(z))
            else:
                z = best_z + 0.15 * rng.normal(size=n)
                z = np.clip(z, -50.0, 50.0)
                z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _direct_peak_descent(p, deadline, lr_init=0.02, max_iters=2000):
    """Directly target and reduce the single maximum peak via projected gradient."""
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size
    best_p = p.copy()
    best_score = _score(p)
    c = _full_convolve(p)
    lr = lr_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        k_peak = int(np.argmax(c))
        M = float(c[k_peak])
        score = float(2.0 * n * M)
        if score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()

        # Gradient of c[k_peak] w.r.t. a:
        # c[k] = sum_i a[i]*a[k-i] (valid indices)
        # dc[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n, j-k != k-j)
        grad = np.zeros(n)
        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)
        for i in range(i_start, i_end + 1):
            j = k_peak - i
            grad[i] += a[j]
            if j != i:
                grad[j] += a[i]

        # Projected gradient step: reduce mass where gradient is high
        # Use multiplicative update for stability
        grad_norm = float(np.abs(grad).max())
        if grad_norm < 1e-15:
            break
        grad_norm = grad / grad_norm
        factor = 1.0 - lr * np.maximum(grad_norm, 0)
        factor = np.clip(factor, 0.05, 1.0)
        p = p * factor
        p = np.maximum(p, 1e-15)
        p = p / p.sum()
        c = _full_convolve(p)

        # Adaptive lr
        if it % 100 == 0 and it > 0:
            if score >= best_score:
                lr = max(0.001, lr * 0.8)
            else:
                lr = min(0.05, lr * 1.1)

    return best_p, best_score


def _lp_refine(a, deadline, t_init=0.35, max_solves=40, m_shrink=0.994):
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    c = _full_convolve(a)
    M = float(c.max())
    t = t_init

    for _ in range(max_solves):
        if time.monotonic() > deadline:
            break
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                rows.extend([k] * len(idx))
                cols.extend(idx.tolist())
                vals.extend((a[k - idx]).tolist())
        A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b_ub_arr = M * np.ones(2 * n - 1)
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
        if not res.success:
            M *= 1.03
            continue
        b = np.maximum(res.x, 0.0)
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score(a_new)
        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()
        a = a_new
        c = _full_convolve(a)
        M = max(float(c.max()) * m_shrink, best_score / (2.0 * n) * 1.002)
        t = max(0.15, t * 0.92)

    return best_a, best_score


def _peak_flood(a, deadline, eta=0.025, n_iter_max=4000):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    c = _full_convolve(a)

    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break
        M = float(c.max())
        score = float(2.0 * n * M)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
        threshold = M * 0.92
        hot_mask = c >= threshold
        if not hot_mask.any():
            hot_mask[int(np.argmax(c))] = True
        heat = np.zeros(n)
        hot_indices = np.where(hot_mask)[0]
        for k in hot_indices:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                partner = k - np.arange(i_start, i_end + 1)
                heat[i_start:i_end + 1] += a[partner]
        hm = float(heat.max())
        if hm < 1e-15:
            break
        heat = heat / hm
        factor = 1.0 - eta * heat
        factor = np.maximum(factor, 0.01)
        a = a * factor
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        c = _full_convolve(a)
        if it % 200 == 0 and it > 0:
            if score >= best_score:
                eta = max(0.002, eta * 0.85)
            else:
                eta = min(0.08, eta * 1.06)

    return best_a, best_score


def _spectral_random(n, rng):
    N = n
    freq = np.fft.rfftfreq(N)
    mask = 1.0 / (1.0 + 5.0 * freq ** 2)
    amp = rng.standard_normal(N // 2 + 1) * mask
    phase = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    spec = amp * np.exp(1j * phase)
    spec[0] = abs(spec[0]) + 1.0
    a = np.fft.irfft(spec * N, N)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine(n, freq, amp):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _recombine(p1, p2, t, rng):
    p = (1 - t) * np.asarray(p1) + t * np.asarray(p2)
    p = np.maximum(p, 1e-12)
    p = p + 0.02 * rng.normal(size=p.size)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    best_seq = None
    best_score = np.inf

    def _update(p, label=""):
        nonlocal best_seq, best_score
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            p = np.maximum(p, 1e-15)
            s = p.sum()
            if s < 0.01:
                return
            p = p / s
            sc = _score(p)
            if np.isfinite(sc) and sc < best_score:
                best_seq = p.copy()
                best_score = float(sc)
                print(f"  BEST: {best_score:.10f} n={p.size} [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent score: {sc_parent:.10f}")

    # Phase 0: Cyclic shift on parent (30s)
    if parent is not None:
        print("Phase 0: Cyclic shift optimization...")
        p_cs, sc_cs = _cyclic_shift_best(parent, max_shifts=500)
        _update(p_cs, 'cyclic_parent')

    # Phase 1: Adam with top-k (200s)
    phase1_end = t_start + 200.0
    print(f"Phase 1: Adam top-k, score={best_score:.10f}...")
    start_p = best_seq if best_seq is not None else parent
    if start_p is None:
        start_p = _make_linear(1024, 2.0)
    p1, sc1 = _adam_optimize(start_p, phase1_end, rng, lr=0.05, beta_frac=0.07,
                              k_top=5, beta_decay=0.94, no_imp_limit=180)
    _update(p1, 'adam1')
    # Cyclic shift after Adam
    if best_seq is not None and time.monotonic() < t_start + 220:
        p_cs2, _ = _cyclic_shift_best(best_seq, max_shifts=300)
        _update(p_cs2, 'cyclic_post_adam1')

    # Phase 2: Direct peak descent (150s)
    phase2_end = t_start + 370.0
    print(f"Phase 2: Direct peak descent, score={best_score:.10f}...")
    if best_seq is not None:
        p2, sc2 = _direct_peak_descent(best_seq, phase2_end, lr_init=0.015, max_iters=2000)
        _update(p2, 'direct_peak')

    # Phase 3: Peak flooding (200s)
    phase3_end = t_start + 570.0
    print(f"Phase 3: Peak flooding, score={best_score:.10f}...")
    if best_seq is not None:
        p3, sc3 = _peak_flood(best_seq, phase3_end, eta=0.025)
        _update(p3, 'flood1')

    # Phase 4: LP refinement (300s)
    phase4_end = t_start + 870.0
    print(f"Phase 4: LP refinement, score={best_score:.10f}...")
    if best_seq is not None and best_seq.size <= 2048:
        p4, sc4 = _lp_refine(best_seq, phase4_end, t_init=0.35, max_solves=45, m_shrink=0.993)
        _update(p4, 'lp1')
        # Adam polish after LP
        if time.monotonic() < phase4_end - 30:
            p4a, _ = _adam_optimize(best_seq, min(phase4_end, time.monotonic() + 50), rng,
                                     lr=0.025, beta_frac=0.04, k_top=3,
                                     beta_decay=0.96, no_imp_limit=100)
            _update(p4a, 'adam_post_lp1')

    # Phase 5: Second round - Adam + shift + direct (200s)
    phase5_end = t_start + 1070.0
    print(f"Phase 5: Round 2, score={best_score:.10f}...")
    if best_seq is not None:
        p5, _ = _adam_optimize(best_seq, min(phase5_end, time.monotonic() + 80), rng,
                                lr=0.03, beta_frac=0.05, k_top=4,
                                beta_decay=0.95, no_imp_limit=120)
        _update(p5, 'adam2')
        if best_seq is not None:
            p5c, _ = _cyclic_shift_best(best_seq, max_shifts=300)
            _update(p5c, 'cyclic_r2')
        if time.monotonic() < phase5_end - 20 and best_seq is not None:
            p5d, _ = _direct_peak_descent(best_seq, min(phase5_end, time.monotonic() + 60),
                                           lr_init=0.01, max_iters=1000)
            _update(p5d, 'direct_r2')

    # Phase 6: Multi-scale exploration (200s)
    phase6_end = t_start + 1270.0
    print(f"Phase 6: Multi-scale, score={best_score:.10f}...")
    if time.monotonic() < phase6_end and best_seq is not None:
        n_cur = best_seq.size
        for n_try in [n_cur * 2, max(256, n_cur // 2)]:
            if time.monotonic() > phase6_end - 30:
                break
            if n_try < 64 or n_try > 4096:
                continue
            x0 = np.linspace(0, 1, n_cur)
            x1 = np.linspace(0, 1, n_try)
            p_int = np.interp(x1, x0, best_seq)
            p_int = np.maximum(p_int, 1e-15) / np.sum(np.maximum(p_int, 1e-15))
            _update(p_int, f'interp_{n_try}')
            if time.monotonic() < phase6_end - 20:
                per = min(60.0, phase6_end - time.monotonic() - 10)
                if per > 15:
                    p_it, _ = _adam_optimize(p_int, time.monotonic() + per, rng,
                                              lr=0.04, beta_frac=0.06, k_top=5,
                                              beta_decay=0.94, no_imp_limit=100)
                    _update(p_it, f'adam_interp_{n_try}')

    # Phase 7: Alternative initializations (200s)
    phase7_end = t_start + 1470.0
    print(f"Phase 7: Alt inits, score={best_score:.10f}...")
    if time.monotonic() < phase7_end:
        n_base = best_seq.size if best_seq is not None else 1024
        alts = []
        for i in range(4):
            alts.append((f'spec_{i}', _spectral_random(n_base, rng)))
        alts.append(('lin2', _make_linear(n_base, 2.0)))
        alts.append(('lin3', _make_linear(n_base, 3.0)))
        alts.append(('cos1', _make_cosine(n_base, 1, 0.7)))
        alts.append(('cos2', _make_cosine(n_base, 2, 0.4)))
        per_alt = max(12.0, (phase7_end - time.monotonic()) / len(alts))
        for label, pa in alts:
            if time.monotonic() > phase7_end - 10:
                break
            per = min(per_alt, phase7_end - time.monotonic() - 5)
            if per < 10:
                break
            p_alt, _ = _adam_optimize(pa, time.monotonic() + per, rng,
                                       lr=0.05, beta_frac=0.08, k_top=5,
                                       beta_decay=0.93, no_imp_limit=100)
            _update(p_alt, f'alt_{label}')

    # Phase 8: Recombination + LP (150s)
    phase8_end = t_start + 1620.0
    print(f"Phase 8: Recombination, score={best_score:.10f}...")
    if time.monotonic() < phase8_end and best_seq is not None:
        pool = [best_seq.copy()]
        for i in range(2):
            pert = best_seq + 0.04 * rng.normal(size=best_seq.size)
            pert = np.maximum(pert, 1e-12)
            pool.append(pert / pert.sum())
        per_rec = max(10.0, (phase8_end - time.monotonic()) / max(1, len(pool) * (len(pool) - 1)))
        for i in range(len(pool)):
            for j in range(i + 1, len(pool)):
                if time.monotonic() > phase8_end - 20:
                    break
                for t_mix in [0.3, 0.5, 0.7]:
                    if time.monotonic() > phase8_end - 10:
                        break
                    rec = _recombine(pool[i], pool[j], t_mix, rng)
                    _update(rec, f'rec_{i}{j}_{t_mix}')
        # Final LP
        if time.monotonic() < phase8_end - 15 and best_seq is not None and best_seq.size <= 2048:
            per_lp = min(50.0, phase8_end - time.monotonic() - 10)
            if per_lp > 15:
                p8l, _ = _lp_refine(best_seq, time.monotonic() + per_lp,
                                     t_init=0.2, max_solves=12, m_shrink=0.998)
                _update(p8l, 'lp_r2')

    # Phase 9: Final intensive polish (remaining ~160s)
    phase9_end = deadline
    print(f"Phase 9: Final polish, score={best_score:.10f}...")
    if time.monotonic() < phase9_end and best_seq is not None:
        remaining = phase9_end - time.monotonic()
        if remaining > 40:
            p9a, _ = _adam_optimize(best_seq, time.monotonic() + remaining * 0.35, rng,
                                     lr=0.012, beta_frac=0.025, k_top=2,
                                     beta_decay=0.97, no_imp_limit=80)
            _update(p9a, 'final_adam')
        if time.monotonic() < phase9_end - 20 and best_seq is not None:
            p9c, _ = _cyclic_shift_best(best_seq, max_shifts=400)
            _update(p9c, 'final_cyclic')
        if time.monotonic() < phase9_end - 15 and best_seq is not None:
            per_d = min(30.0, phase9_end - time.monotonic() - 10)
            if per_d > 10:
                p9d, _ = _direct_peak_descent(best_seq, time.monotonic() + per_d,
                                               lr_init=0.005, max_iters=500)
                _update(p9d, 'final_direct')
        if time.monotonic() < phase9_end - 10 and best_seq is not None and best_seq.size <= 2048:
            per_f = min(20.0, phase9_end - time.monotonic() - 5)
            if per_f > 8:
                p9f, _ = _peak_flood(best_seq, time.monotonic() + per_f, eta=0.01)
                _update(p9f, 'final_flood')
        # Absolute final cyclic shift
        if time.monotonic() < phase9_end - 3 and best_seq is not None:
            p_fc, _ = _cyclic_shift_best(best_seq, max_shifts=500)
            _update(p_fc, 'abs_final_cyclic')

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_linear(1024, 2.0)
        exact_sc = _score(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")
        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass
        return [float(x) for x in final]

    fb = _make_linear(1024, 2.0)
    print(f"FALLBACK: score={_score(fb):.10f}")
    return [float(x) for x in fb]
