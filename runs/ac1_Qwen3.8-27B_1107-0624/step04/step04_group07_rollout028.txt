"""
Multi-scale Adam smooth-max optimization with diverse initialization,
parametric profile search, and iterative refinement for minimizing
2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based convolution, log-space parameterization, and adaptive
learning rates with restart strategies.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _cross_correlate(w, a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w


def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new


def _make_profile(n, ptype=0, p1=1.0, p2=0.0, p3=0.0):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    if ptype == 0:  # linear
        a = 1.0 + p1 * (t - 0.5)
    elif ptype == 1:  # quadratic
        a = 1.0 + p1 * u ** 2
    elif ptype == 2:  # cubic
        a = 1.0 + p1 * u + p2 * u ** 3
    elif ptype == 3:  # exponential decay
        a = np.exp(-p1 * t)
    elif ptype == 4:  # cosine
        a = 1.0 + p1 * np.cos(2 * np.pi * p2 * t)
    elif ptype == 5:  # sine
        a = 1.0 + p1 * np.sin(2 * np.pi * p2 * t)
    elif ptype == 6:  # parabola
        a = u ** 2 + p1
    elif ptype == 7:  # two gaussians
        a = np.exp(-0.5 * ((t - p1) / p3) ** 2) + np.exp(-0.5 * ((t - p2) / p3) ** 2)
    elif ptype == 8:  # gaussian
        a = np.exp(-0.5 * ((t - 0.5) / p1) ** 2)
    elif ptype == 9:  # U shape
        a = u ** 2 + p1
    elif ptype == 10:  # edge heavy
        a = 1.0 / (0.01 + abs(t - 0.5) * 2.0) ** p1
    elif ptype == 11:  # inverse sqrt
        j = np.arange(1, n + 1)
        a = 1.0 / np.sqrt(j)
    elif ptype == 12:  # inverse sqrt symmetric
        j = np.arange(n)
        a = 1.0 / np.sqrt((j + 1) * (n - j))
    elif ptype == 13:  # raised cosine (Hann)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif ptype == 14:  # Hamming
        a = 0.54 + 0.46 * np.cos(2 * np.pi * t)
    elif ptype == 15:  # Blackman
        a = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    elif ptype == 16:  # power
        a = t ** p1 + (1 - t) ** p1
    elif ptype == 17:  # dual freq
        a = 1.0 + p1 * np.cos(2 * np.pi * p2 * t) + p3 * np.cos(2 * np.pi * (p2 + 1) * t)
    elif ptype == 18:  # sinc-like
        x = p1 * u
        a = np.where(np.abs(x) > 1e-10, np.sin(np.pi * x) / (np.pi * x), 1.0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _adam_smooth_max(a, deadline, rng, lr=0.05, beta_init=None,

            v.fill(0)

            continue

def _block_optimization(a, deadline, block_size=32):

            grad = np.zeros(block_size_actual)

                k_j = k_max - j

                    grad[local_j] = 2.0 * a[k_j]

                    best_score = sc_cd
                    print(f"  BEST: {best_score:.10f} n={best_seq.size} [coord]")

    # Final validation and return
    print(f"\nFinal internal score: {best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile(512, 0, 1.5, 0, 0)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile(512, 0, 1.5, 0, 0)
    print("FALLBACK n=512")
    return [float(x) for x in fb]
