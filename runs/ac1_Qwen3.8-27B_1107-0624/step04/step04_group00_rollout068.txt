        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _cross_corr(w, a):
    """Compute h[i] = sum_k w[k]*a[k-i] for i=0..n-1 using FFT."""
    n = len(a)
    N = _next_pow2(3 * n)
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    h_full = np.fft.ifft(W * np.conj(A), N).real
    return h_full[:n]


def _softmax_from_z(z):
    """Convert z to probability vector via softmax."""
    z_shifted = z - np.max(z)
    e = np.exp(z_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _soft_obj_grad(z, beta, n):
    """
    Compute soft-smoothed score and gradient w.r.t. z (log-param).
    Score = 2n * softmax_approx(max(conv(a,a)))
    where a = softmax(z).
    """
    a = _softmax_from_z(z)
    c = _full_convolve(a, a)
    cm = float(c.max())

    # Softmax weights over convolution values
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s_e = float(e.sum())
    if s_e < 1e-300 or not np.isfinite(s_e):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s_e

    soft_max_val = cm + beta * float(np.log(s_e))
    score = 2.0 * n * soft_max_val

    # Gradient: d(soft_max)/da[i] = 2*h[i] where h = cross_corr(w, a)
    h = _cross_corr(w, a)
    grad_a = 2.0 * h

    # Convert to gradient w.r.t. z: grad_z = grad_a*a - a*dot(grad_a, a)
    S = float(np.dot(grad_a, a))
    grad_z = grad_a * a - a * S

    if not np.all(np.isfinite(grad_z)):
        grad_z = np.zeros_like(z)

    return score, grad_z


def _lbfgs_continuation(z0, n, beta_list, deadline, maxiter=2000):
    """L-BFGS with decreasing beta continuation schedule."""
    best_a = None
    best_score = np.inf
    z = z0.copy()

    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break

        remaining = len(beta_list) - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 120.0)
        level_end = time.monotonic() + time_per

        def obj_fn(z_eval):
            return _soft_obj_grad(z_eval, beta, n)[0]

        def grad_fn(z_eval):
            return _soft_obj_grad(z_eval, beta, n)[1]

        maxiter_this = min(maxiter, max(100, int(time_per * 15)))
        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': maxiter_this, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxls': 100, 'maxfun': maxiter_this * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z = result.x.copy()
        except Exception as e:
            print(f"    L-BFGS error at beta={beta:.2e}: {e}")
            break

        a = _softmax_from_z(z)
        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Final evaluation
    a = _softmax_from_z(z)
    sc = _score_norm(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def _linear_ramp(n):
    """Normalized linear ramp: a[i] = (i+1)/T."""
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / a.sum()


def _make_initial_profiles(n, rng):
    """Generate diverse initial profiles for exploration."""
    t = np.linspace(0, 1, n)
    profiles = []

    # Linear ramp (our best known)
    profiles.append(_linear_ramp(n))

    # Power laws
    for p in [0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.5, 2.0]:
        a = np.maximum(t, 1e-10) ** p
        profiles.append(a / a.sum())

    # Linear with modified endpoints
    for eps in [0.05, 0.1, 0.2, 0.3]:
        a = t + eps * t * (1 - t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Exponential
    for rate in [0.5, 1.0, 2.0, 3.0]:
        a = np.exp(rate * t)
        profiles.append(a / a.sum())

    # Asymmetric: steep then flat
    for alpha in [0.3, 0.5, 0.7]:
        a = t ** alpha
        profiles.append(a / a.sum())

    return profiles


def _peak_directed_search(a, deadline, rng, max_rounds=200):
    """
    Directly reduce the convolution peak by adjusting contributing elements.
    Works in the log-parameterization space.
    """
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_from_z(z)
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())
        k_star = int(np.argmax(c))

        # Find elements contributing to the peak
        valid_i = np.arange(max(0, k_star - n + 1), min(n - 1, k_star) + 1)
        contributions = a_cur[valid_i] * a_cur[k_star - valid_i]

        # Try reducing top contributors (in log space)
        top_count = min(15, len(valid_i))
        top_local = np.argsort(contributions)[::-1][:top_count]

        improved = False
        for li in top_local:
            idx = valid_i[li]
            # Try reducing this element
            for delta in [0.01, 0.02, 0.05, 0.1, 0.2]:
                z_trial = z.copy()
                z_trial[idx] -= delta
                a_trial = _softmax_from_z(z_trial)
                sc = _score_norm(a_trial)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try increasing small elements (redistribute mass)
            small_idx = np.argsort(a_cur)[:8]
            for idx in small_idx:
                for delta in [0.01, 0.03, 0.05, 0.1]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = _softmax_from_z(z_trial)
                    sc = _score_norm(a_trial)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Try a random perturbation to escape
            if rnd % 20 == 0:
                scale = 0.03 * (1 + rnd / 100.0)
                z = z + scale * rng.normal(size=n)
                z = z - z.mean()
            else:
                break

    return best_a, best_score


def _adam_polish(a, deadline, rng, lr=0.01, beta=1e-4):
    """Adam-based polish with fixed small beta."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    best_a = a.copy()
    best_score = _score_norm(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr_cur = lr
    stuck_count = 0

    while time.monotonic() < deadline:
        t += 1
        score, grad = _soft_obj_grad(z, beta, n)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        a_cur = _softmax_from_z(z)
        sc = _score_norm(a_cur)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        if t % 500 == 0:
            lr_cur *= 0.9
        if stuck_count > 5000:
            z = z + 0.02 * rng.normal(size=n)
            z = z - z.mean()
            stuck_count = 0
        if stuck_count > 20000:
            break

    return best_a, best_score
