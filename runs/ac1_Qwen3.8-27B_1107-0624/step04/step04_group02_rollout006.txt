    A_flip = np.fft.rfft(a[::-1], N_fft)
    cross = np.fft.irfft(W * A_flip, N_fft)
    grad = 2.0 * p * cross[n - 1:2 * n - 1]
    return grad, c


def _powmean_obj_and_grad(z, n, p):
    """Objective and gradient for p-norm in log-space (softmax parameterization)."""
    a = _softmax_fn(z)
    grad_da, c = _powmean_grad_da(a, p)
    # Project gradient to log-space (tangent space of simplex)
    # d(a_i)/d(z_j) = a_i*(delta_ij - a_j)
    # df/dz_j = a_j * (df/da_j - sum_i a_i * df/da_i)
    mean_grad = float(np.dot(a, grad_da))
    grad_z = a * (grad_da - mean_grad)
    # Objective: (sum c^p)^(1/p)  [we minimize this, or equivalently sum c^p]
    c_clipped = np.maximum(c, 1e-30)
    obj = float(np.sum(c_clipped ** p))
    return obj, grad_z


def _run_powmean_lbfgs(a_init, deadline, p_list, max_wall_time=None):
    """L-BFGS with power mean continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    if max_wall_time is not None:
        per_level = max(5.0, max_wall_time / max(1, len(p_list)))
    else:
        per_level = 60.0

    for p in p_list:
        if time.monotonic() > deadline - 2.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        try:
            result = minimize(
                _powmean_obj_and_grad, z, args=(n, p),
                jac=True,
                method='L-BFGS-B',
                options={'maxiter': 2000, 'ftol': 1e-20, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 10000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z = result.x
                a_new = _softmax_fn(z)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    print(f"    p={p}: score={sc:.10f}")
        except Exception as e:
            print(f"    p={p}: L-BFGS failed: {e}")
            # Try with fewer iterations
            try:
                result = minimize(
                    _powmean_obj_and_grad, z, args=(n, p),
                    jac=True,
                    method='L-BFGS-B',
                    options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-10,
                             'maxls': 30, 'maxfun': 5000}
                )
                if result.x is not None and np.all(np.isfinite(result.x)):
                    z = result.x
                    a_new = _softmax_fn(z)
                    sc = _score_norm_fn(a_new)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
            except Exception:
                pass

    return best_a, best_score


def _solve_lp(f, M):
    """Solve LP: max sum(g) s.t. g >= 0, conv(f,g) <= M."""
    n = len(f)
    # Build constraint matrix A (2n-1 x n): A[k,j] = f[j] if 0 <= k-j < n
    # This is a banded Toeplitz matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    if res.status == 0:
        return res.x
    return None


def _lp_refinement(a_init, deadline, max_iters=8):
    """LP-based refinement: solve LP for best response, interpolate."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 10.0:
            break
        
        c = _full_convolve(a, a)
        M = float(c.max())
        print(f"    LP iter {it}: M={M:.10f}, score={best_score:.10f}")
        
        try:
            g = _solve_lp(a, M)
            if g is None or not np.all(np.isfinite(g)):
                print(f"    LP failed at iter {it}")
                break
            g_sum = float(g.sum())
            if g_sum < 1e-10:
                break
            g = g / g_sum  # Normalize
        except Exception as e:
            print(f"    LP error at iter {it}: {e}")
            break
        
        # Interpolate
        t = 0.5  # Interpolation weight
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-10:
            break
        a_new = a_new / a_new_sum
        
        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            t2 = 0.2
            a_try = (1 - t2) * a + t2 * g
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s > 1e-10:
                a_try = a_try / s
                sc2 = _score_norm_fn(a_try)
                if sc2 < best_score:
                    best_score = sc2
                    best_a = a_try.copy()
                    a = a_try
                else:
                    break
            else:
                break
    
    return best_a, best_score


def _coordinate_descent(a_init, deadline, n_sweeps=50):
    """Coordinate descent: update one element at a time."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline - 2.0:
            break
        
        for j in range(n):
            if time.monotonic() > deadline - 2.0:
                break
            # Try modifying a[j]
            orig = a[j]
            # Try several multipliers
            for mult in [0.7, 0.85, 1.0, 1.15, 1.3]:
                a[j] = orig * mult
                if a[j] < 0:
                    a[j] = 0
                s = a.sum()
                if s < 1e-10:
                    a[j] = orig
                    continue
                a_norm = a / s
                sc = _score_norm_fn(a_norm)
                if sc < best_score:
                    best_score = sc
                    best_a = a_norm.copy()
                    a = a_norm
                else:
                    a[j] = orig
                    break  # Restored, try next j
            # Re-normalize after each j
            s = a.sum()
            if s > 0:
                a = a / s
    
    return best_a, best_score


def _block_local_search(a_init, deadline, rng, n_trials=5000):
    """Block local search with simulated annealing."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    temp = 0.01
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break
        
        block_size = int(rng.integers(2, min(32, n // 2)))
        start = int(rng.integers(0, n - block_size))
        
        a_trial = a.copy()
        pert = rng.uniform(0.8, 1.2, size=block_size)
        a_trial[start:start + block_size] *= pert
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial = a_trial / a_trial.sum()
        
        sc = _score_norm_fn(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
        elif rng.random() < np.exp(-(sc - best_score) / max(temp, 1e-6)):
            a = a_trial.copy()
        
        temp *= 0.999
    
    return best_a, best_score


def _make_uniform(n):
    return np.ones(n) / n


def _make_linear(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian(n, width):
    t = np.linspace(0.0, 1.0, n)
    u = t - 0.5
    a = np.exp(-u ** 2 / (2.0 * width ** 2))
    return a / a.sum()


def _make_raised_cosine(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5) * alpha))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power(n, p):
    t = np.linspace(0.0, 1.0, n)
    u = np.abs(2.0 * t - 1.0)
    a = (1.0 - u * 0.9) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal(n, sep):
    """Two peaks separated by sep fraction of the domain."""
    t = np.linspace(0.0, 1.0, n)
    c1 = 0.5 - sep / 2
    c2 = 0.5 + sep / 2
    a = np.exp(-((t - c1) / 0.08) ** 2) + np.exp(-((t - c2) / 0.08) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_sinc_like(n, freq):
    """Sinc-like profile (first lobe of sinc)."""
    t = np.linspace(-0.5, 0.5, n)
    x = np.pi * freq * t
    a = np.sinc(freq * t) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 0: Diverse initialization
    Phase 1: Power-mean continuation (main optimization)
    Phase 2: LP-based refinement
    Phase 3: Coordinate descent + local search polishing
    Phase 4: Multi-restart and size exploration
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        _update_best(parent, sc_parent, "parent")
    else:
        n_parent = 2048
        parent = _make_linear(n_parent, 1.5)
        print(f"No parent, using linear n={n_parent}")

    n_work = n_parent  # Working size

    # --- Phase 0: Diverse initialization (40s) ---
    phase0_end = min(deadline, t_start + 40.0)
    print("Phase 0: Diverse initialization")
    candidates = []

    profiles = [
        parent,
        _make_linear(n_work, 0.5),
        _make_linear(n_work, 1.0),
        _make_linear(n_work, 2.0),
        _make_linear(n_work, 3.0),
        _make_quadratic(n_work, 1.0, 0.3),
        _make_quadratic(n_work, 1.5, -0.2),
        _make_quadratic(n_work, 0.5, 0.5),
        _make_quadratic(n_work, 2.0, -0.5),
        _make_gaussian(n_work, 0.2),
        _make_gaussian(n_work, 0.3),
        _make_gaussian(n_work, 0.4),
        _make_raised_cosine(n_work, 1.0),
        _make_raised_cosine(n_work, 0.5),
        _make_power(n_work, 1.0),
        _make_power(n_work, 2.0),
        _make_power(n_work, 3.0),
        _make_bimodal(n_work, 0.3),
        _make_bimodal(n_work, 0.5),
        _make_sinc_like(n_work, 2.0),
        _make_sinc_like(n_work, 3.0),
    ]

    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p, f"init{idx}"))
        _update_best(p, sc, f"init{idx}")

    # Sort by score
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}")

    # --- Phase 1: Power mean continuation (600s) ---
    phase1_end = min(deadline, t_start + 700.0)
    print("Phase 1: Power mean continuation")
    
    # Use top 3 candidates
    n_to_opt = 3
    per_cand = max(60.0, (phase1_end - time.monotonic()) / max(1, n_to_opt))
    
    p_list = [4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048]
    
    for ci in range(min(n_to_opt, len(candidates))):
        if time.monotonic() > phase1_end - 10.0:
            break
        sc_c, p_c, label_c = candidates[ci]
        if sc_c > best_score * 1.05:
            break
        
        dl = min(phase1_end - 10.0, time.monotonic() + per_cand)
        print(f"  Optimizing candidate {ci} ({label_c}), score={sc_c:.10f}")
        p_best, sc_best = _run_powmean_lbfgs(p_c, dl, p_list, max_wall_time=per_cand)
        _update_best(p_best, sc_best, f"powmean_{label_c}")

    # Continue on the best
    if time.monotonic() < phase1_end - 30:
        p_list_fine = [128, 256, 512, 1024, 2048, 4096]
        dl = phase1_end - 10.0
        p_best, sc_best = _run_powmean_lbfgs(best_seq, dl, p_list_fine, max_wall_time=200)
        _update_best(p_best, sc_best, "powmean_fine")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: LP refinement (300s) ---
    phase2_end = min(deadline, t_start + 1100.0)
    print("Phase 2: LP refinement")
    
    if time.monotonic() < phase2_end - 60:
        dl = phase2_end - 30.0
        # For LP, use a slightly smaller n if the current n is large
        n_lp = min(n_work, 2048)
        if n_lp < n_work:
            p_lp = _interp_profile(best_seq, n_lp)
        else:
            p_lp = best_seq.copy()
        
        p_best, sc_best = _lp_refinement(p_lp, dl, max_iters=6)
        if sc_best < best_score or n_lp == n_work:
            _update_best(p_best, sc_best, "lp_refined")
        
        # Interpolate back to full n if needed
        if n_lp < n_work and best_seq is not None:
            p_full = _interp_profile(p_best, n_work)
            sc_full = _score_norm_fn(p_full)
            _update_best(p_full, sc_full, "lp_interp_full")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Power mean again on LP result (200s) ---
    phase3_end = min(deadline, t_start + 1300.0)
    print("Phase 3: Post-LP power mean")
    
    if time.monotonic() < phase3_end - 30:
        p_list3 = [256, 512, 1024, 2048, 4096, 8192]
        dl = phase3_end - 10.0
        p_best, sc_best = _run_powmean_lbfgs(best_seq, dl, p_list3, max_wall_time=200)
        _update_best(p_best, sc_best, "post_lp_powmean")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Coordinate descent + local search (150s) ---
    phase4_end = min(deadline, t_start + 1450.0)
    print("Phase 4: Coordinate descent & local search")
    
    if time.monotonic() < phase4_end - 20:
        # Coordinate descent (use subset of coordinates for speed)
        n_cd = min(n_work, 1024)
        p_cd = _interp_profile(best_seq, n_cd) if n_cd < n_work else best_seq.copy()
        dl_cd = min(phase4_end - 60, time.monotonic() + 80.0)
        p_best, sc_best = _coordinate_descent(p_cd, dl_cd, n_sweeps=10)
        if n_cd < n_work:
            p_best_full = _interp_profile(p_best, n_work)
            sc_best = _score_norm_fn(p_best_full)
            _update_best(p_best_full, sc_best, "coord_desc")
        else:
            _update_best(p_best, sc_best, "coord_desc")

        # Local search
        if time.monotonic() < phase4_end - 10:
            dl_ls = phase4_end - 10.0
            p_best, sc_best = _block_local_search(best_seq, dl_ls, rng, n_trials=3000)
            _update_best(p_best, sc_best, "local_search")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multi-restart + size exploration (remaining) ---
    phase5_end = deadline
    print("Phase 5: Multi-restart and size exploration")
    
    if time.monotonic() < phase5_end - 30:
        n_restarts = 4
        per_restart = max(30.0, (phase5_end - time.monotonic() - 30) / max(1, n_restarts))
        
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 20:
                break
            
            # Different noise levels
            noise = 0.02 + 0.02 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=len(best_seq))
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)
            
            dl = min(phase5_end - 20, time.monotonic() + per_restart)
            p_list_r = [64, 128, 256, 512, 1024]
            p_best, sc_best = _run_powmean_lbfgs(p_start, dl, p_list_r, max_wall_time=per_restart)
            _update_best(p_best, sc_best, f"restart{i}")
        
        # Try larger n
        if time.monotonic() < phase5_end - 60:
            n_curr = len(best_seq)
            for target_n in [n_curr * 2]:
                if target_n > 8192:
                    continue
                p_interp = _interp_profile(best_seq, target_n)
                sc_interp = _score_norm_fn(p_interp)
                _update_best(p_interp, sc_interp, f"interp_n{target_n}")
                
                if time.monotonic() < phase5_end - 30:
                    dl = min(phase5_end - 30, time.monotonic() + 120.0)
                    p_list_big = [32, 64, 128, 256, 512]
                    p_best, sc_best = _run_powmean_lbfgs(p_interp, dl, p_list_big, max_wall_time=120)
                    _update_best(p_best, sc_best, f"opt_n{target_n}")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:

        sc_final = _score_fn(final_seq)

        # Validate with evaluator
        if len(final_seq) <= 16384:

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_linear(2048, 1.5)
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n=2048, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
