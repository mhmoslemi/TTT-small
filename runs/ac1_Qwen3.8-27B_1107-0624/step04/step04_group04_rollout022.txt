"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses top-k subgradient with Adam, active-set KKT polish, and diverse
initializations including chirp, two-cluster, and smooth-random profiles.
Focuses on exploring different sequence lengths and optimization basins.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _active_grad(a, k_list):
    """Compute average gradient of active convolution values."""
    n = len(a)
    g = np.zeros(n)
    for k in k_list:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    g /= len(k_list)
    return g


def _adam_optimize(a_init, deadline, lr=0.1, top_k=3, momentum=0.9):
    """Adam-style optimizer with top-k active set subgradient."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    a /= a.sum()
    
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    # Initial active set
    c = _full_convolve(a, a)
    cm = c.max()
    
    step_scale = 1.0  # scales the gradient
    
    for it in range(200000):
        if time.monotonic() > deadline:
            break
        
        t += 1
        
        # Periodically recompute active set
        if t % 50 == 1:
            c = _full_convolve(a, a)
            cm = float(c.max())
            # Use threshold-based active set
            threshold = cm * 0.98
            active = np.where(c >= threshold)[0]
            if len(active) == 0:
                active = np.array([np.argmax(c)])
            # Limit to top_k for numerical stability
            if len(active) > top_k:
                top_idx = np.argsort(c[active])[-top_k:]
                active = active[top_idx]
        
        # Compute gradient
        grad = _active_grad(a, active)
        
        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        
        # Step
        step = lr * step_scale * mh / (np.sqrt(vh) + eps)
        a_new = a - step
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new /= s
        
        # Evaluate
        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce step scale
            step_scale *= 0.9
            if step_scale < 0.01:
                step_scale = 1.0
                # Reset momentum
                m *= 0.5
                v *= 0.5
        
        # Adaptive learning rate
        if t % 500 == 0:
            lr *= 0.98
            step_scale = min(step_scale * 1.05, 1.0)
    
    return best_a, best_score


def _make_chirp_profile(n, freq1, freq2):
    """Chirp profile: smooth sweep from one frequency to another."""
    t = np.linspace(0, 1, n)
    # Linear chirp: instantaneous frequency goes from freq1 to freq2
    phase = 2 * np.pi * (freq1 * t + (freq2 - freq1) * t**2 / 2)
    # Use the absolute value of the cosine as the amplitude envelope
    a = np.abs(np.cos(phase)) ** 2
    a = a + 0.1  # Add small floor to avoid zeros
    return a / a.sum()


def _make_two_cluster(n, width, pos1=0.05, pos2=0.95, sigma=None):
    """Two Gaussian clusters."""
    if sigma is None:
        sigma = width / 3.0
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - pos1) / sigma) ** 2) + np.exp(-((t - pos2) / sigma) ** 2)
    return a / a.sum()


def _make_smooth_random(n, seed, smoothness=0.1):
    """Random profile with smooth envelope."""
    rng = np.random.default_rng(seed)
    # Generate smooth random signal using low-frequency Fourier components
    freqs = np.fft.rfftfreq(n)
    coeffs = rng.normal(size=len(freqs)) + 1j * rng.normal(size=len(freqs))
    # Apply low-pass filter
    cutoff = int(smoothness * n / 2)
    coeffs[cutoff:] = 0
    a = np.fft.irfft(coeffs * n, n)
    a = np.abs(a) + 0.01
    return a / a.sum()


def _make_exp_profile(n, alpha):
    r = 1.0 - alpha
    a = r ** np.arange(n)
    return a / a.sum()


def _make_cosine_bell(n, power=1.0):
    """Cosine bell: a_i = cos^power(pi*x) where x in [-1,1]."""
    t = np.linspace(-1, 1, n)
    a = np.cos(np.pi * t / 2) ** (2 * power)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

"""
Hybrid active-set optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines top-k exact subgradient with L-BFGS on logsumexp smoothing,
KKT linear-system solving, and diverse multi-length initialization.
Key insight: use exact active-set gradients (not softmax approximation)
with backtracking line search on the TRUE objective for accurate descent.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _slack_fn(a):
    """Compute 2*n*max_c - lambda * sum_c = 2*n*max_c - lambda (for sum=1)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() - 1.0)


def _active_grad(a, k_list):
    """Compute average gradient of active convolution positions."""
    n = len(a)
    g = np.zeros(n)
    for k in k_list:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= len(k_list)
    return g


def _get_active_set(c, top_k=3):
    """Get top-k active set from convolution."""
    if top_k >= len(c):
        return np.arange(len(c))
    return np.argsort(c)[-top_k:]


def _lbfgs_smoothed(a_init, deadline, top_k=0):
    """L-BFGS with logsumexp smoothing and beta continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    def softmax_fn(zv):
        zv = zv - np.max(zv)
        e = np.exp(zv)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return np.ones_like(zv) / len(zv)
        return e / s

    def lse_objective(zv, beta, tk):
        av = softmax_fn(zv)
        cv = _full_convolve(av, av)
        cm = float(cv.max())
        if tk > 0 and tk < len(cv):
            idx = np.argsort(cv)[-tk:]
            ct = cv[idx]
            ct_norm = ct - ct.max()
            log_sum = np.log(np.sum(np.exp(ct_norm / beta)))
            return 2.0 * n * (ct.max() + beta * log_sum)
        else:
            cv_shifted = (cv - cm) / beta
            log_sum = np.log(np.sum(np.exp(cv_shifted)))
            return 2.0 * n * (cm + beta * log_sum)

    def lse_gradient(zv, beta, tk):
        av = softmax_fn(zv)
        cv = _full_convolve(av, av)
        cm = float(cv.max())
        if tk > 0 and tk < len(cv):
            idx = np.argsort(cv)[-tk:]
            w = np.zeros(len(cv))
            ct = cv[idx]
            ct_norm = ct - ct.max()
            e = np.exp(ct_norm / beta)
            s = e.sum()
            if s < 1e-300:
                w[idx[0]] = 1.0
            else:
                w[idx] = e / s
        else:
            cv_shifted = (cv - cm) / beta
            e = np.exp(cv_shifted)
            s = e.sum()
            if s < 1e-300:
                idx = int(np.argmax(cv))
                w = np.zeros(len(cv))
                w[idx] = 1.0
            else:
                w = e / s
        W = float(np.dot(w, cv))
        # Gradient: for each active k with weight w[k], grad = 2*a shifted
        ar = av[::-1]
        w_conv = _full_convolve(w, ar)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * av * (h - W)
        if not np.all(np.isfinite(grad)):
            return np.zeros_like(zv)
        return grad

    # Beta continuation
    c = _full_convolve(a_init, a_init)
    mc = float(c.max())

    betas = [b * max(mc, 1e-10) for b in beta_starts]

    per_level = max(5.0, (deadline - time.monotonic()) / len(betas))

    for beta in betas:
        if time.monotonic() > deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level)
        try:
            result = minimize(
                lse_objective, z.copy(), args=(beta, top_k),
                method='L-BFGS-B', jac=lse_gradient,
                options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 20, 'maxfun': 800}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = softmax_fn(result.x)
                if np.all(np.isfinite(a_new)):
                    sc = _score_norm_fn(a_new)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
                        z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _adam_topk(a_init, deadline, lr=0.1, top_k=3, use_line_search=True):
    """Adam optimizer with exact top-k active set gradient and optional line search."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    if a.sum() < 1e-15:
        a = np.ones(n) / n
    else:
        a /= a.sum()

    best_a = a.copy()
    best_score = _score_norm_fn(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8

    c = _full_convolve(a, a)
    cm = float(c.max())
    active = _get_active_set(c, top_k)

    step_scale = 1.0
    stall_count = 0

    for it in range(500000):
        if time.monotonic() > deadline:
            break

        t += 1

        # Recompute active set periodically
        if t % 30 == 1:
            c = _full_convolve(a, a)
            cm = float(c.max())
            # Use threshold: all within 5% of max, capped at top_k
            threshold = cm * 0.97
            active_full = np.where(c >= threshold)[0]
            if len(active_full) > top_k:
                active = _get_active_set(c, top_k)
            else:
                active = active_full
            if len(active) == 0:
                active = np.array([np.argmax(c)])

        # Compute exact gradient of average active set
        grad = _active_grad(a, active)

        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        # Candidate step
        step = lr * step_scale * mh / (np.sqrt(vh) + eps)
        a_new = a - step
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new /= s

        # Line search (optional)
        if use_line_search:
            sc_new = _score_norm_fn(a_new)
            if np.isfinite(sc_new) and sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                stall_count = 0
                step_scale = min(step_scale * 1.02, 2.0)
            else:
                # Try smaller step
                for shrink in [0.5, 0.25, 0.1]:
                    a_try = a - step * shrink
                    a_try = np.maximum(a_try, 0)
                    st = a_try.sum()
                    if st > 1e-15:
                        a_try /= st
                    sc_try = _score_norm_fn(a_try)
                    if np.isfinite(sc_try) and sc_try < best_score:
                        best_score = sc_try
                        best_a = a_try.copy()
                        a = a_try
                        stall_count = 0
                        break
                else:
                    stall_count += 1
                    step_scale *= 0.85
                    if stall_count > 50:
                        step_scale = max(step_scale, 0.01)
                        # Perturb to escape
                        a_pert = a + 0.01 * np.random.default_rng(t).normal(size=n)
                        a_pert = np.maximum(a_pert, 0)
                        a_pert /= a_pert.sum()
                        a = a_pert
                        stall_count = 0
                        m *= 0.5
                        v *= 0.5
        else:
            a = a_new
            sc_new = _score_norm_fn(a)
            if np.isfinite(sc_new) and sc_new < best_score:
                best_score = sc_new
                best_a = a.copy()
                stall_count = 0

        # Adaptive learning rate decay
        if t % 1000 == 0:
            lr *= 0.995

    return best_a, best_score


def _kkt_refine(a_init, deadline, max_iter=200):
    """Refine using KKT conditions: at optimum, active gradients are constant on support."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    a /= a.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        threshold = cm * 0.99
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            break
        if len(active) > 8:
            active = _get_active_set(c, 8)

        # Compute average gradient over active set
        g = _active_grad(a, active)

        # At KKT: for i where a_i > 0, g_i should be constant (= lambda)
        # Direction: reduce a where g > lambda, increase where g < lambda
        support = a > 1e-12
        if not np.any(support):
            break
        g_support = g[support]
        lambda_est = float(np.median(g_support))

        # Descent direction: a_i should decrease where g_i > lambda, increase where g_i < lambda
        direction = np.zeros(n)
        direction[support] = -(g[support] - lambda_est)

        # Also allow zero elements to become positive if g_i < lambda
        zero_mask = ~support
        if np.any(zero_mask):
            direction[zero_mask] = np.maximum(lambda_est - g[zero_mask], 0) * 0.5

        # Step with line search
        best_step = 0.0
        best_sc = best_score
        for step_size in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]:
            a_try = a + step_size * direction
            a_try = np.maximum(a_try, 0)
            st = a_try.sum()
            if st < 1e-15:
                continue
            a_try /= st
            sc = _score_norm_fn(a_try)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_step = step_size
                best_a_try = a_try.copy()

        if best_step > 0:
            a = best_a_try
            best_score = best_sc
            best_a = a.copy()
        else:
            break

    return best_a, best_score


def _make_exp_profile(n, alpha):
    r = 1.0 - alpha
    a = r ** np.arange(n)
    return a / a.sum()


def _make_chirp_profile(n, freq1, freq2):
    t = np.linspace(0, 1, n)
    phase = 2 * np.pi * (freq1 * t + (freq2 - freq1) * t**2 / 2)
    a = np.abs(np.cos(phase)) ** 2
    a = a + 0.05
    return a / a.sum()


def _make_two_cluster(n, width, pos1=0.05, pos2=0.95):
    t = np.linspace(0, 1, n)
    sigma = width / 3.0
    a = np.exp(-((t - pos1) / sigma) ** 2) + np.exp(-((t - pos2) / sigma) ** 2)
    return a / a.sum()


def _make_smooth_random(n, seed, smoothness=0.1):
    rng = np.random.default_rng(seed)
    freqs = np.fft.rfftfreq(n)
    coeffs = rng.normal(size=len(freqs)) + 1j * rng.normal(size=len(freqs))
    cutoff = max(2, int(smoothness * n / 2))
    coeffs[cutoff:] = 0
    coeffs[0] = 0
    a = np.fft.irfft(coeffs * n, n)
    a = np.abs(a) + 0.01
    return a / a.sum()


def _make_cosine_bell(n, power=1.0):
    t = np.linspace(-1, 1, n)
    a = np.cos(np.pi * t / 2) ** (2 * power)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_asymmetric_shift(n, base, shift_frac):
    """Create asymmetric sequence by shifting a symmetric profile."""
    t = np.linspace(-1, 1, n)
    x = t + shift_frac
    a = np.cos(np.pi * np.clip(x, -1.02, 1.02) / 2) ** 2
    mask = np.abs(x) <= 1.0
    a[~mask] = 0
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return base
    return a / s


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _hybrid_optimize(a_init, deadline):
    """Hybrid: L-BFGS smooth -> Adam top-k -> KKT refine."""
    n = len(a_init)
    remaining = deadline - time.monotonic()
    if remaining < 10:
        return a_init, _score_norm_fn(a_init)

    # Phase A: L-BFGS with moderate smoothing
    a1_end = min(deadline, time.monotonic() + remaining * 0.4)
    p1, s1 = _lbfgs_smoothed(a_init, a1_end, top_k=0)

    # Phase B: Adam with top-k exact gradient
    a2_end = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.4)
    p2, s2 = _adam_topk(p1, a2_end, lr=0.08, top_k=3, use_line_search=True)

    # Phase C: KKT refinement
    p3, s3 = _kkt_refine(p2, deadline, max_iter=100)

    return p3, s3


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid optimization: diverse initialization + L-BFGS/Adam/KKT pipeline.
    Explores multiple sequence lengths and uses exact active-set gradients.
    """
    global height_sequence_1

    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:7.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Phase 0: Initial candidates ---
    print("=== Phase 0: Initial candidates ===")
    phase0_end = min(deadline, t_start + 30.0)
    candidates = []

    # Parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm_fn(parent)
                candidates.append((sc, parent.copy(), "parent"))
                update_best(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception:
        pass

    # Diverse profiles at multiple lengths
    target_lengths = [2048, 4096, 8192]
    for n in target_lengths:
        if time.monotonic() > phase0_end:
            break
        profs = [
            (f"chirp{n}_1_5", _make_chirp_profile(n, 1, 5)),
            (f"chirp{n}_2_10", _make_chirp_profile(n, 2, 10)),
            (f"chirp{n}_1_20", _make_chirp_profile(n, 1, 20)),
            (f"2cl{n}_5", _make_two_cluster(n, 0.05)),
            (f"2cl{n}_10", _make_two_cluster(n, 0.1)),
            (f"2cl{n}_20", _make_two_cluster(n, 0.2)),
            (f"sr{n}_05", _make_smooth_random(n, seed + n, 0.05)),
            (f"sr{n}_10", _make_smooth_random(n, seed + n + 1, 0.1)),
            (f"sr{n}_20", _make_smooth_random(n, seed + n + 2, 0.2)),
            (f"sr{n}_50", _make_smooth_random(n, seed + n + 3, 0.5)),
            (f"cb{n}_1", _make_cosine_bell(n, 1.0)),
            (f"cb{n}_2", _make_cosine_bell(n, 2.0)),
            (f"exp{n}_2", _make_exp_profile(n, 0.002)),
            (f"asym{n}", _make_asymmetric_shift(n, _make_cosine_bell(n, 1.0), 0.3)),
        ]
        for label, p in profs:
            sc = _score_norm_fn(p)
            candidates.append((sc, p, label))
            update_best(p, sc, label)

    # Interpolate parent to other lengths if available
    if parent is not None:
        for n in [2048, 4096, 8192]:
            if len(parent) != n:
                p_int = _interp_profile(parent, n)
                sc = _score_norm_fn(p_int)
                candidates.append((sc, p_int, f"parent_int_n{n}"))
                update_best(p_int, sc, f"parent_int_n{n}")

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 0: {len(candidates)} candidates, best={candidates[0][0]:.10f}")

    # --- Phase 1: Hybrid optimization on top diverse candidates ---
    print("=== Phase 1: Hybrid optimization ===")
    phase1_end = min(deadline, t_start + 1000.0)

    used_ns = set()
    opt_count = 0
    max_opts = 4

    for sc, p, label in candidates:
        if time.monotonic() > phase1_end - 100:
            break
        if opt_count >= max_opts:
            break
        n = len(p)
        if n in used_ns:
            continue
        used_ns.add(n)

        remaining = phase1_end - time.monotonic()
        per_time = min(300.0, remaining / max(1, max_opts - opt_count))
        dl = min(phase1_end - 50, time.monotonic() + per_time)

        print(f"  [{label}] n={n}, init_score={sc:.10f}, budget={per_time:.0f}s")
        p_best, sc_best = _hybrid_optimize(p, dl)
        update_best(p_best, sc_best, f"hybrid_{label}")
        opt_count += 1

    # If parent wasn't already optimized
    if parent is not None and len(parent) not in used_ns:
        if time.monotonic() < phase1_end - 100:
            dl = min(phase1_end - 50, time.monotonic() + 250.0)
            print(f"  [parent] n={len(parent)}, budget={250.0:.0f}s")
            p_best, sc_best = _hybrid_optimize(parent, dl)
            update_best(p_best, sc_best, "hybrid_parent")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Focused refinement on global best ---
    print("=== Phase 2: Focused refinement ===")
    phase2_end = min(deadline, t_start + 1400.0)

    if best_seq is not None and time.monotonic() < phase2_end:
        # Adam with different top_k values
        for tk, lr_val in [(2, 0.05), (5, 0.06), (3, 0.04)]:
            if time.monotonic() > phase2_end - 60:
                break
            dl = min(phase2_end - 30, time.monotonic() + 120.0)
            p_best, sc_best = _adam_topk(best_seq, dl, lr=lr_val, top_k=tk, use_line_search=True)
            update_best(p_best, sc_best, f"adam_tk{tk}")

        # KKT refinement
        if time.monotonic() < phase2_end - 30:
            dl = min(phase2_end - 10, time.monotonic() + 80.0)
            p_best, sc_best = _kkt_refine(best_seq, dl, max_iter=200)
            update_best(p_best, sc_best, "kkt_refine")

        # Another L-BFGS pass with top-k
        if time.monotonic() < phase2_end - 30:
            dl = min(phase2_end - 10, time.monotonic() + 100.0)
            p_best, sc_best = _lbfgs_smoothed(best_seq, dl, top_k=3)
            update_best(p_best, sc_best, "lbfgs_tk3")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Multi-restart with structured perturbations ---
    print("=== Phase 3: Multi-restart ===")
    phase3_end = min(deadline, t_start + 1650.0)

    if best_seq is not None and time.monotonic() < phase3_end:
        n_best = len(best_seq)
        restart_strategies = [
            "noise_small", "noise_med", "shift", "flip_smooth", "rescale"
        ]
        per_restart = min(80.0, (phase3_end - time.monotonic()) / 5.0)

        for i, strat in enumerate(restart_strategies):
            if time.monotonic() > phase3_end - 30:
                break

            if strat == "noise_small":
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z + 0.05 * rng.normal(size=n_best)
                p_start = np.exp(z - z.max())
                p_start /= p_start.sum()
            elif strat == "noise_med":
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z + 0.15 * rng.normal(size=n_best)
                p_start = np.exp(z - z.max())
                p_start /= p_start.sum()
            elif strat == "shift":
                # Circular shift by random amount
                shift = rng.integers(n_best // 10, n_best // 2)
                p_start = np.roll(best_seq, shift)
                p_start = np.maximum(p_start, 0)
                p_start /= p_start.sum()
            elif strat == "flip_smooth":
                # Flip and smooth
                p_flipped = best_seq[::-1]
                # Smooth with a small kernel
                kernel = np.array([0.25, 0.5, 0.25])
                p_smooth = np.convolve(p_flipped, kernel, mode='same')
                p_start = np.maximum(p_smooth, 0)
                p_start /= p_start.sum()
            else:  # rescale
                # Non-linear rescaling
                p_start = best_seq ** 1.5
                p_start = np.maximum(p_start, 0)
                p_start /= p_start.sum()

            dl = min(phase3_end - 20, time.monotonic() + per_restart)
            p_best, sc_best = _adam_topk(p_start, dl, lr=0.08, top_k=3, use_line_search=True)
            update_best(p_best, sc_best, f"r_{strat}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Length exploration ---
    print("=== Phase 4: Length exploration ===")
    phase4_end = min(deadline, t_start + 1750.0)

    if best_seq is not None and time.monotonic() < phase4_end:
        n_curr = len(best_seq)
        for tn in [1024, 2048, 4096, 8192, 16384]:
            if time.monotonic() > phase4_end - 40:
                break
            if tn == n_curr:
                continue
            p_int = _interp_profile(best_seq, tn)
            sc_int = _score_norm_fn(p_int)
            update_best(p_int, sc_int, f"interp_n{tn}")

            if sc_int < best_score * 1.03 and time.monotonic() < phase4_end - 40:
                dl = min(phase4_end - 30, time.monotonic() + 80.0)
                p_best, sc_best = _adam_topk(p_int, dl, lr=0.1, top_k=3, use_line_search=True)
                update_best(p_best, sc_best, f"opt_n{tn}")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Final ultra-fine polish ---
    print("=== Phase 5: Final polish ===")

    if best_seq is not None and time.monotonic() < deadline - 10:
        # Very low lr Adam with top_k=2
        dl = min(deadline - 5, time.monotonic() + 80.0)
        p_best, sc_best = _adam_topk(best_seq, dl, lr=0.02, top_k=2, use_line_search=True)
        update_best(p_best, sc_best, "final_tk2")

        # KKT one more time
        if time.monotonic() < deadline - 10:
            dl = min(deadline - 5, time.monotonic() + 40.0)
            p_best, sc_best = _kkt_refine(best_seq, dl, max_iter=100)
            update_best(p_best, sc_best, "final_kkt")

    # --- Final output ---
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = _make_exp_profile(4096, 0.003)
            final_score = _score_norm_fn(final_seq)

        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")

        try:
            seq_list = [float(x) for x in final_seq]
            sc_eval = float(evaluate_sequence(seq_list))
            print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  eval check: {e}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_exp_profile(4096, 0.003)
    print(f"  FALLBACK: n=4096")
    return [float(x) for x in fallback]
