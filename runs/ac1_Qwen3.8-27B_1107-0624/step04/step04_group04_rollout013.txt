```python
import cvxpy as cp

def lp_update(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    # (f * g)_k <= M for all k
    # (f * g)_k = sum_i f_i * g_{k-i}
    # This is a convolution, can be expressed as a matrix product
    # F @ g where F is a convolution matrix
    F = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(n, k+1)):
            F[k, i] = f[i]
    constraints = [F @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CLARABEL)  # or GLOP, ECOS, etc.
    return g.value
```

```python
    def matvec(x):
        W = np.fft.rfft(w, N)
        X = np.fft.rfft(x, N)
        result = np.fft.irfft(W * np.conj(X), N)
        return result[:n] + reg * x

    b = np.ones(n)
    x = np.ones(n) / n
    r = b - matvec(x)
    p = r.copy()
    rs_old = float(r @ r)

    for it in range(max_iter):
        if np.sqrt(rs_old) < tol:
            break
        Ap = matvec(p)
        pAp = float(p @ Ap)
        if pAp < 1e-15:
            break
        alpha = rs_old / pAp
        x += alpha * p
        r = r - alpha * Ap
        rs_new = float(r @ r)
        if not np.isfinite(rs_new) or rs_new >= rs_old:
            p = r
        else:
            beta = rs_new / rs_old
            p = r + beta * p
        rs_old = rs_new

    return x


def _kkt_iteration(a, beta, n_solve=None, reg=1e-8, tol=1e-10):
    """
    One KKT iteration: compute w from softmax, solve linear system, return new a.
    """
    n = len(a)
    c = _full_convolve(a, a)
    # Compute softmax weights
    c_max = float(c.max())
    c_shifted = (c - c_max) / beta
    e = np.exp(c_shifted)
    w = e / e.sum()

    # Solve Gx = 1
    x = _kkt_solve(w, n, tol=tol, max_iter=2000, reg=reg)

    # Project to non-negative and normalize
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s < 1e-15:
        return a.copy()
    x = x / s
    return x


def _kkt_continue(a_init, deadline, betas=None, max_iter_per_beta=100, reg=1e-8):
    """
    KKT iteration with beta continuation.
    """
    if betas is None:
        c0 = _full_convolve(a_init, a_init)
        cmax0 = float(c0.max())
        betas = [b * cmax0 for b in [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]]

    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    for beta in betas:
        if time.monotonic() > deadline:
            break
        for it in range(max_iter_per_beta):
            if time.monotonic() > deadline:
                break
            a_new = _kkt_iteration(a, beta, reg=reg)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
            else:
                # Try with smaller effective step (blend)
                a_blend = 0.5 * a + 0.5 * a_new
                a_blend = np.maximum(a_blend, 0)
                a_blend = a_blend / a_blend.sum()
                sc_blend = _score_norm_fn(a_blend)
                if np.isfinite(sc_blend) and sc_blend < best_score:
                    best_score = sc_blend
                    best_a = a_blend.copy()
                    a = a_blend
                else:
                    # Reduce step further
                    a_blend2 = 0.8 * a + 0.2 * a_new
                    a_blend2 = np.maximum(a_blend2, 0)
                    a_blend2 = a_blend2 / a_blend2.sum()
                    sc_blend2 = _score_norm_fn(a_blend2)
                    if np.isfinite(sc_blend2) and sc_blend2 < best_score:
                        best_score = sc_blend2
                        best_a = a_blend2.copy()
                        a = a_blend2
                    else:
                        break  # No progress, move to next beta

    return best_a, best_score


def _z_gradient(z, beta, n, top_k=0):
    """Gradient of smoothed objective w.r.t. z (logit space)."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())

    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        w = np.zeros_like(c)
        c_top = c[idx]
        c_norm = (c_top - c_top.max()) / beta
        e = np.exp(c_norm)
        s = e.sum()
        if s < 1e-300:
            w[idx[0]] = 1.0
        else:
            w[idx] = e / s
    else:
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300:
            idx = int(np.argmax(c))
            w = np.zeros_like(c)
            w[idx] = 1.0
        else:
            w = e / s

    W = float(np.dot(w, c))
    # h_i = sum_k w_k * a_{k-i} (cross-correlation)
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]

    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _z_objective(z, beta, n, top_k=0):
    """Smoothed objective in logit space."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_norm = c_top - c_top.max()
        log_sum = float(np.log(np.sum(np.exp(c_norm / beta))))
        return 2.0 * n * (c_top.max() + beta * log_sum)
    else:
        c_shifted = (c - cm) / beta
        log_sum = float(np.log(np.sum(np.exp(c_shifted))))
        return 2.0 * n * (cm + beta * log_sum)


def _run_lbfgs(a_init, deadline, max_wall_time=None, top_k=0):
    """L-BFGS with beta continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if max_wall_time is None:
        max_wall_time = max(1.0, deadline - time.monotonic())

    beta_starts = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max(max_c0, 1e-10) for b in beta_starts]

    per_level = max_wall_time / len(betas)

    for beta in betas:
        if time.monotonic() > deadline:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)
        try:
            result = minimize(
                _z_objective, z, args=(beta, n, top_k),
                method='L-BFGS-B', jac=_z_gradient,
                options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 3000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                if np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _adam_polish(a_init, deadline, lr=0.05, top_k=0, n_max_iter=50000):
    """Adam optimizer with adaptive beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-10, 0.05 * max_c)

    adaptive_k = top_k == 0
    cur_k = 5 if adaptive_k else top_k

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        use_k = cur_k if adaptive_k else top_k
        grad = _z_gradient(z, beta, n, use_k)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if t % 5 == 0:
            a = _softmax_fn(z)
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

        if t % 200 == 0:
            beta = max(1e-14, beta * 0.9)
            if adaptive_k and t % 1000 == 0:
                cur_k = min(cur_k + 3, 50)
        if t % 2000 == 0:
            if best_score < 1.51:
                lr = max(0.002, lr * 0.95)

    return best_a, best_score


def _make_exp_profile(n, alpha):
    r = 1.0 - alpha
    a = r ** np.arange(n)
    return a / a.sum()


def _make_uniform(n):
    return np.ones(n) / n


def _make_bell_profile(n, width):
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - 0.5) ** 2) / (2 * width ** 2))
    return a / a.sum()


def _make_symmetric_profile(n, p):
    """Symmetric profile: high in the middle, power decay."""
    t = np.abs(np.linspace(0, 1, n) - 0.5) * 2
    a = (1.0 - t) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _active_set_polish(a_init, deadline, max_iter=3000):
    """
    Polish using active-set KKT: identify active constraints, solve restricted system.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        # Active set with tolerance
        threshold = cm * (1.0 - 1e-4)
        active = np.where(c >= threshold)[0]

        if len(active) == 0:
            break

        # Equal weights on active set
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)

        # Solve on full support (with regularization)
        x = _kkt_solve(w, n, tol=1e-12, max_iter=1000, reg=1e-10)
        x = np.maximum(x, 0.0)
        s = float(x.sum())
        if s < 1e-15:
            break
        x = x / s

        # Line search: blend current and new
        best_t = 0.0
        best_sc = best_score
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * x
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            sc_try = _score_norm_fn(a_try)
            if np.isfinite(sc_try) and sc_try < best_sc:
                best_sc = sc_try
                best_t = t

        if best_t > 0:
            a = (1 - best_t) * a + best_t * x
            a = np.maximum(a, 0)
            a = a / a.sum()
            if best_sc < best_score:
                best_score = best_sc
                best_a = a.copy()
        else:
            # No improvement, tighten threshold
            threshold = cm * (1.0 - 5e-5)
            active = np.where(c >= threshold)[0]
            if len(active) <= 1:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase KKT + gradient optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    """
    global height_sequence_1

    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Phase 0: Initial candidates ---
    print("=== Phase 0: Initial candidates ===")
    candidates = []

    # Try parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm_fn(parent)
                candidates.append((sc, parent.copy(), len(parent), "parent"))
                update_best(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent: {e}")

    # Generate profiles at n=2048 and n=4096
    for n_try in [2048, 4096]:
        profiles = []
        profiles.append(_make_uniform(n_try))
        for alpha in [0.001, 0.003, 0.005, 0.01, 0.02]:
            profiles.append(_make_exp_profile(n_try, alpha))
        for width in [0.1, 0.15, 0.2, 0.3, 0.4]:
            profiles.append(_make_bell_profile(n_try, width))
        for p in [0.5, 1.0, 2.0, 3.0, 5.0]:
            profiles.append(_make_symmetric_profile(n_try, p))

        for idx, p in enumerate(profiles):
            sc = _score_norm_fn(p)
            candidates.append((sc, p, n_try, f"p{n_try}_{idx}"))
            update_best(p, sc, f"p{n_try}_{idx}")

        if time.monotonic() > t_start + 30:
            break

    # Interpolate parent to standard sizes
    if parent is not None:
        for n_interp in [1024, 2048, 4096]:
            if n_interp != len(parent):
                p_i = _interp_profile(parent, n_interp)
                sc_i = _score_norm_fn(p_i)
                candidates.append((sc_i, p_i, n_interp, f"parent_n{n_interp}"))
                update_best(p_i, sc_i, f"parent_n{n_interp}")

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 0 done: best={best_score:.10f}, {len(candidates)} candidates")

    # --- Phase 1: KKT continuation on top candidates ---
    print("=== Phase 1: KKT continuation ===")
    phase1_end = min(deadline, t_start + 500.0)
    count = 0
    seen = set()

    for sc, p, pn, label in candidates:
        if time.monotonic() > phase1_end:
            break
        if count >= 5:
            break
        if sc > best_score * 1.01:
            break
        key = f"{label}_n{pn}"
        if key in seen:
            continue
        seen.add(key)

        remaining = phase1_end - time.monotonic()
        per_time = max(10.0, min(100.0, remaining / max(1, 5 - count)))
        dl = min(phase1_end, time.monotonic() + per_time)
        print(f"  KKT on {label} (n={pn}), budget={per_time:.0f}s")

        p_best, sc_best = _kkt_continue(p, dl, max_iter_per_beta=80)
        update_best(p_best, sc_best, f"kkt_{label}")
        count += 1

    print(f"  Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: L-BFGS polish ---
    print("=== Phase 2: L-BFGS polish ===")
    phase2_end = min(deadline, t_start + 900.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        remaining = phase2_end - time.monotonic()
        dl = min(phase2_end, time.monotonic() + min(200.0, remaining))
        print(f"  L-BFGS on best (n={len(best_seq)}), budget={min(200.0, remaining):.0f}s")
        p_best, sc_best = _run_lbfgs(best_seq, dl, max_wall_time=min(200.0, remaining))
        update_best(p_best, sc_best, "lbfgs_phase2")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Active-set polish ---
    print("=== Phase 3: Active-set polish ===")
    phase3_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        remaining = phase3_end - time.monotonic()
        dl = min(phase3_end, time.monotonic() + min(200.0, remaining))
        print(f"  Active-set polish (n={len(best_seq)}), budget={min(200.0, remaining):.0f}s")
        p_best, sc_best = _active_set_polish(best_seq, dl, max_iter=3000)
        update_best(p_best, sc_best, "active_set")

        # Follow with short KKT
        if time.monotonic() < phase3_end - 20:
            dl2 = min(phase3_end - 5, time.monotonic() + 80.0)
            p_best2, sc_best2 = _kkt_continue(best_seq, dl2, max_iter_per_beta=50)
            update_best(p_best2, sc_best2, "kkt_post_active")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Adam fine-tune ---
    print("=== Phase 4: Adam fine-tune ===")
    phase4_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        remaining = phase4_end - time.monotonic()
        dl = min(phase4_end, time.monotonic() + min(200.0, remaining))
        print(f"  Adam (n={len(best_seq)}), budget={min(200.0, remaining):.0f}s")
        p_best, sc_best = _adam_polish(best_seq, dl, lr=0.03, top_k=0, n_max_iter=50000)
        update_best(p_best, sc_best, "adam_phase4")

        # Second pass with smaller lr
        if time.monotonic() < phase4_end - 30:
            dl2 = min(phase4_end - 10, time.monotonic() + 100.0)
            p_best2, sc_best2 = _adam_polish(best_seq, dl2, lr=0.01, top_k=10, n_max_iter=30000)
            update_best(p_best2, sc_best2, "adam2_phase4")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multi-restart ---
    print("=== Phase 5: Multi-restart ===")
    phase5_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 3
        per_restart = min(60.0, (phase5_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 15:
                break
            noise = [0.03, 0.08, 0.15][i % 3]
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            dl = min(phase5_end - 5, time.monotonic() + per_restart)
            # Quick KKT then L-BFGS
            per_kkt = per_restart * 0.5
            per_lbfgs = per_restart * 0.45

            p_k, sc_k = _kkt_continue(p_start, min(phase5_end - 5, time.monotonic() + per_kkt),
                                       max_iter_per_beta=30)
            update_best(p_k, sc_k, f"restart{i}_kkt")
            if time.monotonic() < phase5_end - 5:
                p_l, sc_l = _run_lbfgs(p_k, min(phase5_end - 5, time.monotonic() + per_lbfgs),
                                        max_wall_time=per_lbfgs, top_k=5)
                update_best(p_l, sc_l, f"restart{i}_lbfgs")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Try different sizes ---
    print("=== Phase 6: Different sizes ===")
    phase6_end = min(deadline, t_start + 1770.0)
    if time.monotonic() < phase6_end and best_seq is not None:
        n_curr = len(best_seq)
        target_ns = [1024, 2048, 4096]
        target_ns = [tn for tn in target_ns if tn != n_curr]

        for tn in target_ns:
            if time.monotonic() > phase6_end - 30:
                break
            p_interp = _interp_profile(best_seq, tn)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{tn}")

            remaining = phase6_end - time.monotonic()
            if remaining < 20:
                break
            per_opt = min(80.0, remaining * 0.5)
            dl = min(phase6_end - 10, time.monotonic() + per_opt)
            p_b, sc_b = _run_lbfgs(p_interp, dl, max_wall_time=per_opt, top_k=0)
            update_best(p_b, sc_b, f"lbfgs_n{tn}")

            if time.monotonic() < phase6_end - 15:
                dl2 = min(phase6_end - 5, time.monotonic() + min(40.0, remaining * 0.3))
                p_b2, sc_b2 = _adam_polish(p_b, dl2, lr=0.02, top_k=5, n_max_iter=15000)
                update_best(p_b2, sc_b2, f"adam_n{tn}")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final KKT + polish ---
    print("=== Phase 7: Final polish ===")
    if time.monotonic() < deadline and best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            # Final KKT with tight beta
            dl = min(deadline - 5, time.monotonic() + min(40.0, remaining * 0.4))
            c_final = _full_convolve(best_seq, best_seq)
            cmax_final = float(c_final.max())
            tight_betas = [b * cmax_final for b in [0.001, 0.0005, 0.0002, 0.0001]]
            p_f, sc_f = _kkt_continue(best_seq, dl, betas=tight_betas, max_iter_per_beta=50, reg=1e-10)
            update_best(p_f, sc_f, "final_kkt")

        if time.monotonic() < deadline - 10:
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                dl2 = min(deadline - 5, time.monotonic() + min(30.0, remaining2))
                p_f2, sc_f2 = _run_lbfgs(best_seq, dl2, max_wall_time=max(5.0, remaining2 - 5), top_k=10)
                update_best(p_f2, sc_f2, "final_lbfgs")

    # --- Final validation ---
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s <= 0.01:
            final_seq = _make_exp_profile(2048, 0.003)
            s = float(final_seq.sum())
        final_score = _score_fn(final_seq)
        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}, sum={s:.4f}")

        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 8192:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  eval check failed: {e}")

        return [float(x) for x in final_seq]

    fallback = _make_exp_profile(2048, 0.003)
    print(f"  FALLBACK: n=2048")
    return [float(x) for x in fallback]
```
