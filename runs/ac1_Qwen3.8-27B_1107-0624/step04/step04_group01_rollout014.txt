from scipy.optimize import linprog
import numpy as np

def lp_direction(a, M):
    """Find the LP direction: max sum(g) s.t. g >= 0, (a*g)[k] <= M for all k."""
    n = len(a)
    # Variables: g[0], ..., g[n-1]
    # Objective: minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: (a*g)[k] <= M for k = 0, ..., 2n-2
    # (a*g)[k] = sum_i a[i]*g[k-i] (valid indices)
    # For constraint k: sum_i a[i]*g[k-i] <= M
    # In terms of g[j]: sum_{i: k-i in [0,n-1]} a[i]*g[k-i] <= M
    # Let j = k-i, so i = k-j: sum_{j: max(0,k-n+1) <= j <= min(n-1,k)} a[k-j]*g[j] <= M
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A_ub[k, j] = a[k-j]
    
    b_ub = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    return None
```

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _cross_convolve(a, g):
    """Compute (a * g) full convolution."""
    return _full_convolve(a, g)

def _lp_direction(a, M, max_iter=50000):

    rows = []

    vals = []
    for j in range(n):
        for k in range(j, j + n):  # k = j + i where i = 0, ..., n-1
            if k < num_constraints:
                rows.append(k)
                cols.append(j)

    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',

If the LP returns g = 0 (or close to it), it means the current M is too tight to allow any progress, and we should increase M.

Actually, the LP is:
max sum(g) s.t. g >= 0, A*g <= M*1

If the optimal g has sum(g) > 0, we can normalize it to sum = 1 and use it as a direction. If sum(g) = 0, we're stuck.

In practice, as long as M is the current max of a*a (which it is), the LP should have a non-trivial solution because we can always take g to be a small multiple of a (since (a*a)[k] <= M means a is "compatible" with M, and g = eps*a gives (a*g) = eps*(a*a) <= eps*M < M).

Wait, (a*(eps*a))[k] = eps*(a*a)[k] <= eps*M < M. So g = eps*a is feasible for any eps > 0. The LP will find something at least as good.

OK so the LP is always feasible and has a non-trivial solution. Good.

Now, the paper's iteration is:
1. f = current (normalized to sum = sqrt(2n) or whatever)
2. M = max(f*f)
3. g_raw = LP solution (max sum(g) s.t. g >= 0, f*g <= M)
4. g = g_raw / sum(g_raw) * sqrt(2n) (normalize g to same L1 norm as f)
5. f_new = (1-t)*f + t*g for small t
6. Repeat

Hybrid LP + Adam optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses the LP-based direction-finding from the literature combined with
Adam gradient descent for polishing.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lp_solve(a, M):
    """Solve: max sum(g) s.t. g >= 0, (a*g)[k] <= M for all k."""
    n = len(a)
    num_c = 2 * n - 1
    
    # Build sparse constraint matrix
    # (a*g)[k] = sum_{j} a[k-j] * g[j] for valid (k,j)
    # A[k, j] = a[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # k ranges from j to j+n-1
        k_start = j
        k_end = min(j + n, num_c)
        for k in range(k_start, k_end):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(num_c, n))
    
    c_obj = -np.ones(n)
    b_ub = np.full(num_c, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 30, 'presolve': True})
    
    if res.success and res.x is not None:
        g = res.x
        g = np.maximum(g, 0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None


def _adam_step(z, lr, t, m_buf, v_buf, b1, b2, eps, grad):
    m_buf = b1 * m_buf + (1.0 - b1) * grad
    v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
    mh = m_buf / (1.0 - b1 ** t)
    vh = v_buf / (1.0 - b2 ** t)
    z_new = z - lr * mh / (np.sqrt(vh) + eps)
    return z_new, m_buf, v_buf

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
