```python
LP-based iterative refinement with Adam polishing for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses linear programs to find optimal "directions" that reduce the peak
autocorrelation, combined with smooth gradient-based polishing.

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _build_constraint_matrix(a):
    """Build the (2n-1) x n constraint matrix A where A[k,j] = a[k-j]."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - (n-1))
        j_max = min(k, n-1)
        if j_min <= j_max:
            A[k, j_min:j_max+1] = a[k - np.arange(j_min, j_max+1)]
    return A

def solve_lp_direction(a, M):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M, b >= 0."""
    n = len(a)
    A = _build_constraint_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return a.copy()  # Fallback

def line_search(a, b, max_iter=50):
    """Find t in [0,1] minimizing score of (1-t)*a + t*b (both sum to 1)."""
    n = len(a)
    ca = _full_convolve(a, a)
    cab = _full_convolve(a, b)
    cb = _full_convolve(b, b)
    
    def obj(t):
        ct = (1-t)**2 * ca + 2*t*(1-t) * cab + t**2 * cb
        return ct.max()
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc, fd = obj(c), obj(d)
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj(d)
    return 0.5 * (lo + hi)

    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = c.max()

        # Gradient w.r.t. z: 2*a[i]*(a[k_star-i] - c[k_star])
        a_shifted = np.zeros(n)
        # a[k_star - i] for i = 0, ..., n-1
        # k_star - i ranges from k_star to k_star-(n-1)
        # Valid when 0 <= k_star - i < n, i.e., k_star-(n-1) <= i <= k_star
        i_min = max(0, k_star - (n-1))
        i_max = min(k_star, n-1)
        if i_min <= i_max:
            a_shifted[i_min:i_max+1] = a[k_star - np.arange(i_min, i_max+1)]
        
        grad_z = 2.0 * a * (a_shifted - cm)

        m_buf = b1 * m_buf + (1-b1) * grad_z
        v_buf = b2 * v_buf + (1-b2) * (grad_z * grad_z)
        mh = m_buf / (1 - b1**t_count)
```

Looking at this problem, I need to minimize `2*n*max(conv(a,a))/sum(a)^2` for non-negative sequences. The key insight is that this is a "peak autocorrelation minimization" problem. The previous approach used Adam with log-sum-exp smoothing. I'll take a different approach using **LP-based iterative refinement** — solving a linear program at each step to find the optimal "dual direction" that reduces the peak, combined with line search and gradient polishing.

The LP at each step is: maximize `sum(b)` subject to `(a*b)[k] <= M` for all lags k, `b >= 0`. This gives a global direction (not just local gradient) that exploits the full constraint structure.

```python
"""
LP-guided iterative refinement with subgradient/Adam polishing for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses linear programs to find optimal dual
directions that reduce peak autocorrelation, combined with smooth gradient
polishing. The key idea: at each step, solve an LP to find the sequence b
that maximizes sum(b) subject to (a*b)[k] <= M, then blend a toward b.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _build_constraint_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[k - np.arange(j_min, j_max + 1)]
    return A


def solve_lp_direction(a, M):
    n = len(a)
    A = _build_constraint_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs')
    if res.success:
        x = np.maximum(res.x, 0)
        if x.sum() > 1e-15:
            return x
    return a.copy()


def line_search_conv(a, b, max_iter=60):
    ca = _full_convolve(a, a)
    cab = _full_convolve(a, b)
    cb = _full_convolve(b, b)

    def obj(t):
        ct = (1 - t) ** 2 * ca + 2 * t * (1 - t) * cab + t * t * cb
        return ct.max()

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c1 = hi - gr * (hi - lo)
    d1 = lo + gr * (hi - lo)
    fc = obj(c1)
    fd = obj(d1)
    for _ in range(max_iter):
        if fc < fd:
            hi = d1
            d1 = c1
            fd = fc
            c1 = hi - gr * (hi - lo)
            fc = obj(c1)
        else:
            lo = c1
            c1 = d1
            fc = fd
            d1 = lo + gr * (hi - lo)
            fd = obj(d1)
    return 0.5 * (lo + hi)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _adam_smooth(a_init, deadline, beta_schedule, lr=0.02, max_iter=200000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_count = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    cur_lr = lr

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_phase = 0
        while it_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_phase += 1
            t_count += 1

            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shift = (c - cm) / beta
            e = np.exp(c_shift)
            s_e = float(e.sum())
            if s_e < 1e-300 or not np.isfinite(s_e):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_e

            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t_count)
            vh = v_buf / (1.0 - b2 ** t_count)

            update = cur_lr * mh / (np.sqrt(vh) + eps)
            un = np.linalg.norm(update)
            if un > 1.0:
                update = update * (1.0 / un)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t_count % 100 == 0:
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

            if it_phase % 10000 == 0:
                cur_lr = max(0.005, cur_lr * 0.95)

    a_final = _softmax(z)
    sc = _score_norm(a_final)
    if sc < best_score:
        best_score = sc
        best_a = a_final.copy()

    return best_a, best_score


def _lp_phase(a, deadline, max_iters=5000):
    """Run LP-based iterative refinement until deadline."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    prev_score = best_score
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()

        c = _full_convolve(a, a)
        M = float(c.max())

        b = solve_lp_direction(a, M)
        b_sum = float(b.sum())
        if b_sum < 1e-10:
            break
        b_norm = b / b_sum

        t_opt = line_search_conv(a, b_norm, max_iter=50)
        a_new = (1.0 - t_opt) * a + t_opt * b_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc = _score_norm(a_new)
        if not np.isfinite(sc):
            break

        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            stagnant = 0
            if it % 20 == 0 or stagnant < 5:
                elapsed = time.monotonic() - t0
                print(f"    LP[{it}]: score={sc:.10f}, t={t_opt:.4f}, "
                      f"iter_time={elapsed:.2f}s, total={time.monotonic()-time_start:.1f}s")
        else:
            stagnant += 1

        a = a_new

        if stagnant > 50:
            # Try a perturbation to escape
            rng = np.random.default_rng(it)
            a_pert = a + 0.01 * rng.standard_normal(n)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            sc_pert = _score_norm(a_pert)
            if sc_pert < best_score:
                a = a_pert
                stagnant = 0
            else:
                break

    return best_a, best_score


def _adam_subgrad(a_init, deadline, lr=0.5, max_iter=50000):
    """Adam with exact subgradient of max (no smoothing)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_count = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    cur_lr = lr

    it = 0
    while it < max_iter and time.monotonic() < deadline:
        it += 1
        t_count += 1

        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        k_star = int(c.argmax())

        a_shifted = np.zeros(n)
        i_min = max(0, k_star - (n - 1))
        i_max = min(k_star, n - 1)
        if i_min <= i_max:
            a_shifted[i_min:i_max + 1] = a[k_star - np.arange(i_min, i_max + 1)]

        grad_z = 2.0 * a * (a_shifted - cm)

        if not np.all(np.isfinite(grad_z)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t_count)
        vh = v_buf / (1.0 - b2 ** t_count)

        update = cur_lr * mh / (np.sqrt(vh) + eps)
        un = np.linalg.norm(update)
        if un > 2.0:
            update = update * (2.0 / un)

        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if it % 100 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

        if it % 20000 == 0:
            cur_lr = max(0.1, cur_lr * 0.8)

    a_final = _softmax(z)
    sc = _score_norm(a_final)
    if sc < best_score:
        best_score = sc
        best_a = a_final.copy()

    return best_a, best_score
```
