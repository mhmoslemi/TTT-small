def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _logsumexp_objective(z, n, beta, topk_blend=0.0):
    """Compute logsumexp-smoothed objective and its gradient."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * cm, None
    log_sum = float(np.log(s))
    obj = 2.0 * n * (cm + beta * log_sum)
    return obj


def _logsumexp_grad(z, n, beta):
    """Compute gradient of logsumexp-smoothed objective."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    g = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(g)):
        return np.zeros_like(z)
    return g

            e_ = np.exp(c_shifted)
            s_ = e_.sum()

            c_ = _full_convolve(a_, a_)

            W = float(np.dot(w, c_))
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_ * (h - W)

                return np.zeros_like(z_)
            return g

def _fista_smoothed(a_init, deadline, rng, lr=0.05, beta_frac=0.001, n_iters=50000):
    """FISTA (Nesterov accelerated) on the logsumexp-smoothed objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    beta = beta_frac * max_c0

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    y = z.copy()
    t = 1.0
    z_prev = z.copy()

    for it in range(n_iters):
        if time.monotonic() > deadline - 1.0:
            break

        # Compute gradient at y
        grad = _logsumexp_grad(y, n, beta)

        # Gradient step
        y = y - lr * grad
        y = np.clip(y, -100, 100)
        y = y - y.mean()

        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z = y + ((t - 1.0) / t_new) * (y - z_prev)
        z_prev = y.copy()
        y = z.copy()
        t = t_new

        # Periodic evaluation
        if (it + 1) % 50 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                # Adapt beta: decrease if improving
                c_curr = _full_convolve(a_new, a_new)
                top3 = np.sort(c_curr)[-3:]
                if len(top3) >= 2 and (top3[0] - top3[-1]) / max(top3[-1], 1e-30) < 0.01:
                    beta *= 0.8
                    lr = max(0.001, lr * 0.95)

    return best_a, best_score

def _active_set_polish(a_init, deadline, rng, k_active=30, n_iters=2000):
    """Polish by directly minimizing the max of top-K active lags using subgradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    a = a_init.copy()
    c = _full_convolve(a, a)
    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)

    for it in range(n_iters):
        if time.monotonic() > deadline - 1.0:
            break

        # Find active lags
        active_set = np.argsort(c)[-k_active:]

        # Compute subgradient averaged over active lags
        grad = np.zeros(n)
        for k_val in active_set:
            k_val = int(k_val)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * a[k_val - hi:k_val - lo + 1][::-1]
        grad /= len(active_set)
        grad *= 4.0 * n

        # Momentum update
        v = momentum * v + grad
        z = z - lr * v
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        # Periodic evaluation
        if (it + 1) % 20 == 0:

            c = _full_convolve(a, a)
            sc = float(2.0 * n * c.max())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                # Decay learning rate
                lr *= 0.995

        # Adaptive: if no improvement for a while, reduce lr
        if it % 200 == 0 and it > 0:
            lr = max(0.001, lr * 0.8)
            momentum = min(0.99, momentum + 0.01)

    return best_a, best_score

                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    t = np.linspace(0.0, 1.0, 2048)
    fallback = np.sin(np.pi * t) ** 2
    fallback = np.maximum(fallback, 1e-12)
    fallback = fallback / fallback.sum()
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n=2048, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
