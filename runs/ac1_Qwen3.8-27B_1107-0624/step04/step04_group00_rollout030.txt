Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines LP-based updates, subgradient methods, L-BFGS continuation,
and coordinate descent for fine polishing.
Key innovations: (1) LP-based multiplicative updates for fast global progress,

import numpy as np
import time
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    m = float(c.max())
    return float(2.0 * n * m)


def _score_direct(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = float(a.sum())
    if s > 1e-15:
        return a / s
    return a.copy()


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _subgrad_max_c(a):
    """Exact subgradient of max(conv(a,a)) at the max position."""
    n = len(a)
    c = _full_convolve(a, a)
    k_star = int(np.argmax(c))
    indices = np.arange(n)
    partners = k_star - indices
    valid = (partners >= 0) & (partners < n)
    grad = np.zeros(n)
    grad[valid] = 2.0 * a[partners[valid]]
    return grad, c


def _smoothed_grad(a, beta):
    """Gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    w = np.exp((c - cm) / beta)
    w = w / w.sum()
    # Gradient: sum_k w_k * 2 * a_{k-i} for each i
    # = 2 * cross_correlate(w, a) at appropriate positions
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    return 2.0 * h, c


def _lp_update(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns b or None."""
    n = len(a)
    # Build constraint matrix A: (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'time_limit': 60})
    if res.success and res.x is not None:
        b = res.x
        if b.sum() > 1e-10:
            return b
    return None


def _multiplicative_weights_step(a, beta, alpha):
    """One step of multiplicative weights (LP-inspired)."""
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-15:
        return a.copy()
    # Weights proportional to violation
    w = np.exp((c - cm) / beta)
    w = w / w.sum()
    # Compute "pressure" g_i = 2 * sum_k w_k * a_{k-i}
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    # Multiplicative update
    a_new = a * np.exp(-alpha * h)
    a_new = np.maximum(a_new, 1e-30)
    return _normalize(a_new)


def _coord_descent_pass(a, max_passes=5, deadline=None):
    """Cyclic coordinate descent with 1D optimization per coordinate."""
    n = len(a)

        # Find optimal a_i given all others fixed
        # a_i appears in c_k for k in [i, i+n-1]

        k_end = min(2 * n - 1, i + n - 1)

        if len(k_range) == 0:
            continue
        
        partner = k_range - i  # indices into a for the partner term
        # Verify partner is valid (should be 0 to n-1)
        if partner.min() < 0 or partner.max() >= n:
            continue

        c_affected = c[k_range].copy()
        a_i = a[i]

        # For k != 2i: c_k includes 2*a_i*a[partner]
        # For k == 2i: c_k includes a_i^2

            a_i_contrib[idx_2i[0]] = a_i * a_i

        all_affected_mask = np.zeros(2 * n - 1, dtype=bool)
        all_affected_mask[k_range] = True
        if all_affected_mask.any():
            unaffected_max = float(c[~all_affected_mask].max()) if (~all_affected_mask).any() else 0.0
        else:
            unaffected_max = 0.0

        # subject to t >= 0
        # Use golden section search
        a_lo = 0.0
        a_hi = max(2.0 * a_i, 4.0 / n)

            c_new = base + 2.0 * t * a[partner]

                c_pt = d_pt
                fc = fd
                d_pt = a_lo + gr * (a_hi - a_lo)

    n0 = len(p0)
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return _normalize(p)


def propose_candidate(seed=42, budget_s=1800):

    Multi-phase optimization:
    Phase 1: LP-based updates (fast global improvement)
    Phase 2: L-BFGS with continuation (smooth refinement)
    Phase 3: Subgradient with momentum (handle non-smooth max)
    Phase 4: Coordinate descent (fine polishing)
    Phase 5: Final multiplicative weights polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def _update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = _normalize(parent_raw)
    except Exception:

        _update_best(parent, sc_parent, "parent")

    n = len(a_cur)

    lp_works = False
    t_lp_start = time.monotonic()
    try:
        c = _full_convolve(a_cur, a_cur)
        M = float(c.max())
        b = _lp_update(a_cur, M)
        lp_time = time.monotonic() - t_lp_start
        print(f"  LP solve time: {lp_time:.1f}s")
        if b is not None:
            lp_works = True
            b_norm = _normalize(b)
            t_mix = 0.5
            a_new = (1 - t_mix) * a_cur + t_mix * b_norm
            a_new = np.maximum(a_new, 1e-15)
            a_new = _normalize(a_new)
            sc_new = _score_norm(a_new)

            a_cur = best_seq.copy()
    except Exception as e:

                    a_new = _normalize(a_new)
                    sc_new = _score_norm(a_new)
                    _update_best(a_new, sc_new, "lp_iter")
                    t_mix = max(0.2, t_mix * 0.95)
                else:

            except Exception:
