def _lp_step(a, max_iter=1):
    """
    One LP-based step: find direction d that minimizes the linearized max convolution.
    a must be normalized (sum=1, non-negative).
    Returns new a or None.
    """
    n = len(a)
    c = _full_convolve(a, a)
    
    # Build the constraint matrix rows
    # For each k in 0..2n-2: c[k] + 2*sum_i a[k-i]*d[i] <= M
    # Variables: x = [d_0, ..., d_{n-1}, M]
    # Minimize M -> c_vec = [0,...,0,1]
    
    # A_ub shape: (2n-1 + n) x (n+1)
    # First 2n-1 rows: convolution constraints
    # Next n rows: -d[i] <= a[i] (i.e., d[i] >= -a[i])
    
    nrows = 2 * n - 1 + n
    ncols = n + 1
    
    c_vec = np.zeros(ncols)
    c_vec[-1] = 1.0  # minimize M
    
    A_ub = np.zeros((nrows, ncols))
    b_ub = np.zeros(nrows)
    
    # Convolution constraints
    for k in range(2 * n - 1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, i] += 2.0 * a[j]
        A_ub[k, n] = -1.0  # coefficient of M
        b_ub[k] = -c[k]
    
    # Non-negativity: d[i] >= -a[i]  =>  -d[i] <= a[i]
    for i in range(n):
        A_ub[2*n-1+i, i] = -1.0
        b_ub[2*n-1+i] = a[i]
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, ncols))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: M >= 0
    bounds = [(None, None)] * n + [(0, None)]
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            return None
        d = res.x[:n]
        # Normalize step
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-12:
            return None
        # Line search
        for alpha in [0.3, 0.2, 0.1, 0.05, 0.02]:
            a_new = a + alpha * d / d_norm * d_norm  # step size alpha * ||d||/||d|| * d = alpha*d

            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:

            a_new = a_new / s
            sc = _score_norm_fn(a_new)
            if sc < _score_norm_fn(a):

        return None
    except Exception:
        return None

Optimization for minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
Uses LP-based linearized max reduction as primary method,
with L-BFGS, Adam, and KKT polishing.
"""
import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    return e / s
