def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):

    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    a_norm = a / s
    c = _full_convolve(a_norm, a_norm)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_conv_grad(a, k_indices, n):
    """Compute the averaged gradient of the top-k convolution values w.r.t. a.
    grad_k[i] = 2*a[k-i] if 0<=k-i<n else 0 (for each k).
    Returns h[i] = (1/len(k_indices)) * sum_k grad_k[i].
    """
    h = np.zeros(n, dtype=np.float64)
    for k_val in k_indices:
        k_int = int(k_val)
        lo = max(0, k_int - (n - 1))
        hi = min(n - 1, k_int)
        if lo <= hi:
            h[lo:hi + 1] += 2.0 * a[k_int - hi:k_int - lo + 1][::-1]
    h /= len(k_indices)
    return h


def _z_objective_smooth(z, beta, n, top_k):
    """Smoothed objective in logit space using top-k logsumexp."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_shift = (c_top - c_top.max()) / beta
        log_sum = float(np.log(np.sum(np.exp(c_shift))))
        val = 2.0 * n * (c_top.max() + beta * log_sum)
    else:
        c_shift = (c - cm) / beta
        log_sum = float(np.log(np.sum(np.exp(c_shift))))
        val = 2.0 * n * (cm + beta * log_sum)
    return val


def _z_gradient_smooth(z, beta, n, top_k):
    """Gradient of smoothed objective w.r.t. z (logit space)."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())

    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_norm = (c_top - c_top.max()) / beta
        e = np.exp(c_norm)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w_active = np.zeros(len(c))
            w_active[idx[0]] = 1.0
        else:
            w_active = np.zeros(len(c))
            w_active[idx] = e / s
        active_vals = idx
    else:
        c_shift = (c - cm) / beta
        e = np.exp(c_shift)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w_active = np.zeros(len(c))
            w_active[int(np.argmax(c))] = 1.0
            active_vals = [int(np.argmax(c))]
        else:
            w_active = e / s
            active_vals = np.where(w_active > 1e-10)[0]

    W = float(np.dot(w_active, c))
    # h[i] = sum_k w_k * 2*a[k-i] (valid range)
    h = np.zeros(n, dtype=np.float64)
    for k_val in active_vals:
        w_k = w_active[int(k_val)]
        if w_k < 1e-12:
            continue
        k_int = int(k_val)
        lo = max(0, k_int - (n - 1))
        hi = min(n - 1, k_int)
        if lo <= hi:
            h[lo:hi + 1] += w_k * 2.0 * a[k_int - hi:k_int - lo + 1][::-1]

    # Gradient in logit space: d/dz = a * (4*n*h - 4*n*W*a) = 4*n*a*(h - W)
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs_adaptive(a_init, deadline, max_wall_time=None, use_top_k=True):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    a = a_init / a_init.sum()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    best_a = a.copy()
    best_score = _score_norm_fn(a)

    c0 = _full_convolve(a, a)
    max_c0 = float(c0.max())

    if max_wall_time is None:
        max_wall_time = max(1.0, deadline - time.monotonic())

    # Adaptive beta: start large, reduce based on progress
    beta = 0.1 * max(max_c0, 1e-10)
    t_start = time.monotonic()
    no_improve_time = 0.0
    last_improve_time = t_start

    cur_k = 5 if use_top_k else 0
    max_k = 50 if use_top_k else 0

    while time.monotonic() < min(deadline, t_start + max_wall_time):
        remaining = min(deadline, t_start + max_wall_time) - time.monotonic()
        if remaining < 2.0:
            break

        # Per-level time budget
        per_level = min(30.0, remaining * 0.4)
        level_deadline = time.monotonic() + per_level

        try:
            result = minimize(
                _z_objective_smooth, z, args=(beta, n, cur_k),
                method='L-BFGS-B', jac=_z_gradient_smooth,
                options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-8,
                         'maxls': 40, 'maxfun': 1000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                if np.all(np.isfinite(a_new)):
                    sc = _score_norm_fn(a_new)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
                        z = z_new.copy()
                        last_improve_time = time.monotonic()
                    else:
                        no_improve_time += per_level
        except Exception:
            pass

        # Adaptive beta reduction
        elapsed_level = time.monotonic() - (level_deadline - per_level)
        if time.monotonic() - last_improve_time > 15.0:
            beta *= 0.5
            no_improve_time = 0.0
        else:
            beta *= 0.7

        # Increase top_k over time
        if use_top_k and int(time.monotonic() - t_start) % 20 < per_level:
            cur_k = min(cur_k + 2, max_k)

        # Update a from best for next iteration
        a = best_a
        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()

    return best_a, best_score


def _kkt_equalize(a_init, deadline, max_iter=20000, top_k_init=5):
    """KKT-based peak equalization with backtracking line search."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    cur_k = top_k_init
    stall_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        # Dynamic active set: top-k peaks
        if cur_k < len(c):
            idx = np.argsort(c)[-cur_k:]
        else:
            idx = np.arange(len(c))

        # Compute KKT direction
        h = np.zeros(n, dtype=np.float64)
        for k_val in idx:
            k_int = int(k_val)
            lo = max(0, k_int - (n - 1))
            hi = min(n - 1, k_int)
            if lo <= hi:
                h[lo:hi + 1] += a[k_int - hi:k_int - lo + 1][::-1]
        h /= len(idx)

        W = float(np.dot(h, a))
        direction = h - W

        # Center the direction (to preserve sum)
        direction -= direction.mean()

        max_dir = float(np.abs(direction).max())
        if max_dir < 1e-15:
            break

        direction = direction / max_dir

        # Backtracking line search
        step = 0.1
        z_curr = np.log(np.maximum(a, 1e-15))
        z_curr = z_curr - z_curr.mean()
        found = False
        for _ls in range(15):
            z_new = z_curr - step * direction
            z_new = np.clip(z_new, -60, 60)
            z_new = z_new - z_new.mean()
            a_new = _softmax_fn(z_new)
            sc = _score_norm_fn(a_new)

            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                z_curr = z_new
                found = True
                stall_count = 0
                break
            step *= 0.5

        if not found:
            stall_count += 1
            if stall_count > 50:
                # Increase top_k and reset
                cur_k = min(cur_k + 3, 100)
                stall_count = 0
                # Small random perturbation
                z_curr = z_curr + 0.01 * np.random.default_rng(it).normal(size=n)
                z_curr = np.clip(z_curr, -60, 60)
                z_curr = z_curr - z_curr.mean()
                a = _softmax_fn(z_curr)

        # Gradually increase top_k
        if it % 200 == 0 and it > 0:
            cur_k = min(cur_k + 1, 100)

        # Early termination if converged
        if it > 500 and best_score < 1.503:
            break

    return best_a, best_score


def _adam_refine(a_init, deadline, lr_init=0.03, top_k_init=5):
    """Adam optimizer with adaptive parameters for final polishing."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    best_a = a.copy()
    best_score = _score_norm_fn(a)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a, a)
    max_c = float(c.max())
    beta = 0.05 * max_c
    cur_k = top_k_init
    lr = lr_init

    t_start = time.monotonic()
    last_improve = t_start

    for it in range(100000):
        if time.monotonic() > deadline:
            break

        t += 1
        a = _softmax_fn(z)
        grad = _z_gradient_smooth(z, beta, n, cur_k)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            last_improve = time.monotonic()

        # Adaptive schedule
        if t % 200 == 0:
            beta = max(1e-12, beta * 0.85)
            if time.monotonic() - last_improve > 20.0:
                lr = max(0.002, lr * 0.9)
                cur_k = min(cur_k + 2, 60)

    return best_a, best_score


def _make_profiles(n, rng):
    """Generate diverse initial profiles."""
    profiles = []

    # Uniform
    profiles.append(np.ones(n) / n)

    # Exponential decay (both directions)
    for alpha in [0.001, 0.003, 0.01, 0.03]:
        a = np.exp(-alpha * np.arange(n) / n)
        profiles.append(a / a.sum())
        a2 = np.exp(-alpha * (n - 1 - np.arange(n)) / n)
        profiles.append(a2 / a2.sum())

    # Linear
    for slope in [-2, -1, 0, 1, 2]:
        t = np.linspace(0, 1, n)
        a = np.maximum(1.0 + slope * (t - 0.5), 0.01)
        profiles.append(a / a.sum())

    # Quadratic / parabolic
    t = np.linspace(0, 1, n)
    for a1, a2 in [(-1, 0), (0, -1), (1, -2), (-2, 1), (0.5, -1)]:
        a = np.maximum(1.0 + a1 * (2*t - 1) + a2 * (2*t - 1)**2, 0.01)
        profiles.append(a / a.sum())

    # Sine-modulated
    for freq in [1, 2, 3]:
        a = 1.0 + 0.5 * np.sin(2 * np.pi * freq * t)
        a = np.maximum(a, 0.01)
        profiles.append(a / a.sum())

    # Random smoothed
    for _ in range(3):
        a = np.abs(rng.normal(size=n)) + 0.1
        # Smooth with a window
        w = np.ones(20) / 20
        a = np.convolve(a, w, mode='same')
        a = np.maximum(a, 0.01)
        profiles.append(a / a.sum())

    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: adaptive L-BFGS + KKT peak equalization + Adam polish.
    Minimizes 2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
    """
    global height_sequence_1

    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Phase 0: Initial candidates ---
    print("=== Phase 0: Initial candidates ===")
    phase0_end = min(deadline, t_start + 30.0)

    # Try parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm_fn(parent)
                update_best(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent: {e}")

    n_primary = 4096
    if parent is not None:
        n_primary = len(parent)

    candidates = []
    profiles = _make_profiles(n_primary, rng)

    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p.copy(), f"prof{idx}"))
        update_best(p, sc, f"prof{idx}")

    # Add parent if available
    if parent is not None:
        candidates.append((_score_norm_fn(parent), parent.copy(), "parent"))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 0 done: best={best_score:.10f}, {len(candidates)} candidates")

    # --- Phase 1: L-BFGS on top candidates ---
    print("=== Phase 1: Adaptive L-BFGS ===")
    phase1_end = min(deadline, t_start + 400.0)
    n_attempts = 0

    for sc, p, label in candidates:
        if time.monotonic() > phase1_end - 30:
            break
        if n_attempts >= 5:
            break
        if sc > best_score * 1.05:
            break

        remaining = phase1_end - time.monotonic()
        per_time = max(30.0, min(120.0, remaining / max(1, 5 - n_attempts)))
        dl = min(phase1_end - 5, time.monotonic() + per_time)

        print(f"  L-BFGS on {label}, budget={per_time:.0f}s")
        p_best, sc_best = _run_lbfgs_adaptive(p, dl, max_wall_time=per_time, use_top_k=True)
        update_best(p_best, sc_best, f"lbfgs_{label}")
        n_attempts += 1

    print(f"  Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: KKT peak equalization ---
    print("=== Phase 2: KKT Peak Equalization ===")
    phase2_end = min(deadline, t_start + 800.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        remaining = phase2_end - time.monotonic()
        dl = min(phase2_end - 5, time.monotonic() + remaining)
        print(f"  KKT on best (n={len(best_seq)}), budget={remaining:.0f}s")
        p_best, sc_best = _kkt_equalize(best_seq, dl, max_iter=30000, top_k_init=5)
        update_best(p_best, sc_best, "kkt1")

        # Second KKT pass with different top_k
        if time.monotonic() < phase2_end - 30:
            remaining = phase2_end - time.monotonic()
            dl = min(phase2_end - 5, time.monotonic() + remaining)
            p_best2, sc_best2 = _kkt_equalize(best_seq, dl, max_iter=20000, top_k_init=15)
            update_best(p_best2, sc_best2, "kkt2")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Adam refinement ---
    print("=== Phase 3: Adam refinement ===")
    phase3_end = min(deadline, t_start + 1100.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        remaining = phase3_end - time.monotonic()
        dl = min(phase3_end - 5, time.monotonic() + remaining)
        print(f"  Adam on best, budget={remaining:.0f}s")
        p_best, sc_best = _adam_refine(best_seq, dl, lr_init=0.02, top_k_init=5)
        update_best(p_best, sc_best, "adam1")

        if time.monotonic() < phase3_end - 30:
            remaining = phase3_end - time.monotonic()
            dl = min(phase3_end - 5, time.monotonic() + remaining)
            p_best2, sc_best2 = _adam_refine(best_seq, dl, lr_init=0.01, top_k_init=10)
            update_best(p_best2, sc_best2, "adam2")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Multi-restart ---
    print("=== Phase 4: Multi-restart ===")
    phase4_end = min(deadline, t_start + 1400.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        n_restarts = 5
        per_restart = min(50.0, (phase4_end - time.monotonic()) / max(1, n_restarts + 1))

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 15:
                break
            noise = [0.03, 0.06, 0.1, 0.15, 0.25][i]
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=len(best_seq))
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            # Quick L-BFGS
            dl1 = min(phase4_end - 10, time.monotonic() + per_restart * 0.6)
            p_m, sc_m = _run_lbfgs_adaptive(p_start, dl1, max_wall_time=per_restart * 0.55, use_top_k=True)
            update_best(p_m, sc_m, f"r{i}_lbfgs")

            # Quick Adam
            if time.monotonic() < phase4_end - 10:
                dl2 = min(phase4_end - 10, time.monotonic() + per_restart * 0.35)
                p_a, sc_a = _adam_refine(p_m, dl2, lr_init=0.02, top_k_init=5)
                update_best(p_a, sc_a, f"r{i}_adam")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Try different lengths ---
    print("=== Phase 5: Different lengths ===")
    phase5_end = min(deadline, t_start + 1650.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        n_curr = len(best_seq)
        target_ns = []
        if n_curr != 8192:
            target_ns.append(8192)
        if n_curr != 2048:
            target_ns.append(2048)

        for tn in target_ns:
            if time.monotonic() > phase5_end - 40:
                break
            p_interp = _interp_profile(best_seq, tn)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{tn}")

            if time.monotonic() < phase5_end - 40:
                remaining = phase5_end - time.monotonic()
                per_opt = min(80.0, remaining * 0.5)
                dl = min(phase5_end - 10, time.monotonic() + per_opt)
                p_best, sc_best = _run_lbfgs_adaptive(p_interp, dl, max_wall_time=per_opt, use_top_k=True)
                update_best(p_best, sc_best, f"lbfgs_n{tn}")

                if time.monotonic() < phase5_end - 20:
                    dl2 = min(phase5_end - 10, time.monotonic() + min(40.0, remaining * 0.3))
                    p_best2, sc_best2 = _kkt_equalize(p_best, dl2, max_iter=5000, top_k_init=10)
                    update_best(p_best2, sc_best2, f"kkt_n{tn}")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Final intensive polish ---
    print("=== Phase 6: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            # KKT with very fine parameters
            dl = min(deadline - 5, time.monotonic() + min(60.0, remaining * 0.4))
            p_best, sc_best = _kkt_equalize(best_seq, dl, max_iter=10000, top_k_init=20)
            update_best(p_best, sc_best, "final_kkt")

            if time.monotonic() < deadline - 10:
                dl2 = min(deadline - 5, time.monotonic() + min(40.0, deadline - time.monotonic() - 5))
                p_best2, sc_best2 = _adam_refine(best_seq, dl2, lr_init=0.005, top_k_init=15)
                update_best(p_best2, sc_best2, "final_adam")

            if time.monotonic() < deadline - 10:
                dl3 = min(deadline - 3, time.monotonic() + max(5.0, deadline - time.monotonic() - 3))
                p_best3, sc_best3 = _run_lbfgs_adaptive(best_seq, dl3, max_wall_time=max(3.0, dl3 - time.monotonic()), use_top_k=True)
                update_best(p_best3, sc_best3, "final_lbfgs")

    # --- Final validation ---
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = np.ones(4096) / 4096
            final_score = 2.0

        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}, sum={s:.4f}")

        # Validate with evaluate_sequence
        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 16384:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < final_score:
                    final_score = sc_eval
        except Exception as e:
            print(f"  evaluate_sequence failed: {e}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = np.ones(4096) / 4096
    print(f"  FALLBACK: n=4096, score={_score_norm_fn(fallback):.10f}")
    return [float(x) for x in fallback]
