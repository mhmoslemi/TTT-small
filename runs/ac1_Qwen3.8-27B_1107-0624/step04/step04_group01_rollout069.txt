```
def _project_simplex(v):
    n=len(v)
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    rho=np.nonzero(u*(np.arange(1,n+1)-1)>cssv-1)[0][-1]
    theta=(cssv[rho]-1)/(rho+1)
    w=np.maximum(v-theta,0)
    return w/w.sum()
```

```python
"""
LP-based active-set optimization, subgradient escape, and Adam polishing
for minimizing evaluate_sequence(a) = 2*n*max(convolve(a,a))/sum(a)^2
over non-negative coefficient sequences.

The main idea is to keep the sequence normalized to sum 1 and minimize
max(conv(a,a)).  At each step we solve a relaxed linear program:

    max sum(b)
    s.t. conv(a/M, b)_k <= 1 for a small active set of lags k,
         b >= 0,

where M = max(conv(a,a)).  The LP is solved with constraint generation:
start from the most active lags, add the most violated lag until the
relaxed LP solution is feasible for the active set or the active set is
truncated.  A line search over convex combinations with the normalized LP
solution produces a safer update.  This is combined with occasional
subgradient escapes, multiple random restarts, a short large-n refinement
phase, and a final Adam polish in log-softmax parameterization.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None

_BOUNDS_CACHE = {}


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return k if k > 1 else 2


def _fft_convolve(a, b):
    n = int(len(a))
    m = int(len(b))
    out_n = n + m - 1
    if out_n <= 0:
        return np.empty(0, dtype=np.float64)
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    if r.size:
        r = np.where(r > 0.0, r, 0.0)
    return r.astype(np.float64, copy=False)


def _norm_sum(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(a > 0.0, a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def _score_sequence(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    if c.size == 0:
        return np.inf
    mb = float(np.max(c))
    return float(2.0 * a.size * mb / (s * s))


def _maxconv_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    s = float(np.sum(a))
    if s <= 1e-300:
        return np.inf
    a = a / s
    c = _fft_convolve(a, a)
    if c.size == 0:
        return np.inf
    return float(np.max(c))


def _get_bounds(n):
    bounds = _BOUNDS_CACHE.get(n)
    if bounds is None:
        bounds = [(0.0, np.inf)] * n
        _BOUNDS_CACHE[n] = bounds
    return bounds


def _conv_rows(a, ks, n):
    """Rows A[k, :] for the constraint conv(a, b)_k <= rhs."""
    A = np.zeros((len(ks), n), dtype=np.float64)
    for r, k in enumerate(ks):
        k = int(k)
        j0 = max(0, k - n + 1)
        j1 = min(k, n - 1)
        if j0 <= j1:
            start = k - j0
            stop = k - j1 - 1
            if stop < 0:
                stop = None
            A[r, j0:j1 + 1] = a[start:stop:-1]
    return A


def _initial_active(c, n, K):
    K = int(max(1, min(K, len(c))))
    idx = np.argsort(c, kind="mergesort")[::-1][:K]
    active = [int(k) for k in idx]
    if (n - 1) not in active:
        active.append(n - 1)
    seen = set()
    out = []
    for k in active:
        if 0 <= k < len(c) and k not in seen:
            seen.add(k)
            out.append(k)
    if not out:
        out = [0]
    return out


def _prune_active(active, conv_array, max_active, force_ks):
    max_active = int(max(2, max_active))
    if len(active) <= max_active:
        return list(active)

    kept = []
    kept_set = set()
    for k in force_ks:
        k = int(k)
        if 0 <= k < len(conv_array) and k not in kept_set:
            kept.append(k)
            kept_set.add(k)

    if len(kept) >= max_active:
        return kept[:max_active]

    vals = np.empty(len(active), dtype=np.float64)
    for i, k in enumerate(active):
        vals[i] = float(conv_array[int(k)])

    order = np.argsort(vals, kind="mergesort")[::-1]
    for pos in order:
        if len(kept) >= max_active:
            break
        k = int(active[int(pos)])
        if k not in kept_set:
            kept.append(k)
            kept_set.add(k)

    return kept


def _lp_solve_scaled(a_scaled, active_ks, n, time_limit):
    if linprog is None:
        return None
    if len(active_ks) == 0:
        return np.ones(n, dtype=np.float64) / n

    A = _conv_rows(a_scaled, active_ks, n)
    b_ub = np.ones(len(active_ks), dtype=np.float64)
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = _get_bounds(n)

    try:
        res = linprog(
            c_obj,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": float(time_limit), "threads": 1},
        )
    except Exception:
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
        except Exception:
            try:
                res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds)
            except Exception:
                return None

    if res is None:
        return None

    ok = getattr(res, "success", None)
    if ok is None:
        ok = (getattr(res, "status", -1) == 0)
    if not ok:
        return None

    b = np.asarray(getattr(res, "x", None), dtype=np.float64)
    if b.size != n or not np.all(np.isfinite(b)):
        return None
    b = np.where(b > 0.0, b, 0.0)
    if float(np.sum(b)) <= 1e-300:
        return None
    return b


def _lp_one_step(a, M, n, c=None, max_active=320, K_init=128, max_inner=30, time_limit=5.0):
    if M <= 1e-300:
        return a.copy()
    if c is None:
        c = _fft_convolve(a, a)

    a_scaled = a / M
    active = _initial_active(c, n, K_init)
    active_set = set(active)
    b = a.copy()

    for _ in range(max_inner):
        b_raw = _lp_solve_scaled(a_scaled, active, n, time_limit)
        if b_raw is None:
            break
        b = b_raw
        bsum = float(np.sum(b))
        if bsum <= 1.0 + 1e-8:
            return b

        conv_scaled = _fft_convolve(a_scaled, b)
        if conv_scaled.size == 0:
            return b
        max_k = int(np.argmax(conv_scaled))
        if float(conv_scaled[max_k]) <= 1.0 + 1e-7:
            return b

        if max_k not in active_set:
            if len(active_set) >= max_active:
                active = _prune_active(active, conv_scaled, max_active, [n - 1, max_k])
                active_set = set(active)
            else:
                active.append(max_k)
                active_set.add(max_k)
        else:
            if float(conv_scaled[max_k]) > 1.0 + 1e-4:
                break
            return b

    return b


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    flags = u * (np.arange(1, n + 1, dtype=np.float64)) > (cssv - 1.0)
    idx = np.nonzero(flags)[0]
    if idx.size == 0:
        return np.ones(n, dtype=np.float64) / n
    r = int(idx[-1])
    theta = (float(cssv[r]) - 1.0) / (r + 1)
    w = np.maximum(v - theta, 0.0)
    s = float(np.sum(w))
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / n
    return w / s


def _subgradient_one(a, active_frac=0.99, max_active=128):
    a = _norm_sum(a)
    n = a.size
    c = _fft_convolve(a, a)
    M = float(np.max(c)) if c.size else np.inf
    if not np.isfinite(M) or M <= 1e-300:
        return a, M

    active_idx = np.flatnonzero(c >= M * active_frac)
    if active_idx.size == 0:
        active_idx = np.array([int(np.argmax(c))], dtype=int)
    elif active_idx.size > max_active:
        active_idx = np.argsort(c, kind="mergesort")[::-1][:max_active]

    A = _conv_rows(a, active_idx, n)
    d = np.sum(A, axis=0) / max(1, active_idx.size)

    best = a.copy()
    bestM = M
    for eta in (0.1, 0.3, 1.0, 3.0, 10.0, 30.0):
        trial = _project_simplex(a - eta * d)
        mm = _maxconv_norm(trial)
        if mm < bestM - 1e-13:
            best = trial
            bestM = mm
    return best, bestM


def _lp_optimize(a, deadline, max_iters=400, max_active=320, K_init=128, no_improve_sec=90.0):
    a = _norm_sum(a)
    best = a.copy()
    best_M = _maxconv_norm(a)
    last_improve = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        if time.monotonic() - last_improve > no_improve_sec:
            break

        c = _fft_convolve(a, a)
        M = float(np.max(c)) if c.size else np.inf
        if not np.isfinite(M):
            break

        b_raw = _lp_one_step(
            a, M, len(a),
            c=c,
            max_active=max_active,
            K_init=K_init,
            max_inner=30,
            time_limit=5.0,
        )
        b = _norm_sum(b_raw)

        new_a = None
        new_M = _maxconv_norm(a)

        for t in (0.01, 0.03, 0.10, 0.25, 0.50, 0.75, 1.00):
            trial = (1.0 - t) * a + t * b
            trial = _norm_sum(trial)
            mm = _maxconv_norm(trial)
            if mm < new_M - 1e-13:
                new_M = mm
                new_a = trial

            trial_s = _try_sym(trial)
            mm_s = _maxconv_norm(trial_s)
            if mm_s < new_M - 1e-13:
                new_M = mm_s
                new_a = trial_s

        if new_a is not None and new_M < best_M - 1e-13:
            a = new_a
            best = a.copy()
            best_M = new_M
            last_improve = time.monotonic()
            if it % 10 == 0:
                print(
                    f"    LP iter {it}: max_conv={best_M:.12f} score={2.0 * len(a) * best_M:.10f} "
                    f"t={time.monotonic() - (deadline - (deadline - last_improve)):.1f}s",
                    flush=True
                )
        elif it % 25 == 24:
            a_sg, m_sg = _subgradient_one(a)
            if m_sg < best_M - 1e-13:
                a = a_sg
                best = a.copy()
                best_M = m_sg
                last_improve = time.monotonic()
                print(f"    subgradient improvement: score={2.0 * len(a) * best_M:.10f}", flush=True)

    return best, best_M


def _softmax_z(z):
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return z
    z = z - np.max(z)
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 1e-300 or not np.isfinite(s):
        return np.ones(z.size, dtype=np.float64) / z.size
    return e / s


def _adam_polish(a_init, deadline, lr=0.002, max_iter=100000):
    a = _norm_sum(a_init)
    n = a.size
    c0 = _fft_convolve(a, a)
    M0 = float(np.max(c0)) if c0.size else np.inf
    if not np.isfinite(M0) or M0 <= 1e-300:
        return a, 0.0

    betas = [M0 * f for f in (0.02, 0.008, 0.003, 0.001, 0.0004, 0.0001)]
    z = np.log(np.maximum(a, 1e-15))
    z = z - np.mean(z)

    best = a.copy()
    best_M = M0
    m_buf = np.zeros(n, dtype=np.float64)
    v_buf = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    per_beta = max(100, max_iter // len(betas))
    no_imp = 0

    for beta in betas:
        if time.monotonic() > deadline:
            break
        steps = 0
        while steps < per_beta:
            if time.monotonic() > deadline:
                break
            steps += 1
            t += 1

            a = _softmax_z(z)
            c = _fft_convolve(a, a)
            cm = float(np.max(c)) if c.size else 0.0
            if cm <= 1e-300:
                break

            c_shift = (c - cm) / beta
            c_shift = np.minimum(c_shift, 0.0)
            e = np.exp(c_shift)
            s = float(np.sum(e))
            if s <= 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            W = float(np.dot(w, c))
            a_rev = a[::-1]
            wc = _fft_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            grad = 4.0 * n * a * (h - W)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            step = lr * mh / (np.sqrt(vh) + eps)
            snorm = float(np.linalg.norm(step))
            if snorm > 0.5:
                step = step * (0.5 / snorm)

            z = z - step
            z = np.clip(z, -40.0, 40.0)
            z = z - np.mean(z)

            if t % 50 == 0:
                sc = _maxconv_norm(a)
                if sc < best_M - 1e-13:
                    best_M = sc
                    best = a.copy()
                    no_imp = 0
                else:
                    no_imp += 50

            if no_imp > 6000:
                break

    a = _softmax_z(z)
    sc = _maxconv_norm(a)
    if sc < best_M - 1e-13:
        best_M = sc
        best = a.copy()

    return best, best_M


def _try_sym(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    s = 0.5 * (a + a[::-1])
    return _norm_sum(s)


def _make_profile(n, kind=0):
    n = int(max(1, n))
    if kind == 0:
        return np.ones(n, dtype=np.float64) / n
    if kind == 1:
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * t)
    elif kind == 2:
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.35 * np.cos(4.0 * np.pi * t)
    else:
        t = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * t) ** 0.7
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _perturb(a, sigma, rng):
    a = np.asarray(a, dtype=np.float64).ravel()
    z = np.log(np.maximum(a, 1e-15)) + sigma * rng.standard_normal(a.size)
    return _softmax_z(z)


def _interp_profile(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    n_new = int(max(1, n_new))
    if n0 == n_new:
        return _norm_sum(p)
    if n0 == 0:
        return np.ones(n_new, dtype=np.float64) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    q = np.interp(x1, x0, p)
    q = np.maximum(q, 1e-12)
    return q / q.sum()


def _consider_existing(best_sc, best_a, a, label, t0):
    if a is None:
        return best_sc, best_a
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return best_sc, best_a
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    sc = _score_sequence(a)
    if np.isfinite(sc) and sc < best_sc:
        print(
            f"  NEW BEST [{label}] {sc:.10f} n={a.size} t={time.monotonic() - t0:.1f}s",
            flush=True
        )
        return float(sc), a.copy()
    return best_sc, best_a


def propose_candidate(seed=42, budget_s=1800):
    t0 = time.monotonic()
    total = max(60.0, float(budget_s))
    deadline = t0 + max(10.0, total - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_a = None
    best_sc = np.inf

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.clip(p, 0.0, 1000.0)
        if p.size > 0 and float(np.sum(p)) > 0.01:
            parent = _norm_sum(p)
    except Exception:
        parent = None

    if parent is not None and 512 <= parent.size <= 8192:
        n_main = parent.size
        a_main = parent.copy()
        best_sc, best_a = _consider_existing(best_sc, best_a, a_main, "parent", t0)
    else:
        n_main = 2048
        if parent is not None:
            a_main = _interp_profile(parent, n_main)
        else:
            a_main = _make_profile(n_main, 0)
        best_sc, best_a = _consider_existing(best_sc, best_a, a_main, "initial", t0)

    a_sym = _try_sym(a_main)
    best_sc, best_a = _consider_existing(best_sc, best_a, a_sym, "init_sym", t0)

    reserve_adam = min(260.0, 0.15 * total)
    if total >= 700.0:
        reserve_large = min(300.0, 0.17 * total)
    else:
        reserve_large = 0.0

    lp_deadline_candidate = deadline - reserve_adam - reserve_large
    lp_deadline = min(t0 + 0.58 * (deadline - t0), lp_deadline_candidate)
    if lp_deadline < t0 + 60.0:
        lp_deadline = min(deadline - 30.0, t0 + 60.0)

    if n_main <= 2048:
        max_active_main = 320
        k_init_main = 128
        lp_iters_main = 600
        no_imp_main = 100.0
    elif n_main <= 4096:
        max_active_main = 200
        k_init_main = 96
        lp_iters_main = 250
        no_imp_main = 70.0
    else:
        max_active_main = 120
        k_init_main = 64
        lp_iters_main = 100
        no_imp_main = 45.0

    if time.monotonic() < lp_deadline - 5.0:
        print(f"  Phase 1: LP optimization n={n_main}, deadline={lp_deadline - t0:.1f}s", flush=True)
        lp_a, lp_M = _lp_optimize(
            a_main,
            lp_deadline,
            max_iters=lp_iters_main,
            max_active=max_active_main,
            K_init=k_init_main,
            no_improve_sec=no_imp_main,
        )
        best_sc, best_a = _consider_existing(best_sc, best_a, lp_a, "lp_main", t0)
        if best_a is not None:
            best_sc, best_a = _consider_existing(best_sc, best_a, _try_sym(best_a), "lp_main_sym", t0)

    main_adam_deadline = deadline - reserve_large
    restart_deadline = min(main_adam_deadline, t0 + 0.75 * (deadline - t0))

    if best_a is not None and time.monotonic() < restart_deadline - 30.0:
        print(
            f"  Phase 2: restarts until t={restart_deadline - t0:.1f}s "
            f"current={best_sc:.10f}",
            flush=True
        )
        max_restarts = 3 if best_sc <= 1.503 else 8
        for r in range(max_restarts):
            if time.monotonic() >= restart_deadline - 35.0:
                break
            sigma = 0.02 + 0.018 * r
            start = _perturb(best_a, sigma, rng)
            if r % 2 == 1:
                start = _try_sym(start)

            n_cur = start.size
            if n_cur <= 2048:
                ra, rk = 240, 112
            elif n_cur <= 4096:
                ra, rk = 150, 80
            else:
                ra, rk = 90, 56

            short_deadline = min(restart_deadline - 20.0, time.monotonic() + 140.0)
            if short_deadline <= time.monotonic():
                continue
            p, m = _lp_optimize(
                start,
                short_deadline,
                max_iters=90,
                max_active=ra,
                K_init=rk,
                no_improve_sec=35.0,
            )
            best_sc, best_a = _consider_existing(best_sc, best_a, p, f"restart_lp_{r}", t0)

            polish_deadline = min(restart_deadline - 8.0, time.monotonic() + 70.0)
            if polish_deadline > time.monotonic() + 10.0 and best_a is not None:
                p2, m2 = _adam_polish(p, polish_deadline, lr=0.002, max_iter=18000)
                best_sc, best_a = _consider_existing(best_sc, best_a, p2, f"restart_adam_{r}", t0)

    if best_a is not None and time.monotonic() < main_adam_deadline - 15.0:
        n_cur = best_a.size
        max_adam_iter = 120000 if n_cur <= 2048 else (60000 if n_cur <= 4096 else 25000)
        print(f"  Phase 3: main Adam polish n={n_cur}", flush=True)
        p, m = _adam_polish(best_a, main_adam_deadline, lr=0.002, max_iter=max_adam_iter)
        best_sc, best_a = _consider_existing(best_sc, best_a, p, "adam_main", t0)
        if best_a is not None:
            best_sc, best_a = _consider_existing(best_sc, best_a, _try_sym(best_a), "adam_main_sym", t0)

    if best_a is not None and time.monotonic() < deadline - 90.0:
        n_cur = best_a.size
        if n_cur < 4096:
            target_n = 4096
        elif n_cur < 8192:
            target_n = 8192
        else:
            target_n = int(n_cur * 1.25)
            if target_n > 8192:
                target_n = 8192

        if target_n > n_cur and target_n <= 8192:
            p = _interp_profile(best_a, target_n)
            best_sc, best_a = _consider_existing(best_sc, best_a, p, f"interp_n{target_n}", t0)

            lp2_deadline = min(deadline - 90.0, time.monotonic() + 90.0)
            if lp2_deadline > time.monotonic() + 15.0:
                if target_n <= 4096:
                    ra, rk = 160, 72
                else:
                    ra, rk = 96, 48
                p2, m2 = _lp_optimize(
                    p,
                    lp2_deadline,
                    max_iters=70,
                    max_active=ra,
                    K_init=rk,
                    no_improve_sec=25.0,
                )
                best_sc, best_a = _consider_existing(best_sc, best_a, p2, f"lp_n{target_n}", t0)

            if time.monotonic() < deadline - 35.0:
                n_polish = best_a.size if best_a is not None else target_n
                max_adam2 = 45000 if n_polish <= 4096 else 18000
                p3, m3 = _adam_polish(best_a, deadline - 10.0, lr=0.0015, max_iter=max_adam2)
                best_sc, best_a = _consider_existing(best_sc, best_a, p3, f"adam_n{target_n}", t0)

    if best_a is None:
        best_a = _make_profile(2048, 0)
        best_sc = _score_sequence(best_a)

    final = np.asarray(best_a, dtype=np.float64).ravel()
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.clip(final, 0.0, 1000.0)
    final = _norm_sum(final)

    our_sc = _score_sequence(final)
    print(f"  Internal final score: {our_sc:.10f}, n={final.size}, t={time.monotonic() - t0:.1f}s", flush=True)

    try:
        ev = float(evaluate_sequence([float(x) for x in final]))
        print(f"  Evaluator score: {ev:.10f}", flush=True)
    except Exception as ex:
        print(f"  Evaluator unavailable or failed: {ex}", flush=True)

    return [float(x) for x in final]
```
