def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)

    """Build the (2n-1) x n convolution matrix."""
    rows = []
    cols = []
    vals = []

        j_start = max(0, k - n + 1)

            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())

        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()
            stall = 0
            if verbose and it % 200 == 0:
                print(f"    LP iter {it}: score={best_score:.10f} M={M:.8f}")
        else:
            stall += 1
            if stall == 50:
                t_base = max(0.03, t_base * 0.6)
            elif stall > 200:
                break

    return best_a, best_score


def _gradient_polish(a0, deadline, rng, lr=0.01, beta_frac=0.05):
    n = len(a0)
    p = np.maximum(a0, 1e-15).copy()
    p = p / p.sum()
    z = np.log(p)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m_ag = np.zeros(n)
    v_ag = np.zeros(n)
    t_adam = 0

    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-4, beta_frac * maxc)
    best_p = p.copy()
    best_z = z.copy()
    best_score = _score_norm(p)
    no_imp = 0

    while time.monotonic() < deadline:
        mval = float(c.max())
        x = (c - mval) / beta
        ex = np.exp(np.minimum(x, 0.0))
        s_ex = float(ex.sum())
        if not np.isfinite(s_ex) or s_ex <= 0:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex

        N = 1
        while N < 2 * n - 1:
            N <<= 1
        fw = np.fft.rfft(w, N)
        fp = np.fft.rfft(p[::-1], N)
        h = np.fft.irfft(fw * fp, N)
        g = h[n - 1:2 * n - 1]

        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m_ag.fill(0)
            v_ag.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m_ag = b1 * m_ag + (1.0 - b1) * gz
        v_ag = b2 * v_ag + (1.0 - b2) * (gz * gz)
        mh = m_ag / (1.0 - b1 ** t_adam)
        vh = v_ag / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m_ag.fill(0)
            v_ag.fill(0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= z.mean()
        e = np.exp(z)
        p = e / e.sum()
        p = np.maximum(p, 1e-15)
        p = p / p.sum()

        c = _full_convolve(p)
        maxc = float(c.max())
        score = _score_norm(p)

        if score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        if t_adam % 20 == 0:
            beta = max(5e-5, beta * 0.95)

        if no_imp > 300:
            z = best_z + 0.03 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p = np.exp(z)
            p = p / p.sum()
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-4, beta_frac * maxc)
            m_ag.fill(0)
            v_ag.fill(0)
            t_adam = 0
            no_imp = 0

    return best_p, best_score


def _make_exponential(n, lam):
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * t
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_quadratic(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + a1 * t + a2 * t * t
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_cosine(n, freq, amp):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(20.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  *** BEST: {best_score:.10f} n={p.size} [{label}]")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    print(f"Started. Parent n={parent.size if parent is not None else 0}, budget={budget_s}s")

    # Phase 1: Quick scan of starting profiles (30s)
    phase1_end = t_start + 30.0
    candidates = []

    if parent is not None:
        candidates.append(('parent', parent.copy(), parent.size))

    for n in [512, 1024, 2048]:
        for lam in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
            a = _make_exponential(n, lam)
            candidates.append((f'exp_{lam}', a, n))
        for eta in [0.5, 1.0, 1.5, 2.0, 3.0]:
            a = _make_linear(n, eta)
            candidates.append((f'lin_{eta}', a, n))
        a = _make_quadratic(n, 1.0, -0.5)
        candidates.append(('quad1', a, n))
        a = _make_quadratic(n, 2.0, -1.0)
        candidates.append(('quad2', a, n))
        a = _make_cosine(n, 1, 0.5)
        candidates.append(('cos1', a, n))
        a = _make_cosine(n, 2, 0.3)
        candidates.append(('cos2', a, n))

    scored = []
    for label, a, n in candidates:
        sc = _score_norm(a)
        scored.append((sc, label, a, n))
    scored.sort(key=lambda x: x[0])

    print(f"Phase 1: {len(scored)} candidates, top-5:")
    for sc, label, a, n in scored[:5]:
        print(f"    {sc:.10f} n={n} [{label}]")

    update(scored[0][2], scored[0][0], scored[0][1])

    # Phase 2: LP iteration on top-3 candidates (most of the time)
    phase2_end = deadline - 90.0
    top_k = min(3, len(scored))
    time_per = max(30.0, (phase2_end - time.monotonic()) / top_k)

    for i in range(top_k):
        if time.monotonic() > phase2_end:
            break
        sc0, label, a0, n0 = scored[i]
        per_end = min(phase2_end, time.monotonic() + time_per)
        print(f"  Phase 2: LP on [{label}] n={n0} score={sc0:.10f} (budget {per_end - time.monotonic():.0f}s)")

        a_best, sc_best = _lp_iterate(a0, per_end - 5.0, verbose=True)
        update(a_best, sc_best, f'LP_{label}')

    # Phase 3: Gradient polish on best (30s)
    if best_seq is not None and time.monotonic() < deadline - 30:
        per = min(40.0, deadline - time.monotonic() - 20)
        if per > 10:
            print(f"  Phase 3: Gradient polish (budget {per:.0f}s)")
            a_gp, sc_gp = _gradient_polish(best_seq, time.monotonic() + per, rng,
                                           lr=0.02, beta_frac=0.04)
            update(a_gp, sc_gp, 'grad_polish')

    # Phase 4: Final LP on absolute best (remaining time)
    if best_seq is not None and time.monotonic() < deadline - 3:
        per = deadline - time.monotonic() - 3
        if per > 20:
            print(f"  Phase 4: Final LP (budget {per:.0f}s)")
            a_fl, sc_fl = _lp_iterate(best_seq, deadline - 2.0, verbose=True)
            update(a_fl, sc_fl, 'final_LP')

    # Final output
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_exponential(1024, 1.5)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"EVAL error: {e}")

        return [float(x) for x in final]

    fb = _make_exponential(1024, 1.5)
    print("FALLBACK n=1024")
    return [float(x) for x in fb]
