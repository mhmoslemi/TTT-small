"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses aggressive L-BFGS with logsumexp continuation, Adam with cyclic LR,
KKT balancing, coordinate refinement, and multi-length search.
Key improvements: better init profiles, more aggressive smoothing schedule,
active-set coordinate descent, and structured perturbation restarts.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_conv_and_weights(a, beta, n, top_k=0):
    """Compute convolution, weights, and intermediate quantities."""
    c = _full_convolve(a, a)
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        w = np.zeros_like(c)
        c_top = c[idx]
        c_norm = (c_top - c_top.max()) / beta
        e = np.exp(c_norm)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w[idx[0]] = 1.0
        else:
            w[idx] = e / s
    else:
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            idx = int(np.argmax(c))
            w = np.zeros_like(c)
            w[idx] = 1.0
        else:
            w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    return w, h, W, c, cm


def _z_gradient(z, beta, n, top_k=0):
    a = _softmax_fn(z)
    w, h, W, c, cm = _compute_conv_and_weights(a, beta, n, top_k)
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _z_objective(z, beta, n, top_k=0):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_norm = c_top - c_top.max()
        log_sum = float(np.log(np.sum(np.exp(c_norm / beta))))
        return 2.0 * n * (c_top.max() + beta * log_sum)
    else:
        c_shifted = (c - cm) / beta
        log_sum = float(np.log(np.sum(np.exp(c_shifted))))
        return 2.0 * n * (cm + beta * log_sum)


def _run_lbfgs(a_init, deadline, max_wall_time=None, top_k=0):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if max_wall_time is None:
        max_wall_time = max(1.0, deadline - time.monotonic())

    beta_starts = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    betas = [b * max(max_c0, 1e-10) for b in beta_starts]

    per_level = max_wall_time / len(betas)

    for beta in betas:
        if time.monotonic() > deadline:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        try:
            result = minimize(
                _z_objective, z, args=(beta, n, top_k),
                method='L-BFGS-B', jac=_z_gradient,
                options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 40, 'maxfun': 2500}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                if np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _adam_polish(a_init, deadline, lr=0.05, top_k=0, n_max_iter=100000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-10, 0.01 * max_c)

    if top_k == 0:
        adaptive_k = True
        cur_k = 3
    else:
        adaptive_k = False
        cur_k = top_k

    # Cyclic LR schedule
    lr_base = lr
    lr_cycle = 0.3
    lr_period = 2000

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break

        t += 1

        a = _softmax_fn(z)
        use_k = cur_k if adaptive_k else top_k
        grad = _z_gradient(z, beta, n, use_k)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        # Cyclic LR
        lr_cur = lr_base * (1.0 + lr_cycle * abs(np.sin(np.pi * t / lr_period)))
        z = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 150 == 0:
            beta = max(1e-14, beta * 0.85)
            if adaptive_k and t % 500 == 0:
                cur_k = min(cur_k + 2, 50)

        if t % 3000 == 0:
            if best_score < 1.51:
                lr_base = max(0.002, lr_base * 0.95)

    return best_a, best_score


def _kkt_balance(a_init, deadline, max_iter=10000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        threshold = cm * 0.99
        active = np.where(c >= threshold)[0]

        if len(active) == 0:
            break

        h = np.zeros(n)
        for k_idx in active:
            k_val = int(k_idx)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                h[lo:hi+1] += a[k_val - hi:k_val - lo + 1][::-1]
        h /= len(active)

        W = float(np.dot(h, a))

        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()

        active_spread = float(c[active].max() - c[active].min()) + 1e-15
        step = 0.02 * active_spread / (cm + 1e-15)
        step = min(step, 0.08)

        h_diff = h - W
        h_max = np.abs(h_diff).max() + 1e-15

        z_new = z - step * h_diff / h_max
        z_new = np.clip(z_new, -80, 80)
        z_new = z_new - z_new.mean()
        a_new = _softmax_fn(z_new)

        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            for retry in range(4):
                step *= 0.5
                z_new = z - step * h_diff / h_max
                z_new = np.clip(z_new, -80, 80)
                z_new = z_new - z_new.mean()
                a_new = _softmax_fn(z_new)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new
                    no_improve = 0
                    break

        if no_improve > 50:
            break

    return best_a, best_score


def _coordinate_descent(a_init, deadline, max_iter=50000):
    """Coordinate descent: optimize one element at a time."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    rng = np.random.default_rng(12345)

    # Precompute the convolution
    c = _full_convolve(a, a)
    cm = float(c.max())
    S = float(a.sum())

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Pick a random coordinate
        i = int(rng.integers(0, n))
        ai = a[i]

        # Compute the gradient-like quantity for this coordinate
        # d/dai [max(b)] ≈ b_k* where k* is argmax, contribution from ai
        # More precisely: if we change ai to ai+da, then
        # b[k] changes by da*(a[k-i] + a[i-k]) for valid k
        # And S changes by da
        # So we want to minimize 2*n*max(b_k + da*c_k) / (S+da)^2

        c_k = np.zeros(2 * n - 1)
        # b[k] = sum_j a[j]*a[k-j], dB[k]/da[i] = a[k-i] + a[i-k] (if valid)
        for k in range(2 * n - 1):
            val = 0.0
            if 0 <= k - i <= n - 1:
                val += a[k - i]
            if 0 <= i - k <= n - 1:
                val += a[i - k]
            c_k[k] = val

        # Try a few step sizes
        best_da = 0.0
        best_local_score = _score_norm_fn(a)

        for frac in [-0.3, -0.15, -0.05, 0.05, 0.15, 0.3]:
            da = frac * ai if ai > 0 else frac * (S / n)
            ai_new = ai + da
            if ai_new < 0:
                continue
            a[i] = ai_new
            c_new = c + da * c_k
            S_new = S + da
            sc = 2.0 * n * c_new.max() / (S_new * S_new) if S_new > 1e-15 else np.inf
            if np.isfinite(sc) and sc < best_local_score:
                best_local_score = sc
                best_da = da
            a[i] = ai
            c = c_new - da * c_k
            S = S - da

        if best_da != 0:
            a[i] += best_da
            c = _full_convolve(a, a)  # Full recompute occasionally
            cm = float(c.max())
            S = float(a.sum())
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

        if it % 1000 == 0:
            c = _full_convolve(a, a)
            cm = float(c.max())
            S = float(a.sum())

    return best_a, best_score


def _make_geom_profile(n, c):
    """Geometric decay: a_i = exp(-c*i/(n-1)), normalized."""
    x = np.arange(n) / max(n - 1, 1)
    a = np.exp(-c * x)
    return a / a.sum()


def _make_double_exp_profile(n, c1, c2):
    """Double exponential: a_i = exp(-c1*i/(n-1) - c2*(i/(n-1))^2)."""
    x = np.arange(n) / max(n - 1, 1)
    a = np.exp(-c1 * x - c2 * x * x)
    return a / a.sum()


def _make_power_geom_profile(n, p, c):
    """Power times geometric: a_i = (i+1)^p * exp(-c*i/(n-1))."""
    x = np.arange(n) / max(n - 1, 1)
    a = (np.arange(n) + 1.0) ** p * np.exp(-c * x)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_sine_mod_profile(n, c, amp, freq):
    """Geometric with sine modulation: a_i = exp(-c*x) * (1 + amp*sin(2*pi*freq*x))."""
    x = np.arange(n) / max(n - 1, 1)
    a = np.exp(-c * x) * (1.0 + amp * np.sin(2.0 * np.pi * freq * x))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_tri_geom_profile(n, c1, c2):
    """Triangular-geometric: exp(-c1*min(x,1-x) - c2*x)."""
    x = np.arange(n) / max(n - 1, 1)
    a = np.exp(-c1 * np.minimum(x, 1.0 - x) - c2 * x)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _make_exp_profile(n, alpha):
    r = 1.0 - alpha
    a = r ** np.arange(n)
    return a / a.sum()


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

    count = 0
    seen_keys = set()
    n_lbfags = 5

    for sc, p, pn, label in candidates:
        if time.monotonic() > phase1_end:
            break
        if count >= n_lbfags:
            break
        if sc > best_score * 1.05:
            break
        key = label
        if key in seen_keys:
            continue
        seen_keys.add(key)

        remaining = phase1_end - time.monotonic()
        per_time = max(30.0, min(150.0, remaining / max(1, n_lbfags - count)))
        dl = min(phase1_end, time.monotonic() + per_time)
        print(f"  L-BFGS on {label} (n={pn}, score={sc:.6f}), budget={per_time:.1f}s")

        p_best, sc_best = _run_lbfgs(p, dl, max_wall_time=per_time, top_k=0)
        update_best(p_best, sc_best, f"lbfgs_{label}")
        count += 1

    print(f"  Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Adam polish ---
    print("=== Phase 2: Adam polish ===")
    phase2_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        remaining = phase2_end - time.monotonic()
        dl = min(phase2_end, time.monotonic() + remaining * 0.5)
        print(f"  Adam pass 1 on best (n={len(best_seq)}), budget={remaining*0.5:.1f}s")
        p_best, sc_best = _adam_polish(best_seq, dl, lr=0.04, top_k=0, n_max_iter=100000)
        update_best(p_best, sc_best, "adam1")

        if time.monotonic() < phase2_end - 10:
            dl2 = min(phase2_end - 5, time.monotonic() + (phase2_end - time.monotonic()) * 0.6)
            p_best2, sc_best2 = _adam_polish(best_seq, dl2, lr=0.02, top_k=8, n_max_iter=60000)
            update_best(p_best2, sc_best2, "adam2")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: KKT balancing ---
    print("=== Phase 3: KKT balancing ===")
    phase3_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        remaining = phase3_end - time.monotonic()
        dl = min(phase3_end, time.monotonic() + remaining * 0.6)
        print(f"  KKT balance, budget={remaining*0.6:.1f}s")
        p_best, sc_best = _kkt_balance(best_seq, dl, max_iter=20000)
        update_best(p_best, sc_best, "kkt1")

        # Second KKT pass with tighter threshold via different init
        if time.monotonic() < phase3_end - 30:
            dl2 = min(phase3_end - 10, time.monotonic() + (phase3_end - time.monotonic()) * 0.5)
            p_best2, sc_best2 = _kkt_balance(best_seq, dl2, max_iter=15000)
            update_best(p_best2, sc_best2, "kkt2")

        # Follow-up L-BFGS after KKT
        if time.monotonic() < phase3_end - 20:
            dl3 = min(phase3_end - 10, time.monotonic() + 80.0)
            p_best3, sc_best3 = _run_lbfgs(best_seq, dl3, max_wall_time=75.0, top_k=5)
            update_best(p_best3, sc_best3, "lbfgs_post_kkt")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Coordinate descent ---
    print("=== Phase 4: Coordinate descent ===")
    phase4_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        remaining = phase4_end - time.monotonic()
        dl = min(phase4_end, time.monotonic() + min(remaining, 300.0))
        print(f"  Coordinate descent, budget={min(remaining,300.0):.1f}s")
        p_best, sc_best = _coordinate_descent(best_seq, dl, max_iter=200000)
        update_best(p_best, sc_best, "coord")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multi-restart ---
    print("=== Phase 5: Multi-restart ===")
    phase5_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 5
        per_restart = min(40.0, (phase5_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            # Structured perturbation
            noise_levels = [0.03, 0.05, 0.08, 0.12, 0.18]
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise = noise_levels[i % len(noise_levels)]

            # Mix of random and structured noise
            if i % 2 == 0:
                z = z + noise * rng.normal(size=n_best)
            else:
                # Low-frequency perturbation
                freq = rng.integers(1, 5)
                phase = rng.uniform(0, 2 * np.pi)
                x = np.arange(n_best) / n_best
                z = z + noise * np.sin(2 * np.pi * freq * x + phase)

            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            dl = min(phase5_end - 5, time.monotonic() + per_restart)
            per_lbfgs = per_restart * 0.55
            per_adam = per_restart * 0.4

            p_m, sc_m = _run_lbfgs(p_start, min(phase5_end - 5, time.monotonic() + per_lbfgs),
                                   max_wall_time=per_lbfgs, top_k=3)
            update_best(p_m, sc_m, f"r{i}_lbfgs")
            if time.monotonic() < phase5_end - 5:
                p_a, sc_a = _adam_polish(p_m, min(phase5_end - 5, time.monotonic() + per_adam),
                                         lr=0.03, top_k=5, n_max_iter=15000)
                update_best(p_a, sc_a, f"r{i}_adam")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Try different lengths ---
    print("=== Phase 6: Different lengths ===")
    phase6_end = min(deadline, t_start + 1750.0)
    if time.monotonic() < phase6_end and best_seq is not None:
        n_curr = len(best_seq)
        target_ns = []
        if n_curr != 8192:
            target_ns.append(8192)
        if n_curr != 2048:
            target_ns.append(2048)

        for tn in target_ns:
            if time.monotonic() > phase6_end - 40:
                break
            p_interp = _interp_profile(best_seq, tn)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{tn}")

            remaining = phase6_end - time.monotonic()
            if remaining < 40:
                break
            per_opt = min(100.0, remaining * 0.5)
            dl = min(phase6_end - 15, time.monotonic() + per_opt)
            p_best, sc_best = _run_lbfgs(p_interp, dl, max_wall_time=per_opt, top_k=0)
            update_best(p_best, sc_best, f"lbfgs_n{tn}")

            if time.monotonic() < phase6_end - 25:
                dl2 = min(phase6_end - 10, time.monotonic() + min(50.0, remaining * 0.25))
                p_best2, sc_best2 = _adam_polish(p_best, dl2, lr=0.03, top_k=5, n_max_iter=15000)
                update_best(p_best2, sc_best2, f"adam_n{tn}")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final polish ---
    print("=== Phase 7: Final polish ===")
    if time.monotonic() < deadline and best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            dl = min(deadline - 5, time.monotonic() + min(50.0, remaining * 0.5))
            p_best, sc_best = _adam_polish(best_seq, dl, lr=0.01, top_k=15, n_max_iter=20000)
            update_best(p_best, sc_best, "final_adam")

        if time.monotonic() < deadline - 10:
            dl2 = min(deadline - 3, time.monotonic() + max(5.0, deadline - time.monotonic() - 5))
            p_best2, sc_best2 = _run_lbfgs(best_seq, dl2, max_wall_time=max(5.0, dl2 - time.monotonic()), top_k=10)
            update_best(p_best2, sc_best2, "final_lbfgs")

    # --- Final validation and return ---
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = _make_geom_profile(4096, 1.256)
            final_score = _score_fn(final_seq)

        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")

        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 16384:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  evaluate_sequence check failed: {e}")

        return [float(x) for x in final_seq]

    fallback = _make_geom_profile(4096, 1.256)
    print(f"  FALLBACK: n=4096, score={_score_norm_fn(fallback):.10f}")
    return [float(x) for x in fallback]
