def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())

def _binomial_sequence(n):
    """Compute coefficients of (1-x)^{-1/2} truncated to length n, normalized."""
    # binom(2k, k) / 4^k
    b = np.zeros(n)
    b[0] = 1.0
    for k in range(1, n):
        b[k] = b[k-1] * (4.0 * k - 2.0) / (4.0 * k)
    b = np.maximum(b, 0)
    return b / b.sum()

def _make_profile(n, kind, param=0.0):
    t = np.linspace(0.0, 1.0, n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'linear':
        a = 1.0 + param * t
    elif kind == 'quadratic':
        u = 2.0 * t - 1.0
        a = 1.0 + param * u * u
    elif kind == 'gaussian':
        a = np.exp(-param * (t - 0.5) ** 2)
    elif kind == 'raised_cosine':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5)))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _run_lbfgs(a_init, deadline, rng, beta_levels=None):
    """L-BFGS with logsumexp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    c0 = _convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if beta_levels is None:
        beta_levels = [0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    betas = [b * max_c0 for b in beta_levels]
    
    remaining = max(10.0, deadline - time.monotonic())
    per_level = remaining / len(betas)
    
    for level_idx, beta in enumerate(betas):
        if time.monotonic() > deadline - 3.0:
            break
        
        # More iterations for smaller beta
        if level_idx < len(betas) // 2:
            maxiter = 500
            maxfun = 3000
        else:
            maxiter = 2000
            maxfun = 10000
        
        def obj_func(z_):
            a_ = _softmax(z_)
            c_ = _convolve(a_, a_)
            cm_ = float(c_.max())
            if beta < 1e-15:
                return 2.0 * n * cm_
            c_shift = (c_ - cm_) / beta
            e_ = np.exp(c_shift)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            return 2.0 * n * (cm_ + beta * float(np.log(s_)))
        
        def grad_func(z_):
            a_ = _softmax(z_)
            c_ = _convolve(a_, a_)
            cm_ = float(c_.max())
            if beta < 1e-15:
                # Gradient at the max point
                k_star = int(np.argmax(c_))
                grad = np.zeros(n)
                lo = max(0, k_star - n + 1)
                hi = min(n - 1, k_star)
                grad[lo:hi+1] = 2.0 * a_[k_star - hi:k_star - lo+1][::-1]
                grad *= 4.0 * n
                return grad
            
            c_shift = (c_ - cm_) / beta
            e_ = np.exp(c_shift)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            
            # Gradient: 4*n * a * (conv(w_rev, a) center - W)
            # where W = dot(w, c)
            W = float(np.dot(w, c_))
            w_rev = w[::-1]
            # h = conv(w_rev, a)[n-1 : 2n-1]
            h = _convolve(w_rev, a_)[n-1:2*n-1]
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g
        
        try:
            result = minimize(
                obj_func, z, jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': maxfun}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax(z_new)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception as e:
            print(f"  L-BFGS level {level_idx} error: {e}")
    
    return best_a, best_score

def _active_set_max(a_init, deadline, rng, n_iters=5000):
    """Subgradient method on the true max with momentum."""
    n = len(a_init)
    a = a_init.copy()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    v = np.zeros(n)  # momentum
    mu = 0.9
    
    for it in range(n_iters):
        if time.monotonic() > deadline - 2.0:
            break
        
        a_curr = _softmax(z)
        c = _convolve(a_curr, a_curr)
        cm = float(c.max())
        
        # Find active set (top few)
        k_top = 5
        top_idx = np.argsort(c)[-k_top:]
        
        # Weighted gradient
        weights = c[top_idx] - c[top_idx[0]]
        weights = np.exp(weights / (0.01 * max(cm, 1e-15)))
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for w_i, k_val in zip(weights, top_idx):
            k_val = int(k_val)
            lo = max(0, k_val - n + 1)
            hi = min(n - 1, k_val)
            grad[lo:hi+1] += w_i * 2.0 * a_curr[k_val - hi:k_val - lo+1][::-1]
        
        grad *= 4.0 * n
        
        # Gradient step with momentum
        step_size = 0.01 * (1.0 / (1.0 + 0.001 * it))
        v = mu * v - step_size * grad
        z = z + v
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        if it % 50 == 0:
            a_new = _softmax(z)
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
    
    return best_a, best_score

def _local_search(a_init, deadline, rng, n_trials=10000):
    """Local search with multiplicative perturbations."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 2.0:
            break
        
        a_trial = a.copy()
        # Random block perturbation
        block_size = rng.integers(2, min(30, n // 8))
        start = rng.integers(0, n - block_size)
        factor = rng.uniform(0.8, 1.25, size=block_size)
        a_trial[start:start+block_size] *= factor
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial /= a_trial.sum()
        
        sc = _score_norm(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
        elif rng.random() < 0.005 and sc < best_score * 1.005:
            a = a_trial.copy()
    
    return best_a, best_score

def _sequential_refine(a_init, deadline, rng, n_passes=10):
    """Exploit nested structure: for each position, optimize given neighbors."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for p in range(n_passes):
        if time.monotonic() > deadline - 2.0:
            break
        
        # Forward pass
        for j in range(n):
            if time.monotonic() > deadline - 2.0:
                break
            a_old = a[j]
            # Try a range of values around a_old
            if a_old < 1e-10:
                candidates = [a_old]
            else:
                candidates = [a_old * f for f in [0.7, 0.85, 0.95, 1.0, 1.05, 1.15, 1.3]]
            
            best_trial = a_old
            best_trial_score = _score_norm(a)
            
            for val in candidates:
                a[j] = max(val, 0)
                sc = _score_norm(a)
                if sc < best_trial_score:
                    best_trial_score = sc
                    best_trial = max(val, 0)
            a[j] = best_trial
        
        # Normalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a /= s
        
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score
