```python
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _correlation(a, p):
    """Compute g_i = sum_k p_k * a_{k-i} for i = 0..len(a)-1"""
    n = len(a)
    N = _next_pow2(len(p) + n)
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    g_full = np.fft.irfft(P * np.conj(A), N)
    return g_full[:n]

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _kkt_fixed_point(a, beta, n_iter, t=0.5, floor=1e-15):
    """
    KKT fixed-point iteration for minimizing max autocorrelation.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    
    for it in range(n_iter):
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Softmax weights
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        e_sum = e.sum()
        if e_sum < 1e-300:
            p = np.zeros_like(c)
            p[int(np.argmax(c))] = 1.0
        else:
            p = e / e_sum
        
        # g_i = sum_k p_k * a_{k-i}
        g = _correlation(a, p)
        
        # sigma = 2 * soft_max(c) / sum(a)
        soft_max = cm + beta * np.log(e_sum)
        sa = float(a.sum())
        sigma = 2.0 * soft_max / sa

        a_new = (1.0 - t) * a + t * g / sigma

        # Normalize (optional, for numerical stability)
        a_new = a_new / a_new.sum() * sa  # Keep same sum
        
        a = a_new
```

```python
"""
KKT Fixed-Point Optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Core innovation: principled fixed-point iteration based on KKT conditions,
where at optimum the weighted-averaged shifted sequence is constant (flat).
Combined with beta continuation, L-BFGS polishing, and multi-restart.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _corr_g(a, p):
    """Compute g_i = sum_k p_k * a_{k-i} for i = 0..len(a)-1.
    This is the cross-correlation of p (length 2n-1) and a (length n)."""
    n = len(a)
    N = _next_pow2(len(p) + n)
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    g_full = np.fft.irfft(P * np.conj(A), N)
    return g_full[:n]


def _grad_kkt_residual(a, p, r):
    """Compute grad_j = sum_i r_i * p_{i+j} for j = 0..n-1.
    This is the cross-correlation of r (length n) and p (length 2n-1)."""
    n = len(a)
    N = _next_pow2(n + len(p))
    R = np.fft.rfft(r, N)
    P = np.fft.rfft(p, N)
    grad_full = np.fft.irfft(R * np.conj(P), N)
    return grad_full[:n]


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _kkt_fixed_point(a_init, deadline, n_iter_max=500000, t=1.0, floor=1e-30):
    """Pure KKT fixed-point: a <- g/mean(g), which converges to g_i = const."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, floor)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        # For fixed-point iteration with t=1, we don't need softmax - just use the argmax
        # But for smoothness, use a small effective active set
        # Use top elements: those within 1% of max
        threshold = cm * 0.999
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Build p from active set (uniform weights)
        p = np.zeros_like(c)
        p[active] = 1.0 / len(active)

        # Compute g
        g = _corr_g(a, p)

        # Update: a_new = g / mean(g)
        mg = float(g.mean())
        if mg < 1e-30:
            break
        a_new = g / mg
        a_new = np.maximum(a_new, floor)
        a_new = a_new / a_new.sum()

        # Check improvement
        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def _kkt_smooth_iteration(a_init, deadline, beta_schedule, t=1.0, floor=1e-30, iters_per_beta=2000):
    """KKT iteration with beta continuation for the softmax weights."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, floor)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for beta_rel in beta_schedule:
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = beta_rel * cm

        for it in range(iters_per_beta):
            if time.monotonic() > deadline:
                break

            c = _full_convolve(a, a)
            cm = float(c.max())
            beta_curr = beta_rel * cm

            # Softmax weights
            c_shifted = (c - cm) / beta_curr
            e = np.exp(c_shifted)
            e_sum = float(e.sum())
            if e_sum < 1e-300:
                p = np.zeros_like(c)
                p[int(np.argmax(c))] = 1.0
            else:
                p = e / e_sum

            # Compute g
            g = _corr_g(a, p)

            # Update: a_new = g / mean(g) with relaxation
            mg = float(g.mean())
            if mg < 1e-30:
                break
            target = g / mg
            target = np.maximum(target, floor)
            target = target / target.sum()

            a_new = (1.0 - t) * a + t * target
            a_new = np.maximum(a_new, floor)
            a_new = a_new / a_new.sum()
            a = a_new

            # Periodic score check
            if it % 100 == 0 or it == iters_per_beta - 1:
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    return best_a, best_score


def _kkt_gradient_descent(a_init, deadline, lr=0.01, n_iter_max=200000, top_k=50):
    """Gradient descent on the KKT residual (making g flat)."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        # Softmax weights with top-k
        if top_k > 0 and top_k < len(c):
            idx = np.argsort(c)[-top_k:]
            c_top = c[idx]
            beta = 0.01 * (c_top.max() - c_top.min() + 1e-15)
            p = np.zeros_like(c)
            c_norm = (c_top - c_top.max()) / beta
            e = np.exp(c_norm)
            p[idx] = e / e.sum()
        else:
            beta = 0.01 * cm
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            p = e / e.sum()

        # Compute g
        g = _corr_g(a, p)
        mg = float(g.mean())
        r = g - mg  # residual (mean zero)

        # Gradient of sum(r^2) w.r.t. a
        grad = _grad_kkt_residual(a, p, r)

        # Update
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        a_sum = float(a_new.sum())
        if a_sum < 1e-15:
            break
        a_new = a_new / a_sum
        a = a_new

        # Score check
        if it % 200 == 0 or it == n_iter_max - 1:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

        # Adaptive lr
        if it % 1000 == 0 and it > 0:
            lr = max(0.001, lr * 0.95)

    return best_a, best_score


def _z_objective(z, beta, n, top_k=0):
    """Smoothed objective in logit space."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_max_top = float(c_top.max())
        c_norm = (c_top - c_max_top) / beta
        log_sum = float(np.log(np.sum(np.exp(c_norm))))
        return 2.0 * n * (c_max_top + beta * log_sum)
    else:
        c_shifted = (c - cm) / beta
        log_sum = float(np.log(np.sum(np.exp(c_shifted))))
        return 2.0 * n * (cm + beta * log_sum)


def _z_gradient(z, beta, n, top_k=0):
    """Gradient of smoothed objective w.r.t. z."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())

    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_max_top = float(c_top.max())
        c_norm = (c_top - c_max_top) / beta
        e = np.exp(c_norm)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[idx[0]] = 1.0
        else:
            w = np.zeros_like(c)
            w[idx] = e / s
    else:
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

    # h_i = sum_k w_k * a_{k-i}
    h = _corr_g(a, w)
    W = float(np.dot(w, c))

    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _lbfgs_polish(a_init, deadline, max_wall_time=None, top_k=0):
    """L-BFGS polish with beta continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if max_wall_time is None:
        max_wall_time = max(1.0, deadline - time.monotonic())

    beta_starts = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max(max_c0, 1e-10) for b in beta_starts]
    per_level = max_wall_time / len(betas)

    for beta in betas:
        if time.monotonic() > deadline:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        try:
            result = minimize(
                _z_objective, z, args=(beta, n, top_k),
                method='L-BFGS-B', jac=_z_gradient,
                options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 2000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                if np.all(np.isfinite(a_new)) and a_new.sum() > 0:
                    a_new = a_new / a_new.sum()
                    sc = _score_norm(a_new)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
                        z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _adam_polish(a_init, deadline, lr=0.03, top_k=10, n_max_iter=50000):
    """Adam optimizer for the smoothed max objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-10, 0.01 * max_c)

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        grad = _z_gradient(z, beta, n, top_k)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 200 == 0:
            beta = max(1e-14, beta * 0.85)
        if t % 2000 == 0 and best_score < 1.51:
            lr = max(0.003, lr * 0.95)

    return best_a, best_score


def _make_exp_profile(n, alpha):
    r = 1.0 - alpha
    a = r ** np.arange(n)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization using KKT fixed-point iteration as the core method.
    Phase 1: KKT fixed-point with continuation (main optimization)
    Phase 2: L-BFGS polish
    Phase 3: Adam polish
    Phase 4: Multi-restart
    Phase 5: Final polish
    """
    global height_sequence_1

    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    print("=== Getting parent ===")
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm(parent)
                update_best(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent: {e}")

    # If no parent, create initial
    if parent is None:
        n_init = 4096
        parent = _make_exp_profile(n_init, 0.003)
        sc = _score_norm(parent)
        update_best(parent, sc, "exp_init")
        print(f"  Fallback init: n={n_init}, score={sc:.10f}")

    n_curr = len(parent)

    # --- Phase 1: KKT Fixed-Point with continuation (60% of time) ---
    print(f"\n=== Phase 1: KKT Fixed-Point (n={n_curr}) ===")
    phase1_end = min(deadline, t_start + float(budget_s) * 0.55)

    # Beta schedule: from coarse to fine
    beta_schedule = [1.0, 0.5, 0.3, 0.2, 0.15, 0.1, 0.08, 0.06, 0.05,
                     0.04, 0.03, 0.025, 0.02, 0.015, 0.012, 0.01, 0.008, 0.006, 0.005, 0.004]

    # Run KKT smooth iteration
    p_best, sc_best = _kkt_smooth_iteration(
        parent, phase1_end, beta_schedule, t=1.0, iters_per_beta=3000
    )
    update_best(p_best, sc_best, "kkt_fp")

    # If still time, run KKT gradient descent (different flavor)
    if time.monotonic() < phase1_end - 60:
        dl2 = min(phase1_end - 10, time.monotonic() + (phase1_end - time.monotonic()) * 0.7)
        p_best2, sc_best2 = _kkt_gradient_descent(
            best_seq if best_seq is not None else parent,
            dl2, lr=0.02, n_iter_max=200000, top_k=30
        )
        update_best(p_best2, sc_best2, "kkt_gd")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: L-BFGS polish (20% of time) ---
    print(f"\n=== Phase 2: L-BFGS Polish ===")
    phase2_end = min(deadline, t_start + float(budget_s) * 0.75)
    if time.monotonic() < phase2_end and best_seq is not None:
        remaining = phase2_end - time.monotonic()
        per_lbfgs = min(200.0, remaining * 0.6)
        dl = min(phase2_end - 10, time.monotonic() + per_lbfgs)
        print(f"  L-BFGS (n={len(best_seq)}), budget={per_lbfgs:.0f}s")
        p_best, sc_best = _lbfgs_polish(best_seq, dl, max_wall_time=per_lbfgs, top_k=0)
        update_best(p_best, sc_best, "lbfgs1")

        # Second L-BFGS with top-k
        if time.monotonic() < phase2_end - 20:
            remaining2 = phase2_end - time.monotonic()
            per2 = min(100.0, remaining2 * 0.8)
            dl2 = min(phase2_end - 5, time.monotonic() + per2)
            p_best2, sc_best2 = _lbfgs_polish(best_seq, dl2, max_wall_time=per2, top_k=10)
            update_best(p_best2, sc_best2, "lbfgs_topk")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Adam polish (10% of time) ---
    print(f"\n=== Phase 3: Adam Polish ===")
    phase3_end = min(deadline, t_start + float(budget_s) * 0.88)
    if time.monotonic() < phase3_end and best_seq is not None:
        remaining = phase3_end - time.monotonic()
        per_adam = min(120.0, remaining * 0.8)
        dl = min(phase3_end - 5, time.monotonic() + per_adam)
        print(f"  Adam (n={len(best_seq)}), budget={per_adam:.0f}s")
        p_best, sc_best = _adam_polish(best_seq, dl, lr=0.03, top_k=10, n_max_iter=50000)
        update_best(p_best, sc_best, "adam1")

        # Second pass with smaller lr
        if time.monotonic() < phase3_end - 10:
            remaining2 = phase3_end - time.monotonic()
            per2 = min(60.0, remaining2 * 0.8)
            dl2 = min(phase3_end - 3, time.monotonic() + per2)
            p_best2, sc_best2 = _adam_polish(best_seq, dl2, lr=0.01, top_k=20, n_max_iter=30000)
            update_best(p_best2, sc_best2, "adam2")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: KKT fixed-point again (exploit) ---
    print(f"\n=== Phase 4: KKT Fixed-Point Re-run ===")
    phase4_end = min(deadline, t_start + float(budget_s) * 0.92)
    if time.monotonic() < phase4_end and best_seq is not None:
        # Use a finer beta schedule since we're already close to optimum
        fine_beta = [0.02, 0.015, 0.012, 0.01, 0.008, 0.006, 0.005, 0.004, 0.003, 0.002]
        remaining = phase4_end - time.monotonic()
        p_best, sc_best = _kkt_smooth_iteration(
            best_seq, phase4_end, fine_beta, t=1.0, iters_per_beta=2000
        )
        update_best(p_best, sc_best, "kkt_fp2")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multi-restart (5% of time) ---
    print(f"\n=== Phase 5: Multi-Restart ===")
    phase5_end = min(deadline, t_start + float(budget_s) * 0.96)
    if time.monotonic() < phase5_end and best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 3
        per_restart = min(30.0, (phase5_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 5:
                break
            noise_levels = [0.02, 0.05, 0.1]
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_levels[i] * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            dl = min(phase5_end - 3, time.monotonic() + per_restart)
            p_m, sc_m = _lbfgs_polish(p_start, dl, max_wall_time=max(5.0, per_restart - 5), top_k=5)
            update_best(p_m, sc_m, f"restart{i}")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Final polish ---
    print(f"\n=== Phase 6: Final Polish ===")
    if time.monotonic() < deadline and best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            dl = min(deadline - 3, time.monotonic() + min(40.0, remaining * 0.6))
            p_best, sc_best = _lbfgs_polish(best_seq, dl, max_wall_time=max(5.0, remaining * 0.4), top_k=5)
            update_best(p_best, sc_best, "final_lbfgs")

        if time.monotonic() < deadline - 5:
            dl2 = min(deadline - 2, time.monotonic() + min(20.0, deadline - time.monotonic() - 3))
            p_best2, sc_best2 = _adam_polish(best_seq, dl2, lr=0.005, top_k=15, n_max_iter=10000)
            update_best(p_best2, sc_best2, "final_adam")

    # --- Final validation and return ---
    print("\n=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            # Scale to a reasonable range (evaluator clips to [0, 1000])
            # The score is scale-invariant, so any positive scaling works
            # Use sum = sqrt(2n) as a natural normalization
            target_sum = np.sqrt(2.0 * len(final_seq))
            final_seq = final_seq * (target_sum / s)
            final_score = _score_full(final_seq)
        else:
            final_seq = _make_exp_profile(4096, 0.003)
            final_score = _score_full(final_seq)

        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")

        # Validate with evaluate_sequence if feasible
        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 10000:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
                # Adjust score reporting if needed
                if np.isfinite(sc_eval) and sc_eval < best_score:
                    pass  # Just for logging
        except Exception as e:
            print(f"  evaluate_sequence check failed: {e}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_exp_profile(4096, 0.003)
    fallback = fallback * np.sqrt(2.0 * 4096)
    print(f"  FALLBACK: n=4096, score={_score_full(fallback):.10f}")
    return [float(x) for x in fallback]
```
