def solve_lp_direction(f, M):
    n = len(f)
    b = cp.Variable(n)
    
    objective = cp.Maximize(cp.sum(b))
    
    constraints = [b >= 0]
    
    # Convolution constraints: (f * b)[k] <= M for k = 0, ..., 2n-2
    # (f * b)[k] = sum_i f[i] * b[k-i] where 0 <= i < n, 0 <= k-i < n
    # So k-i >= 0 => i <= k, and k-i < n => i > k-n
    # i >= 0 and i < n
    # So i ranges from max(0, k-n+1) to min(n-1, k)
    
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # sum_{i=lo}^{hi} f[i] * b[k-i]
        # b[k-i] where i goes from lo to hi, so k-i goes from k-lo to k-hi
        # k-lo = k - max(0, k-n+1)
        # k-hi = k - min(n-1, k)
        
        # Let's index: for i in [lo, hi], the b index is k-i
        # b indices: k-lo, k-lo-1, ..., k-hi (decreasing)
        # So we need f[lo:hi+1] dot b[k-hi : k-lo+1][::-1]

        constraint = cp.sum(f[lo:hi+1] * b[b_indices]) <= M

from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    r = irfft(fa * fb, N)[:out_n]
    return r

def score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def build_conv_matrix(a, n):
    rows = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        rows[k, lo:hi+1] = a[lo:hi+1]
    return rows

def solve_lp(a, M):

    return None, None

```python
import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import time

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    return irfft(rfft(a, N) * rfft(b, N), N)[:out_n]

def score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def score_norm(a):
    """Score assuming sum(a) is already normalized to 1."""
    n = len(a)
    c = full_convolve(a, a)
    return 2.0 * n * c.max()

def build_conv_matrix(a, n):
    """Build sparse convolution matrix T where (T @ b)[k] = (a*b)[k]."""
    row_list = []
    col_list = []
    data_list = []
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        width = hi - lo + 1
        row_list.append(np.full(width, k, dtype=np.int32))
        col_list.append(np.arange(lo, hi+1, dtype=np.int32))
        data_list.append(a[lo:hi+1])
    rows = np.concatenate(row_list)
    cols = np.concatenate(col_list)
    data = np.concatenate(data_list)
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

def lp_direction(a, M):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M."""
    n = len(a)
    T = build_conv_matrix(a, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_obj, A_ub=T, b_ub=b_ub, bounds=bounds,
                        method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if result.status == 0:
            return result.x, -result.fun
    except Exception as e:
        pass
    return None, None
```
