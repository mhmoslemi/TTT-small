```python
"""
Search for low-peak non-negative auto-convolution sequences.

The objective is minimize 2*n*max(a*a) / sum(a)^2.  This program uses a
multi-resolution strategy:
  * small/n/medium length global and local optimization
  * Fourier-accelerated logsumexp-smooth L-BFGS continuation
  * deterministic and random smooth profile generators
  * Adam-style polishing in softmax log-space
  * exact-score local search with pair/block moves

All major helpers are top-level; no closures or lambdas are used.
"""

import numpy as np
from scipy.optimize import minimize
import time


def _now():
    return time.monotonic()


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _auto_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score_array(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if n == 0:
        return float("inf")
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 1e-15:
        return float("inf")
    c = _auto_conv(a)
    m = float(np.max(c))
    if m < 0.0:
        m = 0.0
    val = (2.0 * n * m) / (s * s)
    if not np.isfinite(val):
        return float("inf")
    return float(val)


def _valid_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 1e-12:
        a = np.ones_like(a, dtype=np.float64) / float(a.size)
    else:
        a = a / s
    return a


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.where(np.isfinite(z), z, 0.0)
    mz = float(np.max(z))
    z = z - mz
    e = np.exp(z)
    s = float(np.sum(e))
    if not np.isfinite(s) or s < 1e-300:
        return np.ones_like(z, dtype=np.float64) / float(z.size)
    return e / s


def _tau_fun(z, tau, n, N):
    z = np.asarray(z, dtype=np.float64).ravel()
    a = _softmax(z)
    if not np.isfinite(tau) or tau <= 0.0:
        tau = 1e-20

    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    idx = int(np.argmax(c))
    cm = float(c[idx])

    shift = (c - cm) / tau
    shift = np.minimum(shift, 0.0)
    shift = np.maximum(shift, -745.0)
    e = np.exp(shift)
    s = float(np.sum(e))
    if not np.isfinite(s) or s < 1e-300:
        e = np.zeros_like(c, dtype=np.float64)
        e[idx] = 1.0
        s = 1.0

    w = e / s
    S = cm + tau * float(np.log(s))
    f = float(2.0 * n * S)
    if not np.isfinite(f):
        return (1e100, np.zeros(n, dtype=np.float64))

    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(fw * fa_rev, N)[:2 * n - 1]
    h = conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    g = 4.0 * float(n) * a * (h - W)

    if not np.all(np.isfinite(g)):
        g = np.where(np.isfinite(g), g, 0.0)
    return (f, g)


def _score_key(item):
    return item[0]


def _record_best(best, a, s):
    if a is None or not np.isfinite(s):
        return
    if s < best[1] - 1e-15:
        best[0] = np.asarray(a, dtype=np.float64).copy()
        best[1] = float(s)
        if _now() - best[2] > 5.0:
            best[2] = _now()
            print("new best %.10f n=%d" % (best[1], len(best[0])), flush=True)


def _interp_profile(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size == 0:
        return None
    n0 = len(p)
    if n0 == n_new:
        a = _valid_normalized(p)
        return a
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    q = np.interp(x1, x0, p)
    q = np.maximum(q, 0.0)
    return _valid_normalized(q)


def _is_close_profile(cands, p):
    p = np.asarray(p, dtype=np.float64).ravel()
    for _, q in cands:
        if q.size != p.size:
            continue
        if np.allclose(q, p, rtol=1e-5, atol=1e-8):
            return True
    return False


def _lbfgs_refine(a0, deadline, rng, best, time_budget, tau_rels=None):
    a0 = _valid_normalized(a0)
    if a0 is None:
        return (None, float("inf"))

    n = len(a0)
    N = _next_pow2(2 * n - 1)
    if tau_rels is None:
        tau_rels = [0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001, 0.0004]
    tau_rels = [float(t) for t in tau_rels if t > 0]
    if not tau_rels:
        tau_rels = [0.05]

    end = min(deadline, _now() + max(0.0, float(time_budget)))
    cur_a = a0.copy()
    cur_sc = _score_array(cur_a)
    _record_best(best, cur_a, cur_sc)
    z = np.log(np.maximum(cur_a, 1e-300))
    z = z - float(np.mean(z))

    if n <= 64:
        base_ml = 600
    elif n <= 256:
        base_ml = 400
    elif n <= 1024:
        base_ml = 250
    elif n <= 4096:
        base_ml = 120
    else:
        base_ml = 60

    L = len(tau_rels)
    for idx, rel in enumerate(tau_rels):
        now = _now()
        if now >= end - 2.0:
            break
        level_end = min(end, now + max(2.0, (end - now) / max(1, L - idx)))

        c = _auto_conv(cur_a)
        cm = float(np.max(c))
        if cm <= 0.0 or not np.isfinite(cm):
            cm = 1.0 / max(1, 2 * n - 1)
        tau = max(1e-20, cm * float(rel))

        if idx == 0:
            ml = base_ml * 2
        elif rel >= 0.01:
            ml = base_ml
        else:
            ml = max(30, base_ml // 2)

        stall = 0
        while _now() < level_end - 0.7:
            try:
                res = minimize(
                    _tau_fun,
                    z,
                    args=(tau, n, N),
                    jac=True,
                    method="L-BFGS-B",
                    options={"maxiter": ml, "ftol": 1e-14, "gtol": 1e-8, "maxls": 30},
                )
            except Exception:
                break

            if res is None or res.x is None:
                break
            xnew = np.asarray(res.x, dtype=np.float64)
            if not np.all(np.isfinite(xnew)):
                break

            z = xnew.copy()
            a_trial = _softmax(z)
            sc_trial = _score_array(a_trial)

            improved = False
            if np.isfinite(sc_trial) and sc_trial < cur_sc - 1e-12:
                cur_sc = sc_trial
                cur_a = a_trial.copy()
                improved = True
                _record_best(best, cur_a, cur_sc)

            fun_val = getattr(res, "fun", None)
            if fun_val is None or not np.isfinite(fun_val):
                break

            if bool(res.success):
                if improved:
                    stall = 0
                else:
                    stall += 1
                    if stall >= 3:
                        break
            else:
                stall = 0

    return (cur_a, cur_sc)


def _adam_refine(a0, deadline, best, time_budget, tau_rel=0.003, lr=0.08):
    a0 = _valid_normalized(a0)
    if a0 is None:
        return (None, float("inf"))

    n = len(a0)
    N = _next_pow2(2 * n - 1)
    end = min(deadline, _now() + max(0.0, float(time_budget)))

    best_a = a0.copy()
    best_sc = _score_array(best_a)
    _record_best(best, best_a, best_sc)

    z = np.log(np.maximum(best_a, 1e-300))
    z = z - float(np.mean(z))
    best_z = z.copy()

    cmax = float(np.max(_auto_conv(best_a)))
    if not np.isfinite(cmax) or cmax <= 0.0:
        cmax = 1.0 / max(1, 2 * n - 1)
    tau = max(1e-20, cmax * float(tau_rel))

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    b1, b2, eps = 0.90, 0.999, 1e-9
    t = 0
    noimp = 0
    max_iters = 1000000

    while t < max_iters and _now() < end - 0.5:
        f, g = _tau_fun(z, tau, n, N)
        if not np.isfinite(f):
            break
        if not np.any(np.abs(g) > 1e-15):
            break

        t += 1
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * (g * g)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        z = z - float(lr) * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z = z - float(np.mean(z))

        if t % 25 == 0:
            a_trial = _softmax(z)
            sc_trial = _score_array(a_trial)
            if np.isfinite(sc_trial) and sc_trial < best_sc - 1e-12:
                best_sc = sc_trial
                best_a = a_trial.copy()
                best_z = z.copy()
                noimp = 0
                _record_best(best, best_a, best_sc)
                cmax = float(np.max(_auto_conv(best_a)))
                if np.isfinite(cmax) and cmax > 0.0:
                    tau = max(1e-20, cmax * float(tau_rel))
            else:
                noimp += 1
                if noimp > 50:
                    z = best_z.copy()
                    m = np.zeros(n, dtype=np.float64)
                    v = np.zeros(n, dtype=np.float64)
                    noimp = 0
                    lr = max(0.005, float(lr) * 0.75)

    return (best_a, best_sc)


def _local_search(a0, deadline, best, time_budget, rng):
    a0 = _valid_normalized(a0)
    if a0 is None:
        return (None, float("inf"))

    n = len(a0)
    cur_a = a0.copy()
    cur_sc = _score_array(cur_a)
    _record_best(best, cur_a, cur_sc)

    end = min(deadline, _now() + max(0.0, float(time_budget)))
    sigma = 0.08
    it = 0

    while _now() < end - 0.3:
        it += 1
        mode = int(rng.integers(0, 3))
        trial = None

        if mode == 0:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j and cur_a[i] > 0.0:
                trial = cur_a.copy()
                d = float(rng.uniform(0.02, 0.25)) * float(cur_a[i])
                trial[i] -= d
                trial[j] += d

        elif mode == 1:
            bs_hi = min(33, max(4, n // 8))
            bs = int(rng.integers(3, bs_hi + 1))
            if n - bs >= 0:
                start = int(rng.integers(0, n - bs + 1))
                trial = cur_a.copy()
                trial[start:start + bs] *= np.exp(rng.standard_normal(bs) * sigma)

        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                if cur_a[i] < cur_a[j]:
                    i, j = j, i
                if cur_a[i] > 0.0:
                    trial = cur_a.copy()
                    d = float(rng.uniform(0.05, 0.35)) * float(cur_a[i])
                    trial[i] -= d
                    trial[j] += d

        if trial is None:
            continue

        s = float(np.sum(trial))
        if not np.isfinite(s) or s <= 0.0:
            continue
        trial = trial / s
        sc_trial = _score_array(trial)

        if np.isfinite(sc_trial) and sc_trial < cur_sc - 1e-13:
            cur_a = trial.copy()
            cur_sc = sc_trial
            sigma = max(0.01, sigma * 0.97)
            _record_best(best, cur_a, cur_sc)

    return (cur_a, cur_sc)


def _profile_linear2(n, start, end):
    x = np.linspace(0.0, 1.0, n)
    a = float(start) + (float(end) - float(start)) * x
    a = np.maximum(a, 0.0)
    return _valid_normalized(a)


def _profile_power(n, p_left, p_right):
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-6
    a = (x + eps) ** float(p_left) * ((1.0 - x) + eps) ** float(p_right)
    a = np.maximum(a, 0.0)
    return _valid_normalized(a)


def _profile_quad(n, c1, c2):
    x = np.linspace(0.0, 1.0, n)
    u = 2.0 * x - 1.0
    a = 1.0 + float(c1) * u + float(c2) * u * u
    a = np.maximum(a, 1e-6)
    return _valid_normalized(a)


def _profile_gauss(n, mu, width, skew):
    x = np.linspace(0.0, 1.0, n)
    y = x - float(mu)
    a = np.exp(-(y * y) / (2.0 * float(width) * float(width)) + float(skew) * y)
    a = np.maximum(a, 1e-6)
    return _valid_normalized(a)


def _random_z(n, rng, order=None):
    if order is None:
        order = min(10, n // 2)
    x = np.linspace(0.0, 1.0, n)
    z = rng.standard_normal(n) * 0.05
    z += rng.uniform(-0.8, 0.8) * (x - 0.5)
    z += rng.uniform(-0.3, 0.3)
    for m in range(1, order + 1):
        amp = rng.standard_normal() * (1.0 / (m ** 0.8))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        z += amp * np.cos(2.0 * np.pi * m * x + phase)
        amp2 = rng.standard_normal() * (0.7 / (m ** 0.8))
        phase2 = float(rng.uniform(0.0, 2.0 * np.pi))
        z += amp2 * np.sin(2.0 * np.pi * m * x + phase2)
    return z


def _profile_from_z(z):
    a = _softmax(z)
    return _valid_normalized(a)


def _generate_profiles(n, rng, parent):
    profiles = []

    if parent is not None:
        profiles.append(_interp_profile(parent, n))

    for start, end in [
        (0.0, 1.0), (0.1, 1.0), (0.2, 1.0), (0.35, 1.0),
        (0.5, 1.0), (1.0, 1.0), (2.0, 1.0), (3.0, 1.0),
        (1.0, 0.5), (2.0, 0.5), (3.0, 0.5), (1.0, 0.2), (0.5, 2.0)
    ]:
        profiles.append(_profile_linear2(n, start, end))

    for p, q in [
        (0.0, 1.0), (0.0, 2.0), (0.0, 3.0), (0.0, 4.0),
        (0.5, 1.0), (0.5, 2.0), (0.5, 3.0),
        (1.0, 2.0), (1.0, 3.0), (1.0, 4.0),
        (2.0, 3.0), (2.0, 4.0)
    ]:
        profiles.append(_profile_power(n, p, q))

    for c1, c2 in [
        (-0.5, 0.2), (0.0, 0.2), (0.5, 0.2), (1.0, 0.2),
        (-0.5, -0.2), (0.5, -0.2), (1.0, -0.2),
        (2.0, 0.5), (0.0, 0.5), (-1.0, 0.0)
    ]:
        profiles.append(_profile_quad(n, c1, c2))

    for mu, wd, sk in [
        (0.7, 0.20, 1.0), (0.8, 0.30, -1.0),
        (0.55, 0.25, 1.5), (0.65, 0.20, 2.0),
        (0.45, 0.25, -2.0), (0.30, 0.20, -3.0)
    ]:
        profiles.append(_profile_gauss(n, mu, wd, sk))

    n_rand = 30 if n <= 128 else 20
    for _ in range(n_rand):
        z = _random_z(n, rng)
        profiles.append(_profile_from_z(z))

    return profiles


def _optimize_length(n, deadline, rng, best, time_budget, init_profiles=None, top_k=4, local_frac=0.1):
    if not np.isfinite(time_budget) or time_budget <= 0.0:
        return (None, float("inf"))

    end = min(deadline, _now() + float(time_budget))
    if end <= _now():
        return (None, float("inf"))

    parent = globals().get("height_sequence_1")
    if init_profiles is None:
        init_profiles = _generate_profiles(n, rng, parent)

    scored = []
    for p in init_profiles:
        if p is None:
            continue
        q = np.asarray(p, dtype=np.float64).ravel()
        if q.size == 0:
            continue
        if q.size != n:
            q = _interp_profile(q, n)
        q = _valid_normalized(q)
        if q is None:
            continue
        sc = _score_array(q)
        _record_best(best, q, sc)
        scored.append((sc, q))

    if not scored:
        return (None, float("inf"))

    scored.sort(key=_score_key)
    base = scored[0][0]
    if not np.isfinite(base):
        return (None, float("inf"))

    cand = []
    for sc, q in scored:
        if len(cand) >= top_k:
            break
        if not np.isfinite(sc):
            continue
        if len(cand) > 0 and sc > base * 1.15:
            break
        if not _is_close_profile(cand, q):
            cand.append((sc, q))

    if not cand:
        cand = [scored[0]]

    loc_a = np.asarray(cand[0][1], dtype=np.float64).copy()
    loc_sc = float(cand[0][0])
    _record_best(best, loc_a, loc_sc)

    local_time = 0.0
    if n > 256 and local_frac > 0.0:
        local_time = min(120.0, float(time_budget) * float(local_frac))

    refine_end = min(end, _now() + max(1.0, (end - _now()) - local_time))

    for sc0, q0 in cand:
        now = _now()
        if now >= refine_end - 5.0:
            break

        remaining_cands = max(1, len(cand))
        per = max(5.0, (refine_end - now) / float(remaining_cands))
        arb = min(per, refine_end - now - 2.0, deadline - now - 2.0)
        if sc0 > loc_sc * 1.05:
            arb = min(arb, 25.0)
        if arb < 4.0:
            break

        ra, rs = _lbfgs_refine(q0, deadline, rng, best, arb)
        if ra is not None and np.isfinite(rs) and rs < loc_sc:
            loc_a = np.asarray(ra, dtype=np.float64).copy()
            loc_sc = float(rs)

    if local_time > 3.0 and _now() < end - 3.0:
        la, ls = _local_search(loc_a, deadline, best, min(local_time, end - _now() - 2.0), rng)
        if la is not None and np.isfinite(ls) and ls < loc_sc:
            loc_a = np.asarray(la, dtype=np.float64).copy()
            loc_sc = float(ls)

    return (loc_a, loc_sc)


def _get_parent():
    p = globals().get("height_sequence_1")
    if p is None:
        return None
    return _valid_normalized(p)


def propose_candidate(seed=42, budget_s=1800):
    t0 = _now()
    deadline = t0 + max(1.0, float(budget_s)) - 25.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best = [None, float("inf"), 0.0]

    parent = _get_parent()
    if parent is not None:
        sc_parent = _score_array(parent)
        _record_best(best, parent, sc_parent)
        print("parent: n=%d score=%.10f" % (len(parent), sc_parent), flush=True)

    if parent is not None and len(parent) > 0:
        target_len = int(len(parent))
    else:
        target_len = 4096

    small_a = None
    if _now() < deadline - 40.0:
        small_a, _ = _optimize_length(
            32, deadline, rng, best,
            time_budget=min(60.0, deadline - _now() - 10.0),
            top_k=12,
            local_frac=0.0
        )
        if small_a is not None:
            print("small done: best so far %.10f" % best[1], flush=True)

    medium_len = int(min(1024, max(128, target_len)))
    medium_a = small_a
    if _now() < deadline - 260.0:
        init = []
        if small_a is not None:
            init.append(_interp_profile(small_a, medium_len))
        if parent is not None:
            init.append(_interp_profile(parent, medium_len))
        init.extend(_generate_profiles(medium_len, rng, parent))
        budget = min(260.0, deadline - _now() - 20.0)
        medium_a, _ = _optimize_length(
            medium_len, deadline, rng, best,
            time_budget=budget,
            init_profiles=init,
            top_k=7,
            local_frac=0.08
        )
        if medium_a is not None:
            print("medium done: best so far %.10f" % best[1], flush=True)

    large_len = int(max(512, min(4096, target_len)))
    large_a = medium_a if medium_a is not None else small_a
    if _now() < deadline - 220.0:
        init = []
        for p in (small_a, medium_a, parent):
            if p is not None:
                init.append(_interp_profile(p, large_len))
        init.extend(_generate_profiles(large_len, rng, parent))
        budget = min(820.0, deadline - _now() - 80.0)
        large_a, _ = _optimize_length(
            large_len, deadline, rng, best,
            time_budget=budget,
            init_profiles=init,
            top_k=6,
            local_frac=0.10
        )
        if large_a is not None:
            print("large done: best so far %.10f" % best[1], flush=True)

    if large_a is not None and _now() < deadline - 220.0:
        n2 = int(min(8192, large_len * 2))
        if n2 > large_len:
            init = []
            for p in (large_a, medium_a, small_a, parent):
                if p is not None:
                    init.append(_interp_profile(p, n2))
            init.extend(_generate_profiles(n2, rng, parent))
            budget = min(250.0, deadline - _now() - 80.0)
            n2_a, n2_sc = _optimize_length(
                n2, deadline, rng, best,
                time_budget=budget,
                init_profiles=init,
                top_k=5,
                local_frac=0.08
            )
            if n2_a is not None and np.isfinite(n2_sc):
                print("extended n=%d candidate score=%.10f" % (n2, n2_sc), flush=True)
            _now()

    if best[0] is not None and _now() < deadline - 5.0:
        a = np.asarray(best[0], dtype=np.float64).copy()

        if _now() < deadline - 40.0:
            a, _ = _adam_refine(a, deadline, best, time_budget=min(90.0, deadline - _now() - 20.0), tau_rel=0.003, lr=0.08)

        if _now() < deadline - 25.0:
            a, _ = _local_search(a, deadline, best, min(60.0, deadline - _now() - 10.0), rng)

        if _now() < deadline - 5.0:
            a, _ = _lbfgs_refine(
                a, deadline, rng, best,
                time_budget=min(40.0, deadline - _now() - 4.0),
                tau_rels=[0.003, 0.001, 0.0004, 0.00015]
            )

    if best[0] is not None:
        final = np.asarray(best[0], dtype=np.float64).ravel()
    else:
        final = _profile_linear2(2048, 0.35, 1.0)

    final = _valid_normalized(final)
    if final is None:
        final = np.ones(1024, dtype=np.float64) / 1024.0

    final = np.clip(final, 0.0, 1000.0)
    s = float(np.sum(final))
    if not np.isfinite(s) or s < 0.01:
        final = np.ones(1024, dtype=np.float64) / 1024.0
    else:
        final = final / s

    sc_final = _score_array(final)
    print("FINAL proposed: n=%d score=%.10f" % (len(final), sc_final), flush=True)

    if _now() < deadline - 10.0 and len(final) <= 10000:
        try:
            exact = float(evaluate_sequence([float(x) for x in final]))
            print("FINAL evaluator: %.10f" % exact, flush=True)
        except Exception:
            pass

    return [float(x) for x in final]
```
