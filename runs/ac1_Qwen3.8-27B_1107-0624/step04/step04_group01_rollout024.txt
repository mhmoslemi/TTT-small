import time
from scipy import sparse
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _score_norm(a):

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _build_conv_matrix(a):
    """Build sparse matrix T where (T @ g) = convolve(a, g)."""
    n = len(a)
    # Column j has a[0..n-1] at rows j..j+n-1
    cols = np.repeat(np.arange(n), n)
    rows = np.zeros(n * n, dtype=np.int32)
    for j in range(n):
        rows[j*n:(j+1)*n] = np.arange(j, j + n)
    data = np.tile(a, n)
    T = sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return T

def _solve_lp(a, M):
    """Solve LP: max sum(g) s.t. T@g <= M, g >= 0."""
    n = len(a)
    T = _build_conv_matrix(a)
    # min -sum(g) s.t. T@g <= M, g >= 0
    c_obj = -np.ones(n)
    A_ub = T
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if not res.success:
        return None
    g = res.x
    s = g.sum()
    if s < 1e-15:
        return None
    return g / s  # Normalize to sum=1
