"""
Optimized step function search using projected gradient descent on the probability simplex
with log-sum-exp smoothing, FFT-based convolution, and beta continuation.
Uses direct simplex optimization (no softmax parameterization) for faster convergence.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _cross_corr(w, a):
    """Compute g[i] = sum_k w[k] * a[k-i] for i = 0, ..., len(a)-1.
    w has length 2*len(a)-1."""
    n = len(a)
    N = _next_pow2(2 * n)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    r = np.fft.irfft(W * np.conj(A), N)[:n]
    return r

def _simplex_project(x):
    """Project x onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    csv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (csv - 1))[0][-1]
    theta = (csv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)

def _score_norm(a):
    """Compute 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score(a):
    """Compute the full evaluation score."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _compute_grad_simplex(a, beta):
    """Compute gradient of logsumexp(conv(a,a)/beta) w.r.t. a.
    g[i] = (1/beta) * sum_k softmax(c/beta)[k] * a[k-i]
    """
    n = len(a)
    c = _full_convolve(a, a)
    cm = c.max()
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    g = _cross_corr(w, a) / beta
    return g, c

def _adam_simplex(a_init, deadline, beta_schedule, max_iter=500000, lr=0.1):
    """Adam optimizer on the probability simplex with beta continuation."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            g, c = _compute_grad_simplex(a, beta)
            if not np.all(np.isfinite(g)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * g
            v_buf = b2 * v_buf + (1.0 - b2) * (g * g)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            a_new = a - current_lr * mh / (np.sqrt(vh) + eps)
            a = _simplex_project(a_new)

            if t % 200 == 0:
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

            if it_in_phase % 5000 == 0:
                current_lr = max(0.01, current_lr * 0.95)

    a_final = _simplex_project(a)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score

def _fista_simplex(a_init, deadline, beta_schedule, max_iter=500000):
    """FISTA (accelerated projected gradient) on the simplex."""
    n = len(a_init)
    a = a_init.copy()
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = _score_norm(a)

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        # Compute initial step size
        # Lipschitz constant approximation
        c0 = _full_convolve(y, y)
        L = float(c0.max()) / (beta * max(1e-10, y.max())) + 1.0
        lr = 1.0 / L
        lr = max(lr, 0.001)
        lr = min(lr, 1.0)

        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1

            g, c = _compute_grad_simplex(y, beta)
            if not np.all(np.isfinite(g)):
                break

            a_new = _simplex_project(y - lr * g)

            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = a_new + (t_k - 1.0) / t_new * (a_new - a)
            y = _simplex_project(y)
            a = a_new
            t_k = t_new

            if it_in_phase % 500 == 0:
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    sc_final = _score_norm(a)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cosine":
        t = np.linspace(0, np.pi, n)
        a = np.cos(t / 2.0) ** 2
        a = np.maximum(a, 1e-15)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-15)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _perturb(a, sigma, rng):
    """Add noise and reproject to simplex."""
    a_perturbed = a + sigma * rng.standard_normal(len(a))
    return _simplex_project(a_perturbed)

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()

"""
Optimized step function search using projected Adam/FISTA on the probability simplex
with log-sum-exp smoothing, FFT-based convolution, and beta continuation.
Uses direct simplex optimization for faster convergence, supports multiple sequence
lengths, and includes coordinate-aware polishing for final improvements.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _cross_corr(w, a):
    """Compute g[i] = sum_k w[k] * a[k-i] for i = 0..len(a)-1.
    w has length 2*len(a)-1."""
    n = len(a)
    N = _next_pow2(2 * n)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    r = np.fft.irfft(W * np.conj(A), N)[:n]
    return r


def _simplex_project(x):
    """Project x onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    csv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (csv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (csv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)


def _score_norm(a):
    """Compute 2*n*max(conv(a,a)) for a with sum=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score(a):
    """Compute the full evaluation score 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_grad(a, beta):
    """Gradient of logsumexp(conv(a,a)/beta) w.r.t. a.
    Returns (grad, c) where grad[i] = 2/beta * sum_k w[k]*a[k-i], w=softmax(c/beta)."""
    n = len(a)
    c = _full_convolve(a, a)
    cm = c.max()
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    g = 2.0 * _cross_corr(w, a) / beta
    return g, c


def _adam_simplex(a_init, deadline, beta_schedule, max_iter=500000, lr=0.5):
    """Adam optimizer on the probability simplex with beta continuation."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            g, c = _compute_grad(a, beta)
            if not np.all(np.isfinite(g)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * g
            v_buf = b2 * v_buf + (1.0 - b2) * (g * g)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            a_new = a - current_lr * mh / (np.sqrt(vh) + eps)
            a = _simplex_project(a_new)

            if t % 100 == 0:
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

            if it_in_phase % 10000 == 0:
                current_lr = max(0.05, current_lr * 0.9)

    sc_final = _score_norm(a)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score


def _fista_simplex(a_init, deadline, beta_schedule, max_iter=500000):
    """FISTA (accelerated projected gradient) on the simplex."""
    n = len(a_init)
    a = a_init.copy()
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = _score_norm(a)

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        # Estimate Lipschitz constant
        c0 = _full_convolve(y, y)
        max_c0 = float(c0.max())
        L = 2.0 * max_c0 / (beta * max(1e-10, float(y.max()))) + 1.0
        lr = 1.0 / max(L, 1e-10)
        lr = np.clip(lr, 0.001, 1.0)

        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1

            g, c = _compute_grad(y, beta)
            if not np.all(np.isfinite(g)):
                break

            a_new = _simplex_project(y - lr * g)

            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = a_new + (t_k - 1.0) / t_new * (a_new - a)
            y = _simplex_project(y)
            a = a_new
            t_k = t_new

            if it_in_phase % 200 == 0:
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    sc_final = _score_norm(a)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score


def _gradient_descent_simplex(a_init, deadline, beta, lr, max_iter=200000):
    """Plain projected gradient descent (useful for very fine polish)."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iter):

            if np.isfinite(sc) and sc < best_score:

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cosine":
        t = np.linspace(0, np.pi, n)
        a = np.cos(t / 2.0) ** 2
        a = np.maximum(a, 1e-15)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-15)
        return a / a.sum()
    elif kind == "tukey":
        alpha = 0.3
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        half = alpha / 2.0
        for i in range(n):
            if t[i] < half:
                a[i] = 0.5 + 0.5 * np.cos(np.pi / alpha - np.pi * t[i] / half)
            elif t[i] > 1 - half:
                a[i] = 0.5 + 0.5 * np.cos(np.pi / alpha - np.pi * (1 - t[i]) / half)
        a = np.maximum(a, 1e-15)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    """Add noise and reproject to simplex."""
    a_perturbed = a + sigma * rng.standard_normal(len(a))
    return _simplex_project(a_perturbed)


def _perturb_fourier(a, rng):
    """Add a low-frequency Fourier perturbation."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    freq = rng.integers(1, max(2, n // 8))
    amp = 0.1 * rng.uniform(0.5, 2.0)
    phase = rng.uniform(0, 2 * np.pi)
    t = np.arange(n)
    z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    a_new = np.exp(z)
    return _simplex_project(a_new)


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _coordinate_polish(a, deadline, max_rounds=50):
    """Coordinate descent: try swapping mass between pairs of coordinates."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        # Randomly sample pairs
        for _ in range(n):
            if time.monotonic() > deadline:
                break
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            # Try moving mass from i to j
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                a_trial = best_a.copy()
                transfer = a_trial[i] * frac
                a_trial[i] -= transfer
                a_trial[j] += transfer
                if a_trial[i] < 0:
                    continue
                sc = _score_norm(a_trial)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_trial
                    improved = True
                    break
        if not improved and rnd > 10:
            break

    return best_a, best_score

            bs = [0.01 * max_cs, 0.005 * max_cs, 0.002 * max_cs,
                  0.001 * max_cs, 0.0005 * max_cs, 0.0002 * max_cs]

            p_best, sc_best = _adam_simplex(
                p_start, time.monotonic() + per_strat,
                bs, max_iter=300000, lr=0.3
            )
            update_best(p_best, _score(p_best), f"restart_{name}_n{len(p_best)}")

    print(f"Phase 3 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 4: Fine polish (1200s - 1550s) ---
    phase4_end = min(deadline, t_start + 1550.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        c_best = _full_convolve(best_seq, best_seq)
        max_cb = float(c_best.max())
        beta_fine = [
            0.0002 * max_cb, 0.0001 * max_cb, 0.00005 * max_cb,
            0.00002 * max_cb
        ]
        per_time = (phase4_end - time.monotonic() - 5) / 2
        p_best, sc_best = _adam_simplex(
            best_seq, time.monotonic() + per_time,
            beta_fine, max_iter=500000, lr=0.15
        )
        update_best(p_best, _score(p_best), "fine_adam")

        if time.monotonic() < phase4_end - 5:
            # Very fine gradient descent
            c_cur = _full_convolve(p_best, p_best)
            max_cc = float(c_cur.max())
            beta_very_fine = 0.00001 * max_cc
            lr_fine = 0.01 * max_cc / max(1e-10, float(_compute_grad(p_best, beta_very_fine)[0].max()))
            lr_fine = np.clip(lr_fine, 0.001, 0.1)
            p_best2, sc_best2 = _gradient_descent_simplex(
                p_best, time.monotonic() + (phase4_end - time.monotonic() - 3),
                beta_very_fine, lr_fine, max_iter=200000
            )
            update_best(p_best2, _score(p_best2), "fine_gd")

    print(f"Phase 4 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 5: Coordinate polish (1550s - 1700s) ---
    phase5_end = min(deadline, t_start + 1700.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        p_polished, sc_pol = _coordinate_polish(
            best_seq, time.monotonic() + (phase5_end - time.monotonic() - 3),
            max_rounds=30
        )
        update_best(p_polished, _score(p_polished), "coord_polish")

    print(f"Phase 5 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 6: Ultra-fine final polish (1700s - deadline) ---
    if best_seq is not None and time.monotonic() < deadline - 5:
        c_final = _full_convolve(best_seq, best_seq)
        max_cf = float(c_final.max())
        beta_ultra = [0.00001 * max_cf, 0.000005 * max_cf, 0.000002 * max_cf]
        per_time = deadline - time.monotonic() - 3
        if per_time > 5:
            p_best, sc_best = _adam_simplex(
                best_seq, time.monotonic() + per_time,
                beta_ultra, max_iter=200000, lr=0.05
            )
            update_best(p_best, _score(p_best), "ultra_fine")

    print(f"Phase 6 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256, dtype=np.float64)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256, dtype=np.float64)
    return [float(x) for x in fallback]
