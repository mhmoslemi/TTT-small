"""
Adaptive-temperature spectral flattening optimizer for minimizing
2*n*max(convolve(a,a))/(sum(a))^2.
Uses anti-symmetric modulation initializations, FFT-based gradients,

Target: beat 1.5027 (lower is better).

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b, N=None):
    n, m = len(a), len(b)
    out_n = n + m - 1
    if N is None:
        N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    b = np.fft.irfft(A * A, N)[:2 * n - 1]
    return float(2.0 * n * b.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _make_antisymmetric_profile(n, freq, amp):
    """Create a profile with anti-symmetric modulation about center."""
    t = np.arange(n)
    center = (n - 1) / 2.0
    phase = np.sin(2.0 * np.pi * freq * (t - center) / n)
    a = 1.0 + amp * phase
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _make_mixed_profile(n, freqs, amps):
    """Create a profile with multiple anti-symmetric sinusoidal modulations."""
    t = np.arange(n)
    center = (n - 1) / 2.0
    a = np.ones(n)
    for f, amp in zip(freqs, amps):
        a += amp * np.sin(2.0 * np.pi * f * (t - center) / n)
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _grad_score_z(z, beta, N):
    """
    Compute gradient of smoothed score w.r.t. z (log-parameters).
    a = softmax(z), score = 2n * max(b), b = conv(a,a).
    Smoothed: 2n * (beta * log(sum(exp(b/beta)))).
    Returns: (smoothed_value, gradient_wrt_z)
    """
    n = len(z)
    a = _softmax(z)

    A = np.fft.rfft(a, N)
    b = np.fft.irfft(A * A, N)[:2 * n - 1]
    cm = float(b.max())

    shifted = (b - cm) / beta
    e = np.exp(shifted)
    se = float(e.sum())
    if se < 1e-300 or not np.isfinite(se):
        w = np.zeros_like(b)
        w[int(np.argmax(b))] = 1.0
    else:
        w = e / se

    W_val = float(np.dot(w, b))
    smooth_val = 2.0 * n * (cm + beta * np.log(se)) if se > 0 else 2.0 * n * cm

    # Gradient of smoothed max w.r.t. a:
    # grad_a[j] = 2 * sum_m a[m] * w[j+m] for m=0..n-1
    # = 2 * conv(a_rev, w)[j + n - 1]
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, np.nextafter(N, N * 2) if N < 12286 else N)
    # Actually let's just use a separate N for this
    N_corr = _next_pow2(n + (2 * n - 1) - 1)  # 3n-2
    A_rev = np.fft.rfft(a_rev, N_corr)
    Wf = np.fft.rfft(w, N_corr)
    corr = np.fft.irfft(A_rev * Wf, N_corr)
    grad_a = 2.0 * corr[n - 1:2 * n - 1]

    # d(score)/d(z_j) = 2n * a_j * (grad_a_j - dot(a, grad_a))
    S = float(np.dot(a, grad_a))
    grad_z = 2.0 * n * a * (grad_a - S)

    if not np.all(np.isfinite(grad_z)):
        return smooth_val, np.zeros_like(z)
    return smooth_val, grad_z

        a = _make_antisymmetric_profile(n, freq, 0.5)
        sc = _score_norm(a)
        candidates.append((sc, a, f"anti_f{freq}"))

    # Mixed profiles
    for f1, f2, a1, a2 in [(1, 3, 0.5, 0.3), (2, 5, 0.4, 0.3), (1, 5, 0.5, 0.2),
                            (3, 8, 0.4, 0.3), (2, 7, 0.5, 0.25)]:
        if time.monotonic() > p0_end:
            break
        a = _make_mixed_profile(n, [f1, f2], [a1, a2])
        sc = _score_norm(a)
        candidates.append((sc, a, f"mix_{f1}_{f2}"))

    # Linear profile
    a_lin = _make_linear_profile(n, 1.5)
    sc_lin = _score_norm(a_lin)
    candidates.append((sc_lin, a_lin, "linear"))

    candidates.sort(key=lambda x: x[0])
    print(f"  Top candidate: {candidates[0][3]}, score={candidates[0][0]:.10f}")

    # Start from best candidate (prefer parent if it's already good)
    if parent is not None and len(candidates) > 0:
        z_start = np.log(np.maximum(parent, 1e-15))
        z_start = z_start - z_start.mean()
        a_start = parent.copy()
    else:
        a_start = candidates[0][1].copy()
        z_start = np.log(np.maximum(a_start, 1e-15))
        z_start = z_start - z_start.mean()

    # --- Phase 1: Nesterov continuation (500s) ---
    print("Phase 1: Nesterov continuation...")
    p1_end = min(deadline, t_start + 520.0)

    a0 = _softmax(z_start)
    c0 = _fft_convolve(a0, a0)
    max_c0 = float(c0.max())
    beta_start = 0.02 * max_c0

    if time.monotonic() < p1_end - 10:
        z_best, sc_best, a_best = _nesterov_fast(z_start, p1_end, beta_start, lr_start=0.025)
        update_best(a_best, sc_best, "nesterov_p1")

    # Try second-best candidate too
    if time.monotonic() < p1_end - 30:
        for sc_c, a_c, label in candidates[1:3]:
            if time.monotonic() > p1_end - 10:
                break
            if len(a_c) != n:
                continue
            z_c = np.log(np.maximum(a_c, 1e-15))
            z_c = z_c - z_c.mean()
            a_c0 = _fft_convolve(a_c, a_c)
            mc0 = float(a_c0.max())
            per_dl = min(p1_end - 5, time.monotonic() + 80.0)
            z_best2, sc_best2, a_best2 = _nesterov_fast(z_c, per_dl, 0.02 * mc0, lr_start=0.02)
            update_best(a_best2, sc_best2, f"nesterov_{label}")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Adam polishing (300s) ---
    print("Phase 2: Adam polishing...")
    p2_end = min(deadline, t_start + 820.0)

    if best_seq is not None and time.monotonic() < p2_end:
        z_curr = np.log(np.maximum(best_seq, 1e-15))
        z_curr = z_curr - z_curr.mean()
        per_time = min(280.0, p2_end - time.monotonic() - 5)
        if per_time > 20:
            z_best, sc_best, a_best = _adam_loop_fixed(
                z_curr, min(p2_end - 3, time.monotonic() + per_time),
                lr=0.04, beta_frac=0.008
            )
            update_best(a_best, sc_best, "adam_p2")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Structured restarts (400s) ---
    print("Phase 3: Structured restarts...")
    p3_end = min(deadline, t_start + 1220.0)

    if best_seq is not None and time.monotonic() < p3_end:
        n_best = len(best_seq)
        z_base = np.log(np.maximum(best_seq, 1e-15))
        z_base = z_base - z_base.mean()

        n_restarts = 6
        per_restart = min(60.0, (p3_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > p3_end - 10:
                break

            # Structured perturbation: add a sinusoid
            freq = rng.integers(1, 30)
            amp = 0.03 + 0.02 * i
            t_arr = np.arange(n_best)
            center = (n_best - 1) / 2.0
            perturbation = amp * np.sin(2.0 * np.pi * freq * (t_arr - center) / n_best)
            z_pert = z_base + perturbation
            z_pert = z_pert + 0.02 * rng.normal(size=n_best)
            z_pert = np.clip(z_pert, -40, 40)
            z_pert = z_pert - z_pert.mean()

            per_dl = min(p3_end - 5, time.monotonic() + per_restart)
            a_p = _softmax(z_pert)
            c_p = _fft_convolve(a_p, a_p)
            mc_p = float(c_p.max())
            z_best, sc_best, a_best = _nesterov_fast(
                z_pert, per_dl, 0.015 * mc_p, lr_start=0.015
            )
            update_best(a_best, sc_best, f"restart{i}_f{freq}")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Active-set polish (300s) ---
    print("Phase 4: Active-set polish...")
    p4_end = min(deadline, t_start + 1520.0)

    if best_seq is not None and time.monotonic() < p4_end:
        z_curr = np.log(np.maximum(best_seq, 1e-15))
        z_curr = z_curr - z_curr.mean()
        per_time = min(280.0, p4_end - time.monotonic() - 5)
        if per_time > 20:
            z_best, sc_best, a_best = _active_set_polish(
                z_curr, min(p4_end - 3, time.monotonic() + per_time)
            )
            update_best(a_best, sc_best, "active_set_p4")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Final fine polish (remaining time) ---
    print("Phase 5: Final fine polish...")
    p5_end = deadline

    if best_seq is not None and time.monotonic() < p5_end:
        z_curr = np.log(np.maximum(best_seq, 1e-15))
        z_curr = z_curr - z_curr.mean()

        # Very slow Nesterov with tiny beta
        a_curr = _softmax(z_curr)
        c_curr = _fft_convolve(a_curr, a_curr)
        mc_curr = float(c_curr.max())
        beta_tiny = 1e-5 * mc_curr

        per_time = min(150.0, p5_end - time.monotonic() - 30)
        if per_time > 20:
            z_best, sc_best, a_best = _nesterov_fast(
                z_curr, min(p5_end - 30, time.monotonic() + per_time),
                beta_tiny, lr_start=0.005
            )
            update_best(a_best, sc_best, "fine_nesterov")

        # Final Adam
        z_curr = np.log(np.maximum(best_seq, 1e-15))
        z_curr = z_curr - z_curr.mean()
        if time.monotonic() < p5_end - 10:
            per_time2 = min(100.0, p5_end - time.monotonic() - 5)
            if per_time2 > 10:
                z_best, sc_best, a_best = _adam_loop_fixed(
                    z_curr, min(p5_end - 3, time.monotonic() + per_time2),
                    lr=0.015, beta_frac=0.002
                )
                update_best(a_best, sc_best, "fine_adam")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_antisymmetric_profile(4096, 1, 0.5)

        sc_final = _score_fn_safe(final_seq)
        print(f"FINAL internal: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned inf, using internal: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_antisymmetric_profile(4096, 1, 0.5)
    print(f"FALLBACK: n=4096, score={_score_fn_safe(fallback):.10f}")
    return [float(x) for x in fallback]


def _score_fn_safe(a):
    try:
        n = len(a)
        s = float(a.sum())
        if s < 0.01:
            return np.inf
        c = _fft_convolve(a, a)
        return float(2.0 * n * c.max() / (s * s))
    except Exception:
        return np.inf
