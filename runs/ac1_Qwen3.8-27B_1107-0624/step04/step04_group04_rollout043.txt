def _make_mod_power_profile(n, alpha, slope=0.0):
    i=np.arange(n,dtype=float)
    base=(i+1.0)**(-alpha)
    if slope !=0:
        x=i/(max(1,n-1))
        mod=1.0 + slope*x
        mod=np.maximum(mod, 1e-3)
        base=base*mod
    s=base.sum()
    if s<=0: return np.ones(n)/n

Search for low-peak non-negative step functions minimizing
    2*n*max(convolve(a,a)) / sum(a)^2.

Uses a mixture of:
- explicit and randomised low-dimensional starts,

from scipy import optimize, sparse


def _normalise(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return z.copy()
    z = z - np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    return e / s


def _z_from_a(a):
    a = _normalise(a)
    if a.size == 0:
        return a
    z = np.log(np.maximum(a, 1e-300))
    z = z - np.mean(z)
    return np.clip(z, -60.0, 60.0)


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _self_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    out = 2 * n - 1
    N = _next_pow2(out)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:out]
    return np.maximum(c, 0.0)


def _full_convolve(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n, m = a.size, b.size
    if n == 0 or m == 0:
        return np.array([], dtype=np.float64)
    out = n + m - 1
    N = _next_pow2(out)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:out]
    return np.maximum(c, 0.0)


def _score_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    c = _self_convolve(a)
    return float(2.0 * a.size * c.max())


def _score_norm(a):
    a = _normalise(a)
    if a.size == 0:
        return np.inf
    return _score_sum1(a)


def _adaptive_beta(c, K, min_beta=None):
    cm = float(np.max(c))
    if not np.isfinite(cm):
        cm = 1.0
    if min_beta is None:
        min_beta = max(1e-14, 1e-9 * max(1.0, cm))
    K = int(max(1, min(K, c.size)))
    if K <= 1:
        return min_beta
    if K >= c.size:
        return max(min_beta, 0.02 * max(cm, 1e-30))
    cut = c.size - K
    kth = float(np.partition(c, cut)[cut])
    gap = cm - kth
    if gap <= 1e-18 * max(1.0, cm):
        return min_beta
    beta = gap / math.log(float(K))
    low = min_beta
    high = max(min_beta * 100.0, 0.1 * max(cm, 1e-30))
    if beta < low:
        beta = low
    if beta > high:
        beta = high
    return float(beta)


def _z_obj_full(z, beta, n):
    a = _softmax(z)
    c = _self_convolve(a)
    cm = float(c.max())
    if not np.isfinite(cm):
        return 1e100
    shift = (c - cm) / beta
    e = np.exp(shift)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        lse = cm
    else:
        lse = cm + beta * math.log(s)
    return float(2.0 * n * lse)


def _z_grad_full(z, beta, n):
    a = _softmax(z)
    c = _self_convolve(a)
    cm = float(c.max())
    shift = (c - cm) / beta
    e = np.exp(shift)
    s = float(e.sum())
    L = c.size
    w = np.zeros(L, dtype=np.float64)
    if not np.isfinite(s) or s <= 1e-300:
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros(n, dtype=np.float64)
    return grad


def _lbfgs_one(a_init, beta, deadline, max_wall_time):
    a0 = _normalise(a_init)
    n = a0.size
    if n < 2 or max_wall_time <= 1.0:
        return a0.copy(), _score_norm(a0)

    z = _z_from_a(a0)
    best = a0.copy()
    best_score = _score_norm(a0)

    try:
        t0 = time.monotonic()
        _self_convolve(a0)
        _self_convolve(a0)
        _full_convolve(a0, a0)
        t_eval = (time.monotonic() - t0) / 3.0
        if not np.isfinite(t_eval) or t_eval <= 1e-8:
            t_eval = 1e-4
        maxfun = int(min(500, max(60, max_wall_time / (t_eval * 50.0))))
        maxiter = max(30, maxfun // 4)

        res = optimize.minimize(
            _z_obj_full,
            z,
            args=(beta, n),
            method="L-BFGS-B",
            jac=_z_grad_full,
            options={
                "maxiter": int(maxiter),
                "maxfun": int(maxfun),
                "ftol": 1e-15,
                "gtol": 1e-12,
                "maxls": 20
            }
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            a1 = _softmax(res.x)
            s1 = _score_norm(a1)
            if np.isfinite(s1) and s1 < best_score:
                best = a1.copy()
                best_score = float(s1)
    except Exception:
        pass

    return best, best_score


def _lbfgs_schedule(a_init, deadline, k_start=8, k_max=512, pass_time=30.0, cycles=2):
    a = _normalise(a_init)
    best = a.copy()
    best_score = _score_norm(a)
    n = a.size
    k_max = int(max(2, min(k_max, max(8, n))))
    K = max(1, int(k_start))
    cyc = 0
    while time.monotonic() < deadline and cyc < cycles:
        rem = deadline - time.monotonic()
        if rem <= 5.0:
            break
        c = _self_convolve(a)
        beta = _adaptive_beta(c, K)
        wall = min(pass_time, rem - 2.0)
        if wall < 5.0:
            break
        a1, s1 = _lbfgs_one(a, beta, deadline, wall)
        if s1 < best_score:
            best = a1.copy()
            best_score = float(s1)
            a = a1.copy()
        else:
            a = best.copy()
        K = int(min(k_max, K * 1.5 + 2))
        if K >= k_max:
            K = max(2, k_start // 2)
            cyc += 1
    return best, best_score


def _active_h_uniform(c, K, a):
    n = a.size
    L = c.size
    if K <= 1:
        idxs = np.array([int(np.argmax(c))], dtype=int)
        h = np.zeros(n, dtype=np.float64)
        for kk in idxs:
            lo = max(0, int(kk) - (n - 1))
            hi = min(n - 1, int(kk))
            if lo <= hi:
                h[lo:hi + 1] += a[int(kk) - hi:int(kk) - lo + 1][::-1]
        return h, float(c[idxs[0]])

    if K >= L:
        w = np.full(L, 1.0 / float(L), dtype=np.float64)
        wc = _full_convolve(w, a[::-1])
        h = wc[n - 1:2 * n - 1]
        return h, float(np.mean(c))

    cut = L - K
    idxs = np.argpartition(c, cut)[cut:]
    idxs = idxs.astype(int, copy=False)

    if K <= 64:
        h = np.zeros(n, dtype=np.float64)
        for kk in idxs:
            lo = max(0, int(kk) - (n - 1))
            hi = min(n - 1, int(kk))
            if lo <= hi:
                h[lo:hi + 1] += a[int(kk) - hi:int(kk) - lo + 1][::-1]
        h /= float(K)
        W = float(c[idxs].mean())
        return h, W

    w = np.zeros(L, dtype=np.float64)
    w[idxs] = 1.0 / float(K)
    wc = _full_convolve(w, a[::-1])
    h = wc[n - 1:2 * n - 1]
    W = float(c[idxs].mean())
    return h, W


def _balance_pass(a_init, deadline, K=20, max_iter=100000, initial_step=0.25, use_direct=True):
    a = _normalise(a_init)
    n = a.size
    if n < 2:
        return a.copy(), _score_norm(a)

    z = _z_from_a(a)
    best = a.copy()
    best_score = _score_norm(a)
    step = float(initial_step)
    stall = 0
    it = 0

    while it < max_iter and time.monotonic() < deadline:
        c = _self_convolve(a)
        cmax = float(c.max())
        cur = float(2.0 * n * cmax)
        if np.isfinite(cur) and cur < best_score:
            best_score = cur
            best = a.copy()

        h, W = _active_h_uniform(c, K, a)
        g = h - W

        if use_direct:
            d = -g
        else:
            d = -a * g

        dmax = float(np.max(np.abs(d)))
        if not np.isfinite(dmax) or dmax <= 1e-18:
            break
        d /= dmax

        improved = False
        try_alpha = step
        if try_alpha < 1e-5:
            break

        for _ in range(12):
            zn = z + try_alpha * d
            an = _softmax(zn)
            sc = _score_sum1(an)
            if np.isfinite(sc) and sc < best_score - 1e-15:
                z = zn
                a = an
                best = an.copy()
                best_score = float(sc)
                improved = True
                break
            try_alpha *= 0.5
            if try_alpha < 1e-6:
                break

        if improved:
            step = min(3.0, step * 1.15)
            stall = 0
        else:
            step *= 0.5
            stall += 1
            if stall > 120:
                break
        it += 1

        if it % 500 == 0:
            print("      balance K={:5d} it={:6d} score={:.10f}".format(K, it, best_score))

    return best, best_score


def _balance_schedule(a_init, deadline, k_start=1, k_max=None, pass_time=25.0, use_direct=True, cycles=2):
    a = _normalise(a_init)
    best = a.copy()
    best_score = _score_norm(a)
    n = a.size
    if k_max is None:
        k_max = int(min(2 * n, max(256, n // 2)))
    k_max = max(2, int(k_max))
    K = max(1, int(k_start))
    cyc = 0
    while time.monotonic() < deadline and cyc < cycles:
        rem = deadline - time.monotonic()
        if rem <= 3.0:
            break
        wall = min(pass_time, rem - 2.0)
        if wall < 2.0:
            break
        a1, s1 = _balance_pass(a, time.monotonic() + wall, K=K, use_direct=use_direct)
        if s1 < best_score:
            best = a1.copy()
            best_score = float(s1)
            a = a1.copy()
        else:
            a = best.copy()
        K = int(min(k_max, K * 1.6 + 1))
        if K >= k_max:
            K = max(1, k_start // 2 + 1)
            cyc += 1
            if cyc == 1:
                use_direct = not use_direct
    return best, best_score


def _make_mod_power_profile(n, alpha, slope=0.0):
    n = int(max(2, n))
    i = np.arange(n, dtype=np.float64)
    base = (i + 1.0) ** (-float(alpha))
    if abs(float(slope)) > 1e-12:
        x = i / float(max(1, n - 1))
        mod = 1.0 + float(slope) * x
        mod = np.maximum(mod, 1e-3)
        base *= mod
    s = float(base.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    return base / s


def _make_wave_profile(n, alpha, omega, amp, phase):
    n = int(max(2, n))
    i = np.arange(n, dtype=np.float64)
    base = (i + 1.0) ** (-float(alpha))
    mod = 1.0 + float(amp) * np.cos(float(omega) * i + float(phase))
    mod = np.maximum(mod, 1e-3)
    a = base * mod
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return _make_mod_power_profile(n, alpha)
    return a / s


def _interp_profile(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n_new = int(max(2, n_new))
    if p.size == 0:
        return _make_mod_power_profile(n_new, 0.5)
    if p.size == n_new:
        return _normalise(p)
    x0 = np.linspace(0.0, 1.0, p.size)
    x1 = np.linspace(0.0, 1.0, n_new)
    q = np.interp(x1, x0, p)
    q = np.maximum(q, 0.0)
    s = float(q.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return _make_mod_power_profile(n_new, 0.5)
    return q / s


def _fallback_sequence(n=4096):
    return _make_mod_power_profile(int(n), 0.5)


def _consider_candidate(best_seq, best_score, cand, label, t_start):
    if cand is None:
        return best_seq, best_score
    try:
        cnd = np.asarray(cand, dtype=np.float64).ravel()
    except Exception:
        return best_seq, best_score
    if cnd.size == 0:
        return best_seq, best_score
    if not np.all(np.isfinite(cnd)):
        return best_seq, best_score
    cnd = np.clip(cnd, 0.0, 1000.0)
    s = float(cnd.sum())
    if not np.isfinite(s) or s < 0.01:
        return best_seq, best_score
    cnd = cnd / s
    scv = _score_norm(cnd)
    if not np.isfinite(scv):
        return best_seq, best_score
    if scv < best_score:
        elapsed = time.monotonic() - t_start
        print("  NEW BEST score={:.10f} n={:d} label={} elapsed={:.1f}s".format(scv, len(cnd), label, elapsed))
        return cnd.copy(), float(scv)
    return best_seq, best_score


def _perturb(a, rng, level):
    a = _normalise(a)
    z = _z_from_a(a)
    z = z + float(level) * rng.standard_normal(a.size)
    z = z - np.mean(z)
    return _softmax(np.clip(z, -60.0, 60.0))


def _lp_best_response(a, max_time, top_rows=300, full_if_n=2048):
    a = _normalise(a)
    n = a.size
    if n < 2 or max_time <= 1.0:
        return None, None

    L = 2 * n - 1
    c = _self_convolve(a)
    full = (n <= int(full_if_n))

    if full:
        k_list = np.arange(L, dtype=int)
    else:
        R = int(min(L, max(50, top_rows)))
        R = min(R, L - 1)
        if R <= 0:
            R = 1
        cut = L - R
        idxs = np.argpartition(c, cut)[cut:].astype(int, copy=False)
        rows = [int(x) for x in idxs]
        extras = [n - 1, 0, L - 1]
        for kk in extras:
            kk = int(kk)
            if 0 <= kk < L and kk not in rows:
                rows.append(kk)
        k_list = np.array(rows, dtype=int)

    rows_n = k_list.size
    cols = n + 1
    row_parts = []
    col_parts = []
    data_parts = []

    for ri, kk in enumerate(k_list):
        kk = int(kk)
        lo = max(0, kk - (n - 1))
        hi = min(n - 1, kk)
        cnt = hi - lo + 1
        if cnt > 0:
            cols_j = np.arange(lo, hi + 1, dtype=int)
            vals = a[kk - cols_j].astype(np.float64, copy=False)
            row_parts.append(np.full(cnt, ri, dtype=int))
            col_parts.append(cols_j)
            data_parts.append(vals)
        row_parts.append(np.array([ri], dtype=int))
        col_parts.append(np.array([n], dtype=int))
        data_parts.append(np.array([-1.0], dtype=np.float64))

    try:
        data = np.concatenate(data_parts)
        rows_arr = np.concatenate(row_parts)
        cols_arr = np.concatenate(col_parts)
        A_ub = sparse.coo_matrix((data, (rows_arr, cols_arr)), shape=(rows_n, cols)).tocsc()
    except Exception:
        return None, None

    c_obj = np.zeros(cols, dtype=np.float64)
    c_obj[-1] = 1.0

    try:
        eq_data = np.ones(n, dtype=np.float64)
        eq_row = np.zeros(n, dtype=int)
        eq_col = np.arange(n, dtype=int)
        A_eq = sparse.coo_matrix((eq_data, (eq_row, eq_col)), shape=(1, cols)).tocsc()
        b_eq = np.ones(1, dtype=np.float64)
    except Exception:
        return None, None

    try:
        res = optimize.linprog(
            c_obj,
            A_ub=A_ub,
            b_ub=np.zeros(rows_n, dtype=np.float64),
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=(0.0, None),
            method="highs",
            options={"time_limit": float(max_time)}
        )
    except Exception:
        return None, None

    if res is None or getattr(res, "x", None) is None:
        return None, None

    x = np.asarray(res.x, dtype=np.float64)
    if not np.all(np.isfinite(x)):
        return None, None

    t_opt = float(x[-1]) if x.size > 0 else np.inf
    b = x[:n]
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if not np.isfinite(s) or s < 1e-12:
        return None, t_opt
    b = b / s
    return b, t_opt


def _lp_schedule(a_init, deadline, max_iterations=20, max_lp_time=20.0,
                 top_rows=300, full_if_n=2048):
    a0 = _normalise(a_init)
    best = a0.copy()
    best_score = _score_norm(a0)

    for it in range(int(max_iterations)):
        rem = deadline - time.monotonic()
        if rem <= max_lp_time + 3.0:
            break
        lt = min(float(max_lp_time), rem - 3.0)
        if lt <= 2.0:
            break

        t_start_lp = time.monotonic()
        b, t_opt = _lp_best_response(a0, lt, top_rows=top_rows, full_if_n=full_if_n)
        lp_wall = time.monotonic() - t_start_lp
        if b is None:
            if it == 0:
                print("    LP move unavailable (t_opt=None, wall={:.2f}s)".format(lp_wall))
            break

        a_cur = a0.copy()
        sc_cur = best_score
        local_best = a_cur.copy()
        local_score = _score_norm(a_cur)

        sc_b = _score_norm(b)
        if np.isfinite(sc_b) and sc_b < local_score:
            local_score = sc_b
            local_best = b.copy()

        for alpha in (0.25, 0.5, 0.75):
            m = (1.0 - alpha) * a_cur + alpha * b
            s = float(m.sum())
            if s > 1e-12:
                m = m / s
                sc_m = _score_norm(m)
                if np.isfinite(sc_m) and sc_m < local_score:
                    local_score = sc_m
                    local_best = m.copy()

        a0 = _normalise(local_best)
        if local_score < best_score:
            best = a0.copy()
            best_score = float(local_score)

        if t_opt is not None and np.isfinite(t_opt):
            print("    LP it={} wall={:.2f}s t_opt={:.10f} local={:.10f}".format(
                it, lp_wall, t_opt, best_score))

        rem = deadline - time.monotonic()
        if rem <= 5.0:
            break
        if it > 0 and best_score < 1.45:
            break

    return best, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    budget = max(0.0, float(budget_s) - 25.0)
    deadline = t_start + budget
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Parent.
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            best_seq, best_score = _consider_candidate(best_seq, best_score, parent_raw, "parent", t_start)
    except Exception:
        pass

    # Initial explicit/random starts.
    if time.monotonic() < min(deadline, t_start + 40.0):
        print("=== Phase 0: starts ===")
        n_starts = (2048, 4096)
        for nst in n_starts:
            if time.monotonic() > deadline - 20.0:
                break
            for alpha, slope in (
                (0.5, 0.0), (0.5, -0.5), (0.5, -1.0), (0.5, 0.5),
                (0.45, 0.0), (0.55, 0.0), (0.4, 0.0), (0.6, 0.0)
            ):
                if time.monotonic() > deadline - 10.0:
                    break
                a = _make_mod_power_profile(nst, alpha, slope)
                best_seq, best_score = _consider_candidate(best_seq, best_score, a, "modpow", t_start)

            for omega, amp, phase in (
                (37.0, 0.25, 0.0), (91.0, 0.20, 1.3), (211.0, 0.30, 0.4),
                (53.0, 0.35, 2.2), (127.0, 0.18, 0.7)
            ):
                a = _make_wave_profile(nst, 0.5, omega, amp, phase)
                best_seq, best_score = _consider_candidate(best_seq, best_score, a, "wave", t_start)

            r = rng.normal(size=nst)
            a = _softmax(r - np.mean(r))
            best_seq, best_score = _consider_candidate(best_seq, best_score, a, "rand", t_start)

    # Low-resolution global search.
    low_deadline = min(deadline, t_start + 180.0)
    if time.monotonic() < low_deadline - 10.0:
        print("=== Phase 1: low-res LP + balance ===")
        if best_seq is None:
            a_low = _make_mod_power_profile(1024, 0.5)
        else:
            a_low = _interp_profile(best_seq, 1024)
            best_seq, best_score = _consider_candidate(best_seq, best_score, a_low, "low1", t_start)

        d1 = min(low_deadline, time.monotonic() + 70.0)
        a_low, _ = _lp_schedule(a_low, d1, max_iterations=8, max_lp_time=8.0,
                                top_rows=180, full_if_n=1024)
        best_seq, best_score = _consider_candidate(best_seq, best_score, a_low, "low_lp_1024", t_start)

        d2 = min(low_deadline, time.monotonic() + 100.0)
        a_low, _ = _balance_schedule(a_low, d2, k_start=1, k_max=256, pass_time=18.0)
        best_seq, best_score = _consider_candidate(best_seq, best_score, a_low, "low_bal_1024", t_start)

        if time.monotonic() < low_deadline - 20.0 and best_seq is not None:
            a_low2 = _interp_profile(best_seq, 2048)
            best_seq, best_score = _consider_candidate(best_seq, best_score, a_low2, "low2048", t_start)
            d3 = min(low_deadline, time.monotonic() + 60.0)
            a_low2, _ = _lp_schedule(a_low2, d3, max_iterations=5, max_lp_time=10.0,
                                     top_rows=250, full_if_n=2048)
            best_seq, best_score = _consider_candidate(best_seq, best_score, a_low2, "low_lp_2048", t_start)
            d4 = min(low_deadline, time.monotonic() + 50.0)
            a_low2, _ = _balance_schedule(a_low2, d4, k_start=2, k_max=512, pass_time=14.0)
            best_seq, best_score = _consider_candidate(best_seq, best_score, a_low2, "low_bal_2048", t_start)

    # Primary high-resolution work at 4096.
    if best_seq is None:
        best_seq = _fallback_sequence(4096)
        best_score = _score_norm(best_seq)

    if time.monotonic() < min(deadline, t_start + 700.0) - 20.0:
        print("=== Phase 2: 4096 balance + L-BFGS ===")
        a4 = _interp_profile(best_seq, 4096)
        best_seq, best_score = _consider_candidate(best_seq, best_score, a4, "4096_init", t_start)

        d = min(deadline, t_start + 700.0)
        a4, _ = _balance_schedule(a4, d, k_start=1, k_max=1024, pass_time=22.0, use_direct=True, cycles=2)
        best_seq, best_score = _consider_candidate(best_seq, best_score, a4, "4096_bal_direct", t_start)

        d = min(deadline, t_start + 700.0)
        a4, _ = _lbfgs_schedule(a4, d, k_start=8, k_max=512, pass_time=30.0, cycles=2)
        best_seq, best_score = _consider_candidate(best_seq, best_score, a4, "4096_lbfgs", t_start)

        d = min(deadline, t_start + 700.0)
        a4, _ = _balance_schedule(a4, d, k_start=5, k_max=1024, pass_time=16.0, use_direct=False, cycles=1)
        best_seq, best_score = _consider_candidate(best_seq, best_score, a4, "4096_bal_mult", t_start)

    # Longer 8192 sequence.
    if time.monotonic() < min(deadline, t_start + 1200.0) - 20.0:
        print("=== Phase 3: 8192 balance + L-BFGS ===")
        a8 = _interp_profile(best_seq, 8192)
        best_seq, best_score = _consider_candidate(best_seq, best_score, a8, "8192_init", t_start)

        d = min(deadline, t_start + 1200.0)
        a8, _ = _balance_schedule(a8, d, k_start=1, k_max=2048, pass_time=18.0, use_direct=True, cycles=2)
        best_seq, best_score = _consider_candidate(best_seq, best_score, a8, "8192_bal_direct", t_start)

        d = min(deadline, t_start + 1200.0)
        a8, _ = _lbfgs_schedule(a8, d, k_start=8, k_max=1024, pass_time=25.0, cycles=2)
        best_seq, best_score = _consider_candidate(best_seq, best_score, a8, "8192_lbfgs", t_start)

        d = min(deadline, t_start + 1200.0)
        a8, _ = _balance_schedule(a8, d, k_start=4, k_max=2048, pass_time=12.0, use_direct=False, cycles=1)
        best_seq, best_score = _consider_candidate(best_seq, best_score, a8, "8192_bal_mult", t_start)

    # LP best-response on the current best, using top constraints for large n.
    if time.monotonic() < min(deadline, t_start + 1450.0) - 20.0:
        print("=== Phase 4: LP best-response polishing ===")
        n_lp = len(best_seq)
        if n_lp > 4096:
            n_lp = 4096
        elif n_lp < 2048:
            n_lp = 2048
        a_lp = _interp_profile(best_seq, n_lp)
        best_seq, best_score = _consider_candidate(best_seq, best_score, a_lp, "lp_init", t_start)

        d = min(deadline, t_start + 1450.0)
        a_lp, _ = _lp_schedule(a_lp, d, max_iterations=10, max_lp_time=12.0,
                               top_rows=320, full_if_n=2048)
        best_seq, best_score = _consider_candidate(best_seq, best_score, a_lp, "lp_polish", t_start)

        d = min(d, time.monotonic() + 40.0)
        a_lp, _ = _balance_schedule(a_lp, d, k_start=2, k_max=512, pass_time=12.0)
        best_seq, best_score = _consider_candidate(best_seq, best_score, a_lp, "lp_after_bal", t_start)

    # Random restarts.
    if time.monotonic() < min(deadline, t_start + 1650.0) - 20.0:
        print("=== Phase 5: random restarts ===")
        d = min(deadline, t_start + 1650.0)
        for level in (0.04, 0.08, 0.12, 0.18, 0.06, 0.15):
            if time.monotonic() > d - 20.0:
                break
            ar = _perturb(best_seq, rng, level)
            dr = min(d, time.monotonic() + 22.0)
            ar, _ = _balance_pass(ar, dr, K=8, use_direct=True)
            best_seq, best_score = _consider_candidate(best_seq, best_score, ar, "restart_bal", t_start)

            if time.monotonic() < d - 15.0:
                cr = _self_convolve(ar)
                beta = _adaptive_beta(cr, 20)
                ar1, _ = _lbfgs_one(ar, beta, d, 18.0)
                best_seq, best_score = _consider_candidate(best_seq, best_score, ar1, "restart_lbfgs", t_start)

    # Final polish on the global best.
    if best_seq is not None:
        if len(best_seq) > 8192:
            best_seq = _interp_profile(best_seq, 8192)
            best_score = _score_norm(best_seq)

        print("=== Final polish ===")
        best_seq, best_score = _consider_candidate(best_seq, best_score, best_seq, "final_start", t_start)

        if time.monotonic() < deadline - 10.0:
            a_f = best_seq.copy()
            a_f, _ = _balance_schedule(a_f, deadline - 2.0, k_start=3, k_max=min(1024, a_f.size),
                                       pass_time=8.0, use_direct=True, cycles=1)
            best_seq, best_score = _consider_candidate(best_seq, best_score, a_f, "final_bal", t_start)

        if time.monotonic() < deadline - 10.0:
            a_f = best_seq.copy()
            a_f, _ = _lbfgs_schedule(a_f, deadline - 2.0, k_start=6, k_max=min(512, a_f.size),
                                     pass_time=12.0, cycles=1)
            best_seq, best_score = _consider_candidate(best_seq, best_score, a_f, "final_lbfgs", t_start)

        if time.monotonic() < deadline - 8.0:
            a_f = best_seq.copy()
            a_f, _ = _balance_schedule(a_f, deadline - 1.0, k_start=2, k_max=min(512, a_f.size),
                                       pass_time=6.0, use_direct=False, cycles=1)
            best_seq, best_score = _consider_candidate(best_seq, best_score, a_f, "final_bal2", t_start)

    # Final safety.
    if best_seq is None:
        best_seq = _fallback_sequence(4096)
        best_score = _score_norm(best_seq)

    best_seq = _normalise(best_seq)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if not np.isfinite(s) or s < 0.01:
        best_seq = _fallback_sequence(4096)
        best_seq = _normalise(best_seq)
    else:
        best_seq = best_seq / s

    final_score = _score_norm(best_seq)
    print("FINAL internal score={:.10f} n={}".format(final_score, len(best_seq)))

    # Optional local evaluator check.
    try:
        if len(best_seq) <= 8192 and time.monotonic() < deadline - 1.0:
            ev = float(evaluate_sequence([float(x) for x in best_seq]))
            print("FINAL evaluate_sequence={:.10f}".format(ev))
    except Exception as e:
        print("evaluate_sequence check skipped:", e)

    return [float(x) for x in best_seq]
