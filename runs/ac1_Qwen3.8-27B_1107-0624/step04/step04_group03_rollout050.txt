"""
Minimax-inspired optimizer for auto-convolution peak minimization.
Combines multiplicative weights (MW) for the minimax structure,
log-sum-exp smooth optimization, and top-k subgradient polishing.
Key insight: score = 2n*max(conv(a,a))/sum(a)^2 is scale-invariant,
so we work on the simplex sum(a)=1 and minimize max(conv(a,a)).
"""
import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    """Compute 2n*max(conv(a,a)) assuming sum(a)=1 (scale invariant)."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _full_score(a):
    """Compute the actual score 2*n*max(conv)/sum^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _smooth_gradient(a, w):
    """Gradient of sum(w_k * b_k) w.r.t. a, using FFT.
    b_k = sum_i a_i * a_{k-i}, so db_k/da_j = 2*a_{k-j} (if valid) or a_{k-j} at center.
    More precisely: grad[j] = sum_k w_k * db_k/da_j
    db_k/da_j: term a_j appears in b_k when k-j is valid (gives a_{k-j}),
    and a_j appears as the second factor when k-j valid (same term).
    So db_k/da_j = 2*a_{k-j} if 0<=k-j<n and k-j != j (i.e. 2j != k),
    and = a_j if 2j = k.
    
    Simpler: grad = 2 * sum_k w_k * (convolution of delta_j with... )
    
    Actually, the cleanest: sum_k w_k * b_k = sum_k w_k * sum_i a_i * a_{k-i}
    = sum_i a_i * sum_k w_k * a_{k-i}
    = sum_i a_i * (correlation of a with w at offset i)
    
    So the "Q@a" vector has component i = sum_k w_k * a[k-i] (valid k-i).
    And grad[i] = 2 * (Q@a)[i] (with the factor of 2 from the quadratic form,
    except at the diagonal where it's 1, but for our purposes the factor of 2 is fine
    since it just scales the step).
    
    (Q@a)[i] = sum_j a[j] * w[i+j] for j=0..n-1, i=0..n-1, w defined on [0,2n-2].
    This equals (a_rev * w)[i + n - 1] where a_rev[j] = a[n-1-j] and * is convolution.
    """
    n = len(a)
    a_rev = a[::-1]
    # a_rev * w has length n + (2n-1) - 1 = 3n - 2
    conv = _fft_convolve(a_rev, w)
    # We want conv[i + n - 1] for i = 0..n-1
    h = conv[n - 1:2 * n - 1]
    return 2.0 * h

def _multiplicative_weights(a_init, deadline, n_outer=2000, n_substeps=3, eta_init=0.001):
    """
    Multiplicative weights minimax approach.
    Alternates between:
    1. Minimizing weighted sum of convolution terms (projected gradient on simplex)
    2. Updating weights via exponential weights
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    num_lags = 2 * n - 1
    w = np.ones(num_lags) / num_lags

    best_a = a.copy()
    best_score = _score_norm(a)

    eta = eta_init
    lr = 0.01

    for outer in range(n_outer):
        if time.monotonic() >= deadline:
            break

        # Step 1: Minimize sum_k w_k * b_k(a) over simplex
        # Gradient: 2*(Q@a) where (Q@a)[i] = sum_j a[j]*w[i+j]
        for sub in range(n_substeps):
            h = _smooth_gradient(a, w)  # this gives 2*(Q@a)
            # Projected gradient step on simplex
            a_new = a - lr * h
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a
            else:
                a_new = a_new / s_new

            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            a = a_new

        # Step 2: Update weights
        c = _fft_convolve(a, a)
        # Exponential weights update
        c_shifted = c - c.max()
        w_new = w * np.exp(c_shifted / eta)
        s_w = w_new.sum()
        if s_w < 1e-15 or not np.isfinite(s_w):
            w_new = np.ones_like(c) / num_lags
        else:
            w_new = w_new / s_w
        w = w_new

        # Adaptive eta and lr
        if outer % 100 == 0 and outer > 0:
            eta *= 0.8
            lr *= 0.95

    if time.monotonic() < deadline:
        sc_final = _score_norm(a)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a.copy()

    return best_a, best_score


def _topk_subgradient(a_init, deadline, k=5, lr_init=0.002, momentum=0.9):
    """Top-k subgradient method: average subgradient over top-k lags."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    mom_vec = np.zeros(n)

    lr = lr_init
    it = 0
    stall_count = 0

    while time.monotonic() < deadline:
        it += 1
        a = _softmax_fn(z)
        c = _fft_convolve(a, a)

        # Get top-k lags
        if k < len(c):
            top_k = np.argsort(c)[-k:]
        else:
            top_k = np.arange(len(c))

        w_k = np.zeros_like(c)
        w_k[top_k] = 1.0 / len(top_k)

        # Compute subgradient
        h = _smooth_gradient(a, w_k)
        ga = 4.0 * n * h
        dot = float(np.dot(ga, a))
        grad_z = ga * a - a * dot

        if not np.all(np.isfinite(grad_z)):
            continue

        # Momentum step
        mom_vec = momentum * mom_vec - lr * grad_z
        z = z + mom_vec
        z = np.clip(z, -50, 50)
        z = z - z.mean()

        if it % 50 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm(a_curr)
            if np.isfinite(sc) and sc < best_score:

                best_a = a_curr.copy()
                stall_count = 0
                # Slightly increase momentum
                momentum = min(0.999, momentum + 0.001)
            else:
                stall_count += 1
                if stall_count > 10:
                    lr *= 0.7
                    momentum = 0.85
                    stall_count = 0

    a_final = _softmax_fn(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _nesterov_optimize(a_init, deadline, beta_fracs, lr_init=0.01, mom_init=0.9):
    """Nesterov-accelerated gradient descent in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c = _fft_convolve(a_init, a_init)
    max_c = float(c.max())
    lr = lr_init
    mom = mom_init
    t = 0

    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-15, beta_frac * max_c)

        while time.monotonic() < deadline:
            t += 1
            a_y = _softmax_fn(y)
            c_y = _fft_convolve(a_y, a_y)
            w, cm = _compute_weights(c_y, beta)
            h = _smooth_gradient(a_y, w)
            ga = 4.0 * n * h
            dot = float(np.dot(ga, a_y))
            grad_z = ga * a_y - a_y * dot

            if not np.all(np.isfinite(grad_z)):
                break

            z_new = z - lr * grad_z
            z_new = np.clip(z_new, -50, 50)
            z_new = z_new - z_new.mean()

            y_new = z_new + mom * (z_new - z)
            y_new = np.clip(y_new, -50, 50)
            y_new = y_new - y_new.mean()

            z = z_new
            y = y_new

            if t % 50 == 0:
                a_curr = _softmax_fn(z)
                sc = _score_norm(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()

            if t % 200 == 0:
                mom = min(0.999, mom + 0.0003)

        a_curr = _softmax_fn(z)
        c = _fft_convolve(a_curr, a_curr)
        max_c = float(c.max())
        lr = max(0.001, lr * 0.75)

    a_final = _softmax_fn(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_a.score if hasattr(best_a, 'score') else best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-method optimizer:
    1. LSE smooth optimization (Adam)
    2. Multiplicative weights minimax
    3. Top-k subgradient polishing
    4. Multi-restart with diversity
    Time budget: 1800s total.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            p = np.asarray(p, dtype=np.float64)
            p = np.maximum(p, 0)
            s = p.sum()
            if s < 1e-15:
                return
            p = p / s
            best_seq = p.copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        print(f"Parent: n={n}, score={_full_score(parent):.10f}")
    else:
        n = 4096
        parent = _make_linear(n, 1.0)
        print(f"No parent, using linear profile n={n}")

    # Normalize parent to sum=1 for our internal use
    a0 = parent / parent.sum()
    sc0 = _score_norm(a0)
    update_best(a0, sc0, "parent")

    c0 = _fft_convolve(a0, a0)
    max_c0 = float(c0.max())
    print(f"Initial: n={n}, max_c={max_c0:.8f}, score={sc0:.10f}")

    # --- Phase 1: LSE Adam optimization (400s) ---
    phase1_end = min(deadline, t_start + 400.0)
    beta_fracs1 = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    print("Phase 1: LSE Adam optimization...")
    p1, s1 = _run_lse_optimize(a0, phase1_end, beta_fracs1, lr_init=0.03)
    update_best(p1, s1, "lse_adam")

    # --- Phase 2: L-BFGS with continuation (200s) ---
    phase2_end = min(deadline, t_start + 600.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        print("Phase 2: L-BFGS continuation...")
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        c_best = _fft_convolve(best_seq, best_seq)
        max_cb = float(c_best.max())

        betas_lbfgs = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
        per_beta_time = max(5.0, (phase2_end - time.monotonic()) / len(betas_lbfgs))

        for bf in betas_lbfgs:
            if time.monotonic() >= phase2_end:
                break
            beta = max(1e-15, bf * max_cb)
            dl = min(phase2_end, time.monotonic() + per_beta_time)
            z = _lbfgs_on_z(z, beta, n, maxiter=100)
            a_new = _softmax_fn(z)
            sc_new = _score_norm(a_new)
            if np.isfinite(sc_new) and sc_new < best_score:
                update_best(a_new, sc_new, f"lbfgs_b{bf}")
                c_best = _fft_convolve(a_new, a_new)
                max_cb = float(c_best.max())

    # --- Phase 3: Multiplicative weights (300s) ---
    phase3_end = min(deadline, t_start + 900.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        print("Phase 3: Multiplicative weights minimax...")
        p3, s3 = _multiplicative_weights(best_seq, phase3_end,
                                         n_outer=5000, n_substeps=3, eta_init=0.002)
        update_best(p3, s3, "mw")

    # --- Phase 4: Top-k subgradient (250s) ---
    phase4_end = min(deadline, t_start + 1150.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        print("Phase 4: Top-k subgradient...")
        p4, s4 = _topk_subgradient(best_seq, phase4_end, k=3, lr_init=0.003, momentum=0.9)
        update_best(p4, s4, "topk_k3")

        if time.monotonic() < phase4_end - 60:
            p4b, s4b = _topk_subgradient(best_seq, phase4_end, k=1, lr_init=0.002, momentum=0.92)
            update_best(p4b, s4b, "topk_k1")

    # --- Phase 5: Nesterov with tiny beta (200s) ---
    phase5_end = min(deadline, t_start + 1350.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        print("Phase 5: Nesterov fine polish...")
        beta_fracs5 = [0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
        p5, s5 = _nesterov_optimize(best_seq, phase5_end, beta_fracs5, lr_init=0.005, mom_init=0.95)
        update_best(p5, s5, "nesterov_fine")

    # --- Phase 6: Multi-restart (250s) ---
    phase6_end = min(deadline, t_start + 1600.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        print("Phase 6: Multi-restart...")
        n_restarts = 4
        per_restart = (phase6_end - time.monotonic()) / max(1, n_restarts)

        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 30:
                break
            noise_levels = [0.02, 0.05, 0.1, 0.2]
            noise = noise_levels[i % len(noise_levels)]
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=len(best_seq))
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            per_dl = min(phase6_end - 20, time.monotonic() + per_restart)
            # Quick LSE then topk
            p_r, s_r = _run_lse_optimize(p_start, min(per_dl - 30, time.monotonic() + per_restart * 0.5),
                                         [0.01, 0.005, 0.001, 0.0001], lr_init=0.02)
            if time.monotonic() < per_dl - 10:
                p_r2, s_r2 = _topk_subgradient(p_r, per_dl, k=3, lr_init=0.002, momentum=0.9)
                update_best(p_r2, s_r2, f"restart{i}_topk")
            else:
                update_best(p_r, s_r, f"restart{i}")

    # --- Phase 7: Final polish with remaining time ---
    phase7_end = deadline
    if best_seq is not None and time.monotonic() < phase7_end:
        remaining = phase7_end - time.monotonic()
        if remaining > 30:
            print("Phase 7: Final polish...")
            # Very fine LSE
            per1 = min(remaining * 0.4, 60.0)
            p7, s7 = _run_lse_optimize(best_seq, time.monotonic() + per1,
                                       [0.00005, 0.00002, 0.00001], lr_init=0.003)
            update_best(p7, s7, "final_lse")

            if time.monotonic() < phase7_end - 20:
                per2 = min(phase7_end - time.monotonic() - 10, 60.0)
                p7b, s7b = _topk_subgradient(best_seq, time.monotonic() + per2,
                                             k=1, lr_init=0.001, momentum=0.95)
                update_best(p7b, s7b, "final_topk")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_linear(2048, 1.0)

        sc_final = _full_score(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned inf/NaN, using internal score: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_linear(4096, 1.0)
    print(f"FALLBACK: n=4096, score={_full_score(fallback):.10f}")
    return [float(x) for x in fallback]
