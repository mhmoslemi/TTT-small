"""
Optimized step function coefficient search using multi-phase gradient methods
with log-sum-exp smoothing, direct subgradient, coordinate polish, and random
perturbation restarts. Targets score < 1.5027 starting from parent (~1.506).

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    """Score assuming sum(a)=1 (normalized)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _gradient_smooth(z, beta):
    """Gradient of smooth approximation in softmax parameterization."""
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _objective_smooth(z, beta):
    """Smooth approximation of the score (sum=1 parameterization)."""
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_wrap_obj(z, beta):
    return _objective_smooth(z, beta)


def _lbfgs_wrap_grad(z, beta):
    return _gradient_smooth(z, beta)


def _run_lbfgs(a_init, deadline, max_wall_time=None):
    """L-BFGS with beta continuation schedule."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if max_wall_time is None:
        max_wall_time = min(200.0, max(1.0, deadline - time.monotonic()))

    # Beta schedule: start moderate, end very small
    betas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    betas = [b * max_c0 for b in betas]

    per_level = max_wall_time / len(betas)

    for beta in betas:
        if time.monotonic() >= deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level)
        remaining = level_dl - time.monotonic()
        if remaining < 2:
            continue
        try:
            max_iter = max(50, int(300 * remaining / per_level))
            result = minimize(
                _lbfgs_wrap_obj, z, args=(beta,),
                method='L-BFGS-B', jac=_lbfgs_wrap_grad,
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 40, 'maxfun': max_iter * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x.copy()
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new
        except Exception:
            pass

    return best_a, best_score


def _adam_optimize(a_init, deadline, lr=0.05, beta_init_frac=0.01, beta_decay=0.85):

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, beta_init_frac * max_c)

    check_interval = max(50, n // 10)

    for it in range(500000):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            beta = max(1e-12, beta * beta_decay)
            c_new = _full_convolve(a_curr, a_curr)
            max_c = float(c_new.max())
            beta = max(1e-12, min(beta, 0.01 * max_c))

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

def _subgradient_optimize(a_init, deadline, band_frac=0.01, lr_init=0.01, momentum_init=0.9):
    """Direct subgradient method using active set of near-max lags."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    lr = lr_init
    momentum = np.zeros(n)
    mom_coeff = momentum_init
    t = 0

    check_interval = max(20, n // 20)

    # Running average of subgradients for variance reduction
    grad_avg = np.zeros(n)
    grad_avg_count = 0
    avg_decay = 0.95

    for it in range(500000):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Active set: lags within band_frac of max
        threshold = cm * (1.0 - band_frac)
        active = c >= threshold
        if not np.any(active):
            active[int(np.argmax(c))] = True

        n_active = int(np.sum(active))
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        # Variance-reduced gradient
        grad_avg = avg_decay * grad_avg + (1 - avg_decay) * grad
        grad_avg_count += 1

        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            lr_eff = lr_init / (1.0 + t * 0.001)

        momentum = mom_coeff * momentum - lr_eff * grad_avg
        z = z + momentum
        z = z - z.mean()
        z = np.clip(z, -50, 50)

        # Increase momentum over time
        if t % 500 == 0:
            mom_coeff = min(0.99, mom_coeff + 0.002)

        if t % check_interval == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
                # Reduce learning rate on improvement
                lr_init = lr_init * 0.95

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _coordinate_polish(a_init, deadline, n_sweeps_max=100, eps=1e-3):
    """Try small multiplicative perturbations to individual coordinates."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    rng = np.random.default_rng(12345)

    # Precompute current convolution
    c = _full_convolve(best_a, best_a)
    cm = float(c.max())

    sweeps = 0
    while time.monotonic() < deadline and sweeps < n_sweeps_max:
        sweeps += 1
        improved = False
        # Process coordinates in random order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            if best_a[i] < 1e-10:
                continue
            # Try increasing
            a_try = best_a.copy()
            a_try[i] *= (1.0 + eps)
            sc_try = _score_norm_fn(a_try)
            if sc_try < best_score - 1e-12:
                best_a = a_try
                best_score = sc_try
                improved = True
            else:
                # Try decreasing
                a_try[i] = best_a[i]
                a_try[i] *= (1.0 - eps)
                a_try[i] = max(a_try[i], 0.0)
                sc_try = _score_norm_fn(a_try)
                if sc_try < best_score - 1e-12:
                    best_a = a_try
                    best_score = sc_try
                    improved = True

        if not improved:
            # Decrease eps and try again
            eps *= 0.7
            if eps < 1e-6:
                break

    return best_a, best_score

def _make_profile(n, kind=0):
    """Generate various initial profiles."""
    t = np.linspace(0.0, 1.0, n)
    if kind == 0:
        # Linear
        a = 1.0 + 2.0 * (t - 0.5)
    elif kind == 1:
        # Quadratic
        u = 2.0 * t - 1.0
        a = 1.0 + 0.8 * u + 0.3 * u ** 2
    elif kind == 2:
        # Exponential decay
        a = np.exp(-1.5 * t)
    elif kind == 3:
        # Exponential growth then decay (asymmetric)
        a = np.exp(-2.0 * t) + 0.5 * np.exp(2.0 * (t - 1.0))
    elif kind == 4:
        # Strong exponential
        a = np.exp(-3.0 * t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

    if best_seq is not None and time.monotonic() < phase7_end:
        n_restarts = 3
        per_restart = min(45.0, (phase7_end - time.monotonic()) / max(1, n_restarts))
        n_best = len(best_seq)

        for i in range(n_restarts):
            if time.monotonic() > phase7_end - 10:
                break
            noise_level = 0.02 + 0.015 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_level * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            # Quick optimization on perturbed start
            per_dl = min(phase7_end - 5, time.monotonic() + per_restart)
            p_best, sc_best = _adam_optimize(p_start, per_dl,
                                             lr=0.03, beta_init_frac=0.003, beta_decay=0.85)
            update_best(p_best, sc_best, f"restart_{i}")

    print(f"Phase 7 done: best={best_score:.10f}")

    # --- Phase 8: Final ultra-fine polish (remaining time) ---
    final_end = deadline
    if best_seq is not None and time.monotonic() < final_end:
        remaining = final_end - time.monotonic()
        if remaining > 20:
            # Final Adam with extremely small beta
            per_time = min(60.0, remaining - 10)
            if per_time > 10:
                p_best, sc_best = _adam_optimize(best_seq, min(final_end - 5, time.monotonic() + per_time),
                                                 lr=0.005, beta_init_frac=0.0001, beta_decay=0.75)
                update_best(p_best, sc_best, "final_adam")

            # Final subgradient
            if time.monotonic() < final_end - 15:
                per_time2 = min(40.0, final_end - time.monotonic() - 5)
                if per_time2 > 5:
                    p_best, sc_best = _subgradient_optimize(best_seq, min(final_end - 2, time.monotonic() + per_time2),
                                                            band_frac=0.001, lr_init=0.003, momentum_init=0.95)
                    update_best(p_best, sc_best, "final_subgrad")

    print(f"Phase 8 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profile(4096, 2)

        sc_final = _score_fn(final_seq)
        print(f"FINAL internal: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned non-finite, using internal: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profile(4096, 2)
    print(f"FALLBACK: n=4096, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
