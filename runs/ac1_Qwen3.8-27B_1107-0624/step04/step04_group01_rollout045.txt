def _nesterov_optimize(a_init, deadline, beta_schedule, max_iter=100000, lr=0.01, topk_frac=0.05):
    """Nesterov-accelerated gradient descent in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    v = np.zeros(n)  # velocity for Nesterov
    mu = 0.9  # momentum
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            
            # Nesterov: look-ahead position
            z_lookahead = z - mu * v
            a = _softmax(z_lookahead)
            
            grad, a_check, c, cm = _compute_gradient_softmax(z_lookahead, beta, topk_frac)
            
            if not np.all(np.isfinite(grad)):
                break
            
            v = mu * v - lr * grad
            z = z + v
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if it_in_phase % 100 == 0:
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()
            
            if it_in_phase % 5000 == 0:
                lr = max(0.001, lr * 0.95)
    
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score
