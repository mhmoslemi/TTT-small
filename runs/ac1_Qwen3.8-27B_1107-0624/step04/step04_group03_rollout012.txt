def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

    s = float(a.sum())

def _compute_weights(c, beta):

def _make_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0.0, 1.0, n)

    profiles.append(np.maximum(a, 1e-12) / np.maximum(a, 1e-12).sum())
    # Quadratic (U-shaped)
    a = 1.0 + 0.8 * u ** 2
    profiles.append(a / a.sum())
    # Inverted quadratic (hill)
    a = 1.0 - 0.5 * u ** 2
    profiles.append(np.maximum(a, 1e-12) / np.maximum(a, 1e-12).sum())
    # Cosine window
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    profiles.append(np.maximum(a, 1e-12) / np.maximum(a, 1e-12).sum())

    # Exponential decay from center

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _structured_noise(n, rng, freq_scale=1.0):
    """Generate low-frequency structured noise for restarts."""
    # Create noise in Fourier domain with limited bandwidth
    N = _next_pow2(n)
    noise_spec = np.zeros(N // 2 + 1, dtype=np.float64)

    noise_spec[:max_band] = rng.normal(size=max_band)
    noise = np.fft.irfft(noise_spec, N)[:n]
    return noise

            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n}, score={sc_parent:.10f}")
    else:
        n = 4096
        parent = _make_profiles(n)[0]
        sc_parent = _score_norm_fn(parent)
        print(f"No parent, using default profile n={n}, score={sc_parent:.10f}")

    update_best(parent, sc_parent, "parent")

    # --- Phase 0: Quick profile evaluation (15s) ---
    phase0_end = min(deadline, t_start + 15.0)
    profiles = _make_profiles(n)
    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        update_best(p, sc, f"profile_{idx}")

    print(f"Phase 0 done: best={best_score:.10f}")

    # --- Phase 1: L-BFGS on best candidate (150s) ---
    phase1_end = min(deadline, t_start + 165.0)
    if best_seq is not None:
        per_time = min(150.0, phase1_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _run_lbfgs(best_seq, phase1_end, max_wall_time=per_time)
            update_best(p_best, sc_best, "lbfgs_phase1")
    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Adaptive Nesterov (500s) ---
    phase2_end = min(deadline, t_start + 665.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        p_best, sc_best = _nesterov_adaptive(
            best_seq, phase2_end,
            beta_init_frac=0.01, beta_min_frac=1e-6,
            lr_init=0.02, mom_init=0.9, beta_decay=0.6,
            patience=300, eval_interval=30
        )
        update_best(p_best, sc_best, "nesterov_phase2")
    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Adam fine polishing (300s) ---
    phase3_end = min(deadline, t_start + 965.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        per_time = min(280.0, phase3_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _adam_adaptive(
                best_seq, min(phase3_end, time.monotonic() + per_time),
                lr=0.02, beta_init_frac=0.001
            )
            update_best(p_best, sc_best, "adam_phase3")
    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Active set polish (200s) ---
    phase4_end = min(deadline, t_start + 1165.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        per_time = min(180.0, phase4_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _active_set_polish(
                best_seq, min(phase4_end, time.monotonic() + per_time)
            )
            update_best(p_best, sc_best, "active_set_polish")
    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Structured restarts (350s) ---
    phase5_end = min(deadline, t_start + 1515.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        n_best = len(best_seq)
        n_restarts = 6
        per_restart = min(55.0, (phase5_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            # Structured noise: low frequency for large restarts, high for small
            freq_scale = 0.1 + 0.3 * (i / max(1, n_restarts - 1))
            noise_mag = 0.03 + 0.02 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise = _structured_noise(n_best, rng, freq_scale)
            z = z + noise_mag * noise
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            per_dl = min(phase5_end - 5, time.monotonic() + per_restart)
            if i < 3:
                # First 3 restarts: Nesterov
                p_best, sc_best = _nesterov_adaptive(
                    p_start, per_dl,
                    beta_init_frac=0.005, beta_min_frac=1e-7,
                    lr_init=0.015, mom_init=0.85, beta_decay=0.6,
                    patience=150, eval_interval=20
                )
            else:
                # Last 3 restarts: Adam (faster convergence from good start)
                p_best, sc_best = _adam_adaptive(
                    p_start, per_dl,
                    lr=0.015, beta_init_frac=0.001
                )
            update_best(p_best, sc_best, f"restart_{i}")
    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Try different length (150s) ---
    phase6_end = min(deadline, t_start + 1665.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        n_curr = len(best_seq)
        # Try interpolating to 2x length
        if n_curr <= 4096:
            target_n = min(8192, n_curr * 2)
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            if sc_interp < best_score:
                update_best(p_interp, sc_interp, f"interp_n{target_n}")
                # Optimize at new length
                if time.monotonic() < phase6_end - 30:
                    per_time = min(80.0, phase6_end - time.monotonic() - 20)
                    if per_time > 20:
                        p_best, sc_best = _nesterov_adaptive(
                            p_interp, min(phase6_end - 10, time.monotonic() + per_time),
                            beta_init_frac=0.005, beta_min_frac=1e-7,
                            lr_init=0.01, mom_init=0.85, beta_decay=0.6,
                            patience=100, eval_interval=20
                        )
                        update_best(p_best, sc_best, f"opt_n{target_n}")
    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final fine polish (remaining) ---
    phase7_end = deadline
    if best_seq is not None and time.monotonic() < phase7_end:
        remaining = phase7_end - time.monotonic()
        if remaining > 20:
            # Final Nesterov with tiny beta
            per_time = min(40.0, remaining - 10)
            if per_time > 5:
                p_best, sc_best = _nesterov_adaptive(
                    best_seq, min(phase7_end - 10, time.monotonic() + per_time),
                    beta_init_frac=1e-5, beta_min_frac=1e-8,
                    lr_init=0.005, mom_init=0.95, beta_decay=0.5,
                    patience=100, eval_interval=20
                )
                update_best(p_best, sc_best, "final_nesterov")

        if time.monotonic() < phase7_end - 5:
            per_time = min(20.0, phase7_end - time.monotonic() - 2)
            if per_time > 3:
                p_best, sc_best = _active_set_polish(
                    best_seq, min(phase7_end - 1, time.monotonic() + per_time)
                )
                update_best(p_best, sc_best, "final_active")
    print(f"Phase 7 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profiles(4096)[0]
        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned inf/NaN, using internal: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")
        return [float(x) for x in final_seq]

    fallback = _make_profiles(4096)[0]
    print(f"FALLBACK: n=4096, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
