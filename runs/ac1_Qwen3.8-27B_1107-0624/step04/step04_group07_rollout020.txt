"""
Optimized autocorrelation minimization using multi-scale Adam with FFT-based
smooth-max gradients, diverse shape initialization, and escalation strategy.
Targets score < 1.5027 by exploiting the ~4/3 asymptotic of monotone sequences
at large n, combined with fine-tuning via gradient descent in log-space.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _cross_correlate(w, a):
    """Compute sum_k w[k] * (a*a)'s gradient contribution at each index."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    # grad_i = sum_k w[k] * (a[k-i] + a[i-k]) = 2 * (w * a)[i] (circular, adjusted)
    # Actually: d(a*a)/da_i = (a * reverse(a))[i] + (reverse(a) * a)[i] = 2 * (w_conv_a)[i]
    # More precisely: d(a*a)/da_i[k] = a[k-i] + a[i-k] (with bounds)
    # Sum over k of w[k]*(a[k-i]+a[i-k]) = (w*a_conv)_i + (w*a_conv_rev)_i
    # This equals 2 * (w * a)[i] where * is the appropriate cross-correlation
    h = np.fft.irfft(fw * fa, N)
    # h[j] = sum_k w[k] * a[k-j] (cross-correlation)
    # We want sum_k w[k] * (a[k-i] + a[i-k]) = h[i] + h_rev[i]
    # h[i] = sum_k w[k]*a[k-i], h_rev[i] = sum_k w[k]*a[i-k]
    # By symmetry of convolution: sum_k w[k]*a[i-k] = (w*a_conv)[i] where a_conv = a[::1]
    # Actually h[i] = sum_k w[k] a[k-i] which is the cross-correlation
    # And sum_k w[k] a[i-k] is the convolution (w * a)[i]
    # For real signals: cross-correlate = flip then convolve
    # h[i] = (w * flip(a))[i] = sum_k w[k] flip(a)[i-k] = sum_k w[k] a[k-i] ✓
    # conv(i) = (w * a)[i] = sum_k w[k] a[i-k]
    fa2 = np.fft.rfft(a, N)
    conv = np.fft.irfft(fw * fa2, N)
    # The gradient is h[i] + conv[i] for i = 0,...,n-1
    # But we need to be careful about the full convolution length
    # h has length N, we take h[0:n] for the "a[k-i]" part where i=0..n-1
    # conv has length N, we take conv[0:n] for the "a[i-k]" part
    grad = h[:n] + conv[:n]
    return grad


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _compute_score_and_grad(a, beta):
    """Compute score and gradient of smooth-max in log-space."""
    n = len(a)
    c = _full_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    smooth_max = mval + beta * float(np.log(s))
    score = float(2.0 * n * smooth_max)
    # Gradient in a-space: g_a[i] = 2n * w_conv[i] / (sum_a)^2
    # In log-space: g_z[i] = a[i] * g_a[i]
    w_pad = np.zeros(2 * n - 1)
    w_pad[:] = w
    grad_a = _cross_correlate(w_pad, a)
    sum_a = float(a.sum())
    grad_z = a * (2.0 * n * grad_a / (sum_a * sum_a))
    # Center the gradient (remove mean for invariance to scaling)
    grad_z -= grad_z.mean()
    return score, grad_z, c, mval


def _adam_optimize(a_init, deadline, rng, lr=0.04, beta_init=None,
                   beta_decay=0.97, no_imp_limit=300, lr_decay=0.95,
                   min_lr=0.002, restart_noise=0.1):
    """Adam optimization in log-space with smooth-max."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy()
    a = a / a.sum()

    z = np.log(a)
    z -= z.mean()

    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = _full_convolve(a)
    maxc = float(c.max())
    beta = beta_init if beta_init is not None else 0.05 * maxc

    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    cur_lr = lr

    while time.monotonic() < deadline:
        score, grad_z, c, mval = _compute_score_and_grad(a, beta)

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            a = np.exp(z - z.max())
            a = a / a.sum()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - cur_lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z -= z.mean()

        a = np.exp(z - z.max())
        a = a / a.sum()
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        if np.isfinite(score):
            if score < best_score - 1e-10:
                best_score = score
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    cur_lr = max(min_lr, cur_lr * lr_decay)
        else:
            z = best_z.copy()
            a = np.exp(z - z.max())
            a = a / a.sum()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        if t_adam % 30 == 0:
            beta = max(1e-4, beta * beta_decay)
            # Adaptive beta: shrink if top values are close
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                top5 = c[sidx[:min(5, c.size)]]
                spread = float(top5.std())
                if spread < beta * 0.3:
                    beta = max(1e-4, beta * 0.7)

        if no_imp > no_imp_limit:
            ns = restart_noise * (0.5 + 0.5 * (1 - best_score / 2.0))
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60, 60)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c_new = _full_convolve(a)
            beta = 0.05 * c_new.max()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            cur_lr = lr
            no_imp = 0

    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score


def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new


def _make_profile(n, ptype, p1=1.0, p2=0.0):
    if ptype == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * t
    elif ptype == 'linear_centered':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (t - 0.5)
    elif ptype == 'power':
        t = np.linspace(0, 1, n)
        a = t ** p1 + 1e-6
    elif ptype == 'exp':
        t = np.linspace(0, 1, n)
        a = np.exp(p1 * t)
    elif ptype == 'quad':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * t * t
    elif ptype == 'cubic':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * t + p2 * t * t
    elif ptype == 'sqrt':
        t = np.linspace(0, 1, n)
        a = np.sqrt(t) + 1e-6
    elif ptype == 'log':
        t = np.linspace(0.01, 1, n)
        a = np.log(1 + p1 * t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _coordinate_refine(a, deadline, max_sweeps=20):
    """Simple coordinate descent: try scaling each element."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    factors = [0.5, 0.7, 1.3, 2.0, 0.75, 1.25]

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            old = a[j]
            for f in factors:
                a[j] = max(old * f, 1e-15)
                sc = _score_norm(a)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = old
                    break
            a[j] = old
        if not improved:
            break
    return best_a, best_score


def _block_optimize(a, deadline):
    """Optimize blocks of the sequence simultaneously."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    # Try different block sizes
    for block_size in [4, 8, 16, 32]:
        if time.monotonic() > deadline:
            break
        n_blocks = n // block_size
        for _ in range(3):
            if time.monotonic() > deadline:
                break
            improved = False
            for b in range(n_blocks):
                if time.monotonic() > deadline:
                    break
                start = b * block_size
                end = start + block_size
                old_block = a[start:end].copy()
                # Try various modifications to this block
                for mod in range(4):
                    a[start:end] = old_block * (1.0 + 0.2 * (1 + mod) * (-1) ** (mod % 2))
                    a[start:end] = np.maximum(a[start:end], 1e-15)
                    a = a / a.sum()
                    sc = _score_norm(a)
                    if sc < best_score - 1e-10:
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                a[start:end] = old_block
                a = a / a.sum()
            if not improved:
                break
    return best_a, best_score

                blend_profile = (1.0 + 2.0 * t) / (1.0 + 2.0 * t).sum()
                alpha = 0.2 + 0.1 * (i % 3)
                a_s = (1 - alpha) * best_seq + alpha * blend_profile
                a_s = np.maximum(a_s, 1e-12)
                a_s = a_s / a_s.sum()

            a_r, sc_r = _adam_optimize(
                a_s, time.monotonic() + per, rng,
                lr=0.03, beta_init=0.02,
                beta_decay=0.97, no_imp_limit=150,
                restart_noise=0.1
            )
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  *** NEW BEST (restart{i}): {best_score:.10f}")
            else:
                print(f"  Restart {i}: {sc_r:.10f} (best={best_score:.10f})")

    # === Phase 6: Final polish with Adam at very low lr ===
    print("\nPhase 6: Final polish...")
    if best_seq is not None and time.monotonic() < deadline - 10:
        per = min(60.0, deadline - time.monotonic() - 5)
        if per > 10:
            a_fp, sc_fp = _adam_optimize(
                best_seq, time.monotonic() + per, rng,
                lr=0.01, beta_init=0.01,
                beta_decay=0.98, no_imp_limit=500,
                restart_noise=0.03, min_lr=0.001
            )
            if sc_fp < best_score:
                best_score = sc_fp
                best_seq = a_fp.copy()
                print(f"  *** FINAL BEST: {best_score:.10f}")

    # Final validation and return
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile(1024, 'linear', 2.0, 0)

        exact_sc = _score_norm(final)
        print(f"\nFINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile(1024, 'linear', 2.0, 0)
    print("FALLBACK n=1024 linear")
    return [float(x) for x in fb]
