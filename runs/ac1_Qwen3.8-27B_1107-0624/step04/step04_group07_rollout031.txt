"""
Optimized step function search combining LP fixed-point with golden-section
line search, Adam smooth-max with adaptive temperature, variance-regularized
objectives, multi-scale escalation, and simulated annealing to minimize
2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _cross_correlate(w, a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _conv_stats(a):
    c = _full_convolve(a)
    mx = float(c.max())
    mn = float(c.min()) if c.size > 0 else 0.0
    mean_c = float(c.mean()) if c.size > 0 else 0.0
    var_c = float(c.var()) if c.size > 1 else 0.0
    return mx, mn, mean_c, var_c, c


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w


def _build_conv_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_primal(a, M, n):
    A = _build_conv_matrix(n, a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _golden_section_line_search(a, b, deadline, max_evals=15):
    """Find best t in [0,1] for (1-t)*a + t*b minimizing score."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)

    def eval_t(t):
        a_t = (1.0 - t) * a + t * b
        a_t = np.maximum(a_t, 1e-15)
        a_t = a_t / a_t.sum()
        return _score_norm(a_t)

    f_c = eval_t(c_pt)
    f_d = eval_t(d_pt)
    evals = 2

    for _ in range(max_evals - 2):
        if time.monotonic() > deadline:
            break
        if f_c < f_d:
            hi = d_pt
            d_pt = c_pt
            f_d = f_c
            c_pt = hi - gr * (hi - lo)
            f_c = eval_t(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            f_c = f_d
            d_pt = lo + gr * (hi - lo)
            f_d = eval_t(d_pt)
        evals += 1

    best_t = (lo + hi) / 2.0
    best_sc = min(f_c, f_d, eval_t(0.5))
    # Also check endpoints
    sc_0 = eval_t(0.0)
    sc_1 = eval_t(1.0)
    if sc_0 < best_sc:
        best_t, best_sc = 0.0, sc_0
    if sc_1 < best_sc:
        best_t, best_sc = 1.0, sc_1
    return best_t, best_sc


def _lp_fixed_point(a, deadline, max_iters=60, use_gs=True):
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a)
        M = float(c.max())

        b = _lp_primal(a, M, n)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b = b / s_b

        if use_gs:
            best_t, _ = _golden_section_line_search(a, b, deadline, max_evals=12)
        else:
            best_t = 0.7

        a_new = (1.0 - best_t) * a + best_t * b
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        # Check for convergence
        if abs(sc - best_score) < 1e-10:
            break

    return best_a, best_score


def _adam_smooth_max(a, deadline, rng, lr=0.08, beta_init=None,
                     beta_decay=0.97, no_imp_limit=150, use_var_pen=False):
    n = len(a)
    a = np.maximum(a, 1e-12).copy()
    a = a / a.sum()

    z = np.log(a)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = _full_convolve(a)
    maxc = float(c.max())
    alpha = maxc / max(1, 2 * n - 1)
    beta = beta_init if beta_init is not None else 0.25 * maxc

    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)
        g = _cross_correlate(w, a)

        if use_var_pen:
            # Add gradient of variance penalty
            c_mean = c.mean()
            c_shifted = c - c_mean
            w_var = np.exp(c_shifted / beta)
            w_var = w_var / w_var.sum()
            g_var = _cross_correlate(w_var, a)
            g = 0.7 * g + 0.3 * g_var

        dot = float(np.dot(a, g))
        gz = a * (g - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()

        a = np.exp(z - z.max())
        a = a / a.sum()
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        c = _full_convolve(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.005, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        if t_adam % 30 == 0:
            beta = max(1e-5, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[min(1, c.size-1)]]
                if gap < beta * 0.05:
                    beta = max(1e-5, beta * 0.7)

        if no_imp > no_imp_limit:
            ns = 0.06 + 0.04 * (best_score - 1.5) if best_score > 1.5 else 0.1
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a)
            beta = 0.25 * c.max()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score


def _simulated_annealing(a, deadline, rng, max_steps=5000):
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    temp = 0.001 / n
    cool = 0.9997

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break

        # Random perturbation: modify 1-3 random elements
        n_mod = rng.integers(1, 4)
        idx = rng.integers(0, n, size=n_mod)
        a_try = a.copy()
        for i in idx:
            factor = np.exp(rng.normal(0, 0.3))
            a_try[i] = max(a_try[i] * factor, 1e-15)
        a_try = a_try / a_try.sum()

        sc_try = _score_norm(a_try)

        if sc_try < best_score or rng.random() < np.exp(-(sc_try - best_score) / temp):
            a = a_try
            if sc_try < best_score:
                best_score = sc_try
                best_a = a.copy()

        temp *= cool

    return best_a, best_score


def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new


def _make_profile(n, profile_type, p1=1.0, p2=0.0, p3=0.0):
    if profile_type == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (t - 0.0)
    elif profile_type == 'linear_center':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * t
    elif profile_type == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (2 * t - 1) ** 2
    elif profile_type == 'cubic':
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + p1 * u + p2 * u ** 3
    elif profile_type == 'exp':
        t = np.arange(n, dtype=np.float64)
        a = np.exp(-p1 * t / n)
    elif profile_type == 'exp_rev':
        t = np.arange(n, dtype=np.float64)
        a = np.exp(-p1 * (n - 1 - t) / n)
    elif profile_type == 'cosine':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + p1 * np.cos(2 * np.pi * p2 * t / n)
    elif profile_type == 'sine':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + p1 * np.sin(2 * np.pi * p2 * t / n)
    elif profile_type == 'parabola':
        t = np.linspace(0, 1, n)
        a = (t - 0.5) ** 2 + p1
    elif profile_type == 'two_gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - p1) / p3) ** 2) + np.exp(-0.5 * ((t - p2) / p3) ** 2)
    elif profile_type == 'gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / p1) ** 2)
    elif profile_type == 'U_shape':
        t = np.linspace(0, 1, n)
        a = (2 * t - 1) ** 2 + p1
    elif profile_type == 'edge_heavy':
        t = np.arange(n, dtype=np.float64) / max(1, n - 1)
        a = 1.0 / (0.01 + abs(t - 0.5) * 2.0) ** p1
    elif profile_type == 'one_spike':
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        a[-1] = 1.0 + p1
    elif profile_type == 'kwik':
        # Designed for flat convolution: high at one end, low at other
        t = np.linspace(0, 1, n)
        a = p1 + (1.0 - p1) * t ** p2
    elif profile_type == 'randn_clipped':
        a = 1.0 + p1 * np.tanh(np.random.standard_normal(n))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization: LP fixed-point with golden-section line search,
    Adam smooth-max with adaptive temperature, simulated annealing, and
    multi-scale escalation to minimize 2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    # === Phase 1: Generate diverse initial candidates ===
    print("Phase 1: Generating initial candidates...")
    t_phase1_end = t_start + 60.0
    candidates = []

    n_sizes = [128, 256, 512, 1024]
    profile_defs = [
        ('lin1', 'linear', 1.0, 0, 0),
        ('lin2', 'linear', 2.0, 0, 0),
        ('lin3', 'linear', 3.0, 0, 0),
        ('lin2c', 'linear_center', 2.0, 0, 0),
        ('quad1', 'quadratic', 1.0, 0, 0),
        ('quad2', 'quadratic', 2.0, 0, 0),
        ('quad3', 'quadratic', 3.0, 0, 0),
        ('ub1', 'U_shape', 0.1, 0, 0),
        ('ub2', 'U_shape', 0.3, 0, 0),
        ('ub3', 'U_shape', 0.5, 0, 0),
        ('exp1', 'exp', 1.0, 0, 0),
        ('exp2', 'exp', 2.0, 0, 0),
        ('exp3', 'exp', 3.0, 0, 0),
        ('exp1r', 'exp_rev', 1.0, 0, 0),
        ('exp2r', 'exp_rev', 2.0, 0, 0),
        ('cos1', 'cosine', 0.5, 1, 0),
        ('cos2', 'cosine', 0.5, 2, 0),
        ('sin1', 'sine', 0.5, 1, 0),
        ('kwik1', 'kwik', 0.3, 2.0, 0),
        ('kwik2', 'kwik', 0.2, 3.0, 0),
        ('kwik3', 'kwik', 0.5, 1.5, 0),
        ('spike1', 'one_spike', 2.0, 0, 0),
        ('spike2', 'one_spike', 3.0, 0, 0),
        ('cub1', 'cubic', 1.5, -1.0, 0),
        ('cub2', 'cubic', 2.0, -0.5, 0),
    ]

    for n in n_sizes:
        for (label, ptype, p1, p2, p3) in profile_defs:
            p = _make_profile(n, ptype, p1, p2, p3)
            sc = _score_norm(p)
            candidates.append((sc, p, n, label))
        if time.monotonic() > t_phase1_end:
            break

    # Add parent-based candidates
    if parent is not None:
        for n in [256, 512, 1024, 2048]:
            pi = _interp(parent, n)
            sc = _score_norm(pi)
            candidates.append((sc, pi, n, 'parent'))
        if 0 < pn <= 4096:
            sc = _score_norm(parent)
            candidates.append((sc, parent.copy(), pn, 'parent_exact'))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, best init: {candidates[0][0]:.8f} n={candidates[0][2]} [{candidates[0][3]}]")

    if candidates[0][1] is not None and candidates[0][0] < best_score:
        best_seq = np.asarray(candidates[0][1], dtype=np.float64).copy()
        best_score = float(candidates[0][0])
        print(f"  BEST: {best_score:.10f} n={best_seq.size} [init]")

    # === Phase 2: LP fixed-point on top candidates ===
    print("Phase 2: LP fixed-point optimization...")
    t_phase2_end = t_start + 500.0
    polished = []

    # Sort by (score, n) - prefer small n for LP speed
    candidates_lp = sorted(candidates, key=lambda x: (x[0], x[2]))
    count = 0
    for (sc, p, n, label) in candidates_lp:
        if time.monotonic() > t_phase2_end or count >= 12:
            break
        if n > 1024:
            continue
        time_alloc = max(30.0, min(60.0, 500.0 * (256.0 / float(n))))
        per_end = min(t_phase2_end, time.monotonic() + time_alloc)

        a_lp, sc_lp = _lp_fixed_point(p, per_end, max_iters=50, use_gs=True)
        if sc_lp < best_score:
            best_seq = np.asarray(a_lp, dtype=np.float64).copy()
            best_score = float(sc_lp)
            print(f"  BEST: {best_score:.10f} n={a_lp.size} [LP_{label}_n{n}]")
        polished.append((sc_lp, a_lp, n, label + '_lp'))
        count += 1

    polished.sort(key=lambda x: x[0])
    print(f"  After Phase 2: best={best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    # === Phase 3: Adam refinement ===
    print("Phase 3: Adam smooth-max refinement...")
    t_phase3_end = t_start + 950.0

    for (sc, p, n, label) in polished[:5]:
        if time.monotonic() > t_phase3_end - 30:
            break
        per = min(120.0, t_phase3_end - time.monotonic() - 15)
        if per < 25:
            break

        a_ad, sc_ad = _adam_smooth_max(p, time.monotonic() + per, rng,
                                       lr=0.08, beta_init=0.25 * _full_convolve(p).max(),
                                       beta_decay=0.97, no_imp_limit=150, use_var_pen=True)
        if sc_ad < best_score:
            best_seq = np.asarray(a_ad, dtype=np.float64).copy()
            best_score = float(sc_ad)
            print(f"  BEST: {best_score:.10f} n={a_ad.size} [Adam_{label}]")

        # LP polish after Adam
        if n <= 1024 and time.monotonic() < t_phase3_end - 30:
            per2 = min(40.0, t_phase3_end - time.monotonic() - 15)
            if per2 > 15:
                a_lp2, sc_lp2 = _lp_fixed_point(a_ad, time.monotonic() + per2,
                                                max_iters=20, use_gs=True)
                if sc_lp2 < best_score:
                    best_seq = np.asarray(a_lp2, dtype=np.float64).copy()
                    best_score = float(sc_lp2)
                    print(f"  BEST: {best_score:.10f} n={a_lp2.size} [LPpostAdam_{label}]")

    # Also run Adam directly on parent if available
    if parent is not None and time.monotonic() < t_phase3_end - 60:
        n_pa = min(parent.size, 2048)
        p_pa = _interp(parent, n_pa)
        per = min(100.0, t_phase3_end - time.monotonic() - 30)
        if per > 30:
            a_pa, sc_pa = _adam_smooth_max(p_pa, time.monotonic() + per, rng,
                                           lr=0.07, beta_init=0.3 * _full_convolve(p_pa).max(),
                                           beta_decay=0.97, no_imp_limit=200, use_var_pen=False)
            if sc_pa < best_score:
                best_seq = np.asarray(a_pa, dtype=np.float64).copy()
                best_score = float(sc_pa)
                print(f"  BEST: {best_score:.10f} n={a_pa.size} [Adam_parent]")

    # === Phase 4: Multi-scale escalation ===
    print("Phase 4: Multi-scale escalation...")
    t_phase4_end = t_start + 1350.0

    if best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048, 4096]
        elif n_cur <= 1024:
            target_ns = [2048, 4096]
        elif n_cur <= 2048:
            target_ns = [4096]

        for tn in target_ns:
            if time.monotonic() > t_phase4_end - 60:
                break
            p_int = _interp(best_seq, tn)
            sc_int = _score_norm(p_int)
            print(f"  Interpolated to n={tn}: {sc_int:.8f}")
            if sc_int < best_score:
                best_seq = p_int.copy()
                best_score = sc_int
                print(f"  BEST: {best_score:.10f} n={tn} [interp]")

            # Adam on interpolated (larger n)
            if time.monotonic() < t_phase4_end - 60:
                per = min(100.0, t_phase4_end - time.monotonic() - 30)
                if per > 30:
                    a_e, sc_e = _adam_smooth_max(p_int, time.monotonic() + per, rng,
                                                 lr=0.06, beta_init=0.3 * _full_convolve(p_int).max(),
                                                 beta_decay=0.97, no_imp_limit=250, use_var_pen=True)
                    if sc_e < best_score:
                        best_seq = a_e.copy()
                        best_score = sc_e
                        print(f"  BEST: {best_score:.10f} n={tn} [Adam_esc]")

            # LP if feasible
            if tn <= 1024 and time.monotonic() < t_phase4_end - 30:
                per2 = min(40.0, t_phase4_end - time.monotonic() - 15)
                if per2 > 15:
                    a_le, sc_le = _lp_fixed_point(p_int, time.monotonic() + per2,
                                                  max_iters=15, use_gs=True)
                    if sc_le < best_score:
                        best_seq = a_le.copy()
                        best_score = sc_le
                        print(f"  BEST: {best_score:.10f} n={tn} [LP_esc]")

    # === Phase 5: Simulated annealing + random restarts ===
    print("Phase 5: Annealing and restarts...")
    t_phase5_end = deadline - 80.0

    if best_seq is not None and time.monotonic() < t_phase5_end:
        n_best = best_seq.size
        n_restarts = 4
        per_r = max(30.0, (t_phase5_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > t_phase5_end:
                break
            per = min(per_r, t_phase5_end - time.monotonic() - 10)
            if per < 20:
                break

            # Random perturbation
            ns = 0.04 + 0.03 * i
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            a_s = np.exp(z - z.max())
            a_s = a_s / a_s.sum()

            # Adam on perturbed
            a_r, sc_r = _adam_smooth_max(a_s, time.monotonic() + per * 0.6, rng,
                                         lr=0.08, beta_decay=0.97, no_imp_limit=100)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  BEST: {best_score:.10f} n={n_best} [restart{i]}")

            # Quick LP polish
            if n_best <= 1024 and time.monotonic() < t_phase5_end - 15:
                per2 = min(25.0, t_phase5_end - time.monotonic() - 10)
                if per2 > 10:
                    a_rl, sc_rl = _lp_fixed_point(a_r, time.monotonic() + per2,
                                                  max_iters=10, use_gs=True)
                    if sc_rl < best_score:
                        best_seq = a_rl.copy()
                        best_score = sc_rl
                        print(f"  BEST: {best_score:.10f} n={n_best} [LP_restart{i}]")

            # Simulated annealing on best
            if time.monotonic() < t_phase5_end - 15:
                per3 = min(20.0, t_phase5_end - time.monotonic() - 10)
                if per3 > 10 and n_best <= 2048:
                    a_sa, sc_sa = _simulated_annealing(best_seq, time.monotonic() + per3, rng,
                                                       max_steps=n_best * 5)
                    if sc_sa < best_score:
                        best_seq = a_sa.copy()
                        best_score = sc_sa
                        print(f"  BEST: {best_score:.10f} n={n_best} [SA_restart{i}]")

    # === Phase 6: Final polish ===
    print("Phase 6: Final polish...")
    if time.monotonic() < deadline - 15 and best_seq is not None:
        per = deadline - time.monotonic() - 15
        if per > 20:
            if best_seq.size <= 1024:
                a_fp, sc_fp = _lp_fixed_point(best_seq, time.monotonic() + per,
                                              max_iters=30, use_gs=True)
                if sc_fp < best_score:
                    best_seq = a_fp.copy()
                    best_score = sc_fp
                    print(f"  BEST: {best_score:.10f} n={best_seq.size} [final_LP]")
            else:
                a_fp, sc_fp = _adam_smooth_max(best_seq, time.monotonic() + per, rng,
                                               lr=0.03, beta_decay=0.97, no_imp_limit=80)
                if sc_fp < best_score:
                    best_seq = a_fp.copy()
                    best_score = sc_fp
                    print(f"  BEST: {best_score:.10f} n={best_seq.size} [final_Adam]")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile(512, 'linear_center', 2.0, 0, 0)

        exact_sc = _score_norm(final)
        print(f"\nFINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile(512, 'linear_center', 2.0, 0, 0)
    print("FALLBACK n=512")
    return [float(x) for x in fb]
