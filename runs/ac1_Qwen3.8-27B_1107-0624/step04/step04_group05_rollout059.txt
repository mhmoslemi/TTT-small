            if score < best_score - 1e-15:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.003, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)

            beta = max(3e-5, beta * beta_decay)

            no_imp = 0

    return best_p, best_score


def _peak_reduction_step(a, n_reduce=10, alpha=0.3):
    """Directly reduce the peak by lowering elements that contribute most."""
    n = len(a)
    c = _full_convolve(a)
    k_peak = int(np.argmax(c))
    # Find which (j, k_peak - j) pairs contribute most
    j_start = max(0, k_peak - n + 1)
    j_end = min(n - 1, k_peak)
    if j_start > j_end:
        return a.copy()
    idx = np.arange(j_start, j_end + 1)
    products = a[idx] * a[k_peak - idx]
    # Reduce the top contributors
    top_n = min(n_reduce, len(idx))
    top_rel = np.argsort(products)[::-1][:top_n]
    a_new = a.copy()
    for ti in top_rel:
        j = int(idx[ti])
        partner = k_peak - j
        # Reduce both equally
        reduce_amt = alpha * min(a[j], a[partner]) * 0.5
        a_new[j] = max(1e-15, a[j] - reduce_amt)
        if partner != j:
            a_new[partner] = max(1e-15, a_new[partner] - reduce_amt)
    a_new = a_new / a_new.sum()
    return a_new


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    if n0 == 0:
        return None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def _make_uniform_profile(n):
    return np.ones(n) / n


def _make_cosine_profile(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

"""

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        c = np.convolve(a, a)
    else:
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-9
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 0.0))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _cross_corr(w, p):

    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
