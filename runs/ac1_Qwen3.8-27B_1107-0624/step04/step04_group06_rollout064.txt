"""
Optimization via power-law ramp initialization with continuous exponent refinement
and Adam log-space gradient descent using smooth-max approximation of the convolution peak.
Key insight: a[j] = (j+1)^p / sum gives scores ~0.07 for n=1024, far below previous best.

import numpy as np
import time
from scipy.optimize import minimize_scalar


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c


def _score(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    c = _full_convolve_fft(p)
    sum_a = float(p.sum())
    if sum_a < 1e-10:
        return np.inf
    return float(2.0 * n * c.max() / (sum_a * sum_a))


def _power_ramp(n, p):
    j = np.arange(1, n + 1, dtype=np.float64)
    a = j ** p
    s = float(a.sum())
    if s <= 0:
        return None
    return a / s


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if mval < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    if beta < 1e-12:
        eps_max = mval * 1e-10
        mask = c >= mval - eps_max
        cnt = int(mask.sum())
        if cnt == 0:
            eps_max = mval * 1e-6
            mask = c >= mval - eps_max
            cnt = int(mask.sum())
            if cnt == 0:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
                return mval, w
        w = np.zeros_like(c)
        w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _adam_logit(p0, deadline, rng, lr=0.03, beta_init_frac=0.05, beta_decay=0.95,
                no_imp_limit=200):
    p0 = np.asarray(p0, dtype=np.float64).ravel().copy()
    p0 = np.maximum(p0, 1e-15)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps_adam = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = _full_convolve_fft(p0)
    maxc = float(c.max())
    beta = max(1e-6, beta_init_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = _score(best_p)
    no_imp = 0

    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)
        # Gradient of smooth_max(f*f) w.r.t. f is 2*(f*w_cross) where
        # w_cross[i] = sum_k w[k]*f[k-i] (cross-correlation of w with f)
        # Actually: d/dx_i [f*f](k) = sum over valid j of f[j] if k-i=j... 
        # Let's use: gradient[i] = sum_k w[k] * (f*f)_k_xi = 2 * (w * f_reversed cross-corr)
        # (f*f)(k) = sum_j f[j]*f[k-j], d/dx_i = sum_j delta(j,i)*f[k-j] + f[j]*delta(k-j,i)
        # = f[k-i] (for valid j=i) + f[j] (for valid j=k-i)
        # So gradient[i] = sum_k w[k]*(f[k-i] + f[k-i]) if both valid = 2*sum_k w[k]*f[k-i]
        # This is 2 * (w * f)(i) where (w*f) is cross-correlation
        # Use FFT: cross_corr = ifft(conj(fft(w)) * fft(f)) shifted appropriately
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp = np.fft.rfft(p0, N)
        # cross-correlation: c[i] = sum_k w[k] * f[k-i] for i in 0..n-1
        # This is the convolution of w with reversed f, evaluated at appropriate offsets
        # Actually simpler: grad[i] = sum_k w[k] * f[(k-i) mod? No, need range check
        # Let's just do: grad = 2 * ifft(fft(w) * conj(fft(f_reversed)))
        # Or more directly: grad[i] = 2 * sum_{k: 0<=k-i<n, 0<=k<n} w[k]*f[k-i]
        # = 2 * (w conv f_r)[i + n - 1] where f_r[k] = f[n-1-k]
        # Actually let me just compute it via the full convolution approach:
        # f*f has length 2n-1. grad[i] = 2*sum_k w[k]*d(f*f)/dx_i at k
        # d(f*f)[k]/dx_i = f[k-i] + f[k-i] (if i valid in [0,n-1] and k-i valid)
        # = 2*f[k-i] if 0 <= k-i <= n-1 and 0 <= i <= n-1
        # So grad[i] = 2*sum_{k=i}^{i+n-1} w[k]*f[k-i] (where k goes from i to i+n-1)
        # = 2 * (w * f_r)(i + n - 1) where f_r[m] = f[n-1-m]
        # Let's use: h = ifft(fft(w, N) * fft(f_padded, N)) where f_padded is f zero-padded to N
        # Then h[k] = sum_j w[k-j] * f[j] (circular conv)
        # We want sum_j w[k] * f[k-i-j+... ] 
        # Simplest: grad[0:n] = 2 * ifft(fft(w, 2n) * conj(fft(f, 2n))) [n-1 : 2n-1]
        # Let me just use the direct formula with FFT:
        N2 = _next_pow2(2 * n - 1)
        fw2 = np.fft.rfft(w, N2)
        fp2 = np.fft.rfft(p0, N2)
        # The cross-correlation c[i] = sum_k w[k]*f[k-i] for i in 0..n-1
        # This equals the (full convolution of w with reversed f) at position (n-1+i)
        fr = p0[::-1]
        frf = np.fft.rfft(fr, N2)
        cc = np.fft.irfft(fw2 * frf, N2)
        # cc has length N2, the valid part is cc[0:2n-1]
        # c[i] = cc[i+n-1] for i in 0..n-1? Let me verify with small example.
        # Actually: full_conv(w, fr)[k] = sum_j w[j]*fr[k-j] = sum_j w[j]*f[n-1-(k-j)]
        # = sum_j w[j]*f[n-1-k+j]
        # We want grad[i] = 2*sum_k w[k]*f[k-i]
        # Let's change variable: let m = k-i, so k = m+i
        # grad[i] = 2*sum_m w[m+i]*f[m] for valid m, i
        # = 2*sum_m f[m]*w[m+i]
        # This is (f * w_rev)(i) where w_rev[m] = w[n-1-m]? No...
        # (f * w)(i) = sum_m f[m]*w[i-m]... not quite
        # Let me just use: grad = 2 * cc[n-1 : n-1+n] if the indexing works out
        # For safety, let me use a direct O(n^2) approach for small n, FFT for large
        if n <= 512:
            # Direct computation
            grad = np.zeros(n)
            for i in range(n):
                # grad[i] = 2 * sum_k w[k] * f[k-i] where k-i in [0,n-1] and k in [0, 2n-2]
                k_start = max(0, i)
                k_end = min(2 * n - 2, i + n - 1)
                if k_start <= k_end:
                    m_idx = np.arange(k_start, k_end + 1) - i
                    grad[i] = 2.0 * float(np.dot(w[k_start:k_end + 1], p0[m_idx]))
        else:
            # FFT-based: grad[i] = 2 * sum_m f[m] * w[m+i] for valid range
            # = 2 * (f * w_shifted) ... 
            # Actually the simplest FFT trick: pad w to length 2n-1, compute ifft(fft(w)*conj(fft(f)))
            # Hmm, let me just do the convolution approach
            # w has length 2n-1, f has length n
            # We want for each i in 0..n-1: sum_{m=0}^{n-1} f[m] * w[m+i]
            # = cross_correlation(f, w) evaluated at lag i
            # cross_corr(f,w)[i] = sum_m f[m]*w[m+i] = (f * w_rev)(i + len(w) - 1 - ... )
            # Let me use: ifft(conj(fft(w_zero_padded)) * fft(f_zero_padded))
            Nc = _next_pow2(2 * n - 1 + n)
            w_pad = np.zeros(Nc)
            w_pad[:len(w)] = w
            f_pad = np.zeros(Nc)
            f_pad[:n] = p0
            cw = np.fft.rfft(w_pad, Nc)
            cf = np.fft.rfft(f_pad, Nc)
            cc2 = np.fft.irfft(np.conj(cw) * cf, Nc)
            # cc2[i] = sum_k w_pad[k] * f_pad[k-i] ... circular
            # For our purposes: grad[i] = 2 * cc2[i] for i in 0..n-1 (if the indexing aligns)
            # Let me verify: cc2[i] = sum_k w_pad[k] * f_pad[(k-i) mod Nc]
            # For k in 0..2n-2, f_pad[k-i] = f[k-i] if 0 <= k-i < n else 0
            # So cc2[i] = sum_{k=i}^{i+n-1} w[k]*f[k-i] (when i < n)
            # That's exactly what we want! grad[i] = 2*cc2[i]
            grad = 2.0 * cc2[:n]

        # Now convert to log-space gradient
        # d(score)/dz where z = log(p), p = softmax(z) (after adding constant)
        # Actually we're using p = exp(z) / sum(exp(z)) implicitly via the normalization
        # The constraint is sum(p) = 1, so we work in the simplex
        # Gradient in log-space: gz[i] = p[i] * (grad[p]_i - sum_j p[j]*grad[p]_j)
        vp = grad  # gradient w.r.t. p (unconstrained)
        dot = float(np.dot(p0, vp))
        gz = p0 * (vp - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps_adam)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -30.0, 30.0)
        # Convert back to p
        z_shifted = z - float(np.max(z))
        e = np.exp(z_shifted)
        s = float(e.sum())
        if not np.isfinite(s) or s <= 0:
            p_new = np.ones(n) / n
        else:
            p_new = e / s

        c = _full_convolve_fft(p_new)
        maxc = float(c.max())
        sc = float(2.0 * n * maxc)

        if np.isfinite(sc):
            if sc < best_score - 1e-14:
                best_score = sc
                best_p = p_new.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 50 == 0:
            beta = max(1e-8, beta * beta_decay)

        # Restart if stuck
        if no_imp > no_imp_limit:
            ns = 0.05
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -30.0, 30.0)
            z_shifted = z - float(np.max(z))
            e = np.exp(z_shifted)
            s = float(e.sum())
            if s > 0 and np.isfinite(s):
                p_new = e / s
            else:
                p_new = best_p.copy()
            c = _full_convolve_fft(p_new)
            maxc = float(c.max())
            beta = max(1e-6, beta_init_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            no_imp = 0

        p0 = p_new

    return best_p, best_score


def _refine_p(n, p_low, p_high, max_evals=30):
    """Minimize score over p using golden-section search."""
    def f(p_val):
        a = _power_ramp(n, p_val)
        if a is None:
            return np.inf
        return _score(a)

    res = minimize_scalar(f, bounds=(p_low, p_high), method='bounded',
                          options={'maxiter': max_evals})
    return res.x, res.fun


def _peak_flood(p0, deadline, eta=0.03, max_iter=2000):
    """Multiplicative update to reduce peak contributions."""
    p0 = np.asarray(p0, dtype=np.float64).ravel().copy()
    p0 = np.maximum(p0, 1e-15)
    p0 = p0 / p0.sum()
    n = p0.size
    best_p = p0.copy()
    best_score = _score(p0)
    c = _full_convolve_fft(p0)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        M = float(c.max())
        sc = float(2.0 * n * M)
        if sc < best_score - 1e-14:
            best_score = sc
            best_p = p0.copy()

        # Find positions contributing to peaks
        threshold = M * 0.92
        hot_mask = c >= threshold
        if not hot_mask.any():
            arr = np.zeros(2 * n - 1, dtype=bool)
            arr[int(np.argmax(c))] = True
            hot_mask = arr

        # Compute heat
        heat = np.zeros(n)
        hot_indices = np.where(hot_mask)[0]
        for k in hot_indices:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                partner = k - np.arange(i_start, i_end + 1)
                heat[i_start:i_end + 1] += p0[partner]

        heat_max = float(heat.max())
        if heat_max < 1e-15:
            break
        heat = heat / heat_max

        factor = 1.0 - eta * heat
        factor = np.maximum(factor, 0.01)
        p0 = p0 * factor
        p0 = np.maximum(p0, 1e-15)
        p0 = p0 / p0.sum()

        c = _full_convolve_fft(p0)

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.maximum(p, 0.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = p.copy()
            print(f"  NEW BEST: {sc:.10f} n={p.size} [{label}] t={time.monotonic()-t_start:.1f}s")

    print(f"=== Starting optimization, budget={budget_s}s ===")

    # Phase 1: Sweep over (n, p) to find best power-law ramp
    print("Phase 1: Sweeping power-law ramps...")
    n_candidates = [256, 512, 768, 1024, 1536, 2048, 3072, 4096]
    p_candidates = np.arange(0.5, 8.5, 0.5)

    best_sweep_score = np.inf
    best_sweep_n = 1024
    best_sweep_p = 4.0

    for n in n_candidates:
        if time.monotonic() > deadline - 600:
            break
        for p_val in p_candidates:
            a = _power_ramp(n, p_val)
            if a is None:
                continue
            sc = _score(a)
            if np.isfinite(sc) and sc < best_sweep_score:
                best_sweep_score = sc
                best_sweep_n = n
                best_sweep_p = p_val

    print(f"  Best sweep: n={best_sweep_n}, p={best_sweep_p}, score={best_sweep_score:.8f}")
    a_sweep = _power_ramp(best_sweep_n, best_sweep_p)
    update(a_sweep, 'sweep')

    # Phase 2: Refine p with golden section for the best n (and nearby n values)
    print(f"Phase 2: Refining p around p={best_sweep_p:.3f}...")
    for n_try in [best_sweep_n, best_sweep_n // 2, best_sweep_n * 2]:
        if n_try < 128 or n_try > 16384:
            continue
        if time.monotonic() > deadline - 500:
            break
        p_lo = max(0.1, best_sweep_p - 1.0)
        p_hi = best_sweep_p + 1.0
        p_opt, sc_opt = _refine_p(n_try, p_lo, p_hi, max_evals=25)
        a_opt = _power_ramp(n_try, p_opt)
        update(a_opt, f'refine_p_n{n_try}_p{p_opt:.4f}')

    # Phase 3: Peak flooding on best
    print(f"Phase 3: Peak flooding, current best={best_score:.8f}...")
    if time.monotonic() < deadline - 300:
        pf_end = min(time.monotonic() + 120, deadline - 300)
        p_fl, sc_fl = _peak_flood(best_seq, pf_end, eta=0.03, max_iter=1000)
        update(p_fl, 'peak_flood')

    # Phase 4: Adam log-space optimization
    print(f"Phase 4: Adam log-space, current best={best_score:.8f}...")
    if time.monotonic() < deadline - 100:
        adam_end = min(time.monotonic() + 300, deadline - 100)
        p_ad, sc_ad = _adam_logit(best_seq, adam_end, rng, lr=0.03,
                                  beta_init_frac=0.05, beta_decay=0.95,
                                  no_imp_limit=150)
        update(p_ad, 'adam1')

    # Phase 5: Another round of peak flooding
    print(f"Phase 5: Peak flood round 2, best={best_score:.8f}...")
    if time.monotonic() < deadline - 100:
        pf2_end = min(time.monotonic() + 100, deadline - 100)
        p_fl2, sc_fl2 = _peak_flood(best_seq, pf2_end, eta=0.02, max_iter=800)
        update(p_fl2, 'peak_flood2')

    # Phase 6: Final Adam with smaller learning rate
    print(f"Phase 6: Final Adam polish, best={best_score:.8f}...")
    if time.monotonic() < deadline - 50:
        adam2_end = deadline - 30
        p_ad2, sc_ad2 = _adam_logit(best_seq, adam2_end, rng, lr=0.01,
                                    beta_init_frac=0.03, beta_decay=0.96,
                                    no_imp_limit=100)
        update(p_ad2, 'adam_final')

    # Phase 7: Try alternative n values with the best p
    print(f"Phase 7: Try other scales, best={best_score:.8f}...")
    if time.monotonic() < deadline - 60 and best_seq is not None:
        current_n = best_seq.size
        for n_new in [current_n * 2, current_n // 2]:
            if n_new < 128 or n_new > 16384 or n_new == current_n:
                continue
            if time.monotonic() > deadline - 60:
                break
            # Interpolate
            x0 = np.linspace(0, 1, current_n)
            x1 = np.linspace(0, 1, n_new)
            a_interp = np.interp(x1, x0, best_seq)
            a_interp = np.maximum(a_interp, 1e-15)
            a_interp = a_interp / a_interp.sum()
            update(a_interp, f'interp_n{n_new}')

            # Quick Adam at new scale
            if time.monotonic() < deadline - 60:
                ad_end = min(time.monotonic() + 80, deadline - 60)
                p_new_ad, sc_new = _adam_logit(a_interp, ad_end, rng, lr=0.02,
                                               beta_init_frac=0.04, beta_decay=0.95,
                                               no_imp_limit=80)
                update(p_new_ad, f'adam_n{n_new}')

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        exact_sc = _score(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            print(f"EVALUATE: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    # Fallback: power ramp
    fb = _power_ramp(1024, 4.0)
    print(f"FALLBACK: n=1024, p=4.0, score={_score(fb):.8f}")
    return [float(x) for x in fb]
