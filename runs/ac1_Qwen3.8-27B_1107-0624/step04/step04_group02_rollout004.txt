def _run_pnorm_lbfgs(a_init, deadline, rng, p_schedule=None, max_wall_time=None):
    """L-BFGS with p-norm continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    if p_schedule is None:
        p_schedule = [4, 8, 16, 32, 64, 128, 256, 512]
    
    if max_wall_time is not None:
        per_p = max(5.0, max_wall_time / len(p_schedule))
    else:
        per_p = 60.0
    
    for p_val in p_schedule:
        if time.monotonic() > deadline - 3.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_p)
        
        p_use = float(p_val)
        
        def obj_fn(z_):
            val, _ = _pnorm_obj_and_grad(z_, n, p_use)
            return val
        
        def grad_fn(z_):
            _, g = _pnorm_obj_and_grad(z_, n, p_use)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g
        
        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 500, 'ftol': 1e-20, 'gtol': 1e-12,
                         'maxls': 30, 'maxfun': 3000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                # softmax
                z_s = z_new - np.max(z_new)
                e = np.exp(z_s)
                a_new = e / e.sum()
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass
    
    return best_a, best_score

def _deflation_step(a, deadline, rng, n_steps=500):
    """Post-optimization: reduce the top lags specifically."""
    n = len(a)
    a_curr = a.copy()
    a_curr = a_curr / a_curr.sum()
    best_a = a_curr.copy()
    best_score = _score_norm_fn(a_curr)
    
    lr = 0.01
    
    for step in range(n_steps):
        if time.monotonic() > deadline - 2.0:
            break
        
        c = _full_convolve(a_curr, a_curr)
        cm = c.max()
        
        # Find top-K lags
        K = min(3, len(c))
        top_idx = np.argsort(c)[-K:]
        
        # Compute gradient as average over top-K
        grad = np.zeros(n)
        for k_val in top_idx:
            k_val = int(k_val)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                # d(a*a)_{k_val}/da_i = 2*a_{k_val-i} for valid i
                # grad component for index i: 2 * a_{k_val - i}
                # So grad[i] = 2 * a[k_val - i] for i in [lo, hi]
                # a[k_val - i] for i in [lo, hi] means indices [k_val-hi, k_val-lo] in a
                indices = k_val - np.arange(lo, hi + 1)
                grad[lo:hi + 1] += 2.0 * a_curr[indices]
        grad /= K
        
        # Update
        a_curr = a_curr - lr * grad
        a_curr = np.maximum(a_curr, 1e-15)
        a_curr = a_curr / a_curr.sum()
        
        sc = _score_norm_fn(a_curr)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()
            lr = min(lr * 1.1, 0.1)  # increase lr slightly
        else:
            lr = max(lr * 0.95, 0.001)
    
    return best_a, best_score

def _make_profile(n, profile_type, **kwargs):
    """Create various initial profiles."""
    t = np.linspace(0.0, 1.0, n)
    
    if profile_type == 'linear':
        slope = kwargs.get('slope', 1.5)
        a = 1.0 + slope * (t - 0.5)
    elif profile_type == 'quadratic':
        a1 = kwargs.get('a1', 0.8)
        a2 = kwargs.get('a2', 0.3)
        u = 2.0 * t - 1.0
        a = 1.0 + a1 * u + a2 * u ** 2
    elif profile_type == 'gaussian':
        width = kwargs.get('width', 0.3)
        u = t - 0.5
        a = np.exp(-u ** 2 / (2.0 * width ** 2))
    elif profile_type == 'raised_cosine':
        alpha = kwargs.get('alpha', 1.0)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5) * alpha))
    elif profile_type == 'power':
        p = kwargs.get('p', 2.0)
        u = np.abs(2.0 * t - 1.0)
        a = (1.0 - u) ** p
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

    w = c ** (p - 1)  # length 2n-1
    
    # h_i = sum_{k=0}^{n-1} w[i+k] * a[k] for i = 0, ..., n-1
    # This is cross-correlation of w and a at lags 0, ..., n-1
    L = _next_pow2(3 * n)
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    cross = np.fft.irfft(W * np.conj(A), L)[:n]
    # cross[i] = sum_k w[i+k] * a[k] (approximately, with circular wrap-around if L is too small)
    # But since L >= 3n > (2n-1) + n - 1 = 3n-2, there's no wrap-around.
    
    h = 2.0 * cross  # factor of 2 from the derivative
    
    G = float(np.sum(c ** p)) ** (1.0 / p)
    grad = G ** (1.0 - p) * h
    return grad
