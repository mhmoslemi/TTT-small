def _proj_simplex(v, target_sum=1.0):
    v=np.asarray(v,float).ravel()
    n=v.size
    if n==0: return v.copy()
    if target_sum <=0: return np.zeros(n)
    if not np.all(np.isfinite(v)):
       v=np.where(np.isfinite(v),v,0.0)
    u=np.sort(v)[::-1]
    css=np.cumsum(u)
    idx=np.arange(1,n+1,dtype=float)
    cond = u - (css - target_sum) / idx > 0.0
    valid=np.nonzero(cond)[0]
    if valid.size ==0:
        return np.full(n, target_sum/n)
    rho = int(valid[-1]) + 1

    x = np.maximum(v - theta, 0.0)
    s = x.sum()
    if s > 0 and np.isfinite(s):
        x = x * (target_sum / s)
    else:
        x = np.full(n, target_sum/n)
    return x

```python
"""
Optimizer for minimizing 2*n*max(convolve(a,a))/(sum(a)^2).

Strategy:
- Use FFT-based convolution for fast objective/gradient evaluation.
- Start from the provided parent sequence when available, plus a few analytic profiles.
- Optimize mainly by:
  1) L-BFGS-B on log-softmax variables with a log-sum-exp continuation.
  2) Adam on log-softmax variables with smoothed max.
  3) Multiplicative mirror-descent updates using lag weights concentrated near the maximum.
  4) A short projected-gradient phase and random local block polish.
- Try a longer sequence by interpolation if it looks promising.
"""

import time
import numpy as np
from scipy.optimize import minimize


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (int(n) - 1).bit_length()


def _make_uniform(n):
    if n < 2:
        n = 2
    n = int(n)
    return np.full(n, 1.0 / float(n), dtype=float)


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=float).ravel()
    if a.size == 0:
        return _make_uniform(2048)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-12:
        return _make_uniform(max(2, a.size))
    return a / s


def _softmax(z):
    z = np.asarray(z, dtype=float)
    if z.size == 0:
        return z.copy()
    zm = z - np.max(z)
    e = np.exp(np.clip(zm, -745.0, 709.0))
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return np.ones(z.size, dtype=float) / float(z.size)
    return e / s


def _convolve_self(a):
    a = np.asarray(a, dtype=float).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=float)
    out_n = 2 * n - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:out_n]
    c = np.maximum(c, 0.0)
    return c


def _convolve_ab(a, b):
    a = np.asarray(a, dtype=float).ravel()
    b = np.asarray(b, dtype=float).ravel()
    if a.size == 0 or b.size == 0:
        return np.array([], dtype=float)
    out_n = a.size + b.size - 1
    N = _next_pow2(out_n)
    ra = np.fft.rfft(a, N)
    rb = np.fft.rfft(b, N)
    r = np.fft.irfft(ra * rb, N)[:out_n]
    return np.maximum(r, 0.0)


def _score_norm(a):
    a = np.asarray(a, dtype=float).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve_self(a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _proj_simplex(v, target_sum=1.0):
    v = np.asarray(v, dtype=float).ravel()
    n = v.size
    if n == 0:
        return v.copy()
    if not np.all(np.isfinite(v)):
        v = np.where(np.isfinite(v), v, 0.0)
    if target_sum <= 0.0:
        return np.zeros(n, dtype=float)

    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=float)
    cond = u - (css - float(target_sum)) / idx > 0.0
    valid = np.nonzero(cond)[0]

    if valid.size == 0:
        return np.full(n, target_sum / float(n), dtype=float)

    rho = int(valid[-1]) + 1
    theta = (css[rho - 1] - float(target_sum)) / float(rho)
    x = np.maximum(v - theta, 0.0)
    s = float(x.sum())
    if s > 0.0 and np.isfinite(s):
        x = x * (float(target_sum) / s)
    else:
        x = np.full(n, float(target_sum) / float(n), dtype=float)
    return x


def _profile_linear(n, slope):
    if n < 2:
        return _make_uniform(max(2, n))
    t = np.linspace(0.0, 1.0, int(n), dtype=float)
    a = 1.0 + float(slope) * (t - 0.5)
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return _make_uniform(n)
    return a / s


def _profile_quadratic(n, a1, a2):
    if n < 2:
        return _make_uniform(max(2, n))
    t = np.linspace(0.0, 1.0, int(n), dtype=float)
    u = 2.0 * t - 1.0
    a = 1.0 + float(a1) * u + float(a2) * u * u
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return _make_uniform(n)
    return a / s


def _profile_raised_cosine(n, alpha):
    if n < 2:
        return _make_uniform(max(2, n))
    t = np.linspace(0.0, 1.0, int(n), dtype=float)
    a = 0.25 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5) * float(alpha)))
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return _make_uniform(n)
    return a / s


def _profile_power(n, p):
    if n < 2:
        return _make_uniform(max(2, n))
    t = np.linspace(0.0, 1.0, int(n), dtype=float)
    u = np.abs(2.0 * t - 1.0)
    a = np.power(1.0 - u, float(p)) + 1e-12
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return _make_uniform(n)
    return a / s


def _profile_exponential(n, gamma):
    if n < 2:
        return _make_uniform(max(2, n))
    t = np.linspace(0.0, 1.0, int(n), dtype=float)
    a = np.exp(float(gamma) * (t - 0.5))
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return _make_uniform(n)
    return a / s


def _make_candidate_profiles(n):
    prof = []
    for sl in (0.8, 1.2, 1.6, 2.2):
        prof.append(_profile_linear(n, sl))
    for a1, a2 in ((0.5, 0.2), (0.9, 0.1), (1.3, -0.2), (0.6, 0.4), (1.0, 0.3)):
        prof.append(_profile_quadratic(n, a1, a2))
    for al in (0.8, 1.0, 1.2, 1.5):
        prof.append(_profile_raised_cosine(n, al))
    for p in (1.0, 1.5, 2.0, 2.8):
        prof.append(_profile_power(n, p))
    for g in (-2.5, -1.5, 1.5, 2.5):
        prof.append(_profile_exponential(n, g))
    return prof


def _interp_profile(p, n_new):
    p = np.asarray(p, dtype=float).ravel()
    n_old = p.size
    n_new = int(n_new)
    if n_new < 1:
        n_new = 1
    if n_old == n_new:
        return _normalize_nonneg(p)
    if n_old <= 1:
        return _make_uniform(n_new)
    x0 = np.linspace(0.0, 1.0, n_old, dtype=float)
    x1 = np.linspace(0.0, 1.0, n_new, dtype=float)
    q = np.interp(x1, x0, p)
    q = np.maximum(q, 1e-12)
    s = float(q.sum())
    if s <= 0.0:
        return _make_uniform(n_new)
    return q / s


def _symmetrize(a):
    a = np.asarray(a, dtype=float).ravel().copy()
    a = 0.5 * (a + a[::-1])
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    if s <= 0.0:
        return _make_uniform(a.size)
    return a / s


def _fun_lse_lbfgs(z, beta):
    n = len(z)
    a = _softmax(z)
    c = _convolve_self(a)
    cm = float(c.max())
    b = max(float(beta), 1e-300)
    sh = (c - cm) / b
    e = np.exp(np.clip(sh, -745.0, 709.0))
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return 2.0 * n * cm
    return 2.0 * n * (cm + b * np.log(s))


def _jac_lse_lbfgs(z, beta):
    n = len(z)
    a = _softmax(z)
    c = _convolve_self(a)
    cm = float(c.max())
    b = max(float(beta), 1e-300)
    sh = (c - cm) / b
    e = np.exp(np.clip(sh, -745.0, 709.0))
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p = e / s
    h = _convolve_ab(p, a[::-1])[n - 1:2 * n - 1]
    T = float(np.dot(p, c))
    g = 4.0 * n * a * (h - T)
    if not np.all(np.isfinite(g)):
        g = np.zeros(n, dtype=float)
    return g


def _grad_score_z(z, beta):
    n = len(z)
    a = _softmax(z)
    c = _convolve_self(a)
    cm = float(c.max())
    b = max(float(beta), 1e-300)
    sh = (c - cm) / b
    e = np.exp(np.clip(sh, -745.0, 709.0))
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p = e / s
    h = _convolve_ab(p, a[::-1])[n - 1:2 * n - 1]
    T = float(np.dot(p, c))
    g = 4.0 * n * a * (h - T)
    if not np.all(np.isfinite(g)):
        g = np.zeros(n, dtype=float)
    return g, a, cm


def _grad_lse_a(a, beta):
    n = len(a)
    c = _convolve_self(a)
    cm = float(c.max())
    b = max(float(beta), 1e-300)
    sh = (c - cm) / b
    e = np.exp(np.clip(sh, -745.0, 709.0))
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p = e / s
    h = _convolve_ab(p, a[::-1])[n - 1:2 * n - 1]
    g = 4.0 * n * h
    if not np.all(np.isfinite(g)):
        g = np.zeros(n, dtype=float)
    return g


def _estimate_eval_time(z, beta):
    t0 = time.monotonic()
    _fun_lse_lbfgs(z, beta)
    _jac_lse_lbfgs(z, beta)
    dt = time.monotonic() - t0
    return max(1e-5, float(dt))


def _run_lbfgs_continuation(a_init, deadline, rng, beta_scale=1.0):
    a = _normalize_nonneg(a_init)
    best_a = a.copy()
    best_sc = _score_norm(a)
    if not np.isfinite(best_sc):
        return best_a, best_sc

    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    best_z = z.copy()

    c0 = _convolve_self(a)
    cm0 = float(c0.max())
    if cm0 <= 0.0 or not np.isfinite(cm0):
        cm0 = 1.0 / float(a.size)

    fracs = (
        1.0, 0.75, 0.55, 0.40, 0.30, 0.22, 0.16, 0.12, 0.09, 0.067,
        0.050, 0.038, 0.028, 0.021, 0.016, 0.012, 0.009, 0.0068,
        0.0050, 0.0038, 0.0028, 0.0021, 0.0015, 0.0011, 0.0008,
        0.0006, 0.00045, 0.00033, 0.00025, 0.00018
    )

    beta_start = max(0.25 * float(beta_scale) * cm0, 1e-12)
    eval_time = _estimate_eval_time(z, beta_start)
    remaining = max(10.0, float(deadline) - time.monotonic())
    per_call_time = max(3.0, min(25.0, remaining / float(max(1, len(fracs))))) * 0.8

    no_improve = 0

    for idx, fr in enumerate(fracs):
        if time.monotonic() > float(deadline) - 4.0:
            break

        beta = max(beta_start * float(fr), 1e-12 * cm0)
        maxfun = int(0.65 * per_call_time / max(eval_time, 1e-5))
        maxfun = max(20, min(3000, maxfun))

        opts = {
            "maxfun": int(maxfun),
            "maxiter": int(max(20, maxfun // 2)),
            "ftol": 1e-15,
            "gtol": 1e-10,
            "maxls": 30
        }

        try:
            res = minimize(
                _fun_lse_lbfgs,
                z,
                jac=_jac_lse_lbfgs,
                args=(beta,),
                method="L-BFGS-B",
                options=opts
            )
        except Exception:
            break

        x_new = getattr(res, "x", None)
        if x_new is None:
            continue
        x_new = np.asarray(x_new, dtype=float)
        if x_new.shape != z.shape or not np.all(np.isfinite(x_new)):
            continue

        z2 = np.clip(x_new, -60.0, 60.0)
        z2 = z2 - z2.mean()
        z = z2
        a2 = _softmax(z)
        sc2 = _score_norm(a2)

        if np.isfinite(sc2):
            a = a2
            if sc2 < best_sc - 1e-15:
                best_a = a2.copy()
                best_sc = float(sc2)
                best_z = z.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve >= 8:
                    z = best_z.copy()
                    a = _softmax(z)
                    no_improve = 0

    return best_a, best_sc


def _optimize_adam_z(a_init, deadline, rng, lr_init=0.012):
    a = _normalize_nonneg(a_init)
    best_a = a.copy()
    best_sc = _score_norm(a)
    if not np.isfinite(best_sc):
        return best_a, best_sc

    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    best_z = z.copy()

    fracs = (
        0.40, 0.30, 0.22, 0.16, 0.12, 0.09, 0.067, 0.050, 0.038,
        0.028, 0.021, 0.016, 0.012, 0.009, 0.0068, 0.0050, 0.0038,
        0.0028, 0.0021, 0.0016
    )

    remaining = max(10.0, float(deadline) - time.monotonic())
    per_phase = max(3.0, remaining / float(max(1, len(fracs))) * 0.85)

    for fr in fracs:
        if time.monotonic() > float(deadline) - 2.0:
            break

        a = _softmax(z)
        cur_sc = _score_norm(a)
        if not np.isfinite(cur_sc) or cur_sc > best_sc * 1.005:
            z = best_z.copy()
            a = _softmax(z)
            cur_sc = best_sc

        n = a.size
        cm_now = cur_sc / (2.0 * n)
        beta = max(0.45 * float(fr) * max(cm_now, 1e-15), 1e-12 * max(cm_now, 1e-15))

        m = np.zeros(n, dtype=float)
        v = np.zeros(n, dtype=float)
        step = 0
        b1 = 0.85
        b2 = 0.999
        eps = 1e-8

        if float(fr) > 0.08:
            lr = float(lr_init)
        elif float(fr) > 0.015:
            lr = 0.007
        else:
            lr = 0.0035

        phase_deadline = min(float(deadline) - 2.0, time.monotonic() + per_phase)

        for it in range(1000000):
            if time.monotonic() > phase_deadline:
                break

            g, a_curr, cm = _grad_score_z(z, beta)
            if not np.all(np.isfinite(g)):
                break

            step += 1
            bias1 = 1.0 - b1 ** step
            bias2 = 1.0 - b2 ** step
            m = b1 * m + (1.0 - b1) * g
            v = b2 * v + (1.0 - b2) * (g * g)
            mh = m / bias1
            vh = v / bias2

            z = z - lr * mh / (np.sqrt(vh) + eps)
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if (it + 1) % 40 == 0:
                a2 = _softmax(z)
                sc2 = _score_norm(a2)
                if np.isfinite(sc2) and sc2 < best_sc - 1e-15:
                    best_a = a2.copy()
                    best_sc = float(sc2)
                    best_z = z.copy()

            if (it + 1) % 1000 == 0:
                lr = max(0.0005, lr * 0.985)

        a2 = _softmax(z)
        sc2 = _score_norm(a2)
        if np.isfinite(sc2) and sc2 < best_sc - 1e-15:
            best_a = a2.copy()
            best_sc = float(sc2)
            best_z = z.copy()

    return best_a, best_sc


def _optimize_mirror(a_init, deadline, rng):
    a = _normalize_nonneg(a_init)
    best_a = a.copy()
    best_sc = _score_norm(a)
    if not np.isfinite(best_sc):
        return best_a, best_sc

    cur = a.copy()
    cur_c = _convolve_self(cur)
    cur_cm = float(cur_c.max())
    cur_sc = 2.0 * float(cur.size) * cur_cm

    if not np.isfinite(cur_sc):
        cur_sc = best_sc

    alphas = (
        0.70, 0.50, 0.35, 0.25, 0.18, 0.13, 0.09, 0.065, 0.050,
        0.038, 0.028, 0.021, 0.016, 0.012, 0.009, 0.0068,
        0.0050, 0.0038, 0.0028, 0.0021, 0.0016, 0.0012,
        0.0009, 0.0007
    )

    remaining = max(10.0, float(deadline) - time.monotonic())
    per_phase = max(2.0, remaining / float(max(1, len(alphas))) * 0.9)

    eta = 0.7
    n = cur.size

    for al in alphas:
        if time.monotonic() > float(deadline) - 2.0:
            break

        if cur_cm <= 0.0 or not np.isfinite(cur_cm):
            cur = best_a.copy()
            cur_c = _convolve_self(cur)
            cur_cm = float(cur_c.max())
            cur_sc = 2.0 * float(cur.size) * cur_cm

        tau = max(float(al) * max(cur_cm, 1e-15), 1e-12)
        phase_deadline = min(float(deadline) - 2.0, time.monotonic() + per_phase)
        no_improve = 0

        while time.monotonic() < phase_deadline:
            if cur_cm <= 0.0 or not np.isfinite(cur_cm):
                break

            sh = (cur_c - cur_cm) / tau
            e = np.exp(np.clip(sh, -745.0, 709.0))
            s = float(e.sum())
            if s <= 0.0 or not np.isfinite(s):
                p = np.zeros_like(cur_c)
                p[int(np.argmax(cur_c))] = 1.0
            else:
                p = e / s

            h = _convolve_ab(p, cur[::-1])[n - 1:2 * n - 1]
            hm = float(h.mean())
            hs = float(h.std())
            if hs <= 1e-15 or not np.isfinite(hm) or not np.isfinite(hs):
                break

            h_use = (h - hm) / (hs + 1e-15)
            trial = cur * np.exp(-eta * np.clip(h_use, -5.0, 5.0))
            trial = trial + 1e-12
            trial = trial / trial.sum()

            trial_c = _convolve_self(trial)
            trial_cm = float(trial_c.max())
            trial_sc = 2.0 * n * trial_cm

            if np.isfinite(trial_sc) and trial_sc < cur_sc - 1e-14:
                cur = trial
                cur_c = trial_c
                cur_cm = trial_cm
                cur_sc = float(trial_sc)
                eta = min(eta * 1.08, 6.0)
                no_improve = 0
                if cur_sc < best_sc:
                    best_a = cur.copy()
                    best_sc = float(cur_sc)
            else:
                eta = max(eta * 0.75, 0.02)
                no_improve += 1

                if no_improve >= 50:
                    cur = best_a.copy()
                    cur_c = _convolve_self(cur)
                    cur_cm = float(cur_c.max())
                    cur_sc = 2.0 * n * cur_cm
                    eta = 0.7
                    no_improve = 0

                if no_improve > 250:
                    break

    if cur_sc < best_sc:
        best_a = cur.copy()
        best_sc = float(cur_sc)

    return best_a, best_sc


def _optimize_pgd(a_init, deadline, rng):
    cur = _normalize_nonneg(a_init)
    best_a = cur.copy()
    best_sc = _score_norm(cur)
    if not np.isfinite(best_sc):
        return best_a, best_sc

    cur_sc = best_sc
    fracs = (
        0.20, 0.14, 0.10, 0.075, 0.055, 0.040, 0.030, 0.022,
        0.016, 0.012, 0.009, 0.0068, 0.0050, 0.0038, 0.0028, 0.0021
    )

    remaining = max(10.0, float(deadline) - time.monotonic())
    per_phase = max(2.0, remaining / float(max(1, len(fracs))) * 0.8)
    n = cur.size

    for fr in fracs:
        if time.monotonic() > float(deadline) - 2.0:
            break

        cm_now = cur_sc / (2.0 * n)
        beta = max(0.5 * float(fr) * max(cm_now, 1e-15), 1e-12 * max(cm_now, 1e-15))
        L = max(25.0 * n * float(cur.max()) + 1.0, 1.0)
        fail = 0
        phase_deadline = min(float(deadline) - 2.0, time.monotonic() + per_phase)

        while time.monotonic() < phase_deadline:
            g = _grad_lse_a(cur, beta)
            if not np.all(np.isfinite(g)):
                break

            z = cur - g / L
            x = _proj_simplex(z, 1.0)
            sc = _score_norm(x)

            if np.isfinite(sc) and sc < cur_sc - 1e-14:
                cur = x
                cur_sc = float(sc)
                fail = 0
                L = max(5.0 * n * float(cur.max()) + 1.0, L * 0.9)
                if cur_sc < best_sc:
                    best_a = cur.copy()
                    best_sc = float(cur_sc)
            else:
                L = min(L * 1.25, 1e6)
                fail += 1
                if fail > 30:
                    break

    return best_a, best_sc


def _random_block_search(a_init, deadline, rng):
    cur = _normalize_nonneg(a_init)
    best_a = cur.copy()
    best_sc = _score_norm(cur)
    if not np.isfinite(best_sc):
        return best_a, best_sc

    cur_sc = best_sc
    n = cur.size
    if n < 8:
        return best_a, best_sc

    max_bs = int(max(4, min(128, n // 4)))
    scales = (0.02, 0.04, 0.08, 0.15)

    while time.monotonic() < float(deadline) - 1.0:
        bs = int(rng.integers(2, max_bs + 1))
        start = int(rng.integers(0, max(1, n - bs + 1)))

        for _attempt in range(4):
            neg = 1.0 if rng.random() < 0.5 else -1.0
            scale = neg * float(scales[int(rng.integers(0, 4))])

            if rng.random() < 0.5:
                pert = np.exp(scale * rng.normal(size=bs))
            else:
                tt = np.linspace(0.0, 1.0, bs, dtype=float)
                pert = 1.0 + scale * tt

            trial = cur.copy()
            trial[start:start + bs] *= np.maximum(pert, 1e-3)
            trial = np.maximum(trial, 1e-12)
            s = float(trial.sum())
            if s <= 0.0:
                continue
            trial = trial / s
            sc = _score_norm(trial)

            if np.isfinite(sc) and sc < cur_sc - 1e-14:
                cur = trial
                cur_sc = float(sc)
                if cur_sc < best_sc:
                    best_a = cur.copy()
                    best_sc = float(cur_sc)
                break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    budget_s = max(60.0, float(budget_s))
    deadline = t_start + max(60.0, budget_s - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    parent = None
    try:
        raw = np.asarray(height_sequence_1, dtype=float).ravel()
    except Exception:
        raw = np.array([], dtype=float)

    if raw.size > 0:
        raw = np.where(np.isfinite(raw), raw, 0.0)
        raw = np.maximum(raw, 0.0)
        s = float(raw.sum())
        if s > 0.01:
            parent = raw / s

    if parent is None:
        parent = _make_uniform(2048)
    if parent.size > 16384:
        parent = _interp_profile(parent, 8192)

    best_a = parent.copy()
    best_sc = _score_norm(best_a)
    print(f"initial: n={best_a.size}, score={best_sc:.10f}")

    # Quick deterministic symmetrization.
    sym = _symmetrize(best_a)
    sc = _score_norm(sym)
    if np.isfinite(sc) and sc < best_sc:
        best_a = sym.copy()
        best_sc = float(sc)
        print(f"sym: n={best_a.size}, score={best_sc:.10f}")

    # Analytic profiles at current length.
    n_prof = best_a.size
    if n_prof > 8192:
        n_prof = 8192
    profs = _make_candidate_profiles(n_prof)
    for p in profs:
        if p.size != best_a.size:
            p = _interp_profile(p, best_a.size)
        sc = _score_norm(p)
        if np.isfinite(sc) and sc < best_sc:
            best_a = p.copy()
            best_sc = float(sc)
            print(f"profile: n={best_a.size}, score={best_sc:.10f}")

    # Main LBFGS continuation.
    if time.monotonic() < deadline - 120.0:
        if best_a.size <= 4096:
            lbfgs_budget = 600.0
        elif best_a.size <= 8192:
            lbfgs_budget = 400.0
        else:
            lbfgs_budget = 220.0
        phase_end = min(deadline - 200.0, time.monotonic() + lbfgs_budget)
        a2, sc2 = _run_lbfgs_continuation(best_a, phase_end, rng)
        if np.isfinite(sc2) and sc2 < best_sc:
            best_a = a2.copy()
            best_sc = float(sc2)
            print(f"lbfgs: n={best_a.size}, score={best_sc:.10f}")

    # Try a longer sequence by interpolation if it looks promising.
    if time.monotonic() < deadline - 160.0 and best_a.size < 8192:
        n2 = int(min(8192, max(4096, best_a.size * 2)))
        a2 = _interp_profile(best_a, n2)
        sc2 = _score_norm(a2)
        if np.isfinite(sc2):
            print(f"interpolated n={n2}, score={sc2:.10f}")
            if sc2 < best_sc:
                best_a = a2.copy()
                best_sc = float(sc2)
                print(f"interpolated best: n={best_a.size}, score={best_sc:.10f}")

            if sc2 <= best_sc * 1.02:
                if time.monotonic() < deadline - 100.0:
                    phase_end = min(deadline - 100.0, time.monotonic() + 260.0)
                    a2, sc2 = _run_lbfgs_continuation(a2, phase_end, rng)
                    if np.isfinite(sc2) and sc2 < best_sc:
                        best_a = a2.copy()
                        best_sc = float(sc2)
                        print(f"lbfgs n2: n={best_a.size}, score={best_sc:.10f}")

                if time.monotonic() < deadline - 70.0:
                    a2, sc2 = _optimize_adam_z(a2, min(deadline - 50.0, time.monotonic() + 70.0), rng, lr=0.008)
                    if np.isfinite(sc2) and sc2 < best_sc:
                        best_a = a2.copy()
                        best_sc = float(sc2)
                        print(f"adam n2: n={best_a.size}, score={best_sc:.10f}")

    # Adam polish on the current global best.
    if time.monotonic() < deadline - 100.0:
        if best_a.size <= 4096:
            adam_budget = 260.0
            lr = 0.012
        elif best_a.size <= 8192:
            adam_budget = 160.0
            lr = 0.008
        else:
            adam_budget = 80.0
            lr = 0.005
        phase_end = min(deadline - 70.0, time.monotonic() + adam_budget)
        a2, sc2 = _optimize_adam_z(best_a, phase_end, rng, lr=lr)
        if np.isfinite(sc2) and sc2 < best_sc:
            best_a = a2.copy()
            best_sc = float(sc2)
            print(f"adam global: n={best_a.size}, score={best_sc:.10f}")

    # Mirror descent polish.
    if time.monotonic() < deadline - 40.0:
        mirror_budget = 320.0 if best_a.size <= 4096 else 220.0
        phase_end = min(deadline - 25.0, time.monotonic() + mirror_budget)
        a2, sc2 = _optimize_mirror(best_a, phase_end, rng)
        if np.isfinite(sc2) and sc2 < best_sc:
            best_a = a2.copy()
            best_sc = float(sc2)
            print(f"mirror: n={best_a.size}, score={best_sc:.10f}")

    # Short projected-gradient attempt.
    if time.monotonic() < deadline - 25.0 and best_a.size <= 8192:
        pgd_budget = 70.0 if best_a.size <= 4096 else 45.0
        phase_end = min(deadline - 15.0, time.monotonic() + pgd_budget)
        a2, sc2 = _optimize_pgd(best_a, phase_end, rng)
        if np.isfinite(sc2) and sc2 < best_sc:
            best_a = a2.copy()
            best_sc = float(sc2)
            print(f"pgd: n={best_a.size}, score={best_sc:.10f}")

    # Final random block polish.
    if time.monotonic() < deadline - 8.0:
        a2, sc2 = _random_block_search(best_a, deadline - 4.0, rng)
        if np.isfinite(sc2) and sc2 < best_sc:
            best_a = a2.copy()
            best_sc = float(sc2)
            print(f"block polish: n={best_a.size}, score={best_sc:.10f}")

    # Final symmetrization check.
    sym = _symmetrize(best_a)
    sc = _score_norm(sym)
    if np.isfinite(sc) and sc < best_sc:
        best_a = sym.copy()
        best_sc = float(sc)
        print(f"final sym: n={best_a.size}, score={best_sc:.10f}")

    # Safeguards for return.
    final = np.asarray(best_a, dtype=float).ravel()
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.maximum(final, 0.0)
    s = float(final.sum())

    if s < 0.01:
        final = _make_uniform(4096)
        s = float(final.sum())

    final = final / s

    if final.size > 12000:
        final = _interp_profile(final, 8192)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_uniform(8192)

    final_score = _score_norm(final)
    print(f"FINAL local: n={final.size}, score={final_score:.10f}, sum={float(final.sum()):.6f}")

    # Optional validator sanity check. The returned object is based on our internal score.
    if final.size <= 9000:
        try:
            ev = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(ev):
                print(f"FINAL evaluator: {ev:.10f}")
            else:
                print("FINAL evaluator returned inf; using fallback-safe normalization.")
        except Exception as ex:
            print(f"evaluate_sequence check failed: {ex}")

    return [float(x) for x in final]
```
