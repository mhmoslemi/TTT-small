def _linear_step(p, q_max=128, inc=0.5, alphas=(1.0,0.5,0.25,0.1)):
    p=np.asarray(p,float)
    n=p.size
    if n==0: return None,np.inf
    c=_conv(p); M=float(np.max(c)) if c.size else 0
    if M<=0: return p,np.inf
    threshold=M - max(1e-12, 0.01*M)
    active=np.flatnonzero(c >= threshold)
    if active.size > q_max:
       # top q_max
       part=np.argpartition(c, -q_max)[-q_max:]
       active=part
    if active.size ==0:
       active=np.array([int(np.argmax(c))])
    B=np.zeros((active.size,n), dtype=np.float64)
    for r,k in enumerate(active):
       i0=max(0,k-n+1); i1=min(n-1,k)
       if i0<=i1:
          idx=np.arange(i0,i1+1)
          B[r,i0:i1+1]=2.0*p[k-idx]
    ca=c[active]
    A_ub=np.empty((active.size,n+1), dtype=np.float64)
    A_ub[:,0]=-1.0
    A_ub[:,1:]=B
    b_ub=-ca
    c_obj=np.zeros(n+1); c_obj[0]=1.0
    A_eq=np.zeros((1,n+1)); A_eq[0,1:]=1.0
    b_eq=np.zeros(1)
    lb=np.full(n+1, -np.inf); ub=np.full(n+1, np.inf)
    lb[0]=0.0
    lb[1:]=-0.5*p
    ub[1:]=p*(1.0+inc)
    # ensure numeric bounds
    bounds=(lb,ub)
    try:
       res=linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    except Exception:
       return None, np.inf
    if not res.success or res.x is None: return None, np.inf
    delta=res.x[1:]
    if not np.all(np.isfinite(delta)): return None,np.inf
    best_p=p.copy(); best_sc=_score_normalized(p)
    for alpha in alphas:
       p_new=p + alpha*delta
       p_new=np.maximum(p_new,0)
       p_new=_normalize_nonneg(p_new)
       if p_new is None: continue

          best_sc=sc_new; best_p=p_new.copy()
    if best_p is not None and best_sc < _score_normalized(p) - 1e-12:
       return best_p,best_sc
    return None,best_sc

```python
"""
Multi-stage optimizer for non-negative step-function coefficients.

Strategy:
- Normalize to sum 1; score = 2 * n * max(linear self-convolution).
- Improve a few smooth profiles (beta powers, low-dimensional cosine/log-softmax
  basis, full logit Adam, multiplicative peak-flooding, and local active-set
  linearized LP steps).
- Keep a pool of candidates and finalize with direct convolution validation.
"""
import time
import numpy as np
from scipy.optimize import linprog


def _next_pow2(n):
    n = int(max(1, n))
    k = 1
    while k < n:
        k <<= 1
    return k


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    a = np.where(np.isfinite(a), a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return None
    return a / s


def _score_normalized(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return np.inf
    c = _conv(p)
    if c.size == 0:
        return np.inf
    m = float(np.max(c))
    if not np.isfinite(m) or m < 0.0:
        return np.inf
    return float(2.0 * n * m)


def _score_any(p):
    p = _normalize_nonneg(p)
    if p is None:
        return np.inf
    return _score_normalized(p)


def _conv(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fp = np.fft.rfft(p, N)
    c = np.fft.irfft(fp * fp, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _softmax_from_z(z, floor=1e-15):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.nan_to_num(z)
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(e) / float(e.size)
    return p / s


def _interp_profile(p0, n_new):
    p0 = _normalize_nonneg(p0)
    if p0 is None:
        return None
    n0 = p0.size
    if n0 == n_new:
        return p0
    if n0 <= 1 or n_new <= 0:
        return None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def _power_profile(n, p, q):
    n = int(max(1, n))
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    # x is in (0,1), so logs are safe.
    logf = float(p) * np.log(x) + float(q) * np.log1p(-x)
    logf = np.nan_to_num(logf)
    logf -= float(np.max(logf))
    f = np.exp(logf)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    return f / s


def _power_search(deadline, N=4096, n_coarse=31, n_fine=15, p_max=4.0):
    best_prof = None
    best_sc = np.inf
    best_pv = 0.0
    best_qv = 0.0

    vals = np.linspace(0.0, p_max, n_coarse)
    i = 0
    while i < n_coarse:
        pv = float(vals[i])
        j = 0
        while j < n_coarse:
            qv = float(vals[j])
            if time.monotonic() > deadline:
                break
            prof = _power_profile(N, pv, qv)
            sc = _score_normalized(prof)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_prof = prof.copy()
                best_pv = pv
                best_qv = qv
            j += 1
        i += 1
        if time.monotonic() > deadline:
            break

    if best_prof is not None and time.monotonic() < deadline - 5:
        step = p_max / float(max(1, n_coarse - 1))
        for _round in range(3):
            if time.monotonic() > deadline:
                break
            lo_p = max(0.0, best_pv - 2.0 * step)
            hi_p = min(max(p_max, 5.0), best_pv + 2.0 * step)
            lo_q = max(0.0, best_qv - 2.0 * step)
            hi_q = min(max(p_max, 5.0), best_qv + 2.0 * step)
            gp = np.linspace(lo_p, hi_p, n_fine)
            gq = np.linspace(lo_q, hi_q, n_fine)
            ip = 0
            while ip < n_fine:
                pv = float(gp[ip])
                iq = 0
                while iq < n_fine:
                    qv = float(gq[iq])
                    if time.monotonic() > deadline:
                        break
                    prof = _power_profile(N, pv, qv)
                    sc = _score_normalized(prof)
                    if np.isfinite(sc) and sc < best_sc:
                        best_sc = sc
                        best_prof = prof.copy()
                        best_pv = pv
                        best_qv = qv
                    iq += 1
                ip += 1
                if time.monotonic() > deadline:
                    break

    return best_prof, best_sc


def _weights(c, beta_scale=0.01):
    c = np.asarray(c, dtype=np.float64).ravel()
    K = c.size
    if K == 0:
        return np.array([], dtype=np.float64)
    M = float(np.max(c)) if K else 0.0
    if not np.isfinite(M) or M <= 0.0:
        return np.ones(K, dtype=np.float64) / float(K)
    beta = max(1e-30, float(beta_scale) * M)
    x = (c - M) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros(K, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s


def _heat_sparse(w, a, thresh=1e-12):
    n = a.size
    h = np.zeros(n, dtype=np.float64)
    K = w.size
    if K == 0:
        return h
    idx = np.flatnonzero(w > thresh)
    if idx.size == 0:
        return h
    for k in idx:
        val = float(w[k])
        if val <= 0.0:
            continue
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n - 1, k)
        if i0 <= i1:
            idx2 = np.arange(i0, i1 + 1)
            h[i0:i1 + 1] += val * a[k - idx2]
    return h


def _heat_fft(w, a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    K = 2 * n - 1
    N = _next_pow2(n + K - 1)
    w_rev = w[::-1].copy()
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(fa * fw, N)
    h = conv[n - 1:2 * n - 1][::-1]
    h = np.maximum(h, 0.0)
    return h


def _heat(w, a, max_sparse=96):
    w = np.asarray(w, dtype=np.float64).ravel()
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    if w.size == 0:
        return np.zeros(a.size, dtype=np.float64)
    idx = np.flatnonzero(w > 1e-12)
    if idx.size <= max_sparse:
        return _heat_sparse(w, a)
    return _heat_fft(w, a)


def _adam_logit(p0, deadline, rng, lr=0.05, beta_scale=0.01, beta_decay=0.98,
                no_imp_limit=250, max_sparse=96, noise_scales=None):
    p = _normalize_nonneg(p0)
    if p is None:
        return None, np.inf
    if rng is None:
        rng = np.random.default_rng(12345)
    if noise_scales is None:
        noise_scales = [0.03, 0.1, 0.3, 1.0]

    n = p.size
    z = np.log(np.maximum(p, 1e-15))
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m_adam = np.zeros(n, dtype=np.float64)
    v_adam = np.zeros(n, dtype=np.float64)
    t = 0

    best_p = p.copy()
    best_z = z.copy()
    best_score = _score_normalized(p)
    if not np.isfinite(best_score):
        best_score = np.inf

    no_imp = 0
    base_lr = lr
    cur_lr = lr
    beta = beta_scale
    noise_idx = 0

    while time.monotonic() < deadline:
        c = _conv(p)
        M = float(np.max(c)) if c.size else 0.0
        score = float(2.0 * n * M)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            p = best_p.copy()
            z = best_z.copy()
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            t = 0
            no_imp += 1

        if no_imp > no_imp_limit:
            ns = float(noise_scales[noise_idx % len(noise_scales)])
            noise_idx += 1
            z = best_z + ns * rng.normal(size=n)
            z = np.nan_to_num(z)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            t = 0
            no_imp = 0
            cur_lr = base_lr
            beta = beta_scale
            continue

        w = _weights(c, beta)
        h = _heat(w, p, max_sparse)
        if not np.all(np.isfinite(h)):
            p = best_p.copy()
            z = best_z.copy()
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            t = 0
            no_imp += 1
            continue

        vp = h
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))
        if not np.all(np.isfinite(gz)):
            p = best_p.copy()
            z = best_z.copy()
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            t = 0
            no_imp += 1
            continue

        t += 1
        m_adam = b1 * m_adam + (1.0 - b1) * gz
        v_adam = b2 * v_adam + (1.0 - b2) * (gz * gz)
        mh = m_adam / (1.0 - b1 ** t)
        vh = v_adam / (1.0 - b2 ** t)
        z = z - cur_lr * mh / (np.sqrt(vh) + eps)
        z = np.nan_to_num(z)
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)

        if t % 30 == 0:
            beta = max(0.002, beta * beta_decay)
            if no_imp > no_imp_limit // 3:
                cur_lr = max(0.005, cur_lr * 0.95)

    return best_p, best_score


def _peak_flood(p0, deadline, eta=0.05, eta_min=0.005, eta_max=0.30,
                beta_scale=0.01, beta_decay=0.98, no_imp_limit=300, max_sparse=96):
    p = _normalize_nonneg(p0)
    if p is None:
        return None, np.inf
    n = p.size
    best_p = p.copy()
    best_score = _score_normalized(p)
    if not np.isfinite(best_score):
        best_score = np.inf

    no_imp = 0
    cur_eta = eta
    beta = beta_scale
    it = 0

    while time.monotonic() < deadline:
        c = _conv(p)
        M = float(np.max(c)) if c.size else 0.0
        score = float(2.0 * n * M)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            p = best_p.copy()
            no_imp += 1

        if no_imp > no_imp_limit:
            p = best_p.copy()
            cur_eta = eta
            beta = beta_scale
            no_imp = 0
            continue

        w = _weights(c, beta)
        h = _heat(w, p, max_sparse)
        hmax = float(np.max(h)) if h.size else 0.0
        if not np.isfinite(hmax) or hmax <= 1e-30:
            break

        factor = 1.0 - cur_eta * (h / hmax)
        factor = np.clip(factor, 0.05, 1.0)
        p2 = p * factor
        p2 = _normalize_nonneg(p2)
        if p2 is None:
            p = best_p.copy()
            no_imp += 1
            continue

        sc2 = _score_normalized(p2)
        if np.isfinite(sc2) and sc2 < best_score - 1e-14:
            best_score = sc2
            best_p = p2.copy()
            no_imp = 0
            cur_eta = min(eta_max, cur_eta * 1.05)
        else:
            no_imp += 1
            cur_eta = max(eta_min, cur_eta * 0.90)

        p = p2
        it += 1
        if it % 30 == 0:
            beta = max(0.002, beta * beta_decay)

    return best_p, best_score


def _cosine_phi(N, m):
    N = int(N)
    m = int(m)
    if m <= 0:
        m = 1
    m = min(m, max(1, N - 1))
    i = np.arange(N, dtype=np.float64)
    x = (i + 0.5) / float(N)
    J = np.arange(1, m + 1, dtype=np.float64)[:, None]
    phi = np.cos(np.pi * J * x[None, :])
    phi *= np.sqrt(2.0 / float(N))
    return phi


def _basis_adam(p0, N, m, deadline, rng=None, lr=0.10, beta_scale=0.01,
                beta_decay=0.985, reg=0.0, no_imp_limit=220, max_sparse=96):
    pN = _interp_profile(p0, int(N))
    N = int(N)
    if pN is None:
        pN = np.ones(N, dtype=np.float64) / float(N)

    m = int(m)
    if m <= 0 or m >= N:
        m = max(4, N // 32)

    phi = _cosine_phi(N, m)
    logp = np.log(np.maximum(pN, 1e-15))
    logp -= float(np.mean(logp))
    A = phi.T @ logp
    A = np.nan_to_num(A)
    A = np.clip(A, -20.0, 20.0)

    if rng is None:
        rng = np.random.default_rng(54321)

    m_adam = np.zeros(m, dtype=np.float64)
    v_adam = np.zeros(m, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    P = phi @ A
    P -= float(np.mean(P))
    p = _softmax_from_z(P)

    best_p = p.copy()
    best_A = A.copy()
    best_score = _score_normalized(p)
    if not np.isfinite(best_score):
        best_score = np.inf

    no_imp = 0
    base_lr = lr
    cur_lr = lr
    beta = beta_scale
    noise_scales = [0.10, 0.30, 0.80, 1.60]
    noise_idx = 0

    while time.monotonic() < deadline:
        c = _conv(p)
        M = float(np.max(c)) if c.size else 0.0
        score = float(2.0 * N * M)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_A = A.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            A = best_A.copy()
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            t = 0
            no_imp += 1

        if no_imp > no_imp_limit:
            ns = float(noise_scales[noise_idx % len(noise_scales)])
            noise_idx += 1
            A = best_A + ns * rng.normal(size=m)
            A = np.nan_to_num(A)
            A = np.clip(A, -20.0, 20.0)
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            t = 0
            no_imp = 0
            cur_lr = base_lr
            beta = beta_scale
            continue

        w = _weights(c, beta)
        h = _heat(w, p, max_sparse)
        if not np.all(np.isfinite(h)):
            A = best_A.copy()
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            t = 0
            no_imp += 1
            continue

        vp = h
        dot = float(np.dot(p, vp))
        fp_p = phi.T @ p
        fp_vp = phi.T @ (vp * p)
        g = fp_vp - dot * fp_p
        if reg > 0.0:
            g = g + reg * A

        if not np.all(np.isfinite(g)):
            A = best_A.copy()
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            t = 0
            no_imp += 1
            continue

        t += 1
        m_adam = b1 * m_adam + (1.0 - b1) * g
        v_adam = b2 * v_adam + (1.0 - b2) * (g * g)
        mh = m_adam / (1.0 - b1 ** t)
        vh = v_adam / (1.0 - b2 ** t)
        A = A - cur_lr * mh / (np.sqrt(vh) + eps)
        A = np.nan_to_num(A)
        A = np.clip(A, -20.0, 20.0)

        P = phi @ A
        P -= float(np.mean(P))
        p = _softmax_from_z(P)

        if t % 30 == 0:
            beta = max(0.002, beta * beta_decay)
            if no_imp > no_imp_limit // 3:
                cur_lr = max(0.005, cur_lr * 0.95)

    return best_p, best_score


def _linear_step(p, q_max=128, inc=0.5, alphas=(1.0, 0.5, 0.25, 0.10)):
    p = _normalize_nonneg(p)
    if p is None:
        return None, np.inf
    n = p.size
    if n == 0:
        return None, np.inf
    c = _conv(p)
    M = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(M) or M <= 0.0:
        return None, np.inf

    threshold = M - max(1e-12, 0.01 * M)
    active = np.flatnonzero(c >= threshold)
    if active.size > q_max:
        active = np.argpartition(c, -q_max)[-q_max:]
    if active.size == 0:
        active = np.array([int(np.argmax(c))], dtype=int)

    B = np.zeros((active.size, n), dtype=np.float64)
    for r, k in enumerate(active):
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n - 1, k)
        if i0 <= i1:
            idx = np.arange(i0, i1 + 1)
            B[r, i0:i1 + 1] = 2.0 * p[k - idx]

    ca = c[active]
    A_ub = np.empty((active.size, n + 1), dtype=np.float64)
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = B
    b_ub = -ca

    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[0] = 1.0

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1, dtype=np.float64)

    lb = np.full(n + 1, -np.inf, dtype=np.float64)
    ub = np.full(n + 1, np.inf, dtype=np.float64)
    lb[0] = 0.0
    lb[1:] = -0.5 * p
    ub[1:] = p * (1.0 + inc)
    bounds = (lb, ub)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    except Exception:
        return None, np.inf

    if not res.success or res.x is None:
        return None, np.inf

    delta = np.asarray(res.x[1:], dtype=np.float64)
    if not np.all(np.isfinite(delta)):
        return None, np.inf

    base_sc = _score_normalized(p)
    best_p = p
    best_sc = base_sc
    for alpha in alphas:
        alpha = float(alpha)
        p_new = p + alpha * delta
        p_new = np.maximum(p_new, 0.0)
        p_new = _normalize_nonneg(p_new)
        if p_new is None:
            continue
        sc_new = _score_normalized(p_new)
        if np.isfinite(sc_new) and sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_p = p_new

    if best_sc < base_sc - 1e-12:
        return best_p, best_sc
    return None, base_sc


def _linear_phase(p0, deadline, q_max=128, inc=0.5, max_steps=200):
    p = _normalize_nonneg(p0)
    if p is None:
        return None, np.inf
    best_p = p.copy()
    best_score = _score_normalized(p)
    if not np.isfinite(best_score):
        best_score = np.inf

    cur_q = min(64, q_max)
    cur_inc = inc
    fail = 0
    it = 0
    while it < max_steps and time.monotonic() < deadline:
        p2, sc2 = _linear_step(p, q_max=cur_q, inc=cur_inc)
        it += 1
        if p2 is not None and np.isfinite(sc2) and sc2 < best_score - 1e-14:
            best_score = sc2
            best_p = p2.copy()
            p = p2.copy()
            fail = 0
            cur_inc = min(1.0, cur_inc * 1.05)
            if it % 12 == 0:
                cur_q = min(q_max, cur_q + 16)
        else:
            fail += 1
            cur_inc = max(0.10, cur_inc * 0.85)
            cur_q = max(32, cur_q - 8)
            if fail > 45:
                break
    return best_p, best_score


def _add_candidate(candidates, seq, label=""):
    p = _normalize_nonneg(seq)
    if p is None:
        return
    sc = _score_normalized(p)
    candidates.append((sc, p.copy(), label))
    print(f"  candidate: {label}, n={p.size}, score={sc:.10f}")


def _best_candidate(candidates):
    if not candidates:
        return None, np.inf
    bi = 0
    bv = candidates[0][0]
    for i in range(1, len(candidates)):
        if candidates[i][0] < bv:
            bv = candidates[i][0]
            bi = i
    return candidates[bi][1].copy(), bv


def _select_top(candidates, k):
    items = list(candidates)
    out = []
    do = 0
    while do < k and items:
        bi = 0
        bv = items[0][0]
        for i in range(1, len(items)):
            if items[i][0] < bv:
                bv = items[i][0]
                bi = i
        out.append(items.pop(bi))
        do += 1
    return out


def _exact_score(p):
    p = _normalize_nonneg(p)
    if p is None:
        return np.inf
    n = p.size
    if n == 0:
        return np.inf
    if n <= 4096:
        c = np.convolve(p, p)
    else:
        c = _conv(p)
        c = np.maximum(c, 0.0)
    if c.size == 0:
        return np.inf
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m)


def _evaluate_candidate(p, label=""):
    p = _normalize_nonneg(p)
    if p is None:
        return np.inf
    esc = _exact_score(p)
    try:
        ev = float(evaluate_sequence([float(x) for x in p]))
        if np.isfinite(ev) and ev < esc:
            esc = ev
    except Exception:
        pass
    print(f"  evaluate: {label}, n={p.size}, exact={esc:.10f}")
    return esc


def _finalize(candidates):
    if not candidates:
        return _power_profile(1024, 0.8, 0.0)

    selected = _select_top(candidates, 12)
    # Also evaluate the first few candidates so the original parent is not missed
    # if FFT-based ranking is slightly optimistic.
    extra = []
    if len(candidates) > 0:
        extra.append(candidates[0])
    if len(candidates) > 1:
        extra.append(candidates[1])

    best_p = None
    best_esc = np.inf
    best_label = ""

    for item in selected:
        sc = item[0]
        p = item[1]
        label = item[2]
        esc = _evaluate_candidate(p, label)
        if np.isfinite(esc) and esc < best_esc:
            best_esc = esc
            best_p = p.copy()
            best_label = label

    for item in extra:
        sc = item[0]
        p = item[1]
        label = item[2] + "_extra"
        esc = _evaluate_candidate(p, label)
        if np.isfinite(esc) and esc < best_esc:
            best_esc = esc
            best_p = p.copy()
            best_label = label

    if best_p is None:
        best_p = _power_profile(1024, 0.8, 0.0)
        best_esc = _score_any(best_p)

    print(f"FINAL chosen: {best_label}, n={best_p.size}, score={best_esc:.10f}")
    return best_p


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Return the best non-negative coefficient list found within budget_s."""
    t0 = time.monotonic()
    deadline = t0 + max(60.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    candidates = []

    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = _normalize_nonneg(arr)
    except Exception:
        parent = None

    if parent is None:
        parent = _power_profile(1024, 0.8, 0.0)

    _add_candidate(candidates, parent, "parent")

    # 1) Broad smooth profile search with generalized beta powers.
    end_power = min(deadline, t0 + 70.0)
    if time.monotonic() < end_power - 5.0:
        pp, ps = _power_search(end_power, N=4096, n_coarse=31, n_fine=13, p_max=4.0)
        if pp is not None:
            _add_candidate(candidates, pp, "power")

    best_p, best_sc = _best_candidate(candidates)
    if best_p is not None and best_p.size != 4096:
        p4096 = _interp_profile(best_p, 4096)
        if p4096 is not None:
            _add_candidate(candidates, p4096, "interp_4096")
        best_p, best_sc = _best_candidate(candidates)

    # 2) Low-dimensional log-cosine basis optimization at n=4096.
    end_b64 = min(deadline, t0 + 260.0)
    if best_p is not None and time.monotonic() < end_b64 - 20.0:
        pb, sb = _basis_adam(best_p, 4096, 64, end_b64, rng=rng,
                             lr=0.10, beta_scale=0.012, beta_decay=0.985,
                             no_imp_limit=220)
        if pb is not None:
            _add_candidate(candidates, pb, "basis_64")
        best_p, best_sc = _best_candidate(candidates)

    end_b128 = min(deadline, t0 + 400.0)
    if best_p is not None and time.monotonic() < end_b128 - 20.0:
        pb, sb = _basis_adam(best_p, 4096, 128, end_b128, rng=rng,
                             lr=0.08, beta_scale=0.010, beta_decay=0.985,
                             no_imp_limit=180)
        if pb is not None:
            _add_candidate(candidates, pb, "basis_128")
        best_p, best_sc = _best_candidate(candidates)

    # 3) Full-dimensional logit Adam at n=4096.
    end_a4096 = min(deadline, t0 + 1000.0)
    if best_p is not None and time.monotonic() < end_a4096 - 30.0:
        pa, sa = _adam_logit(best_p, end_a4096, rng,
                             lr=0.04, beta_scale=0.010, beta_decay=0.985,
                             no_imp_limit=300, max_sparse=96)
        if pa is not None:
            _add_candidate(candidates, pa, "adam_4096")
        best_p, best_sc = _best_candidate(candidates)

    # 4) Multiplicative peak flooding at n=4096.
    end_f4096 = min(deadline, t0 + 1150.0)
    if best_p is not None and time.monotonic() < end_f4096 - 20.0:
        pf, sf = _peak_flood(best_p, end_f4096, rng=rng,
                             eta=0.05, beta_scale=0.010, beta_decay=0.98,
                             no_imp_limit=260)
        if pf is not None:
            _add_candidate(candidates, pf, "flood_4096")
        best_p, best_sc = _best_candidate(candidates)

    # 5) Medium scale n=2048, sometimes wins due to better discrete local optima.
    end_a2048 = min(deadline, t0 + 1320.0)
    if best_p is not None and time.monotonic() < end_a2048 - 30.0:
        p2048 = _interp_profile(best_p, 2048)
        if p2048 is not None:
            _add_candidate(candidates, p2048, "interp_2048")
            pa2, sa2 = _adam_logit(p2048, end_a2048, rng,
                                   lr=0.05, beta_scale=0.012, beta_decay=0.985,
                                   no_imp_limit=250, max_sparse=80)
            if pa2 is not None:
                _add_candidate(candidates, pa2, "adam_2048")
            best_p, best_sc = _best_candidate(candidates)

    # 6) Small scale n=1024: include parent-like start and active-set LP steps.
    end_1024 = min(deadline, t0 + 1620.0)
    p1024 = None
    if parent is not None and parent.size == 1024:
        p1024 = parent.copy()
    elif best_p is not None:
        p1024 = _interp_profile(best_p, 1024)

    if p1024 is not None and time.monotonic() < end_1024 - 40.0:
        _add_candidate(candidates, p1024, "start_1024")
        sub_dead = min(end_1024, time.monotonic() + 80.0)
        p10, s10 = _adam_logit(p1024, sub_dead, rng,
                               lr=0.06, beta_scale=0.012, beta_decay=0.985,
                               no_imp_limit=200, max_sparse=80)
        if p10 is not None:
            _add_candidate(candidates, p10, "adam_1024")
            p1024 = p10

        if time.monotonic() < end_1024 - 20.0:
            p_l, s_l = _linear_phase(p1024, end_1024, q_max=128, inc=0.45,
                                     max_steps=120)
            if p_l is not None:
                _add_candidate(candidates, p_l, "linear_1024")

    best_p, best_sc = _best_candidate(candidates)

    # 7) Final polish on the current best candidate.
    if best_p is not None and time.monotonic() < deadline - 20.0:
        rem = deadline - time.monotonic()
        if rem > 40.0:
            dur = rem * 0.55
            pfin, sfin = _adam_logit(best_p, time.monotonic() + dur, rng,
                                     lr=0.02, beta_scale=0.008, beta_decay=0.99,
                                     no_imp_limit=140, max_sparse=80)
            if pfin is not None:
                _add_candidate(candidates, pfin, "final_adam")
            best_p, best_sc = _best_candidate(candidates)

        if best_p is not None and time.monotonic() < deadline - 15.0:
            if best_p.size <= 1536:
                p_l, s_l = _linear_phase(best_p, min(deadline, time.monotonic() + 45.0),
                                         q_max=128, inc=0.35, max_steps=35)
                if p_l is not None:
                    _add_candidate(candidates, p_l, "final_linear")
            else:
                p_f, s_f = _peak_flood(best_p, min(deadline, time.monotonic() + 35.0),
                                       eta=0.025, beta_scale=0.008, beta_decay=0.99,
                                       no_imp_limit=90)
                if p_f is not None:
                    _add_candidate(candidates, p_f, "final_flood")

    final = _finalize(candidates)
    final = _normalize_nonneg(final)
    if final is None:
        final = _power_profile(1024, 0.8, 0.0)

    return [float(x) for x in final]
```
