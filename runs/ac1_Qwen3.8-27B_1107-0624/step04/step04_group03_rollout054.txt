3. Nesterov-accelerated gradient descent in log-space

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _project_simplex(v, target_sum=1.0):
    """Project v onto {a >= 0, sum(a) = target_sum}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    # We want sum(a) = target_sum, a >= 0
    # Equivalent to: find tau such that sum(max(v - tau, 0)) = target_sum
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    a = np.maximum(v - theta, 0.0)
    # Fix any numerical issues
    s = a.sum()
    if s > 0:
        a = a * target_sum / s
    else:
        a = np.ones(n) / n * target_sum
    return a


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _gradient_smooth(z, beta):
    """Gradient of smooth objective w.r.t. log-variables z (softmax parameterization)."""
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _objective_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_obj(z, beta, n):
    return _objective_smooth(z, beta)


def _lbfgs_grad(z, beta, n):
    return _gradient_smooth(z, beta)


def _gradient_direct(a, beta):
    """Gradient of smooth objective w.r.t. a (before projection)."""
    n = len(a)
    s = float(a.sum())
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    # df/da[i] = 4n/S^2 * h[i] - 4n*m_soft/S^3 * a[i] ... but let's use the simpler form

    # But we also need to account for the constraint, so we just use the unconstrained gradient

    m_soft = cm + beta * float(np.log(np.sum(np.exp((c - cm) / beta)))) if beta > 0 else cm
    grad = 4.0 * n * h - 4.0 * n * W
    return grad

                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new
        except Exception:
            pass
        # Update max_c for next beta
        c_curr = _full_convolve(best_a, best_a)
        max_c0 = float(c_curr.max())
        # Rescale remaining betas
        if idx < len(betas) - 1:
            betas[idx+1:] = [b / betas[idx] * b * max_c0 for b in betas[idx+1:]]

    return best_a, best_score


def _nesterov_optimize(a_init, deadline, beta_fracs, lr_init=0.02, mom_init=0.9):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())

    lr = lr_init
    mom = mom_init
    t = 0
    stall_count = 0
    stall_threshold = score_at_stall = best_score

    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-14, beta_frac * max_c)

        while time.monotonic() < deadline:
            t += 1
            grad_y = _gradient_smooth(y, beta)
            if not np.all(np.isfinite(grad_y)):
                grad_y = _gradient_smooth(z, beta)

            z_new = z - lr * grad_y
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -60, 60)

            y_new = z_new + mom * (z_new - z)
            y_new = y_new - y_new.mean()

            z = z_new
            y = y_new

            if t % 30 == 0:
                a_curr = _softmax_fn(z)
                sc = _score_norm_fn(a_curr)
                if np.isfinite(sc):
                    if sc < best_score:
                        best_score = sc
                        best_a = a_curr.copy()
                        stall_count = 0
                        score_at_stall = sc
                    elif sc < score_at_stall * 1.001:
                        stall_count += 1
                    else:
                        stall_count += 1
                        # Revert and reduce lr
                        if stall_count > 5:
                            lr *= 0.7
                            stall_count = 0

            # Adaptive learning rate
            if t % 50 == 0:
                lr = max(0.001, lr * 0.98)
            if t % 200 == 0:
                mom = min(0.9995, mom + 0.0003)

        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        max_c = float(c.max())
        lr = max(0.001, lr * 0.6)

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _direct_simplex_optimize(a_init, deadline, beta_fracs, lr_init=0.001):
    """Optimize in direct space with simplex projection."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()  # normalize to sum=1

    best_a = a.copy()
    best_score = _score_norm_fn(a)

    c = _full_convolve(a, a)
    max_c = float(c.max())

    lr = lr_init

    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-14, beta_frac * max_c)

        t = 0
        while time.monotonic() < deadline:
            t += 1
            a = a / a.sum()  # ensure normalization
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            w = e / e.sum()
            W = float(np.dot(w, c))

            a_rev = a[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]

            # Gradient (unconstrained, S=1)
            grad = 4.0 * n * (h - W)

            # Step and project
            a = a - lr * grad
            a = _project_simplex(a, 1.0)

            if t % 20 == 0:
                sc = _score_norm_fn(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                # Adaptive step
                c = _full_convolve(a, a)
                max_c = float(c.max())
                lr = max(1e-6, lr * 0.995)

        c = _full_convolve(a, a)
        max_c = float(c.max())
        lr = max(1e-6, lr * 0.5)

    a_final = _project_simplex(a, 1.0)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _exact_subgradient_polish(a_init, deadline, n_iter=50000):
    """Polish using exact subgradient (active set = top lags)."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm_fn(a)

    lr_num = 0.1  # Polyak numerator (gap)
    mom = np.zeros(n)
    mom_coeff = 0.7

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        a = a / a.sum()
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Active set: lags within 0.5% of max
        threshold = cm * (1.0 - 0.005)
        active = c >= threshold
        if not np.any(active):
            active[int(np.argmax(c))] = True
        n_active = np.sum(active)

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active

        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]

        # Exact subgradient (S=1): 4n * a * (h - W) where W = max(c)
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        # Polyak step size
        grad_norm_sq = float(np.dot(grad, grad))
        if grad_norm_sq > 1e-30:
            lr = lr_num / grad_norm_sq
        else:
            lr = 0.0

        # Update with momentum
        mom = mom_coeff * mom - lr * grad
        a = a + mom
        a = _project_simplex(a, 1.0)

        if it % 100 == 0:
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                # Update Polyak numerator
                lr_num = max(0.01, (best_score - best_score * 0.99))
            else:
                lr_num *= 0.8

    a_final = _project_simplex(a, 1.0)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_exp_profile(n, rate):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(rate * (t - 0.5))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_sine_profile(n, freq, amp):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS continuation, Nesterov gradient,
    direct-space simplex projection, and exact subgradient polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_base = len(parent)
        print(f"Parent: n={n_base}, score={_score_fn(parent):.10f}")
    else:
        n_base = 4096
        parent = _make_quadratic_profile(n_base, 0.3, 0.5)
        print(f"No parent, using quadratic profile n={n_base}")

    # --- Phase 0: Evaluate diverse starting points (30s) ---
    phase0_end = min(deadline, t_start + 30.0)
    candidates = []

    sc_parent = _score_norm_fn(parent)
    candidates.append((sc_parent, parent.copy(), len(parent), "parent"))
    update_best(parent, sc_parent, "parent")

    n_try = n_base  # Use same length as parent
    profiles = [
        _make_quadratic_profile(n_try, 0.5, 0.3),
        _make_quadratic_profile(n_try, -0.3, 0.8),
        _make_quadratic_profile(n_try, 0.2, 0.6),
        _make_linear_profile(n_try, 1.0),
        _make_linear_profile(n_try, 2.0),
        _make_cubic_profile(n_try, 0.5, -0.3),
        _make_exp_profile(n_try, 2.0),
        _make_exp_profile(n_try, -2.0),
        _make_sine_profile(n_try, 1, 0.3),
        _make_sine_profile(n_try, 2, 0.2),
    ]
    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p, n_try, f"prof{idx}"))

    # Also try interpolating parent to n=8192
    if n_base == 4096 and time.monotonic() < phase0_end - 2:
        p_8192 = _interp_profile(parent, 8192)
        sc_8192 = _score_norm_fn(p_8192)
        candidates.append((sc_8192, p_8192, 8192, "parent_8192"))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: {len(candidates)} candidates, best={best_score:.10f}, n={len(best_seq)}")

    # --- Phase 1: L-BFGS continuation on best candidate (350s) ---
    phase1_end = min(deadline, t_start + 380.0)
    if best_seq is not None and time.monotonic() < phase1_end:
        per_time = min(300.0, phase1_end - time.monotonic() - 10)
        if per_time > 20:
            p_best, sc_best = _run_lbfgs(best_seq, min(phase1_end - 5, time.monotonic() + per_time),
                                          max_wall_time=per_time)
            update_best(p_best, sc_best, "lbfgs_phase1")

    # Try second-best if time allows
    if time.monotonic() < phase1_end - 60 and len(candidates) > 1:
        for sc, p, pn, label in candidates[1:4]:
            if time.monotonic() > phase1_end - 40:
                break
            if len(p) != len(best_seq):
                continue
            per_time = min(40.0, phase1_end - time.monotonic() - 10)
            if per_time > 10:
                p_best, sc_best = _run_lbfgs(p, min(phase1_end - 5, time.monotonic() + per_time),
                                              max_wall_time=per_time)
                update_best(p_best, sc_best, f"lbfgs_{label}")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Nesterov refinement (300s) ---
    phase2_end = min(deadline, t_start + 680.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        beta_fracs = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                      5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]
        per_time = min(280.0, phase2_end - time.monotonic() - 10)
        if per_time > 30:
            p_best, sc_best = _nesterov_optimize(
                best_seq, min(phase2_end - 5, time.monotonic() + per_time),
                beta_fracs, lr_init=0.015, mom_init=0.9)
            update_best(p_best, sc_best, "nesterov_phase2")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Direct space simplex optimization (250s) ---
    phase3_end = min(deadline, t_start + 930.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        per_time = min(230.0, phase3_end - time.monotonic() - 10)
        if per_time > 30:
            beta_fracs_direct = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
            p_best, sc_best = _direct_simplex_optimize(
                best_seq, min(phase3_end - 5, time.monotonic() + per_time),
                beta_fracs_direct, lr_init=0.002)
            update_best(p_best, sc_best, "simplex_phase3")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Exact subgradient polishing (200s) ---
    phase4_end = min(deadline, t_start + 1130.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        per_time = min(180.0, phase4_end - time.monotonic() - 10)
        if per_time > 20:
            p_best, sc_best = _exact_subgradient_polish(
                best_seq, min(phase4_end - 5, time.monotonic() + per_time))
            update_best(p_best, sc_best, "subgrad_phase4")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Restarts with noise (200s) ---
    phase5_end = min(deadline, t_start + 1330.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        n_best = len(best_seq)
        n_restarts = 4
        per_restart = (phase5_end - time.monotonic() - 20) / max(1, n_restarts)

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 15:
                break
            noise_level = 0.03 + 0.02 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_level * rng.normal(size=n_best)
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            per_dl = min(phase5_end - 10, time.monotonic() + per_restart)
            beta_fracs_short = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]
            p_best, sc_best = _nesterov_optimize(p_start, per_dl, beta_fracs_short,
                                                  lr_init=0.01, mom_init=0.85)
            update_best(p_best, sc_best, f"restart{i}")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Try n=8192 (150s) ---
    phase6_end = min(deadline, t_start + 1480.0)
    if best_seq is not None and time.monotonic() < phase6_end and len(best_seq) <= 4096:
        n_curr = len(best_seq)
        target_n = min(8192, n_curr * 2)
        p_interp = _interp_profile(best_seq, target_n)
        sc_interp = _score_norm_fn(p_interp)
        print(f"  Interpolated to n={target_n}: score={sc_interp:.10f}")
        if sc_interp < best_score:
            update_best(p_interp, sc_interp, f"interp_n{target_n}")

        if time.monotonic() < phase6_end - 30:
            per_time = min(100.0, phase6_end - time.monotonic() - 20)
            if per_time > 30:
                p_best, sc_best = _run_lbfgs(p_interp, min(phase6_end - 10, time.monotonic() + per_time),
                                              max_wall_time=per_time)
                update_best(p_best, sc_best, f"lbfgs_n{target_n}")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final fine polish (remaining time) ---
    phase7_end = deadline
    if best_seq is not None and time.monotonic() < phase7_end:
        remaining = phase7_end - time.monotonic()
        if remaining > 60:
            # Nesterov with tiny beta
            per_time = min(80.0, remaining - 40)
            if per_time > 20:
                beta_fracs_tiny = [2e-6, 1e-6, 5e-7, 2e-7, 1e-7]
                p_best, sc_best = _nesterov_optimize(
                    best_seq, min(phase7_end - 40, time.monotonic() + per_time),
                    beta_fracs_tiny, lr_init=0.003, mom_init=0.97)
                update_best(p_best, sc_best, "final_nesterov")

        if time.monotonic() < phase7_end - 30:
            remaining2 = phase7_end - time.monotonic()
            if remaining2 > 30:
                per_time2 = min(50.0, remaining2 - 20)
                if per_time2 > 15:
                    p_best, sc_best = _exact_subgradient_polish(
                        best_seq, min(phase7_end - 10, time.monotonic() + per_time2))
                    update_best(p_best, sc_best, "final_subgrad")

        if time.monotonic() < phase7_end - 20:
            remaining3 = phase7_end - time.monotonic()
            if remaining3 > 25:
                per_time3 = min(30.0, remaining3 - 15)
                if per_time3 > 10:
                    p_best, sc_best = _direct_simplex_optimize(
                        best_seq, min(phase7_end - 5, time.monotonic() + per_time3),
                        [5e-6, 2e-6, 1e-6], lr_init=0.001)
                    update_best(p_best, sc_best, "final_simplex")

    print(f"Phase 7 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        # Try evaluate_sequence for authoritative score

                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned inf/NaN, using internal score: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_quadratic_profile(4096, 0.3, 0.5)
    print(f"FALLBACK: n=4096, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
