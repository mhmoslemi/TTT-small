"""
Hybrid simplex-projected optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive log-sum-exp smoothing, projected gradient in simplex space,
multi-scale n, and multi-restart diversification.
"""

# --- Top-level helper functions ---

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max()) / (s * s)

def _score_norm(a):
    """Assumes sum(a) is approximately 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())

def _softmax_vec(x, beta):
    """Compute softmax(x/beta) with numerical stability."""
    x_shifted = (x - x.max()) / beta
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        out = np.zeros_like(x)
        out[x.argmax()] = 1.0
        return out
    return e / s

def _grad_smoothed(a, c, w, n):
    """Compute gradient of 2n*sum_k w_k * c_k w.r.t. a (in a-space).
    Result: 4n * h where h_j = sum_k w_k * a_{k-j} * [0<=k-j<n]
    """
    # h_j = corr(a, w_rev)[n-1:2n-1][::-1]
    w_rev = w[::-1]
    corr = _full_convolve(a, w_rev)
    h = corr[n-1:2*n-1][::-1]
    return 4.0 * n * h

def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    # Find the threshold
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

    n = len(a_init)
    a = a_init.copy()
    # Normalize to simplex
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_norm(a)

    v_buf = np.zeros(n)
    t = 0

    current_lr = lr
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            
            c = _full_convolve(a, a)
            cm = float(c.max())

            # Compute gradient

            if not np.all(np.isfinite(g)):

            # Adam update

            v_buf = b2 * v_buf + (1.0 - b2) * (g * g)
            mh = m_buf / (1.0 - b1**t)

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)

        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5

    else:

def _perturb(a, sigma, rng):
    n = len(a)
    noise = sigma * rng.standard_normal(n)
    a_new = a + noise
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def _interp_profile(p, n_new):
    n0 = len(p)

    x0 = np.linspace(0, 1, n0)

    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)

Hybrid simplex-projected optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max()) / (s * s)


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())


def _softmax_weights(c, beta):
    x_shifted = (c - c.max()) / beta
    e = np.exp(x_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        out = np.zeros_like(c)
        out[c.argmax()] = 1.0
        return out
    return e / s


def _grad_smoothed(a, c, w, n):
    w_rev = w[::-1]
    corr = _full_convolve(a, w_rev)
    h = corr[n - 1:2 * n - 1][::-1]
    return 4.0 * n * h


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    k = int(rho[-1])
    theta = (cssv[k] - 1.0) / (k + 1.0)
    return np.maximum(v - theta, 0.0)

    v_buf = np.zeros(n)

        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        if lo <= hi:
            idx = k_star - np.arange(lo, hi + 1)
            g[lo:hi + 1] = 4.0 * n * a[idx]

        lr = base_lr / (1.0 + decay * i)
        a_candidate = a - lr * g
        a = _project_simplex(a_candidate)

        if i % 1000 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = _score_norm(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _multi_lag_polish(a_init, deadline, n_lags=3, base_lr=0.0005):
    """Polish using average gradient of top n_lags convolution values."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)
    i = 0

    while time.monotonic() < deadline:
        i += 1
        c = _full_convolve(a, a)
        top_k = np.argsort(c)[::-1][:n_lags]

        g = np.zeros(n)
        for k_star in top_k:
            lo = max(0, k_star - (n - 1))
            hi = min(n - 1, k_star)
            if lo <= hi:
                idx = k_star - np.arange(lo, hi + 1)
                g[lo:hi + 1] += 4.0 * n * a[idx]
        g /= n_lags

        lr = base_lr / (1.0 + 0.0003 * i)
        a = _project_simplex(a - lr * g)

        if i % 500 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = _score_norm(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bicentric":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.25 * (np.cos(2.0 * np.pi * t) + 0.5 * np.cos(4.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.where(t < 0.3, 0.5, np.where(t > 0.7, 0.5, 1.0))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    a_new = a + sigma * rng.standard_normal(n)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses simplex-projected Adam with log-sum-exp smoothing, Nesterov
    acceleration, multi-restart, exact subgradient polishing, and
    multi-scale n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(parent)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        update_best(parent, sc0, "parent")
    else:
        n_main = 2048
        a0 = _make_profile(n_main, "uniform")
        print(f"No parent, using uniform n={n_main}")
        update_best(a0, _score(a0), "uniform")

    # === Phase 1: Main Adam optimization (0-250s) ===
    phase1_end = min(deadline, t_start + 250.0)
    c_init = _full_convolve(a0, a0)
    max_c0 = float(c_init.max())

    beta_main = [
        0.04 * max_c0, 0.02 * max_c0, 0.01 * max_c0,
        0.005 * max_c0, 0.002 * max_c0, 0.001 * max_c0,
        0.0005 * max_c0
    ]
    p1, _ = _adam_simplex(a0, time.monotonic() + (phase1_end - time.monotonic() - 5),
                          beta_main, max_iter=500000, lr=0.025)
    update_best(p1, _score(p1), "adam_main")

    # Nesterov on the result
    if time.monotonic() < phase1_end - 10:
        c1 = _full_convolve(p1, p1)
        mc1 = float(c1.max())
        beta_nest = [0.01 * mc1, 0.005 * mc1, 0.002 * mc1, 0.001 * mc1]
        p2, _ = _nesterov_simplex(p1, time.monotonic() + (phase1_end - time.monotonic() - 3),
                                  beta_nest, max_iter=300000, lr=0.01)
        update_best(p2, _score(p2), "nesterov")

    print(f"Phase 1 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Multi-restart (250s-700s) ===
    phase2_end = min(deadline, t_start + 700.0)

    if best_seq is not None and time.monotonic() < phase2_end:
        n_restarts = 8
        per_restart = (phase2_end - time.monotonic() - 10) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > phase2_end - 10:
                break

            if i == 0:
                p_start = _perturb(best_seq, 0.04, rng)
            elif i == 1:
                p_start = _perturb(best_seq, 0.1, rng)
            elif i == 2:
                n_p = len(best_seq)
                p_start = best_seq.copy()
                freq = rng.integers(2, max(3, n_p // 6))
                tq = np.arange(n_p)
                p_start = p_start + 0.06 * np.cos(2 * np.pi * freq * tq / n_p)
                p_start = np.maximum(p_start, 0.0)
                p_start = p_start / p_start.sum()
            elif i == 3:
                p_start = 0.6 * _make_profile(n_main, "sinusoidal") + 0.4 * best_seq
                p_start = p_start / p_start.sum()
            elif i == 4:
                p_start = _interp_profile(best_seq, min(4096, 2 * n_main))
            elif i == 5:
                p_start = _perturb(best_seq, 0.02, rng)
            elif i == 6:
                p_start = 0.5 * _make_profile(n_main, "tapered") + 0.5 * best_seq
                p_start = p_start / p_start.sum()
            else:
                p_start = _perturb(best_seq, 0.15, rng)

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = [0.02 * max_cs, 0.01 * max_cs, 0.005 * max_cs,
                  0.002 * max_cs, 0.001 * max_cs, 0.0005 * max_cs]

            p_r, _ = _adam_simplex(p_start, time.monotonic() + per_restart,
                                   bs, max_iter=200000, lr=0.02)
            update_best(p_r, _score(p_r), f"restart{i}")

    print(f"Phase 2 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Fine polishing (700s-1200s) ===
    phase3_end = min(deadline, t_start + 1200.0)

    if best_seq is not None and time.monotonic() < phase3_end:
        c_best = _full_convolve(best_seq, best_seq)
        max_cb = float(c_best.max())

        # Ultra-fine Adam
        beta_fine = [0.001 * max_cb, 0.0005 * max_cb, 0.0002 * max_cb, 0.0001 * max_cb]
        t_alloc = (phase3_end - time.monotonic() - 20) * 0.5
        if t_alloc > 20:
            pf, _ = _adam_simplex(best_seq, time.monotonic() + t_alloc,
                                  beta_fine, max_iter=300000, lr=0.008)
            update_best(pf, _score(pf), "fine_adam")

        # Multi-lag polish
        if time.monotonic() < phase3_end - 15:
            t_alloc2 = phase3_end - time.monotonic() - 10
            pm, _ = _multi_lag_polish(best_seq, time.monotonic() + t_alloc2,
                                      n_lags=2, base_lr=0.0008)
            update_best(pm, _score(pm), "multi_lag")

    print(f"Phase 3 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Exact subgradient polish (1200s-1500s) ===
    phase4_end = min(deadline, t_start + 1500.0)
    if best_seq is not None and time.monotonic() < phase4_end - 5:
        t_alloc = phase4_end - time.monotonic() - 3
        pe, _ = _exact_polish(best_seq, time.monotonic() + t_alloc,
                              base_lr=0.0006, decay=0.001)
        update_best(pe, _score(pe), "exact_subgrad")

    print(f"Phase 4 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Try larger n (1500s-1720s) ===
    phase5_end = min(deadline, t_start + 1720.0)
    if best_seq is not None and time.monotonic() < phase5_end - 30:
        for target_n in [4096, 8192]:
            if time.monotonic() > phase5_end - 30:
                break
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{target_n}")

            remaining = phase5_end - time.monotonic() - 15
            if remaining > 30:
                c_int = _full_convolve(p_interp, p_interp)
                max_ci = float(c_int.max())
                bs_int = [0.01 * max_ci, 0.005 * max_ci, 0.002 * max_ci,
                          0.001 * max_ci, 0.0005 * max_ci]
                p_int, _ = _adam_simplex(p_interp, time.monotonic() + remaining,
                                         bs_int, max_iter=150000, lr=0.012)
                update_best(p_int, _score(p_int), f"opt_n{target_n}")

    print(f"Phase 5 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final ultra-fine (1720s-deadline) ===
    if best_seq is not None and time.monotonic() < deadline - 3:
        c_final = _full_convolve(best_seq, best_seq)
        max_cf = float(c_final.max())
        beta_final = [0.00005 * max_cf, 0.00002 * max_cf, 0.00001 * max_cf]
        p_fin, _ = _adam_simplex(best_seq, deadline - 2,
                                 beta_final, max_iter=100000, lr=0.003)
        update_best(p_fin, _score(p_fin), "final_ultra")

        # One more exact polish
        if time.monotonic() < deadline - 2:
            p_ex, _ = _exact_polish(best_seq, deadline - 1, base_lr=0.0003, decay=0.002)
            update_best(p_ex, _score(p_ex), "final_exact")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s_final = float(final.sum())
        if s_final < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator confirms: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
