```python
from scipy.optimize import linprog

def _lp_step(f, M, n):
    # Maximize sum(g) subject to:
    # g >= 0
    # conv(f, g)[k] <= M for all k
    # 
    # conv(f, g)[k] = sum_j f[j] * g[k-j] for valid j, k-j
    # = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[j] * g[k-j]
    
    # Variables: g[0], ..., g[n-1]
    # Objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Inequality constraints: A_ub @ g <= M (for each k)
    # For lag k: sum_j f[j] * g[k-j] <= M
    # Let i = k-j, so j = k-i
    # sum_{i} f[k-i] * g[i] <= M (for valid indices)
    
    A_ub = np.zeros((2*n-1, n))
    b_ub = np.full(2*n-1, M)
    
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            i = k - j
            # g[i] * f[j] contributes to lag k
            A_ub[k, i] += f[j]
    
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None
```

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * c.max()

"""
LP-guided optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find directions that simultaneously reduce
all active autocorrelation peaks, combined with L-BFGS polishing.
"""
import numpy as np
from scipy.optimize import minimize, linprog
from scipy import sparse
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _gradient_smoothed(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _obj_smoothed(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    return 2.0 * n * (cm + beta * float(np.log(np.sum(np.exp((c - cm) / beta)))))


def _solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M, g >= 0."""
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            i = k - j
            rows.append(k)
            cols.append(i)
            vals.append(f[j])
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                     options={'time_limit': 60, 'presolve': True})
    
    if result.success:
        return result.x
    return None


def _lp_iteration(f, n, max_iters=100, deadline=None, t_init=0.5):
    """Perform LP-based iterations to reduce max convolution."""
    best_f = f.copy()
    best_score = _score_norm_fn(f)
    t = t_init
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        S = f.sum()
        f_norm = f / S  # normalize to sum=1
        c = _full_convolve(f_norm, f_norm)
        M = float(c.max())
        
        # Solve LP
        g = _solve_lp(f_norm, M, n)
        if g is None:
            break
        
        S_g = g.sum()
        if S_g < 1e-15 or not np.isfinite(S_g):
            break
        
        # Rescale g to have same sum as f_norm
        g_r = g * (1.0 / S_g)
        
        # Update
        f_new = (1.0 - t) * f_norm + t * g_r
        f_new = np.maximum(f_new, 0)
        
        sc = _score_norm_fn(f_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            f = f_new
            # Slowly decrease step size
            t = max(0.1, t * 0.98)
        else:

            if t < 0.01:

                f = f / f.sum()
        
        if it % 10 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}, t={t:.4f}, M={M:.8f}")
    
    return best_f, best_score

def _make_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []

    for slope in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    
    # Exponential
    for rate in [1.0, 2.0, 3.0, -1.0, -2.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    
    # Quadratic
    for a1, a2 in [(1.0, 0.5), (2.0, -0.3), (-1.0, 0.5)]:
        a = 1.0 + a1 * u + a2 * u ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    
    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()

    x0 = np.linspace(0.0, 1.0, n0)

    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

"""
LP-guided optimization with L-BFGS polishing for minimizing
2*n*max(conv(a,a))/sum(a)^2.
Key idea: Use linear programming to find directions that simultaneously
reduce all active autocorrelation peaks, combined with continuation-based
L-BFGS for fine polishing.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy import sparse
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _gradient_smoothed(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _obj_smoothed(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    logsum = np.log(np.sum(np.exp((c - cm) / beta)))
    return 2.0 * n * (cm + beta * logsum)


def _build_sparse_conv_matrix(f, n):
    """Build sparse matrix A where (A @ g)[k] = conv(f, g)[k]."""
    total_nnz = 0
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        total_nnz += (j_hi - j_lo + 1)

    rows = np.empty(total_nnz, dtype=np.int32)
    cols = np.empty(total_nnz, dtype=np.int32)
    vals = np.empty(total_nnz, dtype=np.float64)

    idx = 0
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        j_range = np.arange(j_lo, j_hi + 1)
        cnt = len(j_range)
        rows[idx:idx + cnt] = k
        cols[idx:idx + cnt] = k - j_range
        vals[idx:idx + cnt] = f[j_range]
        idx += cnt

    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _solve_lp(f, M, n, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0."""
    A = _build_sparse_conv_matrix(f, n)
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds,
                         method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
        if result.success:
            g = result.x
            if g is not None and np.all(np.isfinite(g)) and np.all(g >= -1e-12):
                return np.maximum(g, 0)
    except Exception:
        pass
    return None

    sc_base = _score_norm_fn(a_base)
    update_best(a_base, sc_base, f"base_n{n_base}")

    # Try a few profiles at n_base
    for idx in [0, 2, 4, 6, 8]:
        if time.monotonic() > p1_end - 30:
            break
        p = _make_profiles(n_base)[idx]
        sc_p = _score_norm_fn(p)
        update_best(p, sc_p, f"prof{idx}_n{n_base}")

    # Short L-BFGS from best
    if time.monotonic() < p1_end - 20:
        sub_dl = min(p1_end - 5, time.monotonic() + (p1_end - time.monotonic()) * 0.6)
        p_b, s_b = _run_lbfgs_continuation(best_seq if len(best_seq) == n_base else a_base, sub_dl, max_betas=6)
        update_best(p_b, s_b, "warmup_lbfgs")

    print(f"  Phase 1 done: best={best_score:.10f} at n={len(best_seq)}")

    # --- Phase 2: LP iterations at n=1024 (main phase, ~900s) ---
    p2_end = min(deadline, t_start + 990.0)
    print("Phase 2: LP iterations (n=1024)")

    n_lp = 1024
    a_lp = _interp_profile(best_seq, n_lp)
    sc_lp = _score_norm_fn(a_lp)
    update_best(a_lp, sc_lp, f"lp_start_n{n_lp}")

    # First round of LP
    if time.monotonic() < p2_end - 60:
        dl_lp1 = min(p2_end - 60, time.monotonic() + 500.0)
        p_b, s_b = _lp_iteration(a_lp, n_lp, max_iters=300, deadline=dl_lp1, t_init=0.5)
        update_best(p_b, s_b, "lp_round1")

    # Second round with perturbation
    if time.monotonic() < p2_end - 30 and best_seq is not None:
        a_lp2 = _interp_profile(best_seq, n_lp)
        z2 = np.log(np.maximum(a_lp2, 1e-15)) + 0.03 * rng.normal(size=n_lp)
        z2 = z2 - z2.mean()
        a_lp2 = _softmax_fn(z2)
        dl_lp2 = min(p2_end - 10, time.monotonic() + 300.0)
        p_b2, s_b2 = _lp_iteration(a_lp2, n_lp, max_iters=200, deadline=dl_lp2, t_init=0.4)
        update_best(p_b2, s_b2, "lp_round2_pert")

    # Third round with larger perturbation
    if time.monotonic() < p2_end - 20 and best_seq is not None:
        a_lp3 = _interp_profile(best_seq, n_lp)
        z3 = np.log(np.maximum(a_lp3, 1e-15)) + 0.08 * rng.normal(size=n_lp)
        z3 = z3 - z3.mean()
        a_lp3 = _softmax_fn(z3)
        dl_lp3 = min(p2_end - 5, time.monotonic() + 200.0)
        p_b3, s_b3 = _lp_iteration(a_lp3, n_lp, max_iters=150, deadline=dl_lp3, t_init=0.3)
        update_best(p_b3, s_b3, "lp_round3_bigpert")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Upscale + L-BFGS at n=2048 or 4096 (300s) ---
    p3_end = min(deadline, t_start + 1290.0)
    print("Phase 3: Upscale + L-BFGS polish")

    if time.monotonic() < p3_end and best_seq is not None:
        # Upscale to 2048
        n_up = 2048
        a_up = _interp_profile(best_seq, n_up)
        sc_up = _score_norm_fn(a_up)
        update_best(a_up, sc_up, f"up_n{n_up}")

        if time.monotonic() < p3_end - 30:
            sub_dl = min(p3_end - 10, time.monotonic() + (p3_end - time.monotonic()) * 0.5)
            p_b, s_b = _run_lbfgs_continuation(a_up, sub_dl, max_betas=10)
            update_best(p_b, s_b, f"lbfgs_n{n_up}")

        # Try n=4096
        if time.monotonic() < p3_end - 60:
            n_up2 = 4096
            a_up2 = _interp_profile(best_seq, n_up2)
            sc_up2 = _score_norm_fn(a_up2)
            update_best(a_up2, sc_up2, f"up_n{n_up2}")

            sub_dl2 = min(p3_end - 10, time.monotonic() + (p3_end - time.monotonic()) * 0.6)
            p_b2, s_b2 = _run_lbfgs_continuation(a_up2, sub_dl2, max_betas=8)
            update_best(p_b2, s_b2, f"lbfgs_n{n_up2}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Adaptive GD + Peak surgery (200s) ---
    p4_end = min(deadline, t_start + 1490.0)
    print("Phase 4: Adaptive GD + Peak surgery")

    if time.monotonic() < p4_end and best_seq is not None:
        sub_dl = min(p4_end - 60, time.monotonic() + 100.0)
        p_b, s_b = _adaptive_gd(best_seq, sub_dl, lr_init=0.08)
        update_best(p_b, s_b, "adaptive_gd")

    if time.monotonic() < p4_end - 10 and best_seq is not None:
        sub_dl2 = min(p4_end - 5, time.monotonic() + 90.0)
        p_b2, s_b2 = _peak_surgery(best_seq, sub_dl2, rng)
        update_best(p_b2, s_b2, "peak_surgery")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Final L-BFGS with tiny beta (150s) ---
    p5_end = min(deadline, t_start + 1640.0)
    print("Phase 5: Final fine L-BFGS")

    if time.monotonic() < p5_end and best_seq is not None:
        sub_dl = min(p5_end - 5, time.monotonic() + 140.0)
        p_b, s_b = _run_lbfgs_continuation(best_seq, sub_dl, max_betas=8)
        update_best(p_b, s_b, "final_lbfgs")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Final peak surgery (60s) ---
    p6_end = deadline
    print("Phase 6: Final peak surgery")

    if time.monotonic() < p6_end and best_seq is not None:
        sub_dl = min(p6_end - 10, time.monotonic() + 50.0)
        p_b, s_b = _peak_surgery(best_seq, sub_dl, rng)
        update_best(p_b, s_b, "final_surgery")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profiles(256)[2]

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profiles(256)[2]
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
