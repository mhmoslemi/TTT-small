        if peak < best_peak:
            best_peak = peak
            best_t = t
            # Try to increase t
            if t < 0.9:
                t = min(1.0, t * 1.5)
            else:
                break
        else:
            t *= 0.5
        if t < 0.01:
            break
    
    if best_t > 0:
        f_new = f + best_t * d
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        return f_new, _max_conv(f_new), True
    return f, M, False
