            c_mod_padded = np.zeros(N)
            len_circ = min(len(c_mod_circ), N)
            c_mod_padded[:len_circ] = c_mod_circ[:len_circ]

            P_mod = np.fft.rfft(c_mod_padded)
            # P_mod should be real and non-negative for valid autocorrelation
            # Clip to handle numerical issues
            P_mod = np.maximum(P_mod.real, 0)

            # New amplitude spectrum
            A_mag = np.sqrt(P_mod)
            A_phase = np.angle(A)
            A_new = A_mag * np.exp(1j * A_phase)

            # New time-domain signal
            a_new = np.fft.irfft(A_new, N)[:n]

            # Project to non-negative with relaxation
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new

            # Relaxation to avoid oscillation
            a = (1 - relax) * a + relax * a_new
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                break
            a = a / s

            sc = _score_norm_fn(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a


def _active_set_reduce(a_init, deadline, max_iter=2000, verbose=True):
    """Active-set gradient reduction with bisection line search."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = _score_norm_fn(a)

    # Work in the space where sum(a) = 1
    # Objective: minimize max(conv(a,a))
    # Gradient of c[k] w.r.t. a[i] = a[k-i] if 0 <= k-i < n

    active_thresh = 0.999  # fraction of max to consider "active"

    for it in range(max_iter):
        if time.monotonic() > deadline - 0.05:
            break

        c = _convolve_a_a(a)
        M = float(c.max())

        # Active set: peaks within a small fraction of the max
        threshold = M * active_thresh
        active = np.where(c >= threshold)[0]

        if len(active) == 0:
            break

        # Compute the average gradient over active set
        # grad_k[i] = a[k-i] if 0 <= k-i < n, else 0
        # h[i] = (1/|S|) * sum_{k in S} a[k-i] for valid (k,i)
        h = np.zeros(n, dtype=np.float64)
        for k_idx in active:
            k_val = int(k_idx)
            # h[i] += a[k_val - i] for i where 0 <= k_val - i < n
            # i ranges from max(0, k_val - n + 1) to min(n-1, k_val)
            i_lo = max(0, k_val - n + 1)
            i_hi = min(n - 1, k_val)
            if i_lo <= i_hi:
                # h[i] += a[k_val - i] for i in [i_lo, i_hi]
                # Let j = k_val - i, so j ranges from k_val - i_hi to k_val - i_lo
                j_lo = k_val - i_hi
                j_hi = k_val - i_lo
                h[i_lo:i_hi + 1] += a[j_lo:j_hi + 1][::-1]
        h /= len(active)

        # Remove the component along ones (to maintain sum(a) = 1)
        h_mean = h.mean()
        d = h - h_mean  # direction to subtract from a

        # Bisection line search: find step t such that max(conv(a - t*d, a - t*d)) < M
        # and a - t*d >= 0
        # First find max valid t for non-negativity
        pos_mask = d > 1e-20
        if np.any(pos_mask):
            t_max_pos = float((a[pos_mask] / d[pos_mask]).min())
        else:
            t_max_pos = np.inf
        t_hi = min(t_max_pos * 0.999, 1.0)
        t_lo = 0.0

        # Evaluate at t_hi
        a_trial = a - t_hi * d
        a_trial = np.maximum(a_trial, 0)
        s_trial = a_trial.sum()
        if s_trial > 1e-15:
            a_trial = a_trial / s_trial
        c_trial = _convolve_a_a(a_trial)
        M_trial = float(c_trial.max())

        if M_trial >= M:
            # Even the largest step doesn't help; try smaller
            # This means the direction is bad; reduce activity threshold
            active_thresh = max(0.99, active_thresh * 0.99)
            continue

        # Bisection: find the largest t that reduces M
        for _ in range(8):
            t_mid = 0.5 * (t_lo + t_hi)
            a_trial = a - t_mid * d
            a_trial = np.maximum(a_trial, 0)
            s_trial = a_trial.sum()
            if s_trial > 1e-15:
                a_trial = a_trial / s_trial
            c_trial = _convolve_a_a(a_trial)
            M_trial = float(c_trial.max())
            if M_trial < M:
                t_lo = t_mid
            else:
                t_hi = t_mid

        t_opt = 0.5 * (t_lo + t_hi)
        a_new = a - t_opt * d
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new

        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Didn't improve; adjust strategy
            active_thresh = max(0.99, active_thresh * 0.995)

        # Gradually tighten the active set threshold
        if it % 100 == 0:
            active_thresh = min(0.9999, active_thresh * 1.001)

    return best_a


def _z_objective(z, beta, n, top_k=0):
    a = _softmax_fn(z)
    c = _convolve_a_a(a)
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_norm = c_top - c_top.max()
        log_sum = float(np.log(np.sum(np.exp(c_norm / beta))))
        return 2.0 * n * (c_top.max() + beta * log_sum)
    else:
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        log_sum = float(np.log(np.sum(e)))
        return 2.0 * n * (cm + beta * log_sum)


def _z_gradient(z, beta, n, top_k=0):
    a = _softmax_fn(z)
    c = _convolve_a_a(a)
    cm = float(c.max())

    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        w = np.zeros(len(c))
        c_top = c[idx]
        c_norm = (c_top - c_top.max()) / beta
        e = np.exp(c_norm)
        s = e.sum()
        if s < 1e-300:
            w[idx[0]] = 1.0
        else:
            w[idx] = e / s
    else:
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300:
            idx = int(np.argmax(c))
            w = np.zeros(len(c))
            w[idx] = 1.0
        else:
            w = e / s

    W = float(np.dot(w, c))
    a_rev = a[::-1]
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a_rev, N)
    w_conv = np.fft.irfft(fw * fa_rev, N)[:2 * n - 1]
    h = w_conv[n - 1:2 * n - 1]

    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs(a_init, deadline, max_wall_time=None, top_k=0):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _convolve_a_a(a_init)
    max_c0 = float(c0.max())

    if max_wall_time is None:
        max_wall_time = max(1.0, deadline - time.monotonic())

    beta_starts = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max(max_c0, 1e-10) for b in beta_starts]
    per_level = max_wall_time / len(betas)

    for beta in betas:
        if time.monotonic() > deadline:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)
        try:
            result = minimize(
                _z_objective, z, args=(beta, n, top_k),
                method='L-BFGS-B', jac=_z_gradient,
                options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 40, 'maxfun': 2000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                if np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _adam_polish(a_init, deadline, lr=0.03, top_k=5, n_max_iter=50000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _convolve_a_a(a_init)
    max_c = float(c.max())
    beta = max(1e-10, 0.01 * max_c)

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        grad = _z_gradient(z, beta, n, top_k)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 100 == 0:
            beta = max(1e-14, beta * 0.85)
        if t % 2000 == 0 and best_score < 1.51:
            lr = max(0.002, lr * 0.95)

    return best_a, best_score


def _make_exp_profile(n, alpha):
    r = 1.0 - alpha
    a = r ** np.arange(n)
    return a / a.sum()


def _make_cosine_profile(n):
    t = np.linspace(0.0, np.pi, n)
    a = np.maximum(np.sin(t)**2, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _directed_peak_reduction(a_init, deadline, max_iter=500):
    """Directly target the worst peaks by adjusting specific components of a."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for it in range(max_iter):
        if time.monotonic() > deadline - 0.05:
            break

        c = _convolve_a_a(a)
        M = float(c.max())
        # Find the single worst peak
        k_worst = int(np.argmax(c))

        # The gradient of c[k_worst] w.r.t. a:
        # grad[i] = a[k_worst - i] if 0 <= k_worst - i < n
        i_lo = max(0, k_worst - n + 1)
        i_hi = min(n - 1, k_worst)
        grad = np.zeros(n)
        if i_lo <= i_hi:
            j_lo = k_worst - i_hi
            j_hi = k_worst - i_lo
            grad[i_lo:i_hi + 1] = a[j_lo:j_hi + 1][::-1]

        # Remove mean to maintain sum constraint
        grad = grad - grad.mean()

        # Also add a small component from the second worst peak
        c_copy = c.copy()
        c_copy[k_worst] = -np.inf
        k_second = int(np.argmax(c_copy))
        if c[k_second] > M * 0.95:
            i_lo2 = max(0, k_second - n + 1)
            i_hi2 = min(n - 1, k_second)
            grad2 = np.zeros(n)
            if i_lo2 <= i_hi2:
                j_lo2 = k_second - i_hi2
                j_hi2 = k_second - i_lo2
                grad2[i_lo2:i_hi2 + 1] = a[j_lo2:j_hi2 + 1][::-1]
            grad2 = grad2 - grad2.mean()
            grad = 0.7 * grad + 0.3 * grad2

        # Line search
        d = -grad  # direction to decrease c[k_worst]
        pos_mask = d > 1e-20
        if np.any(pos_mask):
            t_max = float((a[pos_mask] / d[pos_mask]).min()) * 0.9
        else:
            t_max = 0.1
        t_max = min(t_max, 0.05)

        # Binary search for best step
        lo, hi = 0.0, t_max
        for _ in range(6):
            mid = 0.5 * (lo + hi)
            a_try = a + mid * d
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
            c_try = _convolve_a_a(a_try)
            M_try = float(c_try.max())
            if M_try < M:
                lo = mid
            else:
                hi = mid

        t_opt = 0.5 * (lo + hi)
        a_new = a + t_opt * d
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new

        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining spectral (GS), active-set,
    directed peak reduction, and L-BFGS methods.
    """
    global height_sequence_1

    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Phase 0: Initial candidates ---
    print("=== Phase 0: Initialization ===")
    n_base = 4096

    # Try parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm_fn(parent)
                update_best(parent, sc, f"parent_n{len(parent)}")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent: {e}")

    # Interpolate parent to n_base
    if parent is not None and len(parent) != n_base:
        p_interp = _interp_profile(parent, n_base)
        sc = _score_norm_fn(p_interp)
        update_best(p_interp, sc, f"parent_interp_n{n_base}")
    elif parent is not None:
        p_interp = parent.copy()

    if p_interp is None:
        p_interp = _make_exp_profile(n_base, 0.002)
        sc = _score_norm_fn(p_interp)
        update_best(p_interp, sc, "exp_init")

    # --- Phase 1: GS Spectral iterations ---
    print("=== Phase 1: GS Spectral ===")
    gs_end = min(deadline, t_start + 300.0)
    remaining = gs_end - time.monotonic()
    n_gs_iters = max(100, int(remaining / 0.002))  # ~2ms per iter estimate
    print(f"  GS iterations: ~{n_gs_iters}, time budget: {remaining:.0f}s")

    a_gs = _gs_spectral_iterate(p_interp, n_gs_iters, gs_end,
                                 alpha_schedule=[0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01],
                                 relax=0.7)
    sc_gs = _score_norm_fn(a_gs)
    update_best(a_gs, sc_gs, "gs")

    # Also try a second GS run with different parameters
    if time.monotonic() < gs_end - 30:
        a_gs2 = _gs_spectral_iterate(p_interp, n_gs_iters // 2, gs_end,
                                      alpha_schedule=[0.4, 0.25, 0.15, 0.08, 0.03],
                                      relax=0.5)
        sc_gs2 = _score_norm_fn(a_gs2)
        update_best(a_gs2, sc_gs2, "gs_relaxed")

    # Use the best of GS and parent for subsequent phases
    work_a = best_seq.copy() if best_seq is not None else p_interp.copy()

    # --- Phase 2: Active-set reduction ---
    print("=== Phase 2: Active-set reduction ===")
    as_end = min(deadline, t_start + 600.0)
    remaining = as_end - time.monotonic()
    if remaining > 10:
        print(f"  Active-set: time budget {remaining:.0f}s")
        a_as = _active_set_reduce(work_a, as_end, max_iter=5000)
        sc_as = _score_norm_fn(a_as)
        update_best(a_as, sc_as, "active_set")
        work_a = best_seq.copy() if best_seq is not None else work_a

    # --- Phase 3: Directed peak reduction ---
    print("=== Phase 3: Directed peak reduction ===")
    dpr_end = min(deadline, t_start + 800.0)
    remaining = dpr_end - time.monotonic()
    if remaining > 10:
        print(f"  Directed peak reduction: time budget {remaining:.0f}s")
        a_dpr = _directed_peak_reduction(work_a, dpr_end, max_iter=500)
        sc_dpr = _score_norm_fn(a_dpr)
        update_best(a_dpr, sc_dpr, "directed")
        work_a = best_seq.copy() if best_seq is not None else work_a

    # --- Phase 4: L-BFGS polish ---
    print("=== Phase 4: L-BFGS polish ===")
    lbfgs_end = min(deadline, t_start + 1200.0)
    remaining = lbfgs_end - time.monotonic()
    if remaining > 20 and best_seq is not None:
        print(f"  L-BFGS: time budget {remaining:.0f}s")
        per_run = min(120.0, remaining * 0.4)
        dl = min(lbfgs_end - 5, time.monotonic() + per_run)
        a_lb, sc_lb = _run_lbfgs(best_seq, dl, max_wall_time=per_run, top_k=0)
        update_best(a_lb, sc_lb, "lbfgs_full")

        # Second L-BFGS with top_k
        if time.monotonic() < lbfgs_end - 20:
            per_run2 = min(60.0, (lbfgs_end - time.monotonic()) * 0.5)
            dl2 = min(lbfgs_end - 5, time.monotonic() + per_run2)
            a_lb2, sc_lb2 = _run_lbfgs(best_seq, dl2, max_wall_time=per_run2, top_k=10)
            update_best(a_lb2, sc_lb2, "lbfgs_topk")

    # --- Phase 5: Adam polish ---
    print("=== Phase 5: Adam polish ===")
    adam_end = min(deadline, t_start + 1500.0)
    remaining = adam_end - time.monotonic()
    if remaining > 20 and best_seq is not None:
        print(f"  Adam: time budget {remaining:.0f}s")
        per_run = min(200.0, remaining * 0.5)
        dl = min(adam_end - 5, time.monotonic() + per_run)
        a_ad, sc_ad = _adam_polish(best_seq, dl, lr=0.03, top_k=5, n_max_iter=50000)
        update_best(a_ad, sc_ad, "adam1")

        if time.monotonic() < adam_end - 20:
            per_run2 = min(100.0, (adam_end - time.monotonic()) * 0.5)
            dl2 = min(adam_end - 5, time.monotonic() + per_run2)
            a_ad2, sc_ad2 = _adam_polish(best_seq, dl2, lr=0.01, top_k=15, n_max_iter=30000)
            update_best(a_ad2, sc_ad2, "adam2_topk15")

    # --- Phase 6: Active-set again on best ---
    print("=== Phase 6: Active-set round 2 ===")
    as2_end = min(deadline, t_start + 1650.0)
    remaining = as2_end - time.monotonic()
    if remaining > 20 and best_seq is not None:
        print(f"  Active-set round 2: time budget {remaining:.0f}s")
        a_as2 = _active_set_reduce(best_seq, as2_end, max_iter=3000)
        sc_as2 = _score_norm_fn(a_as2)
        update_best(a_as2, sc_as2, "active_set_2")

    # --- Phase 7: Try n=8192 ---
    print("=== Phase 7: Try n=8192 ===")
    n8192_end = min(deadline, t_start + 1750.0)
    remaining = n8192_end - time.monotonic()
    if remaining > 60 and best_seq is not None and len(best_seq) == 4096:
        print(f"  n=8192 attempt: time budget {remaining:.0f}s")
        p_8192 = _interp_profile(best_seq, 8192)
        sc_8192 = _score_norm_fn(p_8192)
        update_best(p_8192, sc_8192, "interp_8192")

        if time.monotonic() < n8192_end - 20:
            a_gs8 = _gs_spectral_iterate(p_8192, 500, n8192_end - 20,
                                          alpha_schedule=[0.3, 0.15, 0.05, 0.02],
                                          relax=0.7)
            sc_gs8 = _score_norm_fn(a_gs8)
            update_best(a_gs8, sc_gs8, "gs_8192")

    # --- Phase 8: Final short L-BFGS ---
    print("=== Phase 8: Final polish ===")
    if time.monotonic() < deadline - 10 and best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            dl = min(deadline - 3, time.monotonic() + min(30.0, remaining * 0.6))
            a_final, sc_final = _run_lbfgs(best_seq, dl, max_wall_time=max(5.0, remaining * 0.5), top_k=8)
            update_best(a_final, sc_final, "final_lbfgs")

    # --- Final validation ---
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = _make_exp_profile(4096, 0.003)
            final_score = _score_fn(final_seq)

        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")

        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 16384:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  evaluate_sequence check failed: {e}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_exp_profile(4096, 0.003)
    print(f"  FALLBACK: n=4096, score={_score_norm_fn(fallback):.10f}")
    return [float(x) for x in fallback]
