    n = len(a)
    # LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0
    # (a*b)[k] = sum_i a[k-i] * b[i] for valid i

    if result.success:

    return None

                t *= 0.5
                continue
            a_new = a_new / s_new
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                a = a_new
                best_a = a_new.copy()
                best_score = sc
                improved = True
                break
            t *= 0.5
            if t < 0.001:
                break
        
        if not improved:
            no_improve += 1
            step_size *= 0.85
            if no_improve > 30:
                break
        else:
            no_improve = 0
            step_size = min(0.5, step_size * 1.03)
    
    return best_a, best_score


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _gradient_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _objective_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _run_lbfgs_continuation(a_init, deadline):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    betas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max_c0 for b in betas]
    
    total_time = deadline - time.monotonic()
    per_level = total_time / len(betas)
    
    for i, beta in enumerate(betas):
        if time.monotonic() >= deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level * 1.5)
        try:
            result = minimize(
                _objective_smooth, z, args=(beta,),
                method='L-BFGS-B', jac=lambda zz, bb: _gradient_smooth(zz, bb),
                options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-9,
                         'maxls': 30, 'maxfun': 1000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x.copy()
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new
        except Exception:
            pass
    
    return best_a, best_score


def _nesterov_optimize(a_init, deadline, beta_fracs, lr_init=0.02, mom_init=0.9):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    
    lr = lr_init
    mom = mom_init
    t = 0
    
    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-12, beta_frac * max_c)
        
        while time.monotonic() < deadline:
            t += 1
            grad_y = _gradient_smooth(y, beta)
            if not np.all(np.isfinite(grad_y)):
                grad_y = _gradient_smooth(z, beta)
            
            z_new = z - lr * grad_y
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -40, 40)
            
            y_new = z_new + mom * (z_new - z)
            y_new = y_new - y_new.mean()
            
            z = z_new
            y = y_new
            
            if t % 100 == 0:
                a_curr = _softmax_fn(z)
                sc = _score_norm_fn(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()
            
            if t % 200 == 0:
                mom = min(0.999, mom + 0.0003)
                lr = max(0.001, lr * 0.95)
        
        c = _full_convolve(_softmax_fn(z), _softmax_fn(z))
        max_c = float(c.max())
        lr = max(0.001, lr * 0.7)
    
    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _subgradient_polish(a_init, deadline, lr_init=0.003, top_k_frac=0.05):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    lr = lr_init
    momentum = np.zeros(n)
    mom_coeff = 0.85
    top_k = max(10, int(top_k_frac * (2 * n - 1)))
    
    for it in range(100000):
        if time.monotonic() > deadline:
            break
        
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        top_idx = np.argsort(c)[-top_k:]
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / top_k
        W = float(np.dot(w, c))
        
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        
        if not np.all(np.isfinite(grad)):
            break
        
        momentum = mom_coeff * momentum - lr * grad
        z = z + momentum
        z = z - z.mean()
        z = np.clip(z, -40, 40)
        
        if it % 200 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            else:
                lr *= 0.9
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")
    
    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    n_base = 4096
    if parent is not None:
        n_base = len(parent)
        print(f"Parent: n={n_base}, score={_score_fn(parent):.10f}")
        best_seq = parent.copy()
        best_score = _score_fn(parent)
    else:
        best_seq = _make_linear_profile(n_base, 1.5)
        best_score = _score_fn(best_seq)
        print(f"No parent, using linear profile n={n_base}, score={best_score:.10f}")
    
    # --- Phase 0: Quick initial optimization with L-BFGS (120s) ---
    phase0_end = min(deadline, t_start + 120.0)
    print(f"Phase 0: L-BFGS initialization (until {phase0_end - t_start:.0f}s)")
    p_best, sc_best = _run_lbfgs_continuation(best_seq, phase0_end)
    update_best(p_best, sc_best, "lbfgs_phase0")
    
    # --- Phase 1: LP-based optimization (600s) ---
    phase1_end = min(deadline, t_start + 720.0)
    print(f"Phase 1: LP-based optimization (until {phase1_end - t_start:.0f}s)")
    
    if time.monotonic() < phase1_end - 60:
        # First try full LP for a few iterations to gauge speed
        test_start = time.monotonic()
        a_test = best_seq.copy() / best_seq.sum()
        c_test = _full_convolve(a_test, a_test)
        M_test = float(c_test.max())
        dir_test = _solve_lp_direction(a_test, M_test, active_indices=np.arange(2*len(a_test)-1), timeout=20.0)
        test_time = time.monotonic() - test_start
        print(f"  LP test: {test_time:.2f}s per solve, success={dir_test is not None}")
        
        if dir_test is not None and test_time < 30:
            # Full LP is feasible
            p_lp, sc_lp = _lp_optimize(best_seq, min(phase1_end, time.monotonic() + (phase1_end - time.monotonic()) * 0.6))
            update_best(p_lp, sc_lp, "lp_full")
        else:
            # Use active set LP
            p_lp, sc_lp = _lp_optimize_active(best_seq, min(phase1_end, time.monotonic() + (phase1_end - time.monotonic()) * 0.7))
            update_best(p_lp, sc_lp, "lp_active")
        
        # Use remaining Phase 1 time for more LP iterations
        if time.monotonic() < phase1_end - 30:
            p_lp2, sc_lp2 = _lp_optimize_active(best_seq, phase1_end - 10)
            update_best(p_lp2, sc_lp2, "lp_active2")
    
    print(f"Phase 1 done: best={best_score:.10f}")
    
    # --- Phase 2: L-BFGS continuation polishing (300s) ---
    phase2_end = min(deadline, t_start + 1020.0)
    print(f"Phase 2: L-BFGS polishing (until {phase2_end - t_start:.0f}s)")
    if time.monotonic() < phase2_end - 30:
        p_best, sc_best = _run_lbfgs_continuation(best_seq, phase2_end - 5)
        update_best(p_best, sc_best, "lbfgs_phase2")
    
    # --- Phase 3: Nesterov acceleration (200s) ---
    phase3_end = min(deadline, t_start + 1220.0)
    print(f"Phase 3: Nesterov (until {phase3_end - t_start:.0f}s)")
    if time.monotonic() < phase3_end - 30:
        beta_fracs = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]
        p_best, sc_best = _nesterov_optimize(best_seq, phase3_end - 5,
                                             beta_fracs, lr_init=0.015, mom_init=0.9)
        update_best(p_best, sc_best, "nesterov_phase3")
    
    # --- Phase 4: Subgradient polishing (150s) ---
    phase4_end = min(deadline, t_start + 1370.0)
    print(f"Phase 4: Subgradient polish (until {phase4_end - t_start:.0f}s)")
    if time.monotonic() < phase4_end - 20:
        p_best, sc_best = _subgradient_polish(best_seq, phase4_end - 5, lr_init=0.002, top_k_frac=0.03)
        update_best(p_best, sc_best, "subgrad_phase4")
    
    # --- Phase 5: Noisy restarts + LP (150s) ---
    phase5_end = min(deadline, t_start + 1520.0)
    print(f"Phase 5: Restarts (until {phase5_end - t_start:.0f}s)")
    if time.monotonic() < phase5_end - 30:
        n_restarts = 3
        per_restart = (phase5_end - time.monotonic() - 10) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 15:
                break
            noise = 0.03 + 0.02 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=len(best_seq))
            z = np.clip(z, -30, 30)
            z = z - z.mean()
            p_start = _softmax_fn(z)
            
            per_dl = min(phase5_end - 10, time.monotonic() + per_restart)
            p_best, sc_best = _nesterov_optimize(p_start, per_dl,
                                                  [0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                                  lr_init=0.01, mom_init=0.85)
            update_best(p_best, sc_best, f"restart{i}")
    
    # --- Phase 6: Try different length (100s) ---
    phase6_end = min(deadline, t_start + 1620.0)
    print(f"Phase 6: Different lengths (until {phase6_end - t_start:.0f}s)")
    if time.monotonic() < phase6_end - 30 and best_seq is not None:
        n_curr = len(best_seq)
        
        for target_n in [2048, 8192]:
            if time.monotonic() > phase6_end - 20:
                break
            if target_n == n_curr:
                continue
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_fn(p_interp)
            print(f"  Interpolated to n={target_n}: score={sc_interp:.10f}")
            
            if sc_interp < best_score * 1.001:
                per_time = min(40.0, phase6_end - time.monotonic() - 10)
                if per_time > 15:
                    p_best, sc_best = _run_lbfgs_continuation(p_interp, time.monotonic() + per_time)
                    update_best(p_best, sc_best, f"lbfgs_n{target_n}")
    
    # --- Phase 7: Final fine polish (remaining) ---
    phase7_end = deadline
    print(f"Phase 7: Final polish (until {phase7_end - t_start:.0f}s)")
    if time.monotonic() < phase7_end - 15 and best_seq is not None:
        remaining = phase7_end - time.monotonic()
        if remaining > 40:
            p_best, sc_best = _nesterov_optimize(
                best_seq, phase7_end - 5,
                [1e-5, 5e-6, 2e-6, 1e-6, 5e-7],
                lr_init=0.005, mom_init=0.95)
            update_best(p_best, sc_best, "final_nesterov")
    
    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_linear_profile(2048, 1.5)
        
        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")
        
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned inf/NaN, using internal: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")
        
        return [float(x) for x in final_seq]
    
    fallback = _make_linear_profile(4096, 1.5)
    print(f"FALLBACK: n=4096, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
