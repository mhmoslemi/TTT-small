**Mirror Descent with KL Divergence:**
We want to minimize $f(a) = \max_k (a*a)_k$ (or smooth approximation) subject to $a \geq 0, \sum a = 1$.

$$a^{(t+1)} = \arg\min_{b \in \Delta} \text{KL}(b|a^{(t)}) + \eta_t f(b)$$

where $\Delta$ is the simplex. For the exponential family (which is what KL gives), the update is:
$$a^{(t+1)}_j = \frac{a^{(t)}_j \exp(-\eta_t g_j)}{\sum_k a^{(t)}_k \exp(-\eta_t g_k)}$$

where $g_j$ is the gradient of $f$ at $a^{(t)}$.

For the smooth approximation $f(a) = \text{LSE}_\beta(a*a) = \beta \log \sum_k \exp((a*a)_k / \beta)$:
The gradient is:
$$\frac{\partial f}{\partial a_j} = \sum_k p_k \frac{\partial (a*a)_k}{\partial a_j} = \sum_k p_k \cdot 2 a_{k-j} \text{ (for valid indices)}$$

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based convolution, L-BFGS continuation, mirror descent,
FISTA with simplex projection, coordinate descent, and multi-restart.
"""
import numpy as np
from scipy.optimize import minimize
import time
import math

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes a is normalized to sum=1. Returns 2n*max(c)."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _convolve(a, b):
    return _fft_convolve(a, b)

def _gradient_lse(a, beta):
    """Gradient of LSE_beta(conv(a,a)) w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a, a)
    cm = c.max()
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p = e / s
    # grad_j = 2 * sum_k p_k * a_{k-j} (for valid k-j)
    # This is 2 * convolve(p, a_reversed)[n-1 : 2n-1]
    a_rev = a[::-1]
    p_conv_a = _fft_convolve(p, a_rev)
    grad = 2.0 * p_conv_a[n-1 : 2*n-1]
    return grad

def _lse_objective(a, beta):
    n = len(a)
    c = _fft_convolve(a, a)
    cm = c.max()
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        return float(2.0 * n * cm)
    return float(2.0 * n * (cm + beta * np.log(s)))
