def solve_lp(a, c_max, solver=' highs'):
    """Solve: max sum(b) s.t. a*b <= c_max (elementwise over all lags), b >= 0"""
    n = len(a)
    # Build the convolution matrix: A[i,j] such that (A @ b)[i] = (a*b)[i]
    # (a*b)[k] = sum_{i=0}^{min(k,n-1)} a[max(0,k-n+1)] * b[k-i] ... 
    # Actually: (a*b)[k] = sum_i a[i] * b[k-i] where b[j]=0 if j<0 or j>=n
    # So (a*b)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * b[k-i]
    # Let j = k-i, then i = k-j, so j ranges from max(0,k-n+1) to min(k,n-1)
    # (a*b)[k] = sum_j a[k-j] * b[j] where 0<=j<n and 0<=k-j<n
    
    # The constraint matrix A has A[k,j] = a[k-j] if max(0,k-n+1) <= j <= min(k,n-1), else 0
    
    # This is a Toeplitz matrix. Let me build it sparse.
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_start = max(0, k-n+1)
        j_end = min(k, n-1)
        for j in range(j_start, j_end+1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    
    A_mat = csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: max sum(b) s.t. A_mat @ b <= c_max, b >= 0
    # In standard form for linprog: min -sum(b) s.t. A_mat @ b <= c_max, b >= 0
    c_vec = -np.ones(n)
    res = linprog(c_vec, A_ub=A_mat, b_ub=np.full(2*n-1, c_max), bounds=[(0, None)]*n, method='highs')
    return res.x if res.success else None

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines simplex-projected Nesterov gradient descent with smooth-max continuation,
LP-based directional improvement, spectral initialization, and multi-restart strategy.
Key improvements over previous: direct simplex optimization (no softmax barrier),
LP refinement, adaptive Polyak step, and symmetry exploitation.
"""
import numpy as np
from scipy.optimize import linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _project_simplex(v, target=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    if not np.any(cond):
        return np.full(n, target / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target) / (rho + 1)
    return np.maximum(v - theta, 0)


def _smooth_grad(a, beta):
    """Compute smooth max value and gradient. Assumes sum(a)=1."""
    n = len(a)
    c = _convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    smooth_val = c_max + beta * np.log(max(s, 1e-300))

    # Gradient: grad[i] = 2 * sum_j a[j] * w[i+j]
    N = _next_pow2(3 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    cross = np.fft.irfft(fw * np.conj(fa), N)
    h = cross[:n]
    grad = 2.0 * h
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n)
    return smooth_val, grad, c_max


def _lp_direction(a, c_max, n_top=150, time_limit=8.0):
    """Solve LP with top constraints to find improvement direction."""
    n = len(a)
    c = _convolve(a, a)
    top_idx = np.argsort(c)[::-1][:n_top]

    A_sub = np.zeros((n_top, n))
    for i, k in enumerate(top_idx):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1)
        A_sub[i, js] = a[k - js]

    try:
        res = linprog(-np.ones(n), A_ub=A_sub, b_ub=np.full(n_top, c_max),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': time_limit})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def _make_profiles(n):
    """Generate various initial profiles normalized to sum=1."""
    profiles = []
    t = np.linspace(0, 1, n)
    u = 2.0 * t - 1.0

    a = np.maximum(1.0 + 1.5 * u, 1e-12)
    profiles.append(a / a.sum())

    a = np.maximum(1.0 + 0.5 * u + 0.8 * u**2, 1e-12)
    profiles.append(a / a.sum())

    a = np.maximum(0.5 * (1.0 + np.cos(np.pi * u)), 1e-12)
    profiles.append(a / a.sum())

    a = np.exp(-3.0 * u**2)
    profiles.append(a / a.sum())

    a = np.maximum(1.0 + 2.0 * u**2, 1e-12)
    profiles.append(a / a.sum())

    a = np.maximum(1.0 - 0.5 * u**2 + 0.3 * u**4, 1e-12)
    profiles.append(a / a.sum())

    return profiles


def _nesterov_smooth(a, deadline, betas_fracs, lr_init=0.5, mom_init=0.3,
                      eval_interval=200, label=""):
    """Nesterov-accelerated gradient descent on smooth max with continuation."""
    n = len(a)
    best_a = a.copy()
    best_norm = _score_norm(a)
    c_init = float(_convolve(a, a).max())

    for beta_frac in betas_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-10, beta_frac * c_init)

        lr = lr_init
        mom = mom_init
        momentum = np.zeros(n)
        y = a.copy()
        it = 0
        smooth_history = []

        while time.monotonic() < deadline:
            it += 1
            a_y = _project_simplex(y, 1.0)
            sv, grad, c_max = _smooth_grad(a_y, beta)

            if not np.all(np.isfinite(grad)):
                break

            # Nesterov step
            a_new = a - lr * (momentum + grad)
            a_new = _project_simplex(a_new, 1.0)
            momentum = (it / (it + 1.0 + mom)) * (momentum + grad) + (mom * it / (it + 1.0 + mom)) * grad

            # Simplified Nesterov
            y = a_new + (it / (it + 1.0 + mom)) * (a_new - a)
            y = _project_simplex(y, 1.0)

            # Check improvement using smooth value
            sv_new, _, _ = _smooth_grad(a_new, beta)
            if sv_new < sv * 1.001:
                a = a_new
                lr = min(lr * 1.01, 5.0)
            else:
                lr *= 0.85
                if lr < 1e-10:
                    break

            if it % eval_interval == 0:
                sc = _score_norm(a)
                if sc < best_norm:
                    best_norm = sc
                    best_a = a.copy()
                    print(f"    [{label}] it={it}, beta={beta:.2e}, score={sc:.10f}")

        # End of beta level
        sc = _score_norm(a)
        if sc < best_norm:
            best_norm = sc
            best_a = a.copy()
        c_init = float(_convolve(best_a, best_a).max())

    return best_a, best_norm


def _lp_phase(a, deadline, max_iters=None, label=""):
    """Iterative LP-based refinement."""
    n = len(a)
    best_a = a.copy()
    best_norm = _score_norm(a)
    it = 0

    if max_iters is None:
        max_iters = 500

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        c = _convolve(a, a)
        c_max = float(c.max())

        b = _lp_direction(a, c_max, n_top=150, time_limit=6.0)
        if b is None:
            continue

        b_sum = float(b.sum())
        if b_sum <= 1e-10:
            continue

        b_norm = b / b_sum

        # Try multiple blend factors
        improved = False
        for t_blend in [0.5, 0.7, 0.9, 1.0, 0.3]:
            a_try = (1.0 - t_blend) * a + t_blend * b_norm
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score_norm(a_try)
            if sc_try < best_norm:
                a = a_try
                best_a = a.copy()
                best_norm = sc_try
                improved = True
                if it % 10 == 0:
                    print(f"    [{label}] LP it={it}, score={best_norm:.10f}")
                break

        if not improved and it % 20 == 0:
            # Try small random perturbation to escape stagnation
            a_pert = a + 0.005 * np.random.default_rng(it).normal(size=n)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            sc_pert = _score_norm(a_pert)
            if sc_pert < best_norm:
                a = a_pert
                best_a = a.copy()
                best_norm = sc_pert

    return best_a, best_norm


def _symmetric_optimize(a, deadline, beta=1e-4, n_iter=10000, label=""):
    """Optimize using symmetry a[i] = a[n-1-i]."""
    n = len(a)
    half = n // 2
    best_a = a.copy()
    best_norm = _score_norm(a)

    # Work with first half
    x = a[:half].copy()
    if n % 2 == 0:
        # a = [x, x_reversed]
        pass
    else:
        # a = [x, middle, x_reversed]
        pass

    # For even n: a[i] = a[n-1-i], so a[i] = x[i] for i < n/2, a[n/2+i] = x[n/2-1-i]
    # For odd n: a[i] = x[i] for i <= n/2, a[n/2+1+i] = x[n/2-1-i]

    def expand(half_x):
        if n % 2 == 0:
            return np.concatenate([half_x, half_x[::-1]])
        else:
            return np.concatenate([half_x, half_x[-2::-1]])

    def pack(full):
        if n % 2 == 0:
            return full[:half]
        else:
            return full[:half + 1]

    momentum = np.zeros_like(x)
    lr = 0.1
    mom = 0.5

    for it in range(n_iter):
        if time.monotonic() >= deadline:
            break
        a_full = expand(x)
        a_full = _project_simplex(a_full, 1.0)
        sv, grad_full, c_max = _smooth_grad(a_full, beta)
        if not np.all(np.isfinite(grad_full)):
            break
        grad_half = pack(grad_full)

        x_new = x - lr * (momentum + grad_half)
        x_new = np.maximum(x_new, 0)

        a_new_full = expand(x_new)
        a_new_full = _project_simplex(a_new_full, 1.0)
        sv_new, _, _ = _smooth_grad(a_new_full, beta)

        if sv_new < sv:
            x = pack(a_new_full)
            momentum = mom * momentum + (1 - mom) * grad_half
            lr = min(lr * 1.01, 5.0)
        else:
            lr *= 0.8
            if lr < 1e-10:
                break

        if it % 200 == 0:
            sc = _score_norm(a_new_full)
            if sc < best_norm:
                best_norm = sc
                best_a = a_new_full.copy()

    return best_a, best_norm


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, lbl=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{lbl}]")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_par = _score_norm(parent)
        print(f"Parent: n={n}, score={sc_par:.10f}")
    else:
        n = 4096
        parent = _make_profiles(n)[0]
        sc_par = _score_norm(parent)
        print(f"No parent, using profile n={n}, score={sc_par:.10f}")

    # Phase 0: Evaluate initial candidates (30s)
    phase0_end = min(deadline, t_start + 30.0)
    update_best(parent, sc_par, "parent")

    for p in _make_profiles(n):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm(p)
        if sc < best_score:
            update_best(p, sc, "profile")

    # Also try symmetric version of parent
    a_sym = (parent + parent[::-1]) / 2.0
    a_sym = a_sym / a_sym.sum()
    sc_sym = _score_norm(a_sym)
    if sc_sym < best_score:
        update_best(a_sym, sc_sym, "symmetric")

    print(f"Phase 0 done: best={best_score:.10f}")

    # Phase 1: Nesterov smooth gradient with continuation (450s)
    phase1_end = min(deadline, t_start + 480.0)
    if best_seq is not None:
        betas1 = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        p_best, sc_best = _nesterov_smooth(best_seq, phase1_end, betas1,
                                           lr_init=0.3, mom_init=0.5,
                                           eval_interval=300, label="P1")
        if sc_best < best_score:
            update_best(p_best, sc_best, "P1_nesterov")

    print(f"Phase 1 done: best={best_score:.10f}")

    # Phase 2: LP-based refinement (400s)
    phase2_end = min(deadline, t_start + 880.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        p_best, sc_best = _lp_phase(best_seq, phase2_end, max_iters=300, label="P2")
        if sc_best < best_score:
            update_best(p_best, sc_best, "P2_lp")

    print(f"Phase 2 done: best={best_score:.10f}")

    # Phase 3: Fine Nesterov polishing (350s)
    phase3_end = min(deadline, t_start + 1230.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        betas3 = [0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
        p_best, sc_best = _nesterov_smooth(best_seq, phase3_end, betas3,
                                           lr_init=0.2, mom_init=0.7,
                                           eval_interval=200, label="P3")
        if sc_best < best_score:
            update_best(p_best, sc_best, "P3_fine")

    print(f"Phase 3 done: best={best_score:.10f}")

    # Phase 4: LP + gradient alternation (250s)
    phase4_end = min(deadline, t_start + 1480.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        a = best_seq.copy()
        n_a = len(a)
        sub_deadline = phase4_end

        # Alternate: 30s gradient, 30s LP
        while time.monotonic() < sub_deadline - 60:
            # Gradient burst
            g_end = min(sub_deadline - 30, time.monotonic() + 30)
            betas_g = [0.00005]
            a, sc_g = _nesterov_smooth(a, g_end, betas_g, lr_init=0.1, mom_init=0.8,
                                       eval_interval=100, label="P4g")
            if sc_g < best_score:
                update_best(a, sc_g, "P4_grad")

            # LP burst
            lp_end = min(sub_deadline, time.monotonic() + 30)
            a, sc_lp = _lp_phase(a, lp_end, max_iters=30, label="P4lp")
            if sc_lp < best_score:
                update_best(a, sc_lp, "P4_lp")

    print(f"Phase 4 done: best={best_score:.10f}")

    # Phase 5: Restarts (200s)
    phase5_end = min(deadline, t_start + 1680.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        n_a = len(best_seq)
        n_restarts = 5
        per_restart = (phase5_end - time.monotonic()) / max(1, n_restarts)

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            noise_lvl = 0.005 + 0.005 * i
            a_r = best_seq + noise_lvl * rng.normal(size=n_a)
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

            per_dl = min(phase5_end - 5, time.monotonic() + per_restart)
            betas_r = [0.0001, 0.00003]
            a_r, sc_r = _nesterov_smooth(a_r, per_dl, betas_r, lr_init=0.1, mom_init=0.6,
                                         eval_interval=100, label=f"R{i}")
            if sc_r < best_score:
                update_best(a_r, sc_r, f"restart{i}")

    print(f"Phase 5 done: best={best_score:.10f}")

    # Phase 6: Final ultra-fine polish (remaining time)
    if best_seq is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            betas_final = [0.000005, 0.000002]
            a_f, sc_f = _nesterov_smooth(best_seq, deadline, betas_final,
                                         lr_init=0.05, mom_init=0.9,
                                         eval_interval=100, label="P6")
            if sc_f < best_score:
                update_best(a_f, sc_f, "final_polish")

    # Final validation and return
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profiles(4096)[0]

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned non-finite: {sc_eval}")
        except Exception as ex:
            print(f"evaluate_sequence exception: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profiles(4096)[0]
    print(f"FALLBACK: n=4096, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
