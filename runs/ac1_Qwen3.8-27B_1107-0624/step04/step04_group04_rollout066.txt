"""
Active-set Newton optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses exact subgradient/Newton directions over the active lag set, Adam polish,
and multi-scale refinement. Key innovation: solves a linear system to compute
the minimum-norm direction that equalizes active peaks while preserving normalization.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_general(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * _fft_convolve(a, a).max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _make_power_profile(n, alpha, beta=0.0):
    i = np.arange(n) + 1.0
    a = i ** (-alpha)
    if beta > 0:
        a = a * (n + 1.0 - i) ** (-beta)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_hann_profile(n):
    x = np.arange(n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-15)
    return p / p.sum()


def _newton_direction(a, active_lags, n):
    """
    Compute minimum-norm direction d such that:
      (a*d)[k] = target_k for each active k
      sum(d) = 0
    where target_k equalizes the active peaks.
    """
    c = _fft_convolve(a, a)
    m = len(active_lags)
    if m == 0:
        return None, np.inf

    active_vals = c[active_lags]
    target_sum = np.sum(active_vals)

    # Build A_mat: (m+1) x n
    # Row j: a shifted so that row_j[i] = a[active_lags[j] - i] if valid
    A = np.zeros((m + 1, n))
    for j, k in enumerate(active_lags):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A[j, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    A[m, :] = 1.0

    # Target vector: want (a*d)[k_j] = (target_sum/m - c[k_j]) / 2 for each j
    # The factor 2 comes from: (a+ad)(a+ad)[k] ≈ c[k] + 2*a*d[k]
    target = np.zeros(m + 1)
    for j in range(m):
        target[j] = (target_sum / m - active_vals[j]) / 2.0
    target[m] = 0.0

    # Solve: d = A^T (A A^T)^{-1} target
    # A A^T is (m+1) x (m+1)
    AT = A.T  # n x (m+1)
    AAT = A @ AT  # (m+1) x (m+1)
    # Add small regularization for stability
    AAT += 1e-12 * np.eye(m + 1)
    try:
        solve_tt = np.linalg.solve(AAT, target)
    except np.linalg.LinAlgError:
        solve_tt = np.linalg.lstsq(AAT, target, rcond=None)[0]
    d = AT @ solve_tt  # n vector

    if not np.all(np.isfinite(d)):
        return None, np.inf
    return d, float(np.linalg.norm(d))


def _project_simplex(a, target=1.0):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.full(len(a), target / len(a))
    return a * (target / s)

        if np.isfinite(sc_new) and sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            stalled = 0
            # Narrow active set as we converge
            if it > 100 and cur_top_k > 1:
                cur_top_k = max(1, cur_top_k // 2)
        else:
            stalled += 1
            if stalled > max_stall:
                # Widen active set and reduce step
                cur_top_k = min(cur_top_k * 2, 30)
                stalled = 0

        # Periodic check
        if it % 500 == 0:
            pass

    return best_a / best_a.sum(), best_score


def _adam_optimize(a_init, deadline, lr=0.03, top_k=5, max_iter=200000):
    """Adam in logit space with adaptive top-k smoothing."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    best_a = a.copy()
    best_score = _score_norm(a)

    c = _fft_convolve(a, a)
    max_c = float(c.max())
    beta = 0.01 * max_c

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_fn(z)
        c = _fft_convolve(a, a)

        if top_k > 0 and top_k < len(c):
            idx = np.argsort(c)[-top_k:]
            ct = c[idx]
            ct_sh = (ct - ct.max()) / beta
            e = np.exp(ct_sh)
            w = np.zeros(len(c))
            w[idx] = e / e.sum()
        else:
            cs = (c - c.max()) / beta
            e = np.exp(cs)
            w = e / e.sum()

        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _fft_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        g = 2.0 * n * a * (h - W)

        if not np.all(np.isfinite(g)):
            g = np.zeros(n)

        m_buf = b1 * m_buf + (1 - b1) * g
        v_buf = b2 * v_buf + (1 - b2) * (g * g)
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 200 == 0:
            beta = max(1e-14, beta * 0.88)
        if t % 5000 == 0 and best_score < 1.51:
            lr = max(0.003, lr * 0.95)

    return best_a / best_a.sum(), best_score


def _lbfgs_smooth(a_init, deadline, top_k=5):
    """L-BFGS with continuation on smoothed objective in logit space."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    best_a = a.copy()
    best_score = _score_norm(a)

    c = _fft_convolve(a, a)
    max_c = float(c.max())
    beta_starts = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    betas = [b * max(max_c, 1e-10) for b in beta_starts]

    per_level = max(1.0, max(0.0, deadline - time.monotonic()) / len(betas))

    for beta in betas:
        if time.monotonic() > deadline:
            break
        lvl_dl = min(deadline, time.monotonic() + per_level)
        try:
            res = minimize(
                _lbfgs_obj_fn, z, args=(beta, n, top_k),
                method='L-BFGS-B', jac=_lbfgs_grad_fn,
                options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-10,
                         'maxls': 20, 'maxfun': 1000}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_new = _softmax_fn(res.x)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = res.x.copy()
        except Exception:
            pass

    return best_a / best_a.sum(), best_score


def _lbfgs_obj_fn(z, beta, n, top_k):
    a = _softmax_fn(z)
    c = _fft_convolve(a, a)
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        ct = c[idx]
        ct_sh = ct - ct.max()
        return 2.0 * n * (ct.max() + beta * np.log(np.sum(np.exp(ct_sh / beta))))
    else:
        cs = (c - c.max()) / beta
        return 2.0 * n * (c.max() + beta * np.log(np.sum(np.exp(cs))))


def _lbfgs_grad_fn(z, beta, n, top_k):
    a = _softmax_fn(z)
    c = _fft_convolve(a, a)
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        ct = c[idx]
        ct_sh = (ct - ct.max()) / beta
        e = np.exp(ct_sh)
        w = np.zeros(len(c))
        w[idx] = e / e.sum()
    else:
        cs = (c - c.max()) / beta
        e = np.exp(cs)
        w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _fft_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    g = 2.0 * n * a * (h - W)
    if not np.all(np.isfinite(g)):
        return np.zeros(n)
    return g


def _subgrad_optimize(a_init, deadline, nesterov=True):
    """
    Projected subgradient with Nesterov momentum.
    Uses exact subgradient (average over active set) without smoothing.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    if nesterov:
        y = a.copy()
        beta_m = (1.0 + np.sqrt(1.0)) / 2.0
    else:
        y = a.copy()
        beta_m = 0.9

    velocity = np.zeros(n)
    t = 0
    lr = 0.02

    for it in range(2000000):
        if time.monotonic() > deadline:
            break
        t += 1

        c = _fft_convolve(y, y)
        cm = float(c.max())
        if cm < 1e-30:
            break

        # Active set: top few
        top_k = min(5, len(c))
        idx = np.argsort(c)[-top_k:]
        active = idx

        # Average subgradient
        d = np.zeros(n)
        for k in active:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                d[lo:hi + 1] += y[k - hi:k - lo + 1][::-1]
        d /= len(active)

        # Center to preserve sum
        d = d - d.mean()
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm

        # Nesterov update
        if nesterov:
            beta_m = (1.0 + np.sqrt(1.0 + 2.0 / (t * t))) / 2.0
            new_y = a + (a - y) * (beta_m - 1.0) / beta_m
            a = _project_simplex(new_y - lr * d, 1.0)
            y = a + (a - new_y) / beta_m
        else:
            velocity = 0.9 * velocity - lr * d
            a = _project_simplex(a + velocity, 1.0)
            y = a.copy()

        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 1000 == 0:
            lr = max(0.001, lr * 0.98)

    return best_a / best_a.sum(), best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: active-set Newton + subgradient + Adam + L-BFGS.
    Explores multiple n values and diverse initial profiles.
    Minimizes 2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
    """
    global height_sequence_1

    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            el = time.monotonic() - t_start
            print("  [{:6.1f}s] NEW BEST: {:.10f} (n={}) [{}]".format(el, best_score, len(p), label))

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")
    phase1_end = min(deadline, t_start + 30.0)
    candidates = []

    # Try parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm(parent)
                candidates.append((sc, parent.copy(), len(parent), "parent"))
                _update(parent, sc, "parent")
                print("  Parent: n={}, score={:.10f}".format(len(parent), sc))
    except Exception as e:
        print("  No parent: {}".format(e))

    # Generate profiles at n=4096
    n_base = 4096
    profiles_list = []
    for alpha in [0.30, 0.35, 0.40, 0.42, 0.44, 0.46, 0.48, 0.49]:
        profiles_list.append(_make_power_profile(n_base, alpha))
    for alpha in [0.10, 0.20, 0.30, 0.40]:
        profiles_list.append(_make_power_profile(n_base, alpha, alpha))
    profiles_list.append(_make_hann_profile(n_base))
    # One-sided with slight other-side
    for alpha, beta in [(0.45, 0.05), (0.48, 0.02), (0.49, 0.01), (0.4, 0.1), (0.35, 0.15)]:
        profiles_list.append(_make_power_profile(n_base, alpha, beta))

    for idx, p in enumerate(profiles_list):
        if time.monotonic() > phase1_end:
            break
        sc = _score_norm(p)
        candidates.append((sc, p, n_base, "prof_{}_a{}".format(idx, alpha if alpha < 1 else 0)))
        _update(p, sc, "prof_{}".format(idx))

    # Interpolate parent to n_base if different length
    if parent is not None and len(parent) != n_base:
        p_int = _interp_profile(parent, n_base)
        sc = _score_norm(p_int)
        candidates.append((sc, p_int, n_base, "parent_interp"))
        _update(p_int, sc, "parent_interp")

    candidates.sort(key=lambda x: x[0])
    print("  Phase 1 done: best={:.10f}, {} candidates".format(best_score, len(candidates)))

    # === Phase 2: Active-set Newton on best candidate ===
    print("=== Phase 2: Active-set Newton ===")
    phase2_end = min(deadline, t_start + 500.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        remaining = phase2_end - time.monotonic()
        dl = min(phase2_end, t_start + remaining)
        print("  Newton on best (n={}), budget={:.1f}s".format(len(best_seq), remaining))
        p_best, sc_best = _newton_optimize(best_seq, dl, top_k_search=1)
        _update(p_best, sc_best, "newton1")

        # Second Newton pass with wider active set
        if time.monotonic() < phase2_end - 30:
            dl2 = min(phase2_end, time.monotonic() + min(120.0, (phase2_end - time.monotonic()) * 0.6))
            p_best2, sc_best2 = _newton_optimize(best_seq, dl2, top_k_search=3)
            _update(p_best2, sc_best2, "newton3")

    print("  Phase 2 done: best={:.10f}".format(best_score))

    # === Phase 3: Subgradient with Nesterov ===
    print("=== Phase 3: Nesterov subgradient ===")
    phase3_end = min(deadline, t_start + 800.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        remaining = phase3_end - time.monotonic()
        dl = min(phase3_end, time.monotonic() + min(200.0, remaining * 0.7))
        print("  Subgradient (n={}), budget={:.1f}s".format(len(best_seq), dl - time.monotonic()))
        p_best, sc_best = _subgrad_optimize(best_seq, dl, nesterov=True)
        _update(p_best, sc_best, "subgrad")

    print("  Phase 3 done: best={:.10f}".format(best_score))

    # === Phase 4: Adam polish ===
    print("=== Phase 4: Adam polish ===")
    phase4_end = min(deadline, t_start + 1100.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        remaining = phase4_end - time.monotonic()
        dl = min(phase4_end, time.monotonic() + min(200.0, remaining * 0.6))
        print("  Adam (n={}), budget={:.1f}s".format(len(best_seq), dl - time.monotonic()))
        p_best, sc_best = _adam_optimize(best_seq, dl, lr=0.03, top_k=5, max_iter=100000)
        _update(p_best, sc_best, "adam1")

        if time.monotonic() < phase4_end - 20:
            dl2 = min(phase4_end - 5, time.monotonic() + min(100.0, (phase4_end - time.monotonic()) * 0.5))
            p_best2, sc_best2 = _adam_optimize(best_seq, dl2, lr=0.01, top_k=10, max_iter=50000)
            _update(p_best2, sc_best2, "adam2")

    print("  Phase 4 done: best={:.10f}".format(best_score))

    # === Phase 5: L-BFGS smooth ===
    print("=== Phase 5: L-BFGS ===")
    phase5_end = min(deadline, t_start + 1400.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        remaining = phase5_end - time.monotonic()
        dl = min(phase5_end, time.monotonic() + min(150.0, remaining * 0.7))
        print("  L-BFGS (n={}), budget={:.1f}s".format(len(best_seq), dl - time.monotonic()))
        p_best, sc_best = _lbfgs_smooth(best_seq, dl, top_k=5)
        _update(p_best, sc_best, "lbfgs")

    print("  Phase 5 done: best={:.10f}".format(best_score))

    # === Phase 6: Multi-n exploration ===
    print("=== Phase 6: Multi-n ===")
    phase6_end = min(deadline, t_start + 1600.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        n_curr = len(best_seq)
        target_ns = []
        if n_curr != 2048:
            target_ns.append(2048)
        if n_curr != 8192:
            target_ns.append(8192)
        if n_curr != 1024:
            target_ns.append(1024)

        for tn in target_ns:
            if time.monotonic() > phase6_end - 20:
                break
            p_int = _interp_profile(best_seq, tn)
            sc_int = _score_norm(p_int)
            _update(p_int, sc_int, "interp_n{}".format(tn))

            remaining = phase6_end - time.monotonic()
            if remaining < 20:
                break
            per_opt = min(80.0, remaining * 0.5)
            dl = min(phase6_end - 5, time.monotonic() + per_opt)
            p_best, sc_best = _newton_optimize(p_int, dl, top_k_search=1)
            _update(p_best, sc_best, "newton_n{}".format(tn))

            if time.monotonic() < phase6_end - 10:
                dl2 = min(phase6_end - 5, time.monotonic() + min(40.0, (phase6_end - time.monotonic()) * 0.4))
                p_best2, sc_best2 = _adam_optimize(p_best, dl2, lr=0.02, top_k=5, max_iter=30000)
                _update(p_best2, sc_best2, "adam_n{}".format(tn))

    print("  Phase 6 done: best={:.10f}".format(best_score))

    # === Phase 7: Final intensive Newton on global best ===
    print("=== Phase 7: Final Newton polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - 10 - time.monotonic()
        if remaining > 10:
            dl = min(deadline - 5, time.monotonic() + min(60.0, remaining * 0.6))
            p_best, sc_best = _newton_optimize(best_seq, dl, top_k_search=1)
            _update(p_best, sc_best, "final_newton")

        if time.monotonic() < deadline - 10:
            dl2 = min(deadline - 5, time.monotonic() + min(30.0, (deadline - 5 - time.monotonic()) * 0.5))
            p_best2, sc_best2 = _adam_optimize(best_seq, dl2, lr=0.005, top_k=3, max_iter=20000)
            _update(p_best2, sc_best2, "final_adam")

    # === Final validation ===
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_power_profile(4096, 0.45)
            s = final_seq.sum()
        final_score = _score_general(final_seq)
        print("  FINAL: score_general={:.10f}, n={}".format(final_score, len(final_seq)))

        # Scale to use full available magnitude (up to 1000 per element)
        # The objective is scale-invariant, but let's keep it reasonable
        max_el = final_seq.max()
        if max_el > 990.0:
            final_seq = final_seq * (990.0 / max_el)

        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 16384:
                sc_eval = float(evaluate_sequence(seq_list))
                print("  evaluate_sequence: {:.10f}".format(sc_eval))
        except Exception as e:
            print("  eval check failed: {}".format(e))

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_power_profile(4096, 0.45)
    print("  FALLBACK: n=4096, score={:.10f}".format(_score_norm(fallback)))
    return [float(x) for x in fallback]
