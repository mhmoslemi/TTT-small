import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):

    return a / s

    return np.where(c >= cm - epsilon)[0]

    """Project v onto the probability simplex {x: x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:

    u = np.sort(v)[::-1]
    csv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (csv - 1))[0][-1]
    theta = (csv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

    s = a_new.sum()
    if s < 1e-300:

    else:

"""
Optimized step function sequence minimizer using exact subgradient methods,
mirror descent (multiplicative updates), adaptive Polyak step sizes,
and multi-phase refinement with progressive step size reduction.
Key innovations over previous approach:
1. Exact subgradient (no log-sum-exp smoothing) for precise optimization
2. FFT-based fast subgradient computation using cross-correlation trick
3. Multi-phase step size schedule with aggressive cooling
4. Nesterov momentum in early phases, none in late phases
5. Active set tracking with adaptive epsilon
6. Symmetry exploitation for final polishing
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _convolve_max(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(c.max()), c


def _normalize(a):
    s = float(a.sum())
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def _subgradient_fast(a, active_indices):
    """
    Fast subgradient computation using FFT cross-correlation.
    Returns (2/|S|) * sum_{k in S} a_{k-j} for j=0..n-1.
    """
    n = len(a)
    S = len(active_indices)
    if S == 0:
        return np.zeros(n)

    b = np.zeros(2 * n - 1)
    b[active_indices] = 1.0 / S

    a_rev = a[::-1]
    conv = _full_convolve(b, a_rev)
    h = conv[n - 1:2 * n - 1]

    return 2.0 * h


def _subgradient_direct(a, active_indices):
    """Direct subgradient computation (for small active sets)."""
    n = len(a)
    S = len(active_indices)
    if S == 0:
        return np.zeros(n)

    g = np.zeros(n)
    for k in active_indices:
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            g[j_lo:j_hi + 1] += a[k - np.arange(j_lo, j_hi + 1)]
    g *= 2.0 / S
    return g


def _mirror_step(a, g, eta):
    """Multiplicative (mirror descent) update step."""
    log_a = np.log(np.maximum(a, 1e-300))
    new_log = log_a - eta * g
    new_log -= new_log.max()
    a_new = np.exp(new_log)
    s = a_new.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones(len(a)) / len(a)
    return a_new / s


def _mirror_step_momentum(a, g, eta, momentum, momentum_decay=0.0):
    """Mirror descent with momentum in log-space."""
    log_a = np.log(np.maximum(a, 1e-300))
    momentum -= eta * g
    new_log = log_a + momentum
    new_log -= new_log.max()
    a_new = np.exp(new_log)
    s = a_new.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones(len(a)) / len(a), momentum
    return a_new / s, momentum

    if epsilon <= 0:

    return np.where(c >= cm - epsilon)[0]

    """

    best_val = _convolve_max(a)[0]

    if n_max_iter is None:

    momentum = np.zeros(n) if use_momentum else None

    c, cm = _convolve_max(a)
    active = _get_active_set(c, cm, active_epsilon_frac * cm)

    total = n_max_iter

        # Adaptive step size (geometric interpolation from eta_init to eta_final)

    c, cm = _convolve_max(a)
    active = _get_active_set(c, cm, active_epsilon_frac * cm)
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break

        if it % recompute_interval == 0:
            c, cm = _convolve_max(a)
            if cm < best_val:
                best_val = cm
                best_a = a.copy()
            active = _get_active_set(c, cm, active_epsilon_frac * cm)
            if len(active) == 0:
                active = np.array([int(np.argmax(c))])

        if len(active) <= 10:
            g = _subgradient_direct(a, active)
        else:
            g = _subgradient_fast(a, active)

        if not np.all(np.isfinite(g)):
            continue

        a = _mirror_step(a, g, eta)

    cm_final = _convolve_max(a)[0]
    if cm_final < best_val:
        best_val = cm_final
        best_a = a.copy()

    return best_a, best_val


def _optimize_with_restarts(a_init, deadline, eta_init, eta_final,
                            n_restarts=3, active_epsilon_frac=0.0):
    """
    Multiple restarts with progressively smaller eta.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_val = _convolve_max(a)[0]

    rng = np.random.default_rng(12345)

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break

        # Progressively smaller step sizes
        eta_r_init = eta_init * (0.3 ** r)
        eta_r_final = eta_final * (0.3 ** r)

        time_budget = (deadline - time.monotonic()) / (n_restarts - r)
        sub_deadline = time.monotonic() + time_budget

        a_r, val_r = _optimize_mirror(
            a, sub_deadline,
            eta_init=max(1e-8, eta_r_init),
            eta_final=max(1e-10, eta_r_final),
            use_momentum=(r == 0),
            momentum_init=0.5,
            momentum_final=0.0,
            active_epsilon_frac=active_epsilon_frac,
            recompute_interval=200,
            n_max_iter=int(1e6 * time_budget)
        )

        if val_r < best_val:
            best_val = val_r
            best_a = a_r.copy()
            a = a_r.copy()
        else:
            # Small perturbation for next restart
            noise = 0.01 / (r + 1)
            log_a = np.log(np.maximum(a, 1e-300))
            log_a += noise * rng.normal(size=n)
            a = _normalize(np.exp(log_a - log_a.max()))

    return best_a, best_val


def _symmetrize(a):
    """Make sequence symmetric: a[i] = (a[i] + a[n-1-i]) / 2, then renormalize."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    return _normalize(a_sym)


def _try_symmetric_optimize(a_init, deadline):
    """Optimize in the symmetric subspace."""
    n = len(a_init)
    # Work with only the first half
    half = (n + 1) // 2

    # Create symmetric initial
    a_sym = _symmetrize(a_init)

    # Define a wrapper that operates on the half-vector
    # For simplicity, just run standard optimization on the symmetric sequence
    # and re-symmetrize periodically
    a = a_sym.copy()
    best_a = a.copy()
    best_val = _convolve_max(a)[0]

    c, cm = _convolve_max(a)
    active = _get_active_set(c, cm, 0.0)
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])

    eta = 0.001
    for it in range(500000):
        if time.monotonic() > deadline:
            break

        if it % 100 == 0:
            c, cm = _convolve_max(a)
            if cm < best_val:
                best_val = cm
                best_a = a.copy()
            active = _get_active_set(c, cm, 0.0)
            if len(active) == 0:
                active = np.array([int(np.argmax(c))])

        if len(active) <= 10:
            g = _subgradient_direct(a, active)
        else:
            g = _subgradient_fast(a, active)

        if not np.all(np.isfinite(g)):
            continue

        a = _mirror_step(a, g, eta)
        # Re-symmetrize every 10 steps
        if it % 10 == 0:
            a = _symmetrize(a)

        # Decrease eta slowly
        if it % 5000 == 0 and eta > 1e-6:
            eta *= 0.95

    cm_final = _convolve_max(a)[0]
    if cm_final < best_val:
        best_val = cm_final
        best_a = a.copy()

    return best_a, best_val


def _polyak_step_size(a, cm, g, cm_low=0.0):
    """Polyak step size: (f(a) - f_low) / ||g||^2"""
    g_norm_sq = float(np.dot(g, g))
    if g_norm_sq < 1e-30:
        return 0.0
    return (cm - cm_low) / g_norm_sq


def _optimize_polyak(a_init, deadline, polyak_cap=0.5, n_max_iter=200000):
    """Optimization with Polyak step size."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_val = _convolve_max(a)[0]

    c, cm = _convolve_max(a)
    active = _get_active_set(c, cm, 0.0)
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])

    # Lower bound: use 0.75 * 2 (empirical)
    cm_low = 0.75

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break

        if it % 200 == 0:
            c, cm = _convolve_max(a)
            if cm < best_val:
                best_val = cm
                best_a = a.copy()
                cm_low = min(cm_low, best_val * 0.99)
            active = _get_active_set(c, cm, 0.0)
            if len(active) == 0:
                active = np.array([int(np.argmax(c))])

        if len(active) <= 10:
            g = _subgradient_direct(a, active)
        else:
            g = _subgradient_fast(a, active)

        if not np.all(np.isfinite(g)):
            continue

        eta = min(polyak_cap, _polyak_step_size(a, cm, g, cm_low))
        eta = max(eta, 1e-8)

        a = _mirror_step(a, g, eta)

    cm_final = _convolve_max(a)[0]
    if cm_final < best_val:
        best_val = cm_final
        best_a = a.copy()

    return best_a, best_val


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer using exact subgradient mirror descent with
    progressive step size reduction, Polyak step sizes, and symmetry exploitation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc_norm, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc_norm):
            s = float(p.sum())
            if s < 1e-15:
                return
            sc = 2.0 * len(p) * sc_norm / (s * s)
            if sc < best_score:
                best_seq = np.asarray(p, dtype=np.float64).copy()
                best_score = float(sc)
                elapsed = time.monotonic() - t_start
                print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc = _score(parent)
        print(f"Parent: n={n}, score={sc:.10f}, max_conv={_convolve_max(parent)[0]:.10f}")
    else:
        n = 4096
        parent = _make_linear_profile(n, 1.5)
        sc = _score(parent)
        print(f"No parent, using linear profile n={n}, score={sc:.10f}")

    # Normalize to sum=1 for optimization
    a_work = _normalize(parent)
    cm_work = _convolve_max(a_work)[0]
    print(f"Working point: sum=1, max_conv={cm_work:.10f}, score={2*n*cm_work:.10f}")

    # --- Phase 0: Try multiple initializations (30s) ---
    phase0_end = min(deadline, t_start + 30.0)
    n_use = n
    candidates = [(cm_work, a_work.copy(), "parent")]

    if time.monotonic() < phase0_end:
        for name, gen_fn in [
            ("sine", lambda: _make_sine_profile(n_use)),
            ("quad", lambda: _make_quadratic_profile(n_use, 0.8, 0.2)),
            ("cos2", lambda: _make_cos_profile(n_use, 2.0)),
            ("cos3", lambda: _make_cos_profile(n_use, 3.0, freq=2.0)),
            ("exp", lambda: _make_exp_profile(n_use, 1.5)),
            ("lin0.5", lambda: _make_linear_profile(n_use, 0.5)),
            ("lin2.0", lambda: _make_linear_profile(n_use, 2.0)),
        ]:
            if time.monotonic() > phase0_end:
                break
            p = gen_fn()
            cm_p = _convolve_max(p)[0]
            candidates.append((cm_p, p, name))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0: best initial = {candidates[0][2]} with max_conv={candidates[0][0]:.10f}")
    a_work = candidates[0][1].copy()
    cm_work = candidates[0][0]
    update_best(a_work, cm_work, f"init_{candidates[0][2]}")

    # --- Phase 1: Mirror descent with momentum, moderate eta (300s) ---
    phase1_end = min(deadline, t_start + 330.0)
    if time.monotonic() < phase1_end:
        print(f"Phase 1: Mirror descent with momentum (eta 0.1->0.01)")
        a_p1, cm_p1 = _optimize_mirror(
            a_work, phase1_end,
            eta_init=0.1, eta_final=0.01,
            use_momentum=True, momentum_init=0.5, momentum_final=0.1,
            active_epsilon_frac=0.01,
            recompute_interval=500,
            n_max_iter=500000
        )
        update_best(a_p1, cm_p1, "phase1_mirror_mom")
        if cm_p1 < cm_work:
            a_work = a_p1.copy()
            cm_work = cm_p1

    print(f"Phase 1 done: score={2*n*cm_work:.10f}")

    # --- Phase 2: Mirror descent without momentum, decreasing eta (400s) ---
    phase2_end = min(deadline, t_start + 730.0)
    if time.monotonic() < phase2_end:
        print(f"Phase 2: Mirror descent, eta 0.01->0.001")
        a_p2, cm_p2 = _optimize_mirror(
            a_work, phase2_end,
            eta_init=0.01, eta_final=0.001,
            use_momentum=False,
            active_epsilon_frac=0.005,
            recompute_interval=300,
            n_max_iter=1000000
        )
        update_best(a_p2, cm_p2, "phase2_mirror")
        if cm_p2 < cm_work:
            a_work = a_p2.copy()
            cm_work = cm_p2

    print(f"Phase 2 done: score={2*n*cm_work:.10f}")

    # --- Phase 3: Fine polishing with very small eta (400s) ---
    phase3_end = min(deadline, t_start + 1130.0)
    if time.monotonic() < phase3_end:
        print(f"Phase 3: Fine polishing (eta 0.001->0.0001)")
        a_p3, cm_p3 = _optimize_mirror(

            a_work = a_p3.copy()
            cm_work = cm_p3

    print(f"Phase 3 done: score={2*n*cm_work:.10f}")

    phase4_end = min(deadline, t_start + 1430.0)
    if time.monotonic() < phase4_end:
        print(f"Phase 4: Ultra-fine polishing (eta=0.0001, exact active set)")
        a_p4, cm_p4 = _optimize_fine(
            a_work, phase4_end,

            n_max_iter=5000000,
            active_epsilon_frac=0.0,
            recompute_interval=50
        )
        update_best(a_p4, cm_p4, "phase4_ultra")
        if cm_p4 < cm_work:
            a_work = a_p4.copy()
            cm_work = cm_p4

    print(f"Phase 4 done: score={2*n*cm_work:.10f}")

    # --- Phase 5: Even finer with smaller eta (200s) ---
    phase5_end = min(deadline, t_start + 1630.0)
    if time.monotonic() < phase5_end:
        print(f"Phase 5: Ultra-ultra fine (eta=0.00002)")
        a_p5, cm_p5 = _optimize_fine(
            a_work, phase5_end,
            eta=0.00002,
            n_max_iter=10000000,
            active_epsilon_frac=0.0,
            recompute_interval=30
        )
        update_best(a_p5, cm_p5, "phase5_uultra")
        if cm_p5 < cm_work:
            a_work = a_p5.copy()
            cm_work = cm_p5

    print(f"Phase 5 done: score={2*n*cm_work:.10f}")

    # --- Phase 6: Symmetry exploitation (100s) ---
    phase6_end = min(deadline, t_start + min(1730.0, deadline - t_start))
    if time.monotonic() < phase6_end - 30:
        print(f"Phase 6: Symmetric optimization")
        a_p6, cm_p6 = _try_symmetric_optimize(a_work, phase6_end - 30)
        update_best(a_p6, cm_p6, "phase6_sym")
        if cm_p6 < cm_work:
            a_work = a_p6.copy()
            cm_work = cm_p6

    # --- Phase 7: Final polish with remaining time ---
    if best_seq is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            print(f"Phase 7: Final polish ({remaining:.0f}s remaining)")
            eta_final = 0.00001 if remaining > 50 else 0.00005
            a_p7, cm_p7 = _optimize_fine(
                a_work, deadline,
                eta=eta_final,
                n_max_iter=int(remaining * 10000),
                active_epsilon_frac=0.0,

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
            # Also try at sum = sqrt(2n) which might give slightly different result
            # due to floating point in evaluate_sequence
        else:
            final_seq = _make_linear_profile(2048, 1.5)

        sc_final = _score(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, max_conv={_convolve_max(_normalize(final_seq))[0]:.10f}")

        try:

            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned inf, using internal: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_linear_profile(2048, 1.5)
    print(f"FALLBACK: n=2048, score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
