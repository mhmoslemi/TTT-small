```python
"""
Search for non-negative step-function coefficients minimizing
2*n*max(conv(a,a))/sum(a)^2.

The algorithm combines:
  - a linearized local LP direction on the probability simplex,
  - short L-BFGS polishing with a log-sum-exp smoothed max convolution,
  - Adam-based beta-continuation optimization for robust refinement,
  - multi-restart perturbations and larger-size interpolation.
"""
import numpy as np
import time
from scipy.optimize import linprog, minimize


_EPS = 1e-15


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _conv(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    out_n = len(a) + len(b) - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    c = _conv(a, a)
    c = np.maximum(c, 0.0)
    return float(2.0 * a.size * float(np.max(c)) / (s * s))


def _norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(64, dtype=np.float64) / 64.0
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < _EPS:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    return a / s


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return np.ones(64, dtype=np.float64) / 64.0
    z = z - float(np.mean(z))
    z = np.clip(z, -100.0, 100.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s < 1e-300:
        return np.ones_like(z, dtype=np.float64) / float(z.size)
    return e / s


def _logit(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 1e-15)
    z = np.log(a)
    if z.size == 0:
        return np.zeros(64, dtype=np.float64)
    return z - float(np.mean(z))


def _cross_w_a(w, a):
    """Compute h[m] = sum_j w[j+m] a[j] for m=0,...,len(a)-1."""
    n = len(a)
    full = _conv(w, a[::-1])
    return full[n - 1:2 * n - 1]


def _softmax_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    c = np.where(np.isfinite(c), c, 0.0)
    cmax = float(np.max(c))
    if cmax <= 0.0:
        w = np.ones_like(c, dtype=np.float64) / float(c.size)
        return w
    if beta <= 1e-15:
        w = np.zeros_like(c, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    x = (c - cmax) / beta
    x = np.where(x > 0.0, 0.0, x)
    e = np.exp(x)
    s = float(e.sum())
    if not np.isfinite(s) or s < 1e-300:
        w = np.zeros_like(c, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _smooth_obj(z, beta):
    a = _softmax(z)
    c = _conv(a, a)
    c = np.maximum(c, 0.0)
    cmax = float(np.max(c))
    if beta <= 1e-15:
        return float(2.0 * len(a) * cmax)
    x = (c - cmax) / beta
    x = np.where(x > 0.0, 0.0, x)
    e = np.exp(x)
    s = float(e.sum())
    if not np.isfinite(s) or s < 1e-300:
        return float(2.0 * len(a) * cmax)
    return float(2.0 * len(a) * beta * (cmax + np.log(s)))


def _smooth_grad(z, beta):
    n = len(z)
    if n == 0:
        return np.zeros(64, dtype=np.float64)
    a = _softmax(z)
    c = _conv(a, a)
    c = np.maximum(c, 0.0)
    w = _softmax_weights(c, beta)
    h = _cross_w_a(w, a)
    W = float(np.dot(w, c))
    grad = 4.0 * float(n) * a * (h - W)
    grad = grad - float(np.mean(grad))
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n, dtype=np.float64)
    return grad


def _lp_linearized_direction(a, c, max_constraints, time_limit=None):
    """
    LP direction from first-order model:
      minimize t
      s.t. 2 dot(a_{k-m}, b) - t <= c_k  for selected k
           sum(b)=1, b>=0.
    """
    a = _norm(a)
    c = np.asarray(c, dtype=np.float64)
    c = np.where(np.isfinite(c), c, 0.0)
    n = len(a)
    total = len(c)
    if n == 0 or total == 0:
        return None, np.inf

    p = int(max_constraints)
    p = max(8, p)
    if p > total:
        p = total

    if p >= total:
        k_idx = np.arange(total, dtype=np.int64)
    else:
        k_idx = np.argpartition(c, total - p)[-p:]
        k_idx = np.asarray(k_idx, dtype=np.int64)

    A = np.zeros((p, n + 1), dtype=np.float64)
    A[:, n] = -1.0

    for r, kk in enumerate(k_idx):
        kk = int(kk)
        lo = max(0, kk - (n - 1))
        hi = min(n - 1, kk)
        if lo <= hi:
            idxs = kk - np.arange(lo, hi + 1, dtype=np.int64)
            A[r, lo:hi + 1] = 2.0 * a[idxs]

    obj = np.zeros(n + 1, dtype=np.float64)
    obj[-1] = 1.0

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    bounds = [(0.0, None) for _ in range(n)] + [(-np.inf, None)]
    b_ub = c[k_idx].astype(np.float64)

    options = {}
    if time_limit is not None:
        options["time_limit"] = float(max(1.0, min(180.0, time_limit)))

    try:
        res = linprog(
            obj,
            A_ub=A,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options=options,
        )
    except Exception:
        try:
            res = linprog(
                obj,
                A_ub=A,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
                options={},
            )
        except Exception:
            return None, np.inf

    if res.success:
        b = res.x[:n]
        t_val = float(res.x[-1])
        if b is not None and np.all(np.isfinite(b)):
            b = np.maximum(b, 0.0)
            s = float(b.sum())
            if s > _EPS:
                return b / s, t_val

    return None, np.inf


def _lp_linearized_update(a, c=None, max_constraints=2048, time_limit=None):
    a = _norm(a)
    base = _score(a)
    if not np.isfinite(base):
        return a, base, False, np.inf

    n = len(a)
    if c is None:
        c = _conv(a, a)
    else:
        c = _norm(c)
        c = np.asarray(c, dtype=np.float64)
    c = np.maximum(c, 0.0)

    current_peak = base / (2.0 * float(n))
    b, pred = _lp_linearized_direction(a, c, max_constraints, time_limit)
    if b is None:
        return a, base, False, pred

    if pred >= current_peak - 1e-12:
        return a, base, False, pred

    best = a
    bests = base
    alphas = [0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.75, 1.0]

    ae = float(np.clip((current_peak - pred) / max(1e-12, current_peak), 0.0, 1.0))
    if 0.0 < ae < 1.0:
        alphas.append(ae)
        alphas.append(max(0.01, 0.5 * ae))
        alphas.append(min(1.0, 2.0 * ae))
    alphas = sorted(set(float(x) for x in alphas))

    for alpha in alphas:
        new = (1.0 - alpha) * a + alpha * b
        new = np.maximum(new, 0.0)
        s = float(new.sum())
        if s <= _EPS:
            continue
        new = new / s
        sc = _score(new)
        if np.isfinite(sc) and sc < bests:
            best = new
            bests = sc

    return best, float(bests), bool(bests < base - 1e-14), float(pred)


def _lbfgs_chunks(a, deadline, betas, maxiter_per_chunk=100, max_chunks_per_beta=3, maxls=15):
    a = _norm(a)
    best = a.copy()
    bests = _score(a)
    z_best = _logit(best)

    for beta in betas:
        if time.monotonic() >= deadline:
            break
        z = z_best.copy()
        chunks = 0
        while time.monotonic() < deadline - 5.0:
            rem = deadline - time.monotonic()
            if rem < 8.0:
                break

            mi = int(maxiter_per_chunk)
            if rem < 20.0:
                mi = min(mi, max(10, mi // 2))
            if rem < 10.0:
                mi = min(mi, 10)

            try:
                res = minimize(
                    _smooth_obj,
                    z,
                    jac=_smooth_grad,
                    args=(float(beta),),
                    method="L-BFGS-B",
                    options={
                        "maxiter": int(mi),
                        "ftol": 1e-12,
                        "gtol": 1e-6,
                        "maxls": int(min(20, maxls)),
                    },
                )
            except Exception:
                break

            if res.x is None:
                break
            znew = np.asarray(res.x, dtype=np.float64)
            if znew.size != len(z) or not np.all(np.isfinite(znew)):
                break

            znew = znew - float(np.mean(znew))
            znew = np.clip(znew, -100.0, 100.0)

            a_new = _softmax(znew)
            sc = _score(a_new)
            if np.isfinite(sc):
                if sc < bests:
                    bests = float(sc)
                    best = a_new.copy()
                    z_best = _logit(best)
                if sc <= bests * 1.02 + 1e-12:
                    z = znew
                else:
                    z = z_best
            else:
                z = z_best

            chunks += 1
            if chunks >= int(max_chunks_per_beta):
                break

    return best, float(bests)


def _adam_optimize(a_init, deadline, beta_schedule, lr=0.01, max_iter_total=200000, eval_every=50):
    a = _norm(a_init)
    n = len(a)
    if n == 0:
        return a, np.inf

    z = _logit(a)
    best = a.copy()
    bests = _score(a)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    current_lr = float(lr)

    for beta in beta_schedule:
        if time.monotonic() >= deadline:
            break
        beta = float(beta)
        it_phase = 0
        while it_phase < int(max_iter_total) and time.monotonic() < deadline - 3.0:
            a = _softmax(z)
            c = _conv(a, a)
            c = np.maximum(c, 0.0)
            w = _softmax_weights(c, beta)
            h = _cross_w_a(w, a)
            W = float(np.dot(w, c))
            grad = 4.0 * float(n) * a * (h - W)
            grad = grad - float(np.mean(grad))
            if not np.all(np.isfinite(grad)):
                break

            t += 1
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * (grad * grad)
            mh = m / (1.0 - b1 ** t)
            vh = v / (1.0 - b2 ** t)

            z = z - current_lr * mh / (np.sqrt(vh) + eps)
            z = z - float(np.mean(z))
            z = np.clip(z, -100.0, 100.0)

            if t % int(eval_every) == 0:
                a_cur = _softmax(z)
                sc = _score(a_cur)
                if np.isfinite(sc) and sc < bests:
                    bests = float(sc)
                    best = a_cur.copy()

            if it_phase % 2000 == 0:
                current_lr = max(0.001, current_lr * 0.98)

            it_phase += 1

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if np.isfinite(sc_final) and sc_final < bests:
        best = a_final
        bests = sc_final

    return best, float(bests)


def _lp_lbfgs_phase(a, deadline, max_constraints, beta_frac, lp_time_limit, lbfgs_iter, lbfgs_chunks):
    a = _norm(a)
    best = a.copy()
    bests = _score(a)
    n = len(a)
    total = 2 * n - 1

    current_mc = int(min(total, max(512, max_constraints)))
    bad = 0
    round_no = 0

    while time.monotonic() < deadline - 20.0:
        round_no += 1
        c = _conv(best, best)
        c = np.maximum(c, 0.0)
        M = float(np.max(c))

        tl = lp_time_limit
        if current_mc > max(2048, n):
            tl = max(tl, 45.0)

        new, sc, imp, pred = _lp_linearized_update(best, c, current_mc, tl)
        if imp:
            best = new
            bests = float(sc)
            bad = 0
            print(f"    LP round {round_no}: score={sc:.10f} pred={pred:.6e} mc={current_mc}")
        else:
            bad += 1
            if bad == 3 and current_mc < total:
                current_mc = min(total, max(1000, 2 * current_mc))
                bad = 0
            elif bad > 15:
                break

        now = time.monotonic()
        polish_end = min(deadline, now + 30.0)
        if polish_end > now + 10.0:
            beta = max(1e-8, float(beta_frac) * M)
            a2, sc2 = _lbfgs_chunks(
                best,
                polish_end,
                [beta],
                maxiter_per_chunk=max(20, int(min(80, lbfgs_iter))),
                max_chunks_per_beta=max(1, int(lbfgs_chunks)),
                maxls=10,
            )
            if sc2 < bests:
                best = a2
                bests = float(sc2)
                print(f"    LBFGS polish round {round_no}: score={sc2:.10f}")

    return best, float(bests)


def _make_profile(n, kind):
    n = int(max(2, n))
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    if kind == "uniform":
        a = np.ones(n, dtype=np.float64)
    elif kind == "sinusoidal":
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * t)
    elif kind == "bimodal":
        a = 1.0 + 0.35 * np.cos(4.0 * np.pi * t)
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.7
    elif kind == "gaussian":
        a = np.exp(-15.0 * (t - 0.5) ** 2)
    elif kind == "edge":
        a = 1.0 / (1.0 + 30.0 * (t - 0.5) ** 2)
    else:
        a = np.ones(n, dtype=np.float64)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp(a, n_new):
    a = _norm(a)
    n_old = len(a)
    n_new = int(max(2, n_new))
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, n_old, dtype=np.float64)
    x1 = np.linspace(0.0, 1.0, n_new, dtype=np.float64)
    y = np.interp(x1, x0, a)
    y = np.maximum(y, 1e-15)
    return y / y.sum()


def _perturb_white(a, sigma, rng):
    a = _norm(a)
    z = _logit(a) + float(sigma) * rng.standard_normal(len(a))
    return _softmax(z)


def _perturb_lowfreq(a, amp, rng):
    a = _norm(a)
    n = len(a)
    z = _logit(a)
    t = np.arange(n, dtype=np.float64)
    f = int(rng.integers(1, max(2, n // 32)))
    phase = float(rng.uniform(0.0, 2.0 * np.pi))
    z = z + float(amp) * np.cos(2.0 * np.pi * f * t / float(n) + phase)
    return _softmax(z)


def _perturb_avg_reverse(a, rng):
    a = _norm(a)
    b = a[::-1].copy()
    mix = 0.5 * a + 0.5 * b
    return _perturb_white(mix, 0.02, rng)


def propose_candidate(seed=42, budget_s=1800):
    t0 = time.monotonic()
    deadline = t0 + max(60.0, float(budget_s) - 60.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.clip(parent_arr, 0.0, 1000.0)
        if parent_arr.size > 0 and float(parent_arr.sum()) > 0.01:
            parent = parent_arr / float(parent_arr.sum())
    except Exception:
        parent = None

    if parent is None or len(parent) < 128:
        parent = _make_profile(2048, "uniform")
    if len(parent) > 16384:
        parent = _interp(parent, 16384)

    best = _norm(parent)
    best_score = _score(best)
    print(f"Initial: n={len(best)}, score={best_score:.10f}")

    # Quick profile scan at current size.
    n = len(best)
    for kind in ["uniform", "sinusoidal", "bimodal", "tapered", "gaussian", "edge"]:
        p = _make_profile(n, kind)
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best = p
            best_score = float(sc)
            print(f"  profile {kind}: score={best_score:.10f}")

    # Phase 1: LP linearized + short LBFGS polish, mostly for n <= 4096.
    phase1_end = min(deadline, t0 + 550.0)
    if len(best) <= 4096 and time.monotonic() < phase1_end - 60.0:
        print("Phase 1: LP + LBFGS refinement")
        best, best_score = _lp_lbfgs_phase(
            best,
            phase1_end,
            max_constraints=len(best) // 2,
            beta_frac=0.03,
            lp_time_limit=20.0,
            lbfgs_iter=70,
            lbfgs_chunks=1,
        )
        print(f"Phase 1 done: score={best_score:.10f}")

    # Phase 2: Adam beta-continuation for deeper local refinement.
    phase2_end = min(deadline, t0 + 1000.0)
    if time.monotonic() < phase2_end - 60.0:
        print("Phase 2: Adam continuation")
        n = len(best)
        cmax = max(1e-12, best_score / (2.0 * float(n)))
        fractions = [0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        beta_sched = [max(1e-9, cmax * float(f)) for f in fractions]
        if n <= 2048:
            lr = 0.012
        elif n <= 4096:
            lr = 0.008
        elif n <= 8192:
            lr = 0.005
        else:
            lr = 0.003
        best, best_score = _adam_optimize(
            best,
            phase2_end,
            beta_sched,
            lr=lr,
            max_iter_total=250000,
            eval_every=40,
        )
        print(f"Phase 2 done: score={best_score:.10f}")

    # Phase 3: multi-restart perturbations.
    phase3_end = min(deadline, t0 + 1350.0)
    if time.monotonic() < phase3_end - 90.0:
        print("Phase 3: multi-restart Adam")
        n = len(best)
        for i in range(14):
            if time.monotonic() >= phase3_end - 40.0:
                break

            if i == 0:
                cand = best.copy()
            elif i % 3 == 0:
                cand = _perturb_white(best, 0.06 + 0.02 * (i % 3), rng)
            elif i % 3 == 1:
                cand = _perturb_lowfreq(best, 0.05 + 0.01 * (i % 3), rng)
            else:
                cand = _perturb_avg_reverse(best, rng)

            sc = _score(cand)
            if not np.isfinite(sc) or sc > best_score * 1.75:
                continue

            cmaxc = max(1e-12, sc / (2.0 * float(n)))
            fractions_short = [0.1, 0.05, 0.02, 0.01, 0.005]
            beta_sched = [max(1e-9, cmaxc * float(f)) for f in fractions_short]
            if n <= 2048:
                lr = 0.008
            elif n <= 4096:
                lr = 0.006
            elif n <= 8192:
                lr = 0.004
            else:
                lr = 0.0025

            per_end = min(phase3_end, time.monotonic() + 70.0)
            cand2, sc2 = _adam_optimize(
                cand,
                per_end,
                beta_sched,
                lr=lr,
                max_iter_total=60000,
                eval_every=30,
            )
            if sc2 < best_score:
                best = cand2
                best_score = float(sc2)
                print(f"  restart {i}: score={best_score:.10f}")

    # Phase 4: try larger sizes.
    large_targets = []
    for target in [4096, 8192, 16384]:
        if time.monotonic() < deadline - 140.0 and len(best) < target:
            large_targets.append(target)

    for idx, target in enumerate(large_targets):
        if time.monotonic() >= deadline - 90.0:
            break

        print(f"Phase 4: trying n={target}")
        cand = _interp(best, target)
        sc = _score(cand)
        cand_best = cand
        cand_best_score = sc

        for kind in ["uniform", "tapered"]:
            p = _make_profile(target, kind)
            scp = _score(p)
            if np.isfinite(scp) and scp < cand_best_score:
                cand_best = p
                cand_best_score = scp

        if cand_best_score < best_score:
            best = cand_best
            best_score = float(cand_best_score)
            print(f"  n={target} initial score={best_score:.10f}")

        remaining_for_opt = [110.0, 80.0, 55.0]
        opt_time = float(remaining_for_opt[min(idx, len(remaining_for_opt) - 1)])
        opt_end = min(deadline - 60.0, time.monotonic() + opt_time)
        if opt_end > time.monotonic() + 40.0:
            n2 = len(cand_best)
            cmax = max(1e-12, cand_best_score / (2.0 * float(n2)))
            fractions = [0.1, 0.05, 0.02, 0.01, 0.005]
            beta_sched = [max(1e-9, cmax * float(f)) for f in fractions]
            if n2 <= 4096:
                lr = 0.008
            elif n2 <= 8192:
                lr = 0.005
            else:
                lr = 0.003
            cand_opt, sc_opt = _adam_optimize(
                cand_best,
                opt_end,
                beta_sched,
                lr=lr,
                max_iter_total=90000,
                eval_every=40,
            )

            if len(cand_opt) <= 8192 and time.monotonic() < deadline - 40.0:
                cmax2 = max(1e-12, sc_opt / (2.0 * float(len(cand_opt))))
                beta = max(1e-9, cmax2 * 0.02)
                mi = 70 if len(cand_opt) <= 4096 else 35
                lbfgs_end = min(deadline - 30.0, time.monotonic() + 25.0)
                cand_opt, sc_opt = _lbfgs_chunks(
                    cand_opt,
                    lbfgs_end,
                    [beta],
                    maxiter_per_chunk=mi,
                    max_chunks_per_beta=2,
                    maxls=12,
                )

            if sc_opt < best_score:
                best = cand_opt
                best_score = float(sc_opt)
                print(f"  optimized n={len(best)} score={best_score:.10f}")

    # Phase 5: final polish.
    if time.monotonic() < deadline - 30.0:
        print("Phase 5: final LBFGS polish")
        n = len(best)
        cmax = max(1e-12, best_score / (2.0 * float(n)))
        betas = [max(1e-10, cmax * float(f)) for f in [0.02, 0.01, 0.005, 0.002, 0.001]]
        if n <= 2048:
            mi = 120
        elif n <= 4096:
            mi = 70
        elif n <= 8192:
            mi = 40
        else:
            mi = 20
        best, best_score = _lbfgs_chunks(
            best,
            deadline,
            betas,
            maxiter_per_chunk=mi,
            max_chunks_per_beta=2,
            maxls=15,
        )

        if time.monotonic() < deadline - 25.0 and len(best) <= 4096:
            tl = max(5.0, deadline - time.monotonic() - 5.0)
            tc = min(15.0, tl)
            mc = min(2 * len(best) - 1, max(512, len(best) // 2))
            new, sc, imp, pred = _lp_linearized_update(best, None, mc, tc)
            if imp and sc < best_score:
                best = new
                best_score = float(sc)
                print(f"  final LP: score={best_score:.10f} pred={pred:.6e}")

    # Final safe upsample: often gives a small free improvement.
    for target in [4096, 8192, 16384]:
        if time.monotonic() >= deadline - 10.0:
            break
        if len(best) < target:
            cand = _interp(best, target)
            sc = _score(cand)
            if np.isfinite(sc) and sc < best_score:
                best = cand
                best_score = float(sc)
                print(f"  upsampled to n={target}: score={best_score:.10f}")

    final = _norm(best)
    final = np.clip(final, 0.0, 1000.0)
    final_score = _score(final)
    print(f"Final internal score: {final_score:.10f}, n={len(final)}")

    try:
        ev = evaluate_sequence([float(x) for x in final])
        print(f"Final evaluator score: {ev:.10f}")
        if not np.isfinite(ev):
            final = _make_profile(256, "uniform")
            final_score = _score(final)
    except Exception as exc:
        print(f"Final evaluator check failed: {exc}")

    if float(final.sum()) < 0.01:
        final = _make_profile(256, "uniform")

    return [float(x) for x in final]
```
