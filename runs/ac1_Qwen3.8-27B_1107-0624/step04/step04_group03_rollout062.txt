from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1

    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm_fn(a):
    """Score assuming sum(a) = 1 (scale-invariant)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm

def _gradient_smooth(z, beta):
    """Gradient of the smoothed objective w.r.t. z (log-space)."""
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

def _objective_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)

def _lbfgs_obj(z, beta, n):
    return _objective_smooth(z, beta)

def _lbfgs_grad(z, beta, n):
    return _gradient_smooth(z, beta)

def _run_lbfgs(a_init, deadline, max_wall_time=None, verbose=True):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if max_wall_time is None:
        max_wall_time = min(300.0, max(1.0, deadline - time.monotonic()))
    
    # More aggressive beta schedule with more levels
    beta_fracs = [0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002, 
                  0.0015, 0.001, 0.0007, 0.0005, 0.0003, 0.0002, 0.0001, 5e-5, 3e-5, 2e-5, 1e-5]
    betas = [b * max_c0 for b in beta_fracs]
    
    per_level = max_wall_time / len(betas)
    
    for i, beta in enumerate(betas):
        if time.monotonic() >= deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level)
        try:
            result = minimize(
                _lbfgs_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_grad,
                options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 3000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x.copy()
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new
                    if verbose:
                        elapsed = time.monotonic() - (deadline - max_wall_time)
                        print(f"    L-BFGS beta={beta:.2e}: score={best_score:.10f}")
        except Exception:
            pass
    
    return best_a, best_score

def _adam_optimize(a_init, deadline, lr=0.05, beta_frac=0.005, n_max_iter=200000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, beta_frac * max_c)
    
    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        
        if not np.all(np.isfinite(grad)):
            continue
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a = _softmax_fn(z)
        
        if t % 100 == 0:
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
            beta = max(1e-12, beta * 0.95)
            c_new = _full_convolve(a, a)
            max_c = float(c_new.max())
    
    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _nesterov_optimize(a_init, deadline, beta_fracs, lr_init=0.02, mom_init=0.9):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    
    lr = lr_init
    t = 0
    
    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-12, beta_frac * max_c)
        
        while time.monotonic() < deadline:
            t += 1
            grad_y = _gradient_smooth(y, beta)
            if not np.all(np.isfinite(grad_y)):
                grad_y = _gradient_smooth(z, beta)
            
            z_new = z - lr * grad_y
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -50, 50)
            
            y_new = z_new + mom_init * (z_new - z)
            y_new = y_new - y_new.mean()
            
            z = z_new
            y = y_new
            
            if t % 200 == 0:
                a_curr = _softmax_fn(z)
                sc = _score_norm_fn(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()
    
    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _active_set_polish(a_init, deadline, top_k=10):
    """Polish by focusing on the top-k peaks of the convolution."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    lr = 0.003
    momentum = np.zeros(n)
    mom_coeff = 0.95
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Use top-k peaks as active set
        k = min(top_k, len(c))
        top_indices = np.argsort(c)[-k:]
        w = np.zeros_like(c)
        w[top_indices] = 1.0 / k
        
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        
        momentum = mom_coeff * momentum - lr * grad
        z = z + momentum
        z = z - z.mean()
        z = np.clip(z, -50, 50)
        
        if it % 50 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            else:
                lr = lr * 0.8
    
    return best_a, best_score

def _make_profile(n, profile_type='exp_decay', param=1.0):
    t = np.linspace(0.0, 1.0, n)
    if profile_type == 'exp_decay':
        a = np.exp(-param * t)
    elif profile_type == 'exp_growth':
        a = np.exp(param * t)
    elif profile_type == 'linear':
        a = 1.0 + param * t
    elif profile_type == 'tilted_hann':
        hann = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = hann * np.exp(param * (t - 0.5))
    elif profile_type == 'tilted_cosine':
        base = 0.5 + 0.5 * np.cos(2 * np.pi * (t - 0.5))
        a = base * np.exp(param * (t - 0.5))
    elif profile_type == 'power':
        a = (t + 0.1) ** param
    elif profile_type == 'inv_power':
        a = (1.1 - t) ** param
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

def _coordinate_descent_polish(a_init, deadline, max_iter=10000):
    """Try coordinate-wise perturbations to find local improvements."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_fn(best_a)
    rng = np.random.default_rng(12345)
    
    # Precompute for efficiency
    a = best_a.copy()
    c = _full_convolve(a, a)
    cm = float(c.max())
    sum_a = float(a.sum())
    
    step = 0.01
    it = 0
    while time.monotonic() < deadline and it < max_iter:
        it += 1
        # Pick a random (or cycling) index
        i = it % n
        old_val = a[i]
        
        # Try increasing and decreasing
        for delta in [step, -step, step*0.5, -step*0.5, step*2, -step*2]:
            new_val = a[i] + delta * old_val
            if new_val < 0:
                continue
            a[i] = new_val
            c = _full_convolve(a, a)
            sc = 2.0 * n * c.max() / (a.sum()**2)
            if sc < best_score:
                best_score = sc
                cm = float(c.max())
                sum_a = float(a.sum())
            else:
                a[i] = old_val
        
        # Adaptively adjust step
        if it % 100 == 0:
            step = step * 0.99
    
    return best_a, best_score

    3. Top-k active set polish
    4. Coordinate descent for final refinement
    5. Diverse multi-restart strategy
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)

    def update_best(p, sc, label=""):

            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()

            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n = len(parent)
        print(f"Parent: n={n}, score={_score_fn(parent):.10f}")
    else:

    phase0_end = min(deadline, t_start + 30.0)
    candidates = []

    # Sort by score

"""
Multi-phase optimizer for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses softmax parameterization, smoothed max (log-sum-exp), adaptive L-BFGS
continuation, Adam polishing, active-set equalization, and multi-restart.
Key improvements: adaptive beta schedule, more L-BFGS iterations, active-set
balancing, diverse initializations, and optional length exploration.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _gradient_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _objective_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_obj(z, beta, n):
    return _objective_smooth(z, beta)


def _lbfgs_grad(z, beta, n):
    return _gradient_smooth(z, beta)


def _run_lbfgs_adaptive(a_init, deadline, max_wall_time=None, verbose=True):
    """L-BFGS with adaptive beta: decrease beta only upon convergence."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if max_wall_time is None:
        max_wall_time = max(1.0, deadline - time.monotonic())

    # Adaptive continuation: start with large beta, decrease upon convergence
    beta = 0.1 * max_c0
    min_beta = 1e-8 * max_c0
    consecutive_non_improvements = 0
    max_levels = 60

    level = 0
    while time.monotonic() < deadline and level < max_levels:
        if beta < min_beta:
            break
        level += 1
        # Limit time per level
        remaining = deadline - time.monotonic()
        level_time = min(remaining, max(max_wall_time / (max_levels - level + 1), 2.0))
        level_dl = time.monotonic() + level_time

        try:
            result = minimize(
                _lbfgs_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_grad,
                options={'maxiter': 800, 'ftol': 1e-16, 'gtol': 1e-11,
                         'maxls': 50, 'maxfun': 4000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x.copy()
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score - 1e-14:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new
                    consecutive_non_improvements = 0
                else:
                    consecutive_non_improvements += 1

                if verbose and level % 5 == 0:
                    print(f"    L-BFGS level {level}, beta={beta:.2e}: score={best_score:.10f}")
        except Exception:
            pass

        # Adaptive beta reduction
        if consecutive_non_improvements >= 2:
            beta *= 0.3
            consecutive_non_improvements = 0
        else:
            beta *= 0.5

        # Update max_c reference
        c_curr = _full_convolve(best_a, best_a)
        max_c0 = float(c_curr.max())
        min_beta = 1e-8 * max_c0

    return best_a, best_score


def _adam_optimize(a_init, deadline, lr=0.03, beta_frac=0.005, n_max_iter=200000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, beta_frac * max_c)
    current_lr = lr

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()

        if t % 100 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            # Adaptive beta decay
            beta = max(1e-12, beta * 0.93)
            c_new = _full_convolve(a_curr, a_curr)
            max_c = float(c_new.max())
            # Slow LR decay
            current_lr = max(0.002, lr * (0.999 ** (t // 100)))

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _nesterov_optimize(a_init, deadline, beta_fracs, lr_init=0.015, mom_init=0.9):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())

    lr = lr_init
    t = 0

    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-12, beta_frac * max_c)

        while time.monotonic() < deadline:
            t += 1
            grad_y = _gradient_smooth(y, beta)
            if not np.all(np.isfinite(grad_y)):
                grad_y = _gradient_smooth(z, beta)

            z_new = z - lr * grad_y
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -50, 50)

            y_new = z_new + mom_init * (z_new - z)
            y_new = y_new - y_new.mean()

            z = z_new
            y = y_new

            if t % 200 == 0:
                a_curr = _softmax_fn(z)
                sc = _score_norm_fn(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _active_set_polish(a_init, deadline, top_k=10):
    """Polish by focusing gradient on the top-k peaks."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    lr = 0.002
    momentum = np.zeros(n)
    mom_coeff = 0.95

    it = 0
    while time.monotonic() < deadline:
        it += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        k = min(top_k, len(c))
        top_indices = np.argsort(c)[-k:]
        w = np.zeros_like(c)
        w[top_indices] = 1.0 / k

        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        momentum = mom_coeff * momentum - lr * grad
        z = z + momentum
        z = z - z.mean()
        z = np.clip(z, -50, 50)

        if it % 50 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            else:
                lr = lr * 0.8

    return best_a, best_score


def _equalization_phase(a_init, deadline):
    """Try to equalize the top peaks of the convolution."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    lr = 0.001
    momentum = np.zeros(n)
    mom_coeff = 0.97

    it = 0
    while time.monotonic() < deadline:
        it += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Use a very narrow active set: only the top peak
        top_idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[top_idx] = 1.0

        # Also include near-max values for stability
        near_max = c >= cm * 0.9995
        if np.sum(near_max) > 1:
            w = np.zeros_like(c)
            w[near_max] = 1.0 / np.sum(near_max)

        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        momentum = mom_coeff * momentum - lr * grad
        z = z + momentum
        z = z - z.mean()
        z = np.clip(z, -50, 50)

        if it % 30 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            elif it % 100 == 0:
                lr = lr * 0.9

    return best_a, best_score


def _make_profile(n, profile_type='exp_decay', param=1.0):
    t = np.linspace(0.0, 1.0, n)
    if profile_type == 'exp_decay':
        a = np.exp(-param * t)
    elif profile_type == 'exp_growth':
        a = np.exp(param * t)
    elif profile_type == 'linear':
        a = 1.0 + param * t
    elif profile_type == 'tilted_hann':
        hann = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = hann * np.exp(param * (t - 0.5))
    elif profile_type == 'tilted_blackman':
        x = 2.0 * t - 1.0
        blackman = 0.42 + 0.5 * np.cos(np.pi * x) + 0.08 * np.cos(2 * np.pi * x)
        a = np.maximum(blackman, 0.01) * np.exp(param * (t - 0.5))
    elif profile_type == 'power':
        a = (t + 0.05) ** param
    elif profile_type == 'inv_power':
        a = (1.05 - t) ** param
    elif profile_type == 'gaussian_tilt':
        a = np.exp(-4.0 * (t - 0.5) ** 2) * np.exp(param * (t - 0.5))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
