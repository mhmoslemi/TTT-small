    h = np.fft.irfft(fw * fp_rev, N)
    return h[n - 1:2 * n - 1]


def _adam_smoothmax(p0, deadline, rng, beta_frac=0.08, beta_decay=0.95,
                    lr=0.05, no_imp_patience=200):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0 + 1e-15)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-4, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0

    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)
        g = _conv_partner_gradient(w, p)

        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z - float(np.max(z)))
        p = (e + 1e-12) / (e + 1e-12).sum()

        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        if t_adam % 20 == 0:
            beta = max(1e-5, beta * beta_decay)

        if no_imp > no_imp_patience:
            ns = 0.05 if best_score < 1.51 else 0.1
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z - float(np.max(z)))
            p = (e + 1e-12) / (e + 1e-12).sum()
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-4, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            no_imp = 0

    return best_p, best_score


def _lp_iteration(a, deadline, m_decrease=0.998, t_blend=0.3, max_iters=10000,
                  m_floor_frac=0.99):
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    M_min = best_score / (2.0 * n) * m_floor_frac

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        M_try = max(M * m_decrease, M_min)

        b, res = _lp_solve_max_sum(a, M_try)
        if b is None:
            b, res = _lp_solve_max_sum(a, M)
            if b is None:
                M = M * 1.01
                continue

        s_b = float(b.sum())
        if not np.isfinite(s_b) or s_b < 1e-10:
            break

        b_norm = b / s_b

        # Try several blend factors via line search
        best_local_a = a.copy()
        best_local_score = _score_norm(a)
        for t in [t_blend, t_blend * 0.5, t_blend * 1.5, 0.1]:
            a_try = (1.0 - t) * a + t * b_norm
            a_try = np.maximum(a_try, 1e-15)
            a_try = a_try / a_try.sum()
            sc_try = _score_norm(a_try)
            if sc_try < best_local_score:
                best_local_score = sc_try
                best_local_a = a_try.copy()

        if best_local_score < best_score - 1e-12:
            best_score = best_local_score
            best_a = best_local_a.copy()

        a = best_local_a
        c = _full_convolve(a)
        current_peak = float(c.max())
        M = max(current_peak, M_min)

        # Adaptive decrease: if we're making progress, decrease faster
        progress = 1.0 - (best_score / (2.0 * n * current_peak)) if current_peak > 0 else 0
        if progress > 0.01:
            m_decrease = min(0.9995, m_decrease + 0.0002)
        else:
            m_decrease = max(0.9999, m_decrease * 0.999)

    return best_a, best_score


def _dual_guided_refinement(a, deadline, max_iters=200):
    """Use LP dual to guide gradient steps toward binding constraints."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    c = _full_convolve(a)
    M = float(c.max())

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Solve LP to get dual variables
        b, res = _lp_solve_max_sum(a, M)
        if b is None:
            M *= 1.01
            continue

        # Extract dual variables (shadow prices) if available
        # Use the constraint tightness as proxy for dual importance
        ab_conv = _full_convolve(a)  # This is conv(a,a), not conv(a,b)
        # Actually we need conv(a, b)
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        fb = np.fft.rfft(b, N)
        ab_c = np.fft.irfft(fa * fb, N)[:2 * n - 1]
        ab_c = np.maximum(ab_c, 0.0)

        # Dual weights: how close each constraint is to binding
        binding = np.clip((M - ab_c) / (M + 1e-10), 0, 1)
        # Invert: constraints that are tight have high weight
        dual_w = 1.0 - binding
        dual_w = np.maximum(dual_w, 0)
        dw_sum = float(dual_w.sum())
        if dw_sum < 1e-10:
            dual_w = np.zeros(2 * n - 1)
            dual_w[int(np.argmax(ab_c))] = 1.0
        else:
            dual_w = dual_w / dw_sum

        # Gradient step using dual-guided weights
        g = _conv_partner_gradient(dual_w, a)
        dot = float(np.dot(a, g))
        direction = a * (g - dot)
        direction -= float(np.mean(direction))

        # Adaptive step size
        dir_norm = float(np.sqrt(np.sum(direction**2)))
        if dir_norm < 1e-15:
            break
        step = 0.3 * a.sum() / (n * dir_norm + 1e-10)

        a_new = a - step * direction
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)

        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        c = _full_convolve(a)
        M = float(c.max())

    return best_a, best_score


def _peak_flood_refined(a, deadline, eta_init=0.03, max_iters=3000):
    """Refined peak flooding with adaptive temperature and multi-scale peaks."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    eta = eta_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = float(c.max())
        score = float(2.0 * n * M)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        # Soft-max weights over near-peak positions
        beta = M * 0.05
        _, w = _smooth_max_weights(c, beta)

        # Compute gradient
        g = _conv_partner_gradient(w, a)
        dot = float(np.dot(a, g))
        heat = np.maximum(a * (g - dot), 0)
        heat_max = float(heat.max())
        if heat_max < 1e-15:
            break
        heat = heat / heat_max

        # Multiplicative update
        factor = 1.0 - eta * heat
        factor = np.maximum(factor, 0.01)
        a = a * factor
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        c = _full_convolve(a)

        if it % 200 == 0 and it > 0:
            if score >= best_score:
                eta = max(0.005, eta * 0.85)
            else:
                eta = min(0.1, eta * 1.1)

    return best_a, best_score


def _make_ramp(n, p=1.0):
    t = np.arange(1, n + 1, dtype=np.float64)
    a = t ** p
    return a / a.sum()


def _make_cosine_asym(n, freq=1, amp=0.7, phase=0.3):
    t = np.arange(n) / n
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * (t + phase))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_piecewise(n, left_frac=0.6, left_height=1.0, right_height=0.3):
    a = np.ones(n)
    cut = int(n * left_frac)
    a[:cut] = left_height
    a[cut:] = right_height
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, label=""):
        nonlocal best_seq, best_score
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            p = np.maximum(p, 0.0)
            s = float(p.sum())
            if s > 0.01:
                p = p / s
                sc = _score_norm(p)
                if np.isfinite(sc) and sc < best_score:
                    best_seq = p.copy()
                    best_score = float(sc)
                    print(f"  NEW BEST: {best_score:.10f} n={p.size} [{label}]")

    # Get parent sequence
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm(parent)
        print(f"Parent: n={parent.size}, score={sc_parent:.10f}")
        _update(parent, 'parent')
    else:
        parent = _make_ramp(1024, 1.0)
        print(f"No parent, using ramp init n=1024")
        _update(parent, 'ramp_init')

    n = best_seq.size if best_seq is not None else 1024

    # Phase 1: Intensive LP iteration (main workhorse)
    p1_end = t_start + 500.0
    print(f"\nPhase 1: LP iteration (target: {p1_end - t_start:.0f}s)")
    if time.monotonic() < p1_end and best_seq is not None:
        p1, sc1 = _lp_iteration(best_seq, p1_end, m_decrease=0.9985,
                                 t_blend=0.3, max_iters=5000)
        _update(p1, 'lp_phase1')

    # Phase 2: Gradient refinement
    p2_end = t_start + 800.0
    print(f"\nPhase 2: Gradient refinement (target: {p2_end - t_start:.0f}s)")
    if time.monotonic() < p2_end and best_seq is not None:
        p2, sc2 = _adam_smoothmax(best_seq, p2_end, rng, beta_frac=0.06,
                                   beta_decay=0.96, lr=0.04, no_imp_patience=150)
        _update(p2, 'grad_phase2')

    # Phase 3: Continued LP iteration
    p3_end = t_start + 1100.0
    print(f"\nPhase 3: LP continued (target: {p3_end - t_start:.0f}s)")
    if time.monotonic() < p3_end and best_seq is not None:
        p3, sc3 = _lp_iteration(best_seq, p3_end, m_decrease=0.9992,
                                 t_blend=0.2, max_iters=5000)
        _update(p3, 'lp_phase3')

    # Phase 4: Dual-guided refinement
    p4_end = t_start + 1300.0
    print(f"\nPhase 4: Dual-guided refinement (target: {p4_end - t_start:.0f}s)")
    if time.monotonic() < p4_end and best_seq is not None:
        p4, sc4 = _dual_guided_refinement(best_seq, p4_end, max_iters=100)
        _update(p4, 'dual_phase4')

    # Phase 5: Fine gradient polish
    p5_end = t_start + 1500.0
    print(f"\nPhase 5: Fine gradient polish (target: {p5_end - t_start:.0f}s)")
    if time.monotonic() < p5_end and best_seq is not None:
        p5, sc5 = _adam_smoothmax(best_seq, p5_end, rng, beta_frac=0.03,
                                   beta_decay=0.97, lr=0.02, no_imp_patience=100)
        _update(p5, 'grad_phase5')

    # Phase 6: Peak flood + final LP
    p6_end = deadline
    print(f"\nPhase 6: Final polish (target: {p6_end - t_start:.0f}s)")
    if time.monotonic() < p6_end - 30 and best_seq is not None:
        p6a, sc6a = _peak_flood_refined(best_seq, time.monotonic() + 60, eta_init=0.02)
        _update(p6a, 'flood_phase6')

    if time.monotonic() < p6_end - 20 and best_seq is not None:
        p6b, sc6b = _lp_iteration(best_seq, p6_end - 5, m_decrease=0.9995,
                                   t_blend=0.15, max_iters=2000)
        _update(p6b, 'lp_final')

    # Phase 7: Try alternative initializations if time remains
    if time.monotonic() < deadline - 60 and best_seq is not None:
        print(f"\nPhase 7: Alt inits exploration")
        alts = [
            ('ramp1.5', _make_ramp(n, 1.5)),
            ('ramp0.7', _make_ramp(n, 0.7)),
            ('cos_asym', _make_cosine_asym(n, 1, 0.7, 0.25)),
            ('piecewise', _make_piecewise(n, 0.5, 1.0, 0.4)),
        ]
        per_alt = max(10.0, (deadline - 60 - time.monotonic()) / len(alts))
        for label, pa in alts:
            if time.monotonic() > deadline - 30:
                break
            per = min(per_alt, deadline - 30 - time.monotonic())
            if per < 10:
                break
            p_alt, sc_alt = _adam_smoothmax(pa, time.monotonic() + per, rng,
                                             beta_frac=0.08, beta_decay=0.95,
                                             lr=0.05, no_imp_patience=80)
            _update(p_alt, f'alt_{label}')

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        exact_sc = _score_norm(final)
        print(f"\nFINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_ramp(1024, 1.0)
    print(f"FALLBACK: ramp n=1024, score={_score_norm(fb):.10f}")
    return [float(x) for x in fb]
