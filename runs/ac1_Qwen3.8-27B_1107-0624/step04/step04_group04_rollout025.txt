Improved multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive logsumexp smoothing with L-BFGS, Adam with growing top-k,
peak equalization, and multi-n exploration.

import numpy as np
from scipy.optimize import minimize
import time

# Helper functions (all top-level)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_norm_fn(a):

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive logsumexp smoothing with L-BFGS, Adam with growing top-k,
peak equalization via KKT conditions, multi-n exploration, and random restarts.
Key improvements over previous: more aggressive smoothing schedule, top-k growth,
peak equalization, larger n exploration, and better hyperparameter tuning.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_active_weights(c, top_k):
    n_c = len(c)
    if top_k <= 0 or top_k >= n_c:
        return None
    if top_k == 1:
        idx = int(np.argmax(c))
        w = np.zeros(n_c)
        w[idx] = 1.0
        return w
    idx = np.argsort(c)[-top_k:]
    c_top = c[idx]
    w = np.zeros(n_c)
    w[idx] = 1.0 / top_k
    return w


def _z_objective_smooth(z, beta, n, top_k):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_norm = c_top - c_top.max()
        log_sum = float(np.log(np.sum(np.exp(c_norm / beta))))
        return 2.0 * n * (c_top.max() + beta * log_sum)
    else:
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n * cm
        log_sum = float(np.log(s))
        return 2.0 * n * (cm + beta * log_sum)


def _z_gradient_smooth(z, beta, n, top_k):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    n_c = len(c)
    if top_k > 0 and top_k < n_c:
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_norm = (c_top - c_top.max()) / beta
        e = np.exp(c_norm)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(n_c)
            w[idx[0]] = 1.0
        else:
            w = np.zeros(n_c)
            w[idx] = e / s
    else:
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            idx = int(np.argmax(c))
            w = np.zeros(n_c)
            w[idx] = 1.0
        else:
            w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs_adaptive(a_init, deadline, top_k_start=1, top_k_end=30):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    # More levels, more gradual
    beta_factors = [0.05, 0.03, 0.02, 0.01, 0.006, 0.004, 0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001]
    betas = [b * max(max_c0, 1e-10) for b in beta_factors]

    # top_k schedule: grow from top_k_start to top_k_end
    n_levels = len(betas)
    for i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        # Grow top_k
        frac = i / max(1, n_levels - 1)
        cur_k = int(top_k_start + frac * (top_k_end - top_k_start))
        cur_k = max(1, cur_k)

        remaining = deadline - time.monotonic()
        per_time = max(3.0, remaining / (n_levels - i))

        try:
            result = minimize(
                _z_objective_smooth, z, args=(beta, n, cur_k),
                method='L-BFGS-B', jac=_z_gradient_smooth,
                options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 3000,
                         'maxfun_per_sec': max(50, int(per_time * 50))}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                if np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _run_adam(a_init, deadline, lr=0.04, top_k_start=1, top_k_end=20, n_max_iter=100000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, 0.02 * max_c)

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        # Adaptive top_k
        frac = min(1.0, t / (n_max_iter * 0.3))
        cur_k = int(top_k_start + frac * (top_k_end - top_k_start))
        cur_k = max(1, cur_k)

        a = _softmax_fn(z)
        grad = _z_gradient_smooth(z, beta, n, cur_k)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a = _softmax_fn(z)

        if t % 10 == 0:
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

        # Adaptive beta decay
        if t % 50 == 0:
            beta = max(1e-16, beta * 0.90)

        # Learning rate decay
        if t % 1000 == 0:
            lr = max(0.002, lr * 0.95)

    # Final evaluation
    a = _softmax_fn(z)
    sc = _score_norm_fn(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def _peak_equalize(a_init, deadline, max_iter=20000):
    """Direct peak equalization: push down the max peak by adjusting a."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    lr = 0.01
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        # Find top peaks (within 0.5% of max)
        threshold = cm * 0.998
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            break

        # Compute the " pressure " from active peaks
        h = np.zeros(n)
        for k_val in active:
            k_val = int(k_val)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                h[lo:hi + 1] += a[k_val - hi:k_val - lo + 1][::-1]
        h /= len(active)

        W = float(np.dot(h, a))

        # Direction: reduce a where h > W, increase where h < W
        # Use logit space for non-negativity
        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()

        diff = h - W
        max_diff = float(np.abs(diff).max())
        if max_diff < 1e-15:
            break

        # Adaptive step size
        active_spread = float(c[active].max() - c[active].min()) + 1e-15
        step = lr * active_spread / (cm + 1e-15)
        step = min(step, 0.03)

        z = z - step * diff / max_diff
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a_new = _softmax_fn(z)

        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Halve the step
            lr = max(0.001, lr * 0.7)
            z = np.log(np.maximum(a, 1e-15))
            z = z - z.mean()
            z = z - (lr * 0.5) * diff / max_diff
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new

        if it > 200 and best_score > sc - 1e-12:
            if it > 500:
                break

    return best_a, best_score


def _make_gaussian_profile(n, sigma, center=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    return a / a.sum()


def _make_bimodal_profile(n, sigma, sep=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - (0.5 - sep)) / sigma) ** 2) + np.exp(-0.5 * ((t - (0.5 + sep)) / sigma) ** 2)
    return a / a.sum()


def _make_cosine_profile(n, power=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.maximum(np.cos(np.pi * (t - 0.5)), 0.0) ** power
    return a / (a.sum() + 1e-30)


def _make_exp_profile(n, alpha):
    r = 1.0 - alpha
    a = r ** np.arange(n)
    return a / a.sum()


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _subgradient_step(a, n_stops=5):
    """Take a few exact subgradient steps on the max peak only."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    lr = 0.005

    for _ in range(n_stops):
        c = _full_convolve(a, a)
        cm = float(c.max())
        k_star = int(np.argmax(c))

        # Subgradient of max w.r.t. a_i is 2*a_{k_star - i} (for valid i)
        h = np.zeros(n)
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        if lo <= hi:
            h[lo:hi + 1] = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]

        # Gradient of full objective (normalized)
        S = float(a.sum())
        # dF/da_i = 2n*(h_i * S - 2*cm) / S^3  ... actually let's use logit space
        # In logit space: z_i -= lr * (h_i * S - 2*cm) / (S * a_i) * (some scale)
        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()

        grad_z = h * S - 2.0 * cm
        # Normalize gradient
        mg = float(np.abs(grad_z).max())
        if mg > 1e-15:
            z = z - lr * grad_z / mg
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                a = a_new
            else:
                lr *= 0.5

    return best_a, best_sc

            remaining2 = phase2_end - time.monotonic()
            dl2 = min(phase2_end - 5, time.monotonic() + remaining2 * 0.6)
            p_best2, sc_best2 = _run_adam(best_seq, dl2, lr=0.01, top_k_start=5, top_k_end=30, n_max_iter=50000)
            update_best(p_best2, sc_best2, "adam2")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Peak equalization ---
    print("=== Phase 3: Peak equalization ===")
    phase3_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        remaining = phase3_end - time.monotonic()
        dl = min(phase3_end, time.monotonic() + remaining * 0.6)
        print(f"  Peak eq on best, budget={remaining:.1f}s")
        p_best, sc_best = _peak_equalize(best_seq, dl, max_iter=30000)
        update_best(p_best, sc_best, "peak_eq")

        # Subgradient steps
        if time.monotonic() < phase3_end - 5:
            p_sg, sc_sg = _subgradient_step(best_seq, n_stops=20)
            update_best(p_sg, sc_sg, "subgrad")

        # Another L-BFGS round with top-k
        if time.monotonic() < phase3_end - 10:
            dl2 = min(phase3_end - 5, time.monotonic() + 60.0)
            p_best2, sc_best2 = _run_lbfgs_adaptive(best_seq, dl2, top_k_start=5, top_k_end=50)
            update_best(p_best2, sc_best2, "lbfgs_post_eq")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Random restarts ---
    print("=== Phase 4: Random restarts ===")
    phase4_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 5
        per_restart = min(50.0, (phase4_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 10:
                break
            noise_levels = [0.03, 0.05, 0.08, 0.12, 0.2]
            noise = noise_levels[i]
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=n_best)
            z = np.clip(z, -40, 40)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            dl = min(phase4_end - 5, time.monotonic() + per_restart)
            # Quick L-BFGS
            per_lbfgs = per_restart * 0.55
            p_m, sc_m = _run_lbfgs_adaptive(p_start, min(phase4_end - 5, time.monotonic() + per_lbfgs),
                                            top_k_start=1, top_k_end=20)
            update_best(p_m, sc_m, f"r{i}_lbfgs")

            if time.monotonic() < phase4_end - 5:
                per_adam = min(20.0, phase4_end - 5 - time.monotonic())
                p_a, sc_a = _run_adam(p_m, time.monotonic() + per_adam,
                                       lr=0.02, top_k_start=3, top_k_end=15, n_max_iter=20000)
                update_best(p_a, sc_a, f"r{i}_adam")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Try different lengths ---
    print("=== Phase 5: Different lengths ===")
    phase5_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        n_curr = len(best_seq)
        target_ns = []
        for tn in [1024, 2048, 4096, 8192, 16384]:
            if tn != n_curr and tn not in target_ns:
                target_ns.append(tn)

        for tn in target_ns[:3]:
            if time.monotonic() > phase5_end - 30:
                break
            p_interp = _interp_profile(best_seq, tn)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{tn}")

            remaining = phase5_end - time.monotonic()
            if remaining < 30:
                break
            per_opt = min(80.0, remaining * 0.5)
            dl = min(phase5_end - 10, time.monotonic() + per_opt)
            p_best, sc_best = _run_lbfgs_adaptive(p_interp, dl, top_k_start=1, top_k_end=30)
            update_best(p_best, sc_best, f"lbfgs_n{tn}")

            if time.monotonic() < phase5_end - 15:
                dl2 = min(phase5_end - 10, time.monotonic() + min(30.0, (phase5_end - time.monotonic()) * 0.4))
                p_best2, sc_best2 = _run_adam(p_best, dl2, lr=0.02, top_k_start=3, top_k_end=20, n_max_iter=15000)
                update_best(p_best2, sc_best2, f"adam_n{tn}")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Final intensive polish ---
    print("=== Phase 6: Final polish ===")
    if time.monotonic() < deadline and best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            # Adam with very small lr
            dl = min(deadline - 5, time.monotonic() + min(40.0, remaining * 0.4))
            p_best, sc_best = _run_adam(best_seq, dl, lr=0.005, top_k_start=10, top_k_end=40, n_max_iter=30000)
            update_best(p_best, sc_best, "final_adam")

        if time.monotonic() < deadline - 10:
            # Peak equalization
            dl2 = min(deadline - 5, time.monotonic() + min(20.0, (deadline - time.monotonic()) * 0.5))
            p_best2, sc_best2 = _peak_equalize(best_seq, dl2, max_iter=10000)
            update_best(p_best2, sc_best2, "final_peak_eq")

        if time.monotonic() < deadline - 5:
            # Final L-BFGS
            dl3 = min(deadline - 2, time.monotonic() + min(15.0, deadline - time.monotonic() - 2))
            p_best3, sc_best3 = _run_lbfgs_adaptive(best_seq, dl3, top_k_start=10, top_k_end=60)
            update_best(p_best3, sc_best3, "final_lbfgs")

    # --- Final validation and return ---
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = _make_exp_profile(4096, 0.003)
            final_score = _score_fn(final_seq)

        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")

        # Validate with evaluate_sequence if feasible
        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 16384:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  evaluate_sequence check failed: {e}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_cosine_profile(4096, 1.0)
    print(f"  FALLBACK: n=4096, score={_score_norm_fn(fallback):.10f}")
    return [float(x) for x in fallback]
