The optimality condition is: `∂f/∂a_i = λ` for all `i` with `a_i > 0` (and `∂f/∂a_i >= λ` for `a_i = 0`).

So at the optimum, the gradient of the objective (at the active set) should be constant across all support positions.

This gives us a "water-filling" interpretation: we want to "equalize" the gradient across all positions.

Algorithm:
1. Compute `b = conv(a, a)`.
2. Find the active set `K` (positions where `b_k` is near max).
3. Compute the gradient `g = sum_{k in K} w_k * g_k` (with softmax weights).
4. Update: `a_i <- a_i - η * (g_i - mean(g))` (the mean subtraction enforces the sum constraint).
5. Project onto the positive orthant: `a_i <- max(a_i, 0)`.
6. Re-normalize: `a <- a / sum(a)`.

This is essentially projected gradient descent with a constraint.

For convergence, we want a good step size η. We can use:
- Fixed η (maybe decreasing over time).
- Backtracking line search.
- Adam (adapts η per coordinate).

I think Adam with a decaying learning rate will work well here.

Optimization of non-negative step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Adam with log-parameterization + exact subgradient, LP-based Frank-Wolfe refinement,
peak-targeted coordinate descent, and multi-scale exploration.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-30:
        return np.ones_like(a) / a.size
    return a / s


def _soft_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    m = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= m - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return w
    x = (c - m) / beta
    x = np.clip(x, -40.0, 0.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    return w


def _gradient_from_weights(w, a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = np.zeros(N)
    a_rev[:n] = a[::-1]
    conv = np.fft.irfft(np.fft.rfft(w_pad, N) * np.fft.rfft(a_rev, N), N)
    return 2.0 * conv[n - 1:2 * n - 1]


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    mu = np.zeros(n)
    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    rho = np.nonzero(v * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    mu = np.maximum(v - theta, 0)
    return mu


def _adam_logit(a_init, deadline, rng, lr=0.03, beta_frac=0.06, beta_decay=0.94,

    a = a / a.sum()
    n = a.size
    if base_lr is None:
        base_lr = lr

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    c = _full_convolve(a)
    beta = max(1e-4, beta_frac * float(c.max()))

    best_a = a.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * c.max())

    cur_lr = lr

        maxc = float(c.max())

            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                cur_lr = max(0.003, cur_lr * 0.92)

        grad = _gradient_from_weights(w, a)
        grad -= float(np.mean(grad))

        if not np.all(np.isfinite(grad)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            continue

        t += 1
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)

        z = z - cur_lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40.0, 40.0)

        a = p

            m.fill(0.0)

def _lp_solve_active(a, M, n_active=None):
    n = len(a)
    c = _full_convolve(a)
    if n_active is not None and n_active < len(c):
        threshold = np.partition(c, len(c) - n_active)[len(c) - n_active]
        active_mask = c >= threshold
    else:
        active_mask = np.ones(len(c), dtype=bool)

    active_k = np.where(active_mask)[0]
    rows = []
    cols = []
    vals = []
    for k in active_k:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([int(k) for _ in idx])
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())

    n_rows = len(active_k)
    A_ub = csr_matrix((vals, (rows, cols)), shape=(n_rows, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(n_rows)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a_init, deadline, t_mix=0.35, m_shrink=0.998, max_solves=60):
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    c = _full_convolve(a)
    M = float(c.max())
    t = t_mix
    n_solves = 0

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1

        b = _lp_solve_active(a, M, n_active=512)
        if b is None:
            b = _lp_solve_active(a, M * 1.01, n_active=512)
            if b is None:
                break

        s_b = float(b.sum())
        if s_b < 1e-10:
            break

        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score(a_new)

        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        c = _full_convolve(a)
        M_new = float(c.max())
        M = max(M_new * m_shrink, best_score / (2.0 * n) * 1.003)
        t = max(0.12, t * 0.93)

    return best_a, best_score


def _peak_coordinate_descent(a_init, deadline, n_steps=2000):
    a = np.asarray(a_init, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng(777)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a)
        M = float(c.max())
        peak_k = int(np.argmax(c))
        score = 2.0 * n * M

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        if step % 50 == 0 and step > 0:
            continue

        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        if i_start > i_end:
            continue

        idx = np.arange(i_start, i_end + 1)
        contrib = a[idx] * a[peak_k - idx]
        total = float(contrib.sum())
        if total < 1e-15:
            continue

        # Find the position with highest contribution
        max_contrib_idx = int(np.argmax(contrib))
        i1 = i_start + max_contrib_idx

        # Find a target position with low contribution to nearby peaks
        partner_set = set()
        for kk in range(max(0, peak_k - 3), min(2 * n - 1, peak_k + 4)):
            for ii in range(max(0, kk - n + 1), min(n, kk + 1)):
                partner_set.add(kk - ii)

        candidates = []
        for i in range(n):
            if i == i1 or i in partner_set:
                continue
            # Estimate contribution to peak if we move mass here
            new_contrib = a[i] * a[peak_k - i] if 0 <= peak_k - i < n else 0.0
            candidates.append((new_contrib, i))

        if not candidates:
            candidates = [(0.0, i) for i in range(n) if i != i1]
        if not candidates:
            continue

        candidates.sort()
        target = candidates[0][1]

        frac = min(0.15, 0.03 + 0.05 * (a[i1] * n))
        amount = a[i1] * frac
        a[i1] = max(a[i1] - amount, 1e-15)
        a[target] += amount
        a = a / a.sum()

    return best_a, best_score

def _interp_to_n(a, n_new):
