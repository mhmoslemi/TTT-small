def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)

def _adaptive_beta(c):
    """Choose beta based on peak structure."""
    cm = float(c.max())
    if cm < 1e-15:
        return 1e-10
    sorted_c = np.sort(c)[::-1]
    k = min(20, len(sorted_c) - 1)
    gap = sorted_c[0] - sorted_c[k]
    if gap < 0.05 * cm:
        # Peaks are balanced, use moderate beta to see all of them
        return max(1e-10, 0.1 * cm)
    elif gap > 0.5 * cm:
        # One peak dominates, use small beta to focus on it
        return max(1e-10, 0.01 * cm)
    else:
        # Intermediate, use half the gap
        return max(1e-10, 0.5 * gap)

def _fourier_optimize(n, deadline, rng):
    """Optimize a smooth Fourier-parameterized profile."""
    u = 2.0 * np.linspace(0, 1, n) - 1.0  # in [-1, 1]
    # Basis: 1, u, u^2, u^3, u^4, u^5 (or cos terms)
    K = 5
    basis = np.ones((n, K + 1))
    for k in range(1, K + 1):
        basis[:, k] = u ** k

    best_score = np.inf
    best_alpha = np.zeros(K + 1)

    # Try multiple initial alpha values
    inits = [
        np.zeros(K + 1),
        np.array([0, 1.0, 0, 0, 0, 0]),
        np.array([0, -1.0, 0, 0, 0, 0]),
        np.array([0, 0, 1.0, 0, 0, 0]),
        np.array([0, 1.0, 0.5, 0, 0, 0]),
        np.array([0, -1.0, 0.5, 0, 0, 0]),
        np.array([0, 2.0, -1.0, 0, 0, 0]),
        np.array([0, -2.0, -1.0, 0, 0, 0]),
        np.array([0, 0.5, 0.3, 0.1, 0, 0]),
        np.array([0, -0.5, 0.3, -0.1, 0, 0]),
    ]

    def fourier_score(alpha):
        log_a = basis @ alpha
        log_a = log_a - np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s < 1e-10:
            return np.inf
        a = a / s
        return _score_norm_fn(a)

    for init in inits:
        if time.monotonic() > deadline:
            break
        try:
            res = minimize(fourier_score, init, method='Nelder-Mead',
                           options={'maxiter': 5000, 'xatol': 1e-10, 'fatol': 1e-12})
            if res.fun < best_score:
                best_score = res.fun
                best_alpha = res.x.copy()
        except Exception:
            pass

    # Construct the sequence
    log_a = basis @ best_alpha
    log_a = log_a - np.max(log_a)
    a = np.exp(log_a)
    a = a / a.sum()
    return a, best_score

def _make_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []

    # Linear ramps
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Quadratic
    for a1, a2 in [(1.0, 0.5), (2.0, -0.3), (1.5, 0.2), (-1.0, 0.5)]:
        a = 1.0 + a1 * u + a2 * u ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Exponential
    for rate in [1.0, 2.0, 3.0, -1.0, -2.0, -3.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())

    # Cosine power
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())

    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
