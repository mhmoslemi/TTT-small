def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-15, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _adam_logit(p0, deadline, rng, lr=0.05, beta_frac=0.08, beta_decay=0.93,
                no_imp_limit=250, noise_scale=None, use_cosine_lr=True):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    total_steps_estimate = max(100, int((deadline - time.monotonic()) / 0.05))
    step_count = 0

    while time.monotonic() < deadline:
        _, w = _smooth_max_and_weights(c, beta)
        g = _conv_cross_corr(w, p)

        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        step_count += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)

        current_lr = lr
        if use_cosine_lr:
            progress = min(1.0, step_count / total_steps_estimate)
            current_lr = lr * 0.5 * (1.0 + np.cos(np.pi * progress))
            current_lr = max(current_lr, lr * 0.05)

        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)

        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 30 == 0:
            beta = max(1e-5, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.05:
                    beta = max(1e-5, beta * 0.5)

        limit = max(80, no_imp_limit)
        if no_imp > limit:
            if noise_scale is None:

                    ns = 0.05
                elif best_score < 1.53:
                    ns = 0.10
                else:
                    ns = 0.20
            else:

            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            step_count = 0
            no_imp = 0

    return best_p, best_score


def _subgradient_projection(a, deadline, n_iter=2000, step_init=0.01):
    """Projected subgradient method for minimizing max autocorrelation."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a)
        M = float(c.max())
        score = float(2.0 * n * M)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        # Find all near-max positions
        threshold = M - 1e-10 * M
        hot_mask = c >= threshold
        if not hot_mask.any():
            hot_mask[int(np.argmax(c))] = True

        # Subgradient: weighted average of gradients at hot positions
        # grad_k[i] = a[k-i] + a[i-(k-n+1)] ... (cross-correlation structure)
        # For (a*a)[k], d/d a[i] = a[k-i] (if 0<=k-i<n) + a[i-(2n-2-k)] (if valid)
        # More precisely: (a*a)[k] = sum_j a[j]*a[k-j], d/d a[i] = a[k-i] (if 0<=k-i<n) + a[i-(k-n+1)]... 
        # Actually for linear conv: (a*a)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[j]*a[k-j]
        # d/d a[i] = a[k-i] if max(0,k-n+1) <= i <= min(n-1,k) and k-i is in [0,n-1]
        # = a[k-i] if max(0,k-n+1) <= i <= min(n-1,k)
        # Plus: d/d a[i] also gets contribution from a[i]*a[k-i] where i is the second index
        # = a[i] * d(a[k-i])/d(a[i]) ... 
        # Actually (a*a)[k] = sum_j a[j] * a[k-j]
        # d/d a[i] = a[k-i] (from j=i term) + (if k-i is valid, a[k-(k-i)] = a[i] ... no)
        # Let's just use: gradient of (a*a)[k] w.r.t. a is the "flipped and shifted" a
        # Specifically: grad_k = shift(correlate(a, e_k)) where e_k is the unit vector
        # For our purposes: grad_k[i] = a[k-i] if 0 <= k-i < n, else 0
        # PLUS: grad_k[i] += a[i-(2n-2-k)] if 0 <= i-(2n-2-k) < n
        # Wait, that's not right either. Let me just compute it directly.
        # (a*a)[k] = sum_j a[j]*a[k-j] (valid j)
        # d/d a[i]: the term a[j]*a[k-j] contributes if j=i (gives a[k-i]) or k-j=i (gives a[j]=a[k-i])
        # So if j=i and k-i is valid: contribution a[k-i]
        # If k-j=i, i.e., j=k-i, and j is valid: contribution a[k-i]
        # These are the same when k-i is valid! So grad_k[i] = a[k-i] if 0<=k-i<n AND i is in valid range for j
        # Valid range for j: max(0, k-n+1) <= j <= min(n-1, k)
        # So grad_k[i] = a[k-i] if max(0,k-n+1) <= i <= min(n-1,k), else 0
        
        grad = np.zeros(n)
        hot_indices = np.where(hot_mask)[0]
        n_hot = len(hot_indices)
        for k in hot_indices:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                partner = k - np.arange(i_start, i_end + 1)
                # Each of these positions contributes a[partner] to the gradient
                # But also the "second" index contributes: for position i, the partner is k-i
                # The gradient entry at position i is a[k-i] (from the first factor) + a[i-(k-n+1)]... 
                # Actually I think for the "cross term" we need to be more careful.
                # (a*a)[k] = sum_j a[j]*a[k-j]
                # This is a quadratic form. The gradient at position i is:
                # = a[k-i] (if k-i in [0,n-1] and i in [max(0,k-n+1), min(n-1,k)])
                #   + a[i-(k-(n-1))] ... hmm this is getting confusing.
                # 
                # Let me just use the simple formula: grad[i] = a[k-i] for valid i
                grad[i_start:i_end + 1] += a[partner]
        
        grad /= n_hot
        # Scale: the actual gradient of max is in the direction of grad (for the active max)
        # But we need to account for the factor of 2 (since (a*a)[k] has each pair counted once)
        # Actually for (a*a)[k] = sum_j a[j]*a[k-j], d/d a[i] = a[k-i] if valid (each term appears once)
        # So the gradient is just a[k-i] (not 2*a[k-i])
        
        # Adaptive step size
        step = step_init / (1.0 + 0.1 * it)
        
        a_new = a - step * grad
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        sc_new = _score_norm(a_new)
        if sc_new < score:
            a = a_new
        else:
            # Halve step size
            step_init = max(1e-6, step_init * 0.99)
            a_new = a - 0.5 * step * grad
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            a = a_new

    return best_a, best_score


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a, deadline, t_start=0.4, max_solves=100, m_shrink=0.995, min_M_frac=0.95):
    """LP refinement with adaptive M schedule."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    M_target = best_score / (2.0 * n) * 1.001
    n_solves = 0
    t = t_start
    no_imp_count = 0

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1

        # Solve LP
        b = _lp_solve(a, M)
        if b is None:
            # Try slightly larger M
            M = M * 1.02
            if M > float(c.max()) * 1.1:
                M = float(c.max()) * 1.01
            b = _lp_solve(a, M)
            if b is None:
                break

        s_b = float(b.sum())
        if s_b < 1e-10:
            break

        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)

        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()
            no_imp_count = 0
        else:
            no_imp_count += 1
            if no_imp_count > 20:
                # Reset M to current peak
                a = best_a.copy()
                c = _full_convolve(a)
                M = float(c.max())
                t = t_start
                no_imp_count = 0
                continue

        a = a_new
        c = _full_convolve(a)
        M_current = float(c.max())

        # Adaptive M: decrease towards target
        M = max(M_current * m_shrink, M_target)
        if M < best_score / (2.0 * n) * 0.99:
            M = best_score / (2.0 * n) * 1.001

        # Decrease step size
        t = max(0.1, t * 0.95)

        # If M has been stuck, try a fresh start with tighter M
        if n_solves % 30 == 0 and no_imp_count > 10:
            a = best_a.copy()
            c = _full_convolve(a)
            M = float(c.max()) * 0.998
            t = t_start * 0.7

    return best_a, best_score

def _interp_profile(p0, n_new):

def _next_pow2(n):
    k = 1

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c
