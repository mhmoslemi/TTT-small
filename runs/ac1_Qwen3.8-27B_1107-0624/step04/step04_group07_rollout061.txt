import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _top_m_smooth_grad(c, a, beta, top_m):
    """Compute gradient of top-m smooth max of c w.r.t. a using FFT cross-correlation."""
    n = len(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, _grad_single_peak(c, a, int(np.argmax(c)))

    # Use top-m weights
    if top_m is not None and top_m < len(c):
        idx = np.argsort(c)[::-1][:top_m]
        w = np.zeros_like(c)
        w[idx] = ex[idx]
        w = w / w.sum()
    else:
        w = ex / s

    sm_val = mval + beta * np.log(float(ex[idx].sum()) if top_m is not None and top_m < len(c) else s)

    # Gradient: grad[j] = sum_{k=0}^{2n-2} w[k] * a[k-j] (for valid k-j in [0,n-1])
    # This is cross-correlation: grad[j] = sum_m a[m] * w[m+j]
    # Using FFT: grad = ifft(conj(fft(a)) * fft(w)) with padding
    N = _next_pow2(2 * n + (2 * n - 1) - 1)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    grad = np.fft.irfft(np.conj(fa) * fw, N)[:n]

    return sm_val, grad


def _grad_single_peak(c, a, k_star):
    """Gradient of c[k_star] w.r.t. a = shifted/reversed a."""
    n = len(a)
    grad = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(n - 1, k_star)
    for j in range(j_start, j_end + 1):
        grad[j] += a[k_star - j]
        if k_star - j == j:
            grad[j] += a[j]
    return grad

    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()

    z = np.log(a)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = _full_convolve(a)
    maxc = float(c.max())
    beta = beta_init if beta_init is not None else 0.05 * maxc
    base_lr = lr

    best_score = float(2.0 * n * maxc)
    no_imp = 0

        # grad_z = a * (grad_a - dot(a, grad_a))

            m.fill(0)

        m = b1 * m + (1 - b1) * grad_z

        a = np.maximum(a, 1e-15)

def _block_coord_descent(a, deadline, block_size=4, max_sweeps=100):
    """Block coordinate descent: optimize groups of coordinates jointly."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for start in range(0, n, block_size):
            if time.monotonic() > deadline:
                break
            end = min(start + block_size, n)
            block = a[start:end].copy()

            def block_obj(x):
                a_trial = a.copy()
                a_trial[start:end] = np.maximum(x, 1e-15)
                s = a_trial.sum()
                if s < 0.01:
                    return 1e10
                a_trial = a_trial / s
                return _score(a_trial)

            res = minimize(block_obj, block, method='Nelder-Mead',
                           options={'maxiter': 50, 'xatol': 1e-8, 'fatol': 1e-10})
            if res.success and res.fun < best_score - 1e-10:
                a[start:end] = np.maximum(res.x, 1e-15)
                s = a.sum()
                a = a / s
                best_score = res.fun
                best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_score

def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new


def _make_profile(n, ptype, p1=1.0, p2=0.0, p3=0.0):

    elif ptype == 'two_bump':

    elif ptype == 'cos':
        a = 1.0 + p1 * np.cos(2 * np.pi * p2 * t)
    elif ptype == 'sin_asym':

    else:

    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    # === Phase 1: Generate diverse initial candidates ===
    print("Phase 1: Generating candidates...")
    candidates = []

    # Various n sizes
    n_sizes = [128, 256, 512, 1024, 2048]

    profile_defs = [
        ('lin05', 'lin', 0.5, 0, 0),
        ('lin10', 'lin', 1.0, 0, 0),
        ('lin15', 'lin', 1.5, 0, 0),
        ('lin20', 'lin', 2.0, 0, 0),
        ('lin25', 'lin', 2.5, 0, 0),
        ('lin30', 'lin', 3.0, 0, 0),
        ('pow1', 'pow', 1.0, 0, 0),
        ('pow2', 'pow', 2.0, 0, 0),
        ('pow05', 'pow', 0.5, 0, 0),
        ('powr1', 'pow_rev', 1.0, 0, 0),
        ('powr2', 'pow_rev', 2.0, 0, 0),
        ('exp1', 'exp', 1.0, 0, 0),
        ('exp2', 'exp', 2.0, 0, 0),
        ('exp3', 'exp', 3.0, 0, 0),
        ('exp_rev2', 'exp_rev', 2.0, 0, 0),
        ('qa1', 'quad_asym', 1.0, 0.5, 0),
        ('qa2', 'quad_asym', 2.0, 1.0, 0),
        ('qa3', 'quad_asym', 1.5, -0.5, 0),
        ('bump30', 'bump', 0.3, 0.15, 0),
        ('bump70', 'bump', 0.7, 0.15, 0),
        ('bump50', 'bump', 0.5, 0.2, 0),
        ('tb35', 'two_bump', 0.35, 0.65, 0.15),
        ('tb30', 'two_bump', 0.3, 0.7, 0.2),
        ('cos1', 'cos', 0.5, 1, 0),
        ('cos2', 'cos', 0.3, 2, 0),
        ('sina1', 'sin_asym', 0.5, 0.5, 0),
        ('sina2', 'sin_asym', 0.8, -0.3, 0),
        ('sig5', 'sigmoid', 5.0, 0.5, 0),
        ('sig8', 'sigmoid', 8.0, 0.4, 0),
        ('flap07', 'flap', 0.7, 0, 0),
        ('flap05', 'flap', 0.5, 0, 0),
        ('edge05', 'edge', 0.5, 0, 0),
        ('edge1', 'edge', 1.0, 0, 0),
        ('ch1', 'center_heavy', 1.0, 0, 0),
        ('ch3', 'center_heavy', 3.0, 0, 0),
    ]

    # Also add random smooth profiles
    for _ in range(20):
        prof = _random_smooth_profile(256, rng, n_freq=rng.integers(2, 5))
        candidates.append((_score(prof), prof, 256, 'rand'))

    for n in n_sizes:
        for (label, ptype, p1, p2, p3) in profile_defs:
            p = _make_profile(n, ptype, p1, p2, p3)
            sc = _score(p)
            candidates.append((sc, p, n, label))

    if parent is not None:
        for n in [256, 512, 1024, 2048, 4096]:
            pi = _interp(parent, n)
            sc = _score(pi)
            candidates.append((sc, pi, n, 'parent'))
        if 0 < pn <= 4096:
            sc = _score(parent)
            candidates.append((sc, parent.copy(), pn, 'parent_exact'))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, best init: {candidates[0][0]:.8f} n={candidates[0][2]} [{candidates[0][3]}]")

    if candidates[0][0] < best_score:
        best_seq = np.asarray(candidates[0][1], dtype=np.float64).copy()
        best_score = float(candidates[0][0])
        print(f"  BEST: {best_score:.10f} n={best_seq.size} [init]")

    # === Phase 2: Adam optimization on top candidates ===
    print("Phase 2: Adam optimization...")
    phase2_end = t_start + 600.0
    polished = []

    # Sort by (score, n) - prefer smaller n for speed, then by score
    candidates_sorted = sorted(candidates, key=lambda x: (x[0] * max(1, x[2] / 256), x[2]))

    count = 0
    for (sc, p, n, label) in candidates_sorted:
        if time.monotonic() > phase2_end or count >= 12:
            break
        if n > 2048:
            continue

        # Time allocation: more time for smaller n, less for larger
        time_alloc = max(30.0, min(90.0, 600.0 * (256.0 / float(n)) ** 0.7))
        per_end = min(phase2_end, time.monotonic() + time_alloc)

        # Adam with top-m
        top_m = max(3, n // 64)
        a_ad, sc_ad = _adam_optimize(p, per_end, rng, top_m=top_m, lr=0.04,
                                     beta_decay=0.97, no_imp_limit=250)
        if sc_ad < best_score:
            best_seq = np.asarray(a_ad, dtype=np.float64).copy()
            best_score = float(sc_ad)
            print(f"  BEST: {best_score:.10f} n={a_ad.size} [Adam_{label}_n{n}]")
        polished.append((sc_ad, a_ad, n, label + '_adam'))
        count += 1

    polished.sort(key=lambda x: x[0])
    print(f"  After Phase 2: best={best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    # === Phase 3: Block coordinate descent refinement ===
    print("Phase 3: Block coordinate descent...")
    phase3_end = t_start + 1000.0

    for (sc, p, n, label) in polished[:5]:
        if time.monotonic() > phase3_end - 30:
            break
        per = min(120.0, phase3_end - time.monotonic() - 10)
        if per < 30:
            break

        # Block size depends on n
        bs = max(2, min(8, 512 // max(1, n // 64)))
        a_bc, sc_bc = _block_coord_descent(p, time.monotonic() + per, block_size=bs, max_sweeps=50)
        if sc_bc < best_score:
            best_seq = np.asarray(a_bc, dtype=np.float64).copy()
            best_score = float(sc_bc)
            print(f"  BEST: {best_score:.10f} n={a_bc.size} [BlockCD_{label}]")

        # Quick 1D coordinate descent after block
        if time.monotonic() < phase3_end - 20:
            per2 = min(60.0, phase3_end - time.monotonic() - 10)
            if per2 > 20:
                a_cd, sc_cd = _coord_descent_1d(a_bc, time.monotonic() + per2, max_sweeps=20)
                if sc_cd < best_score:
                    best_seq = np.asarray(a_cd, dtype=np.float64).copy()
                    best_score = float(sc_cd)
                    print(f"  BEST: {best_score:.10f} n={a_cd.size} [CD1d_{label}]")

    # === Phase 4: Second round of Adam on best results ===
    print("Phase 4: Second Adam pass...")
    phase4_end = t_start + 1300.0

    if best_seq is not None:
        # Re-run Adam on the best with different hyperparams
        configs = [
            {'top_m': 5, 'lr': 0.02, 'beta_decay': 0.98, 'no_imp_limit': 400},
            {'top_m': 12, 'lr': 0.05, 'beta_decay': 0.96, 'no_imp_limit': 200},
            {'top_m': 3, 'lr': 0.01, 'beta_decay': 0.99, 'no_imp_limit': 500},
        ]
        for cfg in configs:
            if time.monotonic() > phase4_end - 30:
                break
            per = min(80.0, phase4_end - time.monotonic() - 10)
            if per < 25:
                break
            a_ad2, sc_ad2 = _adam_optimize(best_seq, time.monotonic() + per, rng, **cfg)
            if sc_ad2 < best_score:
                best_seq = np.asarray(a_ad2, dtype=np.float64).copy()
                best_score = float(sc_ad2)
                print(f"  BEST: {best_score:.10f} n={best_seq.size} [Adam2_{cfg['top_m']}]")

    # === Phase 5: Perturbation restarts ===
    print("Phase 5: Perturbation restarts...")
    phase5_end = t_start + 1550.0

    if best_seq is not None and time.monotonic() < phase5_end:
        n_best = best_seq.size
        n_restarts = 6
        per_r = max(20.0, (phase5_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            per = min(per_r, phase5_end - time.monotonic() - 5)
            if per < 15:
                break

            # Different perturbation strategies
            ns = 0.04 + 0.04 * i
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()

            if i % 3 == 0:
                z = z + ns * rng.normal(size=n_best)
            elif i % 3 == 1:
                # Multiplicative perturbation
                log_mult = rng.normal(size=n_best)
                z = z + ns * log_mult
            else:
                # Block perturbation
                block = n_best // 4
                for _ in range(3):
                    s = rng.integers(0, n_best - block)
                    z[s:s+block] += ns * rng.uniform(-1, 1, size=block)

            z = np.clip(z, -40, 40)
            z -= z.mean()
            a_s = np.exp(z - z.max())
            a_s = a_s / a_s.sum()

            top_m = max(3, n_best // 64)
            a_r, sc_r = _adam_optimize(a_s, time.monotonic() + per, rng, top_m=top_m,
                                       lr=0.03, beta_decay=0.97, no_imp_limit=150)
            if sc_r < best_score:
                best_seq = np.asarray(a_r, dtype=np.float64).copy()
                best_score = float(sc_r)
                print(f"  BEST: {best_score:.10f} n={n_best} [restart{i}]")

    # === Phase 6: Scale up ===
    print("Phase 6: Scale up...")
    phase6_end = deadline - 60.0

    if best_seq is not None and time.monotonic() < phase6_end:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048]
        elif n_cur <= 1024:
            target_ns = [2048, 4096]
        elif n_cur <= 2048:
            target_ns = [4096]

        for tn in target_ns:
            if time.monotonic() > phase6_end - 60:
                break
            p_int = _interp(best_seq, tn)
            sc_int = _score(p_int)
            print(f"  Interpolated to n={tn}: {sc_int:.8f}")
            if sc_int < best_score:
                best_seq = p_int.copy()
                best_score = sc_int
                print(f"  BEST: {best_score:.10f} n={tn} [interp]")

            # Adam on interpolated
            if time.monotonic() < phase6_end - 40:
                per = min(60.0, phase6_end - time.monotonic() - 20)
                if per > 20:
                    top_m = max(3, tn // 64)
                    a_e, sc_e = _adam_optimize(p_int, time.monotonic() + per, rng,
                                               top_m=top_m, lr=0.02,
                                               beta_decay=0.98, no_imp_limit=200)
                    if sc_e < best_score:
                        best_seq = a_e.copy()
                        best_score = sc_e
                        print(f"  BEST: {best_score:.10f} n={tn} [Adam_esc]")

    # === Phase 7: Final refinement ===
    print("Phase 7: Final refinement...")
    if time.monotonic() < deadline - 10 and best_seq is not None:
        per = deadline - time.monotonic() - 10
        if per > 20:
            # Final block CD
            if best_seq.size <= 2048:
                bs = max(2, min(4, 256 // max(1, best_seq.size // 128)))
                a_fp, sc_fp = _block_coord_descent(best_seq, time.monotonic() + per * 0.6,
                                                   block_size=bs, max_sweeps=30)
                if sc_fp < best_score:
                    best_seq = a_fp.copy()
                    best_score = sc_fp
                    print(f"  BEST: {best_score:.10f} n={best_seq.size} [final_BCD]")

            # Final quick Adam
            if time.monotonic() < deadline - 15:
                per2 = deadline - time.monotonic() - 15
                if per2 > 15:
                    top_m = max(3, best_seq.size // 64)
                    a_fa, sc_fa = _adam_optimize(best_seq, time.monotonic() + per2, rng,
                                                 top_m=top_m, lr=0.01,
                                                 beta_decay=0.99, no_imp_limit=100)
                    if sc_fa < best_score:
                        best_seq = a_fa.copy()
                        best_score = sc_fa
                        print(f"  BEST: {best_score:.10f} n={best_seq.size} [final_Adam]")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile(512, 'lin', 1.5, 0, 0)

        exact_sc = _score(final)
        print(f"\nFINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile(512, 'lin', 1.5, 0, 0)
    print("FALLBACK n=512")
    return [float(x) for x in fb]
