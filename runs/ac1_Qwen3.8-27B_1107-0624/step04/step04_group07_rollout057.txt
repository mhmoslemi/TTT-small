2. Find the "linear minimizer": `s = argmin_{b in simplex} <g, b>`
   - This means finding the coordinate `j` that minimizes `g[j]`, and setting `s = e_j` (the j-th standard basis vector).
   - Wait, that's not quite right. The linear minimizer over the simplex of `<g, b>` is to put all mass on the coordinate with the smallest `g[j]`.
3. Update: `a <- (1-t)*a + t*s`

But this would converge to a vertex of the simplex (a point mass), which is bad. So Frank-Wolfe isn't directly applicable here.

What about **mirror descent with entropy regularization**? This is equivalent to optimizing in log space:
- Maintain `z = log(a) - log(sum(exp(a)))` (normalized log)
- Gradient step: `z <- z - eta * grad`
- Where grad is the gradient of the score with respect to `a`, projected appropriately

This is essentially what the Adam method in the previous code does.

Let me think about what specific improvements I can make:

1. **Adaptive beta for smooth max**: The previous code decays beta over time. I could use a beta that adapts to the gap between the top-1 and top-2 values of c.

2. **Multi-scale gradient**: Use the gradient at multiple beta values and combine them.

3. **Restart strategy**: When stuck, restart with a larger perturbation and reset the Adam state.

4. **n escalation**: More aggressively try larger n values.

5. **Profile diversity**: Try more exotic profiles, like:
   - `a[i] = 1/(1 + (i/n)^p)` (power-law decay)
   - `a[i] = sin(pi * i/n)^p` (raised sinusoid)
   - `a[i] = exp(-|i/n - alpha|^q)` (generalized Gaussian)
   - `a[i] = 1 + eps * cos(2*pi*f*i/n)` (cosine perturbation of uniform)

6. **Iterated function system**: Apply the LP fixed-point multiple times with different parameters.

"""
Multi-scale Adam optimization with adaptive smooth-max for minimizing
2*n*max(conv(a,a))/(sum(a))^2 using FFT-based convolution and
multi-n escalation strategy.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())

def _cross_correlate(w, a):
    """Compute the gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]

def _smooth_max(c, beta):
    """Compute smooth max and weights."""
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip for numerical stability
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w

def _adam_optimize(a, deadline, rng, lr=0.05, beta_init=None,
                   beta_decay=0.97, no_imp_limit=300, top_k=None):
    """Adam optimization with smooth-max (or top-k) gradient."""
    n = len(a)
    a = np.maximum(a, 1e-12).copy()
    a = a / a.sum()
    
    z = np.log(a)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    c = _full_convolve(a)
    maxc = float(c.max())
    score = 2.0 * n * maxc
    beta = beta_init if beta_init is not None else 0.05 * maxc
    
    best_z = z.copy()
    best_score = float(score)
    best_a = a.copy()
    no_imp = 0
    base_lr = lr
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # Compute weights and gradient
        if top_k is not None and top_k < len(c):
            # Top-k averaging
            idx = np.argpartition(c, -top_k)[-top_k:]
            w = np.zeros_like(c)
            w[idx] = 1.0 / top_k
        else:
            _, w = _smooth_max(c, beta)
        
        g = _cross_correlate(w, a)
        
        # Center the gradient (projection onto the simplex tangent space)
        dot = float(np.dot(a, g))
        gz = a * (g - dot)
        gz -= gz.mean()
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40, 40)
        z -= z.mean()
        
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        c = _full_convolve(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        
        if np.isfinite(score):
            if score < best_score - 1e-10:
                best_score = score
                best_z = z.copy()
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.005, lr * 0.9)
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        # Adaptive beta
        if it % 30 == 0:
            beta = max(1e-5, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.05:
                    beta = max(1e-5, beta * 0.5)
        
        # Restart
        if no_imp > no_imp_limit:
            ns = 0.1
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -40, 40)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a)
            beta = 0.05 * c.max()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0
    
    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score

def _subgradient_descent(a, deadline, rng, lr_init=0.01, lr_decay=0.999,
                         top_k=5, max_iters=5000):
    """Subgradient descent with top-k averaging and projection onto simplex."""
    n = len(a)
    a = np.maximum(a, 1e-12).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = lr_init
    momentum = np.zeros(n)
    mu = 0.9
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a)
        # Top-k weights
        if top_k < len(c):
            idx = np.argpartition(c, -top_k)[-top_k:]
            w = np.zeros_like(c)
            w[idx] = 1.0 / top_k
        else:
            w = np.ones_like(c) / len(c)
        
        g = _cross_correlate(w, a)
        # Project gradient: subtract mean (tangent to simplex)
        g = g - g.mean()
        
        # Momentum
        momentum = mu * momentum + (1 - mu) * g
        momentum = momentum - momentum.mean()
        
        # Update in log space
        z = np.log(np.maximum(a, 1e-15))
        z -= z.mean()
        z = z - lr * momentum
        z = np.clip(z, -30, 30)
        a_new = np.exp(z - z.max())
        a_new = a_new / a_new.sum()
        
        score = _score_norm(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Don't update, decay lr
            lr = lr * lr_decay
            a = a  # keep old a
        
        if lr < 1e-6:
            break
    
    return best_a, best_score

def _coord_descent(a, deadline, max_sweeps=30):
    """Coordinate descent trying multiplicative changes to each element."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            for factor in [0.5, 0.7, 1.3, 2.0]:
                a[j] = max(old_val * factor, 1e-15)
                a = a / a.sum()
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    old_val = a[j]
                else:
                    a[j] = old_val
                    a = a / a.sum()
                    break
        if not improved:
            break
    
    return best_a, best_score

def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new

def _make_profile(n, ptype, p1=1.0, p2=0.0, p3=0.0):
    """Generate various profile types."""
    t = np.linspace(0, 1, n)
    i = np.arange(n)
    
    if ptype == 'linear':
        a = 1.0 + p1 * (t - 0.5)
    elif ptype == 'quadratic':
        a = 1.0 + p1 * (2 * t - 1) ** 2
    elif ptype == 'power':
        # Power law: a[i] = (i+1)^p
        a = (i + 1) ** p1
    elif ptype == 'power_rev':
        # Power law reversed
        a = (n - i) ** p1
    elif ptype == 'sig':
        # Sigmoid
        a = 1.0 / (1.0 + np.exp(-p1 * (t - p2)))
    elif ptype == 'gauss':
        a = np.exp(-0.5 * ((t - p2) / p1) ** 2)
    elif ptype == 'exp_decay':
        a = np.exp(-p1 * t)
    elif ptype == 'cosine':
        a = 1.0 + p1 * np.cos(2 * np.pi * p2 * t)
    elif ptype == 'sine':
        a = 1.0 + p1 * np.sin(2 * np.pi * p2 * t)
    elif ptype == 'raised_sin':
        a = np.sin(np.pi * t) ** p1
    elif ptype == 'U':
        a = (2 * t - 1) ** 2 + p1
    elif ptype == 'bimodal':
        a = np.exp(-0.5*((t-p1)/0.1)**2) + np.exp(-0.5*((t-p2)/0.1)**2)
    elif ptype == 'edge':
        a = 1.0 / (0.05 + 2.0 * np.abs(t - 0.5)) ** p1
    elif ptype == 'plateau':
        a = np.ones(n)
        a[:int(p1*n)] *= p2
        a[int(p2*n):] *= p3  # Hmm, this doesn't make sense with the params
        a = np.ones(n)
        a[int(p1*n):int(p2*n)] = p3
    elif ptype == 'ramp':
        # Asymmetric ramp
        a = np.maximum(p1 * t, p2 * (1 - t))
    elif ptype == 'step':
        a = np.ones(n)
        a[:int(p1*n)] = p2
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 1e-12)
    return a / a.sum()

"""
Multi-scale Adam optimization with adaptive smooth-max, top-k subgradient,
Nesterov momentum, and aggressive n-escalation for minimizing
2*n*max(conv(a,a))/(sum(a))^2.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _cross_correlate(w, a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]


def _smooth_max(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w


def _adam_nesterov(a, deadline, rng, lr=0.05, beta_init=None,
                   beta_decay=0.97, no_imp_limit=250, top_k=None,
                   nesterov_mu=0.9):
    n = len(a)
    a = np.maximum(a, 1e-12).copy()
    a = a / a.sum()
    z = np.log(a)
    z -= z.mean()
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    c = _full_convolve(a)
    maxc = float(c.max())
    score = 2.0 * n * maxc
    beta = beta_init if beta_init is not None else 0.05 * maxc
    best_z = z.copy()
    best_score = float(score)
    best_a = a.copy()
    no_imp = 0
    base_lr = lr
    it = 0
    while time.monotonic() < deadline:
        it += 1
        if top_k is not None and top_k < len(c):
            idx = np.argpartition(c, -top_k)[-top_k:]
            w = np.zeros_like(c)
            w[idx] = 1.0 / top_k
        else:
            _, w = _smooth_max(c, beta)
        g = _cross_correlate(w, a)
        dot = float(np.dot(a, g))
        gz = a * (g - dot)
        gz -= gz.mean()
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * nesterov_mu * (mh + 0.1 * (1 - b1) * gz) / (np.sqrt(vh) + eps)
        z = np.clip(z, -40, 40)
        z -= z.mean()
        a = np.exp(z - z.max())
        a = a / a.sum()
        c = _full_convolve(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        if np.isfinite(score):
            if score < best_score - 1e-10:
                best_score = score
                best_z = z.copy()
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 80 == 0:
                    lr = max(0.005, lr * 0.92)
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0

        if it % 25 == 0:

            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a)
            beta = 0.05 * c.max()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0
    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score


def _targeted_descent(a, deadline, max_iters=2000):
    """Targeted coordinate descent: find the lag with max c[k], then reduce it."""
    n = len(a)
    a = np.maximum(a, 1e-12).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a)
        k_star = int(np.argmax(c))
        # c[k_star] = sum_i a[i]*a[k_star - i] for valid i
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        idx = np.arange(j_start, j_end + 1)
        products = a[idx] * a[k_star - idx]
        # Find the pair with the largest product and reduce it
        max_pos = int(np.argmax(products))
        i1 = idx[max_pos]
        i2 = k_star - i1
        # Try reducing the larger of the two
        if a[i1] >= a[i2]:
            for frac in [0.7, 0.5, 0.3]:
                a_try = a.copy()
                a_try[i1] *= frac
                a_try = a_try / a_try.sum()
                sc = _score_norm(a_try)
                if sc < best_score:
                    best_score = sc
                    best_a = a_try.copy()
                    a = a_try
                    break
        else:
            for frac in [0.7, 0.5, 0.3]:
                a_try = a.copy()
                a_try[i2] *= frac
                a_try = a_try / a_try.sum()
                sc = _score_norm(a_try)
                if sc < best_score:
                    best_score = sc
                    best_a = a_try.copy()
                    a = a_try
                    break
    return best_a, best_score


def _coord_descent(a, deadline, max_sweeps=20):
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            for factor in [0.6, 0.8, 1.25, 1.6]:
                a[j] = max(old_val * factor, 1e-15)
                a = a / a.sum()
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    old_val = a[j]
                else:
                    a[j] = old_val
                    a = a / a.sum()
                    break
        if not improved:
            break
    return best_a, best_score


def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new
