    Compute the subgradient of the top-K average of conv(a,a).
    Returns (gradient, current_max).
    O(K*n) time.
    """
    n = len(a)
    b = _fft_convolve(a, a)
    k_vals = np.argsort(b)[-K:]
    grad = np.zeros(n)

        hi = min(n - 1, kv)

    # df/da[j] = 2n * (1/p) * (sum b^p)^((1/p)-1) * sum_k p * b[k]^(p-1) * db[k]/da[j]
    #          = 2n * (sum b^p)^((1-p)/p) * sum_k b[k]^(p-1) * db[k]/da[j]
    
    # Let C = (sum b^p)^((1-p)/p)
    # Let u[k] = b[k]^(p-1)
    # grad[j] = 2n * C * sum_k u[k] * db[k]/da[j]
    
    # db[k]/da[j] = 2*a[k-j] (if 0<=k-j<n and k!=2j) + a[j] (if k=2j)
    #             = 2*a[k-j] - a[j]*[k==2j]
    
    # sum_k u[k] * db[k]/da[j] = 2*sum_k u[k]*a[k-j] - a[j]*u[2j] (if 2j < len(u))
    #                           = 2*h[j] - a[j]*u[2j]
    
    u = b  # u[k] = b[k]^(p-1), but we already computed b_pow = b^(p-1)
    # Actually u[k] = b_pow[k] / b_pow_sum... no, u[k] = b[k]^(p-1)
    u = b ** (p - 1) if p != 1 else np.ones_like(b)
    
    C = (b_avg := (b_pow_sum) ** (p / (p - 1)))  # (sum b^p) = b_pow_sum * max(b) ... no
    # sum_k b[k]^p = (b^(p-1) * b).sum() = (b_pow * b).sum()
    sum_bp = float((b_pow * b).sum())
    C = sum_bp ** ((1.0 - p) / p)
    
    grad = 2.0 * n * C * 2.0 * h
    # Diagonal correction
    for j in range(n):
        k_diag = 2 * j
        if k_diag < len(u):
            grad[j] -= 2.0 * n * C * a[j] * u[k_diag]
    
    current_score = float(2.0 * n * sum_bp ** (1.0 / p))
    return grad, current_score, float(b.max())


def _topk_direct_gradient(a, K):
    """
    Compute subgradient of top-K average of conv(a,a). O(K*n).
    Returns (gradient, max_b).
    """
    n = len(a)
    b = _fft_convolve(a, a)
    k_vals = np.argsort(b)[-K:]
    grad = np.zeros(n)
    for kv in k_vals:
        kv = int(kv)
        lo = max(0, kv - n + 1)
        hi = min(n - 1, kv)
        grad[lo:hi + 1] += 2.0 * a[kv - hi:kv - lo + 1][::-1]
        if kv % 2 == 0:
            j_diag = kv // 2
            if lo <= j_diag <= hi:
                grad[j_diag] -= a[j_diag]
    grad /= K
    return grad, float(b.max())


def _accelerated_pgd(a_init, deadline, p_start=200, p_end=50, K_start=10, K_end=1,
                     n_inner=5000, use_nesterov=True):
    """
    Accelerated projected gradient descent with adaptive power-sum / top-K smoothing.
    Works on the simplex directly.
    """
    n = len(a_init)
    a = a_init.copy()
    a = _project_simplex(a)
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    # Nesterov variables
    if use_nesterov:
        y = a.copy()
        beta_prev = 0.5
    else:
        y = a.copy()
        beta_prev = 0.5
        momentum = np.zeros(n)
    
    t = 0
    lr_base = 0.5  # Will be scaled
    
    start_time = time.monotonic()
    
    while time.monotonic() < deadline - 0.5:
        t += 1
        
        # Adaptive parameters
        frac = min(1.0, t / n_inner)
        p = int(p_start - (p_start - p_end) * frac)
        K = max(1, int(K_start - (K_start - K_end) * frac))
        
        # Compute gradient at y (Nesterov) or a
        if p >= 10:
            grad, curr_score, max_b = _power_sum_gradient(y, p)
        else:
            grad, max_b = _topk_direct_gradient(y, K)
            curr_score = 2.0 * n * max_b
        
        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n)
            curr_score = _score_norm(a)
        
        # Polyak step size with lower bound
        lower_bound = 2.0 * n / (2 * n - 1)  # Theoretical minimum
        gap = max(curr_score - lower_bound, 0.001)
        g_norm_sq = float(np.dot(grad, grad))
        if g_norm_sq < 1e-30:
            eta = 1e-6
        else:
            eta = gap / g_norm_sq
            # Cap the step size to avoid overshooting
            a_norm = float(np.linalg.norm(a))
            g_norm = math.sqrt(g_norm_sq) if g_norm_sq > 0 else 1e-15
            eta = min(eta, 0.5 / max(g_norm, 1e-10) if g_norm > 1e-10 else 1.0)
        
        # Nesterov momentum
        if use_nesterov:
            beta_curr = (1.0 + math.sqrt(1.0 + 4.0 * beta_prev ** 2)) / 2.0
            theta = (1.0 - beta_curr) / (1.0 + beta_prev)
            y_new = a + theta * (y - a)
        else:
            y_new = a.copy()
        
        # Gradient step
        a_trial = y_new - eta * grad
        a_new = _project_simplex(a_trial)
        
        # Check improvement
        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        # Always accept if it improves the objective (or use averaging)
        a = a_new
        if use_nesterov:
            y = y_new  # Will be updated next iteration
            beta_prev = beta_curr
        else:
            momentum = 0.9 * momentum + 0.1 * grad
            y = a - 0.01 * momentum
        
        # Periodic check
        if t % 500 == 0:
            a = _project_simplex(a)
            sc_a = _score_norm(a)
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()
    
    return best_a, best_score


def _lbfgs_smooth(a_init, deadline, p_values=None):
    """L-BFGS with power-sum smooth approximation (continuation)."""
    n = len(a_init)
    
    if p_values is None:
        p_values = [100, 50, 25, 15, 10, 7, 5]
    
    # Work in log-space with softmax
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    for p in p_values:
        if time.monotonic() > deadline - 2.0:
            break
        level_dl = min(deadline, time.monotonic() + max(5.0, (deadline - time.monotonic()) / (len(p_values) + 1)))
        
        def obj(z_):
            a_ = _softmax_local(z_)
            b_ = _fft_convolve(a_, a_)
            b_ = np.maximum(b_, 0.0)
            if p == 1:
                return 2.0 * n * b_.max()
            val = float((b_**p).sum())
            return 2.0 * n * val**(1.0/p)
        
        def grad_fn(z_):
            a_ = _softmax_local(z_)
            b_ = _fft_convolve(a_, a_)
            b_ = np.maximum(b_, 0.0)
            if p == 1:
                # Subgradient at max
                k_star = int(np.argmax(b_))
                g = np.zeros(n)
                lo = max(0, k_star - n + 1)
                hi = min(n - 1, k_star)
                g[lo:hi+1] = 2.0 * a_[k_star-hi:k_star-lo+1][::-1]
                if k_star % 2 == 0 and 0 <= k_star//2 < n:
                    j_d = k_star // 2
                    if lo <= j_d <= hi:
                        g[j_d] = a_[j_d]
                return 2.0 * n * g
            b_pow = b_ ** (p - 1)
            b_sum_p = float(b_.sum()**p / b_.sum()**(p-1))  # Not quite right
            # Actually: sum b^p
            sum_bp = float((b_**p).sum())
            if sum_bp < 1e-300:
                return np.zeros(n)
            u = b_ ** (p - 1)
            C = sum_bp ** ((1.0 - p) / p)
            
            # Compute h[j] = sum_k u[k] * a_[k-j] via convolution
            u_norm = u  # Already proportional to what we need
            flip_a = a_[::-1]
            N2 = _next_pow2(len(u) + n - 1)
            conv_val = np.fft.irfft(np.fft.rfft(u, N2) * np.fft.rfft(flip_a, N2), N2)[:len(u) + n - 1]
            h = conv_val[n-1:n-1+n]
            
            grad_a = 2.0 * n * C * 2.0 * h
            for j in range(n):
                k_d = 2 * j
                if k_d < len(u):
                    grad_a[j] -= 2.0 * n * C * a_[j] * u[k_d]
            
            # Chain rule: d/da * da/dz where a = softmax(z)
            # da_j/dz_k = a_j * (delta_jk - a_k)
            # dz gradient = a * (grad_a - a.dot(grad_a))
            inner = float(np.dot(a_, grad_a))
            grad_z = a_ * (grad_a - inner)
            return grad_z
        
        try:
            result = minimize(
                obj, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-8,
                         'maxls': 30, 'maxfun': 5000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_local(result.x)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _softmax_local(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _local_search(a_init, deadline, rng, n_trials=10000):
    """Block perturbation local search on the simplex."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 0.5:
            break
        
        block_size = int(rng.integers(3, min(30, n // 4)))
        start = int(rng.integers(0, n - block_size))
        
        a_trial = a.copy()
        # Random multiplicative perturbation
        pert = rng.uniform(0.7, 1.3, size=block_size)
        a_trial[start:start + block_size] *= pert
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial /= s
        
        sc = _score_norm(a_trial)
        if sc < best_score * (1.0 - 1e-9):
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
    
    return best_a, best_score


def _make_profile(n, profile_type, *args):
    """Generate various profiles."""
    if profile_type == 'uniform':
        return np.ones(n) / n
    elif profile_type == 'linear':
        slope = args[0] if args else 1.0
        t = np.linspace(0, 1, n)
        a = 1.0 + slope * (2 * t - 1)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif profile_type == 'quadratic':
        c1, c2 = (args + (0.0, 0.0))[:2]
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + c1 * u + c2 * u * u
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif profile_type == 'gaussian':
        sigma = args[0] if args else 0.3
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
        return a / a.sum()
    elif profile_type == 'hamming':
        k = np.arange(n)
        a = 0.54 - 0.46 * np.cos(2 * np.pi * k / max(1, n - 1))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif profile_type == 'welch':
        k = np.arange(n)
        mid = (n - 1) / 2.0
        a = 1 - ((k - mid) / max(1, mid)) ** 2
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif profile_type == 'raised_cos':
        alpha = args[0] if args else 1.0
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * alpha * (t - 0.5)))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif profile_type == 'bump':
        width = args[0] if args else 0.2
        amplitude = args[1] if len(args) > 1 else 0.5
        t = np.linspace(0, 1, n)
        a = 1.0 + amplitude * np.exp(-0.5 * ((t - 0.5) / width) ** 2)
        return a / a.sum()
    elif profile_type == 'power':
        p = args[0] if args else 2.0
        t = np.linspace(0, 1, n)
        a = t**p + (1 - t)**p + 0.1
        return a / a.sum()
    elif profile_type == 'sine_tilt':
        amp = args[0] if args else 0.3
        t = np.linspace(0, 1, n)
        a = 1.0 + amp * np.sin(np.pi * t) * (2 * t - 1)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = len(p)
    if n0 == n_new:
        return p / p.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
