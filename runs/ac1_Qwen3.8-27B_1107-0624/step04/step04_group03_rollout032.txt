    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)

Multi-phase optimizer for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Combines L-BFGS continuation, LP-based Frank-Wolfe directions, FISTA,

import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)

    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _make_profile(n, ptype, **kw):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if ptype == 'uniform':
        a = np.ones(n)
    elif ptype == 'linear':
        slope = kw.get('slope', 1.5)
        a = 1.0 + slope * (t - 0.5)
    elif ptype == 'quadratic':
        a1 = kw.get('a1', 1.0)
        a2 = kw.get('a2', 0.2)
        a = 1.0 + a1 * u + a2 * u ** 2
    elif ptype == 'cubic':
        a1 = kw.get('a1', 0.8)
        a3 = kw.get('a3', -0.4)
        a = 1.0 + a1 * u + a3 * u ** 3
    elif ptype == 'gaussian':
        sigma = kw.get('sigma', 0.3)
        a = np.exp(-u ** 2 / (2 * sigma ** 2))
    elif ptype == 'half_sine':
        a = np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
    elif ptype == 'parabolic':
        a = 1.0 - u ** 2
        a = np.maximum(a, 1e-10)
    elif ptype == 'trapezoid':
        w = kw.get('w', 0.5)
        a = np.where(np.abs(u) < w, 1.0, np.maximum(0, 1 - (np.abs(u) - w) / (1 - w)))
        a = np.maximum(a, 1e-10)
    elif ptype == 'cosine_bell':
        a = 0.5 * (1.0 + np.cos(np.pi * u))
        a = np.maximum(a, 1e-10)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _subgradient_with_weights(a, w):
    n = len(a)
    W = float(np.dot(w, _full_convolve(a, a)))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    return 2.0 * a * (h - W)


def _lp_direction(a, M, K=100, max_time=5.0):
    """Solve LP: max sum(b) s.t. T_a[:,topK] @ b <= M, b >= 0.
    Returns normalized direction b (sum=1) or None."""
    n = len(a)
    t0 = time.monotonic()
    c = _full_convolve(a, a)
    top_k = np.argsort(c)[-K:]

    A_ub = np.zeros((K, n))
    for k_idx in range(K):
        k = int(top_k[k_idx])
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_end >= j_start:
            A_ub[k_idx, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]

    c_obj = -np.ones(n)
    b_ub = np.full(K, M)

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                         bounds=[(0, None)] * n, method='highs',
                         options={'time_limit': max_time, 'presolve': True})
        if time.monotonic() - t0 > max_time * 1.5:
            return None
        if result.success and result.x is not None:
            b = np.maximum(result.x, 0)
            s = b.sum()
            if s > 1e-15:
                b = b / s
                if np.all(np.isfinite(b)):
                    return b
    except Exception:
        pass
    return None


def _active_set_weights(a, K=50, max_time=3.0):
    """Solve dual LP to get optimal weights on top-K lags.
    Returns weight vector w (length 2n-1) concentrated on top-K lags."""
    n = len(a)
    t0 = time.monotonic()
    c = _full_convolve(a, a)
    M = float(c.max())
    top_k = np.argsort(c)[-K:]

    c_obj = M * np.ones(K)
    A_neg = np.zeros((n, K))
    for k_idx in range(K):
        k = int(top_k[k_idx])
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_end >= j_start:
            A_neg[j_start:j_end + 1, k_idx] = -a[k - j_end:k - j_start + 1][::-1]

    b_ub = -np.ones(n)
    try:
        result = linprog(c_obj, A_ub=A_neg, b_ub=b_ub,
                         bounds=[(0, None)] * K, method='highs',
                         options={'time_limit': max_time})
        if result.success and result.x is not None:
            y = np.maximum(result.x, 0)
            w = np.zeros(2 * n - 1)
            for k_idx in range(K):
                w[int(top_k[k_idx])] = y[k_idx]
            s = w.sum()
            if s > 1e-30:
                w = w / s
            else:
                w[int(np.argmax(c))] = 1.0
            return w
    except Exception:
        pass

    if time.monotonic() - t0 > max_time:
        return None
    w = np.zeros(2 * n - 1)
    w[int(np.argmax(c))] = 1.0
    return w


def _run_lbfgs(a_init, deadline, max_wall_time=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if max_wall_time is None:
        max_wall_time = min(300.0, max(1.0, deadline - time.monotonic()))

    betas = [0.1, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015,
             0.0008, 0.0004, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6]
    betas = [b * max_c0 for b in betas]
    betas = [b for b in betas if b > 1e-15]

    per_level = max_wall_time / max(1, len(betas))

    for beta in betas:
        if time.monotonic() >= deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level)
        try:
            result = minimize(
                _objective_smooth, z, args=(beta,),
                method='L-BFGS-B', jac=_gradient_smooth,
                options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 3000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z = result.x.copy()
                a_new = _softmax_fn(z)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
        except Exception:
            pass

    a_final = _softmax_fn(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _fista_optimize(a_init, deadline, beta_init_frac=0.01, n_inner=5000):
    """FISTA on the smoothed problem with adaptive beta."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())

    a = a_init.copy()
    a_prev = a_init.copy()
    t_mom = 1.0
    lr = 0.5 / (4.0 * n * max(a_init.max(), 1e-10))

    beta = max(1e-12, beta_init_frac * max_c)

    for it in range(n_inner):
        if time.monotonic() >= deadline:
            break

        a_curr = a
        c = _full_convolve(a_curr, a_curr)
        w, cm = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        a_rev = a_curr[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a_curr * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2)) / 2.0
        y = a + ((t_mom - 1.0) / t_new) * (a - a_prev)
        a_new = y - lr * grad
        a_new = _project_simplex(a_new)

        a_prev = a
        a = a_new
        t_mom = t_new

        if it % 100 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

        if it % 500 == 0:
            beta = max(1e-15, beta * 0.7)
            c_new = _full_convolve(a, a)
            max_c = float(c_new.max())

    a_final = a
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _lw_optimize(a_init, deadline, K=100, n_iter=2000):
    """LP-based Frank-Wolfe: at each step, solve LP for direction b,
    then update a = (1-t)*a + t*b."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    a = a_init.copy()

    lr = 0.1
    for it in range(n_iter):
        if time.monotonic() >= deadline:
            break

        c = _full_convolve(a, a)
        M = float(c.max())

        b = _lp_direction(a, M, K=K, max_time=2.0)
        if b is None:
            break

        t = 2.0 / (it + 2) * lr
        a_new = (1.0 - t) * a + t * b
        a_new = _project_simplex(a_new)

        sc = _score_norm(a_new)
        if np.isfinite(sc):
            a = a_new
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def _subgradient_optimize(a_init, deadline, use_lp=True, K=50, n_iter=10000):
    """Projected subgradient with (optionally) LP-based active set weights."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    a = a_init.copy()

    lr = 0.05
    momentum = np.zeros(n)
    mom_coeff = 0.8

    c = _full_convolve(a, a)
    max_c = float(c.max())

    for it in range(n_iter):
        if time.monotonic() >= deadline:
            break

        c = _full_convolve(a, a)
        M = float(c.max())

        if use_lp and it % 10 == 0:
            w = _active_set_weights(a, K=K, max_time=1.0)
            if w is None:
                w, cm = _compute_weights(c, 0.01 * M)
        else:
            w, cm = _compute_weights(c, 0.005 * M)

        grad = _subgradient_with_weights(a, w)

        if not np.all(np.isfinite(grad)):
            continue

        momentum = mom_coeff * momentum - lr * grad
        a_new = _project_simplex(a + momentum)

        if it % 50 == 0:
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
            else:
                lr *= 0.95
                a = a_new
        else:
            a = a_new

        if it % 500 == 0:
            mom_coeff = min(0.95, mom_coeff + 0.01)

    sc_final = _score_norm(a)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score


def _nesterov_optimize(a_init, deadline, beta_fracs, lr_init=0.02, mom_init=0.9):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())

    lr = lr_init
    mom = mom_init

    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-14, beta_frac * max_c)

        while time.monotonic() < deadline:
            grad_y = _gradient_smooth(y, beta)
            if not np.all(np.isfinite(grad_y)):
                grad_y = _gradient_smooth(z, beta)

            z_new = z - lr * grad_y
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -60, 60)

            y_new = z_new + mom * (z_new - z)
            y_new = y_new - y_new.mean()

            z = z_new
            y = y_new

            if True:
                a_curr = _softmax_fn(z)
                sc = _score_norm(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()

        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        max_c = float(c.max())
        lr = max(0.001, lr * 0.6)

    a_final = _softmax_fn(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS continuation -> LP Frank-Wolfe ->
    FISTA -> active-set subgradient -> aggressive restarts -> final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s)) - 20.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_parent = _score_norm(parent)
        print(f"Parent: n={n}, score={2.0*n*sc_parent/(n*n):.10f} (norm={sc_parent:.10f})")
    else:
        n = 4096
        parent = _make_profile(n, 'cosine_bell')
        sc_parent = _score_norm(parent)
        print(f"No parent, using cosine_bell n={n}, score={_score_fn(parent):.10f}")

    update_best(parent, sc_parent, "parent")

    # --- Phase 0: Evaluate multiple profiles (30s) ---
    phase0_end = min(deadline, t_start + 30.0)
    candidates = [(sc_parent, parent.copy(), "parent")]

    profiles_to_try = [
        ('gaussian', {'sigma': 0.35}),
        ('gaussian', {'sigma': 0.45}),
        ('half_sine', {}),
        ('parabolic', {}),
        ('cosine_bell', {}),
        ('trapezoid', {'w': 0.3}),
        ('trapezoid', {'w': 0.6}),
        ('quadratic', {'a1': 0.0, 'a2': 0.5}),
        ('quadratic', {'a1': 1.5, 'a2': -0.3}),
    ]
    for ptype, kwargs in profiles_to_try:
        if time.monotonic() > phase0_end:
            break
        p = _make_profile(n, ptype, **kwargs)
        sc = _score_norm(p)
        candidates.append((sc, p, f"{ptype}"))
        if sc < best_score:
            best_seq = p.copy()
            best_score = sc
            print(f"  Phase0: {ptype} score={sc:.10f}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best_score={best_score:.10f}")

    # --- Phase 1: L-BFGS continuation (400s) ---
    phase1_end = min(deadline, t_start + 430.0)
    if best_seq is not None and time.monotonic() < phase1_end:
        per_time = min(380.0, phase1_end - time.monotonic())
        if per_time > 10:
            p_best, sc_best = _run_lbfgs(best_seq, min(phase1_end - 5, time.monotonic() + per_time),
                                         max_wall_time=per_time)
            update_best(p_best, sc_best, "lbfgs_p1")

    # Also try L-BFGS on 2nd best candidate
    if time.monotonic() < phase1_end - 60:
        for sc_c, p_c, label in candidates[1:3]:
            if time.monotonic() > phase1_end - 30:
                break
            per_time = min(80.0, phase1_end - time.monotonic() - 10)
            if per_time > 10:
                p_best, sc_best = _run_lbfgs(p_c, min(phase1_end - 10, time.monotonic() + per_time),
                                             max_wall_time=per_time)
                update_best(p_best, sc_best, f"lbfgs_{label}")

    print(f"Phase 1 done: best_score={best_score:.10f}")

    # --- Phase 2: LP-based Frank-Wolfe (300s) ---
    phase2_end = min(deadline, t_start + 730.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        per_time = min(280.0, phase2_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _lw_optimize(best_seq, min(phase2_end - 5, time.monotonic() + per_time),
                                           K=80, n_iter=500)
            update_best(p_best, sc_best, "lp_fw_p2")

    print(f"Phase 2 done: best_score={best_score:.10f}")

    # --- Phase 3: FISTA (300s) ---
    phase3_end = min(deadline, t_start + 1030.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        per_time = min(280.0, phase3_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _fista_optimize(best_seq, min(phase3_end - 5, time.monotonic() + per_time),
                                              beta_init_frac=0.005, n_inner=50000)
            update_best(p_best, sc_best, "fista_p3")

    # Also try Nesterov if time
    if time.monotonic() < phase3_end - 30:
        beta_fracs = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
        p_best, sc_best = _nesterov_optimize(best_seq, min(phase3_end - 5, time.monotonic()),
                                             beta_fracs, lr_init=0.012, mom_init=0.88)
        update_best(p_best, sc_best, "nesterov_p3")

    print(f"Phase 3 done: best_score={best_score:.10f}")

    # --- Phase 4: Active-set subgradient (250s) ---
    phase4_end = min(deadline, t_start + 1280.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        per_time = min(230.0, phase4_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _subgradient_optimize(best_seq, min(phase4_end - 5, time.monotonic() + per_time),
                                                    use_lp=True, K=40, n_iter=20000)
            update_best(p_best, sc_best, "activeset_p4")

    print(f"Phase 4 done: best_score={best_score:.10f}")

    # --- Phase 5: Aggressive restarts (350s) ---
    phase5_end = min(deadline, t_start + 1630.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        n_best = len(best_seq)
        n_restarts = 6
        per_restart = min(55.0, (phase5_end - time.monotonic()) / max(1, n_restarts))

        perturbations = [
            ('gaussian_noise', 0.03),
            ('gaussian_noise', 0.08),
            ('gaussian_noise', 0.15),
            ('sinusoidal', 0.1),
            ('swap_pairs', 50),
            ('mirror_blend', 0.3),
        ]

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            ptype, param = perturbations[i % len(perturbations)]

            if ptype == 'gaussian_noise':
                noise = param * rng.normal(size=n_best)
                p_start = best_seq + noise * best_seq
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            elif ptype == 'sinusoidal':
                freq = rng.integers(3, 20)
                phase = rng.uniform(0, 2 * np.pi)
                t = np.linspace(0, 1, n_best)
                pert = 1.0 + param * np.sin(2 * np.pi * freq * t + phase)
                p_start = best_seq * pert
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            elif ptype == 'swap_pairs':
                p_start = best_seq.copy()
                for _ in range(param):
                    i1, i2 = rng.integers(0, n_best, 2)
                    p_start[i1], p_start[i2] = p_start[i2], p_start[i1]
                p_start = p_start / p_start.sum()
            elif ptype == 'mirror_blend':
                p_rev = best_seq[::-1]
                p_start = (1.0 - param) * best_seq + param * p_rev
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            else:
                p_start = best_seq.copy()

            sc_start = _score_norm(p_start)
            if sc_start < best_score:
                update_best(p_start, sc_start, f"restart{i}_init")

            per_dl = min(phase5_end - 5, time.monotonic() + per_restart)
            beta_fracs_short = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]
            p_best, sc_best = _nesterov_optimize(p_start, per_dl, beta_fracs_short,
                                                 lr_init=0.01, mom_init=0.85)
            update_best(p_best, sc_best, f"restart{i}_{ptype}")

            # Quick FISTA polish
            if time.monotonic() < phase5_end - 15:
                p_best2, sc_best2 = _fista_optimize(p_best, min(phase5_end - 5, time.monotonic() + 15),
                                                    beta_init_frac=0.001, n_inner=10000)
                update_best(p_best2, sc_best2, f"restart{i}_fista")

    print(f"Phase 5 done: best_score={best_score:.10f}")

    # --- Phase 6: Try different length (100s) ---
    phase6_end = min(deadline, t_start + 1730.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        n_curr = len(best_seq)
        for target_n in [n_curr * 2, n_curr // 2]:
            if target_n < 256 or target_n > 8192:
                continue
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm(p_interp)
            if sc_interp < best_score:
                update_best(p_interp, sc_interp, f"interp_n{target_n}")
            elif time.monotonic() < phase6_end - 20:
                per_time = min(40.0, phase6_end - time.monotonic() - 10)
                if per_time > 10:
                    p_best, sc_best = _run_lbfgs(p_interp, min(phase6_end - 5, time.monotonic() + per_time),
                                                 max_wall_time=per_time)
                    update_best(p_best, sc_best, f"lbfgs_n{target_n}")

    print(f"Phase 6 done: best_score={best_score:.10f}")

    # --- Phase 7: Final fine polish (remaining) ---
    phase7_end = deadline
    if best_seq is not None and time.monotonic() < phase7_end:
        remaining = phase7_end - time.monotonic()
        if remaining > 50:
            # Final Nesterov with tiny beta
            per_time = min(80.0, remaining - 30)
            beta_fracs_final = [1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]
            p_best, sc_best = _nesterov_optimize(best_seq, min(phase7_end - 30, time.monotonic() + per_time),
                                                  beta_fracs_final, lr_init=0.003, mom_init=0.96)
            update_best(p_best, sc_best, "final_nesterov")

        if time.monotonic() < phase7_end - 15:
            per_time = min(50.0, phase7_end - time.monotonic() - 5)
            if per_time > 10:
                p_best, sc_best = _fista_optimize(best_seq, min(phase7_end - 5, time.monotonic() + per_time),
                                                  beta_init_frac=0.0005, n_inner=20000)
                update_best(p_best, sc_best, "final_fista")

        if time.monotonic() < phase7_end - 10:
            per_time = min(30.0, phase7_end - time.monotonic() - 3)
            if per_time > 5:
                p_best, sc_best = _subgradient_optimize(best_seq, min(phase7_end - 2, time.monotonic() + per_time),
                                                        use_lp=True, K=30, n_iter=5000)
                update_best(p_best, sc_best, "final_subgrad")

    print(f"Phase 7 done: best_score={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profile(4096, 'cosine_bell')

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, norm_score={_score_norm(final_seq):.10f}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned non-finite, using internal: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    fallback = _make_profile(4096, 'cosine_bell')
    print(f"FALLBACK: n=4096, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
