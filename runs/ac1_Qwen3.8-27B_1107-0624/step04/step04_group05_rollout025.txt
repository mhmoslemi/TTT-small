        s_b = float(b.sum())
        if s_b < 1e-10:

                a_try = a_try / a_try.sum()
                sc_try = _score_norm(a_try)

                    best_t = tc

def _adam_logit(p0, deadline, rng, lr=0.05, beta_frac=0.08, beta_decay=0.93,
                no_imp_limit=500, base_lr=None):
    """Adam optimizer in logit space with adaptive temperature for smoothed max."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    if base_lr is None:
        base_lr = lr

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-4, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    lr_cur = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_and_weights(c, beta)
        g = _conv_cross_corr(w, p)

        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr_cur * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr_cur = max(0.003, lr_cur * 0.92)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 30 == 0:
            beta = max(5e-5, beta * beta_decay)
            if c.size >= 2:
                top5 = np.sort(c)[::-1][:min(5, c.size)]
                gap = top5[0] - top5[-1]
                if gap < beta * 0.05:
                    beta = max(5e-5, beta * 0.7)

        if no_imp > no_imp_limit:
            if best_score < 1.51:
                ns = 0.04
            elif best_score < 1.53:
                ns = 0.08
            else:
                ns = 0.15
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-4, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr_cur = base_lr
            no_imp = 0

    return best_p, best_score

    c = _full_convolve(a)
    maxc = float(c.max())
    peak_indices = np.where(c >= maxc * 0.999)[0]

    trial = 0
    stall = 0
    while time.monotonic() < deadline and trial < n_trials:
        trial += 1
        if stall > 80:
            break

        # Pick a peak lag
        k_star = int(rng.choice(peak_indices)) if len(peak_indices) > 0 else n - 1

        # Compute contribution of each position i to b[k_star]
        # b[k_star] = sum_i a[i] * a[k_star - i] (valid indices)
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        total_contrib = float(contrib.sum())
        if total_contrib < 1e-15:
            break

        # Find high-contrib and low-contrib positions
        high_mask = contrib > np.percentile(contrib[contrib > 0], 90) if (contrib > 0).sum() > 10 else contrib > 0
        low_mask = contrib < np.percentile(contrib[contrib > 0], 10) if (contrib > 0).sum() > 10 else np.zeros(n, dtype=bool)

        high_positions = np.where(high_mask)[0]
        low_positions = np.where(low_mask)[0]

        if len(high_positions) == 0 or len(low_positions) == 0:
            stall += 1
            continue

        # Move mass from a high-contrib position to a low-contrib position
        i_from = int(rng.choice(high_positions))
        i_to = int(rng.choice(low_positions))

        # Try small moves in both directions
        best_sc = current_score
        best_a = None
        for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
            delta = frac * a[i_from]
            if delta < 1e-15:
                continue
            a_try = a.copy()
            a_try[i_from] -= delta
            a_try[i_to] += delta
            if a_try[i_from] < 0:
                continue
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            sc = _score_norm(a_try)
            if sc < best_sc:
                best_sc = sc
                best_a = a_try.copy()

        if best_a is not None and best_sc < current_score - 1e-12:
            a = best_a
            current_best = a.copy()
            current_score = best_sc
            c = _full_convolve(a)
            maxc = float(c.max())
            peak_indices = np.where(c >= maxc * 0.999)[0]
            stall = 0
        else:
            stall += 1

    return current_best, current_score


def _alternating_lp_adam(a, deadline, rng, max_rounds=20):
    """Alternate between LP refinement and Adam polishing."""
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    round_time = 60.0

    for rnd in range(max_rounds):
        if time.monotonic() >= deadline:
            break

        # LP phase
        lp_deadline = min(time.monotonic() + round_time * 0.6, deadline - 10)
        if time.monotonic() < lp_deadline:
            p_lp, sc_lp = _lp_refine_adaptive(current_best, lp_deadline, max_solves=30, line_search=True)
            if sc_lp < current_score:
                current_score = sc_lp
                current_best = p_lp.copy()

        # Adam phase
        ad_deadline = min(time.monotonic() + round_time * 0.4, deadline - 5)
        if time.monotonic() < ad_deadline:
            p_ad, sc_ad = _adam_logit(current_best, ad_deadline, rng, lr=0.03,
                                      beta_frac=0.05, beta_decay=0.95, no_imp_limit=200)
            if sc_ad < current_score:
                current_score = sc_ad
                current_best = p_ad.copy()

        # Peak targeted
        if time.monotonic() < deadline - 30:
            pt_deadline = min(time.monotonic() + 30, deadline - 10)
            p_pt, sc_pt = _peak_targeted_search(current_best, pt_deadline, rng, n_trials=100)
            if sc_pt < current_score:
                current_score = sc_pt
                current_best = p_pt.copy()

    return current_best, current_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}]")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Started. Parent n={pn}, budget={budget_s}s")

    # Determine working lengths
    lengths_to_try = []
    if parent is not None and 0 < pn <= 4096:
        lengths_to_try.append(pn)
    if pn != 1024:
        lengths_to_try.append(1024)
    if pn != 2048:
        lengths_to_try.append(2048)

    # Prepare initial candidates for each length
    init_candidates = {}
    for nl in lengths_to_try:
        cands = []
        if parent is not None:
            if pn == nl:
                sc = _score_norm(parent)
                cands.append((sc, parent.copy(), 'parent'))
            else:
                pi = _interp_profile(parent, nl)
                if pi is not None:
                    sc = _score_norm(pi)
                    cands.append((sc, pi, 'interp'))
        # Add some profile-based candidates
        profiles = [
            _make_profile_linear(nl, 1.0),
            _make_profile_linear(nl, 1.5),
            _make_profile_linear(nl, 2.0),
            _make_profile_cosine(nl, 1, 0.5),
            _make_profile_cosine(nl, 2, 0.3),
            _make_profile_quadratic(nl, 1.0, -0.5),
        ]
        for i, prof in enumerate(profiles):
            sc = _score_norm(prof)
            cands.append((sc, prof, f'prof{i}'))
        cands.sort(key=lambda x: x[0])
        init_candidates[nl] = cands

    # Phase 1: Main optimization on best length (first 800s)
    phase1_deadline = t_start + 800.0
    # Start with the best initial candidate
    all_inits = []
    for nl in lengths_to_try:
        for sc, p, label in init_candidates[nl]:
            all_inits.append((sc, p, nl, label))
    all_inits.sort(key=lambda x: x[0])

    if all_inits:
        sc0, p0, n0, label0 = all_inits[0]
        print(f"Starting with n={n0} score={sc0:.10f} [{label0}]")
        _update(p0, sc0, 'init')

        # LP refinement on best
        lp_end = min(phase1_deadline - 60, t_start + 300)
        if best_seq is not None and best_seq.size <= 2048:
            print(f"  LP Phase A n={best_seq.size}")
            p_lp, sc_lp = _lp_refine_adaptive(best_seq, lp_end, max_solves=50, line_search=True)
            _update(p_lp, sc_lp, 'LP_A')

        # Adam polish
        if time.monotonic() < phase1_deadline - 30 and best_seq is not None:
            ad_end = min(phase1_deadline - 10, time.monotonic() + 200)
            print(f"  Adam Phase A n={best_seq.size}")
            p_ad, sc_ad = _adam_logit(best_seq, ad_end, rng, lr=0.04, beta_frac=0.06,
                                      beta_decay=0.94, no_imp_limit=300)
            _update(p_ad, sc_ad, 'Adam_A')

        # More LP
        if time.monotonic() < phase1_deadline - 10 and best_seq is not None and best_seq.size <= 2048:
            lp_end2 = min(phase1_deadline - 5, time.monotonic() + 150)
            print(f"  LP Phase B n={best_seq.size}")
            p_lp2, sc_lp2 = _lp_refine_adaptive(best_seq, lp_end2, max_solves=40, line_search=True)
            _update(p_lp2, sc_lp2, 'LP_B')

    # Phase 2: Try other lengths (800s-1100s)
    phase2_deadline = t_start + 1100.0
    tried_lengths = {best_seq.size} if best_seq is not None else set()
    for nl in lengths_to_try:
        if nl in tried_lengths or time.monotonic() >= phase2_deadline - 60:
            continue
        if nl > 2048:
            continue
        candidates = init_candidates.get(nl, [])
        if not candidates:
            continue
        sc_c, p_c, _ = candidates[0]
        print(f"  Trying n={nl}, init score={sc_c:.10f}")
        # Quick LP
        lp_end_c = min(phase2_deadline - 30, time.monotonic() + 80)
        if time.monotonic() < lp_end_c:
            p_c2, sc_c2 = _lp_refine_adaptive(p_c, lp_end_c, max_solves=25, line_search=True)
            _update(p_c2, sc_c2, f'LP_n{nl}')

        # Quick Adam
        if time.monotonic() < phase2_deadline - 10 and best_seq is not None:
            ad_end_c = min(phase2_deadline - 10, time.monotonic() + 80)
            p_c3, sc_c3 = _adam_logit(p_c2 if 'p_c2' in dir() else p_c, ad_end_c, rng,
                                       lr=0.04, beta_frac=0.06, beta_decay=0.94, no_imp_limit=200)
            _update(p_c3, sc_c3, f'Adam_n{nl}')

    # Phase 3: Alternating LP + Adam + Peak search (1100s-1500s)
    phase3_deadline = t_start + 1500.0
    if time.monotonic() < phase3_deadline and best_seq is not None:
        print(f"  Alternating phase, n={best_seq.size}, score={best_score:.10f}")
        p_alt, sc_alt = _alternating_lp_adam(best_seq, phase3_deadline, rng, max_rounds=15)
        _update(p_alt, sc_alt, 'alternating')

    # Phase 4: Multi-restart with annealing (1500s-1770s)
    phase4_deadline = deadline - 15.0
    if time.monotonic() < phase4_deadline and best_seq is not None:
        n_best = best_seq.size
        n_restarts = 6
        per_restart = max(10.0, (phase4_deadline - time.monotonic()) / (n_restarts + 1))
        print(f"  Multi-restart phase, n={n_best}")

        for i in range(n_restarts):
            if time.monotonic() > phase4_deadline - 10:
                break
            ns = 0.02 + 0.03 * i  # Increasing noise
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -60, 60)
            z -= z.mean()
            p_s = _softmax_from_z(z)

            # Adam
            ad_end_r = min(time.monotonic() + per_restart * 0.5, phase4_deadline - 20)
            if time.monotonic() < ad_end_r:
                p_r, sc_r = _adam_logit(p_s, ad_end_r, rng, lr=0.03, beta_frac=0.05,
                                         beta_decay=0.95, no_imp_limit=150)
                _update(p_r, sc_r, f'restart{i}')

            # LP on best
            if time.monotonic() < phase4_deadline - 10 and best_seq.size <= 2048:
                lp_end_r = min(time.monotonic() + 20, phase4_deadline - 8)
                if time.monotonic() < lp_end_r:
                    p_lr, sc_lr = _lp_refine_adaptive(best_seq, lp_end_r, max_solves=12, line_search=True)
                    _update(p_lr, sc_lr, f'LP_r{i}')

    # Phase 5: Final intensive LP on absolute best
    if time.monotonic() < deadline - 3 and best_seq is not None and best_seq.size <= 2048:
        per_final = deadline - time.monotonic() - 3
        if per_final > 15:
            print(f"  Final LP polish, score={best_score:.10f}")
            p_f, sc_f = _lp_refine_adaptive(best_seq, time.monotonic() + per_final * 0.7,
                                             max_solves=15, line_search=True)
            _update(p_f, sc_f, 'final_LP')

        if time.monotonic() < deadline - 3:
            per_f2 = deadline - time.monotonic() - 3
            if per_f2 > 8:
                p_f2, sc_f2 = _adam_logit(best_seq, time.monotonic() + per_f2, rng,
                                           lr=0.01, beta_frac=0.03, beta_decay=0.97,
                                           no_imp_limit=80)
                _update(p_f2, sc_f2, 'final_adam')

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(1024, 1.5)
            s = float(final.sum())
            final = final / s

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile_linear(1024, 1.5)
    print(f"FALLBACK n=1024")
    return [float(x) for x in fb]
