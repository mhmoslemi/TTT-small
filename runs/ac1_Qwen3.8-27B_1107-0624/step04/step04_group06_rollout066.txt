def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(p):

def _grad_smooth_max(p, beta):
    """Gradient of smooth_max(convolve(p,p)) w.r.t. p."""
    n = len(p)
    c = _full_convolve(p)
    mval, w = _smooth_max_weights(c, beta)

    # Use FFT: cross-correlation of w (len 2n-1) and p (len n)
    N = _next_pow2(3 * n)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p, N)
    R = np.fft.irfft(np.conj(fw) * fp, N)
    grad = 2.0 * R[:n]
    return mval, grad

def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def _gradient_mutation(p, rng, lr=0.03, beta_frac=0.08, noise_frac=0.05):

    gnorm = float(np.sqrt(np.sum(grad_p ** 2)))

def _crossover(p1, p2, rng):
    """Convex combination crossover."""
    alpha = rng.beta(2.0, 2.0)
    child = alpha * p1 + (1.0 - alpha) * p2
    child = np.maximum(child, 0.0)
    s = float(child.sum())
    if s < 1e-30:
        child = (p1 + p2) / 2.0
        s = float(child.sum())
    return child / s


def _swap_mutation(p, rng, n_swaps=5):
    """Move small amounts of mass between random positions."""
    n = len(p)
    p_new = p.copy()
    for _ in range(n_swaps):
        i = rng.integers(0, n)
        j = rng.integers(0, n)

            continue
        frac = rng.uniform(0.01, 0.05)
        amount = p_new[i] * frac
        p_new[i] -= amount
        p_new[j] += amount
    p_new = np.maximum(p_new, 0.0)
    p_new = p_new / p_new.sum()
    return p_new


def _block_mutation(p, rng):

def _resample(p, n_new, rng):
    """Resample a sequence to a new length using random加权."""
    n_old = len(p)
    if n_old == n_new:
        return p.copy()
    # Use cumulative distribution for resampling
    cdf = np.cumsum(p) / p.sum()
    u = rng.uniform(0, 1, size=n_new)
    idx = np.searchsorted(cdf, u)
    idx = np.clip(idx, 0, n_old - 1)
    p_new = p[idx]
    p_new = p_new / p_new.sum()
    return p_new

                    if i_s <= i_e:
                        partner = k - np.arange(i_s, i_e + 1)
                        heat[i_s:i_e + 1] += p_parent[partner]
                heat_max = float(heat.max())
                if heat_max > 1e-15:
                    heat = heat / heat_max
                    factor = 1.0 - 0.05 * heat
                    factor = np.maximum(factor, 0.01)
                    p_child = p_parent * factor
                    p_child = np.maximum(p_child, 1e-15)
                    p_child = p_child / p_child.sum()
                else:
                    p_child = _swap_mutation(p_parent, rng, n_swaps=3)
            elif r < 0.85:
                # Swap + block mutation
                parent_idx = rng.integers(0, max(2, pop_size // 3))
                p_parent = population[parent_idx]
                p_child = _swap_mutation(p_parent, rng, n_swaps=rng.integers(3, 8))
                if rng.random() < 0.5:
                    p_child = _block_mutation(p_child, rng)
            else:
                # Full random restart (rare)
                r2 = rng.random()
                if r2 < 0.3:
                    p_child = _random_sequence(n_ga, rng, alpha=rng.uniform(1.0, 4.0))
                elif r2 < 0.5:
                    eta = rng.uniform(-3.0, 4.0)
                    p_child = _make_linear(n_ga, eta)
                elif r2 < 0.7:
                    freq = rng.integers(1, 5)
                    amp = rng.uniform(0.5, 2.0)
                    p_child = _make_cosine(n_ga, freq, amp)
                else:
                    # Resample best
                    p_child = _resample(best_seq, n_ga, rng)

            sc_child = _score(p_child)
            offspring.append(p_child)
            offspring_scores.append(sc_child)

        # Selection: keep best from population + offspring
        all_inds = population + offspring
        all_scores = scores + offspring_scores
        # Elitism: keep top 2 from current population
        n_keep = 2
        # Fill rest with best offspring
        order_off = np.argsort(offspring_scores)
        n_off_keep = pop_size - n_keep
        keep_off = [offspring[i] for i in order_off[:n_off_keep]]
        keep_off_scores = [offspring_scores[i] for i in order_off[:n_off_keep]]

        population = [population[i] for i in order[:n_keep]] + keep_off
        scores = [scores[i] for i in order[:n_keep]] + keep_off_scores

        # Update global best
        for i, ind in enumerate(population):
            if scores[i] < best_score:
                best_seq = ind.copy()
                best_score = scores[i]
                no_global_imp = 0
                print(f"  GA gen {ga_generations}: NEW BEST {best_score:.10f}")
                break
        else:
            no_global_imp += 1

        # Periodic intensive local search on best
        if ga_generations % 10 == 0 and time.monotonic() < phase4_end - 60:
            time_for_ls = min(30.0, phase4_end - time.monotonic() - 30)
            if time_for_ls > 15:
                z_best = _to_logit(best_seq)
                p_ls, z_ls, sc_ls = _adam_logit_opt(z_best, time.monotonic() + time_for_ls,
                                                    rng, lr=0.03, beta_frac=0.05,
                                                    beta_decay=0.95, no_imp_limit=100)
                _update_candidate(p_ls, f'ga_ls_gen{ga_generations}')
                # Also try peak flood
                if time.monotonic() < phase4_end - 20:
                    pf_time = min(15.0, phase4_end - time.monotonic() - 10)
                    if pf_time > 10:
                        p_pf, sc_pf = _peak_flood(best_seq, time.monotonic() + pf_time, eta=0.03)
                        _update_candidate(p_pf, f'ga_pf_gen{ga_generations}')

        print(f"  GA gen {ga_generations}: best={best_score:.10f}, pop_min={min(scores):.6f}")

    # --- Phase 5: Try different lengths (200s) ---
    phase5_end = t_start + 1220.0
    print(f"Phase 5: Multi-length, score={best_score:.10f}...")

    if time.monotonic() < phase5_end:
        # Try n=512 and n=2048
        for n_try in [512, 2048]:
            if time.monotonic() > phase5_end - 30:
                break
            if n_try == best_seq.size:
                continue

            # Resample to new length
            p_resampled = _resample(best_seq, n_try, rng)
            sc_r = _score(p_resampled)
            print(f"  n={n_try} resampled: {sc_r:.10f}")
            _update_candidate(p_resampled, f'resample_n{n_try}')

            # Adam polish at new length
            per_adam = min(60.0, phase5_end - time.monotonic() - 20)
            if per_adam > 20:
                z_r = _to_logit(p_resampled)
                pa, za, sa = _adam_logit_opt(z_r, time.monotonic() + per_adam,
                                             rng, lr=0.04, beta_frac=0.06, beta_decay=0.94)
                _update_candidate(pa, f'adam_n{n_try}')

            # Peak flood at new length
            if time.monotonic() < phase5_end - 15:
                per_pf = min(30.0, phase5_end - time.monotonic() - 10)
                if per_pf > 15:
                    ppf, spf = _peak_flood(_from_logit(za) if 'za' in dir() else p_resampled,
                                           time.monotonic() + per_pf, eta=0.03)
                    _update_candidate(ppf, f'flood_n{n_try}')

    # --- Phase 6: Alternative structured inits (150s) ---
    phase6_end = t_start + 1370.0
    print(f"Phase 6: Alt structured inits, score={best_score:.10f}...")

    if time.monotonic() < phase6_end:
        n_alt = best_seq.size
        alts = []
        # Various parameter combinations
        for eta in [1.5, 2.5, 4.0, -1.5, -3.0]:
            alts.append((f'lin{eta}', _make_linear(n_alt, eta)))
        for freq, amp in [(1, 1.0), (1, 2.0), (3, 0.8), (5, 0.5), (2, 1.2)]:
            alts.append((f'cos{freq}_{amp}', _make_cosine(n_alt, freq, amp)))
        for eta_q in [0.5, 1.0, 2.0, -0.5]:
            alts.append((f'quad{eta_q}', _make_quadratic(n_alt, eta_q)))
        for a1, a3 in [(1.0, -0.5), (2.0, -1.0), (0.5, 0.5), (3.0, -2.0)]:
            alts.append((f'cub{a1}_{a3}', _make_cubic(n_alt, a1, a3)))

        # Score them
        scored_alts = []
        for label, a in alts:
            sc = _score(a)
            scored_alts.append((sc, label, a))
        scored_alts.sort(key=lambda x: x[0])

        # Try top 5 with Adam
        per_alt = max(10.0, (phase6_end - time.monotonic()) / 5.0)
        for sc_a, label, a in scored_alts[:5]:
            if time.monotonic() > phase6_end - 10:
                break
            per = min(per_alt, phase6_end - time.monotonic() - 5)
            if per < 8:
                break
            z_a = _to_logit(a)
            pa, za, sa = _adam_logit_opt(z_a, time.monotonic() + per,
                                         rng, lr=0.05, beta_frac=0.07, beta_decay=0.93)
            _update_candidate(pa, f'alt_{label}')

    # --- Phase 7: Final intensive polish (remaining time) ---
    phase7_end = deadline
    print(f"Phase 7: Final polish, score={best_score:.10f}...")

    if time.monotonic() < phase7_end and best_seq is not None:
        # Intensive Adam with very small lr
        remaining = phase7_end - time.monotonic()
        if remaining > 50:
            per1 = remaining * 0.35
            z_final = _to_logit(best_seq)
            pf1, zf1, sf1 = _adam_logit_opt(z_final, time.monotonic() + per1,
                                            rng, lr=0.015, beta_frac=0.04,
                                            beta_decay=0.96, no_imp_limit=150,
                                            noise_scale=0.03)
            _update_candidate(pf1, 'final_adam1')

        # Peak flood
        if time.monotonic() < phase7_end - 30:
            per2 = min(50.0, phase7_end - time.monotonic() - 20)
            if per2 > 20:
                p_pf, sc_pf = _peak_flood(best_seq, time.monotonic() + per2, eta=0.02)
                _update_candidate(p_pf, 'final_flood')

        # LP refinement
        if time.monotonic() < phase7_end - 30 and best_seq.size <= 2048:
            per3 = min(40.0, phase7_end - time.monotonic() - 20)
            if per3 > 15:
                p_l, sc_l = _lp_refine(best_seq, time.monotonic() + per3,
                                       t_start=0.2, max_solves=8, m_shrink=0.999)
                _update_candidate(p_l, 'final_lp')

        # Final Adam
        if time.monotonic() < phase7_end - 5:
            per4 = phase7_end - time.monotonic() - 5
            if per4 > 10:
                z_f2 = _to_logit(best_seq)
                pf2, zf2, sf2 = _adam_logit_opt(z_f2, time.monotonic() + per4,
                                                rng, lr=0.008, beta_frac=0.03,
                                                beta_decay=0.97, no_imp_limit=80,
                                                noise_scale=0.02)
                _update_candidate(pf2, 'final_adam2')

        # One more peak flood
        if time.monotonic() < phase7_end - 5:
            per5 = min(15.0, phase7_end - time.monotonic() - 3)
            if per5 > 8:
                p_pf2, sc_pf2 = _peak_flood(best_seq, time.monotonic() + per5, eta=0.01)
                _update_candidate(p_pf2, 'final_flood2')

    # --- Final validation ---
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s > 0.01:
        final = final / s
    else:
        final = _make_linear(1024, 2.0)

    exact_sc = _score(final)
    print(f"FINAL: score={exact_sc:.10f} n={final.size}")

    try:
        eval_sc = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(eval_sc):
            print(f"EVAL: {eval_sc:.10f}")
            if eval_sc < best_score:
                best_score = eval_sc
            # If eval is different, the eval is what matters
            # But our score should match eval since we're computing the same thing
    except Exception as e:
        print(f"Eval check failed: {e}")

    return [float(x) for x in final]
