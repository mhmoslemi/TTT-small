Let me also think about one more idea: **dual-based weight update**.

The LP dual has variables (shadow prices) for each constraint b[k] <= M. If a constraint is tight (b[k] = M), its dual variable is positive. The dual variable λ[k] tells us how much the objective would improve if we could reduce M at that constraint.

In the primal-dual framework, the update to the sequence would be:
a[i] ← a[i] * (1 - eta * sum_k λ[k] * a[k-i])

This is exactly the multiplicative update I described earlier, but with the weights λ[k] coming from the LP dual instead of just being uniform over the top peaks.

The advantage: the LP dual gives the exact "importance" of each constraint, leading to a more informed update.

def _lp_dual_update(a, deadline, n_solves_max=100, t_start=0.3, m_shrink=0.99):
    """LP-based multiplicative update using dual variables."""
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    c = _full_convolve(a)
    M = c.max()
    t = t_start
    
    for solve_idx in range(n_solves_max):
        if time.monotonic() > deadline:
            break
        
        # Solve LP: maximize sum(b) s.t. convolve(a, b) <= M, b >= 0
        # Get the dual variables
        b_sol, duals = _lp_solve_with_duals(a, M)
        
        if b_sol is None:
            # Try slightly larger M
            M *= 1.005
            continue
        
        s_b = b_sol.sum()
        if s_b < 1e-10:
            break
        
        # Dual-based update
        # The dual variables tell us which constraints are binding
        # Compute the "pressure" on each position
        if duals is not None:
            # pressure[i] = sum_k duals[k] * a[k-i] (for valid k, i)
            # This is a cross-correlation of duals with a
            pressure = _conv_cross_corr(duals, a)
            pressure = np.maximum(pressure, 0)
            p_max = pressure.max()
            if p_max > 1e-15:
                eta = 0.1  # step size
                a = a * (1 - eta * pressure / p_max)
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
        
        # Also blend with LP solution
        b_norm = b_sol / s_b
        a = (1 - t) * a + t * b_norm
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        c = _full_convolve(a)
        M = max(c.max() * m_shrink, best_score / (2*n) * 1.001)
        t = max(0.1, t * 0.95)
    
    return best_a, best_score

The question is: what's the best way to use this gradient?

Options:
1. **Gradient descent on a directly**: a ← a - lr * g, then clip to [0, ∞) and renormalize. Problem: this can create negative values and the renormalization distorts the gradient.
2. **Gradient descent on log(a)**: z ← z - lr * (g / a), then a = exp(z) / sum(exp(z)). This is what the Adam approach does (approximately). It's smooth and preserves positivity.
3. **Multiplicative update**: a ← a * exp(-eta * g / max(g)). This is a "mirror descent" step with entropy regularization. It's always positive and can be more effective for problems with the right structure.
4. **Proximal gradient**: Combine the gradient step with a proximal operator (like projection onto the simplex). 

I think the multiplicative update (option 3) is the most natural for this problem because:

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _normalize_nonneg(a, eps=1e-15):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, eps, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _cross_corr(w, a):
    """Compute cross-correlation (w * a)[m] = sum_k w[k] * a[k-m] for valid indices."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]


def _peak_gradient(c, a, top_k=5, beta=None):
    """Compute gradient direction for peak reduction.
    Returns g[m] such that increasing a[m] in direction g[m] increases the peak."""
    n = len(a)
    if beta is None or beta < 1e-8:
        # Hard top-k
        if top_k >= len(c):
            w = np.ones(len(c)) / len(c)
        else:
            idx = np.argsort(c)[::-1][:top_k]
            w = np.zeros(len(c))
            w[idx] = 1.0 / top_k
    else:
        c_shifted = c - c.max()
        w = np.exp(c_shifted / beta)
        s = w.sum()
        if s < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = w / s
    g = _cross_corr(w, a) * 2.0
    return g


def _multiplicative_reduce(a, deadline, eta0=0.3, top_k=5, beta_start=None,
                           beta_decay=0.95, patience=200, line_search=True,
                           ls_stages=5):
    """Multiplicative peak reduction with adaptive parameters and line search."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)

    if beta_start is None:
        beta_start = c.max() * 0.1
    beta = beta_start
    eta = eta0
    no_imp = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        M = float(c.max())
        score = float(2.0 * n * M)

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > patience:
                # Restart with perturbation
                a = best_a.copy()
                rng = np.random.default_rng(it * 12345)
                noise = 0.05 * rng.normal(size=n)
                a = a * np.exp(noise)
                a = a / a.sum()
                c = _full_convolve(a)
                beta = max(1e-6, beta_start * 0.5)
                eta = eta0
                no_imp = 0
                continue

        # Compute gradient direction
        g = _peak_gradient(c, a, top_k=top_k, beta=beta)
        g_max = float(g.max())
        if g_max < 1e-15:
            break
        g_norm = g / g_max

        if line_search:
            # Try multiple step sizes
            scores = []
            candidates = []
            for stage in range(ls_stages):
                e = eta * (0.5 ** stage)
                if e < 1e-6:
                    break
                a_trial = a * (1.0 - e * g_norm)
                a_trial = np.maximum(a_trial, 1e-15)
                a_trial = a_trial / a_trial.sum()
                sc_trial = _score_norm(a_trial)
                scores.append(sc_trial)
                candidates.append(e)
            best_idx = int(np.argmin(scores))
            e_best = candidates[best_idx]
            a = a * (1.0 - e_best * g_norm)
            # Adapt eta
            if best_idx == 0:
                eta = min(1.0, eta * 1.1)
            elif best_idx == ls_stages - 1:
                eta = max(0.01, eta * 0.5)
            else:
                eta = eta * 0.8
        else:
            a = a * (1.0 - eta * g_norm)

        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        c = _full_convolve(a)

        if it % 20 == 0:
            beta = max(1e-8, beta * beta_decay)

    return best_a, best_score


def _cosine_shape_optimize(n, deadline, n_modes=8, max_iter=500):
    """Optimize sequence shape in cosine basis."""
    rng = np.random.default_rng(777)
    best_score = np.inf
    best_coeffs = None

    # Try multiple random initializations
    n_starts = 10
    per_start = max(2.0, (deadline - time.monotonic()) / n_starts)

    for start_idx in range(n_starts):
        if time.monotonic() > deadline - 5:
            break

        # Initialize coefficients
        if start_idx == 0:
            # Start with a decreasing ramp-like shape
            coeffs = np.zeros(n_modes)
            for m in range(n_modes):
                # Fourier coefficients of a decreasing exponential
                if m == 0:
                    coeffs[m] = 0.0
                else:
                    coeffs[m] = (-1)**m * 1.5 / m
        elif start_idx == 1:
            # Linear ramp
            coeffs = np.zeros(n_modes)
            coeffs[1] = 2.0
            coeffs[2] = -1.0
        elif start_idx == 2:
            # Exponential decay
            coeffs = np.zeros(n_modes)
            coeffs[1] = 3.0
            coeffs[2] = 2.0
            coeffs[3] = 1.0
        else:
            coeffs = rng.normal(0, 0.5, n_modes)

        def _loss(c_vec):
            i = np.arange(n)
            log_a = np.zeros(n)
            for m in range(n_modes):
                if m == 0:
                    log_a += c_vec[m]
                else:
                    log_a += c_vec[m] * np.cos(2 * np.pi * m * i / n)
            log_a -= log_a.max()
            a = np.exp(log_a)
            s = a.sum()
            if s < 1e-10:
                return 1e10
            a = a / s
            c = _full_convolve(a)
            return float(2.0 * n * c.max())

        res = minimize(_loss, coeffs, method='Nelder-Mead',
                       options={'maxiter': max(20, int(per_start * 50)),
                                'xatol': 1e-6, 'fatol': 1e-8})
        if res.fun < best_score:
            best_score = res.fun
            best_coeffs = res.x.copy()

    if best_coeffs is None:
        return None, np.inf

    # Reconstruct
    i = np.arange(n)
    log_a = np.zeros(n)
    for m in range(n_modes):
        if m == 0:
            log_a += best_coeffs[m]
        else:
            log_a += best_coeffs[m] * np.cos(2 * np.pi * m * i / n)
    log_a -= log_a.max()
    a = np.exp(log_a)
    a = a / a.sum()
    return a, best_score


def _cosine_shape_fine(n, a_init, deadline, n_modes=12, max_iter=2000):
    """Fine-tune shape in cosine basis around a given sequence."""
    # Project a_init onto cosine basis to get initial coefficients

        A[:, m] = np.cos(2 * np.pi * m * i / n)
    coeffs, _, _, _ = np.linalg.lstsq(A, log_a, rcond=None)

    def _loss(c_vec):
        log_a = np.zeros(n)
        for m in range(n_modes):
            if m == 0:
                log_a += c_vec[m]
            else:
                log_a += c_vec[m] * np.cos(2 * np.pi * m * i / n)

        s = a.sum()
        if s < 1e-10:
            return 1e10
        a = a / s
        c = _full_convolve(a)
        return float(2.0 * n * c.max())

    res = minimize(_loss, coeffs, method='Nelder-Mead',
                   options={'maxiter': max(50, int(max_iter)),
                            'xatol': 1e-8, 'fatol': 1e-10})

    log_a = np.zeros(n)
    for m in range(n_modes):
        if m == 0:
            log_a += res.x[m]
        else:
            log_a += res.x[m] * np.cos(2 * np.pi * m * i / n)

    a = np.exp(log_a)
    a = a / a.sum()
    return a, res.fun


def _lp_solve(a, M, get_duals=False):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        x = np.maximum(res.x, 0.0)
        if get_duals:
            try:
                duals = np.maximum(res.ineqlin.marginals, 0.0)
                return x, duals
            except Exception:
                return x, None
        return x
    return None


def _lp_refine(a, deadline, t_init=0.35, max_solves=80, m_shrink=0.995,
               blend_mode='linear'):
    """LP refinement with dual-guided multiplicative updates."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    t = t_init
    n_solves = 0

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1

        b, duals = _lp_solve(a, M, get_duals=True)
        if b is None:
            M *= 1.01
            if M > 1.0:
                break
            continue

        s_b = float(b.sum())
        if s_b < 1e-10:
            break

        b_norm = b / s_b

        if blend_mode == 'dual':
            # Use dual variables for multiplicative update
            if duals is not None and np.any(duals > 1e-10):
                # Normalize duals
                d_norm = duals / (duals.max() + 1e-30)
                # Compute pressure
                pressure = _cross_corr(d_norm, a)
                pressure = np.maximum(pressure, 0)
                p_max = pressure.max()
                if p_max > 1e-15:
                    eta = 0.15
                    a_dual = a * (1.0 - eta * pressure / p_max)
                    a_dual = np.maximum(a_dual, 1e-15)
                    a_dual = a_dual / a_dual.sum()
                    a_new = (1.0 - t) * a + t * a_dual
                else:
                    a_new = (1.0 - t) * a + t * b_norm
            else:
                a_new = (1.0 - t) * a + t * b_norm
        else:
            a_new = (1.0 - t) * a + t * b_norm

        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc = _score_norm(a_new)

        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()

        a = a_new
        c = _full_convolve(a)
        M = max(float(c.max()) * m_shrink, best_score / (2.0 * n) * 1.002)
        t = max(0.08, t * 0.94)

    return best_a, best_score


def _adam_logit(p0, deadline, rng, lr=0.04, beta_frac=0.06, beta_decay=0.94,
                no_imp_limit=250, top_k=3):
    """Adam on logit parameters with multi-peak gradient."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-4, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        g = _peak_gradient(c, p, top_k=top_k, beta=beta)
        # Convert to gradient on log-parameters
        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m_adam = b1 * m_adam + (1.0 - b1) * gz
        v_adam = b2 * v_adam + (1.0 - b2) * (gz * gz)
        mh = m_adam / (1.0 - b1 ** t_adam)
        vh = v_adam / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = np.exp(z - z.max())
        p = p / p.sum()
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.003, lr * 0.95)
        else:
            z = best_z.copy()
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 40 == 0:
            beta = max(1e-6, beta * beta_decay)

        if no_imp > no_imp_limit:

            p = np.exp(z - z.max())
            p = p / p.sum()
            c = _full_convolve(p)
            beta = max(1e-4, beta_frac * float(c.max()))
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def _make_exponential(n, rate):
    """Exponential decay: a[i] = exp(-rate * i/(n-1))."""
    t = np.arange(n) / max(1, n - 1)
    a = np.exp(-rate * t)
    return a / a.sum()


def _make_bump(n, center, width):
    """Gaussian bump."""
    t = np.arange(n) / max(1, n - 1)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    return a / a.sum()

    if best_seq is not None and time.monotonic() < phase2_end:
        p_mult, sc_mult = _multiplicative_reduce(best_seq, phase2_end, eta0=0.25,
                                                  top_k=5, beta_start=None,
                                                  beta_decay=0.96, patience=300,
                                                  line_search=True, ls_stages=4)
        _update(p_mult, 'multiplicative')

    # --- Phase 3: LP refinement (300s) ---
    phase3_end = t_start + 900.0
    print(f"Phase 3: LP refinement, score={best_score:.10f}")
    if best_seq is not None and best_seq.size <= 2048 and time.monotonic() < phase3_end:
        p_lp, sc_lp = _lp_refine(best_seq, phase3_end, t_init=0.3,
                                  max_solves=60, m_shrink=0.996, blend_mode='dual')
        _update(p_lp, 'lp_dual')

        # Quick multiplicative after LP
        if time.monotonic() < phase3_end - 50:
            p_post = _multiplicative_reduce(best_seq, min(phase3_end, time.monotonic() + 80),
                                            eta0=0.15, top_k=3, beta_decay=0.97,
                                            patience=200, line_search=True, ls_stages=3)
            _update(p_post[0], 'post_lp_mult')

    # --- Phase 4: Adam polish (200s) ---
    phase4_end = t_start + 1100.0
    print(f"Phase 4: Adam polish, score={best_score:.10f}")
    if best_seq is not None and time.monotonic() < phase4_end:
        p_adam, sc_adam = _adam_logit(best_seq, phase4_end, rng, lr=0.03,
                                       beta_frac=0.05, beta_decay=0.95,
                                       no_imp_limit=200, top_k=3)
        _update(p_adam, 'adam_polish')

    # --- Phase 5: Multi-scale exploration (300s) ---
    phase5_end = t_start + 1400.0
    print(f"Phase 5: Multi-scale, score={best_score:.10f}")
    if best_seq is not None and time.monotonic() < phase5_end:
        for n_try in [1536, 2048, 3072]:
            if time.monotonic() > phase5_end - 60:
                break
            if n_try <= best_seq.size:
                continue
            p_int = _interp_profile(best_seq, n_try)
            if p_int is not None:
                sc_int = _score_norm(p_int)
                print(f"  Interp to n={n_try}: {sc_int:.10f}")
                _update(p_int, f'interp_{n_try}')

                # Quick multiplicative at new scale
                per = min(60.0, phase5_end - time.monotonic() - 20)
                if per > 30:
                    p_new, sc_new = _multiplicative_reduce(p_int, time.monotonic() + per,
                                                            eta0=0.2, top_k=5,
                                                            beta_decay=0.95,
                                                            patience=150, line_search=False)
                    _update(p_new, f'mult_{n_try}')

    # --- Phase 6: Alternative initializations (150s) ---
    phase6_end = t_start + 1550.0
    print(f"Phase 6: Alt inits, score={best_score:.10f}")
    if time.monotonic() < phase6_end:
        n_alt = 1024
        alts = [
            ('exp2', _make_exponential(n_alt, 2.0)),
            ('exp3', _make_exponential(n_alt, 3.0)),
            ('exp4', _make_exponential(n_alt, 4.0)),
            ('exp5', _make_exponential(n_alt, 5.0)),
            ('bump06', _make_bump(n_alt, 0.6, 0.25)),
            ('bump07', _make_bump(n_alt, 0.7, 0.2)),
            ('bump04', _make_bump(n_alt, 0.4, 0.3)),
        ]
        per_alt = max(10.0, (phase6_end - time.monotonic()) / len(alts))
        for label, pa in alts:
            if time.monotonic() > phase6_end - 10:
                break
            per = min(per_alt, phase6_end - time.monotonic() - 5)
            if per < 8:
                break
            p_alt, sc_alt = _multiplicative_reduce(pa, time.monotonic() + per,
                                                    eta0=0.3, top_k=5,
                                                    beta_decay=0.94, patience=100,
                                                    line_search=False)
            _update(p_alt, f'alt_{label}')

    # --- Phase 7: Final intensive polish (remaining) ---
    phase7 = deadline
    print(f"Phase 7: Final polish, score={best_score:.10f}, time left={phase7 - time.monotonic():.0f}s")
    if best_seq is not None and time.monotonic() < phase7:
        remaining = phase7 - time.monotonic()
        if remaining > 120:
            # Fine multiplicative
            per = remaining * 0.4
            p_f1, sc_f1 = _multiplicative_reduce(best_seq, time.monotonic() + per,
                                                  eta0=0.1, top_k=3, beta_decay=0.97,
                                                  patience=150, line_search=True, ls_stages=5)
            _update(p_f1, 'final_mult1')

        if time.monotonic() < phase7 - 40 and best_seq.size <= 2048:
            per = min(50.0, phase7 - time.monotonic() - 10)
            if per > 20:
                p_f2, sc_f2 = _lp_refine(best_seq, time.monotonic() + per,
                                          t_init=0.15, max_solves=15, m_shrink=0.999,
                                          blend_mode='dual')
                _update(p_f2, 'final_lp')

        if time.monotonic() < phase7 - 10:
            per = phase7 - time.monotonic() - 10
            if per > 20:
                p_f3, sc_f3 = _adam_logit(best_seq, time.monotonic() + per, rng,
                                           lr=0.01, beta_frac=0.03, beta_decay=0.97,
                                           no_imp_limit=50, top_k=2)
                _update(p_f3, 'final_adam')

        # Ultimate fine multiplicative
        if time.monotonic() < phase7 - 5:
            per = phase7 - time.monotonic() - 5
            if per > 10:
                p_f4, sc_f4 = _multiplicative_reduce(best_seq, time.monotonic() + per,
                                                      eta0=0.05, top_k=2, beta_decay=0.98,
                                                      patience=100, line_search=True, ls_stages=5)
                _update(p_f4, 'final_mult2')

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_exponential(1024, 2.0)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL confirms: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    # Fallback
    fb = _make_exponential(1024, 2.0)
    print(f"FALLBACK n=1024, score={_score_norm(fb):.10f}")
    return [float(x) for x in fb]
