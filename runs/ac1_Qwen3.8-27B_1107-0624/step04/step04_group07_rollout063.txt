def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _cross_correlate_grad(w, a):
    """Compute sum_k w[k] * a[k-j] for all j using FFT.
    This gives (1/2) * gradient of sum_k w[k]*c[k] w.r.t. a[j].
    Since dc[k]/da[j] = 2*a[k-j], the gradient of sum w[k]*c[k] is sum_k w[k]*2*a[k-j].
    So this function returns the full gradient (with factor 2).
    """
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    # We want g[j] = sum_k w[k] * a[k-j] (with a[.] = 0 outside [0,n-1])
    # This is the cross-correlation of w and a: g[j] = sum_k w[k]*a[k-j]
    # = (w * a_reversed)[j + n - 1] in linear convolution terms
    # Using FFT: ifft(fft(w) * conj(fft(a))) gives circular cross-correlation
    # Better: g = ifft(fft(w) * fft(a_reversed)) shifted appropriately
    # Actually: g[j] = sum_k w[k] a[k-j] = sum_i w[i+j] a[i] = (rev(w) * a)[n-1-j] in circ
        
    # Simplest: use the fact that g[j] = sum_k w[k]*a[k-j]
    # = ifft(conj(fft(w)) * fft(a))[j] in circular sense (with padding)
    aa = np.fft.irfft(np.conj(fw) * fa, N)
    # For linear case with N >= 2n-1, the first n values are correct
    return aa[:n]


def _score_norm(a):

def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w

def _interp_seq(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new

        g_raw = _cross_correlate_grad(w, a)
        
        # In log-space: dz = a * g (chain rule: da = a*dz, so d(obj)/dz = a*d(obj)/da)
        dot = float(np.dot(a, g_raw))
        gz = a * (g_raw - dot)
        gz -= gz.mean()  # Remove mean to preserve normalization

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1**t_adam)
        vh = v / (1 - b2**t_adam)
        
        # Nesterov: look-ahead
        v_nest = momentum * v_nest + (1 - momentum) * mh
        z = z - lr * (v_nest + momentum * mh) / (np.sqrt(vh) + eps)
        z = np.clip(z, -40, 40)
        z -= z.mean()

        a = np.exp(z - z.max())
        a = a / a.sum()
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        c = _full_convolve(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-10:
                best_score = score
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.002, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            v_nest.fill(0)
            t_adam = 0
            continue

        # Adaptive beta
        if t_adam % 30 == 0:
            beta = max(1e-4, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.3:
                    beta = max(1e-4, beta * 0.6)

        # Restart if stuck
        if no_imp > no_imp_limit:
            ns = 0.05 + 0.05 * min(5, no_imp // no_imp_limit)
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -40, 40)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a)
            beta = 0.08 * c.max()
            m.fill(0)
            v.fill(0)
            v_nest.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score


def _build_conv_matrix_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_fixed_point(a, deadline, max_iters=30):
    """LP-based fixed-point: minimize max(conv(a,b)) by optimizing b."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a)
        M = float(c.max())

        # LP: maximize sum(b) s.t. conv(a,b) <= M, b >= 0
        A = _build_conv_matrix_sparse(n, a)
        c_obj = -np.ones(n)
        b_ub = M * np.ones(2 * n - 1)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={"time_limit": max(5, deadline - time.monotonic() - 2)})
        except Exception:
            break
        
        if not res.success or res.x is None:
            break

        b = np.maximum(res.x, 0.0)
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b = b / s_b

        # Line search
        best_t = 0.0
        best_sc = best_score
        current_a = a.copy()
        for t_cand in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_t = (1.0 - t_cand) * current_a + t_cand * b
            a_t = np.maximum(a_t, 1e-15)
            a_t = a_t / a_t.sum()
            sc_t = _score_norm(a_t)
            if sc_t < best_sc:
                best_sc = sc_t
                best_t = t_cand
                best_a = a_t.copy()

        if best_t == 0.0:
            break

        a = best_a.copy()
        sc = best_sc
        if sc < best_score:
            best_score = sc

    return best_a, best_score


def _coord_descent_bisection(a, deadline, max_sweeps=10):
    """Coordinate descent with bisection on each coordinate."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            
            # Try a range of multiplicative factors
            best_factor = 1.0
            best_local = best_score
            for factor in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 3.0]:
                a[j] = max(old_val * factor, 1e-15)
                a_norm = a / a.sum()
                sc = _score_norm(a_norm)
                if sc < best_local:
                    best_local = sc
                    best_factor = factor
            if best_factor != 1.0:
                a[j] = max(old_val * best_factor, 1e-15)
                a = a / a.sum()
                if best_local < best_score:
                    best_score = best_local
                    best_a = a.copy()
                    improved = True
            else:
                a[j] = old_val
        if not improved:
            break

    return best_a, best_score


def _gredient_descent_projected(a, deadline, lr_init=0.01, max_iters=5000):
    """Direct gradient descent with projection onto simplex."""
    n = len(a)
    a = np.maximum(a, 1e-12).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = lr_init
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a)
        M = float(c.max())
        # Soft weights
        beta = 0.02 * M
        _, w = _smooth_max_weights(c, beta)
        
        # Gradient of 2n*sum(w*c) w.r.t. a (with S=1 constraint)
        g = _cross_correlate_grad(w, a) * 2.0  # factor 2 from dc/da
        dot = float(np.dot(a, g))
        g = a * (g - dot)  # project to tangent space of simplex
        g -= g.mean()

        # Backtracking line search
        step = lr
        found = False
        for _ in range(10):
            a_try = a - step * g
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s > 1e-10:
                a_try = a_try / s
                sc = _score_norm(a_try)
                if sc < best_score * 1.001:  # allow slight increase
                    a = a_try
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        no_imp = 0
                    else:
                        no_imp += 1
                    found = True
                    break
            step *= 0.5

        if not found:
            no_imp += 1
            lr = max(1e-6, lr * 0.9)

        if no_imp > 200:
            lr = lr_init
            no_imp = 0
            # Add small perturbation
            a = a + 0.01 * np.random.default_rng(it).normal(size=n)
            a = np.maximum(a, 1e-12)
            a = a / a.sum()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
    Combines smooth-max Adam with Nesterov momentum, novel profile families,
    LP fixed-point polish, projected gradient descent, coordinate descent,
    and multi-scale escalation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    # === Phase 1: Generate diverse initial candidates ===
    print("Phase 1: Generating candidates...")
    candidates = []

    n_sizes = [128, 256, 384, 512, 768, 1024]
    
    profile_defs = [
        ('lin1', 'linear', 1.0),
        ('lin2', 'linear', 2.0),
        ('lin3', 'linear', 3.0),
        ('lin5', 'linear', 5.0),
        ('quad1', 'quadratic', 1.0),
        ('quad2', 'quadratic', 2.0),
        ('quad3_asym', 'quadratic', 2.0),
        ('cubic1', 'cubic', 1.5),
        ('cubic2', 'cubic', 2.5),
        ('exp1', 'exp_dec', 1.0),
        ('exp2', 'exp_dec', 2.0),
        ('exp4', 'exp_dec', 4.0),
        ('exp_inc1', 'exp_inc', 1.0),
        ('exp_inc2', 'exp_inc', 2.0),
        ('cos1', 'cosine', 0.8),
        ('cos2', 'cosine', 0.6),
        ('sin1', 'sine', 0.8),
        ('sin2', 'sine', 0.6),
        ('para1', 'parabola', 1.0),
        ('para3', 'parabola', 3.0),
        ('U1', 'U_shape', 1.0),
        ('U3', 'U_shape', 3.0),
        ('edge1', 'edge_heavy', 1.0),
        ('edge2', 'edge_heavy', 2.0),
        ('edge3', 'edge_heavy', 3.0),
        ('gauss03', 'gauss', 0.3),
        ('gauss05', 'gauss', 0.5),
        ('tg1', 'two_gauss', 0.3),
        ('tg2', 'two_gauss', 0.25),
        ('kron1', 'kronecker', 0.61803),
        ('kron2', 'kronecker', 0.70711),
        ('kron3', 'kronecker', 0.5),
        ('chirp1', 'chirp', 0.8),
        ('chirp2', 'chirp', 0.5),
        ('w1', 'w_shape', 0.8),
        ('w2', 'w_shape', 1.5),
        ('m1', 'm_shape', 0.8),
        ('m2', 'm_shape', 1.5),
        ('ramp1', 'ramp_cos', 0.5),
        ('ramp2', 'ramp_cos', 0.8),
        ('ag1', 'asym_gauss', 0.3),
        ('ag2', 'asym_gauss', 0.5),
        ('pl05', 'power_law', 0.5),
        ('pl1', 'power_law', 1.0),
        ('step1', 'step_asym', 0.3),
        ('step2', 'step_asym', 0.5),
    ]

    for n in n_sizes:
        for (label, ptype, p1) in profile_defs:
            p = _make_profile(n, ptype, p1)
            # Add some variation for some types
            if ptype == 'quadratic':
                p2_val = 0.5 if 'asym' not in label else -0.5
                p = _make_profile(n, ptype, p1, p2_val)
            elif ptype == 'cubic':
                p = _make_profile(n, ptype, p1, -p1*0.7)
            elif ptype == 'two_gauss':
                p = _make_profile(n, ptype, p1, 1.0-p1, 0.15)
            elif ptype == 'kronecker':
                p = _make_profile(n, ptype, p1, 1.5)
            elif ptype == 'chirp':
                p = _make_profile(n, ptype, p1, 3.0)
            elif ptype == 'ramp_cos':
                p = _make_profile(n, ptype, p1, 2.0)
            elif ptype == 'asym_gauss':
                p = _make_profile(n, ptype, p1, 0.15)
            elif ptype == 'step_asym':
                p = _make_profile(n, ptype, p1, 0.5)
            
            sc = _score_norm(p)
            candidates.append((sc, p, n, label))

    # Parent-based candidates
    if parent is not None:
        for n in [256, 384, 512, 768, 1024, 1536, 2048]:
            pi = _interp_seq(parent, n)
            sc = _score_norm(pi)
            candidates.append((sc, pi, n, f'parent_n{n}'))
        if 0 < pn <= 4096:
            sc = _score_norm(parent)
            candidates.append((sc, parent.copy(), pn, 'parent_exact'))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, best init: {candidates[0][0]:.8f} n={candidates[0][2]} [{candidates[0][3]}]")

    if candidates[0][0] < best_score:
        best_seq = np.asarray(candidates[0][1], dtype=np.float64).copy()
        best_score = float(candidates[0][0])
        print(f"  BEST: {best_score:.10f} n={best_seq.size} [init]")

    # === Phase 2: Adam+Nesterov on top candidates ===
    print("Phase 2: Adam+Nesterov optimization...")
    phase2_end = t_start + 600.0
    polished = []

    # Process candidates in order of initial score, but prefer smaller n for speed
    for (sc, p, n, label) in candidates[:30]:
        if time.monotonic() > phase2_end - 30:
            break
        # Time allocation based on n
        if n <= 256:
            time_alloc = 60.0
        elif n <= 512:
            time_alloc = 45.0
        elif n <= 768:
            time_alloc = 35.0
        else:
            time_alloc = 25.0
        
        per_end = min(phase2_end - 10, time.monotonic() + time_alloc)
        
        a_ad, sc_ad = _adam_nesterov(p, per_end, rng, lr=0.03,
                                      beta_decay=0.97, no_imp_limit=200)
        if sc_ad < best_score:
            best_seq = np.asarray(a_ad, dtype=np.float64).copy()
            best_score = float(sc_ad)
            print(f"  BEST: {best_score:.10f} n={a_ad.size} [Adam_{label}_n{n}]")
        polished.append((sc_ad, a_ad, n, label + '_adam'))

        # Also try projected gradient descent for variety
        if n <= 512 and time.monotonic() < phase2_end - 20:
            per_pg = min(20.0, phase2_end - time.monotonic() - 10)
            if per_pg > 10:
                a_pg, sc_pg = _gredient_descent_projected(a_ad, time.monotonic() + per_pg,
                                                          lr_init=0.005)
                if sc_pg < best_score:
                    best_seq = np.asarray(a_pg, dtype=np.float64).copy()
                    best_score = float(sc_pg)
                    print(f"  BEST: {best_score:.10f} n={a_pg.size} [ProjGrad_{label}]")
                polished.append((sc_pg, a_pg, n, label + '_pg'))

    polished.sort(key=lambda x: x[0])
    print(f"  After Phase 2: best={best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    # === Phase 3: LP polish on small-n best ===
    print("Phase 3: LP polish...")
    phase3_end = t_start + 900.0

    for (sc, p, n, label) in polished[:6]:
        if time.monotonic() > phase3_end - 20:
            break
        if n > 768:
            continue
        per = min(40.0, phase3_end - time.monotonic() - 10)
        if per < 15:
            break
        a_lp, sc_lp = _lp_fixed_point(p, time.monotonic() + per, max_iters=20)
        if sc_lp < best_score:
            best_seq = np.asarray(a_lp, dtype=np.float64).copy()
            best_score = float(sc_lp)
            print(f"  BEST: {best_score:.10f} n={a_lp.size} [LP_{label}]")

    # === Phase 4: Multi-scale escalation ===
    print("Phase 4: Multi-scale escalation...")
    phase4_end = t_start + 1300.0

    if best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 384:
            target_ns = [512, 768, 1024]
        elif n_cur <= 512:
            target_ns = [768, 1024, 1536]
        elif n_cur <= 768:
            target_ns = [1024, 1536]
        elif n_cur <= 1024:
            target_ns = [1536, 2048]
        else:
            target_ns = []

        for tn in target_ns:
            if time.monotonic() > phase4_end - 40:
                break
            p_int = _interp_seq(best_seq, tn)
            sc_int = _score_norm(p_int)
            print(f"  Interp to n={tn}: {sc_int:.8f}")
            if sc_int < best_score:
                best_seq = p_int.copy()
                best_score = sc_int
                print(f"  BEST: {best_score:.10f} n={tn} [interp]")

            # Adam on interpolated
            if time.monotonic() < phase4_end - 30:
                per = min(60.0, phase4_end - time.monotonic() - 20)
                if per > 20:
                    a_e, sc_e = _adam_nesterov(p_int, time.monotonic() + per, rng,
                                                lr=0.02, beta_decay=0.97, no_imp_limit=250)
                    if sc_e < best_score:
                        best_seq = a_e.copy()
                        best_score = sc_e
                        print(f"  BEST: {best_score:.10f} n={tn} [Adam_esc]")

            # Coordinate descent on escalated
            if tn <= 1024 and time.monotonic() < phase4_end - 15:
                per_cd = min(20.0, phase4_end - time.monotonic() - 10)
                if per_cd > 10:
                    a_cd, sc_cd = _coord_descent_bisection(
                        best_seq if best_seq.size == tn else _interp_seq(best_seq, tn),
                        time.monotonic() + per_cd, max_sweeps=5)
                    if sc_cd < best_score:
                        best_seq = a_cd.copy()
                        best_score = sc_cd
                        print(f"  BEST: {best_score:.10f} n={tn} [CD_esc]")

    # === Phase 5: More Adam iterations on best ===
    print("Phase 5: Extended Adam on best...")
    phase5_end = t_start + 1550.0

    if best_seq is not None and time.monotonic() < phase5_end:
        n_best = best_seq.size
        # Multiple Adam runs with different hyperparameters
        lr_options = [0.02, 0.03, 0.05]
        for i, lr_i in enumerate(lr_options):
            if time.monotonic() > phase5_end - 20:
                break
            per = min(50.0, phase5_end - time.monotonic() - 10)
            if per < 15:
                break
            # Add small noise to break symmetry
            a_noisy = best_seq + 0.02 * rng.normal(size=n_best)
            a_noisy = np.maximum(a_noisy, 1e-12)
            a_noisy = a_noisy / a_noisy.sum()
            
            a_ax, sc_ax = _adam_nesterov(a_noisy, time.monotonic() + per, rng,
                                          lr=lr_i, beta_decay=0.96, no_imp_limit=150)
            if sc_ax < best_score:
                best_seq = a_ax.copy()
                best_score = sc_ax
                print(f"  BEST: {best_score:.10f} n={n_best} [Adam5_{i}]")

        # LP if small enough
        if n_best <= 512 and time.monotonic() < phase5_end - 15:
            per = min(30.0, phase5_end - time.monotonic() - 10)
            if per > 10:
                a_lp5, sc_lp5 = _lp_fixed_point(best_seq, time.monotonic() + per, max_iters=15)
                if sc_lp5 < best_score:
                    best_seq = a_lp5.copy()
                    best_score = sc_lp5
                    print(f"  BEST: {best_score:.10f} n={n_best} [LP5]")

    # === Phase 6: Random restarts ===
    print("Phase 6: Random restarts...")
    phase6_end = deadline - 60.0

    if best_seq is not None and time.monotonic() < phase6_end:
        n_best = best_seq.size
        n_restarts = 4
        per_r = max(15.0, (phase6_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase6_end:
                break
            per = min(per_r, phase6_end - time.monotonic() - 5)
            if per < 10:
                break

            ns = 0.03 + 0.04 * i
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -40, 40)
            z -= z.mean()
            a_s = np.exp(z - z.max())
            a_s = a_s / a_s.sum()

            a_r, sc_r = _adam_nesterov(a_s, time.monotonic() + per, rng,
                                        lr=0.03, beta_decay=0.96, no_imp_limit=120)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  BEST: {best_score:.10f} n={n_best} [restart{i}]")

    # === Phase 7: Coordinate descent final polish ===
    print("Phase 7: Coordinate descent polish...")
    if time.monotonic() < deadline - 20 and best_seq is not None:
        per_cd = deadline - time.monotonic() - 20
        if per_cd > 10:
            a_cd, sc_cd = _coord_descent_bisection(best_seq, time.monotonic() + per_cd,
                                                    max_sweeps=8)
            if sc_cd < best_score:
                best_seq = a_cd.copy()
                best_score = sc_cd
                print(f"  BEST: {best_score:.10f} n={best_seq.size} [CD_final]")

    # === Final validation ===
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile(512, 'linear', 1.5)

        exact_sc = _score_norm(final)
        print(f"\nFINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile(512, 'linear', 2.0)
    print(f"FALLBACK n=512, score={_score_norm(fb):.8f}")
    return [float(x) for x in fb]
