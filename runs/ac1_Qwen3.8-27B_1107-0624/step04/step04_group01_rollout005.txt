def _compute_grad_z(a, c, w, n):
    """Gradient of (2n/β)*log(sum(exp(β*c))) w.r.t. z, where a=softmax(z).

    # conv(w, a_reversed) gives cross-correlation
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)

            break

                w = e / s

            if not np.all(np.isfinite(grad)):

            m_buf = b1 * m_buf + (1.0 - b1) * grad

                        continue
                    a_new = a_new / s
                    sc = _score_norm(a_new)
                    if sc < best_score:
                        best_score = sc
                        a = a_new.copy()
                        improved = True
                        break
        
        if not improved:
            break
    
    return a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(parent)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        update_best(parent, sc0, "parent")
    else:
        n_main = 2048
        a0 = _make_profile(n_main, "uniform")
        print(f"No parent, using uniform n={n_main}")

    # === Phase 0: Quick exploration (0-30s) ===
    phase0_end = min(deadline, t_start + 30.0)
    
    # Try a few quick gradient steps on the parent
    c0 = _full_convolve(a0, a0)
    max_c0 = float(c0.max())
    beta_quick = [0.01 * max_c0, 0.005 * max_c0]
    p_q, sc_q = _adam_optimize(a0, phase0_end, beta_quick, max_iter=5000, lr=0.01)
    update_best(p_q, _score(p_q), "quick_adam")

    # Try asymmetric initializations
    for kind in ["linear_ramp", "geometric", "two_peaks"]:
        if time.monotonic() > phase0_end:
            break
        p = _make_profile(n_main, kind)
        sc = _score(p)
        update_best(p, sc, f"init_{kind}")

    print(f"Phase 0: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 1: LP-based optimization (30s - 500s) ===
    phase1_end = min(deadline, t_start + 500.0)
    
    if HAS_SCIPY and best_seq is not None and time.monotonic() < phase1_end:
        print(f"Starting LP optimization...")
        f_current = best_seq.copy()
        f_current = f_current / f_current.sum()
        n = len(f_current)
        
        best_lp_f = f_current.copy()
        best_lp_score = _score_norm(f_current)
        
        # Adaptive step size
        t_step = 0.2
        
        for it in range(500):
            if time.monotonic() > phase1_end:
                break
            
            c = _full_convolve(f_current, f_current)
            M = float(c.max())
            
            # Build LP constraint matrix
            num_rows = 2 * n - 1
            row_idx = []
            col_idx = []
            data = []
            for j in range(n):
                i_start = j
                i_end = min(j + n - 1, 2 * n - 2)
                if i_end >= i_start:
                    rows_j = np.arange(i_start, i_end + 1)
                    row_idx.extend(rows_j.tolist())
                    col_idx.extend([j] * len(rows_j))
                    data.extend([f_current[j]] * len(rows_j))
            
            A = csr_array((data, (row_idx, col_idx)), shape=(num_rows, n))
            b_vec = np.full(num_rows, M)
            
            t0 = time.monotonic()
            res = linprog(-np.ones(n), A_ub=A, b_ub=b_vec, bounds=(0, None), method='highs')
            t_lp = time.monotonic() - t0
            
            if it == 0 or it % 10 == 0:
                print(f"  LP iter {it}: t={t_lp:.2f}s, M={M:.8f}, score={_score_norm(f_current):.10f}")
            
            if not res.success:
                print(f"  LP failed at iter {it}: {res.message}")
                break
            
            b = res.x
            s_b = float(b.sum())
            if s_b < 1e-15:
                break
            
            b_rescaled = b / s_b
            
            # Update with adaptive step
            f_new = (1 - t_step) * f_current + t_step * b_rescaled
            f_new = np.maximum(f_new, 0)
            s_new = f_new.sum()
            if s_new < 1e-15:
                break
            f_new = f_new / s_new
            
            sc = _score_norm(f_new)
            
            if sc < best_lp_score:
                best_lp_score = sc
                best_lp_f = f_new.copy()
                # Allow larger step
                t_step = min(0.4, t_step * 1.05)
            else:
                # Reduce step
                t_step = max(0.02, t_step * 0.7)
            
            f_current = f_new
        
        update_best(best_lp_f, _score(best_lp_f), "lp_opt")
        print(f"LP optimization done: best_score={best_lp_score:.10f}")

    print(f"Phase 1: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Adam fine-tuning (500s - 1100s) ===
    phase2_end = min(deadline, t_start + 1100.0)
    
    if best_seq is not None and time.monotonic() < phase2_end:
        c_best = _full_convolve(best_seq, best_seq)
        max_cb = float(c_best.max())
        
        beta_schedule = [
            0.005 * max_cb, 0.002 * max_cb, 0.001 * max_cb,
            0.0005 * max_cb, 0.0002 * max_cb, 0.0001 * max_cb
        ]
        per_time = phase2_end - time.monotonic() - 10
        p_best, sc_best = _adam_optimize(
            best_seq, time.monotonic() + per_time,
            beta_schedule, max_iter=300000, lr=0.01
        )
        update_best(p_best, _score(p_best), "adam_fine")

    print(f"Phase 2: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Multi-restart (1100s - 1500s) ===
    phase3_end = min(deadline, t_start + 1500.0)
    
    if best_seq is not None and time.monotonic() < phase3_end:
        n_restarts = 4
        per_restart = (phase3_end - time.monotonic() - 20) / n_restarts
        
        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 15:
                break
            
            sigma = 0.02 + 0.01 * i
            p_start = _perturb(best_seq, sigma, rng)
            
            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = [0.003 * max_cs, 0.001 * max_cs, 0.0005 * max_cs,
                  0.0002 * max_cs, 0.0001 * max_cs]
            
            p_best, sc_best = _adam_optimize(
                p_start, time.monotonic() + per_restart,
                bs, max_iter=100000, lr=0.008
            )
            update_best(p_best, _score(p_best), f"restart{i}")

    print(f"Phase 3: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Coordinate descent (1500s - 1650s) ===
    phase4_end = min(deadline, t_start + 1650.0)
    
    if best_seq is not None and time.monotonic() < phase4_end:
        a_cd, sc_cd = _coord_descent(best_seq, phase4_end, max_rounds=50)
        update_best(a_cd, _score(a_cd), "coord_descent")

    print(f"Phase 4: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Try larger n (1650s - 1760s) ===
    phase5_end = min(deadline, t_start + 1760.0)
    
    if best_seq is not None and len(best_seq) <= 2048 and time.monotonic() < phase5_end - 30:
        for target_n in [3072, 4096]:
            if time.monotonic() > phase5_end - 30:
                break
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{target_n}")
            
            remaining = phase5_end - time.monotonic() - 20
            if remaining > 30:
                c_int = _full_convolve(p_interp, p_interp)
                max_ci = float(c_int.max())
                bs_int = [0.003 * max_ci, 0.001 * max_ci, 0.0005 * max_ci]
                p_best, sc_best = _adam_optimize(
                    p_interp, time.monotonic() + min(remaining, 60),
                    bs_int, max_iter=50000, lr=0.005
                )
                update_best(p_best, _score(p_best), f"opt_n{target_n}")

    print(f"Phase 5: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final polish (1760s - deadline) ===
    if best_seq is not None and time.monotonic() < deadline - 5:
        c_final = _full_convolve(best_seq, best_seq)
        max_cf = float(c_final.max())
        beta_final = [0.0001 * max_cf, 0.00005 * max_cf, 0.00002 * max_cf]
        per_time = deadline - time.monotonic() - 3
        if per_time > 5:
            p_best, sc_best = _adam_optimize(
                best_seq, time.monotonic() + per_time,
                beta_final, max_iter=100000, lr=0.002
            )
            update_best(p_best, _score(p_best), "final_polish")

    print(f"Phase 6: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum
