LP-guided peak shaving optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal descent directions, combined with

from scipy.optimize import linprog, minimize
from scipy.signal import fftconvolve
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    """Score assuming sum(a) = 1: returns 2*n*max(a*a)."""
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())

def _score_fn(a):
    """Full score: 2*n*max(a*a)/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max()) / (s * s)

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    m = np.max(z)
    e = np.exp(z - m)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

    """Score with sum(a)=1: 2*n*max(a*a)."""

"""
LP-guided peak shaving with active-set gradient optimization for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key innovation: uses linear programming to find directions that simultaneously
reduce all active convolution peaks, combined with momentum-based gradient polish.
"""
import numpy as np
from scipy.optimize import linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max()) / (s * s)


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    m = np.max(z)
    e = np.exp(z - m)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _compute_directions(c, a, k_active):
    """Compute the average peak-shaving direction for top-k_active peaks."""
    n = len(a)
    total_conv = len(c)
    idx = np.argsort(c)[-k_active:]
    d = np.zeros(n)
    for k_val in idx:
        k_val = int(k_val)
        lo = max(0, k_val - n + 1)
        hi = min(n - 1, k_val)
        if lo <= hi:
            d[lo:hi + 1] += a[k_val - hi:k_val - lo + 1][::-1]
    return d, idx


def _lp_direction(a, top_k=100, time_limit=15.0):
    """Find LP-optimal direction minimizing max(a*g)[k] over active peaks."""
    n = len(a)
    c = _full_convolve(a, a)
    M = float(c.max())
    if M < 1e-30:
        return None, None

    idx = np.argsort(c)[-top_k:]

    # Build constraint matrix efficiently
    A = np.zeros((len(idx), n))
    for row in range(len(idx)):
        k_val = int(idx[row])
        lo = max(0, k_val - n + 1)
        hi = min(n - 1, k_val)
        if lo <= hi:
            A[row, lo:hi + 1] = a[k_val - hi:k_val - lo + 1][::-1]

    # LP: min z s.t. A@g <= M, sum(g)=1, g>=0
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((len(idx), 1))])
    b_ub = np.full(len(idx), M)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'time_limit': time_limit, 'dual_feasibility_tolerance': 1e-8})
        if result.success:
            g = np.maximum(result.x[:n], 0)
            gs = g.sum()
            if gs > 1e-15:
                g /= gs
            return g, float(result.fun)
    except Exception:
        pass
    return None, None
