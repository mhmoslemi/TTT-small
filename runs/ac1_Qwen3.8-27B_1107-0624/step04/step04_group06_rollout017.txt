2. Solve the linear system: for k in S, sum_{i in S} A[k,i] * b[i] = M

3. Check if all b[i] >= 0 for i in S
4. If not, remove the most negative b from S and go to step 2
5. Also check dual feasibility: for k not in S, (A*b)[k] <= M
   If violated, add the most violated k to S and go to step 2

```python
def active_set_lp(a, M, tol=1e-10, max_iter=10000):
    n = len(a)
    N = 2 * n - 1  # length of convolution
    
    # Initialize: all indices active
    S = set(range(N))
    b = np.zeros(N)
    
    for iteration in range(max_iter):
        # Solve the linear system on S
        S_list = sorted(S)
        m = len(S_list)
        if m == 0:
            return None
        
        # Build the sub-matrix A[S, S]
        # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
        # We need A[k, i] for k in S_list, i in S_list
        # Note: A is Toeplitz: A[k, i] = a[k-i] depends only on k-i
        
        # Build the matrix
        As = np.zeros((m, m))
        for idx_k, k in enumerate(S_list):
            for idx_i, i in enumerate(S_list):
                d = k - i
                if 0 <= d < n:
                    As[idx_k, idx_i] = a[d]
        
        # Solve As @ b_S = M * ones
        try:
            b_S = np.linalg.solve(As, M * np.ones(m))
        except np.linalg.LinAlgError:
            # Fallback: use lstsq
            b_S, _, _, _ = np.linalg.lstsq(As, M * np.ones(m), rcond=None)
        
        # Check if all b_S >= 0
        min_idx = np.argmin(b_S)
        if b_S[min_idx] < -tol:
            # Remove the most negative
            S.remove(S_list[min_idx])
            b = np.zeros(N)
            for idx, i in enumerate(S_list):
                b[i] = b_S[idx]
            continue
        
        # All b_S >= 0 (within tolerance), set b
        b = np.zeros(N)
        for idx, i in enumerate(S_list):
            b[i] = max(0, b_S[idx])
        
        # Check dual feasibility: for k not in S, (A*b)[k] <= M
        # Compute A*b for all k
        Ab = np.convolve(a, b, mode='full')
        
        # Find violations
        inactive = [k for k in range(N) if k not in S]
        if inactive:
            max_violation = max(Ab[k] - M for k in inactive)
            if max_violation > tol:
                # Add the most violated
                worst_k = max(inactive, key=lambda k: Ab[k] - M)
                S.add(worst_k)
                continue
        
        # Both primal and dual feasibility satisfied
        return b
    
    return None  # Didn't converge
```

def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

    lr_cur = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_and_grad(c, beta)
        g = _conv_cross_corr(w, p)

        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr_cur * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr_cur = max(0.002, lr_cur * 0.93)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 30 == 0:
            beta = max(1e-6, beta * beta_decay)

        if no_imp > no_imp_limit:
            ns = 0.08 if best_score < 1.52 else 0.15
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr_cur = base_lr
            no_imp = 0

    return best_p, best_score


def _active_set_lp(a, M, max_active=400, tol=1e-10):
    """Solve: max sum(b) s.t. a*b <= M, b >= 0 using active set method."""
    n = len(a)
    N = 2 * n - 1

    # Start with all indices active, but limit to top contributors
    # Initial guess: uniform b
    b = np.ones(N) / N * M  # small initial
    b = np.maximum(b, 0)

    # Iteratively determine active set
    active = set(range(N))
    if N > max_active * 3:
        # Pre-select: keep indices where the "natural" b would be large
        # Use the convolution of a with itself as a guide
        aa = _full_convolve(a)
        # Keep indices with highest aa values
        top_k = min(N, max_active * 2)
        top_indices = np.argsort(aa)[::-1][:top_k]
        active = set(top_indices.tolist())

    for iteration in range(200):
        S_list = sorted(active)
        m_active = len(S_list)

        if m_active == 0:
            return np.zeros(N)

        # Build sub-matrix A[S, S] where A[k,i] = a[k-i] if valid
        As = np.zeros((m_active, m_active))
        for idx_k, k in enumerate(S_list):
            for idx_i, i in enumerate(S_list):
                d = k - i
                if 0 <= d < n:
                    As[idx_k, idx_i] = a[d]

        # Solve As @ b_S = M * ones
        rhs = M * np.ones(m_active)
        try:
            b_S = np.linalg.solve(As, rhs)
        except np.linalg.LinAlgError:
            try:
                b_S, _, _, _ = np.linalg.lstsq(As, rhs, rcond=None)
            except:
                return None

        # Check non-negativity
        min_val = float(b_S.min())
        if min_val < -tol:
            # Remove most negative
            worst_local = int(np.argmin(b_S))
            active.discard(S_list[worst_local])
            # Rebuild b with current solution (keeping non-negative parts)
            b = np.zeros(N)
            for idx, i in enumerate(S_list):
                b[i] = max(0.0, b_S[idx])
            continue

        # Set b values
        b = np.zeros(N)
        for idx, i in enumerate(S_list):
            b[i] = max(0.0, b_S[idx])

        # Check dual feasibility: for k not in active, (a*b)[k] <= M
        Ab = np.convolve(a, b, mode='full')
        inactive = [k for k in range(N) if k not in active]
        if inactive:
            violations = [Ab[k] - M for k in inactive]
            max_viol = max(violations)
            if max_viol > M * 1e-6:
                # Add most violated
                worst_k = inactive[int(np.argmax(violations))]
                active.add(worst_k)
                continue

        # Converged
        return b

    return b  # Return best effort


def _lp_refine_aggressive(a, deadline, m_shrink=0.99, t_start=0.5, max_solves=100):
    """Aggressive LP refinement with decreasing M and active-set method."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    t = t_start
    n_solves = 0
    no_imp = 0

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1

        b = _active_set_lp(a, M)
        if b is None:
            # Fallback to scipy LP for smaller problems
            if n <= 256:
                b = _scipy_lp(a, M)
            if b is None:
                break

        s_b = float(b.sum())
        if s_b < 1e-10:
            break

        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)

        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        a = a_new
        c = _full_convolve(a)
        M_new = float(c.max())

        # Shrink M target
        M = max(M_new * m_shrink, best_score / (2.0 * n) * 1.001)
        t = max(0.1, t * 0.95)

        if no_imp > 30:
            break

    return best_a, best_score


def _scipy_lp(a, M):
    """Fallback LP using scipy for smaller problems."""
    n = len(a)
    N = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(N):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(N, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(N)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None
