    constraints = [sum(g) >= 0]  # trivial

    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _saml_subgrad(a, active_lags):
    """Compute SAML subgradient for active lags."""
    n = len(a)
    subgrad = np.zeros(n)
    for k in active_lags:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            subgrad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    subgrad /= len(active_lags)
    return subgrad


def _saml_optimize(a_init, deadline, n_active=5, eta0=0.3, decay=0.999, min_eta=1e-5):
    """SAML optimization: minimize max(conv(a,a)) s.t. sum(a)=1, a>=0."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)
    eta = eta0
    stall_count = 0
    stall_threshold = 200
    stall_patience = 5

    while time.monotonic() < deadline - 2.0:
        c = _full_convolve(a, a)
        max_c = float(c.max())
        current_score = 2.0 * n * max_c

        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > stall_threshold * stall_patience:
                break
            if stall_count > stall_threshold:
                eta = max(min_eta, eta * 0.95)
                stall_patience += 1
                stall_threshold += 200

        # Find active lags
        top_idx = np.argsort(c)[-n_active:]
        subgrad = _saml_subgrad(a, top_idx)

        # Multiplicative update
        a = a * np.exp(-eta * subgrad)
        a = np.maximum(a, 1e-30)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

        eta = max(min_eta, eta * decay)

    return best_a, best_score


def _saml_adaptive(a_init, deadline, max_active=15):
    """SAML with adaptive active set size and step size."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)
    eta = 0.2
    n_act = 5

    for it in range(1000000):
        if time.monotonic() > deadline - 2.0:
            break

        c = _full_convolve(a, a)
        max_c = float(c.max())
        current_score = 2.0 * n * max_c

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()

        # Adaptive active set: use lags within 5% of max
        threshold = max_c * 0.95
        active_mask = c >= threshold
        n_active_actual = int(active_mask.sum())
        if n_active_actual < 1:
            active_mask = np.zeros_like(c, dtype=bool)
            active_mask[int(np.argmax(c))] = True
            n_active_actual = 1
        n_active_actual = min(n_active_actual, max_active)

        active_idx = np.where(active_mask)[0]
        # If too many, take top n_active_actual
        if len(active_idx) > n_active_actual:
            active_vals = c[active_idx]
            top_in_active = np.argsort(active_vals)[-n_active_actual:]
            active_idx = active_idx[top_in_active]

        subgrad = _saml_subgrad(a, active_idx)

        # Adaptive step size
        max_subgrad = float(subgrad.max())
        if max_subgrad > 0:
            eta = min(0.5, max(eta * 0.9995, 0.05 / max_subgrad))

        a = a * np.exp(-eta * subgrad)
        a = np.maximum(a, 1e-30)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    return best_a, best_score


def _lbfgs_logsumexp(a_init, deadline, betas=None):
    """L-BFGS on logsumexp-smoothed objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if betas is None:
        betas = [0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005]
    betas = [b * max_c0 for b in betas]

    time_per = max(10.0, (deadline - time.monotonic()) / len(betas))

    for beta in betas:
        if time.monotonic() > deadline - 3.0:
            break
        level_deadline = min(deadline, time.monotonic() + time_per)

        def obj_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            shifted = (c_ - cm_) / beta
            e_ = np.exp(shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            return 2.0 * n * (cm_ + beta * float(np.log(s_)))

        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            shifted = (c_ - cm_) / beta
            e_ = np.exp(shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            W = float(np.dot(w, c_))
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g

        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 40, 'maxfun': 3000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _local_refine(a_init, deadline, block_size=8, n_iter=50000):
    """Local refinement: try adjusting small blocks."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)

    for it in range(n_iter):
        if time.monotonic() > deadline - 2.0:
            break

        # Try adjusting a random block
        bs = min(block_size, n // 4)
        start = rng.integers(0, n - bs)
        a_trial = a.copy()

        # Multiplicative perturbation
        pert = rng.uniform(0.9, 1.1, size=bs)
        a_trial[start:start + bs] *= pert
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial /= a_trial.sum()

        sc = _score_norm(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()

    return best_a, best_score


def _make_linear_profile(n, slope, offset=0.0):
    t = np.linspace(0.0, 1.0, n)
    a = offset + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_exp_profile(n, rate):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(rate * t)
    return a / a.sum()


def _make_power_profile(n, p):
    t = np.linspace(0.0, 1.0, n)
    a = (t + 1e-8) ** p
    return a / a.sum()


def _make_raised_cos_asym(n, asym=0.3):
    t = np.linspace(0.0, 1.0, n)
    u = t + asym * (t - 0.5)
    u = np.clip(u, 0.0, 1.0)
    a = 0.5 * (1.0 + np.cos(np.pi * (2.0 * u - 1.0)))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_waterfall(n, frac=0.3):
    """High in right portion, tapering to 0 at both ends."""
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    # Rising part
    rise_end = int(n * (1.0 - frac))
    a[:rise_end] = np.linspace(0.0, 1.0, rise_end)
    # Falling part
    a[rise_end:] = np.ones(n - rise_end) * np.exp(-np.linspace(0, 2, n - rise_end))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: diverse initialization, SAML subgradient method,
    L-BFGS logsumexp refinement, and local search.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 4096
        parent = _make_linear_profile(n_parent, 1.0, offset=0.5)
        sc_parent = _score_norm(parent)
        print(f"No parent, using linear n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init_linear")

    n_target = n_parent

    # --- Phase 1: Diverse initialization (40s) ---
    phase1_end = min(deadline, t_start + 40.0)
    candidates = []

    # Add parent
    candidates.append((best_score, best_seq.copy(), "parent"))

    # Diverse profiles at n_target
    profiles = [
        (_make_linear_profile(n_target, 0.5, offset=0.5), "lin_0.5"),
        (_make_linear_profile(n_target, 1.0, offset=0.0), "lin_1.0_0off"),
        (_make_linear_profile(n_target, 1.0, offset=0.5), "lin_1.0"),
        (_make_linear_profile(n_target, 1.5, offset=0.3), "lin_1.5"),
        (_make_linear_profile(n_target, 2.0, offset=0.2), "lin_2.0"),
        (_make_exp_profile(n_target, 1.0), "exp_1.0"),
        (_make_exp_profile(n_target, 1.5), "exp_1.5"),
        (_make_exp_profile(n_target, 2.0), "exp_2.0"),
        (_make_exp_profile(n_target, 0.7), "exp_0.7"),
        (_make_power_profile(n_target, 1.5), "pow_1.5"),
        (_make_power_profile(n_target, 2.0), "pow_2.0"),
        (_make_power_profile(n_target, 0.5), "pow_0.5"),
        (_make_raised_cos_asym(n_target, 0.2), "rcos_0.2"),
        (_make_raised_cos_asym(n_target, 0.4), "rcos_0.4"),
        (_make_waterfall(n_target, 0.3), "waterfall_0.3"),
        (_make_waterfall(n_target, 0.5), "waterfall_0.5"),
    ]

    for idx, (p, label) in enumerate(profiles):
        if time.monotonic() > phase1_end:
            break
        sc = _score_norm(p)
        candidates.append((sc, p, label))
        update_best(p, sc, label)

    # Try n=2048
    for slope, off in [(1.0, 0.0), (1.5, 0.3)]:
        if time.monotonic() > phase1_end:
            break
        p = _make_linear_profile(2048, slope, offset=off)
        sc = _score_norm(p)
        candidates.append((sc, p, f"n2048_{slope}_{off}"))
        update_best(p, sc, f"n2048_{slope}_{off}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done: best={best_score:.10f}, n_cands={len(candidates)}")

    # --- Phase 2: SAML on top candidates (800s) ---
    phase2_end = min(deadline, t_start + 840.0)
    n_saml = 3
    per_saml = max(60.0, (phase2_end - time.monotonic()) / max(1, n_saml))
    count = 0
    seen_labels = set()

    for sc, p, label in candidates:
        if time.monotonic() > phase2_end - 10.0:
            break
        if count >= n_saml:
            break
        if sc > best_score * 1.05:
            break
        if label in seen_labels:
            continue
        seen_labels.add(label)

        dl = min(phase2_end - 5.0, time.monotonic() + per_saml)
        p_best, sc_best = _saml_optimize(p, dl, n_active=5, eta0=0.2, decay=0.9995)
        update_best(p_best, sc_best, f"saml_{label}")

        # Second SAML with different params
        if time.monotonic() < phase2_end - 10.0:
            dl2 = min(phase2_end - 5.0, time.monotonic() + per_saml * 0.5)
            p_best2, sc_best2 = _saml_adaptive(p_best, dl2, max_active=10)
            update_best(p_best2, sc_best2, f"saml2_{label}")

        count += 1

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: L-BFGS refinement (300s) ---
    phase3_end = min(deadline, t_start + 1140.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        dl = phase3_end - 5.0
        p_best, sc_best = _lbfgs_logsumexp(best_seq, dl)
        update_best(p_best, sc_best, "lbfgs_1")

        if time.monotonic() < phase3_end - 30.0:
            dl2 = min(phase3_end - 5.0, time.monotonic() + 120.0)
            p_best2, sc_best2 = _lbfgs_logsumexp(p_best, dl2,
                                                   betas=[0.01, 0.005, 0.002, 0.001])
            update_best(p_best2, sc_best2, "lbfgs_2")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: More SAML with tighter params (200s) ---
    phase4_end = min(deadline, t_start + 1340.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        dl = phase4_end - 5.0
        p_best, sc_best = _saml_optimize(best_seq, dl, n_active=3, eta0=0.1,
                                         decay=0.9998, min_eta=1e-6)
        update_best(p_best, sc_best, "saml_fine")

        if time.monotonic() < phase4_end - 10.0:
            dl2 = min(phase4_end - 5.0, time.monotonic() + 80.0)
            p_best2, sc_best2 = _saml_adaptive(p_best, dl2, max_active=8)
            update_best(p_best2, sc_best2, "saml_adap_fine")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Local refinement (200s) ---
    phase5_end = min(deadline, t_start + 1540.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        dl = phase5_end - 5.0
        p_best, sc_best = _local_refine(best_seq, dl, block_size=6, n_iter=50000)
        update_best(p_best, sc_best, "local_1")

        if time.monotonic() < phase5_end - 10.0:
            dl2 = min(phase5_end - 5.0, time.monotonic() + 80.0)
            p_best2, sc_best2 = _local_refine(p_best, dl2, block_size=12, n_iter=20000)
            update_best(p_best2, sc_best2, "local_2")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Try different n (150s) ---
    phase6_end = min(deadline, t_start + 1690.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        n_curr = len(best_seq)
        # Try interpolating to different lengths
        for n_try in [2048, 2560, 3072, 8192]:
            if time.monotonic() > phase6_end - 10.0:
                break
            if n_try == n_curr:
                continue
            p_interp = _interp_profile(best_seq, n_try)
            sc = _score_norm(p_interp)
            update_best(p_interp, sc, f"interp_{n_try}")

            if sc < best_score + 0.01 and time.monotonic() < phase6_end - 15.0:
                dl = min(phase6_end - 10.0, time.monotonic() + 40.0)
                p_best, sc_best = _saml_optimize(p_interp, dl, n_active=5, eta0=0.15)
                update_best(p_best, sc_best, f"saml_interp_{n_try}")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final polishing (remaining time) ---
    if best_seq is not None:
        remaining = max(10.0, deadline - time.monotonic())

        # Final L-BFGS
        if remaining > 30.0:
            dl = min(deadline - 3.0, time.monotonic() + remaining * 0.4)
            p_best, sc_best = _lbfgs_logsumexp(best_seq, dl,
                                                betas=[0.005, 0.002, 0.001, 0.0005])
            update_best(p_best, sc_best, "final_lbfgs")

        # Final SAML
        if time.monotonic() < deadline - 20.0:
            dl = min(deadline - 3.0, time.monotonic() + (deadline - time.monotonic()) * 0.3)
            p_best, sc_best = _saml_optimize(best_seq, dl, n_active=4, eta0=0.05,
                                             decay=0.9999, min_eta=1e-7)
            update_best(p_best, sc_best, "final_saml")

        # Final local
        if time.monotonic() < deadline - 10.0:
            dl = min(deadline - 2.0, time.monotonic() + 30.0)
            p_best, sc_best = _local_refine(best_seq, dl, block_size=4, n_iter=10000)
            update_best(p_best, sc_best, "final_local")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:

            s = float(final_seq.sum())
        final_seq = final_seq / s
        # Scale up so sum is comfortable (> 0.01)
        if s < 1.0:
            final_seq = final_seq * (s / s)  # already normalized

        sc_final = _score(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        # Verify with evaluator
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator confirms: {sc_eval:.10f}")

                print(f"Evaluator returned non-finite: {sc_eval}, using internal score")

            print(f"evaluate_sequence issue: {ex}")

    fallback = _make_linear_profile(4096, 1.0, offset=0.5)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
