2*n*max(conv(a,a))/(sum(a))^2. Uses active-set Newton steps on the

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    """Full score computation."""
    n = len(a)
    s = float(a.sum())

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _make_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    
    # Linear ramps
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    
    # Exponential
    for rate in [1.0, 2.0, 3.0, 5.0, 7.0, -1.0, -2.0, -3.0, -5.0]:
        a = np.exp(rate * (t - 0.5))
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    
    # Power law
    for p in [0.3, 0.5, 0.7, 1.5, 2.0, 3.0]:
        a = np.maximum(t, 1e-12) ** p
        profiles.append(a / a.sum())
        a2 = np.maximum(1-t, 1e-12) ** p
        profiles.append(a2 / a2.sum())
    
    # Bump (off-center)
    for center in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for width in [0.1, 0.2, 0.3, 0.5]:
            a = np.exp(-((t - center) / width) ** 2)
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())
    
    # Product of ramp and exponential (asymmetric bump)
    for c in [0.3, 0.5, 0.7]:
        for w in [0.2, 0.4, 0.6]:
            a = t * np.exp(-((t - c) / w) ** 2)
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())
    
    return profiles


def _interp_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
