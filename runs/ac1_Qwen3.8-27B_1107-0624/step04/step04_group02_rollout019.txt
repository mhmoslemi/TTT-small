            break
        
        c, cm = _convolve_and_max(a)
        
        # Find all lags within 1% of max
        threshold = cm * 0.999
        peak_lags = np.where(c >= threshold)[0]
        
        # Subgradient: average over peak lags
        g = np.zeros(n)
        for k in peak_lags:
            k = int(k)
            lo = max(0, k - (n-1))
            hi = min(n-1, k)
            if lo <= hi:
                g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        g /= len(peak_lags)
        
        # Polyak step size
        s = a.sum()
        # Projected gradient step
        lr = 0.01 * s / (np.linalg.norm(g) + 1e-15)
        lr = min(lr, 0.1)
        
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
        count += 1
        a_avg = (count * a_avg + a) / (count + 1)
        
        if count % 1000 == 0 and count > 100:
            sc_avg = _score_norm_fn(a_avg)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = a_avg.copy()
    
    return best_a, best_score

def _coordinate_descent(a_init, deadline, rng, n_rounds=10):
    """Coordinate descent: optimize each coordinate with 1D search."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    for round_idx in range(n_rounds):
        if time.monotonic() > deadline - 5:
            break
        
        # Process coordinates in random order
        order = rng.permutation(n)
        
        for idx in order:
            if time.monotonic() > deadline - 5:
                break
            
            j = int(idx)
            a_old_j = a[j]
            
            # Compute baseline convolution (without a_j contributions)
            # c_k = sum_i a_i * a_{k-i}
            # Terms involving a_j: a_j * a_{k-j} (when valid)
            # If a_j = 0: c_k^{(0)} = c_k - a_j * a_{k-j} * (1 + [j≠k/2])
            # Actually let's just compute c with a_j = 0
            a_trial = a.copy()
            a_j_save = a_trial[j]
            a_trial[j] = 0.0
            c_base = _full_convolve(a_trial, a_trial)
            
            # Now c(x) = c_base + x * a_{k-j} * 2 (for linear terms) + x^2 (for k=2j)
            # More precisely:
            # c_k(x) = c_base[k] + x * a_{k-j} (if 0<=k-j<n and k-j≠j)
            #                    + x * a_{k-j} (if 0<=k-j<n and k-j=j, i.e., this is x^2)
            # Wait: c_k = sum_i a_i * a_{k-i}
            # With a_j = x:
            # Terms with i=j: x * a_{k-j} (if 0 <= k-j < n)
            # Terms with k-i=j (i=k-j): a_{k-j} * x (if 0 <= k-j < n)
            # If j ≠ k-j: two terms, total 2x*a_{k-j}
            # If j = k-j (k=2j): one term x^2
            # So c_k(x) = c_base[k] + 2x*a_{k-j} for k≠2j (with 0<=k-j<n)
            #            c_2j(x) = c_base[2j] + x^2
            
            # Define: linear_k = 2*a_{k-j} if 0<=k-j<n and k≠2j, else 0
            #         quad = 1 if k=2j, else 0
            # c_k(x) = c_base[k] + linear_k * x + quad * x^2
            
            # We want to minimize max_k c_k(x) over x >= 0
            # This is a 1D minimax problem.
            
            # The function max_k (c_base[k] + linear_k * x + quad * x^2) is convex in x (max of convex functions)
            # We can use ternary search or golden section search.
            
            # Find a reasonable range for x
            x_lo = 0.0
            x_hi = max(a_j_save * 3, 5 * a.mean())
            
            def f_1d(x):
                # Compute max of c_k(x)
                c_x = c_base.copy()
                if 0 <= 2*j < len(c_base):
                    c_x[2*j] += x * x
                # Linear terms
                k_lo = max(0, j)
                k_hi = min(len(c_base)-1, j + n - 1)
                # For k in [j, j+n-1]: linear_k = 2*a_{k-j} (if k != 2j)
                for k_idx in range(k_lo, k_hi + 1):
                    if k_idx != 2 * j:
                        c_x[k_idx] += 2.0 * x * a_trial[k_idx - j]
                return c_x.max()
            
            # Golden section search
            gr = (np.sqrt(5) - 1) / 2  # golden ratio
            a_gs = x_lo
            b_gs = x_hi
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)
            fc = f_1d(c_gs)
            fd = f_1d(d_gs)
            
            for _ in range(50):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = f_1d(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = f_1d(d_gs)
                if b_gs - a_gs < 1e-12 * max(1, x_hi):
                    break
            
            x_opt = (a_gs + b_gs) / 2
            x_opt = max(0, x_opt)
            
            # Apply update
            a[j] = x_opt
            sc = _score_norm_fn(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            
            # Re-normalize occasionally
            if (j + 1) % 100 == 0:
                s = a.sum()
                if s > 0:
                    a = a / s
                    best_score = min(best_score, _score_norm_fn(a))
                    if _score_norm_fn(a) < best_score:
                        best_a = a.copy()
    
    return best_a, best_score
