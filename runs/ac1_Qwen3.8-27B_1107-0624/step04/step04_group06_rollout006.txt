LP refinement, and multi-scale search to minimize 2*n*max(conv(a,a))/sum(a)^2.

from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())

def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    val = mval + beta * np.log(s)
    w = ex / s
    return val, w

def _softmax_from_z(z, floor=1e-30):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0:

    else:

    return p

    N = _next_pow2(4 * n - 3)

    w_pad = np.zeros(N)
    w_pad[:2*n-1] = w
    fa = np.fft.fft(a_pad)
    fw = np.fft.fft(w_pad)
    R = np.fft.ifft(np.conj(fa) * fw).real[:n]

        dot = float(np.dot(a, g_a))

def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def _lp_refine(a, deadline, max_solves=30, t_init=0.3):
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    t = t_init
    
    for _ in range(max_solves):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a)
        M = float(c.max())
        
        b = _lp_solve(a, M)
        if b is None:
            b = _lp_solve(a, M * 1.02)
            if b is None:
                break
        
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        
        b = b / s_b
        a_new = (1.0 - t) * a + t * b
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        sc_new = _score_norm(a_new)
        
        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()
        
        a = a_new
        t = max(0.1, t * 0.9)
    
    return best_a, best_score

def _interp_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, n_old - 1, n_old)
    x_new = np.linspace(0, n_new - 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    return a_new
