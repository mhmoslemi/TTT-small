def falling_profile(length, exponent=1.5):
    """Generate exponential decay profile normalized to sqrt(2 * length)"""
    t = np.linspace(0, 1, length)
    a = (t + 0.5) ** (-exponent)
    norm = np.sqrt(2 * length)
    total = np.sum(a)
    if abs(total) < 1e-10:
        return np.ones(length) / length * norm
    return a / total * norm

            best_score_actual, forced = score_and_sequence(best_seq)
            if np.isfinite(best_score_actual) and best_score_actual < best_score:
                best_score = best_score_actual
        except:
            # Handle optimization failure
            pass

    return best_seq.tolist()
