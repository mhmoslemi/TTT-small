    # Defensive checks from the submit function

    sequence = [min(1000.0, x) for x in sequence]

    start_time = time.time()
    while time.time() < start_time + time_quota:
        # Randomized weight stepping
        step_direction = np.random.choice([-1, 0, 1], size=n)
        amount = np.random.rand(n) * 0.1

        candidate = candidate * (np.abs(1-step_direction)) + (step_direction==1) * candidate
