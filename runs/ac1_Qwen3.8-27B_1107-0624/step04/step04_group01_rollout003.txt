"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS, Adam with beta-continuation, coordinate descent, and peak-shaving.
Starts from parent sequence and explores multiple n values.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score(a):
    """Compute the full score 2*n*max(conv)/sum^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_dL_da(a, c, w, n):
    """Compute dL/da where L = LSE(conv(a,a), beta).
    dL/da[m] = 2 * sum_k w[k] * a[k-m] (for valid k-m)
    This equals 2 * (w * a_rev)[m + n - 1] extracted appropriately.
    """
    a_rev = a[::-1]
    corr = _full_convolve(w, a_rev)
    # corr has length (2n-1) + n - 1 = 3n-2
    # dL/da[m] = 2 * sum_{k=m}^{m+n-1} w[k] * a[k-m]
    # = 2 * corr[m + n - 1] (checking: corr[m+n-1] = sum_j w[j]*a_rev[m+n-1-j] = sum_j w[j]*a[n-1-(m+n-1-j)] = sum_j w[j]*a[j-m])
    # Let i = j-m: sum_i w[i+m]*a[i] -- yes this is what we want
    dL_da = 2.0 * corr[n-1:2*n-1]
    return dL_da

def _adam_obj_and_grad(z, beta, n):
    """Compute objective and gradient w.r.t. z for Adam-style optimization.
    Objective: 2*n*LSE(conv(a,a), beta) with a = softmax(z) (so sum(a)=1).
    """
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    w = e / s
    
    # Smoothed max value
    L = cm + beta * float(np.log(s))
    
    # Gradient
    dL_da = _compute_dL_da(a, c, w, n)
    dJ_da = 2.0 * n * dL_da
    W = float(np.dot(a, dJ_da))
    grad = a * (dJ_da - W)
    
    obj = float(2.0 * n * L)
    return obj, grad

def _adam_optimize(z_init, deadline, beta_schedule, max_iter=200000, lr=0.01, verbose=False):
    """Adam optimizer with beta-continuation."""
    n = len(z_init)
    z = z_init.copy()
    
    best_a = _softmax(z)
    best_norm_score = _score_norm(best_a)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            
            obj, grad = _adam_obj_and_grad(z, beta, n)
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            # Clip
            max_update = 1.0
            un = np.linalg.norm(update)
            if un > max_update:
                update = update * (max_update / un)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if it_in_phase % 200 == 0:
                a = _softmax(z)
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_norm_score:
                    best_norm_score = sc
                    best_a = a.copy()
                    if verbose:
                        print(f"    beta={beta:.6f}, iter={it_in_phase}, score={sc:.10f}")
            
            if it_in_phase % 5000 == 0:
                current_lr = max(0.001, current_lr * 0.95)
    
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_norm_score:
        best_norm_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_norm_score

def _lbfgs_optimize(a_init, deadline, beta, max_iter=5000):
    """L-BFGS optimization using scipy, working in log-space with softmax."""
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):
        obj, _ = _adam_obj_and_grad(z, beta, n)
        return obj
    
    def grad(z):
        _, g = _adam_obj_and_grad(z, beta, n)
        return g
    
    try:
        res = minimize(fun, z0, method='L-BFGS-B', jac=grad,
                      options={'maxiter': max_iter, 'maxfun': max_iter * 2,
                             'ftol': 1e-12, 'gtol': 1e-8,
                             'iprint': 0})
        z_opt = res.x
        a_opt = _softmax(z_opt)
        return a_opt
    except Exception as e:
        print(f"  L-BFGS error: {e}")
        return a_init

def _coordinate_ascent(a, c, n, num_steps=100):
    """
    Coordinate descent: for each element a[m], find the best value that reduces
    the max of the convolution. This is a 1D optimization.
    
    We parameterize a[m] = alpha * a0[m] (scaling) and find optimal alpha.
    """
    a = a.copy()
    best_score = _score_norm(a)
    
    for step in range(num_steps):
        improved = False
        # Try each coordinate (randomly shuffled)
        idx = np.random.permutation(n)
        count = 0
        for m in idx:
            # Try different scaling factors
            for alpha in [0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5]:
                a_new = a.copy()
                a_new[m] = alpha * a[m]
                a_new = np.maximum(a_new, 0)
                if a_new.sum() < 1e-10:
                    continue
                a_new = a_new / a_new.sum()  # renormalize
                sc = _score_norm(a_new)
                if sc < best_score - 1e-10:
                    a = a_new
                    best_score = sc
                    improved = True
                    break
            count += 1
            if count >= n:
                break
        if not improved:
            break
    
    return a, best_score

def _peak_shaving(a, c, n, num_rounds=50):
    """
    Peak shaving: identify the top peaks of the convolution and try to
    reduce them by adjusting the contributing elements.
    """
    a = a.copy()
    best_score = _score_norm(a)
    
    for round_idx in range(num_rounds):
        c = _full_convolve(a, a)
        M = float(c.max())
        # Find lags where c is close to max
        threshold = M * 0.95
        peak_lags = np.where(c >= threshold)[0]
        
        for k in peak_lags:
            # c[k] = sum_i a[i]*a[k-i]
            # For each contributing pair (i, k-i), try reducing one of them
            valid_i = range(max(0, k - n + 1), min(n, k + 1))
            for i in valid_i:
                j = k - i
                if j < 0 or j >= n:
                    continue
                # Try reducing a[i]
                for factor in [0.7, 0.8, 0.9]:
                    a_new = a.copy()
                    a_new[i] *= factor
                    a_new = np.maximum(a_new, 0)
                    s = a_new.sum()
                    if s < 1e-10:
                        continue
                    a_new /= s
                    sc = _score_norm(a_new)
                    if sc < best_score - 1e-12:
                        a = a_new
                        best_score = sc
                        break
        
        # Check time
        if time.monotonic() > time.time():  # placeholder
            break
    
    return a, best_score

def _make_profile(n, kind="uniform"):

        z_init = z_init - z_init.mean()
        
        per_time = phase2_end - time.monotonic() - 10
        a_best, sc_best = _adam_optimize(
            z_init, time.monotonic() + per_time,
            beta_schedule, max_iter=300000, lr=0.015, verbose=True
        )
        update_best(a_best, _score(a_best), "adam_main")
    
    print(f"Phase 2 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # --- Phase 3: L-BFGS on refined solution (500s - 700s) ---
    phase3_end = min(deadline, t_start + 700.0)
    print(f"\n=== Phase 3: L-BFGS refine (t=500s to 700s) ===")
    
    if best_seq is not None and time.monotonic() < phase3_end - 10:
        c_ref = _full_convolve(best_seq, best_seq)
        max_cr = float(c_ref.max())
        for beta_frac in [0.005, 0.002, 0.001]:
            if time.monotonic() > phase3_end - 5:
                break
            beta_r = beta_frac * max_cr
            a_ref = _lbfgs_optimize(best_seq, phase3_end - 5, beta_r, max_iter=3000)
            update_best(a_ref, _score(a_ref), f"lbfgs_beta{beta_frac}")
            if best_seq is not None:
                a_ref = a_ref  # best_seq already updated
    
    print(f"Phase 3 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # --- Phase 4: Multi-restart (700s - 1100s) ---
    phase4_end = min(deadline, t_start + 1100.0)
    print(f"\n=== Phase 4: Multi-restart (t=700s to 1100s) ===")
    
    if best_seq is not None and time.monotonic() < phase4_end:
        n_restarts = 6
        per_restart = (phase4_end - time.monotonic() - 20) / n_restarts
        
        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 15:
                break
            
            if i % 2 == 0:
                p_start = _perturb(best_seq, 0.03 + 0.01 * i, rng)
            else:
                # Fourier perturbation
                n_s = len(best_seq)
                z_s = np.log(np.maximum(best_seq, 1e-15))
                freq = rng.integers(1, n_s // 4)
                amp = 0.05 * rng.uniform(0.5, 2.0)
                t_arr = np.arange(n_s)
                z_s = z_s + amp * np.cos(2.0 * np.pi * freq * t_arr / n_s)
                z_s = z_s - z_s.mean()
                p_start = _softmax(z_s)
            
            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = [0.02 * max_cs, 0.01 * max_cs, 0.005 * max_cs,
                  0.002 * max_cs, 0.001 * max_cs, 0.0005 * max_cs]
            
            z_s = np.log(np.maximum(p_start, 1e-15))
            z_s = z_s - z_s.mean()
            
            p_best, sc_best = _adam_optimize(
                z_s, time.monotonic() + per_restart,
                bs, max_iter=100000, lr=0.01
            )
            update_best(p_best, _score(p_best), f"restart{i}")
    
    print(f"Phase 4 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # --- Phase 5: Try n=4096 (1100s - 1400s) ---
    phase5_end = min(deadline, t_start + 1400.0)
    print(f"\n=== Phase 5: n=4096 (t=1100s to 1400s) ===")
    
    if best_seq is not None and len(best_seq) < 4096 and time.monotonic() < phase5_end - 30:
        target_n = 4096
        p_interp = _interp_profile(best_seq, target_n)
        sc_interp = _score(p_interp)
        update_best(p_interp, sc_interp, f"interp_n{target_n}")
        
        remaining = phase5_end - time.monotonic() - 20
        if remaining > 30:
            c_int = _full_convolve(p_interp, p_interp)
            max_ci = float(c_int.max())
            bs_int = [0.005 * max_ci, 0.002 * max_ci, 0.001 * max_ci,
                      0.0005 * max_ci, 0.0002 * max_ci]
            z_int = np.log(np.maximum(p_interp, 1e-15))
            z_int = z_int - z_int.mean()
            
            p_best, sc_best = _adam_optimize(
                z_int, time.monotonic() + remaining,
                bs_int, max_iter=150000, lr=0.008
            )
            update_best(p_best, _score(p_best), f"opt_n{target_n}")
    
    # Also try going back to n=2048 from n=4096 solution if it was better
    if best_seq is not None and len(best_seq) == 4096:
        p_2048 = _interp_profile(best_seq, 2048)
        sc_2048 = _score(p_2048)
        # Don't update best (we want to keep 4096), but check
        print(f"  n=2048 from 4096: score={sc_2048:.10f}")
    
    print(f"Phase 5 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # --- Phase 6: Fine polish with coordinate descent (1400s - 1650s) ---
    phase6_end = min(deadline, t_start + 1650.0)
    print(f"\n=== Phase 6: Coordinate descent polish (t=1400s to 1650s) ===")
    
    if best_seq is not None and time.monotonic() < phase6_end - 10:
        a_cd = best_seq.copy()
        c_cd = _full_convolve(a_cd, a_cd)
        n_cd = len(a_cd)
        
        # Coordinate descent: try to adjust individual elements
        improved_any = True
        while improved_any and time.monotonic() < phase6_end - 5:
            improved_any = False
            # Use a rotating order
            start_idx = int(time.monotonic() * 1000) % n_cd
            order = np.concatenate([np.arange(start_idx, n_cd), np.arange(0, start_idx)])
            
            for m in order:
                if time.monotonic() > phase6_end - 5:
                    break
                base_val = a_cd[m]
                if base_val < 1e-15:
                    continue
                # Try a few scaling factors
                for alpha in [0.85, 0.9, 0.95, 1.05, 1.1, 1.15]:
                    a_try = a_cd.copy()
                    a_try[m] = base_val * alpha
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    sc_try = _score_norm(a_try)
                    if sc_try < _score_norm(a_cd) - 1e-12:
                        a_cd = a_try
                        improved_any = True
                        break
    
        sc_cd = _score(a_cd)
        update_best(a_cd, sc_cd, "coord_descent")
