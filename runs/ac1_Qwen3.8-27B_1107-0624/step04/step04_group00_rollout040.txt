"""
Optimized step function search using active-set-aware smooth max,
multi-n exploration, and continuation-based L-BFGS with adaptive
peak tracking. Key improvements over previous approach:
- Top-k aware smoothing that focuses on the critical peaks
- Direct active-set identification and Newton refinement

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _topk_smooth_obj(z, beta, n, topk):
    """Objective: smooth max of top-k convolution values (scale-invariant)."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    # Get top-k values
    k = min(topk, len(c))
    sorted_idx = np.argpartition(c, -k)[-k:]
    top = c[sorted_idx]
    m = top.max()
    return 2.0 * n * (m + beta * np.log(np.sum(np.exp((top - m) / beta))))


def _topk_smooth_grad(z, beta, n, topk):
    """Gradient of top-k smooth max w.r.t. log-space variables z."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    k = min(topk, len(c))
    sorted_idx = np.argpartition(c, -k)[-k:]
    top = c[sorted_idx]
    m = top.max()
    # Softmax weights over top-k
    w_top = np.exp((top - m) / beta)
    w_top = w_top / w_top.sum()
    # Full weight vector over c
    w = np.zeros(len(c))
    w[sorted_idx] = w_top
    # Gradient of smooth_max w.r.t. c: w
    # Gradient of c[k] w.r.t. a[j]: 2*a[k-j] (if valid)
    # So grad_c = sum_k w[k] * 2*a[k-j] for each j
    # This is 2 * conv(w, a_reversed) centered
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    # Gradient of objective w.r.t. a: 2n * 2*h (from chain rule through c)
    # Wait: obj = 2n * smooth_max(c), dc/da_j = 2*sum_k w[k]*a[k-j] = 2*h[j]
    # d(obj)/da_j = 2n * 2*h[j] = 4n*h[j]
    # d(da_j)/dz_j = a_j
    # d(obj)/dz_j = 4n*h[j]*a[j]
    grad = 4.0 * n * a * h
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs_topk(a_init, deadline, rng, topk=3, beta_list=None):
    """L-BFGS with top-k smooth max objective and continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if beta_list is None:
        # Continuation: start with moderate beta, end with very small
        beta_list = [0.1, 0.05, 0.03, 0.02, 0.015, 0.01, 0.008,
                     0.006, 0.005, 0.004, 0.003, 0.002, 0.0015,
                     0.001, 0.0008, 0.0006, 0.0005, 0.0004,
                     0.0003, 0.0002, 0.00015, 0.0001, 0.00005]
        beta_list = [b * max_c0 for b in beta_list]

    n_betas = len(beta_list)
    total_time = deadline - time.monotonic()
    if total_time <= 0:
        return best_a, best_score

    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline - 5:
            break

        # Time allocation: more time for smaller betas
        remaining = n_betas - idx
        frac = (n_betas - remaining) / n_betas  # progress so far
        time_left = deadline - time.monotonic()
        # Allocate proportionally to remaining betas, weighted toward small beta
        weight = (idx + 1) ** 1.5  # favor later (smaller beta) stages
        total_weight = sum((i + 1) ** 1.5 for i in range(idx, n_betas))
        time_alloc = time_left * weight / total_weight if total_weight > 0 else 10
        time_alloc = min(time_alloc, 120.0)
        level_deadline = time.monotonic() + time_alloc

        # Scale topk based on beta: smaller beta -> fewer peaks matter
        effective_topk = max(2, min(topk, int(2 + 10 * beta / max_c0)))

        maxiter = max(200, int(5000 * time_alloc / 60.0))
        maxfun = max(1000, maxiter * 5)

        try:
            result = minimize(
                _topk_smooth_obj, z, args=(beta, n, effective_topk),
                method='L-BFGS-B', jac=_topk_smooth_grad,
                options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': maxfun}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                if np.all(a_new >= -1e-10) and np.all(np.isfinite(a_new)):
                    a_new = np.maximum(a_new, 0)
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
                        print(f"    beta={beta/max_c0:.6f}, score={s_new:.10f}, iters={result.nit}")
        except Exception as e:
            print(f"    L-BFGS error at beta={beta/max_c0:.6f}: {e}")

    return best_a, best_score


def _run_adam_topk(a_init, deadline, rng, topk=3, lr_init=0.1, max_iter=100000):
    """Adam optimizer with top-k smooth max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c = _full_convolve(a_init, a_init)
    max_c0 = float(c.max())

    # Adaptive beta schedule
    beta_start = 0.05 * max_c0
    beta_end = 1e-5 * max_c0

    m_v = np.zeros(n, dtype=np.float64)
    v_v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Adaptive beta: interpolate from start to end
        frac = min(1.0, it / (max_iter * 0.8))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = max(beta, 1e-10 * max_c0)

        t += 1
        grad = _topk_smooth_grad(z, beta, n, topk)
        if not np.all(np.isfinite(grad)):
            break

        m_v = b1 * m_v + (1 - b1) * grad
        v_v = b2 * v_v + (1 - b2) * (grad * grad)
        m_hat = m_v / (1 - b1 ** t)
        v_hat = v_v / (1 - b2 ** t)

        # Learning rate with decay
        lr = lr_init * (0.5 ** (it / 5000.0))
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Periodic kick if stuck
        if it > 0 and it % 10000 == 0:
            z = z + 0.02 * rng.normal(size=n)
            z = z - z.mean()

    return best_a, best_score


def _direct_peak_equalization(a_init, deadline, rng, n_peaks=3):

            aa = _softmax_fn(z)

            s = aa.sum()
            return 2.0 * n * max(cc[active]).item() / (s * s)

        # This is non-smooth; use Nelder-Mead instead

def _make_structured_profiles(n, rng):
    """Generate diverse structured initial profiles."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1  # in [-1, 1]
    profiles = []

    # Linear ramps (various)
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Quadratic
    for a1, a2 in [(1.0, 0.5), (2.0, -0.3), (1.5, 0.2), (-1.0, 0.5),
                   (3.0, 1.0), (-2.0, 0.8), (0.5, -0.3)]:
        a = 1.0 + a1 * u + a2 * u ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Cubic
    for a1, a2, a3 in [(1.0, 0.5, 0.1), (2.0, -0.5, 0.3), (-1.0, 1.0, -0.2)]:
        a = 1.0 + a1 * u + a2 * u ** 2 + a3 * u ** 3
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Exponential
    for rate in [1.0, 2.0, 3.0, 5.0, -1.0, -2.0, -3.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())

    # Power law
    for p in [0.5, 0.7, 1.5, 2.0, 3.0]:
        a = (t + 0.1) ** p
        profiles.append(a / a.sum())
        a = (1 - t + 0.1) ** p
        profiles.append(a / a.sum())

    # Cosine variants
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())

    # Shifted cosine (asymmetric)
    for shift in [0.1, 0.2, 0.3, -0.1, -0.2, -0.3]:
        a = np.maximum(np.cos(np.pi * (u - shift) / (2 - abs(shift))), 1e-12) ** 2
        profiles.append(a / a.sum())

    # Gaussian-like (asymmetric)
    for center, width in [(0.4, 0.2), (0.6, 0.2), (0.5, 0.15), (0.3, 0.3), (0.7, 0.3)]:
        a = np.exp(-((t - center) / width) ** 2)
        profiles.append(a / a.sum())

    # Double-hump
    for c1, c2 in [(0.3, 0.7), (0.25, 0.6), (0.4, 0.8)]:
        a = np.exp(-40 * (t - c1) ** 2) + np.exp(-40 * (t - c2) ** 2)
        profiles.append(a / a.sum())

    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _fourier_parameterize(n, K=6):
    """Create Fourier basis for smooth profiles."""
    t = np.linspace(0, 2 * np.pi, n)
    basis = np.ones((n, 2 * K + 1))
    for k in range(1, K + 1):
        basis[:, 2 * k - 1] = np.sin(k * t)
        basis[:, 2 * k] = np.cos(k * t)
    return basis


def _optimize_fourier_profile(n, deadline, rng, K=5):
    """Optimize a Fourier-parameterized profile."""
    basis = _fourier_parameterize(n, K)
    dim = basis.shape[1]

    best_score = np.inf
    best_alpha = np.zeros(dim)

    def fourier_score_alpha(alpha):
        log_a = basis @ alpha
        log_a = log_a - np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s < 1e-10:
            return np.inf
        a = a / s
        return _score_norm_fn(a)

    # Try multiple initializations
    n_inits = 20
    t_start_phase = time.monotonic()
    time_per_init = (deadline - t_start_phase) / max(1, n_inits)

    for init_idx in range(n_inits):
        if time.monotonic() > deadline - 5:
            break

        if init_idx == 0:
            alpha0 = np.zeros(dim)
        elif init_idx == 1:
            alpha0 = np.zeros(dim)
            alpha0[2] = 1.0  # cos(t)
        elif init_idx == 2:
            alpha0 = np.zeros(dim)
            alpha0[1] = 1.0  # sin(t)
        elif init_idx == 3:
            alpha0 = np.zeros(dim)
            alpha0[4] = -0.5  # -cos(2t)
        elif init_idx == 4:
            alpha0 = np.zeros(dim)
            alpha0[2] = 1.5
            alpha0[4] = -0.3
        else:
            alpha0 = rng.normal(0, 0.5, size=dim)

        per_dl = min(deadline - 2, time.monotonic() + time_per_init)
        try:
            res = minimize(fourier_score_alpha, alpha0, method='Nelder-Mead',
                           options={'maxiter': 3000, 'xatol': 1e-12, 'fatol': 1e-14})
            if res.fun < best_score:
                best_score = res.fun
                best_alpha = res.x.copy()
        except Exception:

    a = a / a.sum()

            break

        if r == 0:
            a_cur = a_start.copy()
            topk = 3
        elif r == 1:
            # Perturb: focus on making it more asymmetric
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            # Add a linear tilt
            tilt = rng.uniform(0.5, 2.0) * rng.choice([-1, 1])
            z += tilt * np.linspace(-0.5, 0.5, n_best)
            z = z - z.mean()
            a_cur = _softmax_fn(z)
            topk = 3
        else:
            # Larger perturbation
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z += 0.1 * rng.normal(size=n_best)
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            a_cur = _softmax_fn(z)
            topk = 4

        per_dl = min(phase2_end - 5, time.monotonic() + per_restart - 5)
        label = f"lbfgs_topk_r{r}_n{n_best}"
        p_best, sc_best = _run_lbfgs_topk(a_cur, per_dl, rng, topk=topk)
        update_best(p_best, sc_best, label)

    print(f"  Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Adam with adaptive beta (300s) ---
    phase3_end = min(deadline, t_start + 1220.0)
    print("Phase 3: Adam optimization")

    if time.monotonic() < phase3_end and best_seq is not None:
        per_dl = min(phase3_end - 5, time.monotonic() + (phase3_end - time.monotonic()) * 0.6)
        p_best, sc_best = _run_adam_topk(best_seq, per_dl, rng, topk=3, lr_init=0.08)
        update_best(p_best, sc_best, "adam_topk3")

        if time.monotonic() < phase3_end - 20:
            per_dl2 = min(phase3_end - 5, time.monotonic() + (phase3_end - time.monotonic()) * 0.4)
            p_best2, sc_best2 = _run_adam_topk(best_seq, per_dl2, rng, topk=5, lr_init=0.05)
            update_best(p_best2, sc_best2, "adam_topk5")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Try interpolation to other n values (200s) ---
    phase4_end = min(deadline, t_start + 1420.0)
    print("Phase 4: Multi-n exploration")

    if time.monotonic() < phase4_end and best_seq is not None:
        n_curr = len(best_seq)
        for n_new in [n_curr // 2, n_curr * 2, n_curr + 512, n_curr - 512]:
            if time.monotonic() > phase4_end - 30:
                break
            if n_new < 512 or n_new > 8192:
                continue
            if n_new == n_curr:
                continue

            p_interp = _interp_profile(best_seq, n_new)
            sc_interp = _score_norm_fn(p_interp)
            if sc_interp < best_score:
                update_best(p_interp, sc_interp, f"interp_n{n_new}")

            # Quick L-BFGS on the interpolated profile
            if time.monotonic() < phase4_end - 30 and sc_interp < best_score + 0.1:
                per_dl = min(phase4_end - 10, time.monotonic() + 60)
                p_best, sc_best = _run_lbfgs_topk(p_interp, per_dl, rng, topk=3)
                update_best(p_best, sc_best, f"lbfgs_interp_n{n_new}")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Direct peak equalization (150s) ---
    phase5_end = min(deadline, t_start + 1570.0)
    print("Phase 5: Peak equalization")

    if time.monotonic() < phase5_end and best_seq is not None:
        per_dl = min(phase5_end - 5, time.monotonic() + (phase5_end - time.monotonic()) * 0.9)
        p_best, sc_best = _direct_peak_equalization(best_seq, per_dl, rng, n_peaks=3)
        update_best(p_best, sc_best, "peak_eq")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Final ultra-fine polish (remaining time) ---
    phase6_end = deadline
    print("Phase 6: Final polish")

    if time.monotonic() < phase6_end and best_seq is not None:
        # Very fine L-BFGS with smallest beta
        c_final = _full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())

        # Create a beta list that's very fine
        fine_betas = [0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
        fine_beta_list = [b * max_c_final for b in fine_betas]

        per_dl = min(phase6_end - 10, time.monotonic() + (phase6_end - time.monotonic()) * 0.5)
        p_best, sc_best = _run_lbfgs_topk(best_seq, per_dl, rng, topk=2, beta_list=fine_beta_list)
        update_best(p_best, sc_best, "final_fine_lbfgs")

        # Final Adam with very small lr
        if time.monotonic() < phase6_end - 10:
            per_dl2 = min(phase6_end - 5, time.monotonic() + (phase6_end - time.monotonic()) * 0.5)
            p_best2, sc_best2 = _run_adam_topk(best_seq, per_dl2, rng, topk=2, lr_init=0.01, max_iter=200000)
            update_best(p_best2, sc_best2, "final_adam_slow")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            # The score is scale-invariant, but let's keep sum reasonable
            # for numerical stability in the evaluator
            pass
        else:
            final_seq = np.ones(256) / 256.0

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 8192:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                    if sc_eval < best_score:
                        best_score = sc_eval
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = np.linspace(1, 2, 2048)
    fallback = fallback / fallback.sum()
    print(f"FALLBACK: n=2048, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
