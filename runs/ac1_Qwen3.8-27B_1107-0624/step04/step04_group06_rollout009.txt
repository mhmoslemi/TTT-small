import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())

def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-15, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s

def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p

def _gradient_at_peak(p, k):
    """Compute subgradient of (f*f)(k) with respect to f."""
    n = len(p)
    g = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    for i in range(i_start, i_end + 1):
        j = k - i
        if i == j:
            g[i] += 2.0 * p[i]
        else:
            g[i] += 2.0 * p[j]
    return g

def _gradient_smooth(p, c, beta):
    """Compute smooth gradient using soft-max weighting over near-peak positions."""
    n = len(p)
    mval = float(c.max())
    if beta < 1e-14:
        # Hard max: use only the argmax
        k_star = int(np.argmax(c))
        return _gradient_at_peak(p, k_star)
    
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        k_star = int(np.argmax(c))
        return _gradient_at_peak(p, k_star)
    w = ex / s
    
    # Weighted sum of subgradients
    g = np.zeros(n)
    # Only consider positions with significant weight
    active = np.where(w > 1e-8)[0]
    for k in active:
        g += w[k] * _gradient_at_peak(p, int(k))
    return g

def _lp_solve_with_dual(a, M):
    """Solve LP and return both primal and dual (shadow prices)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        primal = np.maximum(res.x, 0.0)
        # Dual variables (shadow prices) for the inequality constraints
        dual = np.zeros(2 * n - 1)
        if res.unequality.marginals is not None:
            dual = np.maximum(-res.unequality.marginals, 0.0)  # negated because of sign convention
        return primal, dual
    return None, None

        sc_new = _score_norm(a_new)

def _make_profile(n, type='linear', param=1.0):
    if type == 'linear':
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + param * (t - 0.5)
    elif type == 'quadratic':
        t = np.linspace(0.0, 1.0, n)
        u = 2.0 * t - 1.0
        a = 1.0 + param * u * u
    elif type == 'cubic':
        t = np.linspace(0.0, 1.0, n)
        u = 2.0 * t - 1.0
        a = 1.0 + param * u + 0.3 * param * u ** 3
    elif type == 'cosine':
        t = np.arange(n)
        a = 1.0 + param * np.cos(2.0 * np.pi * t / n)
    elif type == 'ramp':
        t = np.linspace(0.0, 1.0, n)
        a = 0.1 + param * t
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))

    c = _full_convolve(a)
    # Only include constraints where c[k] > threshold_frac * M
    active_k = np.where(c > threshold_frac * M)[0]

    cols = []
    vals = []

        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())

    if res.success:
        return np.maximum(res.x, 0.0)

"""
Optimized step function search using:
1. FFT-based fast smooth gradient descent with momentum
2. Reduced-constraint LP refinement with adaptive M scheduling
3. Multiplicative peak-flooding
4. Multi-scale interpolation (1024 -> 2048)
5. Structured restarts and alternative initializations
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-15, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _gradient_at_peak(p, k):
    n = len(p)
    g = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start > i_end:
        return g
    idx = np.arange(i_start, i_end + 1)
    partners = k - idx
    g[idx] = 2.0 * p[partners]
    return g


def _gradient_smooth_fast(p, c, beta):
    n = len(p)
    mval = float(c.max())
    if beta < 1e-14:
        k_star = int(np.argmax(c))
        return _gradient_at_peak(p, k_star)
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        k_star = int(np.argmax(c))
        return _gradient_at_peak(p, k_star)
    w = ex / s
    N = _next_pow2(3 * n - 2)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(fw * fp, N)[:3 * n - 2]
    g = 2.0 * conv[n - 1:2 * n - 1]
    return g


def _lp_solve_reduced(a, M, threshold_frac=0.85):
    n = len(a)
    c = _full_convolve(a)
    active_k = np.where(c > threshold_frac * M)[0]
    if len(active_k) == 0:
        active_k = np.array([int(np.argmax(c))])
    rows = []
    cols = []
    vals = []
    for k in active_k:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([int(k)] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(len(active_k))
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_solve_full(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a, deadline, max_solves=60, m_shrink=0.998, t_init=0.4, t_decay=0.95,
               use_reduced=True, threshold_frac=0.85):
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    t = t_init
    n_solves = 0
    last_improve = 0
    stall_count = 0

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        t0 = time.monotonic()

        if use_reduced:
            b = _lp_solve_reduced(a, M, threshold_frac)
        else:
            b = _lp_solve_full(a, M)

        solve_time = time.monotonic() - t0

        if b is None:
            if use_reduced:
                b = _lp_solve_full(a, M)
            if b is None:
                M *= 1.01
                stall_count += 1
                if stall_count > 10:
                    break
                continue

        s_b = float(b.sum())
        if s_b < 1e-10:
            break

        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)

        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()
            last_improve = n_solves
            stall_count = 0

        a = a_new
        c = _full_convolve(a)
        M_new = float(c.max())

        M = max(M_new * m_shrink, best_score / (2.0 * n) * 1.003)
        t = max(0.1, t * t_decay)

        if n_solves - last_improve > 15:
            M = float(c.max()) * 1.005
            t = max(0.15, t_init * 0.6)
            stall_count += 1
            if stall_count > 5:
                break

        if solve_time > 25 and n > 1024:
            use_reduced = True
            threshold_frac = max(0.9, threshold_frac - 0.02)

        if n_solves % 10 == 0:
            print(f"    LP iter {n_solves}: score={best_score:.10f}, M={M:.6f}, t={t:.3f}, solve_time={solve_time:.1f}s")

    return best_a, best_score


def _gradient_descent(a, deadline, rng, lr=0.04, momentum=0.88, beta_frac=0.08,
                      beta_decay=0.94, no_imp_limit=200):
    p0 = np.asarray(a, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    base_momentum = momentum

    v = np.zeros(n)

    while time.monotonic() < deadline:
        g = _gradient_smooth_fast(p, c, beta)

        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            v.fill(0.0)
            continue

        v = momentum * v - lr * gz
        z = z + v

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            v.fill(0.0)
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 80 == 0:
                    lr = max(0.001, lr * 0.93)
        else:
            z = best_z.copy()
            v.fill(0.0)
            continue

        if no_imp % 40 == 0:
            beta = max(1e-6, beta * beta_decay)

        if no_imp > no_imp_limit:
            if best_score < 1.51:
                ns = 0.04
            elif best_score < 1.53:
                ns = 0.08
            else:
                ns = 0.15
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            v.fill(0.0)
            lr = base_lr
            momentum = min(0.95, base_momentum + 0.05)
            no_imp = 0

    return best_p, best_score


def _peak_flood(a, deadline, eta=0.03, n_iter_max=10000):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)

    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break

        M = float(c.max())
        score = float(2.0 * n * M)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        threshold = M * 0.88
        hot_mask = c >= threshold
        if not hot_mask.any():
            hot_mask[int(np.argmax(c))] = True

        heat = np.zeros(n)
        hot_indices = np.where(hot_mask)[0]
        for k in hot_indices:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                partner = k - np.arange(i_start, i_end + 1)
                heat[i_start:i_end + 1] += a[partner]

        heat_max = float(heat.max())
        if heat_max < 1e-15:
            break
        heat = heat / heat_max

        factor = 1.0 - eta * heat
        factor = np.maximum(factor, 0.01)
        a = a * factor
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        c = _full_convolve(a)

        if it % 200 == 0 and it > 0:
            if score >= best_score:
                eta = max(0.003, eta * 0.9)
            else:
                eta = min(0.1, eta * 1.05)

    return best_a, best_score


def _make_profile(n, ptype, param=1.0):
    if ptype == 'linear':
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + param * (t - 0.5)
    elif ptype == 'quadratic':
        t = np.linspace(0.0, 1.0, n)
        u = 2.0 * t - 1.0
        a = 1.0 + param * u * u
    elif ptype == 'cubic':
        t = np.linspace(0.0, 1.0, n)
        u = 2.0 * t - 1.0
        a = 1.0 + param * u + 0.3 * param * u ** 3
    elif ptype == 'cosine':
        t = np.arange(n)
        a = 1.0 + param * np.cos(2.0 * np.pi * t / n)
    elif ptype == 'ramp':
        t = np.linspace(0.0, 1.0, n)
        a = 0.1 + param * t
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))
