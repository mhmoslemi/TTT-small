import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _gradient_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _objective_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_obj(z, beta, n):
    return _objective_smooth(z, beta)


def _lbfgs_grad(z, beta, n):
    return _gradient_smooth(z, beta)


def _run_lbfgs(a_init, deadline, max_wall_time=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if max_wall_time is None:
        max_wall_time = min(300.0, max(1.0, deadline - time.monotonic()))

    betas = [0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008, 0.0004,
             0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]
    betas = [b * max_c0 for b in betas]

    per_level = max_wall_time / len(betas)

    for idx_beta, beta in enumerate(betas):
        if time.monotonic() >= deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level * (1 + 0.3 * idx_beta))
        try:
            result = minimize(
                _lbfgs_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_grad,
                options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 40, 'maxfun': 2000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x.copy()
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new
        except Exception:
            pass

    return best_a, best_score


def _subgradient_primal(a, deadline, n_iter=50000, tol_active=1e-6, use_bb=True):
    """
    Subgradient method in primal space with exact active set.
    Works with a >= 0, normalizes to sum = sqrt(2n).
    Uses Barzilai-Borwein step size or fixed decreasing schedule.
    """
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    a = a.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n * target_sum
    else:
        a = a / s * target_sum

    best_a = a.copy()
    best_score = _score_norm_fn(a)

    lr_init = target_sum * 0.01
    lr = lr_init
    momentum = np.zeros(n)
    mom_coeff = 0.7

    prev_a = a.copy()
    prev_grad = None

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_score = 2.0 * n * c_max / (target_sum ** 2)

        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()

        # Active set: lags within tolerance of max
        threshold = c_max - max(tol_active * c_max, 1e-10)
        active_mask = c >= threshold
        n_active = int(active_mask.sum())
        if n_active == 0:
            active_mask[int(np.argmax(c))] = True
            n_active = 1

        # Compute subgradient: average of grad_k for active k
        # grad_k(a)[i] = 2*a[k-i] if max(0,k-n+1) <= i <= min(n-1,k), else 0
        # This can be computed efficiently using the convolution structure
        # grad = 2 * sum_{k in active} a_rev_shifted[k]
        # Where a_rev_shifted[k][i] = a[k-i] if valid, else 0

        # Efficient computation: grad[i] = 2 * sum_{k in active, |k-i| < n} a[k-i] * indicator
        # = 2 * sum_{j: k=i+j in active, 0<=j<n} a[j]
        # = 2 * (a * w_active)[i] where w_active is the active mask convolved appropriately

        # Actually: grad[i] = 2 * sum_{k in A} a[k-i] (if 0 <= k-i < n)
        # = 2 * sum_{j: i+j in A, 0 <= j < n} a[j]
        # This is 2 * (a * w_A_flipped)[n-1] ... let me think more carefully.

        # grad_k(a)[i] = 2*a[k-i] if 0 <= k-i < n (and i in valid range for k)
        # Sum over k in A: grad[i] = 2 * sum_{k in A, 0 <= k-i < n} a[k-i]
        # Let j = k-i, so k = i+j. grad[i] = 2 * sum_{j: 0<=j<n, i+j in A} a[j]
        # = 2 * sum_{j=0}^{n-1} a[j] * w_A[i+j]
        # where w_A is the active mask extended to length 2n-1.

        # This is a cross-correlation: grad[i] = 2 * sum_j a[j] * w_A[i+j]
        # = 2 * (full_convolve(a, w_A_reversed))[i + n - 1] ... 

        # Simpler: just loop over active k values (there shouldn't be too many)
        active_indices = np.where(active_mask)[0]
        if len(active_indices) > 200:
            # Too many active, subsample
            active_indices = active_indices[np.linspace(0, len(active_indices)-1, 200, dtype=int)]

        grad = np.zeros(n)
        for k in active_indices:
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            # grad[lo:hi] += 2 * a[k-lo : k-hi+1][::-1]
            # For i in [lo, hi-1]: grad[i] += 2 * a[k-i]
            # a[k-i] for i in [lo, hi-1] gives a[k-lo], a[k-lo-1], ..., a[k-hi+1]
            # which is a[k-lo : k-hi+1 : -1] = a[lo:hi][::-1] reversed... 
            # Let's just do: for i in range(lo, hi): grad[i] += 2*a[k-i]
            # Vectorized: grad[lo:hi] += 2 * a[k-hi+1:k-lo+1][::-1]
            # a[k-i] for i=lo: a[k-lo]; for i=hi-1: a[k-hi+1]
            # So the sequence is a[k-lo], a[k-lo-1], ..., a[k-hi+1]
            # = a[k-hi+1:k-lo+1][::-1]
            seg = a[k - hi + 1: k - lo + 1][::-1]
            grad[lo:hi] += 2.0 * seg

        grad /= n_active

        # Barzilai-Borwein step size
        if use_bb and prev_grad is not None:
            s_vec = a - prev_a
            y_vec = grad - prev_grad
            s_norm2 = float(np.dot(s_vec, s_vec))
            y_norm2 = float(np.dot(y_vec, y_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm2 > 1e-20 and y_norm2 > 1e-20:
                # BB+ (use max of the two BB steps)
                bb1 = s_norm2 / sy if sy > 1e-20 else lr
                bb2 = sy / y_norm2 if y_norm2 > 1e-20 else lr
                lr_new = max(1e-8, min(10.0 * lr_init, max(bb1, bb2)))
                lr = 0.9 * lr + 0.1 * lr_new  # Smoothed
            else:
                lr *= 0.95

        # Momentum step
        momentum = mom_coeff * momentum - lr * grad
        a_new = a + momentum

        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)

        # Normalize
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) / n * target_sum
        else:
            a_new = a_new / s_new * target_sum

        prev_a = a.copy()
        prev_grad = grad.copy()
        a = a_new

        # Slow down momentum over time
        if it % 5000 == 0:
            mom_coeff = min(0.95, mom_coeff + 0.01)

        # Periodic evaluation
        if it % 100 == 0:
            c_check = _full_convolve(a, a)
            sc = 2.0 * n * c_check.max() / (target_sum ** 2)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        # Decrease tolerance periodically
        if it % 5000 == 0:
            tol_active *= 0.8

    # Final check
    c_final = _full_convolve(a, a)
    sc_final = 2.0 * n * c_final.max() / (target_sum ** 2)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score


def _coordinate_descent(a, deadline, n_passes=20, step_frac=0.05):
    """
    Coordinate descent: for each element, try increasing/decreasing by small amounts.
    Works in primal space with normalization.
    """
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    a = a.copy()
    s = a.sum()
    a = a / s * target_sum

    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for pass_idx in range(n_passes):
        if time.monotonic() > deadline:
            break

        # Randomize order
        perm = np.random.permutation(n)

        for i in perm:
            if time.monotonic() > deadline:
                break

            orig_val = a[i]
            if orig_val < 1e-10:
                continue

            # Try several perturbations
            best_local = orig_val
            best_local_score = _full_convolve(a, a).max()

            for frac in [step_frac, -step_frac, step_frac/3, -step_frac/3,
                         step_frac/10, -step_frac/10]:
                a[i] = max(0.0, orig_val * (1.0 + frac))
                s = a.sum()
                if s < 1e-15:
                    a[i] = orig_val
                    break
                a_norm = a / s * target_sum
                c_max = _full_convolve(a_norm, a_norm).max()
                if c_max < best_local_score:
                    best_local_score = c_max
                    best_local = a[i]
            a[i] = best_local

        # Re-normalize after full pass
        s = a.sum()
        a = a / s * target_sum
        sc = _score_norm_fn(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Reduce step size
        step_frac *= 0.7

    return best_a, best_score


def _nesterov_primal(a, deadline, n_max_iter=100000):
    """
    Nesterov-accelerated subgradient in primal space.
    """
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    a = a.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    a = a / s * target_sum

    y = a.copy()  # Nesterov lookahead
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    lr = target_sum * 0.005
    mom = 0.8

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(y, y)
        c_max = float(c.max())
        threshold = c_max - max(1e-4 * c_max, 1e-10)
        active_mask = c >= threshold
        if not np.any(active_mask):
            active_mask[int(np.argmax(c))] = True

        active_indices = np.where(active_mask)[0]
        if len(active_indices) > 300:
            active_indices = active_indices[np.linspace(0, len(active_indices)-1, 300, dtype=int)]

        grad = np.zeros(n)
        for k in active_indices:
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            seg = y[k - hi + 1: k - lo + 1][::-1]
            grad[lo:hi] += 2.0 * seg
        grad /= max(1, len(active_indices))

        # Nesterov step
        a_new = y - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new * target_sum

        t_val = (1.0 + np.sqrt(1.0 + 4.0 * mom * mom)) / 2.0
        y = a_new + mom * (a_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a_new.copy()
        else:
            y = y / s_y * target_sum

        a = a_new

        if it % 200 == 0:
            c_check = _full_convolve(a, a)
            sc = 2.0 * n * c_check.max() / (target_sum ** 2)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if it % 5000 == 0:
            lr *= 0.9
            mom = min(0.99, mom + 0.01)

    c_final = _full_convolve(a, a)
    sc_final = 2.0 * n * c_final.max() / (target_sum ** 2)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, lr=0.03, n_max_iter=100000):
    """Adam optimizer in log-space with smooth approximation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, 0.01 * max_c)

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a = _softmax_fn(z)

        if t % 200 == 0:
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
            beta = max(1e-14, beta * 0.85)
            c_new = _full_convolve(a, a)
            max_c = float(c_new.max())

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_exp_profile(n, rate):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(rate * t)
    return a / a.sum()


def _make_power_profile(n, alpha):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 0.1
    return a / a.sum()


def _normalize_to_target(a, target_sum):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a) * target_sum
    return a / s * target_sum


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid optimizer: parametric init + L-BFGS continuation + primal subgradient
    + coordinate descent + final L-BFGS polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print("  [{:.1f}s] NEW BEST: {:.10f} (n={}) [{}]".format(elapsed, best_score, len(p), label))

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        print("Parent: n={}, score={:.10f}".format(n, _score_fn(parent)))
    else:
        n = 4096
        parent = _make_linear_profile(n, 2.0)
        print("No parent, using linear profile n={}".format(n))

    # --- Phase 0: Multiple initializations (30s) ---
    phase0_end = min(deadline, t_start + 30.0)
    candidates = []

    sc_parent = _score_norm_fn(parent)
    candidates.append((sc_parent, parent.copy(), "parent"))
    update_best(parent, sc_parent, "parent")

    if parent is not None:
        n = len(parent)

    profiles = [
        _make_linear_profile(n, 1.0),
        _make_linear_profile(n, 2.0),
        _make_linear_profile(n, 3.0),
        _make_quadratic_profile(n, 1.5, 0.3),
        _make_quadratic_profile(n, 2.0, -0.2),
        _make_exp_profile(n, 2.0),
        _make_exp_profile(n, 4.0),
        _make_power_profile(n, 1.5),
        _make_power_profile(n, 2.0),
        _make_power_profile(n, 0.5),
    ]

    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        # Pad or truncate to match n
        if len(p) != n:
            from numpy import interp
            x0 = np.linspace(0, 1, len(p))
            x1 = np.linspace(0, 1, n)
            p = np.interp(x1, x0, p)
            p = np.maximum(p, 1e-12)
            p = p / p.sum()
        sc = _score_norm_fn(p)
        candidates.append((sc, p, "prof{}".format(idx)))

    candidates.sort(key=lambda x: x[0])
    print("Phase 0 done: best={:.10f}".format(best_score))

    # --- Phase 1: L-BFGS continuation on top candidates (350s) ---
    phase1_end = min(deadline, t_start + 380.0)
    if best_seq is not None:
        per_time = min(200.0, phase1_end - time.monotonic() - 10)
        if per_time > 20:
            p_best, sc_best = _run_lbfgs(best_seq, phase1_end, max_wall_time=per_time)
            update_best(p_best, sc_best, "lbfgs_p1")

    # Try second best if time allows
    if time.monotonic() < phase1_end - 60:
        for sc, p, label in candidates[1:3]:
            if time.monotonic() > phase1_end - 10:
                break
            per_time = min(80.0, phase1_end - time.monotonic() - 5)
            if per_time > 15:
                p_best, sc_best = _run_lbfgs(p, min(phase1_end - 5, time.monotonic() + per_time),
                                              max_wall_time=per_time)
                update_best(p_best, sc_best, "lbfgs_{}".format(label))

    print("Phase 1 done: best={:.10f}".format(best_score))

    # --- Phase 2: Primal subgradient with BB step (400s) ---
    phase2_end = min(deadline, t_start + 780.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        target_sum = np.sqrt(2.0 * len(best_seq))
        a_init = _normalize_to_target(best_seq, target_sum)
        per_time = phase2_end - time.monotonic() - 5
        if per_time > 30:
            p_best, sc_best = _subgradient_primal(
                a_init, min(phase2_end - 5, time.monotonic() + per_time),
                n_iter=200000, tol_active=1e-5, use_bb=True
            )
            # Convert back: score_norm = 2n * max_c (since sum = sqrt(2n), score = max_c)
            # But _score_norm_fn computes 2n * max_c, and the actual score = 2n*max_c/S^2 = max_c when S=sqrt(2n)
            # So sc_best from _subgradient_primal is already the normalized score
            update_best(p_best, sc_best, "subgrad_p2")

    print("Phase 2 done: best={:.10f}".format(best_score))

    # --- Phase 3: Nesterov primal (250s) ---
    phase3_end = min(deadline, t_start + 1030.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        target_sum = np.sqrt(2.0 * len(best_seq))
        a_init = _normalize_to_target(best_seq, target_sum)
        per_time = phase3_end - time.monotonic() - 5
        if per_time > 20:
            p_best, sc_best = _nesterov_primal(
                a_init, min(phase3_end - 5, time.monotonic() + per_time),
                n_max_iter=100000
            )
            update_best(p_best, sc_best, "nesterov_p3")

    print("Phase 3 done: best={:.10f}".format(best_score))

    # --- Phase 4: L-BFGS refinement with tiny beta (250s) ---
    phase4_end = min(deadline, t_start + 1280.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        per_time = min(150.0, phase4_end - time.monotonic() - 10)
        if per_time > 10:
            p_best, sc_best = _run_lbfgs(
                best_seq, min(phase4_end - 5, time.monotonic() + per_time),
                max_wall_time=per_time
            )
            update_best(p_best, sc_best, "lbfgs_p4")

    # --- Phase 5: Coordinate descent (200s) ---
    phase5_end = min(deadline, t_start + 1480.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        target_sum = np.sqrt(2.0 * len(best_seq))
        a_init = _normalize_to_target(best_seq, target_sum)
        per_time = phase5_end - time.monotonic() - 5
        if per_time > 10:
            p_best, sc_best = _coordinate_descent(
                a_init, min(phase5_end - 5, time.monotonic() + per_time),
                n_passes=50, step_frac=0.03
            )
            update_best(p_best, sc_best, "coord_p5")

    print("Phase 5 done: best={:.10f}".format(best_score))

    # --- Phase 6: Adam in log-space (150s) ---
    phase6_end = min(deadline, t_start + 1630.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        per_time = phase6_end - time.monotonic() - 5
        if per_time > 10:
            p_best, sc_best = _adam_logspace(
                best_seq, min(phase6_end - 5, time.monotonic() + per_time),
                lr=0.02, n_max_iter=100000
            )
            update_best(p_best, sc_best, "adam_p6")

    # --- Phase 7: Final L-BFGS + subgradient polish (remaining) ---
    phase7_end = deadline
    if best_seq is not None and time.monotonic() < phase7_end:
        remaining = phase7_end - time.monotonic()
        if remaining > 20:
            # Final L-BFGS
            per_time = min(50.0, remaining - 10)
            if per_time > 5:
                p_best, sc_best = _run_lbfgs(
                    best_seq, min(phase7_end - 10, time.monotonic() + per_time),
                    max_wall_time=per_time
                )
                update_best(p_best, sc_best, "final_lbfgs")

            # Final subgradient
            if time.monotonic() < phase7_end - 10:
                target_sum = np.sqrt(2.0 * len(best_seq))
                a_init = _normalize_to_target(best_seq, target_sum)
                per_time2 = phase7_end - time.monotonic() - 5
                if per_time2 > 5:
                    p_best, sc_best = _subgradient_primal(
                        a_init, min(phase7_end - 3, time.monotonic() + per_time2),
                        n_iter=100000, tol_active=1e-7, use_bb=False
                    )
                    update_best(p_best, sc_best, "final_subgrad")

    print("Phase 7 done: best={:.10f}".format(best_score))

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_linear_profile(256, 2.0)

        sc_final = _score_fn(final_seq)
        print("FINAL: score={:.10f}, n={}".format(sc_final, len(final_seq)))

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print("Evaluator score: {:.10f}".format(sc_eval))
            else:
                print("Evaluator returned inf/NaN, using internal: {:.10f}".format(sc_final))
        except Exception as ex:
            print("evaluate_sequence failed: {}".format(ex))

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_linear_profile(4096, 2.0)
    print("FALLBACK: n=4096, score={:.10f}".format(_score_fn(fallback)))
    return [float(x) for x in fallback]
