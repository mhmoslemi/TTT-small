    s = float(ex.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s
    sm_val = mval + beta * float(np.log(s))
    
    # Gradient: g_i = 2 * sum_k w_k * a_{k-i}
    g = 2.0 * _conv_cross_corr(w, a)
    
    # Project to tangent space of simplex (logit space)
    dot = float(np.dot(a, g))
    grad = 2.0 * n * a * (g - dot)
    
    return 2.0 * n * sm_val, grad


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _lbfgs_smooth_max(z0, beta, n, maxiter=500):
    """Run L-BFGS-B on the smooth-max objective."""
    a0 = _softmax(z0)
    
    def obj_and_grad(z):
        z_clipped = np.clip(z, -50, 50)
        val, grad = _smooth_max_obj_and_grad(z_clipped, beta, n)
        if not np.isfinite(val):
            return 1e10, np.zeros_like(z)
        return val, grad
    
    res = minimize(obj_and_grad, z0, method='L-BFGS-B',
                   options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8,
                            'maxls': 50})
    return np.clip(res.x, -50, 50), res


def _lbfgs_continuation(a0, deadline, beta_start=0.15, beta_end=0.01,
                         n_steps=8, n=None):
    """L-BFGS-B with decreasing beta (continuation method)."""
    if n is None:
        n = len(a0)
    z = np.log(np.maximum(a0, 1e-30))
    z -= float(np.mean(z))
    z = np.clip(z, -50, 50)
    
    best_a = a0.copy()
    best_score = _score_norm(a0)
    
    betas = np.geomspace(beta_start, beta_end, n_steps)
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        print(f"  L-BFGS-B: beta={beta:.4f}")
        z, res = _lbfgs_smooth_max(z, float(beta), n, maxiter=300)
        a = _softmax(z)
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        print(f"    score={sc:.10f}, success={res.success}")
    
    return best_a, best_score


def _adam_optimize(a0, deadline, beta_frac=0.06, lr=0.04, max_no_imp=150, rng=None):
    """Adam optimizer on logit parameterization with smooth-max."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-30))
    z -= float(np.mean(z))
    z = np.clip(z, -50, 50)
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    a = a0.copy()
    c = _full_convolve(a)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)
    
    best_a = a0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    
    while time.monotonic() < deadline:
        mval = float(c.max())
        x = (c - mval) / beta
        ex = np.exp(np.clip(x, -50, 50))
        s = float(ex.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s
        
        g = 2.0 * _conv_cross_corr(w, a)
        dot = float(np.dot(a, g))
        grad = a * (g - dot)
        grad -= float(np.mean(grad))
        
        if not np.all(np.isfinite(grad)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            continue
        
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= float(np.mean(z))
        a = _softmax(z)
        
        c = _full_convolve(a)
        maxc = float(c.max())
        score = 2.0 * n * maxc
        
        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.001, lr * 0.93)
        
        if t % 50 == 0:
            beta = max(1e-5, beta * 0.94)
        
        if no_imp > max_no_imp:
            ns = 0.05 if best_score < 1.51 else 0.1
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= float(np.mean(z))
            a = _softmax(z)
            c = _full_convolve(a)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            lr = 0.04
            no_imp = 0
    
    return best_a, best_score


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a, deadline, t_init=0.35, max_solves=30):
    """LP-based refinement."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    t = t_init
    n_solves = 0
    
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(a, M)
        if b is None:
            b = _lp_solve(a, M * 1.005)
            if b is None:
                break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)
        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()
        a = a_new
        c = _full_convolve(a)
        M = max(float(c.max()) * 0.999, best_score / (2.0 * n) * 1.001)
        t = max(0.1, t * 0.93)
    
    return best_a, best_score


def _peak_flood(a, deadline, eta=0.03):
    """Multiplicative peak flooding."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    
    for it in range(3000):
        if time.monotonic() > deadline:
            break
        M = float(c.max())
        score = 2.0 * n * M
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
        
        threshold = M * 0.95
        hot_mask = c >= threshold
        if not hot_mask.any():
            hot_mask[int(np.argmax(c))] = True
        
        hot_indices = np.where(hot_mask)[0]
        heat = np.zeros(n)
        for k in hot_indices:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                partner = k - np.arange(i_start, i_end + 1)
                heat[i_start:i_end + 1] += a[partner]
        
        heat_max = float(heat.max())
        if heat_max < 1e-15:
            break
        heat = heat / heat_max
        factor = np.maximum(1.0 - eta * heat, 0.01)
        a = a * factor
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        c = _full_convolve(a)
        
        if it % 200 == 0 and it > 0 and score >= best_score:
            eta = max(0.005, eta * 0.9)
    
    return best_a, best_score

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-300:
        return np.ones_like(e) / e.size
    return e / s


def _conv_cross_corr(w, p):
    """Compute cross-correlation h[i] = sum_k w[k] * p[k-i] for i=0..n-1."""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

    c = _full_convolve(a)

    ex = np.exp(np.clip(x, -50.0, 50.0))

        s = float(ex.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s

        h = _conv_cross_corr(w, a)
        dot_h = float(np.dot(a, h))
        grad = a * (h - dot_h)
        grad = grad - float(np.mean(grad))

        if not np.all(np.isfinite(grad)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            continue

        t += 1
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z = z - float(np.mean(z))
        a = _softmax(z)

        c = _full_convolve(a)
        maxc = float(c.max())
        score = 2.0 * n * maxc

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 80 == 0:
                lr = max(0.001, lr * 0.92)

        if t % 40 == 0:
            beta = max(1e-5, beta * 0.93)

        if no_imp > max_no_imp:
            ns = 0.03 if best_score < 1.505 else 0.06
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z = z - float(np.mean(z))
            a = _softmax(z)
            c = _full_convolve(a)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            lr = 0.04
            no_imp = 0

    return best_a, best_score


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a, deadline, t_init=0.35, max_solves=30, shrink=0.998):
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    t = t_init

    for _ in range(max_solves):
        if time.monotonic() > deadline:
            break
        b = _lp_solve(a, M)
        if b is None:
            b = _lp_solve(a, M * 1.002)
            if b is None:
                break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)
        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()
        a = a_new
        c = _full_convolve(a)
        M = max(float(c.max()) * shrink, best_score / (2.0 * n) * 1.001)
        t = max(0.08, t * 0.92)

    return best_a, best_score


def _peak_flood(a, deadline, eta=0.03):
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)

    for it in range(3000):
        if time.monotonic() > deadline:
            break
        M = float(c.max())
        score = 2.0 * n * M
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        threshold = M * 0.94
        hot_mask = c >= threshold
        if not hot_mask.any():
            hot_mask[int(np.argmax(c))] = True

        hot_indices = np.where(hot_mask)[0]
        heat = np.zeros(n)
        for k in hot_indices:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                partner = k - np.arange(i_start, i_end + 1)
                heat[i_start:i_end + 1] += a[partner]

        heat_max = float(heat.max())
        if heat_max < 1e-15:
            break
        heat = heat / heat_max
        factor = np.maximum(1.0 - eta * heat, 0.01)
        a = a * factor
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        c = _full_convolve(a)

        if it % 150 == 0 and it > 0 and score >= best_score:
            eta = max(0.005, eta * 0.88)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, label=""):
        nonlocal best_seq, best_score
        if p is not None and p.size > 0:
            p = np.asarray(p, dtype=np.float64).ravel()
            p = np.maximum(p, 0.0)
            s = float(p.sum())
            if s > 0.01:
                p = p / s
            else:
                return
            sc = _score_norm(p)
            if np.isfinite(sc) and sc < best_score:
                best_seq = p.copy()
                best_score = float(sc)
                print(f"  >> BEST: {best_score:.10f} n={p.size} [{label}]")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    if parent is not None:
        ps = _score_norm(parent)
        print(f"Parent score: {ps:.10f}")
        _update(parent, 'parent')

    if best_seq is None:
        n = 1024
        t_arr = np.linspace(0.0, 1.0, n)
        best_seq = 1.0 + 2.0 * (t_arr - 0.5)
        best_seq = np.maximum(best_seq, 1e-10)
        best_seq = best_seq / best_seq.sum()
        best_score = _score_norm(best_seq)
        print(f"Initialized linear profile, score={best_score:.10f}")

    # Phase 1: Quick Adam (50s)
    p1_end = t_start + 50.0
    print("Phase 1: Adam warmup...")
    p1, sc1 = _adam_optimize(best_seq, p1_end, beta_frac=0.05, lr=0.05, max_no_imp=80, rng=rng)
    _update(p1, 'adam1')

    # Phase 2: L-BFGS-B continuation (350s)
    p2_end = t_start + 400.0
    print(f"Phase 2: L-BFGS-B continuation, score={best_score:.10f}...")
    p2, sc2 = _lbfgs_continuation(best_seq, p2_end, beta_start=0.1, beta_end=0.005, n_steps=10)
    _update(p2, 'lbfgs_cont')

    # Phase 3: Peak flooding (150s)
    p3_end = t_start + 550.0
    print(f"Phase 3: Peak flooding, score={best_score:.10f}...")
    p3, sc3 = _peak_flood(best_seq, p3_end, eta=0.025)
    _update(p3, 'flood1')

    # Phase 4: Multi-restart (300s)
    p4_end = t_start + 850.0
    print(f"Phase 4: Multi-restart, score={best_score:.10f}...")
    n_restarts = 8
    per_r = (p4_end - time.monotonic()) / n_restarts
    for r in range(n_restarts):
        if time.monotonic() > p4_end - 40:
            break
        ns = 0.02 + 0.015 * r
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - float(np.mean(z))
        z_n = z + ns * rng.normal(size=len(z))
        z_n = np.clip(z_n, -50.0, 50.0)
        z_n = z_n - float(np.mean(z_n))
        a_n = _softmax(z_n)
        per = min(per_r * 0.7, p4_end - time.monotonic() - 10)
        if per < 10:
            break
        p4a, _ = _adam_optimize(a_n, time.monotonic() + per, beta_frac=0.04, lr=0.03, max_no_imp=60, rng=rng)
        _update(p4a, f'r{r}_adam')
        if time.monotonic() < p4_end - 20:
            per2 = min(per_r * 0.3, p4_end - time.monotonic() - 5)
            if per2 > 10:
                p4b, _ = _lbfgs_continuation(best_seq, time.monotonic() + per2,
                                              beta_start=0.04, beta_end=0.005, n_steps=4)
                _update(p4b, f'r{r}_lbfgs')

    # Phase 5: LP refinement (250s)
    p5_end = t_start + 1100.0
    print(f"Phase 5: LP refinement, score={best_score:.10f}...")
    if best_seq.size <= 2048:
        p5, sc5 = _lp_refine(best_seq, p5_end, t_init=0.3, max_solves=25, shrink=0.997)
        _update(p5, 'lp1')

    # Phase 6: Second L-BFGS pass (100s)
    p6_end = t_start + 1200.0
    print(f"Phase 6: L-BFGS-B round 2, score={best_score:.10f}...")
    p6, sc6 = _lbfgs_continuation(best_seq, p6_end, beta_start=0.03, beta_end=0.003, n_steps=5)
    _update(p6, 'lbfgs2')

    # Phase 7: Final polish (remaining)
    p7_end = deadline
    print(f"Phase 7: Final polish, score={best_score:.10f}...")
    if time.monotonic() < p7_end - 20:
        per = min(60.0, p7_end - time.monotonic() - 30)
        if per > 20:
            p7a, _ = _adam_optimize(best_seq, time.monotonic() + per,
                                     beta_frac=0.015, lr=0.01, max_no_imp=50, rng=rng)
            _update(p7a, 'final_adam')

    if time.monotonic() < p7_end - 15 and best_seq.size <= 2048:
        p7b, _ = _lp_refine(best_seq, p7_end, t_init=0.12, max_solves=10, shrink=0.999)
        _update(p7b, 'final_lp')

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            n = 1024
            t_arr = np.linspace(0.0, 1.0, n)
            final = 1.0 + 2.0 * (t_arr - 0.5)
            final = np.maximum(final, 1e-10)
            final = final / final.sum()

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    # Fallback
    n = 1024
    t_arr = np.linspace(0.0, 1.0, n)
    fb = 1.0 + 2.0 * (t_arr - 0.5)
    fb = np.maximum(fb, 1e-10)
    fb = fb / fb.sum()
    print(f"FALLBACK n={n}, score={_score_norm(fb):.10f}")
    return [float(x) for x in fb]
