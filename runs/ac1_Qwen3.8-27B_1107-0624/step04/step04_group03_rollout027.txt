    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    n, m = len(a), len(b)

LP-based convex optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses iterative LP refinement where at each step, we solve a linear program
to find the best "direction" that reduces the peak of the self-convolution.
Combined with smooth optimization (L-BFGS with softmax) for final polishing.
"""
import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm_fn(a):
    """Assumes sum(a) = 1, returns 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _build_conv_matrix(f):
    """Build matrix C where (C@g)[k] = (np.convolve(f,g))[k]"""
    n = len(f)
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        f_start = k - j_end + 1
        f_end = k - j_start
        C[k, j_start:j_end] = f[f_start:f_end+1][::-1]
    return C

def _solve_lp(f, M, timeout=30):
    """
    Solve: max sum(g) s.t. g >= 0, convolve(f, g) <= M
    """
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    C = _build_conv_matrix(f)
    constraints = [C @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None:
                return np.maximum(g.value, 0)
        except Exception:
            continue
    return None

def _lp_refine(a, deadline, max_iters=200):
    """
    Iterative LP refinement.
    a: initial sequence with sum(a) = 1
    Returns refined sequence.
    """
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    t = time.monotonic()
    M = float(_full_convolve(a, a).max())
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g_star = _solve_lp(a, M)
        if g_star is None:
            print(f"  LP failed at iter {it}")
            break
        
        S = float(g_star.sum())
        if S < 1.0 + 1e-6:
            # No improvement possible
            print(f"  LP gave S={S:.6f} <= 1, stopping at iter {it}")
            break
        
        g_norm = g_star / S
        
        # Try multiple step sizes
        best_t_score = best_score
        best_t_a = best_a.copy()
        
        for t_val in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t_val) * a + t_val * g_norm
            sc = _score_norm_fn(a_try)
            if sc < best_t_score:
                best_t_score = sc
                best_t_a = a_try.copy()
                best_t = t_val
        
        if best_t_score >= best_score:
            # No improvement
            break
        
        a = best_t_a
        best_a = a.copy()
        best_score = best_t_score
        M = float(_full_convolve(a, a).max())
        
        elapsed = time.monotonic() - t
        print(f"  LP iter {it}: score={best_score:.10f}, S={S:.4f}, t={best_t:.2f}, time={elapsed:.1f}s")
        
        if best_score < 1.5027:
            print("  Target reached!")
            break
    
    return best_a, best_score

def _compute_weights(c, beta):

def _gradient_smooth(z, beta):

    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]

    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad
