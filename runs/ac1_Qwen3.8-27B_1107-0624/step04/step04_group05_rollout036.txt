"""
Optimized step function search using aggressive LP best-response with 
over-relaxation, subgradient polishing, multiplicative weights update,
and multi-scale escalation. Targets score < 1.5027.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c

def _score(p):
    """Compute the score for a normalized (sum=1) non-negative sequence."""
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())

def _build_conv_matrix_sparse(a):
    """Build sparse matrix A such that (A @ b)[k] = (a*b)[k] for linear convolution."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def _lp_best_response(a, M):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""
    n = len(a)
    A = _build_conv_matrix_sparse(a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

    return None

def _lp_best_response_tight(a, M, tight_indices):
    """Solve LP with only tight constraints (faster)."""
    n = len(a)
    k = len(tight_indices)
    if k == 0:
        # No constraints, just maximize sum with b >= 0, unbounded
        # Return uniform
        return np.ones(n)
    A_full = _build_conv_matrix_sparse(a)

    A = A_full[tight_indices, :]

    return None

def _project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / w.sum()

def _smooth_max(c, beta):
    """Compute smooth max and weights."""
    c = np.asarray(c, dtype=np.float64).ravel()
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 0.0))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w

def _cross_corr(w, p):
    """Compute cross-correlation of w and p (for gradient computation)."""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def _subgradient_step(p, lr, beta):
    """One step of subgradient descent on smooth max of autoconv."""
    n = len(p)
    c = _full_convolve(p)
    _, w = _smooth_max(c, beta)
    g = _cross_corr(w, p)  # gradient of smooth max w.r.t. p
    # Constrained gradient: remove mean (for sum constraint)
    g = g - float(np.mean(g))
    # Update
    p_new = p - lr * g
    p_new = _project_simplex(p_new)
    return p_new

def _multiplicative_weights_step(p, lr, beta):
    """Multiplicative weights update."""
    n = len(p)
    c = _full_convolve(p)
    _, w = _smooth_max(c, beta)
    g = _cross_corr(w, p)
    # Mirror descent in entropic regularization
    log_p = np.log(np.maximum(p, 1e-300))
    log_p_new = log_p - lr * (g - float(np.mean(g)))
    log_p_new -= float(np.mean(log_p_new))
    p_new = np.exp(log_p_new)
    p_new = p_new / p_new.sum()
    p_new = np.maximum(p_new, 1e-300)
    p_new = p_new / p_new.sum()
    return p_new

def _interp_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    if s > 0:
        p1 = p1 / s
    return p1

"""
Optimized step function search combining LP best-response with constraint 
subsetting, adaptive step sizes, deconvolution polishing, multi-scale 
escalation, and aggressive restarts. Targets score < 1.5027.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c


def _score(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_solve_full(a, M):
    n = len(a)
    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_solve_subset(a, M, constraint_indices):
    n = len(a)
    A_full = _build_conv_matrix(a)
    A = A_full[constraint_indices, :]
    c_obj = -np.ones(n)
    b_ub = M * np.ones(len(constraint_indices))
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    mask = u * rho_vals > (cssv - 1)
    if not mask.any():
        return np.ones(n) / n
    rho = np.nonzero(mask)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 0.0))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s


def _cross_corr_signals(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _gradient_step(p, lr, beta, method='subgrad'):
    n = len(p)
    c = _full_convolve(p)
    w = _smooth_max_weights(c, beta)
    g = _cross_corr_signals(w, p)
    g = g - float(np.mean(g))
    
    if method == 'subgrad':
        p_new = p - lr * g
        p_new = _project_simplex(p_new)
    elif method == 'mirror':
        log_p = np.log(np.maximum(p, 1e-300))
        log_p = log_p - lr * g
        log_p = log_p - float(np.mean(log_p))
        p_new = np.exp(np.clip(log_p, -50, 50))
        s = p_new.sum()
        if s < 1e-30:
            p_new = np.ones(n) / n
        else:
            p_new = p_new / s
    else:
        p_new = p
    return p_new


def _interp_to_length(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    if s > 0:
        p1 = p1 / s
    else:
        p1 = np.ones(n_new) / n_new
    return p1
