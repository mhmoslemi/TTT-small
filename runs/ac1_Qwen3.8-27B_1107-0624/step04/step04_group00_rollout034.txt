import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _md_with_lse(a_init, deadline, rng, beta_init_frac=0.1, lr_init=0.05,

def _interp_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 1e-15)
    return a_new / a_new.sum()


def _make_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    
    # Uniform
    profiles.append(np.ones(n) / n)
    
    # Linear ramps (asymmetric)
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        for direction in [1, -1]:
            a = 1.0 + direction * slope * (t - 0.5)
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())
    
    # Quadratic (asymmetric)
    for a1, a2 in [(1.0, 0.5), (2.0, -0.3), (-1.5, 0.3), (3.0, -1.0), (-3.0, -1.0)]:
        a = 1.0 + a1 * u + a2 * u ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    
    # Exponential
    for rate in [1.0, 2.0, 3.0, 5.0, -1.0, -2.0, -3.0, -5.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    
    # Power law
    for p in [0.5, 2, 3, 5]:
        a = np.maximum(t, 1e-12) ** p
        profiles.append(a / a.sum())
        a = np.maximum(1 - t, 1e-12) ** p
        profiles.append(a / a.sum())
    
    # Bimodal
    for center in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for width in [0.1, 0.2, 0.3]:
            a = np.exp(-((t - center) / width) ** 2)
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())
    
    return profiles
