def _compute_grad(c_weights, a):
    """Compute gradient of sum_k c_weights[k]*conv(a,a)[k] w.r.t. a"""
    n = len(a)
    # grad[j] = sum_k c_weights[k] * (a[k-j] + a[j-k])
    # = conv(c_weights, a)[j] + cross_corr(c_weights, a)[j]
    # where conv is full linear convolution (take first n)
    # and cross_corr[j] = sum_m c_weights[m+j] * a[m]
    
    # Full convolution of c_weights (len 2n-1) and a (len n): length 3n-2
    N = _next_pow2(3*n - 2)
    fcw = np.fft.rfft(c_weights, N)
    fa = np.fft.rfft(a, N)
    full_conv = np.fft.irfft(fcw * fa, N)
    
    # T2[j] = full_conv[j] for j=0..n-1
    T2 = full_conv[:n]
    
    # T1[j] = cross_corr(c_weights, a)[j] = sum_m c_weights[m+j] * a[m]
    # = full_conv(c_weights_rev, a)[j + 2n-2]
    cw_rev = c_weights[::-1]
    fcw_r = np.fft.rfft(cw_rev, N)
    full_conv_r = np.fft.irfft(fcw_r * fa, N)
    T1 = full_conv_r[2*n-2 : 2*n-2+n]
    
    return T1 + T2
