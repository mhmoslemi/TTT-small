Let me analyze this problem carefully.

The objective is to minimize:
$$\frac{2n \cdot \max(\text{conv}(a, a))}{(\sum a)^2}$$

where $a = (a_0, \ldots, a_{n-1})$ is a non-negative sequence.

Key observations:

1. The objective is scale-invariant. If we multiply $a$ by $c > 0$, the numerator scales by $c^2$ and the denominator scales by $c^2$, so the ratio is unchanged. This means we can normalize $\sum a = 1$ or $\sum a = \sqrt{2n}$ or anything else.

2. For a uniform sequence $a_i = 1/n$ (so $\sum a = 1$):
   - $\text{conv}(a,a)[k] = \sum_{i} a_i a_{k-i}$
   - For $k = 0$: $a_0^2 = 1/n^2$
   - For $k = n-1$: $n \cdot (1/n)^2 = 1/n$
   - For $k = 2n-2$: $a_{n-1}^2 = 1/n^2$
   - So $\max \text{conv} = 1/n$
   - Score = $2n \cdot (1/n) / 1 = 2$

So the uniform sequence gives score = 2. The previous best was 1.506, so we're doing much better than uniform.

3. For a "Barker-like" or "flat-spaced" sequence, we want the autocorrelation to be as flat as possible. This is related to the "minimum peak sidelobe" problem in signal processing.

4. The key insight: the score is $2n \cdot \max_k \text{conv}(a,a)[k] / (\sum a)^2$. Since conv(a,a) has $2n-1$ elements, and we want to minimize the maximum, we want the convolution to be as "flat" as possible.

5. Note that $\sum_k \text{conv}(a,a)[k] = (\sum a)^2$. So the average value of the convolution is $(\sum a)^2 / (2n-1)$. If we could make all convolution values equal, the max would be $(\sum a)^2 / (2n-1)$, giving a score of $2n/(2n-1) \approx 1$. So the theoretical lower bound is approximately 1.

            w = e_w / s_w

        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        a = e / e.sum()  # reuse
        z_shift = z - z.max()
        e = np.exp(z_shift)
        a = e / e.sum()

        sc = _norm_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 2000 == 0:
            lr = max(0.01, lr * 0.95)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-method optimization:
    1. Successive Linearization (LP-based cutting planes) - main method
    2. Log-sum-exp gradient descent - auxiliary
    3. Subgradient polish - final refinement
    Multiple initializations and restarts to avoid local optima.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: score={best_score:.10f}, n={len(p)}, t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={sc:.10f}")
        update_best(parent, sc)
        n_main = len(parent)
    else:
        n_main = 4096
        parent = np.ones(n_main) / n_main
        update_best(parent, _score(parent))

    print(f"Main n = {n_main}")

    # Phase 1: Successive Linearization from parent (primary method)
    # This is the main workhorse - allocate most time here
    phase1_deadline = min(deadline, t_start + 900.0)
    print("Phase 1: Successive Linearization from parent")
    if time.monotonic() < phase1_deadline:
        a_best_l, sc_best_l = _run_successive_lin(parent, phase1_deadline, max_iter=100000)
        sc_true = _score(a_best_l)
        update_best(a_best_l, sc_true)

    print(f"  Phase 1 done: best={best_score:.10f}")

    # Phase 2: Successive Linearization from other inits
    phase2_deadline = min(deadline, t_start + 1400.0)
    print("Phase 2: Successive Linearization from diverse inits")
    if time.monotonic() < phase2_deadline:
        inits = _make_initializations(n_main, rng)
        # Include the current best as well
        inits.insert(0, best_seq.copy())
        # Shuffle to get diversity
        for idx in range(len(inits) - 1, 0, -1):
            j = rng.integers(0, idx + 1)
            inits[idx], inits[j] = inits[j], inits[idx]

        per_init_time = (phase2_deadline - time.monotonic()) / len(inits)
        per_init_time = max(per_init_time, 20.0)  # at least 20s each

        for i, init_a in enumerate(inits):
            if time.monotonic() > phase2_deadline - 10:
                break
            if per_init_time < 30 and i > 3:
                break  # Don't spend too little on each
            per_dl = min(phase2_deadline - 5, time.monotonic() + per_init_time)
            a_res, sc_res = _run_successive_lin(init_a, per_dl, max_iter=20000)
            sc_true = _score(a_res)
            update_best(a_res, sc_true)

    print(f"  Phase 2 done: best={best_score:.10f}")

    # Phase 3: Log-sum-exp GD polish
    phase3_deadline = min(deadline, t_start + 1650.0)
    print("Phase 3: Log-sum-exp GD polish")
    if time.monotonic() < phase3_deadline and best_seq is not None:
        per_dl = min(phase3_deadline - 5, time.monotonic() + 200.0)
        a_res, sc_res = _run_logsumexp_gd(best_seq, per_dl, rng)
        sc_true = _score(a_res)
        update_best(a_res, sc_true)

    print(f"  Phase 3 done: best={best_score:.10f}")

    # Phase 4: Subgradient polish
    phase4_deadline = min(deadline, t_start + 1750.0)
    print("Phase 4: Subgradient polish")
    if time.monotonic() < phase4_deadline and best_seq is not None:
        per_dl = min(phase4_deadline - 5, time.monotonic() + 80.0)
        a_res, sc_res = _subgradient_polish(best_seq, per_dl)
        sc_true = _score(a_res)
        update_best(a_res, sc_true)

    print(f"  Phase 4 done: best={best_score:.10f}")

    # Phase 5: One more round of successive linearization on best
    phase5_deadline = deadline
    print("Phase 5: Final successive linearization")
    if time.monotonic() < phase5_deadline and best_seq is not None:
        per_dl = min(phase5_deadline - 10, time.monotonic() + (phase5_deadline - time.monotonic()) * 0.8)
        if per_dl > time.monotonic() + 20:
            a_res, sc_res = _run_successive_lin(best_seq, per_dl, max_iter=50000)
            sc_true = _score(a_res)
            update_best(a_res, sc_true)

    print(f"  Phase 5 done: best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            pass  # Already normalized
        else:
            final_seq = final_seq / s
        sc_final = _score(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        # Validate with evaluator
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Evaluate failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = np.ones(256) / 256
    print(f"FALLBACK: n=256, score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
