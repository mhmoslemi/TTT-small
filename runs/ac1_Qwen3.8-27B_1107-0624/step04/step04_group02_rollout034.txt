Step function optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses fine log-sum-exp continuation with L-BFGS, KKT-based local equalization,
and multi-restart strategy. Focused on polishing the parent solution.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_obj(z, n, beta):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * cm
    return 2.0 * n * (cm + beta * float(np.log(s)))


def _smooth_grad(z, n, beta):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:

    g = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(g)):
        return np.zeros_like(z)
    return g

                     maxiter_start, maxiter_end, gtol_start, gtol_end, per_level_time):
    """L-BFGS with fine log-sum-exp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    beta_start = beta_start_scale * max_c0

    for i in range(num_betas):
        if time.monotonic() > deadline - 2.0:
            break
        
        beta = beta_start * (beta_ratio ** i)
        maxiter = int(maxiter_start + (maxiter_end - maxiter_start) * (i / max(1, num_betas - 1)))
        gtol = gtol_start * (gtol_end / gtol_start) ** (i / max(1, num_betas - 1))
