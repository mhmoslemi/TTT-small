"""
Multi-strategy optimizer for min 2n*max(conv(a,a))/sum(a)^2 on non-negative sequences.
Combines L-BFGS continuation on log-softmax parameterization, Nesterov projected
gradient on the simplex, and active-set subgradient polishing. Uses FFT-based
convolutions and a time-budgeted multi-phase schedule with restarts.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_conv(a, b):
    n = len(a)
    m = len(b)
    out_len = n + m - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_len]

def _autocorr(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    r = np.fft.irfft(fa * fa, N)
    return r[:2 * n - 1]

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _autocorr(a)
    return 2.0 * n * c.max() / (s * s)

def _score_norm(a):
    """Score assuming sum(a)=1: 2n*max(c)."""
    n = len(a)
    c = _autocorr(a)
    return 2.0 * n * c.max()

def _project_simplex(v):
    """Project v onto {x>=0, sum(x)=1} in O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones(len(z)) / len(z)
    return e / s

def _grad_wrt_a(a, w):
    """
    Gradient of sum_k w[k]*(a*a)[k] with respect to a.
    Returns g where g[j] = 2*sum_d a[d]*w[j+d].
    Computed as 2*convolve(a[::-1], w)[n-1:2n-1].
    """
    n = len(a)
    a_rev = a[::-1]
    conv = _fft_conv(a_rev, w)
    return 2.0 * conv[n - 1:2 * n - 1]

def _smooth_weights(c, beta):
    """Compute softmax weights over lags."""
    c_max = c.max()
    shifted = (c - c_max) / beta
    # Clip for numerical stability
    shifted = np.minimum(shifted, 0.0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = e / s
    return w, c_max

def _smooth_obj_val(c, beta):
    """Value of soft-max approximation: c_max + beta*logsumexp((c-c_max)/beta)."""
    c_max = c.max()
    shifted = (c - c_max) / beta
    shifted = np.minimum(shifted, 0.0)
    return c_max + beta * np.log(np.sum(np.exp(shifted)))

def _lbfgs_fun(z, beta, n):
    """Objective for L-BFGS: smooth max of autocorrelation, 2n-scaled."""
    a = _softmax(z)
    c = _autocorr(a)
    val = _smooth_obj_val(c, beta)
    return 2.0 * n * val

def _lbfgs_jac(z, beta, n):
    """Gradient of smooth objective w.r.t. z (log-softmax parameterization)."""
    a = _softmax(z)
    c = _autocorr(a)
    w, c_max = _smooth_weights(c, beta)
    g = _grad_wrt_a(a, w)
    # df/dz_j = a_j * (g_j - dot(a, g))
    dot_ag = np.dot(a, g)
    dz = a * (g - dot_ag)
    return 2.0 * n * dz

def _run_lbfgs_from_z(z, beta, n, deadline, maxiter=500):
    """Run L-BFGS-B from initial z with given beta."""
    def fun(z_):
        return _lbfgs_fun(z_, beta, n)
    
    def jac(z_):
        return _lbfgs_jac(z_, beta, n)
    
    try:
        result = minimize(fun, z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'ftol': 1e-16,
                                   'gtol': 1e-10, 'maxls': 40, 'maxfun': 2000})
        if result.x is not None and np.all(np.isfinite(result.x)):
            return result.x
    except Exception:
        pass
    return z

def _nesterov_simplex(a_init, deadline, beta_schedule, lr_init=0.1, mom_init=0.9):
    """
    Nesterov-accelerated projected gradient on the simplex.
    Uses smooth approximation with continuation on beta.
    """
    n = len(a_init)
    a = a_init.copy()
    v = np.zeros(n)  # momentum
    beta_idx = 0
    best_a = a.copy()
    best_sc = _score_norm(a)
    it = 0
    
    while time.monotonic() < deadline and beta_idx < len(beta_schedule):
        beta = beta_schedule[beta_idx]
        # Run for a fixed number of iterations or until deadline
        iters_per_beta = int((deadline - time.monotonic()) * 0.8 / max(1, len(beta_schedule) - beta_idx))
        iters_per_beta = max(10, min(iters_per_beta, 50000))
        
        for inner in range(iters_per_beta):
            if time.monotonic() >= deadline:
                break
            it += 1
            a_cur = _project_simplex(a)
            c = _autocorr(a_cur)
            w, c_max = _smooth_weights(c, beta)
            g = _grad_wrt_a(a_cur, w)
            
            # Nesterov momentum step
            v = mom_init * v - lr_init / (n + 2) * g
            a = a_cur + v
            a = _project_simplex(a)
            
            # Periodic evaluation
            if it % 1000 == 0:
                sc = _score_norm(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
        
        beta_idx += 1
    
    # Final eval
    sc = _score_norm(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    
    return best_a, best_sc

def _active_set_grad_opt(a_init, deadline, lr=0.001, mom=0.8, top_frac=0.01):
    """
    Projected gradient with hard active set (near-max lags).
    """
    n = len(a_init)
    a = a_init.copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = _score_norm(a)
    it = 0
    
    while time.monotonic() < deadline:
        a = _project_simplex(a)
        c = _autocorr(a)
        c_max = c.max()
        
        # Active set: top fraction of lags
        k_active = max(1, int(top_frac * (2 * n - 1)))
        top_indices = np.argpartition(c, -k_active)[-k_active:]
        
        w = np.zeros(len(c))
        w[top_indices] = 1.0 / k_active
        
        g = _grad_wrt_a(a, w)
        
        # Momentum step
        v = mom * v - lr * g
        a = a + v
        a = _project_simplex(a)
        
        it += 1
        if it % 500 == 0:
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        
    sc = _score_norm(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    
    return best_a, best_sc

def _make_u_shape(n, p=2.0):
    """Create a U-shaped (bowl) initial sequence."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + 3.0 * np.abs(u) ** p
    a = np.maximum(a, 0.01)
    return a / a.sum()

def _make_sine_shape(n, freq=1.0, p=2.0):
    """Create a sine-based initial sequence."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = np.sin(np.pi * freq * np.abs(u)) ** p + 0.1
    a = np.maximum(a, 0.01)
    return a / a.sum()

def _interp_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 0 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    s = p.sum()
    return p / s if s > 0 else np.ones(n_new) / n_new
