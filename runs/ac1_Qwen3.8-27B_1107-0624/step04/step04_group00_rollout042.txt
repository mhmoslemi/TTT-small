        inits.append(a / a.sum())
    
    # Cosine window
    for p in [0.5, 1.0, 1.5, 2.0]:
        a = np.maximum(np.cos(np.pi * u / 2.0), 1e-10) ** p
        inits.append(a / a.sum())
    
    # Exponential taper from ends
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        a = np.exp(-alpha * np.abs(u))
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    # Sine-based
    for freq in [1, 2, 3]:
        a = 1.0 + 0.3 * np.sin(freq * np.pi * u)
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    # Random smooth (via random polynomial coefficients)
    for _ in range(5):
        coeffs = rng.normal(0, 0.2, size=7)
        a = np.ones(n)
        for k in range(1, 7):
            a = a + coeffs[k] * u**k
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    return inits


def _lbfgs_continuation(a_init, deadline, beta_base, n_betas=None):
    """L-BFGS with continuation on beta (from large to small)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if n_betas is None:
        n_betas = 12
    
    # Create beta schedule: from beta_base*max_c0 down to very small
    betas = [beta_base * max_c0 * (0.5 ** i) for i in range(n_betas)]
    betas[-1] = 1e-6 * max_c0
    
    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - 5:
            break
        
        remaining = n_betas - i
        remaining_time = deadline - time.monotonic() - 5
        per_beta_time = remaining_time / remaining
        
        try:
            res = minimize(
                _smoothed_obj_grad, z, args=(beta, n),
                method='L-BFGS-B', jac=True,
                options={'maxiter': int(max(50, per_beta_time * 20)),
                         'ftol': 1e-20, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': int(max(200, per_beta_time * 50))}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                z = res.x
                a_new = _softmax_fn(z)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _adam_optimize(a_init, deadline, lr_init=0.1, beta_init_ratio=0.05):
    """Adam optimizer for the smoothed objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    lr = lr_init
    
    c0 = _full_convolve(a_init, a_init)
    beta = beta_init_ratio * float(c0.max())
    
    stuck_count = 0
    
    for it in range(200000):
        if time.monotonic() > deadline:
            break
        
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Adaptive beta: decrease over time, adapt to peak structure
        beta = max(1e-7, beta_init_ratio * cm * (0.9995 ** t))
        
        logits = (c - cm) / beta
        w = np.exp(logits)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= w_sum
        
        grad_a = 2.0 * n * _conv_grad(a, w)
        grad_z = a * (grad_a - np.dot(grad_a, a))
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100, 100)
        z = z - z.mean()
        
        # Periodic restart/kick if stuck
        if t % 5000 == 0 and stuck_count > 3:
            z = z + 0.03 * np.random.default_rng(t).normal(size=n)
            z = z - z.mean()
            lr = max(0.01, lr * 0.8)
        
        if t % 1000 == 0:
            lr = max(0.005, lr * 0.97)
        
        a_new = _softmax_fn(z)
        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            stuck_count = 0
        else:
            stuck_count += 1
    
    return best_a, best_score


def _peak_equalization(a_init, deadline, n_iter=50000):
    """
    Direct peak equalization: iteratively identify the top peaks and
    apply subgradient steps to reduce them while trying to keep
    lower peaks from increasing.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    # Momentum for the subgradient
    momentum = np.zeros(n)
    mu = 0.9
    lr = 0.05
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Find top K peaks and use their average subgradient
        K = min(5, len(c) - 1)
        top_indices = np.argsort(c)[::-1][:K]
        weights = np.zeros(len(c))
        for i in range(K):
            weights[top_indices[i]] = (K - i) / (K * (K + 1) / 2.0)
        weights /= weights.sum()
        
        # Compute weighted subgradient
        grad_a = 2.0 * n * _conv_grad(a, weights)
        grad_z = a * (grad_a - np.dot(grad_a, a))
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        momentum = mu * momentum + (1 - mu) * grad_z
        z = z - lr * momentum
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        # Adaptive learning rate
        if it % 1000 == 0:
            lr = max(0.001, lr * 0.98)
        
        a_new = _softmax_fn(z)
        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
    
    return best_a, best_score


def _peak_surgery(a_init, deadline):
    """Direct peak surgery: find contributors to the max peak and reduce them."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    for outer in range(200):
        if time.monotonic() > deadline:
            break
        
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        k_star = int(np.argmax(c))
        
        # Find top contributing pairs to this peak
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        valid_i = np.arange(i_min, i_max + 1)
        contributions = a[valid_i] * a[k_star - valid_i]
        top_idx = valid_i[np.argsort(contributions)[::-1][:8]]
        
        improved = False
        for idx in top_idx:
            for delta in [0.002, 0.005, 0.01, 0.02, 0.05]:
                z_trial = z.copy()
                z_trial[idx] -= delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm(a_trial)
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try increasing low values (redistribution)
            a_cur = _softmax_fn(z)
            low_idx = np.argsort(a_cur)[:5]
            for idx in low_idx:
                for delta in [0.005, 0.01, 0.02]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm(a_trial)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            break
    
    return best_a, best_score


def _polynomial_optimize(n, deadline, max_degree=8):
    """Optimize a polynomial-parameterized sequence using Nelder-Mead."""
    t = np.linspace(0, 1, n)
    u = 2.0 * t - 1.0
    
    def poly_score(coeffs):
        log_a = coeffs[0] * np.ones(n)
        for k in range(1, len(coeffs)):
            log_a = log_a + coeffs[k] * u**k
        log_a = log_a - log_a.max()
        a = np.exp(log_a)
        a = np.maximum(a, 1e-15)
        s = a.sum()
        if s < 1e-10:
            return np.inf
        a = a / s
        c = _full_convolve(a, a)
        return 2.0 * n * c.max()
    
    best_score = np.inf
    best_coeffs = np.zeros(max_degree + 1)
    
    # Try various starting points
    rng = np.random.default_rng(123)
    starts = [np.zeros(max_degree + 1)]
    for _ in range(10):
        s = rng.normal(0, 0.3, max_degree + 1)
        s[0] = 0
        starts.append(s)
    
    for start in starts:
        if time.monotonic() > deadline:
            break
        try:
            res = minimize(poly_score, start, method='Nelder-Mead',
                          options={'maxiter': 3000, 'xatol': 1e-10, 'fatol': 1e-12})
            if res.fun < best_score:
                best_score = res.fun
                best_coeffs = res.x.copy()
                print(f"    Poly opt: {best_score:.10f}", flush=True)
                # Continue from this point with another round
                res2 = minimize(poly_score, res.x, method='Nelder-Mead',
                               options={'maxiter': 2000, 'xatol': 1e-12, 'fatol': 1e-14})
                if res2.fun < best_score:
                    best_score = res2.fun
                    best_coeffs = res2.x.copy()
        except Exception:
            pass
    
    # Construct sequence
    log_a = best_coeffs[0] * np.ones(n)
    for k in range(1, len(best_coeffs)):
        log_a = log_a + best_coeffs[k] * u**k
    log_a = log_a - log_a.max()
    a = np.exp(log_a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    return a, best_score


def _interp_to_n(p, n_new):
    """Interpolate a profile to a new length."""
    p = np.asarray(p, dtype=np.float64)
    n0 = len(p)
    if n0 == n_new:
        return p / p.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Main optimization driver. Uses multiple methods and initializations
    to find the best sequence within the time budget.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  [{label}] score={best_score:.10f} n={len(p)} t={time.monotonic()-t_start:.0f}s", flush=True)
    
    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    # Determine working length
    if parent is not None and len(parent) >= 1024:
        n_work = len(parent)
    else:
        n_work = 4096
    
    print(f"Working with n={n_work}", flush=True)
    
    if parent is not None:
        sc_p = _score_norm(parent)
        update_best(parent, sc_p, "parent")
    else:
        parent = np.ones(n_work) / n_work
        update_best(parent, _score_norm(parent), "uniform")
    
    # --- Phase 1: Quick scan of initializations (120s) ---
    phase1_end = min(deadline, t_start + 120.0)
    print("Phase 1: Initialization scan", flush=True)
    
    candidates = []
    if parent is not None:
        candidates.append(("parent", parent.copy()))
    
    # Generate and evaluate diverse inits
    inits = _make_init_profiles(n_work, rng)
    for i, init in enumerate(inits):
        if time.monotonic() > phase1_end - 30:
            break
        sc = _score_norm(init)
        if sc < 2.0:  # Only keep if better than uniform
            candidates.append((f"init_{i}", init))
        if sc < best_score:
            update_best(init, sc, f"init_{i}")
    
    # Sort candidates by score
    scored = []
    for name, seq in candidates:
        sc = _score_norm(seq)
        scored.append((sc, name, seq))
    scored.sort(key=lambda x: x[0])
    
    print(f"  Found {len(scored)} candidates, best={scored[0][0]:.10f} ({scored[0][1]})", flush=True)
    
    # --- Phase 2: L-BFGS continuation on top candidates (600s) ---
    phase2_end = min(deadline, t_start + 720.0)
    print("Phase 2: L-BFGS continuation", flush=True)
    
    n_top = min(4, len(scored))
    per_candidate_time = (phase2_end - time.monotonic() - 10) / max(1, n_top)
    
    for ci in range(n_top):
        if time.monotonic() > phase2_end - 20:
            break
        
        sc_c, name_c, seq_c = scored[ci]
        per_dl = min(phase2_end - 10, time.monotonic() + per_candidate_time)
        
        # L-BFGS with moderate beta first
        p_best, sc_best = _lbfgs_continuation(seq_c, per_dl, beta_base=0.03, n_betas=10)
        update_best(p_best, sc_best, f"lbfgs_{name_c}")
        
        # Also update scored list for subsequent phases
        if sc_best < scored[ci][0]:
            scored[ci] = (sc_best, name_c, p_best)
    
    # --- Phase 3: Deep L-BFGS with fine continuation on best (400s) ---
    phase3_end = min(deadline, t_start + 1120.0)
    print("Phase 3: Fine L-BFGS on best", flush=True)
    
    if best_seq is not None:
        per_dl = min(phase3_end - 10, time.monotonic() + (phase3_end - time.monotonic() - 10))
        
        # Fine continuation starting from very small beta
        p_best, sc_best = _lbfgs_continuation(best_seq, per_dl, beta_base=0.005, n_betas=15)
        update_best(p_best, sc_best, "fine_lbfgs")
        
        # Multi-restart: perturb and re-optimize
        if time.monotonic() < phase3_end - 60:
            for r in range(3):
                if time.monotonic() > phase3_end - 30:
                    break
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z - z.mean()
                noise = 0.02 * (r + 1) * rng.normal(size=n_work)
                z = z + noise
                z = np.clip(z, -60, 60)
                z = z - z.mean()
                a_pert = _softmax_fn(z)
                
                per_dl_r = min(phase3_end - 10, time.monotonic() + 60.0)
                p_r, sc_r = _lbfgs_continuation(a_pert, per_dl_r, beta_base=0.003, n_betas=8)
                update_best(p_r, sc_r, f"lbfgs_restart{r}")
    
    # --- Phase 4: Adam with very small beta (250s) ---
    phase4_end = min(deadline, t_start + 1370.0)
    print("Phase 4: Adam optimization", flush=True)
    
    if best_seq is not None:
        per_dl = min(phase4_end - 10, time.monotonic() + (phase4_end - time.monotonic() - 10))
        
        p_best, sc_best = _adam_optimize(best_seq, per_dl, lr_init=0.08, beta_init_ratio=0.005)
        update_best(p_best, sc_best, "adam")
        
        # Second Adam run with different params
        if time.monotonic() < phase4_end - 50:
            per_dl2 = min(phase4_end - 10, time.monotonic() + (phase4_end - time.monotonic() - 10))
            p_best2, sc_best2 = _adam_optimize(best_seq, per_dl2, lr_init=0.03, beta_init_ratio=0.002)
            update_best(p_best2, sc_best2, "adam_slow")
    
    # --- Phase 5: Peak equalization (150s) ---
    phase5_end = min(deadline, t_start + 1520.0)
    print("Phase 5: Peak equalization", flush=True)
    
    if best_seq is not None and time.monotonic() < phase5_end:
        per_dl = min(phase5_end - 10, time.monotonic() + (phase5_end - time.monotonic() - 10))
        p_best, sc_best = _peak_equalization(best_seq, per_dl, n_iter=100000)
        update_best(p_best, sc_best, "peak_eq")
    
    # --- Phase 6: Peak surgery (100s) ---
    phase6_end = min(deadline, t_start + 1620.0)
    print("Phase 6: Peak surgery", flush=True)
    
    if best_seq is not None and time.monotonic() < phase6_end:
        per_dl = min(phase6_end - 10, time.monotonic() + (phase6_end - time.monotonic() - 10))
        p_best, sc_best = _peak_surgery(best_seq, per_dl)
        update_best(p_best, sc_best, "surgery")
    
    # --- Phase 7: Final L-BFGS polish (100s) ---
    phase7_end = deadline
    print("Phase 7: Final polish", flush=True)
    
    if best_seq is not None and time.monotonic() < phase7_end - 20:
        per_dl = min(phase7_end - 15, time.monotonic() + (phase7_end - time.monotonic() - 15))
        p_best, sc_best = _lbfgs_continuation(best_seq, per_dl, beta_base=0.001, n_betas=6)
        update_best(p_best, sc_best, "final_lbfgs")
        
        # Final Adam micro-step
        if time.monotonic() < phase7_end - 5:
            per_dl2 = min(phase7_end - 2, time.monotonic() + (phase7_end - time.monotonic() - 2) * 0.5)
            p_best2, sc_best2 = _adam_optimize(best_seq, per_dl2, lr_init=0.01, beta_init_ratio=0.001)
            update_best(p_best2, sc_best2, "final_adam")
    
    # --- Try polynomial optimization if time remains ---
    if time.monotonic() < deadline - 30 and n_work <= 4096:
        print("Bonus: Polynomial optimization", flush=True)
        per_dl = min(deadline - 20, time.monotonic() + 50.0)
        p_poly, sc_poly = _polynomial_optimize(n_work, per_dl, max_degree=6)
        update_best(p_poly, sc_poly, "polynomial")
    
    # --- Final validation ---
    print(f"\nFinal best score (norm): {best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}", flush=True)
    
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = np.ones(n_work) / n_work
        
        sc_final = _score_full(final_seq)
        print(f"FINAL normalized score: {sc_final:.10f}", flush=True)
        
        # Validate with evaluator if sequence is not too long
        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                print(f"Evaluator score: {sc_eval:.10f}", flush=True)
            except Exception as ex:
                print(f"evaluate_sequence error: {ex}", flush=True)
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = np.ones(256) / 256
    print(f"FALLBACK: n=256, score={_score_full(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
