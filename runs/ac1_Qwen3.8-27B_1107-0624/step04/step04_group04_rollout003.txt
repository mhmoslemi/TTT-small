"""
Optimized step function search using adaptive active-set gradients with
L-BFGS continuation, KKT balancing, and multi-restart strategy.
Exploits scale invariance (sum=1) for cleaner gradient computation.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k *= 2
    return max(2, k)


def _conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _conv_self(a):
    n = len(a)
    out_n = 2 * n - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    r = np.fft.irfft(fa * fa, N)[:out_n]
    return r


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_unit(a):
    """Score assuming sum(a)=1: 2n * max(conv(a,a))."""
    n = len(a)
    c = _conv_self(a)
    return float(2.0 * n * c.max())


def _score_general(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-15:
        return np.inf
    c = _conv_self(a)
    return float(2.0 * n * c.max() / (S * S))


def _compute_active_weights(c, beta, top_k=0):
    """Compute softmax weights over convolution values (active set)."""
    n_c = len(c)
    if top_k > 0 and top_k < n_c:
        idx = np.argpartition(c, -top_k)[-top_k:]
        w = np.zeros(n_c)
        c_top = c[idx]
        c_norm = (c_top - c_top.max()) / beta
        e = np.exp(c_norm)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w[idx[np.argmax(c_top)]] = 1.0
        else:
            w[idx] = e / s
        W = float(np.dot(w, c))
    else:
        c_max = c.max()
        c_shifted = (c - c_max) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(n_c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        W = float(np.dot(w, c))
    return w, W


def _grad_z(z, beta, n, top_k=0):
    """
    Gradient of smoothed objective in z-space.
    Objective: 2n * LSE_beta(c) where c = conv(softmax(z), softmax(z)), sum=1.
    Gradient: 2n * a_i * (h_i - W)
    where h_i = sum_k w_k * a[k-i], W = sum_k w_k * c_k.
    """
    a = _softmax(z)
    c = _conv_self(a)
    w, W = _compute_active_weights(c, beta, top_k)

    # h_i = sum_k w[k] * a[k-i] = conv(w, a_rev)[n-1+i]
    a_rev = a[::-1]
    ha = _conv(w, a_rev)
    h = ha[n - 1:2 * n - 1]

    grad = 2.0 * n * a * (h - W)

    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _obj_z(z, beta, n, top_k=0):
    """Smoothed objective in z-space (sum=1 normalization)."""
    a = _softmax(z)
    c = _conv_self(a)
    w, W = _compute_active_weights(c, beta, top_k)
    if top_k > 0 and top_k < len(c):
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_top = c[idx]
        log_sum = float(np.log(np.sum(np.exp((c_top - c_top.max()) / beta))))
        return float(2.0 * n * (c_top.max() + beta * log_sum))
    else:
        c_max = c.max()
        c_shifted = (c - c_max) / beta
        log_sum = float(np.log(np.sum(np.exp(c_shifted))))
        return float(2.0 * n * (c_max + beta * log_sum))


def _lbfgs_optimize(a_init, deadline, max_time=None, top_k=0, beta_schedule=None):
    """L-BFGS with beta continuation in logit space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_unit(a_init)

    if max_time is None:
        max_time = max(1.0, deadline - time.monotonic())

    if beta_schedule is None:
        c0 = _conv_self(a_init)
        c_scale = max(float(c0.max()), 1e-8)
        beta_schedule = [b * c_scale for b in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]]

    per_beta = max_time / len(beta_schedule)

    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline:
            break
        b_deadline = min(deadline, time.monotonic() + per_beta)
        remaining = b_deadline - time.monotonic()
        if remaining < 1.0:
            break

        try:
            res = minimize(
                _obj_z, z, args=(beta, n, top_k),
                method='L-BFGS-B', jac=_grad_z,
                options={
                    'maxiter': max(20, int(1000 * remaining / per_beta)),
                    'ftol': 1e-18,
                    'gtol': 1e-12,
                    'maxls': 50,

                    'iprint': 0

            )

                    s = _score_unit(a_new)
                    if np.isfinite(s) and s < best_score:
                        best_score = s
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _adam_optimize(a_init, deadline, lr=0.1, top_k=0, beta_init=None):

        c0 = _conv_self(a_init)

    else:
        beta = beta_init

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

        t += 1

        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        z = z - z.mean()

            else:

            if t % 200 == 0:
                beta = max(1e-15, beta * 0.85)
            if stagnation > 2000:

def _kkt_optimize(a_init, deadline, max_iter=50000, tol=1e-12):
    """
    KKT-based optimization: directly enforce the stationarity condition
    h_i = W for all i with a_i > 0.
    Uses iterative projection in logit space.
    """
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_unit(a_init)

    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    a = _softmax(z)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _conv_self(a)
        c_max = float(c.max())
        # Active set: positions within relative tolerance of max
        thresh = c_max * (1.0 - 1e-6)
        active = np.where(c >= thresh)[0]

        if len(active) == 0:
            break

        # Uniform weights over active set (simplest KKT condition)
        w = np.zeros(len(c))
        w[active] = 1.0 / len(active)

        # Compute h_i = sum_k w[k] * a[k-i]
        a_rev = a[::-1]
        ha = _conv(w, a_rev)
        h = ha[n - 1:2 * n - 1]
        W = float(np.dot(w, c))

        # Residual: r_i = h_i - W. We want r_i = 0 for all i.
        r = h - W
        r_max = float(np.abs(r).max())

        if r_max < tol:
            break

        # Update: z_i -= step * r_i (push a_i down where h_i > W, up where h_i < W)
        step = min(0.5, r_max * 0.8 / (float(np.abs(r).mean()) + 1e-15))
        step = min(step, 0.3)

        z = z - step * (r / (r_max + 1e-15))
        z = np.clip(z, -100, 100)
        z = z - z.mean()
        a_new = _softmax(z)

        s = _score_unit(a_new)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_a = a_new.copy()
        else:
            # Roll back and reduce step
            z = z + step * (r / (r_max + 1e-15))
            z = z - z.mean()
            step *= 0.5
            z = z - step * (r / (r_max + 1e-15))
            z = np.clip(z, -100, 100)
            z = z - z.mean()
            a_new = _softmax(z)
            s = _score_unit(a_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_a = a_new.copy()

        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()

        # Convergence check
        if it > 100 and r_max < 1e-10:
            break

    return best_a, best_score


def _make_profiles(n):
    """Generate diverse initial profiles."""
    profiles = []
    # Uniform
    profiles.append(np.ones(n) / n)
    # Linear (increasing)
    t = np.linspace(0, 1, n)
    a = 1.0 + 0.5 * t
    profiles.append(a / a.sum())
    # Linear (decreasing)
    a = 1.0 + 0.5 * (1 - t)
    profiles.append(a / a.sum())
    # Quadratic (U-shape)
    a = 1.0 + 2.0 * (2 * t - 1) ** 2
    profiles.append(a / a.sum())
    # Quadratic (inverted U)
    a = 1.0 + 1.5 * (1 - (2 * t - 1) ** 2)
    profiles.append(a / a.sum())
    # Exponential (fast decay)
    a = np.exp(-3.0 * t)
    profiles.append(a / a.sum())
    # Exponential (slow decay)
    a = np.exp(-0.5 * t)
    profiles.append(a / a.sum())
    # Sine shape
    a = 1.0 + 0.3 * np.sin(2 * np.pi * t)
    a = np.maximum(a, 0.01)
    profiles.append(a / a.sum())
    # Bathtub (high at edges)
    a = 1.0 / (0.1 + np.abs(t - 0.5) * 2)
    profiles.append(a / a.sum())
    # Two-hump
    a = np.exp(-20 * t ** 2) + np.exp(-20 * (t - 1) ** 2)
    profiles.append(a / a.sum())
    return profiles


def _interp_profile(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = len(p)
    if n0 == n_new:
        return p / p.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization:
    1. Generate diverse profiles at multiple lengths.
    2. L-BFGS with beta continuation.
    3. Adam with adaptive beta.
    4. KKT balancing.
    5. Multi-restart from best.
    6. Try different lengths.
    """
    global height_sequence_1

    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            el = time.monotonic() - t0
            print(f"  [{el:7.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # === Phase 0: Initial candidates ===
    print("=== Phase 0: Initial candidates ===")

    # Try parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            S = parent_raw.sum()
            if S > 0.01:
                parent = parent_raw / S
                sc = _score_unit(parent)
                update_best(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent: {e}")

    # Generate profiles at n=4096
    n_main = 4096
    profiles = _make_profiles(n_main)
    for i, p in enumerate(profiles):
        if time.monotonic() > deadline:
            break
        sc = _score_unit(p)
        update_best(p, sc, f"prof{i}_n{n_main}")

    # Also try n=2048
    n_alt = 2048
    profiles_alt = _make_profiles(n_alt)
    for i, p in enumerate(profiles_alt[:5]):
        if time.monotonic() > deadline:
            break
        sc = _score_unit(p)
        update_best(p, sc, f"prof{i}_n{n_alt}")

    # Interpolate parent to n_main if different length
    if parent is not None and len(parent) != n_main:
        p_ie = _interp_profile(parent, n_main)
        sc_ie = _score_unit(p_ie)
        update_best(p_ie, sc_ie, "parent_interp")

    print(f"  Phase 0 done: best={best_score:.10f}")

    # === Phase 1: L-BFGS on top candidates ===
    print("=== Phase 1: L-BFGS optimization ===")
    phase1_end = min(deadline, t0 + 600.0)

    # Sort candidates by score, optimize top few
    candidates = [(best_score, best_seq, n_main, "best")]
    # Add a few diverse ones
    for i, p in enumerate(profiles[:4]):
        sc = _score_unit(p)
        if sc < best_score * 1.05:
            candidates.append((sc, p, n_main, f"p{i}"))

    candidates.sort(key=lambda x: x[0])

    for ci, (sc, p, pn, label) in enumerate(candidates[:3]):
        if time.monotonic() > phase1_end - 30:
            break
        remaining = phase1_end - time.monotonic()
        per_time = max(30.0, remaining / (3 - ci))
        dl = min(phase1_end - 5, time.monotonic() + per_time)
        print(f"  L-BFGS #{ci} [{label}] n={pn}, budget={per_time:.0f}s")

        p_b, sc_b = _lbfgs_optimize(p, dl, max_time=per_time, top_k=0)
        update_best(p_b, sc_b, f"lbfgs_{label}")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Adam polish ===
    print("=== Phase 2: Adam polish ===")
    phase2_end = min(deadline, t0 + 900.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        remaining = phase2_end - time.monotonic()
        # Adam with full active set (no top_k restriction)
        dl = min(phase2_end - 10, time.monotonic() + remaining * 0.6)
        p_b, sc_b = _adam_optimize(best_seq, dl, lr=0.08, top_k=0)
        update_best(p_b, sc_b, "adam_full")

        # Adam with small top_k
        if time.monotonic() < phase2_end - 10:
            dl2 = min(phase2_end - 5, time.monotonic() + (phase2_end - time.monotonic()) * 0.5)
            p_b2, sc_b2 = _adam_optimize(best_seq, dl2, lr=0.05, top_k=5)
            update_best(p_b2, sc_b2, "adam_top5")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: KKT balancing ===
    print("=== Phase 3: KKT balancing ===")
    phase3_end = min(deadline, t0 + 1200.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        remaining = phase3_end - time.monotonic()
        dl = min(phase3_end - 5, time.monotonic() + remaining)
        p_b, sc_b = _kkt_optimize(best_seq, dl, max_iter=80000)
        update_best(p_b, sc_b, "kkt")

        # Follow-up L-BFGS after KKT
        if time.monotonic() < phase3_end - 10:
            dl2 = min(phase3_end - 5, time.monotonic() + min(90.0, (phase3_end - time.monotonic()) * 0.5))
            p_b2, sc_b2 = _lbfgs_optimize(best_seq, dl2, max_time=min(80.0, dl2 - time.monotonic()), top_k=3)
            update_best(p_b2, sc_b2, "lbfgs_post_kkt")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Multi-restart ===
    print("=== Phase 4: Multi-restart ===")
    phase4_end = min(deadline, t0 + 1500.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        n_best = len(best_seq)
        n_restarts = 5
        per_restart = min(60.0, max(20.0, (phase4_end - time.monotonic()) / n_restarts))

        for ri in range(n_restarts):
            if time.monotonic() > phase4_end - 15:
                break
            noise = [0.02, 0.05, 0.1, 0.15, 0.25][ri]
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax(z)

            dl = min(phase4_end - 10, time.monotonic() + per_restart)
            # Quick L-BFGS
            per_lb = per_restart * 0.6
            p_m, sc_m = _lbfgs_optimize(p_start, min(phase4_end - 10, time.monotonic() + per_lb),
                                         max_time=per_lb, top_k=5)
            update_best(p_m, sc_m, f"r{ri}_lbfgs(n={noise})")

            if time.monotonic() < phase4_end - 10:
                per_ad = min(20.0, phase4_end - 10 - time.monotonic())
                p_a, sc_a = _adam_optimize(p_m, time.monotonic() + per_ad, lr=0.03, top_k=3)
                update_best(p_a, sc_a, f"r{ri}_adam")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Try different lengths ===
    print("=== Phase 5: Different lengths ===")
    phase5_end = min(deadline, t0 + 1700.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        n_curr = len(best_seq)
        target_ns = [2048, 8192]
        target_ns = [tn for tn in target_ns if tn != n_curr]

        for tn in target_ns:
            if time.monotonic() > phase5_end - 40:
                break
            p_t = _interp_profile(best_seq, tn)
            sc_t = _score_unit(p_t)
            update_best(p_t, sc_t, f"interp_n{tn}")

            remaining = phase5_end - time.monotonic()
            if remaining < 30:
                break
            per_opt = min(100.0, remaining * 0.5)
            dl = min(phase5_end - 10, time.monotonic() + per_opt)
            p_b, sc_b = _lbfgs_optimize(p_t, dl, max_time=per_opt, top_k=0)
            update_best(p_b, sc_b, f"lbfgs_n{tn}")

            if time.monotonic() < phase5_end - 20:
                dl2 = min(phase5_end - 10, time.monotonic() + min(50.0, phase5_end - time.monotonic() - 10))
                p_b2, sc_b2 = _adam_optimize(p_b, dl2, lr=0.04, top_k=3)
                update_best(p_b2, sc_b2, f"adam_n{tn}")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Final polishing ===
    print("=== Phase 6: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            # KKT polish
            dl = min(deadline - 5, time.monotonic() + min(60.0, remaining * 0.4))
            p_b, sc_b = _kkt_optimize(best_seq, dl, max_iter=30000)
            update_best(p_b, sc_b, "final_kkt")

        if time.monotonic() < deadline - 10:
            # Short L-BFGS with small top_k
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                dl2 = min(deadline - 3, time.monotonic() + min(40.0, remaining2 * 0.5))
                p_b2, sc_b2 = _lbfgs_optimize(best_seq, dl2, max_time=max(5.0, dl2 - time.monotonic()), top_k=5)
                update_best(p_b2, sc_b2, "final_lbfgs")

        if time.monotonic() < deadline - 5:
            # Very short Adam
            remaining3 = deadline - time.monotonic()
            if remaining3 > 5:
                dl3 = min(deadline - 2, time.monotonic() + min(20.0, remaining3 * 0.8))
                p_b3, sc_b3 = _adam_optimize(best_seq, dl3, lr=0.01, top_k=1)
                update_best(p_b3, sc_b3, "final_adam")

    # === Final output ===
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        S = float(final_seq.sum())
        if S > 0.01:
            final_score = _score_general(final_seq)
        else:
            final_seq = _make_profiles(2048)[0]
            final_score = _score_general(final_seq)
        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}, sum={S:.4f}")

        # Validate
        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 16384:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  eval check: {e}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profiles(2048)[0]
    print(f"  FALLBACK: n=2048, score={_score_general(fallback):.10f}")
    return [float(x) for x in fallback]
