"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses exact subgradient of the dominant convolution peak with L-BFGS,
adaptive multi-peak gradient, and strategic multi-start refinement.
Key innovations: exact peak gradient exploitation, peak-identity tracking,
and structured asymmetric initializations.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _grad_single_peak(a, k_star, n):
    g = np.zeros(n)
    j_min = max(0, k_star - n + 1)
    j_max = min(n - 1, k_star)
    js = np.arange(j_min, j_max + 1)
    g[js] = a[k_star - js] + a[js]
    return g


def _grad_top_k(a, c, K, n):
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n), 0, 0.0
    sorted_idx = np.argsort(c)[::-1][:K]
    weights = c[sorted_idx]
    weights = weights / weights.sum()
    g = np.zeros(n)
    for idx, w in zip(sorted_idx, weights):
        k = int(idx)
        g += w * _grad_single_peak(a, k, n)
    return g, sorted_idx[0], cm


def _obj_exact_peak(z, k_star, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c[k_star])


def _grad_exact_peak(z, k_star, n):
    a = _softmax(z)
    g_a = _grad_single_peak(a, k_star, n)
    W = float(np.dot(g_a, a))
    return 2.0 * n * a * (g_a - W)


def _obj_soft_peaks(z, K, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    sorted_c = np.sort(c)[::-1][:K]
    # Use log-sum-exp of top-K peaks
    m = float(sorted_c[0])
    return 2.0 * n * (m + float(np.log(np.sum(np.exp(sorted_c - m)))))


def _grad_soft_peaks(z, K, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    # Weight peaks by their relative magnitude (softmax over sorted peaks)
    sorted_idx = np.argsort(c)[::-1][:K]
    peak_vals = c[sorted_idx]
    m = float(peak_vals[0])
    e = np.exp(peak_vals - m)
    w = e / e.sum()
    g = np.zeros(n)
    for idx, wi in zip(sorted_idx, w):
        k = int(idx)
        g += wi * _grad_single_peak(a, k, n)
    W = float(np.dot(g, a))
    return 2.0 * n * a * (g - W)


def _run_lbfgs_peak(a_init, k_star, n, deadline, maxiter=5000):
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    elapsed = time.monotonic() - (deadline - maxiter * 0.01)

    try:
        result = minimize(
            _obj_exact_peak, z, args=(k_star, n),
            method='L-BFGS-B', jac=_grad_exact_peak,
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                     'maxls': 30, 'maxfun': maxiter * 3}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            a_new = _softmax(result.x)
            s_new = _score_norm(a_new)
            if np.isfinite(s_new) and s_new < best_score:
                best_score = s_new
                best_a = a_new.copy()
    except Exception:
        pass

    return best_a, best_score


def _run_lbfgs_multi(a_init, K, n, deadline, maxiter=5000):

def _adaptive_peak_tracking(a_init, deadline, n, max_rounds=200):
    """Iteratively optimize for the current max peak, tracking peak identity."""
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))
        cm = float(c[k_star])
        
        # Check gap to second peak
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[1] if len(c_sorted) > 1 else cm

        if gap < 0.05 * cm:
            # Multiple peaks are close, use multi-peak gradient
            K = min(5, len(c_sorted))
            g = _grad_soft_peaks(z, K, n)
        else:
            # Single dominant peak
            g = _grad_exact_peak(z, k_star, n)

        if not np.all(np.isfinite(g)):
            break

        # Adam-like update
        z = z - 0.1 * g / (np.abs(g).max() + 1e-10)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        a = _softmax(z)
        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _make_asymmetric_profiles(n):
    """Generate diverse asymmetric initial profiles."""
    t = np.linspace(0, 1, n)
    profiles = []

    # Linear ramps (various slopes)
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Reversed ramps
    for slope in [0.5, 1.0, 1.5, 2.0]:
        a = 1.0 + slope * (0.5 - t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Quadratic (asymmetric)
    for a1, a2 in [(1.0, 0.5), (2.0, -0.3), (1.5, 0.2), (0.5, 1.0), (-1.0, 0.5)]:
        u = 2 * t - 1
        a = 1.0 + a1 * u + a2 * u ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Exponential (various rates)
    for rate in [1.0, 2.0, 3.0, 5.0, -1.0, -2.0, -3.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())

    # Multi-modal
    for center in [0.2, 0.3, 0.4, 0.6, 0.7, 0.8]:
        for width in [0.1, 0.15, 0.25]:
            a = np.exp(-((t - center) / width) ** 2) + 0.1
            profiles.append(a / a.sum())

    # Triangular (off-center)
    for peak_pos in [0.2, 0.3, 0.4, 0.6, 0.7, 0.8]:
        a = 1.0 - np.abs(t - peak_pos) / max(peak_pos, 1.0 - peak_pos)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

            c_work = _full_convolve(a_work, a_work)
            sorted_c = np.sort(c_work)[::-1]
            if len(sorted_c) > 1 and sorted_c[1] > 0.9 * sorted_c[0]:
                # Second peak is close, optimize for it
                second_idx = int(np.where(c_work == sorted_c[1])[0][0])
                # Find the second distinct position
                mask = c_work >= sorted_c[1] - 1e-10
                positions = np.where(mask)[0]
                if len(positions) > 1:
                    k_second = int(positions[1])
                    per_dl = min(phase6_end - 5, time.monotonic() + 30.0)
                    p_best, sc_best = _run_lbfgs_peak(a_work, k_second, n_final, per_dl, maxiter=2000)
                    update_best(p_best, sc_best, "second_peak")

        # Final multi-peak L-BFGS
        if time.monotonic() < phase6_end - 15:
            per_dl = min(phase6_end - 5, time.monotonic() + 30.0)
            p_best, sc_best = _run_lbfgs_multi(best_seq, 3, n_final, per_dl, maxiter=2000)
            update_best(p_best, sc_best, "final_multi")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_asymmetric_profiles(256)[2]

        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        # Validate with evaluator
        if len(final_seq) <= 5000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                else:
                    print("WARNING: Evaluator returned inf!")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_asymmetric_profiles(256)[2]
    print(f"\nFALLBACK: n=256, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
