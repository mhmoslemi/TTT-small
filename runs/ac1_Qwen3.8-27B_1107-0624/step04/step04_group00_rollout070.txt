```python
"""
Search for a non-negative sequence minimizing
  2*n*max(convolve(a,a))/sum(a)^2.

The program combines:
  - low-dimensional Chebyshev/edge basis profile optimization,
  - full L-BFGS continuation in a log-softmax parameterization,
  - exact sparse top-peak subgradient Adam with line search,
  - golden-section mass-transfer local search.
"""

import time
import numpy as np
from scipy.optimize import minimize


def _next_pow2(n):
    n = int(max(1, n))
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    if out_n <= 0:
        return np.zeros(0, dtype=np.float64)
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    out = np.fft.irfft(fa * fb, N)[:out_n]
    return out


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    n = z.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if not np.all(np.isfinite(z)):
        return np.ones(n, dtype=np.float64) / float(n)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return float("inf")
    s = float(a.sum())
    if s < 1e-15 or not np.isfinite(s):
        return float("inf")
    c = _full_convolve(a, a)
    m = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m / (s * s))


def _as_vector(x):
    arr = np.asarray(x, dtype=np.float64).ravel()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if arr.size == 0 or s < 1e-12:
        return None
    return arr / s


def _default_sequence(n):
    n = int(max(2, n))
    j = np.arange(n, dtype=np.float64)
    a = j + 1.0
    return a / a.sum()


def _initial_profiles(n):
    n = int(max(2, n))
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    out = []

    for slope in (-3.0, -2.0, -1.0, -0.5, 0.5, 1.0, 1.5, 2.0, 3.0):
        a = 1.0 + slope * u
        a = np.maximum(a, 1e-12)
        out.append(a / a.sum())

    for c1 in (-2.0, -1.0, -0.5, 0.5, 1.0, 2.0):
        a = 1.0 + 0.5 * u + c1 * u * u
        a = np.maximum(a, 1e-12)
        out.append(a / a.sum())

    for rate in (-3.0, -2.0, -1.0, 1.0, 2.0, 3.0):
        a = np.exp(rate * u)
        out.append(a / a.sum())

    for p in (0.5, 1.0, 2.0, 4.0):
        a = np.maximum(np.cos(0.5 * np.pi * u), 1e-12) ** p
        out.append(a / a.sum())

    j = np.arange(n, dtype=np.float64)
    a = j + 1.0
    out.append(a / a.sum())
    a = float(n) - j
    out.append(a / a.sum())
    a = 1.0 + 1.2 * (float(n) - 1.0 - j) / float(n)
    out.append(a / a.sum())
    a = 1.0 + 1.2 * j / float(n)
    out.append(a / a.sum())
    return out


def _interp_profile(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    n_new = int(max(2, n_new))
    if n0 == 0:
        return np.ones(n_new, dtype=np.float64) / float(n_new)
    if n0 == 1:
        val = max(float(p[0]), 0.0)
        if val <= 0.0:
            val = 1.0 / float(n_new)
        return np.full(n_new, val, dtype=np.float64)
    if n0 == n_new:
        p = np.maximum(p, 0.0)
        s = float(p.sum())
        if s <= 0.0:
            return np.ones(n_new, dtype=np.float64) / float(n_new)
        return p / s
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    q = np.interp(x1, x0, np.maximum(p, 0.0))
    q = np.maximum(q, 1e-12)
    return q / q.sum()


def _cheb_basis(n, d):
    n = int(max(1, n))
    d = int(max(0, d))
    if n == 1:
        return np.ones((1, d + 1), dtype=np.float64)
    u = 2.0 * np.arange(n, dtype=np.float64) / float(n - 1) - 1.0
    x = np.empty((n, d + 1), dtype=np.float64)
    x[:, 0] = 1.0
    if d >= 1:
        x[:, 1] = u
    if d >= 2:
        x0 = np.ones(n, dtype=np.float64)
        x1 = u.copy()
        for k in range(2, d + 1):
            x2 = 2.0 * u * x1 - x0
            x[:, k] = x2
            x0, x1 = x1, x2
    return x


def _profile_basis(n, d):
    X = _cheb_basis(n, d)
    n = int(X.shape[0])
    if n == 1:
        extras = np.ones((1, 4), dtype=np.float64)
    else:
        u = 2.0 * np.arange(n, dtype=np.float64) / float(n - 1) - 1.0
        left_loc = np.exp(-12.0 * np.maximum(0.0, u + 1.0))
        right_loc = np.exp(-12.0 * np.maximum(0.0, 1.0 - u))
        left_half = np.exp(-6.0 * np.maximum(0.0, -u))
        right_half = np.exp(-6.0 * np.maximum(0.0, u))
        extras = np.column_stack((left_loc, right_loc, left_half, right_half))
    return np.column_stack((X, extras))


def _weights_and_value(c, beta, topK):
    c = np.asarray(c, dtype=np.float64).ravel()
    L = c.size
    w = np.zeros(L, dtype=np.float64)
    if L == 0:
        return w, 0.0
    m = float(np.max(c))
    if not np.isfinite(m):
        return w, m

    if beta is None or beta <= 0.0:
        tol = max(1e-15, 1e-12 * abs(m))
        if topK is None or topK <= 0 or topK >= L:
            idx = np.where(c >= m - tol)[0]
        else:
            kk = int(max(1, min(topK, L)))
            part = np.argpartition(c, L - kk)[-kk:]
            vals = c[part]
            mid = float(np.max(vals))
            idx = part[vals >= mid - tol]
        if idx.size == 0:
            idx = np.array([int(np.argmax(c))], dtype=int)
        w[idx] = 1.0 / float(idx.size)
        return w, m

    if topK is None or topK <= 0 or topK >= L:
        shifted = (c - m) / beta
        e = np.exp(shifted)
        s = float(e.sum())
        if s <= 0.0 or not np.isfinite(s):
            tol = max(1e-15, 1e-12 * abs(m))
            idx = np.where(c >= m - tol)[0]
            if idx.size == 0:
                idx = np.array([int(np.argmax(c))], dtype=int)
            w[idx] = 1.0 / float(idx.size)
            return w, m
        w = e / s
        F = m + beta * float(np.log(s))
        return w, float(F)

    kk = int(max(1, min(topK, L)))
    part = np.argpartition(c, L - kk)[-kk:]
    vals = c[part]
    mid = float(np.max(vals))
    shifted = (vals - mid) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        tol = max(1e-15, 1e-12 * abs(mid))
        good = vals >= (mid - tol)
        idx = part[good]
        if idx.size == 0:
            idx = part[np.argmax(vals)]
        w[idx] = 1.0 / float(idx.size)
        return w, mid
    vals = e / s
    w[part] = vals
    F = mid + beta * float(np.log(s))
    return w, float(F)


def _sparse_h(a, c, idx, wvals):
    a = np.asarray(a, dtype=np.float64).ravel()
    c = np.asarray(c, dtype=np.float64).ravel()
    idx = np.asarray(idx, dtype=int).ravel()
    wvals = np.asarray(wvals, dtype=np.float64).ravel()
    n = a.size
    h = np.zeros(n, dtype=np.float64)
    W = 0.0
    for kk, ww in zip(idx, wvals):
        if ww == 0.0:
            continue
        kk = int(kk)
        if kk < 0 or kk >= len(c):
            continue
        i0 = max(0, kk - n + 1)
        i1 = min(n - 1, kk)
        if i0 > i1:
            continue
        r_start = kk - i0
        r_end = kk - i1
        seg = a[r_end:r_start + 1][::-1]
        h[i0:i1 + 1] += ww * seg
        W += ww * float(c[kk])
    return h, float(W)


def _h_from_weights(a, c, w, topK):
    n = a.size
    L = c.size
    if topK is not None and 0 < topK < L:
        nz = np.nonzero(w)[0]
        if nz.size > 0 and nz.size <= 512:
            return _sparse_h(a, c, nz, w[nz])
    h = _full_convolve(w, a[::-1])[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return h, W


def _full_obj(z, beta, topK, n):
    a = _softmax(z)
    if a.sum() < 1e-15:
        return 1e100
    c = _full_convolve(a, a)
    w, F = _weights_and_value(c, beta, topK)
    sc = 2.0 * n * F
    if not np.isfinite(sc):
        return 1e100
    return float(sc)


def _full_grad(z, beta, topK, n):
    a = _softmax(z)
    if a.sum() < 1e-15:
        return np.zeros(n, dtype=np.float64)
    c = _full_convolve(a, a)
    w, _ = _weights_and_value(c, beta, topK)
    h, W = _h_from_weights(a, c, w, topK)
    grad = 4.0 * float(n) * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros(n, dtype=np.float64)
    return grad


def _profile_a(coeffs, X):
    coeffs = np.asarray(coeffs, dtype=np.float64).ravel()
    if coeffs.size != X.shape[1]:
        coeffs = np.zeros(X.shape[1], dtype=np.float64)
    z = X @ coeffs
    if not np.all(np.isfinite(z)):
        return np.ones(X.shape[0], dtype=np.float64) / float(X.shape[0])
    return _softmax(z)


def _profile_obj(coeffs, n, beta, topK, X):
    a = _profile_a(coeffs, X)
    if a.sum() < 1e-15:
        return 1e100
    c = _full_convolve(a, a)
    _, F = _weights_and_value(c, beta, topK)
    sc = 2.0 * n * F
    if not np.isfinite(sc):
        return 1e100
    return float(sc)


def _profile_jac(coeffs, n, beta, topK, X):
    a = _profile_a(coeffs, X)
    if a.sum() < 1e-15:
        return np.zeros(X.shape[1], dtype=np.float64)
    c = _full_convolve(a, a)
    w, _ = _weights_and_value(c, beta, topK)
    h, W = _h_from_weights(a, c, w, topK)
    grad_z = 4.0 * float(n) * a * (h - W)
    if not np.all(np.isfinite(grad_z)):
        return np.zeros(X.shape[1], dtype=np.float64)
    grad = np.dot(X.T, grad_z)
    if not np.all(np.isfinite(grad)):
        return np.zeros(X.shape[1], dtype=np.float64)
    return grad


def _measure_eval_time(a):
    t = time.perf_counter()
    _ = _full_convolve(a, a)
    dt = time.perf_counter() - t
    return max(1e-5, dt)


def _profile_lbfgs(n, X, a_init, deadline, rng):
    n = int(n)
    a_init = np.asarray(a_init, dtype=np.float64).ravel()
    if a_init.size != n or a_init.sum() < 1e-15:
        a_init = _default_sequence(n)
    best_a = a_init.copy()
    best_sc = _score_norm(best_a)
    c0 = _full_convolve(a_init, a_init)
    base_c = float(np.max(c0)) if c0.size else 1e-3
    if base_c <= 1e-15 or not np.isfinite(base_c):
        base_c = 1e-3
    eval_time = _measure_eval_time(a_init)
    dim = X.shape[1]

    inits = []
    z = np.log(np.maximum(a_init, 1e-300))
    z = z - float(np.mean(z))
    try:
        fit = np.linalg.lstsq(X, z, rcond=None)[0]
        if np.all(np.isfinite(fit)):
            inits.append(fit.astype(np.float64))
    except Exception:
        pass

    inits.append(np.zeros(dim, dtype=np.float64))

    for v1 in (1.0, 2.0, -1.0, -2.0):
        v = np.zeros(dim, dtype=np.float64)
        v[1] = v1
        inits.append(v)

    for v2 in (1.0, 2.0, -1.0, -2.0):
        v = np.zeros(dim, dtype=np.float64)
        v[1] = 0.5 * v2
        v[2] = v2
        inits.append(v)

    if dim >= 4:
        il = dim - 4
        ir = dim - 3
        ilh = dim - 2
        irh = dim - 1
        edge_specs = (
            (2.0, 0.0, 0.0, 0.0),
            (-2.0, 0.0, 0.0, 0.0),
            (0.0, 2.0, 0.0, 0.0),
            (0.0, -2.0, 0.0, 0.0),
            (0.0, 0.0, 2.0, 0.0),
            (0.0, 0.0, 0.0, 2.0),
            (2.0, 0.0, -1.0, 1.0),
            (-2.0, 0.0, 1.0, -1.0),
            (1.0, 1.0, 0.0, 0.0),
        )
        for lv, rv, lh, rh in edge_specs:
            v = np.zeros(dim, dtype=np.float64)
            v[il] = lv
            v[ir] = rv
            v[ilh] = lh
            v[irh] = rh
            inits.append(v)

    for _ in range(5):
        v = 0.45 * rng.normal(size=dim)
        inits.append(v)

    inits = inits[:12]
    settings = [
        (5e-3, 0),
        (2e-3, 0),
        (1e-3, 0),
        (5e-4, 128),
        (2e-4, 128),
        (1e-4, 128),
    ]

    for ci, coeffs0 in enumerate(inits):
        if time.monotonic() >= deadline:
            break
        coeffs = np.asarray(coeffs0, dtype=np.float64).copy()
        for bj, (frac, tk) in enumerate(settings):
            now = time.monotonic()
            if now >= deadline:
                break
            remaining = max(0.0, deadline - now)
            total_left = max(1, (len(inits) - ci) * max(1, len(settings) - bj))
            target = min(20.0, remaining / float(total_left))
            if target < 2.0:
                break
            maxfun = int(min(800, max(150, target / (eval_time * 3.5))))
            beta = frac * base_c
            if beta < 1e-20:
                beta = 1e-20
            try:
                res = minimize(
                    _profile_obj,
                    coeffs,
                    args=(n, beta, tk, X),
                    jac=_profile_jac,
                    method="L-BFGS-B",
                    options={
                        "maxiter": 220,
                        "maxfun": maxfun,
                        "ftol": 1e-18,
                        "gtol": 1e-12,
                        "maxls": 20,
                    },
                )
                if res.x is not None and np.all(np.isfinite(res.x)):
                    coeffs = np.asarray(res.x, dtype=np.float64).copy()
                    a = _profile_a(coeffs, X)
                    sc = _score_norm(a)
                    if np.isfinite(sc) and sc < best_sc:
                        best_a = a.copy()
                        best_sc = float(sc)
                        print(f"    profile improve: sc={best_sc:.10f}")
            except Exception:
                break
    return best_a, float(best_sc)


def _full_lbfgs(a_init, deadline, rng, settings=None):
    a_init = np.asarray(a_init, dtype=np.float64).ravel()
    n = int(a_init.size)
    if n < 2 or a_init.sum() < 1e-15:
        return a_init.copy(), _score_norm(a_init)

    if settings is None:
        settings = [
            (5e-3, 0),
            (2e-3, 0),
            (1e-3, 0),
            (5e-4, 0),
            (2e-4, 64),
            (1e-4, 128),
            (5e-5, 128),
            (2e-5, 192),
            (1e-5, 256),
            (5e-6, 256),
            (2e-6, 256),
        ]

    z = np.log(np.maximum(a_init, 1e-300))
    z = z - float(np.mean(z))
    best_a = a_init.copy()
    best_sc = _score_norm(best_a)

    c0 = _full_convolve(a_init, a_init)
    base_c = float(np.max(c0)) if c0.size else 1e-3
    if base_c <= 1e-15 or not np.isfinite(base_c):
        base_c = 1e-3

    eval_time = _measure_eval_time(a_init)

    for idx, (frac, tk) in enumerate(settings):
        now = time.monotonic()
        if now >= deadline:
            break
        remaining = max(0.0, deadline - now)
        total_left = max(1, len(settings) - idx)
        target = min(70.0, remaining / float(total_left))
        if target < 4.0:
            break
        factor = 2.6 if (tk is not None and 0 < tk < 2 * n) else 3.5
        maxfun = int(min(8000, max(250, target / (eval_time * factor))))
        beta = frac * base_c
        if beta < 1e-20:
            beta = 1e-20

        try:
            res = minimize(
                _full_obj,
                z,
                args=(beta, tk, n),
                jac=_full_grad,
                method="L-BFGS-B",
                options={
                    "maxiter": 350,
                    "maxfun": maxfun,
                    "ftol": 1e-18,
                    "gtol": 1e-12,
                    "maxls": 30,
                },
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                novo = np.asarray(res.x, dtype=np.float64).copy()
                a = _softmax(novo)
                sc = _score_norm(a)
                if np.isfinite(sc):
                    z = novo
                    if sc < best_sc:
                        best_a = a.copy()
                        best_sc = float(sc)
                        print(f"    full lbfgs improve: sc={best_sc:.10f}")
        except Exception:
            break
    return best_a, float(best_sc)


def _get_top_indices(c, topK, tol=1e-10):
    c = np.asarray(c, dtype=np.float64).ravel()
    L = c.size
    if L == 0:
        return np.array([], dtype=int)
    cm = float(np.max(c))
    tolabs = max(1e-15, float(tol) * abs(cm))
    close = np.where(c >= cm - tolabs)[0]
    kk = int(max(1, min(topK, L)))
    if close.size > 0 and close.size <= max(4, kk * 3):
        return close.astype(int)
    if kk >= L:
        return np.arange(L, dtype=int)
    part = np.argpartition(c, L - kk)[-kk:]
    return part.astype(int)


def _run_adam_sparse(a_init, deadline, rng, topK=8, tol=1e-10, lr=0.1):
    a_init = np.asarray(a_init, dtype=np.float64).ravel()
    n = int(a_init.size)
    if n < 2 or a_init.sum() < 1e-15:
        return a_init.copy(), _score_norm(a_init)

    best_a = a_init.copy()
    best_sc = _score_norm(best_a)

    z = np.log(np.maximum(a_init, 1e-300))
    z = z - float(np.mean(z))

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    lr_cur = float(lr)
    no_improve = 0

    while time.monotonic() < deadline:
        a = _softmax(z)
        c = _full_convolve(a, a)
        cur_sc = float(2.0 * n * np.max(c)) if c.size else float("inf")
        if np.isfinite(cur_sc) and cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        idx = _get_top_indices(c, int(topK), float(tol))
        if idx.size == 0:
            break
        wvals = np.full(idx.size, 1.0 / float(idx.size), dtype=np.float64)
        h, W = _sparse_h(a, c, idx, wvals)
        grad = 4.0 * float(n) * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        gnorm = float(np.linalg.norm(grad))
        if gnorm <= 1e-18:
            break

        t += 1
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        step = mh / (np.sqrt(vh) + eps)

        accepted = False
        s = lr_cur
        base = z
        for _ in range(6):
            zt = base - s * step
            zt = zt - float(np.mean(zt))
            at = _softmax(zt)
            st = _score_norm(at)
            if np.isfinite(st) and st < cur_sc - 1e-16:
                z = zt
                accepted = True
                break
            s *= 0.5

        if accepted:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 1500:
                z = np.log(np.maximum(best_a, 1e-300))
                z = z - float(np.mean(z))
                z = z + 0.02 * rng.normal(size=n)
                z = z - float(np.mean(z))
                lr_cur = max(0.005, 0.8 * lr_cur)
                no_improve = 0
                if lr_cur < 0.008:
                    break
        if t % 2000 == 0:
            lr_cur = max(0.005, 0.95 * lr_cur)

    return best_a, float(best_sc)


def _transfer_score(a, i, j, eps):
    if eps <= 0.0:
        return _score_norm(a)
    i = int(i)
    j = int(j)
    eps = float(eps)
    if eps > a[i] * (1.0 - 1e-12):
        return float("inf")
    b = a.copy()
    b[i] -= eps
    b[j] += eps
    if b[i] < -1e-12 or b[j] < -1e-12 or (not np.all(np.isfinite(b))):
        return float("inf")
    if np.any(b < 0.0):
        b = np.maximum(b, 0.0)
    s = float(b.sum())
    if s <= 1e-15:
        return float("inf")
    b = b / s
    return _score_norm(b)


def _rank_transfer_pairs(h, a, k=20):
    h = np.asarray(h, dtype=np.float64).ravel()
    a = np.asarray(a, dtype=np.float64).ravel()
    n = h.size
    if n == 0:
        return []
    k = int(max(4, min(k, n)))
    src = np.argsort(h)[::-1][:k]
    tgt = np.argsort(h)[:k]
    pairs = []
    for ii in src:
        i = int(ii)
        ai = float(a[i])
        if ai <= 1e-12:
            continue
        for jj in tgt:
            j = int(jj)
            if i == j:
                continue
            d = float(h[i]) - float(h[j])
            if d <= 0.0:
                continue
            potential = d * min(ai, 0.1)
            pairs.append((potential, i, j))
    pairs.sort(reverse=True)
    return pairs[:8]


def _golden_transfer(a, i, j, max_eps, iters, deadline):
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = float(max_eps)
    c1 = hi - inv_phi * (hi - lo)
    c2 = lo + inv_phi * (hi - lo)
    fc1 = _transfer_score(a, i, j, c1)
    fc2 = _transfer_score(a, i, j, c2)
    for _ in range(int(iters)):
        if time.monotonic() >= deadline:
            break
        if not np.isfinite(fc1):
            hi = c2
            c2 = c1
            fc2 = fc1
            c1 = hi - inv_phi * (hi - lo)
            fc1 = _transfer_score(a, i, j, c1)
            continue
        if not np.isfinite(fc2):
            hi = c1
            c1 = c2
            fc1 = fc2
            c2 = lo + inv_phi * (hi - lo)
            fc2 = _transfer_score(a, i, j, c2)
            continue
        if fc1 < fc2:
            hi = c2
            c2 = c1
            fc2 = fc1
            c1 = hi - inv_phi * (hi - lo)
            fc1 = _transfer_score(a, i, j, c1)
        else:
            lo = c1
            c1 = c2
            fc1 = fc2
            c2 = lo + inv_phi * (hi - lo)
            fc2 = _transfer_score(a, i, j, c2)
    eps = 0.5 * (lo + hi)
    sc = _transfer_score(a, i, j, eps)
    return float(eps), float(sc)


def _run_transfer(a_init, deadline, rng, topK=16, tol=1e-10, golden_iters=20, max_rounds=300):
    a_init = np.asarray(a_init, dtype=np.float64).ravel()
    n = int(a_init.size)
    if n < 2 or a_init.sum() < 1e-15:
        return a_init.copy(), _score_norm(a_init)

    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(best_a)
    no_improve = 0

    for _ in range(int(max_rounds)):
        now = time.monotonic()
        if now >= deadline:
            break
        c = _full_convolve(a, a)
        cur_sc = float(2.0 * n * np.max(c)) if c.size else float("inf")
        if np.isfinite(cur_sc) and cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        idx = _get_top_indices(c, int(topK), float(tol))
        if idx.size == 0:
            break
        wvals = np.full(idx.size, 1.0 / float(idx.size), dtype=np.float64)
        h, _ = _sparse_h(a, c, idx, wvals)
        pairs = _rank_transfer_pairs(h, a, k=20)
        improved = False
        for _, i, j in pairs[:5]:
            max_eps = 0.5 * float(a[i])
            if max_eps <= 1e-13:
                continue
            eps, sc = _golden_transfer(a, i, j, max_eps, golden_iters, time.monotonic() + 3.0)
            if np.isfinite(sc) and sc < best_sc - 1e-15:
                b = a.copy()
                b[i] -= eps
                b[j] += eps
                if b[i] < 0.0:
                    eps = 0.999 * float(a[i])
                    b = a.copy()
                    b[i] -= eps
                    b[j] += eps
                if np.min(b) < 0.0:
                    continue
                s = float(b.sum())
                if s <= 1e-15:
                    continue
                b = b / s
                a = b
                best_a = b.copy()
                best_sc = _score_norm(b)
                improved = True
                break
        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

    return best_a, float(best_sc)


def _random_restart_lbfgs(a_init, deadline, rng, scale=0.05):
    a_init = np.asarray(a_init, dtype=np.float64).ravel()
    n = int(a_init.size)
    if n < 2 or a_init.size != a_init.size or a_init.sum() < 1e-15:
        return a_init.copy(), _score_norm(a_init)
    z = np.log(np.maximum(a_init, 1e-300))
    z = z - float(np.mean(z))
    z = z + float(scale) * rng.normal(size=n)
    z = z - float(np.mean(z))
    a = _softmax(z)
    settings = [
        (5e-4, 64),
        (1e-4, 128),
        (2e-5, 192),
        (1e-5, 256),
    ]
    return _full_lbfgs(a, deadline, rng, settings=settings)


def propose_candidate(seed=42, budget_s=1800):
    """Main search routine. Returns a list of nonnegative floats."""
    t0 = time.monotonic()
    budget = float(max(60.0, budget_s))
    deadline = t0 + max(10.0, budget - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best = None
    best_sc = float("inf")

    try:
        parent = _as_vector(height_sequence_1)
    except Exception:
        parent = None
    if parent is None or parent.size < 16:
        parent = _default_sequence(4096)
    s = float(parent.sum())
    if s < 1e-12:
        parent = _default_sequence(4096)
        s = float(parent.sum())
    parent = parent / s

    best = parent.copy()
    best_sc = _score_norm(parent)
    print(f"parent: n={best.size}, score={best_sc:.10f}")

    n_parent = int(best.size)
    n_other = max(256, n_parent // 2) if n_parent > 512 else max(256, 2 * n_parent if n_parent < 1024 else n_parent)
    n_candidates = [n_parent]
    if n_other != n_parent and n_other <= 8192:
        n_candidates.append(n_other)

    # Quick profile candidates.
    for n in n_candidates:
        if time.monotonic() >= deadline:
            break
        for p in _initial_profiles(n):
            sc = _score_norm(p)
            if np.isfinite(sc) and sc < best_sc:
                best = np.asarray(p, dtype=np.float64).copy()
                best_sc = float(sc)
                print(f"profile initial: n={n}, score={best_sc:.10f}")

    # Phase 1: low-dimensional profile optimization.
    profile_end = min(deadline, t0 + 0.25 * budget)
    if time.monotonic() < profile_end:
        n_first = int(best.size)
        X = _profile_basis(n_first, 8)
        a, sc = _profile_lbfgs(n_first, X, best.copy(), min(profile_end, t0 + min(330.0, (profile_end - t0) * 0.78)), rng)
        if np.isfinite(sc) and sc < best_sc:
            best = a.copy()
            best_sc = float(sc)

        if time.monotonic() < profile_end:
            for n in n_candidates:
                if int(n) == int(best.size):
                    continue
                init_alt = _interp_profile(best, n)
                X_alt = _profile_basis(n, 8)
                a, sc = _profile_lbfgs(n, X_alt, init_alt, min(profile_end, time.monotonic() + 70.0), rng)
                if np.isfinite(sc) and sc < best_sc:
                    best = a.copy()
                    best_sc = float(sc)
                    print(f"profile alt: n={n}, score={best_sc:.10f}")
                if time.monotonic() >= profile_end:
                    break

    print(f"after profiles: n={best.size}, best={best_sc:.10f}, t={time.monotonic()-t0:.0f}s")

    # Phase 2: full L-BFGS continuation.
    full_end = min(deadline, t0 + 0.62 * budget)
    if time.monotonic() < full_end:
        a, sc = _full_lbfgs(best, full_end, rng)
        if np.isfinite(sc) and sc < best_sc:
            best = a.copy()
            best_sc = float(sc)

    # Phase 3: golden-section mass transfer.
    transfer_end = min(deadline, t0 + 0.80 * budget)
    if time.monotonic() < transfer_end:
        a, sc = _run_transfer(best, transfer_end, rng, topK=16, tol=1e-10, golden_iters=20, max_rounds=320)
        if np.isfinite(sc) and sc < best_sc:
            best = a.copy()
            best_sc = float(sc)

    # Phase 4: sparse top-peak Adam.
    adam_end = min(deadline, t0 + 0.93 * budget)
    if time.monotonic() < adam_end:
        remaining = max(10.0, adam_end - time.monotonic())
        dl1 = min(adam_end - 10.0, time.monotonic() + max(20.0, 0.45 * remaining))
        a, sc = _run_adam_sparse(best, dl1, rng, topK=4, tol=1e-10, lr=0.10)
        if np.isfinite(sc) and sc < best_sc:
            best = a.copy()
            best_sc = float(sc)
        if time.monotonic() < adam_end - 10.0:
            remaining = max(10.0, adam_end - time.monotonic())
            dl2 = min(adam_end, time.monotonic() + max(20.0, 0.55 * remaining))
            a, sc = _run_adam_sparse(best, dl2, rng, topK=16, tol=1e-8, lr=0.05)
            if np.isfinite(sc) and sc < best_sc:
                best = a.copy()
                best_sc = float(sc)

    # Phase 5: random restart if time permits.
    if time.monotonic() < deadline - 120.0:
        restart_end = min(deadline, time.monotonic() + max(30.0, 0.08 * budget))
        a, sc = _random_restart_lbfgs(best, restart_end, rng, scale=0.04)
        if np.isfinite(sc) and sc < best_sc:
            best = a.copy()
            best_sc = float(sc)

    # Phase 6: final fine LBFGS and transfer polish.
    if time.monotonic() < deadline - 20.0:
        final_settings = [
            (1e-4, 128),
            (5e-5, 192),
            (2e-5, 256),
            (1e-5, 256),
            (5e-6, 256),
        ]
        fin_end = min(deadline, time.monotonic() + max(20.0, 0.07 * budget))
        a, sc = _full_lbfgs(best, fin_end, rng, settings=final_settings)
        if np.isfinite(sc) and sc < best_sc:
            best = a.copy()
            best_sc = float(sc)

    if time.monotonic() < deadline - 5.0:
        fin_end = min(deadline, time.monotonic() + max(5.0, 0.04 * budget))
        a, sc = _run_transfer(best, fin_end, rng, topK=32, tol=5e-9, golden_iters=16, max_rounds=120)
        if np.isfinite(sc) and sc < best_sc:
            best = a.copy()
            best_sc = float(sc)

    # Final formatting.
    final = np.asarray(best, dtype=np.float64).ravel()
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.clip(final, 0.0, 1000.0)
    s = float(final.sum())
    if s < 1e-6:
        final = _default_sequence(2048)
    else:
        final = final / s

    final_sc = _score_norm(final)
    print(f"FINAL internal: n={final.size}, score={final_sc:.10f}, best_internal={best_sc:.10f}")

    if final.size <= 5000:
        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            print(f"FINAL evaluate_sequence: {eval_sc:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

    return [float(x) for x in final]
```
