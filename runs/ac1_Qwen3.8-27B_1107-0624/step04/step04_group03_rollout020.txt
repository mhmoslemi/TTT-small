def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _grad_wrt_a(a, w):
    n = len(a)
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    return h


def _grad_smooth_z(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    h = _grad_wrt_a(a, w)
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)

def _lbfgs_grad(z, beta, n):

def _make_raised_cosine_profile(n):

    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5)))

def _make_flat_top_profile(n, transition=0.1):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    edge = int(transition * n)
    if edge > 0:

        a[-(edge + 1):] = 0.5 * (1 - np.cos(np.pi * ramp[::-1]))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_polynomial_profile(n, order=2):

    else:

    a = np.maximum(a, 1e-15)

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

    best_score = _score_norm_fn(best_a)
    beta = beta_init
    lr = lr_init
    t = 0

    c = _full_convolve(_softmax_fn(z), _softmax_fn(z))
    max_c = float(c.max())

    while time.monotonic() < deadline:
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        h = _grad_wrt_a(a, w)
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        v = momentum * v - lr * grad
        z_new = z + v
        z_new = z_new - z_new.mean()
        z_new = np.clip(z_new, -60, 60)

        # Check if step improved (periodically)
        if t % eval_interval == 0:
            a_curr = _softmax_fn(z_new)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
                z = z_new
            else:
                # Revert and reduce step
                lr *= 0.7
                v *= 0.5
                z = z_new  # still take the step but reduce future steps
        else:
            z = z_new

        # Decay schedule
        if t % 200 == 0:
            beta *= beta_decay
            max_c = float(_full_convolve(_softmax_fn(z), _softmax_fn(z)).max())
            lr = max(0.0001, lr * lr_decay)
            if beta < 1e-10 * max_c:
                beta = 1e-10 * max_c

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _adam_gd(z_init, deadline, beta_init, lr=0.02, eval_interval=50):
    n = len(z_init)
    z = z_init.copy()
    best_a = _softmax_fn(z)
    best_score = _score_norm_fn(best_a)
    beta = beta_init

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8

    while time.monotonic() < deadline:
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        h = _grad_wrt_a(a, w)
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = z - z.mean()
        z = np.clip(z, -60, 60)

        if t % eval_interval == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            beta = max(1e-12, beta * 0.97)

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _active_set_polish(a_init, deadline, eval_interval=100):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(best_a)

    v = np.zeros(n)
    mom = 0.85
    lr = 0.003

    for t in range(1000000):
        if time.monotonic() > deadline:
            break

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Active set: lags within 0.5% of max
        threshold = cm * 0.995
        active = c >= threshold
        n_active = int(np.sum(active))
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        W = float(np.dot(w, c))
        h = _grad_wrt_a(a, w)
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        v = mom * v - lr * grad
        z = z + v
        z = z - z.mean()
        z = np.clip(z, -60, 60)

        if t % eval_interval == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            else:
                lr *= 0.8
                v *= 0.3

        if t % 500 == 0:
            lr = max(0.0001, lr * 0.95)
            mom = min(0.95, mom + 0.001)

    return best_a, best_score


def _run_lbfgs_phased(a_init, deadline, max_time=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if max_time is None:
        max_time = min(300.0, max(1.0, deadline - time.monotonic()))

    # Aggressive beta schedule
    beta_fracs = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005,
                  0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    betas = [b * max_c0 for b in beta_fracs]

    per_level = max_time / len(betas)

    for beta in betas:
        if time.monotonic() >= deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level * 1.5)
        try:
            result = minimize(
                _lbfgs_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_grad,
                options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 2000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x.copy()
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new
                    # Update max_c for next beta
                    c_new = _full_convolve(a_new, a_new)
                    max_c0 = float(c_new.max())
                    # Rescale remaining betas
                    remaining_idx = betas.index(beta) + 1
                    for i in range(remaining_idx, len(betas)):
                        betas[i] = (betas[i] / max_c0) * float(_full_convolve(a_new, a_new).max())
        except Exception:
            pass

    return best_a, best_score

def _random_restart(a_init, noise_level, rng):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    z = z + noise_level * rng.normal(size=n)
    z = np.clip(z, -60, 60)
    z = z - z.mean()
    return _softmax_fn(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n_target = 4096
    if parent is not None:
        n_target = len(parent)
        print(f"Parent: n={n_target}, score={_score_fn(parent):.10f}")
    else:
        print(f"No parent, using n={n_target}")

    # --- Phase 0: Multiple initializations (30s) ---
    phase0_end = min(deadline, t_start + 30.0)
    candidates = []

    if parent is not None:
        sc = _score_norm_fn(parent)
        candidates.append((sc, parent.copy(), "parent"))
        _update_best(parent, sc, "parent")

    n_use = n_target
    profiles_list = [
        ("cos_sq", _make_cosine_sq_profile(n_use)),
        ("gauss_0.3", _make_gaussian_profile(n_use, 0.3)),
        ("gauss_0.2", _make_gaussian_profile(n_use, 0.2)),
        ("gauss_0.25", _make_gaussian_profile(n_use, 0.25)),
        ("raised_cos", _make_raised_cosine_profile(n_use)),
        ("flat_top", _make_flat_top_profile(n_use, 0.1)),
        ("poly2", _make_polynomial_profile(n_use, 2)),
        ("poly4", _make_polynomial_profile(n_use, 4)),
    ]

    for name, p in profiles_list:
        if time.monotonic() > phase0_end:
            break
        # Interpolate to target length if needed
        if len(p) != n_target:
            p = _interp_profile(p, n_target)
        sc = _score_norm_fn(p)
        candidates.append((sc, p, name))
        _update_best(p, sc, f"init_{name}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}, candidates={len(candidates)}")
    if len(candidates) > 0:
        print(f"  Top candidate: {candidates[0][2]} with score {candidates[0][0]:.10f}")

    # --- Phase 1: L-BFGS on best candidate (350s) ---
    phase1_end = min(deadline, t_start + 380.0)
    if best_seq is not None and time.monotonic() < phase1_end:
        per_time = min(300.0, phase1_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _run_lbfgs_phased(best_seq, min(phase1_end, time.monotonic() + per_time),
                                                 max_time=per_time)
            _update_best(p_best, sc_best, "lbfgs_p1")

    # Try second best if time allows
    if time.monotonic() < phase1_end - 30:
        for sc, p, label in candidates:
            if time.monotonic() > phase1_end - 15:
                break
            if len(p) != len(best_seq):
                continue
            if label == "parent" and sc >= best_score:
                continue
            per_time = min(60.0, phase1_end - time.monotonic() - 10)
            if per_time > 15:
                p_best, sc_best = _run_lbfgs_phased(p, min(phase1_end - 5, time.monotonic() + per_time),
                                                     max_time=per_time)
                _update_best(p_best, sc_best, f"lbfgs_{label}")
            if time.monotonic() > phase1_end - 20:
                break

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Momentum GD (350s) ---
    phase2_end = min(deadline, t_start + 730.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        c = _full_convolve(best_seq, best_seq)
        max_c = float(c.max())
        beta_init = 0.005 * max_c
        per_time = min(300.0, phase2_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _momentum_gd(z, min(phase2_end - 5, time.monotonic() + per_time),
                                            beta_init, lr_init=0.008, momentum=0.92)
            _update_best(p_best, sc_best, "momentum_p2")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Adam GD (300s) ---
    phase3_end = min(deadline, t_start + 1030.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        c = _full_convolve(best_seq, best_seq)
        max_c = float(c.max())
        beta_init = 0.003 * max_c
        per_time = min(250.0, phase3_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _adam_gd(z, min(phase3_end - 5, time.monotonic() + per_time),
                                        beta_init, lr=0.015)
            _update_best(p_best, sc_best, "adam_p3")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Random restarts (250s) ---
    phase4_end = min(deadline, t_start + 1280.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        n_restarts = 4
        per_restart = min(60.0, (phase4_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 10:
                break
            noise_level = 0.03 + 0.02 * i
            p_start = _random_restart(best_seq, noise_level, rng)
            z = np.log(np.maximum(p_start, 1e-15))
            z = z - z.mean()
            c = _full_convolve(p_start, p_start)
            max_c = float(c.max())
            beta_init = 0.003 * max_c
            per_dl = min(phase4_end - 5, time.monotonic() + per_restart)
            p_best, sc_best = _momentum_gd(z, per_dl, beta_init, lr_init=0.006, momentum=0.9)
            _update_best(p_best, sc_best, f"restart{i}")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Active set polishing (200s) ---
    phase5_end = min(deadline, t_start + 1480.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        per_time = min(150.0, phase5_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _active_set_polish(best_seq, min(phase5_end - 5, time.monotonic() + per_time))
            _update_best(p_best, sc_best, "active_set_p5")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: L-BFGS on different lengths (150s) ---
    phase6_end = min(deadline, t_start + 1630.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        n_curr = len(best_seq)
        # Try a larger length
        if n_curr <= 4096:
            target_n = min(8192, n_curr * 2)
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            if sc_interp < best_score:
                _update_best(p_interp, sc_interp, f"interp_n{target_n}")
            if time.monotonic() < phase6_end - 30:
                per_time = min(80.0, phase6_end - time.monotonic() - 20)
                if per_time > 20:
                    z = np.log(np.maximum(p_interp, 1e-15))
                    z = z - z.mean()
                    c = _full_convolve(p_interp, p_interp)
                    max_c = float(c.max())
                    beta_init = 0.005 * max_c
                    p_best, sc_best = _momentum_gd(z, min(phase6_end - 10, time.monotonic() + per_time),
                                                    beta_init, lr_init=0.005, momentum=0.9)
                    _update_best(p_best, sc_best, f"opt_n{target_n}")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final fine polish (remaining) ---
    phase7_end = deadline
    if best_seq is not None and time.monotonic() < phase7_end:
        remaining = phase7_end - time.monotonic()
        if remaining > 30:
            # Very fine L-BFGS
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            c = _full_convolve(best_seq, best_seq)
            max_c = float(c.max())
            betas_fine = [max_c * 1e-5, max_c * 5e-6, max_c * 2e-6, max_c * 1e-6]
            per_level = min(30.0, (remaining - 10) / len(betas_fine))

            for beta in betas_fine:
                if time.monotonic() >= phase7_end - 5:
                    break
                level_dl = min(phase7_end - 3, time.monotonic() + per_level)
                n = len(z)
                try:
                    result = minimize(
                        _lbfgs_obj, z, args=(beta, n),
                        method='L-BFGS-B', jac=_lbfgs_grad,
                        options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-14,
                                 'maxls': 30, 'maxfun': 500}
                    )
                    if result.x is not None and np.all(np.isfinite(result.x)):
                        z_new = result.x.copy()
                        a_new = _softmax_fn(z_new)
                        s_new = _score_norm_fn(a_new)
                        if np.isfinite(s_new) and s_new < best_score:
                            best_score = s_new
                            best_seq = a_new.copy()
                            z = z_new
                            print(f"  Fine polish improvement: {best_score:.10f}")
                except Exception:
                    pass

        # Final active set with very tight threshold
        if time.monotonic() < phase7_end - 10:
            per_time = min(40.0, phase7_end - time.monotonic() - 5)
            if per_time > 10:
                p_best, sc_best = _active_set_polish(best_seq, min(phase7_end - 2, time.monotonic() + per_time))
                _update_best(p_best, sc_best, "final_active")

    print(f"Phase 7 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            # Don't normalize - the evaluator handles it, but let's keep values reasonable
            pass
        else:
            final_seq = _make_cosine_sq_profile(4096)

        sc_final = _score_fn(final_seq)
        print(f"FINAL internal score: {sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned non-finite, using internal: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_cosine_sq_profile(4096)
    print(f"FALLBACK: n=4096, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
