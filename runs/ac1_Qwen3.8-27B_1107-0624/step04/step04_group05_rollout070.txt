    N = _next_pow2(2 * n - 1)
    fa = rfft(np.asarray(a, dtype=np.float64), N)
    c = irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score_direct(a):
    n = len(a)
    c = _full_convolve_fft(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def _make_power_law(n, alpha):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** alpha
    s = a.sum()
    return a / s


def _smooth_max_grad(c, beta):
    """Returns (smooth_max_value, gradient_weights) for soft-max approximation."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 0.0))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    val = mval + beta * np.log(s)
    w = ex / s
    return val, w


def _adam_refine(p0, deadline, rng, lr=0.02, beta_frac=0.1, beta_decay=0.95,
                 no_imp_limit=500, max_iters=50000):
    """Adam optimizer in logit-space to minimize the smooth-max proxy of the score."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-15)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = _full_convolve_fft(p0)
    maxc = float(c.max())
    beta = max(1e-5, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0

    N = _next_pow2(2 * n - 1)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Compute smooth max and its gradient w.r.t. c
        _, w = _smooth_max_grad(c, beta)

        # Gradient of c_k = sum_i p[i]*p[k-i] w.r.t. p[j]:
        # dc/dp[j] = sum_k w[k] * (p[k-j] + p[j-k]) (valid terms)
        # Use FFT: grad[j] = 2 * (w * p)[j] restricted to valid range
        # More precisely: grad_p = 2 * cross_correlation(w_conv, p)
        # where w_conv is the convolution kernel weights

        # The convolution c = conv(p, p), so dc/dp = conv(w_total, p) + conv(w_total_rev, p)
        # where w_total = 2*w (symmetric since c is the self-convolution)
        # But c[k] = sum_i p[i]*p[k-i], so dc/dp[j] = sum_k w[k]*(p[k-j] + p[k+j-n+1... ])
        # Simpler: use the fact that d(conv(p,p))[j]/dp = conv(2*w_pad, p) appropriately

        # Actually: c[k] = sum_i p[i] p[k-i]
        # dc[k]/dp[j] = p[k-j] + p[j-k] (when valid)
        # dc/dp[j] = sum_k w[k] * (p[k-j] (if 0<=k-j<n) + p[j-k] (if 0<=j-k<n))
        # = sum_k w[k]*p[k-j] (k-j in [0,n-1]) + sum_k w[k]*p[j-k] (j-k in [0,n-1])
        # First term: sum_{k' = j}^{j+2n-2} w[k']*p[k'-j] where k'-j in [0,n-1]
        #   = (w * p)[j + (n-1)] in the full convolution sense... let me use FFT.

        # Use the identity: if c = conv(p, p), then grad_p[j] = 2 * corr(w, p)[j]
        # where corr(w,p)[j] = sum_k w[k]*p[k+j] (cross-correlation)
        # This is equivalent to conv(w_flipped, p) at the right offset.

        fw = rfft(w, N)
        fp = rfft(p, N)
        # cross-correlation: corr(w, p)[d] = sum_k w[k] * p[k-d]
        # = irfft(conj(rfft(w)) * rfft(p))
        cp = irfft(np.conj(fw) * fp, N)
        # cp[d] = sum_k w[k] * p[k-d] for d = 0, ..., N-1 (with wraparound)
        # We want grad_p[j] = sum_k w[k]*(p[k-j] + p[j-k])
        # = cp[j] + sum_k w[k]*p[j-k] = cp[j] + cp[-j mod N]... but for our range:
        # sum_k w[k]*p[j-k]: let m=j-k, = sum_m w[j-m]*p[m] for valid j-m
        # This is conv(w, p)[j] = irfft(fw * fp)[j]

        conv_wp = irfft(fw * fp, N)

        grad_p = 2.0 * (cp[n - 1:n - 1 + n] + conv_wp[:n])
        # Actually let me be more careful.
        # cp[d] for d=0,...,2n-2 gives sum_k w[k]*p[k-d] (proper cross-correlation without wraparound if N is large enough)
        # conv_wp[d] for d=0,...,2n-2 gives sum_k w[k]*p[d-k] = sum_k w[k]*p[k-(d)]...
        # Hmm, let me just use: grad_p[j] = cp[j] + conv_wp[j] where both are taken at index j
        # But we need to check the indexing carefully.

        # Let me use a cleaner approach:
        # c[k] = sum_{i: max(0,k-n+1)<=i<=min(k,n-1)} p[i]*p[k-i]
        # dc[k]/dp[j] = p[k-j] (if 0<=k-j<n) + p[j-k] (if 0<=j-k<n)
        # grad_p[j] = sum_k w[k] * dc[k]/dp[j]
        #            = sum_k w[k]*p[k-j] (valid) + sum_k w[k]*p[j-k] (valid)
        # First sum: let d=k-j, d ranges over [j-(n-1), j+(2n-2-j)] ∩ [0, 2n-2] for valid k
        #   = sum_d w[d+j]*p[d] for d in [0,n-1] and d+j in [0,2n-2]
        #   This is (w_shifted_correlated with p) at position j
        #   = (cross-correlation of w and p, offset by j)
        #   In FFT terms: irfft(conj(fw)*fp)[j] gives sum_k w[k]*p[k-j]... 
        #   Actually irfft(conj(FW)*FP)[d] = sum_k w[k]*p[k+d]... no.

        # Let me just do it the simple way for correctness:
        # irfft(conj(fw) * fp)[d] = sum_k conj(w)[k] * p[(k+d) mod N]... 
        # No, the convolution theorem says: irfft(FW * FP)[n] = sum_k w[k]*p[n-k]
        # And irfft(conj(FW) * FP)[n] = sum_k w[-k]*p[n-k] = sum_k w[k]*p[n+k] (by substitution)

        # So: conv_wp[n] = sum_k w[k]*p[n-k] (standard convolution)
        #     cp[n] = sum_k w[k]*p[n+k] (cross-correlation, positive lag)

        # First term of grad: sum_k w[k]*p[k-j] for valid k,j
        #   Let m = k-j: sum_m w[m+j]*p[m] for m in [0,n-1]
        #   This is sum_m p[m]*w[m+j] = (p * w_flipped)[j]... 
        #   = sum_m p[m]*w[m+j]
        #   In conv terms: conv(p, w_prime)[j] where w_prime[d] = w[-d] = w[2n-2-d]...
        #   This is getting too complicated analytically. Let me use a different approach.

        # DUAL approach: gradient via cross-correlation with the convolution kernel
        # The score is 2*n*max(c)/(sum p)^2. With sum=1 fixed (projection),
        # d(score)/d(p[j]) = 2n * (w[k_max]*p[k_max-j] + w[k_max]*p[j-(k_max-n+1)]) / 1 (for the max term)
        # For smooth max: d(score)/d(p[j]) = 2n * sum_k w[k] * (p[k-j]*1[k-j<n] + p[j-k]*1[j-k<n])
        # The term p[k-j] with 0<=k-j<n means k in [j, j+n-1]:
        #   sum_{k=j}^{j+n-1} w[k]*p[k-j] (where k-j in [0,n-1])
        #   = sum_{m=0}^{n-1} w[m+j]*p[m] where m+j <= 2n-2
        # The term p[j-k] with 0<=j-k<n means k in [j-(n-1), j]:
        #   sum_{k=max(0,j-n+1)}^{j} w[k]*p[j-k]
        #   = sum_{m=0}^{n-1} w[j-m]*p[m] where j-m >= 0

        # So grad_p[j] = sum_{m=0}^{n-1} p[m]*(w[m+j] + w[j-m]) (with w out of range = 0)
        # This is a cross-correlation of p with the "combined kernel" h_j = w[m+j]+w[j-m]
        # But h depends on j, so it's not a simple convolution.

        # SIMPLEST correct approach: use the fact that c = conv(p,p)
        # and dc/dp is a linear operator that can be implemented as:
        # grad_p = 2 * Ifft(conj(F_p) .* something)... no.

        # Let me use the most straightforward method:
        # grad_p[j] = d(sum_k w[k]*c[k])/dp[j]
        # c[k] = sum_i p[i]*p[k-i]
        # c[k]/dp[j] = p[k-j] (if 0<=k-j<=n-1) + p[j-k] (if 0<=j-k<=n-1)
        # So grad_p[j] = sum_k w[k]*p[k-j]*1[0<=k-j<n] + sum_k w[k]*p[j-k]*1[0<=j-k<n]

        # First sum: sum_k w[k]*p[k-j] where k-j in [0,n-1], i.e., k in [j, j+n-1] ∩ [0, 2n-2]
        #   = sum_{m=0}^{n-1} w[m+j]*p[m] (with w out of [0,2n-2] treated as 0)
        # This is: for each m, multiply p[m] by w[m+j], sum over m.
        # = dot(p, w[j:j+n]) (where w is of length 2n-1, w[j:j+n] is the slice)

        # Second sum: sum_k w[k]*p[j-k] where j-k in [0,n-1], i.e., k in [j-(n-1), j] ∩ [0, 2n-2]
        #   = sum_{m=0}^{n-1} w[j-m]*p[m] (with w out of [0,2n-2] treated as 0)
        # = dot(p, w_rev[j]) where w_rev[j-m] = w[j-m]

        # So: grad_p[j] = dot(p, w[j:j+n]) + dot(p, w[j-n+1:j+1]) (with appropriate clamping)
        # This is: grad_p = conv(w, p) + conv(w_flipped, p) evaluated at the right indices
        # conv(w,p)[j] = sum_k w[k]*p[j-k] for k in [0,2n-2], j-k in [0,n-1]
        #   = sum_{m=max(0,j-2n+2)}^{min(j,n-1)} w[j-m]*p[m]
        # This is the second sum! So second_sum = conv(w,p)[j] for j in [0,n-1].

        # First sum: sum_{m=0}^{n-1} w[m+j]*p[m]
        #   = sum_{k=j}^{j+n-1} w[k]*p[k-j]
        #   Let's define v = w reversed: v[k] = w[2n-2-k]
        #   sum_{m=0}^{n-1} w[m+j]*p[m] = sum_{m=0}^{n-1} v[2n-2-m-j]*p[m]
        #   = sum_{m=0}^{n-1} p[m]*v[(2n-2-j)-m]
        #   = conv(v, p)[(2n-2-j)] restricted to valid range... 
        #   Actually conv(v,p)[d] = sum_k v[k]*p[d-k] for valid k,d-k
        #   = sum_{k: 0<=k<2n-1, 0<=d-k<n} v[k]*p[d-k]
        #   For d = 2n-2-j: k in [max(0, 2n-2-j-(n-1)), min(2n-2-j, 2n-2)] ∩ [0,2n-2]
        #   = sum_k v[k]*p[2n-2-j-k]
        #   Let m = 2n-2-j-k: k = 2n-2-j-m
        #   = sum_m v[2n-2-j-m]*p[m] for m in [0,n-1] and 2n-2-j-m in [0,2n-2]
        #   = sum_m w[m+j]*p[m] ✓ (since v[2n-2-(m+j)] = w[m+j])

        # So first_sum[j] = conv(v, p)[2n-2-j] where v = w[::-1]

        # Therefore: grad_p[j] = conv(v, p)[2n-2-j] + conv(w, p)[j] for j in [0, n-1]
        # where v = w[::-1] (reversed w)

        # Compute via FFT:
        fv = rfft(w[::-1], N)
        conv_wp2 = irfft(fw * fp, N)  # conv(w, p)
        conv_vp2 = irfft(fv * fp, N)  # conv(v, p)

        # conv(w,p)[j] for j in [0, n-1]: take indices 0 to n-1
        # conv(v,p)[2n-2-j] for j in [0, n-1]: take indices 2n-2 down to n-1, reverse
        term2 = conv_wp2[:n]
        term1 = conv_vp2[2*n-2:n-1:-1]

        grad_p = 2.0 * (term1 + term2)

        # Project gradient to simplex (sum p = 1 constraint):
        # The Lagrangian gives: effective gradient = grad_p - mean(grad_p) * p / sum(p)
        # Actually for the constraint sum(p)=1, the projected gradient is:
        # gz = p * (grad_p - dot(p, grad_p)/dot(p,p))... no.
        # For softmax parameterization z = log(p), the update is on z.
        # d(score)/dz[j] = p[j] * (d(score)/dp[j] - sum_k p[k]*d(score)/dp[k])
        # (The constraint is handled by the softmax: sum of p = 1 automatically)

        dot_p_grad = float(np.dot(p, grad_p))
        gz = p * (grad_p - dot_p_grad)
        gz -= float(np.mean(gz))  # center for numerical stability

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.max(z))
        e = np.exp(z)
        p = e / e.sum()
        p = np.maximum(p, 1e-15)
        p = p / p.sum()

        c = _full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                beta = max(1e-6, beta * beta_decay)
            if no_imp > no_imp_limit:
                # Restart with noise
                ns = 0.05
                z = best_z + ns * rng.normal(size=n)
                z = np.clip(z, -50, 50)
                z -= float(np.max(z))
                e = np.exp(z)
                p = e / e.sum()
                z_best = z.copy()
                c = _full_convolve_fft(p)
                maxc = float(c.max())
                beta = max(1e-5, beta_frac * maxc)
                m.fill(0.0)
                v.fill(0.0)
                t_adam = 0
                no_imp = 0

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}]")

    print(f"Started. Budget={budget_s}s, deadline in {deadline-t_start:.1f}s")

    # Phase 1: Grid search over power law parameters
    phase1_end = t_start + 120.0
    print("Phase 1: Power law grid search")

    ns_list = [256, 512, 1024, 2048, 4096]
    alphas_list = np.arange(1.0, 25.0, 0.25)

    for n in ns_list:
        for alpha in alphas_list:
            if time.monotonic() > phase1_end:
                break
            a = _make_power_law(n, alpha)
            sc = _score_direct(a)
            if sc < best_score:
                best_seq = a.copy()
                best_score = sc
                print(f"  n={n}, alpha={alpha:.2f}: {sc:.8f}")

    print(f"Phase 1 done. Best: {best_score:.10f}")

    # Phase 2: Finer grid around best
    if best_seq is not None:
        n_best = best_seq.size
        # Find the alpha that produced best_seq
        # Re-search with finer grid
        phase2_end = t_start + 300.0
        for n_try in [n_best // 2, n_best, n_best * 2]:
            if n_try < 128 or n_try > 8192:
                continue
            for alpha in np.arange(1.0, 25.0, 0.1):
                if time.monotonic() > phase2_end:
                    break
                a = _make_power_law(n_try, alpha)
                sc = _score_direct(a)
                if sc < best_score:
                    best_seq = a.copy()
                    best_score = sc
                    print(f"  Phase2: n={n_try}, alpha={alpha:.2f}: {sc:.8f}")

        # Also try modified profiles: truncated power law
        for n_try in [n_best, n_best * 2]:
            if n_try > 8192:
                continue
            for cap_frac in [0.1, 0.2, 0.3, 0.5]:
                if time.monotonic() > phase2_end:
                    break
                a = _make_power_law(n_try, 5.0)
                # Truncate: set top cap_frac*n elements to their min value
                n_cap = max(1, int(n_try * cap_frac))
                threshold = a[-n_cap]
                a[-n_cap:] = threshold
                a = a / a.sum()
                sc = _score_direct(a)
                if sc < best_score:
                    best_seq = a.copy()
                    best_score = sc
                    print(f"  Phase2: n={n_try}, cap={cap_frac}: {sc:.8f}")

    print(f"Phase 2 done. Best: {best_score:.10f}")

    # Phase 3: Adam refinement
    if best_seq is not None and time.monotonic() < deadline - 60:
        phase3_end = deadline - 60.0
        n_cur = best_seq.size
        print(f"Phase 3: Adam refinement, n={n_cur}")

        # Run Adam with different betas
        per_adam = min(180.0, phase3_end - time.monotonic() - 10)
        if per_adam > 30:
            p_r, sc_r = _adam_refine(best_seq, time.monotonic() + per_adam, rng,
                                     lr=0.01, beta_frac=0.05, beta_decay=0.97,
                                     no_imp_limit=1000)
            _update(p_r, sc_r, 'adam1')

        # Try with a different learning rate
        if time.monotonic() < phase3_end - 30:
            per2 = min(120.0, phase3_end - time.monotonic() - 10)
            if per2 > 20:
                p_r2, sc_r2 = _adam_refine(best_seq, time.monotonic() + per2, rng,
                                           lr=0.005, beta_frac=0.03, beta_decay=0.98,
                                           no_imp_limit=2000)
                _update(p_r2, sc_r2, 'adam2')

    print(f"Phase 3 done. Best: {best_score:.10f}")

    # Phase 4: Try larger n with interpolation
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_cur = best_seq.size
        for n_new in [n_cur * 2, n_cur * 4]:
            if n_new > 16384 or time.monotonic() > deadline - 60:
                break
            # Interpolate
            x0 = np.linspace(0, 1, n_cur)
            x1 = np.linspace(0, 1, n_new)
            p_int = np.interp(x1, x0, best_seq)
            p_int = np.maximum(p_int, 1e-15)
            p_int = p_int / p_int.sum()
            sc = _score_direct(p_int)
            _update(p_int, sc, f'interp_n{n_new}')

            # Quick Adam on interpolated
            if time.monotonic() < deadline - 30:
                per = min(60.0, deadline - time.monotonic() - 20)
                if per > 15:
                    p_e, sc_e = _adam_refine(p_int, time.monotonic() + per, rng,
                                             lr=0.005, beta_frac=0.04, beta_decay=0.98,
                                             no_imp_limit=500)
                    _update(p_e, sc_e, f'escalate_n{n_new}')

    # Final: verify and return
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
            # Rescale so sum is a "nice" value to avoid floating point issues
            final = final * 100.0  # sum = 100, well within [0.01, 1000*n]
        else:
            final = _make_power_law(1024, 5.0)

        exact_sc = _score_direct(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size} sum={final.sum():.2f}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_power_law(1024, 5.0)
    print("FALLBACK: power law n=1024 alpha=5")
    return [float(x) for x in fb]
