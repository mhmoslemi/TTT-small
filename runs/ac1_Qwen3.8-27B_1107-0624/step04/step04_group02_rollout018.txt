def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return _score_norm_fn(a) / (s * s)


def _project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - target_sum)).astype(int)
    if len(rho) == 0 or rho[0] == 0:
        return np.full(n, target_sum / n)
    rho = rho[0]
    theta = (cssv[rho - 1] - target_sum) / rho
    proj = np.maximum(v - theta, 0.0)
    # Fix numerical issues
    proj_sum = proj.sum()
    if proj_sum > 0:
        proj = proj * (target_sum / proj_sum)
    else:
        proj = np.full(n, target_sum / n)
    return proj

def _active_set_refine(a_init, deadline, rng, k_active=8):
    """Refine by focusing on the active set of maximum lags."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()

    # Identify active lags

        iterations += 1
        c = _full_convolve(a, a)
        cm = c.max()
        
        # Get top K active lags
        top_k = min(k_active, len(c))
        top_idx = np.argsort(c)[-top_k:]
        
        # Subgradient from active set (uniform weights on active)
        grad = np.zeros(n)
        for k_val in top_idx:
            k_val = int(k_val)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                # (B_k a)_j = a[k-j] for j in [lo, hi]
                # k-j ranges from k-hi to k-lo
                grad[lo:hi + 1] += a[k_val - hi:k_val - lo + 1][::-1]
        grad *= 2.0 / top_k
        
        # Projected gradient step with backtracking
        lr = 0.1 / (np.linalg.norm(grad) + 1e-10)
        for attempt in range(10):
            a_trial = a - lr * grad
            a_trial = _project_simplex(a_trial, 1.0)
            sc = _score_norm_fn(a_trial)
            if sc < cm:  # Feasible descent (reducing the max)
                break
            lr *= 0.5
        
        a = a_trial
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Check convergence
        if iterations > 50 and cm - best_score < 1e-8:
            break

def _structured_local_search(a_init, deadline, rng, n_trials=20000):
    """Local search with structured perturbations targeting high-convolution regions."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    # Identify which regions of a contribute most to the peak
    c = _full_convolve(a, a)
    top_lag = int(np.argmax(c))
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 2:
            break
        
        # Strategy: perturb elements that contribute to the peak lag
        if it % 3 == 0:
            # Perturb elements near the peak contribution
            k = top_lag
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            width = hi - lo + 1
            if width > 0:
                idx = lo + rng.integers(0, width)
                a[idx] *= rng.uniform(0.7, 1.3)
        elif it % 3 == 1:
            # Random block perturbation
            block_size = rng.integers(2, min(50, n // 8))
            start = rng.integers(0, n - block_size)
            a[start:start + block_size] *= rng.uniform(0.9, 1.1, size=block_size)
        else:
            # Swap-based perturbation: redistribute mass between two positions
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0 and a[j] > 0:
                mass = a[i] * rng.uniform(0.1, 0.3)
                a[i] -= mass
                a[j] += mass
                a = np.maximum(a, 0)
        
        a = _project_simplex(a, 1.0)
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            c = _full_convolve(a, a)
            top_lag = int(np.argmax(c))
        else:
            # Revert bad moves (with small probability of accepting)
            if rng.random() > 0.02:
                a = best_a.copy()
    
    return best_a, best_score


def _make_profile(n, profile_type, **kwargs):
    """Generate various parametric profiles."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    
    if profile_type == 'linear':
        slope = kwargs.get('slope', 1.5)
        a = 1.0 + slope * (t - 0.5)
    elif profile_type == 'quadratic':
        a1 = kwargs.get('a1', 0.8)
        a2 = kwargs.get('a2', 0.3)
        a = 1.0 + a1 * u + a2 * u ** 2
    elif profile_type == 'cubic':
        a1 = kwargs.get('a1', 0.8)
        a3 = kwargs.get('a3', -0.4)
        a = 1.0 + a1 * u + a3 * u ** 3
    elif profile_type == 'gaussian':
        width = kwargs.get('width', 0.3)
        a = np.exp(-u ** 2 / (2.0 * width ** 2))
    elif profile_type == 'raised_cosine':
        alpha = kwargs.get('alpha', 1.0)
        a = 0.5 * (1.0 + np.cos(np.pi * u * alpha))
    elif profile_type == 'power':
        p = kwargs.get('p', 2.0)
        a = (1.0 - np.abs(u)) ** p
    elif profile_type == 'sinc_like':
        width = kwargs.get('width', 0.2)
        x = u / width
        a = np.sinc(x) ** 2  # squared sinc
        a = np.maximum(a, 0)
    elif profile_type == 'two_gaussian':
        center1 = kwargs.get('c1', -0.3)
        center2 = kwargs.get('c2', 0.3)
        w = kwargs.get('width', 0.15)
        a = np.exp(-(u - center1) ** 2 / (2 * w ** 2)) + np.exp(-(u - center2) ** 2 / (2 * w ** 2))
    else:
        a = np.ones(n) * 0.5 * (1.0 + np.cos(np.pi * u))
    
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _lbfgs_smooth(z, n, beta):
    """Objective and gradient for L-BFGS on log-parameterized variable."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    cm = c.max()
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    obj = 2.0 * n * (cm + beta * np.log(s) if s > 0 else cm)
    
    # Gradient: 2*n * sum_k w_k * (B_k a)
    # (B_k a)_j = a[k-j] valid
    N = _next_pow2(n + len(c) - 1)
    Na = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    cross = np.fft.irfft(np.conj(Na) * W, N)
    grad_a = 2.0 * n * cross[:n]
    
    # Chain rule: z -> a via softmax-like (exp/log param)
    # a_j = exp(z_j) / sum(exp(z_k))
    # da_j/dz_i = a_j * (delta_ji - a_i)

    if not np.all(np.isfinite(grad_z)):
        grad_z = np.zeros_like(z)
    
    return obj, grad_z

        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")
    
    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    n_base = 4096
    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_parent}, score_norm={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
        n_base = n_parent
    else:
        parent = _make_profile(n_base, 'raised_cosine', alpha=1.0)
        sc = _score_norm_fn(parent)
        update_best(parent, sc, "default_parent")
        n_base = n_base
    
    # --- Phase 1: Diverse initialization (60s) ---
    phase1_end = min(deadline, t_start + 60.0)
    candidates = [(best_score, best_seq.copy(), "parent")]
    
    profiles_to_try = [
        ('raised_cosine', {'alpha': 0.7}),
        ('raised_cosine', {'alpha': 1.3}),
        ('raised_cosine', {'alpha': 2.0}),
        ('linear', {'slope': 1.0}),
        ('linear', {'slope': 2.0}),
        ('linear', {'slope': 3.0}),
        ('quadratic', {'a1': 1.0, 'a2': -0.5}),
        ('quadratic', {'a1': 1.5, 'a2': 0.3}),
        ('cubic', {'a1': 1.0, 'a3': -0.5}),
        ('gaussian', {'width': 0.25}),
        ('gaussian', {'width': 0.4}),
        ('power', {'p': 1.5}),
        ('power', {'p': 3.0}),
        ('power', {'p': 5.0}),
        ('sinc_like', {'width': 0.15}),
        ('sinc_like', {'width': 0.3}),
        ('two_gaussian', {'c1': -0.3, 'c2': 0.3, 'width': 0.15}),
        ('two_gaussian', {'c1': -0.5, 'c2': 0.5, 'width': 0.2}),
    ]
    
    for ptype, pkw in profiles_to_try:
        if time.monotonic() > phase1_end:
            break
        p = _make_profile(n_base, ptype, **pkw)
        sc = _score_norm_fn(p)
        candidates.append((sc, p, f"{ptype}_{str(pkw)[:20]}"))
        update_best(p, sc, f"init_{ptype}")
    
    # Also try different lengths
    for sn in [2048, 8192]:
        if time.monotonic() > phase1_end:
            break
        for ptype, pkw in [('raised_cosine', {'alpha': 1.0}), ('linear', {'slope': 1.5}), ('power', {'p': 2.0})]:
            p = _make_profile(sn, ptype, **pkw)
            sc = _score_norm_fn(p)
            candidates.append((sc, p, f"n{sn}_{ptype}"))
            update_best(p, sc, f"n{sn}_{ptype}")
    
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done: best={best_score:.10f}, {len(candidates)} candidates")
    
    # --- Phase 2: Accelerated logsumexp on top candidates (400s) ---
    phase2_end = min(deadline, t_start + 460.0)
    n_to_opt = 5
    per_cand = max(30.0, (phase2_end - time.monotonic()) / max(1, n_to_opt))
    count = 0
    seen_labels = set()
    
    for sc, p, label in candidates:
        if time.monotonic() > phase2_end - 10:
            break
        if count >= n_to_opt:
            break
        if sc > best_score * 1.05:
            break
        key = (len(p), round(sc, 6))
        if key in seen_labels:
            continue
        seen_labels.add(key)
        
        dl = min(phase2_end - 5, time.monotonic() + per_cand)
        p_best, sc_best = _accel_logsumexp(p, dl, rng, n_betas=10, beta_scale=1.0)
        update_best(p_best, sc_best, f"accel_{label[:20]}")
        count += 1
    
    print(f"Phase 2 done: best={best_score:.10f}")
    
    # --- Phase 3: L-BFGS on best (300s) ---
    phase3_end = min(deadline, t_start + 760.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        dl = min(phase3_end - 5, time.monotonic() + 150.0)
        p_best, sc_best = _run_lbfgs_on_z(best_seq, dl, beta=0.1, max_wall_time=150.0)
        update_best(p_best, sc_best, "lbfgs_0.1")
        
        if time.monotonic() < phase3_end - 30:
            dl2 = min(phase3_end - 5, time.monotonic() + 140.0)
            p_best2, sc_best2 = _run_lbfgs_on_z(best_seq, dl2, beta=0.03, max_wall_time=140.0)
            update_best(p_best2, sc_best2, "lbfgs_0.03")
    
    print(f"Phase 3 done: best={best_score:.10f}")
    
    # --- Phase 4: Active set refinement (200s) ---
    phase4_end = min(deadline, t_start + 960.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        # Try different active set sizes
        for k_act in [4, 8, 16]:
            if time.monotonic() > phase4_end - 15:
                break
            dl = min(phase4_end - 5, time.monotonic() + 50.0)
            p_best, sc_best = _active_set_refine(best_seq, dl, rng, k_active=k_act)
            update_best(p_best, sc_best, f"active_k{k_act}")
    
    print(f"Phase 4 done: best={best_score:.10f}")
    
    # --- Phase 5: L-BFGS again on refined (200s) ---
    phase5_end = min(deadline, t_start + 1160.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        dl = min(phase5_end - 5, time.monotonic() + 100.0)
        p_best, sc_best = _run_lbfgs_on_z(best_seq, dl, beta=0.05, max_wall_time=100.0)
        update_best(p_best, sc_best, "lbfgs_refine1")
        
        if time.monotonic() < phase5_end - 20:
            dl2 = min(phase5_end - 5, time.monotonic() + 90.0)
            p_best2, sc_best2 = _accel_logsumexp(best_seq, dl2, rng, n_betas=6, beta_scale=0.5)
            update_best(p_best2, sc_best2, "accel_refine1")
    
    print(f"Phase 5 done: best={best_score:.10f}")
    
    # --- Phase 6: Structured local search (200s) ---
    phase6_end = min(deadline, t_start + 1360.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        dl = phase6_end - 5
        p_best, sc_best = _structured_local_search(best_seq, dl, rng, n_trials=15000)
        update_best(p_best, sc_best, "local_search")
    
    print(f"Phase 6 done: best={best_score:.10f}")
    
    # --- Phase 7: Multi-restart (200s) ---
    phase7_end = min(deadline, t_start + 1560.0)
    if best_seq is not None and time.monotonic() < phase7_end:
        n_restarts = 4
        per_restart = max(20.0, (phase7_end - time.monotonic()) / max(1, n_restarts))
        
        for i in range(n_restarts):
            if time.monotonic() > phase7_end - 10:
                break
            
            # Different noise levels
            noise = 0.02 + 0.02 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=len(z))
            z = np.clip(z, -80, 80)
            a_start = np.exp(z)
            a_start = a_start / a_start.sum()
            
            dl = min(phase7_end - 5, time.monotonic() + per_restart)
            # Alternate between methods
            if i % 2 == 0:
                p_best, sc_best = _accel_logsumexp(a_start, dl, rng, n_betas=6, beta_scale=0.7)
            else:
                p_best, sc_best = _run_lbfgs_on_z(a_start, dl, beta=0.08, max_wall_time=per_restart - 5)
            update_best(p_best, sc_best, f"restart{i}")
    
    print(f"Phase 7 done: best={best_score:.10f}")
    
    # --- Phase 8: Final polish with L-BFGS + local (remaining) ---
    if best_seq is not None:
        remaining = max(10.0, deadline - time.monotonic())
        
        # Final L-BFGS
        if remaining > 30:
            dl = min(deadline - 10, time.monotonic() + remaining * 0.4)
            p_best, sc_best = _run_lbfgs_on_z(best_seq, dl, beta=0.02, max_wall_time=remaining * 0.4)
            update_best(p_best, sc_best, "final_lbfgs")
        
        # Final active set
        if remaining > 20:
            dl = min(deadline - 10, time.monotonic() + remaining * 0.3)
            p_best, sc_best = _active_set_refine(best_seq, dl, rng, k_active=12)
            update_best(p_best, sc_best, "final_active")
        
        # Final local search
        if time.monotonic() < deadline - 10:
            dl = deadline - 5
            p_best, sc_best = _structured_local_search(best_seq, dl, rng, n_trials=5000)
            update_best(p_best, sc_best, "final_local")
    
    print(f"All phases done: best={best_score:.10f}")
    
    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_profile(2048, 'raised_cosine', alpha=1.0)
            s = float(final_seq.sum())
        final_seq = final_seq / s  # Normalize to sum=1
        
        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")
        
        # Validate with evaluator
        if len(final_seq) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                else:
                    print(f"Evaluator returned inf, trying smaller version")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_profile(2048, 'raised_cosine', alpha=1.0)
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n=2048, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
