"""
LP-based + Exponentiated Gradient optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Implements active-set LP iterations, mirror descent (exponentiated gradient),
L-BFGS continuation, and Adam polishing with aggressive time management.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix, csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_profile(n, width):
    t = np.linspace(0.0, 1.0, n)
    u = t - 0.5
    a = np.exp(-u ** 2 / (2.0 * width ** 2))
    return a / a.sum()


def _make_raised_cosine(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5) * alpha))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power_profile(n, p):
    t = np.linspace(0.0, 1.0, n)
    u = np.abs(2.0 * t - 1.0)
    a = (1.0 - u) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _build_active_matrix(a, active_lags):
    """Build sparse matrix A where A[i,j] = a[k-j] for active_lags[i]=k."""
    n = len(a)
    m = len(active_lags)
    A = lil_matrix((m, n))
    for i, k in enumerate(active_lags):
        k = int(k)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[i, j] = a[k - j]
    return A.tocsr()


def _lp_active_step(a, M, active_lags, time_limit=10.0):
    """Solve LP: max sum(b) s.t. conv(a,b)[k] <= M for active k, b>=0, sum(b)=1."""
    n = len(a)
    active_lags = [int(k) for k in active_lags]
    m = len(active_lags)
    if m == 0:
        return None

    # Build constraint matrix
    t0 = time.monotonic()
    A = _build_active_matrix(a, active_lags)

    # Use linprog with HiGHS
    c_obj = -np.ones(n)
    b_ub = np.full(m, M)
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])

    try:
        result = linprog(
            c_obj,
            A_ub=A,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=[(0, None)] * n,
            method='highs',
            options={'time_limit': time_limit, 'presolve': True}
        )
        if result.success and result.x is not None:
            b = np.maximum(result.x, 0)
            s = b.sum()
            if s > 1e-10:
                b = b / s
                return b
    except Exception:
        pass
    return None


def _eg_step(a, active_lags, eta):
    """Exponentiated gradient step for active lags."""
    n = len(a)
    # Compute grad[j] = 2 * sum_{k in active} a[k-j] (valid)
    I = np.zeros(2 * n - 1)
    for k in active_lags:
        k = int(k)
        if 0 <= k <= 2 * n - 2:
            I[k] = 1.0

    # S[j] = sum_k I[k] * a[k-j] = correlation
    # This is conv(I_reversed, a) ... let's just compute it
    # S[j] = sum_{k valid} I[k] * a[k-j] where 0 <= k-j <= n-1
    # Equivalently: for each active k, add a[k-j] to S[j] for valid j
    S = np.zeros(n)
    for k in active_lags:
        k = int(k)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        S[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]

    # Multiplicative update
    a_new = a * np.exp(-eta * S / max(1, len(active_lags)))
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    return a_new


def _lp_iteration(a, deadline, max_iters=200, active_frac=0.05, min_active=10, max_active=200):
    """LP-based iterative optimization with active set."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    M_prev = best_score / (2.0 * n)  # max of conv(a,a)

    t_step = 0.1
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline - 2.0:
            break

        c = _full_convolve(a, a)
        M = float(c.max())

        # Find active lags
        threshold = M * (1.0 - active_frac)
        active_mask = c >= threshold
        active_lags = np.where(active_mask)[0]

        # Limit active set size
        if len(active_lags) > max_active:
            top_idx = np.argsort(c)[-max_active:]
            active_lags = top_idx
        elif len(active_lags) < min_active:
            top_idx = np.argsort(c)[-min_active:]
            active_lags = top_idx

        # Solve LP
        b = _lp_active_step(a, M, active_lags, time_limit=5.0)
        if b is None:
            # Fallback: use EG step
            b = _eg_step(a, active_lags, eta=t_step)

        # Line search for step size
        best_t = t_step * 0.1
        best_new_score = _score_norm_fn(a)
        for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
            a_trial = (1.0 - frac) * a + frac * b
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            sc = _score_norm_fn(a_trial)
            if sc < best_new_score:
                best_new_score = sc
                best_t = frac

        a_new = (1.0 - best_t) * a + best_t * b
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s

        if best_new_score < best_score:
            best_score = best_new_score
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
            # Decay step size slightly
            t_step = max(0.02, t_step * 0.95)
            active_frac = max(0.01, active_frac * 0.99)
        else:
            no_improve += 1
            t_step = max(0.02, t_step * 0.8)
            if no_improve > 20:
                break

        if it % 10 == 0:
            print(f"    LP iter {it}: score={best_score:.10f}, M={M:.6f}, "
                  f"active={len(active_lags)}, t={best_t:.3f}")

    return best_a, best_score


def _eg_iteration(a, deadline, max_iters=5000, eta_init=0.05, eta_decay=0.9999):
    """Exponentiated gradient (mirror descent) iteration."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    eta = eta_init
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline - 1.0:
            break

        c = _full_convolve(a, a)
        M = float(c.max())

        # Active lags
        threshold = M * 0.99
        active_lags = np.where(c >= threshold)[0]
        if len(active_lags) > 500:
            active_lags = np.argsort(c)[-500:]

        a_new = _eg_step(a, active_lags, eta)
        sc = _score_norm_fn(a_new)

        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Reduce eta more aggressively
            eta *= 0.95
            if no_improve > 100:
                break

        eta *= eta_decay

        if it % 500 == 0:
            print(f"    EG iter {it}: score={best_score:.10f}, eta={eta:.6f}")

    return best_a, best_score


def _adam_smooth(a_init, deadline, lr=0.02, k_top=5, temp=0.01):
    """Adam with smoothed objective (logsumexp + top-K blend)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8

    for it in range(100000):
        if time.monotonic() > deadline - 1.0:
            break

        t += 1
        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        cm = float(c.max())

        # Compute gradient of smooth objective
        # Use softmax weighting over lag values
        c_shifted = (c - cm) / temp
        exp_c = np.exp(c_shifted)
        w = exp_c / exp_c.sum()

        # Gradient w.r.t. z (via a)
        # d(max-smooth)/da = 4n * a * (h - W) where h = conv(w_reversed, a)
        # and W = dot(w, c)
        W = float(np.dot(w, c))
        # h[j] = sum_k w[k] * a[k-j] (valid) = correlation of w and a
        # = conv(w_reversed_padded, a)[n-1 : 2n-1] approximately
        # More precisely: h[j] = sum_k w[k] * a[k-j] for 0<=k-j<=n-1
        # This is the cross-correlation
        # Use FFT: h = real(ifft(fft(w, N) * conj(fft(a, N))))[n-1:2n-1]
        N = _next_pow2(2 * n)
        fw = np.fft.rfft(w, N)
        fa = np.fft.rfft(a_curr, N)
        h_full = np.fft.irfft(fw * np.conj(fa), N)
        h = h_full[n - 1:2 * n - 1]

        da = 4.0 * n * a_curr * (h - W)

        # Chain rule: dz = da / a (for softmax parameterization)
        dz = da / (a_curr + 1e-30)

        # Also subtract mean to stay on the simplex (redundant but helps stability)
        dz = dz - dz.mean()

        m = b1 * m + (1 - b1) * dz
        v = b2 * v + (1 - b2) * (dz * dz)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if t % 20 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()

        if t % 5000 == 0:
            lr = max(0.001, lr * 0.8)
            temp = max(0.001, temp * 0.9)

    return best_a, best_score


def _lbfgs_continuation(a_init, deadline, max_wall_time=60.0, beta_scale=1.0):
    """L-BFGS with logsumexp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    betas = [b * max_c0 * beta_scale for b in beta_starts]

    per_level = max(5.0, max_wall_time / len(betas))

    for beta in betas:
        if time.monotonic() > deadline - 2.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        def obj(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            log_sum = float(np.log(s_))
            return 2.0 * n * (cm_ + beta * log_sum)

        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            W = float(np.dot(w, c_))
            N = _next_pow2(2 * n)
            fw = np.fft.rfft(w, N)
            fa = np.fft.rfft(a_, N)
            h_full = np.fft.irfft(fw * np.conj(fa), N)
            h = h_full[n - 1:2 * n - 1]
            da = 4.0 * n * a_ * (h - W)
            dz = da / (a_ + 1e-30)
            dz = dz - dz.mean()
            return dz

        try:
            result = minimize(
                obj, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 500, 'ftol': 1e-20, 'gtol': 1e-12,
                         'maxls': 30, 'maxfun': 3000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _local_search(a_init, deadline, rng, n_trials=2000):
    """Local search with block perturbations."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break

        block_size = int(rng.integers(5, min(30, n // 4)))
        start = int(rng.integers(0, n - block_size))

        a_trial = a.copy()
        perturbation = rng.uniform(0.9, 1.1, size=block_size)
        a_trial[start:start + block_size] *= perturbation
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial /= a_trial.sum()

        sc = _score_norm_fn(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()

    return best_a, best_score

        dl = deadline - 10.0
        p_best, sc_best = _adam_smooth(best_seq, dl, lr=0.005, k_top=2, temp=0.001)
        _update_best(p_best, sc_best, "final_adam")

    # === Try different n ===
    if best_seq is not None and time.monotonic() < deadline - 30:
        print("\n=== Trying different lengths ===")
        n_curr = len(best_seq)
        for target_n in [2048, 8192]:
            if abs(target_n - n_curr) < 10 or time.monotonic() > deadline - 20:
                continue
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            _update_best(p_interp, sc_interp, f"interp_n{target_n}")

            if time.monotonic() < deadline - 20:
                dl = min(deadline - 15, time.monotonic() + 30.0)
                p_best, sc_best = _eg_iteration(p_interp, dl, max_iters=2000, eta_init=0.02)
                _update_best(p_best, sc_best, f"eg_n{target_n}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_linear_profile(4096, 1.5)
            s = float(final_seq.sum())
        final_seq = final_seq / s

        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        if len(final_seq) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    fallback = _make_linear_profile(4096, 1.5)
    print(f"FALLBACK: n=4096, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
