from scipy.optimize import minimize
import time

def _fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1 << (out_n - 1).bit_length()
    N = max(N, 2)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    """Compute the normalized score (assuming sum=1)."""
    n = len(a)
    b = _fft_convolve(a, a)
    return 2.0 * n * b.max()

    """Score with normalization."""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    return _score(a_norm)

"""
Optimized step-function minimizer targeting score < 1.5027.
Uses blended logsumexp/top-K smoothing, FFT-based gradients,
adaptive L-BFGS continuation, peak-shaving local search,
coordinate descent, and multi-restart with diverse profiles.
Exploits symmetry and multi-scale initialization.

import numpy as np
from scipy.optimize import minimize
import time


def _fftconv(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    out_len = n + m - 1
    N = 1 << (out_len - 1).bit_length()
    N = max(N, 2)
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:out_len]


def _softmax(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm(a):
    """Score assuming sum(a)=1, i.e., 2n*max(a*a)."""
    n = len(a)
    b = _fftconv(a, a)
    return float(2.0 * n * b.max())


def _score_full(a):
    """Full score: 2n*max(a*a)/sum(a)^2."""
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a_n = a / s
    return _score_norm(a_n)


def _ls_obj(z, n, beta):
    """Logsumexp-smoothed objective."""
    a = _softmax(z)
    b = _fftconv(a, a)
    bm = float(b.max())
    sh = (b - bm) / beta
    e = np.exp(sh)
    s = float(e.sum())
    if s < 1e-300:
        return 2.0 * n * bm
    return 2.0 * n * (bm + beta * np.log(s))


def _ls_grad(z, n, beta):
    """Gradient of logsumexp-smoothed objective."""
    a = _softmax(z)
    b = _fftconv(a, a)
    bm = float(b.max())
    sh = (b - bm) / beta
    e = np.exp(sh)
    s = float(e.sum())
    if s < 1e-300:
        w = np.zeros_like(b)
        w[int(np.argmax(b))] = 1.0
    else:
        w = e / s
    # grad_j = 4n * sum_k w_k * a[k-j] (cross-correlation)
    # DFT: IFFT(conj(FFT(w)) * FFT(a))
    N = 1 << (len(w) + n - 1 - 1).bit_length()
    N = max(N, 2)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    grad = 4.0 * n * np.fft.irfft(np.conj(W) * A, N)[:n]
    if not np.all(np.isfinite(grad)):
        return np.zeros(n)
    return grad


def _topk_obj(z, n, K):
    """Top-K average smoothed objective."""
    a = _softmax(z)
    b = _fftconv(a, a)
    K_actual = min(K, len(b))
    top = np.sort(b)[-K_actual:]
    return float(2.0 * n * top.mean())


def _topk_grad(z, n, K):
    """Gradient of top-K average."""
    a = _softmax(z)
    b = _fftconv(a, a)
    K_actual = min(K, len(b))
    top_idx = np.argsort(b)[-K_actual:]
    # grad = (1/K) * sum_{k in top} 4n * a shifted by k
    grad = np.zeros(n)
    for k_val in top_idx:
        k_val = int(k_val)
        # 4n * a[k_val - j] for valid j
        lo = max(0, k_val - n + 1)
        hi = min(n - 1, k_val)
        if lo <= hi:
            # a[k_val - j] for j from lo to hi
            # k_val - j ranges from k_val-hi to k_val-lo
            idx = k_val - hi : k_val - lo + 1
            # Reverse because j increases means k_val-j decreases
            grad[lo:hi+1] += 4.0 * n * a[idx][::-1]
    grad /= K_actual
    return grad


def _blended_obj(z, n, beta, K, alpha_ls):
    """Blend of logsumexp and top-K smoothed objectives."""
    obj_ls = _ls_obj(z, n, beta)
    obj_tk = _topk_obj(z, n, K)
    return alpha_ls * obj_ls + (1.0 - alpha_ls) * obj_tk


def _blended_grad(z, n, beta, K, alpha_ls):
    """Gradient of blended objective."""
    g_ls = _ls_grad(z, n, beta)
    g_tk = _topk_grad(z, n, K)
    return alpha_ls * g_ls + (1.0 - alpha_ls) * g_tk


def _lbfgs_optimize(z0, n, deadline, betas, K=5, alpha_ls=0.7):
    """L-BFGS with continuation on blended objective."""
    z = z0.copy()
    best_z = z0.copy()
    best_a = _softmax(z0)
    best_sc = _score_norm(best_a)
    
    for beta in betas:
        if time.monotonic() > deadline - 2:
            break
        
        def obj(z_):
            return _blended_obj(z_, n, beta, K, alpha_ls)
        
        def grad(z_):
            return _blended_grad(z_, n, beta, K, alpha_ls)
        
        try:
            res = minimize(
                obj, z, jac=grad,
                method='L-BFGS-B',
                options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-8,
                         'maxls': 30, 'maxfun': 2000}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_new = _softmax(res.x)
                sc_new = _score_norm(a_new)
                if np.isfinite(sc_new) and sc_new < best_sc:
                    best_sc = sc_new
                    best_a = a_new.copy()
                    best_z = res.x.copy()
                    z = res.x.copy()
        except Exception:
            pass
    
    return best_a, best_sc, best_z


def _adam_optimize(z0, n, deadline, lr=0.01, beta_ls=0.01, K=5, alpha_ls=0.5):
    """Adam optimization on blended objective."""
    n_var = len(z0)
    z = z0.copy()
    m = np.zeros(n_var)
    v = np.zeros(n_var)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    best_a = _softmax(z0)
    best_sc = _score_norm(best_a)
    
    while time.monotonic() < deadline - 1:
        t += 1
        g = _blended_grad(z, n, beta_ls, K, alpha_ls)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * (g * g)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100, 100)
        z = z - z.mean()
        
        if t % 5 == 0:
            a_new = _softmax(z)
            sc = _score_norm(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
        
        if t % 500 == 0:
            lr *= 0.95
    
    return best_a, best_sc


def _peak_shave(a, deadline, rng, n_iter=5000):
    """Local search: try single-element perturbations to reduce peak."""
    n = len(a)
    a_curr = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a_curr)
    
    for it in range(n_iter):
        if time.monotonic() > deadline - 1:
            break
        
        b = _fftconv(a_curr, a_curr)
        k_peak = int(np.argmax(b))
        
        # Find the most contributing element to the peak
        lo = max(0, k_peak - n + 1)
        hi = min(n - 1, k_peak)
        if lo > hi:
            break
        
        # Contribution of each a_j to b[k_peak]
        j_range = np.arange(lo, hi + 1)
        contributions = a_curr[j_range] * a_curr[k_peak - j_range][::-1]
        # Wait, a_curr[k_peak - j_range] gives a[k_peak - lo], ..., a[k_peak - hi]
        # But we want a[k_peak - j] for j in j_range
        # a[k_peak - j_range] = a[k_peak - lo], a[k_peak - (lo+1)], ..., a[k_peak - hi]
        # This is in decreasing order of index, which is correct.
        
        j_best = j_range[np.argmax(contributions)]
        
        # Try reducing a[j_best] by a small amount
        delta = a_curr[j_best] * 0.01 * rng.uniform(0.5, 1.5)
        j_best = int(j_best)
        
        # Try adding to a few candidate positions
        # Choose positions that don't contribute much to the peak
        # (i.e., a[k_peak - j] is small)
        
        # Random candidate
        j_add = int(rng.integers(0, n))
        if j_add == j_best:
            continue
        
        a_trial = a_curr.copy()
        a_trial[j_best] -= delta
        a_trial[j_add] += delta
        
        if a_trial[j_best] < 0:
            continue
        
        sc = _score_norm(a_trial)
        if sc < best_sc:
            best_sc = sc
            best_a = a_trial.copy()
            a_curr = a_trial.copy()
    
    return best_a, best_sc


def _coord_descent(a, deadline, rng, n_outer=200):
    """Coordinate descent: optimize pairs of elements."""
    n = len(a)
    a_curr = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a_curr)
    
    for outer in range(n_outer):
        if time.monotonic() > deadline - 1:
            break
        
        # Pick a random pair
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        
        # Try fractional moves
        for frac in [0.3, 0.5, 0.7]:
            delta = frac * min(a_curr[i], a_curr[j])
            for sign in [1, -1]:
                a_trial = a_curr.copy()
                a_trial[i] -= sign * delta
                a_trial[j] += sign * delta
                if a_trial[i] < 0 or a_trial[j] < 0:
                    continue
                sc = _score_norm(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    break
        # Update a_curr to best
        if best_sc < _score_norm(a_curr):
            a_curr = best_a.copy()
    
    return best_a, best_sc
