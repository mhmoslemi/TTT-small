"""
Optimized subgradient + smoothed Nesterov method for minimizing peak autocorrelation.
Key improvements: direct unsmoothed subgradient with Nesterov acceleration,
adaptive active set, aggressive multi-restart, and better hyperparameter scheduling.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _subgradient_z(a, c, active_frac=0.01):
    """Compute subgradient of max(c) w.r.t. z where a = softmax(z).
    Uses soft active set: lags within active_frac of max."""
    n = len(a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros_like(a), cm
    threshold = cm * (1.0 - active_frac)
    active = c >= threshold
    count = np.sum(active)
    if count == 0:
        active[int(np.argmax(c))] = True
        count = 1
    w = np.zeros_like(c)
    w[active] = 1.0 / count
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(a), cm
    return grad, cm


def _nesterov_subgrad(a_init, deadline, active_frac=0.01, lr_init=2.0, mom_init=0.9,
                      lr_decay=0.999, check_interval=20):

    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    lr = lr_init
    mom = mom_init
    t = 0
    stall_count = 0
    prev_best = best_score

    while time.monotonic() < deadline:
        t += 1
        a_y = _softmax(y)
        c = _full_convolve(a_y, a_y)
        grad, cm = _subgradient_z(a_y, c, active_frac)

        z_new = z - lr * grad
        z_new = z_new - z_new.mean()
        z_new = np.clip(z_new, -60, 60)

        y_new = z_new + mom * (z_new - z)
        y_new = y_new - y_new.mean()

        z = z_new
        y = y_new

        lr *= lr_decay

        if t % check_interval == 0:

            sc = _score_norm(a_curr)
            if np.isfinite(sc) and sc < best_score:

                best_a = a_curr.copy()
                stall_count = 0

def _smoothed_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _smoothed_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _lbfgs_smoothed(a_init, deadline, betas, max_iter_per_level=200):

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    for beta in betas:
        if time.monotonic() >= deadline:
            break
        level_dl = min(deadline, time.monotonic() + max(5.0, (deadline - time.monotonic()) / len(betas)))

        def obj(z2):
            return _smoothed_obj(z2, beta, n)

        def grad(z2):
            return _smoothed_grad(z2, beta, n)

        try:

            if result.x is not None and np.all(np.isfinite(result.x)):

                a_new = _softmax(z_new)

                if np.isfinite(sc) and sc < best_score:

                    best_a = a_new.copy()
                z = z_new
        except Exception:
            pass

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:

            continue

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        if t % decay_interval == 0:
            beta *= beta_decay
            beta = max(1e-10, beta)

        if t % 100 == 0:
            a_curr = _softmax(z)
            sc = _score_norm(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _make_profile(n, kind="linear", param=1.5):
    t = np.linspace(0.0, 1.0, n)
    if kind == "linear":
        a = 1.0 + param * (t - 0.5)
    elif kind == "quad":
        u = 2.0 * t - 1.0
        a = 1.0 + 0.3 * u + param * u * u
    elif kind == "cos2":
        a = np.cos(t * np.pi / 2.0) ** 2
    elif kind == "exp_ramp":
        a = np.exp(param * (t - 0.5))
    elif kind == "cubic":
        u = 2.0 * t - 1.0
        a = 1.0 + param * u - 0.3 * u ** 3
    else:
        a = 1.0 + param * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 0: Initialization with diverse profiles (20s)
    Phase 1: Nesterov subgradient on best init (400s)
    Phase 2: L-BFGS smoothed continuation (300s)
    Phase 3: Adam fine-tuning (250s)
    Phase 4: Multi-restart with perturbation (400s)
    Phase 5: Final Nesterov polish (remaining)
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Phase 0: Get parent and generate candidates (20s) ---
    phase0_end = min(deadline, t_start + 20.0)

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n = 4096
    if parent is not None:
        if len(parent) != n:
            parent = _interp_profile(parent, n)
        print(f"Parent: n={n}, score={_score_full(parent):.10f}")
    else:
        parent = _make_profile(n, "linear", 1.5)
        print(f"No parent, using linear profile n={n}")

    candidates = []
    sc_p = _score_norm(parent)
    candidates.append((sc_p, parent.copy(), "parent"))
    update_best(parent, sc_p, "parent")

    profiles_to_try = [
        ("linear", 1.0), ("linear", 1.5), ("linear", 2.0), ("linear", 2.5),
        ("quad", 0.3), ("cos2", 1.0), ("exp_ramp", 1.0), ("exp_ramp", 1.5),
        ("cubic", 1.0),
    ]
    for kind, param in profiles_to_try:
        if time.monotonic() > phase0_end:
            break
        p = _make_profile(n, kind, param)
        sc = _score_norm(p)
        candidates.append((sc, p, f"{kind}_{param}"))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f} (n={len(best_seq)})")
    for sc, p, label in candidates[:5]:
        print(f"    {label}: {sc:.10f}")

    # --- Phase 1: Nesterov subgradient on best candidate (400s) ---
    phase1_end = min(deadline, t_start + 420.0)
    if best_seq is not None and time.monotonic() < phase1_end:
        per_time = phase1_end - time.monotonic()
        if per_time > 30:
            p_best, sc_best = _nesterov_subgrad(best_seq, min(phase1_end, time.monotonic() + per_time),
                                                active_frac=0.01, lr_init=3.0, mom_init=0.9, lr_decay=0.9995)
            update_best(p_best, sc_best, "nesterov_p1")

            # Second run with different params
            if time.monotonic() < phase1_end - 30:
                per_time2 = phase1_end - time.monotonic() - 5
                if per_time2 > 30:
                    p_best2, sc_best2 = _nesterov_subgrad(best_seq, time.monotonic() + per_time2,
                                                          active_frac=0.05, lr_init=5.0, mom_init=0.85, lr_decay=0.999)
                    update_best(p_best2, sc_best2, "nesterov_p1b")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: L-BFGS smoothed (300s) ---
    phase2_end = min(deadline, t_start + 720.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        c0 = _full_convolve(best_seq, best_seq)
        max_c0 = float(c0.max())
        betas = [b * max_c0 for b in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]]
        per_time = phase2_end - time.monotonic()
        if per_time > 20:
            p_best, sc_best = _lbfgs_smoothed(best_seq, min(phase2_end, time.monotonic() + per_time),
                                              betas, max_iter_per_level=150)
            update_best(p_best, sc_best, "lbfgs_p2")

        # Also try on second-best candidate if time allows
        if time.monotonic() < phase2_end - 60:
            for sc_c, p_c, label in candidates[1:3]:
                if time.monotonic() > phase2_end - 30:
                    break
                if abs(len(p_c) - n) > 0:
                    continue
                per_t = min(50.0, phase2_end - time.monotonic() - 10)
                if per_t > 15:
                    betas_short = [b * max_c0 for b in [0.03, 0.01, 0.005, 0.002, 0.001]]
                    p_b, sc_b = _lbfgs_smoothed(p_c, time.monotonic() + per_t, betas_short, max_iter_per_level=100)
                    update_best(p_b, sc_b, f"lbfgs_{label}")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Adam fine-tuning (250s) ---
    phase3_end = min(deadline, t_start + 970.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        per_time = min(200.0, phase3_end - time.monotonic() - 10)
        if per_time > 20:
            p_best, sc_best = _adam_smoothed(best_seq, time.monotonic() + per_time,
                                             lr=0.15, beta_init=0.01, beta_decay=0.93, decay_interval=300)
            update_best(p_best, sc_best, "adam_p3")

        # Second Adam run with different params
        if time.monotonic() < phase3_end - 30:
            per_t2 = min(80.0, phase3_end - time.monotonic() - 10)
            if per_t2 > 15:
                p_b2, sc_b2 = _adam_smoothed(best_seq, time.monotonic() + per_t2,
                                             lr=0.05, beta_init=0.005, beta_decay=0.9, decay_interval=200)
                update_best(p_b2, sc_b2, "adam_p3b")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Multi-restart (400s) ---
    phase4_end = min(deadline, t_start + 1370.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        n_restarts = 5
        per_restart = (phase4_end - time.monotonic() - 10) / max(1, n_restarts)

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 15:
                break
            # Different perturbation strategies
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()

            if i == 0:
                noise = 0.05 * rng.normal(size=n)
            elif i == 1:
                noise = 0.1 * rng.normal(size=n)
            elif i == 2:
                # Structured perturbation: shift the profile
                shift = rng.uniform(-0.1, 0.1)
                t = np.linspace(0, 1, n)
                noise = shift * np.sin(2 * np.pi * t)
            elif i == 3:
                # Blend with a different profile
                alt = _make_profile(n, "exp_ramp", 1.2)
                blend = 0.2
                z_noise = blend * np.log(np.maximum(alt, 1e-15))
                noise = z_noise - z_noise.mean()
            else:
                noise = 0.15 * rng.normal(size=n)

            z = z + noise
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            p_start = _softmax(z)

            # Nesterov subgradient on perturbed start
            per_dl = min(phase4_end - 10, time.monotonic() + per_restart)
            p_best, sc_best = _nesterov_subgrad(p_start, per_dl,
                                                active_frac=0.02, lr_init=3.0, mom_init=0.9, lr_decay=0.9995)
            update_best(p_best, sc_best, f"restart{i}")

            # Quick smoothed polish
            if time.monotonic() < phase4_end - 20:
                c_p = _full_convolve(p_best, p_best)
                mc_p = float(c_p.max())
                betas_r = [b * mc_p for b in [0.01, 0.005, 0.002, 0.001]]
                per_dl2 = min(phase4_end - 10, time.monotonic() + 30.0)
                p_b2, sc_b2 = _lbfgs_smoothed(p_best, per_dl2, betas_r, max_iter_per_level=80)
                update_best(p_b2, sc_b2, f"restart{i}_polish")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Nesterov on second-best candidate (cross-pollination) (100s) ---
    phase5_end = min(deadline, t_start + 1470.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        # Try blending best with other candidates
        for sc_c, p_c, label in candidates[1:4]:
            if time.monotonic() > phase5_end - 20:
                break
            if len(p_c) != n:
                continue
            # Create blend
            for blend in [0.3, 0.5]:
                if time.monotonic() > phase5_end - 10:
                    break
                p_blend = (1 - blend) * best_seq + blend * p_c
                p_blend = np.maximum(p_blend, 1e-12)
                p_blend = p_blend / p_blend.sum()
                sc_blend = _score_norm(p_blend)
                # Nesterov on blend
                per_dl = min(phase5_end - 10, time.monotonic() + 20.0)
                p_b, sc_b = _nesterov_subgrad(p_blend, per_dl,
                                              active_frac=0.01, lr_init=3.0, mom_init=0.9, lr_decay=0.9995)
                update_best(p_b, sc_b, f"blend_{label}_{blend}")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Final fine Nesterov polish (remaining) ---
    phase6_end = deadline
    if best_seq is not None and time.monotonic() < phase6_end:
        remaining = phase6_end - time.monotonic()
        if remaining > 30:
            # Very gentle Nesterov with tight active set
            per_time = min(remaining - 10, 120.0)
            if per_time > 15:
                p_best, sc_best = _nesterov_subgrad(best_seq, time.monotonic() + per_time,
                                                    active_frac=0.005, lr_init=1.0, mom_init=0.95, lr_decay=0.9999)
                update_best(p_best, sc_best, "final_nesterov")

            # Very small beta Adam
            if time.monotonic() < phase6_end - 10:
                per_t2 = min(phase6_end - time.monotonic() - 5, 60.0)
                if per_t2 > 10:
                    p_b2, sc_b2 = _adam_smoothed(best_seq, time.monotonic() + per_t2,
                                                 lr=0.03, beta_init=0.003, beta_decay=0.9, decay_interval=150)
                    update_best(p_b2, sc_b2, "final_adam")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profile(256, "linear", 1.5)

        sc_final = _score_full(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned non-finite, using internal: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profile(4096, "linear", 1.5)
    print(f"FALLBACK: n=4096, score={_score_full(fallback):.10f}")
    return [float(x) for x in fallback]
