"""
Optimization of step function heights to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses log-parameterization with soft-L1 smoothing, multi-beta L-BFGS continuation,
Adam gradient descent with adaptive learning rate, and coordinate descent polish.
Starts from the parent sequence if available, with multi-restart exploration.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _fft_convolve(a, b):
    n_out = len(a) + len(b) - 1
    n_fft = 1
    while n_fft < n_out:
        n_fft <<= 1
    fa = np.fft.rfft(a, n_fft)
    fb = np.fft.rfft(b, n_fft)
    return np.fft.irfft(fa * fb, n_fft)[:n_out]


def _normalized_score(a):
    c = _fft_convolve(a, a)
    return float(2.0 * len(a) * np.max(c))


def _softmax(vec):
    vec = vec - np.max(vec)
    e = np.exp(vec)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones(len(vec)) / len(vec)
    return e / s


def _init_from_parent():
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0 and np.isfinite(raw).all():
            raw = np.clip(raw, 0.0, 1000.0)
            s = raw.sum()
            if s > 0.01:
                return raw / s
    except Exception:
        pass
    return None


def _make_profiles(n):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    profiles = []
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for rate in [1.0, 2.0, 3.0, 5.0, 10.0, -1.0, -2.0, -3.0, -5.0, -10.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * u / 2.0), 1e-12) ** p
        profiles.append(a / a.sum())
    a = np.maximum(1.0 + 0.5 * u + 0.3 * u ** 2, 1e-12)
    profiles.append(a / a.sum())
    a = np.maximum(1.0 - 0.5 * u + 0.3 * u ** 2, 1e-12)
    profiles.append(a / a.sum())
    for ratio in [0.3, 0.5, 0.7]:
        half = n // 2
        a = np.ones(n)
        a[:half] = ratio
        profiles.append(a / a.sum())
    return profiles


def _adaptive_beta(c):
    cm = float(np.max(c))
    if cm < 1e-15:
        return 1e-10
    sorted_c = np.sort(c)[::-1]
    k = min(30, len(sorted_c) - 1)
    gap = sorted_c[0] - sorted_c[k]
    if gap < 0.03 * cm:
        return max(1e-10, 0.05 * cm)
    elif gap > 0.3 * cm:
        return max(1e-10, 0.005 * cm)
    else:
        return max(1e-10, 0.3 * gap)


def _lbfgs_optimize(a_init, beta, n, deadline, maxiter=5000):
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _normalized_score(a_init)
    try:
        def obj(zv):
            av = _softmax(zv)
            cv = _fft_convolve(av, av)
            cmv = float(np.max(cv))
            shifted = (cv - cmv) / beta
            log_sum = float(np.max(shifted) + np.log(np.sum(np.exp(shifted - np.max(shifted)))))
            return 2.0 * n * (cmv + beta * log_sum)

        def grad(zv):
            av = _softmax(zv)
            cv = _fft_convolve(av, av)
            cmv = float(np.max(cv))
            shifted = (cv - cmv) / beta
            mx = float(np.max(shifted))
            e = np.exp(shifted - mx)
            w = e / e.sum()
            w_rev = w[::-1]
            h_full = _fft_convolve(w_rev, av)
            h = h_full[n - 1:2 * n - 1]
            W = float(np.dot(w, cv))
            g = 4.0 * n * av * (h - W)
            return g

        res = minimize(obj, z, method='L-BFGS-B', jac=grad,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-10, 'maxls': 50, 'maxfun': 20000})
        if res.x is not None and np.all(np.isfinite(res.x)):
            a_new = _softmax(res.x)
            sc_new = _normalized_score(a_new)
            if np.isfinite(sc_new) and sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
    except Exception:
        pass
    return best_a, best_score


def _run_continuation(a_init, deadline, rng):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _normalized_score(a_init)
    c0 = _fft_convolve(a_init, a_init)
    max_c0 = float(np.max(c0))
    beta_list = [0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.035, 0.025,
                 0.018, 0.013, 0.01, 0.008, 0.006, 0.005, 0.004,
                 0.003, 0.0025, 0.002, 0.0015, 0.0012, 0.001,
                 0.0008, 0.0006, 0.0005, 0.0004, 0.0003,
                 0.0002, 0.00015, 0.0001, 0.00007, 0.00005, 0.00003, 0.00002]
    beta_list = [b * max_c0 for b in beta_list]
    n_betas = len(beta_list)
    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        frac = max(0.01, 2.0 / (remaining + 2.0))
        time_alloc = (deadline - time.monotonic()) * frac
        time_alloc = min(time_alloc, 80.0)
        lvl_dl = time.monotonic() + max(5.0, time_alloc)
        a_new, sc_new = _lbfgs_optimize(best_a, beta, n, lvl_dl,
                                         maxiter=min(5000, max(500, int(time_alloc * 100))))
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
    return best_a, best_score


def _adam_optimize(a_init, deadline, rng, lr_init=0.2):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _normalized_score(a_init)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    kick_counter = 0
    while time.monotonic() < deadline:
        t += 1
        a = _softmax(z)
        c = _fft_convolve(a, a)
        beta = _adaptive_beta(c)
        cm = float(np.max(c))
        shifted = (c - cm) / beta
        mx = float(np.max(shifted))
        e = np.exp(shifted - mx)
        w = e / e.sum()
        w_rev = w[::-1]
        h_full = _fft_convolve(w_rev, a)
        h = h_full[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        lr_eff = lr * min(1.0, max(0.1, cm / max(1e-15, best_score / (2.0 * n))))
        z = z - lr_eff * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()
        a = _softmax(z)
        sc = _normalized_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            m *= 0.5
            v *= 0.5
        if t % 1000 == 0:
            lr *= 0.95
        if t % 2000 == 0 and lr < 0.05:
            z = z + 0.03 * rng.normal(size=n)
            z = z - z.mean()
            lr = max(lr, 0.1)
            kick_counter += 1
            if kick_counter > 10:
                break
    return best_a, best_score


def _coordinate_descent(a_init, deadline, rng, n_iterations=5000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _normalized_score(a)
    for it in range(n_iterations):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        cm = float(np.max(c))
        k_peak = int(np.argmax(c))
        valid = np.arange(max(0, k_peak - n + 1), min(n - 1, k_peak) + 1)
        if len(valid) == 0:
            break
        contrib = a[valid] * a[k_peak - valid]
        order = np.argsort(contrib)[::-1]
        n_try = min(5, len(order))
        improved = False
        for oi in range(n_try):
            idx = int(valid[order[oi]])
            a_trial = a.copy()
            a_trial[idx] *= 0.95
            a_trial /= a_trial.sum()
            sc = _normalized_score(a_trial)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
                improved = True
                break
        if not improved:
            c2 = _fft_convolve(a, a)
            k_low = int(np.argmin(c2))
            valid2 = np.arange(max(0, k_low - n + 1), min(n - 1, k_low) + 1)
            if len(valid2) > 0:
                contrib2 = a[valid2] * a[k_low - valid2]
                order2 = np.argsort(contrib2)[::-1]
                for oi in range(min(3, len(order2))):
                    idx = int(valid2[order2[oi]])
                    a_trial = a.copy()
                    a_trial[idx] *= 1.02
                    a_trial /= a_trial.sum()
                    sc = _normalized_score(a_trial)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        improved = True
                        break
        if not improved and it > 100:
            small_idx = np.argsort(a)[:3]
            found = False
            for idx in small_idx:
                a_trial = a.copy()
                a_trial[idx] *= 1.1
                a_trial /= a_trial.sum()
                sc = _normalized_score(a_trial)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    found = True
                    break
            if not found:
                break
    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print("  NEW BEST: norm={:.10f} (n={}) [{}] t={:.0f}s".format(
                best_score, len(p), label, time.monotonic() - t_start))

    parent = _init_from_parent()
    if parent is not None:
        sc_p = _normalized_score(parent)
        print("Parent: n={}, norm_score={:.10f}".format(len(parent), sc_p))
        update_best(parent, sc_p, "parent")
    else:
        n0 = 2048
        parent = _make_profiles(n0)[5]
        update_best(parent, _normalized_score(parent), "default")

    # Phase 1: Profile exploration (60s)
    phase1_end = min(deadline, t_start + 60.0)
    print("Phase 1: Profile exploration")
    n_cur = len(best_seq)
    profiles = _make_profiles(n_cur)
    for i, p in enumerate(profiles):
        if time.monotonic() > phase1_end:
            break
        sc = _normalized_score(p)
        update_best(p, sc, "prof{}".format(i))
    print("  Phase 1 done: best={:.10f}".format(best_score))

    # Phase 2: L-BFGS continuation (700s)
    phase2_end = min(deadline, t_start + 760.0)
    print("Phase 2: L-BFGS continuation")
    a_start = best_seq.copy()
    n_restarts = 3
    per_restart = (phase2_end - time.monotonic()) / max(1, n_restarts)
    for r in range(n_restarts):
        if time.monotonic() > phase2_end - 10:
            break
        if r == 0:
            a_cur = a_start
        else:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise = 0.05 * (r + 1)
            z = z + noise * rng.normal(size=n_cur)
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()
            a_cur = _softmax(z)
        dl = min(phase2_end - 5, time.monotonic() + per_restart)
        p_best, sc_best = _run_continuation(a_cur, dl, rng)
        update_best(p_best, sc_best, "cont_r{}".format(r))
    print("  Phase 2 done: best={:.10f}".format(best_score))

    # Phase 3: Adam GD (300s)
    phase3_end = min(deadline, t_start + 1060.0)
    print("Phase 3: Adam GD")
    if time.monotonic() < phase3_end:
        dl = min(phase3_end - 5, time.monotonic() + 280.0)
        p_best, sc_best = _adam_optimize(best_seq, dl, rng, lr_init=0.15)
        update_best(p_best, sc_best, "adam")
        if time.monotonic() < phase3_end - 30:
            dl2 = min(phase3_end - 5, time.monotonic() + 150.0)
            p_best2, sc_best2 = _adam_optimize(best_seq, dl2, rng, lr_init=0.05)
            update_best(p_best2, sc_best2, "adam_slow")
    print("  Phase 3 done: best={:.10f}".format(best_score))

    # Phase 4: Coordinate descent (200s)
    phase4_end = min(deadline, t_start + 1260.0)
    print("Phase 4: Coordinate descent")
    if time.monotonic() < phase4_end:
        dl = min(phase4_end - 5, time.monotonic() + 180.0)
        p_best, sc_best = _coordinate_descent(best_seq, dl, rng)
        update_best(p_best, sc_best, "coord")
    print("  Phase 4 done: best={:.10f}".format(best_score))

    # Phase 5: Try larger n (200s)
    phase5_end = min(deadline, t_start + 1460.0)
    print("Phase 5: Larger n")
    if time.monotonic() < phase5_end:
        n_curr = len(best_seq)
        if n_curr <= 2048:
            n_new = 4096
            p_interp = _interp_profile(best_seq, n_new)
            sc_interp = _normalized_score(p_interp)
            update_best(p_interp, sc_interp, "interp_n{}".format(n_new))
            if time.monotonic() < phase5_end - 30:
                dl = min(phase5_end - 10, time.monotonic() + 150.0)
                p_best, sc_best = _run_continuation(p_interp, dl, rng)
                update_best(p_best, sc_best, "cont_n{}".format(n_new))
    print("  Phase 5 done: best={:.10f}".format(best_score))

    # Phase 6: Final fine polish (remaining)
    phase6_end = deadline
    print("Phase 6: Final polish")
    if time.monotonic() < phase6_end:
        n_final = len(best_seq)
        c_final = _fft_convolve(best_seq, best_seq)
        max_c_final = float(np.max(c_final))
        beta_final = 1e-5 * max_c_final
        dl = min(phase6_end - 10, time.monotonic() + (phase6_end - time.monotonic()) * 0.5)
        p_best, sc_best = _lbfgs_optimize(best_seq, beta_final, n_final, dl, maxiter=10000)
        update_best(p_best, sc_best, "final_lbfgs")
        if time.monotonic() < phase6_end - 10:
            dl2 = min(phase6_end - 5, time.monotonic() + (phase6_end - time.monotonic()) * 0.4)
            p_best2, sc_best2 = _adam_optimize(best_seq, dl2, rng, lr_init=0.03)
            update_best(p_best2, sc_best2, "final_adam")
    print("  Phase 6 done: best={:.10f}".format(best_score))

    # Final: one more coordinate descent pass
    if best_seq is not None and time.monotonic() < deadline - 30:
        dl = min(deadline - 10, time.monotonic() + 40.0)
        p_best, sc_best = _coordinate_descent(best_seq, dl, rng)
        update_best(p_best, sc_best, "final_coord")

    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profiles(256)[2]
        print("FINAL: norm_score={:.10f}, n={}".format(_normalized_score(final_seq), len(final_seq)))
        return [float(x) for x in final_seq]
    fallback = _make_profiles(256)[2]
    print("FALLBACK: n=256, score={:.10f}".format(_normalized_score(fallback)))
    return [float(x) for x in fallback]
