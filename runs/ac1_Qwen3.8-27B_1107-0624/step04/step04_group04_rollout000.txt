    n = len(a)
    a = a.copy()
    s = a.sum()
    a = a / s  # normalize to sum = 1
    best_a = a.copy()
    best_score = score(a)
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        c = convolve(a, a)
        M = c.max()
        
        # Find active lags
        active = np.argsort(c)[-K:]
        
        # Build LP constraint matrix
        # For each active lag k, the constraint is:
        #   sum_j A[k,j] * b_j <= M
        # where A[k,j] = a[k-j] if 0 <= k-j < n, else 0
        
        A_mat = np.zeros((K, n))
        for i, k in enumerate(active):
            k = int(k)
            # b_j contributes to c(k) if 0 <= k-j < n, i.e., k-n+1 <= j <= k
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                A_mat[i, j] = a[k - j]
        
        # Solve LP: max sum(b) s.t. A_mat @ b <= M * ones(K), b >= 0
        # Also add: sum(b) <= 2 (to keep bounded)
        # Actually, let's add bounds: 0 <= b_j <= 2*max(a)
        b_max = 2.0 * a.max()
        
        from scipy.optimize import linprog
        res = linprog(
            c=-np.ones(n),
            A_ub=A_mat,
            b_ub=np.full(K, M),
            bounds=[(0, b_max)] * n,
            method='highs'
        )
        
        if res.success:
            b = res.x
            b_sum = b.sum()
            if b_sum > 0:
                b = b / b_sum  # normalize
                # Tempered update
                t = 0.1  # step size
                a_new = (1 - t) * a + t * b
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                
                sc = score(a_new)

                    best_score = sc

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: Active-set LP refinement based on the SOTA paper's approach,
combined with gradient-based polishing and diverse profile initialization.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_norm_fn(a):
    """Score with sum(a) = 1 (i.e., 2*n*max(c))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_fn(a):
    """Full score: 2*n*max(c)/(sum a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    """Normalize to sum = 1."""
    s = a.sum()
    if s < 1e-30:
        return a
    return a / s

def _active_set_lp_step(a, K=50, use_lp=True):
    """
    One step of the active-set LP refinement.
    Returns the new sequence (normalized) and the LP solution quality.
    """
    n = len(a)
    c = _full_convolve(a, a)
    M = float(c.max())
    
    if M < 1e-15:
        return a.copy(), 0.0
    
    # Find top-K active lags
    active = np.argsort(c)[-K:]
    
    # Build constraint matrix
    A_mat = np.zeros((K, n))
    for i in range(K):
        k = int(active[i])
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            A_mat[i, j_lo:j_hi+1] = a[k-j_hi:k-j_lo+1][::-1]
    
    if not use_lp:
        # Fallback: uniform weight on active set
        h = A_mat.T @ (np.ones(K) / K)
        # h[j] = (1/K) * sum_k a[k-j]
        # Update direction: reduce where h is large
        return h, M
    
    # Solve LP: max sum(b) s.t. A_mat @ b <= M * ones(K), 0 <= b <= b_max
    b_max = 3.0 * a.max()  # upper bound to ensure boundedness
    
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0, b_max)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            b = res.x
            b_sum = float(b.sum())
            if b_sum > 1e-15:
                return b / b_sum, b_sum
            else:
                return a.copy(), 0.0
        else:
            return a.copy(), 0.0
    except Exception:
        return a.copy(), 0.0

def _lp_refine(a_init, deadline, K_list=None, t_list=None, max_iter=200):
    """
    Active-set LP refinement with adaptive K and step size.
    """
    if K_list is None:
        K_list = [20, 30, 50, 80, 100]
    if t_list is None:
        t_list = [0.1, 0.05, 0.02, 0.01]
    
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    it = 0
    K_idx = 0
    t_idx = 0
    no_improve = 0
    
    while time.monotonic() < deadline and it < max_iter:
        K = K_list[min(K_idx, len(K_list) - 1)]
        t = t_list[min(t_idx, len(t_list) - 1)]
        
        b, lp_quality = _active_set_lp_step(a, K=K, use_lp=True)
        
        if lp_quality == 0:
            # LP failed, try uniform weight
            h, M = _active_set_lp_step(a, K=K, use_lp=False)
            # Update in log space
            z = np.log(np.maximum(a, 1e-15))
            z = z - z.mean()
            h_centered = h - h.mean()
            step = 0.01 * t
            z = z - step * h_centered / (np.abs(h_centered).max() + 1e-15)
            z = np.clip(z, -80, 80)
            z = z - z.mean()
            a_new = _softmax_fn(z)
        else:
            a_new = (1 - t) * a + t * b
            a_new = np.maximum(a_new, 0)
            a_new = _normalize(a_new)
        
        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Decrease step size gradually
            if no_improve == 0 and it > 10:
                t_idx = min(t_idx + 1, len(t_list) - 1)
                if t_idx >= len(t_list) - 1:
                    K_idx = min(K_idx + 1, len(K_list) - 1)
        else:
            no_improve += 1
            # Try smaller step
            t_idx = min(t_idx + 1, len(t_list) - 1)
            if no_improve > 20:
                # Reset
                t_idx = 0
                K_idx = max(0, K_idx - 1)
                no_improve = 0
                # Add small perturbation
                a = a + 0.001 * np.random.default_rng(it).normal(size=n)
                a = np.maximum(a, 0)
                a = _normalize(a)
        
        it += 1
        if it % 50 == 0:
            print(f"    LP refine: it={it}, score={best_score:.10f}, K={K}, t={t}")
    
    return best_a, best_score

def _grad_refine(a_init, deadline, lr=0.05, beta_init=None, max_iter=50000):
    """Gradient-based refinement with adaptive beta and line search."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    if beta_init is None:
        c = _full_convolve(a, a)
        beta_init = 0.01 * float(c.max())
    beta = beta_init
    
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_count += 1
        
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Smoothed max gradient
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = e / s
        
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n-1:2*n-1]
        
        # Gradient in z-space: a_j * (h_j - W) * 2n
        grad = 2.0 * n * a * (h - W)
        
        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t_count)
        vh = v / (1 - b2 ** t_count)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)
        
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Adaptive beta
        if t_count % 50 == 0:
            beta = max(1e-10, beta * 0.9)
        
        # Adaptive lr
        if t_count % 500 == 0 and best_score < 1.51:
            lr = max(0.002, lr * 0.95)
    
    return best_a, best_score

def _make_profiles(n):
    """Generate diverse initial profiles."""
    profiles = []
    
    # Uniform (baseline, score = 2)
    profiles.append(np.ones(n) / n)
    
    # Exponential decay
    for alpha in [0.001, 0.003, 0.005, 0.01, 0.02]:
        a = np.exp(-alpha * np.arange(n))
        profiles.append(a / a.sum())
    
    # Symmetric exponential (peak in middle)
    for alpha in [0.001, 0.003, 0.005]:
        mid = (n - 1) / 2.0
        a = np.exp(-alpha * np.abs(np.arange(n) - mid))
        profiles.append(a / a.sum())
    
    # Cosine modulation
    for A in [0.1, 0.3, 0.5, 0.7]:
        for freq in [1, 2, 3]:
            t = np.arange(n) / n
            a = 1.0 + A * np.cos(2 * np.pi * freq * t)
            a = np.maximum(a, 0.01)
            profiles.append(a / a.sum())
    
    # Linear
    for slope in [-1.0, -0.5, 0.5, 1.0]:
        t = np.arange(n) / (n - 1)
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 0.01)
        profiles.append(a / a.sum())
    
    # Quadratic (U-shaped and inverted)
    for c2 in [-0.5, -1.0, 0.5, 1.0]:
        t = 2.0 * np.arange(n) / (n - 1) - 1.0
        a = 1.0 + c2 * t ** 2
        a = np.maximum(a, 0.01)
        profiles.append(a / a.sum())
    
    # Power law
    for p in [0.5, 1.0, 1.5, 2.0, -0.5, -1.0]:
        a = (np.arange(n) + 1) ** p
        a = np.maximum(a, 1e-6)
        profiles.append(a / a.sum())
    
    # "Tapered" - smooth step up then down
    for width in [0.2, 0.3, 0.5]:
        t = np.arange(n) / (n - 1)
        a = np.ones(n)
        # Taper first 'width' fraction
        taper = np.minimum(1.0, np.minimum(t / width, (1 - t) / width))
        taper = np.maximum(taper, 0.01)
        a = taper
        profiles.append(a / a.sum())
    
    # Spline-like (smooth bump)
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.3]:
            t = np.arange(n) / (n - 1)
            a = np.exp(-((t - center) / width) ** 2)
            profiles.append(a / a.sum())
    
    return profiles

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return _normalize(p)

def _lbfgs_refine(a_init, deadline, max_time=None):
    """L-BFGS refinement with beta continuation."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    c0 = _full_convolve(a, a)
    max_c0 = float(c0.max())
    
    if max_time is None:
        max_time = max(1.0, deadline - time.monotonic())
    
    beta_starts = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    betas = [b * max(max_c0, 1e-10) for b in beta_starts]
    per_level = max_time / len(betas)
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level)
        
        def obj(z_cur):
            a_cur = _softmax_fn(z_cur)
            c = _full_convolve(a_cur, a_cur)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            log_sum = float(np.log(np.sum(np.exp(c_shifted))))
            return 2.0 * n * (cm + beta * log_sum)
        
        def grad(z_cur):
            a_cur = _softmax_fn(z_cur)
            c = _full_convolve(a_cur, a_cur)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300:
                w = np.zeros_like(c)
                w[np.argmax(c)] = 1.0
            else:
                w = e / s
            W = float(np.dot(w, c))
            a_rev = a_cur[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n-1:2*n-1]
            g = 2.0 * n * a_cur * (h - W)
            return g
        
        try:
            result = minimize(
                obj, z, jac=grad,
                method='L-BFGS-B',
                options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-8, 'maxls': 30}

            if result.x is not None and np.all(np.isfinite(result.x)):

                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:

                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score
