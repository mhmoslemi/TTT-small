Combines LP-based direction finding (maximizing sum(g) s.t. f*g <= M, g>=0)

import numpy as np

import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(f):
    n = len(f)
    row_parts = []
    col_parts = []
    data_parts = []
    for j in range(n):
        rows = np.arange(j, j + n)
        row_parts.append(rows)
        col_parts.append(np.full(n, j))
        data_parts.append(f)
    row_indices = np.concatenate(row_parts)
    col_indices = np.concatenate(col_parts)
    data = np.concatenate(data_parts)
    return sp.csr_matrix((data, (row_indices, col_indices)), shape=(2 * n - 1, n))


def _lp_direction(f, M, g_max, time_limit=20):
    n = len(f)
    T = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, g_max)] * n
    try:
        result = linprog(c_obj, A_ub=T, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'time_limit': time_limit, 'presolve': True})
        if result.success and result.x is not None:
            g = np.maximum(0.0, result.x)
            if np.all(np.isfinite(g)) and g.sum() > 1e-10:
                return g
    except Exception as e:
        print(f"  LP error: {e}")
    return None


def _subgradient_step(f, top_k=5):
    n = len(f)
    c = _full_convolve(f, f)
    cm = float(c.max())
    idx = np.argsort(c)[-top_k:]
    grad = np.zeros(n)
    for k in idx:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        m_lo = k - hi
        m_hi = k - lo
        seg = f[m_lo:m_hi + 1][::-1]
        grad[lo:hi + 1] += 2.0 * seg
    gn = np.linalg.norm(grad)
    if gn > 1e-15:

    return grad, cm

    if best_seq is not None and time.monotonic() < phase1_end:
        p_best, sc_best = _lp_iteration(best_seq, phase1_end, max_lp_calls=300)
        update_best(p_best, sc_best, "LP_phase1")

    # Phase 2: Smooth L-BFGS refinement (200s)
    phase2_end = min(deadline, t_start + 630.0)
    print(f"Phase 2: L-BFGS smooth refinement (target {phase2_end - time.monotonic():.0f}s)")
    if best_seq is not None and time.monotonic() < phase2_end:
        p_best, sc_best = _lbfgs_smooth(best_seq, phase2_end, max_time=phase2_end - time.monotonic() - 5)
        update_best(p_best, sc_best, "lbfgs_phase2")

    # Phase 3: More LP iterations (300s)
    phase3_end = min(deadline, t_start + 930.0)
    print(f"Phase 3: LP refinement (target {phase3_end - time.monotonic():.0f}s)")
    if best_seq is not None and time.monotonic() < phase3_end:
        p_best, sc_best = _lp_iteration(best_seq, phase3_end, max_lp_calls=200)
        update_best(p_best, sc_best, "LP_phase3")

    # Phase 4: Gradient polish (100s)
    phase4_end = min(deadline, t_start + 1030.0)
    print(f"Phase 4: Gradient polish (target {phase4_end - time.monotonic():.0f}s)")
    if best_seq is not None and time.monotonic() < phase4_end:
        p_best, sc_best = _gradient_polish(best_seq, phase4_end, lr=0.003, n_iter=5000)
        update_best(p_best, sc_best, "grad_polish")

    # Phase 5: Nesterov smooth with tiny beta (200s)
    phase5_end = min(deadline, t_start + 1230.0)
    print(f"Phase 5: Nesterov fine (target {phase5_end - time.monotonic():.0f}s)")
    if best_seq is not None and time.monotonic() < phase5_end:
        bf = [1e-4, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]
        p_best, sc_best = _nesterov_smooth(best_seq, phase5_end, bf, lr=0.008, mom=0.92)
        update_best(p_best, sc_best, "nesterov_fine")

    # Phase 6: Try longer sequence (300s)
    phase6_end = min(deadline, t_start + 1530.0)
    print(f"Phase 6: Longer sequence (target {phase6_end - time.monotonic():.0f}s)")
    if best_seq is not None and time.monotonic() < phase6_end:
        n_curr = len(best_seq)
        target_n = max(4096, n_curr * 2)
        if target_n <= 8192:
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            print(f"  Interpolated to n={target_n}: score={sc_interp:.10f}")
            if sc_interp < best_score:
                update_best(p_interp, sc_interp, f"interp_n{target_n}")
            # Try LP on longer sequence
            if time.monotonic() < phase6_end - 60:
                p_best, sc_best = _lp_iteration(p_interp, min(phase6_end - 10, time.monotonic() + 180), max_lp_calls=50)
                update_best(p_best, sc_best, f"LP_n{target_n}")
            # Try L-BFGS on longer
            if time.monotonic() < phase6_end - 30:
                p_best, sc_best = _lbfgs_smooth(p_interp, min(phase6_end - 5, time.monotonic() + 100), max_time=100)
                update_best(p_best, sc_best, f"lbfgs_n{target_n}")

    # Phase 7: Final LP + polish (remaining)
    phase7_end = deadline
    print(f"Phase 7: Final (target {phase7_end - time.monotonic():.0f}s)")
    if best_seq is not None and time.monotonic() < phase7_end:
        remaining = phase7_end - time.monotonic()
        if remaining > 60:
            p_best, sc_best = _lp_iteration(best_seq, min(phase7_end - 30, time.monotonic() + remaining - 40), max_lp_calls=100)
            update_best(p_best, sc_best, "LP_final")
        if time.monotonic() < phase7_end - 20:
            bf_final = [1e-6, 5e-7, 2e-7]
            p_best, sc_best = _nesterov_smooth(best_seq, min(phase7_end - 5, time.monotonic() + 40), bf_final, lr=0.003, mom=0.95)
            update_best(p_best, sc_best, "nesterov_final")

    print(f"All phases done. Best score: {best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profile(4096, "linear")
        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned inf/NaN, internal: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profile(4096, "linear")
    print(f"FALLBACK: n=4096, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
