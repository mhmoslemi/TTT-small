def lp_peak_reduction(a, M, method='highs'):
    n = len(a)
    # Variables: b_0, ..., b_{n-1}
    # Objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: conv(a, b)_k <= M for k = 0, ..., 2n-2
    # conv(a,b)_k = sum_j a_j * b_{k-j} (where b_i = 0 outside [0,n-1])
    # For each k, this is a linear function of b
    
    # A_ub shape: (2n-1, n)
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = a[idx]
    b_ub = np.full(2*n-1, M)
    
    # b >= 0 is handled by bounds
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
    
    if result.success:

    return None

But the LP doesn't enforce this. The LP only ensures conv(a,b) <= M, not conv(b,b) <= M.

So for the mixing to work, we either need:
1. t to be small enough that t^2 * (max(conv(b,b)) - M) < 2tM * (1 - max(conv(a,b))/M) (the "first-order" improvement)
2. Or add an additional constraint to the LP

Option 1 is simpler. For small t, the linear term (in t) dominates:
First-order change: 2t(1-t)(M' - M) where M' = max(conv(a,b)) <= M
If M' < M, we get improvement for any small t.
If M' = M (binding constraint), the first-order term is 0 and we need the second-order term to be negative, which requires max(conv(b,b)) < M.

So in practice, we should:
- If max(conv(a,b)) < M: use a moderate t (e.g., t=0.1 to 0.3)
- If max(conv(a,b)) ≈ M: use a very small t (e.g., t=0.01)

Alternatively, we can add the constraint conv(b,b) <= M to the LP, but that makes it non-linear (since conv(b,b) is quadratic in b). So we can't do that directly.

A compromise: after solving the LP, check max(conv(b,b)). If it's too large, reduce t.

def _lp_direction(a, M, active_tol=0.02, max_active=50):
    """Solve LP to find direction b that maximizes sum(b) subject to conv(a,b) <= M."""
    n = len(a)
    c_aa = _conv(a)
    
    # Find active constraints
    cm = c_aa.max()
    threshold = cm * (1.0 - active_tol)
    active_mask = c_aa >= threshold
    active_idx = np.where(active_mask)[0]
    
    if len(active_idx) > max_active:

        order = np.argsort(top_vals)[::-1][:max_active]
        active_idx = active_idx[order]

    m = len(active_idx)
    A = np.zeros((m, n))
    for i, k in enumerate(active_idx):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if js <= je:
            # A[i, js:je+1] = a[k-je : k-js+1][::-1]
            A[i, js:je+1] = a[k - np.arange(js, je+1)]
    
    # Solve LP
    c_obj = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, cm
    return None, cm

    # For efficiency, note that sum_k w_k * a_{k-i} is the i-th entry of the "anti-convolution"
    # which equals conv(w, a_reversed) at position (n-1+i) ... let me verify:
    # conv(w, a_rev)[m] = sum_k w[k] * a_rev[m-k] = sum_k w[k] * a[n-1-m+k]
    # Set n-1-m+k = j => k = j - n + 1 + m
    # conv(w, a_rev)[m] = sum_j w[j-n+1+m] * a[j]
    # For m = n-1-i: conv(w,a_rev)[n-1-i] = sum_j w[j-i] * a[j]
    # That's sum_j a_j * w_{j-i} = sum_k a_k * w_{k-i} ... YES!
    # So sum_k w_k * a_{k-i} ... hmm, indices are swapped.
    # sum_k w_k a_{k-i}: let l = k-i, = sum_l w_{l+i} a_l = sum_l a_l w_{l+i}
    # conv(w, a_rev)[n-1-i] = sum_l a_l w_{l-i}
    # These are different (w_{l+i} vs w_{l-i}).
    # OK let me just do: sum_k w_k a_{k-i} = (conv of w and a) at a specific position
    # conv(w, a)[m] = sum_k w[k] a[m-k]
    # Set m-k = k'-i => k = m - k' + i
    # This is getting too convoluted. Let me just compute the gradient numerically
    # using the fact that for our purpose, the gradient of the smoothed max
    # w.r.t. log-softmax parameters is:
    # d obj / d z_i = d obj / d a_i * a_i (chain rule for softmax)
    # And d obj / d a_i = 2n * sum_k w_k * (a_{k-i} + a_{i-k}) [with bounds]
    
    # Efficient computation:
    # sum_k w_k * a_{k-i} for all i at once:
    # This is the cross-correlation of w and a, which equals conv(w, a_flip) shifted
    a_flip = a[::-1]
    cross = _conv(w, a_flip)  # length len(w)+len(a)-1
    # cross[m] = sum_k w[k] * a_flip[m-k] = sum_k w[k] * a[n-1-m+k]
    # We want for each i: sum_k w[k] * a[k-i]
    # Let j = k-i: sum_j w[j+i] * a[j]
    # cross[n-1+i] = sum_k w[k] * a[n-1-(n-1+i)+k] = sum_k w[k] * a[k-i]
    # YES! So sum_k w[k]*a[k-i] = cross[n-1+i] for valid i
    h1 = cross[n - 1:2 * n - 1]  # h1[i] = sum_k w[k] * a[k-i]
    
    # Similarly, sum_k w[k] * a[i-k]:
    # = sum_k w[k] * a[(-k+i)] 
    # Let l = -k+i => k = i-l: sum_l w[i-l] * a[l]
    # cross[m] = sum_k w[k] * a[n-1-m+k], set n-1-m+k = l, k = l - n + 1 + m
    # cross[n-1-i] = sum_k w[k] * a[n-1-(n-1-i)+k] = sum_k w[k] * a[i+k]
    # Hmm that's a[i+k] not a[i-k].
    # Actually by symmetry: if w is symmetric (w[k] = w[2n-2-k] since c is symmetric),
    # then sum_k w[k]*a[i-k] = sum_k w[2n-2-k]*a[i-k] = sum_j w[j]*a[i-(2n-2-j)]
    # This is getting messy. Since w is derived from c which is symmetric, w IS symmetric.
    # So sum_k w[k]*a[i-k] = sum_k w[2n-2-k]*a[i-k], let j=2n-2-k:
    # = sum_j w[j]*a[i-2n+2+j] ... this is out of range for most j.
    # 
    # Simpler: since c is symmetric (c[k]=c[2n-2-k]), w is also symmetric.
    # sum_k w[k]*a[i-k]: let m = i-k, k = i-m:
    # = sum_m w[i-m]*a[m] = sum_m a[m]*w[i-m]
    # cross_conv = conv(a, w)[m] = sum_k a[k]*w[m-k]
    # Set m-k = i-m => k = m - i + m = 2m - i ... no.
    # conv(a,w)[i] = sum_k a[k]*w[i-k] = sum_m a[m]*w[i-m] ... YES!
    cross2 = _conv(a, w)  # length 2n-1
    h2 = cross2[:n]  # h2[i] = sum_k a[k]*w[i-k] = sum_m w[i-m]*a[m] = sum_k w[k]*a[i-k]
    
    # Wait, I need to be more careful. Let me re-derive:
    # d/d a_i [sum_k w_k * c_k] where c_k = sum_j a_j * a_{k-j}
    # = sum_k w_k * (a_{k-i} + a_{i-k}) [with a_j = 0 for j outside [0,n-1]]
    # = sum_k w_k * a_{k-i} + sum_k w_k * a_{i-k}
    # First term T1_i = sum_k w_k * a_{k-i}:
    #   Valid k: 0<=k<len(w) and 0<=k-i<n => i <= k <= i+n-1
    #   = sum_{k=i}^{i+n-1} w_k * a_{k-i} (with k bounded by len(w)-1=2n-2)
    # Second term T2_i = sum_k w_k * a_{i-k}:
    #   Valid k: 0<=k<len(w) and 0<=i-k<n => i-n+1 <= k <= i
    #   = sum_{k=max(0,i-n+1)}^{min(i,2n-2)} w_k * a_{i-k}
    
    # For T1: this is a cross-correlation. T1_i = sum_k w_k * a_{k-i}
    # = (w ✻ a)_i (cross-correlation at lag i)
    # = conv(w, a_flip)[n-1+i] where a_flip = a[::-1]
    # Let me verify: conv(w, a_flip)[m] = sum_k w_k * a_flip[m-k] = sum_k w_k * a[n-1-m+k]
    # For m = n-1+i: sum_k w_k * a[n-1-(n-1+i)+k] = sum_k w_k * a[k-i] ✓
    
    # For T2: T2_i = sum_k w_k * a_{i-k}
    # = conv(a, w)[i] ... let me check: conv(a,w)[m] = sum_k a_k * w[m-k]
    # For m = i: sum_k a_k * w[i-k] = sum_j w[j] * a[i-j] ... let j = i-k, k=i-j:
    # = sum_j a[i-j] * w[j] = sum_j w_j * a[i-j] ✓
    
    # So: grad_a_i = 2n * (T1_i + T2_i) = 2n * (conv(w, a_flip)[n-1+i] + conv(a, w)[i])
    # And grad_z_i = grad_a_i * a_i (softmax chain rule)
    
    # But wait, I also need to account for the fact that we're using w = exp((c-cm)/beta) / sum(...)
    # which depends on a. For the gradient of the smoothed max, we use:
    # d/d a [max_smooth(c)] = sum_k w_k * d c_k / d a
    # where w_k = exp((c_k - cm)/beta) / sum_j exp((c_j - cm)/beta)
    # We treat w as fixed (it's the standard approach for the log-sum-exp gradient)
    
    # Actually, the gradient of the log-sum-exp is:
    # d/d x [beta * log(sum exp((x_i - m)/beta))] = sum_i w_i * dx_i/dx where m = max(x)
    # So the gradient of the smoothed max objective w.r.t. a is:
    # 2n * sum_k w_k * d c_k / d a = 2n * (T1 + T2)
    
    # For the log-softmax parameterization:
    # d obj / d z_i = d obj / d a_i * d a_i / d z_i = grad_a_i * a_i * (1 - a_i) ... 
    # No, for softmax: a_i = exp(z_i)/sum(exp(z)), d a_i / d z_j = a_i * (delta_ij - a_j)
    # So d obj / d z_j = sum_i grad_a_i * a_i * (delta_ij - a_j)
    # = a_j * (sum_i grad_a_i * a_i - a_j * sum_i grad_a_i)
    # Hmm, this is getting complex. Let me just use:
    # d obj / d z_j = a_j * grad_a_j - a_j * sum_i (a_i * grad_a_i)
    # = a_j * (grad_a_j - sum_i a_i * grad_a_i)
    
    grad_a = 2.0 * n * (h1 + h2)
    
    # Now convert to gradient w.r.t. z (log-softmax parameters)
    # Using the fact that the objective only depends on a up to scale (it's homogeneous of degree 2
    # in the numerator and degree 2 in the denominator), we can simplify:
    # Actually for the normalized case (sum a = 1), the gradient w.r.t. z is:
    # d obj / d z_j = a_j * grad_a_j - a_j * sum_i a_i * grad_a_i
    W = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - W)
    
    if not np.all(np.isfinite(grad_z)):
        return np.zeros_like(z)
    return grad_z


def _adapt_gd(a_init, deadline, lr_init=0.2, verbose=True):
    """Adaptive gradient descent with automatic beta scheduling."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    t_start_local = time.monotonic()

    for it in range(100000):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _conv(a)
        cm = float(c.max())

        # Adaptive beta
        sorted_c = np.sort(c)[::-1]
        k = min(10, len(sorted_c) - 1)
        gap = sorted_c[0] - sorted_c[k]
        beta = max(1e-10, 0.5 * gap if gap > 0.05 * cm else 0.1 * cm)

        w = np.exp((c - cm) / beta)
        w = w / w.sum()

        a_flip = a[::-1]
        cross1 = _conv(w, a_flip)
        h1 = cross1[n - 1:2 * n - 1]
        cross2 = _conv(a, w)
        h2 = cross2[:n]

        grad_a = 2.0 * n * (h1 + h2)
        W = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W)

        if not np.all(np.isfinite(grad_z)):
            break

        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        a_new = _softmax_fn(z)
        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        if t % 500 == 0:
            lr = max(0.01, lr * 0.97)

    if verbose:
        print(f"  AGD done: iters={t}, best_score={best_score:.10f}")
    return best_a, best_score


def _lbfgs_polish(a_init, deadline, verbose=True):
    """L-BFGS with continuation on beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    t_start_local = time.monotonic()

    c0 = _conv(a_init)
    cm0 = float(c0.max())

    beta_list = [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    beta_list = [b * cm0 for b in beta_list]

    n_betas = len(beta_list)
    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 120.0)
        level_dl = time.monotonic() + time_budget

        try:
            res = minimize(
                _grad_smooth_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_grad_smooth_grad,
                options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': 10000}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_new = _softmax_fn(res.x)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = res.x.copy()
                    if verbose:
                        print(f"  L-BFGS beta={beta:.2e}: score={sc:.10f}")
        except Exception:
            pass

    if verbose:
        print(f"  L-BFGS done: best_score={best_score:.10f}")
    return best_a, best_score


def _peak_surgery(a, deadline, verbose=True):
    """Direct peak reduction by modifying individual elements."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    t_start_local = time.monotonic()

    for outer in range(200):
        if time.monotonic() > deadline:
            break
        a_cur = _softmax_fn(z)
        c = _conv(a_cur)
        cm = float(c.max())
        k_star = int(np.argmax(c))

        valid_i = np.arange(max(0, k_star - n + 1), min(n - 1, k_star) + 1)
        contributions = a_cur[valid_i] * a_cur[k_star - valid_i]
        top_idx = valid_i[np.argsort(contributions)[::-1][:15]]

        improved = False
        for idx in top_idx:
            for delta in [0.005, 0.01, 0.02, 0.05, 0.1]:
                z_trial = z.copy()
                z_trial[idx] -= delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm(a_trial)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            small_idx = np.argsort(a_cur)[:10]
            for idx in small_idx:
                for delta in [0.01, 0.03, 0.05]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm(a_trial)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            break

    if verbose:
        print(f"  Peak surgery done: score={best_score:.10f}")
    return best_a, best_score


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for a1, a2 in [(1.0, 0.5), (2.0, -0.3), (1.5, 0.2), (-1.0, 0.5), (0.0, 1.0), (0.0, -1.0)]:
        a = 1.0 + a1 * u + a2 * u ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for rate in [1.0, 2.0, 3.0, -1.0, -2.0, -3.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())
    return profiles


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage optimization combining active-set LP peak reduction,
    adaptive gradient descent, and L-BFGS polishing.
    The LP approach finds directions that provably reduce the peak,
    while gradient methods handle fine-scale optimization.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm(parent)
        print(f"Parent loaded: n={n_parent}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 4096
        parent = _make_profiles(n_parent)[2]
        update_best(parent, _score_norm(parent), "default")

    # --- Phase 0: Quick profile exploration (30s) ---
    phase0_end = min(deadline, t_start + 30.0)
    print("\n=== Phase 0: Profile exploration ===")
    n_curr = len(best_seq)
    for idx, p in enumerate(_make_profiles(n_curr)[:10]):
        sc = _score_norm(p)
        update_best(p, sc, f"profile{idx}")

    # --- Phase 1: Initial LP optimization (300s) ---
    phase1_end = min(deadline, t_start + 330.0)
    print(f"\n=== Phase 1: LP optimization (loose) ===")
    a_work = best_seq.copy()
    p_best, sc_best = _lp_optimize(a_work, phase1_end, verbose=True)
    update_best(p_best, sc_best, "lp_loose")

    # --- Phase 2: Gradient polish (200s) ---
    phase2_end = min(deadline, t_start + 530.0)
    print(f"\n=== Phase 2: Adaptive GD ===")
    if time.monotonic() < phase2_end:
        p_best, sc_best = _adapt_gd(best_seq, phase2_end, lr_init=0.15, verbose=True)
        update_best(p_best, sc_best, "agd_1")

    # --- Phase 3: Second LP round (500s) ---
    phase3_end = min(deadline, t_start + 1030.0)
    print(f"\n=== Phase 3: LP optimization (tight) ===")
    if time.monotonic() < phase3_end:
        a_work = best_seq.copy()
        p_best, sc_best = _lp_optimize(a_work, phase3_end, verbose=True)
        update_best(p_best, sc_best, "lp_tight")

    # --- Phase 4: L-BFGS fine continuation (350s) ---
    phase4_end = min(deadline, t_start + 1380.0)
    print(f"\n=== Phase 4: L-BFGS fine polish ===")
    if time.monotonic() < phase4_end:
        p_best, sc_best = _lbfgs_polish(best_seq, phase4_end, verbose=True)
        update_best(p_best, sc_best, "lbfgs_fine")

    # --- Phase 5: Peak surgery + final GD (200s) ---
    phase5_end = min(deadline, t_start + 1580.0)
    print(f"\n=== Phase 5: Peak surgery ===")
    if time.monotonic() < phase5_end:
        p_best, sc_best = _peak_surgery(best_seq, min(phase5_end, time.monotonic() + 100.0), verbose=True)
        update_best(p_best, sc_best, "peak_surgery")
        if time.monotonic() < phase5_end:
            p_best2, sc_best2 = _adapt_gd(best_seq, phase5_end, lr_init=0.05, verbose=True)
            update_best(p_best2, sc_best2, "agd_final")

    # --- Phase 6: Final LP polish (remaining) ---
    phase6_end = deadline
    print(f"\n=== Phase 6: Final LP polish ===")
    if time.monotonic() < phase6_end:
        a_work = best_seq.copy()
        p_best, sc_best = _lp_optimize(a_work, phase6_end, verbose=True)
        update_best(p_best, sc_best, "lp_final")

    # --- Final validation ---
    print(f"\n=== Final Result ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_full(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profiles(256)[2]
    print(f"FALLBACK: n=256, score={_score_full(fallback):.10f}")
    return [float(x) for x in fallback]
