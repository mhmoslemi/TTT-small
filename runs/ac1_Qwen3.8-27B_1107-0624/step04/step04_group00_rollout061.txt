One more thought: the score is `2n * max_conv / sum^2`. Since we can scale `a` freely, let's normalize `sum(a) = 1` and minimize `2n * max_conv`.

For the polynomial parametrization, let me use:
`a[i] = exp(sum_{k=0}^{K} c_k * (i/n - 0.5)^k) / sum_j exp(sum_k c_k * (j/n - 0.5)^k)`

This ensures `a >= 0` and `sum(a) = 1`. The parameters are `c_0, ..., c_K`.

For K=0: uniform
For K=1: linear (exponential in i)
For K=2: quadratic (exponential of a parabola) - this gives U-shape or inverted-U
For K=3: cubic
etc.

The key shapes:
- K=2 with positive c_2: U-shaped (high at edges, low in middle)
- K=2 with negative c_2: inverted-U (low at edges, high in middle)
- K=1: monotone (high at one edge)

For minimizing peak autocorrelation, I expect the U-shape (K=2, positive c_2) to be good, possibly with some odd-order terms for asymmetry.

Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses polynomial parametrization for global exploration,

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    """Score with sum normalized to 1: 2n * max(conv(a,a))"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a = a / s
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_raw(a):
    """The actual evaluation function score."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _logsumexp(x, beta):
    """Compute logsumexp(x/beta) * beta (smoothed max)."""
    x = np.asarray(x, dtype=np.float64)
    xm = x.max()
    if xm < -1e30:
        return 0.0
    shifted = (x - xm) / beta
    # Clamp for numerical stability
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return float(xm)
    return float(xm + beta * np.log(s))

def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _poly_profile(coeffs, n):
    """Generate a profile from polynomial coefficients."""
    t = np.linspace(0, 1, n) - 0.5
    log_a = np.zeros(n)
    for k, c in enumerate(coeffs):
        log_a += c * t**k
    log_a -= log_a.max()
    a = np.exp(log_a)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s

def _poly_score(coeffs, n):
    a = _poly_profile(coeffs, n)
    return _score_norm(a)

def _poly_grad(coeffs, n):
    """Gradient of _score_norm w.r.t. polynomial coefficients.
    Uses finite differences for simplicity (K is small)."""
    eps = 1e-7
    g = np.zeros(len(coeffs))
    f0 = _poly_score(coeffs, n)
    for i in range(len(coeffs)):
        cp = coeffs.copy()
        cp[i] += eps
        g[i] = (_poly_score(cp, n) - f0) / eps
    return g

def _run_adaptive_gd(a_init, deadline, rng, lr=0.1, max_iter=100000):
    """Adam-like gradient descent with adaptive beta."""
    n = len(a_init)
    a_init = a_init / a_init.sum()
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Adaptive beta
        sorted_c = np.sort(c)[::-1]
        k = min(10, len(sorted_c)-1)
        gap = sorted_c[0] - sorted_c[k]
        beta = max(1e-10, 0.5 * gap) if gap > 0.05*cm else max(1e-10, 0.1*cm)
        
        # Compute gradient
        shifted = np.clip((c - cm) / beta, -500, 500)
        w = np.exp(shifted)
        w = w / w.sum()
        w_rev = w[::-1]
        h_full = _full_convolve(w_rev, a)
        h = h_full[2*n-2 : n-1 : -1]
        W = float(np.dot(a, h))
        grad = 2.0 * n * a * (h - W)
        
        if not np.all(np.isfinite(grad)):
            break
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        
        # Adaptive LR
        lr_adapt = lr * min(1.0, max(0.1, cm / max(1e-15, best_score / (2*n))))
        z = z - lr_adapt * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        a = _softmax(z)
        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Periodic learning rate decay
        if t % 2000 == 0:
            lr = max(0.01, lr * 0.9)
        
        # Periodic kick
        if t % 10000 == 0:
            z = z + 0.03 * rng.normal(size=n)
            z = z - z.mean()
    
    return best_a, best_score

def _peak_targeted_search(a_init, deadline, rng, n_peaks=20, n_steps=200):
    """Directly target and reduce the top peaks."""
    n = len(a_init)
    a = a_init / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Find top peaks
        top_k = np.argsort(c)[::-1][:n_peaks]
        
        # For each top peak, find the most contributing coordinate
        improved = False
        for k in top_k:
            # conv(a,a)_k = sum_i a[i] * a[k-i]
            valid_i = np.arange(max(0, k - n + 1), min(n - 1, k) + 1)
            contribs = a[valid_i] * a[k - valid_i]
            top_i = valid_i[np.argmax(contribs)]
            
            # Try reducing a[top_i]
            for frac in [0.01, 0.02, 0.05, 0.1]:
                a_trial = a.copy()
                reduction = a_trial[top_i] * frac
                a_trial[top_i] -= reduction
                # Redistribute to a random small element
                small_idx = np.argmin(a_trial)
                a_trial[small_idx] += reduction
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                
                sc = _score_norm(a_trial)
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try increasing a small element (to spread mass)
            small_idx = np.argsort(a)[:5]
            for idx in small_idx:
                for frac in [0.01, 0.05, 0.1]:
                    a_trial = a.copy()
                    # Take from the largest element
                    large_idx = np.argmax(a_trial)
                    transfer = a_trial[large_idx] * frac
                    a_trial[large_idx] -= transfer
                    a_trial[idx] += transfer
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = _score_norm(a_trial)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            # Small random perturbation
            z = np.log(np.maximum(a, 1e-15))
            z = z + 0.02 * rng.normal(size=n)
            z = z - z.mean()
            a_new = _softmax(z)
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
            else:
                break
    
    return best_a, best_score

def _poly_explore(n, deadline, max_K=10):
    """Explore polynomial parameterizations of various degrees."""
    best_a = None
    best_score = np.inf
    
    for K in range(0, max_K + 1):
        if time.monotonic() > deadline - 10:
            break
        
        # Multiple initializations for each K
        n_inits = 5
        for init_idx in range(n_inits):
            if time.monotonic() > deadline - 5:
                break
            
            coeffs = np.zeros(K + 1)
            if init_idx == 0:
                pass  # zeros = uniform
            else:
                # Random initialization with emphasis on even terms (U-shape)
                for k in range(1, K + 1):
                    if k % 2 == 0:
                        coeffs[k] = np.random.uniform(0.5, 3.0) * (1 if np.random.rand() > 0.3 else -1)
                    else:
                        coeffs[k] = np.random.uniform(-1.0, 1.0)
            
            try:
                res = minimize(_poly_score, coeffs, args=(n,),
                              method='Nelder-Mead',
                              options={'maxiter': 5000, 'xatol': 1e-12, 'fatol': 1e-14,
                                       'adaptive': True})
                if res.fun < best_score:
                    best_score = res.fun
                    best_a = _poly_profile(res.x, n)
            except Exception:
                pass
        
        # Also try L-BFGS on the polynomial params
        coeffs = np.zeros(K + 1)
        if K >= 2:
            coeffs[2] = 2.0  # Start with U-shape
        for init_idx in range(3):
            if time.monotonic() > deadline - 5:
                break
            if init_idx > 0:
                c = np.zeros(K+1)
                for k in range(1, K+1):
                    c[k] = np.random.uniform(-2, 2)
                coeffs = c
            
            try:
                res = minimize(_poly_score, coeffs, args=(n,),
                              method='Nelder-Mead',
                              options={'maxiter': 3000, 'xatol': 1e-12, 'fatol': 1e-14})
                if res.fun < best_score:
                    best_score = res.fun
                    best_a = _poly_profile(res.x, n)
            except Exception:
                pass
    
    if best_a is None:
        best_a = np.ones(n) / n
        best_score = _score_norm(best_a)
    
    return best_a, best_score

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines polynomial parametrization for global exploration, spectral flattening
via alternating projections, L-BFGS with ultra-fine continuation,
adaptive gradient descent, and peak-targeted coordinate search.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    """Score with sum normalized to 1: 2n * max(conv(a,a))."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a = a / s
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_raw(a):
    """The actual evaluation function score."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _logsumexp_smooth(c, beta):
    """Compute smooth approximation to max(c)."""
    c = np.asarray(c, dtype=np.float64)
    cm = c.max()
    shifted = np.clip((c - cm) / beta, -500, 500)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return float(cm)
    return float(cm + beta * np.log(s))


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _compute_h(w, a):
    """Compute h[j] = sum_k w[k] * a[k-j] for j=0,...,n-1."""
    n = len(a)
    w_rev = w[::-1]
    conv_full = _full_convolve(w_rev, a)
    h = conv_full[n - 1 :: -1]
    return h[:n]


def _smoothed_obj(z, beta, n):
    """Smoothed objective: 2n * logsumexp_smooth(conv(softmax(z),softmax(z)), beta)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    return _logsumexp_smooth(c, beta) * 2.0 * n


def _smoothed_grad(z, beta, n):
    """Gradient of smoothed objective w.r.t. z."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = c.max()
    shifted = np.clip((c - cm) / beta, -500, 500)
    w = np.exp(shifted)
    w = w / w.sum()
    h = _compute_h(w, a)
    W = float(np.dot(w, c))
    g = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(g)):
        return np.zeros_like(z)
    return g


def _poly_profile(coeffs, n):
    """Generate a profile from polynomial coefficients (log-space)."""
    t = np.linspace(0, 1, n) - 0.5
    log_a = np.zeros(n, dtype=np.float64)
    for k, c in enumerate(coeffs):
        log_a += c * (t ** k)
    log_a -= log_a.max()
    a = np.exp(log_a)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s

def _run_adaptive_gd(a_init, deadline, rng, lr_init=0.1, max_iter=100000):
    """Adam-like gradient descent with adaptive beta."""
    n = len(a_init)
    a_init = a_init / a_init.sum()
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Adaptive beta: focus on the gap between top peaks
        sorted_c = np.sort(c)[::-1]
        k = min(15, len(sorted_c) - 1)
        gap = sorted_c[0] - sorted_c[k]
        if gap < 0.05 * cm:
            beta = max(1e-10, 0.1 * cm)
        elif gap > 0.5 * cm:
            beta = max(1e-10, 0.01 * cm)
        else:
            beta = max(1e-10, 0.5 * gap)

        # Compute gradient

        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        # Adaptive learning rate

    n = len(a_init)
    a = a_init / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

            break

            valid_i = np.arange(max(0, k - n + 1), min(n - 1, k) + 1)
            if len(valid_i) == 0:

            contribs = a[valid_i] * a[k - valid_i]
            top_idx_list = valid_i[np.argsort(contribs)[::-1][:5]]

                    if reduction < 1e-12:

                    a_trial[top_i] -= reduction
                    # Redistribute to a few small elements

                    a_trial[small_idx] += reduction / 3.0
                    a_trial = np.maximum(a_trial, 0.0)
                    ts = a_trial.sum()
                    if ts < 1e-15:

                        a = a_trial.copy()
                        improved = True

            if improved:

            # Try mass redistribution
            small_idx = np.argsort(a)[:5]
            large_idx = np.argsort(a)[::-1][:3]
            for si in small_idx:
                for li in large_idx:
                    if si == li:
                        continue
                    for frac in [0.005, 0.01, 0.02]:
                        a_trial = a.copy()
                        transfer = a_trial[li] * frac
                        a_trial[li] -= transfer
                        a_trial[si] += transfer
                        a_trial = np.maximum(a_trial, 0.0)
                        ts = a_trial.sum()
                        if ts < 1e-15:
                            continue
                        a_trial = a_trial / ts
                        sc = _score_norm(a_trial)
                        if sc < best_score:
                            best_score = sc
                            best_a = a_trial.copy()
                            a = a_trial.copy()
                            improved = True
                            no_improve_count = 0
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            no_improve_count += 1
            if no_improve_count > 20:
                # Random restart from current best
                z = np.log(np.maximum(best_a, 1e-15))
                z = z + 0.05 * rng.normal(size=n)
                z = z - z.mean()
                a = _softmax(z)
                no_improve_count = 0
                if _score_norm(a) < best_score:
                    best_score = _score_norm(a)
                    best_a = a.copy()

    return best_a, best_score


def _poly_explore(n, deadline, rng, max_K=10):
    """Explore polynomial parameterizations of various degrees."""
    best_a = np.ones(n) / n
    best_score = _score_norm(best_a)

    for K in range(0, max_K + 1):
        if time.monotonic() > deadline - 10:
            break

        n_inits = max(3, 8 - K // 2)
        for init_idx in range(n_inits):
            if time.monotonic() > deadline - 5:
                break

            coeffs = np.zeros(K + 1)
            if init_idx == 0:
                pass  # uniform
            elif init_idx == 1:
                coeffs[2] = 2.0  # U-shape
                if K >= 4:
                    coeffs[4] = 1.0
            elif init_idx == 2:
                coeffs[2] = 3.0
                if K >= 3:
                    coeffs[3] = 0.5
                if K >= 4:
                    coeffs[4] = 1.5
            else:
                for k in range(1, K + 1):
                    if k % 2 == 0:
                        coeffs[k] = rng.uniform(0.5, 3.0) * (1 if rng.random() > 0.2 else -1)
                    else:
                        coeffs[k] = rng.uniform(-1.5, 1.5)

            try:
                res = minimize(_poly_score, coeffs, args=(n,),
                              method='Nelder-Mead',
                              options={'maxiter': 3000, 'xatol': 1e-13, 'fatol': 1e-15,
                                       'adaptive': True})
                if res.fun < best_score:
                    best_score = res.fun
                    best_a = _poly_profile(res.x, n)
            except Exception:
                pass

    return best_a, best_score


def _spectral_flatten(a_init, n_iters=200, deadline=None):
    """Iteratively flatten the spectrum while maintaining non-negativity."""
    n = len(a_init)
    a = a_init / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Compute FFT
        A = np.fft.fft(a)
        mag = np.abs(A)
        mean_mag = mag.mean()

        # Flatten: set all magnitudes to the mean
        new_mag = np.full_like(mag, mean_mag)
        # Keep phase
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)

        # Inverse FFT
        a_new = np.real(np.fft.ifft(A_new))

        # Project: non-negativity and normalization
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining polynomial parametrization,
    spectral flattening, L-BFGS with ultra-fine continuation,
    adaptive gradient descent, and peak-targeted coordinate search.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n_parent = len(parent) if parent is not None else 4096

    if parent is not None:
        sc_parent = _score_norm(parent)
        print(f"Parent: n={n_parent}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")

    # === Phase 1: Polynomial exploration (120s) ===
    phase1_end = min(deadline, t_start + 120.0)
    print("Phase 1: Polynomial exploration")

    a_poly, sc_poly = _poly_explore(n_parent, phase1_end, rng, max_K=10)
    update_best(a_poly, sc_poly, f"poly_n{n_parent}")

    # === Phase 2: Spectral flattening (150s) ===
    phase2_end = min(deadline, t_start + 270.0)
    print("Phase 2: Spectral flattening")

    if time.monotonic() < phase2_end:
        a_spec, sc_spec = _spectral_flatten(best_seq, n_iters=100, deadline=phase2_end)
        update_best(a_spec, sc_spec, "spectral_flatten")

    # === Phase 3: L-BFGS continuation (450s) ===
    phase3_end = min(deadline, t_start + 720.0)
    print("Phase 3: L-BFGS continuation")

    n_best = len(best_seq)
    a_start = best_seq.copy()

    for r in range(3):
        if time.monotonic() > phase3_end - 30:
            break

        if r == 0:
            a_cur = a_start.copy()
        else:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise = 0.04 * r
            z = z + noise * rng.normal(size=n_best)
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            a_cur = _softmax(z)

        per_dl = min(phase3_end - 10, time.monotonic() + (phase3_end - time.monotonic()) / (3 - r))
        p_best, sc_best = _run_lbfgs_continuation(a_cur, per_dl, label=f"r{r}")
        update_best(p_best, sc_best, f"lbfgs_r{r}")

    # === Phase 4: Adaptive GD (250s) ===
    phase4_end = min(deadline, t_start + 970.0)
    print("Phase 4: Adaptive GD")

    if time.monotonic() < phase4_end:
        per_dl = min(phase4_end - 10, time.monotonic() + 220.0)
        p_best, sc_best = _run_adaptive_gd(best_seq, per_dl, rng, lr_init=0.08)
        update_best(p_best, sc_best, "adaptive_gd")

    # === Phase 5: Peak targeted search (150s) ===
    phase5_end = min(deadline, t_start + 1120.0)
    print("Phase 5: Peak targeted search")

    if time.monotonic() < phase5_end:
        per_dl = min(phase5_end - 10, time.monotonic() + 130.0)
        p_best, sc_best = _peak_targeted_search(best_seq, per_dl, rng, n_peaks=30, max_steps=300)
        update_best(p_best, sc_best, "peak_search")

    # === Phase 6: Very fine L-BFGS (second round) (200s) ===
    phase6_end = min(deadline, t_start + 1320.0)
    print("Phase 6: Very fine L-BFGS")

    if time.monotonic() < phase6_end:
        # Use very small betas only
        c = _full_convolve(best_seq, best_seq)
        cm = float(c.max())
        fine_betas = [b * cm for b in [0.0005, 0.0003, 0.0002, 0.0001, 0.00005, 0.00003, 0.00001, 0.000005]]

        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        a_cur = best_seq.copy()
        sc_cur = _score_norm(a_cur)

        for beta in fine_betas:
            if time.monotonic() > phase6_end - 5:
                break
            try:
                res = minimize(_smoothed_obj, z, args=(beta, n),
                              method='L-BFGS-B', jac=_smoothed_grad,
                              options={'maxiter': 5000, 'ftol': 1e-22, 'gtol': 1e-16,
                                       'maxls': 200, 'maxfun': 20000})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    z = np.asarray(res.x)
                    a_new = _softmax(z)
                    sc = _score_norm(a_new)
                    if np.isfinite(sc) and sc < sc_cur:
                        sc_cur = sc
                        a_cur = a_new.copy()
            except Exception:
                pass

        update_best(a_cur, sc_cur, "fine_lbfgs")

    # === Phase 7: Try n=2048 and interpolate back (100s) ===
    phase7_end = min(deadline, t_start + 1420.0)

        final_betas = [b * cm for b in [0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000001]]

        a_cur = best_seq.copy()
        sc_cur = _score_norm(a_cur)

        for beta in final_betas:

            try:

                              method='L-BFGS-B', jac=_smoothed_grad,
                              options={'maxiter': 5000, 'ftol': 1e-22, 'gtol': 1e-16,
                                       'maxls': 200, 'maxfun': 20000})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    z = np.asarray(res.x)
                    a_new = _softmax(z)
                    sc = _score_norm(a_new)
