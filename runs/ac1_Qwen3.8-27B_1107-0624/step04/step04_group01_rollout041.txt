"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a)=1, returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_linear_ramp(n):
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / a.sum()


def _make_power_ramp(n, alpha):
    a = np.arange(1, n + 1, dtype=np.float64)
    a = a ** alpha
    return a / a.sum()


def _make_sin_ramp(n):
    """Ramp modulated by sine for smoother transition."""
    t = np.linspace(0, 1, n)
    a = t + 0.1 * np.sin(np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_quadratic_ramp(n):
    t = np.linspace(0.01, 1, n)
    a = t ** 2
    return a / a.sum()


def _compute_grad(a, c, w, n):
    """
    Gradient of smooth objective w.r.t. log-space params z.
    a is normalized (sum=1), c=a*a (full conv), w=softmax(c/beta).
    Returns: 4*n*a*(h - W) where h[m]=sum_k w[k]*a[k-m], W=dot(w,c).
    """
    a_flip = a[::-1]
    wc = _full_convolve(w, a_flip)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adam_optimize(a_init, deadline, beta_schedule, max_iter=200000, lr=0.01):
    """Adam optimizer in log-space with beta-continuation for smooth max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    prev_best = best_score

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_e = float(e.sum())
            if s_e < 1e-300 or not np.isfinite(s_e):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_e

            grad = _compute_grad(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 200 == 0 or it_in_phase < 5:
                a_check = _softmax(z)
                sc = _score_norm(a_check)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_check.copy()

            if it_in_phase % 5000 == 0:
                if best_score < prev_best * 0.999:
                    current_lr = min(0.05, current_lr * 1.05)
                else:
                    current_lr = max(0.002, current_lr * 0.97)
                prev_best = best_score

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    n_freqs = rng.integers(1, max(2, n // 8))
    for _ in range(n_freqs):
        freq = rng.integers(1, n // 2)
        amp = rng.uniform(0.02, 0.15) * (1.0 if rng.random() < 0.5 else -1.0)
        phase = rng.uniform(0, 2 * np.pi)
        t = np.arange(n)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_local(a, rng, width=50):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    center = rng.integers(0, n)
    mask = np.zeros(n)
    lo = max(0, center - width)
    hi = min(n, center + width)
    t = np.linspace(-1, 1, hi - lo)
    mask[lo:hi] = np.cos(np.pi * t / 2) ** 2
    z = z + 0.1 * mask * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)
