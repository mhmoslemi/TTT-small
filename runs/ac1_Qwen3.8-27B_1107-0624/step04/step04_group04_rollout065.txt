Let me be very careful about the cross-correlation FFT. Let me define:

"""
Active-set subgradient optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses FFT-based autocorrelation, adaptive active set with growing K, momentum,
multiple diverse initializations, dual-LP weight refinement, and Adam polishing.
"""
import numpy as np
from scipy import signal
from scipy.optimize import linprog
import time


def _fft_autocorr(a):
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _fft_xcorr_valid(B, a):
    n = len(a)
    N = 1
    while N < 3 * n - 2:
        N *= 2
    Bf = B[::-1]
    fa = np.fft.rfft(a, N)
    fB = np.fft.rfft(Bf, N)
    conv = np.fft.irfft(fa * fB, N)
    return conv[2 * n - 2:3 * n - 2]


def _subgrad_uni(a, active):
    n = len(a)
    B = np.zeros(2 * n - 1)
    B[active] = 1.0
    g = _fft_xcorr_valid(B, a)
    g /= len(active)
    return g


def _subgrad_wt(a, active, w):
    n = len(a)
    B = np.zeros(2 * n - 1)
    B[active] = w
    g = _fft_xcorr_valid(B, a)
    ts = w.sum()
    if ts > 1e-30:
        g /= ts
    return g


def _dual_lp(a, active):
    n = len(a)
    K = len(active)
    if K < 2 or K > 120:
        return np.ones(K) / K
    A = np.zeros((n, K))
    for ki in range(K):
        k = active[ki]
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A[lo:hi + 1, ki] = a[k - np.arange(lo, hi + 1)]
    try:
        res = linprog(np.ones(K), A_ub=-A, b_ub=-np.ones(n),
                      method='highs', options={'presolve': True, 'time_limit': 3})
        if res.status == 0 and res.x is not None:
            w = res.x
            if np.all(np.isfinite(w)) and w.sum() > 1e-30:
                return w / w.sum()
    except Exception:
        pass
    return np.ones(K) / K


def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_autocorr(a)
    return 2.0 * n * float(c.max()) / (s * s)


def _score1(a):
    n = len(a)
    c = _fft_autocorr(a)
    return 2.0 * n * float(c.max())


def _mk_exp(n, eps):
    r = 1.0 - eps / n
    if r <= 0:
        return np.ones(n) / n
    a = r ** np.arange(n)
    return a / a.sum()


def _mk_sin(n, freq, depth, ph):
    base = _mk_exp(n, 2.0)
    t = np.arange(n, dtype=np.float64) / n
    mod = 1.0 + depth * np.sin(2.0 * np.pi * freq * t + ph)
    a = base * np.maximum(mod, 0.01)
    return a / a.sum()


def _mk_cos(n, gamma):
    t = np.arange(n, dtype=np.float64) / n
    a = 0.5 * (1.0 + np.cos(np.pi * np.abs(2.0 * t - 1.0) ** gamma))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _mk_quad(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u * u
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _sg_opt(a0, deadline, k0, k1, mom, dual, dfrq, mxit):
    a = a0.copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score1(a)
    c = _fft_autocorr(a)
    cm = float(c.max())
    v = np.zeros(n)
    alpha = 0.3
    stall = 0

    for it in range(mxit):
        if time.monotonic() > deadline:
            break

        prog = min(1.0, it / (0.8 * mxit))
        K = int(k0 + (k1 - k0) * prog)
        if K > 2 * n - 1:
            K = 2 * n - 1
        if K < 1:
            K = 1

        if K == 2 * n - 1:
            active = np.arange(2 * n - 1)
        else:
            active = np.argpartition(c, -K)[-K:]

        if dual and it % dfrq == 0 and 2 <= K <= 100:
            w = _dual_lp(a, active)
            g = _subgrad_wt(a, active, w)
        else:
            g = _subgrad_uni(a, active)

        gm = float(np.mean(g))
        d = gm - g
        v = mom * v + (1.0 - mom) * d

        neg = v < 0
        if neg.any():
            tm = float((a[neg] / (-v[neg])).min())
        else:
            tm = 1e10
        t = min(0.5 * alpha, tm * 0.85)
        if t < 1e-15:
            v *= 0.5
            alpha *= 0.9
            continue

        a_new = a + t * v
        a_new = np.maximum(a_new, 0.0)
        c_new = _fft_autocorr(a_new)
        cm_new = float(c_new.max())

        if cm_new < cm + 1e-14:
            a = a_new
            c = c_new
            cm = cm_new
            stall = 0
        else:
            t *= 0.3
            a_new = a + t * v
            a_new = np.maximum(a_new, 0.0)
            c_new = _fft_autocorr(a_new)
            cm_new = float(c_new.max())
            if cm_new < cm:
                a = a_new
                c = c_new
                cm = cm_new
                stall = 0
            else:
                alpha *= 0.9
                v *= 0.5
                stall += 1
                if stall > 3000:
                    break

        if 2.0 * n * cm < best_sc:
            best_sc = 2.0 * n * cm
            best_a = a.copy()

    return best_a, best_sc


def _adam_opt(a0, deadline, lr, tk, mxit):
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-15))
    z = z - z.mean()
    m = np.zeros(n)
    vv = np.zeros(n)
    tc = 0
    b1, b2, eps_a = 0.9, 0.999, 1e-8
    best_a = a0.copy()
    best_sc = _score1(a0)

    for it in range(mxit):
        if time.monotonic() > deadline:
            break
        tc += 1

        z_max = z.max()
        a = np.exp(z - z_max)
        a /= a.sum()

        c = _fft_autocorr(a)
        cm = c.max()
        if tk >= 2 * n - 1:
            active = np.arange(2 * n - 1)
        else:
            active = np.argpartition(c, -tk)[-tk:]

        g = _subgrad_uni(a, active)
        gz = a * (g - float(np.dot(a, g)))

        m = b1 * m + (1.0 - b1) * gz
        vv = b2 * vv + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** tc)
        vh = vv / (1.0 - b2 ** tc)
        z = z - lr * mh / (np.sqrt(vh) + eps_a)
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        z_max2 = z.max()
        a_new = np.exp(z - z_max2)
        a_new /= a_new.sum()
        sc = _score1(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

    return best_a, best_sc
