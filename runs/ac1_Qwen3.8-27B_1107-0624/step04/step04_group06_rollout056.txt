                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.1:
                    beta = max(1e-6, beta * 0.4)

        limit = max(100, no_imp_limit)
        if no_imp > limit:
            if best_score < 1.51:
                ns = 0.04
            elif best_score < 1.53:
                ns = 0.08
            else:
                ns = 0.15
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            if use_symmetry:
                p = _make_symmetric(p)
                z = np.log(p + 1e-30)
                z -= float(np.mean(z))
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _lp_solve_sym(a, M):
    """LP solve with symmetry on b. Maximize sum(b) s.t. (a*b)_k <= M, b>=0, b symmetric."""
    n = len(a)
    n_half = (n + 1) // 2
    n_cons = n  # By symmetry, only need k=0..n-1

    # Build constraint matrix A (n_cons x n_half)
    # (a*b)_k = sum_j a_j * b_{k-j} where b is symmetric
    # B_j for j in 0..n_half-1, with b_j = b_{n-1-j} for the full sequence
    rows = []
    cols = []
    vals = []

    for k in range(n_cons):
        for j in range(n_half):
            # b_j contributes to (a*b)_k through a_{k-j} (if valid) and a_{k-(n-1-j)} (if valid)
            # Full b index is j for the first half
            # In the full sequence, b_full[t] = b_half[min(t, n-1-t)]
            # (a*b_full)_k = sum_t a[t] * b_full[k-t]
            # b_full[k-t] = b_half[min(k-t, n-1-(k-t))] if 0<=k-t<n
            # This is complex; let's just build the full matrix and then symmetrize

    # Simpler approach: build full A (n_cons x n), solve, then average with flip
    # But that defeats the purpose. Let's build the symmetric LP properly.

    # Actually, let's just build the full LP. For n<=2048 it should be fast enough.
    rows = []
    cols = []
    vals = []
    n_full = 2 * n - 1
    for k in range(n_full):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())

    A_ub = csr_matrix((vals, (rows, cols)), shape=(n_full, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(n_full)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_solve_sym_fast(a, M):
    """Faster LP: exploit that a is symmetric to reduce constraints."""
    n = len(a)
    # By symmetry of a, (a*b)_k = (a*b)_{2n-2-k} if b is symmetric
    # So we only need constraints k = 0..n-1
    # And we enforce b symmetric by solving with n/2 variables

    n_half = (n + 1) // 2
    n_cons = n  # k = 0, 1, ..., n-1

    # Build A: for each constraint k, A[k, j_half] = coefficient of b_half[j_half]
    # b_full[t] = b_half[min(t, n-1-t)]
    # (a*b_full)_k = sum_{t=0}^{n-1} a[t] * b_full[k-t] (where b_full[q]=0 if q<0 or q>=n)
    # = sum_{t=0}^{n-1} a[t] * b_half[min(k-t, n-1-(k-t))] (if 0<=k-t<n)

    rows = []
    cols = []
    vals = []

    for k in range(n_cons):
        t_start = max(0, k - n + 1)
        t_end = min(n - 1, k)
        for t in range(t_start, t_end + 1):
            q = k - t  # index into b_full
            jh = min(q, n - 1 - q)  # index into b_half
            rows.append(k)
            cols.append(jh)
            vals.append(a[t])

    A_ub = csr_matrix((vals, (rows, cols)), shape=(n_cons, n_half))
    c_obj = -np.ones(n_half)
    b_ub_arr = M * np.ones(n_cons)
    bounds = [(0, None)] * n_half
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        b_half = np.maximum(res.x, 0.0)
        # Reconstruct full b
        b_full = np.zeros(n)
        for j in range(n_half):
            b_full[j] = b_half[j]
            b_full[n - 1 - j] = b_half[j]
        if n % 2 == 0:
            # Even: b_full[j] = b_half[j] for j < n/2, b_full[n-1-j] = b_half[j]
            # This double-counts, so divide by 2
            b_full = b_full / 2.0
        # Normalize
        s = b_full.sum()
        if s > 1e-15:
            b_full = b_full / s
        return b_full
    return None


def _lp_refine(a, deadline, t_init=0.35, max_solves=80, use_sym=False):
    """LP refinement with adaptive M reduction."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0
    t = t_init
    stall = 0

    solver = _lp_solve_sym_fast if use_sym else _lp_solve_sym

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1

        b = solver(a, M)
        if b is None:
            # Try larger M
            b = solver(a, M * 1.02)
            if b is None:
                b = solver(a, M * 1.05)
                if b is None:
                    break

        s_b = float(b.sum())
        if s_b < 1e-10:
            break

        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)

        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 10:
                t = max(0.1, t * 0.7)
                stall = 0

        a = a_new
        c = _full_convolve(a)
        M = float(c.max()) * 0.998
        M = max(M, best_score / (2.0 * n) * 1.001)
        t = max(0.12, t * 0.94)

    return best_a, best_score


def _peak_flood(a, deadline, eta=0.04, n_iter_max=3000):
    """Multiplicative peak-flooding with adaptive threshold."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    eta_cur = eta

    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break

        M = float(c.max())
        score = float(2.0 * n * M)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        # Adaptive threshold
        threshold = M * 0.93
        hot_mask = c >= threshold
        if not hot_mask.any():
            hot_mask[int(np.argmax(c))] = True

        # Compute heat
        heat = np.zeros(n)
        hot_indices = np.where(hot_mask)[0]
        for k in hot_indices:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                partner = k - np.arange(i_start, i_end + 1)
                heat[i_start:i_end + 1] += a[partner]

        heat_max = float(heat.max())
        if heat_max < 1e-15:
            break
        heat = heat / heat_max

        factor = 1.0 - eta_cur * heat
        factor = np.maximum(factor, 0.02)
        a = a * factor
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        c = _full_convolve(a)

        if it % 80 == 0 and it > 0:
            if score >= best_score:
                eta_cur = max(0.005, eta_cur * 0.85)
            else:
                eta_cur = min(0.2, eta_cur * 1.1)

    return best_a, best_score


def _make_hann(n):
    k = np.arange(n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * k / (n - 1)))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_hamming(n):
    k = np.arange(n)
    a = 0.54 + 0.46 * np.cos(2.0 * np.pi * k / (n - 1))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_blackman(n):
    k = np.arange(n)
    a = 0.42 + 0.5 * np.cos(2.0 * np.pi * k / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * k / (n - 1))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_kaiser(n, alpha):
    k = np.arange(n)
    m = n - 1
    u = 2.0 * k / m - 1.0
    # Modified Bessel function I_0
    # Use series approximation for numerical stability
    x = alpha * np.sqrt(np.maximum(1.0 - u * u, 0.0))
    # I_0(x) approximation for x in [0, alpha]
    # Use numpy's implementation if available, otherwise series
    try:
        from scipy.special import i0
        b = i0(x)
    except ImportError:
        b = np.exp(x - 0.4 * x * x / (1.0 + x))  # rough approx
    a = b
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_poly(n, order=4):
    """Polynomial window: (1 - (2t-1)^order) style"""
    t = np.arange(n) / (n - 1)
    u = 2.0 * t - 1.0
    a = 1.0 - np.abs(u) ** order
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian(n, sigma_frac=0.3):
    k = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n - 1) / 2.0
    a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc):
            sc = _score_norm(p)
            if sc < best_score:
                best_seq = np.asarray(p, dtype=np.float64).copy()
                best_score = float(sc)
                print(f"  BEST: {best_score:.10f} n={p.size} [{label}] t={time.monotonic()-t_start:.0f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    if parent is not None:
        sc_parent = _score_norm(parent)
        print(f"Parent score: {sc_parent:.10f}")
        _update(parent, sc_parent, 'parent')

    # --- Phase 1: Quick Adam on parent (120s) ---
    phase1_end = t_start + 120.0
    print("Phase 1: Adam on parent...")
    if parent is not None and pn <= 2048:
        p1, sc1 = _adam_logit(parent, phase1_end, rng, lr=0.05, beta_frac=0.06,
                              beta_decay=0.94, no_imp_limit=200, use_symmetry=True)
        _update(p1, _score_norm(p1), 'adam_parent_sym')

    # --- Phase 2: Try larger n with good inits (300s) ---
    phase2_end = t_start + 420.0
    print(f"Phase 2: Multi-scale, score={best_score:.10f}...")

    inits_2048 = [
        ('hann', _make_hann(2048)),
        ('hamming', _make_hamming(2048)),
        ('blackman', _make_blackman(2048)),
        ('kaiser_5', _make_kaiser(2048, 5.0)),
        ('poly6', _make_poly(2048, 6)),
    ]

    # Also try interpolation from parent
    if parent is not None:
        p_interp = _interp_profile(parent, 2048)
        if p_interp is not None:
            inits_2048.insert(0, ('parent_2048', p_interp))

    per_init = max(15.0, (phase2_end - time.monotonic()) / len(inits_2048))
    for label, pa in inits_2048:
        if time.monotonic() > phase2_end - 10:
            break
        per = min(per_init, phase2_end - time.monotonic() - 5)
        if per < 10:
            break
        pa_sym = _make_symmetric(pa)
        p_r, sc_r = _adam_logit(pa_sym, time.monotonic() + per, rng,
                                 lr=0.05, beta_frac=0.06,
                                 beta_decay=0.94, no_imp_limit=150, use_symmetry=True)
        _update(p_r, _score_norm(p_r), f'n2048_{label}')

    # Try n=4096 if time allows
    if time.monotonic() < phase2_end - 40:
        pa = _make_hann(4096)
        pa_sym = _make_symmetric(pa)
        per = min(50.0, phase2_end - time.monotonic() - 5)
        if per > 15:
            p_r, sc_r = _adam_logit(pa_sym, time.monotonic() + per, rng,
                                     lr=0.04, beta_frac=0.05,
                                     beta_decay=0.95, no_imp_limit=150, use_symmetry=True)
            _update(p_r, _score_norm(p_r), 'n4096_hann')

    # Also try parent-scale LP refinement
    if parent is not None and pn <= 1024 and time.monotonic() < phase2_end - 30:
        p_lp, sc_lp = _lp_refine(parent, time.monotonic() + 40, t_init=0.3,
                                  max_solves=30, use_sym=True)
        _update(p_lp, sc_lp, 'lp_parent')

    # --- Phase 3: Intensive LP refinement on best (500s) ---
    phase3_end = t_start + 920.0
    print(f"Phase 3: LP refinement, score={best_score:.10f}...")
    if best_seq is not None and best_seq.size <= 4096:
        # Symmetric LP refinement
        bs = _make_symmetric(best_seq)
        per = min(200.0, phase3_end - time.monotonic() - 30)
        if per > 30:
            p3, sc3 = _lp_refine(bs, time.monotonic() + per, t_init=0.35,
                                  max_solves=100, use_sym=True)
            _update(p3, sc3, 'lp_sym')

        # Full LP refinement (non-symmetric)
        if time.monotonic() < phase3_end - 30 and best_seq.size <= 2048:
            per2 = min(120.0, phase3_end - time.monotonic() - 20)
            if per2 > 20:
                p3b, sc3b = _lp_refine(best_seq, time.monotonic() + per2, t_init=0.3,
                                        max_solves=60, use_sym=False)
                _update(p3b, sc3b, 'lp_full')

        # Adam after LP
        if time.monotonic() < phase3_end - 20:
            per3 = min(80.0, phase3_end - time.monotonic() - 10)
            if per3 > 15:
                p3c, sc3c = _adam_logit(best_seq, time.monotonic() + per3, rng,
                                         lr=0.03, beta_frac=0.04,
                                         beta_decay=0.96, no_imp_limit=120, use_symmetry=True)
                _update(p3c, sc3c, 'post_lp_adam')

    # --- Phase 4: Peak flooding + LP alternating (300s) ---
    phase4_end = t_start + 1220.0
    print(f"Phase 4: Flood+LP, score={best_score:.10f}...")
    if best_seq is not None:

        # LP after flood
        if time.monotonic() < phase4_end - 20 and best_seq.size <= 2048:
            per2 = min(80.0, phase4_end - time.monotonic() - 15)
            if per2 > 15:
                p4l, sc4l = _lp_refine(best_seq, time.monotonic() + per2, t_init=0.25,
                                        max_solves=40, use_sym=True)
                _update(p4l, sc4l, 'lp_after_flood')

        # Another flood round
        if time.monotonic() < phase4_end - 20:
            per3 = min(60.0, phase4_end - time.monotonic() - 10)
            if per3 > 15 and best_seq.size <= 4096:
                p4f2, sc4f2 = _peak_flood(best_seq, time.monotonic() + per3, eta=0.02)
                _update(p4f2, sc4f2, 'flood2')

    # --- Phase 5: Try alternative inits at best n (150s) ---
    phase5_end = t_start + 1370.0
    print(f"Phase 5: Alt inits, score={best_score:.10f}...")
    if time.monotonic() < phase5_end:
        n_try = best_seq.size if best_seq is not None else 1024
        if n_try > 4096:
            n_try = 2048
        n_try = min(n_try, 2048)

        alts = [
            ('poly4', _make_poly(n_try, 4)),
            ('poly8', _make_poly(n_try, 8)),
            ('gauss_025', _make_gaussian(n_try, 0.25)),
            ('gauss_035', _make_gaussian(n_try, 0.35)),
            ('kaiser_4', _make_kaiser(n_try, 4.0)),
            ('kaiser_6', _make_kaiser(n_try, 6.0)),
        ]

        per_alt = max(12.0, (phase5_end - time.monotonic()) / len(alts))
        for label, pa in alts:
            if time.monotonic() > phase5_end - 10:
                break
            pa_sym = _make_symmetric(pa)
            per = min(per_alt, phase5_end - time.monotonic() - 5)
            if per < 8:
                break
            p_alt, sc_alt = _adam_logit(pa_sym, time.monotonic() + per, rng,
                                         lr=0.05, beta_frac=0.07,
                                         beta_decay=0.93, no_imp_limit=100, use_symmetry=True)
            _update(p_alt, sc_alt, f'alt_{label}_n{n_try}')

    # --- Phase 6: Final intensive polish (remaining) ---
    phase6_end = deadline
    print(f"Phase 6: Final polish, score={best_score:.10f}...")
    if time.monotonic() < phase6_end and best_seq is not None:
        remaining = phase6_end - time.monotonic()

        # Intensive Adam with very small learning rate
        if remaining > 100:
            per1 = remaining * 0.3
            p6a, sc6a = _adam_logit(best_seq, time.monotonic() + per1, rng,
                                     lr=0.015, beta_frac=0.03,
                                     beta_decay=0.97, no_imp_limit=80, use_symmetry=True)
            _update(p6a, sc6a, 'final_adam1')

        # LP with very tight M
        if time.monotonic() < phase6_end - 30 and best_seq.size <= 2048:
            per2 = min(60.0, phase6_end - time.monotonic() - 15)
            if per2 > 20:
                p6l, sc6l = _lp_refine(best_seq, time.monotonic() + per2, t_init=0.15,
                                        max_solves=30, use_sym=True)
                _update(p6l, sc6l, 'final_lp')

        # Very fine Adam
        if time.monotonic() < phase6_end - 10:
            per3 = phase6_end - time.monotonic() - 5
            if per3 > 15:
                p6b, sc6b = _adam_logit(best_seq, time.monotonic() + per3, rng,
                                         lr=0.008, beta_frac=0.02,
                                         beta_decay=0.98, no_imp_limit=40, use_symmetry=True)
                _update(p6b, sc6b, 'final_adam2')

        # One more LP
        if time.monotonic() < phase6_end - 10 and best_seq.size <= 2048:
            per4 = phase6_end - time.monotonic() - 5
            if per4 > 15:
                p6c, sc6c = _lp_refine(best_seq, time.monotonic() + per4, t_init=0.1,
                                        max_solves=10, use_sym=True)
                _update(p6c, sc6c, 'final_lp2')

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_hann(1024)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"EVAL error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_hann(1024)
    print(f"FALLBACK n=1024, score={_score_norm(fb):.10f}")
    return [float(x) for x in fb]
