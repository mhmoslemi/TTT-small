def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 512:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _smooth_max(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    mval = float(c.max())
    if beta < 1e-13:
        return mval
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 0.0))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        return mval
    return mval + beta * float(np.log(s))


def _smooth_max_grad(c, beta):

def _build_conv_matrix(a):
    """Build sparse matrix A such that A @ b = convolve(a, b)."""
    n = len(a)

    rows = []
    cols = []
    vals = []

            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())

def _profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _profile_exp(n, lam):
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _profile_cosine(n, freq, amp):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _profile_quadratic(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + a1 * t + a2 * t * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _profile_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x1, x0, p0))


def _frank_wolfe_step(a, deadline_chunk):
    """
    Frank-Wolfe (conditional gradient) step.
    The linear minimization oracle over the simplex for f(b) = max_k conv(a,b)_k
    is: min_b max_k conv(a,b)_k s.t. b in simplex
    = min M s.t. A(a)*b <= M*1, sum(b)=1, b>=0
    This is an LP! But it's equivalent to finding the b that minimizes the max.
    Actually, the FW linear step minimizes <grad_f(a), b> where grad_f is a subgradient.
    For f(a) = max_k conv(a,a)_k, a subgradient at a is 2*a' where a' is the LP best response
    corresponding to the maximizer k.
    
    Actually, let's use the standard FW: 
    - Compute subgradient g of f at current point a
    - Solve: min_b <g, b> s.t. b in simplex -> b* = e_{argmin(g_i)}
    - Update: a_new = (1-t)*a + t*b*
    
    But this is very crude. A better approach is the "away steps" FW.
    
    For our problem, the subgradient of max_k conv(a,a)_k at a is:
    2 * conv(w, a) where w is uniform over the argmax set of conv(a,a).
    
    Let's implement this.
    """
    n = len(a)
    c = _full_convolve(a)
    maxc = float(c.max())
    # Subgradient: 2 * sum_k w_k * d/d a [conv(a,a)_k] = 2 * conv(w, a) where w = uniform on argmax
    mask = c >= maxc - 1e-9
    w = np.zeros_like(c)
    w[mask] = 1.0 / mask.sum()
    grad = 2.0 * _conv_cross(w, a)
    
    # FW step: b* = e_{argmin grad_i}
    # This is too crude. Let's use the LP oracle instead:
    # min_b <grad, b> s.t. b in simplex
    # The minimum is achieved at b* = e_{argmin(grad_i)}
    j_min = int(np.argmin(grad))
    b_star = np.zeros(n)
    b_star[j_min] = 1.0
    
    # Actually this is too aggressive. Let's use the LP best response instead.
    # The LP best response gives: max sum(b) s.t. conv(a,b) <= maxc, b >= 0
    # This is equivalent to finding the "flattest" b that keeps conv(a,b) bounded.
    b = _lp_best_response(a, maxc)
    if b is not None and b.sum() > 1e-10:
        b = b / b.sum()
        return b
    return None

"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 512:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _smooth_max_grad(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    mval = float(c.max())
    if beta < 1e-13:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-9
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 0.0))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-14):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -70.0, 70.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    p = np.maximum(p, 1e-15)
    s = float(p.sum())
    if s < 1e-30:
        return None
    return p / s


def _build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(n_rows):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        if j_min <= j_max:
            idx = np.arange(j_min, j_max + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(n_rows, n))
    return A


def _lp_best_response(a, M):
    n = len(a)
    A_ub = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_min_peak(a, M_init):
    """Solve: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0. Returns (b_opt, M_opt)."""
    n = len(a)
    A_conv = _build_conv_matrix(a)
    # Variables: b_0,...,b_{n-1}, M
    # Minimize: M (last variable)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    # Inequalities: A_conv @ b - M <= 0 for each row
    # Build constraint matrix: [A_conv | -I_{2n-1}] @ [b; M] <= 0
    from scipy.sparse import eye as speye, hstack
    I_neg = -speye(2 * n - 1)
    A_ub = hstack([A_conv, I_neg])
    b_ub = np.zeros(2 * n - 1)
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, M_init * 2)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.success:
        b_opt = np.maximum(res.x[:n], 0.0)
        M_opt = float(res.x[-1])
        return b_opt, M_opt
    return None, None


def _lp_refine(a, deadline, max_iters=150, use_min_peak=False):
    """Iterative LP best-response with adaptive step size."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    stall = 0
    t = 0.4

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        if use_min_peak and it % 3 == 2:
            b, M_new = _lp_min_peak(a, M)
            if b is not None and b.sum() > 1e-10:
                b = b / b.sum()
                M = max(M_new, 1e-12)
            else:
                b = _lp_best_response(a, M)
                if b is None or b.sum() < 1e-10:
                    break
                b = b / b.sum()
        else:
            b = _lp_best_response(a, M)
            if b is None or b.sum() < 1e-10:
                break
            b = b / b.sum()

        # Line search over step size
        best_t = t
        best_sc = best_score
        for tc in [0.15, 0.25, 0.35, 0.5, 0.65, 0.8]:
            a_try = (1 - tc) * a + tc * b
            a_try = np.maximum(a_try, 1e-15)
            a_try = a_try / a_try.sum()
            sc = _score_norm(a_try)
            if sc < best_sc:
                best_sc = sc
                best_t = tc

        a_new = (1 - best_t) * a + best_t * b
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)

        if sc_new < best_score - 1e-13:
            best_score = sc_new
            best_a = a_new.copy()
            stall = 0
            c = _full_convolve(a_new)
            M = float(c.max())
            t = min(0.8, best_t * 1.1)
        else:
            stall += 1
            t = max(0.1, t * 0.85)
            if stall > 12:
                break

        a = a_new

    return best_a, best_score


def _adam_logit(p0, deadline, rng, lr=0.05, beta_frac=0.08, beta_decay=0.93,
                no_imp_limit=350, min_beta=1e-5):
    n = p0.size
    p = np.maximum(p0, 1e-12).copy()
    p = p / p.sum()
    z = np.log(p.copy())
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(min_beta, beta_frac * maxc)

    best_p = p.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_grad(c, beta)
        g = _conv_cross(w, p)
        df_dp = 2.0 * g
        dot = float(np.dot(p, df_dp))
        gz = p * (df_dp - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -70.0, 70.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 150 == 0:
                    lr = max(0.005, lr * 0.92)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 25 == 0:
            beta = max(min_beta, beta * beta_decay)

        if no_imp > no_imp_limit:
            ns = 0.04 if best_score < 1.51 else 0.08
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -70.0, 70.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(min_beta, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score
