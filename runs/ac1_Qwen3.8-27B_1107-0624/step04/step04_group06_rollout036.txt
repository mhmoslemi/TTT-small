def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _normalized_convolve(a):
    """Compute convolution of a with itself, with a normalized to sum=1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    s = a.sum()
    if s < 1e-30:
        return np.zeros(2*len(a)-1)
    a = a / s
    return _full_convolve(a)


def _smooth_max_weights(c, beta):

        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:

        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w
