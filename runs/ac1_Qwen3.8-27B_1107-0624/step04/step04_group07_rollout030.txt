```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP fixed-point with continuation,
Adam with Nesterov momentum, coordinate descent, and multi-scale escalation.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    return np.maximum(c, 0.0)

def _score(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())

def _cross_corr(w, a):
    """Compute cross-correlation: (w * a_flipped)[n-1:2n-1]"""
    n = len(a)
    N = _next_pow2(2*n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n-1:2*n-1]

def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * np.log(s), w

def _build_conv_matrix(n, a):
    rows, cols, vals = [], [], []
    for k in range(2*n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def _lp_solve(a, M, n):
    A = _build_conv_matrix(n, a)
    res = linprog(-np.ones(n), A_ub=A, b_ub=M*np.ones(2*n-1),
                  bounds=[(0, None)]*n, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0), float(res.x.sum())
    return None, 0.0

def _lp_fixed_point(a, deadline, max_iters=60, line_search=True):
    """LP fixed-point with line search and adaptive M."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a)
        M = float(c.max())
        b, S = _lp_solve(a, M, n)
        if b is None or S < 1e-10:
            break
        b = b / b.sum()
        
        # Line search
        if line_search:
            best_t, best_s = 0.5, _score(a)
            for t in [0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
                a_t = np.maximum((1-t)*a + t*b, 1e-15)
                a_t = a_t / a_t.sum()
                s_t = _score(a_t)
                if s_t < best_s:
                    best_s, best_t = s_t, t
            a = np.maximum((1-best_t)*a + best_t*b, 1e-15)
            a = a / a.sum()
        else:
            a = b
        
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if best_sc - sc > -1e-10 and it > 5:
            # Converged
            break
    
    return best_a, best_sc

def _adam_smooth_max(a, deadline, rng, lr=0.05, beta_init=None, 
                     beta_decay=0.95, no_imp_limit=150, nesterov=True):
    """Adam with optional Nesterov momentum and adaptive beta."""
    n = len(a)
    a = np.maximum(a, 1e-12).copy()
    a = a / a.sum()
    z = np.log(a)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    c = _full_convolve(a)
    maxc = float(c.max())
    beta = beta_init if beta_init is not None else 0.1 * maxc
    
    best_z = z.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    z_prev = z.copy()
    
    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)
        g = _cross_corr(w, a)
        
        if nesterov:
            z_look = z + b1 * m  # Nesterov look-ahead
            a_look = np.exp(z_look - z_look.max())
            a_look = a_look / a_look.sum()
            c_look = _full_convolve(a_look)
            _, w_look = _smooth_max_weights(c_look, beta)
            g_look = _cross_corr(w_look, a_look)
            dot = float(np.dot(a_look, g_look))
            gz = a_look * (g_look - dot)
        else:
            dot = float(np.dot(a, g))
            gz = a * (g - dot)
        
        gz -= gz.mean()
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0); v.fill(0); t_adam = 0
            continue
        
        t_adam += 1
        m = b1*m + (1-b1)*gz
        v = b2*v + (1-b2)*(gz*gz)
        mh = m / (1 - b1**t_adam)
        vh = v / (1 - b2**t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()
        
        a = np.exp(z - z.max())
        a = a / a.sum()
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        c = _full_convolve(a)
        maxc = float(c.max())
        score = 2.0 * n * maxc
        
        if score < best_score - 1e-12:
            best_score = score
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 150 == 0:
                lr = max(0.002, lr * 0.9)
        
        # Adaptive beta
        if t_adam % 50 == 0:
            beta = max(1e-4, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.1:
                    beta = max(1e-4, beta * 0.5)
        
        # Restart
        if no_imp > no_imp_limit:
            ns = 0.06 + 0.04 * (no_imp // no_imp_limit)
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a)
            beta = 0.1 * c.max()
            m.fill(0); v.fill(0); t_adam = 0
            lr = 0.05
            no_imp = 0
    
    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score

def _coord_descent(a, deadline, max_sweeps=3, top_k=None):
    """Coordinate descent with 1D optimization per coordinate."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    if top_k is None:
        top_k = n
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a)
        # Focus on coordinates with largest gradient contribution
        grad_mag = np.zeros(n)
        for k in range(2*n-1):
            if c[k] < best_sc / (2*n) - c.max() + c.max():  # all lags
                j_lo = max(0, k-n+1)
                j_hi = min(n-1, k)
                for j in range(j_lo, j_hi+1):
                    grad_mag[j] += 2.0 * c[k] * a[k-j]
        # Actually, let's just use uniform coordinates for simplicity
        # and limit to top_k
        order = np.argsort(a)[::-1]  # Process largest first
        order = order[:top_k]
        
        for j in order:
            if time.monotonic() > deadline:
                break
            old = a[j]
            # Binary search for optimal delta
            lo, hi = -old + 1e-15, old  # delta in [-old, old] to keep a[j] >= 0
            # Actually, let's try a few multipliers
            best_local = best_sc
            best_val = old
            for factor in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 1.7, 2.0, 3.0]:
                new_val = old * factor
                if new_val < 0:
                    continue
                a[j] = new_val
                a_n = a / a.sum()  # Renormalize
                sc = _score(a_n)
                if sc < best_local:
                    best_local = sc
                    best_val = new_val
            a[j] = best_val
            if best_local < best_sc:
                best_sc = best_local
                best_a = a_n.copy()
                a = best_a.copy()
    
    return best_a, best_sc

def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new

def _make_profile(n, ptype, p1=1.0, p2=0.0, p3=0.0):
    t = np.linspace(0, 1, n)
    if ptype == 'linear':
        a = 1.0 + p1*(t - 0.5)
    elif ptype == 'quadratic':
        a = 1.0 + p1*(2*t-1)**2
    elif ptype == 'cosine':
        a = 1.0 + p1*np.cos(2*np.pi*p2*t)
    elif ptype == 'sine':
        a = 1.0 + p1*np.sin(2*np.pi*p2*t)
    elif ptype == 'U_shape':
        a = (2*t-1)**2 + p1
    elif ptype == 'parabola':
        a = (t-0.5)**2 + p1
    elif ptype == 'gauss':
        a = np.exp(-0.5*((t-0.5)/p1)**2)
    elif ptype == 'edge_heavy':
        a = 1.0 / (0.01 + np.abs(t-0.5)*2.0)**p1
    elif ptype == 'two_gauss':
        a = np.exp(-0.5*((t-p1)/p3)**2) + np.exp(-0.5*((t-p2)/p3)**2)
    elif ptype == 'cubic':

        a = 1.0 + p1*u + p2*u**3
    elif ptype == 'periodic':
        a = 1.0 + p1*np.cos(2*np.pi*t/p2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()
```

```python
def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new


def _make_profile(n, ptype, p1=1.0, p2=0.0, p3=0.0):
    t = np.linspace(0, 1, n)
    if ptype == 'linear':
        a = 1.0 + p1 * (t - 0.5)
    elif ptype == 'quadratic':
        a = 1.0 + p1 * (2 * t - 1) ** 2
    elif ptype == 'cosine':
        a = 1.0 + p1 * np.cos(2 * np.pi * p2 * t)
    elif ptype == 'sine':
        a = 1.0 + p1 * np.sin(2 * np.pi * p2 * t)
    elif ptype == 'U_shape':
        a = (2 * t - 1) ** 2 + p1
    elif ptype == 'parabola':
        a = (t - 0.5) ** 2 + p1
    elif ptype == 'gauss':
        a = np.exp(-0.5 * ((t - 0.5) / p1) ** 2)
    elif ptype == 'edge_heavy':
        a = 1.0 / (0.01 + np.abs(t - 0.5) * 2.0) ** p1

    elif ptype == 'w_shape':
        a = np.abs(np.sin(np.pi * p2 * t)) ** p1
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

                best_seq = p_int.copy()
            print(f"  Escalate to n={tn}: {sc_int:.8f}")

            if time.monotonic() < phase4_end - 30:
                per = min(100.0, phase4_end - time.monotonic() - 20)
                if per > 20:
                    a_e, sc_e = _adam_smooth_max(p_int, time.monotonic() + per, rng,
                                                 lr=0.04, beta_decay=0.95,
                                                 no_imp_limit=200, nesterov=True)
                    if sc_e < best_score:
                        best_score = sc_e
                        best_seq = a_e.copy()
                        print(f"  Adam_esc: {sc_e:.10f} n={tn} <- BEST")

                    # LP if feasible
                    if tn <= 768 and time.monotonic() < phase4_end - 20:
                        per2 = min(40.0, phase4_end - time.monotonic() - 10)
                        if per2 > 10:
                            a_le, sc_le = _lp_fixed_point(a_e, time.monotonic() + per2, max_iters=20)
                            if sc_le < best_score:
                                best_score = sc_le
                                best_seq = a_le.copy()
                                print(f"  LP_esc: {sc_le:.10f} n={tn} <- BEST")

    # Phase 5: Intensive LP + Adam alternation on best
    print("Phase 5: Intensive optimization")
    phase5_end = t_start + 1450.0
    if best_seq is not None and time.monotonic() < phase5_end:
        n_best = best_seq.size
        # If n is too large for LP, interpolate down for LP, then back up
        n_lp = min(n_best, 768)
        if n_lp < n_best:
            p_lp = _interp(best_seq, n_lp)
        else:
            p_lp = best_seq.copy()

        # Alternate LP and Adam
        for round_i in range(4):
            if time.monotonic() > phase5_end - 30:
                break
            # LP step
            if time.monotonic() < phase5_end - 30:
                per = min(80.0, phase5_end - time.monotonic() - 30)
                if per > 10:
                    a_r, sc_r = _lp_fixed_point(p_lp, time.monotonic() + per, max_iters=40)
                    if sc_r < best_score * (n_lp / n_best):
                        print(f"  Round {round_i} LP: {sc_r:.10f} n={n_lp}")
                    if n_lp == n_best:
                        if sc_r < best_score:
                            best_score = sc_r
                            best_seq = a_r.copy()
                    else:
                        p_up = _interp(a_r, n_best)
                        sc_up = _score(p_up)
                        if sc_up < best_score:
                            best_score = sc_up
                            best_seq = p_up.copy()
                            print(f"  Round {round_i} LP+up: {sc_up:.10f} n={n_best} <- BEST")
                        p_lp = a_r.copy()
                    else:
                        p_lp = a_r.copy()

            # Adam step at full n
            if time.monotonic() < phase5_end - 10:
                per = min(60.0, phase5_end - time.monotonic() - 10)
                if per > 10:
                    a_ad, sc_ad = _adam_smooth_max(best_seq, time.monotonic() + per, rng,
                                                  lr=0.03, beta_decay=0.96,
                                                  no_imp_limit=100, nesterov=True)
                    if sc_ad < best_score:
                        best_score = sc_ad
                        best_seq = a_ad.copy()
                        print(f"  Round {round_i} Adam: {sc_ad:.10f} n={n_best} <- BEST")

    # Phase 6: Random restarts
    print("Phase 6: Random restarts")
    phase6_end = deadline - 60.0
    if best_seq is not None and time.monotonic() < phase6_end:
        n_best = best_seq.size
        n_restarts = 8
        per_r = max(10.0, (phase6_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase6_end:
                break
            per = min(per_r, phase6_end - time.monotonic() - 5)
            if per < 8:
                break
            ns = 0.04 + 0.03 * i
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            a_s = np.exp(z - z.max())
            a_s = a_s / a_s.sum()

            a_r, sc_r = _adam_smooth_max(a_s, time.monotonic() + per, rng,
                                         lr=0.05, beta_decay=0.94,
                                         no_imp_limit=100, nesterov=True)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart {i}: {sc_r:.10f} n={n_best} <- BEST")

    # Phase 7: Final LP polish
    print("Phase 7: Final polish")
    if time.monotonic() < deadline - 10 and best_seq is not None:
        per = deadline - time.monotonic() - 10
        n_fin = min(best_seq.size, 768)
        p_fin = _interp(best_seq, n_fin) if n_fin < best_seq.size else best_seq.copy()
        if per > 10:
            a_fp, sc_fp = _lp_fixed_point(p_fin, time.monotonic() + per, max_iters=30)
            if n_fin == best_seq.size:
                if sc_fp < best_score:
                    best_score = sc_fp
                    best_seq = a_fp.copy()
                    print(f"  Final: {sc_fp:.10f} n={n_fin} <- BEST")
            else:
                p_up = _interp(a_fp, best_seq.size)
                sc_up = _score(p_up)
                if sc_up < best_score:
                    best_score = sc_up
                    best_seq = p_up.copy()
                    print(f"  Final+up: {sc_up:.10f} n={best_seq.size} <- BEST")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile(512, 'linear', 1.5, 0, 0)
        exact_sc = _score(final)
        print(f"\nFINAL: score={exact_sc:.10f} n={final.size}")
        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")
        return [float(x) for x in final]

    fb = _make_profile(512, 'linear', 1.5, 0, 0)
    print("FALLBACK n=512")
    return [float(x) for x in fb]
```
