"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses U-shaped profiles, improved active-set balancing, adaptive Adam,
coordinate descent polish, and multi-length exploration.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _compute_active_dir(a, active_indices):
    """Compute average gradient of active c_k w.r.t. a.
    Returns d where d_j = (1/|A|) * sum_{k in A} (a[k-j] + a[j-k])
    """
    n = len(a)
    d = np.zeros(n)
    for k in active_indices:
        k = int(k)
        lo1 = max(0, k - (n - 1))
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            d[lo1:hi1 + 1] += a[k - hi1:k - lo1 + 1][::-1]
        lo2 = max(0, k)
        hi2 = min(n - 1, k + (n - 1))
        if lo2 <= hi2:
            d[lo2:hi2 + 1] += a[lo2 - k:hi2 - k + 1]
    d /= len(active_indices)
    return d


def _make_profiles(n):
    """Generate diverse initial profiles of length n."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0

    # U-shaped profiles (more weight at edges)
    for w in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        p = 1.0 + w * u ** 2
        profiles.append(p / p.sum())

    # Cosine profiles
    for w in [0.3, 0.5, 0.7, 1.0]:
        p = 1.0 + w * np.cos(np.pi * u)
        p = np.maximum(p, 1e-10)
        profiles.append(p / p.sum())

    # Inverted cosine (U-shaped via cos)
    for w in [0.3, 0.5, 0.7, 1.0]:
        p = 1.0 - w * np.cos(np.pi * u)
        p = np.maximum(p, 1e-10)
        profiles.append(p / p.sum())

    # Power profiles
    for p_exp in [0.5, 1.0, 1.5, 2.0, 3.0, -0.5, -1.0]:
        p = (np.abs(u) + 0.01) ** p_exp
        profiles.append(p / p.sum())

    # Sine-based
    p = np.sin(np.pi * t) ** 2
    profiles.append(p / p.sum())
    p = np.cos(np.pi * (t - 0.5)) ** 2
    profiles.append(p / p.sum())

    # Exponential edge concentration
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        d = np.minimum(t, 1.0 - t)
        p = np.exp(-alpha * d * n / 100.0)
        profiles.append(p / p.sum())

    # Hann window
    p = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    profiles.append(p / p.sum())

    # Flat (uniform)
    profiles.append(np.ones(n) / n)

    # Bimodal
    for gap in [0.1, 0.2, 0.3]:
        centers = np.array([gap, 1.0 - gap])
        p = np.zeros(n)
        for c_val in centers:
            p += np.exp(-((t - c_val) ** 2) / (0.02 * n))
        p = np.maximum(p, 1e-10)
        profiles.append(p / p.sum())

    return profiles


def _adam_optimize(a_init, deadline, lr=0.04, top_k=5, max_iter=50000):
    """Adam optimizer in logit space with adaptive beta and top-k smoothing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-10, 0.01 * max_c)

    cur_k = top_k if top_k > 0 else 5
    no_improve = 0
    patience = 500

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Compute smoothed gradient using top-k
        if cur_k > 0 and cur_k < len(c):
            idx = np.argsort(c)[-cur_k:]
            w = np.zeros(len(c))
            c_top = c[idx]
            c_norm = (c_top - c_top.max()) / beta
            e = np.exp(c_norm)
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w[idx[0]] = 1.0
            else:
                w[idx] = e / s
        else:
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                idx_max = int(np.argmax(c))
                w = np.zeros(len(c))
                w[idx_max] = 1.0
            else:
                w = e / s

        W = float(np.dot(w, c))

        # Compute gradient: grad_j = 2n * a_j * (h_j - W) where h_j = sum_k w_k * a|k-j|
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]

        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive adjustments
        if t % 200 == 0:
            beta = max(1e-15, beta * 0.85)
            cur_k = min(cur_k + 1, 50)

        if no_improve > patience:
            lr = max(0.002, lr * 0.7)
            no_improve = 0
            if lr < 0.003:
                break

    return best_a, best_score


def _active_set_optimize(a_init, deadline, max_iter=20000):
    """Active-set balancing with adaptive step size and tightening threshold."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    delta_frac = 0.05
    base_step = 0.02
    stall_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        threshold = cm * (1.0 - delta_frac)
        active = np.where(c >= threshold)[0]

        if len(active) == 0:
            break
        if len(active) > 200:
            # Use only the very top
            top_idx = np.argsort(c)[-100:]
            active = top_idx

        d = _compute_active_dir(a, active)
        d_centered = d - d.mean()
        d_norm = float(np.sqrt(np.sum(d_centered ** 2)))
        if d_norm < 1e-15:
            break

        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()

        # Adaptive step
        step = base_step * (cm / (d_norm * (float(a.sum()) + 1e-15)))
        step = min(step, 0.05)

        # Try multiple step sizes (line search)
        found = False
        for attempt in range(4):
            z_new = z - step * d_centered / d_norm
            z_new = np.clip(z_new, -50, 50)
            z_new = z_new - z_new.mean()
            a_new = _softmax_fn(z_new)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                z = z_new.copy()
                found = True
                stall_count = 0
                break
            step *= 0.4

        if not found:
            stall_count += 1
            if stall_count > 50:
                # Widen active set
                delta_frac = min(0.3, delta_frac * 1.5)
                stall_count = 0
                base_step = max(0.003, base_step * 0.5)
                if base_step < 0.003:
                    break

        # Gradually tighten
        if it % 500 == 0 and it > 0:
            delta_frac = max(0.002, delta_frac * 0.8)

    return best_a, best_score


def _coordinate_descent(a_init, deadline, max_rounds=100):
    """Coordinate descent: adjust one element at a time for final polish."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    rng = np.random.default_rng(12345)

    for rd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Random order of coordinates
        order = rng.permutation(n)
        improved = 0

        for j in order:
            if time.monotonic() > deadline:
                break
            j = int(j)

            # Try scaling a[j] by various factors
            base_val = a[j]
            if base_val < 1e-15:
                continue

            best_local_val = base_val
            best_local_score = best_score

            for factor in [0.9, 0.8, 0.7, 1.1, 1.2, 1.3]:
                a[j] = base_val * factor
                sc = _score_norm_fn(a)
                if np.isfinite(sc) and sc < best_local_score:
                    best_local_score = sc
                    best_local_val = base_val * factor
            a[j] = best_local_val

            if best_local_score < best_score:
                best_score = best_local_score
                best_a = a.copy()
                improved += 1

        if improved == 0:
            break

        # Shrink perturbation as we converge
        # (handled by the fixed factors above; could add finer factors)

    return best_a, best_score


def _lbfgs_optimize(a_init, deadline, top_k=5):
    """L-BFGS with logsumexp smoothing in logit space."""
    n = len(a_init)

    def _z_obj(z, beta, tk):
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        if tk > 0 and tk < len(c):
            idx = np.argsort(c)[-tk:]
            c_top = c[idx]
            c_norm = c_top - c_top.max()
            log_sum = float(np.log(np.sum(np.exp(c_norm / beta))))
            return 2.0 * n * (c_top.max() + beta * log_sum)
        else:
            c_shifted = (c - cm) / beta
            log_sum = float(np.log(np.sum(np.exp(c_shifted))))
            return 2.0 * n * (cm + beta * log_sum)

    def _z_grad(z, beta, tk):
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        if tk > 0 and tk < len(c):
            idx = np.argsort(c)[-tk:]
            w = np.zeros(len(c))
            c_top = c[idx]
            c_norm = (c_top - c_top.max()) / beta
            e = np.exp(c_norm)
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w[idx[0]] = 1.0
            else:
                w[idx] = e / s
        else:
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                idx_max = int(np.argmax(c))
                w = np.zeros(len(c))
                w[idx_max] = 1.0
            else:
                w = e / s
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            return np.zeros(n)
        return grad

    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]
    betas = [b * max(max_c0, 1e-10) for b in beta_starts]

    remaining = max(1.0, deadline - time.monotonic())
    per_level = remaining / len(betas)

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level)
        cur_tk = min(3 + i, top_k) if top_k > 0 else 3

        try:
            result = minimize(
                _z_obj, z, args=(beta, cur_tk),
                method='L-BFGS-B', jac=_z_grad,
                options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 30, 'maxfun': 1000,
                         'iprint': -1}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-10)
    return p / p.sum()

            dl = min(phase6_end, time.monotonic() + remaining)
            print(f"  Coord descent, budget={remaining:.0f}s")
            p_best, sc_best = _coordinate_descent(best_seq, dl, max_rounds=50)
            _update(p_best, sc_best, "coord_descent")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Try different length ---
    print("=== Phase 7: Different length ===")
    phase7_end = min(deadline, t_start + 1750.0)
    if best_seq is not None and time.monotonic() < phase7_end:
        n_curr = len(best_seq)
        alt_n = 2048 if n_curr == 4096 else 4096
        if n_curr != alt_n:
            p_alt = _interp_profile(best_seq, alt_n)
            sc_alt = _score_norm_fn(p_alt)
            _update(p_alt, sc_alt, f"interp_n{alt_n}")

            if time.monotonic() < phase7_end - 30:
                dl = min(phase7_end - 5, time.monotonic() + 60.0)
                p_b, s_b = _adam_optimize(p_alt, dl, lr=0.03, top_k=5, max_iter=20000)
                _update(p_b, s_b, f"adam_n{alt_n}")

    print(f"  Phase 7 done: best={best_score:.10f}")

    # --- Phase 8: Final polish ---
    print("=== Phase 8: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            dl = min(deadline - 5, time.monotonic() + min(50.0, remaining * 0.6))
            p_best, sc_best = _active_set_optimize(best_seq, dl, max_iter=3000)
            _update(p_best, sc_best, "final_as")

        if time.monotonic() < deadline - 10:
            dl2 = min(deadline - 3, time.monotonic() + min(30.0, deadline - time.monotonic() - 5))
            p_best2, sc_best2 = _adam_optimize(best_seq, dl2, lr=0.005, top_k=20, max_iter=10000)
            _update(p_best2, sc_best2, "final_adam")

    # --- Final output ---
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = np.ones(4096) / 4096.0
            final_score = _score_fn(final_seq)

        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")

        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 10000:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  evaluate check: {e}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profiles(4096)[0]
    print(f"  FALLBACK: n=4096")
    return [float(x) for x in fallback]
