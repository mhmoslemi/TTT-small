    z_new = z - lr * mh / (np.sqrt(vh) + eps)
    return z_new, m, v


def _adam_smooth_max(z, deadline, rng, lr=0.06, no_imp_limit=250,
                     name=""):
    """Core smooth-max gradient descent in log-space with adaptive beta."""
    n = len(z)
    a = _from_z(z)

    c = _full_convolve(a)
    maxc = float(c.max())
    beta = max(1e-8, 0.02 / n)

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    base_lr = lr

    best_score = _score_normalized(a)
    best_z = z.copy()
    no_imp = 0
    total_iters = 0

    while time.monotonic() < deadline:
        # Compute smooth max weights
        w = _smooth_max_weights(c, beta)

        # Gradient in a-space: 2 * cross_correlate(w, a)
        g_a = 2.0 * _cross_correlate(w, a)

        # Gradient in z-space: g_a * a (chain rule), centered
        gz = g_a * a - np.mean(g_a * a)

        # Safety check
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            t_adam = 0
            continue

        # Adam update
        t_adam += 1
        total_iters += 1
        z, m_adam, v_adam = _adam_step(z, gz, m_adam, v_adam, t_adam, lr)
        z = np.clip(z, -60, 60)
        z -= z.mean()

        # Convert to a
        a = _from_z(z)

        # Score
        c = _full_convolve(a)
        maxc = float(c.max())
        score = 2.0 * n * maxc

        if np.isfinite(score) and score < best_score - 1e-10:
            best_score = score
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Adaptive beta: cosine-like decay
        if total_iters % 40 == 0:
            progress = min(1.0, total_iters / (no_imp_limit * 3.0))
            beta_floor = 1e-7 / n
            beta = beta_floor + (0.02 / n - beta_floor) * (1.0 - progress) * (1.0 + np.cos(np.pi * progress)) / 2.0
            # Also adapt based on gap
            srt = np.sort(c)[::-1]
            if len(srt) >= 2:
                gap = srt[0] - srt[1]
                if gap < beta * 0.3:
                    beta = max(beta_floor, beta * 0.6)

        # Learning rate scheduling
        if no_imp > 0 and no_imp % 150 == 0:
            lr = max(0.002, lr * 0.93)

        # Reset if stuck
        if no_imp > no_imp_limit:
            noise_scale = 0.12 if best_score < 1.55 else 0.2
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -60, 60)
            z -= z.mean()
            a = _from_z(z)
            c = _full_convolve(a)
            beta = 0.03 / n
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    a_best = _from_z(best_z)
    return a_best, best_score


def _spectral_shaping(a, deadline, rng, n_steps=2000, name=""):
    """Spectral shaping: flatten FFT magnitude, project to non-negative."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    a = a.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_score = _score_normalized(a)
    best_a = a.copy()
    alpha = 0.7  # Start moderately aggressive
    alpha_min = 0.3

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        # FFT
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Flatten: mag_new = mag^alpha (alpha < 1 flattens)
        mag_new = mag ** alpha
        A_new = mag_new * np.exp(1j * phase)

        # IFFT
        a_new = np.fft.irfft(A_new, N)[:n]

        # Soft projection to non-negative
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        # Mix with current to avoid oscillation
        a_curr = 0.5 * a + 0.5 * a_new
        a_curr = np.maximum(a_curr, 1e-15)
        a_curr = a_curr / a_curr.sum()

        score = _score_normalized(a_curr)
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_curr.copy()
            a = a_curr
            # Keep alpha or increase slightly
            alpha = min(0.9, alpha * 1.02)
        else:
            # Reject, make less aggressive
            alpha = max(alpha_min, alpha * 0.98)
            a = a_curr  # Still take the step to explore

        # Periodically re-randomize phase for exploration
        if step % 500 == 499:
            A = np.fft.rfft(a, N)
            mag = np.abs(A)
            # Add small random phase perturbation
            dp = 0.1 * rng.normal(size=mag.shape)
            A = mag * np.exp(1j * (np.angle(A) + dp))
            a = np.fft.irfft(A, N)[:n]
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a, best_score


def _interp_to_n(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new


def _make_init(n, init_type, rng):
    """Generate initial sequence of length n."""
    if init_type == 0:  # uniform
        a = np.ones(n)
    elif init_type == 1:  # linear ramp (anti-symmetric tendency)
        a = np.linspace(0.1, 1.0, n)
    elif init_type == 2:  # strong ramp
        a = np.linspace(0.01, 1.0, n)
    elif init_type == 3:  # quadratic
        t = np.linspace(0, 1, n)
        a = 0.1 + (t - 0.3) ** 2
    elif init_type == 4:  # cosine (one period)
        t = np.arange(n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * t / n)
    elif init_type == 5:  # random
        a = np.abs(rng.normal(size=n)) + 0.1
    elif init_type == 6:  # U-shape
        t = np.linspace(0, 1, n)
        a = (t - 0.5) ** 2 * 4 + 0.1
    elif init_type == 7:  # exponential decay
        a = np.exp(-1.5 * np.arange(n) / n)
    elif init_type == 8:  # two peaks
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.2) ** 2) + np.exp(-20 * (t - 0.8) ** 2) + 0.1
    elif init_type == 9:  # sawtooth
        t = np.arange(n) / n
        a = np.abs(2 * t - 1) + 0.1
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: diverse initialization -> smooth-max gradient
    descent with adaptive annealing -> spectral shaping -> multi-scale
    escalation -> final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    print(f"Parent n={parent.size if parent is not None else 0}, budget={budget_s}s")

    # === Phase 1: Diverse initialization (30s) ===
    print("Phase 1: Diverse initialization...")
    p1_end = t_start + 30.0

    n_values = [512, 1024, 2048]
    init_types = list(range(10))
    candidates = []

    for n in n_values:
        for it in init_types:
            if time.monotonic() > p1_end:
                break
            a = _make_init(n, it, rng)
            sc = _score_normalized(a)
            candidates.append((sc, a, n, f"init{it}_n{n}"))

    # Add parent-based candidates
    if parent is not None:
        for n in [512, 1024, 2048, 4096]:
            if time.monotonic() > p1_end:
                break
            pi = _interp_to_n(parent, n)
            sc = _score_normalized(pi)
            candidates.append((sc, pi, n, f"parent_n{n}"))
        if parent.size in n_values:
            sc = _score_normalized(parent)
            candidates.append((sc, parent.copy(), parent.size, "parent_exact"))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, best: {candidates[0][0]:.8f} n={candidates[0][2]} [{candidates[0][3]}]")

    for sc, a, n, lbl in candidates[:5]:
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
    if best_seq is not None:
        print(f"  BEST after init: {best_score:.10f} n={best_seq.size}")

    # === Phase 2: Adam smooth max on best candidates (700s) ===
    print("Phase 2: Adam smooth-max optimization...")
    p2_end = t_start + 730.0

    # Pick top candidates at n=1024 primarily
    cands_1024 = [(sc, a, n, lbl) for sc, a, n, lbl in candidates if n == 1024]
    if not cands_1024:
        cands_1024 = [(sc, _interp_to_n(a, 1024), 1024, lbl + "_i1024")
                      for sc, a, n, lbl in candidates[:5]]
    cands_1024.sort(key=lambda x: x[0])

    # Run Adam on top 3 candidates at n=1024
    for idx, (sc, a, n, lbl) in enumerate(cands_1024[:3]):
        if time.monotonic() > p2_end - 30:
            break
        remaining = p2_end - time.monotonic() - 10
        if remaining < 30:
            break
        alloc = min(remaining, 250.0)

        z = np.log(np.maximum(a, 1e-12))
        z -= z.mean()

        a_out, sc_out = _adam_smooth_max(
            z, time.monotonic() + alloc, rng,
            lr=0.06, no_imp_limit=250, name=f"adam1024_{lbl}"
        )
        if sc_out < best_score:
            best_score = sc_out
            best_seq = a_out.copy()
            print(f"  BEST: {best_score:.10f} n=1024 [{lbl} adam]")

    # Also try n=512 for a bit
    cands_512 = [(sc, a, n, lbl) for sc, a, n, lbl in candidates if n == 512]
    if cands_512 and time.monotonic() < p2_end - 40:
        cands_512.sort(key=lambda x: x[0])
        sc, a, n, lbl = cands_512[0]
        if time.monotonic() < p2_end - 40:
            z = np.log(np.maximum(a, 1e-12))
            z -= z.mean()
            a_out, sc_out = _adam_smooth_max(
                z, time.monotonic() + min(80.0, p2_end - time.monotonic() - 20),
                rng, lr=0.06, no_imp_limit=200, name="adam512"
            )
            if sc_out < best_score:
                best_score = sc_out
                best_seq = a_out.copy()
                print(f"  BEST: {best_score:.10f} n=512 [{lbl} adam]")

    # Also try n=2048
    cands_2048 = [(sc, a, n, lbl) for sc, a, n, lbl in candidates if n == 2048]
    if cands_2048 and time.monotonic() < p2_end - 40:
        cands_2048.sort(key=lambda x: x[0])
        sc, a, n, lbl = cands_2048[0]
        if time.monotonic() < p2_end - 40:
            z = np.log(np.maximum(a, 1e-12))
            z -= z.mean()
            a_out, sc_out = _adam_smooth_max(
                z, time.monotonic() + min(100.0, p2_end - time.monotonic() - 20),
                rng, lr=0.05, no_imp_limit=300, name="adam2048"
            )
            if sc_out < best_score:
                best_score = sc_out
                best_seq = a_out.copy()
                print(f"  BEST: {best_score:.10f} n=2048 [{lbl} adam]")

    print(f"  After Phase 2: best={best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    # === Phase 3: Spectral shaping refinement (300s) ===
    print("Phase 3: Spectral shaping...")
    p3_end = t_start + 1030.0

    if best_seq is not None and time.monotonic() < p3_end:
        a_ss, sc_ss = _spectral_shaping(
            best_seq, time.monotonic() + min(200.0, p3_end - time.monotonic() - 10),
            rng, n_steps=3000, name="spectral"
        )
        if sc_ss < best_score:
            best_score = sc_ss
            best_seq = a_ss.copy()
            print(f"  BEST: {best_score:.10f} n={best_seq.size} [spectral]")

        # Follow up with Adam on spectrally shaped result
        if time.monotonic() < p3_end - 20:
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            a_out, sc_out = _adam_smooth_max(
                z, time.monotonic() + min(80.0, p3_end - time.monotonic() - 5),
                rng, lr=0.04, no_imp_limit=200, name="adam_post_spectral"
            )
            if sc_out < best_score:
                best_score = sc_out
                best_seq = a_out.copy()
                print(f"  BEST: {best_score:.10f} n={best_seq.size} [adam_post_spec]")

    # === Phase 4: Multi-scale escalation (300s) ===
    print("Phase 4: Multi-scale escalation...")
    p4_end = t_start + 1330.0

    if best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 1024:
            target_ns = [2048, 4096]
        elif n_cur <= 2048:
            target_ns = [4096]
        else:
            target_ns = []

        for tn in target_ns:
            if time.monotonic() > p4_end - 60:
                break
            p_int = _interp_to_n(best_seq, tn)
            sc_int = _score_normalized(p_int)
            print(f"  Interp to n={tn}: {sc_int:.8f}")
            if sc_int < best_score:
                best_score = sc_int
                best_seq = p_int.copy()
                print(f"  BEST: {best_score:.10f} n={tn} [interp]")

            # Adam at new scale
            if time.monotonic() < p4_end - 80:
                z = np.log(np.maximum(p_int, 1e-12))
                z -= z.mean()
                alloc = min(150.0, p4_end - time.monotonic() - 40)
                if alloc > 30:
                    a_out, sc_out = _adam_smooth_max(
                        z, time.monotonic() + alloc, rng,
                        lr=0.05, no_imp_limit=300, name=f"adam_esc_n{tn}"
                    )
                    if sc_out < best_score:
                        best_score = sc_out
                        best_seq = a_out.copy()
                        print(f"  BEST: {best_score:.10f} n={tn} [adam_esc]")

    # === Phase 5: Random perturbations + re-optimization (150s) ===
    print("Phase 5: Perturbation restarts...")
    p5_end = t_start + 1480.0

    if best_seq is not None and time.monotonic() < p5_end:
        n_best = best_seq.size
        n_restarts = 4
        per_r = max(20.0, (p5_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > p5_end - 10:
                break
            per = min(per_r, p5_end - time.monotonic() - 5)
            if per < 15:
                break

            noise_scales = [0.08, 0.15, 0.25, 0.4]
            ns = noise_scales[i % len(noise_scales)]
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -60, 60)
            z -= z.mean()
            a_s = _from_z(z)

            a_out, sc_out = _adam_smooth_max(
                z, time.monotonic() + per, rng,
                lr=0.06, no_imp_limit=200, name=f"restart{i}"
            )
            if sc_out < best_score:
                best_score = sc_out
                best_seq = a_out.copy()
                print(f"  BEST: {best_score:.10f} n={n_best} [restart{i} ns={ns}]")

    # === Phase 6: Final combined polish (remaining time) ===
    print("Phase 6: Final polish...")
    if time.monotonic() < deadline - 5 and best_seq is not None:
        remaining = deadline - time.monotonic() - 5
        n_best = best_seq.size

        if remaining > 30:
            # Try spectral shaping one more time
            if remaining > 60:
                a_ss, sc_ss = _spectral_shaping(
                    best_seq, time.monotonic() + min(50.0, remaining - 20),
                    rng, n_steps=2000, name="final_spectral"
                )
                if sc_ss < best_score:
                    best_score = sc_ss
                    best_seq = a_ss.copy()

            # Final Adam
            if remaining > 40 and time.monotonic() < deadline - 10:
                z = np.log(np.maximum(best_seq, 1e-12))
                z -= z.mean()
                alloc = min(60.0, deadline - time.monotonic() - 5)
                if alloc > 20:
                    a_out, sc_out = _adam_smooth_max(
                        z, time.monotonic() + alloc, rng,
                        lr=0.03, no_imp_limit=150, name="final_adam"
                    )
                    if sc_out < best_score:
                        best_score = sc_out
                        best_seq = a_out.copy()
                        print(f"  BEST: {best_score:.10f} n={n_best} [final_adam]")

    # === Final validation and return ===
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_init(512, 0, rng)

        exact_sc = _score_general(final)
        print(f"\nFINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_init(512, 1, rng)
    print(f"FALLBACK n=512, score={_score_general(fb):.6f}")
    return [float(x) for x in fb]
