"""
Optimizes step function heights to minimize 2*n*max(conv(a,a))/sum(a)^2
using smooth-max L-BFGS with adaptive temperature, FFT gradients,
and multi-phase search strategy.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _exact_score(a):
    """2*n*max(conv(a,a)) assuming sum(a)=1"""
    n = a.size
    if n == 0:
        return np.inf
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smooth_max_obj_grad(z, beta, n, N):
    """
    Compute smooth max of conv(softmax(z), softmax(z)) and its gradient w.r.t. z.
    z: log-parameters
    beta: temperature
    n: sequence length
    N: FFT size (pre-computed)
    Returns (objective_value, gradient)
    """
    a = _softmax(z)
    # Compute convolution
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    
    # Smooth max
    m = c.max()
    x = (c - m) / beta
    x = np.clip(x, -50, 50)  # numerical stability
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = ex / s
    val = m + beta * np.log(s)
    
    # Gradient w.r.t. a: 2 * cross_correlation(w, a)
    fw = np.fft.rfft(w, N)
    fa_full = np.fft.rfft(a, N)
    cross = np.fft.irfft(fw * np.conj(fa_full), N)[:n]
    grad_a = 2.0 * cross
    
    # Chain rule: grad_z = a * (grad_a - dot(a, grad_a))
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)
    
    return val, grad_z


def _optimize_from_z(z0, beta_start, deadline, beta_end=None, max_runs=30):

"""
Smooth-max L-BFGS optimization with adaptive temperature schedule,
FFT-based analytical gradients, softmax parameterization for constraint
handling, subgradient descent on the true max, and multi-phase search
to minimize 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _exact_score(a):
    n = a.size
    if n == 0:
        return np.inf
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smooth_max_obj_grad(z, beta, n, N):
    a = _softmax(z)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    m = c.max()
    x = (c - m) / beta
    x = np.clip(x, -60, 60)
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = ex / s
    val = m + beta * np.log(s)
    fw = np.fft.rfft(w, N)
    fa2 = np.fft.rfft(a, N)
    cross = np.fft.irfft(fw * np.conj(fa2), N)[:n]
    grad_a = 2.0 * cross
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)
    return val, grad_z


def _lbfgs_smooth_max(z0, beta_start, deadline, max_cycles=40, beta_decay=0.65):
    n = len(z0)
    N = _next_pow2(4 * n)
    best_z = z0.copy()
    best_a = _softmax(z0)
    best_score = _exact_score(best_a)
    beta = beta_start
    c0 = _full_convolve(best_a)
    beta_end = max(1e-5 * c0.max(), 1e-10)

    for cycle in range(max_cycles):
        if time.monotonic() > deadline:
            break
        if beta < beta_end:
            break

        def obj(z):
            v, _ = _smooth_max_obj_grad(z, beta, n, N)
            return v

        def grad(z):
            _, g = _smooth_max_obj_grad(z, beta, n, N)
            return g

        try:
            res = minimize(obj, z0.copy(), jac=grad, method='L-BFGS-B',
                           options={'maxiter': 50, 'ftol': 1e-14, 'gtol': 1e-9,
                                    'maxls': 50})
            if res.x is not None and np.all(np.isfinite(res.x)):
                z0 = res.x
        except Exception:
            pass

        a = _softmax(z0)
        score = _exact_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            best_z = z0.copy()

        beta *= beta_decay

    return best_z, best_a, best_score


def _subgradient_descent(a_init, deadline, max_iters=5000, lr_start=0.01, lr_decay=0.999):

    a = a / a.sum()
    best_a = a.copy()

    lr = lr_start

        max_c = c.max()
        peak_indices = np.where(c >= max_c * (1.0 - 1e-6))[0]
        w = np.zeros_like(c)
        w[peak_indices] = 1.0 / len(peak_indices)
        fw = np.fft.rfft(w, N)
        fa = np.fft.rfft(a, N)
        cross = np.fft.irfft(fw * np.conj(fa), N)[:n]
        grad = 2.0 * cross
        dot = np.dot(a, grad)
        grad_centered = grad - dot
        a = a - lr * grad_centered
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

def _make_profiles(n, rng):

"""
Smooth-max L-BFGS optimization with adaptive temperature, FFT gradients,
softmax constraints, subgradient descent, and multi-phase search to
minimize 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _exact_score(a):
    n = a.size
    if n == 0:
        return np.inf
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smooth_max_obj_grad(z, beta, n, N):
    a = _softmax(z)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    m = c.max()
    x = (c - m) / beta
    x = np.clip(x, -60, 60)
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = ex / s
    val = m + beta * np.log(s)
    fw = np.fft.rfft(w, N)
    fa2 = np.fft.rfft(a, N)
    cross = np.fft.irfft(fw * np.conj(fa2), N)[:n]
    grad_a = 2.0 * cross
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)
    return val, grad_z


def _lbfgs_smooth_max(z0, beta_start, deadline, max_cycles=40, beta_decay=0.65):
    n = len(z0)
    N = _next_pow2(4 * n)
    best_z = z0.copy()
    a_cur = _softmax(z0)
    best_a = a_cur.copy()
    best_score = _exact_score(best_a)
    beta = beta_start
    c0 = _full_convolve(a_cur)
    beta_end = max(0.002 * c0.max(), 1e-10)

    for cycle in range(max_cycles):
        if time.monotonic() > deadline:
            break
        if beta < beta_end:
            break

        def obj(z):
            v, _ = _smooth_max_obj_grad(z, beta, n, N)
            return v

        def grad(z):
            _, g = _smooth_max_obj_grad(z, beta, n, N)
            return g

        try:
            res = minimize(obj, z0.copy(), jac=grad, method='L-BFGS-B',
                           options={'maxiter': 40, 'ftol': 1e-14, 'gtol': 1e-9,
                                    'maxls': 40})
            if res.x is not None and np.all(np.isfinite(res.x)):
                z0 = res.x
        except Exception:
            pass

        a_cur = _softmax(z0)
        score = _exact_score(a_cur)
        if score < best_score:
            best_score = score
            best_a = a_cur.copy()
            best_z = z0.copy()

        beta *= beta_decay

    return best_z, best_a, best_score


def _subgradient_descent(a_init, deadline, max_iters=8000, lr_start=0.005, lr_decay=0.9995):
    n = a_init.size
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _exact_score(a)
    lr = lr_start
    N = _next_pow2(2 * n - 1)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a)
        max_c = c.max()
        peak_mask = c >= max_c * (1.0 - 1e-5)
        w = np.zeros_like(c)
        w[peak_mask] = 1.0 / peak_mask.sum()
        fw = np.fft.rfft(w, N)
        fa = np.fft.rfft(a, N)
        cross = np.fft.irfft(fw * np.conj(fa), N)[:n]
        grad = 2.0 * cross
        dot = np.dot(a, grad)
        a = a - lr * (grad - dot)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        if it % 50 == 0:
            score = _exact_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
        lr *= lr_decay
        if lr < 1e-9:
            break

    score = _exact_score(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()
    return best_a, best_score
