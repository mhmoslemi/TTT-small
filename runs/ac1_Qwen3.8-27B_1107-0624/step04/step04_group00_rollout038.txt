"""
LP-based iterative optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses the linear programming approach to find optimal descent directions,
combined with gradient-based polishing for fine-tuning.
"""
import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import diags
from scipy.sparse.diags import diags as sparse_diags
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _build_conv_matrix(a):
    """Build sparse convolution matrix where (A@g)[k] = (a*g)[k]."""
    n = len(a)
    # Diagonal offset -d has value a[d] repeated n times
    diagonals = [a[d] * np.ones(n) for d in range(n)]
    offsets = list(range(0, -n, -1))
    A = diags(diagonals, offsets, shape=(2*n-1, n), format='csr')
    return A

def _solve_lp(a, M):
    """Solve the LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = res.x
            if g is not None and np.all(g >= -1e-15) and np.all(np.isfinite(g)):
                g = np.maximum(g, 0)
                return g
    except Exception as e:
        print(f"  LP error: {e}")
    return None

def _lp_iteration(a, M_tolerance=0.01, max_lp_iter=3):
    """One iteration of the LP-based descent."""
    n = len(a)
    c = _full_convolve(a, a)
    M = float(c.max())
    
    if M < 1e-15:
        return None, M
    
    g = _solve_lp(a, M)
    if g is None or g.sum() < 1e-15:
        return None, M
    
    # Normalize g to have same sum as a
    a_sum = a.sum()
    g = g / g.sum() * a_sum
    
    return g, M

def _line_search(a, direction, base_score, n_t=10):
    """Line search: find best t in (0, 1] such that score((1-t)a + t*direction) is minimized."""
    best_t = 0
    best_score = base_score
    
    for t in np.linspace(1.0, 0.05, n_t):
        a_new = (1 - t) * a + t * direction
        if np.any(a_new < -1e-15):
            continue
        a_new = np.maximum(a_new, 0)
        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_t = t
    
    if best_t > 0:
        return (1 - best_t) * a + best_t * direction, best_t, best_score
    return a, 0, base_score

def _adaptive_beta(c):
    cm = float(c.max())
    if cm < 1e-15:
        return 1e-10
    sorted_c = np.sort(c)[::-1]
    k = min(20, len(sorted_c) - 1)
    gap = sorted_c[0] - sorted_c[k]
    if gap < 0.05 * cm:
        return max(1e-10, 0.1 * cm)
    elif gap > 0.5 * cm:
        return max(1e-10, 0.01 * cm)
    else:
        return max(1e-10, 0.5 * gap)

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _smooth_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return cm
    return cm + beta * np.log(s)

def _smooth_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n-1:2*n-1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

    best_score = _score_norm_fn(a)

    max_c0 = float(c0.max())
    
    if beta_init is None:
        beta = max(1e-10, 0.01 * max_c0)
    
    # Continuation: decrease beta
    betas = [max_c0 * f for f in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]]
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        remaining = len(betas[betas.index(beta):])
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 60.0)
        level_dl = time.monotonic() + time_per
        
        try:
            result = minimize(
                _smooth_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_smooth_grad,
                options={'maxiter': 2000, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 5000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:

        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300:

"""
LP-based iterative optimization minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find globally optimal descent directions for
the max-autocorrelation problem, combined with spectral initialization,
constraint generation for speed, and gradient-based polishing.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import diags
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _build_conv_matrix(a):
    n = len(a)
    diagonals = [a[d] * np.ones(n) for d in range(n)]
    offsets = list(range(0, -n, -1))
    A = diags(diagonals, offsets, shape=(2 * n - 1, n), format='csr')
    return A


def _build_conv_matrix_subset(a, row_indices):
    n = len(a)
    rows = len(row_indices)
    diagonals_data = []
    offsets_list = []
    for d in range(n):
        diag_vals = np.zeros(rows)
        # Row k has A[k,j] = a[k-j] for valid j
        # For subset rows, for each row index r (actual position row_indices[r]):
        #   A[row_indices[r], j] = a[row_indices[r] - j] if 0 <= row_indices[r]-j <= n-1
        # Diagonal offset -d means j = row - d, so A[row, row-d] = a[d]
        # For subset: positions where j = row_indices[r] - d is in [0, n-1]
        for r, k in enumerate(row_indices):
            j = k - d
            if 0 <= j <= n - 1:
                diag_vals[r] = a[d]
        if np.any(diag_vals != 0):
            diagonals_data.append(diag_vals)
            offsets_list.append(-d)
    if not diagonals_data:
        return diags([], [], shape=(rows, n), format='csr')
    A = diags(diagonals_data, offsets_list, shape=(rows, n), format='csr')
    return A


def _solve_full_lp(a, M):
    n = len(a)
    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            if np.all(np.isfinite(g)) and g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def _solve_subset_lp(a, M, row_indices, max_rounds=10):
    n = len(a)
    c_full = _full_convolve(a, np.ones(n) / n)
    active_rows = list(row_indices)
    
    for round_idx in range(max_rounds):
        A_sub = _build_conv_matrix_subset(a, active_rows)
        c_obj = -np.ones(n)
        b_ub = np.full(len(active_rows), M)
        bounds = [(0, None)] * n
        try:
            res = linprog(c_obj, A_ub=A_sub, b_ub=b_ub, bounds=bounds, method='highs')
            if not res.success or res.x is None:
                break
            g = np.maximum(res.x, 0)
            if g.sum() < 1e-15:
                break
        except Exception:
            break
        
        # Check if any constraint is violated
        ag = _full_convolve(a, g)
        violations = np.where(ag > M * (1 + 1e-10))[0]
        new_violations = [int(v) for v in violations if v not in set(active_rows)]
        
        if not new_violations:
            return g
        # Add violated constraints (up to 20 at a time)
        new_violations.sort(key=lambda v: -ag[v])
        active_rows.extend(new_violations[:20])
    
    if 'g' in dir() and g.sum() > 1e-15:
        return g
    return None


def _line_search(a, direction, base_score, t_values=None):
    if t_values is None:
        t_values = [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
    best_t = 0
    best_score = base_score
    best_a = a
    for t in t_values:
        a_new = (1 - t) * a + t * direction
        if np.any(a_new < -1e-12):
            continue
        a_new = np.maximum(a_new, 0)
        sc = _score_norm_fn(a_new)
        if sc < best_score - 1e-15:
            best_score = sc
            best_t = t
            best_a = a_new
    if best_t > 0:
        return best_a, best_t, best_score
    return a, 0, base_score


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _adaptive_beta(c):
    cm = float(c.max())
    if cm < 1e-15:
        return 1e-10
    sorted_c = np.sort(c)[::-1]
    k = min(20, len(sorted_c) - 1)
    gap = sorted_c[0] - sorted_c[k]
    if gap < 0.05 * cm:
        return max(1e-10, 0.1 * cm)
    elif gap > 0.5 * cm:
        return max(1e-10, 0.01 * cm)
    else:
        return max(1e-10, 0.5 * gap)


def _smooth_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return cm
    return cm + beta * np.log(s)


def _smooth_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
