"""
Optimization combining: (1) L-BFGS with analytic gradient on smooth-max objective,
(2) active-set uniform gradient refinement, (3) LP-based structural updates,
(4) structured multi-scale exploration, and (5) final fine-tuning.
Key innovation: L-BFGS exploits curvature info unlike previous Adam approach,
and active-set gradient focuses uniformly on binding constraints.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score_of(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return np.inf
    c = _full_convolve(p)
    s = float(p.sum())
    if s < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return None
    return a / s


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - float(np.max(z))
    z = np.clip(z, -50.0, 50.0)
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0:
        return np.ones_like(e) / float(e.size)
    return p / s


def _smooth_max_val_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _compute_h_fft(f, w):
    """h_j = sum_k w_k * f_{k-j} for j=0..n-1 (cross-correlation at lag -j)"""
    n = len(f)
    len_w = len(w)
    N = _next_pow2(n + len_w - 1)
    F = np.fft.rfft(f, N)
    W = np.fft.rfft(w, N)
    xcorr = np.fft.irfft(F * np.conj(W), N)
    h = np.zeros(n)
    h[0] = xcorr[0]
    if n > 1:
        h[1:] = xcorr[N - 1: N - n: -1]
    return h


def _smooth_max_score_and_grad(z, n, beta):
    """Compute score = 2n * smooth_max(f*f) and its gradient w.r.t. z."""
    f = _softmax_from_z(z)
    c = _full_convolve(f)
    mval, w = _smooth_max_val_weights(c, beta)
    score = 2.0 * n * mval
    h = _compute_h_fft(f, w)
    s_wc = float(np.dot(w, c))
    grad = 4.0 * n * f * (h - s_wc)
    # Remove mean component (gauge invariance: z + const gives same f)
    grad = grad - float(np.mean(grad))
    return score, grad


def _lbfgs_optimize(f0, deadline, beta_init, beta_min, beta_decay, max_iter):
    """L-BFGS optimization in log-space with adaptive beta."""
    n = len(f0)
    f0 = np.maximum(f0, 1e-12)
    f0 = f0 / f0.sum()
    z0 = np.log(f0)
    z0 -= float(np.mean(z0))

    best_z = z0.copy()
    best_score = float(2.0 * n * _full_convolve(f0).max())
    best_f = f0.copy()

    beta = beta_init
    it = 0
    no_improve = 0

    while time.monotonic() < deadline and it < max_iter:
        def fun_and_grad(z):
            sc, g = _smooth_max_score_and_grad(z, n, beta)
            if not np.isfinite(sc):
                sc = 1e10
                g = np.zeros_like(z)
            return sc, g

        res = minimize(fun_and_grad, z.copy(), method='L-BFGS-B',
                       jac=True, options={'maxiter': 50, 'maxls': 50,
                                          'ftol': 1e-10, 'gtol': 1e-8,
                                          'maxfun': 200})

        if res.success or np.isfinite(res.fun):
            z_new = res.x
            z_new = np.clip(z_new, -30.0, 30.0)
            z_new -= float(np.mean(z_new))
            f_new = _softmax_from_z(z_new)
            sc_new = float(2.0 * n * _full_convolve(f_new).max())

            if np.isfinite(sc_new) and sc_new < best_score - 1e-12:
                best_score = sc_new
                best_f = f_new.copy()
                best_z = z_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        it += 1

        # Adaptive beta
        if it % 10 == 0:
            beta = max(beta_min, beta * beta_decay)
            f_now = _softmax_from_z(best_z)
            c_now = _full_convolve(f_now)
            m_now = float(c_now.max())
            # Adjust beta relative to the scale of c
            beta = max(beta_min, 0.05 * m_now * (0.5 + 0.5 * beta / max(beta_init, 1e-10)))

        if no_improve >= 15:
            break

    return best_f, best_score


def _active_set_gradient(f, deadline, tol_frac=0.02, n_steps=2000, lr_init=0.01):
    """Active-set gradient: uniformly reduce all near-peak convolution values."""
    n = len(f)
    f = np.asarray(f, dtype=np.float64).ravel().copy()
    f = np.maximum(f, 1e-12)
    f = f / f.sum()

    best_f = f.copy()
    best_score = float(2.0 * n * _full_convolve(f).max())
    c = _full_convolve(f)
    M = float(c.max())
    lr = lr_init
    momentum = np.zeros(n)
    mu = 0.9

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        # Active set: positions where c is within tol_frac of max
        threshold = M * (1.0 - tol_frac)
        active = c >= threshold
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > n:
            # If too many active, use top-n
            idx = np.argsort(c)[::-1][:n]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = n

        # h_j = sum_{k in active} f_{k-j} / n_active
        # = (1/n_active) * sum_{k in active} f_{k-j}
        # This is cross-correlation of (active_mask/n_active) and f
        w_active = active.astype(np.float64) / n_active
        h = _compute_h_fft(f, w_active)
        s_wc = float(np.dot(w_active, c))

        # Gradient of (1/n_active * sum_{k in active} c_k) w.r.t. z
        # = 2*f_j*(h_j - s_wc), then score gradient = 2n * that
        # For optimization, we just need the direction: 2*f*(h - s_wc)
        grad_z = 2.0 * f * (h - s_wc)
        grad_z -= float(np.mean(grad_z))

        # Momentum update
        momentum = mu * momentum + (1.0 - mu) * grad_z
        z = np.log(np.maximum(f, 1e-12))
        z -= float(np.mean(z))
        z -= lr * momentum
        z = np.clip(z, -30.0, 30.0)
        z -= float(np.mean(z))
        f_new = _softmax_from_z(z)
        c = _full_convolve(f_new)
        M = float(c.max())
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            # Increase learning rate slightly
            lr = min(0.05, lr * 1.02)
        else:
            f = f_new  # Still update to keep momentum going
            lr = max(0.001, lr * 0.95)

    return best_f, best_score


def _peak_flood(f, deadline, eta=0.04, n_iter_max=3000):
    """Multiplicative peak-flooding updates."""
    a = np.asarray(f, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = float(2.0 * n * _full_convolve(a).max())
    c = _full_convolve(a)

    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break

        M = float(c.max())
        threshold = M * 0.93
        hot_mask = c >= threshold
        n_hot = int(hot_mask.sum())
        if n_hot == 0:
            hot_mask[int(np.argmax(c))] = True

        # Compute heat: for each position i, sum of a[partner] over hot k
        heat = np.zeros(n)
        hot_indices = np.where(hot_mask)[0]
        for k in hot_indices:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                partner = k - np.arange(i_start, i_end + 1)
                heat[i_start:i_end + 1] += a[partner]

        heat_max = float(heat.max())
        if heat_max < 1e-15:
            break
        heat = heat / heat_max

        factor = 1.0 - eta * heat
        factor = np.maximum(factor, 0.01)
        a = a * factor
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        c = _full_convolve(a)

        score = 2.0 * n * float(c.max())
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        if it % 200 == 0 and it > 0:
            if score >= best_score:
                eta = max(0.005, eta * 0.9)
            else:
                eta = min(0.15, eta * 1.05)

    return best_a, best_score


def _lp_solve(f, M):
    """Solve LP: max sum(b) s.t. f*b <= M, b >= 0"""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(f[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_refine(f, deadline, t_init=0.4, max_solves=30, m_shrink=0.998):
    """LP-based refinement with decreasing M."""
    n = len(f)
    a = np.asarray(f, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = float(2.0 * n * _full_convolve(a).max())
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0
    t = t_init

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(a, M)
        if b is None:
            b = _lp_solve(a, M * 1.02)
            if b is None:
                break

        s_b = float(b.sum())
        if s_b < 1e-10:
            break

        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = float(2.0 * n * _full_convolve(a_new).max())

        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        c = _full_convolve(a)
        M_new = float(c.max())
        M = max(M_new * m_shrink, best_score / (2.0 * n) * 1.003)
        t = max(0.15, t * 0.93)

    return best_a, best_score


def _adam_logit(f0, deadline, rng, lr=0.04, beta_frac=0.06, beta_decay=0.94,
                no_imp_limit=200):
    """Adam in log-space (kept from previous, slightly tuned)."""
    p0 = np.asarray(f0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_val_weights(c, beta)
        h = _compute_h_fft(p, w)
        s_wc = float(np.dot(w, c))
        vp = 2.0 * p * (h - s_wc)
        vp -= float(np.mean(vp))

        if not np.all(np.isfinite(vp)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * vp
        v = b2 * v + (1.0 - b2) * (vp * vp)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -30.0, 30.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.002, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 20 == 0:
            beta = max(1e-5, beta * beta_decay)

        if no_imp > no_imp_limit:
            ns = 0.08 if best_score < 1.51 else 0.15
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -30.0, 30.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _make_profile_gaussian(n, width):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_power(n, alpha):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = np.abs(u) ** alpha
    a = a + a.min() + 0.01
    return a / a.sum()


def _make_profile_triangular(n, center_frac=0.5):
    t = np.linspace(0.0, 1.0, n)
    center = center_frac
    a = 1.0 - np.abs(t - center) / max(center, 1.0 - center)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x1, x0, p0))

        _update(f5, sc5, 'flood1')

        # Active set after flood
        if time.monotonic() < phase5_end - 20:
            f5a, sc5a = _active_set_gradient(best_seq, min(phase5_end, time.monotonic() + 60),
                                             tol_frac=0.02, n_steps=2000, lr_init=0.005)
            _update(f5a, sc5a, 'active2')

    # === Phase 6: Final intensive polish (remaining time) ===
    phase6_end = deadline
    print(f"Phase 6: Final polish, score={best_score:.10f}")
    if best_seq is not None:
        remaining = phase6_end - time.monotonic()
        if remaining > 100:
            # L-BFGS with very small beta for fine-tuning
            per1 = remaining * 0.4
            f6a, sc6a = _lbfgs_optimize(best_seq, time.monotonic() + per1,
                                        beta_init=0.02, beta_min=0.002,
                                        beta_decay=0.93, max_iter=200)
            _update(f6a, sc6a, 'final_lbfgs1')

        if time.monotonic() < phase6_end - 60 and best_seq.size <= 2048:
            per2 = min(80.0, phase6_end - time.monotonic() - 20)
            if per2 > 20:
                f6b, sc6b = _lp_refine(best_seq, time.monotonic() + per2,
                                       t_init=0.2, max_solves=10, m_shrink=0.999)
                _update(f6b, sc6b, 'final_lp')

        if time.monotonic() < phase6_end - 30:
            per3 = phase6_end - time.monotonic() - 30
            if per3 > 20:
                f6c, sc6c = _lbfgs_optimize(best_seq, time.monotonic() + per3,
                                            beta_init=0.01, beta_min=0.001,
                                            beta_decay=0.92, max_iter=300)
                _update(f6c, sc6c, 'final_lbfgs2')

        # Very final Adam micro-tune
        if time.monotonic() < phase6_end - 10:
            per4 = phase6_end - time.monotonic() - 10
            if per4 > 15:
                f6d, sc6d = _adam_logit(best_seq, time.monotonic() + per4,
                                        rng, lr=0.008, beta_frac=0.02,
                                        beta_decay=0.97, no_imp_limit=80)
                _update(f6d, sc6d, 'final_adam')

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_gaussian(1024, 300.0)

        exact_sc = _score_of(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")
        print(f"Time used: {time.monotonic() - t_start:.1f}s / {budget_s}s")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"EVAL error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile_gaussian(1024, 300.0)
    print(f"FALLBACK n=1024, score={_score_of(fb):.10f}")
    return [float(x) for x in fb]
