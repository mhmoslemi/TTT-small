        a_t = (1-t)*a + t*b
        a_t = np.maximum(a_t, 0)
        s = a_t.sum()
        if s < 1e-10:
            return np.inf
        a_t = a_t / s
        return _score_norm(a_t)

"""
LP-alternating optimization with golden-section line search, subgradient momentum,
and coordinate descent for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Focuses on aggressive LP iterations with active-set reduction and careful 1D line search.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time
import sys


def _next_pow2(n):
    if n <= 1:
        return 1
    k = 1
    while k < n:
        k <<= 1
    return k


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _cross_correlate(w, a):
    """Compute cross-correlation: result[j] = sum_k w[k]*a[k-j] aligned to length n."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w


def _build_conv_matrix(a):
    """Build sparse matrix A where A @ b = conv(a, b). Shape: (2n-1, n)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            # conv(a,b)[k] = sum_i a[i]*b[k-i], so coefficient of b[j] is a[k-j]
            coeff = a[k - idx].astype(np.float64)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(coeff.tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _build_conv_matrix_active(a, active_idx):
    """Build sparse matrix with only active constraints."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(active_idx):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeff = a[k - idx].astype(np.float64)
            rows.extend([i] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(coeff.tolist())
    return csr_matrix((vals, (rows, cols)), shape=(len(active_idx), n))


def _lp_solve(a, M, n, use_active=False, active_frac=0.3):
    """Solve: max sum(b) s.t. conv(a,b)[active] <= M, b >= 0."""
    if use_active:
        c = _full_convolve(a)
        threshold = c.max() * (1.0 - active_frac)
        active_idx = np.where(c >= threshold)[0]
        if len(active_idx) < 5:
            active_idx = np.argsort(c)[-10:]
        A = _build_conv_matrix_active(a, active_idx)
        b_ub = M * np.ones(len(active_idx))
    else:
        A = _build_conv_matrix(a)
        b_ub = M * np.ones(2 * n - 1)

    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30.0})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_solve_full(a, M, n):
    """Solve LP with all constraints (no active set reduction)."""
    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60.0})
        if res.success and res.x is not None:

def _golden_section_search(a, b, n, max_evals=30):
    """Find t in [0,1] minimizing score((1-t)*a + t*b)."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)

    def f(t):
        a_t = (1.0 - t) * a + t * b
        a_t = np.maximum(a_t, 1e-15)
        s = a_t.sum()
        if s < 1e-10:
            return np.inf
        a_t = a_t / s
        return _score_norm(a_t)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)

    for _ in range(max_evals):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)

    t_best = (lo + hi) / 2.0
    a_best = (1.0 - t_best) * a + t_best * b
    a_best = np.maximum(a_best, 1e-15)
    a_best = a_best / a_best.sum()
    sc = _score_norm(a_best)
    return a_best, sc, t_best


def _lp_alternating(a, deadline, max_iters=200, use_active=True,
                    active_frac=0.3, ls_evals=25, verbose=False):
    """LP alternating method with golden section line search."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    prev_score = best_score

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a)
        M = float(c.max())

        # Solve LP
        if use_active and n <= 2048:
            b = _lp_solve(a, M, n, use_active=True, active_frac=active_frac)
        else:
            b = _lp_solve_full(a, M, n)

        if b is None:
            break

        s_b = float(b.sum())
        if s_b < 1e-10:
            break

        b = b / s_b

        # Check if b is essentially the same as a (convergence)
        if np.allclose(a, b, atol=1e-10):
            break

        # Golden section line search
        a_new, sc_new, t_best = _golden_section_search(a, b, n, max_evals=ls_evals)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            if verbose and (it % 10 == 0 or sc_new < 1.51):
                print(f"    LP it={it}: score={sc_new:.10f}, t={t_best:.4f}, M={M:.8f}")

            # Check for near-convergence
            if prev_score - sc_new < 1e-9:
                if verbose:
                    print(f"    Near convergence at it={it}")
                break
            prev_score = sc_new
        else:
            # No improvement - try smaller step
            a_half, sc_half, _ = _golden_section_search(a, 0.5 * a + 0.5 * b, n, max_evals=10)
            if sc_half < best_score:
                best_score = sc_half
                best_a = a_half.copy()
                a = a_half
            else:
                # Stuck, break
                break

    return best_a, best_score


def _adam_smooth_max(a, deadline, rng, lr=0.03, beta_init=None,
                     beta_decay=0.96, no_imp_limit=300, momentum=0.9):
    """Adam smooth-max optimization in log-space."""
    n = len(a)
    a = np.maximum(a, 1e-12).copy().astype(np.float64)
    a = a / a.sum()

    z = np.log(a.copy())
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = _full_convolve(a)
    maxc = float(c.max())
    beta = beta_init if beta_init is not None else 0.05 * maxc

    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        if time.monotonic() > deadline:
            break

        _, w = _smooth_max_weights(c, beta)
        g = _cross_correlate(w, a)

        gz = a * (g - float(np.dot(a, g)))
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z -= z.mean()

        a = np.exp(z - z.max())
        a = a / a.sum()
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        c = _full_convolve(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-10:
                best_score = score
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.002, lr * 0.9)

            if t_adam % 20 == 0:
                beta = max(1e-5, beta * beta_decay)

            if no_imp > no_imp_limit:
                ns = 0.05
                z = best_z + ns * rng.normal(size=n)
                z = np.clip(z, -60, 60)
                z -= z.mean()
                a = np.exp(z - z.max())
                a = a / a.sum()
                c = _full_convolve(a)
                beta = 0.05 * c.max()
                m.fill(0)
                v.fill(0)
                t_adam = 0
                lr = base_lr
                no_imp = 0
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score


def _subgrad_momentum(a, deadline, rng, lr=0.05, mu=0.7, top_frac=0.15):
    """Subgradient descent with momentum using top-fraction of max positions."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        c = _full_convolve(a)
        M = c.max()
        n_top = max(3, int(top_frac * len(c)))
        top_idx = np.argsort(c)[-n_top:]
        w = np.zeros(len(c))
        w[top_idx] = 1.0 / n_top

        g = _cross_correlate(w, a)
        g = g - np.mean(g)

        v = mu * v - lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc = _score_norm(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                lr = max(0.001, lr * 0.9)
            if no_imp > 300:
                a = best_a + 0.03 * rng.normal(size=n)
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                v.fill(0)
                lr = base_lr
                no_imp = 0

    return best_a, best_score


def _coord_descent(a, deadline, max_sweeps=20):
    """Coordinate descent: try scaling each element by factors."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    factors = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            for factor in factors:
                a[j] = max(old_val * factor, 1e-15)
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = old_val
                    break
            a[j] = old_val
        if not improved:
            break

    return best_a, best_score


def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new


def _make_profile(n, ptype, p1=1.0, p2=0.0, p3=0.0):
    if ptype == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (t - 0.5)
    elif ptype == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (2 * t - 1) ** 2
    elif ptype == 'cubic':
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + p1 * u + p2 * u ** 3
    elif ptype == 'gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / p1) ** 2)
    elif ptype == 'cosine':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + p1 * np.cos(2 * np.pi * p2 * t / n)
    elif ptype == 'U_shape':
        t = np.linspace(0, 1, n)
        a = (2 * t - 1) ** 2 + p1
    elif ptype == 'edge_heavy':
        t = np.arange(n, dtype=np.float64) / max(1, n - 1)
        a = 1.0 / (0.01 + abs(t - 0.5) * 2.0) ** p1
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP alternating with golden-section line search,
    Adam smooth-max, subgradient momentum, coordinate descent, and multi-scale escalation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    # === Phase 1: Initialize candidates ===
    print("Phase 1: Generating candidates...")
    t_phase1 = time.monotonic()

    candidates = []

    if parent is not None:
        for n_try in [512, 1024, 1536, 2048]:
            pi = _interp(parent, n_try)
            sc = _score_norm(pi)
            candidates.append((sc, pi, n_try, 'parent'))
        if 0 < pn <= 4096:
            sc = _score_norm(parent)
            candidates.append((sc, parent.copy(), pn, 'parent_exact'))
            print(f"  Parent exact: score={sc:.10f}")

    # Profile-based candidates
    n_sizes = [256, 512, 1024, 2048]
    profiles = [
        ('lin1.0', 'linear', 1.0, 0, 0),
        ('lin1.5', 'linear', 1.5, 0, 0),
        ('lin2.0', 'linear', 2.0, 0, 0),
        ('quad1', 'quadratic', 1.0, 0, 0),
        ('quad2', 'quadratic', 2.0, 0, 0),
        ('cos05_1', 'cosine', 0.5, 1, 0),
        ('cos05_2', 'cosine', 0.5, 2, 0),
        ('U05', 'U_shape', 0.5, 0, 0),
        ('g03', 'gauss', 0.3, 0, 0),
        ('edge1', 'edge_heavy', 1.0, 0, 0),
    ]

    for n in n_sizes:
        for (label, ptype, p1, p2, p3) in profiles:
            p = _make_profile(n, ptype, p1, p2, p3)
            sc = _score_norm(p)
            candidates.append((sc, p, n, label))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, best: {candidates[0][0]:.8f} n={candidates[0][2]} [{candidates[0][3]}]")

    if candidates[0][0] < best_score:
        best_seq = candidates[0][1].copy()
        best_score = candidates[0][0]
        print(f"  INITIAL BEST: {best_score:.10f} n={best_seq.size}")

    # === Phase 2: LP alternating on top candidates ===
    print("\nPhase 2: LP alternating optimization...")
    t_phase2_end = t_start + 1000.0

    # Sort by (score, n) - prefer good score and smaller n for LP speed
    lp_candidates = sorted(candidates, key=lambda x: (x[0], x[2]))

    # Pick top candidates for full LP treatment
    lp_targets = []
    seen = set()
    for (sc, p, n, label) in lp_candidates:
        key = (n, label[:4])
        if key not in seen:
            lp_targets.append((sc, p, n, label))
            seen.add(key)
        if len(lp_targets) >= 8:
            break

    for idx, (sc, p, n, label) in enumerate(lp_targets):
        if time.monotonic() > t_phase2_end - 60:
            break
        if n > 2048:
            continue

        # Time allocation: more time for smaller n (faster LP)
        if n <= 512:
            time_alloc = 200.0
        elif n <= 1024:
            time_alloc = 300.0
        else:
            time_alloc = 150.0

        per_end = min(t_phase2_end, time.monotonic() + time_alloc)
        print(f"  [{idx}] LP on n={n} [{label}] init_score={sc:.8f}, time={time_alloc:.0f}s")

        # Quick LP first (with active set)
        a_lp, sc_lp = _lp_alternating(p, per_end, max_iters=300,
                                       use_active=True, active_frac=0.3,
                                       ls_evals=25, verbose=True)
        if sc_lp < best_score:
            best_seq = a_lp.copy()
            best_score = sc_lp
            print(f"    ** NEW BEST: {best_score:.10f} n={a_lp.size} [LP_active_{label}]")

    # Second round: full LP (all constraints) on the very best
    if time.monotonic() < t_phase2_end - 30 and best_seq is not None:
        n_best = min(best_seq.size, 1024)
        p_best = _interp(best_seq, n_best) if best_seq.size != n_best else best_seq.copy()
        sc_best = _score_norm(p_best)
        if sc_best < best_score:
            best_seq = p_best.copy()
            best_score = sc_best

        print(f"  Full LP polish on n={n_best}...")
        per_end = min(t_phase2_end, time.monotonic() + 120.0)
        a_fl, sc_fl = _lp_alternating(p_best, per_end, max_iters=100,
                                       use_active=False, ls_evals=20, verbose=True)
        if sc_fl < best_score:
            best_seq = a_fl.copy()
            best_score = sc_fl
            print(f"    ** NEW BEST: {best_score:.10f} n={a_fl.size} [LP_full]")

    print(f"\n  After Phase 2: best={best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    # === Phase 3: Adam smooth-max refinement ===
    print("\nPhase 3: Adam refinement...")
    t_phase3_end = t_start + 1400.0

    if best_seq is not None and time.monotonic() < t_phase3_end:
        per = min(200.0, t_phase3_end - time.monotonic() - 30)
        if per > 50:
            a_ad, sc_ad = _adam_smooth_max(best_seq, time.monotonic() + per, rng,
                                           lr=0.03, beta_decay=0.97, no_imp_limit=400)
            if sc_ad < best_score:
                best_seq = a_ad.copy()
                best_score = sc_ad
                print(f"  ** NEW BEST: {best_score:.10f} n={a_ad.size} [Adam]")

    # Subgradient momentum
    if best_seq is not None and time.monotonic() < t_phase3_end - 30:
        per = min(100.0, t_phase3_end - time.monotonic() - 10)
        if per > 30:
            a_sg, sc_sg = _subgrad_momentum(best_seq, time.monotonic() + per, rng,
                                            lr=0.03, mu=0.7, top_frac=0.2)
            if sc_sg < best_score:
                best_seq = a_sg.copy()
                best_score = sc_sg
                print(f"  ** NEW BEST: {best_score:.10f} n={a_sg.size} [Subgrad]")

    # === Phase 4: Multi-scale escalation ===
    print("\nPhase 4: Multi-scale escalation...")
    t_phase4_end = t_start + 1600.0

    if best_seq is not None and time.monotonic() < t_phase4_end:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048]
        elif n_cur <= 1024:
            target_ns = [2048, 3072]
        elif n_cur <= 2048:
            target_ns = [4096]

        for tn in target_ns:
            if time.monotonic() > t_phase4_end - 60:
                break
            if tn > 4096:
                continue

            p_int = _interp(best_seq, tn)
            sc_int = _score_norm(p_int)
            print(f"  n={tn}: score={sc_int:.10f}")

            if sc_int < best_score:
                best_seq = p_int.copy()
                best_score = sc_int
                print(f"  ** NEW BEST: {best_score:.10f} n={tn} [interp]")

            # Quick LP on interpolated (if feasible)
            if tn <= 1024 and time.monotonic() < t_phase4_end - 60:
                per = min(80.0, t_phase4_end - time.monotonic() - 30)
                if per > 30:
                    a_e, sc_e = _lp_alternating(p_int, time.monotonic() + per,
                                                 max_iters=50, use_active=True,
                                                 active_frac=0.3, ls_evals=15)
                    if sc_e < best_score:
                        best_seq = a_e.copy()
                        best_score = sc_e
                        print(f"  ** NEW BEST: {best_score:.10f} n={tn} [LP_esc]")
            elif tn > 1024 and time.monotonic() < t_phase4_end - 40:
                # Use Adam for larger n
                per = min(60.0, t_phase4_end - time.monotonic() - 20)
                if per > 20:
                    a_e, sc_e = _adam_smooth_max(p_int, time.monotonic() + per, rng,
                                                  lr=0.02, beta_decay=0.97, no_imp_limit=200)
                    if sc_e < best_score:
                        best_seq = a_e.copy()
                        best_score = sc_e
                        print(f"  ** NEW BEST: {best_score:.10f} n={tn} [Adam_esc]")

    # === Phase 5: Coordinate descent polish ===
    print("\nPhase 5: Coordinate descent polish...")
    t_phase5_end = deadline - 30.0

    if best_seq is not None and time.monotonic() < t_phase5_end:
        per = min(120.0, t_phase5_end - time.monotonic())
        if per > 20:
            a_cd, sc_cd = _coord_descent(best_seq, time.monotonic() + per, max_sweeps=10)
            if sc_cd < best_score:
                best_seq = a_cd.copy()
                best_score = sc_cd
                print(f"  ** NEW BEST: {best_score:.10f} n={a_cd.size} [coord_descent]")

    # === Phase 6: Final LP polish on best ===
    print("\nPhase 6: Final polish...")
    if best_seq is not None and time.monotonic() < deadline - 15:
        n_final = min(best_seq.size, 1024)
        p_final = _interp(best_seq, n_final) if best_seq.size != n_final else best_seq.copy()
        sc_final = _score_norm(p_final)
        if sc_final < best_score:
            best_seq = p_final.copy()
            best_score = sc_final

        per = deadline - time.monotonic() - 5
        if per > 20 and n_final <= 1024:
            a_fp, sc_fp = _lp_alternating(p_final, time.monotonic() + per,
                                           max_iters=30, use_active=True,
                                           active_frac=0.2, ls_evals=20)
            if sc_fp < best_score:
                best_seq = a_fp.copy()
                best_score = sc_fp
                print(f"  ** NEW BEST: {best_score:.10f} n={a_fp.size} [final_LP]")

    # === Final validation ===
    print(f"\n=== FINAL ===")
    if best_seq is not None:
        final = np.clip(best_seq.astype(np.float64), 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile(512, 'linear', 1.5, 0.0, 0.0)

        exact_sc = _score_norm(final)
        print(f"  Score: {exact_sc:.10f}, n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            print(f"  Eval: {eval_sc:.10f}")
        except Exception as e:
            print(f"  Eval error: {e}")

        # Ensure no NaN or inf
        if not np.all(np.isfinite(final)):
            final = _make_profile(512, 'linear', 1.5, 0.0, 0.0)

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile(512, 'linear', 1.5, 0.0, 0.0)
    print(f"  FALLBACK: n=512, score={_score_norm(fb):.10f}")
    return [float(x) for x in fb]
