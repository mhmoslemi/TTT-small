Optimized step function search using direct subgradient methods with

import time
from scipy.optimize import minimize, minimize_scalar

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    """Compute the full score: 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1 (normalized)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _subgradient(c, a, tol_frac=0.0):
    """
    Compute subgradient of max(c) w.r.t. a.
    Returns (grad, active_count) where grad[i] = (1/|A|) sum_{k in A} a[k-i]
    """
    n = len(a)
    c_max = c.max()
    if tol_frac == 0.0:
        active = (c == c_max)
    else:
        active = (c >= c_max * (1 - tol_frac))
    if not np.any(active):
        active[int(np.argmax(c))] = True
    
    k_active = np.where(active)[0]
    # grad[i] = (1/|A|) sum_{k in A} a[k-i] (for valid i, k-i)
    grad = np.zeros(n)
    for k in k_active:
        # c[k] = sum_i a[i]*a[k-i], dc[k]/da[i] = a[k-i]
        # valid i: 0 <= i < n and 0 <= k-i < n
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            grad[i_min:i_max+1] += a[k - np.arange(i_min, i_max+1)]
    grad /= len(k_active)
    return grad, len(k_active)

def _subgradient_fft(c, a, tol_frac=0.0):
    """FFT-based subgradient computation."""
    n = len(a)
    c_max = c.max()
    if tol_frac == 0.0:
        active = (c == c_max)
    else:
        active = (c >= c_max * (1 - tol_frac))
    if not np.any(active):
        active[int(np.argmax(c))] = True
    
    # grad[i] = (1/|A|) sum_{k in A} a[k-i]
    # This is a cross-correlation of 'active' (as a filter) with a
    # Specifically: conv(active, a_rev)[n-1 + i] gives sum_k active[k]*a_rev[n-1+i-k]
    # a_rev[j] = a[n-1-j]
    # conv(active, a_rev)[n-1+i] = sum_k active[k] * a_rev[n-1+i-k] = sum_k active[k] * a[n-1-(n-1+i-k)] = sum_k active[k] * a[k-i]
    # That's what we want!
    
    a_rev = a[::-1]
    N = _next_pow2(2*n)
    A = np.fft.rfft(active.astype(np.float64), N)
    B = np.fft.rfft(a_rev, N)
    conv_result = np.fft.irfft(A * B, N)
    # conv_result[n-1+i] = sum_k active[k] * a[k-i]
    grad = conv_result[n-1:n-1+n].copy()
    grad /= active.sum()
    return grad

def _make_profile(n, profile_type="cos2", **kwargs):
    t = np.linspace(0, 1, n)
    if profile_type == "uniform":
        a = np.ones(n)
    elif profile_type == "cos2":
        a = np.cos(np.pi * t / 2)**2
    elif profile_type == "sin2":
        a = np.sin(np.pi * t)**2
    elif profile_type == "cos4":
        a = np.cos(np.pi * t / 2)**4
    elif profile_type == "bell":
        sigma = kwargs.get("sigma", 0.3)
        a = np.exp(-((t - 0.5)**2) / (2*sigma**2))
    elif profile_type == "u_shape":
        a = (2*t - 1)**2 + kwargs.get("floor", 0.1)
    elif profile_type == "linear":
        slope = kwargs.get("slope", 1.0)
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 0.01)
    elif profile_type == "polynomial":
        coeffs = kwargs.get("coeffs", [1, 0, 0])
        u = 2*t - 1
        a = np.zeros(n)
        for j, cj in enumerate(coeffs):
            a += cj * u**j
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _optimize_subgradient(a_init, deadline, tol_frac=0.01, lr_init=0.01, patience=200):
    """
    Direct subgradient method with adaptive step size.
    a is normalized to sum=1 throughout.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    lr = lr_init
    stall_count = 0
    momentum = np.zeros(n)
    mom = 0.9
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        c = _full_convolve(a, a)
        grad = _subgradient_fft(c, a, tol_frac)
        
        # Normalize gradient (remove mean to preserve sum)
        grad = grad - grad.mean()
        # Scale gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        
        # Momentum
        momentum = mom * momentum + grad
        momentum = momentum - momentum.mean()
        
        # Step
        a_new = a + lr * momentum
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count % 50 == 0:
                lr *= 0.8
                # Revert to best
                a = best_a.copy()
        
        if stall_count > patience:
            break
        
        a = a_new
    
    return best_a, best_score

def _optimize_smooth(a_init, deadline, beta_init=0.01):
    """
    Smooth optimization using log-sum-exp approximation.
    Works on the log-space (softmax parameterization).
    """
    from scipy.special import softmax
    
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score_norm = _score_norm(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    beta = beta_init * max_c0
    
    # Adam optimizer
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    lr = 0.05
    
    while time.monotonic() < deadline:
        t += 1
        a = softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        
        # Smooth approximation gradient
        c_shifted = (c - c_max) / beta
        e = np.exp(c_shifted)
        w = e / e.sum()
        W = float(np.dot(w, c))
        
        # grad of smooth objective w.r.t. a
        a_rev = a[::-1]
        N = _next_pow2(2*n)
        Wf = np.fft.rfft(w, N)
        Af = np.fft.rfft(a_rev, N)
        w_conv = np.fft.irfft(Wf * Af, N)
        h = w_conv[n-1:n-1+n]
        
        # d(obj)/da_i = 4*n*a_i*(h_i - W) (for normalized a, obj = 2*n*smooth_max)
        grad_a = 4.0 * n * a * (h - W)
        
        # Transform to z-space: da/dz = a(I - a*a^T)
        # So grad_z = (I - a*a^T) @ (grad_a / a) = grad_a/a - mean(grad_a/a) * a ... 
        # Actually: d smooth_obj / dz_i = sum_j (d smooth_obj / da_j) * (da_j / dz_i)
        # da_j / dz_i = a_i * (delta_ij - a_j)
        # So grad_z_i = sum_j grad_a_j * a_i * (delta_ij - a_j) = a_i * (grad_a_i - sum_j grad_a_j * a_j)
        # = a_i * (grad_a_i - dot(grad_a, a))
        dot_ga_a = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - dot_ga_a)
        
        if not np.all(np.isfinite(grad_z)):
            grad_z = np.zeros(n)
        
        # Adam step
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = z - z.mean()
        
        if t % 100 == 0:
            a_curr = softmax(z)
            sc = _score_norm(a_curr)
            if np.isfinite(sc) and sc < best_score_norm:
                best_score_norm = sc
                best_a = a_curr.copy()
            # Decay beta
            beta *= 0.95
            c_new = _full_convolve(a_curr, a_curr)
            max_c_new = float(c_new.max())
            # Reset Adam
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
        
        if t % 500 == 0:
            lr = max(0.001, lr * 0.9)
    
    a_final = softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score_norm:
        best_score_norm = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score_norm

def _polynomial_parametrization(n, degree=10):
    """Create a polynomial parametrization for the sequence."""
    t = np.linspace(0, 1, n)
    u = 2*t - 1  # [-1, 1]
    # Legendre polynomials
    V = np.zeros((degree+1, n))
    V[0, :] = 1
    if degree >= 1:
        V[1, :] = u
    for j in range(2, degree+1):
        V[j, :] = ((2*j-1)*u*V[j-1, :] - (j-1)*V[j-2, :]) / j
    return V, u

def _optimize_polynomial(V, n, deadline, degree=10):
    """
    Optimize polynomial coefficients to minimize the score.
    a[i] = max(0, sum_j c_j * V[j][i]) / norm
    """
    from scipy.optimize import minimize as sp_min
    
    def obj(c):
        a = V.T @ c
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            return 1e10
        a = a / s
        return _score_norm(a)
    
    # Initial guess: constant (only c_0 nonzero)
    c0 = np.zeros(degree+1)
    c0[0] = 1.0
    
    # Use Nelder-Mead (derivative-free) for robustness
    best_c = c0.copy()
    best_val = obj(c0)
    
    # Try L-BFGS-B first
    try:
        result = sp_min(obj, c0, method='Nelder-Mead', 
                        options={'maxiter': 2000, 'xatol': 1e-10, 'fatol': 1e-10})
        if result.fun < best_val:
            best_val = result.fun
            best_c = result.x.copy()
    except Exception:
        pass
    
    a = V.T @ best_c
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a, best_val

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Adam-based smooth optimization (log-sum-exp approximation)
2. Projected subgradient on the exact non-smooth objective with simplex projection
3. Nesterov-accelerated smooth refinement
4. Multi-restart strategy with structured perturbations
5. Adaptive active-set management
Key improvements over previous: backtracking line search, better step size
scheduling, adaptive tolerance, and projected gradient on simplex.
"""
import numpy as np
from scipy.special import softmax
from scipy.ndimage import gaussian_filter1d
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgradient_fft(c, a, tol_abs=None, tol_frac=0.0):
    """Compute subgradient of max(c) w.r.t. a using FFT."""
    n = len(a)
    c_max = float(c.max())
    if tol_abs is not None:
        active = (c >= c_max - tol_abs).astype(np.float64)
    elif tol_frac > 0:
        active = (c >= c_max * (1.0 - tol_frac)).astype(np.float64)
    else:
        active = (c == c_max).astype(np.float64)
    if active.sum() == 0:
        active[int(np.argmax(c))] = 1.0

    # g[i] = sum_k active[k] * a[k-i] / active.sum()
    # Using: conv(active, a_rev)[n-1+i] = sum_k active[k]*a[k-i]
    a_rev = a[::-1]
    N = _next_pow2(2 * n)
    A = np.fft.rfft(active, N)
    B = np.fft.rfft(a_rev, N)
    conv_result = np.fft.irfft(A * B, N)
    grad = conv_result[n - 1:n - 1 + n].copy()
    grad /= active.sum()
    return grad


def _smooth_grad_z(z, beta):
    """Gradient of smooth objective w.r.t. z (log-space)."""
    a = softmax(z)
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_shifted = (c - c_max) / beta
    e = np.exp(c_shifted)
    w = e / e.sum()
    W = float(np.dot(w, c))

    a_rev = a[::-1]
    N = _next_pow2(2 * n)
    Wf = np.fft.rfft(w, N)
    Af = np.fft.rfft(a_rev, N)
    w_conv = np.fft.irfft(Wf * Af, N)
    h = w_conv[n - 1:n - 1 + n]

    # d(2n*smooth_max)/da_i = 4*n*a_i*(h_i - W)
    grad_a = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad_a)):
        return np.zeros(n)

    # Transform to z-space
    dot_ga_a = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - dot_ga_a)
    return grad_z


def _make_profile(n, profile_type="cos2"):
    t = np.linspace(0, 1, n)
    if profile_type == "uniform":
        a = np.ones(n)
    elif profile_type == "cos2":
        a = np.cos(np.pi * t / 2) ** 2
    elif profile_type == "sin2":
        a = np.sin(np.pi * t) ** 2
    elif profile_type == "cos4":
        a = np.cos(np.pi * t / 2) ** 4
    elif profile_type == "bell":
        a = np.exp(-2.0 * (t - 0.5) ** 2)
    elif profile_type == "u_shape":
        a = (2 * t - 1) ** 2 + 0.1
    elif profile_type == "linear":
        a = 1.0 + 1.5 * (t - 0.5)
        a = np.maximum(a, 0.01)
    elif profile_type == "cos2_mod":
        a = np.cos(np.pi * t / 2) ** 2 * (1 + 0.3 * np.sin(2 * np.pi * t))
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _adam_smooth(a_init, deadline, beta_init=0.005, lr=0.05, beta_decay=0.97):
    """Adam optimization on softmax parameterization with smooth max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    beta = beta_init * max_c0

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8

    stall = 0
    patience = 500

    while time.monotonic() < deadline:
        t += 1
        grad_z = _smooth_grad_z(z, beta)

        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = z - z.mean()

        if t % 200 == 0:
            a_curr = softmax(z)
            sc = _score_norm(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
                stall = 0
            else:
                stall += 1

            # Decay beta and reset momentum
            beta *= beta_decay
            c_new = _full_convolve(best_a, best_a)
            max_c_new = float(c_new.max())
            beta = max(beta, 1e-10 * max_c_new)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = max(0.002, lr * 0.92)

            if stall > patience:
                break

    a_final = softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _projected_subgradient(a_init, deadline, tol_frac_init=0.05, lr_init=0.005):
    """
    Projected subgradient with simplex projection and backtracking line search.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)

    lr = lr_init
    tol_frac = tol_frac_init
    stall = 0
    patience = 500

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive tolerance: start wide, narrow over time
        g = _subgradient_fft(c, a, tol_frac=tol_frac)

        # Backtracking line search
        alpha = lr
        a_curr = a
        sc_curr = c_max
        improved = False
        for _ in range(10):
            a_trial = a_curr - alpha * g
            a_trial = _project_simplex(a_trial)
            c_trial = _full_convolve(a_trial, a_trial)
            sc_trial = float(c_trial.max())
            if sc_trial < sc_curr:
                improved = True
                break
            alpha *= 0.5
            a_curr = a_trial
            sc_curr = sc_trial

        if improved:
            a = a_trial
            sc = float(_full_convolve(a, a).max())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            lr = max(1e-6, lr * 0.8)

        # Narrow tolerance over time
        tol_frac = max(0.001, tol_frac * 0.995)

        if stall > patience:
            break

    return best_a, best_score

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

                update_best(p, sc, f"{prof}_n{n_try}")
    print(f"Phase 0 done: best={best_score:.10f}")

    # --- Phase 1: Adam smooth optimization (250s) ---
    phase1_end = min(deadline, t_start + 265.0)
    if best_seq is not None:
        p_best, sc_best = _adam_smooth(best_seq, phase1_end, beta_init=0.005, lr=0.05, beta_decay=0.97)
        update_best(p_best, sc_best, "adam_smooth_P1")
    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Projected subgradient (200s) ---
    phase2_end = min(deadline, t_start + 465.0)
    if best_seq is not None:
        p_best, sc_best = _projected_subgradient(best_seq, phase2_end, tol_frac_init=0.03, lr_init=0.003)
        update_best(p_best, sc_best, "proj_subgrad_P2")
    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Nesterov smooth (180s) ---
    phase3_end = min(deadline, t_start + 645.0)
    if best_seq is not None:
        p_best, sc_best = _nesterov_smooth(best_seq, phase3_end, beta_init=0.002, lr_init=0.015, mom_init=0.9)
        update_best(p_best, sc_best, "nesterov_P3")
    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Subgradient with narrower tolerance (150s) ---
    phase4_end = min(deadline, t_start + 795.0)
    if best_seq is not None:
        p_best, sc_best = _projected_subgradient(best_seq, phase4_end, tol_frac_init=0.005, lr_init=0.002)
        update_best(p_best, sc_best, "proj_subgrad_P4")
    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multi-restart (350s) ---
    phase5_end = min(deadline, t_start + 1145.0)
    if best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 5
        per_restart = (phase5_end - time.monotonic()) / max(1, n_restarts)

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 20:
                break
            per_dl = min(phase5_end - 10, time.monotonic() + per_restart * 0.7)

            if i == 0:
                # Smooth noise + Nesterov
                z = np.log(np.maximum(best_seq, 1e-15))
                noise = rng.normal(size=n_best)
                noise = gaussian_filter1d(noise, sigma=max(1, n_best // 100))
                z = z + 0.08 * noise
                z = z - z.mean()
                p_start = softmax(z)
                p_best, sc_best = _nesterov_smooth(p_start, per_dl, beta_init=0.003, lr_init=0.01, mom_init=0.85)
                update_best(p_best, sc_best, f"restart_nest_{i}")

            elif i == 1:
                # Flip + Adam
                p_start = best_seq[::-1].copy()
                p_best, sc_best = _adam_smooth(p_start, per_dl, beta_init=0.005, lr=0.04)
                update_best(p_best, sc_best, f"restart_flip_{i}")

            elif i == 2:
                # Multiply by smooth modulator + subgradient
                t = np.linspace(0, 1, n_best)
                mod = 1.0 + 0.15 * np.sin(2 * np.pi * rng.uniform(1, 4) * t + rng.uniform(0, 2 * np.pi))
                p_start = best_seq * mod
                p_start = np.maximum(p_start, 0)
                p_start = p_start / p_start.sum()
                p_best, sc_best = _projected_subgradient(p_start, per_dl, tol_frac_init=0.02, lr_init=0.004)
                update_best(p_best, sc_best, f"restart_mod_{i}")

            elif i == 3:
                # Blend with a different profile
                p_alt = _make_profile(n_best, "sin2")
                p_start = 0.7 * best_seq + 0.3 * p_alt
                p_start = p_start / p_start.sum()
                p_best, sc_best = _adam_smooth(p_start, per_dl, beta_init=0.004, lr=0.04)
                update_best(p_best, sc_best, f"restart_blend_{i}")

            else:
                # Stronger perturbation + Nesterov
                z = np.log(np.maximum(best_seq, 1e-15))
                noise = rng.normal(size=n_best)
                noise = gaussian_filter1d(noise, sigma=max(1, n_best // 50))
                z = z + 0.15 * noise
                z = z - z.mean()
                p_start = softmax(z)
                p_best, sc_best = _nesterov_smooth(p_start, per_dl, beta_init=0.005, lr_init=0.012, mom_init=0.88)
                update_best(p_best, sc_best, f"restart_strong_{i}")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Try different lengths (200s) ---
    phase6_end = min(deadline, t_start + 1345.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        n_curr = len(best_seq)
        target_ns = []
        if n_curr <= 4096:
            target_ns.append(n_curr * 2)
        if n_curr >= 2048:
            target_ns.append(max(1024, n_curr // 2))
        target_ns = [tn for tn in target_ns if tn <= 16384 and tn != n_curr]

        for target_n in target_ns:
            if time.monotonic() > phase6_end - 30:
                break
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score(p_interp)
            if sc_interp < best_score:
                update_best(p_interp, sc_interp, f"interp_n{target_n}")

            per_dl = min(phase6_end - 10, time.monotonic() + 80)
            if per_dl - time.monotonic() > 20:
                p_best, sc_best = _adam_smooth(p_interp, per_dl, beta_init=0.005, lr=0.04)
                update_best(p_best, sc_best, f"adam_n{target_n}")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final alternating polish (remaining time) ---
    phase7_end = deadline
    if best_seq is not None and time.monotonic() < phase7_end:
        # Alternate between Adam (short) and subgradient (short)
        while time.monotonic() < phase7_end - 10:
            remaining = phase7_end - time.monotonic()
            chunk = min(60.0, remaining - 5)
            if chunk < 10:
                break

            # Adam
            per_dl = min(phase7_end - 5, time.monotonic() + chunk * 0.5)
