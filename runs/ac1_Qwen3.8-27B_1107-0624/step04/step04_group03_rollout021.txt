LP-based optimizer for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses the "best response" LP from the state of the art, combined with
gradient-based polishing.
"""
import numpy as np
from scipy.optimize import minimize, linprog
from scipy import sparse
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    """Score with sum(a) = 1 (scale invariant)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_fn(a):
    """Score as in evaluate_sequence."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _build_conv_matrix(f, n):
    """
    Build the (2n-1) x n matrix A where A[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    This is the matrix such that A @ g = convolve(f, g).
    """
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def _solve_response_lp(f, M, solver='highs'):
    """
    Solve: max sum(g) s.t. g >= 0, f*g <= M (componentwise)
    Returns the optimal g.
    """
    n = len(f)
    A = _build_conv_matrix(f, n)
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})
    
    if res.status == 0:
        return res.x
    return None

```python
"""
LP-based optimizer with gradient polishing for minimizing
2*n*max(convolve(a,a))/sum(a)^2 with non-negative sequences.
"""
import numpy as np
from scipy.optimize import minimize
from scipy import sparse
from scipy.optimize import linprog
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _build_conv_matrix(f, n):
    """Build (2n-1) x n matrix A where A @ g = convolve(f, g)."""
    # Column j has values f[0],...,f[n-1] at rows j, j+1, ..., j+n-1
    all_cols = np.repeat(np.arange(n), n)
    all_rows = all_cols + np.tile(np.arange(n), n)
    all_vals = np.tile(f, n)
    A = sparse.csr_matrix((all_vals, (all_rows, all_cols)), shape=(2*n-1, n))
    return A

def _solve_response_lp(f, M, max_time=30.0):
    """Solve max sum(g) s.t. A@f*g <= M, g >= 0."""
    n = len(f)
    A = _build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)]*n, method='highs',
                      options={'time_limit': max_time})
        if res.status == 0 and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def _lp_phase(f, deadline, max_iters=200):
    """LP-based best response iteration."""
    f = f.copy()
    s = f.sum()
    if s < 1e-15:
        return f
    f = f / s
    n = len(f)
    
    M = float((_full_convolve(f, f)).max())
    best_score = _score_norm(f)
    best_f = f.copy()
    
    no_improve_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 10:
            break
        
        g = _solve_response_lp(f, M * 1.001, max_time=20.0)
        if g is None or g.sum() < 1e-10:
            no_improve_count += 1
            if no_improve_count >= 3:
                break
            continue
        
        g = g / g.sum()  # normalize to sum 1
        
        # Line search
        best_t = 0.0
        best_sc = best_score
        for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
            f_try = (1-t)*f + t*g
            sc = _score_norm(f_try)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_t = t
        
        if best_t == 0.0:
            no_improve_count += 1
            if no_improve_count >= 5:
                break
            # Try very small t
            for t in [0.01, 0.02, 0.05]:
                f_try = (1-t)*f + t*g
                sc = _score_norm(f_try)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_t = t
                    break
            if best_t == 0.0:
                break
        
        f = (1-best_t)*f + best_t*g
        M = float(_full_convolve(f, f).max())
        
        if _score_norm(f) < best_score:
            best_score = _score_norm(f)
            best_f = f.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        if it % 10 == 0:
            print(f"  LP iter {it}: score={_score_norm(f):.10f}, M={M:.10f}, t={best_t}")
    
    return best_f

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm

def _gradient_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n-1:2*n-1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

def _objective_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)

def _nesterov_optimize(a_init, deadline, beta_fracs, lr_init=0.01, mom_init=0.9):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    
    lr = lr_init
    mom = mom_init
    t = 0
    stagnation = 0
    
    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-12, beta_frac * max_c)
        
        while time.monotonic() < deadline:
            t += 1
            grad_y = _gradient_smooth(y, beta)
            if not np.all(np.isfinite(grad_y)):
                grad_y = _gradient_smooth(z, beta)
            
            z_new = z - lr * grad_y
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -50, 50)
            
            y_new = z_new + mom * (z_new - z)
            y_new = y_new - y_new.mean()
            
            z = z_new
            y = y_new
            
            if t % 200 == 0:
                a_curr = _softmax_fn(z)
                sc = _score_norm(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()
                    stagnation = 0
                else:
                    stagnation += 1
                if stagnation > 5:
                    break
            
            if t % 500 == 0:
                mom = min(0.999, mom + 0.001)
                lr = max(0.001, lr * 0.95)
        
        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        max_c = float(c.max())
    
    a_final = _softmax_fn(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _adam_optimize(a_init, deadline, lr=0.03, beta_frac=0.01):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, beta_frac * max_c)
    
    stagnant = 0
    
    for it in range(1000000):
        if time.monotonic() > deadline:
            break
        t += 1
        
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n-1:2*n-1]
        grad = 4.0 * n * a * (h - W)
        
        if not np.all(np.isfinite(grad)):
            continue
        
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*(grad*grad)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        
        if t % 300 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
                stagnant = 0
            else:
                stagnant += 1
            beta = max(1e-15, beta * 0.85)
            if stagnant > 3:
                break
    
    a_final = _softmax_fn(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _make_bathtub_profile(n, curvature=1.0):
    t = np.linspace(0, 1, n)
    u = 2*t - 1
    a = 1.0 + curvature * u**2
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_linear_profile(n, slope):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope*(t - 0.5)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0, 1, n)
    u = 2*t - 1
    a = 1.0 + a1*u + a2*u**2
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-10)
    return p / p.sum()
