```python
"""
Search algorithm for minimizing 2*n*max(conv(a,a))/sum(a)^2.

The code uses several complementary methods:
1. Interpolation of the parent sequence to larger lengths (usually 8192/16384).
2. L-BFGS on smooth surrogates:
   - p-norm of the convolution coefficients,
   - log-sum-exp smoothing,
   - active-set stages minimizing the sum of squares of the current top peaks.
3. Nonsmooth mirror-descent Adam using subgradients from top active lags.
4. Targeted local search around the current largest convolution coefficients.
5. A few diverse beta/exponential/random initializations to avoid local minima.

All returned values are nonnegative and normalized to sum 1.
"""

import time
import numpy as np
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    n = int(max(1, n))
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    out_n = int(a.size + b.size - 1)
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return np.ones_like(a, dtype=np.float64) / max(1, a.size)
    return a / s


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.where(np.isfinite(z), z, 0.0)
    z = z - z.max()
    ex = np.exp(z)
    s = float(ex.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return np.ones_like(z, dtype=np.float64) / max(1, z.size)
    return ex / s


def _log_from_a(a):
    a = _normalize(a)
    z = np.log(np.maximum(a, 1e-300))
    z = z - float(z.mean())
    return z


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    n = int(a.size)
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    cm = float(np.max(c))
    if not np.isfinite(cm):
        return np.inf
    return float(2.0 * n * cm / (s * s))


def _quad_conv_weight_a(w, a):
    """
    Return h of length n where
        h[j] = sum_k w[k] * a[k-j]
    over valid 0 <= k-j < n.
    This is the linear derivative kernel of the convolution square.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    out_n = int(w.size + n - 1)
    N = _next_pow2(out_n)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    r = np.fft.irfft(fw * fa, N)
    return r[n - 1:2 * n - 1]


def _topk_direct_h(c, a, K, weight_mode):
    """
    Fast direct construction of h for top-K active lags.
    weight_mode:
      0 = equal weights on top K
      1 = softmax-like weights on top K
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    c = np.asarray(c, dtype=np.float64).ravel()
    c = np.maximum(c, 0.0)
    n = int(a.size)
    L = int(c.size)
    K = int(min(K, L))
    h = np.zeros(n, dtype=np.float64)
    if K <= 0:
        return h

    if K >= L:
        idx = np.arange(L)
        vals = c
    else:
        idx = np.argpartition(c, -K)[-K:]
        vals = c[idx]

    if weight_mode == 1:
        vmax = float(vals.max())
        vmin = float(vals.min())
        spread = vmax - vmin
        if spread <= 1e-300:
            w = np.ones(K, dtype=np.float64) / K
        else:
            temp = 0.25 * spread + 1e-12 * max(1e-300, vmax)
            xv = (vals - vmax) / temp
            ex = np.exp(xv)
            s = float(ex.sum())
            if s > 1e-300 and np.isfinite(s):
                w = ex / s
            else:
                w = np.ones(K, dtype=np.float64) / K
    else:
        w = np.ones(K, dtype=np.float64) / K

    for ii in range(K):
        kk = int(idx[ii])
        wk = float(w[ii])
        if wk == 0.0:
            continue
        lo = kk - (n - 1)
        if lo < 0:
            lo = 0
        hi = kk
        if hi > n - 1:
            hi = n - 1
        if lo <= hi:
            h[lo:hi + 1] += wk * a[kk - hi:kk - lo + 1][::-1]
    return h


def _smooth_value(a, mode, param, scale, active=None):
    """
    Value of the smooth objective:
      mode 0: log-sum-exp with beta=param
      mode 1: p-norm with p=param
      mode 2: sum of squares on active indices
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    c = _full_convolve(a, a)
    c = np.maximum(c, 0.0)
    mode = int(mode)
    scale = float(scale)

    if mode == 0:
        beta = float(param)
        cm = float(c.max())
        if beta <= 1e-300 or cm <= 0.0:
            return scale * cm
        x = (c - cm) / beta
        ex = np.exp(x)
        s = float(ex.sum())
        if s <= 1e-300 or not np.isfinite(s):
            return scale * cm
        return float(scale * (cm + beta * np.log(s)))

    if mode == 2:
        if active is None:
            idx = np.arange(c.size)
        else:
            idx = np.asarray(active, dtype=int).ravel()
            idx = idx[(idx >= 0) & (idx < c.size)]
        if idx.size == 0:
            return 0.0
        ca = c[idx]
        return float(scale * np.dot(ca, ca))

    # mode 1: p-norm
    p = float(param)
    if p <= 1.0:
        p = 1.0
    c_safe = np.maximum(c, 1e-300)
    logc = np.log(c_safe)
    maxlog = float(logc.max())
    scaled = p * (logc - maxlog)
    ex = np.exp(scaled)
    s = float(ex.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return 0.0
    logF = maxlog + np.log(s) / p
    if logF > 700.0:
        return np.inf
    return float(scale * np.exp(logF))


def _obj_grad(z, payload):
    """
    Objective and gradient for L-BFGS on log-softmax parameterization.

    payload = (mode, param, scale, active)
    """
    mode = int(payload[0])
    param = float(payload[1])
    scale = float(payload[2])
    active = payload[3]

    a = _softmax(z)
    n = int(a.size)
    c = _full_convolve(a, a)
    c = np.maximum(c, 0.0)

    if mode == 0:
        beta = param
        cm = float(c.max())
        if beta <= 1e-300 or cm <= 0.0:
            w = np.zeros(c.size, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
            obj = scale * cm
        else:
            x = (c - cm) / beta
            ex = np.exp(x)
            s = float(ex.sum())
            if s <= 1e-300 or not np.isfinite(s):
                w = np.zeros(c.size, dtype=np.float64)
                w[int(np.argmax(c))] = 1.0
                obj = scale * cm
            else:
                w = ex / s
                obj = scale * (cm + beta * np.log(s))
        factor = w

    elif mode == 2:
        if active is None:
            idx = np.arange(c.size)
        else:
            idx = np.asarray(active, dtype=int).ravel()
            idx = idx[(idx >= 0) & (idx < c.size)]
        factor = np.zeros(c.size, dtype=np.float64)
        if idx.size == 0:
            obj = 0.0
        else:
            ca = c[idx]
            factor[idx] = 2.0 * ca
            obj = scale * float(np.dot(ca, ca))

    else:
        p = param
        c_safe = np.maximum(c, 1e-300)
        logc = np.log(c_safe)
        maxlog = float(logc.max())
        scaled = p * (logc - maxlog)
        ex = np.exp(scaled)
        s = float(ex.sum())
        if s <= 1e-300 or not np.isfinite(s):
            factor = np.zeros(c.size, dtype=np.float64)
            factor[int(np.argmax(c))] = 1.0
            obj = scale * float(c.max())
        else:
            logF = maxlog + np.log(s) / p
            logfactor = (p - 1.0) * (logc - logF)
            logfactor = np.minimum(logfactor, 0.0)
            factor = np.exp(logfactor)
            s2 = float(factor.sum())
            if s2 <= 1e-300 or not np.isfinite(s2):
                factor = np.zeros(c.size, dtype=np.float64)
                factor[int(np.argmax(c))] = 1.0
            obj = scale * float(np.exp(logF)) if logF < 700.0 else np.inf

    h = _quad_conv_weight_a(factor, a)
    grad_a = 2.0 * h
    gz = a * (grad_a - a * np.dot(a, grad_a))
    gz = gz * scale

    if not np.all(np.isfinite(gz)):
        gz = np.zeros(n, dtype=np.float64)
    if not np.isfinite(obj):
        obj = 1e100
    return float(obj), gz


def _lbfgs_stage(a, deadline, mode, param, scale, maxiter, active=None):
    """
    Run one L-BFGS stage on the log-softmax variables.
    Returns (continuation_a, best_exact_a, best_exact_score).
    """
    a = _normalize(a)
    z = _log_from_a(a)

    best_a = a.copy()
    best_sc = _score_norm(a)
    cont_a = a.copy()

    if active is not None:
        active = tuple([int(i) for i in active])

    payload = (int(mode), float(param), float(scale), active)
    n = int(a.size)

    if a.size <= 6144:
        imax = int(min(110, maxiter))
    elif a.size <= 12288:
        imax = int(min(75, maxiter))
    else:
        imax = int(min(50, maxiter))

    try:
        val_init = _smooth_value(a, mode, param, scale, active)

        opts = {
            "maxiter": max(10, imax),
            "ftol": 1e-15,
            "gtol": 1e-8,
            "maxls": 8,
            "maxfun": int(max(100, imax * 5)),
        }

        res = minimize(
            _obj_grad,
            z,
            args=(payload,),
            jac=True,
            method="L-BFGS-B",
            options=opts,
        )

        if res.x is not None and np.all(np.isfinite(res.x)):
            a_final = _softmax(res.x)
            a_final = _normalize(a_final)
            sc_final = _score_norm(a_final)
            if np.isfinite(sc_final) and sc_final < best_sc:
                best_a = a_final.copy()
                best_sc = sc_final

            val_final = _smooth_value(a_final, mode, param, scale, active)
            if (
                np.isfinite(val_final)
                and (not np.isfinite(val_init) or val_final <= val_init)
            ):
                cont_a = a_final.copy()
            else:
                cont_a = a.copy()
    except Exception:
        cont_a = a.copy()

    return cont_a, best_a, float(best_sc)


def _p_continuation(a, deadline, n):
    """
    Smooth/active-set continuation.
    Returns (best_by_exact, best_score, current_a, current_score).
    """
    a = _normalize(a)
    cur = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)

    if n <= 6144:
        imax = 110
    elif n <= 12288:
        imax = 75
    else:
        imax = 50

    K1 = 32 if n <= 8192 else 16
    K2 = 16 if n <= 8192 else 8
    K3 = 8 if n <= 8192 else 4

    steps = [
        ("active", K1),
        ("p", 2.0),
        ("active", K2),
        ("p", 4.0),
        ("active", K3),
        ("p", 8.0),
        ("p", 16.0),
        ("lse", 0.08),
        ("lse", 0.03),
        ("lse", 0.01),
    ]

    last_time = time.monotonic()
    for step in steps:
        if time.monotonic() >= deadline - 0.5:
            break

        kind = step[0]
        active = None

        if kind == "active":
            K = int(step[1])
            c = _full_convolve(cur, cur)
            c = np.maximum(c, 0.0)
            K_eff = min(K, int(c.size))
            if K_eff == 0:
                continue
            if K_eff >= c.size:
                idx = np.arange(c.size)
            else:
                idx = np.argpartition(c, -K_eff)[-K_eff:]
            active = tuple([int(i) for i in idx])
            scale = float(n) * float(n)
            mode = 2
            param = 0.0

        elif kind == "p":
            pval = float(step[1])
            mode = 1
            param = pval
            scale = float(n) ** max(0.0, 1.0 - 1.0 / max(1.0, pval))

        else:
            # lse
            c = _full_convolve(cur, cur)
            cm = float(np.max(c))
            frac = float(step[1])
            mode = 0
            param = max(cm * frac, 1e-12)
            scale = 2.0 * float(n)

        cont, best_stage, best_stage_sc = _lbfgs_stage(
            cur, deadline, mode, param, scale, imax, active
        )

        if np.isfinite(best_stage_sc) and best_stage_sc < best_sc:
            best_a = best_stage.copy()
            best_sc = best_stage_sc

        if np.all(np.isfinite(cont)):
            cur = cont.copy()
            sc_cur = _score_norm(cur)
            if np.isfinite(sc_cur) and sc_cur < best_sc:
                best_a = cur.copy()
                best_sc = sc_cur

        # Guard: if a stage took too long, remaining steps will be skipped naturally.
        last_time = time.monotonic()

    cur_sc = _score_norm(cur)
    return best_a, float(best_sc), cur, float(cur_sc)


def _adam_mirror_subgrad(a, deadline, rng, topk=5, weight_mode=0, lr=0.1):
    """
    Adam mirror descent on log-variables using top-K subgradients of the exact max.
    """
    a = _normalize(a)
    n = int(a.size)
    z = _log_from_a(a)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    bias1 = 1.0
    bias2 = 1.0
    b1 = 0.90
    b2 = 0.999
    t = 0
    last_improve = 0
    stuck_limit = 4000
    lr_min = 0.01
    lr_decay = 0.9995

    c0 = _full_convolve(a, a)
    best_a = a.copy()
    best_sc = float(2.0 * n * np.max(c0))

    while time.monotonic() < deadline:
        a = _softmax(z)
        a = _normalize(a)

        c = _full_convolve(a, a)
        c = np.maximum(c, 0.0)
        cm = float(c.max())
        sc = float(2.0 * n * cm)

        if np.isfinite(sc) and sc < best_sc:
            best_a = a.copy()
            best_sc = sc
            last_improve = t

        if t - last_improve > stuck_limit:
            z = z + 0.05 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            m.fill(0.0)
            v.fill(0.0)
            bias1 = 1.0
            bias2 = 1.0
            t = 0
            last_improve = 0
            lr = max(lr, 0.05)

        h = _topk_direct_h(c, a, topk, weight_mode)
        grad_a = 4.0 * float(n) * h
        gz = a * (grad_a - a * np.dot(a, grad_a))

        if not np.all(np.isfinite(gz)):
            gz = np.zeros(n, dtype=np.float64)

        t += 1
        bias1 = 1.0 - b1 * bias1
        bias2 = 1.0 - b2 * bias2

        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)

        mh = m / bias1
        vh = v / bias2
        step = lr * mh / (np.sqrt(vh) + 1e-8)

        z = z - step
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if t % 2000 == 0:
            lr = max(lr_min, lr * lr_decay)

    return best_a, float(best_sc)


def _local_search(a, deadline, rng):
    """
    Targeted/random local search.
    """
    a = _normalize(a)
    n = int(a.size)
    if n == 0:
        return a, np.inf
    if n == 1:
        return a, _score_norm(a)

    best_a = a.copy()
    best_sc = _score_norm(a)

    c0 = _full_convolve(best_a, best_a)
    top_count = min(5, int(c0.size))
    if top_count >= c0.size:
        top_lags = np.arange(c0.size)
    else:
        top_lags = np.argpartition(c0, -top_count)[-top_count:]

    trials = 0
    while time.monotonic() < deadline:
        trials += 1

        a_try = None
        r = rng.random()

        if r < 0.55:
            # Targeted move around one of the largest lags.
            if trials % 50 == 0 or top_lags.size == 0:
                cc = _full_convolve(best_a, best_a)
                tc = min(5, int(cc.size))
                if tc >= cc.size:
                    top_lags = np.arange(cc.size)
                else:
                    top_lags = np.argpartition(cc, -tc)[-tc:]

            if top_lags.size == 0:
                continue
            k = int(rng.choice(top_lags))

            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if hi < lo:
                continue

            prod = best_a[lo:hi + 1] * best_a[k - hi:k - lo + 1][::-1]
            m = min(24, int(prod.size))
            if m <= 0:
                continue
            if m < prod.size:
                pi = np.argpartition(prod, -m)[-m:]
                vals = prod[pi]
            else:
                pi = np.arange(prod.size)
                vals = prod

            svals = float(vals.sum())
            if svals > 1e-300:
                p = np.maximum(vals, 0.0)
                p = p / p.sum()
                choice = int(rng.choice(len(pi), p=p))
            else:
                choice = int(rng.integers(0, len(pi)))
            i_local = int(pi[choice])
            i = lo + i_local

            j = int(rng.integers(0, n))
            if j == i:
                j = (j + 1) % n

            f = float(rng.uniform(0.02, 0.28))
            a_try = best_a.copy()
            transfer = f * a_try[i]
            a_try[i] -= transfer
            a_try[j] += transfer
            a_try = _normalize(a_try)

        else:
            # Random multiplicative block move.
            max_block = min(50, max(2, n // 10 + 2))
            block = int(rng.integers(2, max_block + 1))
            if block >= n:
                block = max(1, n)
            end = n - block + 1
            if end <= 1:
                start = 0
            else:
                start = int(rng.integers(0, end))
            a_try = best_a.copy()
            pert = rng.uniform(0.70, 1.30, size=block)
            a_try[start:start + block] *= pert
            a_try = _normalize(a_try)

        if a_try is None:
            continue

        sc = _score_norm(a_try)
        if np.isfinite(sc) and sc < best_sc:
            best_a = a_try.copy()
            best_sc = sc
            if trials % 25 == 0:
                cc = _full_convolve(best_a, best_a)
                tc = min(5, int(cc.size))
                if tc >= cc.size:
                    top_lags = np.arange(cc.size)
                else:
                    top_lags = np.argpartition(cc, -tc)[-tc:]

    return best_a, float(best_sc)


def _deconvolve_profile(a_init, deadline, iters_max=30, relax=0.25, tilt=0.0, lam_frac=0.02):
    """
    Newton-like deconvolution toward a flat (or gently tilted) target square.
    Useful as a diverse initializer; not guaranteed to improve from an already good point.
    """
    a = _normalize(a_init)
    n = int(a.size)
    if n == 0:
        return a, np.inf
    L = 2 * n - 1
    N = _next_pow2(L)

    t = np.linspace(0.0, 1.0, L)
    u = np.ones(L, dtype=np.float64)
    if abs(tilt) > 0.0:
        u = u * np.exp(float(tilt) * (t - 0.5))
    u = u / u.sum()

    best_a = a.copy()
    best_sc = _score_norm(a)

    for _ in range(int(max(1, iters_max))):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c = np.maximum(c, 0.0)
        cm = float(c.max())
        sc = float(2.0 * n * cm)
        if np.isfinite(sc) and sc < best_sc:
            best_a = a.copy()
            best_sc = sc

        residual = u - c
        residual -= residual.sum() / float(L)

        A = np.fft.rfft(a, N)
        R = np.fft.rfft(residual, N)
        absA = np.abs(A)
        maxabs = float(absA.max()) if absA.size else 1.0
        if maxabs <= 1e-300:
            break

        denom = absA * absA + float(lam_frac) * maxabs * maxabs
        D = 0.5 * np.conj(A) * R / denom
        D[absA < 0.01 * maxabs] = 0.0

        d = np.fft.irfft(D, N)[:n].real
        d -= d.mean()
        dmax = float(np.max(np.abs(d)))
        if dmax <= 1e-300:
            break

        accepted = False
        for frac in (1.0, 0.5, 0.25, 0.125):
            step = frac * float(relax) / dmax
            a_try = np.maximum(a + step * d, 0.0)
            a_try = _normalize(a_try)
            sc_try = _score_norm(a_try)
            if np.isfinite(sc_try) and sc_try < sc:
                a = a_try
                sc = sc_try
                accepted = True
                break

        # If completely stuck, try a tiny perturbation.
        if not accepted:
            d = rng_noise = None  # no rng here; deterministic break after few failures
            # Keep it simple: break if no smaller step accepts.
            if cm >= best_sc - 1e-15 and _ > 3:
                pass

    return best_a, float(best_sc)


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, int(n))
    a = 1.0 + float(slope) * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return _normalize(a)


def _make_beta_profile(n, alpha, beta):
    n = int(n)
    t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    a = np.exp((float(alpha) - 1.0) * np.log(t) + (float(beta) - 1.0) * np.log(1.0 - t))
    a = np.maximum(a, 1e-300)
    return _normalize(a)


def _make_exp_profile(n, rate):
    n = int(n)
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(float(rate) * (t - 0.5))
    return _normalize(a)


def _make_gauss_profile(n, center, width):
    n = int(n)
    t = np.linspace(0.0, 1.0, n)
    u = t - float(center)
    a = np.exp(-(u * u) / (2.0 * float(width) * float(width)))
    return _normalize(a)


def _interp_profile(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = int(a.size)
    n_new = int(n_new)
    if n0 == 0:
        return _make_linear_profile(n_new, 1.2)
    if n0 == n_new:
        return _normalize(a)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    return _normalize(p)


def _random_profile_start(n, rng, trials, deadline):
    n = int(n)
    best_a = None
    best_sc = np.inf

    for _ in range(int(max(1, trials))):
        if time.monotonic() > deadline:
            break
        r = float(rng.random())
        if r < 0.40:
            alpha = float(rng.uniform(0.5, 4.0))
            beta = float(rng.uniform(0.5, 4.0))
            a = _make_beta_profile(n, alpha, beta)
        elif r < 0.70:
            rate = float(rng.uniform(-3.0, 3.0))
            a = _make_exp_profile(n, rate)
        else:
            center = float(rng.uniform(0.20, 0.80))
            width = float(rng.uniform(0.10, 0.45))
            a = _make_gauss_profile(n, center, width)

        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_sc:
            best_a = a
            best_sc = sc

    if best_a is None:
        best_a = _make_beta_profile(n, 1.5, 0.8)
        best_sc = _score_norm(best_a)
    return best_a, float(best_sc)


def _load_parent():
    obj = globals().get("height_sequence_1", None)
    if obj is None:
        return None
    try:
        arr = np.asarray(obj, dtype=np.float64).ravel()
        if arr.size == 0:
            return None
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(arr.sum())
        if s <= 1e-300:
            return None
        return arr / s
    except Exception:
        return None


def _polish_sequence(a, deadline, rng):
    """
    Run a local polishing pipeline on one candidate sequence.
    """
    a = _normalize(a)
    n = int(a.size)
    if n == 0:
        return a, np.inf

    if time.monotonic() >= deadline:
        return a, _score_norm(a)

    t0 = time.monotonic()
    allowed = float(max(0.0, deadline - t0))
    if allowed < 5.0:
        return a, _score_norm(a)

    best_a = a.copy()
    best_sc = _score_norm(a)

    # 1) Smooth/active L-BFGS continuation.
    p_limit = min(allowed * 0.35, 240.0)
    p_deadline = min(float(deadline), t0 + p_limit)
    if p_deadline > t0 + 5.0:
        best_p, sc_p, cur_p, sc_cur = _p_continuation(best_a, p_deadline, n)
        if np.isfinite(sc_p) and sc_p < best_sc:
            best_a = best_p.copy()
            best_sc = sc_p
        if np.isfinite(sc_cur) and sc_cur < best_sc:
            best_a = cur_p.copy()
            best_sc = sc_cur

    # 2) Nonsmooth Adam subgradient.
    t_adam_start = time.monotonic()
    adam_allowed = float(max(0.0, deadline - t_adam_start))
    adam_deadline = min(float(deadline), t_adam_start + max(0.0, adam_allowed * 0.60))

    if n <= 6144:
        configs = [
            (32, 0, 0.10, 0.15),
            (16, 1, 0.08, 0.25),
            (8, 0, 0.06, 0.25),
            (4, 1, 0.04, 0.20),
            (2, 0, 0.02, 0.15),
        ]
    elif n <= 12288:
        configs = [
            (16, 1, 0.08, 0.25),
            (8, 0, 0.06, 0.30),
            (4, 1, 0.04, 0.25),
            (2, 0, 0.02, 0.20),
        ]
    else:
        configs = [
            (8, 1, 0.06, 0.35),
            (4, 0, 0.04, 0.35),
            (2, 0, 0.02, 0.30),
        ]

    adam_total = float(max(0.0, adam_deadline - t_adam_start))
    cur_time = t_adam_start
    if adam_total > 5.0:
        for cfg in configs:
            if time.monotonic() >= adam_deadline - 0.5:
                break
            if cur_time < time.monotonic():
                cur_time = time.monotonic()
            dl = min(adam_deadline, cur_time + float(cfg[3]) * adam_total)
            if dl <= time.monotonic():
                dl = min(adam_deadline, time.monotonic() + 2.0)
            if dl > time.monotonic():
                a_try, sc_try = _adam_mirror_subgrad(
                    best_a,
                    dl,
                    rng,
                    topk=int(cfg[0]),
                    weight_mode=int(cfg[1]),
                    lr=float(cfg[2]),
                )
                if np.isfinite(sc_try) and sc_try < best_sc:
                    best_a = a_try.copy()
                    best_sc = sc_try
            cur_time = time.monotonic()

    # 3) Targeted local search.
    if time.monotonic() < deadline - 2.0:
        a_try, sc_try = _local_search(best_a, float(deadline - 2.0), rng)
        if np.isfinite(sc_try) and sc_try < best_sc:
            best_a = a_try.copy()
            best_sc = sc_try

    return best_a, float(best_sc)


def propose_candidate(seed=42, budget_s=1800):
    """
    Entry point. Uses the 1800 second budget to search for a good nonnegative sequence.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    parent = _load_parent()
    if parent is not None:
        best_a = parent
        best_sc = _score_norm(best_a)
        print(f"Parent loaded: n={int(best_a.size)}, score={best_sc:.10f}")
    else:
        best_a = _make_linear_profile(4096, 1.5)
        best_sc = _score_norm(best_a)
        print(f"No parent; using linear profile: n={int(best_a.size)}, score={best_sc:.10f}")

    # 1) Quick polish of the parent length.
    if time.monotonic() < deadline - 350.0:
        dl = min(float(deadline - 350.0), t_start + 120.0)
        a_try, sc_try = _polish_sequence(best_a, dl, rng)
        if np.isfinite(sc_try) and sc_try < best_sc:
            best_a = a_try.copy()
            best_sc = sc_try
            print(f"Parent polish improved: n={int(best_a.size)}, score={best_sc:.10f}")

    # 2) Main search at n=8192 (a sweet spot for speed / quality).
    n_main = 8192
    if int(best_a.size) != n_main:
        a_main = _interp_profile(best_a, n_main)
    else:
        a_main = best_a.copy()

    # Quick deconvolution initialization at main length.
    if time.monotonic() < deadline - 500.0:
        dl = min(float(deadline - 500.0), time.monotonic() + 30.0)
        a_dec, sc_dec = _deconvolve_profile(a_main, dl, iters_max=25, relax=0.20, tilt=0.0)
        if np.isfinite(sc_dec) and sc_dec < best_sc:
            best_a = a_dec.copy()
            best_sc = sc_dec
            print(f"Deconvolution improved global best: n={int(best_a.size)}, score={best_sc:.10f}")
        sc_main_now = _score_norm(a_main)
        if np.isfinite(sc_dec) and sc_dec < sc_main_now:
            a_main = a_dec.copy()

    remaining_after_base = float(max(0.0, deadline - time.monotonic()))
    main_limit = min(900.0, max(60.0, remaining_after_base - 350.0))
    if main_limit > 30.0 and time.monotonic() < deadline - 300.0:
        dl = min(float(deadline - 300.0), time.monotonic() + main_limit)
        a_try, sc_try = _polish_sequence(a_main, dl, rng)
        if np.isfinite(sc_try) and sc_try < best_sc:
            best_a = a_try.copy()
            best_sc = sc_try
            print(f"Main n={n_main} polish improved: score={best_sc:.10f}")

    # 3) A few diverse starts at main length.
    alt_end = min(float(deadline - 200.0), time.monotonic() + 250.0)
    if values:  # placeholder removed below
        pass

    if alt_end > time.monotonic() + 60.0:
        starts = [
            _make_beta_profile(n_main, 1.7, 0.8),
            _make_beta_profile(n_main, 0.8, 1.7),
            _make_exp_profile(n_main, 1.8),
            _make_exp_profile(n_main, -1.8),
        ]

        if time.monotonic() < alt_end - 60.0:
            a_rand, sc_rand = _random_profile_start(
                n_main,
                rng,
                300,
                min(float(alt_end - 60.0), time.monotonic() + 15.0),
            )
            if a_rand is not None:
                starts.append(a_rand)

        for a_start in starts:
            if time.monotonic() >= alt_end - 30.0:
                break
            dl = min(float(alt_end), time.monotonic() + 70.0)
            a_try, sc_try = _polish_sequence(a_start, dl, rng)
            if np.isfinite(sc_try) and sc_try < best_sc:
                best_a = a_try.copy()
                best_sc = sc_try
                print(f"Alternative start improved: n={int(best_a.size)}, score={best_sc:.10f}")

    # 4) Try a larger length, usually 16384.
    if int(best_a.size) < 16384 and time.monotonic() < deadline - 180.0:
        n_large = 16384
        a_large = _interp_profile(best_a, n_large)
        dl = min(float(deadline - 100.0), time.monotonic() + 250.0)
        a_try, sc_try = _polish_sequence(a_large, dl, rng)
        if np.isfinite(sc_try) and sc_try < best_sc:
            best_a = a_try.copy()
            best_sc = sc_try
            print(f"Large n={n_large} improved: score={best_sc:.10f}")

    # 5) Final polish on the current global best.
    if time.monotonic() < deadline - 15.0:
        dl = float(deadline - 15.0)
        a_try, sc_try = _polish_sequence(best_a, dl, rng)
        if np.isfinite(sc_try) and sc_try < best_sc:
            best_a = a_try.copy()
            best_sc = sc_try
            print(f"Final polish improved: n={int(best_a.size)}, score={best_sc:.10f}")

    # Final safety normalization and validation.
    final_a = _normalize(best_a)
    final_sc = _score_norm(final_a)
    if not np.isfinite(final_sc):
        final_a = _make_linear_profile(4096, 1.5)
        final_sc = _score_norm(final_a)

    print(f"FINAL FFT score: {final_sc:.10f}, n={int(final_a.size)}, sum={float(final_a.sum()):.10f}")

    eval_fn = globals().get("evaluate_sequence", None)
    if eval_fn is not None and int(final_a.size) <= 12000:
        try:
            eval_sc = float(eval_fn([float(x) for x in final_a]))
            if np.isfinite(eval_sc):
                print(f"Evaluate_sequence score: {eval_sc:.10f}")
            else:
                print(f"Evaluate_sequence returned non-finite: {eval_sc}")
        except Exception as ex:
            print(f"Failed to call evaluate_sequence in self-test: {ex}")

    return [float(x) for x in final_a]
```
