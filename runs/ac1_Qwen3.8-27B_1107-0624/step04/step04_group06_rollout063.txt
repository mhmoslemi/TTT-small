```python
"""
Search and polish nonnegative step-function heights for the objective

    score(a) = 2 * n * max_k (a * a)_k / (sum(a)^2)

The program combines:
  - frequency-domain smooth-max gradient descent on log ratios (Adam logistics),
  - L-BFGS polishing of log-sum-exp approximations to the peak,
  - active-constraint LP directions (max-sum and minimax recall-min),
  - multi-resolution interpolation and perturbations across sequence lengths,
  - final low-temperature polishing.

It always returns the best normalized nonnegative sequence found before the
deadline.
"""

import time

import numpy as np
from scipy.optimize import linprog, minimize


# -----------------------------
# Basic utilities
# -----------------------------

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)

    if n <= 128:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)

    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score_norm(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-300:
        return np.inf
    c = _full_convolve(p)
    if c.size == 0:
        return np.inf
    return float((2.0 * n * float(np.max(c))) / (s * s))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _softmax_from_z(z, floor=1e-15):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.array([], dtype=np.float64)
    if not np.all(np.isfinite(z)):
        z = np.nan_to_num(z, nan=0.0, posinf=50.0, neginf=-50.0)
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e, dtype=np.float64) / float(e.size)
    else:
        p = p / s
    return p


def _p_to_z(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    p = np.maximum(p, 1e-30)
    z = np.log(p)
    z -= float(np.mean(z))
    return z


def _cand_key(item):
    return item[0]


def _candidate_update(p, best_seq, best_score):
    if p is None:
        return best_seq, best_score
    pn = _normalize(p)
    if pn is None:
        return best_seq, best_score
    sc = _score_norm(pn)
    if np.isfinite(sc) and sc < best_score:
        return pn.copy(), float(sc)
    return best_seq, best_score


def _update_and_print(p, best_seq, best_score, label):
    old = best_score
    best_seq, best_score = _candidate_update(p, best_seq, best_score)
    if best_score < old and best_seq is not None:
        print(f"  improved [{label}]: score={best_score:.10f} n={best_seq.size}")
    return best_seq, best_score

def _profile_linear(n, eta=2.0, shift=0.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t + shift)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _profile_exp(n, gamma=2.0, reverse=False):
    t = np.linspace(0.0, 1.0, n)
    if reverse:
        t = 1.0 - t
    a = np.exp(gamma * t)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _profile_inv(n, beta=3.0, reverse=False):
    t = np.linspace(0.0, 1.0, n)
    if reverse:
        t = 1.0 - t
    a = 1.0 / (1.0 + beta * t)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _profile_cospi(n, amp=1.0):
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(np.pi * x)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _profile_cubic(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u * u * u
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == 0:
        return None
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, np.maximum(p0, 0.0))
    return _normalize(p1)


def _perturb_profile(p, amp, rng, window=15):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n < 12:
        return _normalize(p)
    window = int(max(3, min(window, max(3, n // 256))))
    z = _p_to_z(p)
    noise = rng.normal(size=n)
    kern = np.ones(window, dtype=np.float64) / float(window)
    for _ in range(3):
        noise = np.convolve(noise, kern, mode="same")
    std = float(np.std(noise))
    if std > 1e-12:
        noise = (noise - float(np.mean(noise))) / std
    else:
        noise = np.zeros_like(noise)
    z = z + amp * noise
    return _softmax_from_z(z)

def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([], dtype=np.float64)
    mval = float(c.max())
    beta = float(beta)

    if not np.isfinite(beta) or beta <= 1e-15:
        tol = max(1e-12, 1e-7 * mval)
        mask = c >= (mval - tol)
        cnt = int(mask.sum())
        w = np.zeros_like(c, dtype=np.float64)
        if cnt <= 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(cnt)
        return mval, w

    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    """
    Returns g_j = sum_i p_i * w_{i+j} for j=0,...,n-1.
    This is half of the derivative of sum_k w_k conv(p,p)_k w.r.t. p_j.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _smooth_obj_grad(z, beta, n):
    if z.size != n:
        z = np.resize(z, n)
    if not np.all(np.isfinite(z)):
        return 1e100, np.zeros(n, dtype=np.float64)

    p = _softmax_from_z(z)
    c = _full_convolve(p)
    if c.size == 0:
        return 1e100, np.zeros(n, dtype=np.float64)

    val, w = _smooth_max_weights(c, beta)
    score = 2.0 * float(n) * val
    heat = _conv_cross_corr(w, p)
    dot = float(np.dot(p, heat))
    grad_z = (4.0 * float(n)) * p * (heat - dot)

    if not np.isfinite(score):
        return 1e100, np.zeros(n, dtype=np.float64)
    if not np.all(np.isfinite(grad_z)):
        grad_z = np.nan_to_num(grad_z, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad_z

def _lbfgs_smooth(p0, deadline, rng=None, betas=None):
    p0 = _normalize(p0)
    if p0 is None:
        return None, np.inf

    n = p0.size
    best = p0.copy()
    best_score = _score_norm(best)
    best_z = _p_to_z(best)

    c = _full_convolve(best)
    M = float(c.max()) if c.size else 1e-8
    if not np.isfinite(M) or M <= 0.0:
        M = 1e-8

    if betas is None:
        betas = [0.30 * M, 0.20 * M, 0.10 * M, 0.05 * M, 0.025 * M, 0.012 * M, 0.006 * M]
    else:
        betas = [max(float(x), 1e-15) for x in betas]

    for beta in betas:
        if time.monotonic() > deadline - 10.0:
            break

        z0 = best_z.copy()
        if rng is not None:
            # Very small perturbations can avoid exact repeats in repeated calls.
            z0 = z0 + 0.02 * rng.normal(size=n)

        rem = deadline - time.monotonic()
        if rem <= 15.0:
            break

        if rem > 90.0:
            maxiter = 220
            maxfun = 1500
        elif rem > 40.0:
            maxiter = 120
            maxfun = 700
        else:
            maxiter = 60
            maxfun = 300

        try:
            res = minimize(
                _smooth_obj_grad,
                z0,
                args=(beta, n),
                method="L-BFGS-B",
                jac=True,
                options={
                    "maxiter": maxiter,
                    "maxfun": maxfun,
                    "ftol": 1e-14,
                    "gtol": 1e-8,
                    "maxls": 10,
                    "disp": False,
                },
            )
            if res is not None and res.x is not None and np.all(np.isfinite(res.x)):
                p_cand = _softmax_from_z(res.x)
                sc_cand = _score_norm(p_cand)
                if np.isfinite(sc_cand) and sc_cand < best_score - 1e-14:
                    best = p_cand.copy()
                    best_score = float(sc_cand)
                    best_z = _p_to_z(best)
        except Exception:
            pass

    return best, best_score


def _adam_logit(p0, deadline, rng, lr=0.05, beta_frac=0.08, beta_decay=0.93,
                no_imp_limit=250, noise_scale=None):
    p0 = _normalize(p0)
    if p0 is None:
        return None, np.inf
    if rng is None:
        rng = np.random.default_rng()

    n = p0.size
    p = p0.copy()
    z = _p_to_z(p)
    best = p.copy()
    best_score = _score_norm(p)
    best_z = z.copy()

    c = _full_convolve(p)
    M = float(c.max()) if c.size else 1e-8
    if not np.isfinite(M) or M <= 0.0:
        M = 1e-8

    if beta_frac is None:
        beta_frac = 0.08
    beta = 0.0 if float(beta_frac) <= 0.0 else max(1e-14, float(beta_frac) * M)

    b1, b2, eps = 0.90, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t_adam = 0
    base_lr = float(lr) if base_lr is not None and np.isfinite(base_lr) else 0.05
    vr = float(lr) if lr is not None and np.isfinite(lr) else 0.05
    no_imp = 0

    def _auto_noise():
        nonlocal best_score, n
        if noise_scale is not None:
            return float(noise_scale)
        if best_score < 1.51:
            ns = 0.07
        elif best_score < 1.53:
            ns = 0.12
        else:
            ns = 0.22
        ns *= float(np.sqrt(1024.0 / max(n, 128)))**0.5
        return ns

    while time.monotonic() < deadline:
        val, w = _smooth_max_weights(c, beta)
        heat = _conv_cross_corr(w, p)
        dot = float(np.dot(p, heat))
        grad_z = p * (heat - dot)

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mhat = m / (1.0 - b1 ** t_adam)
        vhat = v / (1.0 - b2 ** t_adam)
        z = z - vr * mhat / (np.sqrt(vhat) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)

        c = _full_convolve(p)
        maxc = float(c.max()) if c.size else np.inf
        score = 2.0 * float(n) * maxc

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = float(score)
                best = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    vr = max(0.002, vr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if beta > 0.0:
            beta = beta * beta_decay
            if beta < 1e-3 * maxc:
                beta = 0.0

        limit = max(60, int(no_imp_limit))
        if no_imp > limit:
            ns = _auto_noise()
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max()) if c.size else 1e-8
            beta = 0.0 if float(beta_frac) <= 0.0 else max(1e-14, float(beta_frac) * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            vr = base_lr
            no_imp = 0

    return best, best_score

def _prepare_active(n, active):
    js_list = []
    ids_list = []
    for k in active:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        js_list.append(js)
        ids_list.append(k - np.arange(js, je + 1, dtype=np.int64))
    return js_list, ids_list


def _lp_maxsum_dense(A, b_ub, time_limit=None):
    if A.size == 0:
        return None, 0.0
    n = A.shape[1]
    obj = -np.ones(n, dtype=np.float64)
    b_ub = np.asarray(b_ub, dtype=np.float64)

    opts = {"presolve": True}
    if time_limit is not None and time_limit > 0.0:
        opts["time_limit"] = float(min(max(float(time_limit), 0.1), 60.0))

    try:
        res = linprog(
            c=obj,
            A_ub=A,
            b_ub=b_ub,
            bounds=(0.0, None),
            method="highs",
            options=opts,
        )
    except Exception:
        return None, 0.0

    if res is None or res.x is None or not getattr(res, "success", False):
        return None, 0.0

    b = np.asarray(res.x, dtype=np.float64)
    b = np.maximum(b, 0.0)
    b = np.nan_to_num(b, nan=0.0, posinf=0.0, neginf=0.0)
    s = float(b.sum())
    if s <= 1e-12:
        return None, 0.0
    return b / s, s


def _lp_minu_dense(A, time_limit=None):
    if A.size == 0:
        return None, np.inf
    m, n = A.shape

    A_full = np.empty((m, n + 1), dtype=np.float64)
    A_full[:, :n] = A
    A_full[:, n] = -1.0

    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0

    b_ub = np.zeros(m, dtype=np.float64)
    A_eq = np.ones((1, n), dtype=np.float64)
    b_eq = np.ones(1, dtype=np.float64)

    bounds = [(0.0, None)] * n + [(0.0, None)]

    opts = {"presolve": True}
    if time_limit is not None and time_limit > 0.0:
        opts["time_limit"] = float(min(max(float(time_limit), 0.1), 60.0))

    try:
        res = linprog(
            c=c_obj,
            A_ub=A_full,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options=opts,
        )
    except Exception:
        return None, np.inf

    if res is None or res.x is None or not getattr(res, "success", False):
        return None, np.inf

    b = np.asarray(res.x[:n], dtype=np.float64)
    b = np.maximum(b, 0.0)
    b = np.nan_to_num(b, nan=0.0, posinf=0.0, neginf=0.0)
    s = float(b.sum())
    if s <= 1e-12:
        return None, np.inf
    uval = float(res.x[n]) if np.isfinite(res.x[n]) else np.inf
    return b / s, uval


def _lp_refine(a, deadline, max_iter=80, K=0, t0=0.4, scale=10000.0, rebuild_every=6):
    a = _normalize(a)
    if a is None:
        return None, np.inf

    n = a.size
    if n > 3000:
        return a, _score_norm(a)

    if K <= 0:
        if n <= 800:
            K = 2 * n - 1
        elif n <= 1400:
            K = 2 * n - 1
        elif n <= 2200:
            K = 1600
        else:
            K = 1000
    K = int(max(1, min(K, 2 * n - 1)))

    best = a.copy()
    best_score = _score_norm(best)
    c = _full_convolve(a)
    M = float(c.max()) if c.size else _score_norm(a) / (2.0 * n)
    if not np.isfinite(M) or M <= 0.0:
        M = best_score / (2.0 * n)

    active = np.argsort(c)[::-1][:K]
    js_list, ids_list = _prepare_active(n, active)
    A = np.zeros((K, n), dtype=np.float64)

    it = 0
    no_imp = 0
    step = float(t0)
    scale = float(scale)

    while time.monotonic() < deadline and it < max_iter:
        it += 1

        a_work = a * scale
        A.fill(0.0)
        for r in range(K):
            js = js_list[r]
            ids = ids_list[r]
            A[r, js:js + ids.size] = a_work[ids]

        rem = deadline - time.monotonic()
        if rem < 1.0:
            break

        tl = max(0.5, min(15.0, 0.12 * rem))

        use_minu = (no_imp > 3) and (it % 3 == 0)
        g = None

        if use_minu:
            g, _ = _lp_minu_dense(A, tl)

        if g is None:
            delta = 0.002 + 0.08 * min(1.0, no_imp / 20.0)
            target = M * (1.0 - delta)
            if not np.isfinite(target) or target <= 0.0:
                target = M
            g, raw = _lp_maxsum_dense(A, target * scale, tl)

            if g is None or raw < 0.05:
                g, raw = _lp_maxsum_dense(A, M * scale, max(tl, 2.0))

            if g is None:
                g, _ = _lp_minu_dense(A, max(tl, 2.0))

        if g is None:
            break

        cur_score = _score_norm(a)
        best_cand = a
        best_cand_score = cur_score

        steps = (step, 0.1, 0.2, 0.4, 0.6, 1.0)
        for s in steps:
            if s <= 0.0 or s > 1.0:
                continue
            cand = (1.0 - s) * a + s * g
            cand = _normalize(cand)
            if cand is None:
                continue
            sc = _score_norm(cand)
            if np.isfinite(sc) and sc < best_cand_score:
                best_cand_score = float(sc)
                best_cand = cand

        if best_cand_score < cur_score - 1e-14:
            a = best_cand.copy()
            if best_cand_score < best_score - 1e-14:
                best_score = float(best_cand_score)
                best = a.copy()
                no_imp = 0
            else:
                no_imp += 1
            step = max(0.08, step * 0.95)
        else:
            no_imp += 1
            step = max(0.08, step * 0.90)

        c = _full_convolve(a)
        M = float(c.max()) if c.size else np.inf
        if not np.isfinite(M) or M <= 0.0:
            M = best_score / (2.0 * n)

        if no_imp > rebuild_every or it % max(6, rebuild_every * 2) == 0:
            active = np.argsort(c)[::-1][:K]
            js_list, ids_list = _prepare_active(n, active)

        if no_imp > 80:
            break

    return best, best_score

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    budget = max(30.0, float(budget_s) - 25.0)
    deadline = t_start + budget

    rng = np.random.default_rng(int(seed) % (2**31 - 1))

    best_seq = None
    best_score = np.inf

    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if arr.size > 0:
            arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
            arr = np.clip(arr, 0.0, 1000.0)
            sass = float(arr.sum())
            if np.isfinite(sass) and sass > 0.01:
                parent = arr / sass
    except Exception:
        parent = None

    print(f"Budget: {budget:.1f}s")

    if parent is not None:
        best_seq, best_score = _update_and_print(parent, best_seq, best_score, "parent")
    else:
        print("No parent supplied; using fallback profile search.")

    if best_seq is None:
        init_items = [
            _profile_linear(512, 2.0),
            _profile_linear(1024, 2.0),
            _profile_linear(1024, 4.0),
            _profile_cospi(512, 1.0),
            _profile_cospi(1024, 1.0),
            _profile_exp(512, 2.0),
            _profile_exp(1024, 2.0),
            _profile_inv(1024, 3.0),
        ]
        for p in init_items:
            best_seq, best_score = _candidate_update(p, best_seq, best_score)

    if best_seq is None:
        best_seq = _profile_linear(1024, 2.0)
        best_score = _score_norm(best_seq)

    print(f"Initial best: score={best_score:.10f} n={best_seq.size}")

    # Time chunks, scaled to requested budget.
    A_end = t_start + min(200.0, 0.12 * budget)
    B_end = t_start + min(520.0, 0.30 * budget)
    C_end = t_start + min(1150.0, 0.66 * budget)

    # ---------------- Phase A: polish current best ----------------
    if time.monotonic() < A_end - 20.0:
        n = best_seq.size

        if n <= 4096:
            end_lb = min(A_end, t_start + (90.0 if n <= 2048 else 60.0))
            if time.monotonic() < end_lb - 20.0:
                p, sc = _lbfgs_smooth(best_seq, end_lb, rng)
                best_seq, best_score = _update_and_print(p, best_seq, best_score, "phaseA_lbfgs")

        if time.monotonic() < A_end - 15.0:
            n = best_seq.size
            lr = 0.06 * float((1024.0 / max(n, 128)) ** 0.25)
            p, sc = _adam_logit(best_seq, A_end, rng, lr=lr, beta_frac=0.08,
                                beta_decay=0.93, no_imp_limit=250)
            best_seq, best_score = _update_and_print(p, best_seq, best_score, "phaseA_adam")

    # ---------------- Phase B: LP if medium size ----------------
    if time.monotonic() < B_end - 25.0:
        n = best_seq.size
        if n <= 2048:
            end_lp = min(B_end, time.monotonic() + 210.0)
            if time.monotonic() < end_lp - 20.0:
                p, sc = _lp_refine(best_seq, end_lp, max_iter=90, K=0, t0=0.45,
                                   scale=10000.0, rebuild_every=6)
                best_seq, best_score = _update_and_print(p, best_seq, best_score, "phaseB_lp")

            if time.monotonic() < B_end - 20.0:
                n = best_seq.size
                lr = 0.025 * float((1024.0 / max(n, 128)) ** 0.25)
                p, sc = _adam_logit(best_seq, B_end, rng, lr=lr, beta_frac=0.05,
                                    beta_decay=0.95, no_imp_limit=120)
                best_seq, best_score = _update_and_print(p, best_seq, best_score, "phaseB_adam")
        else:
            n = best_seq.size
            lr = 0.035 * float((1024.0 / max(n, 128)) ** 0.25)
            p, sc = _adam_logit(best_seq, B_end, rng, lr=lr, beta_frac=0.06,
                                beta_decay=0.94, no_imp_limit=200)
            best_seq, best_score = _update_and_print(p, best_seq, best_score, "phaseB_adam_large")

    # ---------------- Phase C: multi-resolution scan ----------------
    if time.monotonic() < C_end - 30.0:
        scan_ns = [128, 256, 512, 1024, 2048, 4096, 8192]
        candidates = []

        for n_try in scan_ns:
            p_base = _interp_profile(best_seq, n_try)
            if p_base is None:
                continue
            sc_base = _score_norm(p_base)

            p_pert = _perturb_profile(p_base, 0.08, rng, window=max(3, n_try // 256))
            sc_pert = _score_norm(p_pert)

            if sc_pert < sc_base:
                p_base, sc_base = p_pert, sc_pert

            candidates.append((float(sc_base), int(n_try), p_base))

        candidates.sort(key=_cand_key)
        top = candidates[:3]

        print("Multi-resolution scan top:", [(round(x[0], 10), x[1]) for x in top])

        for idx, item in enumerate(top):
            if time.monotonic() > C_end - 35.0:
                break

            _, n_try, p_init = item
            best_seq, best_score = _update_and_print(p_init, best_seq, best_score, f"scan_n{n_try}")

            lr = 0.06 * float((1024.0 / max(n_try, 128)) ** 0.25)

            if n_try <= 2048 and time.monotonic() < C_end - 90.0:
                end_lb = min(C_end - 45.0, time.monotonic() + 30.0)
                if end_lb > time.monotonic() + 15.0:
                    p, sc = _lbfgs_smooth(p_init, end_lb, rng)
                    best_seq, best_score = _update_and_print(p, best_seq, best_score, f"scan_n{n_try}_lbfgs")

            remaining_phase = C_end - time.monotonic()
            left_candidates = max(1, len(top) - idx)
            per = max(30.0, min(200.0, remaining_phase / float(left_candidates) - 20.0))

            if per > 25.0:
                p, sc = _adam_logit(p_init, time.monotonic() + per, rng, lr=lr,
                                    beta_frac=0.10, beta_decay=0.93, no_imp_limit=220)
                best_seq, best_score = _update_and_print(p, best_seq, best_score, f"scan_n{n_try}_adam")

            if best_seq.size <= 1024 and time.monotonic() < C_end - 35.0:
                per_lp = min(45.0, C_end - time.monotonic() - 12.0)
                if per_lp > 18.0:
                    p, sc = _lp_refine(best_seq, time.monotonic() + per_lp, max_iter=18, K=0, t0=0.35)
                    best_seq, best_score = _update_and_print(p, best_seq, best_score, f"scan_n{n_try}_lp")

    # ---------------- Phase D: alternative asymmetric initializers ----------------
    alt_end = min(deadline - 250.0, t_start + 0.80 * budget)
    if time.monotonic() < alt_end - 30.0:
        alt_ns = [512, 1024]
        alt_profiles = [
            (_profile_linear(512, 5.0), "lin5_n512"),
            (_profile_exp(1024, 3.0, reverse=True), "exp3rev_n1024"),
            (_profile_inv(1024, 5.0), "inv5_n1024"),
            (_profile_cubic(512, 2.2, -1.8), "cubic_n512"),
            (_profile_cospi(1024, 0.9), "cospi09_n1024"),
            (_profile_exp(2048, 2.2), "exp22_n2048"),
        ]

        # Filter profiles by target n to keep list reasonable.
        filtered = []
        for n_use in alt_ns:
            for p, lab in alt_profiles:
                if p.size == n_use:
                    filtered.append((p, lab))

        if not filtered:
            filtered = [(_profile_linear(1024, 3.0), "lin3_n1024")]

        per = max(8.0, min(22.0, (alt_end - time.monotonic()) / max(1, len(filtered))))

        for p, lab in filtered:
            if time.monotonic() > alt_end - 10.0:
                break
            best_seq, best_score = _update_and_print(p, best_seq, best_score, "alt_" + lab)
            lr = 0.05 * float((1024.0 / max(p.size, 128)) ** 0.25)
            p2, sc2 = _adam_logit(p, time.monotonic() + per, rng, lr=lr,
                                  beta_frac=0.12, beta_decay=0.93, no_imp_limit=80)
            best_seq, best_score = _update_and_print(p2, best_seq, best_score, "altAdam_" + lab)

    # ---------------- Phase E: final intensive polish ----------------
    cycles = 0
    while time.monotonic() < deadline - 55.0 and cycles < 5:
        cycles += 1
        n = best_seq.size

        if n <= 2048:
            per_lp = min(80.0, deadline - time.monotonic() - 70.0)
            if per_lp > 30.0:
                p, sc = _lp_refine(best_seq, time.monotonic() + per_lp, max_iter=22,
                                   t0=0.30, scale=10000.0, rebuild_every=5)
                best_seq, best_score = _update_and_print(p, best_seq, best_score,
                                                         f"finalLP_cyc{cycles}")

        if time.monotonic() < deadline - 90.0:
            n = best_seq.size
            lr = 0.02 * float((1024.0 / max(n, 128)) ** 0.25)
            per = min(170.0, deadline - time.monotonic() - 70.0)
            if per > 35.0:
                p, sc = _adam_logit(best_seq, time.monotonic() + per, rng, lr=lr,
                                    beta_frac=0.045, beta_decay=0.95, no_imp_limit=130,
                                    noise_scale=0.06)
                best_seq, best_score = _update_and_print(p, best_seq, best_score,
                                                         f"finalAdam_cyc{cycles}")

        if best_seq.size <= 4096 and time.monotonic() < deadline - 70.0:
            per = min(70.0, deadline - time.monotonic() - 55.0)
            if per > 25.0:
                p, sc = _lbfgs_smooth(best_seq, time.monotonic() + per, rng)
                best_seq, best_score = _update_and_print(p, best_seq, best_score,
                                                         f"finalLBFGS_cyc{cycles}")

        if time.monotonic() < deadline - 45.0:
            n = best_seq.size
            lr = 0.008 * float((1024.0 / max(n, 128)) ** 0.25)
            per = min(90.0, deadline - time.monotonic() - 30.0)
            if per > 25.0:
                p, sc = _adam_logit(best_seq, time.monotonic() + per, rng, lr=lr,
                                    beta_frac=0.0, beta_decay=1.0, no_imp_limit=80,
                                    noise_scale=0.035)
                best_seq, best_score = _update_and_print(p, best_seq, best_score,
                                                         f"finalExact_cyc{cycles}")

    # ---------------- Final validation and return ----------------
    final = _normalize(best_seq)
    if final is None:
        final = _profile_linear(1024, 2.0)

    final = np.nan_to_num(final, nan=0.0, posinf=1000.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    s = float(final.sum())
    if not np.isfinite(s) or s <= 0.01:
        final = _profile_linear(1024, 2.0)
    else:
        final = final / s

    internal_score = _score_norm(final)
    print(f"FINAL internal estimate: score={internal_score:.10f} n={final.size}")

    try:
        ext = evaluate_sequence([float(x) for x in final])
        print(f"FINAL evaluate_sequence: {ext}")
    except Exception:
        pass

    return [float(x) for x in final]
```
