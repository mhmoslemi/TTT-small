L-BFGS optimization with adaptive log-sum-exp smoothing for minimizing

import time
from scipy.optimize import minimize
from scipy.fft import fftn, ifftn

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    c = _full_convolve(a, a)

    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _lbfgs_objective(z, n, beta):
    """
    Compute smoothed score and gradient in log-space.
    z: log-space variables (a = softmax(z))
    Returns: (score, gradient)
    """
    # Softmax
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    a = e / e.sum()
    
    # Convolution
    c = _full_convolve(a, a)
    cm = float(c.max())
    
    # Log-sum-exp smoothing
    c_shifted = (c - cm) / beta
    e_c = np.exp(c_shifted)
    w = e_c / e_c.sum()
    
    # Smoothed max
    ms = cm + beta * np.log(e_c.sum())
    
    # Score (normalized, sum(a)=1)
    score = 2.0 * n * ms
    
    # Gradient
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]  # cross-correlation
    W = float(np.dot(w, c))
    
    grad = 4.0 * n * a * (h - W)
    
    return score, grad

def _lbfgs_optimize(a_init, deadline, beta_schedule, max_iter=5000):
    """
    Run L-BFGS-B optimization with beta continuation.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        result = minimize(
            _lbfgs_objective,
            z,
            args=(n, beta),
            method='L-BFGS-B',
            jac=True,
            options={
                'maxiter': max_iter,
                'maxfun': max_iter * 3,
                'ftol': 1e-12,
                'gtol': 1e-8,
                'iprint': 0
            }
        )
        
        z = result.x
        z = z - z.mean()
        
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"    L-BFGS beta={beta:.6f}: score={sc:.10f} (iters={result.nit})")
    
    return best_a, best_score

    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    a_new = np.exp(z - z.max())
    a_new = np.maximum(a_new, 0.0)
    return a_new / a_new.sum()


def _perturb_fourier(a, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    n_freqs = rng.integers(1, 4)
    for _ in range(n_freqs):
        freq = rng.integers(1, n // 3)
        amp = 0.02 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        t = np.arange(n)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    a_new = np.exp(z - z.max())
    a_new = np.maximum(a_new, 0.0)
    return a_new / a_new.sum()


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

        nonlocal best_seq, best_score

            return

        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # Load parent

    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        sc0 = _score(parent)
        print(f"Parent: n={n_main}, score={sc0:.10f}")

    # === Phase 2: Adam + L-BFGS hybrid multi-restart (350s - 800s) ===
    phase2_end = min(deadline, t_start + 800.0)
