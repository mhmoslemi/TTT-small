"""
Mode-aware multi-phase optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Key innovations: (1) Fourier-mode subspace optimization for global structure,
(2) adaptive active-set Adam with Hessian-informed steps, (3) KKT balancing
with weighted active sets, (4) diverse multi-restart, (5) length exploration.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_conv_and_grad(a, beta, top_k=0):
    """Compute convolution, softmax weights, and gradient h."""
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        w = np.zeros_like(c)
        c_top = c[idx]
        c_norm = (c_top - c_top.max()) / beta
        e = np.exp(c_norm)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w[idx[np.argmax(c_top)]] = 1.0
        else:
            w[idx] = e / s
    else:
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            idx = int(np.argmax(c))
            w = np.zeros_like(c)
            w[idx] = 1.0
        else:
            w = e / s
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return c, w, h, W, cm


def _mode_feature_matrix(n, num_modes):
    """Build Fourier mode feature matrix."""
    t = np.arange(n, dtype=np.float64)
    parts = []
    for j in range(1, num_modes + 1):
        parts.append(np.cos(2.0 * np.pi * j * t / n))
        parts.append(np.sin(2.0 * np.pi * j * t / n))
    return np.column_stack(parts)


def _mode_optimize(a_init, deadline, num_modes=15, top_k=0):
    """Optimize in Fourier mode subspace using L-BFGS."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    F = _mode_feature_matrix(n, num_modes)
    coeffs, _, _, _ = np.linalg.lstsq(F, z, rcond=None)
    c0 = float(z.mean())

    def _mode_objective(params):
        c0_val = params[0]
        A = params[1:]
        z_new = c0_val + F @ A
        a_new = _softmax_fn(z_new)
        return _score_norm_fn(a_new)

    def _mode_gradient(params):
        c0_val = params[0]
        A = params[1:]
        z_new = c0_val + F @ A
        a_new = _softmax_fn(z_new)
        c, w, h, W, cm = _compute_conv_and_grad(a_new, 0.01 * cm, top_k)
        grad_a = 2.0 * n * a_new * (h - W)
        dc0 = float(np.sum(grad_a))
        dA = F.T @ grad_a
        return np.concatenate([[dc0], dA])

    best_a = _softmax_fn(z)
    best_score = _score_norm_fn(best_a)

    if time.monotonic() < deadline:
        try:
            result = minimize(
                _mode_objective,
                np.concatenate([[c0], coeffs]),
                method='L-BFGS-B',
                jac=_mode_gradient,
                options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxls': 50, 'maxfun': 2000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x[0] + F @ result.x[1:]
                a_new = _softmax_fn(z_new)
                sc_new = _score_norm_fn(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _adam_optimize(a_init, deadline, lr=0.05, top_k=0, beta_init_scale=0.05,
                   beta_decay=0.97, n_max_iter=100000):
    """Adam optimizer with adaptive beta and top-k active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-10, beta_init_scale * max_c)

    if top_k == 0:
        cur_k = 5
    else:
        cur_k = top_k

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        a = _softmax_fn(z)

        use_k = cur_k
        c, w, h, W, cm = _compute_conv_and_grad(a, beta, use_k)
        grad = 2.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 50 == 0:
            beta = max(1e-14, beta * beta_decay)
            if top_k == 0 and t % 200 == 0:
                cur_k = min(cur_k + 3, 50)

        if best_score < 1.52 and t % 500 == 0:
            lr = max(0.002, lr * 0.98)
            beta_decay = 0.99

    return best_a, best_score


def _kkt_balance(a_init, deadline, active_frac=0.1, max_iter=5000):
    """KKT-based balancing with weighted active set."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        # Weighted active set: use top `active_frac` of the convolution
        num_active = max(3, int(active_frac * len(c)))
        idx_sorted = np.argsort(c)[-num_active:]
        active_vals = c[idx_sorted]
        weights = (active_vals - active_vals.min() + 1e-10)
        weights = weights / weights.sum()

        # Compute weighted average of shifted a
        h = np.zeros(n)
        for k_idx, wt in zip(idx_sorted, weights):
            k_val = int(k_idx)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                h[lo:hi + 1] += wt * a[k_val - hi:k_val - lo + 1][::-1]

        W = float(np.dot(h, a))

        # Direction: reduce where h > W, increase where h < W
        diff = h - W
        diff_max = float(np.abs(diff).max())
        if diff_max < 1e-15:
            break

        # Adaptive step size
        step = 0.02 * cm / (diff_max * n + 1e-15)
        step = min(step, 0.03)

        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()
        z = z - step * diff / (diff_max + 1e-15)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a_new = _softmax_fn(z)

        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Halve step
            z = np.log(np.maximum(a, 1e-15))
            z = z - z.mean()
            z = z - (step * 0.5) * diff / (diff_max + 1e-15)
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new

        # Convergence check
        if it > 50:
            active_spread = float(active_vals.max() - active_vals.min())
            if active_spread / (cm + 1e-15) < 0.001:
                break

    return best_a, best_score


def _lp_refine(a_init, deadline, max_iter=200):
    """Iterative LP-based refinement using active set linearization."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        # Active set: top peaks within 2% of max
        threshold = cm * 0.98
        active = np.where(c >= threshold)[0]
        num_active = min(len(active), max(3, 2 * n // 10))
        if num_active < len(active):
            top_idx = np.argsort(c[active])[-num_active:]
            active = active[top_idx]

        # Linearize: at current a, c(k) = sum_i a_i * a[k-i]
        # Gradient of c(k) w.r.t. a: for each i, d c(k)/d a_i = a[k-i] (if valid)
        # We want to find direction d such that for active k:
        #   sum_i d_i * a[k-i] + sum_i a_i * d[k-i] < 0 (decrease all active peaks)
        # Approx: use d = -a * (h - W) where h is the weighted average shift

        h = np.zeros(n)
        for k_idx in active:
            k_val = int(k_idx)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                h[lo:hi + 1] += a[k_val - hi:k_val - lo + 1][::-1]
        h /= len(active)

        W = float(np.dot(h, a))
        diff = h - W
        diff_max = float(np.abs(diff).max())
        if diff_max < 1e-14:
            break

        step = 0.01
        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()
        z = z - step * diff / (diff_max + 1e-15)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a_new = _softmax_fn(z)
        sc = _score_norm_fn(a_new)

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            step *= 0.5
            z = np.log(np.maximum(a, 1e-15))
            z = z - z.mean()
            z = z - step * diff / (diff_max + 1e-15)
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new

    return best_a, best_score


def _make_smooth_profile(n, shape_type, param):
    """Generate various smooth profiles."""
    t = np.linspace(0.0, 1.0, n)
    if shape_type == 'exp':
        a = np.exp(-param * t)
    elif shape_type == 'linear':
        a = 1.0 + param * (t - 0.5)
        a = np.maximum(a, 0.01)
    elif shape_type == 'quad':
        u = 2.0 * t - 1.0
        a = 1.0 + param * u ** 2
        a = np.maximum(a, 0.01)
    elif shape_type == 'cosh':
        u = 2.0 * t - 1.0
        a = np.cosh(param * u)
    elif shape_type == 'arch':
        a = 1.0 + param / (t * (1 - t) + 0.01)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

                z += 0.03 * rng.normal(size=n_best)
            else:
                # Blend with a different profile
                alt = _make_smooth_profile(n_best, 'cosh', rng.uniform(1, 4))
                z_blend = 0.7 * z + 0.3 * np.log(np.maximum(alt, 1e-6))
                z = z_blend

            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            dl = min(phase4_end - 5, time.monotonic() + per_restart)
            # Quick Adam
            p_r, sc_r = _adam_optimize(p_start, dl, lr=0.03, top_k=5,
                                       beta_init_scale=0.02, beta_decay=0.98,
                                       n_max_iter=30000)
            update_best(p_r, sc_r, f"restart_{i}")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Try different lengths ===
    print("=== Phase 5: Length exploration ===")
    phase5_end = min(deadline, t_start + 1720.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        n_curr = len(best_seq)
        target_lengths = [l for l in [2048, 4096, 8192] if l != n_curr]

        for tn in target_lengths[:2]:
            if time.monotonic() > phase5_end - 30:
                break
            p_interp = _interp_profile(best_seq, tn)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{tn}")

            if time.monotonic() < phase5_end - 30:
                dl = min(phase5_end - 10, time.monotonic() + min(90.0, (phase5_end - time.monotonic()) * 0.5))
                p_ad, sc_ad = _adam_optimize(p_interp, dl, lr=0.03, top_k=5,
                                             beta_init_scale=0.02, beta_decay=0.98,
                                             n_max_iter=50000)
                update_best(p_ad, sc_ad, f"adam_n{tn}")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Final intensive polish ===
    print("=== Phase 6: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            # Very slow Adam
            dl = min(deadline - 5, time.monotonic() + min(80.0, remaining * 0.5))
            p_fp, sc_fp = _adam_optimize(best_seq, dl, lr=0.005, top_k=20,
                                         beta_init_scale=0.003, beta_decay=0.997,
                                         n_max_iter=30000)
            update_best(p_fp, sc_fp, "final_adam")

        if time.monotonic() < deadline - 15:
            # KKT with very wide active set
            dl2 = min(deadline - 5, time.monotonic() + min(40.0, deadline - time.monotonic() - 5))
            p_fk, sc_fk = _kkt_balance(best_seq, dl2, active_frac=0.3, max_iter=2000)
            update_best(p_fk, sc_fk, "final_kkt")

    # === Final validation ===
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = _make_smooth_profile(4096, 'exp', 1.0)
            final_score = _score_fn(final_seq)

        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")

        # Validate
        try:
            seq_list = [float(x) for x in final_seq]
            sc_eval = float(evaluate_sequence(seq_list))
            print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  eval check: {e}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_smooth_profile(4096, 'exp', 1.0)
    print(f"  FALLBACK: n=4096, score={_score_norm_fn(fallback):.10f}")
    return [float(x) for x in fallback]
