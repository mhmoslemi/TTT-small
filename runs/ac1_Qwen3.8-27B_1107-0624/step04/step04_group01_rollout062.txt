```python
def _build_A(a):
    n = len(a)
    # Create indices
    i_idx = np.arange(n)  # column indices (0 to n-1)
    k_idx = np.arange(2*n-1)  # row indices (0 to 2n-2)
    
    # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    # k - i ranges from -(n-1) to 2n-2
    diff = k_idx[:, None] - i_idx[None, :]  # shape (2n-1, n)
    
    # Mask: 0 <= diff < n
    mask = (diff >= 0) & (diff < n)
    A = np.zeros_like(diff, dtype=np.float64)
    A[mask] = a[diff[mask]]
    return A
```

```python
"""
LP-based best response + Adam gradient fine-tuning for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses game-theoretic best response iteration
(LP) for coarse optimization, then analytical gradient descent with
log-sum-exp beta-continuation for fine polishing. Multi-restart for robustness.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(a, N) if b is a else np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _build_A_sparse(a):
    n = len(a)
    num_rows = 2 * n - 1
    # Column i has entries at rows i..i+n-1 with values a[0..n-1]
    indices = np.tile(np.arange(n), n) + np.repeat(np.arange(n), n)
    data = np.tile(a, n)
    indptr = (np.arange(n + 1) * n).astype(np.int32)
    return csr_matrix((data.astype(np.float64), indices.astype(np.int32), indptr),
                      shape=(num_rows, n))


def _lp_best_response(a, M, timeout=None):
    n = len(a)
    A = _build_A_sparse(a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    try:
        opts = {}
        if timeout is not None:
            opts['time_limit'] = timeout
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs', options=opts)
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _compute_smoothed_grad(a, beta):
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # grad[i] = 4*n * sum_j a[j] * w[i+j]
    a_rev = a[::-1]
    conv = _full_convolve(a_rev, w)
    h = conv[n - 1:2 * n - 1]
    grad = 4.0 * n * h
    return grad, w, cm + beta * np.log(max(s, 1e-300))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered_asym":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5 * (1.0 + 0.3 * (1.0 - 2.0 * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered_asym_strong":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5 * (1.0 + 0.5 * (1.0 - 2.0 * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian_asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.35) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp_taper":
        t = np.linspace(0, 1, n)
        a = (0.5 + t) * np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    return np.ones(n) / n


def _adam_step(z, grad, m_buf, v_buf, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    m_buf = b1 * m_buf + (1.0 - b1) * grad
    v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
    mh = m_buf / (1.0 - b1 ** t)
    vh = v_buf / (1.0 - b2 ** t)
    z = z - lr * mh / (np.sqrt(vh) + eps)
    return z, m_buf, v_buf


def _grad_optimize(a_init, deadline, beta_schedule, max_iter=200000, lr=0.01):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_phase = 0
        while it_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_phase += 1
            t += 1
            a = _softmax(z)
            grad, w, smoothed_val = _compute_smoothed_grad(a, beta)
            if not np.all(np.isfinite(grad)):
                break
            # Clip gradient
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e10:
                grad = grad * (1e10 / gnorm)
            z, m_buf, v_buf = _adam_step(z, grad, m_buf, v_buf, t, lr)
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            if t % 100 == 0:
                sc = _score(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = float(sc)
                    best_a = a.copy()
            if it_phase % 5000 == 0:
                lr = max(0.001, lr * 0.95)
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = float(sc_final)
        best_a = a_final.copy()
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            if float(parent_raw.sum()) > 0.01:
                parent = parent_raw / float(parent_raw.sum())
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        print(f"Parent: n={n_main}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_main = 2048
        print(f"No parent, n={n_main}")

    # Phase 0: Initialization (0-40s)
    phase0_end = min(deadline, t_start + 40.0)
    for kind in ["tapered_asym", "tapered_asym_strong", "gaussian_asym", "tapered", "ramp_taper"]:
        if time.monotonic() > phase0_end:
            break
        p = _make_profile(n_main, kind)
        update_best(p, _score(p), f"init_{kind}")

    if parent is not None and _score(parent) < 1.6:
        a_cur = parent.copy()
    elif best_seq is not None:
        a_cur = best_seq / best_seq.sum()
    else:
        a_cur = _make_profile(n_main, "tapered_asym")

    print(f"Phase 0: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 1: LP-based best response (40s - 600s)
    phase1_end = min(deadline, t_start + 600.0)
    a_cur = np.maximum(a_cur, 0.0)
    a_cur = a_cur / a_cur.sum()
    n = len(a_cur)

    # Choose LP n (might reduce if parent is too large)
    lp_n = n
    if lp_n > 2048:
        # Interpolate down for faster LP
        x0 = np.linspace(0, 1, n)
        x1 = np.linspace(0, 1, lp_n)
        a_lp = _make_profile(lp_n, "tapered_asym")  # fallback
        # Actually just subsample
        a_lp = a_cur[::(n // lp_n)] if n % lp_n == 0 else np.interp(x1, x0, a_cur)
        a_lp = np.maximum(a_lp, 1e-10)
        a_lp = a_lp / a_lp.sum()
    else:
        a_lp = a_cur.copy()

    lp_n = len(a_lp)
    print(f"LP phase: n={lp_n}, t={time.monotonic()-t_start:.1f}s")

    lp_iter = 0
    prev_lp_score = _score(a_lp)
    mix = 0.5
    no_improve = 0

    while time.monotonic() < phase1_end - 15:
        lp_iter += 1
        c_lp = _full_convolve(a_lp, a_lp)
        M = float(c_lp.max())

        b_star = _lp_best_response(a_lp, M, timeout=30)
        if b_star is None:
            print(f"  LP failed at iter {lp_iter}, switching to smaller n")
            if lp_n > 512:
                a_lp = a_lp[::2]
                a_lp = np.maximum(a_lp, 1e-10)
                a_lp = a_lp / a_lp.sum()
                lp_n = len(a_lp)
                mix = 0.5
            else:
                break
            continue

        s_b = float(b_star.sum())
        if s_b < 1e-15:
            break
        b_norm = b_star / s_b

        a_new = (1.0 - mix) * a_lp + mix * b_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)

        if sc_new < prev_lp_score - 1e-12:
            a_lp = a_new
            prev_lp_score = sc_new
            no_improve = 0
            if lp_n == n:
                update_best(a_lp, sc_new, f"lp{lp_iter}")
        else:
            no_improve += 1

        if no_improve > 8:
            mix = max(0.02, mix * 0.7)
            no_improve = 0
            if mix < 0.03:
                print(f"  LP converging, mix={mix:.4f}, stopping LP phase")
                break

        if lp_iter % 10 == 0:
            print(f"  LP iter {lp_iter}: score={sc_new:.10f}, M={M:.6f}, mix={mix:.3f}, t={time.monotonic()-t_start:.1f}s")

    # Interpolate LP result back to full n if needed
    if lp_n != n and a_lp is not None:
        x0 = np.linspace(0, 1, lp_n)
        x1 = np.linspace(0, 1, n)
        a_full = np.interp(x1, x0, a_lp)
        a_full = np.maximum(a_full, 1e-10)
        a_full = a_full / a_full.sum()
        update_best(a_full, _score(a_full), f"lp_interp_n{n}")
    elif lp_n == n:
        update_best(a_lp, _score(a_lp), "lp_final")

    print(f"Phase 1 (LP): best={best_score:.10f}, iters={lp_iter}, t={time.monotonic()-t_start:.1f}s")

    # Phase 2: Gradient fine-tuning (600s - 1200s)
    phase2_end = min(deadline, t_start + 1200.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        a_opt = best_seq / best_seq.sum()
        n_opt = len(a_opt)
        c_opt = _full_convolve(a_opt, a_opt)
        max_c = float(c_opt.max())
        beta_sched = [0.04 * max_c, 0.02 * max_c, 0.01 * max_c,
                      0.005 * max_c, 0.002 * max_c, 0.001 * max_c,
                      0.0005 * max_c, 0.0002 * max_c]
        p_best, sc_best = _grad_optimize(a_opt, phase2_end - 5, beta_sched,
                                         max_iter=100000, lr=0.012)
        update_best(p_best, _score(p_best), "grad_main")

    print(f"Phase 2 (grad): best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 3: Multi-restart (1200s - 1600s)
    phase3_end = min(deadline, t_start + 1600.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        n_restarts = 6
        per_restart = (phase3_end - time.monotonic() - 15) / max(1, n_restarts)
        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 5:
                break
            sigma = 0.02 + 0.015 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z + sigma * rng.standard_normal(len(z))
            z = z - z.mean()
            a_start = _softmax(z)
            n_r = len(a_start)
            c_r = _full_convolve(a_start, a_start)
            beta_r = [0.02 * float(c_r.max()), 0.005 * float(c_r.max()),
                      0.001 * float(c_r.max())]
            p_r, sc_r = _grad_optimize(a_start, time.monotonic() + per_restart,
                                       beta_r, max_iter=30000, lr=0.01)
            update_best(p_r, _score(p_r), f"restart{i}")

    print(f"Phase 3 (restarts): best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 4: Final polish (1600s - deadline)
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_fin = best_seq / best_seq.sum()
        c_fin = _full_convolve(a_fin, a_fin)
        max_cf = float(c_fin.max())
        beta_fin = [0.001 * max_cf, 0.0003 * max_cf, 0.0001 * max_cf,
                    0.00003 * max_cf, 0.00001 * max_cf]
        p_fin, sc_fin = _grad_optimize(a_fin, deadline - 3, beta_fin,
                                       max_iter=100000, lr=0.004)
        update_best(p_fin, _score(p_fin), "final_polish")

    print(f"Phase 4 (polish): best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
