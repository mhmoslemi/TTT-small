"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:
1. Active-set Newton method with sparse KKT solve
2. SLP (Successive Linearization) direction finding via LP
3. More aggressive beta continuation
4. Multi-length exploration including n=8192
5. Better line search in active-set balancing
"""
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import spsolve
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _convolve_shift(w, a, n):
    """Compute h_i = sum_k w_k * a[k-i] for valid indices.
    This is the gradient contribution from active set weights w on convolution peaks."""
    h = np.zeros(n)
    for k_idx, w_val in enumerate(w):
        if w_val == 0.0:
            continue
        k = int(k_idx)
        # a[k - i] for i = 0..n-1, valid when 0 <= k-i <= n-1
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            h[lo:hi + 1] += w_val * a[k - hi:k - lo + 1][::-1]
    return h


def _z_objective(z, beta, n, top_k=0):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_norm = c_top - c_top.max()
        log_sum = float(np.log(np.sum(np.exp(c_norm / beta))))
        return 2.0 * n * (c_top.max() + beta * log_sum)
    else:
        c_shifted = (c - cm) / beta
        log_sum = float(np.log(np.sum(np.exp(c_shifted))))
        return 2.0 * n * (cm + beta * log_sum)


def _z_gradient(z, beta, n, top_k=0):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_norm = (c_top - c_top.max()) / beta
        e = np.exp(c_norm)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(len(c))
            w[idx[0]] = 1.0
        else:
            w = np.zeros(len(c))
            w[idx] = e / s
    else:
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            idx = int(np.argmax(c))
            w = np.zeros(len(c))
            w[idx] = 1.0
        else:
            w = e / s
    W = float(np.dot(w, c))
    h = _convolve_shift(w, a, n)
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs(a_init, deadline, max_wall_time=None, top_k=0):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    if max_wall_time is None:
        max_wall_time = max(1.0, deadline - time.monotonic())
    beta_starts = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max(max_c0, 1e-10) for b in beta_starts]
    per_level = max_wall_time / len(betas)
    for beta in betas:
        if time.monotonic() > deadline:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)
        try:
            result = minimize(
                _z_objective, z, args=(beta, n, top_k),
                method='L-BFGS-B', jac=_z_gradient,
                options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 40, 'maxfun': 2000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                if np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass
    return best_a, best_score


def _adam_polish(a_init, deadline, lr=0.05, top_k=0, n_max_iter=50000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-10, 0.01 * max_c)
    adaptive_k = (top_k == 0)
    cur_k = 3 if adaptive_k else top_k
    stalling = 0
    best_at_stall = best_score
    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        use_k = cur_k if adaptive_k else top_k
        grad = _z_gradient(z, beta, n, use_k)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            stalling = 0
            best_at_stall = best_score
        else:
            stalling += 1
        if t % 50 == 0:
            beta = max(1e-14, beta * 0.85)
            if adaptive_k and t % 300 == 0:
                cur_k = min(cur_k + 2, 50)
        if stalling > 2000:
            if best_score < 1.51:
                lr = max(0.002, lr * 0.95)
            stalling = 0
    return best_a, best_score


def _active_set_balance(a_init, deadline, max_iter=20000, k_active=None):
    """Active-set balancing with line search and optimal weights."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    if k_active is None:
        k_active = min(20, n)
    
    stall_count = 0
    stall_threshold = best_score
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Find active set: peaks within 0.5% of max
        threshold = cm * 0.9995
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0:
            active = np.argmax(c)
            active = np.array([active])
        
        # Limit active set size
        if len(active) > k_active:
            top_idx = np.argsort(c)[::-1][:k_active]
            active = top_idx
        
        active_sorted = np.sort(active)
        n_active = len(active_sorted)
        
        # Compute optimal weights: minimize max rate of increase
        # Simple approach: weight proportional to how much above mean
        c_active = c[active_sorted]
        c_mean = float(c_active.mean())
        c_spread = float(c_active.max() - c_active.min())
        if c_spread > 1e-15:
            w = (c_active - c_mean + c_spread) / (2.0 * c_spread * n_active)
            w = w / w.sum()
        else:
            w = np.ones(n_active) / n_active
        
        # Compute h = weighted average shift
        h = _convolve_shift(w, a, n)
        W = float(np.dot(w, c_active))
        
        # Direction: reduce where h > W, increase where h < W
        diff = h - W
        max_diff = float(np.abs(diff).max())
        if max_diff < 1e-15:
            break
        
        # Try multiple step sizes via line search
        best_step_sc = best_score
        best_step_a = a
        base_z = np.log(np.maximum(a, 1e-15))
        base_z = base_z - base_z.mean()
        
        step_candidates = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5]
        for step in step_candidates:
            z_new = base_z - step * diff / max_diff
            z_new = np.clip(z_new, -60, 60)
            z_new = z_new - z_new.mean()
            a_new = _softmax_fn(z_new)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_step_sc:
                best_step_sc = sc
                best_step_a = a_new.copy()
        
        if np.isfinite(best_step_sc) and best_step_sc < best_score:
            best_score = best_step_sc
            best_a = best_step_a.copy()
            a = best_step_a
            stall_count = 0
            stall_threshold = best_score
        else:
            a = best_step_a
            stall_count += 1
            if stall_count > 5000:
                break
            if stall_count % 1000 == 0:
                # Reset with smaller effective step
                a = best_a.copy()
    
    return best_a, best_score


def _newton_active_set(a_init, deadline, max_iter=500, k_active=15):
    """Newton's method on the KKT system for a fixed active set."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Identify active set
        active = np.argsort(c)[::-1][:k_active]
        active_sorted = np.sort(active)
        n_active = len(active_sorted)
        
        # Current residual of KKT conditions
        # Compute h with uniform weights
        w = np.ones(n_active) / n_active
        h = _convolve_shift(w, a, n)
        W = float(np.dot(w, c[active_sorted]))
        
        # The gradient of the smoothed objective (exact max approximation)
        # At optimum: h_i should be constant for all i (since p_i > 0 for all i)
        # Residual: h_i - W
        residual = h - W
        res_norm = float(np.abs(residual).max())
        
        if res_norm < 1e-10 * (cm + 1e-15):
            break
        
        # Newton-like step: solve (J^T J) d = -J^T r approximately
        # J is the Jacobian of residual w.r.t. z (logit space)
        # residual_i = h_i - W = sum_k w_k * a[k-i] - W
        # In logit space: da_i/dz_i = a_i(1-a_i), da_i/dz_j = 0 (for i != j)
        # This is complex, so use a simplified gradient step with adaptive step size
        
        # Compute effective gradient in z-space
        grad_z = a * residual
        grad_norm = float(np.sqrt(np.dot(grad_z, grad_z)))
        if grad_norm < 1e-15:
            break
        
        # Adaptive step size
        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()
        
        # Try several step sizes
        best_sc = best_score
        best_a_new = a
        for alpha in [0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
            z_try = z - alpha * grad_z / (grad_norm / max_diff)
            z_try = np.clip(z_try, -50, 50)
            z_try = z_try - z_try.mean()
            a_try = _softmax_fn(z_try)
            sc = _score_norm_fn(a_try)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a_new = a_try.copy()
        
        if best_sc < best_score:
            best_score = best_sc
            best_a = best_a_new.copy()
            a = best_a_new
        else:
            a = best_a_new
            if it > 100:
                break
    
    return best_a, best_score


def _slp_direction(a, c, active, n):
    """Find SLP direction using LP: minimize max of linearized active constraints."""
    # Linearize: c_k(p + t*d) ≈ c_k(p) + t * sum_i (dc_k/dp_i) * d_i
    # dc_k/dp_i = 2 * p[k-i] (if valid)
    # We want: for all k in active: sum_i 2*p[k-i]*d_i < 0
    # Subject to: sum_i d_i = 0, ||d||_inf <= 1
    
    # Simplified: find d that minimizes max_k {sum_i 2*p[k-i]*d_i}
    # This is a linear program in d
    
    # For efficiency, use a simpler approach:
    # d_i = -h_i + mean(h) where h_i = (1/|A|) sum_{k in A} 2*p[k-i]
    n_active = len(active)
    h = np.zeros(n)
    for k_idx in active:
        k = int(k_idx)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            h[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    h /= n_active
    d = -(h - h.mean())
    d_norm = float(np.abs(d).max())
    if d_norm > 0:
        d /= d_norm
    return d


def _sweep_lengths(a_best, t_start, deadline, rng):
    """Try different sequence lengths by interpolation + optimization."""
    n_best = len(a_best)
    results = []
    
    target_ns = [1024, 2048, 4096, 8192, 16384]
    target_ns = [tn for tn in target_ns if tn != n_best]
    target_ns.sort(key=lambda x: abs(x - n_best))
    
    for tn in target_ns[:3]:
        if time.monotonic() > deadline - 60:
            break
        
        # Interpolate
        x0 = np.linspace(0.0, 1.0, n_best)
        x1 = np.linspace(0.0, 1.0, tn)
        p_interp = np.interp(x1, x0, a_best)
        p_interp = np.maximum(p_interp, 1e-12)
        p_interp = p_interp / p_interp.sum()
        
        sc_interp = _score_norm_fn(p_interp)
        results.append((sc_interp, p_interp.copy(), tn))
        
        # Quick optimization
        remaining = deadline - time.monotonic() - 30
        if remaining > 30:
            per_opt = min(80.0, remaining * 0.5)
            dl = min(deadline - 30, time.monotonic() + per_opt)
            p_opt, sc_opt = _run_lbfgs(p_interp, dl, max_wall_time=per_opt, top_k=0)
            results.append((sc_opt, p_opt, tn))
            
            if time.monotonic() < deadline - 40:
                dl2 = min(deadline - 30, time.monotonic() + min(40.0, remaining * 0.3))
                p_opt2, sc_opt2 = _adam_polish(p_opt, dl2, lr=0.03, top_k=5, n_max_iter=15000)
                results.append((sc_opt2, p_opt2, tn))
    
    return results


def _make_window_profile(n, type_='kaiser', param=10.0):
    """Create various window function profiles."""
    t = np.arange(n) / max(1, n - 1)
    if type_ == 'kaiser':
        from scipy.special import i0
        a = i0(param * np.sqrt(np.maximum(0, 1 - (2 * t - 1) ** 2))) / i0(param)
    elif type_ == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif type_ == 'hamming':
        a = 0.54 - 0.46 * np.cos(2 * np.pi * t)
    elif type_ == 'blackman':
        a = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    elif type_ == 'gaussian':
        sigma = param / (n - 1)
        center = (n - 1) / 2.0
        a = np.exp(-0.5 * ((t * (n - 1) - center) / (sigma * (n - 1))) ** 2)
    elif type_ == 'uniform':
        a = np.ones(n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_fourier_profile(n, freqs=None, amplitudes=None, phase=0):
    """Create a profile based on a sum of cosines (always positive if dominant DC)."""
    if freqs is None:
        freqs = [0, 1, 2, 3, 4, 5]
    if amplitudes is None:
        amplitudes = [1.0, 0.3, 0.2, 0.1, 0.05, 0.02]
    t = np.linspace(0, 2 * np.pi, n, endpoint=False)
    a = np.zeros(n)
    for f, amp in zip(freqs, amplitudes):
        a += amp * np.cos(f * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _run_with_active_set(a_init, deadline, k_active=None):
    """Combined active-set optimization: balance + Newton + re-identify."""
    n = len(a_init)
    if k_active is None:
        k_active = min(30, n // 10)
    
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    # Alternating: balance steps then Newton steps
    cycle_len = 500
    cycle = 0
    
    while time.monotonic() < deadline:
        # Active set balancing phase
        bal_deadline = min(deadline, time.monotonic() + 30)
        a, sc = _active_set_balance(a, bal_deadline, max_iter=cycle_len, k_active=k_active)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Newton refinement
        newton_deadline = min(deadline, time.monotonic() + 20)
        a2, sc2 = _newton_active_set(a, newton_deadline, max_iter=200, k_active=k_active)
        if sc2 < best_score:
            best_score = sc2
            best_a = a2.copy()
            a = a2
        
        cycle += 1
        if cycle > 20:  # Safety
            break
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining L-BFGS, Adam, active-set Newton,
    and SLP methods to minimize 2*n*max(conv(a,a))/sum(a)^2.
    """
    global height_sequence_1
    
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")
    
    # ==================== Phase 0: Initialization ====================
    print("=== Phase 0: Initialization ===")
    phase0_end = min(deadline, t_start + 30.0)
    candidates = []
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm_fn(parent)
                candidates.append((sc, parent.copy(), len(parent), "parent"))
                update_best(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent: {e}")
    
    # Determine main search length
    if parent is not None:
        n_main = len(parent)
    else:
        n_main = 4096
    
    # Generate diverse profiles
    profiles = []
    for wn, wp in [('kaiser', 5.0), ('kaiser', 10.0), ('kaiser', 15.0),
                   ('hann', 0), ('hamming', 0), ('blackman', 0),
                   ('gaussian', 3.0), ('gaussian', 5.0)]:
        if time.monotonic() < phase0_end:
            profiles.append(_make_window_profile(n_main, wn, wp))
    
    # Fourier profiles
    for f1 in [0.5, 1.0, 2.0]:
        for f2 in [1.0, 2.0, 3.0]:
            if f1 != f2:
                a = _make_fourier_profile(n_main, freqs=[0, f1, f2], amplitudes=[1.0, 0.3, 0.15])
                profiles.append(a)
    
    # Exponential profiles
    for alpha in [0.001, 0.003, 0.005, 0.01, 0.02]:
        if time.monotonic() < phase0_end:
            r = 1.0 - alpha
            a = r ** np.arange(n_main)
            a = a / a.sum()
            profiles.append(a)
    
    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p, n_main, f"prof{idx}"))
        update_best(p, sc, f"prof{idx}")
    
    # If parent exists at different length, interpolate
    if parent is not None and len(parent) != n_main:
        x0 = np.linspace(0.0, 1.0, len(parent))
        x1 = np.linspace(0.0, 1.0, n_main)
        p_interp = np.interp(x1, x0, parent)
        p_interp = np.maximum(p_interp, 1e-12)
        p_interp = p_interp / p_interp.sum()
        sc = _score_norm_fn(p_interp)
        candidates.append((sc, p_interp, n_main, "parent_interp"))
        update_best(p_interp, sc, "parent_interp")
    
    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 0 done: best={best_score:.10f}, {len(candidates)} candidates")
    
    # ==================== Phase 1: L-BFGS Continuation ====================
    print("=== Phase 1: L-BFGS Continuation ===")
    phase1_end = min(deadline, t_start + 450.0)
    count = 0
    seen = set()
    
    for sc, p, pn, label in candidates:
        if time.monotonic() > phase1_end:
            break
        if count >= 3:
            break
        if sc > best_score * 1.05:
            break
        key = f"{label}_{pn}"
        if key in seen:
            continue
        seen.add(key)
        
        remaining = phase1_end - time.monotonic()
        per_time = max(10.0, min(150.0, remaining / max(1, 3 - count)))
        dl = min(phase1_end, time.monotonic() + per_time)
        print(f"  L-BFGS on {label} (n={pn}), budget={per_time:.1f}s")
        
        p_best, sc_best = _run_lbfgs(p, dl, max_wall_time=per_time, top_k=0)
        update_best(p_best, sc_best, f"lbfgs_{label}")
        count += 1
    
    print(f"  Phase 1 done: best={best_score:.10f}")
    
    # ==================== Phase 2: Adam Polish ====================
    print("=== Phase 2: Adam Polish ===")
    phase2_end = min(deadline, t_start + 800.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        remaining = phase2_end - time.monotonic()
        dl = min(phase2_end, time.monotonic() + remaining * 0.6)
        print(f"  Adam pass 1 (n={len(best_seq)}), budget={remaining*0.6:.1f}s")
        p_best, sc_best = _adam_polish(best_seq, dl, lr=0.05, top_k=0, n_max_iter=100000)
        update_best(p_best, sc_best, "adam1")
        
        if time.monotonic() < phase2_end - 30:
            dl2 = min(phase2_end - 10, time.monotonic() + (phase2_end - time.monotonic()) * 0.5)
            print(f"  Adam pass 2 (top_k=10), budget={dl2 - time.monotonic():.1f}s")
            p_best2, sc_best2 = _adam_polish(best_seq, dl2, lr=0.02, top_k=10, n_max_iter=50000)
            update_best(p_best2, sc_best2, "adam2")
    
    print(f"  Phase 2 done: best={best_score:.10f}")
    
    # ==================== Phase 3: Active-Set Optimization ====================
    print("=== Phase 3: Active-Set Optimization ===")
    phase3_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        remaining = phase3_end - time.monotonic()
        print(f"  Active-set optimization, budget={remaining:.1f}s")
        
        # Try different active set sizes
        for k_act in [10, 20, 50]:
            if time.monotonic() > phase3_end - 20:
                break
            dl = min(phase3_end - 10, time.monotonic() + min(80.0, remaining * 0.3))
            print(f"    Active-set balance with k={k_act}")
            p_best, sc_best = _active_set_balance(best_seq, dl, max_iter=20000, k_active=k_act)
            update_best(p_best, sc_best, f"aset_k{k_act}")
        
        # Newton refinement
        if time.monotonic() < phase3_end - 30:
            dl = min(phase3_end - 10, time.monotonic() + 60.0)
            print(f"    Newton active-set, k=20")
            p_best, sc_best = _newton_active_set(best_seq, dl, max_iter=300, k_active=20)
            update_best(p_best, sc_best, "newton_k20")
    
    print(f"  Phase 3 done: best={best_score:.10f}")
    
    # ==================== Phase 4: Multi-Restart ====================
    print("=== Phase 4: Multi-Restart ===")
    phase4_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 5
        per_restart = min(50.0, (phase4_end - time.monotonic()) / max(1, n_restarts))
        
        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 10:
                break
            # Different noise structures
            noise_strategies = [
                lambda: 0.05 * rng.normal(size=n_best),
                lambda: 0.1 * rng.normal(size=n_best),
                lambda: 0.03 * rng.standard_t(df=3, size=n_best),
                lambda: 0.08 * (rng.random(n_best) - 0.5),
                lambda: 0.02 * rng.normal(size=n_best) + 0.05 * rng.normal(size=n_best) ** 2,
            ]
            noise = noise_strategies[i % len(noise_strategies)]()
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise
            z = np.clip(z, -40, 40)
            z = z - z.mean()
            p_start = _softmax_fn(z)
            
            dl = min(phase4_end - 5, time.monotonic() + per_restart)
            per_lbfgs = per_restart * 0.5
            per_adam = per_restart * 0.45
            p_m, sc_m = _run_lbfgs(p_start, min(phase4_end - 5, time.monotonic() + per_lbfgs),
                                   max_wall_time=per_lbfgs, top_k=5)
            update_best(p_m, sc_m, f"r{i}_lbfgs")
            if time.monotonic() < phase4_end - 5:
                p_a, sc_a = _adam_polish(p_m, min(phase4_end - 5, time.monotonic() + per_adam),
                                         lr=0.03, top_k=10, n_max_iter=15000)
                update_best(p_a, sc_a, f"r{i}_adam")
    
    print(f"  Phase 4 done: best={best_score:.10f}")
    
    # ==================== Phase 5: Multi-Length ====================
    print("=== Phase 5: Multi-Length Exploration ===")
    phase5_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        results = _sweep_lengths(best_seq, t_start, phase5_end, rng)
        for sc, p, tn in results:
            update_best(p, sc, f"len{tn}")
    
    print(f"  Phase 5 done: best={best_score:.10f}")
    
    # ==================== Phase 6: Final Polish ====================
    print("=== Phase 6: Final Polish ===")
    if time.monotonic() < deadline and best_seq is not None:
        remaining = deadline - time.monotonic()
        
        if remaining > 20:
            dl = min(deadline - 5, time.monotonic() + min(40.0, remaining * 0.4))
            p_best, sc_best = _adam_polish(best_seq, dl, lr=0.008, top_k=20, n_max_iter=50000)
            update_best(p_best, sc_best, "final_adam")
        
        if time.monotonic() < deadline - 10:
            dl2 = min(deadline - 5, time.monotonic() + min(30.0, deadline - time.monotonic() - 5))
            p_best2, sc_best2 = _active_set_balance(best_seq, dl2, max_iter=10000, k_active=30)
            update_best(p_best2, sc_best2, "final_aset")
        
        if time.monotonic() < deadline - 5:
            dl3 = min(deadline - 3, time.monotonic() + min(20.0, deadline - time.monotonic() - 3))
            p_best3, sc_best3 = _run_lbfgs(best_seq, dl3, max_wall_time=max(3.0, dl3 - time.monotonic()), top_k=20)
            update_best(p_best3, sc_best3, "final_lbfgs")
    
    # ==================== Final Validation ====================
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = _make_window_profile(4096, 'kaiser', 10.0)
            final_score = _score_fn(final_seq)
        
        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")
        
        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 16384:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  evaluate_sequence check failed: {e}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_window_profile(4096, 'kaiser', 10.0)
    print(f"  FALLBACK: n=4096, score={_score_norm_fn(fallback):.10f}")
    return [float(x) for x in fallback]
