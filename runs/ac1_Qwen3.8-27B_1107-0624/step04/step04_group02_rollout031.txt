    # sum_k w[k] * a[k-j] * 1[0<=k-j<=n-1]
    # Let i = k-j: sum_i w[i+j] * a[i] * 1[0<=i<=n-1]
    # = sum_{i=0}^{n-1} a[i] * w[i+j]
    
    # This is the cross-correlation of a and w at lag j.
    # In terms of convolution: cross_corr(a, w)[j] = (rev(a) * w)[j + n - 1]
    # where rev(a)[i] = a[n-1-i]
    
    a_rev = a[::-1]  # length n
    # cross_corr = convolve(a_rev, w)  # length n + (2n-1) - 1 = 3n-2
    # We want cross_corr[j + n - 1] for j = 0, ..., n-1
    # So we want indices n-1 to 2n-2 of the convolution
    
    cross = _full_convolve(a_rev, w)  # length 3n-2
    grad_a = 4.0 * n * cross[n-1:2*n-1]  # length n
    
    # Gradient w.r.t. z (softmax parameterization)
    # dg/dz[j] = a[j] * (grad_a[j] - sum_i a[i]*grad_a[i])
    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)
    
    return float(obj), grad_z, w, cm
