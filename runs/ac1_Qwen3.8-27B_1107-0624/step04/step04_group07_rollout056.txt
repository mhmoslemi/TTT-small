    # cross[i] = sum_{m=0}^{n-1} w[i+m] * a[m]
    
    # Pad both to length N >= (2n-1) + n - 1 = 3n-2
    N = _next_pow2(3 * n - 2)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    # Cross-correlation: IFFT(FFT(w) * conj(FFT(a)))
    cross_full = np.fft.irfft(W * np.conj(A), N)
    # The cross-correlation at lag i is cross_full[i] for i >= 0
    return cross_full[:n]

def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w

def _adam_smooth_max(a, deadline, rng, lr=0.05, beta_init=None,
                     beta_decay=0.95, no_imp_limit=300):
    n = len(a)
    a = np.maximum(a, 1e-12).copy()
    a = a / a.sum()
    
    z = np.log(a)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    c = _full_convolve(a)
    maxc = float(c.max())
    beta = beta_init if beta_init is not None else 0.05 * maxc
    
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    
    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)
        g = _cross_correl_w_a(w, a)
        
        # Project gradient to tangent space of simplex
        dot = float(np.dot(a, g))
        gz = a * (g - dot)
        gz -= gz.mean()
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()
        
        a = np.exp(z - z.max())
        a = a / a.sum()
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        c = _full_convolve(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        
        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.002, lr * 0.92)
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        if t_adam % 30 == 0:
            beta = max(1e-4, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.1:
                    beta = max(1e-4, beta * 0.5)
        
        if no_imp > no_imp_limit:
            ns = 0.05 + 0.05 * (no_imp // no_imp_limit)
            ns = min(ns, 0.3)
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a)
            beta = 0.05 * c.max()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0
    
    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score

def _coord_descent_exact(a, deadline, max_sweeps=100):

            # Split into j=i and j≠i:
            # = a[i]*a[k-i]*1_{0<=k-i<n} + sum_{j≠i} a[j]*a[k-j]*1_{0<=k-j<n}

Optimized search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Adam smooth-max, LP fixed-point, coordinate descent, and multi-n exploration.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())

def _smooth_max_beta(c, beta):
    """Compute softmax weights for smooth max approximation."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * np.log(s), w

    """Compute gradient of smooth_max(conv(a,a)) w.r.t. a.
    Returns (gradient, smooth_max_value).
    """
