"""
Multi-phase optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses improved L-BFGS with logsumexp, active-set projected gradient with
backtracking line search, and multi-resolution warm start at n=8192.
Key improvements: larger sequence length, exact active-set optimization,
adaptive step sizes, and direct subgradient method.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_active_gradient(a, active_indices, n):
    """
    Compute exact subgradient of max_{k in A} conv(a,a)[k] w.r.t. a.
    grad[i] = (2/|A|) * sum_{k in A, 0<=k-i<n} a[k-i]
    Computed via FFT: grad = 2/|A| * conv(v, a_rev)[n-1 : 2n-1]
    """
    if len(active_indices) == 0:
        return np.zeros(n)
    c_len = 2 * n - 1
    v = np.zeros(c_len)
    v[active_indices] = 1.0
    a_rev = a[::-1]
    vc = _full_convolve(v, a_rev)
    # corr[i] = vc[i + n - 1]
    grad = (2.0 / len(active_indices)) * vc[n - 1:2 * n - 1]
    return grad


def _z_gradient_active(z, a, n, active_indices):
    """Gradient of max active conv value w.r.t. z (logit space)."""
    G = _compute_active_gradient(a, active_indices, n)
    S = float(a.sum())
    H = float(np.dot(G, a))
    grad_z = a * (G - H / S)
    return grad_z


def _z_objective_smooth(z, beta, n, top_k):
    """Logsumexp-smoothed objective in logit space."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        cm = float(c_top.max())
        shifted = (c_top - cm) / beta
        e = np.exp(shifted)
        ls = float(np.log(np.sum(e)))
        return 2.0 * n * (cm + beta * ls) / (float(a.sum()) ** 2)
    else:
        cm = float(c.max())
        shifted = (c - cm) / beta
        e = np.exp(shifted)
        ls = float(np.log(np.sum(e)))
        return 2.0 * n * (cm + beta * ls) / (float(a.sum()) ** 2)


def _z_gradient_smooth(z, beta, n, top_k):
    """Gradient of logsumexp-smoothed objective."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    S = float(a.sum())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        cm = float(c_top.max())
        shifted = (c_top - cm) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(len(c))
            w[idx[0]] = 1.0
        else:
            w = np.zeros(len(c))
            w[idx] = e / s
    else:
        cm = float(c.max())
        shifted = (c - cm) / beta
        e = np.exp(shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            idx = int(np.argmax(c))
            w = np.zeros(len(c))
            w[idx] = 1.0
        else:
            w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    # Gradient of 2*n*W/S^2 w.r.t. a is 2*n*(h/S^2 - 2*W*a/S^3)
    # Gradient w.r.t. z: a * (2n*h/S^2 - 4n*W*something)
    # Actually let's use: d/dz_j [f(a(z))] where f = 2n*W/S^2
    # df/da_i = 2n*(h_i/S^2 - 2*W*a_i/S^3)
    # df/dz_j = sum_i (df/da_i) * (a_i*delta_ij - a_i*a_j/S)
    #          = a_j*(df/da_j) - (a_j/S)*sum_i(df/da_i)*a_i
    G = 2.0 * n * (h / (S * S) - 2.0 * W * a / (S ** 3))
    H = float(np.dot(G, a))
    grad_z = a * (G - H / S)
    if not np.all(np.isfinite(grad_z)):
        return np.zeros(n)
    return grad_z


def _active_set_grad_descent(a_init, deadline, max_iter=50000, top_k_start=5,
                             top_k_max=100, momentum=0.9):
    """
    Core optimization: active-set gradient descent with momentum and
    backtracking line search. Works in logit space.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    target_sum = float(np.sqrt(2.0 * n))

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init) * (a_init.sum() / target_sum) ** 2
    # Actually let's just track the normalized score
    best_norm_score = _score_norm_fn(a_init)

    v = np.zeros(n)  # momentum
    t = 0
    cur_k = top_k_start

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_fn(z) * target_sum

        # Compute convolution and find active set
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Score check
        sc = cm  # Since sum(a) = sqrt(2n), score = max_conv
        if sc < best_norm_score:
            best_norm_score = sc
            best_a = a.copy()

        # Find active set
        if cur_k < len(c):
            idx = np.argsort(c)[-cur_k:]
            active = idx
        else:
            active = np.arange(len(c))

        # Compute gradient in z-space
        G = _compute_active_gradient(a, active, n)
        S = target_sum
        H = float(np.dot(G, a))
        grad_z = a * (G - H / S)

        # Nesterov momentum
        v = momentum * v + (1 - momentum) * grad_z
        direction = -v

        # Backtracking line search
        z_old = z.copy()
        f_old = sc
        step = 0.1
        for ls in range(20):
            z_try = z_old + step * direction
            z_try = np.clip(z_try, -60, 60)
            z_try = z_try - z_try.mean()
            a_try = _softmax_fn(z_try) * target_sum
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 0.0001 * step * float(np.dot(direction, grad_z)):
                break
            step *= 0.5
        else:
            step = 0.001

        z = z_old + step * direction
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        # Adaptively grow active set
        if t % 200 == 0 and cur_k < top_k_max:
            cur_k = min(cur_k + 3, top_k_max)

        # Periodic restart with small noise to escape local minima
        if t % 5000 == 0 and it > 1000:
            rng = np.random.default_rng(t)
            z = z + 0.02 * rng.normal(size=n)
            z = z - z.mean()

    return best_a, best_norm_score


def _lbfgs_run(a_init, deadline, max_wall_time, top_k=0):
    """Run L-BFGS with beta continuation on a normalized sequence."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    target_sum = float(np.sqrt(2.0 * n))

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if max_wall_time is None:
        max_wall_time = max(1.0, deadline - time.monotonic())

    beta_starts = [0.03, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    betas = [b * max(max_c0, 1e-10) for b in beta_starts]

    per_level = max_wall_time / len(betas)

    for beta in betas:
        if time.monotonic() > deadline:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        # Objective and gradient for L-BFGS (working with raw a scaling)
        def obj(z_val):
            a_val = _softmax_fn(z_val) * target_sum
            return _z_objective_smooth(z_val, beta, n, top_k)

        def grad(z_val):
            a_val = _softmax_fn(z_val) * target_sum
            return _z_gradient_smooth(z_val, beta, n, top_k)

        try:
            result = minimize(
                obj, z, jac=grad,
                method='L-BFGS-B',
                options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 40, 'maxfun': 1000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new) * target_sum
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _adam_polish(a_init, deadline, lr=0.03, top_k=5, n_max_iter=50000):
    """Adam optimizer in logit space with adaptive top-k."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    target_sum = float(np.sqrt(2.0 * n))

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    cur_k = top_k

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_fn(z) * target_sum
        grad = _z_gradient_smooth(z, 0.001 * target_sum, n, cur_k)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        if t % 50 == 0:
            a_check = _softmax_fn(z) * target_sum
            sc = _score_norm_fn(a_check)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_check.copy()

        if t % 500 == 0:
            cur_k = min(cur_k + 2, 50)
            if best_score < 1.51:
                lr = max(0.005, lr * 0.98)

    return best_a, best_score


def _project_onto_simplex(a, s):
    """Project a onto {x >= 0, sum(x) = s}."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    # Find t such that sum(max(a - t, 0)) = s
    a_sorted = np.sort(a)[::-1]
    cumsum = np.cumsum(a_sorted)
    rho = np.searchsorted(cumsum - np.arange(1, n + 1), s, side='right')
    if rho == 0:
        rho = 1
    t = (cumsum[rho - 1] - s) / rho
    proj = np.maximum(a - t, 0)
    # Normalize to exactly sum = s
    ps = float(proj.sum())
    if ps > 0:
        proj = proj * (s / ps)
    return proj


def _exact_active_set_opt(a_init, deadline, max_iter=20000):

            a_try = a + step * direction
            a_try = _project_onto_simplex(a_try, target_sum)
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < cm - 0.0001 * step * float(-np.dot(direction, G)):
                a = a_try
                break
            step *= 0.5

        if it % 500 == 0 and cur_k < 80:
            cur_k = min(cur_k + 5, 80)

        # Periodic noisy restart
        if it % 3000 == 0 and it > 1000:
            rng = np.random.default_rng(it)
            a = _project_onto_simplex(a + 0.01 * target_sum * rng.normal(size=n), target_sum)

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _make_gaussian_profile(n, center, width):
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    return a / a.sum()


def _make_bimodal_profile(n, c1, c2, w1, w2, mix):
    t = np.arange(n, dtype=np.float64)
    a = mix * np.exp(-0.5 * ((t - c1) / w1) ** 2) + (1 - mix) * np.exp(-0.5 * ((t - c2) / w2) ** 2)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization for minimizing the autocorrelation score.
    Uses L-BFGS with logsumexp, active-set projected gradient, and
    multi-resolution at n=8192.
    """
    global height_sequence_1

    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Phase 0: Initialization ---
    print("=== Phase 0: Initialization ===")
    n_target = 8192

    # Try parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm_fn(parent)
                update_best(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent: {e}")

    candidates = []

    # Interpolate parent to target length
    if parent is not None:
        p_interp = _interp_profile(parent, n_target)
        sc = _score_norm_fn(p_interp)
        candidates.append((sc, p_interp, n_target, "parent_interp"))
        update_best(p_interp, sc, "parent_interp")

    # Generate diverse profiles
    profiles = []
    # Gaussian profiles
    for center_frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for width_frac in [0.1, 0.15, 0.2, 0.25, 0.3]:
            a = _make_gaussian_profile(n_target, center_frac * n_target, width_frac * n_target)
            profiles.append(a)
    # Bimodal
    for mix in [0.3, 0.5, 0.7]:
        a = _make_bimodal_profile(n_target, 0.3 * n_target, 0.7 * n_target,
                                   0.15 * n_target, 0.15 * n_target, mix)
        profiles.append(a)
    # Linear ramp
    for slope in [0.5, 1.0, 2.0, 3.0]:
        t = np.linspace(0, 1, n_target)
        a = 1.0 + slope * t
        profiles.append(a / a.sum())
        a = 1.0 + slope * (1 - t)
        profiles.append(a / a.sum())
    # Exponential
    for alpha in [0.001, 0.002, 0.005, 0.01]:
        a = np.exp(-alpha * np.arange(n_target))
        profiles.append(a / a.sum())
        a = np.exp(-alpha * (n_target - 1 - np.arange(n_target)))
        profiles.append(a / a.sum())

    phase0_end = min(deadline, t_start + 30.0)
    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p, n_target, f"prof{idx}"))
        if sc < best_score:
            update_best(p, sc, f"prof{idx}")

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 0 done: best={best_score:.10f}, {len(candidates)} candidates")

    # --- Phase 1: L-BFGS on top candidates ---
    print("=== Phase 1: L-BFGS ===")
    phase1_end = min(deadline, t_start + 400.0)
    count = 0
    seen = set()

    for sc, p, pn, label in candidates:
        if time.monotonic() > phase1_end or count >= 5:
            break
        if sc > best_score * 1.05:
            break
        key = f"{label}"
        if key in seen:
            continue
        seen.add(key)

        remaining = phase1_end - time.monotonic()
        if remaining < 30:
            break
        per_time = min(80.0, remaining / max(1, 5 - count))
        dl = min(phase1_end, time.monotonic() + per_time)
        print(f"  L-BFGS {label} (n={pn}), budget={per_time:.0f}s, start_score={sc:.10f}")

        p_best, sc_best = _lbfgs_run(p, dl, max_wall_time=per_time, top_k=0)
        update_best(p_best, sc_best, f"lbfgs_{label}")
        count += 1

    print(f"  Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Active-set gradient descent ---
    print("=== Phase 2: Active-set gradient descent ===")
    phase2_end = min(deadline, t_start + 900.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        remaining = phase2_end - time.monotonic()
        print(f"  Active-set GD, budget={remaining:.0f}s")
        p_best, sc_best = _active_set_grad_descent(
            best_seq, deadline=phase2_end,
            max_iter=30000, top_k_start=10, top_k_max=200,
            momentum=0.9
        )
        update_best(p_best, sc_best, "active_gd")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Adam polish ---
    print("=== Phase 3: Adam polish ===")
    phase3_end = min(deadline, t_start + 1200.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        remaining = phase3_end - time.monotonic()
        if remaining > 30:
            dl = min(phase3_end - 5, time.monotonic() + remaining)
            p_best, sc_best = _adam_polish(best_seq, dl, lr=0.03, top_k=10,
                                           n_max_iter=40000)
            update_best(p_best, sc_best, "adam1")
            # Second pass with smaller lr
            if time.monotonic() < phase3_end - 30:
                dl2 = min(phase3_end - 5, time.monotonic() + (phase3_end - time.monotonic()) * 0.6)
                p_best2, sc_best2 = _adam_polish(best_seq, dl2, lr=0.01, top_k=20,
                                                 n_max_iter=20000)
                update_best(p_best2, sc_best2, "adam2")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Exact active-set optimization ---
    print("=== Phase 4: Exact active-set optimization ===")
    phase4_end = min(deadline, t_start + 1500.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        remaining = phase4_end - time.monotonic()
        if remaining > 30:
            p_best, sc_best = _exact_active_set_opt(best_seq, deadline=phase4_end,
                                                    max_iter=15000)
            update_best(p_best, sc_best, "exact_as")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multi-restart ---
    print("=== Phase 5: Multi-restart ===")
    phase5_end = min(deadline, t_start + 1650.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        n_best = len(best_seq)
        n_restarts = 3
        per_restart = min(60.0, (phase5_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 15:
                break
            noise_levels = [0.03, 0.05, 0.08]
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_levels[i] * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z) * float(np.sqrt(2.0 * n_best))

            dl = min(phase5_end - 10, time.monotonic() + per_restart * 0.6)
            p_m, sc_m = _lbfgs_run(p_start, dl, max_wall_time=per_restart * 0.5, top_k=5)
            update_best(p_m, sc_m, f"r{i}_lbfgs")
            if time.monotonic() < phase5_end - 10:
                dl2 = min(phase5_end - 10, time.monotonic() + per_restart * 0.35)
                p_a, sc_a = _active_set_grad_descent(p_m, dl2, max_iter=10000,
                                                      top_k_start=10, top_k_max=50,
                                                      momentum=0.9)
                update_best(p_a, sc_a, f"r{i}_gd")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Final polish ---
    print("=== Phase 6: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            dl = min(deadline - 5, time.monotonic() + min(40.0, remaining * 0.6))
            p_best, sc_best = _adam_polish(best_seq, dl, lr=0.005, top_k=30,
                                           n_max_iter=15000)
            update_best(p_best, sc_best, "final_adam")

        if time.monotonic() < deadline - 10:
            dl2 = min(deadline - 5, time.monotonic() + min(30.0, deadline - time.monotonic() - 5))
            p_best2, sc_best2 = _lbfgs_run(best_seq, dl2,
                                           max_wall_time=max(5.0, dl2 - time.monotonic()),
                                           top_k=10)
            update_best(p_best2, sc_best2, "final_lbfgs")

    # --- Final: Validate and return ---
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = _make_gaussian_profile(8192, 0.5 * 8192, 0.2 * 8192)
            final_score = _score_fn(final_seq)

        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")

        # Validate with evaluate_sequence
        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 10000:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"  evaluate_sequence check failed: {e}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_gaussian_profile(8192, 0.5 * 8192, 0.2 * 8192)
    print(f"  FALLBACK: n=8192, score={_score_norm_fn(fallback):.10f}")
    return [float(x) for x in fallback]
