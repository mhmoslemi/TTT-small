import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _langs_rule_grad(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _langs_obj(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _compute_power_grad(z, p):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    c_clipped = np.maximum(c, 1e-30)
    m = len(c)
    c_p = c_clipped ** (p - 1)
    sum_c_p = float(np.sum(c_p))
    if sum_c_p < 1e-300:
        return np.zeros(n)
    w = c_p / sum_c_p
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c_clipped))
    grad_a = 4.0 * n * (h - W)
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)
    if not np.all(np.isfinite(grad_z)):
        return np.zeros(n)
    return grad_z


def _power_obj(z, p):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    c_clipped = np.maximum(c, 1e-30)
    val = (np.sum(c_clipped ** p) / len(c)) ** (1.0 / p)
    return 2.0 * n * val


def _topk_subgradient_step(a, K, lr):
    n = len(a)
    c = _full_convolve(a, a)
    topK = np.argpartition(c, -K)[-K:] if K < len(c) else np.arange(len(c))
    w = np.zeros_like(c)
    w[topK] = 1.0 / K
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * (h - W)
    a_new = _project_simplex(a - lr * grad)
    return a_new, c, topK


def _bb_step_size(a_curr, a_prev, g_curr, g_prev):
    s = a_curr - a_prev
    y = g_curr - g_prev
    sy = np.dot(s, y)
    yy = np.dot(y, y)
    if yy < 1e-30 or sy <= 1e-30:
        return 0.01
    return max(1e-8, min(1.0, sy / yy))


def _multiplicative_update(a, K, lr):
    n = len(a)
    c = _full_convolve(a, a)
    topK = np.argpartition(c, -K)[-K:] if K < len(c) else np.arange(len(c))
    w = np.zeros_like(c)
    w[topK] = 1.0 / K
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    contrib = h - W
    a_new = a * np.exp(-lr * contrib)
    a_new = np.maximum(a_new, 1e-30)
    a_new = a_new / a_new.sum()
    return a_new


def _make_profiles(n):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    profiles = []
    # Linear
    a = 1.0 + 1.5 * (t - 0.5)
    profiles.append(np.maximum(a, 1e-12) / np.maximum(a, 1e-12).sum())
    # Quadratic
    a = 1.0 + 0.8 * u + 0.3 * u ** 2
    profiles.append(np.maximum(a, 1e-12) / np.maximum(a, 1e-12).sum())
    # Cubic
    a = 1.0 + 0.6 * u - 0.3 * u ** 3
    profiles.append(np.maximum(a, 1e-12) / np.maximum(a, 1e-12).sum())
    # Sinc-like
    x = u * np.pi * 2
    a = np.sinc(x / np.pi)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    # Raised cosine
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    # Exponential decay from center
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (n * 0.2))
    profiles.append(a / a.sum())
    # Two-hump
    a = np.exp(-2.0 * (t - 0.25) ** 2) + np.exp(-2.0 * (t - 0.75) ** 2)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    # Uniform
    profiles.append(np.ones(n) / n)
    return profiles

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        print(f"Parent loaded: n={n}, score={_score_fn(parent):.10f}")
    else:
        n = 4096
        parent = _make_profiles(n)[0]
        print(f"No parent, using default n={n}")

    # Phase 0: Quick evaluation of diverse candidates (30s)
    phase0_end = min(deadline, t_start + 30.0)
    candidates = []
    sc_parent = _score_norm(parent)
    candidates.append((sc_parent, parent.copy(), "parent"))
    update_best(parent, sc_parent, "parent")

    # Try different n values
    for sn in [2048, 4096, 8192]:
        if time.monotonic() > phase0_end:
            break
        if sn == n:
            continue
        p_interp = _interp_profile(parent, sn)
        sc = _score_norm(p_interp)
        candidates.append((sc, p_interp, f"interp_n{sn}"))

    # Try different profiles
    for sn in [2048, 4096]:
        if time.monotonic() > phase0_end:
            break
        profiles = _make_profiles(sn)
        for idx, p in enumerate(profiles):
            sc = _score_norm(p)
            candidates.append((sc, p, f"prof{idx}_n{sn}"))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: {len(candidates)} candidates, best={best_score:.10f} (n={len(best_seq)})")

    # Phase 1: L-BFGS log-sum-exp on best candidate (350s)
    phase1_end = min(deadline, t_start + 380.0)
    if best_seq is not None and time.monotonic() < phase1_end:
        per_time = min(340.0, phase1_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _lbfgs_optimize(best_seq, phase1_end, per_time, method='langs')
            update_best(p_best, sc_best, "lbfgs_langs")

    # Phase 2: L-BFGS power-smoothed (250s)
    phase2_end = min(deadline, t_start + 630.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        per_time = min(240.0, phase2_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _lbfgs_optimize(best_seq, phase2_end, per_time, method='power')
            update_best(p_best, sc_best, "lbfgs_power")

    # Phase 3: Top-K subgradient with BB step (250s)
    phase3_end = min(deadline, t_start + 880.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        per_time = min(240.0, phase3_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _topk_bb_optime(best_seq, min(phase3_end, time.monotonic() + per_time), K=30)
            update_best(p_best, sc_best, "topk_bb")

    # Phase 4: Nesterov with aggressive continuation (250s)
    phase4_end = min(deadline, t_start + 1130.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        per_time = min(240.0, phase4_end - time.monotonic())
        if per_time > 20:
            beta_fracs = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6]
            p_best, sc_best = _nesterov_opt(best_seq, min(phase4_end, time.monotonic() + per_time),
                                            beta_fracs, lr_init=0.01, mom_init=0.9)
            update_best(p_best, sc_best, "nesterov")

    # Phase 5: Multiplicative polish (150s)
    phase5_end = min(deadline, t_start + 1280.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        per_time = min(140.0, phase5_end - time.monotonic())
        if per_time > 10:
            p_best, sc_best = _multiplicative_polish(best_seq, min(phase5_end, time.monotonic() + per_time),
                                                     K=50, lr_init=0.03)
            update_best(p_best, sc_best, "multiplicative")

    # Phase 6: Adam fine polish (150s)
    phase6_end = min(deadline, t_start + 1430.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        per_time = min(140.0, phase6_end - time.monotonic())
        if per_time > 10:
            p_best, sc_best = _adam_opt(best_seq, min(phase6_end, time.monotonic() + per_time),
                                        lr=0.02, beta_frac=0.003)
            update_best(p_best, sc_best, "adam")

    # Phase 7: Multi-restart with perturbations (200s)
    phase7_end = min(deadline, t_start + 1630.0)
    if best_seq is not None and time.monotonic() < phase7_end:
        n_best = len(best_seq)
        n_restarts = 5
        per_restart = max(5.0, (phase7_end - time.monotonic()) / n_restarts - 5)

        for i in range(n_restarts):
            if time.monotonic() > phase7_end - 5:
                break
            noise_level = 0.03 + 0.02 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise = rng.normal(size=n_best) * noise_level
            # Add structured noise: smooth perturbation
            noise_smooth = np.convolve(noise, np.ones(7) / 7, mode='same')
            z = z + 0.5 * noise + 0.5 * noise_smooth
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            # Quick L-BFGS + Nesterov
            per_dl = min(phase7_end - 3, time.monotonic() + per_restart * 0.5)
            if per_dl - time.monotonic() > 5:
                p_best, sc_best = _lbfgs_optimize(p_start, per_dl, per_restart * 0.4, method='langs')
                update_best(p_best, sc_best, f"restart{i}_lbfgs")

            per_dl2 = min(phase7_end - 2, time.monotonic() + per_restart * 0.4)
            if per_dl2 - time.monotonic() > 5:
                bf = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
                p_best, sc_best = _nesterov_opt(p_start, per_dl2, bf, lr_init=0.008, mom_init=0.85)
                update_best(p_best, sc_best, f"restart{i}_nest")

    # Phase 8: Final ultra-fine polish (remaining - 25s buffer)
    phase8_end = deadline
    if best_seq is not None and time.monotonic() < phase8_end:
        remaining = phase8_end - time.monotonic()
        if remaining > 20:
            # Nesterov with very small beta
            per_time = min(60.0, remaining - 15)
            if per_time > 10:
                bf = [1e-5, 5e-6, 2e-6, 1e-6, 5e-7]
                p_best, sc_best = _nesterov_opt(best_seq, min(phase8_end - 10, time.monotonic() + per_time),
                                                bf, lr_init=0.003, mom_init=0.95)
                update_best(p_best, sc_best, "final_nesterov")

            if time.monotonic() < phase8_end - 5:
                per_time2 = min(30.0, phase8_end - time.monotonic() - 3)
                if per_time2 > 5:
                    p_best, sc_best = _adam_opt(best_seq, min(phase8_end - 2, time.monotonic() + per_time2),
                                                lr=0.005, beta_frac=0.001)
                    update_best(p_best, sc_best, "final_adam")

    print(f"All phases done. Best score: {best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profiles(4096)[0]

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence call issue: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profiles(4096)[0]
    print(f"FALLBACK: n=4096, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
