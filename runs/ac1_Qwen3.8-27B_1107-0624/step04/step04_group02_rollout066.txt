"""
Multi-phase optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses binomial-sequence init, L-BFGS logsumexp continuation,
active-set subgradient, sequential coordinate descent, local search.
"""
import numpy as np
from scipy.optimize import minimize
import time
import functools

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _binomial_sequence(n):
    b = np.zeros(n, dtype=np.float64)
    b[0] = 1.0
    for k in range(1, n):
        b[k] = b[k - 1] * (4.0 * k - 2.0) / (4.0 * k)
    b = np.maximum(b, 0.0)
    return b / b.sum()

def _make_profile(n, kind, param=0.0):
    t = np.linspace(0.0, 1.0, n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'linear':
        a = 1.0 + param * t
    elif kind == 'quadratic':
        u = 2.0 * t - 1.0
        a = 1.0 + param * u * u
    elif kind == 'gaussian':
        a = np.exp(-param * (t - 0.5) ** 2)
    elif kind == 'raised_cosine':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5)))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _obj_lse(z, n, beta):
    a = _softmax(z)
    c = _convolve(a, a)
    cm = float(c.max())
    c_shift = (c - cm) / beta
    e = np.exp(c_shift)
    S = float(e.sum())
    if S < 1e-300:
        return 2.0 * n * cm
    return 2.0 * n * (cm + beta * float(np.log(S)))

def _grad_lse(z, n, beta):
    a = _softmax(z)
    c = _convolve(a, a)
    cm = float(c.max())
    c_shift = (c - cm) / beta
    e = np.exp(c_shift)
    S = float(e.sum())
    if S < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / S
    a_rev = a[::-1]
    conv_w = _convolve(w, a_rev)
    g1_raw = conv_w[n - 1:2 * n - 1]
    W_dot_c = float(np.dot(w, c))
    gz = 4.0 * n * a * (g1_raw - W_dot_c)
    if not np.all(np.isfinite(gz)):
        return np.zeros_like(z)
    return gz

def _run_lbfgs(a_init, deadline, beta_fractions=None):

        weights = np.exp(weights)
