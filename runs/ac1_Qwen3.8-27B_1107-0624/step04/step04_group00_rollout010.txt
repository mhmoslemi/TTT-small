"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using multi-phase optimization:
1. L-BFGS with geometric log-sum-exp continuation (halving beta each phase)
2. Active-set gradient descent tracking the argmax of convolution
3. Peak balancing to equalize near-maximal peaks
4. Fine polish with exact-max gradient steps
Key innovation: geometric beta schedule + active-set tracking avoids
the plateau problem of fixed-beta log-sum-exp, and peak balancing
exploits the structure that optimal solutions have multiple equal peaks.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    shifted = (c - cm) / beta
    lse = cm + beta * float(np.log(np.sum(np.exp(shifted))))
    return 2.0 * n * lse


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _exact_max_grad(z, n, top_k=1):
    """Compute gradient of max of convolution using top-k peaks."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    # Find top-k peaks
    top_indices = np.argsort(c)[::-1][:top_k]
    # If peaks are close, use weighted average
    top_vals = c[top_indices]
    weights = top_vals - cm  # all <= 0, top one is 0
    # Use a soft weighting: weight proportional to how close to max
    sigma = max(0.01 * cm, 1e-10)
    w_soft = np.exp((top_vals - cm) / sigma)
    w_soft = w_soft / w_soft.sum()
    # Weighted sum of gradients
    a_rev = a[::-1]
    total_h = np.zeros(n)
    for k_idx, wt in zip(top_indices, w_soft):
        k = int(k_idx)
        # d(conv(a,a)_k)/da_j = 2*a_{k-j} for valid j
        h = np.zeros(n)
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                h[j] = 2.0 * a[kj]
        total_h += wt * h
    W = float(np.dot(w_soft, c[top_indices]))
    grad = 4.0 * n * a * (total_h - W)
    if not np.all(np.isfinite(grad)):
        # Fallback to single peak
        k_star = int(np.argmax(c))
        h = np.zeros(n)
        for j in range(n):
            kj = k_star - j
            if 0 <= kj < n:
                h[j] = 2.0 * a[kj]
        W = cm
        grad = 4.0 * n * a * (h - W)
    return grad


def _active_set_gd(a_init, deadline, rng, lr_init=0.1, top_k=2, max_iter=100000):
    """Active-set gradient descent with Adam on exact max gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        grad = _exact_max_grad(z, n, top_k=top_k)
        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -70, 70)
        z_new = z_new - z_new.mean()
        a_new = _softmax_fn(z_new)

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            z = z_new.copy()
        else:
            # Rejected - halve learning rate
            lr = max(0.001, lr * 0.7)

        # Slowly increase lr back

            z = z + 0.02 * rng.normal(size=n)

            lr = lr_init * 0.5

        if gap < 0.01 * cm:

            c_no_max = c.copy()

    # Geometric schedule: halve beta each phase

    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline - 5:
            break

        remaining = n_phases - idx
        time_per_phase = max(5.0, (deadline - time.monotonic()) / max(1, remaining))
        time_per_phase = min(time_per_phase, 120.0)
        phase_dl = time.monotonic() + time_per_phase

        try:

                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,

            )
            if result.x is not None and np.all(np.isfinite(result.x)):

        except Exception as ex:

        profiles.append(a / a.sum())

    # Power-law
    for p in [0.5, 1.0, 2.0, 3.0, -0.5]:
        a = np.maximum(1.0 + p * (t - 0.5), 1e-15)
        a = a ** 2  # square to make smoother
        profiles.append(a / a.sum())

    # Asymmetric bell
    for center in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for width in [0.2, 0.3, 0.4, 0.5]:
            a = np.exp(-((t - center) / width) ** 2)
            a = np.maximum(a, 1e-15)
            profiles.append(a / a.sum())

    # Sigmoid-based
    for center in [0.3, 0.5, 0.7]:
        for steep in [5, 10, 20]:
            a = 1.0 / (1.0 + np.exp(-steep * (t - center)))
            a = np.maximum(a, 1e-15)
            profiles.append(a / a.sum())

    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search to minimize 2*n*max(conv(a,a))/sum(a)^2.
    Uses geometric L-BFGS continuation, active-set GD, and peak balancing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}, actual={_score_fn(parent):.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n = 4096
        parent = _fourier_profile(n, [2.0])[0]
        update_best(parent, _score_norm_fn(parent), "default")

    n_best = len(best_seq)

    # --- Phase 1: Quick profile exploration (60s) ---
    phase1_end = min(deadline, t_start + 60.0)
    print("Phase 1: Profile exploration")

    profiles = _make_diverse_profiles(n_best)
    for idx, p in enumerate(profiles[:12]):
        sc = _score_norm_fn(p)
        update_best(p, sc, f"prof{idx}")

    # Also try interpolating parent to different lengths
    for n_try in [2048, 4096, 8192]:
        if n_try == n_best:
            continue
        p_interp = _interp_profile(best_seq, n_try)
        sc = _score_norm_fn(p_interp)
        update_best(p_interp, sc, f"interp_n{n_try}")

    print(f"  Phase 1 done: best={best_score:.10f} t={time.monotonic()-t_start:.0f}s")

    # --- Phase 2: L-BFGS continuation from best (600s) ---
    phase2_end = min(deadline, t_start + 660.0)
    print("Phase 2: L-BFGS continuation")

    n_use = len(best_seq)
    a_start = best_seq.copy()
    a_res, sc_res = _lbfgs_continuation(a_start, phase2_end, rng)
    update_best(a_res, sc_res, "lbfgs_main")

    print(f"  Phase 2 done: best={best_score:.10f} t={time.monotonic()-t_start:.0f}s")

    # --- Phase 3: Active-set GD (400s) ---
    phase3_end = min(deadline, t_start + 1060.0)
    print("Phase 3: Active-set GD")

    if time.monotonic() < phase3_end and best_seq is not None:
        # Multiple rounds with different top_k
        for top_k in [1, 2, 3]:
            if time.monotonic() > phase3_end - 60:
                break
            per_dl = min(phase3_end - 10, time.monotonic() + 120.0)
            a_res, sc_res = _active_set_gd(best_seq, per_dl, rng, lr_init=0.08, top_k=top_k)
            update_best(a_res, sc_res, f"active_k{top_k}")

    print(f"  Phase 3 done: best={best_score:.10f} t={time.monotonic()-t_start:.0f}s")

    # --- Phase 4: Peak balancing (250s) ---
    phase4_end = min(deadline, t_start + 1310.0)
    print("Phase 4: Peak balancing")

    if time.monotonic() < phase4_end and best_seq is not None:
        per_dl = min(phase4_end - 10, time.monotonic() + 230.0)
        a_res, sc_res = _peak_balance(best_seq, per_dl, rng, n_steps=200)
        update_best(a_res, sc_res, "peak_balance")

    print(f"  Phase 4 done: best={best_score:.10f} t={time.monotonic()-t_start:.0f}s")

    # --- Phase 5: L-BFGS restarts (200s) ---
    phase5_end = min(deadline, t_start + 1510.0)
    print("Phase 5: L-BFGS restarts")

    if time.monotonic() < phase5_end and best_seq is not None:
        per_dl = min(phase5_end - 10, time.monotonic() + 180.0)
        a_res, sc_res = _lbfgs_restart(best_seq, per_dl, rng, n_restarts=3)
        update_best(a_res, sc_res, "lbfgs_restarts")

    print(f"  Phase 5 done: best={best_score:.10f} t={time.monotonic()-t_start:.0f}s")

    # --- Phase 6: Final fine polish (remaining) ---
    phase6_end = deadline
    print("Phase 6: Final polish")

    if time.monotonic() < phase6_end and best_seq is not None:
        # Very small step active-set GD
        remaining_time = phase6_end - time.monotonic()
        half = remaining_time / 2

        if half > 20:
            per_dl = time.monotonic() + half
            a_res, sc_res = _active_set_gd(best_seq, per_dl, rng, lr_init=0.02, top_k=2, max_iter=200000)
            update_best(a_res, sc_res, "final_active")

        if time.monotonic() < phase6_end - 10:
            per_dl = min(phase6_end - 5, time.monotonic() + (phase6_end - time.monotonic()) * 0.5)
            a_res, sc_res = _peak_balance(best_seq, per_dl, rng, n_steps=300)
            update_best(a_res, sc_res, "final_balance")

    print(f"  Phase 6 done: best={best_score:.10f} t={time.monotonic()-t_start:.0f}s")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _fourier_profile(256, [1.5])[0]

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        # Validate with evaluator if length allows
        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _fourier_profile(256, [1.5])[0]
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
