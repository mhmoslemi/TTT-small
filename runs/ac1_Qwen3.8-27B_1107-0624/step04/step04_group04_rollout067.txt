"""
Step-function coefficient search for minimizing
    2 * n * max(convolve(a, a)) / sum(a)^2
using adaptive logit-space Adam, active-lag equalization, cross-length
refinement at n=4096/8192, and a final smoothed L-BFGS polish.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _autocorr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    out_len = 2 * n - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:out_len]
    return c


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        a = np.ones_like(a, dtype=np.float64)
        s = float(a.size)
    return a / s


def _score_norm(a):
    """Objective assuming sum(a)=1. Use only on normalized sequences."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if n == 0:
        return np.inf
    c = _autocorr(a)
    if c.size == 0:
        return np.inf
    cm = float(c.max())
    if not np.isfinite(cm):
        return np.inf
    return float(2.0 * n * cm)


def _score_general(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.inf
    c = _autocorr(a)
    cm = float(c.max())
    if not np.isfinite(cm):
        return np.inf
    return float(2.0 * n * cm / (s * s))


def _softmax_vec(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return z
    z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones_like(z, dtype=np.float64) / float(len(z))
    return e / s


def _interp_profile(a, m):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = len(a)
    m = int(m)
    if m <= 0:
        return a
    if n0 == 0:
        return np.ones(m, dtype=np.float64)
    if n0 == m:
        return _normalize(a)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, m)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 0.0)
    return _normalize(p)


def _tilt_profile(a, gamma):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if n == 0:
        return a
    t = np.linspace(0.0, 1.0, n)
    b = a * np.exp(float(gamma) * (t - 0.5))
    b = np.maximum(b, 0.0)
    return _normalize(b)


def _profile_geometric(n, p):
    n = int(max(1, n))
    p = float(p)
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-p * x / float(n))
    a = np.maximum(a, 1e-300)
    return a / a.sum()


def _profile_parabolic(n, center, width):
    n = int(max(1, n))
    x = np.linspace(0.0, 1.0, n)
    u = (x - float(center)) / max(1e-12, float(width))
    a = np.maximum(0.0, 1.0 - u * u)
    return _normalize(a)


def _candidate_key(item):
    return item[0]


def _top_indices(c, m):
    c = np.asarray(c, dtype=np.float64).ravel()
    L = len(c)
    m = int(m)
    if m <= 0 or L == 0:
        return np.array([], dtype=np.int64)
    if m >= L:
        return np.arange(L - 1, -1, -1, dtype=np.int64)
    if m == 1:
        return np.array([int(np.argmax(c))], dtype=np.int64)
    return np.argpartition(c, L - m)[-m:].astype(np.int64)


def _top_weights(c, m, beta_ratio):
    idx = _top_indices(c, m)
    if idx.size == 0:
        return idx, np.array([], dtype=np.float64)
    if idx.size == 1:
        return idx, np.array([1.0], dtype=np.float64)
    cmax = float(c.max())
    beta = max(1e-300, float(beta_ratio) * max(1e-300, cmax))
    vals = (c[idx] - cmax) / beta
    vals = vals - float(np.max(vals))
    w = np.exp(vals)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 1e-300:
        w = np.ones(len(idx), dtype=np.float64)
    else:
        w = w / s
    return idx, w


def _h_top_direct(a, idx, wvals):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    h = np.zeros(n, dtype=np.float64)
    idx = np.asarray(idx, dtype=np.int64).ravel()
    wvals = np.asarray(wvals, dtype=np.float64).ravel()
    m = min(len(idx), len(wvals))
    if m == 0:
        return h
    for j in range(m):
        val = float(wvals[j])
        if val == 0.0 or not np.isfinite(val):
            continue
        k = int(idx[j])
        lo = 0 if k < n else k - (n - 1)
        hi = n - 1 if k >= n else k
        if lo < 0:
            lo = 0
        if hi >= n:
            hi = n - 1
        if lo > hi:
            continue
        h[lo:hi + 1] += val * a[k - hi:k - lo + 1][::-1]
    return h


def _active_h_direct(a, idx, ridge_rel=1e-6):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    idx = np.asarray(idx, dtype=np.int64).ravel()
    m = len(idx)
    if m == 0:
        return np.zeros(n, dtype=np.float64)
    if m == 1:
        return _h_top_direct(a, idx, np.array([1.0], dtype=np.float64))

    B = np.zeros((n, m), dtype=np.float64)
    for j in range(m):
        k = int(idx[j])
        lo = 0 if k < n else k - (n - 1)
        hi = n - 1 if k >= n else k
        if lo < 0:
            lo = 0
        if hi >= n:
            hi = n - 1
        if lo > hi:
            continue
        B[lo:hi + 1, j] = a[k - hi:k - lo + 1][::-1]

    G = B.T @ B
    scale = float(np.trace(G)) / float(m)
    if not np.isfinite(scale) or scale <= 0.0:
        scale = 1.0
    ridge = max(1e-12, float(ridge_rel) * scale)
    G = G + ridge * np.eye(m, dtype=np.float64)
    ones = np.ones(m, dtype=np.float64)

    try:
        y = np.linalg.solve(G, ones)
    except Exception:
        try:
            y = np.linalg.lstsq(G, ones, rcond=None)[0]
        except Exception:
            y = ones / float(m)

    if not np.all(np.isfinite(y)):
        y = ones / float(m)

    # Keep the active weights nonnegative; this is a heuristic active-set blend.
    y = np.where(y > 0.0, y, 0.0)
    s = float(y.sum())
    if not np.isfinite(s) or s <= 1e-300:
        y = ones / float(m)

    h = B @ y
    if not np.all(np.isfinite(h)):
        h = _h_top_direct(a, idx, np.ones(m, dtype=np.float64) / float(m))
    return h


def _adaptive_opt(a0, deadline, lr=0.08, max_top=32, active_every=40,
                  ridge_rel=1e-6, use_active=True, seed=0, label=""):
    """
    Long-running logit-space Adam with top-lag softmax gradient and
    periodic active-lag equalization directions. Tracks the best normalized
    sequence found.
    """
    a = _normalize(a0)
    n = len(a)
    if n == 0:
        return a, np.inf

    z = np.log(np.maximum(a, 1e-300))
    z = z - float(z.mean())

    best_a = a.copy()
    best_score = _score_norm(a)
    if not np.isfinite(best_score):
        best_score = 1e100

    m_mom = np.zeros(n, dtype=np.float64)
    v_mom = np.zeros(n, dtype=np.float64)
    t = 0
    it = 0
    no_improve = 0
    lr_cur = float(lr)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    last_print = 0.0

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1
        t += 1

        a = _softmax_vec(z)
        c = _autocorr(a)
        cmax = float(c.max()) if c.size else 0.0
        score = 2.0 * n * cmax

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 1500:
            lr_cur = max(0.005, lr_cur * 0.75)
            m_mom *= 0.5
            v_mom *= 0.5

        # Adaptive number of active lags.
        base_top = 1 + it // 800
        imp_top = 1 + no_improve // 2500
        top_m = int(max(1, min(max_top, max(base_top, imp_top))))

        if use_active and (it % active_every == 0) and top_m >= 2 and top_m <= 24:
            idx = _top_indices(c, top_m)
            try:
                h = _active_h_direct(a, idx, ridge_rel)
            except Exception:
                h = _h_top_direct(a, idx, np.ones(max(1, len(idx)), dtype=np.float64) / max(1, len(idx)))
        else:
            idx, w = _top_weights(c, top_m, 0.5)
            h = _h_top_direct(a, idx, w)

        hmax = float(np.max(np.abs(h))) if len(h) else 0.0
        if not np.isfinite(hmax) or hmax <= 1e-300:
            h = np.zeros_like(a, dtype=np.float64)
        else:
            h = h / hmax

        C = float(np.dot(h, a))
        if not np.isfinite(C):
            C = 0.0
        g = a * (h - C)
        g = np.where(np.isfinite(g), g, 0.0)

        m_mom = 0.9 * m_mom + 0.1 * g
        v_mom = 0.999 * v_mom + 0.001 * (g * g)
        b1_t = 1.0 - (0.9 ** t)
        b2_t = 1.0 - (0.999 ** t)
        mh = m_mom / b1_t
        vh = v_mom / b2_t
        z = z - lr_cur * mh / (np.sqrt(vh) + 1e-8)
        z = np.where(np.isfinite(z), z, 0.0)
        z = np.clip(z, -80.0, 80.0)
        z -= float(np.mean(z))

        # Periodic reheats from the current best to avoid plateaus.
        if no_improve > 9000:
            base = np.log(np.maximum(best_a, 1e-300))
            base -= float(base.mean())
            z = base + 0.06 * rng.standard_normal(n)
            z = np.clip(z, -80.0, 80.0)
            z -= float(np.mean(z))
            m_mom = np.zeros(n, dtype=np.float64)
            v_mom = np.zeros(n, dtype=np.float64)
            lr_cur = 0.05
            no_improve = 0
        elif no_improve > 3000:
            z = z + 0.02 * rng.standard_normal(n)
            z = np.clip(z, -80.0, 80.0)
            z -= float(np.mean(z))
            no_improve = 0

        now = time.monotonic()
        if (it % 1000 == 0) or (now - last_print >= 30.0):
            print(f"    adapt {label} it={it} score={score:.10f} best={best_score:.10f} top={top_m} lr={lr_cur:.4f}")
            last_print = now

    if best_a.size != n:
        best_a = _normalize(a0)
        best_score = _score_norm(best_a)
    return best_a, best_score


def _lbfgs_objective(z, beta_ratio, top):
    a = _softmax_vec(z)
    n = len(a)
    c = _autocorr(a)
    L = len(c)
    top = int(min(max(1, int(top)), L))
    cmax = float(c.max()) if L else 0.0
    idx = _top_indices(c, top)
    if idx.size == 0:
        return 2.0 * n * cmax
    beta = max(1e-300, float(beta_ratio) * max(1e-300, cmax))
    vals = (c[idx] - cmax) / beta
    vals = vals - float(np.max(vals))
    s = float(np.sum(np.exp(vals)))
    if not np.isfinite(s) or s <= 1e-300:
        return 2.0 * n * cmax
    return float(2.0 * n * (cmax + beta * np.log(s)))


def _lbfgs_gradient(z, beta_ratio, top):
    a = _softmax_vec(z)
    n = len(a)
    c = _autocorr(a)
    L = len(c)
    top = int(min(max(1, int(top)), L))
    cmax = float(c.max()) if L else 0.0
    idx = _top_indices(c, top)
    if idx.size == 0:
        return np.zeros(n, dtype=np.float64)
    if idx.size == 1:
        w = np.array([1.0], dtype=np.float64)
    else:
        beta = max(1e-300, float(beta_ratio) * max(1e-300, cmax))
        vals = (c[idx] - cmax) / beta
        vals = vals - float(np.max(vals))
        w = np.exp(vals)
        s = float(w.sum())
        if not np.isfinite(s) or s <= 1e-300:
            w = np.ones(len(idx), dtype=np.float64)
        else:
            w = w / s
    h = _h_top_direct(a, idx, w)
    C = float(np.dot(h, a))
    if not np.isfinite(C):
        C = 0.0
    g = 4.0 * n * a * (h - C)
    g = np.where(np.isfinite(g), g, 0.0)
    return g


def _lbfgs_polish(a0, deadline, top=8, beta_ratio=0.25, maxfun=80):
    a = _normalize(a0)
    n = len(a)
    if n == 0:
        return a, np.inf
    z = np.log(np.maximum(a, 1e-300))
    z = z - float(z.mean())
    best_a = a.copy()
    best_score = _score_norm(a)
    if not np.isfinite(best_score):
        best_score = 1e100

    ratios = [0.5, 0.25, 0.12, 0.06, 0.03]
    mf = int(max(30, maxfun))

    for br in ratios:
        if time.monotonic() >= deadline:
            break
        try:
            res = minimize(
                _lbfgs_objective,
                z,
                args=(br, int(top)),
                method='L-BFGS-B',
                jac=_lbfgs_gradient,
                options={'maxiter': 60, 'maxfun': mf, 'ftol': 1e-16, 'gtol': 1e-10}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                z_new = np.asarray(res.x, dtype=np.float64).ravel()
                z_new = z_new - float(np.mean(z_new))
                a_new = _softmax_vec(z_new)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass
        mf = int(max(20, int(0.7 * mf)))

    return best_a, best_score

            _consider(_interp_profile(parent, nt), f"parent_n{nt}")
        else:
            _consider(parent, f"parent_n{nt}")

    for p in (0.5, 1.0, 1.8, 2.8):
        if time.monotonic() > cand_end:
            break
        for nt in (4096, 8192):
            _consider(_profile_geometric(nt, p), f"geo_p{p}_n{nt}")

    for center, width in ((0.25, 0.35), (0.35, 0.5), (0.5, 0.6), (0.65, 0.5), (0.75, 0.35)):
        if time.monotonic() > cand_end:
            break
        _consider(_profile_parabolic(4096, center, width), f"parab_{center}_{width}")

    candidates.sort(key=_candidate_key)

    if best_seq is None:
        best_seq = parent.copy()
        best_score = _score_norm(best_seq)

    # Select a 4096-length starting point.
    init4096 = None
    init4096_score = best_score
    for sc, a, label in candidates:
        if a.size == 4096 and sc < init4096_score:
            init4096 = a.copy()
            init4096_score = sc
    if init4096 is None:
        init4096 = _interp_profile(parent, 4096)

    print(f"  Phase 0 done: best={best_score:.10f}, init4096={init4096_score:.10f}")

    # Phase 1: intensive optimization at n=4096.
    phase1_end = min(deadline, t0 + 700.0)
    if time.monotonic() < phase1_end:
        print("=== Phase 1: adaptive opt n=4096 ===")
        a_out, s_out = _adaptive_opt(
            init4096,
            phase1_end,
            lr=0.08,
            max_top=32,
            active_every=40,
            ridge_rel=1e-6,
            use_active=True,
            seed=int(seed) + 1,
            label="4096"
        )
        if np.isfinite(s_out) and s_out < best_score:
            best_score = s_out
            best_seq = a_out.copy()
            print(f"  Phase 1 improved: {best_score:.10f} n={best_seq.size}")

    # Phase 2: lift to n=8192 and optimize.
    phase2_end = min(deadline, t0 + 1400.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        print("=== Phase 2: adaptive opt n=8192 ===")
        init8192 = best_seq.copy() if best_seq.size == 8192 else _interp_profile(best_seq, 8192)
        a_out, s_out = _adaptive_opt(
            init8192,
            phase2_end,
            lr=0.05,
            max_top=24,
            active_every=55,
            ridge_rel=1e-6,
            use_active=True,
            seed=int(seed) + 2,
            label="8192"
        )
        if np.isfinite(s_out) and s_out < best_score:
            best_score = s_out
            best_seq = a_out.copy()
            print(f"  Phase 2 improved: {best_score:.10f} n={best_seq.size}")

    # Phase 3: if 4096 still wins, do short reheats there; otherwise continue on current length.
    phase3_end = min(deadline, t0 + 1650.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        if best_seq.size == 4096:
            print("=== Phase 3: 4096 reheats ===")
            base = np.log(np.maximum(best_seq, 1e-300))
            base -= float(base.mean())
            remaining = phase3_end - time.monotonic()
            for i in range(min(3, max(0, int(remaining // 80)))):
                if time.monotonic() > phase3_end - 20:
                    break
                noise = 0.025 + 0.01 * i
                z = base + noise * rng.standard_normal(base.size)
                z = np.clip(z, -80.0, 80.0)
                z -= float(np.mean(z))
                a0 = _softmax_vec(z)
                dl = min(phase3_end - 10, time.monotonic() + 70.0)
                a_out, s_out = _adaptive_opt(
                    a0,
                    dl,
                    lr=0.055,
                    max_top=16,
                    active_every=45,
                    ridge_rel=1e-6,
                    use_active=True,
                    seed=int(seed) + 10 + i,
                    label=f"reh4k_{i}"
                )
                if np.isfinite(s_out) and s_out < best_score:
                    best_score = s_out
                    best_seq = a_out.copy()
                    print(f"  Phase 3 improved: {best_score:.10f} n={best_seq.size}")
        else:
            print("=== Phase 3: continue current length ===")
            dl = min(phase3_end - 10, time.monotonic() + 180.0)
            a_out, s_out = _adaptive_opt(
                best_seq,
                dl,
                lr=0.035,
                max_top=14,
                active_every=55,
                ridge_rel=1e-6,
                use_active=True,
                seed=int(seed) + 20,
                label="cont"
            )
            if np.isfinite(s_out) and s_out < best_score:
                best_score = s_out
                best_seq = a_out.copy()
                print(f"  Phase 3 improved: {best_score:.10f} n={best_seq.size}")

    # Phase 4: final polish on the best sequence.
    if best_seq is not None and time.monotonic() < deadline - 15.0:
        print("=== Phase 4: final polish ===")
        dl = min(deadline - 25.0, time.monotonic() + 80.0)
        a_out, s_out = _lbfgs_polish(
            best_seq,
            dl,
            top=8,
            beta_ratio=0.2,
            maxfun=70
        )
        if np.isfinite(s_out) and s_out < best_score:
            best_score = s_out
            best_seq = a_out.copy()
            print(f"  L-BFGS improved: {best_score:.10f} n={best_seq.size}")

        if time.monotonic() < deadline - 10.0:
            dl2 = min(deadline - 5.0, time.monotonic() + 50.0)
            a_out, s_out = _adaptive_opt(
                best_seq,
                dl2,
                lr=0.018,
                max_top=10,
                active_every=35,
                ridge_rel=1e-6,
                use_active=True,
                seed=int(seed) + 100,
                label="final"
            )
            if np.isfinite(s_out) and s_out < best_score:
                best_score = s_out
                best_seq = a_out.copy()
                print(f"  final adaptive improved: {best_score:.10f} n={best_seq.size}")

    # Final validation/fallback.
    if best_seq is None:
        final_seq = _profile_geometric(4096, 1.0)
    else:
        final_seq = np.asarray(best_seq, dtype=np.float64).ravel()
        final_seq = np.where(np.isfinite(final_seq), final_seq, 0.0)
        final_seq = np.clip(final_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if not np.isfinite(s) or s <= 0.01:
            final_seq = _profile_geometric(4096, 1.0)

    final_seq = _normalize(final_seq)
    final_score = _score_norm(final_seq)
    print(f"  FINAL local estimate: score={final_score:.10f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
