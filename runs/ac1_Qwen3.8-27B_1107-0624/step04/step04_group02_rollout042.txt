def _solve_reduced_lp(a, M, n_constraints=100):
    n = len(a)
    c = _full_convolve(a, a)
    
    # Select the most important constraints (largest values of c)
    # Actually, we want to constrain the lags that are most likely to be violated
    # after the step. Since we're looking for b such that a*b <= M,
    # and the current a*a has max M, the binding constraints are at the lags
    # where a*a is close to M.
    
    # For safety, include all lags where a*a > M * 0.9 (within 10% of max)
    threshold = M * 0.9
    active = np.where(c > threshold)[0]
    
    # Also add a random sample of other lags for robustness
    n_active = len(active)
    if n_active < n_constraints:
        remaining = np.setdiff1d(np.arange(2*n-1), active)
        extra = np.random.choice(remaining, size=min(n_constraints - n_active, len(remaining)), replace=False)
        active = np.concatenate([active, extra])
    
    # Build reduced constraint matrix
    T_reduced = _build_reduced_conv_matrix(a, active)
    
    result = linprog(
        c=-np.ones(n),
        A_ub=T_reduced,
        b_ub=np.full(len(active), M),
        bounds=[(0, None)] * n,
        method='highs'
    )
    
    if result.success:
        b = result.x
        # Check full convolution
        check = _full_convolve(a, b)
        if check.max() <= M * 1.001:
            return b
        else:
            # Add violated constraints and re-solve (one more iteration)
            violated = np.where(check > M)[0]
            all_active = np.union1d(active, violated)
            T_full = _build_reduced_conv_matrix(a, all_active)
            result2 = linprog(
                c=-np.ones(n),
                A_ub=T_full,
                b_ub=np.full(len(all_active), M),
                bounds=[(0, None)] * n,
                method='highs'
            )
            if result2.success:
                return result2.x
    return None
