def _next_pow2(n):

def _full_convolve(a, b):

    out_n = n + m - 1
    N = _next_pow2(out_n)

    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines: (1) L-BFGS-B with log-sum-exp smoothing in softmax space,
(2) projected subgradient active-set method on the simplex,
(3) multi-restart diversification, (4) adaptive n (2048 vs 4096).
Key insight: we want flat autocorrelation, so the optimal sequence has
pseudo-random/spreading-code structure.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp(obj, beta):
    m = obj.max()
    return m + np.log(np.exp((obj - m) / beta).sum()) * beta


def _objective_and_grad_z(z, beta):
    """Compute smoothed objective and gradient in z-space (softmax param)."""
    a = _softmax(z)
    n = len(a)
    c = _full_convolve(a, a)
    c_max = c.max()
    c_shifted = (c - c_max) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s

    # Objective: 2n/beta * logsumexp(beta*c) = 2n*(cmax + beta*log(sum(exp((c-cmax)/beta))))
    obj = 2.0 * n * (c_max + beta * np.log(s))

    # h[j] = sum_m a[m] * w[m+j]  (j = 0..n-1)
    N = _next_pow2(n + len(c))
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]

    W = float(np.dot(w, c))
    grad = 2.0 * n * a * (h - W)

    return obj, grad


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() < 0.01:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _lbfgs_optimize(a_init, deadline, beta_schedule, max_iter=50000, lr_init=None):
    """L-BFGS-B optimization in softmax space with log-sum-exp smoothing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break

        def fun_grad(z_in):
            obj, grad = _objective_and_grad_z(z_in, beta)
            if not np.all(np.isfinite(grad)):
                return obj + 1e10, np.zeros_like(z_in)
            return obj, grad

        def fun_only(z_in):
            return _objective_and_grad_z(z_in, beta)[0]

        try:
            res = minimize(
                fun_only, z, jac=fun_grad, method='L-BFGS-B',
                bounds=[(-50.0, 50.0)] * n,
                options={'maxiter': max_iter, 'maxfun': max_iter * 2,
                         'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 50}
            )
        except Exception:
            break

        z_res = res.x
        z_res = np.clip(z_res, -50.0, 50.0)
        z_res = z_res - z_res.mean()
        a_res = _softmax(z_res)
        sc = _score_norm(a_res)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_res.copy()
        z = z_res

    return best_a, best_score


def _subgradient_optimize(a_init, deadline, max_iter=50000):
    """Projected subgradient method on the simplex with active-set strategy."""
    n = len(a_init)
    a = a_init.copy()
    a = _project_simplex(a)
    best_a = a.copy()
    best_score = _score_norm(a)

    alpha = 1.0 / np.sqrt(n)
    gamma = 0.5
    stall = 0
    prev_best = best_score

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()

        # Active set: lags within 1% of max
        threshold = c_max * 0.99
        active_mask = c >= threshold
        n_active = active_mask.sum()
        w = np.zeros_like(c)
        w[active_mask] = 1.0 / n_active

        # Subgradient: g[i] = sum_k w[k] * a[k-i] (for valid k-i)
        # g[i] = sum_{j: 0<=i+j<2n-1} w[i+j] * a[j]
        # = cross_corr(w, a)[i]
        N = _next_pow2(n + len(c))
        fa = np.fft.rfft(a, N)
        fw = np.fft.rfft(w, N)
        g = np.fft.irfft(np.conj(fa) * fw, N)[:n]

        # Step size decay
        alpha = alpha * (1.0 - 1e-6) + 0.01 / (n * (it + 1) ** 0.5)

        # Update
        a_new = _project_simplex(a - alpha * g)

        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            stall = 0
            alpha = min(alpha * 1.05, 2.0 / n)
        else:
            stall += 1
            if stall > 500:
                alpha *= 0.7
                stall = 0

        a = a_new

        if it % 1000 == 0:
            # Occasional "jolt" to escape local minima
            if stall > 2000:
                noise = 0.01 * np.random.standard_normal(n)
                a = _project_simplex(a + noise)
                stall = 0

    return best_a, best_score


def _adam_optimize(a_init, deadline, beta_schedule, max_iter=200000, lr=0.01):
    """Adam optimizer in log-space (similar to previous but with improvements)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_phase = 0
        while it_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_phase += 1
            t += 1

            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = c.max()
            cs = (c - cm) / beta
            e = np.exp(cs)
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            N = _next_pow2(n + len(c))
            fa = np.fft.rfft(a, N)
            fw = np.fft.rfft(w, N)
            h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
            W = float(np.dot(w, c))
            grad = 2.0 * n * a * (h - W)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1 - b1) * grad
            v_buf = b2 * v_buf + (1 - b2) * grad * grad
            mh = m_buf / (1 - b1 ** t)
            vh = v_buf / (1 - b2 ** t)

            update = lr * mh / (np.sqrt(vh) + eps)
            unorm = np.linalg.norm(update)
            if unorm > 1.0:
                update *= 1.0 / unorm

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 200 == 0:
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    a_final = _softmax(z)
    sc = _score_norm(a_final)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a_final.copy()

    return best_a, best_score

def _perturb_fourier(a, rng):
    """Add Fourier modes to the log-space perturbation."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(parent)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        update_best(parent, sc0, "parent")
    else:
        n_main = 2048
        a0 = np.ones(n_main) / n_main
        print(f"No parent, using uniform n={n_main}")

    # === Phase 1: L-BFGS-B with log-sum-exp (0 - 500s) ===
    print("Phase 1: L-BFGS-B optimization")
    phase1_end = min(deadline, t_start + 500.0)

    c0 = _full_convolve(a0, a0)
    max_c0 = float(c0.max())

    # Beta schedule: start smooth, get sharper
    beta_schedule_1 = [
        0.05 * max_c0, 0.025 * max_c0, 0.012 * max_c0,
        0.006 * max_c0, 0.003 * max_c0, 0.0015 * max_c0,
        0.0008 * max_c0, 0.0004 * max_c0, 0.0002 * max_c0
    ]

    time_per_beta = (phase1_end - t_start) / len(beta_schedule_1)
    for i, beta in enumerate(beta_schedule_1):
        phase_deadline = t_start + (i + 1) * time_per_beta
        # Re-initialize z from best so far
        if best_seq is not None and i > 0:
            a_start = best_seq / best_seq.sum()
        else:
            a_start = a0
        p_best, sc_best = _lbfgs_optimize(
            a_start, phase_deadline, [beta], max_iter=3000
        )
        update_best(p_best, _score(p_best), f"lbfgs_b{i}")

    print(f"Phase 1 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Subgradient active-set (500s - 900s) ===
    print("Phase 2: Subgradient active-set")
    phase2_end = min(deadline, t_start + 900.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        a_sg = best_seq / best_seq.sum()
        p_best, sc_best = _subgradient_optimize(
            a_sg, phase2_end, max_iter=50000
        )
        update_best(p_best, _score(p_best), "subgradient")

    # Also try Adam from a perturbed point
    if time.monotonic() < phase2_end - 30 and best_seq is not None:
        a_pert = _perturb(best_seq, 0.04, rng)
        c_p = _full_convolve(a_pert, a_pert)
        mc_p = float(c_p.max())
        bs = [0.02 * mc_p, 0.01 * mc_p, 0.005 * mc_p, 0.002 * mc_p, 0.001 * mc_p]
        time_p = phase2_end - time.monotonic() - 10
        p_best, sc_best = _adam_optimize(
            a_pert, t_start + time_p, bs, max_iter=100000, lr=0.012
        )
        update_best(p_best, _score(p_best), "adam_perturb")

    print(f"Phase 2 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Multi-restart L-BFGS-B (900s - 1300s) ===
    print("Phase 3: Multi-restart")
    phase3_end = min(deadline, t_start + 1300.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        n_restarts = 6
        per_restart = (phase3_end - time.monotonic() - 10) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 20:
                break

            if i % 3 == 0:
                a_start = _perturb(best_seq, 0.03 + 0.01 * i, rng)
            elif i % 3 == 1:
                a_start = _perturb_fourier(best_seq, rng)
            else:
                # Structured: mix best with uniform
                a_start = 0.7 * best_seq + 0.3 * np.ones(len(best_seq)) / len(best_seq)
                a_start = a_start / a_start.sum()

            c_s = _full_convolve(a_start, a_start)
            mc_s = float(c_s.max())
            bs = [0.01 * mc_s, 0.005 * mc_s, 0.002 * mc_s, 0.001 * mc_s, 0.0005 * mc_s]

            p_best, sc_best = _lbfgs_optimize(
                a_start, time.monotonic() + per_restart, bs, max_iter=5000
            )
            update_best(p_best, _score(p_best), f"restart{i}")

    print(f"Phase 3 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Fine polish with very low beta (1300s - 1550s) ===
    print("Phase 4: Fine polish")
    phase4_end = min(deadline, t_start + 1550.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        a_fine = best_seq / best_seq.sum()
        c_f = _full_convolve(a_fine, a_fine)
        mc_f = float(c_f.max())
        bs_fine = [0.0005 * mc_f, 0.0002 * mc_f, 0.0001 * mc_f, 0.00005 * mc_f]

        # L-BFGS-B polish
        p_best, sc_best = _lbfgs_optimize(
            a_fine, phase4_end - 30, bs_fine, max_iter=10000
        )
        update_best(p_best, _score(p_best), "fine_lbfgs")

        # Subgradient polish
        if time.monotonic() < phase4_end - 30:
            p_best2, sc_best2 = _subgradient_optimize(
                p_best / p_best.sum(), phase4_end, max_iter=30000
            )
            update_best(p_best2, _score(p_best2), "fine_sg")

    print(f"Phase 4 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Try larger n (1550s - 1750s) ===
    print("Phase 5: Try larger n")
    phase5_end = min(deadline, t_start + 1750.0)
    if best_seq is not None and len(best_seq) < 4096 and time.monotonic() < phase5_end - 60:
        target_n = 4096
        p_interp = _interp_profile(best_seq, target_n)
        sc_interp = _score(p_interp)
        update_best(p_interp, sc_interp, f"interp_n{target_n}")

        if time.monotonic() < phase5_end - 60:
            c_int = _full_convolve(p_interp, p_interp)
            mc_int = float(c_int.max())
            bs_int = [0.005 * mc_int, 0.002 * mc_int, 0.001 * mc_int, 0.0005 * mc_int]
            p_best, sc_best = _lbfgs_optimize(
                p_interp, phase5_end, bs_int, max_iter=5000
            )
            update_best(p_best, _score(p_best), f"opt_n{target_n}")

    print(f"Phase 5 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final ultra-fine polish (1750s - deadline) ===
    print("Phase 6: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_fin = best_seq / best_seq.sum()
        c_fin = _full_convolve(a_fin, a_fin)
        mc_fin = float(c_fin.max())

        # Very fine subgradient
        bs_final = [0.00005 * mc_fin, 0.00002 * mc_fin, 0.00001 * mc_fin]
        p_best, sc_best = _lbfgs_optimize(
            a_fin, deadline, bs_final, max_iter=5000
        )
        update_best(p_best, _score(p_best), "final_lbfgs")

        # Quick subgradient
        if time.monotonic() < deadline - 5:
            p_best2, sc_best2 = _subgradient_optimize(
                p_best / p_best.sum(), deadline + 2, max_iter=10000
            )
            update_best(p_best2, _score(p_best2), "final_sg")

    print(f"Phase 6 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
