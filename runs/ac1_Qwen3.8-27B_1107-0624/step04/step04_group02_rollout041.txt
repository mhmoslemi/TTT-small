LP-based optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses a primal LP iteration combined with L-BFGS polishing.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
from scipy.fft import next_fast_len
import time

def _full_convolve(a, b):
    """Full convolution using FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_fast_len(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Compute max(conv(a,a)) for normalized input (sum=1)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _build_lp_matrix(f, n):
    """Build the constraint matrix A for the LP.
    A[k, i] = f[k-i] if 0 <= k-i <= n-1, else 0.
    Shape: (2n-1, n)
    """
    # For each row k, the non-zero entries are at columns i where 0 <= k-i <= n-1
    # i.e., max(0, k-n+1) <= i <= min(k, n-1)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        for i in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

    """Solve one LP iteration.
    max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns g (the LP solution).

    b = M * np.ones(2 * n - 1)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)

    return None

        g = g / g.sum()

                continue
            f_trial = f_trial / f_trial.sum()
            sc = _score_norm(f_trial)

                best_t = t

"""
LP-based + L-BFGS hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses primal LP iteration (best-response direction) combined with logsumexp L-BFGS
smoothing for polishing. Exploits the non-palindromic structure to achieve scores
well below 2 by reducing the peak of the self-convolution.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
from scipy.fft import next_fast_len
import time


def _full_convolve(a, b):
    """Full convolution using FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_fast_len(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_norm(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    """Full score: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _build_lp_matrix(f):
    """Build constraint matrix for LP: A[k,i] = f[k-i] if valid."""
    n = len(f)
    total_rows = 2 * n - 1
    # Count non-zeros per row
    row_counts = np.minimum(np.minimum(np.arange(total_rows) + 1, n),
                            total_rows - np.arange(total_rows))
    nnz = int(row_counts.sum())
    rows = np.zeros(nnz, dtype=np.int32)
    cols = np.zeros(nnz, dtype=np.int32)
    vals = np.zeros(nnz, dtype=np.float64)
    idx = 0
    for k in range(total_rows):
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        cnt = i_hi - i_lo + 1
        rows[idx:idx + cnt] = k
        cols[idx:idx + cnt] = np.arange(i_lo, i_hi + 1)
        vals[idx:idx + cnt] = f[k - i_hi:k - i_lo + 1]
        idx += cnt
    return csr_matrix((vals, (rows, cols)), shape=(total_rows, n))


def _lp_solve(f, M):
    """Solve: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    A = _build_lp_matrix(f)
    b_ub = M * np.ones(2 * n - 1)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs',
                         options={'presolve': True, 'time_limit': 5.0})
        if result.success and result.x is not None:
            return np.maximum(result.x, 0)
    except Exception:
        pass
    return None


def _lp_optimize(f_init, deadline, verbose=True):
    """LP-based iterative optimization."""
    n = len(f_init)
    f = np.array(f_init, dtype=np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
        s = 1.0
    f = f / s

    best_f = f.copy()
    best_score = _score_norm(f)
    no_improve = 0
    max_no_improve = 200

    for it in range(50000):
        if time.monotonic() > deadline - 3.0:
            break

        conv = _full_convolve(f, f)
        M = float(conv.max())

        g = _lp_solve(f, M)
        if g is None or g.sum() < 1e-15:
            break

        g = g / g.sum()

        # Line search for best blend
        best_t = 0.0
        best_sc = best_score
        for t in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            f_trial = (1.0 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0.0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial = f_trial / st
            sc = _score_norm(f_trial)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_t = t

        if best_t == 0.0:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
            continue

        f = (1.0 - best_t) * f + best_t * g
        f = np.maximum(f, 0.0)
        f = f / f.sum()

        sc = _score_norm(f)
        if sc < best_score - 1e-12:
            best_score = sc
            best_f = f.copy()
            no_improve = 0
            if verbose and (it < 20 or it % 100 == 0):
                print(f"    LP iter {it}: score={best_score:.10f}, t={best_t}")
        else:
            no_improve += 1
            # Nudge f back
            f = (best_f + f) / 2.0
            f = np.maximum(f, 0.0)
            f = f / f.sum()

    return best_f, best_score


def _logsumexp_obj(z, n, beta):
    """Logsumexp smoothed objective in log-space."""
    z = z - np.max(z)
    e = np.exp(z)
    a = e / e.sum()
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_sh = (c - cm) / beta
    e2 = np.exp(c_sh)
    s2 = e2.sum()
    if s2 < 1e-300:
        return 2.0 * n * cm
    return 2.0 * n * (cm + beta * np.log(s2))


def _logsumexp_grad(z, n, beta):
    """Gradient of logsumexp smoothed objective."""
    z = z - np.max(z)
    e = np.exp(z)
    a = e / e.sum()
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_sh = (c - cm) / beta
    e2 = np.exp(c_sh)
    s2 = e2.sum()
    if s2 < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e2 / s2
    W = float(np.dot(w, c))
    # grad of (w^T c) wrt a_i = 2 * sum_k w_k * a[k-i] (via cross-correlation)
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    g = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(g)):
        g = np.zeros(n)
    return g


def _lbfgs_smoothing(f_init, deadline, beta_start=0.05, beta_end=0.001):
    """L-BFGS with logsumexp continuation."""
    n = len(f_init)
    z = np.log(np.maximum(f_init, 1e-15))
    z = z - z.mean()

    best_a = f_init.copy()
    best_score = _score_norm(f_init)

    conv = _full_convolve(f_init, f_init)
    max_c = float(conv.max())

    betas = np.geomspace(beta_start * max_c, beta_end * max_c, 8)
    if time.monotonic() > deadline - 5:
        return best_a, best_score

    per_beta = max(3.0, (deadline - time.monotonic() - 3) / len(betas))

    for beta in betas:
        if time.monotonic() > deadline - 3:
            break
        bd = min(deadline - 1.0, time.monotonic() + per_beta)

        def obj(z_):
            return _logsumexp_obj(z_, n, beta)

        def grad(z_):
            return _logsumexp_grad(z_, n, beta)

        try:
            res = minimize(obj, z, jac=grad, method='L-BFGS-B',
                           options={'maxiter': 2000, 'ftol': 1e-18,
                                    'gtol': 1e-10, 'maxls': 50,
                                    'maxfun': 5000, 'max_time': per_beta})
            if res.x is not None and np.all(np.isfinite(res.x)):
                z_new = res.x
                z_new = z_new - z_new.mean()
                a_new = np.exp(z_new)
                a_new = a_new / a_new.sum()
                sc = _score_norm(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _make_init(n, rng, style=0):
    """Generate diverse initial profiles."""
    if style == 0:
        # Uniform
        return np.ones(n) / n
    elif style == 1:
        # Linear ramp up
        a = np.linspace(0.5, 2.0, n)
        return a / a.sum()
    elif style == 2:
        # Linear ramp down
        a = np.linspace(2.0, 0.5, n)
        return a / a.sum()
    elif style == 3:
        # Two peaks at edges
        a = np.ones(n)
        a[:n // 3] = 2.0
        a[2 * n // 3:] = 2.0
        return a / a.sum()
    elif style == 4:
        # Gaussian centered
        t = np.linspace(0, 1, n)
        a = np.exp(-4 * (t - 0.5) ** 2)
        return a / a.sum()
    elif style == 5:
        # Random smooth
        a = np.abs(rng.standard_normal(n))
        # Smooth it
        from numpy import cumsum
        a = a - a.mean()
        a = np.abs(a) + 0.1
        return a / a.sum()
    elif style == 6:
        # Alternating with noise
        a = np.ones(n) * (1.0 + 0.3 * rng.standard_normal(n))
        a = np.maximum(a, 0.1)
        return a / a.sum()
    elif style == 7:
        # Edge-concentrated (non-palindromic)
        a = np.zeros(n)
        a[0:n // 4] = 2.0 + rng.random(n // 4) * 0.5
        a[n // 2:n // 2 + n // 5] = 0.5 + rng.random(n // 5) * 0.3
        a[3 * n // 4:] = 1.5 + rng.random(n - 3 * n // 4) * 0.5
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    elif style == 8:
        # Sinusoidal modulation (non-palindromic)
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.sin(3 * np.pi * t) + 0.1 * np.sin(7 * np.pi * t)
        a = np.maximum(a, 0.1)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _local_search(f_init, deadline, rng, n_trials=2000):
    """Local search with random perturbations."""
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_score = _score_norm(f)
    T = 0.001
    T_min = 0.0001

    for it in range(n_trials):
        if time.monotonic() > deadline - 2.0:
            break

        f_trial = f.copy()
        # Random block perturbation
        bs = rng.integers(3, max(4, n // 16))
        start = rng.integers(0, n - bs)
        f_trial[start:start + bs] *= rng.uniform(0.7, 1.3, bs)
        f_trial = np.maximum(f_trial, 1e-15)
        f_trial = f_trial / f_trial.sum()

        sc = _score_norm(f_trial)
        if sc < best_score:
            best_score = sc
            best_f = f_trial.copy()
            f = f_trial.copy()
        elif rng.random() < np.exp(-(sc - best_score) / T):
            f = f_trial.copy()

        T = max(T_min, T * 0.999)

    return best_f, best_score

        f_best, sc_best = _lp_optimize(parent, p1_end - 2, verbose=True)
        _update(f_best, sc_best, "LP_parent")

    # --- Phase 2: Continue LP with smaller step refinement (15% of time) ---
    p2_end = t_start + 0.75 * (budget_s - 20)
    print(f"\nPhase 2: LP refinement (budget={max(0, p2_end - time.monotonic()):.0f}s)")

    if best_seq is not None and time.monotonic() < p2_end - 5:
        f_best2, sc_best2 = _lp_optimize(best_seq, p2_end - 2, verbose=False)
        _update(f_best2, sc_best2, "LP_refine")

    # --- Phase 3: L-BFGS smoothing (15% of time) ---
    p3_end = t_start + 0.90 * (budget_s - 20)
    print(f"\nPhase 3: L-BFGS smoothing (budget={max(0, p3_end - time.monotonic()):.0f}s)")

    if best_seq is not None and time.monotonic() < p3_end - 5:
        f_lbfgs, sc_lbfgs = _lbfgs_smoothing(best_seq, p3_end - 2,
                                             beta_start=0.03, beta_end=0.0005)
        _update(f_lbfgs, sc_lbfgs, "L-BFGS")

    # --- Phase 4: Local search (8% of time) ---
    p4_end = t_start + 0.98 * (budget_s - 20)
    print(f"\nPhase 4: Local search (budget={max(0, p4_end - time.monotonic()):.0f}s)")

    if best_seq is not None and time.monotonic() < p4_end - 5:
        f_ls, sc_ls = _local_search(best_seq, p4_end - 2, rng,
                                    n_trials=min(5000, int((p4_end - time.monotonic()) * 200)))
        _update(f_ls, sc_ls, "local_search")

    # --- Phase 5: If time remains, try restart with different init ---
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic() - 30
        if remaining > 60:
            print(f"\nPhase 5: Restart with different init (budget={remaining:.0f}s)")
            # Perturb best and re-optimize
            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise = 0.05
            z_pert = z + noise * rng.standard_normal(n)
            z_pert = np.clip(z_pert, -50, 50)
            z_pert = z_pert - z_pert.mean()
            f_pert = np.exp(z_pert)
            f_pert = f_pert / f_pert.sum()

            dl = min(deadline - 5, time.monotonic() + remaining * 0.6)
            f_r, sc_r = _lp_optimize(f_pert, dl, verbose=False)
            _update(f_r, sc_r, "restart_LP")

            if time.monotonic() < deadline - 20:
                dl2 = min(deadline - 3, time.monotonic() + remaining * 0.35)
                f_r2, sc_r2 = _lbfgs_smoothing(f_r, dl2, beta_start=0.02, beta_end=0.0003)
                _update(f_r2, sc_r2, "restart_LBFGS")

    # --- Final validation and return ---
    print(f"\n{'='*60}")
    print(f"FINAL: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")
    print(f"{'='*60}")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(2048) / 2048
            s = 1.0
        final = final / s  # Normalize sum to 1

        sc_final = _score_full(final)
        print(f"Final score (full): {sc_final:.10f}")

        # Validate with evaluator
        if len(final) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"Evaluation error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fb = np.ones(2048) / 2048
    return [float(x) for x in fb]
