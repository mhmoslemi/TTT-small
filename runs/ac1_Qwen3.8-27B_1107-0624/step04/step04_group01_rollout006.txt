import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    # Assumes sum(a) = 1
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _build_lp_matrix(a):
    """Build the constraint matrix A for the LP.
    A[k, m] = a[k-m] if 0 <= k-m < n, else 0.
    Shape: (2n-1) x n
    """
    n = len(a)
    rows = []
    cols = []
    vals = []
    
    for m in range(n):
        # A[k, m] = a[k-m] for k = m, m+1, ..., m+n-1 (clipped to 0..2n-2)
        k_start = m
        k_end = min(2*n-1, m+n)
        for k in range(k_start, k_end):
            idx = k - m  # 0 <= idx < n
            rows.append(k)
            cols.append(m)
            vals.append(a[idx])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def _solve_lp(a, M):
    """Solve: max sum(g) s.t. A@_g <= M*1, g >= 0.
    Returns g (the solution) or None if infeasible.
    """
    n = len(a)
    A = _build_lp_matrix(a)
    
    # Minimize -sum(g) subject to A@g <= M, g >= 0
    c = -np.ones(n)
    A_ub = A
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    return res.x

"""
LP-based Frank-Wolfe optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses scipy's HiGHS solver for the linear subproblem, combined with Adam gradient
polishing for fine-tuning. The LP subproblem finds the optimal direction to
reduce peak autocorrelation, and the line search ensures monotone improvement.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _build_conv_indices(n):

            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

            # Find position in flat array

def _solve_lp_fast(a, M, n):
    """Optimized LP solve with pre-structured matrix."""
    rows = np.empty(0, dtype=np.int32)
    cols = np.empty(0, dtype=np.int32)
    vals = np.empty(0, dtype=np.float64)

    # Total non-zeros: sum over j of min(n, 2n-1-j) = n^2 - (n-1)*n/2 = n(n+1)/2
    total_nz = n * (n + 1) // 2
    rows = np.empty(total_nz, dtype=np.int32)
    cols = np.empty(total_nz, dtype=np.int32)
    vals = np.empty(total_nz, dtype=np.float64)

    pos = 0
    for j in range(n):
        k_end = min(2 * n - 1, j + n)
        for k in range(j, k_end):
            idx = k - j
            rows[pos] = k
            cols[pos] = j
            vals[pos] = a[idx]
            pos += 1

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

            return res.x
    except Exception:

def _lp_frank_wolfe(a, time_limit, verbose=True):
    """LP-based Frank-Wolfe optimization. a normalized to sum=1."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)
    t0 = time.monotonic()
    it = 0
    stagnant = 0

    while time.monotonic() < time_limit:
        c = _full_convolve(a, a)
        M = float(c.max())

        if M < 1e-15:
            break

        # Solve LP with small buffer
        g = _solve_lp_fast(a, M * (1.0 + 1e-10), n)
        if g is None:
            g = _solve_lp_fast(a, M * (1.0 + 1e-6), n)
        if g is None:
            g = _solve_lp_fast(a, M * (1.0 + 1e-4), n)
        if g is None:
            if verbose:
                print(f"  LP infeasible at iter {it}, breaking.")
            break

        g_sum = float(g.sum())
        if g_sum < 1e-12:
            break
        g = g / g_sum

        # Line search: coarse then fine
        best_t = 0.0
        best_sc = best_score
        grid = [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z(a, c, w, n):
    """Gradient of smoothed objective w.r.t. log(a)."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adam_polish(a_init, time_limit, lr=0.01, beta_schedule=None, verbose=True):
    """Adam optimizer in log-space with log-sum-exp smoothing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c_init = _full_convolve(a_init, a_init)
    max_c = float(c_init.max())

    if beta_schedule is None:
        beta_schedule = [0.01 * max_c, 0.005 * max_c, 0.002 * max_c,
                         0.001 * max_c, 0.0005 * max_c, 0.0002 * max_c,
                         0.0001 * max_c, 0.00005 * max_c]

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_count = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr

    for beta in beta_schedule:
        if time.monotonic() > time_limit:
            break
        phase_iters = 0
        max_phase = 50000
        while phase_iters < max_phase:
            if time.monotonic() > time_limit:
                break
            phase_iters += 1
            t_count += 1

            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros(len(c))
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad = _compute_grad_z(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t_count)
            vh = v_buf / (1.0 - b2 ** t_count)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t_count % 100 == 0:
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

            if phase_iters % 5000 == 0:
                current_lr = max(0.002, current_lr * 0.95)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    if verbose:
        print(f"  Adam polish: final score={best_score:.8f}, iters={t_count}")

    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score_norm(a0)
        print(f"Parent: n={n_main}, norm_score={sc0:.10f}, actual_score={_score(a0):.10f}")
        update_best(a0, _score(a0))
    else:
        n_main = 2048
        a0 = _make_profile(n_main, "uniform")
        print(f"No parent, using uniform n={n_main}")
        update_best(a0, _score(a0))

    # --- Phase 0: Quick exploration (0-20s) ---
    p0_end = min(deadline, t_start + 20.0)
    start_candidates = [a0.copy()]

    if parent is not None:
        for sigma in [0.03, 0.08, 0.15]:
            if time.monotonic() > p0_end:
                break
            p = _perturb(a0, sigma, rng)
            start_candidates.append(p)
            update_best(p, _score(p))

    for kind in ["sinusoidal", "tapered", "bimodal"]:
        if time.monotonic() > p0_end:
            break
        p = _make_profile(n_main, kind)
        update_best(p, _score(p))
        start_candidates.append(p)

    # Find best starting candidate
    best_start = a0.copy()
    best_start_sc = _score_norm(a0)
    for p in start_candidates:
        sc = _score_norm(p)
        if sc < best_start_sc:
            best_start_sc = sc
            best_start = p.copy()

    print(f"Phase 0: best_start_score={best_start_sc:.8f}, t={time.monotonic()-t_start:.1f}s")

    # --- Phase 1: LP-based Frank-Wolfe (20s - 800s) ---
    p1_end = min(deadline, t_start + 700.0)
    if time.monotonic() < p1_end:
        print(f"Phase 1: LP-FW starting, score={best_start_sc:.8f}")
        a_fw, sc_fw = _lp_frank_wolfe(best_start, p1_end, verbose=True)
        update_best(a_fw, _score(a_fw))

    # --- Phase 2: Adam polish (700s - 1050s) ---
    p2_end = min(deadline, t_start + 1050.0)
    if best_seq is not None and time.monotonic() < p2_end:
        print(f"Phase 2: Adam polish starting")
        a_norm = best_seq / best_seq.sum()
        a_p, sc_p = _adam_polish(a_norm, p2_end, lr=0.008, verbose=True)
        update_best(a_p, _score(a_p))

    # --- Phase 3: Multi-restart with LP-FW (1050s - 1400s) ---
    p3_end = min(deadline, t_start + 1400.0)
    if best_seq is not None and time.monotonic() < p3_end - 30:
        print(f"Phase 3: Multi-restart starting")
        n_restarts = 5
        per_restart = (p3_end - time.monotonic() - 20) / n_restarts
        a_norm = best_seq / best_seq.sum()

        for i in range(n_restarts):
            if time.monotonic() > p3_end - 20:
                break
            if per_restart < 5:
                break

            # Different perturbation
            if i % 2 == 0:
                p_start = _perturb(a_norm, 0.03 + 0.02 * i, rng)
            else:
                # Structured: circular shift in log-space
                z = np.log(np.maximum(a_norm, 1e-15))
                shift = rng.integers(1, max(2, n_main // 8))
                z = np.roll(z, int(shift))
                z = z - z.mean()
                p_start = _softmax(z)

            a_r, sc_r = _lp_frank_wolfe(p_start, time.monotonic() + per_restart, verbose=False)
            update_best(a_r, _score(a_r))

    # --- Phase 4: Try larger n (1400s - 1650s) ---
    p4_end = min(deadline, t_start + 1650.0)
    if best_seq is not None and len(best_seq) <= 2048 and time.monotonic() < p4_end - 30:
        print(f"Phase 4: Trying n=4096")
        target_n = 4096
        p_interp = _interp_profile(best_seq, target_n)
        update_best(p_interp, _score(p_interp))

        remaining = p4_end - time.monotonic() - 15
        if remaining > 30:
            a_l, sc_l = _lp_frank_wolfe(p_interp, time.monotonic() + remaining, verbose=False)
            update_best(a_l, _score(a_l))

    # --- Phase 5: Final Adam ultra-polish (1650s - deadline) ---
    if best_seq is not None and time.monotonic() < deadline - 5:
        print(f"Phase 5: Final polish")
        a_norm = best_seq / best_seq.sum()
        remaining = deadline - time.monotonic() - 3
        if remaining > 10:
            a_f, sc_f = _adam_polish(a_norm, time.monotonic() + remaining, lr=0.003,
                                     beta_schedule=None, verbose=True)
            update_best(a_f, _score(a_f))

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
