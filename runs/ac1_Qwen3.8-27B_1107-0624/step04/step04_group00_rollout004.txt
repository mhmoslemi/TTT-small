**Direct Peak Reduction (DPR) method:**

State: a (length n, sum = 1)
Target: minimize max_k (a*a)[k]

At each iteration:
1. Compute c = conv(a, a)
2. Identify the set of "active" peaks: A = {k : c[k] >= (1-eps) * max(c)} for some small eps (e.g., 0.01)
3. Compute the gradient as a uniform mixture of gradients for k in A:
   grad[j] = (1/|A|) * sum_{k in A} 2*a[k-j] * I(0 <= k-j < n)
4. Project grad onto the tangent space of {sum(a) = 1}: grad_proj = grad - mean(grad)
5. Take step: a_new = a - lr * grad_proj
6. Project onto a >= 0: a_new = max(a_new, 0)
7. Renormalize: a_new = a_new / sum(a_new)
8. Check if max(conv(a_new, a_new)) < max(c). If not, reduce lr and retry.
9. If max decreased, accept; otherwise, try a different direction.

The key advantage of using a mixture of active peaks (rather than just the single max) is that it balances the reduction across multiple peaks, potentially leading to a lower overall maximum.

This is essentially the **subgradient method** applied to the max function, with the subgradient chosen as a uniform mixture of the active gradients.

    # h[j] = sum_{m=0}^{n-1} c[j+m]^(p-1) * a[m]
    # Then dS/da[j] = 2*p*h[j]
    
    # I already computed g using c_p_minus_1 = c^(p-1), so g[j] = h[j]. Good.
    
    # Now da/dz for softmax:
    # da[j']/dz[j] = a[j'] * (delta_{j'j} - a[j])
    # dS/dz[j] = sum_{j'} (dS/da[j']) * (da[j']/dz[j])
    # = sum_{j'} 2*p*h[j'] * a[j'] * (delta_{j'j} - a[j])
    # = 2*p*a[j]*h[j] - 2*p*a[j]*sum_{j'} h[j']*a[j']
    # = 2*p*a[j]*(h[j] - H)
    # where H = sum_j h[j]*a[j] = dot(h, a)
    
    H = float(np.dot(g, a))
    grad_a = 2.0 * p * a * (g - H)
    
    # Total gradient w.r.t. z:
    # d/dz of 2n * S^(1/p) = 2n * (1/p) * S^(1/p-1) * dS/dz
    # = 2n * (1/p) * S^(1/p-1) * grad_a
    
    coeff = 2.0 * n * (1.0 / p) * S ** (1.0 / p - 1.0)
    grad = coeff * grad_a
    
    if not np.all(np.isfinite(grad)):
        return np.zeros(n)
    return grad


def _soft_max_obj(z, beta, n):
    """Log-sum-exp smoothed max objective."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = c.max()
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    log_sum = cm / beta + np.log(e.sum())
    return 2.0 * n * beta * log_sum


def _soft_max_grad(z, beta, n):
    """Gradient of log-sum-exp smoothed max."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = c.max()
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    w = e / e.sum()
    
    # w_conv[j] = sum_m w[j+m] * a[m] (cross-correlation)
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n-1:2*n-1]
    
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    
    if not np.all(np.isfinite(grad)):
        return np.zeros(n)
    return grad


def _run_power_mean_lbfgs(a_init, deadline, p_list=None):
    """L-BFGS optimization using power mean continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    if p_list is None:
        p_list = [4, 8, 16, 32, 64, 128, 256, 512, 1024]
    
    for idx, p in enumerate(p_list):
        if time.monotonic() > deadline:
            break
        
        remaining = len(p_list) - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        level_dl = time.monotonic() + min(time_per, 60.0)
        
        try:
            result = minimize(
                _power_mean_obj, z, args=(p, n),
                method='L-BFGS-B', jac=_power_mean_grad,
                options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 50, 'maxfun': 10000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score
