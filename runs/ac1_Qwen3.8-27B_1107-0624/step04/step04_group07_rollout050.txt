"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines log-space Adam with smooth-max gradient, flattening LP, spectral
equalization, multi-scale escalation, and diverse initialization.
Uses FFT-based gradient computation for O(n log n) per iteration.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a):
    """Full convolution of a with itself via FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    c = _convolve(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def _score_norm1(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    c = _convolve(a)
    return float(2.0 * n * c.max())


def _gradient_cross_corr(w, a):
    """
    Compute cross-correlation (w ⋆ a)[j] = sum_k w[k]*a[k-j] for j=0..n-1.
    This gives dF/da when F = sum_k w[k]*c[k] and c = conv(a,a).
    """
    n = len(a)
    nw = len(w)
    N = _next_pow2(nw + n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    # Cross-correlation: ifft(FFT(w) * conj(FFT(a)))
    cc = np.fft.irfft(fw * np.conj(fa), N)
    return cc[:n]


def _adam_logspace(a_init, deadline, rng, lr=0.05, beta_init=None,
                   beta_decay=0.93, min_beta=1e-6, no_imp_limit=150,
                   max_z=40.0, verbose=False):
    """
    Adam optimization in log-space with smooth-max objective.
    a is the probability vector (sum=1, all positive).
    We optimize z = log(a) (centered), then a = softmax(z).
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy()
    s = a.sum()
    a = a / s

    z = np.log(a)
    z -= z.mean()

    b1, b2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = _convolve(a)
    maxc = float(c.max())
    beta = beta_init if beta_init is not None else 0.05 * maxc

    best_z = z.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        # Compute smooth-max weights
        c_shifted = c - c.max()
        x = c_shifted / beta
        x -= x.max()
        ex = np.exp(x)
        sx = ex.sum()
        if sx < 1e-30 or not np.isfinite(sx):
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / sx

        # Gradient: cross-correlation of w with a
        grad = _gradient_cross_corr(w, a)

        # Project to log-space: dF/dz[j] = a[j] * (grad[j] - <a, grad>)
        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)
        # Center gz for numerical stability
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        # Adam update
        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -max_z, max_z)
        z -= z.mean()

        # Convert back to a
        a = np.exp(z - z.max())
        a = a / a.sum()
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        # Evaluate
        c = _convolve(a)
        maxc = float(c.max())
        score = 2.0 * n * maxc

        if np.isfinite(score):
            if score < best_score - 1e-10:
                best_score = score
                best_z = z.copy()
                no_imp = 0
                if verbose and (t_adam % 100 == 0):
                    print(f"    Adam t={t_adam}: score={score:.10f}")
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.005, lr * 0.9)
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        # Temperature decay
        if t_adam % 30 == 0:
            beta = max(min_beta * maxc, beta * beta_decay)
            # Adaptive: if top-2 gap is small, reduce beta more
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.05:
                    beta = max(min_beta * maxc, beta * 0.5)

        # Restart on stagnation
        if no_imp > no_imp_limit:
            ns = 0.1 if best_score < 1.51 else 0.2
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -max_z, max_z)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _convolve(a)
            beta = 0.05 * c.max()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score


def _lp_flatten(a, deadline, max_iters=20):
    """
    LP-based flattening: find b that maximizes min_k conv(a,b)[k].
    Then step towards b.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm1(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _convolve(a)
        target = c.mean()  # We want conv(a,b) to be close to target

        # Build convolution matrix: A[k,j] = a[k-j] if 0<=k-j<n
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                rows.extend([k] * len(idx))
                cols.extend(idx.tolist())
                vals.extend(a[k - idx].tolist())
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

        # LP: maximize t s.t. A@b >= t, b >= 0, sum(b) = 1
        # Variables: b[0..n-1], t
        # Minimize: -t
        c_obj = np.zeros(n + 1)
        c_obj[-1] = -1.0

        # A_ub @ [b, t] <= b_ub
        # A@b - t >= 0  =>  -A@b + t <= 0
        A_ub = -np.hstack([A.toarray(), np.eye(2 * n - 1, 1)])
        b_ub = np.zeros(2 * n - 1)

        # Equality: sum(b) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n + [(-np.inf, np.inf)]

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

        if not res.success:
            break

        b = np.maximum(res.x[:n], 0.0)
        s_b = b.sum()
        if s_b < 1e-10:
            break
        b = b / s_b

        # Line search
        best_t = 0.5
        best_sc = _score_norm1(a)
        for t_candidate in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_t = (1.0 - t_candidate) * a + t_candidate * b
            a_t = np.maximum(a_t, 1e-15)
            a_t = a_t / a_t.sum()
            sc_t = _score_norm1(a_t)
            if sc_t < best_sc:
                best_sc = sc_t
                best_t = t_candidate

        a = (1.0 - best_t) * a + best_t * b
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        sc = _score_norm1(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score

        # We want a_new such that |FFT(a_new)|^2 is closer to flat

        # But this might give negative values, so use soft approach

def _make_profile(n, ptype, p1=1.0, p2=0.0, p3=0.0):
    """Generate various profile shapes."""
    if ptype == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (t - 0.5)
    elif ptype == 'asymmetric_lin':
        t = np.linspace(0, 1, n)
        a = p1 + (1 - p1) * t
    elif ptype == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (2 * t - 1) ** 2
    elif ptype == 'cubic':
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + p1 * u + p2 * u ** 3
    elif ptype == 'cosine':
        t = np.arange(n)
        a = 1.0 + p1 * np.cos(2 * np.pi * p2 * t / n)
    elif ptype == 'sine':
        t = np.arange(n)
        a = 1.0 + p1 * np.sin(2 * np.pi * p2 * t / n)
    elif ptype == 'parabola':
        t = np.linspace(0, 1, n)
        a = (t - 0.5) ** 2 + p1
    elif ptype == 'gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / p1) ** 2)
    elif ptype == 'two_gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - p1) / p3) ** 2) + np.exp(-0.5 * ((t - p2) / p3) ** 2)
    elif ptype == 'U_shape':
        t = np.linspace(0, 1, n)
        a = (2 * t - 1) ** 2 + p1
    elif ptype == 'edge_heavy':
        t = np.arange(n) / max(1, n - 1)
        a = 1.0 / (0.05 + abs(t - 0.5) * 2.0) ** p1
    elif ptype == 'zigzag':
        t = np.arange(n)
        a = 1.0 + p1 * np.abs(2 * (t / n - np.floor(t / n + 0.5))) 
    elif ptype == 'sawtooth':
        t = np.arange(n)
        a = 1.0 + p1 * (2 * np.abs(t / n - np.floor(t / n + 0.5)) - 1)
    elif ptype == 'bimodal':
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.3) ** 2) / (2 * p1 ** 2)) + np.exp(-((t - 0.7) ** 2) / (2 * p1 ** 2))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def _interp(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new

        ('tg30', 'two_gauss', 0.3, 0.7, 0.15),
        ('tg25', 'two_gauss', 0.25, 0.75, 0.2),
        ('U02', 'U_shape', 0.2, 0, 0),
        ('U05', 'U_shape', 0.5, 0, 0),
        ('U10', 'U_shape', 1.0, 0, 0),
        ('edge1', 'edge_heavy', 1.0, 0, 0),
        ('edge15', 'edge_heavy', 1.5, 0, 0),
        ('edge2', 'edge_heavy', 2.0, 0, 0),
        ('zig1', 'zigzag', 1.0, 0, 0),
        ('saw05', 'sawtooth', 0.5, 0, 0),
        ('bim01', 'bimodal', 0.1, 0, 0),
        ('bim015', 'bimodal', 0.15, 0, 0),
    ]

    for n in n_sizes:
        for (label, ptype, p1, p2, p3) in profiles:
            p = _make_profile(n, ptype, p1, p2, p3)
            sc = _score_norm1(p)
            candidates.append((sc, p, n, label))
        if time.monotonic() > t_p1_end - 10:
            break

    # Parent-based candidates
    if parent is not None:
        for n in [256, 512, 1024, 2048]:
            pi = _interp(parent, n)
            sc = _score_norm1(pi)
            candidates.append((sc, pi, n, 'parent'))
        if 0 < parent.size <= 4096:
            sc = _score_norm1(parent)
            candidates.append((sc, parent.copy(), parent.size, 'parent_exact'))

    # Random candidates
    for n in [256, 512, 1024, 2048]:
        for _ in range(5):
            a = rng.dirichlet(np.ones(n) * 1.0)
            sc = _score_norm1(a)
            candidates.append((sc, a, n, 'rand'))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, best: {candidates[0][0]:.8f} n={candidates[0][2]} [{candidates[0][3]}]")

    _update_best(candidates[0][1], 'init')

    # === Phase 2: Adam optimization on top candidates ===
    print("Phase 2: Adam optimization...")
    t_p2_end = t_start + 700.0

    # Select diverse top candidates (different n and profiles)
    selected = []
    seen_n = set()
    for sc, p, n, label in candidates:
        if len(selected) >= 12:
            break
        # Allow up to 3 per n value
        if seen_n.get(n, 0) >= 3:
            continue
        seen_n[n] = seen_n.get(n, 0) + 1
        selected.append((sc, p, n, label))

    for idx, (sc, p, n, label) in enumerate(selected):
        if time.monotonic() > t_p2_end - 20:
            break

        # Allocate time: smaller n gets more time for LP later
        rem = t_p2_end - time.monotonic() - 10
        if rem < 15:
            break
        per = min(rem / max(1, len(selected) - idx), max(30.0, 500.0 / n))
        per = max(15.0, min(per, 120.0))

        a_ad, sc_ad = _adam_logspace(p, time.monotonic() + per, rng,
                                     lr=0.05, beta_decay=0.93, no_imp_limit=120)
        if sc_ad < best_score:
            _update_best(a_ad, f'Adam_{label}_n{n}')

        # If n is small enough, try LP polishing
        if n <= 1024 and time.monotonic() < t_p2_end - 30:
            per_lp = min(40.0, t_p2_end - time.monotonic() - 15)
            if per_lp > 15:
                a_lp, sc_lp = _lp_maximize(a_ad, time.monotonic() + per_lp, max_iters=20)
                if sc_lp < best_score:
                    _update_best(a_lp, f'LP_{label}_n{n}')

                # Also try flatten LP
                if time.monotonic() < t_p2_end - 15:
                    per_fl = min(30.0, t_p2_end - time.monotonic() - 10)
                    if per_fl > 10:
                        a_fl, sc_fl = _lp_flatten(a_lp, time.monotonic() + per_fl, max_iters=10)
                        if sc_fl < best_score:
                            _update_best(a_fl, f'FlatLP_{label}_n{n}')

    print(f"  After Phase 2: best={best_score:.10f}")

    # === Phase 3: Spectral equalization + refinement ===
    print("Phase 3: Spectral equalization...")
    t_p3_end = t_start + 1000.0

    if best_seq is not None:
        # Spectral equalization on current best
        per = min(80.0, t_p3_end - time.monotonic() - 20)
        if per > 20:
            a_se, sc_se = _spectral_equalize(best_seq, time.monotonic() + per, max_iters=100)
            if sc_se < best_score:
                _update_best(a_se, 'spectral_eq')

        # More Adam refinement on the best
        n_best = best_seq.size
        for lr_try in [0.03, 0.05, 0.08]:
            if time.monotonic() > t_p3_end - 30:
                break
            per = min(60.0, t_p3_end - time.monotonic() - 15)
            if per < 20:
                break
            a_r, sc_r = _adam_logspace(best_seq, time.monotonic() + per, rng,
                                       lr=lr_try, beta_decay=0.95, no_imp_limit=200)
            if sc_r < best_score:
                _update_best(a_r, f'refine_lr{lr_try}')

    # === Phase 4: Multi-scale escalation ===
    print("Phase 4: Multi-scale escalation...")
    t_p4_end = t_start + 1350.0

    if best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048, 4096]
        elif n_cur <= 1024:
            target_ns = [2048, 4096]
        elif n_cur <= 2048:
            target_ns = [4096]

        for tn in target_ns:
            if time.monotonic() > t_p4_end - 60:
                break
            p_int = _interp(best_seq, tn)
            sc_int = _score_norm1(p_int)
            print(f"  n={tn}: interp score={sc_int:.8f}")
            if sc_int < best_score:
                _update_best(p_int, f'interp_n{tn}')

            # Adam on interpolated
            if time.monotonic() < t_p4_end - 40:
                per = min(100.0, t_p4_end - time.monotonic() - 30)
                if per > 25:
                    a_e, sc_e = _adam_logspace(p_int, time.monotonic() + per, rng,
                                               lr=0.04, beta_decay=0.94, no_imp_limit=250)
                    if sc_e < best_score:
                        _update_best(a_e, f'Adam_esc_n{tn}')

            # LP if feasible
            if tn <= 1024 and time.monotonic() < t_p4_end - 30:
                per2 = min(40.0, t_p4_end - time.monotonic() - 15)
                if per2 > 15:
                    a_le, sc_le = _lp_maximize(p_int, time.monotonic() + per2, max_iters=15)
                    if sc_le < best_score:
                        _update_best(a_le, f'LP_esc_n{tn}')

    # === Phase 5: Random restarts from best ===
    print("Phase 5: Random restarts...")
    t_p5_end = t_start + 1550.0

    if best_seq is not None and time.monotonic() < t_p5_end:
        n_best = best_seq.size
        n_restarts = 8
        per_r = max(10.0, (t_p5_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > t_p5_end - 5:
                break
            per = min(per_r, t_p5_end - time.monotonic() - 5)
            if per < 10:
                break

            # Different noise strategies
            noise_levels = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.8]
            ns = noise_levels[i % len(noise_levels)]

            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -40, 40)
            z -= z.mean()
            a_s = np.exp(z - z.max())
            a_s = a_s / a_s.sum()

            a_r, sc_r = _adam_logspace(a_s, time.monotonic() + per, rng,
                                       lr=0.05, beta_decay=0.92, no_imp_limit=100)
            if sc_r < best_score:
                _update_best(a_r, f'restart{i}_ns{ns}')

            # Quick LP polish
            if n_best <= 1024 and time.monotonic() < t_p5_end - 10:
                per2 = min(15.0, t_p5_end - time.monotonic() - 5)
                if per2 > 8:
                    a_rl, sc_rl = _lp_maximize(a_r, time.monotonic() + per2, max_iters=8)
                    if sc_rl < best_score:
                        _update_best(a_rl, f'LP_restart{i}')

    # === Phase 6: Final polish ===
    print("Phase 6: Final polish...")
    if time.monotonic() < deadline - 15 and best_seq is not None:
        per = deadline - time.monotonic() - 15
        n_b = best_seq.size

        if n_b <= 1024:
            # LP polish
            if per > 20:
                a_fp, sc_fp = _lp_maximize(best_seq, time.monotonic() + per * 0.6, max_iters=20)
                if sc_fp < best_score:
                    _update_best(a_fp, 'final_LP')

            # Adam polish
            if time.monotonic() < deadline - 10:
                per2 = deadline - time.monotonic() - 10
                a_f2, sc_f2 = _adam_logspace(best_seq, time.monotonic() + per2, rng,
                                             lr=0.02, beta_decay=0.96, no_imp_limit=50)
                if sc_f2 < best_score:
                    _update_best(a_f2, 'final_Adam')
        else:
            # Adam only for large n
            a_fp, sc_fp = _adam_logspace(best_seq, time.monotonic() + per, rng,
                                         lr=0.02, beta_decay=0.96, no_imp_limit=50)
            if sc_fp < best_score:
                _update_best(a_fp, 'final_Adam_large')

    # Final validation
    print(f"\nFinal best: {best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s

        exact_sc = _score_norm1(final)
        print(f"  Final score: {exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"  EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"  Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile(512, 'linear', 1.0, 0, 0)
    print("FALLBACK n=512")
    return [float(x) for x in fb]
