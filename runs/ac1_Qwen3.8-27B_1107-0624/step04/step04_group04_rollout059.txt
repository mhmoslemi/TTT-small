```python
def full_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:

    N = max(N, 2)
    fa = rfft(a, N)
    fb = rfft(b, N)
    r = irfft(fa * fb, N)[:out_n]
    return r
```

```python
SQP-LP based optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses sequential linearization of the max-convolution objective, solving
LP subproblems at each step with an adaptively growing active set.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    r = irfft(fa * fb, N)[:out_n]
    return r


def score_fn(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def score_norm(a):
    """Assume sum(a) ~ 1, compute 2*n*max(conv(a,a))."""
    n = len(a)
    c = full_convolve(a, a)
    return float(2.0 * n * c.max())


def compute_h_k(a, k, n):
    """Compute h_k[i] = a[k-i] if 0 <= k-i < n, else 0."""
    h = np.zeros(n)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo <= hi:

    return h
```

Looking at this problem, I need to minimize `2*n*max(convolve(a,a))/sum(a)^2` where `a` is non-negative. The previous approach used L-BFGS with logsumexp smoothing + Adam + KKT balancing to achieve 1.506189. 

My key insight is to use a **Sequential Quadratic Programming with LP subproblems** approach: at each iteration, I linearize the quadratic constraint `C_k(a) ≤ t` at the current point, yielding a linear constraint. This gives an LP that can be solved exactly and efficiently. The active set (positions near the max of the convolution) is adaptively grown. This avoids the smoothing bias and gradient concentration issues of the logsumexp approach.

```python
"""
SQP-LP optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Linearizes the max-convolution objective at the current iterate, solving
LP subproblems with an adaptively managed active set. Combines multi-restart
with Adam polishing for final refinement.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    r = irfft(fa * fb, N)[:out_n]
    return r


def score_fn(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def score_norm(a):
    """Assume sum(a)~1, compute 2*n*max(conv(a,a))."""
    n = len(a)
    c = full_convolve(a, a)
    return float(2.0 * n * c.max())


def softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def sqp_lp_solve(a, active_idx):
    """
    Solve one SQP step via LP.
    Linearizes C_k(a) <= t for active k, adds sum(a)=1, a>=0.
    Returns (a_new, t_value) or (None, None).
    """
    n = len(a)
    C = full_convolve(a, a)
    cm = C.max()
    
    # Build LP: minimize t
    # Variables: [a_0,...,a_{n-1}, t]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_rows = []
    b_rows = []
    for k in active_idx:
        k = int(k)
        # h_k[i] = a[k-i] if 0 <= k-i < n
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        h = np.zeros(n)
        if lo <= hi:
            h[lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
        row = 2.0 * h
        row = np.append(row, -1.0)
        A_rows.append(row)
        b_rows.append(C[k])
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'time_limit': 30, 'presolve': True})
        if result.success and result.x is not None:
            a_new = np.maximum(result.x[:n], 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
            return a_new, float(result.x[-1])
    except Exception:
        pass
    return None, None


def sqp_optimize(a_init, deadline, max_iter=300, start_active=30,
                 grow_every=5, grow_factor=1.15, max_active=300,
                 min_improvement=1e-13, verbose=True):
    """
    Run SQP-LP optimization loop with adaptive active set.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = score_norm(a)
    
    C = full_convolve(a, a)
    n_active = start_active
    active_idx = np.argsort(C)[-n_active:]
    
    stagnation = 0
    consecutive_fail = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t0 = time.monotonic()
        a_new, t_lp = sqp_lp_solve(a, active_idx)
        
        if a_new is None:
            consecutive_fail += 1
            # Try smaller active set
            if consecutive_fail > 2 and n_active > 10:
                n_active = max(10, n_active // 2)
                C = full_convolve(a, a)
                active_idx = np.argsort(C)[-n_active:]
                consecutive_fail = 0
                continue
            if consecutive_fail > 10:
                break
            continue
        consecutive_fail = 0
        
        sc_new = score_norm(a_new)
        
        if sc_new < best_score - min_improvement:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            stagnation = 0
        else:
            # Line search: try fractional steps
            accepted = False
            for alpha in [0.5, 0.25, 0.125, 0.0625]:
                a_frac = (1.0 - alpha) * a + alpha * a_new
                a_frac = np.maximum(a_frac, 0.0)
                sf = a_frac.sum()
                if sf > 1e-15:
                    a_frac = a_frac / sf
                    sc_frac = score_norm(a_frac)
                    if sc_frac < best_score - min_improvement:
                        best_score = sc_frac
                        best_a = a_frac.copy()
                        a = a_frac
                        stagnation = 0
                        accepted = True
                        break
            if not accepted:
                stagnation += 1
        
        # Update active set
        C = full_convolve(a, a)
        
        # Grow active set periodically
        if it % grow_every == 0 and n_active < max_active:
            n_active = min(int(n_active * grow_factor) + 1, max_active)
        
        # Also add positions that are close to max (within 3%)
        threshold = C.max() * 0.97
        near_active = np.where(C >= threshold)[0]
        combined = np.unique(np.concatenate([active_idx, near_active]))
        
        # Keep top n_active by C value
        if len(combined) > n_active:
            combined = combined[np.argsort(C[combined])[::-1][:n_active]]
        active_idx = combined
        
        if verbose and it % 50 == 0:
            el = time.monotonic() - t0
            print(f"    SQP it={it}: score={best_score:.10f}, active={n_active}, "
                  f"stag={stagnation}, t/it={el:.3f}s")
        
        if stagnation > 40:
            break
    
    return best_a, best_score


def adam_polish(a_init, deadline, lr=0.02, top_k=20, n_max_iter=40000):
    """Adam with top-k active set for final polishing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = score_norm(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    C0 = full_convolve(a_init, a_init)
    beta = max(1e-10, 0.005 * C0.max())
    
    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = softmax_fn(z)
        
        C = full_convolve(a, a)
        cm = C.max()
        
        # Top-k softmax weights
        idx = np.argsort(C)[-top_k:]
        w = np.zeros(len(C))
        c_top = C[idx]
        c_norm = (c_top - c_top.max()) / beta
        e = np.exp(c_norm)
        s = e.sum()
        if s < 1e-300:
            w[idx[0]] = 1.0
        else:
            w[idx] = e / s
        
        W = float(np.dot(w, C))
        
        # Gradient computation
        a_rev = a[::-1]
        w_conv = full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            grad = np.zeros_like(z)
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        a = softmax_fn(z)
        sc = score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if t % 200 == 0:
            beta = max(1e-14, beta * 0.85)
        if t % 2000 == 0 and best_score < 1.51:
            lr = max(0.003, lr * 0.95)
    
    return best_a, best_score


def interp_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

            grow_factor=1.1, max_active=400, min_improvement=1e-14,
            verbose=False)
        update_best(a_f, sc_f, "sqp_final")
    
    # === Phase 5: Try n=2048 if parent was 4096 ===
    print("=== Phase 5: Alternative length ===")
    phase5_end = min(deadline, t_start + 1750.0)
    
    if time.monotonic() < phase5_end and best_seq is not None and len(best_seq) != 2048:
        p_interp = interp_profile(best_seq, 2048)
        sc_interp = score_norm(p_interp)
        update_best(p_interp, sc_interp, f"n2048_interp")
        
        remaining = phase5_end - time.monotonic()
        if remaining > 60:
            dl = min(phase5_end - 5, time.monotonic() + min(120.0, remaining))
            a_alt, sc_alt = sqp_optimize(
                p_interp, dl,
                max_iter=100, start_active=30, grow_every=5,
                grow_factor=1.15, max_active=200, min_improvement=1e-13,
                verbose=False)
            update_best(a_alt, sc_alt, "n2048_sqp")
    
    # === Final validation ===
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = score_fn(final_seq)
        else:
            final_seq = np.exp(-0.005 * np.arange(2048))
            final_seq = final_seq / final_seq.sum()
            final_score = score_fn(final_seq)
        
        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")
        
        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 10000:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  evaluate_sequence check: {e}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    print("  Using fallback")
    fallback = np.exp(-0.005 * np.arange(2048))
    fallback = fallback / fallback.sum()
    print(f"  FALLBACK: n=2048, score={score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
```
