```python
"""
Multi-phase optimizer for minimizing 2n*max(convolve(a,a))/sum(a)^2.
Uses top-k smooth approximation with continuation, Adam optimization,
and active-set refinement. Fixed n=4096 throughout.
"""
import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    """Compute 2n * max(conv(a,a)) without the (sum a)^2 denominator."""
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())

def _score_full(a):
    """Compute the full score 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max()) / (s * s)

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s
```

```python
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_parent = _score_norm(parent)
        print(f"Parent: n={n}, score={sc_parent:.10f}")
    else:
        n = 4096
        parent = _make_exp_profile(n, 1.5)
        print(f"No parent, using exp profile n={n}")

    # --- Phase 0: Diverse initialization (30s) ---
    phase0_end = min(deadline, t_start + 30.0)
    candidates = []

    if parent is not None:
        sc_p = _score_norm(parent)
        candidates.append((sc_p, parent.copy(), "parent"))
        update_best(parent, sc_p, "parent")

    n_target = 4096
    if parent is not None and len(parent) != n_target:
        p_interp = _interp_profile(parent, n_target)
        sc_i = _score_norm(p_interp)
        candidates.append((sc_i, p_interp, "parent_interp"))
        update_best(p_interp, sc_i, "parent_interp")

    profiles = [
        (_make_exp_profile(n_target, 0.5), "exp0.5"),
        (_make_exp_profile(n_target, 1.0), "exp1.0"),
        (_make_exp_profile(n_target, 1.5), "exp1.5"),
        (_make_exp_profile(n_target, 2.0), "exp2.0"),
        (_make_exp_profile(n_target, 3.0), "exp3.0"),
        (_make_exp_profile(n_target, 4.0), "exp4.0"),
        (_make_skew_profile(n_target, 1.0), "skew1"),
        (_make_skew_profile(n_target, 2.0), "skew2"),
        (_make_skew_profile(n_target, 3.0), "skew3"),
        (_make_linear_profile(n_target, 2.0), "lin2"),
        (_make_linear_profile(n_target, 3.0), "lin3"),
        (_make_quadratic_profile(n_target, 1.0, 0.3), "quad1"),
        (_make_quadratic_profile(n_target, 0.5, 0.5), "quad2"),
    ]

    for p, label in profiles:
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm(p)
        candidates.append((sc, p, label))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0: best_init_score={candidates[0][0]:.10f} [{candidates[0][2]}]")
    for sc, p, label in candidates[:5]:
        print(f"  {label}: {sc:.10f}")

    # Start from the best candidate
    if best_seq is None:
        best_seq = candidates[0][1].copy()
        best_score = candidates[0][0]

    # --- Phase 1: L-BFGS continuation (400s) ---
    phase1_end = min(deadline, t_start + 430.0)
    if time.monotonic() < phase1_end:
        per_time = min(350.0, phase1_end - time.monotonic() - 10)
        if per_time > 30:
            p_best, sc_best = _run_lbfgs(best_seq, phase1_end - 5, max_wall_time=per_time)
            update_best(p_best, sc_best, "lbfgs_p1")

    # Also try second-best if time allows
    if time.monotonic() < phase1_end - 60:
        for sc, p, label in candidates[1:4]:
            if time.monotonic() > phase1_end - 30:
                break
            if sc > best_score * 1.05:
                continue
            per_time = min(50.0, phase1_end - time.monotonic() - 15)
            if per_time > 10:
                p_best, sc_best = _run_lbfgs(p, phase1_end - 10, max_wall_time=per_time)
                update_best(p_best, sc_best, f"lbfgs_{label}")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Top-k subgradient (350s) ---
    phase2_end = min(deadline, t_start + 780.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        k_schedule = [100, 50, 30, 20, 15, 10, 8, 6, 5, 4, 3]
        p_best, sc_best = _topk_optimize(best_seq, phase2_end - 5, k_schedule,
                                         lr_init=0.006, mom_init=0.88)
        update_best(p_best, sc_best, "topk_p2")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Adam polishing (300s) ---
    phase3_end = min(deadline, t_start + 1080.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        per_time = min(280.0, phase3_end - time.monotonic() - 10)
        if per_time > 30:
            p_best, sc_best = _adam_optimize(best_seq, min(phase3_end - 5, time.monotonic() + per_time),
                                             lr_init=0.025, beta_frac_init=0.008)
            update_best(p_best, sc_best, "adam_p3")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Top-k with very small k (250s) ---
    phase4_end = min(deadline, t_start + 1330.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        k_schedule_fine = [5, 4, 3, 3, 2, 2]
        p_best, sc_best = _topk_optimize(best_seq, phase4_end - 5, k_schedule_fine,
                                         lr_init=0.003, mom_init=0.95)
        update_best(p_best, sc_best, "topk_fine_p4")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multi-restart (250s) ---
    phase5_end = min(deadline, t_start + 1580.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        n_best = len(best_seq)
        n_restarts = 4
        per_restart = min(60.0, (phase5_end - time.monotonic()) / max(1, n_restarts) - 5)

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 15:
                break
            noise_levels = [0.03, 0.05, 0.08, 0.12]
            nl = noise_levels[i % len(noise_levels)]
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + nl * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            per_dl = min(phase5_end - 10, time.monotonic() + per_restart)
            k_short = [10, 5, 3]
            p_best, sc_best = _topk_optimize(p_start, per_dl, k_short,
                                             lr_init=0.005, mom_init=0.85)
            update_best(p_best, sc_best, f"restart{i}")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Final fine polish (remaining time, ~200s) ---
    phase6_end = deadline
    if best_seq is not None and time.monotonic() < phase6_end:
        # Adam with very small beta
        remaining = phase6_end - time.monotonic()
        if remaining > 60:
            per_time = min(100.0, remaining - 30)
            p_best, sc_best = _adam_optimize(best_seq, min(phase6_end - 30, time.monotonic() + per_time),
                                             lr_init=0.01, beta_frac_init=0.002)
            update_best(p_best, sc_best, "final_adam")

        if time.monotonic() < phase6_end - 20:
            remaining2 = phase6_end - time.monotonic()
            if remaining2 > 30:
                per_time2 = min(60.0, remaining2 - 10)
                k_very_fine = [3, 2, 2, 1]
                p_best, sc_best = _topk_optimize(best_seq, min(phase6_end - 5, time.monotonic() + per_time2),
                                                 k_very_fine, lr_init=0.002, mom_init=0.97)
                update_best(p_best, sc_best, "final_topk")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_exp_profile(4096, 1.5)

        sc_final = _score_full(final_seq)
        print(f"FINAL internal score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"FINAL evaluator score={sc_eval:.10f}")
            else:
                print(f"Evaluator returned inf/NaN")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_exp_profile(4096, 1.5)
    sc_fb = _score_full(fallback)
    print(f"FALLBACK: n=4096, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
```
