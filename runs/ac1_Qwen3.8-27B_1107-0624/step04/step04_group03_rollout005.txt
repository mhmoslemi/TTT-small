def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _autocorr(a):
    """Compute full autocorrelation a * a (convolve with itself)."""
    return _full_convolve(a, a)


def _score_norm(a):
    """Score with sum(a) normalized to 1: 2*n*max(autocorr)."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    c = _autocorr(a_norm)
    return float(2.0 * n * c.max())


def _score_raw(a):
    """Original score: 2*n*max(autocorr)/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _autocorr(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _log_sum_exp(c, beta):
    """Compute log(sum(exp(c/beta))) * beta + max(c) in a stable way."""
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(np.clip(c_shifted, -50, 50))
    s = float(e.sum())
    if s < 1e-300:
        return cm
    return cm + beta * np.log(s)


def _smooth_objective(z, beta):
    a = _softmax(z)
    n = len(a)
    c = _autocorr(a)
    return 2.0 * n * _log_sum_exp(c, beta)


def _smooth_gradient(z, beta):

    n = len(a)
    c = _autocorr(a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(np.clip(c_shifted, -50, 50))
    s = float(e.sum())

        w[int(np.argmax(c))] = 1.0

    W = float(np.dot(w, c))

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()

    p = np.interp(x1, x0, p0)

    return p / p.sum()

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _autocorr(a):
    return _full_convolve(a, a)


def _score_norm(a):
    n = len(a)
    s = float(a.sum())

    e = np.exp(np.clip(c_shifted, -50, 50))

def _dual_direction(a_init, iterations=200):
    """
    Approximate the LP dual direction: find b in simplex maximizing sum(b)
    subject to (a*b)_k <= M for all k. Use projected gradient on b.
    """
    n = len(a_init)
    a_norm = a_init / a_init.sum()
    c = _autocorr(a_norm)
    M = float(c.max())

    b = np.ones(n) / n
    lr = 0.01
    for it in range(iterations):
        # Compute a * b (cross-correlation)
        ab = _full_convolve(a_norm, b)
        # Gradient of constraint violation
        violation = np.maximum(0, ab - M)
        # We want to increase sum(b) while keeping ab <= M
        # Subgradient of max(0, ab - M) w.r.t. b:
        a_rev = a_norm[::-1]
        viol_conv = _full_convolve(violation, a_rev)
        grad = viol_conv[n - 1:2 * n - 1]
        # Update: increase b where gradient is small, decrease where large
        b = b + lr * (1.0 / n - grad)
        b = np.maximum(b, 0)
        b = b / b.sum()
    return b

                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_parent = _score_norm(parent)
        print(f"Parent: n={n}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n = 4096
        parent = _make_cos_profile(n)
        print(f"No parent, using cos profile n={n}")

    n_use = len(parent)

    # --- Phase 0: Diverse initialization evaluation (25s) ---
    phase0_end = min(deadline, t_start + 25.0)
    init_candidates = [(parent.copy(), "parent")]

    if n_use == 4096:
        init_candidates.append((_make_cos_profile(n_use), "cos"))
        init_candidates.append((_make_gaussian_profile(n_use, 0.25), "gauss_025"))
        init_candidates.append((_make_gaussian_profile(n_use, 0.35), "gauss_035"))
        init_candidates.append((_make_gaussian_profile(n_use, 0.45), "gauss_045"))
        init_candidates.append((_make_tent_profile(n_use), "tent"))
        init_candidates.append((_make_raised_cosine(n_use), "raised_cos"))
        init_candidates.append((_make_bump_profile(n_use, 0.7), "bump_07"))
        init_candidates.append((_make_linear_profile(n_use, 2.0), "linear_2"))

    for p, label in init_candidates:
        sc = _score_norm(p)
        print(f"  Init '{label}': score={sc:.10f}")
        update_best(p, sc, f"init_{label}")

    print(f"Phase 0 done: best={best_score:.10f}")

    # --- Phase 1: L-BFGS on best (320s) ---
    phase1_end = min(deadline, t_start + 345.0)
    if best_seq is not None and time.monotonic() < phase1_end:
        p_best, sc_best = _lbfgs_continuation(best_seq, phase1_end)
        update_best(p_best, sc_best, "lbfgs_p1")
    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: L-BFGS on second-best init (100s) ---
    phase2_end = min(deadline, t_start + 445.0)
    if time.monotonic() < phase2_end:
        for p, label in init_candidates[1:4]:
            if time.monotonic() > phase2_end - 10:
                break
            sc_init = _score_norm(p)
            if sc_init < best_score + 0.08:
                p_b, sc_b = _lbfgs_continuation(p, phase2_end)
                update_best(p_b, sc_b, f"lbfgs2_{label}")
    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Mirror descent (250s) ---
    phase3_end = min(deadline, t_start + 695.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        c_curr = _autocorr(best_seq)
        max_c = float(c_curr.max())
        p_best, sc_best = _mirror_descent(
            best_seq, phase3_end,
            beta_init_frac=0.008, beta_min_frac=1e-9,
            beta_decay=0.93, lr_init=0.008, mom_init=0.90
        )
        update_best(p_best, sc_best, "mirror_p3")
    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Adam (200s) ---
    phase4_end = min(deadline, t_start + 895.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        p_best, sc_best = _adam_optimize(best_seq, phase4_end, lr=0.012, beta_frac=0.003)
        update_best(p_best, sc_best, "adam_p4")
    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Active set polish (150s) ---
    phase5_end = min(deadline, t_start + 1045.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        p_best, sc_best = _active_set_polish(best_seq, phase5_end, top_pct=0.5)
        update_best(p_best, sc_best, "active_p5")
    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Multi-restart (280s) ---
    phase6_end = min(deadline, t_start + 1325.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        n_best = len(best_seq)
        n_restarts = 6
        per_restart = min(45.0, (phase6_end - time.monotonic()) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 15:
                break
            noise_level = 0.02 + 0.015 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            if i % 3 == 0:
                z = z + noise_level * rng.normal(size=n_best)
            elif i % 3 == 1:
                n_pert = n_best // 8
                idx = rng.choice(n_best, size=n_pert, replace=False)
                z[idx] += noise_level * rng.normal(size=n_pert)
            else:
                # Low-frequency perturbation
                freq = rng.integers(1, 5)
                t_arr = np.arange(n_best) / n_best
                z = z + noise_level * np.sin(2 * np.pi * freq * t_arr + rng.uniform(0, 2*np.pi))
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax(z)
            per_dl = min(phase6_end - 10, time.monotonic() + per_restart)
            p_b, sc_b = _lbfgs_continuation(p_start, per_dl)
            update_best(p_b, sc_b, f"restart{i}")
    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final fine polish (remaining) ---
    phase7_end = deadline
    if best_seq is not None and time.monotonic() < phase7_end:
        remaining = phase7_end - time.monotonic()
        if remaining > 50:
            per_time = min(55.0, remaining - 25)
            if per_time > 15:
                c_curr = _autocorr(best_seq)
                mc = float(c_curr.max())
                p_best, sc_best = _mirror_descent(
                    best_seq, min(phase7_end - 25, time.monotonic() + per_time),
                    beta_init_frac=0.001, beta_min_frac=1e-11,
                    beta_decay=0.92, lr_init=0.004, mom_init=0.93
                )
                update_best(p_best, sc_best, "fine_mirror_p7")

        if time.monotonic() < phase7_end - 25:
            per_time2 = min(45.0, phase7_end - time.monotonic() - 10)
            if per_time2 > 10:
                p_best, sc_best = _active_set_polish(best_seq, min(phase7_end - 10, time.monotonic() + per_time2), top_pct=0.3)
                update_best(p_best, sc_best, "fine_active_p7")

        if time.monotonic() < phase7_end - 10:
            per_time3 = min(35.0, phase7_end - time.monotonic() - 5)
            if per_time3 > 8:
                p_best, sc_best = _adam_optimize(best_seq, min(phase7_end - 2, time.monotonic() + per_time3), lr=0.004, beta_frac=0.001)
                update_best(p_best, sc_best, "fine_adam_p7")

    print(f"Phase 7 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_cos_profile(4096)
        sc_final = _score_raw(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator non-finite, internal: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")
        return [float(x) for x in final_seq]

    fallback = _make_cos_profile(4096)
    print(f"FALLBACK: n=4096, score={_score_raw(fallback):.10f}")
    return [float(x) for x in fallback]
