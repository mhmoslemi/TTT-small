import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 0.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):

        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w

        p = np.exp(z - z.max())
        p = p / p.sum()
        c = _full_convolve(p)
        maxc = float(c.max())
        sc = float(2.0 * n * maxc)

        if np.isfinite(sc) and sc < best_score - 1e-13:
            best_score = sc
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Adaptive beta
        if t_adam % 25 == 0:
            beta = max(1e-6, beta * beta_decay)
            if c.size >= 2:
                sorted_c = np.sort(c)[::-1]
                gap = sorted_c[0] - sorted_c[1]
                if gap < beta * 0.03:
                    beta = max(1e-6, beta * 0.5)

        # Periodic LP
        if use_lp and time.monotonic() - last_lp_time > lp_interval and no_imp > 50:
            last_lp_time = time.monotonic()
            M = float(c.max()) * 1.001
            b_sol = _lp_solve_quick(p, M)
            if b_sol is not None and b_sol.sum() > 1e-10:
                b_norm = b_sol / b_sol.sum()
                for t_blend in [0.6, 0.4, 0.25]:
                    p_trial = (1 - t_blend) * p + t_blend * b_norm
                    p_trial = np.maximum(p_trial, 1e-15)
                    p_trial = p_trial / p_trial.sum()
                    sc_trial = _score(p_trial)
                    if sc_trial < best_score - 1e-12:
                        best_score = sc_trial
                        best_p = p_trial.copy()
                        best_z = np.log(np.maximum(p_trial, 1e-15))
                        best_z -= np.mean(best_z)
                        p = p_trial.copy()
                        c = _full_convolve(p)
                        maxc = float(c.max())
                        beta = max(1e-3, beta_frac * maxc)
                        no_imp = 0
                        break

        # Restart if stuck
        if no_imp > no_imp_limit:
            ns = 0.05 if best_score < 1.51 else 0.1
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z -= float(np.mean(z))
            p = np.exp(z - z.max())
            p = p / p.sum()
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _lp_solve_quick(a, M):
    """Fast LP: max sum(b) s.t. (a*b)_k <= M, b >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend((a[k - idx]).tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    obj = -np.ones(n)
    res = linprog(obj, A_ub=A, b_ub=M * np.ones(2 * n - 1),
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': 30})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _peak_peel(a, deadline, n_peaks=5):
    """Directly reduce top peaks by redistributing mass from contributing positions."""
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng(999)

    while time.monotonic() < deadline:
        c = _full_convolve(a)
        M = float(c.max())
        sc = float(2.0 * n * M)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()

        # Get top peaks
        peak_indices = np.argsort(c)[::-1][:n_peaks]

        # For each peak, find dominant contributors and reduce them
        improved = False
        for k in peak_indices:
            c = _full_convolve(a)
            if c[k] < M * 0.8:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            contrib = a[js:je + 1] * a[k - np.arange(js, je + 1)]
            total_contrib = float(contrib.sum())
            if total_contrib < 1e-20:
                continue

            # Find the top contributor
            top_local = int(np.argmax(contrib))
            i = js + top_local

            # Find a safe target: position not contributing to top peaks
            contrib_to_peaks = np.zeros(n)
            for kp in peak_indices[:3]:
                s2 = max(0, kp - n + 1)
                e2 = min(n - 1, kp)
                for ii in range(s2, e2 + 1):
                    contrib_to_peaks[ii] += a[kp - ii]

            # Target: position with low contribution to peaks and low current value
            target_score = contrib_to_peaks * 0.5 + a * 0.5
            # Exclude position i
            target_score[i] = np.inf
            target = int(np.argmin(target_score))

            # Move small fraction of mass
            frac = 0.005 + 0.015 * (a[i] * n)
            frac = min(frac, 0.15)
            amount = a[i] * frac
            a[i] -= amount
            a[target] += amount
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            improved = True

        if not improved:
            # Perturb and retry
            idx = rng.integers(0, n)
            a[idx] *= 0.95
            a = a / a.sum()

        # Safety: if score got too bad, revert
        c = _full_convolve(a)
        if float(c.max()) > M * 1.5:
            a = best_a.copy()

    return best_a, best_score


def _lp_refine(a, deadline, max_solves=30):
    """LP refinement with adaptive step size and M shrinking."""
    n = a.size
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve_quick(a, M * 1.002)
        if b is None or b.sum() < 1e-10:
            break

        b_norm = b / b.sum()
        # Try multiple blend factors
        for t_blend in [0.5, 0.7, 0.3]:
            a_trial = (1 - t_blend) * a + t_blend * b_norm
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial = a_trial / a_trial.sum()
            sc = _score(a_trial)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
                c = _full_convolve(a)
                M = float(c.max()) * 0.998
                break
        else:
            # No improvement, shrink M more aggressively
            M = M * 0.995
            if M < best_score / (2.0 * n) * 1.001:
                break

    return best_a, best_score


def _interp_to_size(a, n_new):
    n0 = a.size
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x1, x0, a))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _try_update(p, label):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        p = p / s
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} n={p.size} [{label}] t={time.monotonic()-t_start:.0f}s")

    # Get parent
    parent = None
    try:
        pr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if pr.size > 0:
            pr = np.clip(pr, 0.0, 1000.0)
            if pr.sum() > 0.01:
                parent = pr / pr.sum()
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={parent.size}, score={_score(parent):.10f}")
    else:
        n0 = 1024
        t = np.linspace(0, 1, n0)
        parent = np.maximum(1.0 + 2.0 * (t - 0.5), 1e-6)
        parent = parent / parent.sum()
        print(f"No parent, using linear init n={n0}, score={_score(parent):.10f}")

    _try_update(parent, 'parent')

    # Phase 1: Fast Adam polish (250s)
    p1_end = t_start + 250.0
    print(f"Phase 1: Adam polish (250s) score={best_score:.10f}")
    p1, sc1 = _adam_optimize(parent, min(p1_end, deadline), rng, lr=0.05,
                             beta_frac=0.08, beta_decay=0.94, no_imp_limit=300,
                             use_lp=False)
    _try_update(p1, 'adam1')

    # Phase 2: Adam + LP (400s)
    p2_end = t_start + 650.0
    print(f"Phase 2: Adam+LP (400s) score={best_score:.10f}")
    if best_seq is not None:
        p2, sc2 = _adam_optimize(best_seq, min(p2_end, deadline), rng, lr=0.04,
                                 beta_frac=0.06, beta_decay=0.95, no_imp_limit=400,
                                 use_lp=True, lp_interval=45.0)
        _try_update(p2, 'adam2_lp')

    # Phase 3: LP intensive (250s)
    p3_end = t_start + 900.0
    print(f"Phase 3: LP intensive (250s) score={best_score:.10f}")
    if best_seq is not None and best_seq.size <= 2048:
        p3, sc3 = _lp_refine(best_seq, min(p3_end, deadline), max_solves=40)
        _try_update(p3, 'lp_heavy')

        # Quick Adam after LP
        if time.monotonic() < p3_end - 20:
            rem = p3_end - time.monotonic() - 10
            if rem > 30:
                p3a, _ = _adam_optimize(best_seq, time.monotonic() + rem, rng,
                                        lr=0.03, beta_frac=0.05, beta_decay=0.96,
                                        no_imp_limit=200, use_lp=False)
                _try_update(p3a, 'post_lp_adam')

    # Phase 4: Peak peeling (200s)
    p4_end = t_start + 1100.0
    print(f"Phase 4: Peak peeling (200s) score={best_score:.10f}")
    if best_seq is not None:
        p4, sc4 = _peak_peel(best_seq, min(p4_end, deadline), n_peaks=5)
        _try_update(p4, 'peel')

        # Adam after peeling
        if time.monotonic() < p4_end - 20:
            rem = p4_end - time.monotonic() - 10
            if rem > 30:
                p4a, _ = _adam_optimize(best_seq, time.monotonic() + rem, rng,
                                        lr=0.02, beta_frac=0.04, beta_decay=0.97,
                                        no_imp_limit=150, use_lp=False)
                _try_update(p4a, 'post_peel_adam')

    # Phase 5: Try n=2048 (250s)
    p5_end = t_start + 1350.0
    print(f"Phase 5: Multi-scale n=2048 (250s) score={best_score:.10f}")
    if time.monotonic() < p5_end and best_seq is not None and best_seq.size <= 1024:
        n_new = 2048
        p_int = _interp_to_size(best_seq, n_new)
        _try_update(p_int, f'interp_{n_new}')

        if p_int is not None:
            rem = min(120.0, p5_end - time.monotonic() - 15)
            if rem > 40:
                p5, _ = _adam_optimize(p_int, time.monotonic() + rem, rng,
                                       lr=0.04, beta_frac=0.07, beta_decay=0.94,
                                       no_imp_limit=250, use_lp=False)
                _try_update(p5, f'adam_n{n_new}')

            if time.monotonic() < p5_end - 30:
                rem2 = min(60.0, p5_end - time.monotonic() - 10)
                if rem2 > 25:
                    p5l, _ = _lp_refine(best_seq, time.monotonic() + rem2, max_solves=15)
                    _try_update(p5l, 'lp_n1024')

    # Phase 6: Final intensive polish (remaining)
    p6_end = deadline
    print(f"Phase 6: Final polish score={best_score:.10f}")
    if best_seq is not None and time.monotonic() < p6_end - 10:
        # Alternate Adam and LP
        while time.monotonic() < p6_end - 10:
            rem = p6_end - time.monotonic() - 10
            if rem < 20:
                break
            # Adam
            chunk = min(rem * 0.5, 80.0)
            if chunk > 15:
                pa, _ = _adam_optimize(best_seq, time.monotonic() + chunk, rng,
                                       lr=0.015, beta_frac=0.03, beta_decay=0.97,
                                       no_imp_limit=100, use_lp=False)
                _try_update(pa, 'final_adam')
            # LP
            if time.monotonic() < p6_end - 10 and best_seq.size <= 2048:
                rem2 = p6_end - time.monotonic() - 10
                if rem2 > 30:
                    pl, _ = _lp_refine(best_seq, time.monotonic() + min(50, rem2),
                                       max_solves=8)
                    _try_update(pl, 'final_lp')

    # Final
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            n0 = 1024
            t = np.linspace(0, 1, n0)
            final = np.maximum(1.0 + 2.0*(t - 0.5), 1e-6)
            final = final / final.sum()
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f} n={final.size}")
        try:
            ev = float(evaluate_sequence([float(x) for x in final]))
            print(f"EVAL confirms: {ev:.10f}")
        except Exception:
            pass
        return [float(x) for x in final]

    # Fallback
    n0 = 1024
    t = np.linspace(0, 1, n0)
    fb = np.maximum(1.0 + 2.0 * (t - 0.5), 1e-6)
    fb = fb / fb.sum()
    print(f"FALLBACK score={_score(fb):.10f}")
    return [float(x) for x in fb]
