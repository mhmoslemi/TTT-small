"""
Step function optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive logsumexp-topK continuation with L-BFGS, mirror descent polishing,
multi-scale n exploration, and coordinate descent refinement.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _obj_grad(z, n, beta, k_top):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shift = (c - cm) / beta
    e = np.exp(c_shift)
    s = e.sum()
    if s < 1e-300:
        W_ls = np.zeros_like(c)
        W_ls[int(np.argmax(c))] = 1.0
        ls_val = cm
    else:
        W_ls = e / s
        ls_val = cm + beta * float(np.log(s))
    k_actual = min(k_top, len(c))
    top_idx = np.argsort(c)[-k_actual:]
    W_k = np.zeros_like(c)
    W_k[top_idx] = 1.0 / k_actual
    topk_val = float(c[top_idx].mean())
    obj = 2.0 * n * (0.5 * ls_val + 0.5 * topk_val)
    a_flip = a[::-1]
    h_ls = _full_convolve(W_ls, a_flip)[n - 1:2 * n - 1]
    h_k = _full_convolve(W_k, a_flip)[n - 1:2 * n - 1]
    W_val_ls = float(np.dot(W_ls, c))
    W_val_k = float(np.dot(W_k, c))
    grad_a_ls = 4.0 * n * h_ls
    grad_a_k = 4.0 * n * h_k
    grad_a = 0.5 * grad_a_ls + 0.5 * grad_a_k
    W_combined = 0.5 * W_val_ls + 0.5 * W_val_k
    h_combined = 0.5 * h_ls + 0.5 * h_k
    grad_z = a * (grad_a - 4.0 * n * W_combined * np.ones(n))
    grad_z = a * (4.0 * n * h_combined - 4.0 * n * W_combined)
    if not np.all(np.isfinite(grad_z)):
        grad_z = np.zeros_like(z)
    return obj, grad_z


def _obj_grad_pure(z, n, beta):
    """Pure logsumexp objective and gradient."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shift = (c - cm) / beta
    e = np.exp(c_shift)
    s = e.sum()
    if s < 1e-300:
        W = np.zeros_like(c)
        W[int(np.argmax(c))] = 1.0
        val = cm
    else:
        W = e / s
        val = cm + beta * float(np.log(s))
    obj = 2.0 * n * val
    a_flip = a[::-1]
    h = _full_convolve(W, a_flip)[n - 1:2 * n - 1]
    W_val = float(np.dot(W, c))
    grad_z = a * (4.0 * n * h - 4.0 * n * W_val)
    if not np.all(np.isfinite(grad_z)):
        grad_z = np.zeros_like(z)
    return obj, grad_z


def _obj_grad_topk(z, n, k_top):
    """Pure top-K average objective and gradient."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    k_actual = min(k_top, len(c))
    top_idx = np.argsort(c)[-k_actual:]
    W = np.zeros_like(c)
    W[top_idx] = 1.0 / k_actual
    val = float(c[top_idx].mean())
    obj = 2.0 * n * val
    a_flip = a[::-1]
    h = _full_convolve(W, a_flip)[n - 1:2 * n - 1]
    W_val = float(np.dot(W, c))
    grad_z = a * (4.0 * n * h - 4.0 * n * W_val)
    if not np.all(np.isfinite(grad_z)):
        grad_z = np.zeros_like(z)
    return obj, grad_z


def _run_continuation(a_init, deadline, beta_schedule, k_schedule):
    """Run L-BFGS with logsumexp-topK continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    n_levels = len(beta_schedule)

    for level in range(n_levels):
        if time.monotonic() > deadline - 2.0:
            break
        beta = beta_schedule[level]
        k_top = k_schedule[min(level, len(k_schedule) - 1)]
        remaining_levels = n_levels - level
        time_per_level = max(3.0, (deadline - time.monotonic()) / remaining_levels)

        if beta < 1e-6 * _score_norm(a_init):
            # Use pure top-K
            def obj_fn(z_):
                return _obj_grad_topk(z_, n, k_top)[0]

            def jac_fn(z_):
                return _obj_grad_topk(z_, n, k_top)[1]
        else:
            def obj_fn(z_):
                return _obj_grad(z_, n, beta, k_top)[0]

            def jac_fn(z_):
                return _obj_grad(z_, n, beta, k_top)[1]

        try:
            result = minimize(
                obj_fn, z, jac=jac_fn,
                method='L-BFGS-B',
                options={'maxiter': 800, 'ftol': 1e-20, 'gtol': 1e-12,
                         'maxls': 80, 'maxfun': 5000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z = result.x
        except Exception:
            pass

        a_new = _softmax_fn(z)
        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()

    return best_a, best_score


def _mirror_descent(a_init, deadline, n_steps=5000, eta0=0.5, decay=0.9999):
    """Multiplicative mirror descent for final polishing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    eta = eta0

    for it in range(n_steps):
        if time.monotonic() > deadline - 1.0:
            break
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        k_actual = 3
        top_idx = np.argsort(c)[-k_actual:]
        W = np.zeros_like(c)
        W[top_idx] = 1.0 / k_actual
        a_flip = a[::-1]
        h = _full_convolve(W, a_flip)[n - 1:2 * n - 1]
        W_val = float(np.dot(W, c))
        grad_z = a * (4.0 * n * h - 4.0 * n * W_val)
        # Scale gradient
        grad_scale = float(np.max(np.abs(grad_z))) + 1e-15
        eta = eta0 / (1.0 + 0.01 * it)
        z = z - eta * grad_z / grad_scale
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        if it % 50 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
    return best_a, best_score


def _coordinate_descent(a_init, deadline, n_sweeps=5):
    """Cyclic coordinate descent: adjust one element at a time."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(42)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline - 1.0:
            break
        order = rng.permutation(n)
        for idx in order:
            if time.monotonic() > deadline - 1.0:
                break
            old_val = a[idx]
            # Try a few perturbed values
            for mult in [0.8, 0.9, 1.1, 1.2, 0.7, 1.3]:
                a_trial = a.copy()
                a_trial[idx] = old_val * mult
                a_trial = np.maximum(a_trial, 1e-15)
                a_trial /= a_trial.sum()
                sc = _score_norm(a_trial)
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    break
    return best_a, best_score


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power_profile(n, p):
    t = np.linspace(0.0, 1.0, n)
    u = np.abs(2.0 * t - 1.0)
    a = (1.0 - u * 0.5) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_raised_cosine(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5) * alpha))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_sinusoidal(n, freq, phase=0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.3 * np.sin(2.0 * np.pi * freq * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

    # Phase 5: Coordinate descent (100s)
    phase5_end = min(deadline, t_start + 1220.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        dl = phase5_end - 5.0
        p_best, sc_best = _coordinate_descent(best_seq, dl, n_sweeps=3)
        update_best(p_best, sc_best, "coord")

    print(f"Phase 5 done: best={best_score:.10f}")

    # Phase 6: Multi-restart (200s)
    phase6_end = min(deadline, t_start + 1420.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        n_restarts = 4
        per_restart = max(30.0, (phase6_end - time.monotonic()) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 10:
                break
            noise = 0.02 + 0.04 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=len(best_seq))
            z = np.clip(z, -40, 40)
            z = z - z.mean()
            p_start = _softmax_fn(z)
            dl = min(phase6_end - 5, time.monotonic() + per_restart)
            c_re = _full_convolve(p_start, p_start)
            max_cr = float(c_re.max())
            r_betas = [0.005 * max_cr, 0.002 * max_cr, 0.001 * max_cr,
                       0.0005 * max_cr, 0.0002 * max_cr, 0.0001 * max_cr]
            r_ks = [5, 3, 2, 2, 1, 1]
            p_best, sc_best = _run_continuation(p_start, dl, r_betas, r_ks)
            update_best(p_best, sc_best, f"restart{i}")

    print(f"Phase 6 done: best={best_score:.10f}")

    # Phase 7: Try different n (200s)
    phase7_end = min(deadline, t_start + 1620.0)
    if best_seq is not None and time.monotonic() < phase7_end:
        n_curr = len(best_seq)
        for target_n in [2048, 8192]:
            if target_n == n_curr or time.monotonic() > phase7_end - 30:
                continue
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{target_n}")
            if sc_interp < best_score * 1.01 and time.monotonic() < phase7_end - 30:
                c_ti = _full_convolve(p_interp, p_interp)
                max_cti = float(c_ti.max())
                ti_betas = [0.003 * max_cti, 0.001 * max_cti, 0.0005 * max_cti,
                            0.0002 * max_cti, 0.0001 * max_cti]
                ti_ks = [5, 3, 2, 1, 1]
                dl = min(phase7_end - 10, time.monotonic() + 120.0)
                p_best, sc_best = _run_continuation(p_interp, dl, ti_betas, ti_ks)
                update_best(p_best, sc_best, f"opt_n{target_n}")

    print(f"Phase 7 done: best={best_score:.10f}")

    # Phase 8: Final intensive polishing (remaining)
    if best_seq is not None:
        remaining = max(10.0, deadline - time.monotonic())
        n_polish = 3
        per_polish = remaining / max(1, n_polish)

        for i in range(n_polish):
            if time.monotonic() > deadline - 5:
                break
            if i == 0:
                # Very fine continuation
                c_pf = _full_convolve(best_seq, best_seq)
                max_cpf = float(c_pf.max())
                pf_betas = [0.0001 * max_cpf, 0.00005 * max_cpf, 0.00002 * max_cpf,
                            0.00001 * max_cpf, 0.000005 * max_cpf]
                pf_ks = [2, 1, 1, 1, 1]
                dl = min(deadline - 5, time.monotonic() + per_polish)
                p_best, sc_best = _run_continuation(best_seq, dl, pf_betas, pf_ks)
                update_best(p_best, sc_best, f"final_cont_{i}")
            elif i == 1:
                dl = min(deadline - 5, time.monotonic() + per_polish)
                p_best, sc_best = _mirror_descent(best_seq, dl, n_steps=8000, eta0=0.1)
                update_best(p_best, sc_best, f"final_mirror_{i}")
            else:
                dl = min(deadline - 5, time.monotonic() + per_polish)
                p_best, sc_best = _coordinate_descent(best_seq, dl, n_sweeps=2)
                update_best(p_best, sc_best, f"final_coord_{i}")

    # Final validation and return
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_linear_profile(4096, 1.5)
            s = float(final_seq.sum())
        final_seq = final_seq / s
        sc_final = _score(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        if len(final_seq) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_linear_profile(4096, 1.5)
    sc_fb = _score(fallback)
    print(f"FALLBACK: n=4096, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
