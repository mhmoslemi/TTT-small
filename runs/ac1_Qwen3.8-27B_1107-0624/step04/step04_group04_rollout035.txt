"""
Optimization of step function heights to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses power-sum smoothing with L-BFGS continuation, spectral initialization,
peak shaving, and multi-scale restarts.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _power_sum_obj(z, p, n):
    """Power-sum smoothed objective in logit space."""
    a = _softmax(z)
    c = _fft_convolve(a, a)
    if p == np.inf or p > 1e6:
        return float(2.0 * n * c.max())
    cp = c ** p
    sp = float(np.sum(cp)) ** (1.0 / p)
    return float(2.0 * n * sp)

    """Gradient of power-sum smoothed objective in logit space."""

        sp = (float(np.sum(c ** p))) ** (1.0 / p)
        sp_factor = sp ** (1.0 - p) * 2.0
        # h_i = sum_k cp[k] * a[k-i], computed via correlation
        N = _next_pow2(3 * n)
        W = np.fft.rfft(cp, N)

def _logsumexp_obj(z, beta, n):
    """Logsumexp smoothed objective in logit space."""
    a = _softmax(z)
    c = _fft_convolve(a, a)
    cm = float(c.max())
    c_shift = (c - cm) / beta
    e = np.exp(c_shift)
    ls = float(np.log(np.sum(e)))
    return float(2.0 * n * (cm + beta * ls))


def _logsumexp_grad(z, beta, n, top_k=0):
    """Gradient of logsumexp smoothed objective in logit space."""
    a = _softmax(z)
    c = _fft_convolve(a, a)
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        w = np.zeros_like(c)
        c_norm = (c_top - c_top.max()) / beta
        e = np.exp(c_norm)
        s = e.sum()
        if s < 1e-300:
            w[idx[-1]] = 1.0
        else:
            w[idx] = e / s
        W = float(np.dot(w, c))
    else:
        c_shift = (c - cm) / beta
        e = np.exp(c_shift)
        s = float(e.sum())
        if s < 1e-300:
            idx = int(np.argmax(c))
            w = np.zeros_like(c)
            w[idx] = 1.0
            W = cm
        else:
            w = e / s
            W = float(np.dot(w, c))
    # h_i = sum_k w_k * a[k-i] via correlation
    N = _next_pow2(3 * n)
    Wft = np.fft.rfft(w, N)
    Aft = np.fft.rfft(a, N)
    H = np.fft.irfft(Wft * np.conj(Aft), N)[:n]
    h = 2.0 * H  # factor of 2 from derivative of c_k
    dot_ah = float(np.dot(a, h))
    grad = a * (h - dot_ah)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _make_sine_profile(n, freq, phase=0.0, contrast=0.5, decay=0.0):
    """Sine-modulated profile with optional exponential decay."""
    t = np.arange(n)
    base = 1.0 + contrast * np.sin(2.0 * np.pi * freq * t / n + phase)
    if decay > 0:
        base = base * np.exp(-decay * t / n)
    base = np.maximum(base, 1e-10)
    return base / base.sum()


def _make_ramp_profile(n, slope):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_bimodal_profile(n, center, width, height):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + height * np.exp(-((t - center) / width) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    a = np.asarray(a, dtype=np.float64)
    n0 = len(a)
    if n0 == new_n:
        return a / a.sum()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, new_n)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _lbfgs_optimize(a_init, deadline, p_schedule=None, beta_schedule=None,
                    top_k=0, max_iter_total=2000):
    """Run L-BFGS with continuation on smoothing parameter."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    if p_schedule is not None:
        per_level = max(50, max_iter_total // len(p_schedule))
        for p in p_schedule:
            if time.monotonic() > deadline:
                break
            try:
                result = minimize(
                    _power_sum_obj, z, args=(p, n),
                    method='L-BFGS-B', jac=_power_sum_grad,
                    options={'maxiter': per_level, 'ftol': 1e-16,
                             'gtol': 1e-10, 'maxls': 20, 'maxfun': per_level * 2}
                )
                if result.x is not None and np.all(np.isfinite(result.x)):
                    z = result.x
                    a_new = _softmax(z)
                    sc = _score_norm(a_new)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
            except Exception:
                pass
    elif beta_schedule is not None:
        per_level = max(50, max_iter_total // len(beta_schedule))
        for beta in beta_schedule:
            if time.monotonic() > deadline:
                break
            try:
                result = minimize(
                    _logsumexp_obj, z, args=(beta, n),
                    method='L-BFGS-B', jac=_logsumexp_grad,
                    options={'maxiter': per_level, 'ftol': 1e-16,
                             'gtol': 1e-10, 'maxls': 20, 'maxfun': per_level * 2}
                )
                if result.x is not None and np.all(np.isfinite(result.x)):
                    z = result.x
                    a_new = _softmax(z)
                    sc = _score_norm(a_new)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
                        z = result.x.copy()
            except Exception:
                pass
    return best_a, best_score


def _adam_optimize(a_init, deadline, lr=0.03, p=16, n_iter=50000, top_k=0):
    """Adam optimizer with power-sum smoothing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    # Adaptive p: start moderate, increase over time
    cur_p = float(p)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax(z)
        grad = _power_sum_grad(z, cur_p, n)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        if t % 200 == 0:
            a = _softmax(z)
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
            # Increase p over time
            cur_p = min(cur_p * 1.15, 512.0)
            if best_score < 1.51:
                lr = max(0.005, lr * 0.98)

    # Final check
    a = _softmax(z)
    sc = _score_norm(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def _peak_shave(a_init, deadline, n_iter=2000):
    """Directly reduce elements contributing to the peak of the convolution."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        cm = float(c.max())
        # Active set: top peaks within 0.1% of max
        threshold = cm * 0.999
        active = np.where(c >= threshold)[0]

        if len(active) == 0:
            break

        # Compute pressure: h_i = mean over active k of a[k-i]
        h = np.zeros(n)
        for k in active[:50]:  # Limit to top 50 active for efficiency
            k = int(k)
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo <= hi:
                h[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
        h /= min(len(active), 50)

        # Find the direction that reduces the peak
        # Reduce a_i where h_i is above average, increase where below
        h_mean = float(np.dot(h, a))  # Weighted average
        diff = h - h_mean

        # Adaptive step size
        max_diff = float(np.abs(diff).max())
        if max_diff < 1e-15:
            break
        step = 0.02

        # Update in logit space
        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()
        z = z - step * diff / max_diff
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a_new = _softmax(z)

        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            z = np.log(np.maximum(a, 1e-15))
            z = z - z.mean()
            z = z - step * 0.3 * diff / max_diff
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            a_new = _softmax(z)
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
            else:
                step *= 0.5
                if step < 1e-6:
                    break

        # Progressively tighten
        if it % 200 == 199:
            step *= 0.9

    return best_a, best_score


def _subgradient_step(a, eta, n_active=10):
    """One projected subgradient step."""
    n = len(a)
    c = _fft_convolve(a, a)
    # Top n_active peaks
    idx = np.argsort(c)[-n_active:]
    # Subgradient: average of gradients at active peaks
    h = np.zeros(n)
    for k in idx:
        k = int(k)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            h[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    h /= len(idx) * 2.0  # The 2 is from the derivative

    # Projected gradient step on simplex
    a_new = a - eta * h
    # Project onto simplex {x >= 0, sum x = 1}
    a_new = _project_simplex(a_new)
    return a_new


def _project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    csv = mu.cumsum()
    rho = np.nonzero(mu * np.arange(1, n + 1) > (csv - 1))[0][-1]
    theta = (csv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: spectral initialization + power-sum L-BFGS continuation

    global height_sequence_1

            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # === Phase 0: Initializations ===
    print("=== Phase 0: Initializations ===")
    phase0_end = min(deadline, t_start + 30.0)
    candidates = []

    # Try parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm(parent)
                candidates.append((sc, parent.copy(), "parent"))
                update_best(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent: {e}")

    # Generate diverse profiles at n=4096
    n_base = 4096
    profiles = []
    # Sine profiles
    for freq in [1, 2, 3, 5, 8]:
        for contrast in [0.3, 0.5, 0.7]:
            for phase in [0.0, np.pi / 2]:
                profiles.append(_make_sine_profile(n_base, freq, phase, contrast))
    # Ramp profiles
    for slope in [-2.0, -1.0, -0.5, 0.5, 1.0, 2.0]:
        profiles.append(_make_ramp_profile(n_base, slope))
    # Bimodal
    for center in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for width in [0.1, 0.2, 0.3]:
            for height in [0.5, 1.0, 2.0]:
                profiles.append(_make_bimodal_profile(n_base, center * n_base, width * n_base, height))
    # Uniform
    profiles.append(np.ones(n_base) / n_base)
    # Power profiles
    for p in [0.5, 1.0, 2.0, -1.0]:
        a = (np.arange(n_base) + 1.0) ** p
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm(p)
        if np.isfinite(sc):
            candidates.append((sc, p, f"prof{idx}"))
            update_best(p, sc, f"prof{idx}")

    # Interpolate parent to n_base if needed
    if parent is not None and len(parent) != n_base:
        p_interp = _interp_to_length(parent, n_base)
        sc = _score_norm(p_interp)
        candidates.append((sc, p_interp, "parent_interp"))
        update_best(p_interp, sc, "parent_interp")

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 0 done: best={best_score:.10f}, {len(candidates)} candidates")

    # === Phase 1: L-BFGS with power-sum continuation ===
    print("=== Phase 1: L-BFGS power-sum continuation ===")
    phase1_end = min(deadline, t_start + 400.0)
    n_opt = 0

        print(f"  L-BFGS on {label} (n={len(p)}), budget={per_time:.0f}s")

        # Power-sum continuation: p = 2, 4, 8, 16, 32, 64
        p_sched = [2, 4, 8, 16, 32, 64]
        p_best, sc_best = _lbfgs_optimize(p, dl, p_schedule=p_sched, max_iter_total=1500)
        update_best(p_best, sc_best, f"ps_{label}")
        n_opt += 1

    # === Phase 2: Adam with increasing p ===
    print("=== Phase 2: Adam optimization ===")

    fallback = _make_sine_profile(4096, 3, 0.0, 0.5)
    print(f"  FALLBACK: n=4096, score={_score_norm(fallback):.10f}")
    return [float(x) for x in fallback]
