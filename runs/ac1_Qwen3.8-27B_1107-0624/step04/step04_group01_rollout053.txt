```python
"""
Hybrid LP-guided + Adam + L-BFGS optimization for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses softmax parameterization, adaptive beta-continuation for
log-sum-exp smoothing, LP-based direction kicks, and multi-restart.
"""

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import minimize as scipy_minimize
import cvxpy as cp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z(a, c, w, n):
    """Compute gradient of smoothed objective w.r.t. z."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _build_conv_matrix(n, a):
    """Build sparse (2n-1) x n matrix A where (A@b)[k] = (a*b)[k]."""
    d_idx, j_idx = np.meshgrid(np.arange(n), np.arange(n), indexing='ij')
    rows = (d_idx + j_idx).flatten()
    cols = j_idx.flatten()
    vals = a[d_idx.flatten()]
    return sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))


def _solve_lp(a, M):
    """LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k."""
    n = len(a)
    A = _build_conv_matrix(n, a)
    b = cp.Variable(n)
    obj = cp.Maximize(cp.sum(b))
    constraints = [b >= 0, A @ b <= M]
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=10000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None
    if b.value is None:
        return None
    b_val = np.maximum(0, np.array(b.value).flatten())
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def _adam_optimize(a_init, deadline, beta_list, max_iter, lr):
    """Adam in log-space with given beta schedule."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        while True:
            if time.monotonic() > deadline:
                break
            t += 1
            if t > max_iter:
                break
            
            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if not np.isfinite(s) or s < 1e-300:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            
            grad = _compute_grad_z(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = lr * mh / (np.sqrt(vh) + eps)
            un = np.linalg.norm(update)
            if un > 0.5:
                update *= 0.5 / un
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % 100 == 0:
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
    
    a_f = _softmax(z)
    sc_f = _score_norm(a_f)
    if np.isfinite(sc_f) and sc_f < best_sc:
        best_a = a_f.copy()
        best_sc = sc_f
    return best_a, best_sc


def _lbfgs_polish(a_init, deadline, beta=0.01, max_iter=500):
    """L-BFGS-B polish in log-space."""
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def obj_and_grad(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = c.max()
        # Smooth max
        c_shift = beta * (c - cm)
        e = np.exp(c_shift)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
            val = 2.0 * n * cm
        else:
            w = e / s
            val = 2.0 * n * (cm + (1.0/beta) * np.log(s))
        
        grad_z = _compute_grad_z(a, c, w, n)
        return val, grad_z
    
    result = scipy_minimize(
        obj_and_grad, z0, method='L-BFGS-B',
        jac=True, options={'maxiter': max_iter, 'maxfun': 2000,
                          'ftol': 1e-12, 'gtol': 1e-8}
    )
    
    if result.success and result.x is not None:
        z_opt = result.x
        z_opt = np.clip(z_opt, -50, 50)
        z_opt = z_opt - z_opt.mean()
        a_opt = _softmax(z_opt)
        sc = _score_norm(a_opt)
        if np.isfinite(sc):
            return a_opt, sc
    
    return a_init, _score_norm(a_init)
