def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _score(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    N = _next_pow2(2*n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    c = np.maximum(c, 0)
    return float(2*n*c.max() / (S*S))

def _gradient(a, beta):
    """
    Compute gradient of smooth objective w.r.t. a.
    Returns (grad, M_smooth) where grad = df/da.
    """
    n = len(a)
    S = a.sum()
    N = _next_pow2(2*n - 1)

    fa = np.fft.rfft(a, N)

2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses log-parameterization, Adam optimizer, multi-start, and coordinate polish.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _convolve_fft(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return 1e10
    c = _convolve_fft(a)
    return float(2.0 * n * c.max() / (S * S))

def _score_norm(a):
    n = len(a)
    c = _convolve_fft(a)
    return float(2.0 * n * c.max())

def _grad_smooth_max(a, beta):
    """Returns (gradient_in_a_space, M_smooth, c) where gradient is df/da for f=2n*max_smooth."""
    n = len(a)
    S = a.sum()
    N = _next_pow2(2 * n - 1)
    c = _convolve_fft(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = ex.sum()
    if not np.isfinite(s) or s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        M_smooth = mval
    else:
        w = ex / s
        M_smooth = mval + beta * float(np.log(s))
    A = np.fft.fft(a, N)
    W = np.fft.fft(w, N)
    r = np.fft.ifft(np.conj(A) * W).real[:n]
    grad = 4.0 * n * r / (S * S) - 4.0 * n * M_smooth / (S ** 3)
    return grad, M_smooth, c, r

    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)

        a = a / a.sum()

        sc = _score_norm(a)

        if np.isfinite(sc) and sc < best_score - 1e-13:
            best_score = sc
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.005, lr * 0.92)

        if t_adam % 30 == 0:
            beta = max(1e-7, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.2:
                    beta = max(1e-7, beta * 0.5)

        if no_imp > stagnation_limit:
            ns = 0.06 + 0.04 * min(no_imp / 500.0, 3.0)
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -max_z, max_z)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            c_new = _convolve_fft(a)
            beta = 0.1 * c_new.max()
            lr = base_lr
            no_imp = 0

    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score

def _coord_descent(a, deadline, max_sweeps=5):
    """Coordinate descent with ternary search on each coordinate."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        for j in range(n):
            if time.monotonic() > deadline:
                break
            old = a[j]
            best_local = old
            best_local_sc = best_score
            for factor in [0.3, 0.5, 0.7, 1.3, 1.7, 2.5, 4.0]:
                a[j] = max(old * factor, 1e-15)
                sc = _score_norm(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_local = a[j]
            a[j] = best_local
            if best_local != old:
                a = a / a.sum()
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    return best_a, best_score

def _pairwise_swap(a, deadline, rng, n_pairs=200):
    """Try swapping/scaling pairs of coordinates."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()

    for _ in range(n_pairs):
        if time.monotonic() > deadline:
            break
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        for factor in [0.5, 0.7, 1.3, 1.5, 2.0]:
            save_i, save_j = a[i], a[j]
            a[i] = save_i * factor
            a[j] = save_j / factor
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            a[i], a[j] = save_i, save_j
        a = best_a.copy()

    return best_a, best_score

def _lp_step(a, deadline, max_iters=20):
    """LP-based fixed point: find b maximizing sum(b) s.t. conv(a,b) <= M."""
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    c = _convolve_fft(a)
    M = float(c.max())

    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Build constraint matrix: conv(a, b)[k] <= M for all k
        # conv(a,b)[k] = sum_i a[i]*b[k-i], so row k of A has A[k,j] = a[k-j] if valid
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
        A_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

        c_obj = -np.ones(n)
        b_ub = M * np.ones(2 * n - 1)
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            break
        b = np.maximum(res.x, 0)
        sb = b.sum()
        if sb < 1e-10:
            break
        b = b / sb

        # Line search
        best_t = 0.0
        best_sc = best_score
        for tc in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_t = (1 - tc) * a + tc * b
            a_t = np.maximum(a_t, 1e-15)
            a_t = a_t / a_t.sum()
            sc_t = _score_norm(a_t)
            if sc_t < best_sc:
                best_sc = sc_t
                best_t = tc

        if best_t > 0:
            a = (1 - best_t) * a + best_t * b
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        else:
            break

    return best_a, best_score

        ('exp4', 'exp', 4.0),
        ('cos2', 'cosine', 0.7, 2), ('cos3', 'cosine', 0.6, 3),
