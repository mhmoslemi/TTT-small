def build_conv_matrix(f, n_f, n_b):
    """Build the matrix A such that A @ b = conv(f, b)."""
    # conv(f, b)[k] = sum_j f[k-j] * b[j] for valid indices
    # A[k, j] = f[k-j] if 0 <= k-j < n_f, else 0
    A = np.zeros((n_f + n_b - 1, n_b))

            idx = k - j
            if 0 <= idx < n_f:

def build_conv_matrix(f, n_f, n_b):

LP-based iterative refinement with active-set constraints and L-BFGS

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.signal import correlate
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))
