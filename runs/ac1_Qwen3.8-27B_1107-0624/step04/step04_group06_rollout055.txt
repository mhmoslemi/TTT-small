"""
Optimization using L_p norm annealing with Adam, combined with peak flooding,
LP refinement with dual-guided restarts, and multi-scale exploration.
The key innovation is annealing from a smooth L_p objective (small p) to the
true L_infinity objective, which helps escape local minima by providing
gradient information from all near-peak lags simultaneously.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    c = _full_convolve(p)
    s = float(p.sum())
    if s < 1e-30:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def _score_direct(p):
    """Score assuming p is already normalized to sum=1."""
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    a = a / s
    a = np.clip(a, 1e-15, None)
    return a / a.sum()


def _softmax_from_z(z, floor=1e-14):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _grad_lp_fft(a, c, p):
    """
    Compute gradient of (sum_k c_k^p)^(1/p) with respect to a.
    Uses FFT-based cross-correlation for efficiency.
    g_i = sum_j a_j * c_{j+i}^{p-1}
    """
    n = len(a)
    if p == 1:
        w = np.ones(2 * n - 1)
    else:
        w = c ** (p - 1)
    # g_i = sum_{j=0}^{n-1} a_j * w_{j+i}
    # Using: g = conv(a, w_rev)[n-1:2n-1][::-1]
    w_rev = w[::-1]
    N = _next_pow2(n + (2 * n - 1) - 1)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w_rev, N)
    conv_result = np.fft.irfft(fa * fw, N)
    g = conv_result[n - 1:2 * n - 1][::-1]
    # Ensure non-negative (numerical)
    g = np.maximum(g, 0.0)
    # Scalar factor: norm^(1/p - 1)
    if p == 1:
        norm = float(c.sum())
    else:
        norm = float(np.sum(c ** p))
    if norm < 1e-30:
        return np.zeros(n)
    factor = norm ** (1.0 / p - 1.0)
    return factor * g


def _smooth_max_weights(c, beta):
    """Compute smooth max value and softmax weights."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-10:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _cross_corr_fft(w, a):
    """
    Compute g_i = sum_k w_k * a_{k-i} for i=0..n-1.
    This is the 'responsibility' gradient for the smooth max.
    """
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa_rev, N)
    return h[n - 1:2 * n - 1]


def _adam_smooth_max(p0, deadline, rng, lr=0.04, beta_frac=0.06, beta_decay=0.94,
                     no_imp_limit=250, mode='smooth'):
    """Adam optimization on the smooth max of the convolution."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        if mode == 'smooth':
            _, w = _smooth_max_weights(c, beta)
            g = _cross_corr_fft(w, p)
        else:
            # L_p mode
            g = _grad_lp_fft(p, c, 8)

        # Softmax gradient
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 150 == 0:
                    lr = max(0.001, lr * 0.93)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 40 == 0:
            beta = max(1e-5, beta * beta_decay)

        if no_imp > no_imp_limit:
            # Restart with noise
            ns = 0.08 if best_score < 1.51 else 0.15
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _adam_lp_anneal(p0, deadline, rng, p_values=None, steps_per_p=400,
                    lr=0.05, no_imp_limit=200):
    """
    L_p norm annealing: minimize (sum c_k^p)^(1/p) for increasing p.
    This provides a smoother optimization landscape at low p.
    """
    if p_values is None:
        p_values = [2, 4, 8, 16, 32, 64, 128]

    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    best_p = p0.copy()
    best_z = z.copy()
    best_score = _score_direct(p0)

    b1, b2, eps = 0.9, 0.999, 1e-8

    for p_val in p_values:
        if time.monotonic() >= deadline:
            break

        # Reset optimizer state for new p
        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0
        no_imp = 0
        cur_lr = lr
        z = best_z.copy()
        p = best_p.copy()
        c = _full_convolve(p)

        print(f"  L_p anneal: p={p_val}, score={best_score:.10f}")

        for step in range(steps_per_p):
            if time.monotonic() >= deadline:
                break

            # Compute L_p gradient
            g = _grad_lp_fft(p, c, p_val)

            # Softmax gradient
            dot = float(np.dot(p, g))
            gz = p * (g - dot)
            gz -= float(np.mean(gz))

            if not np.all(np.isfinite(gz)):
                z = best_z.copy()
                m.fill(0.0)
                v.fill(0.0)
                t_adam = 0
                no_imp = 0
                continue

            t_adam += 1
            m = b1 * m + (1.0 - b1) * gz
            v = b2 * v + (1.0 - b2) * (gz * gz)
            mh = m / (1.0 - b1 ** t_adam)
            vh = v / (1.0 - b2 ** t_adam)
            z = z - cur_lr * mh / (np.sqrt(vh) + eps)

            if not np.all(np.isfinite(z)):
                z = best_z.copy()
                m.fill(0.0)
                v.fill(0.0)
                t_adam = 0
                continue

            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            score = _score_direct(p)

            if np.isfinite(score):
                if score < best_score - 1e-14:
                    best_score = score
                    best_p = p.copy()
                    best_z = z.copy()
                    no_imp = 0
                else:
                    no_imp += 1
                    if no_imp % 100 == 0:
                        cur_lr = max(0.001, cur_lr * 0.9)
            else:
                z = best_z.copy()
                m.fill(0.0)
                v.fill(0.0)
                t_adam = 0
                continue

            if no_imp > no_imp_limit:
                # Restart
                ns = 0.1
                z = best_z + ns * rng.normal(size=n)
                z = np.clip(z, -60.0, 60.0)
                z -= float(np.mean(z))
                p = _softmax_from_z(z)
                c = _full_convolve(p)
                m.fill(0.0)
                v.fill(0.0)
                t_adam = 0
                cur_lr = lr
                no_imp = 0

    return best_p, best_score


def _peak_flood(a, deadline, eta=0.04, n_iter_max=5000):
    """Multiplicative peak-flooding updates."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score_direct(a)
    c = _full_convolve(a)

    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break

        M = float(c.max())
        score = float(2.0 * n * M)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        threshold = M * 0.92
        hot_mask = c >= threshold
        n_hot = int(hot_mask.sum())
        if n_hot == 0:
            hot_mask[int(np.argmax(c))] = True

        hot_indices = np.where(hot_mask)[0]
        heat = np.zeros(n)
        for k in hot_indices:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                partner = k - np.arange(i_start, i_end + 1)
                heat[i_start:i_end + 1] += a[partner]

        heat_max = float(heat.max())
        if heat_max < 1e-15:
            break
        heat = heat / heat_max

        factor = 1.0 - eta * heat
        factor = np.maximum(factor, 0.01)
        a = a * factor
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        c = _full_convolve(a)

        if it % 200 == 0 and it > 0:
            if score >= best_score:
                eta = max(0.003, eta * 0.85)
            else:
                eta = min(0.2, eta * 1.08)

    return best_a, best_score


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a, deadline, t_start=0.35, max_solves=30, m_shrink=0.998):
    """LP refinement with decreasing M schedule."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_direct(a)
    c = _full_convolve(a)
    M = float(c.max())

    for _ in range(max_solves):
        if time.monotonic() > deadline:
            break

        b = _lp_solve(a, M)
        if b is None:
            b = _lp_solve(a, M * 1.005)
            if b is None:
                break

        s_b = float(b.sum())
        if s_b < 1e-10:
            break

        b_norm = b / s_b
        a_new = (1.0 - t_start) * a + t_start * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_direct(a_new)

        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        c = _full_convolve(a)
        M_new = float(c.max())
        M = max(M_new * m_shrink, best_score / (2.0 * n) * 1.002)
        t_start = max(0.1, t_start * 0.9)

    return best_a, best_score


def _dual_guided_update(a, deadline, rng, n_iters=100):
    """Use LP dual variables to guide point updates."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_direct(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a)
        M = float(c.max())
        score = 2.0 * n * M

        # Find peak lags
        peak_threshold = M * 0.9
        peak_lags = np.where(c >= peak_threshold)[0]

        if len(peak_lags) == 0:
            peak_lags = np.array([int(np.argmax(c))])

        # For each peak lag, identify the most contributing pair
        for k in peak_lags[:3]:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k - idx]
            top_i = idx[int(np.argmax(contrib))]

            # Find a "cold" position to move mass to
            # Compute per-position heat across all peaks
            heat = np.zeros(n)
            for kk in peak_lags:
                ks = max(0, kk - n + 1)
                ke = min(n - 1, kk)
                ii = np.arange(ks, ke + 1)
                heat[ii] += a[kk - ii]

            cold_mask = heat < float(np.median(heat))
            cold_positions = np.where(cold_mask)[0]
            if len(cold_positions) == 0:
                cold_positions = np.arange(n)

            # Move small amount of mass from hot to cold
            target = cold_positions[rng.integers(0, len(cold_positions))]
            amount = a[top_i] * 0.02
            a[top_i] -= amount
            a[target] += amount
            a = np.maximum(a, 1e-15)
            a = a / a.sum()

        c = _full_convolve(a)
        sc = _score_direct(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _make_profile(n, type='uniform', **kwargs):
    if type == 'uniform':
        return np.ones(n) / n
    elif type == 'linear':
        t = np.arange(n) / (n - 1)
        eta = kwargs.get('eta', 2.0)
        a = 1.0 + eta * (t - 0.5)
        return np.maximum(a, 1e-6) / np.maximum(a, 1e-6).sum()
    elif type == 'cosine':
        t = np.arange(n)
        freq = kwargs.get('freq', 1)
        amp = kwargs.get('amp', 1.0)
        a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
        return np.maximum(a, 1e-6) / np.maximum(a, 1e-6).sum()
    elif type == 'bathtub':
        t = np.arange(n) / (n - 1)
        a = 1.0 + kwargs.get('depth', 1.0) * (2 * t - 1) ** 2
        return a / a.sum()
    elif type == 'tophat':
        # High in middle, low at edges
        t = np.arange(n) / (n - 1)
        a = 1.0 - kwargs.get('depth', 0.5) * (2 * t - 1) ** 2
        return np.maximum(a, 1e-6) / np.maximum(a, 1e-6).sum()
    return np.ones(n) / n


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        p = p / s
        sc = _score_direct(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} n={p.size} [{label}] t={time.monotonic()-t_start:.1f}s")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent: n={pn}, budget={budget_s}s")

    if parent is not None:
        sc_p = _score_direct(parent)
        print(f"Parent score: {sc_p:.10f}")
        best_seq = parent.copy()
        best_score = sc_p

    # === Phase 1: L_p annealing on parent (500s) ===
    phase1_end = t_start + 500.0
    print(f"\nPhase 1: L_p annealing...")
    if parent is not None:
        p1, sc1 = _adam_lp_anneal(parent, phase1_end, rng,
                                   p_values=[2, 4, 8, 16, 32, 64],
                                   steps_per_p=500, lr=0.06, no_imp_limit=200)
        _update(p1, 'lp_anneal')

    # === Phase 2: Smooth max Adam polish (300s) ===
    phase2_end = t_start + 800.0
    print(f"\nPhase 2: Smooth max Adam...")
    if best_seq is not None:
        p2, sc2 = _adam_smooth_max(best_seq, phase2_end, rng,
                                    lr=0.04, beta_frac=0.05, beta_decay=0.95,
                                    no_imp_limit=250, mode='smooth')
        _update(p2, 'adam_smooth')

    # === Phase 3: Peak flooding (200s) ===
    phase3_end = t_start + 1000.0
    print(f"\nPhase 3: Peak flooding...")
    if best_seq is not None:
        p3, sc3 = _peak_flood(best_seq, phase3_end, eta=0.03)
        _update(p3, 'flood')

    # === Phase 4: LP refinement (200s) ===
    phase4_end = t_start + 1200.0
    print(f"\nPhase 4: LP refinement...")
    if best_seq is not None and best_seq.size <= 2048:
        p4, sc4 = _lp_refine(best_seq, phase4_end, t_start=0.3, max_solves=20, m_shrink=0.999)
        _update(p4, 'lp_refine')

    # === Phase 5: Dual-guided + Adam (200s) ===
    phase5_end = t_start + 1400.0
    print(f"\nPhase 5: Dual-guided update...")
    if best_seq is not None:
        p5, sc5 = _dual_guided_update(best_seq, min(phase5_end, time.monotonic() + 80), rng, n_iters=200)
        _update(p5, 'dual_guide')

        if time.monotonic() < phase5_end - 20:
            p5a, sc5a = _adam_smooth_max(best_seq, phase5_end, rng,
                                          lr=0.02, beta_frac=0.03, beta_decay=0.97,
                                          no_imp_limit=150, mode='smooth')
            _update(p5a, 'adam_smooth2')

    # === Phase 6: Multi-scale exploration (200s) ===
    phase6_end = t_start + 1600.0
    print(f"\nPhase 6: Multi-scale...")
    if time.monotonic() < phase6_end:
        # Try different n values
        n_options = [512, 1024, 2048]
        if best_seq is not None:
            n_options = [n for n in n_options if n != best_seq.size]

        for n_try in n_options[:2]:
            if time.monotonic() > phase6_end - 30:
                break

            if best_seq is not None:
                p_scale = _interp_profile(best_seq, n_try)
            else:
                p_scale = _make_profile(n_try, 'uniform')

            sc_scale = _score_direct(p_scale)
            print(f"  n={n_try} interp score: {sc_scale:.10f}")
            _update(p_scale, f'interp_n{n_try}')

            # Quick L_p anneal at new scale
            per = min(60.0, phase6_end - time.monotonic() - 10)
            if per > 20:
                p_ns, sc_ns = _adam_lp_anneal(p_scale, time.monotonic() + per, rng,
                                               p_values=[4, 8, 16],
                                               steps_per_p=200, lr=0.05, no_imp_limit=100)
                _update(p_ns, f'lp_n{n_try}')

    # === Phase 7: Alternative starting points (100s) ===
    phase7_end = deadline - 50.0
    print(f"\nPhase 7: Alt starts...")
    if time.monotonic() < phase7_end:
        n_alt = 1024
        alts = [
            ('bathtub1', _make_profile(n_alt, 'bathtub', depth=1.5)),
            ('bathtub2', _make_profile(n_alt, 'bathtub', depth=2.5)),
            ('tophat1', _make_profile(n_alt, 'tophat', depth=0.8)),
            ('cos2', _make_profile(n_alt, 'cosine', freq=2, amp=0.6)),
            ('lin3', _make_profile(n_alt, 'linear', eta=3.0)),
        ]
        per_alt = max(10.0, (phase7_end - time.monotonic()) / len(alts))
        for label, pa in alts:
            if time.monotonic() > phase7_end - 5:
                break
            per = min(per_alt, phase7_end - time.monotonic() - 5)
            if per < 8:
                break
            p_alt, sc_alt = _adam_lp_anneal(pa, time.monotonic() + per, rng,
                                             p_values=[4, 16],
                                             steps_per_p=150, lr=0.05, no_imp_limit=80)
            _update(p_alt, f'alt_{label}')

    # === Phase 8: Final intensive polish (remaining time) ===
    print(f"\nPhase 8: Final polish...")
    if time.monotonic() < deadline and best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            # Alternate between different methods
            chunks = max(2, int(remaining // 40))
            chunks = min(chunks, 5)
            for ci in range(chunks):
                if time.monotonic() > deadline - 5:
                    break
                per = min(40.0, deadline - time.monotonic() - 5)
                if per < 10:
                    break

                if ci % 3 == 0:
                    p_f, sc_f = _adam_smooth_max(best_seq, time.monotonic() + per, rng,
                                                  lr=0.015, beta_frac=0.02, beta_decay=0.98,
                                                  no_imp_limit=100, mode='smooth')
                    _update(p_f, f'final_adam_{ci}')
                elif ci % 3 == 1:
                    p_f, sc_f = _peak_flood(best_seq, time.monotonic() + per, eta=0.02)
                    _update(p_f, f'final_flood_{ci}')
                else:
                    if best_seq.size <= 2048:
                        p_f, sc_f = _lp_refine(best_seq, time.monotonic() + per,
                                                t_start=0.2, max_solves=8, m_shrink=0.9995)
                        _update(p_f, f'final_lp_{ci}')

    # === Final validation ===
    print(f"\n{'='*60}")
    print(f"Final best score: {best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        exact_sc = _score_direct(final)
        print(f"Exact score: {exact_sc:.10f}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluate_sequence score: {eval_sc:.10f}")
        except Exception as e:
            print(f"Evaluation error: {e}")

        return [float(x) for x in final]

    # Fallback
    print("FALLBACK: uniform n=1024")
    fb = np.ones(1024) / 1024
    return [float(x) for x in fb]
