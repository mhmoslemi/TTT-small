    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _cross_correlate(w, a):
    """Compute sum_k w[k] * a[k-j] for all j, using FFT."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    # We need g[j] = sum_k w[k] * a[k-j] for 0<=j<n
    # This is the cross-correlation of w with a, offset appropriately.
    # g[j] = sum_k w[k] * a[k-j] = (w * a_reversed)[n-1+j] (approximately)
    # Let's use: g = ifft(fft(w) * conj(fft(a))) gives correlation
    # Actually, let me think more carefully.
    # sum_k w[k] * a[k-j] where a is defined for 0<=k-j<n
    # Let b[m] = a[m], extended with zeros.
    # g[j] = sum_k w[k] * b[k-j]
    # This is (w * b_reversed)[n-1-j] + ... hmm.
    # Alternatively: g[j] = sum_m w[m+j] * a[m] (substituting m = k-j)
    # = cross-correlation of w and a at lag j.
    # np.correlate(w, a, mode='full')[j+n-1] gives this.
    # But for efficiency, use FFT.
    # g[j] = sum_m w[m+j] * a[m]
    # = ifft(fft(w, N) * conj(fft(a, N)))[j] (with proper indexing)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    # The circular cross-correlation: ifft(fw * conj(fa))[j] = sum_m w[m] * a[(m-j) mod N]
    # We need the linear version.
    corr = np.fft.irfft(fw * np.conj(fa), N)
    # corr[j] = sum_m w[m] * a[(m-j) mod N] (circular)
    # For linear: corr[j] for j=0,...,n-1 gives sum_m w[m] * a[m-j] (with a=0 outside [0,n-1])
    # But we want g[j] = sum_m w[m+j] * a[m] = sum_k w[k] * a[k-j] (with k=m+j)
    # = corr... let me think again.
    # ifft(fft(w) * conj(fft(a)))[d] = sum_m w[m] * a[(m-d) mod N]
    # For the linear case (N large enough): = sum_m w[m] * a[m-d]
    # We want: g[j] = sum_k w[k] * a[k-j]
    # Let d = -j: sum_m w[m] * a[m+j] ... no.
    # Let me be explicit: g[j] = sum_k w[k] * a[k-j]
    # = sum_m w[m+j] * a[m] (sub m = k-j)
    # The cross-correlation (w cross a)[j] = sum_m w[m+j] * a[m]
    # = ifft(fft(w) * conj(fft(a)))[-j mod N] ... 
    
    # Actually, the standard result is:
    # np.correlate(w, a, 'full')[n-1+j] = sum_m w[m+j] * a[m] for the linear case
    # In FFT: ifft(fft(w_padded) * conj(fft(a_padded))) gives circular cross-correlation
    # where the result at index j is sum_m w[m] * conj(a)[(m-j) mod N]
    # For real: sum_m w[m] * a[(m-j) mod N]
    # We want sum_m w[m+j] * a[m] = sum_k w[k] * a[k-j] (with k=m+j)
    # This is the cross-correlation at lag -j (or equivalently, the result at index N-j)
    
    # Let me just use a direct approach with FFT:
    # g = sum_j w[k] * a[k-j] for each j
    # This is (w * a_flip)[n-1-j] where a_flip = a[::-1]
    # (w * a_flip)[m] = sum_k w[k] * a_flip[m-k] = sum_k w[k] * a[n-1-(m-k)]
    # For m = n-1-j: sum_k w[k] * a[n-1-n+1+j-k] = sum_k w[k] * a[j-k]
    # That's not quite right either.
    
    # Let me just do it the simple way:
    # g[j] = sum_k w[k] * a[k-j] where 0<=k<2n-1 and 0<=k-j<n
    # = sum_{k=j}^{j+n-1} w[k] * a[k-j] (k ranges from j to j+n-1, and k-j ranges from 0 to n-1)
    # This is a convolution of w with a, evaluated at specific points.
    # (w * a)[m] = sum_k w[k] * a[m-k]
    # g[j] = sum_k w[k] * a[k-j] = sum_m w[j+m] * a[m] (sub m=k-j)
    # = (a * w)[j] (cross-correlation of a and w at lag j)
    # = ifft(fft(a) * conj(fft(w)))[j] (circular, with zero padding for linear)
    
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    # cross-correlation: ifft(fa * conj(fw))[j] = sum_m a[m] * w[m-j] (circular)
    # = sum_m a[m] * w[(m-j) mod N]
    # We want: sum_m a[m] * w[m+j] = cross-correlation at lag -j
    # = ifft(fa * conj(fw))[N-j] (for j > 0)
    # Hmm, this is getting confusing. Let me just use a different formulation.
    
    # g[j] = sum_k w[k] * a[k-j]
    # Define w_ext of length N (zero-padded), a_ext of length N (zero-panned).
    # Then g[j] = sum_k w_ext[k] * a_ext[k-j] (linear, since N is large enough)
    # This is the cross-correlation of w_ext and a_ext at lag j.
    # In FFT: ifft(fft(w_ext) * conj(fft(a_ext)))[j] gives circular cross-correlation
    # = sum_m w_ext[m] * a_ext[(m-j) mod N]
    # For linear (N >= len(w) + len(a) - 1): = sum_m w_ext[m] * a_ext[m-j] (with 0 outside range)
    # = sum_m w[m] * a[m-j]
    # We want sum_k w[k] * a[k-j] = sum_m w[m] * a[m-j] ✓ (just rename k to m)
    
    # So: g[j] = ifft(fft(w) * conj(fft(a)))[j] for j = 0, ..., n-1
    result = np.fft.irfft(fw * np.conj(fa), N)
    return result[:n]

def _score(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())

def _smooth_max_grad(c, beta):
    """Return (smooth_max_value, weights) for soft-max of c with temperature beta."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w

def _gradient_on_simplex(a, beta):
    """Compute gradient of soft-max score on the simplex (sum a = 1, a >= 0)."""
    n = len(a)
    c = _full_convolve(a)
    sm_val, w = _smooth_max_grad(c, beta)
    # g[j] = sum_k w[k] * a[k-j]
    g = _cross_correlate(w, a)
    # grad[j] = 4n * (g[j] - sm_val * a[j])  [for S=1]
    grad = 4.0 * n * (g - sm_val * a)
    return grad, sm_val

def _project_simplex(a):
    """Project a onto the simplex {x >= 0, sum x = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def _line_search(a, grad, max_lr=1.0, min_lr=1e-6, n_trials=10):
    """Find step size by backtracking line search."""
    base_score = _score(a)
    for trial in range(n_trials):
        lr = max_lr * (0.5 ** trial)
        a_new = a - lr * grad
        a_new = _project_simplex(a_new)
        sc = _score(a_new)
        if sc < base_score:
            return a_new, lr
    return a, max_lr * (0.5 ** n_trials)

def _gradient_descent(a, deadline, rng, beta_init=0.05, beta_min=0.001, 
                      beta_decay=0.995, lr_init=0.1, no_imp_limit=100):
    """FFT-based gradient descent with simplex projection and line search."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    
    beta = beta_init
    lr = lr_init
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        grad, sm_val = _gradient_on_simplex(a, beta)
        
        # Center the gradient (remove mean to maintain sum)
        grad_centered = grad - grad.mean()
        
        # Line search
        a_new, lr_found = _line_search(a, grad_centered, max_lr=lr, n_trials=8)
        
        if a_new is not a:
            sc = _score(a_new)
            if sc < best_score - 1e-12:
                a = a_new
                best_score = sc
                best_a = a.copy()
                no_imp = 0
                lr = min(1.0, lr_found * 1.1)
            else:
                lr = max(lr_init * 0.1, lr * 0.7)
                no_imp += 1
        else:
            no_imp += 1
            lr = max(lr_init * 0.1, lr * 0.5)
        
        # Anneal beta
        if it % 20 == 0:
            beta = max(beta_min, beta * beta_decay)
        
        # Restart if stuck
        if no_imp > no_imp_limit:
            # Add noise
            noise_scale = 0.1
            a = best_a + noise_scale * rng.normal(size=n)
            a = _project_simplex(a)
            beta = max(beta_min * 10, beta * 2)
            lr = lr_init
            no_imp = 0
            # Re-evaluate
            sc = _score(a)
            if sc > best_score * 1.05:
                a = best_a.copy()
    
    return best_a, best_score

def _lp_fixed_point(a, deadline, max_iters=30):
    """LP-based fixed point iteration."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a)
        M = float(c.max())
        
        # Build the LP: max sum(b) s.t. conv(a,b)_k <= M, b >= 0
        # conv(a,b)_k = sum_j a[j] * b[k-j] (valid range)
        # For each k: sum_j a[j] * b[k-j] <= M
        # This is: A @ b <= M * ones
        # where A[k, j] = a[k-j] if 0<=k-j<n, else 0
        
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                rows.extend([k] * len(idx))
                cols.extend(idx.tolist())
                vals.extend(a[k - idx].tolist())
        
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b_ub = M * np.ones(2 * n - 1)
        bounds = [(0, None)] * n
        
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            break
        
        b = np.maximum(res.x, 0.0)
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b = b / s_b
        
        # Line search for blending
        best_t = 1.0
        best_sc = _score(a)
        for t_candidate in [0.3, 0.5, 0.7, 1.0]:
            a_t = (1.0 - t_candidate) * a + t_candidate * b
            a_t = np.maximum(a_t, 1e-15)
            a_t = a_t / a_t.sum()
            sc_t = _score(a_t)
            if sc_t < best_sc:
                best_sc = sc_t
                best_t = t_candidate
        
        a = (1.0 - best_t) * a + best_t * b
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score

def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()

def _make_profile(n, ptype, p1=1.0, p2=0.0, p3=0.0):
    if ptype == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (t - 0.5)
    elif ptype == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (2*t - 1)**2
    elif ptype == 'cosine':
        t = np.arange(n)
        a = 1.0 + p1 * np.cos(2*np.pi*p2*t/n)
    elif ptype == 'sine':
        t = np.arange(n)
        a = 1.0 + p1 * np.sin(2*np.pi*p2*t/n)
    elif ptype == 'gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5*((t-0.5)/p1)**2)
    elif ptype == 'U_shape':
        t = np.linspace(0, 1, n)
        a = (2*t - 1)**2 + p1
    elif ptype == 'edge_heavy':
        t = np.arange(n) / max(1, n-1)
        a = 1.0 / (0.01 + abs(t - 0.5)*2.0)**p1
    elif ptype == 'two_gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5*((t-p1)/p3)**2) + np.exp(-0.5*((t-p2)/p3)**2)
    elif ptype == 'ramp_sq':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * t**2
    elif ptype == 'ramp_sqrt':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * np.sqrt(t)
    elif ptype == 'modulated_cos':
        t = np.linspace(0, 1, n)
        env = 1.0 - p2 * np.abs(2*t - 1)
        a = env * (1.0 + p1 * np.cos(2*np.pi*p3*t))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()
