"""
Step-function auto-convolution peak minimizer.

Score = 2*n*max(conv(a,a)) / (sum a)^2  (scale-invariant, so we work at sum=1).
Strategy: start from the provided parent, then run aggressive LP best-response
cycles (maximize sum(b) s.t. a*b <= M, b>=0) with adaptive/over-relaxed mixing,
peak-reduction attempts (solving at a smaller M), constraint subsetting for
diverse directions, multi-scale escalation, gradient (subgrad + mirror)
polishing, and restarts, all strictly time-guarded to finish before the deadline.
All helpers are top-level with no closures and no lambdas.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c


def _score(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_solve_with_matrix(A, M):
    n = A.shape[1]
    c_obj = -np.ones(n)
    b_ub = M * np.ones(A.shape[0])
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        x = np.asarray(res.x, dtype=np.float64)
        return np.maximum(x, 0.0)
    return None


def _lp_solve_full(a, M):
    A = _build_conv_matrix(a)
    return _lp_solve_with_matrix(A, M)


def _lp_solve_subset(a, M, row_sel):
    A_full = _build_conv_matrix(a)
    A = A_full[row_sel, :]
    return _lp_solve_with_matrix(A, M)


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    mask = u * rho_vals > (cssv - 1)
    if not mask.any():
        return np.ones(n) / n
    rho = np.nonzero(mask)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 0.0))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s


def _cross_corr_signals(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _gradient_step(p, lr, beta, use_mirror):
    n = len(p)
    c = _full_convolve(p)
    w = _smooth_max_weights(c, beta)
    g = _cross_corr_signals(w, p)
    g = g - float(np.mean(g))
    if not np.all(np.isfinite(g)):
        return p
    if use_mirror:
        log_p = np.log(np.maximum(p, 1e-300))
        log_p = log_p - lr * g
        log_p = log_p - float(np.mean(log_p))
        p_new = np.exp(np.clip(log_p, -50.0, 50.0))
        s = p_new.sum()
        if s < 1e-30:
            return np.ones(n) / n
        return p_new / s
    p_new = p - lr * g
    return _project_simplex(p_new)


def _interp_to_length(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    if s > 1e-30:
        return p1 / s
    return np.ones(n_new) / n_new


def _mix(p, b, t):
    n = p.size
    q = (1.0 - t) * p + t * b
    q = np.maximum(q, 0.0)
    s = q.sum()
    if s < 1e-30:
        return p
    return q / s
