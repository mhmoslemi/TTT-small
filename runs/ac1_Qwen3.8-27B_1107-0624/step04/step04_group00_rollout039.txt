The geometric mean is `GM_K = (prod_{j=1}^K c_{(j)})^{1/K} = exp((1/K) * sum_{j=1}^K log(c_{(j)}))`.

The issue is that the "top-K" set changes as we optimize, making the objective non-smooth. To handle this, we can use a soft version:

This is efficient and handles the non-negativity constraint exactly.

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_lse(a, c, beta):
    """Gradient of soft-max objective w.r.t. a (for sum(a)=1 normalization)."""
    n = len(a)
    cm = float(c.max())
    w = np.exp((c - cm) / beta)
    s = w.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _obj_lse(a, beta):
    """Smoothed max objective (for the subproblem with sum(a)=1)."""
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    logsum = beta * np.log(np.sum(np.exp((c - cm) / beta)))
    return 2.0 * n * (cm + logsum)


def _grad_obj_lse(z, beta, n):
    """Gradient for L-BFGS in softmax parameterization."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    w = np.exp((c - cm) / beta)
    s = w.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 2.0 * n * a * (h - W)

def _fun_obj_lse(z, beta, n):
    """Objective for L-BFGS in softmax parameterization."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    logsum = beta * np.log(np.sum(np.exp((c - cm) / beta)))
    return 2.0 * n * (cm + logsum)


def _run_subgrad(a_init, deadline, rng, n_iter_max=20000):
    """Subgradient method with multiple descent directions and adaptive step size."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    star = best_score
    prev_best = best_score

    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break
        if it % 200 == 0 and best_score == prev_best:
            star *= 0.95

        c = _full_convolve(a, a)
        c_max = float(c.max())
        score_cur = 2.0 * n * c_max

        if score_cur < best_score - 1e-12:
            best_score = score_cur
            best_a = a.copy()
            star = best_score

        k_star = int(np.argmax(c))

        # Direction 1: worst peak gradient (centered)
        gc = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            gc[i] = a[k_star - i]
        gc = gc - gc.mean()

        # Direction 2: centroid of top peaks
        top_k_vals = np.sort(c)[::-1][:min(5, len(c))]
        threshold = top_k_vals[-1] * 0.99
        active = np.where(c >= threshold)[0]
        g_cent = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                g_cent[i] += a[k - i] / len(active)
        g_cent = g_cent - g_cent.mean()

        # Direction 3: Frank-Wolfe (negative min of worst peak gradient)
        g_worst = np.zeros(n)
        for i in range(lo, hi + 1):
            g_worst[i] = a[k_star - i]
        i_min = int(np.argmin(g_worst))
        e_i = np.zeros(n)
        e_i[i_min] = 1.0

        # Try each direction
        base_eta = 0.02 / (1.0 + 0.01 * it)
        directions = [gc, g_cent, e_i - e_i.mean()]
        nls = [3, 3, 2]

        improved = False
        for d, n_ls in zip(directions, nls):
            if np.all(d == 0):
                continue
            d_norm = np.linalg.norm(d)
            if d_norm < 1e-15:
                continue
            d = d / d_norm
            found = False
            for j in range(n_ls):
                eta = base_eta / (j + 1)
                a_trial = _simplex_project(a - eta * d)
                sc = _score_norm_fn(a_trial)
                if sc < score_cur - 1e-12:
                    a = a_trial
                    score_cur = sc
                    improved = True
                    found = True
                    break
            if found:
                break

        if not improved and it > 100:
            # Add noise to escape
            noise = 0.01 / (1.0 + 0.005 * it)
            z = np.log(np.maximum(a, 1e-15))
            z = z + noise * rng.standard_normal(n)
            z = z - z.mean()
            a = _softmax_fn(z)

    return best_a, best_score


def _run_lbfgs_continuation(a_init, deadline, beta_start, beta_end, n_levels, maxiter_per_level=2000):
    """L-BFGS with continuation in beta (softmax parameterization)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    betas = np.geomspace(beta_start, beta_end, n_levels)

    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = n_levels - idx
        time_left = deadline - time.monotonic()
        time_budget = time_left / remaining

        try:
            res = minimize(
                _fun_obj_lse, z, args=(beta, n),
                method='L-BFGS-B', jac=_grad_obj_lse,
                options={'maxiter': maxiter_per_level, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxls': 50, 'maxfun': 10000}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_new = _softmax_fn(res.x)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                z = res.x.copy()
        except Exception:
            pass

        # After each beta level, also try a few subgradient steps
        if time.monotonic() < deadline - 2:
            sg_deadline = min(deadline - 1, time.monotonic() + time_budget * 0.3)
            a_sg, sc_sg = _run_subgrad(best_a, sg_deadline, np.random.default_rng(42 + idx), n_iter_max=2000)
            if sc_sg < best_score:
                best_score = sc_sg
                best_a = a_sg.copy()
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()

    return best_a, best_score


def _peak_equalization(a_init, deadline, top_k=8, max_rounds=50):
    """Equalize the top-K convolution peaks using cyclic coordinate descent."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Find top-K positions
        top_positions = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_positions]

        # If already well-equalized, try finer adjustment
        if top_vals[0] / max(top_vals[-1], 1e-15) < 1.001:
            # Peaks are equalized; try to reduce overall level
            k_star = top_positions[0]
        else:
            # Reduce the highest peak
            k_star = top_positions[0]

        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            contrib[i] = a[i] * a[k_star - i]

        total = float(contrib[lo:hi+1].sum())
        if total < 1e-15:
            break

        target_frac = 0.97  # try to reduce this peak by 3%
        c_target = c_max * target_frac

        # Binary search on uniform reduction factor
        lo_f, hi_f = 0.0, 1.0
        for _ in range(30):
            mid_f = (lo_f + hi_f) / 2.0
            a_trial = a.copy()
            a_trial[lo:hi+1] *= mid_f
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                lo_f = mid_f
                continue
            a_trial /= s_trial
            c_trial = _full_convolve(a_trial, a_trial)
            # Check if target peak is reduced enough
            idx_trial = k_star
            if c_trial[idx_trial] > c_target:
                lo_f = mid_f
            else:
                hi_f = mid_f

        a_new = a.copy()
        a_new[lo:hi+1] *= hi_f
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new /= s_new
            sc_new = _score_norm_fn(a_new)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new.copy()

    return best_a, best_score

def _make_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []

    for slope in [0.5, 1.0, 1.5, 2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    for p in [0.5, 1.0, 1.5, 2.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())

    for rate in [1.0, 2.0, -1.0, -2.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())

    # Asymmetric: more mass at one end
    for skew in [0.3, 0.5, 0.7]:
        a = 1.0 + skew * u + 0.1 * u ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n_parent = len(parent) if parent is not None else 2048
    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_parent}, norm_score={sc_parent:.10f}, actual={_score_fn(parent):.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        parent = _make_profiles(2048)[3]
        update_best(parent, _score_norm_fn(parent), "default")

    # === Phase 1: L-BFGS continuation on parent length (600s) ===
    phase1_end = min(deadline, t_start + 600.0)
    print(f"Phase 1: L-BFGS continuation n={n_parent}")

    if time.monotonic() < phase1_end and best_seq is not None:
        c0 = _full_convolve(best_seq, best_seq)
        max_c0 = float(c0.max())
        beta_start = 0.1 * max_c0
        beta_end = 1e-6 * max_c0
        n_levels = 25  # Fine continuation

        p1, sc1 = _run_lbfgs_continuation(best_seq, phase1_end, beta_start, beta_end, n_levels)
        update_best(p1, sc1, "lbfgs_continuation")

    print(f"  Phase 1 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # === Phase 2: Subgradient polish (200s) ===
    phase2_end = min(deadline, t_start + 800.0)
    print(f"Phase 2: Subgradient polish")

    if time.monotonic() < phase2_end and best_seq is not None:
        p2, sc2 = _run_subgrad(best_seq, phase2_end, rng, n_iter_max=30000)
        update_best(p2, sc2, "subgrad")

    print(f"  Phase 2 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # === Phase 3: Peak equalization (100s) ===
    phase3_end = min(deadline, t_start + 900.0)
    print(f"Phase 3: Peak equalization")

    if time.monotonic() < phase3_end and best_seq is not None:
        p3, sc3 = _peak_equalization(best_seq, phase3_end, top_k=10, max_rounds=100)
        update_best(p3, sc3, "peak_eq")

    print(f"  Phase 3 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # === Phase 4: Try larger n=8192 (300s) ===
    phase4_end = min(deadline, t_start + 1200.0)
    print(f"Phase 4: Try n=8192")

    if time.monotonic() < phase4_end and best_seq is not None:
        n_curr = len(best_seq)
        if n_curr < 8192:
            p_interp = _interp_profile(best_seq, 8192)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, "interp_8192")

            if time.monotonic() < phase4_end - 60:
                c_int = _full_convolve(p_interp, p_interp)
                beta_s = 0.05 * float(c_int.max())
                beta_e = 1e-5 * float(c_int.max())
                p4, sc4 = _run_lbfgs_continuation(p_interp, phase4_end, beta_s, beta_e, 15, maxiter_per_level=1000)
                update_best(p4, sc4, "lbfgs_8192")

    print(f"  Phase 4 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # === Phase 5: Singleton elimination (100s) ===
    phase5_end = min(deadline, t_start + 1300.0)
    print(f"Phase 5: Singleton elimination")

    if time.monotonic() < phase5_end and best_seq is not None:
        p5, sc5 = _singleton_elimination(best_seq, phase5_end, n_remove=32)
        update_best(p5, sc5, "singleton")

    print(f"  Phase 5 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # === Phase 6: Final combined polish (remaining time) ===
    phase6_end = deadline
    print(f"Phase 6: Final polish")

    if time.monotonic() < phase6_end and best_seq is not None:
        # Another round of subgradient with very small steps
        p6, sc6 = _run_subgrad(best_seq, min(phase6_end - 20, time.monotonic() + 150), rng, n_iter_max=50000)
        update_best(p6, sc6, "final_subgrad")

        # Peak equalization again
        if time.monotonic() < phase6_end - 10:
            p7, sc7 = _peak_equalization(best_seq, min(phase6_end - 5, time.monotonic() + 80), top_k=15, max_rounds=80)
            update_best(p7, sc7, "final_peak_eq")

        # One more L-BFGS at very small beta
        if time.monotonic() < phase6_end - 10:
            n_f = len(best_seq)
            c_f = _full_convolve(best_seq, best_seq)
            beta_very_small = 1e-7 * float(c_f.max())
            z_f = np.log(np.maximum(best_seq, 1e-15))
            z_f = z_f - z_f.mean()
            try:
                res = minimize(
                    _fun_obj_lse, z_f, args=(beta_very_small, n_f),
                    method='L-BFGS-B', jac=_grad_obj_lse,
                    options={'maxiter': 5000, 'ftol': 1e-20, 'gtol': 1e-14,
                             'maxls': 50, 'maxfun': 20000}
                )
                if res.x is not None and np.all(np.isfinite(res.x)):
                    a_f = _softmax_fn(res.x)
                    sc_f = _score_norm_fn(a_f)
                    update_best(a_f, sc_f, "final_lbfgs_tiny")
            except Exception:
                pass

    print(f"  Phase 6 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profiles(256)[2]

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 5000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")
        else:
            print(f"Sequence too long for evaluator check (n={len(final_seq)}), using internal score.")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profiles(256)[2]
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
