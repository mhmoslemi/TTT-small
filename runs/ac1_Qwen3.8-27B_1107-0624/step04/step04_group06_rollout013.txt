def _lp_solve(a, M):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)_k <= M for all k."""
    n = len(a)
    Nc = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(Nc):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            j_idx = np.arange(j_start, j_end + 1)
            # (a*b)_k = sum_j a[k-j] * b[j]
            # So coefficient of b[j] is a[k-j]
            a_vals = a[k - j_idx]
            rows.extend([k] * len(j_idx))
            cols.extend(j_idx.tolist())
            vals.extend(a_vals.tolist())
    
    A = csr_matrix((vals, (rows, cols)), shape=(Nc, n))
    c_obj = -np.ones(n)
    b_ub = M * np.ones(Nc)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

Optimized step function coefficient search using aggressive LP alternation,

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _full_convolve(a)
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-30:
        return None
    return a / s


def _build_conv_matrix(a):
    n = len(a)
    Nc = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(Nc):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            j_idx = np.arange(j_start, j_end + 1)
            a_vals = a[k - j_idx]
            rows.extend([k] * len(j_idx))
            cols.extend(j_idx.tolist())
            vals.extend(a_vals.tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(Nc, n))
    return A


def _lp_solve(a, M):
    n = len(a)
    Nc = 2 * n - 1
    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(Nc)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0), res
    return None, res


def _lp_solve_fast(a, M):
    n = len(a)
    Nc = 2 * n - 1
    rows = np.empty(n * n + n, dtype=np.int32)
    cols = np.empty(n * n + n, dtype=np.int32)
    vals = np.empty(n * n + n, dtype=np.float64)

        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        j_idx = np.arange(j_start, j_end + 1)
        a_vals = a[k - j_idx]
        cnt = len(j_idx)
        rows[ptr:ptr + cnt] = k
        cols[ptr:ptr + cnt] = j_idx
        vals[ptr:ptr + cnt] = a_vals
        ptr += cnt
    A = csr_matrix((vals[:ptr], (rows[:ptr], cols[:ptr])), shape=(Nc, n))
    c_obj = -np.ones(n)
    b_ub = M * np.ones(Nc)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

        b_norm = b / s_b
        sc_b = float(2.0 * n * _full_convolve(b_norm).max())

        if np.isfinite(sc_b) and sc_b < best_score - 1e-14:
            best_score = sc_b
            best_a = b_norm.copy()

        # Identify critical lags from the current convolution
        c_best = _full_convolve(best_a)
        M_best = float(c_best.max())
        threshold = M_best * 0.9
        critical_mask = c_best >= threshold
        n_critical = int(critical_mask.sum())

        if n_critical > 0 and n_critical < n:
            # Perturb positions that contribute most to critical lags
            heat = np.zeros(n)
            critical_indices = np.where(critical_mask)[0]
            for k in critical_indices:
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                j_idx = np.arange(j_start, j_end + 1)
                contrib = best_a[j_idx] * best_a[k - j_idx]
                heat[j_start:j_end + 1] += contrib

            if heat.max() > 1e-20:
                heat = heat / heat.max()
                # Reduce mass at hot positions, spread to cold
                eta = 0.03 + 0.02 * (outer / n_outer)
                factor = 1.0 - eta * heat
                factor = np.maximum(factor, 0.05)
                a_new = best_a * factor
                a_new = np.maximum(a_new, 1e-15)
                a_new = a_new / a_new.sum()
                sc_new = _score(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    a = best_a.copy()

        # Also try LP from the potentially improved point
        c_cur = _full_convolve(a)
        M_cur = float(c_cur.max())
        b2 = _lp_solve_fast(a, M_cur)
        if b2 is not None:
            s_b2 = float(b2.sum())
            if s_b2 > 1e-10:
                b2_norm = b2 / s_b2
                sc_b2 = float(2.0 * n * _full_convolve(b2_norm).max())
                if np.isfinite(sc_b2) and sc_b2 < best_score - 1e-14:
                    best_score = sc_b2
                    best_a = b2_norm.copy()

        a = best_a.copy()

    return best_a, best_score


def _multi_scale_attempt(a_best, n_target, deadline, rng):
    if a_best is None:
        return None, np.inf
    x_old = np.linspace(0, 1, a_best.size)
    x_new = np.linspace(0, 1, n_target)
    p_new = np.interp(x_new, x_old, a_best)
    p_new = np.maximum(p_new, 0.0)
    s = float(p_new.sum())
    if s < 0.01:
        return None, np.inf
    p_new = p_new / s
    sc = _score(p_new)
    return p_new, sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _try_update(p, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.maximum(p, 0.0)
        s = float(p.sum())
        if s < 0.01:
            return
        p = p / s
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}] t={elapsed:.0f}s")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is None:
        n = 1024
        t = np.linspace(0, 1, n)
        parent = (1.0 + 1.5 * np.cos(np.pi * t)) ** 2
        parent = np.maximum(parent, 1e-6)
        parent = parent / parent.sum()

    print(f"Parent: n={parent.size}, score={_score(parent):.10f}")
    _try_update(parent, "parent")

    # Phase 1: Quick Adam (200s)
    p1_end = t_start + 200.0
    print(f"\nPhase 1: Adam optimization (0 to 200s)")
    p1, sc1 = _adam_smooth_max(parent, p1_end, rng, lr=0.05, beta_frac=0.1,
                                beta_decay=0.93, no_imp_limit=200)
    _try_update(p1, "adam1")

    # Phase 2: LP alternation - main workhorse (500s)
    p2_end = t_start + 700.0
    print(f"\nPhase 2: LP alternation (200s to 700s)")
    start_score = best_score
    if best_seq is not None:
        p2, sc2 = _lp_alternation(best_seq, p2_end, t_init=0.75, t_min=0.25,
                                   t_decay=0.997, full_update_prob=0.4, n_restart=60)
        _try_update(p2, "lp1")
    print(f"  LP phase improvement: {start_score - best_score:.10f}")

    # Phase 3: Adam fine-tuning (200s)
    p3_end = t_start + 900.0
    print(f"\nPhase 3: Adam fine-tuning (700s to 900s)")
    if best_seq is not None:
        p3, sc3 = _adam_smooth_max(best_seq, p3_end, rng, lr=0.015, beta_frac=0.03,
                                    beta_decay=0.96, no_imp_limit=300, lr_min=0.001)
        _try_update(p3, "adam2")

    # Phase 4: Dual-guided LP refinement (300s)
    p4_end = t_start + 1200.0
    print(f"\nPhase 4: Dual-guided refinement (900s to 1200s)")
    if best_seq is not None:
        p4, sc4 = _dual_guided_refine(best_seq, p4_end, n_outer=30)
        _try_update(p4, "dual_lp")

    # Phase 5: Another LP pass with different params (200s)
    p5_end = t_start + 1400.0
    print(f"\nPhase 5: LP pass 2 (1200s to 1400s)")
    if best_seq is not None:
        p5, sc5 = _lp_alternation(best_seq, p5_end, t_init=0.5, t_min=0.2,
                                   t_decay=0.998, full_update_prob=0.5, n_restart=50)
        _try_update(p5, "lp2")

    # Phase 6: Multi-scale (200s)
    p6_end = deadline - 30.0
    print(f"\nPhase 6: Multi-scale (1400s onwards)")
    if best_seq is not None:
        for n_target in [2048, 4096]:
            if time.monotonic() > p6_end - 60:
                break
            p_new, sc_new = _multi_scale_attempt(best_seq, n_target, p6_end, rng)
            if p_new is not None:
                print(f"  Interp n={n_target}: score={sc_new:.10f}")
                if sc_new < best_score:
                    _try_update(p_new, f"interp_{n_target}")
                    # Quick Adam on the new scale
                    if time.monotonic() < p6_end - 30:
                        p6a, sc6a = _adam_smooth_max(p_new, min(p6_end - 10, deadline),
                                                      rng, lr=0.02, beta_frac=0.06,
                                                      no_imp_limit=100)
                        _try_update(p6a, f"adam_n{n_target}")

    # Phase 7: Final polish (remaining)
    if time.monotonic() < deadline - 5 and best_seq is not None:
        print(f"\nPhase 7: Final polish")
        remaining = deadline - time.monotonic() - 5
        if remaining > 30:
            p7, sc7 = _adam_smooth_max(best_seq, deadline - 3, rng, lr=0.005,
                                        beta_frac=0.02, beta_decay=0.97,
                                        no_imp_limit=100, lr_min=0.001)
            _try_update(p7, "final_adam")

    # Return best
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            n = 1024
            t = np.linspace(0, 1, n)
            final = (1.0 + 1.5 * np.cos(np.pi * t)) ** 2
            final = np.maximum(final, 1e-6)
            final = final / final.sum()

        final_score = _score(final)
        print(f"\nFINAL: score={final_score:.10f} n={final.size}")
        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            print(f"EVALUATE: {eval_sc:.10f}")
        except Exception:
            pass
        return [float(x) for x in final]

    # Fallback
    n = 1024
    t = np.linspace(0, 1, n)
    fb = (1.0 + 1.5 * np.cos(np.pi * t)) ** 2
    fb = np.maximum(fb, 1e-6)
    fb = fb / fb.sum()
    print(f"FALLBACK: n={n}, score={_score(fb):.10f}")
    return [float(x) for x in fb]
