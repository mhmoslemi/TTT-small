def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    m = 2 * n - 1

    k_idx = np.arange(m)[:, None]  # (m, 1)

    diff = k_idx - j_idx  # (m, n)

LP-based optimizer with L-BFGS polishing for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal update directions, combined with
smooth optimization for fine-tuning.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _build_conv_matrix(f):
    """Build matrix C such that C @ g = conv(f, g) (full convolution)."""
    n = len(f)
    m = 2 * n - 1
    k_idx = np.arange(m)
    j_idx = np.arange(n)
    # C[k, j] = f[k-j] if 0 <= k-j <= n-1
    diff = k_idx[:, None] - j_idx[None, :]
    mask = (diff >= 0) & (diff < n)
    C = np.zeros((m, n), dtype=np.float64)
    C[mask] = f[diff[mask]]
    return C

def _solve_lp_for_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g)_k <= M for all k, g >= 0."""
    n = len(f)
    m = 2 * n - 1
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    A_ub = C
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True})
    if result.success and result.x is not None:
        return np.maximum(0, result.x)
    return None

    betas = [b * max_c0 * beta_scale for b in beta_starts]

    if max_wall_time is not None:
        per_level = max(3.0, max_wall_time / len(betas))
    else:

        if time.monotonic() > deadline - 2.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            log_sum = float(np.log(s_))
            return 2.0 * n * (cm_ + beta * log_sum)

                    best_a = a_new.copy()
                    z = z_new.copy()

"""
LP-based optimizer with L-BFGS polishing for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: uses linear programming to find principled descent directions by
exploiting the bilinear structure of convolution (linear in each argument).
Combined with logsumexp-continuation L-BFGS for fine polishing.
"""
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import coo_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix_sparse(f):
    n = len(f)
    m = 2 * n - 1
    k_idx = np.arange(m)
    j_idx = np.arange(n)
    diff = k_idx[:, None] - j_idx[None, :]
    mask = (diff >= 0) & (diff < n)
    rows = np.where(mask)[0]
    cols = np.where(mask)[1]
    vals = f[diff[mask]]
    return coo_matrix((vals, (rows, cols)), shape=(m, n)).tocsr()


def _solve_lp_direction(f, M, time_limit=30.0):
    n = len(f)
    m = 2 * n - 1
    C = _build_conv_matrix_sparse(f)
    c_obj = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    t0 = time.monotonic()
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None, 0.0
    elapsed = time.monotonic() - t0
    if result.success and result.x is not None and np.all(np.isfinite(result.x)):
        return np.maximum(0, result.x), elapsed
    return None, elapsed


def _lp_iteration(a_init, deadline, rng, max_iters=40, t_start=0.5, t_end=0.03, lp_time_limit=20.0):
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = _score_fn(a)
    print(f"  LP iter: n={n}, initial score={best_score:.10f}")

    prev_score = best_score
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        c = _full_convolve(a, a)
        M = float(c.max())

        g, lp_time = _solve_lp_direction(a, M, time_limit=lp_time_limit)
        if g is None or np.sum(g) < 1e-15:
            print(f"  LP failed/timeout at iter {it} ({lp_time:.1f}s)")
            break

        g = g / g.sum()

        # Try multiple step sizes via line search
        t_schedule = t_start * (t_end / t_start) ** (it / max(1, max_iters - 1))
        best_t = 0
        best_sc = best_score
        for t_try in [t_schedule, t_schedule * 0.7, t_schedule * 0.5, t_schedule * 0.3, t_schedule * 0.1, 1.0]:
            a_try = (1.0 - t_try) * a + t_try * g
            a_try = np.maximum(a_try, 0)
            ss = a_try.sum()
            if ss < 1e-15:
                continue
            a_try = a_try / ss
            sc_try = _score_fn(a_try)
            if np.isfinite(sc_try) and sc_try < best_sc:
                best_sc = sc_try
                best_t = t_try

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            if best_sc < best_score:
                best_score = best_sc
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1
        else:
            stall_count += 1

        if it % 5 == 0:
            print(f"  LP iter {it}: score={best_score:.10f}, M={M:.6f}, t={best_t:.3f}, lp_t={lp_time:.1f}s")

        if stall_count >= 8:
            print(f"  LP stalled after {it} iterations")
            break

    return best_a, best_score


def _run_lbfgs_continuation(a_init, deadline, rng, max_wall_time=None, beta_scale=1.0):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.00025, 0.00012, 0.00006]
    betas = [b * max_c0 * beta_scale for b in beta_starts]

    if max_wall_time is not None:
        per_level = max(2.0, max_wall_time / len(betas))
    else:
        per_level = 20.0

    for beta in betas:
        if time.monotonic() > deadline - 2.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        def obj(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            log_sum = float(np.log(s_))
            return 2.0 * n * (cm_ + beta * log_sum)

        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            W = float(np.dot(w, c_))
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g

        try:
            result = minimize(
                obj, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 800, 'ftol': 1e-20, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 5000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _adam_refined(a_init, deadline, rng, lr=0.01, k_top=5):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    while time.monotonic() < deadline - 1.0:
        t += 1
        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)

        # Top-K weighted gradient
        k_actual = min(k_top, len(c))
        top_idx = np.argsort(c)[-k_actual:]
        grad = np.zeros(n)
        for k_val in top_idx:
            k_val = int(k_val)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * a_curr[k_val - hi:k_val - lo + 1][::-1]
        grad /= k_actual
        grad *= 4.0 * n

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -70, 70)
        z = z - z.mean()

        if t % 20 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()

        if t % 500 == 0:
            lr = max(0.002, lr * 0.98)

    return best_a, best_score


def _local_search(a_init, deadline, rng, n_trials=8000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_fn(a)

    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break

        if it < n_trials // 3:
            block_size = int(rng.integers(2, min(15, n // 8)))
        elif it < 2 * n_trials // 3:
            block_size = int(rng.integers(5, min(40, n // 4)))
        else:
            block_size = int(rng.integers(15, min(80, n // 2)))

        start = int(rng.integers(0, max(1, n - block_size + 1)))

        a_trial = a.copy()
        perturbation = rng.uniform(0.75, 1.25, size=block_size)
        a_trial[start:start + block_size] *= perturbation
        a_trial = np.maximum(a_trial, 1e-15)
        ss = a_trial.sum()
        if ss < 1e-15:
            continue
        a_trial /= ss

        sc = _score_fn(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
        elif rng.random() < 0.003 and sc < best_score * 1.003:
            a = a_trial.copy()

    return best_a, best_score


def _make_profiles(n):
    profiles = []
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0

    for slope in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        a = np.maximum(1.0 + slope * (t - 0.5), 1e-12)
        profiles.append(a / a.sum())

    for a1, a2 in [(0.5, 0.5), (1.0, -0.3), (1.5, 0.1), (0.8, 0.8), (2.0, -0.5)]:
        a = np.maximum(1.0 + a1 * u + a2 * u**2, 1e-12)
        profiles.append(a / a.sum())

    for width in [0.12, 0.18, 0.25, 0.33, 0.42]:
        a = np.exp(-u**2 / (2.0 * width**2))
        profiles.append(a / a.sum())

    for alpha in [0.5, 0.8, 1.0, 1.5]:
        a = np.maximum(0.5 * (1.0 + np.cos(np.pi * u * alpha)), 1e-12)
        profiles.append(a / a.sum())

    for p in [1.5, 2.0, 3.0, 4.0, 6.0]:
        a = np.maximum((1.0 - np.abs(u)) ** p, 1e-12)
        profiles.append(a / a.sum())

    a = np.maximum(1.0 - np.abs(u), 1e-12)
    profiles.append(a / a.sum())

    for cw in [0.15, 0.2, 0.3]:
        a = np.exp(-((t - 0.25) / cw)**2) + np.exp(-((t - 0.75) / cw)**2) + 0.3 * np.exp(-((t - 0.5) / (cw*1.5))**2)
        profiles.append(a / a.sum())

    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

        n_parent = 2048
        parent = _make_profiles(n_parent)[8]
        print(f"No parent, using default n={n_parent}")

    # === Phase 1: Diverse initialization (15s) ===
    phase1_end = min(deadline, t_start + 15.0)
    n_base = 2048

    profiles = _make_profiles(n_base)
    for idx, p in enumerate(profiles):
        if time.monotonic() > phase1_end:
            break
        sc = _score_fn(p)
        update_best(p, sc, f"init_{idx}")

    if parent is not None and len(parent) != n_base:
        p_resized = _interp_profile(parent, n_base)
        sc_r = _score_fn(p_resized)
        update_best(p_resized, sc_r, "parent_resized")

    print(f"Phase 1 done: best={best_score:.10f}")

    # === Phase 2: LP iteration at n=2048 (250s) ===
    phase2_end = min(deadline, t_start + 265.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        a_start = best_seq.copy() if len(best_seq) == n_base else _interp_profile(best_seq, n_base)
        p_best, sc_best = _lp_iteration(a_start, phase2_end, rng,
                                        max_iters=50, t_start=0.5, t_end=0.02,
                                        lp_time_limit=15.0)
        update_best(p_best, sc_best, "lp_2048")

    print(f"Phase 2 done: best={best_score:.10f}")

    # === Phase 3: L-BFGS at n=2048 (180s) ===
    phase3_end = min(deadline, t_start + 445.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        a_opt = best_seq.copy() if len(best_seq) == n_base else _interp_profile(best_seq, n_base)
        dl = phase3_end - 5.0
        p_best, sc_norm = _run_lbfgs_continuation(a_opt, dl, rng, max_wall_time=160.0, beta_scale=1.0)
        sc = _score_fn(p_best)
        update_best(p_best, sc, "lbfgs_2048")

    print(f"Phase 3 done: best={best_score:.10f}")

    # === Phase 4: LP at n=4096 (350s) ===
    phase4_end = min(deadline, t_start + 795.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        n_large = 4096
        a_large = _interp_profile(best_seq, n_large)
        sc_large = _score_fn(a_large)
        update_best(a_large, sc_large, "interp_4096")

        if time.monotonic() < phase4_end - 60:
            p_best, sc_best = _lp_iteration(a_large, phase4_end, rng,
                                            max_iters=20, t_start=0.4, t_end=0.02,
                                            lp_time_limit=25.0)
            update_best(p_best, sc_best, "lp_4096")

    print(f"Phase 4 done: best={best_score:.10f}")

    # === Phase 5: L-BFGS on best (200s) ===
    phase5_end = min(deadline, t_start + 995.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        dl = phase5_end - 5.0
        p_best, sc_norm = _run_lbfgs_continuation(best_seq, dl, rng,
                                                  max_wall_time=180.0, beta_scale=0.7)
        sc = _score_fn(p_best)
        update_best(p_best, sc, "lbfgs_best")

    print(f"Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Adam refinement (120s) ===
    phase6_end = min(deadline, t_start + 1115.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        dl = phase6_end - 5.0
        p_best, sc_norm = _adam_refined(best_seq, dl, rng, lr=0.008, k_top=4)
        sc = _score_fn(p_best)
        update_best(p_best, sc, "adam_refined")

    print(f"Phase 6 done: best={best_score:.10f}")

    # === Phase 7: Local search (120s) ===
    phase7_end = min(deadline, t_start + 1235.0)
    if best_seq is not None and time.monotonic() < phase7_end:
        p_best, sc_best = _local_search(best_seq, phase7_end, rng, n_trials=6000)
        update_best(p_best, sc_best, "local_search")

    print(f"Phase 7 done: best={best_score:.10f}")

    # === Phase 8: Multi-restart with quick LP + L-BFGS (150s) ===
    phase8_end = min(deadline, t_start + 1385.0)
    if best_seq is not None and time.monotonic() < phase8_end:
        n_restarts = 4
        per_restart = max(25.0, (phase8_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase8_end - 10:
                break
            noise = 0.03 + 0.02 * i
            n_r = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=n_r)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            dl = min(phase8_end - 5, time.monotonic() + per_restart)
            # Quick LP (very few iterations)
            p_best, sc_best = _lp_iteration(p_start, dl + 5, rng,
                                            max_iters=8, t_start=0.3, t_end=0.1,
                                            lp_time_limit=8.0)
            # Quick L-BFGS
            if time.monotonic() < dl - 10:
                p_lb, sc_lb_norm = _run_lbfgs_continuation(p_best, dl, rng,
                                                          max_wall_time=per_restart * 0.4,
                                                          beta_scale=0.5)
                sc_lb = _score_fn(p_lb)
                if sc_lb < sc_best:
                    p_best = p_lb
                    sc_best = sc_lb
            update_best(p_best, sc_best, f"restart_{i}")

    print(f"Phase 8 done: best={best_score:.10f}")

    # === Phase 9: Final intensive polishing (remaining time) ===
    if best_seq is not None:
        remaining = max(10.0, deadline - time.monotonic())
        n_polish = 3
        per_polish = remaining / n_polish

        for i in range(n_polish):
            if time.monotonic() > deadline - 5:
                break
            if i == 0:
                dl = min(deadline - 5, time.monotonic() + per_polish)
                p_best, sc_norm = _run_lbfgs_continuation(best_seq, dl, rng,
                                                         max_wall_time=per_polish,
                                                         beta_scale=0.3)
                sc = _score_fn(p_best)
                update_best(p_best, sc, f"final_lbfgs_{i}")
            elif i == 1:
                dl = min(deadline - 5, time.monotonic() + per_polish)
                p_best, sc_best = _local_search(best_seq, dl, rng, n_trials=4000)
                update_best(p_best, sc_best, f"final_local_{i}")
            else:
                dl = min(deadline - 5, time.monotonic() + per_polish)
                p_best, sc_norm = _adam_refined(best_seq, dl, rng, lr=0.004, k_top=3)
                sc = _score_fn(p_best)
                update_best(p_best, sc, f"final_adam_{i}")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            n_fb = 2048
            t = np.linspace(0, 1, n_fb)
            final_seq = np.maximum(1.0 + 1.5 * (t - 0.5), 1e-12)
            final_seq = final_seq / final_seq.sum()
            s = float(final_seq.sum())
        final_seq = final_seq / s

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        if len(final_seq) <= 8192:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 2048
    t = np.linspace(0, 1, n_fb)
    fallback = np.maximum(1.0 + 1.5 * (t - 0.5), 1e-12)
    fallback = fallback / fallback.sum()
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n={n_fb}, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
