For the KKT solver, here's the exact approach:
- Given active set A = {k_1, ..., k_m}
- The KKT conditions are:

  (ii) (a*a)[k] ≤ M for all k not in A
  (iii) a[i] ≥ 0 for all i
  (iv) For all i with a[i] > 0: sum_{j: k_j - i ∈ [0,n-1]} λ_j * a[k_j - i] = μ (some constant)

Condition (iv) is the key. It says that the "weighted sum of shifts of a" is constant on the support of a.

where a[j] = 0 for j outside [0, n-1].

Actually, we have:
- n equations from (iv): sum_j λ_j a[k_j - i] = μ for i = 0,...,n-1
- 1 equation from normalization: sum a[i] = 1
- m equations from (i): (a*a)[k_j] = M for j = 1,...,m
- 1 equation: M is the common value

Total: n + 1 + m + 1 = n + m + 2 equations in n + m + 2 unknowns (a[0..n-1], λ_1..λ_m, μ, M).

def fft_convolve(a, b):

    r = np.fft.irfft(fa * fb, N)[:out_n]

def lbfgs_smooth(a_init, deadline, top_k=10):

                ct = c_val[idx]
                ct_shifted = ct - ct.max()
                return 2.0 * n * (ct.max() + beta * np.log(np.sum(np.exp(ct_shifted / beta))))
            else:
                cs = (c_val - c_val.max()) / beta
                return 2.0 * n * (c_val.max() + beta * np.log(np.sum(np.exp(cs))))
        
        def grad(z_val):
            a_val = softmax(z_val)
            c_val = fft_convolve(a_val, a_val)
            if top_k > 0 and top_k < len(c_val):
                idx = np.argsort(c_val)[-top_k:]
                ct = c_val[idx]
                ct_shifted = (ct - ct.max()) / beta
                e = np.exp(ct_shifted)
                w = np.zeros_like(c_val)
                w[idx] = e / e.sum()
            else:
                cs = (c_val - c_val.max()) / beta
                e = np.exp(cs)
                w = e / e.sum()
            W = np.dot(w, c_val)
            a_rev = a_val[::-1]
            w_conv = fft_convolve(w, a_rev)
            h = w_conv[n-1:2*n-1]
            g = 2.0 * n * a_val * (h - W)
            return g if np.all(np.isfinite(g)) else np.zeros(n)
        
        try:
            result = minimize(
                obj, z, jac=grad,
                method='L-BFGS-B',
                options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-10, 'maxls': 20, 'maxfun': 1000}

def adam_optimize(a_init, deadline, lr=0.03, top_k=10, max_iter=100000):
    """Adam optimizer in logit space with top-k smoothing."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    best_a = a.copy()
    best_score = score_normalized(a)
    
    c = fft_convolve(a, a)
    max_c = float(c.max())
    beta = 0.01 * max_c
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        
        a = softmax(z)
        c = fft_convolve(a, a)
        
        if top_k > 0 and top_k < len(c):
            idx = np.argsort(c)[-top_k:]
            ct = c[idx]
            ct_shifted = (ct - ct.max()) / beta
            e = np.exp(ct_shifted)
            w = np.zeros_like(c)
            w[idx] = e / e.sum()
        else:
            cs = (c - c.max()) / beta
            e = np.exp(cs)
            w = e / e.sum()
        
        W = np.dot(w, c)
        a_rev = a[::-1]
        w_conv = fft_convolve(w, a_rev)
        h = w_conv[n-1:2*n-1]
        g = 2.0 * n * a * (h - W)
        
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)
        
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * (g * g)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a = softmax(z)
        
        sc = score_normalized(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if t % 200 == 0:
            beta = max(1e-14, beta * 0.9)
        
        if t % 5000 == 0:
            if best_score < 1.51:
                lr = max(0.005, lr * 0.95)
    
    return best_a / best_a.sum(), best_score

def interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-15)
    return p / p.sum()
