def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    """Compute 2*n*max(conv(a,a)) without normalization (scale-invariant)."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max()) / (s * s)

    n = len(a_init)

    best_score = _score_norm(a_init)

    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t += 1
        obj, grad = _objective_and_grad(z, beta, n, None)
        
        if not np.all(np.isfinite(grad)):
            break
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        
        a = np.exp(z)
        a = np.maximum(a, 0.0)

            best_score = sc
            best_a = a.copy()
