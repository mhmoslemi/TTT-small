```python
multi-restart, and multi-resolution (n=2048, n=4096) strategies.

import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    # Compute using FFT: conv(a_rev, w)[n-1:2n-1]
    a_rev = a[::-1]
    N = _next_pow2(2 * n + 2 * (2 * n - 1) - 1)  # need N >= n + (2n-1) - 1 = 3n-2
    # Actually N >= max(len(a_rev) + len(w) - 1) = n + (2n-1) - 1 = 3n-2
    N = _next_pow2(3 * n)  # safe upper bound
    A = np.fft.rfft(a_rev, N)
    W = np.fft.rfft(w, N)
    C = np.fft.irfft(A * W, N)
    grad_fn = C[n-1:2*n-1]  # length n
    
    grad_c = (4.0 * n / beta) * grad_fn
    grad_z = a * (grad_c - np.dot(a, grad_c))
    
    return a, score, grad_z
```

```python
def _lp_best_response(f, M):
    """
    Solve: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    f has length n, g has length n.
    """
    import cvxpy as cp
    
    n = len(f)
    g = cp.Variable(n)
    
    constraints = [g >= 0]
    
    # (f * g)[k] = sum_j f[j] * g[k-j] <= M
    # For k = 0, ..., 2n-2:
    # (f*g)[k] = sum_{j=max(0,k-n+1)}^{min(k,n-1)} f[j] * g[k-j]
    
    for k in range(2 * n - 1):
        # Build the linear expression: sum_j f[j] * g[k-j]
        expr = 0
        for j in range(max(0, k - n + 1), min(k, n - 1) + 1):
            expr += f[j] * g[k - j]
        constraints.append(expr <= M)
    
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve()
    
    if g.value is not None:
        return np.maximum(g.value, 0)
    return np.ones(n) / n
```

```python
def _build_conv_matrix(f):
    """Build the matrix A such that A @ g = conv(f, g)."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(k, n-1)+1):
            A[k, j] = f[j]
    return A

def _lp_best_response(f, M):
    import cvxpy as cp
    n = len(f)
    A = _build_conv_matrix(f)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Probe(objective, constraints)
    prob.solve(solver=cp.CLARABEL)  # or ECOS, SCS
    if g.value is not None:
        return np.maximum(g.value, 0)
    return np.ones(n) / n
```

```python
    cols = []

        j_end = min(k, n-1)

            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A
```

Actually, one more thought: I should also try **direct subgradient optimization** of the non-smooth objective as a final polish step. The idea is:

1. Compute `c = conv(a, a)`.
2. Find all `k` where `c[k] = max(c)` (or within `epsilon` of max).
3. Compute the subgradient: `grad[m] = sum_{k in argmax} 2*a[k-m]` (normalized).
4. Take a step in the negative subgradient direction.
5. Project back to the simplex `{a >= 0, sum(a) = 1}`.

The projection onto the simplex is:
- Compute `a - step * grad`.
- Find the largest `tau` such that `a - step * grad - tau >= 0` and `sum(a - step * grad - tau) = 1`.
- This is: `tau = (sum(a - step*grad) - 1) / n` if all `a - step*grad - tau >= 0`, otherwise solve a smaller problem.

Actually, the projection onto the simplex is:
`proj_simplex(v) = max(v - tau, 0)` where `tau` is chosen so that `sum(proj) = 1`.

This can be found by sorting: if `v` sorted in decreasing order is `v_(1) >= v_(2) >= ... >= v_(n)`, then `tau = (sum_{i=1}^k v_(i) - 1) / k` where `k` is the largest index such that `v_(k) > (sum_{i=1}^k v_(i) - 1) / k`.

This is O(n log n) per step.
