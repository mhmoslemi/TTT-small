    return np.maximum(c, 0.0)

    c = _full_convolve(p)
    return float(2.0 * n * c.max())

    """Build the convolution matrix A such that (A @ b)_k = (a*b)_k"""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):

        j_end = min(n - 1, k)

            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A
