"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses polynomial parameterization for initialization, L-BFGS continuation with
log-sum-exp smoothing, Nesterov-accelerated gradient descent with backtracking
line search, Adam polishing, and structured multi-restart strategy.
Key improvements: fixed bugs, adaptive beta schedule, line search, polynomial
initialization, and more aggressive fine-tuning phases.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _gradient_a(a, w, n):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c=conv(a,a). Result is 4*n*h."""
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    return 4.0 * n * h


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _objective_smooth_z(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _gradient_smooth_z(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, _ = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _lbfgs_obj(z, beta):
    return _objective_smooth_z(z, beta)


def _lbfgs_grad(z, beta):
    return _gradient_smooth_z(z, beta)


def _run_lbfgs(a_init, deadline, max_wall_time=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))

    best_a = a_init.copy()

    if max_wall_time is None:

    betas = [0.1, 0.05, 0.025, 0.01, 0.005, 0.002, 0.001,

    betas = [b * max_c0 for b in betas]

    per_level = max(1.0, max_wall_time / len(betas))

def _adam_optimize(a_init, deadline, lr=0.03, beta_frac=0.005):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    v = np.zeros(n, dtype=np.float64)

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-15, beta_frac * max_c)
    t = 0

    while time.monotonic() < deadline:
        t += 1

        grad = _gradient_smooth_z(z, beta)
        if not np.all(np.isfinite(grad)):
            continue

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        if t % 200 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            beta = max(1e-15, beta * 0.9)
            c_new = _full_convolve(a_curr, a_curr)
            max_c = float(c_new.max())
            if t % 1000 == 0:
                lr = max(0.001, lr * 0.95)

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _make_polynomial_profile(n, coeffs):
    """Create profile from polynomial coefficients ( ascending order)."""
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for i, c in enumerate(coeffs):
        a += c * t ** i
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_power_profile(n, p):
    """Power-law decay: a[i] proportional to (i+1)^(-p)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    return a / a.sum()


def _make_hann_profile(n):
    """Hann window profile."""
    t = np.linspace(0.0, np.pi, n)
    a = 0.5 * (1 + np.cos(t))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_asymmetric_profile(n, alpha):
    """Asymmetric profile: high on left, decaying."""
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 - t) ** alpha
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _poly_obj(coeffs, n):

        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        momentum = mom_coeff * momentum - lr * grad
        z = z + momentum
        z = z - z.mean()
        z = np.clip(z, -60, 60)

        if it % 200 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            else:
                lr = lr * 0.85

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: polynomial init -> L-BFGS continuation -> Nesterov
    with adaptive step -> Adam polish -> multi-restart -> fine active-set.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Phase 0: Initialization (30s) ---
    phase0_end = min(deadline, t_start + 30.0)
    n_default = 4096

    candidates = []

    # Try parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_p = _score_norm_fn(parent)
        candidates.append((sc_p, parent.copy(), len(parent), "parent"))
        print(f"Parent: n={len(parent)}, score={sc_p:.10f}")
    else:
        print("No parent available, using synthetic profiles.")

    # Try various profiles at n=4096
    profiles = [
        ("poly_p1.2", _make_power_profile(n_default, 1.2)),
        ("poly_p1.5", _make_power_profile(n_default, 1.5)),
        ("poly_p2.0", _make_power_profile(n_default, 2.0)),
        ("asy_1.5", _make_asymmetric_profile(n_default, 1.5)),
        ("asy_2.0", _make_asymmetric_profile(n_default, 2.0)),
        ("asy_3.0", _make_asymmetric_profile(n_default, 3.0)),
        ("lin_1.5", _make_linear_profile(n_default, 1.5)),
        ("quad", _make_quadratic_profile(n_default, 1.2, -0.3)),
    ]

    for name, p in profiles:
        if time.monotonic() > phase0_end - 2:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p, n_default, name))
        print(f"  Profile {name}: score={sc:.10f}")

    # Try polynomial optimization (quick)
    for degree in [2, 3]:
        if time.monotonic() > phase0_end - 3:
            break
        coeffs, sc = _poly_optimize(n_default, degree, min(phase0_end, time.monotonic() + 5))
        p = _make_polynomial_profile(n_default, coeffs)
        if time.monotonic() < phase0_end - 1:
            sc_check = _score_norm_fn(p)
            candidates.append((sc_check, p, n_default, f"poly_d{degree}"))
            print(f"  Poly degree {degree}: score={sc_check:.10f}, coeffs={coeffs}")

    # Sort by score
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 best: {candidates[0][0]:.10f} ({candidates[0][3]})")

    # Use best candidate
    if candidates:
        best_seq = candidates[0][1].copy()
        best_score = candidates[0][0]
        print(f"Starting with: {candidates[0][3]}, score={best_score:.10f}, n={len(best_seq)}")
    else:
        best_seq = _make_power_profile(n_default, 1.5)
        best_score = _score_norm_fn(best_seq)
        print(f"Default start, score={best_score:.10f}")

    # --- Phase 1: L-BFGS continuation (400s) ---
    phase1_end = min(deadline, t_start + 430.0)
    if time.monotonic() < phase1_end:
        per_time = min(350.0, phase1_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _run_lbfgs(best_seq, phase1_end, max_wall_time=per_time)
            update_best(p_best, sc_best, "lbfgs_phase1")

    # Also try L-BFGS on second-best if time allows
    if len(candidates) > 1 and time.monotonic() < phase1_end - 30:
        for sc, p, pn, label in candidates[1:3]:
            if time.monotonic() > phase1_end - 15:
                break
            if len(p) != len(best_seq):
                continue
            per_time = min(40.0, phase1_end - time.monotonic() - 10)
            if per_time > 10:
                p_best, sc_best = _run_lbfgs(p, phase1_end - 5, max_wall_time=per_time)
                update_best(p_best, sc_best, f"lbfgs_{label}")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Nesterov with aggressive continuation (500s) ---
    phase2_end = min(deadline, t_start + 930.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        beta_fracs = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                      5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]
        per_time = min(450.0, phase2_end - time.monotonic())
        if per_time > 30:
            p_best, sc_best = _nesterov_optimize(
                best_seq, min(phase2_end - 5, time.monotonic() + per_time),
                beta_fracs, lr_init=0.012, mom_init=0.88
            )
            update_best(p_best, sc_best, "nesterov_phase2")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Adam polishing (300s) ---
    phase3_end = min(deadline, t_start + 1230.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        per_time = min(280.0, phase3_end - time.monotonic())
        if per_time > 30:
            p_best, sc_best = _adam_optimize(
                best_seq, min(phase3_end - 5, time.monotonic() + per_time),
                lr=0.025, beta_frac=0.003
            )
            update_best(p_best, sc_best, "adam_phase3")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Multi-restart (250s) ---
    phase4_end = min(deadline, t_start + 1480.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        n_best = len(best_seq)
        n_restarts = 5
        per_restart = max(10.0, (phase4_end - time.monotonic() - 10) / n_restarts)

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 10:
                break
            # Structured perturbation
            if i == 0:
                noise_level = 0.03
            elif i == 1:
                noise_level = 0.05
            elif i == 2:
                noise_level = 0.08
            elif i == 3:
                noise_level = 0.12
            else:
                noise_level = 0.15

            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            # Add structured noise: low-frequency + high-frequency
            z_noise = rng.normal(size=n_best) * noise_level
            # Add a low-frequency component
            low_f = np.sin(np.linspace(0, 2 * np.pi, n_best, endpoint=False)) * noise_level * 0.5
            z = z + z_noise + low_f
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            per_dl = min(phase4_end - 5, time.monotonic() + per_restart)
            beta_fracs_short = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]
            p_best, sc_best = _nesterov_optimize(
                p_start, per_dl, beta_fracs_short, lr_init=0.01, mom_init=0.85
            )
            update_best(p_best, sc_best, f"restart{i}_nl{noise_level}")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Fine polishing (remaining time) ---
    phase5_end = deadline
    if best_seq is not None and time.monotonic() < phase5_end:
        remaining = phase5_end - time.monotonic()

        # Nesterov with tiny beta
        if remaining > 60:
            per_time = min(120.0, remaining - 60)
            beta_fracs_tiny = [1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]
            p_best, sc_best = _nesterov_optimize(
                best_seq, min(phase5_end - 60, time.monotonic() + per_time),
                beta_fracs_tiny, lr_init=0.004, mom_init=0.95
            )
            update_best(p_best, sc_best, "fine_nesterov")

        # Adam with tiny beta
        if time.monotonic() < phase5_end - 40:
            per_time2 = min(80.0, phase5_end - time.monotonic() - 30)
            if per_time2 > 20:
                p_best, sc_best = _adam_optimize(
                    best_seq, min(phase5_end - 30, time.monotonic() + per_time2),
                    lr=0.008, beta_frac=0.001
                )
                update_best(p_best, sc_best, "fine_adam")

        # Active set polish
        if time.monotonic() < phase5_end - 10:
            per_time3 = min(60.0, phase5_end - time.monotonic() - 5)
            if per_time3 > 15:
                p_best, sc_best = _active_set_polish(
                    best_seq, min(phase5_end - 2, time.monotonic() + per_time3)
                )
                update_best(p_best, sc_best, "active_set")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_power_profile(2048, 1.5)

        sc_final = _score_fn(final_seq)
        print(f"FINAL internal score: {sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned non-finite, using internal: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_power_profile(2048, 1.5)
    print(f"FALLBACK: n=2048, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
