2*n*max(conv(a,a))/sum(a)^2. Uses linear programming to find optimal

        k <<= 1

    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]

    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _build_lp_matrix(n):
    """Build the sparse constraint matrix structure (without values) for LP with sequence length n."""
    # For sequence f of length n, the constraint matrix A has shape (2n-1, n)
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # We return the row and column indices of nonzeros
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            rows.append(k)
            cols.append(j)

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z(a, c, w, n):
    """Compute gradient of smoothed max objective w.r.t. log-space z."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adam_optimize(a_init, deadline, beta_schedule, max_iter=200000, lr=0.01):
    """Adam optimizer in log-space with beta-continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_count = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t_count += 1

            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad = _compute_grad_z(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t_count)
            vh = v_buf / (1.0 - b2 ** t_count)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t_count % 100 == 0:

"""

2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):

    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _build_sparse_A(n, f):
    """Build sparse matrix for LP constraints: A[k,j] = f[k-j] if valid."""
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n, k + 1)
        for j in range(j_lo, j_hi):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    A = csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))
    return A


def _lp_solve_direction(f, M):
    """Solve: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0."""
    n = len(f)
    A = _build_sparse_A(n, f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def _lp_optimize(f_init, deadline, max_iters=300, verbose=True):
    """LP-based Frank-Wolfe: iteratively compute LP direction and blend."""
    n = len(f_init)
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return f_init, _score(f_init)
    f = f / s

    best_f = f.copy()
    best_score = _score(f)

        if time.monotonic() > deadline:
            break

        c = _full_convolve(f, f)
        M = float(c.max())
        if M < 1e-15:
            break

        g = _lp_solve_direction(f, M)
        if g is None:
            if verbose:
                print(f"  LP failed at iter {it}, stopping")
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Line search over step sizes
        best_t = 0.0
        best_sc = best_score
        step_candidates = [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]

            f_candidate = (1.0 - t) * f + t * g

            if sc_candidate < best_sc:
                best_sc = sc_candidate
                best_t = t

        if best_t > 0.0:

        elif verbose:
            print(f"  LP iter {it}: no improvement found (M={M:.6f})")

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _adam_grad(a, c, w, n):
    """Gradient of smoothed objective w.r.t. z (log-space)."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adam_optimize(a_init, deadline, beta_schedule, max_iter=200000, lr=0.01):
    """Adam in log-space with beta-continuation on smoothed max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_count = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_phase = 0
        while it_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_phase += 1
            t_count += 1

            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shift = (c - cm) / beta
            e = np.exp(c_shift)
            s_e = float(e.sum())
            if s_e < 1e-300 or not np.isfinite(s_e):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_e

            grad = _adam_grad(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t_count)
            vh = v_buf / (1.0 - b2 ** t_count)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            un = np.linalg.norm(update)
            if un > 0.5:
                update = update * (0.5 / un)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t_count % 100 == 0:
                sc = _score(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

            if it_phase % 5000 == 0:
                current_lr = max(0.001, current_lr * 0.97)

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _make_profile(n, kind):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    return np.ones(n) / n

    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid LP (Frank-Wolfe) + Adam optimization.
    Phase 1: LP-based optimization from parent (0-500s)
    Phase 2: LP from perturbed best (500-800s)
    Phase 3: Adam fine polishing (800-1300s)
    Phase 4: LP refinement after Adam (1300-1500s)
    Phase 5: Final Adam polish (1500-1770s)
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            if parent_raw.sum() > 0.01:
                parent = parent_raw / parent_raw.sum()
    except Exception:
        parent = None

    if parent is not None:
        update_best(parent, "parent")
        print(f"Parent loaded: n={len(parent)}, score={best_score:.10f}")
    else:
        n_init = 2048
        update_best(_make_profile(n_init, "uniform"), "uniform")
        print(f"No parent, using uniform n={n_init}")

    # === Phase 1: LP optimization (0 - 500s) ===
    phase1_end = min(deadline, t_start + 500.0)
    if best_seq is not None and time.monotonic() < phase1_end - 30:
        print(f"\nPhase 1: LP optimization (target {phase1_end - t_start:.0f}s)")
        p_lp, sc_lp = _lp_optimize(best_seq, phase1_end, max_iters=500, verbose=True)
        update_best(p_lp, "lp_phase1")

    # === Phase 2: LP from perturbed best (500s - 800s) ===
    phase2_end = min(deadline, t_start + 800.0)
    if best_seq is not None and time.monotonic() < phase2_end - 20:
        print(f"\nPhase 2: LP from perturbed start (target {phase2_end - t_start:.0f}s)")
        for perturb_idx in range(3):
            remaining = phase2_end - time.monotonic() - 20
            if remaining < 60:
                break
            sigma = [0.03, 0.06, 0.10][perturb_idx % 3]
            p_start = _perturb(best_seq, sigma, rng)
            p_lp, sc_lp = _lp_optimize(p_start, time.monotonic() + remaining * 0.4,
                                       max_iters=100, verbose=False)
            update_best(p_lp, f"lp_perturb{perturb_idx}")

    # === Phase 3: Adam fine polishing (800s - 1350s) ===
    phase3_end = min(deadline, t_start + 1350.0)
    if best_seq is not None and time.monotonic() < phase3_end - 30:
        print(f"\nPhase 3: Adam polishing (target {phase3_end - t_start:.0f}s)")
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        max_cb = float(c_best.max())
        beta_schedule = [
            0.01 * max_cb, 0.005 * max_cb, 0.002 * max_cb,
            0.001 * max_cb, 0.0005 * max_cb, 0.0002 * max_cb,
            0.0001 * max_cb, 0.00005 * max_cb
        ]
        p_adam, sc_adam = _adam_optimize(
            best_seq, phase3_end, beta_schedule,
            max_iter=500000, lr=0.012
        )
        update_best(p_adam, "adam_phase3")

    # === Phase 4: LP refinement after Adam (1350s - 1550s) ===
    phase4_end = min(deadline, t_start + 1550.0)
    if best_seq is not None and time.monotonic() < phase4_end - 30:
        print(f"\nPhase 4: LP refinement (target {phase4_end - t_start:.0f}s)")
        p_lp2, sc_lp2 = _lp_optimize(best_seq, phase4_end, max_iters=200, verbose=True)
        update_best(p_lp2, "lp_phase4")

    # === Phase 5: Final Adam polish (1550s - 1770s) ===
    phase5_end = min(deadline, t_start + 1770.0)
    if best_seq is not None and time.monotonic() < phase5_end - 10:
        print(f"\nPhase 5: Final Adam polish (target {phase5_end - t_start:.0f}s)")
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        max_cb = float(c_best.max())
        beta_fine = [
            0.0001 * max_cb, 0.00005 * max_cb,
            0.00002 * max_cb, 0.00001 * max_cb
        ]
        p_final, sc_final = _adam_optimize(
            best_seq, phase5_end, beta_fine,
            max_iter=300000, lr=0.005
        )
        update_best(p_final, "final_adam")

    # Try a quick larger-n interpolation if time permits
    if best_seq is not None and len(best_seq) < 4096 and time.monotonic() < deadline - 30:
        print(f"\nBonus: Try n=4096 interpolation")
        p_interp = _interp_profile(best_seq, 4096)
        sc_interp = _score(p_interp)
        if sc_interp < best_score:
            update_best(p_interp, "interp_4096")
            # Quick Adam polish on the interpolated version
            if time.monotonic() < deadline - 15:
                c_i = _full_convolve(p_interp, p_interp)
                mci = float(c_i.max())
                bs_i = [0.005 * mci, 0.002 * mci, 0.001 * mci]
                p_opt, sc_opt = _adam_optimize(
                    p_interp, deadline, bs_i, max_iter=100000, lr=0.008
                )
                update_best(p_opt, "opt_4096")

    # Final validation and return
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        sc = _score(final)
        print(f"\nFINAL: score={sc:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
