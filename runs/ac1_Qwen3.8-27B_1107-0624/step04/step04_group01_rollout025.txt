import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

L-BFGS optimization with log-sum-exp continuation for minimizing
2*n*max(conv(a,a))/sum(a)^2. Combines L-BFGS (second-order) with
Adam (first-order) for robustness, along with multi-restart and
n-expansion strategies.

import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):

    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_h(w, a, n):
    """h[k] = sum_m w[m] * a[m-k] for k in range(n)."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return wc[n-1:2*n-1]


def _lbfgs_obj(z, n, beta):

    c = _full_convolve(a, a)
    cm = c.max()
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    S = float(e.sum())
    if S < 1e-300 or not np.isfinite(S):
        return 2.0 * n * cm
    return 2.0 * n / beta * (np.log(S) + cm / beta)


def _lbfgs_grad(z, n, beta):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = c.max()
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    S = float(e.sum())
    if S < 1e-300 or not np.isfinite(S):
        return np.zeros_like(z)
    w = e / S
    h = _compute_h(w, a, n)
    W = float(np.dot(w, c))
    return 4.0 * n * a / (beta * beta) * (h - W)


def _adam_step(z, n, beta, m_buf, v_buf, t, lr):
    """One Adam step in z-space. Returns updated z, m_buf, v_buf, a, score."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = c.max()
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    S = float(e.sum())
    if S < 1e-300 or not np.isfinite(S):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / S
    
    h = _compute_h(w, a, n)
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    
    if not np.all(np.isfinite(grad)):
        return z, m_buf, v_buf, t
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m_buf = b1 * m_buf + (1.0 - b1) * grad
    v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
    mh = m_buf / (1.0 - b1 ** t)
    vh = v_buf / (1.0 - b2 ** t)
    
    update = lr * mh / (np.sqrt(vh) + eps)
    z = z - update
    z = np.clip(z, -50.0, 50.0)
    z = z - z.mean()
    
    return z, m_buf, v_buf, t + 1


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "rick":
        t = np.linspace(0, 1, n)
        center = 0.5
        a = 1.0 / (1.0 + 10.0 * (t - center) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

    z = z + sigma * rng.standard_normal(len(a))

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _lbfgs_phase(a_init, beta_schedule, max_iter=5000, gtol=1e-7):
    """Run L-BFGS with beta continuation. Returns best a found."""
    n = len(a_init)

    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    for beta in beta_schedule:
        result = minimize(
            _lbfgs_obj, z, args=(n, beta),
            jac=_lbfgs_grad,
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'gtol': gtol, 'maxls': 100,
                     'maxfun': max_iter * 3}
        )
        z = result.x
        a = _softmax(z)
        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score


def _adam_phase(a_init, deadline, beta_schedule, max_iter=100000, lr=0.01):
    """Adam optimization in log-space. Returns best a found."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:

        for _ in range(max_iter):

            z, m_buf, v_buf, t = _adam_step(z, n, beta, m_buf, v_buf, t, lr)

                sc = _score_norm(a)

                    best_a = a.copy()

    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

"""
L-BFGS with log-sum-exp continuation for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses second-order (L-BFGS) optimization in log-softmax space with beta continuation,
combined with Adam for diversification, n-expansion, and multi-restart strategies.
Key improvement over previous: L-BFGS converges much faster than Adam for smooth
phases, enabling many more optimization restarts within the time budget.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    cm = float(c.max())
    if not np.isfinite(cm):
        return np.inf
    return float(2.0 * n * cm / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if not np.isfinite(cm):
        return np.inf
    return float(2.0 * n * cm)


def _compute_h(w, a, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return wc[n - 1:2 * n - 1]


def _lbfgs_obj(z, n, beta):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = c.max()
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    S = float(e.sum())
    if S < 1e-300 or not np.isfinite(S):
        return 2.0 * n * cm
    return float(2.0 * n / beta * (np.log(S) + cm / beta))


def _lbfgs_grad(z, n, beta):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = c.max()
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    S = float(e.sum())
    if S < 1e-300 or not np.isfinite(S):
        return np.zeros_like(z)
    w = e / S
    h = _compute_h(w, a, n)
    W = float(np.dot(w, c))
    grad = 4.0 * n * a / (beta * beta) * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "rick":
        t = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + 10.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    n_freqs = max(1, n // 8)
    for _ in range(n_freqs):
        freq = rng.integers(1, max(2, n // 2))
        amp = 0.02 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        t = np.arange(n)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_shift(a, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    shift = rng.integers(1, max(2, n // 8))
    z = np.roll(z, int(shift))
    z = z - z.mean()
    return _softmax(z)


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _lbfgs_phase(a_init, beta_schedule, max_iter=5000, gtol=1e-8):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    for beta in beta_schedule:
        try:
            result = minimize(
                _lbfgs_obj, z, args=(n, beta),
                jac=_lbfgs_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'gtol': gtol, 'maxls': 100,
                         'maxfun': max_iter * 5}
            )
            z_new = result.x
            if not np.all(np.isfinite(z_new)):
                continue
            a_new = _softmax(z_new)
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
        except Exception:
            continue
    return best_a, best_score


def _adam_phase(a_init, deadline, beta_schedule, max_iter=100000, lr=0.012):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        for _ in range(max_iter):
            if time.monotonic() > deadline:
                break
            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = c.max()
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            S = float(e.sum())
            if S < 1e-300 or not np.isfinite(S):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / S
            h = _compute_h(w, a, n)
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break
            t += 1
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            update = current_lr * mh / (np.sqrt(vh) + eps)
            max_update = 0.5
            update_norm = np.linalg.norm(update)
            if update_norm > max_update:
                update = update * (max_update / update_norm)
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            if t % 200 == 0:
                a_check = _softmax(z)
                sc = _score_norm(a_check)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_check.copy()
            if t % 5000 == 0:
                current_lr = max(0.002, current_lr * 0.95)
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

    if p is None:

    p = np.asarray(p, dtype=np.float64).ravel()
    p = np.clip(p, 0.0, 1000.0)

    if sc is None:

def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
