"""
Hybrid LP + Adam optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Novel LP-based direction finding combined with smooth-max Adam refinement.
The LP finds the optimal direction g that maximizes sum(g) subject to ||f*g||_inf <= M,
providing globally-optimal step directions that complement local gradient methods.
"""
import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def full_convolve(a, b):
    n_out = len(a) + len(b) - 1
    N = _next_pow2(n_out)
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:n_out]


def score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def score_norm(a):
    """Score with sum=1 normalization (2*n*max_c)."""
    n = len(a)
    c = full_convolve(a, a)
    return float(2.0 * n * c.max())


def softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def lp_solve_direction(f, M, max_time=30.0):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and ||conv(f,g)||_inf <= M.
    Returns the optimal g (normalized to sum=1) or None if infeasible/too slow.
    """
    n = len(f)
    n_con = 2 * n - 1
    t0 = time.monotonic()
    
    # Build constraint matrix A: A[k,j] = f[k-j] if 0<=k-j<n else 0
    rows = []
    cols = []
    vals = []
    for k in range(n_con):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    
    A = csr_matrix((vals, (rows, cols)), shape=(n_con, n))
    
    # Maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    b_ub = np.full(n_con, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': max_time - (time.monotonic() - t0),
                               'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def lp_iteration(f, step_size=0.3, max_time=30.0):
    """
    One LP-based iteration: find best direction g, blend with f.
    Returns (new_f, improved) where improved indicates score decreased.
    """
    n = len(f)
    c = full_convolve(f, f)
    M = float(c.max())
    
    g = lp_solve_direction(f, M, max_time=max_time)
    if g is None:
        return f, False
    
    # Rescale g to match sum of f
    s_f = f.sum()
    g = g * (s_f / g.sum())
    
    # Try different step sizes
    best_new = f
    best_sc = score_norm(f)
    for t in [step_size, step_size * 0.5, step_size * 0.25, step_size * 2.0]:
        candidate = (1.0 - t) * f + t * g
        candidate = np.maximum(candidate, 0.0)
        if candidate.sum() < 1e-15:
            continue
        candidate = candidate / candidate.sum()
        sc = score_norm(candidate)
        if sc < best_sc:
            best_sc = sc
            best_new = candidate
    
    improved = best_sc < best_sc  # always true, check against original
    return best_new, best_sc < score_norm(f)


def adam_smooth_max(a_init, deadline, beta_schedule, max_iter=200000, lr=0.01, clip_z=50.0):
    """Adam optimizer in log-space with log-sum-exp smooth max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_count = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        iter_count = 0
        while iter_count < max_iter:
            if time.monotonic() > deadline:
                break
            iter_count += 1
            t_count += 1
            
            a = softmax_fn(z)
            c = full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_e = float(e.sum())
            if s_e < 1e-300 or not np.isfinite(s_e):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_e
            
            a_rev = a[::-1]
            wc = full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t_count)
            vh = v_buf / (1.0 - b2 ** t_count)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            up_norm = np.linalg.norm(update)
            if up_norm > 0.5:
                update = update * (0.5 / up_norm)
            
            z = z - update
            z = np.clip(z, -clip_z, clip_z)
            z = z - z.mean()
            
            if t_count % 50 == 0:
                sc = score_norm(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()
            
            if iter_count % 5000 == 0:
                current_lr = max(0.002, current_lr * 0.98)
    
    a_final = softmax_fn(z)
    sc_final = score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine_taper":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - 2.0 * np.abs(t - 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return softmax_fn(z)


def interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid LP + Adam optimization to minimize 2*n*max(conv(a,a))/sum(a)^2.
    Uses LP-based direction finding for large structural moves,
    combined with smooth-max Adam for fine refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = score_fn(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")
    
    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = score_fn(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        update_best(a0, sc0, "parent")
    else:
        n_main = 2048
        a0 = make_profile(n_main, "uniform")
        print(f"No parent, using uniform n={n_main}")
    
    # --- Phase 0: Quick exploration (0-40s) ---
    phase0_end = min(deadline, t_start + 40.0)
    
    for kind in ["uniform", "sinusoidal", "cosine_taper", "gaussian", "bimodal"]:
        if time.monotonic() > phase0_end:
            break
        p = make_profile(n_main, kind)
        sc = score_fn(p)
        update_best(p, sc, f"init_{kind}")
        if best_seq is not None and best_score > 1.51:
            # Quick Adam on this
            c_t = full_convolve(p, p)
            max_ct = float(c_t.max())
            bs = [0.02 * max_ct, 0.01 * max_ct, 0.005 * max_ct]
            p_best, _ = adam_smooth_max(p, min(phase0_end, time.monotonic() + 5.0), bs, max_iter=5000, lr=0.02)
            update_best(p_best, None, f"quick_{kind}")
    
    # Small perturbations of parent
    if parent is not None:
        for sigma in [0.03, 0.06, 0.1]:
            if time.monotonic() > phase0_end:
                break
            p = perturb_log(parent, sigma, rng)
            update_best(p, None, f"perturb_{sigma}")
    
    # Quick Adam on parent
    if parent is not None:
        c_p = full_convolve(parent, parent)
        max_cp = float(c_p.max())
        bs_quick = [0.02 * max_cp, 0.01 * max_cp, 0.005 * max_cp, 0.002 * max_cp]
        t_budget = min(phase0_end, time.monotonic() + 20.0)
        p_best, _ = adam_smooth_max(parent, t_budget, bs_quick, max_iter=50000, lr=0.015)
        update_best(p_best, None, "quick_adam")
    
    print(f"Phase 0 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # --- Phase 1: LP-based optimization (40s - 500s) ---
    phase1_end = min(deadline, t_start + 500.0)
    
    if best_seq is not None and time.monotonic() < phase1_end:
        current = best_seq.copy()
        current = current / current.sum()
        sc_current = score_norm(current)
        
        n_lp_iters = 0
        no_improve_count = 0
        step_size = 0.4
        
        while time.monotonic() < phase1_end - 10:
            n_lp_iters += 1
            new_f, improved = lp_iteration(current, step_size=step_size, max_time=15.0)
            
            if improved:
                sc_new = score_norm(new_f)
                update_best(new_f, score_fn(new_f), f"lp_iter{n_lp_iters}")
                current = new_f
                no_improve_count = 0
                # Adapt step size
                if sc_new < sc_current * 0.99:
                    step_size = min(0.8, step_size * 1.1)
                else:
                    step_size = max(0.1, step_size * 0.9)
                sc_current = sc_new
            else:
                no_improve_count += 1
                step_size = max(0.05, step_size * 0.7)
                
                # If stuck, perturb and try again
                if no_improve_count >= 5:
                    current = perturb_log(current, 0.05 * no_improve_count * 0.5, rng)
                    sc_current = score_norm(current)
                    no_improve_count = 0
                    step_size = 0.3
                    
                if no_improve_count >= 20:
                    break
            
            if n_lp_iters % 10 == 0:
                print(f"  LP iter {n_lp_iters}: score_norm={sc_current:.10f}, step={step_size:.3f} t={time.monotonic()-t_start:.1f}s")
    
    print(f"Phase 1 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # --- Phase 2: Adam refinement (500s - 900s) ---
    phase2_end = min(deadline, t_start + 900.0)
    
    if best_seq is not None and time.monotonic() < phase2_end:
        c_best = full_convolve(best_seq, best_seq)
        max_cb = float(c_best.max())
        beta_schedule = [
            0.01 * max_cb, 0.005 * max_cb, 0.002 * max_cb,
            0.001 * max_cb, 0.0005 * max_cb, 0.0002 * max_cb
        ]
        t_budget = min(phase2_end, time.monotonic() + 350.0)
        p_best, _ = adam_smooth_max(best_seq, t_budget, beta_schedule, max_iter=300000, lr=0.01)
        update_best(p_best, None, "adam_refine")
        
        # Second Adam pass with even lower beta
        if time.monotonic() < phase2_end - 30:
            c_best2 = full_convolve(best_seq, best_seq)
            max_cb2 = float(c_best2.max())
            beta_ultra = [0.0002 * max_cb2, 0.0001 * max_cb2, 0.00005 * max_cb2]
            p_best2, _ = adam_smooth_max(best_seq, min(phase2_end, time.monotonic() + 25.0), beta_ultra, max_iter=100000, lr=0.005)
            update_best(p_best2, None, "adam_ultra")
    
    print(f"Phase 2 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # --- Phase 3: More LP iterations from improved point (900s - 1200s) ---
    phase3_end = min(deadline, t_start + 1200.0)
    
    if best_seq is not None and time.monotonic() < phase3_end:
        current = best_seq.copy()
        current = current / current.sum()
        sc_current = score_norm(current)
        step_size = 0.3
        no_improve = 0
        n_iter = 0
        
        while time.monotonic() < phase3_end - 10 and no_improve < 30:
            n_iter += 1
            new_f, improved = lp_iteration(current, step_size=step_size, max_time=15.0)
            if improved:
                sc_new = score_norm(new_f)
                update_best(new_f, score_fn(new_f), f"lp2_iter{n_iter}")
                current = new_f
                sc_current = sc_new
                no_improve = 0
            else:
                no_improve += 1
                step_size = max(0.05, step_size * 0.85)
                if no_improve % 5 == 0:
                    current = perturb_log(current, 0.03, rng)
                    sc_current = score_norm(current)
    
    # Interleave Adam between LP
    if best_seq is not None and time.monotonic() < phase3_end - 20:
        c_b = full_convolve(best_seq, best_seq)
        max_c = float(c_b.max())
        bs = [0.001 * max_c, 0.0005 * max_c, 0.0002 * max_c]
        p_best, _ = adam_smooth_max(best_seq, min(phase3_end, time.monotonic() + 20.0), bs, max_iter=50000, lr=0.008)
        update_best(p_best, None, "mid_adam")
    
    print(f"Phase 3 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # --- Phase 4: Multi-restart with diverse strategies (1200s - 1550s) ---
    phase4_end = min(deadline, t_start + 1550.0)
    
    if best_seq is not None and time.monotonic() < phase4_end:
        n_restarts = 6
        per_restart = (phase4_end - time.monotonic() - 30) / max(1, n_restarts)
        
        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 20:
                break
            
            # Different perturbation strategies
            if i == 0:
                p_start = perturb_log(best_seq, 0.04, rng)
            elif i == 1:
                p_start = perturb_log(best_seq, 0.08, rng)
            elif i == 2:
                # Fourier perturbation
                n = len(best_seq)
                z = np.log(np.maximum(best_seq, 1e-15))
                freq = rng.integers(1, max(2, n // 8))
                amp = 0.05 * rng.uniform(0.5, 2.0)
                t_arr = np.arange(n)
                z = z + amp * np.cos(2.0 * np.pi * freq * t_arr / n + rng.uniform(0, 2 * np.pi))
                z = z - z.mean()
                p_start = softmax_fn(z)
            elif i == 3:
                # Cyclical shift
                shift = rng.integers(1, max(2, len(best_seq) // 16))
                p_start = np.roll(best_seq, shift)
                p_start = p_start / p_start.sum()
            elif i == 4:
                # Blend with a structured profile
                prof = make_profile(len(best_seq), "cosine_taper")
                p_start = 0.7 * best_seq + 0.3 * prof
                p_start = np.maximum(p_start, 1e-10) / np.maximum(p_start, 1e-10).sum()
            else:
                p_start = perturb_log(best_seq, 0.02, rng)
            
            # Quick LP + Adam combo
            per_budget = min(per_restart, 50.0)
            
            # LP step
            if time.monotonic() < t_start + budget_s - 50:
                p_lp, imp = lp_iteration(p_start, step_size=0.3, max_time=10.0)
                if imp:
                    p_start = p_lp
            
            # Adam refinement
            c_s = full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs_r = [0.01 * max_cs, 0.005 * max_cs, 0.002 * max_cs, 0.001 * max_cs]
            t_rem = min(phase4_end - 20, time.monotonic() + per_budget)
            p_best, _ = adam_smooth_max(p_start, t_rem, bs_r, max_iter=80000, lr=0.012)
            update_best(p_best, None, f"restart{i}")
    
    print(f"Phase 4 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # --- Phase 5: Try larger n (1550s - 1700s) ---
    phase5_end = min(deadline, t_start + 1700.0)
    
    if best_seq is not None and len(best_seq) < 4096 and time.monotonic() < phase5_end - 30:
        target_n = min(4096, len(best_seq) * 2)
        p_interp = interp_profile(best_seq, target_n)
        sc_interp = score_fn(p_interp)
        update_best(p_interp, sc_interp, f"interp_n{target_n}")
        
        if time.monotonic() < phase5_end - 40:
            c_int = full_convolve(p_interp, p_interp)
            max_ci = float(c_int.max())
            bs_int = [0.01 * max_ci, 0.005 * max_ci, 0.002 * max_ci, 0.001 * max_ci, 0.0005 * max_ci]
            t_rem = min(phase5_end, time.monotonic() + 120.0)
            p_best, _ = adam_smooth_max(p_interp, t_rem, bs_int, max_iter=150000, lr=0.008)
            update_best(p_best, None, f"opt_n{target_n}")
    
    # Also try n=8192 if time and n allow
    if best_seq is not None and len(best_seq) < 8192 and time.monotonic() < phase5_end - 30:
        target_n2 = 8192
        p_interp2 = interp_profile(best_seq, target_n2)
        sc2 = score_fn(p_interp2)
        update_best(p_interp2, sc2, f"interp_n{target_n2}")
        
        if time.monotonic() < phase5_end - 20:
            c_int2 = full_convolve(p_interp2, p_interp2)
            max_ci2 = float(c_int2.max())
            bs_int2 = [0.005 * max_ci2, 0.002 * max_ci2, 0.001 * max_ci2]
            t_rem2 = min(phase5_end, time.monotonic() + 60.0)
            p_best2, _ = adam_smooth_max(p_interp2, t_rem2, bs_int2, max_iter=80000, lr=0.006)
            update_best(p_best2, None, f"opt_n{target_n2}")
    
    print(f"Phase 5 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # --- Phase 6: Final fine polish (1700s - deadline) ---
    if best_seq is not None and time.monotonic() < deadline - 5:
        # One more LP pass
        if time.monotonic() < deadline - 30:
            current = best_seq / best_seq.sum()
            new_f, imp = lp_iteration(current, step_size=0.2, max_time=10.0)
            if imp:
                update_best(new_f, None, "final_lp")
        
        # Ultra-fine Adam
        c_final = full_convolve(best_seq, best_seq)
        max_cf = float(c_final.max())
        beta_fin = [0.0001 * max_cf, 0.00005 * max_cf, 0.00002 * max_cf]
        t_rem = min(deadline, time.monotonic() + 30.0)
        if t_rem > time.monotonic() + 5:
            p_best, _ = adam_smooth_max(best_seq, t_rem, beta_fin, max_iter=100000, lr=0.003)
            update_best(p_best, None, "final_polish")
    
    print(f"Phase 6 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = make_profile(256, "uniform")
        sc_final = score_fn(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        
        return [float(x) for x in final]
    
    # Fallback
    fallback = make_profile(1024, "uniform")
    print(f"FALLBACK: score={score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
