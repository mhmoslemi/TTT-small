    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _gradient_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _objective_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_obj(z, beta, n):
    return _objective_smooth(z, beta)


def _lbfgs_grad(z, beta, n):
    return _gradient_smooth(z, beta)


def _run_lbfgs(a_init, deadline, max_wall_time=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    if max_wall_time is None:
        max_wall_time = min(120.0, max(1.0, deadline - time.monotonic()))
    betas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max_c0 for b in betas]
    per_level = max_wall_time / len(betas)
    for beta in betas:
        if time.monotonic() >= deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level)
        try:
            result = minimize(
                _lbfgs_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_grad,
                options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxls': 20, 'maxfun': 800}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x.copy()
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new
        except Exception:
            pass
    return best_a, best_score


def _nesterov_optimize(a_init, deadline, beta_fracs, lr_init=0.02, mom_init=0.9):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    lr = lr_init
    mom = mom_init
    t = 0
    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-12, beta_frac * max_c)
        while time.monotonic() < deadline:
            t += 1
            grad_y = _gradient_smooth(y, beta)
            if not np.all(np.isfinite(grad_y)):
                grad_y = _gradient_smooth(z, beta)
            z_new = z - lr * grad_y
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -50, 50)
            y_new = z_new + mom * (z_new - z)
            y_new = y_new - y_new.mean()
            z = z_new
            y = y_new
            if t % 30 == 0:
                a_curr = _softmax_fn(z)
                sc = _score_norm_fn(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()
            if t % 200 == 0:
                mom = min(0.999, mom + 0.0003)
        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        max_c = float(c.max())
        lr = max(0.001, lr * 0.6)
    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _adam_optimize(a_init, deadline, lr=0.05, beta_frac=0.01):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, beta_frac * max_c)
    while time.monotonic() < deadline:
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            continue
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        if t % 100 == 0:
            a = _softmax_fn(z)
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
            beta = max(1e-12, beta * 0.85)
            c_new = _full_convolve(a, a)
            max_c = float(c_new.max())
    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lp_direction(a, M, n_max_vars=None):
    """Find LP direction: max sum(g) s.t. convolve(a,g) <= M, g >= 0.
    Uses reduced constraint set (top active lags only) for speed."""
    n = len(a)
    c_ref = _full_convolve(a, a)
    # Get active lags (top 2% or at least 20)
    n_lags = len(c_ref)
    n_active = max(20, n_lags // 50)
    top_idx = np.argsort(c_ref)[-n_active:]
    # Build constraint matrix for active lags only
    # A[k, i] = a[i] if 0 <= k-i < n
    rows = []
    for k in top_idx:
        row = np.zeros(n)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            row[i] = a[i]
        rows.append(row)
    A = np.array(rows)
    b_vec = np.full(n_active, M)
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            g = res.x
            if np.all(g >= -1e-10):
                g = np.maximum(g, 0)
                s = g.sum()
                if s > 1e-10:
                    return g / s
    except Exception:
        pass
    return None


def _lp_guided_optimize(a_init, deadline, n_lp_solves=10):
    """Use LP direction finding periodically, with Nesterov between LP solves."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    lr = 0.01
    mom = 0.9
    t = 0
    time_per_lp = (deadline - time.monotonic()) / max(1, n_lp_solves)
    for lp_iter in range(n_lp_solves):
        if time.monotonic() >= deadline:
            break
        a = _softmax_fn(z)
        s = a.sum()
        M = float(_full_convolve(a, a).max())
        # Get LP direction
        g = _lp_direction(a, M)
        lp_end = min(deadline, time.monotonic() + time_per_lp)
        if g is not None:
            # Blend LP direction with gradient
            target_z = np.log(np.maximum(g, 1e-15))
            target_z = target_z - target_z.mean()
            # Interpolate towards LP direction
            z = 0.7 * z + 0.3 * target_z
            z = z - z.mean()
            y = z.copy()
        # Nesterov between LP calls
        beta = 0.005 * M
        while time.monotonic() < lp_end:
            t += 1
            grad = _gradient_smooth(y, beta)
            if not np.all(np.isfinite(grad)):
                break
            z_new = z - lr * grad
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -50, 50)
            y_new = z_new + mom * (z_new - z)
            y_new = y_new - y_new.mean()
            z, y = z_new, y_new
            if t % 20 == 0:
                a_curr = _softmax_fn(z)
                sc = _score_norm_fn(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()
            if t % 50 == 0:
                mom = min(0.995, mom + 0.001)
        lr = max(0.002, lr * 0.8)
    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _active_set_subgradient(a_init, deadline, n_active_frac=0.05):
    """Subgradient method on the active set of near-maximum lags."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    lr = 0.003
    momentum = np.zeros(n)
    mom_coeff = 0.85
    n_active = max(10, int(n_active_frac * (2 * n - 1)))
    while time.monotonic() < deadline:
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        # Active set: top n_active lags
        top_idx = np.argsort(c)[-n_active:]
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / n_active
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        momentum = mom_coeff * momentum - lr * grad
        z = z + momentum
        z = z - z.mean()
        z = np.clip(z, -50, 50)
        if int(time.monotonic() * 10) % 5 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            else:
                lr = max(0.0001, lr * 0.95)
    return best_a, best_score


def _make_profile(n, profile_type, **kwargs):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if profile_type == 'linear':
        slope = kwargs.get('slope', 1.5)
        a = 1.0 + slope * (t - 0.5)
    elif profile_type == 'quadratic':
        a1 = kwargs.get('a1', 1.0)
        a2 = kwargs.get('a2', 0.2)
        a = 1.0 + a1 * u + a2 * u ** 2
    elif profile_type == 'cubic':
        a1 = kwargs.get('a1', 0.8)
        a3 = kwargs.get('a3', -0.4)
        a = 1.0 + a1 * u + a3 * u ** 3
    elif profile_type == 'power':
        alpha = kwargs.get('alpha', 1.5)
        a = t ** alpha
    elif profile_type == 'exp':
        rate = kwargs.get('rate', 2.0)
        a = np.exp(-rate * (t - 0.3) ** 2)
    elif profile_type == 'sin':
        freq = kwargs.get('freq', 1.0)
        amp = kwargs.get('amp', 0.3)
        a = 1.0 + amp * np.sin(freq * np.pi * t)
    elif profile_type == 'asym_power':
        alpha = kwargs.get('alpha', 2.0)
        a = t ** alpha * (1 - t) ** 0.3
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _barzilai_borwein_step(z, z_prev, grad, grad_prev, lr_old):
    """BB step size for Nesterov-like methods."""
    s = z - z_prev
    y = grad - grad_prev
    ss = float(np.dot(s, s))
    ys = float(np.dot(y, s))
    if ss < 1e-15 or abs(ys) < 1e-15:
        return lr_old
    if ys > 0:
        lr_new = ss / ys
    else:
        yy = float(np.dot(y, y))
        if yy < 1e-15:
            return lr_old
        lr_new = abs(ys) / yy
    return float(np.clip(lr_new, 0.001, 0.5))


def _advanced_nesterov(a_init, deadline, beta_schedule, lr_init=0.015):
    """Nesterov with BB step size and adaptive beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()
    z_prev = z.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    lr = lr_init
    mom = 0.9
    t = 0
    grad_prev = np.zeros(n)
    for beta in beta_schedule:
        if time.monotonic() >= deadline:
            break
        beta = float(beta)
        while time.monotonic() < deadline:
            t += 1
            grad_y = _gradient_smooth(y, beta)
            if not np.all(np.isfinite(grad_y)):
                grad_y = np.zeros(n)
            # BB step size
            if t > 2:
                lr = _barzilai_borwein_step(z, z_prev, grad_y, grad_prev, lr)
            z_prev = z.copy()
            z_new = z - lr * grad_y
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -50, 50)
            y_new = z_new + mom * (z_new - z)
            y_new = y_new - y_new.mean()
            z, y = z_new, y_new
            grad_prev = grad_y
            if t % 25 == 0:
                a_curr = _softmax_fn(z)
                sc = _score_norm_fn(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()
            if t % 100 == 0:
                mom = min(0.999, mom + 0.0002)
    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coordinate_descent_polish(a_init, deadline, n_rounds=50):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_fn(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    for rnd in range(n_rounds):
        if time.monotonic() >= deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() >= deadline:
                break
            # Try small perturbations of a[i]
            old_val = a[i]
            for factor in [0.9, 0.95, 1.05, 1.1]:
                a[i] = old_val * factor
                a[i] = max(0, a[i])
                s = a.sum()
                if s < 1e-10:
                    a[i] = old_val
                    continue
                a_norm = a / s
                sc = _score_fn(a_norm)
                if np.isfinite(sc) and sc < best_score - 1e-10:
                    best_score = sc
                    best_a = a_norm.copy()
                    improved = True
                    break
            if not improved:
                a[i] = old_val
        # Renormalize
        s = a.sum()
        if s > 0:
            a = a / s
        # Update best
        sc = _score_fn(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: LP-guided search + Nesterov continuation +
    active-set polishing + multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:7.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_parent}, normalized score={sc_parent:.10f}, actual score={_score_fn(parent):.10f}")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, 'linear', slope=1.5)
        print(f"No parent, using linear profile n={n_parent}")

    # --- Phase 0: Multi-n, multi-profile exploration (60s) ---
    phase0_end = min(deadline, t_start + 60.0)
    candidates = []
    update_best(parent, _score_norm_fn(parent), "parent")

    n_values = [2048, 3072, 4096, 5120, 6144]
    profile_configs = [
        ('linear', {'slope': 1.0}),
        ('linear', {'slope': 1.5}),
        ('linear', {'slope': 2.0}),
        ('quadratic', {'a1': 0.8, 'a2': 0.3}),
        ('cubic', {'a1': 0.6, 'a3': -0.3}),
        ('power', {'alpha': 1.5}),
        ('power', {'alpha': 2.0}),
        ('asym_power', {'alpha': 2.0}),
    ]

    for n_val in n_values:
        if time.monotonic() > phase0_end:
            break
        for ptype, pkw in profile_configs:
            if time.monotonic() > phase0_end:
                break
            p = _make_profile(n_val, ptype, **pkw)
            sc = _score_norm_fn(p)
            candidates.append((sc, p.copy(), n_val, f"{ptype}_n{n_val}"))
            if sc < best_score:
                update_best(p, sc, f"init_{ptype}_n{n_val}")

    # Also try interpolating parent to other n values
    if parent is not None:
        for n_val in [3072, 4096, 5120]:
            if time.monotonic() > phase0_end:
                break
            p_interp = _interp_profile(parent, n_val)
            sc = _score_norm_fn(p_interp)
            candidates.append((sc, p_interp, n_val, f"interp_n{n_val}"))
            if sc < best_score:
                update_best(p_interp, sc, f"interp_n{n_val}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f} (n={len(best_seq) if best_seq is not None else 0})")

    # --- Phase 1: LP-guided optimization on best (200s) ---
    phase1_end = min(deadline, t_start + 260.0)
    if best_seq is not None and time.monotonic() < phase1_end:
        per_time = phase1_end - time.monotonic()
        if per_time > 60:
            p_best, sc_best = _lp_guided_optimize(best_seq, min(phase1_end - 5, time.monotonic() + per_time),
                                                   n_lp_solves=5)
            update_best(p_best, sc_best, "lp_phase1")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Advanced Nesterov with BB step (400s) ---
    phase2_end = min(deadline, t_start + 660.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        per_time = phase2_end - time.monotonic()
        if per_time > 30:
            c0 = _full_convolve(best_seq, best_seq)
            max_c0 = float(c0.max())
            beta_schedule = [max_c0 * f for f in
                             [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]]
            p_best, sc_best = _advanced_nesterov(best_seq, min(phase2_end - 5, time.monotonic() + per_time),
                                                  beta_schedule, lr_init=0.01)
            update_best(p_best, sc_best, "adv_nesterov")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Adam with aggressive beta decay (350s) ---
    phase3_end = min(deadline, t_start + 1010.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        per_time = phase3_end - time.monotonic()
        if per_time > 30:
            p_best, sc_best = _adam_optimize(best_seq, min(phase3_end - 5, time.monotonic() + per_time),
                                              lr=0.03, beta_frac=0.005)
            update_best(p_best, sc_best, "adam_phase3")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Nesterov restarts from perturbed best (250s) ---
    phase4_end = min(deadline, t_start + 1260.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        n_best = len(best_seq)
        n_restarts = 5
        per_restart = (phase4_end - time.monotonic()) / max(1, n_restarts)
        c_b = _full_convolve(best_seq, best_seq)
        max_c_b = float(c_b.max())

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 10:
                break
            noise_levels = [0.03, 0.05, 0.08, 0.12, 0.15]
            noise_level = noise_levels[i % len(noise_levels)]
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_level * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)
            per_dl = min(phase4_end - 5, time.monotonic() + per_restart * 0.9)
            beta_fracs = [max_c_b * f for f in [0.003, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]]
            p_best, sc_best = _nesterov_optimize(p_start, per_dl, beta_fracs / max_c_b,
                                                  lr_init=0.008, mom_init=0.85)
            update_best(p_best, sc_best, f"restart{i}_n{noise_level}")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Active-set subgradient polish (200s) ---
    phase5_end = min(deadline, t_start + 1460.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        per_time = phase5_end - time.monotonic()
        if per_time > 30:
            p_best, sc_best = _active_set_subgradient(best_seq, min(phase5_end - 5, time.monotonic() + per_time),
                                                       n_active_frac=0.03)
            update_best(p_best, sc_best, "active_set")

    # Also try with different active set size
    if best_seq is not None and time.monotonic() < phase5_end - 20:
        per_time = phase5_end - time.monotonic() - 5
        if per_time > 20:
            p_best, sc_best = _active_set_subgradient(best_seq, min(phase5_end - 2, time.monotonic() + per_time),
                                                       n_active_frac=0.01)
            update_best(p_best, sc_best, "active_set_tight")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Try different n values (200s) ---
    phase6_end = min(deadline, t_start + 1660.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        n_curr = len(best_seq)
        # Try nearby n values
        for target_n in [n_curr - 512, n_curr + 512, n_curr // 2, n_curr * 2]:
            if time.monotonic() > phase6_end - 10:
                break
            if target_n < 512 or target_n > 16384:
                continue
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            label = f"interp_n{target_n}"
            if sc_interp < best_score:
                update_best(p_interp, sc_interp, label)
            # Quick optimization at this n
            if time.monotonic() < phase6_end - 20:
                per_time = min(40.0, phase6_end - time.monotonic() - 10)
                if per_time > 10:
                    c_i = _full_convolve(p_interp, p_interp)
                    max_ci = float(c_i.max())
                    betas = [max_ci * f for f in [0.005, 0.002, 0.001, 0.0005]]
                    p_best, sc_best = _nesterov_optimize(p_interp, min(phase6_end - 5, time.monotonic() + per_time),
                                                          betas / max_ci, lr_init=0.01, mom_init=0.9)
                    update_best(p_best, sc_best, f"opt_n{target_n}")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final fine polish (remaining time) ---
    phase7_end = deadline
    if best_seq is not None and time.monotonic() < phase7_end:
        remaining = phase7_end - time.monotonic()
        if remaining > 30:
            c_f = _full_convolve(best_seq, best_seq)
            max_cf = float(c_f.max())
            # Very fine Nesterov
            per_time = min(50.0, remaining - 20)
            if per_time > 10:
                betas_fine = [max_cf * f for f in [1e-5, 5e-6, 2e-6, 1e-6]]
                p_best, sc_best = _nesterov_optimize(best_seq, min(phase7_end - 30, time.monotonic() + per_time),
                                                      betas_fine / max_cf, lr_init=0.003, mom_init=0.97)
                update_best(p_best, sc_best, "final_nesterov_fine")

            # Fine Adam
            if time.monotonic() < phase7_end - 15:
                per_time2 = min(30.0, phase7_end - time.monotonic() - 5)
                if per_time2 > 5:
                    p_best, sc_best = _adam_optimize(best_seq, min(phase7_end - 3, time.monotonic() + per_time2),
                                                      lr=0.003, beta_frac=0.0005)
                    update_best(p_best, sc_best, "final_adam_fine")

    print(f"Phase 7 done: best={best_score:.10f}")

    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profile(4096, 'linear', slope=1.5)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned non-finite, using internal: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profile(4096, 'linear', slope=1.5)
    print(f"FALLBACK: n=4096, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
