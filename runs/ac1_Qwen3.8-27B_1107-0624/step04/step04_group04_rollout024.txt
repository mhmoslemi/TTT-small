"""
Subgradient optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses exact subgradient with Nesterov momentum, adaptive active set,
QP-based weight optimization, and KKT balancing for final precision.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 2
    while k < n:
        k <<= 1
    return k


def _full_convolve(a, b):
    n_out = len(a) + len(b) - 1
    N = _next_pow2(n_out)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n_out]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return 2.0 * n * float(np.max(c)) / (s * s)


def _max_c_val(a):
    c = _full_convolve(a, a)
    return float(np.max(c))


def _compute_h(a, active_idx, weights):
    """Compute h[i] = sum_k w_k * a[k-i] for valid k-i."""
    n = len(a)
    h = np.zeros(n)
    for k, w in zip(active_idx, weights):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            h[lo:hi + 1] += w * a[k - hi:k - lo + 1][::-1]
    return h


def _optimize_weights(a, active_idx):
    """Find weights w>=0, sum w=1 minimizing variance of h = sum w_k * v_k."""
    n = len(a)
    m = len(active_idx)
    if m == 1:
        return np.ones(1)
    if m <= 2:
        return np.ones(m) / m

    # Build V matrix: V[i, j] = a[k_j - i] if valid
    # For efficiency, only build on a subset of rows if n is large
    V = np.zeros((n, m))
    for j, k in enumerate(active_idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            V[lo:hi + 1, j] = a[k - hi:k - lo + 1][::-1]

    # Minimize ||V w - mean(Vw)||^2 subject to w>=0, sum w=1
    # Equivalent to minimizing w^T V^T V w - 2 * (1^T V w / n) * (1^T V w / n) ... 
    # Simpler: minimize w^T (V^T V) w subject to constraints, 
    # since the mean term is fixed given the sum constraint.
    # Actually, minimize sum_i (sum_j V[i,j] w_j - lambda)^2 where lambda = (a^T V w)
    # This is complex. Use a simpler approach: minimize ||V w - c*1||^2
    # where c is chosen optimally. The optimal c = (sum a_i * (Vw)_i) / (sum a_i)
    # For now, just minimize w^T V^T D V w where D = diag(a) (weighted LSQ)
    
    # Use projected gradient on the simplex for weight optimization
    # Objective: f(w) = ||V^T D V w - (1^T D V w / n) * 1||^2 ... too complex
    # Simpler: just minimize the variance of h weighted by a
    # h = V w, target: h[i] = lambda for all i
    # lambda = sum_i a[i] h[i] = a^T V w
    # Objective: sum_i a[i] (h[i] - lambda)^2 = (Vw)^T D V (Vw) - ((Vw)^T D 1)^2
    # = w^T V^T D V w - (w^T V^T D 1)^2
    
    D_V = V.T * a  # (m x n) * (n,) = (m x n)
    A_mat = V.T @ (a[:, None] * V)  # m x m: V^T D V
    b_vec = V.T @ (a * np.ones(n))  # m: V^T D 1 ... wait
    # Actually b_vec[j] = sum_i a[i] * V[i,j] = sum_i a[i] * a[k_j - i] = c[k_j]
    # So b_vec is just c[active_idx]!
    
    # Objective: w^T A_mat w - (b_vec^T w)^2
    # This is not a standard QP because of the -(b^T w)^2 term.
    # For simplicity, use uniform weights or a simple heuristic.
    
    # Heuristic: weight inversely proportional to the "spread" contribution
    # Each v_k has a certain variance; weights should be higher for those with lower variance
    variances = np.sum(a * V ** 2, axis=0) - (np.sum(a * V, axis=0)) ** 2
    variances = np.maximum(variances, 1e-30)
    w = 1.0 / variances
    w = w / w.sum()
    return w


def _subgradient_momentum(a_init, time_budget, top_k_start=100, top_k_end=5, eta0=0.05, decay=0.998, momentum=0.9):
    """
    Subgradient descent with Nesterov momentum and adaptive active set.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    a = _softmax(z)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score(a)

    z_prev = z.copy()
    eta = eta0
    start_time = time.monotonic()
    deadline = start_time + time_budget
    total_iters = max(1, int((time_budget - 1) / 0.01))

    it = 0
    no_improve = 0
    while time.monotonic() < deadline:
        it += 1
        c = _full_convolve(a, a)
        max_c = float(c.max())

        # Adaptive top-k: start broad, narrow over time
        frac = min(1.0, it / max(1, total_iters // 2))
        top_k = max(top_k_end, int(top_k_start * (1 - frac) + top_k_end * frac))
        top_k = min(top_k, len(c))

        # Find active set: top-k lags
        if top_k < len(c):
            active_idx = np.argsort(c)[-top_k:]
            active_sorted = sorted(map(int, active_idx))
        else:
            active_sorted = list(range(len(c)))

        m = len(active_sorted)
        # Get weights (simple heuristic)
        if m <= 3:
            w = np.ones(m) / m
        else:
            # Weight by proximity to max: closer to max gets more weight
            c_active = c[active_sorted]
            c_max = c_active.max()
            w = np.maximum(0, c_active - c_max * 0.9)
            if w.sum() < 1e-15:
                w = np.ones(m) / m
            else:
                w = w / w.sum()

        h = _compute_h(a, active_sorted, w)
        lam = float(np.dot(h, a))

        # Nesterov momentum update
        grad_z = h - lam
        z_new = z + momentum * (z - z_prev) - eta * grad_z
        z_new = np.clip(z_new, -50, 50)
        z_new = z_new - z_new.mean()
        a_new = _softmax(z_new)
        a_new = a_new / a_new.sum()

        sc = _score(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Backtracking: reduce step
            if no_improve > 20:
                eta *= 0.7
                no_improve = 0

        z_prev = z
        z = z_new
        a = a_new
        eta *= decay

        if it % 2000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    [SG] it={it}, score={best_score:.10f}, eta={eta:.6f}, top_k={top_k}, elapsed={elapsed:.1f}s")

    return best_a, best_score


def _kkt_balance_v2(a_init, time_budget, max_iter=50000):
    """
    Improved KKT balancing with weight optimization and adaptive active set.
    At KKT: weighted sum of active gradients should be constant on support.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    a = _softmax(z)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score(a)

    start_time = time.monotonic()
    deadline = start_time + time_budget

    # Start with a broad active set, narrow over time
    it = 0
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        max_c = float(c.max())

        # Progressive active set narrowing
        progress = min(1.0, it / (max_iter * 0.5))
        if progress < 0.3:
            thresh_frac = 0.95
        elif progress < 0.6:
            thresh_frac = 0.98
        elif progress < 0.8:
            thresh_frac = 0.995
        else:
            thresh_frac = 0.999

        threshold = max_c * thresh_frac
        active = np.where(c >= threshold)[0]
        active_sorted = sorted(map(int, active))

        if len(active_sorted) == 0:
            active_sorted = [int(np.argmax(c))]

        m = len(active_sorted)

        # Compute weights: try to make h as flat as possible
        if m <= 5:
            w = np.ones(m) / m
        else:
            # Use the c-values as a guide: lags with c closer to max get higher weight
            c_active = np.array([c[k] for k in active_sorted])
            spread = c_active.max() - c_active
            w = spread ** 2 + 1e-10
            w = w / w.sum()

        h = _compute_h(a, active_sorted, w)
        lam = float(np.dot(h, a))

        # Compute the "flatness" of h
        h_spread = float(np.max(np.abs(h - lam)))

        # Step size: proportional to the spread
        step = 0.01 * h_spread / (max_c + 1e-15)
        step = min(step, 0.03)

        # Update in logit space
        z_new = z - step * (h - lam)
        z_new = np.clip(z_new, -60, 60)
        z_new = z_new - z_new.mean()
        a_new = _softmax(z_new)
        a_new = a_new / a_new.sum()

        sc = _score(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            z = z_new
        else:
            # Reduce step and retry
            step *= 0.5
            z_new = z - step * (h - lam)
            z_new = np.clip(z_new, -60, 60)
            z_new = z_new - z_new.mean()
            a_new = _softmax(z_new)
            a_new = a_new / a_new.sum()
            sc = _score(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                z = z_new

        if it % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    [KKT] it={it}, score={best_score:.10f}, active={m}, spread={h_spread:.2e}, elapsed={elapsed:.1f}s")

    return best_a, best_score
