def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm_fn(a):
    """Score with sum(a)=1 normalization: 2n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_fn(a):
    """Score as in evaluator: 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

"""
SLP (Successive Linear Programming) with active-set management combined
with L-BFGS log-sum-exp polishing for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Key insight: the optimal solution is asymmetric, with a ramp-like profile that
minimizes the center convolution peak while controlling off-center peaks.
"""
import numpy as np
from scipy.optimize import linprog, minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    lse = cm / beta + np.log(np.sum(np.exp(vals - vals.max())))
    return 2.0 * n * lse


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    w = np.exp(vals - vals.max())
    w = w / w.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    h = _full_convolve(w, a_rev)[n-1:2*n-1]
    return 4.0 * n * a * (h - W)


def _slp_solve(a, tol_active=0.005, max_active=8):
    """One SLP iteration using LP."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    # Active set
    threshold = c_max * (1.0 - tol_active)
    active = np.where(c >= threshold)[0]
    if len(active) > max_active:
        active = active[np.argsort(c[active])[::-1][:max_active]]
    
    m = len(active)
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Constraints: 2*sum_i a[k-i]*a_new[i] - t <= c[k] for k in active
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    for j, k in enumerate(active):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        # a[k-i] for i in [i_lo, i_hi]
        # When i=i_lo: k-i = k-i_lo
        # When i=i_hi: k-i = k-i_hi
       coeffs = a[k - i_hi : k - i_lo + 1][::-1] * 2.0
        A_ub[j, i_lo:i_hi+1] = coeffs
        A_ub[j, n] = -1.0
        b_ub[j] = c[k]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(-np.inf, np.inf)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success and res.x is not None:
        a_new = np.maximum(res.x[:n], 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        else:
            return None
        return a_new
    return None


def _nesterov_gd(a_init, deadline, lr=0.1, momentum=0.9):
    """Nesterov-accelerated subgradient descent on max(conv(a,a))."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()
    v = np.zeros(n)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    t = 0
    
    while time.monotonic() < deadline:
        t += 1
        a = _softmax(y)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))
        
        # Subgradient of max(c) w.r.t. a
        i_lo = max(0, k_star - n + 1)
        i_hi = min(n - 1, k_star)
        grad_a = np.zeros(n)
        grad_a[i_lo:i_hi+1] = 2.0 * a[k_star - i_hi : k_star - i_lo + 1][::-1]
        
        # Convert to z-gradient
        grad_z = a * (grad_a - np.dot(grad_a, a))
        
        # Nesterov update
        lr_t = lr / (1.0 + 0.01 * t)
        v = momentum * v + lr_t * grad_z
        z = z - v
        y = z + momentum * (z - y)  # Actually Nesterov: y = z + (t/(t+3)) * (z - y_prev)
        # Simpler: just use heavy ball
        z = z - z.mean()
        z = np.clip(z, -50, 50)
        
        a = _softmax(z)
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score


def _make_profiles(n):
