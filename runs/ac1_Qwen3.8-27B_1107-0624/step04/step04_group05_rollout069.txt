def _next_pow2(n):

def _build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_best_response(a, M):
    n = len(a)
    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None

def _make_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    return np.maximum(a, 1e-12) / np.maximum(a, 1e-12).sum()


def _make_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    return np.maximum(a, 1e-12) / np.maximum(a, 1e-12).sum()


def _make_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    return a / a.sum()


def _make_quad(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + a1 * t + a2 * t * t
    return np.maximum(a, 1e-12) / np.maximum(a, 1e-12).sum()


def _make_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    return np.maximum(a, 1e-12) / np.maximum(a, 1e-12).sum()


def _make_exp(n, lam):
    t = np.arange(n)
    a = np.exp(-lam * t / n)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))
