import numpy as np

from scipy.sparse import csr_matrix

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 1e-15)
    s = float(a.sum())
    if s < 1e-30:
        return None
    return a / s


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-10:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(np.clip(x, -50, 0))
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _conv_grad(w, p):
    """Gradient of sum_k w[k] * conv(p,p)[k] w.r.t. p.
    conv(p,p)[k] = sum_i p[i]*p[k-i], so d/dp[j] = sum_k w[k] * (p[k-j] + p[j-k]) (with bounds)
    This equals 2 * (w * p_reversed)[shifted] for the full derivative."""
    n = len(p)
    # d/dp[j] of sum_k w_k * conv_k = sum_k w_k * (p[k-j] + p[j-k]) for valid indices
    # = 2 * sum_k w_k * p[k-j] for the symmetric part
    # Actually: conv_k = sum_i p_i * p_{k-i}
    # d(conv_k)/dp_j = p_{k-j} + p_{j-k} (when both indices valid)
    # So grad_j = sum_k w_k * (p_{k-j} + p_{j-k})
    # By symmetry of the sum, this = 2 * sum_k w_k * p_{k-j} (when k-j valid)
    # = 2 * (w * p)[j] in a specific convolution sense
    # Let's use the FFT approach: grad = 2 * ifft(fft(w) * fft(p))[n-1:2n-1] cross-correlation
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p, N)
    h = np.fft.irfft(fw * fp, N)
    # The gradient w.r.t. p is: for each j, sum_k w[k] * (p[k-j] if k-j in [0,n-1]) + (p[j-k] if j-k in [0,n-1])
    # This equals 2 * cross_correlation(w, p)[j] where cross_corr[j] = sum_k w[k] * p[j-k]
    # cross_correlation = ifft(fft(w) * conj(fft(p)))... 
    # Actually for real signals: ifft(fft(w) * fft(p_rev)) gives cross-correlation
    # Let me just compute it directly via the full convolution approach
    # grad[j] = sum_k w[k] * p[k-j] + sum_k w[k] * p[j-k]
    # First term: (w * p)_j with proper indexing = cross-correlation
    # For efficiency, use FFT:
    # sum_k w[k]*p[k-j] for k-j in [0,n-1] means k in [j, j+n-1]
    # This is the cross-correlation of w and p at lag j
    # corr[j] = sum_k w[k] * p[k-j] = ifft(fft(w) * conj(fft(p)))[j] ... 
    # For real: ifft(fft(w) * fft(flip(p)))
    p_rev = p[::-1]
    fp_r = np.fft.rfft(p_rev, N)
    h2 = np.fft.irfft(fw * fp_r, N)
    # h2[j] = sum_k w[k] * p_rev[k-j] = sum_k w[k] * p[n-1-(k-j)] = sum_k w[k] * p[n-1-k+j]
    # This is getting confusing. Let me use a simpler approach.
    # grad[j] = 2 * sum_{k} w[k] * p[max(0, min(n-1, k-j))] ... no that's not right either.
    
    # Simplest correct approach:
    # conv(k) = sum_i p[i] * p[k-i] for i in [max(0,k-n+1), min(n-1,k)]
    # d conv(k) / d p[j] = p[k-j] if 0 <= k-j <= n-1, else 0
    #   + p[j-k] if 0 <= j-k <= n-1, else 0
    # grad[j] = sum_k w[k] * [p[k-j]*1(0<=k-j<=n-1) + p[j-k]*1(0<=j-k<=n-1)]
    # = sum_k w[k]*p[k-j]*1(0<=k-j<=n-1) + sum_k w[k]*p[j-k]*1(0<=j-k<=n-1)
    # Let m = k-j: first sum = sum_m w[m+j]*p[m]*1(0<=m<=n-1, 0<=m+j<=2n-2)
    #   = sum_m w[m+j]*p[m] for m in [max(0,-j), min(n-1, 2n-2-j)]
    #   Since j in [0,n-1]: m in [0, min(n-1, 2n-2-j)]
    # Second sum: let m = j-k: = sum_m w[j-m]*p[m]*1(0<=m<=n-1, 0<=j-m<=2n-2)
    #   Since j in [0,n-1]: m in [max(0, j-(2n-2)), min(n-1, j)] = [0, min(n-1, j)]
    
    # For efficiency with FFT:
    # First term: (w zero-padded to 2n-1) convolved with p, take positions [0, n-1]
    # This is: ifft(fft(w, 2n-1) * fft(p, 2n-1))[0:n] ... gives sum_k w[k]*p[k-j]? No.
    # ifft(fft(a)*fft(b))[j] = sum_k a[k]*b[(j-k) mod N] (circular)
    # With zero-padding to 2n-1, it becomes linear for the valid range.
    
    # Let me just use the direct formula with FFT:
    # grad = 2 * cross_corr(w, p) restricted to [0, n-1]
    # where cross_corr(w,p)[j] = sum_k w[k] * p[k-j] (with valid range)
    # cross_corr = ifft(fft(w, N) * conj(fft(p, N))) for circular, but with zero-padding it's linear
    # For real: ifft(fft(w) * conj(fft(p))) = ifft(fft(w) * fft(p)) since p is real... no that's wrong.
    # conj(fft(p)) = fft(p[::-1]) * exp(-i*2*pi*k*(n-1)/N) ... this is getting messy.
    
    # Let me just use the cross-correlation via the formula:
    # cross_corr[j] = sum_k w[k] * p[k-j] for k-j in [0, n-1]
    # = sum_{m=0}^{n-1} w[m+j] * p[m] (substituting m = k-j, so k = m+j)
    # This is a "sliding dot product" - can be computed as:
    # cross_corr = conv(w_padded, p_flipped) where w_padded has appropriate zeros
    
    # Actually the cleanest: use the fact that for the gradient of the smooth max:
    # grad[j] = 2 * sum_k w[k] * p[|k-j| valid partner]
    # Let me just compute it via two cross-correlations:
    
    # cross1[j] = sum_k w[k] * p[k-j] for valid k-j
    # = sum_{m} w[m+j] * p[m] for m=0..n-1, m+j in 0..2n-2
    # This is the cross-correlation of w (length 2n-1) and p (length n)
    # = ifft(fft(w, N) * fft(p[::-1], N))[N - (n-1) : N - (n-1) + n] ... 
    
    # I'll use a simpler approach that's O(n log n):
    # Note that sum_k w[k]*p[k-j] for k-j in [0,n-1] is the same as:
    # (w * p_flipped)[2n-2-j] where * is linear convolution and p_flipped = p[::-1]
    # Because (w * p_flipped)[m] = sum_i w[i] * p_flipped[m-i] = sum_i w[i] * p[n-1-m+i]
    # Let j substitute: sum_i w[i] * p[n-1-(2n-2-j)+i] = sum_i w[i] * p[j-1+i]
    # Hmm this is getting too complicated. Let me just use the FFT-based gradient directly.
    
    # The gradient of f(p) = sum_k w_k * conv(p,p)_k w.r.t. p_j is:
    # 2 * sum_k w_k * p_{k-j} (where p_{k-j} is 0 if k-j not in [0,n-1])
    # This is exactly 2 * (w * p)_j in the sense of cross-correlation.
    # Cross-correlation can be computed as: ifft(fft(w) * conj(fft(p)))
    # For real p: conj(fft(p)) = fft(p) conjugated element-wise
    # In numpy: np.conj(np.fft.rfft(p, N)) gives the conjugate of the rfft
    # ifft(fft(w) * conj(fft(p))) should give the cross-correlation
    
    fp_full = np.fft.rfft(p, N)
    h_cross = np.fft.irfft(fw * np.conj(fp_full), N)
    # h_cross[j] = sum_k w[k] * p[k-j] (with circular interpretation, but zero-padded to be linear)
    # We want h_cross[j] for j = 0, ..., n-1
    grad = 2.0 * h_cross[:n]
    return grad


def _adam_optimize(p0, deadline, rng, lr=0.05, beta_frac=0.08, beta_decay=0.94,
                   no_imp_limit=250, apply_nonneg=True):
    """Adam optimization in logit space with smooth-max objective."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0

    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)
        g = _conv_grad(w, p)

        # Project gradient to simplex tangent space
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = np.exp(z)
        p = p / p.sum()
        if apply_nonneg:
            p = np.maximum(p, 1e-15)
            p = p / p.sum()

        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 300 == 0:
                    lr = max(0.002, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 50 == 0:
            beta = max(1e-5, beta * beta_decay)

        if no_imp > no_imp_limit:
            ns = 0.05 if best_score < 1.51 else 0.1
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = np.exp(z)
            p = p / p.sum()
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = 0.05
            no_imp = 0

    return best_p, best_score


def _lp_solve(a, M):
    """Solve LP: maximize sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a, deadline, t_start=0.4, max_solves=60, m_shrink=0.998):
    """LP-based refinement with adaptive M."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(a, M)
        if b is None:
            b = _lp_solve(a, M * 1.02)
            if b is None:
                break

        s_b = float(b.sum())
        if s_b < 1e-10:
            break

        b_norm = b / s_b
        a_new = (1.0 - t_start) * a + t_start * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score(a_new)

        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        c = _full_convolve(a)
        M_new = float(c.max())
        M = max(M_new * m_shrink, best_score / (2.0 * n) * 1.002)
        t_start = max(0.1, t_start * 0.93)

    return best_a, best_score


def _subgradient_shave(a, deadline, n_iter=500):
    """Direct subgradient descent on the max of the convolution."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    lr = 0.01

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a)
        M = float(c.max())
        score = float(2.0 * n * M)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            lr = min(0.05, lr * 1.02)
        else:
            lr = max(0.001, lr * 0.98)

        # Find peak positions
        peak_mask = c >= M * 0.99
        peak_indices = np.where(peak_mask)[0]

        # Subgradient of max at peak: for peak at position k,
        # d c[k] / d a[j] = a[k-j] + a[j-k] (valid terms)
        grad = np.zeros(n)
        for k in peak_indices:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                partner = k - j
                if 0 <= partner < n:
                    grad[j] += a[partner]
                if j != partner and 0 <= j - k + n - 1 < n:
                    pass  # already counted by symmetry in conv
            # Actually for conv: d c[k]/d a[j] = a[k-j] if 0<=k-j<n, 
            #   plus we need to account for the fact that a[j] appears in c[k] 
            #   as both p[j]*p[k-j] and p[k-j]*p[j] (same term since it's self-conv)
            # Wait: c[k] = sum_i a[i]*a[k-i]. If j is one of the indices:
            # a[j] appears as a[j]*a[k-j] (when i=j) and as a[k-j]*a[j] (when i=k-j)
            # If k-j == j (i.e., k=2j), these are the same term, so contribution is 2*a[j]... no.
            # d/d a[j] of sum_i a[i]*a[k-i]:
            # = a[k-j] (from the term where i=j: a[j]*a[k-j], derivative w.r.t. a[j] is a[k-j])
            #   + a[j] (from the term where i=k-j: a[k-j]*a[j], derivative w.r.t. a[j] is a[k-j])
            # Wait no: d/d a[j] of a[i]*a[k-i]:
            # If i = j: derivative is a[k-j]
            # If k-i = j, i.e., i = k-j: derivative is a[i] = a[k-j]
            # If j = k-j (i.e., k=2j): both conditions give the same term, derivative is 2*a[j]... 
            # no, it's just a[j] counted once in the sum.
            # Actually: sum_i a[i]*a[k-i]. The variable a[j] appears in:
            # - term i=j: a[j]*a[k-j], contrib to derivative: a[k-j]
            # - term i=k-j: a[k-j]*a[j], contrib to derivative: a[k-j]
            # If j = k-j (k=2j): only one term a[j]*a[j], derivative is 2*a[j]
            # So: d c[k]/d a[j] = a[k-j] + a[k-j] = 2*a[k-j] if j != k-j
            #                                 = 2*a[j] if j == k-j
            # In both cases: 2*a[k-j] (since if j=k-j then k-j=j)
            # Wait that's just 2*a[k-j] always! Because the two contributions are both a[k-j].
            
        # I made an error above. Let me redo:
        # c[k] = sum_i a[i] * a[k-i]
        # d c[k] / d a[j] = a[k-j] (from i=j) + a[k-j] (from i=k-j, if k-j != j)
        # If k-j == j: the term is a[j]^2, derivative is 2*a[j] = 2*a[k-j]. Same formula!
        # So d c[k]/d a[j] = 2*a[k-j] if 0 <= k-j < n, else a[k-j] if only one term...
        # Actually if 0 <= k-j < n: both i=j and i=k-j are valid (unless they're the same),
        #   so derivative = 2*a[k-j]
        # If k-j < 0 or k-j >= n: only one of i=j or i=k-j is valid
        #   (specifically, i=j is valid if 0<=j<n which is always true, and i=k-j requires 0<=k-j<n)
        #   So if k-j not in [0,n-1]: derivative = a[k-j]... but k-j is out of range so this term doesn't exist.
        #   Only the i=j term survives: derivative = a[k-j]... but that's 0 since k-j out of range.
        #   Wait, I need to be more careful.
        # 
        # c[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i] * a[k-i]
        # d/d a[j]: 
        #   - if 0 <= k-j < n: the term i=j contributes a[j]*a[k-j], derivative = a[k-j]
        #   - if 0 <= j-k < n (i.e., j >= k): the term i=k-j... wait i must satisfy 0<=i<n and 0<=k-i<n
        #     i = k-j satisfies 0<=k-i<n iff 0<=j<n which is true.
        #     And 0<=i<n iff 0<=k-j<n.
        #     So the term i=k-j exists iff 0<=k-j<n, same condition!
        #     In this case, derivative of a[k-j]*a[j] w.r.t. a[j] = a[k-j]
        #   - total: if 0<=k-j<n: derivative = a[k-j] + a[k-j] = 2*a[k-j]
        #     (unless k-j==j, in which case it's still 2*a[j] because d(a[j]^2)/da[j] = 2*a[j])
        #   - if not (0<=k-j<n): derivative = 0
        #
        # So: d c[k] / d a[j] = 2 * a[k-j] if 0 <= k-j < n, else 0.
        
        grad = np.zeros(n)
        n_peaks = len(peak_indices)
        for k in peak_indices:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            partners = k - np.arange(j_min, j_max + 1)
            grad[j_min:j_max + 1] += 2.0 * a[partners]
        
        if n_peaks > 1:
            grad /= n_peaks  # average the subgradients from multiple peaks
        
        # Project to simplex tangent and normalize
        grad_mean = float(np.mean(grad))
        grad = grad - grad_mean
        norm = float(np.linalg.norm(grad))
        if norm < 1e-15:
            continue
        grad = grad / norm
        
        a = a - lr * grad
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

    return best_a, best_score


def _peak_flood(a, deadline, eta=0.04, n_iter_max=3000):
    """Multiplicative peak-flooding."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    c = _full_convolve(a)

    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break

        M = float(c.max())
        score = float(2.0 * n * M)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        threshold = M * 0.92
        hot_mask = c >= threshold
        if not hot_mask.any():
            hot_mask[int(np.argmax(c))] = True

        heat = np.zeros(n)
        hot_indices = np.where(hot_mask)[0]
        for k in hot_indices:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                partner = k - np.arange(i_start, i_end + 1)
                heat[i_start:i_end + 1] += a[partner]

        heat_max = float(heat.max())
        if heat_max < 1e-15:
            break
        heat = heat / heat_max

        factor = 1.0 - eta * heat
        factor = np.maximum(factor, 0.01)
        a = a * factor
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        c = _full_convolve(a)

        if it % 200 == 0 and it > 0:
            if score >= best_score:
                eta = max(0.005, eta * 0.9)

    return best_a, best_score


def _make_profile(n, profile_type='linear', param=2.0):
    t = np.linspace(0.0, 1.0, n)
    if profile_type == 'linear':
        a = 1.0 + param * (t - 0.5)
    elif profile_type == 'quadratic':
        a = 1.0 + param * (t - 0.5) ** 2
    elif profile_type == 'cosine':
        a = 1.0 + param * np.cos(2.0 * np.pi * t)
    elif profile_type == 'cubic':
        u = 2.0 * t - 1.0
        a = 1.0 + param * u ** 3
    elif profile_type == 'flat':
        a = np.ones(n)
    else:
        a = 1.0 + param * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, label=""):
        nonlocal best_seq, best_score
        if p is not None and p.size > 0:
            sc = _score(p)
            if np.isfinite(sc) and sc < best_score:
                best_seq = np.asarray(p, dtype=np.float64).copy()
                best_score = float(sc)
                print(f"  BEST: {best_score:.10f} n={p.size} [{label}] t={time.monotonic()-t_start:.0f}s")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    if parent is not None:
        sc_p = _score(parent)
        print(f"Parent score: {sc_p:.10f}")
        update(parent, 'parent')

    # === Phase 1: Adam polish on parent (250s) ===
    ph1_end = t_start + 250.0
    print(f"\nPhase 1: Adam on parent (t={ph1_end - t_start:.0f}s)...")
    if parent is not None:
        p1, sc1 = _adam_optimize(parent, ph1_end, rng, lr=0.05, beta_frac=0.07,
                                  beta_decay=0.94, no_imp_limit=200)
        update(p1, 'adam_parent')

                                        beta_decay=0.95, no_imp_limit=150)
            update(p1b, 'adam_parent2')
    else:
        p1 = _make_profile(1024, 'linear', 2.0)
        update(p1, 'init')

    # === Phase 2: LP refinement (350s) ===
    ph2_end = t_start + 600.0
    print(f"\nPhase 2: LP refinement (until t={ph2_end - t_start:.0f}s)...")
    if best_seq is not None and best_seq.size <= 2048:
        p2, sc2 = _lp_refine(best_seq, ph2_end, t_start=0.35, max_solves=50, m_shrink=0.998)
        update(p2, 'lp_main')
        # Quick Adam after LP
        if time.monotonic() < ph2_end - 20:
            p2a, sc2a = _adam_optimize(best_seq, ph2_end, rng, lr=0.02, beta_frac=0.04,
                                        beta_decay=0.96, no_imp_limit=100)
            update(p2a, 'post_lp_adam')

    # === Phase 3: Peak flooding + subgradient (200s) ===
    ph3_end = t_start + 800.0
    print(f"\nPhase 3: Flood + subgrad (until t={ph3_end - t_start:.0f}s)...")
    if best_seq is not None:
        half = (ph3_end - time.monotonic()) / 2
        p3, sc3 = _peak_flood(best_seq, time.monotonic() + half, eta=0.03)
        update(p3, 'flood')
        if time.monotonic() < ph3_end - 10:
            p3s, sc3s = _subgradient_shave(best_seq, ph3_end, n_iter=300)
            update(p3s, 'subgrad')

    # === Phase 4: Try larger n = 2048 (350s) ===
    ph4_end = t_start + 1150.0
    print(f"\nPhase 4: n=2048 (until t={ph4_end - t_start:.0f}s)...")
    if time.monotonic() < ph4_end and best_seq is not None:
        n_new = 2048
        if best_seq.size < n_new:
            p_int = _interp_profile(best_seq, n_new)

    if time.monotonic() < ph6_end and best_seq is not None:
        remaining = ph6_end - time.monotonic()

                                      lr=0.01, beta_frac=0.03, beta_decay=0.97,
                                      no_imp_limit=80)

        # LP
        if time.monotonic() < ph6_end - 30 and best_seq.size <= 2048:
            per2 = min(80.0, ph6_end - time.monotonic() - 10)

                update(p6l, 'final_lp')

        # Another Adam
        if time.monotonic() < ph6_end - 10:
            per3 = ph6_end - time.monotonic() - 5
            if per3 > 20:
                p6b, sc6b = _adam_optimize(best_seq, time.monotonic() + per3, rng,
                                            lr=0.005, beta_frac=0.02, beta_decay=0.98,
                                            no_imp_limit=50)

        # Peak flood
        if time.monotonic() < ph6_end - 10:

                update(p6f, 'final_flood')

    # Final validation
    if best_seq is not None:

    fb = _make_profile(1024, 'linear', 2.0)

    return [float(x) for x in fb]
