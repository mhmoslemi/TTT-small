"""
LP-based active set optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal descent directions, combined with
gradient-based polishing and multi-scale exploration.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    """Score with sum(a) assumed to be 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _lp_direction(a, top_k=200, max_frac=0.2):
    """
    Solve LP to find direction d that minimizes the linearized max autocorrelation.
    Returns (d, target_t) or (None, None) if LP fails.
    """
    n = len(a)
    c = _full_convolve(a, a)
    # Find top_k lags
    if top_k >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argsort(c)[-top_k:]
        idx = np.sort(idx.astype(int))
    
    # Build constraint matrix: A[row, i] = dc[m]/da[i] = a[m-i] if 0 <= m-i < n
    A = np.zeros((len(idx), n))
    for row, m in enumerate(idx):
        m = int(m)
        i_lo = max(0, m - n + 1)
        i_hi = min(n - 1, m)
        if i_lo <= i_hi:
            # A[row, i] = a[m - i] for i in [i_lo, i_hi]
            # m - i ranges from m - i_lo to m - i_hi
            # i_lo = max(0, m-n+1), so m - i_lo = min(m, n-1)
            # i_hi = min(n-1, m), so m - i_hi = max(0, m-n+1)
            A[row, i_lo:i_hi+1] = a[m - i_hi : m - i_lo + 1][::-1]
    
    # LP: min t
    # s.t. A @ d - t <= -c[idx]
    #      1^T d = 0
    #      -a[i] <= d[i] <= max_frac * a[i]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((len(idx), n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = -c[idx]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    lower = np.empty(n + 1)
    lower[:n] = -a
    lower[n] = -np.inf
    upper = np.empty(n + 1)
    upper[:n] = max_frac * a
    upper[n] = np.inf
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

                         options={'presolve': True})
        if result.success:
            return result.x[:n].copy(), float(result.x[n])
    except Exception:
        pass
    return None, None

def _kkt_polish(a_init, deadline, max_iter=5000):
    """KKT-based polishing using least-squares for active set multipliers."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Active set: lags where c is within 1% of max
        threshold = cm * 0.99
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0:
            break
        
        # Build gradient matrix for active set
        # G[i, j] = grad(c[active[j]])[i] = a[active[j] - i] if valid
        G = np.zeros((n, len(active)))
        for j, m in enumerate(active):
            m = int(m)
            i_lo = max(0, m - n + 1)
            i_hi = min(n - 1, m)
            G[i_lo:i_hi+1, j] = a[m - i_hi : m - i_lo + 1][::-1]
        
        # Solve: find lambda (len(active) vector) and nu (scalar)
        # such that G @ lambda ≈ nu * 1 (for indices where a > 0)
        # This is: [G, -1] @ [lambda; nu] ≈ 0 (least squares)
        # Or: minimize ||G @ lambda - nu * 1||^2
        # Equivalent to: minimize ||[G, -1] @ [lambda; nu]||^2
        
        M = np.hstack([G, -np.ones((n, 1))])
        # Least squares: minimize ||M @ x||^2 s.t. sum(lambda) = 1, lambda >= 0
        # For simplicity, just do unconstrained LS first
        try:
            x, residuals, rank, sv = np.linalg.lstsq(M, np.zeros(n), rcond=None)
            lambda_vec = x[:len(active)]
            nu = x[-1]
            # Normalize lambda
            lambda_sum = lambda_vec.sum()
            if lambda_sum > 1e-10:
                lambda_vec = lambda_vec / lambda_sum
        except Exception:
            lambda_vec = np.ones(len(active)) / len(active)
            nu = 0.0

def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _build_grad_matrix(a, lags):

    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    LP-based active set optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses LP for direction finding, KKT least-squares for balancing, Adam for polish.
    """
    global height_sequence_1
    
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")
    
    # --- Load parent ---
    print("=== Loading initial candidates ===")
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm_fn(parent)
                update_best(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent: {e}")
    
    if parent is None:
        parent = _make_exp_profile(4096, 0.003)
        sc = _score_norm_fn(parent)
        update_best(parent, sc, "default")
        print(f"  Default: n={len(parent)}, score={sc:.10f}")
    
    a_curr = parent.copy()
    n_curr = len(a_curr)
    
    # --- Phase 1: LP-based optimization (main workhorse) ---
    print(f"=== Phase 1: LP optimization (n={n_curr}) ===")
    phase1_end = min(deadline, t_start + 900.0)
    if time.monotonic() < phase1_end:
        a_best, sc_best = _lp_optimize(a_curr, phase1_end,
                                        top_k_schedule=[50, 100, 200, 400, 600],
                                        max_frac=0.25)
        update_best(a_best, sc_best, "lp_main")
    
    # --- Phase 2: KKT polishing ---
    print(f"=== Phase 2: KKT polish ===")
    phase2_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        a_best, sc_best = _kkt_polish(best_seq, phase2_end, max_iter=10000)
        update_best(a_best, sc_best, "kkt")
    
    # --- Phase 3: Adam polish ---
    print(f"=== Phase 3: Adam polish ===")
    phase3_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        remaining = phase3_end - time.monotonic()
        if remaining > 30:
            a_best, sc_best = _adam_polish(best_seq, min(phase3_end - 5, time.monotonic() + remaining * 0.7),
                                            lr=0.02, n_max_iter=50000)
            update_best(a_best, sc_best, "adam1")
        if time.monotonic() < phase3_end - 10:
            a_best2, sc_best2 = _adam_polish(best_seq, min(phase3_end - 5, time.monotonic() + 120),
                                              lr=0.008, n_max_iter=20000)
            update_best(a_best2, sc_best2, "adam2")
    
    # --- Phase 4: L-BFGS polish ---
    print(f"=== Phase 4: L-BFGS polish ===")
    phase4_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        remaining = phase4_end - time.monotonic()
        if remaining > 20:
            a_best, sc_best = _lbfgs_polish(best_seq, min(phase4_end - 5, time.monotonic() + min(120, remaining)),
                                              top_k=10)
            update_best(a_best, sc_best, "lbfgs")
    
    # --- Phase 5: Short LP burst on best ---
    print(f"=== Phase 5: Final LP burst ===")
    if time.monotonic() < deadline - 5 and best_seq is not None:
        burst_end = min(deadline - 3, time.monotonic() + 120)
        a_best, sc_best = _lp_optimize(best_seq, burst_end,
                                        top_k_schedule=[200, 300, 500],
                                        max_frac=0.1)
        update_best(a_best, sc_best, "lp_final")
    
    # --- Final validation ---
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = _make_exp_profile(4096, 0.003)
            final_score = _score_fn(final_seq)
        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")
        
        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 10000:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  eval check failed: {e}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_exp_profile(4096, 0.003)
    print(f"  FALLBACK: n=4096, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
