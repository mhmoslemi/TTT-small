    m = np.zeros(n)

    t = 0

        # Subtract mean to make it sum to zero (tangent to the simplex)
        grad = grad - grad.mean()
        
        # Adam update
        t += 1
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*(grad*grad)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        
        a_new = a - lr * mh / (np.sqrt(vh) + eps)
        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-10:
            a_new = best_a.copy()
            continue
        a_new = a_new / a_new_sum

        score = 2.0 * n * c.max()
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        
        # Adaptive beta
        if t % 50 == 0:
            beta *= 0.97
            M = c.max()
            beta = max(1e-6, min(beta, 0.1 * M))
    
    return best_a, best_score

Optimized non-negative sequence search minimizing peak self-convolution.
Key improvements: value-space Adam with FFT gradient, fast peak flooding,
constraint-reduced LP, multi-scale search (n=512..4096).
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a):
    """Full self-convolution via FFT."""
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    """Objective: 2n * max(a*a), assuming sum(a)=1."""
    return 2.0 * a.size * float(_fft_conv(a).max())


def _grad_from_weights(w, a):
    """Gradient of smooth-max objective given convolution weights w."""
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(n + 2 * n - 1)
    far = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(far * fw, N)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad


def _smooth_max_weights(c, beta, k_top=None):
    """Compute smooth max value and weights on convolution positions."""
    c = np.asarray(c, dtype=np.float64)
    if k_top is not None and k_top < c.size:
        top_idx = np.argpartition(c, -k_top)[-k_top:]
        top_vals = c[top_idx]
    else:
        top_idx = np.arange(c.size)
        top_vals = c
    m = float(top_vals.max())
    if beta < 1e-14:
        w = np.zeros(c.size)
        mask = top_vals >= m - 1e-10
        w[top_idx[mask]] = 1.0 / max(1, int(mask.sum()))
        return m, w
    x = (top_vals - m) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(c.size)
        w[top_idx[int(np.argmax(top_vals))]] = 1.0
        return m, w
    sm_val = m + beta * np.log(s)
    w = np.zeros(c.size)
    w[top_idx] = ex / s
    return sm_val, w


def _adam_values(a, deadline, rng, lr=0.03, beta_frac=0.06, k_top=200,
                 beta_decay=0.96, no_imp_limit=200, noise_scale=None):
    """Adam optimizer in value space with simplex projection."""
    n = a.size
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_conv(a)
    best_score = 2.0 * n * float(c.max())
    M = float(c.max())
    beta = max(1e-6, beta_frac * M)

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta, k_top)
        grad = _grad_from_weights(w, a)
        grad -= grad.mean()

        if not np.all(np.isfinite(grad)):
            a = best_a.copy()
            m_adam.fill(0)
            v_adam.fill(0)
            t = 0
            c = _fft_conv(a)
            continue

        t += 1
        m_adam = b1 * m_adam + (1 - b1) * grad
        v_adam = b2 * v_adam + (1 - b2) * (grad * grad)
        mh = m_adam / (1.0 - b1 ** t)
        vh = v_adam / (1.0 - b2 ** t)

        a_new = a - lr * mh / (np.sqrt(vh) + eps)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = best_a.copy()
            continue
        a_new = a_new / sn

        c = _fft_conv(a_new)
        score = 2.0 * n * float(c.max())

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        a = a_new

            beta = max(1e-6, beta_frac * M)

            v_adam.fill(0)
            v_adam.fill(0)

            lr = base_lr
            no_imp = 0

def _peak_flood_fast(a, deadline, eta=0.05, max_iter=3000):
    """FFT-based peak flooding: reduce mass at positions contributing to peaks."""
    n = a.size
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    c = _fft_conv(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        M = float(c.max())
        score = 2.0 * n * M
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()

        # Identify hot positions
        thresh = M * 0.92
        hot = (c >= thresh).astype(np.float64)
        if hot.sum() == 0:
            hot[int(np.argmax(c))] = 1.0

        # Compute heat via FFT: heat[j] = sum_k hot[k] * a[k-j] (valid range)
        # = (a_rev * hot)[j + n - 1]
        a_rev = a[::-1]
        N = _next_pow2(3 * n - 1)
        heat = np.fft.irfft(np.fft.rfft(a_rev, N) * np.fft.rfft(hot, N), N)
        heat = heat[n - 1:2 * n - 1]
        heat = np.maximum(heat, 0.0)
        hmax = heat.max()
        if hmax < 1e-15:
            break
        heat /= hmax

        factor = 1.0 - eta * heat
        factor = np.maximum(factor, 0.01)
        a = a * factor
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        c = _fft_conv(a)

        if it % 200 == 0 and it > 0:
            if score >= best_score * (1 - 1e-10):
                eta = max(0.005, eta * 0.9)

    return best_a, best_score


def _lp_solve_reduced(a, M, k_constraints=30):
    """LP with only top-k constraints for speed."""
    n = a.size
    c = _fft_conv(a)
    top_k = min(k_constraints, 2 * n - 1)
    top_idx = np.argpartition(c, -top_k)[-top_k:]
    top_idx.sort()

    rows, cols, vals = [], [], []
    for k in top_idx:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())

    if len(rows) == 0:
        return None

    # Remap rows to 0..len(top_idx)-1
    row_map = {k: i for i, k in enumerate(top_idx)}
    rows_remapped = [row_map[r] for r in rows]

    A_ub = csr_matrix((vals, (rows_remapped, cols)), shape=(len(top_idx), n))
    c_obj = -np.ones(n)
    b_ub = M * np.ones(len(top_idx))
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_refine(a, deadline, t_init=0.4, max_solves=30, m_shrink=0.997, k_constr=30):
    """LP-based refinement with decreasing M."""
    n = a.size
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    c = _fft_conv(a)
    M = float(c.max())
    t = t_init
    n_solves = 0

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve_reduced(a, M, k_constr)
        if b is None:
            b = _lp_solve_reduced(a, M * 1.02, k_constr)
            if b is None:
                break

        s_b = b.sum()
        if s_b < 1e-10:
            break
        b = b / s_b
        a_new = (1.0 - t) * a + t * b
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc = _score(a_new)

        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        c = _fft_conv(a)
        M_new = float(c.max())
        M = max(M_new * m_shrink, best_score / (2.0 * n) * 1.003)
        t = max(0.1, t * 0.9)

    return best_a, best_score


def _init_gaussian(n, width_frac=0.4):
    """Gaussian bump initialization."""
    x = np.arange(n) - (n - 1) / 2.0
    sigma = width_frac * n / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _init_raised_cosine(n):
    """Raised cosine initialization."""
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (np.arange(n) + 0.5) / n))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _init_linear_taper(n, asym=1.5):
    """Asymmetric linear taper: high at one end."""
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 + asym * t) * (1.0 - 0.5 * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _init_power_taper(n, alpha=0.5):
    """Power-law taper."""
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 - t) ** alpha + 0.1
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_to_new_length(a, n_new):
    """Interpolate a profile to a new length."""
    n_old = a.size
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, n_old)
    x1 = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()


def _multi_method_optimize(a, deadline, rng, phase_times=None):
    """Run multiple optimization methods in sequence, tracking best."""
    n = a.size
    best_a = a.copy()
    best_score = _score(a)

    if phase_times is None:
        n_phases = 4
        total = max(1.0, deadline - time.monotonic())
        phase_times = [total / n_phases] * n_phases

    # Phase 1: Adam
    t1 = time.monotonic() + phase_times[0]
    a1, s1 = _adam_values(best_a, t1, rng, lr=0.03, beta_frac=0.06, k_top=200)
    if s1 < best_score:
        best_score, best_a = s1, a1
    print(f"    Adam: {best_score:.10f} (n={n})")

    # Phase 2: Peak flooding
    t2 = time.monotonic() + phase_times[1]
    a2, s2 = _peak_flood_fast(best_a, t2, eta=0.04)
    if s2 < best_score:
        best_score, best_a = s2, a2
    print(f"    Flood: {best_score:.10f} (n={n})")

    # Phase 3: LP
    t3 = time.monotonic() + phase_times[2]
    a3, s3 = _lp_refine(best_a, t3, t_init=0.35, max_solves=25, m_shrink=0.997, k_constr=30)
    if s3 < best_score:
        best_score, best_a = s3, a3
    print(f"    LP: {best_score:.10f} (n={n})")

    # Phase 4: Final Adam
    a4, s4 = _adam_values(best_a, deadline, rng, lr=0.01, beta_frac=0.03, k_top=100,
                          no_imp_limit=100)
    if s4 < best_score:
        best_score, best_a = s4, a4
    print(f"    Final: {best_score:.10f} (n={n})")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main entry point: search for best non-negative sequence."""
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(a, sc, label=""):
        nonlocal best_seq, best_score
        if a is not None and np.isfinite(sc):
            sc = _score(a)
            if sc < best_score:
                best_seq = a.copy()
                best_score = float(sc)
                print(f"  NEW BEST: {best_score:.10f} n={a.size} [{label}]")

    # Get parent if available
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 10:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            if parent_raw.sum() > 0.01:
                parent = parent_raw / parent_raw.sum()
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={parent.size}, score={sc_parent:.10f}")

    # === Strategy: try n=1024 (polish parent) and n=2048 (explore larger) ===
    n_values = [1024, 2048]
    if parent is not None and parent.size not in n_values and parent.size >= 256:
        n_values = [parent.size] + [x for x in n_values if x != parent.size]

    # Allocate time: 50% to first n, 35% to second, 15% reserve
    total_time = deadline - time.monotonic()
    time_alloc = {}
    remaining = total_time
    for i, n in enumerate(n_values):
        frac = 0.5 if i == 0 else (0.35 if i == 1 else 0.2)
        t_alloc = min(remaining * frac, remaining - 30)
        time_alloc[n] = t_alloc
        remaining -= t_alloc

    for n in n_values:
        if time.monotonic() > deadline - 60:
            break
        t_phase_end = time.monotonic() + time_alloc.get(n, 100)
        print(f"\n=== Optimizing n={n} (budget {time_alloc.get(n, 100):.0f}s) ===")

        # Initialize
        if parent is not None and parent.size == n:
            a_init = parent.copy()
            label_init = "parent"
        elif parent is not None:
            a_init = _interp_to_new_length(parent, n)
            label_init = f"interp_from_n{parent.size}"
        else:
            a_init = _init_gaussian(n, 0.4)
            label_init = "gaussian"

        sc_init = _score(a_init)
        print(f"  Init ({label_init}): {sc_init:.10f}")
        _update(a_init, sc_init, label_init)

        # Run multi-method optimization
        phase_budget = t_phase_end - time.monotonic()
        if phase_budget > 30:
            phases = [phase_budget * 0.3, phase_budget * 0.25, phase_budget * 0.25, phase_budget * 0.2]
            a_opt, sc_opt = _multi_method_optimize(a_init, t_phase_end, rng, phases)
            _update(a_opt, sc_opt, f"opt_n{n}")

        # Quick secondary initialization and polish
        if time.monotonic() < t_phase_end - 30:
            a_alt = _init_raised_cosine(n)
            sc_alt = _score(a_alt)
            if sc_alt < sc_init:
                print(f"  Alt init (raised_cos): {sc_alt:.10f}")
                t_sub = min(60.0, t_phase_end - time.monotonic() - 10)
                if t_sub > 20:
                    a_alt_opt, sc_alt_opt = _adam_values(a_alt, time.monotonic() + t_sub,
                                                         rng, lr=0.02, beta_frac=0.05, k_top=150)
                    _update(a_alt_opt, sc_alt_opt, f"alt_n{n}")

    # === Final polish on the best found ===
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Final polish on n={best_seq.size} (score={best_score:.10f}) ===")
        remaining = deadline - time.monotonic() - 10

        # Adam with small learning rate
        t1 = time.monotonic() + remaining * 0.35
        a_f1, s_f1 = _adam_values(best_seq, t1, rng, lr=0.008, beta_frac=0.02,
                                   k_top=50, no_imp_limit=80)
        _update(a_f1, s_f1, "final_adam")

        # Peak flood
        if time.monotonic() < deadline - 30:
            t2 = time.monotonic() + min(60.0, deadline - time.monotonic() - 20)
            a_f2, s_f2 = _peak_flood_fast(best_seq, t2, eta=0.02, max_iter=2000)
            _update(a_f2, s_f2, "final_flood")

        # LP
        if time.monotonic() < deadline - 30 and best_seq.size <= 2048:
            t3 = time.monotonic() + min(45.0, deadline - time.monotonic() - 15)
            a_f3, s_f3 = _lp_refine(best_seq, t3, t_init=0.2, max_solves=8,
                                     m_shrink=0.999, k_constr=20)
            _update(a_f3, s_f3, "final_lp")

        # One more Adam
        if time.monotonic() < deadline - 15:
            t4 = deadline - 10
            a_f4, s_f4 = _adam_values(best_seq, t4, rng, lr=0.005, beta_frac=0.015,
                                       k_top=30, no_imp_limit=50)
            _update(a_f4, s_f4, "final_adam2")

    # Final validation and return
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = final.sum()
        if s > 0.01:
            final = final / s
        else:
            final = _init_gaussian(1024, 0.4)

        exact_sc = _score(final)
        print(f"\nFINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    # Fallback
    fb = _init_gaussian(1024, 0.4)
    print(f"FALLBACK n=1024, score={_score(fb):.10f}")
    return [float(x) for x in fb]
