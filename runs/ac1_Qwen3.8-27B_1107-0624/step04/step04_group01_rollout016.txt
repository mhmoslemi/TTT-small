def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    """Score without the sum normalization (assumes sum=1)."""
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _log_sum_exp(c, beta):
    """Smooth approximation of max(c) using log-sum-exp."""
    cm = float(c.max())
    if beta < 1e-15:
        return cm
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if not np.isfinite(s):
        return cm
    return cm + beta * np.log(s)

"""
Multi-strategy optimization: spectral compression + L-BFGS + evolutionary polishing
for minimizing 2*n*max(conv(a,a))/sum(a)^2 with non-negative sequences.
Key innovations: (1) iterative spectral compression to flatten the power spectrum,
(2) L-BFGS with log-sum-exp smooth max, (3) structured restarts with Fourier mutations.

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _spectral_compress(a, gamma, n_fft=None):
    """
    Apply spectral compression to flatten the power spectrum.
    Compresses |A[f]|^2 by raising to power gamma < 1.
    Preserves DC component and phase.
    """
    n = len(a)
    if n_fft is None:
        n_fft = _next_pow2(2 * n - 1)
    
    a_pad = np.zeros(n_fft)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    
    # Preserve DC
    A0 = A[0].copy()
    
    P = np.abs(A) ** 2
    # Target: compress dynamic range
    # P_new[f] = P[f]^gamma * mean(P[1:])^(1-gamma) for f > 0
    P_non_dc = P[1:]
    mean_P = float(np.mean(P_non_dc))
    if mean_P < 1e-30:
        return a.copy()
    
    P_new = np.copy(P)
    P_new[0] = P[0]  # Keep DC
    P_new[1:] = (P[1:] ** gamma) * (mean_P ** (1 - gamma))
    
    # Reconstruct with same phase
    phase = A / np.abs(A + 1e-300)
    A_new = phase * np.sqrt(P_new)
    A_new[0] = A0
    
    a_full = np.fft.irfft(A_new, n_fft)
    a_new = a_full[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _spectral_compress_iterative(a, deadline, gamma_init=0.7, gamma_min=0.3):
    """Iteratively apply spectral compression with decreasing gamma."""
    n = len(a)
    n_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    gamma = gamma_init
    iteration = 0
    while time.monotonic() < deadline:
        a_new = _spectral_compress(a, gamma, n_fft)
        sc = _score_norm(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Score got worse, reduce compression strength or try different gamma
            gamma = min(1.0, gamma + 0.05)
            if gamma > 1.0:
                gamma = gamma_min + 0.1
                a = best_a
        
        # Gradually increase compression
        if iteration % 50 == 0:
            gamma = max(gamma_min, gamma - 0.02)
        iteration += 1
        if iteration % 200 == 0:
            print(f"    Spectral iter {iteration}: score={best_sc:.10f}, gamma={gamma:.3f}")
    
    return best_a, best_sc


def _fourier_perturb(a, rng, n_modes=5, amp=0.1):
    """Add random Fourier modes to perturb the sequence."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        phase = rng.uniform(0, 2 * np.pi)
        a_mode = amp * rng.uniform(0.3, 1.0)
        t = np.arange(n)
        z = z + a_mode * np.sin(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _lbfgs_optimize(a_init, deadline, beta_init=0.01, max_iter=5000):
    """
    L-BFGS optimization in log-space with smooth max (log-sum-exp).
    """
    n = len(a_init)
    n_fft = _next_pow2(2 * n - 1)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_sc = _score_norm(a_init)
    
    def objective(z_vec):
        a_vec = _softmax(z_vec)
        a_pad = np.zeros(n_fft)
        a_pad[:n] = a_vec
        A = np.fft.rfft(a_pad)
        P = np.abs(A) ** 2
        c = np.fft.irfft(P, n_fft)[:2 * n - 1]
        cm = float(c.max())
        shifted = (c - cm) / beta_init
        e = np.exp(shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            return cm
        lse = cm + beta_init * np.log(s)
        return 2.0 * n * lse
    
    def gradient(z_vec):
        a_vec = _softmax(z_vec)
        a_pad = np.zeros(n_fft)
        a_pad[:n] = a_vec
        A = np.fft.rfft(a_pad)
        P = np.abs(A) ** 2
        c = np.fft.irfft(P, n_fft)[:2 * n - 1]
        cm = float(c.max())
        shifted = (c - cm) / beta_init
        e = np.exp(shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        
        # Compute grad_a[m] = 2n * sum_k w[k] * dc[k]/da[m]
        # dc[k]/da[m] = 2*a[k-m] (when valid and 2m != k) + a[m] (when 2m == k, correction)
        # Use cross-correlation: h[m] = sum_k w[k] * a[k-m]
        W = np.fft.rfft(w, n_fft)
        Av = np.fft.rfft(a_vec, n_fft)
        cc = np.fft.irfft(np.conj(W) * Av, n_fft)
        h = cc[:n]
        
        grad_a = 2.0 * h * (2.0 * n)
        even_k = 2 * np.arange(n)
        valid = even_k < (2 * n - 1)
        if np.any(valid):
            grad_a[valid] -= w[even_k[valid]] * a_vec[valid] * (2.0 * n)
        
        # Chain rule: grad_z[l] = a[l] * (grad_a[l] - dot(grad_a, a))
        dot_ga = float(np.dot(grad_a, a_vec))
        grad_z = a_vec * (grad_a - dot_ga)
        
        return grad_z
    
    # Run L-BFGS with multiple beta values (continuation)
    betas = [beta_init, beta_init * 0.5, beta_init * 0.25, beta_init * 0.1, beta_init * 0.05]
    
    for beta in betas:
        beta_init = beta  # Update for inner functions
        if time.monotonic() > deadline:
            break
        
        options = {
            'maxiter': max_iter,
            'ftol': 1e-12,
            'gtol': 1e-8,
            'maxls': 20,
            'maxcor': 30,
        }
        
        try:
            res = scipy_minimize(
                objective, z, jac=gradient,
                method='L-BFGS-B',
                bounds=[(-30, 30)] * n,
                options=options
            )
            z = res.x
            a_final = _softmax(z)
            sc = _score_norm(a_final)
            if sc < best_sc:
                best_sc = sc
                best_a = a_final.copy()
                print(f"    L-BFGS beta={beta:.5f}: score={sc:.10f}")
        except Exception as ex:
            print(f"    L-BFGS error at beta={beta:.5f}: {ex}")
            break
        
        # Check time
        if time.monotonic() > deadline:
            break
    
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc


def _adam_optimize(a_init, deadline, beta_schedule, max_iter=100000, lr=0.01):
    """Adam optimizer in log-space (similar to previous but cleaned up)."""
    n = len(a_init)
    n_fft = _next_pow2(2 * n - 1)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_sc = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            
            a = _softmax(z)
            a_pad = np.zeros(n_fft)
            a_pad[:n] = a
            A = np.fft.rfft(a_pad)
            P = np.abs(A) ** 2
            c = np.fft.irfft(P, n_fft)[:2 * n - 1]
            cm = float(c.max())
            shifted = (c - cm) / beta
            e = np.exp(shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros(2 * n - 1)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            
            # Gradient
            W = np.fft.rfft(w, n_fft)
            Av = np.fft.rfft(a, n_fft)
            cc = np.fft.irfft(np.conj(W) * Av, n_fft)
            h = cc[:n]
            grad_a = 2.0 * h * (2.0 * n)
            even_k = 2 * np.arange(n)
            valid = even_k < (2 * n - 1)
            if np.any(valid):
                grad_a[valid] -= w[even_k[valid]] * a[valid] * (2.0 * n)
            dot_ga = float(np.dot(grad_a, a))
            grad_z = a * (grad_a - dot_ga)
            
            if not np.all(np.isfinite(grad_z)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            
            z = z - update

            if t % 200 == 0:
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_sc:

                    best_a = a.copy()
                
                if it_in_phase % 5000 == 0:

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)

        best_a = a_final.copy()
    
    return best_a, best_sc


def _direct_flatten(a, deadline, n_steps=1000):
    """
    Direct iterative flattening: at each step, reduce the peak convolution value
    by adjusting the sequence elements that contribute most to it.
    """
    n = len(a)
    n_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    a_work = a.copy()
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        a_pad = np.zeros(n_fft)
        a_pad[:n] = a_work
        A = np.fft.rfft(a_pad)
        P = np.abs(A) ** 2
        c = np.fft.irfft(P, n_fft)[:2 * n - 1]
        c_max = float(c.max())
        
        # Find contributing elements
        # c[k] = sum_j a[j]*a[k-j]
        # The peak is at k*. To reduce c[k*], we need to reduce a[j]*a[k*-j]
        k_star = int(np.argmax(c))
        
        # Find the j that contributes most to c[k_star]
        j_range = np.arange(max(0, k_star - n + 1), min(k_star, n - 1) + 1)
        if len(j_range) == 0:
            break
        contribs = a_work[j_range] * a_work[k_star - j_range]
        j_max = j_range[np.argmax(contribs)]
        
        # Reduce a[j_max] slightly, redistribute to a[(n-1)-j_max] or other low-contribution positions
        # Actually, let's use a simpler approach: reduce the element most responsible
        # for the peak, and increase elements at positions that don't contribute to the peak
        
        # Find positions that don't contribute to the top convolution values
        top_k = np.argsort(c)[-min(5, n):]
        low_contrib = np.zeros(n)
        for k in top_k:
            for j in range(max(0, k - n + 1), min(k, n - 1) + 1):
                low_contrib[j] += 1
                low_contrib[k - j] += 1
        
        # Increase elements with low contribution, decrease those with high
        step_size = 0.001 * a_work.max()
        a_work[j_max] -= step_size
        # Find a low-contribution position to increase
        candidates = np.where(low_contrib == 0)[0]
        if len(candidates) > 0:
            a_work[candidates[0]] += step_size
        else:
            a_work[np.argmin(low_contrib)] += step_size
        
        a_work = np.maximum(a_work, 1e-15)
        a_work = a_work / a_work.sum()
        
        sc = _score_norm(a_work)
        if sc < best_sc:
            best_sc = sc
            best_a = a_work.copy()
    
    return best_a, best_sc

                p_best, sc_best = _lbfgs_optimize(
                    a_start, time.monotonic() + per_time,
                    beta_init=0.01, max_iter=1000
                )
                update_best(p_best, _score(p_best))
            
            print(f"  Restart {i} ({label}): best={best_score:.10f}")
    
    print(f"Phase 4 done: best={best_score:.10f}")
    
    # --- Phase 5: Fine polish with very low beta (1500s - 1750s) ---
    print(f"\n=== Phase 5: Fine Polish ===")
    phase5_end = min(deadline, t_start + 1750.0)
    
    if best_seq is not None and time.monotonic() < phase5_end - 30:
        c_best = _full_convolve(best_seq, best_seq)
        max_cb = float(c_best.max())
        beta_schedule = [
            0.001 * max_cb, 0.0005 * max_cb, 0.0002 * max_cb,
            0.0001 * max_cb, 0.00005 * max_cb
        ]
        per_time = phase5_end - time.monotonic() - 15
        p_best, sc_best = _adam_optimize(
            best_seq, time.monotonic() + per_time,
            beta_schedule, max_iter=200000, lr=0.005
        )
        update_best(p_best, _score(p_best))
    
    # Final L-BFGS with very low beta
    if time.monotonic() < phase5_end - 10 and best_seq is not None:
        per_time = phase5_end - time.monotonic() - 5
        if per_time > 10:
            p_best, sc_best = _lbfgs_optimize(
                best_seq, time.monotonic() + per_time,
                beta_init=0.003, max_iter=500
            )
            update_best(p_best, _score(p_best))
    
    print(f"Phase 5 done: best={best_score:.10f}")
    
    # --- Phase 6: Try larger n (1750s - deadline) ---
    print(f"\n=== Phase 6: Size Increase ===")
    if best_seq is not None and len(best_seq) < 4096 and time.monotonic() < deadline - 30:
        target_n = 4096
        x0 = np.linspace(0, 1, len(best_seq))
        x1 = np.linspace(0, 1, target_n)
        p_interp = np.interp(x1, x0, best_seq)
        p_interp = np.maximum(p_interp, 1e-12)
        p_interp = p_interp / p_interp.sum()
        update_best(p_interp, _score(p_interp))
        
        if time.monotonic() < deadline - 20:
            c_int = _full_convolve(p_interp, p_interp)
            max_ci = float(c_int.max())
            beta_schedule = [0.005 * max_ci, 0.002 * max_ci, 0.001 * max_ci]
            per_time = deadline - time.monotonic() - 10
            if per_time > 20:
                p_best, sc_best = _adam_optimize(
                    p_interp, time.monotonic() + per_time,
                    beta_schedule, max_iter=100000, lr=0.008
                )
                update_best(p_best, _score(p_best))
    
    # --- Final validation ---
    print(f"\n=== Final ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        
        return [float(x) for x in final]
    
    # Fallback
    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
