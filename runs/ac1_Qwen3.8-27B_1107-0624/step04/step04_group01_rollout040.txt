import numpy as np
from scipy.optimize import minimize
import time

def softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    N = 1
    while N < n:
        N *= 2
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n]

def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def smooth_max_grad(z, beta):
    """Compute the gradient of 2n * smooth_max(conv(a,a)) w.r.t. z, where a = softmax(z)."""
    a = softmax(z)
    n = len(a)
    c = fft_convolve(a, a)
    max_c = c.max()
    shifted = (c - max_c) / beta
    e = np.exp(shifted)
    w = e / e.sum()

    # Use the formula from the previous code
    a_rev = a[::-1]
    wc = fft_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = np.dot(w, c)
    
    grad = 2.0 * n * a * (h - W)
    return grad, c, max_c, w

def optimize_lbfgs(a0, deadline, n_beta_steps=20):
    """Optimize using L-BFGS with log-sum-exp continuation."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-15))
    z = z - z.mean()
    
    best_z = z.copy()
    best_score = score_of(a0)
    
    c0 = fft_convolve(a0, a0)
    max_c0 = c0.max()
    
    # Beta schedule: from smooth to sharp

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    """Compute correct gradient of 2n*smoothed_max(conv(a,a)) w.r.t. z where a=softmax(z).

    e = np.exp(shifted)

        return 2.0 * max_c, np.zeros_like(z)
    smooth_val = 2.0 * (max_c + beta * np.log(s))
    w = e / s
    w_rev = w[::-1]
    ac = _full_convolve(a, w_rev)
    s_vec = ac[n - 1:2 * n - 1][::-1]
    h = 2.0 * s_vec
    W = 2.0 * float(np.dot(w, c))
    grad = 2.0 * n * (h - a * W)
    return smooth_val, grad


def _adam_optimize(a0, deadline, beta_schedule, max_iter=200000, lr=0.01):
    """Adam optimizer in log-space with correct gradient."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-15))
    z = z - z.mean()

    best_a = a0.copy()
    best_score = _score_norm(a0) = _score(a0)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            grad, a, c, w, _ = _compute_grad_correct(z, beta)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            max_update = 0.5
            update_norm = np.linalg.norm(update)
            if update_norm > max_update:
                update = update * (max_update / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 100 == 0:
                sc = _score(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _lbfgs_optimize(a0, deadline, n_beta_steps=15):
    """L-BFGS optimization with log-sum-exp continuation."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-15))
    z = z - z.mean()

    best_z = z.copy()
    best_score = _score(a0)

    c0 = _full_convolve(a0, a0)
    max_c0 = float(c0.max())

    # Beta from smooth (large) to sharp (small)
    betas = np.geomspace(0.1 * max_c0, 0.0002 * max_c0, n_beta_steps)

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break

        def fun(z_eval):
            val, _ = _objective_and_grad(z_eval, beta)
            return val

        def grad(z_eval):
            _, g = _objective_and_grad(z_eval, beta)
            return g

        max_iter_phase = max(50, min(500, int((deadline - time.monotonic()) / 0.5)))
        try:
            result = minimize(fun, z, jac=grad, method='L-BFGS-B',
                             options={'maxiter': max_iter_phase, 'gtol': 1e-7,
                                      'ftol': 1e-10, 'maxfun': 10000})
            z = result.x
        except Exception:
            pass

        a = _softmax(z)
        sc = _score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_z = z.copy()
            print(f"    L-BFGS beta={beta:.6f}: score={sc:.10f}")

    return _softmax(best_z), best_score


def _active_set_polish(a, deadline, max_iter=5000):
    """Polish by minimizing the average of top-k convolution values."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    best_a = a.copy()
    best_score = _score(a)

    # Find the number of "active" lags (near max)
    for top_k in [5, 10, 20, 50, 100]:
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        max_c = c.max()
        # Active set: lags where c[k] >= max_c * 0.95
        threshold = max_c * 0.95
        active = c >= threshold

        if active.sum() < 1:
            continue
        if active.sum() > 2 * n:
            # Use top-k instead
            idx = np.argsort(c)[-min(top_k, 2*n-1):]
            active = np.zeros_like(c)
            active[idx] = 1

        # Optimize: minimize sum of c[k] for active k
        # Use Adam
        lr = 0.005
        m_buf = np.zeros(n)
        v_buf = np.zeros(n)
        t_adam = 0
        b1, b2, eps = 0.9, 0.999, 1e-8

        iters_per_set = max(100, min(2000, int((deadline - time.monotonic()) / 0.05)))
        tset = 0
        while tset < iters_per_set:
            if time.monotonic() > deadline:
                break
            tset += 1
            t_adam += 1
            a = _softmax(z)
            c = _full_convolve(a, a)
            max_c = c.max()
            threshold = max_c * 0.98
            active = c >= threshold
            if active.sum() < 1:
                active[np.argmax(c)] = 1
            w = active / active.sum()

            w_rev = w[::-1]
            ac = _full_convolve(a, w_rev)
            s_vec = ac[n - 1:2 * n - 1][::-1]
            h = 2.0 * s_vec
            W = 2.0 * float(np.dot(w, c))
            grad = 2.0 * n * (h - a * W)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t_adam)
            vh = v_buf / (1.0 - b2 ** t_adam)
            z = z - lr * mh / (np.sqrt(vh) + eps)
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

        a = _softmax(z)
        sc = _score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"    Active-set (k~{active.sum()}): score={sc:.10f}")

    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.6 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edge_heavy":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * (2.0 * t - 1.0) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "center_heavy":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * (1.0 - (2.0 * t - 1.0) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-method optimization: L-BFGS continuation, Adam, active-set polishing,
    and multi-scale search to minimize 2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        sc0 = _score(parent)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        update_best(parent, sc0, "parent")
    else:
        n_main = 2048
        parent = _make_profile(n_main, "uniform")
        sc0 = _score(parent)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        update_best(parent, sc0, "uniform_init")

    # --- Phase 1: L-BFGS on parent (0 - 300s) ---
    phase1_end = min(deadline, t_start + 300.0)
    print(f"Phase 1: L-BFGS on parent, budget until t={phase1_end - t_start:.0f}s")
    a_lbfgs, sc_lbfgs = _lbfgs_optimize(parent, phase1_end, n_beta_steps=12)
    update_best(a_lbfgs, sc_lbfgs, "lbfgs_parent")

    # --- Phase 2: Adam continuation on best (300s - 700s) ---
    phase2_end = min(deadline, t_start + 700.0)
    print(f"Phase 2: Adam continuation, budget until t={phase2_end - t_start:.0f}s")
    if best_seq is not None:
        c_best = _full_convolve(best_seq, best_seq)
        max_cb = float(c_best.max())
        beta_schedule = [
            0.05 * max_cb, 0.02 * max_cb, 0.01 * max_cb,
            0.005 * max_cb, 0.002 * max_cb, 0.001 * max_cb,
            0.0005 * max_cb, 0.0002 * max_cb, 0.0001 * max_cb
        ]
        a_adam, sc_adam = _adam_optimize(best_seq, phase2_end, beta_schedule,
                                         max_iter=200000, lr=0.012)
        update_best(a_adam, sc_adam, "adam_cont")

    # --- Phase 3: Multi-restart with different initializations (700s - 1200s) ---
    phase3_end = min(deadline, t_start + 1200.0)
    print(f"Phase 3: Multi-restart, budget until t={phase3_end - t_start:.0f}s")

    restart_configs = [
        ("perturb_0.03", lambda: _perturb(best_seq, 0.03, rng)),
        ("perturb_0.08", lambda: _perturb(best_seq, 0.08, rng)),
        ("sinusoidal", lambda: _make_profile(n_main, "sinusoidal")),
        ("edge_heavy", lambda: _make_profile(n_main, "edge_heavy")),
        ("center_heavy", lambda: _make_profile(n_main, "center_heavy")),
        ("tapered", lambda: _make_profile(n_main, "tapered")),
        ("bimodal", lambda: _make_profile(n_main, "bimodal")),
        ("perturb_0.15", lambda: _perturb(best_seq, 0.15, rng)),
    ]

    per_restart = (phase3_end - time.monotonic() - 10) / max(1, len(restart_configs))
    for i, (label, gen) in enumerate(restart_configs):
        if time.monotonic() > phase3_end - 5:
            break
        a_init = gen()
        sc_init = _score(a_init)
        if sc_init < best_score:
            update_best(a_init, sc_init, f"init_{label}")

        c_s = _full_convolve(a_init, a_init)
        max_cs = float(c_s.max())
        bs = [0.03 * max_cs, 0.01 * max_cs, 0.005 * max_cs,
              0.002 * max_cs, 0.001 * max_cs]
        remaining = min(per_restart, phase3_end - time.monotonic() - 5)
        if remaining > 10:
            a_r, sc_r = _lbfgs_optimize(a_init, time.monotonic() + remaining, n_beta_steps=8)
            update_best(a_r, sc_r, f"lbfgs_{label}")

    # --- Phase 4: Active-set polishing (1200s - 1600s) ---
    phase4_end = min(deadline, t_start + 1600.0)
    print(f"Phase 4: Active-set polish, budget until t={phase4_end - t_start:.0f}s")
    if best_seq is not None:
        a_polished, sc_polish = _active_set_polish(best_seq, phase4_end, max_iter=5000)
        update_best(a_polished, sc_polish, "active_set")

        # Second round of active-set on the polished result
        if time.monotonic() < phase4_end - 30:
            a_polished2, sc_polish2 = _active_set_polish(a_polished, phase4_end, max_iter=3000)
            update_best(a_polished2, sc_polish2, "active_set_2")

    # --- Phase 5: Try different n values (1600s - 1750s) ---
    phase5_end = min(deadline, t_start + 1750.0)
    print(f"Phase 5: Try different n, budget until t={phase5_end - t_start:.0f}s")
    if best_seq is not None:
        for target_n in [4096, 8192]:
            if time.monotonic() > phase5_end - 30:
                break
            if len(best_seq) >= target_n:
                continue
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{target_n}")

            remaining = phase5_end - time.monotonic() - 10
            if remaining > 20 and best_seq is not None:
                c_int = _full_convolve(p_interp, p_interp)
                max_ci = float(c_int.max())
                bs_int = [0.02 * max_ci, 0.01 * max_ci, 0.005 * max_ci,
                          0.002 * max_ci, 0.001 * max_ci]
                a_int, sc_int = _lbfgs_optimize(p_interp, time.monotonic() + remaining,
                                                n_beta_steps=6)
                update_best(a_int, sc_int, f"lbfgs_n{target_n}")

    # --- Phase 6: Final ultra-fine polish (1750s - deadline) ---
    if best_seq is not None and time.monotonic() < deadline - 5:
        print(f"Phase 6: Final polish")
        c_final = _full_convolve(best_seq, best_seq)
        max_cf = float(c_final.max())

        # Very fine Adam
        beta_final = [0.0001 * max_cf, 0.00005 * max_cf, 0.00002 * max_cf,
                      0.00001 * max_cf, 0.000005 * max_cf]
        remaining = deadline - time.monotonic() - 3
        if remaining > 10:
            a_fin, sc_fin = _adam_optimize(best_seq, time.monotonic() + remaining,
                                           beta_final, max_iter=100000, lr=0.003)
            update_best(a_fin, sc_fin, "final_adam")

        # Active set final
        if time.monotonic() < deadline - 5:
            a_as, sc_as = _active_set_polish(best_seq, deadline, max_iter=2000)
            update_best(a_as, sc_as, "final_active")

    # --- Final validation ---
    print(f"\n=== RESULTS ===")
    print(f"Best score: {best_score:.10f}")
    print(f"Best n: {len(best_seq) if best_seq is not None else 'N/A'}")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"Final score (internal): {sc_final:.10f}, n={len(final)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Final score (evaluator): {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
