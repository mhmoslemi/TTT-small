"""
Fast FFT-based gradient optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses logsumexp continuation with Adam optimizer, diverse initializations,
and iterative peak-refinement. Exploits O(n log n) gradient computation via FFT.
Includes symmetry exploitation and multi-restart strategy.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _obj_and_grad_a(z, beta):
    """Compute smoothed objective and gradient wrt a (not z)."""
    n = len(z)
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        obj = 2.0 * n * cm
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        obj = 2.0 * n * (cm + beta * float(np.log(s)))
        w = e / s
    # Cross-correlation h[i] = sum_k w[k]*a[k-i] via FFT
    N = _next_pow2(2 * n + n)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    h = np.fft.irfft(W * np.conj(A), N)[:2 * n - 1]
    grad_a = (4.0 * n / beta) * h
    return obj, grad_a, a, w, c


def _obj_and_grad_z(z, beta):
    """Compute smoothed objective and gradient wrt z."""
    obj, grad_a, a, w, c = _obj_and_grad_a(z, beta)
    ga_dot_a = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - ga_dot_a)
    if not np.all(np.isfinite(grad_z)):
        grad_z = np.zeros_like(z)
    return obj, grad_z

    best_a = _softmax(z)
    best_score = _score_norm(best_a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init

    for beta in beta_schedule:
        iters = iters_per_beta
        for it in range(iters):
            if time.monotonic() > deadline - 1.0:
                return best_a, best_score

            t += 1
            obj, grad_z = _obj_and_grad_z(z, beta)

            m = b1 * m + (1 - b1) * grad_z
            v = b2 * v + (1 - b2) * (grad_z * grad_z)
            mh = m / (1 - b1 ** t)
            vh = v / (1 - b2 ** t)
            z = z - lr * mh / (np.sqrt(vh) + eps)
            z = np.clip(z, -80, 80)
            z = z - z.mean()

            if (it + 1) % 50 == 0:
                a_new = _softmax(z)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()

        lr = max(0.001, lr * lr_decay)

    return best_a, best_score


def _adam_aggressive(z_init, deadline, rng, lr=0.02, beta_start=0.1, beta_end=0.001):
    """Aggressive Adam with many beta levels."""
    n = len(z_init)
    log_betas = np.linspace(np.log(beta_start), np.log(beta_end), 15)
    beta_schedule = np.exp(log_betas)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_a = _softmax(z_init)
    best_score = _score_norm(best_a)
    z = z_init.copy()

    for beta in beta_schedule:
        if time.monotonic() > deadline - 2.0:
            break
        n_iters = max(100, min(5000, int((deadline - time.monotonic()) * 2000)))
        for it in range(n_iters):
            if time.monotonic() > deadline - 1.0:
                break
            t += 1
            obj, grad_z = _obj_and_grad_z(z, beta)
            m = b1 * m + (1 - b1) * grad_z
            v = b2 * v + (1 - b2) * (grad_z * grad_z)
            mh = m / (1 - b1 ** t)
            vh = v / (1 - b2 ** t)
            z = z - lr * mh / (np.sqrt(vh) + eps)
            z = np.clip(z, -80, 80)
            z = z - z.mean()

            if (it + 1) % 20 == 0:
                a_new = _softmax(z)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
        lr = max(0.002, lr * 0.8)

    return best_a, best_score


def _peak_refinement(a_init, deadline, rng, n_iter=5000):
    """Direct peak-reduction local search.
    At each step, find the peak of conv(a,a) and reduce it by
    adjusting a in the negative gradient direction, then project."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(n_iter):
        if time.monotonic() > deadline - 1.0:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        # Find top-K peaks
        k_top = min(5, len(c))
        top_idx = np.argsort(c)[-k_top:]
        top_idx = [int(x) for x in top_idx if c[x] > 0.5 * cm]
        if not top_idx:
            top_idx = [int(np.argmax(c))]

        # Compute weighted gradient (subgradient-like)
        grad = np.zeros(n)
        for k_val in top_idx:
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * a[k_val - hi:k_val - lo + 1][::-1]
        grad /= len(top_idx)

        # Step in negative gradient direction
        step = 0.01 / (np.linalg.norm(grad) + 1e-10)
        a = a - step * grad
        # Project to simplex
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        else:
            a = np.ones(n) / n

        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        else:
            a = best_a.copy()

    return best_a, best_score


def _make_raised_cosine(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    u = (t - 0.5) * 2.0
    a = 0.5 * (1.0 + np.cos(np.pi * u * alpha))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian(n, width=0.3):
    t = np.linspace(0.0, 1.0, n)
    u = t - 0.5
    a = np.exp(-u ** 2 / (2.0 * width ** 2))
    return a / a.sum()


def _make_parabolic(n, p=2.0):
    t = np.linspace(0.0, 1.0, n)
    u = np.abs(2.0 * t - 1.0)
    a = (1.0 - u ** p)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear(n, slope=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_fejer(n):
    """Fejer-kernel-like profile (triangle in frequency domain)."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = (1.0 - u ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bumped(n, center=0.5, width=0.4):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - center) / width) ** 2)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy fast optimizer using FFT-based gradients.
    Combines Adam with logsumexp continuation, peak refinement,
    and multi-restart with diverse profiles.
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"[t={time.monotonic()-t_start:.1f}s]")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n = 4096

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm(parent)
        print(f"Parent: n={n_parent}, norm_score={sc_parent:.10f}")
        # Interpolate parent to our target n
        parent_at_n = _interp_profile(parent, n)
        sc_parent_n = _score_norm(parent_at_n)
        update_best(parent_at_n, sc_parent_n, "parent_interp")
        # Also try original length
        if n_parent != n:
            update_best(parent, sc_parent, "parent_orig")
    else:
        parent_at_n = _make_raised_cosine(n)
        print(f"No parent, using raised cosine n={n}")

    print(f"Phase 0: Diverse initialization")
    phase0_end = min(deadline, t_start + 30.0)

    # Generate diverse candidates
    candidates = []
    profiles = [
        ("raised_cos_1", _make_raised_cosine(n, 1.0)),
        ("raised_cos_0.7", _make_raised_cosine(n, 0.7)),
        ("gauss_0.25", _make_gaussian(n, 0.25)),
        ("gauss_0.35", _make_gaussian(n, 0.35)),
        ("gauss_0.45", _make_gaussian(n, 0.45)),
        ("parabolic_2", _make_parabolic(n, 2.0)),
        ("parabolic_3", _make_parabolic(n, 3.0)),
        ("parabolic_4", _make_parabolic(n, 4.0)),
        ("linear_1", _make_linear(n, 1.0)),
        ("linear_2", _make_linear(n, 2.0)),
        ("fejer", _make_fejer(n)),
        ("bumped_0.4", _make_bumped(n, 0.5, 0.4)),
        ("bumped_0.6", _make_bumped(n, 0.5, 0.6)),
    ]

    for name, p in profiles:
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm(p)
        candidates.append((sc, p, name))
        update_best(p, sc, name)

    # Also add parent
    sc_p = _score_norm(parent_at_n)
    candidates.append((sc_p, parent_at_n, "parent"))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}, top-3: "
          f"{[(c[2], f'{c[0]:.6f}') for c in candidates[:3]]}")

    # Phase 1: Adam optimization on top candidates
    print(f"Phase 1: Adam optimization on top candidates")
    phase1_end = min(deadline, t_start + 600.0)
    n_top = min(4, len(candidates))
    per_cand = (phase1_end - time.monotonic()) / max(1, n_top)

    for i in range(n_top):
        if time.monotonic() > phase1_end - 5:
            break
        sc, p, name = candidates[i]
        if sc > best_score * 1.01:
            break
        z = np.log(np.maximum(p, 1e-15))
        z = z - z.mean()
        dl = min(phase1_end - 2, time.monotonic() + per_cand)
        a_best, sc_best = _adam_optimize(
            z, dl, rng,
            beta_schedule=[0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005],
            iters_per_beta=3000,
            lr_init=0.02
        )
        update_best(a_best, sc_best, f"adam_{name}")

    print(f"Phase 1 done: best={best_score:.10f} [t={time.monotonic()-t_start:.1f}s]")

    # Phase 2: Aggressive Adam with more beta levels
    print(f"Phase 2: Aggressive Adam continuation")
    phase2_end = min(deadline, t_start + 1000.0)
    if best_seq is not None:
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        dl = phase2_end - 5
        a_best, sc_best = _adam_aggressive(
            z, dl, rng,
            lr=0.015, beta_start=0.02, beta_end=0.0003,
        )
        update_best(a_best, sc_best, "aggressive_adam")

    print(f"Phase 2 done: best={best_score:.10f} [t={time.monotonic()-t_start:.1f}s]")

    # Phase 3: Peak refinement (direct non-smooth method)
    print(f"Phase 3: Peak refinement")
    phase3_end = min(deadline, t_start + 1200.0)
    if best_seq is not None:
        dl = phase3_end - 5
        a_best, sc_best = _peak_refinement(best_seq, dl, rng, n_iter=20000)
        update_best(a_best, sc_best, "peak_refine")

    print(f"Phase 3 done: best={best_score:.10f} [t={time.monotonic()-t_start:.1f}s]")

    # Phase 4: Multi-restart with perturbation
    print(f"Phase 4: Multi-restart")
    phase4_end = min(deadline, t_start + 1500.0)
    n_restarts = 4
    per_restart = max(10.0, (phase4_end - time.monotonic()) / max(1, n_restarts))

    for i in range(n_restarts):
        if time.monotonic() > phase4_end - 5:
            break
        noise = 0.02 + 0.02 * i
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        z = z + noise * rng.normal(size=len(best_seq))
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        dl = min(phase4_end - 3, time.monotonic() + per_restart)
        a_best, sc_best = _adam_optimize(
            z, dl, rng,
            beta_schedule=[0.03, 0.01, 0.005, 0.002, 0.001],
            iters_per_beta=2000,
            lr_init=0.015
        )
        update_best(a_best, sc_best, f"restart_{i}")

    print(f"Phase 4 done: best={best_score:.10f} [t={time.monotonic()-t_start:.1f}s]")

    # Phase 5: Final polishing with very small beta
    print(f"Phase 5: Final polishing")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            dl = deadline - 5
            # Very fine beta schedule
            a_best, sc_best = _adam_optimize(
                z, dl, rng,
                beta_schedule=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
                iters_per_beta=2000,
                lr_init=0.005,
                lr_decay=0.9995
            )
            update_best(a_best, sc_best, "final_polish")

        # One more peak refinement
        if time.monotonic() < deadline - 10:
            a_best, sc_best = _peak_refinement(best_seq, deadline - 2, rng, n_iter=10000)
            update_best(a_best, sc_best, "final_peak")

    print(f"Phase 5 done: best={best_score:.10f} [t={time.monotonic()-t_start:.1f}s]")

    # Final validation and return
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_raised_cosine(n)
            s = float(final_seq.sum())
        final_seq = final_seq / s

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        if len(final_seq) <= 8192:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence exception: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_raised_cosine(n)
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n={n}, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
