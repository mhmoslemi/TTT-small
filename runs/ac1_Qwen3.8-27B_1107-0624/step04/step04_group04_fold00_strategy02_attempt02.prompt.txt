def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):

    s = float(a.sum())

        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_smooth_grad(a, beta, n, top_k=0):
    """Compute softmax-weights and gradient for the logsumexp-smoothed max."""
    c = _full_convolve(a, a)
    cm = float(c.max())

def _z_gradient(z, beta, n, top_k=0):
    """Gradient of the smoothed objective with respect to z (logit space)."""
    a = _softmax_fn(z)
    w, h, W, c, cm = _compute_smooth_grad(a, beta, n, top_k)
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _z_objective(z, beta, n, top_k=0):
    """Smoothed objective value in logit space."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_norm = c_top - c_top.max()
        log_sum = float(np.log(np.sum(np.exp(c_norm / beta))))
        return 2.0 * n * (c_top.max() + beta * log_sum)
    else:
        c_shifted = (c - cm) / beta
        log_sum = float(np.log(np.sum(np.exp(c_shifted))))
        return 2.0 * n * (cm + beta * log_sum)

def _adam_polish(a_init, deadline, lr=0.05, top_k=0, n_max_iter=50000):
    """Adam optimizer with adaptive beta for the smoothed max objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-10, 0.01 * max_c)

    # Adaptive top_k: start small, grow over time
    if top_k == 0:
        adaptive_k = True
        cur_k = 3
    else:
        adaptive_k = False
        cur_k = top_k

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break

        t += 1

        a = _softmax_fn(z)
        use_k = cur_k if adaptive_k else top_k
        grad = _z_gradient(z, beta, n, use_k)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:

            best_a = a.copy()

        if t % 100 == 0:

                cur_k = min(cur_k + 2, 30)

        if t % 2000 == 0:
            if best_score < 1.51:
                lr = max(0.003, lr * 0.97)

    return best_a, best_score


def _kkt_balance(a_init, deadline, max_iter=5000):
    """KKT-based balancing: at optimum, weighted sum of active gradients is constant."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        # Active set: indices where c is within 1% of max
        threshold = cm * 0.995
        active = np.where(c >= threshold)[0]

        if len(active) == 0:
            break

        # Compute the "balanced" gradient direction
        # For each active k, the gradient of c(k) w.r.t. a is 2*a_rev_shifted
        # We want to find a direction d such that for all active k:
        #   c(k + d) < c(k) (all active peaks decrease)
        # Approximate: move a in direction that equalizes the active c values

        # Compute h_i = (1/|A|) * sum_{k in A} a[(k-i) mod n] (circular shift)
        # Actually: h_i = (1/|A|) * sum_{k in A} a[k-i] if 0<=k-i<n
        h = np.zeros(n)
        for k_idx in active:
            k_val = int(k_idx)
            # a[k_val - i] for i = 0..n-1, where valid means 0 <= k_val-i <= n-1
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                h[lo:hi+1] += a[k_val - hi:k_val - lo + 1][::-1]
        h /= len(active)

        W = float(np.dot(h, a))  # should be cm/|A| approximately

        # Update direction: reduce a where h > W, increase where h < W
        # But maintain non-negativity via logit space
        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()

        # Step size: adaptive based on the "spread" of active values
        active_spread = float(c[active].max() - c[active].min()) + 1e-15
        step = 0.01 * active_spread / (cm + 1e-15)
        step = min(step, 0.05)

        z = z - step * (h - W) / (np.abs(h - W).max() + 1e-15)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a_new = _softmax_fn(z)

        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce step and retry
            step *= 0.5
            z = np.log(np.maximum(a, 1e-15))
            z = z - z.mean()
            z = z - step * (h - W) / (np.abs(h - W).max() + 1e-15)
            z = np.clip(z, -80, 80)
            z = z - z.mean()
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new

        if it > 100 and (best_score - sc if np.isfinite(sc) else 0) < 1e-10:
            break

    return best_a, best_score


def _make_exp_profile(n, alpha):
    """Exponential decay profile: a_i = (1-alpha)^i, normalized."""
    r = 1.0 - alpha
    a = r ** np.arange(n)
    return a / a.sum()


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power_profile(n, p):
    """Power profile: a_i = (i+1)^p, normalized."""
    a = (np.arange(n) + 1.0) ** p
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
