3. **Multiple restarts**: I'll try several different initializations (Hann, Gaussian, Blackman, etc.) and keep the best.

4. **Adaptive step size**: I'll monitor the objective and adjust the step size accordingly.

5. **Active set refinement**: In the final phase, I'll use a direct subgradient approach on the active set.

Let me also consider: what if the optimal sequence has a specific structure, like being a "window" multiplied by a "chirp" or some other modulated signal?

For non-negative sequences, a natural structure is:
`a_i = w_i * (1 + eps * cos(2*pi*f*i + phi))`

where `w_i` is a window (Hann, Gaussian, etc.) and the cosine modulates it slightly to flatten the autocorrelation.

This gives us two hyperparameters (f, phi) and the window shape to explore.

"""
Multi-phase optimizer for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses Hann/Gaussian window initialization, adaptive log-sum-exp smoothing,
Nesterov-accelerated gradient descent, and multi-restart strategy.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _gradient_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _objective_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _make_hann(n):
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian(n, sigma_frac=0.3):
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_blackman(n):
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_modulated(n, f, phi, window='hann'):
    i = np.arange(n, dtype=np.float64)
    if window == 'hann':
        w = 0.5 * (1.0 + np.cos(2.0 * np.pi * i / (n - 1)))
    elif window == 'gaussian':
        center = (n - 1) / 2.0
        sigma = 0.3 * n
        w = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    else:
        w = np.ones(n)
    eps = 0.3
    a = w * (1.0 + eps * np.cos(2.0 * np.pi * f * i / n + phi))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _nesterov_optimize(a_init, deadline, beta_fracs, lr_init, mom_init, check_interval=50):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())

    lr = lr_init
    mom = mom_init
    t = 0

    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-12, beta_frac * max_c)

        while time.monotonic() < deadline:
            t += 1
            grad_y = _gradient_smooth(y, beta)
            if not np.all(np.isfinite(grad_y)):
                grad_y = _gradient_smooth(z, beta)

            z_new = z - lr * grad_y
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -50, 50)

            y_new = z_new + mom * (z_new - z)
            y_new = y_new - y_new.mean()

            z = z_new
            y = y_new

            if t % check_interval == 0:
                a_curr = _softmax_fn(z)
                sc = _score_norm_fn(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()

            if t % 200 == 0:
                mom = min(0.999, mom + 0.0003)

        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        max_c = float(c.max())
        lr = max(0.001, lr * 0.6)

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _adam_optimize(a_init, deadline, lr, beta_frac, check_interval=100):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, beta_frac * max_c)

    while time.monotonic() < deadline:
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()

        if t % check_interval == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            beta = max(1e-12, beta * 0.85)
            c_new = _full_convolve(a_curr, a_curr)
            max_c = float(c_new.max())

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _active_set_polish(a_init, deadline, lr_init=0.01, check_interval=50):
    """Direct subgradient on the active set with Nesterov momentum."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    lr = lr_init
    mom = 0.9
    t = 0

    while time.monotonic() < deadline:
        t += 1
        a = _softmax_fn(y)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Active set: lags within 0.1% of max
        threshold = cm * (1.0 - 1e-3)
        active = c >= threshold
        n_active = int(np.sum(active))
        
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        z_new = z - lr * grad
        z_new = z_new - z_new.mean()
        z_new = np.clip(z_new, -50, 50)

        y_new = z_new + mom * (z_new - z)
        y_new = y_new - y_new.mean()

        z = z_new
        y = y_new

        if t % check_interval == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            else:
                lr = lr * 0.8

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _lbfgs_phase(z_init, beta, deadline, n):
    """Run L-BFGS-B on the smoothed objective."""
    result = minimize(
        _objective_smooth, z_init, args=(beta,),
        method='L-BFGS-B', jac=_gradient_smooth,
        options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-12,
                 'maxls': 50, 'maxfun': 2000}
    )
    if result.x is not None and np.all(np.isfinite(result.x)):
        z = result.x.copy()
        a = _softmax_fn(z)
        return a
    return None
