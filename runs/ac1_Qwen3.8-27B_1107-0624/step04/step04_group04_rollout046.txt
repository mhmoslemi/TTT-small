def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)

def _softmax(z):

    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:

    return e / s

def _project_simplex(x):
    """Project x onto the probability simplex {x : sum(x)=1, x>=0}."""
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)

        s = float(e.sum())
        if s < 1e-300:
            idx = int(np.argmax(c))
            w = np.zeros_like(c)
            w[idx] = 1.0
        else:
            w = e / s
    
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n-1:2*n-1]
    
    # Gradient of 2*n*W w.r.t. a is 2*n*h (wait, let me reconsider)
    # W = sum_k w[k] * c[k]
    # dW/da[j] = sum_k w[k] * dc[k]/da[j] = sum_k w[k] * 2*a[k-j] (if valid)
    # = 2 * (w * a_rev)[j + n - 1] = 2 * h[j]
    # d/dz of W = (I - a a^T) @ 2*n*h_... 
    # Wait, the objective is 2*n*W, so d(2nW)/da = 2n * 2 * h = 4n * h
    # d(2nW)/dz = (I - aa^T) @ 4n*h
    # But we need to be careful: the objective is 2n * (max + beta*logsumexp), 
    # and the gradient of the beta*logsumexp part is 2n * beta * (w - W/c_max... )
    # Actually, let me just use the standard result:
    # grad_z = (I - aa^T) @ grad_a
    # where grad_a = 4*n*h (for the full logsumexp smoothing)
    
    grad_a = 4.0 * n * h
    a_dot = float(np.dot(a, grad_a))
    grad_z = grad_a - a * a_dot
    
    return grad_z

def _refined_subgradient_descent(a_init, deadline, rel_tol_start=0.1, 
                                   rel_tol_end=1e-8, momentum=0.5, lr_start=0.05):
    """
    Nesterov-momentum projected subgradient descent with decreasing active set tolerance.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    v = np.zeros(n)
    t = 0
    lr = lr_start
    
    while time.monotonic() < deadline:
        t += 1
        # Progress-based tolerance
        progress = min(1.0, t / 10000)
        rel_tol = rel_tol_start * (rel_tol_end / rel_tol_start) ** progress
        
        # Compute subgradient
        g, active, c, cm = _active_set_subgradient(a, rel_tol)
        
        # Tangent space projection (subtract mean to maintain sum=1)
        g_tan = g - g.mean()
        
        # Nesterov update
        v = momentum * v + lr * g_tan
        a_new = a - v
        a_new = _project_simplex(a_new)
        
        # Check improvement
        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce step
            lr *= 0.8
            a_new = a - 0.1 * v
            a_new = _project_simplex(a_new)
            if np.isfinite(_score_norm(a_new)) and _score_norm(a_new) < best_score:
                best_score = _score_norm(a_new)
                best_a = a_new.copy()
                a = a_new
        
        # Decay momentum
        if t % 500 == 0:
            momentum = max(0.1, momentum * 0.95)
            lr = max(0.001, lr * 0.98)
    
    return best_a, best_score

def _kkt_solve(a_init, deadline, tol=1e-10, max_iter=1000):
    """
    KKT-based refinement: iterate until the active set gradients are constant 
    (on the support of a).
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Active set: within 0.1% of max
        thresh = cm * 0.999
        active = np.where(c >= thresh)[0]
        
        if len(active) < 1:
            break
        
        # Compute average gradient over active set
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / len(active)
        w_conv = _full_convolve(w, a_rev)
        g = 2.0 * w_conv[n-1:2*n-1]
        
        # Check KKT: g should be constant on support(a)
        support = np.where(a > 1e-10)[0]
        if len(support) == 0:
            break
        
        g_support = g[support]
        g_var = float(g_support.max() - g_support.min())
        
        if g_var < tol * cm:
            # KKT conditions approximately satisfied
            break
        
        # Newton-like step: adjust a to make g constant on support
        # d(g)/da is the Hessian of the average active quadratic
        # Approximate: use a damping step
        g_tan = g - g.mean()
        step = 0.01 / (np.abs(g_tan).max() + 1e-15)
        
        a_new = a - step * g_tan
        a_new = _project_simplex(a_new)
        
        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            step *= 0.1
            a_new = a - step * g_tan
            a_new = _project_simplex(a_new)
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
    
    return best_a, best_score

def _make_hann_window(n):
    a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    return a / a.sum()

def _make_gauss_window(n, sigma_ratio=0.3):
    center = (n - 1) / 2.0
    x = np.arange(n) - center
    a = np.exp(-x**2 / (2 * (sigma_ratio * n) ** 2))
    return a / a.sum()

def _make_triangular_window(n):
    center = (n - 1) / 2.0
    x = np.abs(np.arange(n) - center) / (center + 1)
    a = np.maximum(0, 1 - x)
    return a / a.sum()

def _make_flat_edges(n, edge_ratio=0.1):
    edge = int(edge_ratio * n)
    a = np.ones(n)
    for i in range(edge):
        a[i] = (i + 1) / (edge + 1)
        a[n-1-i] = (i + 1) / (edge + 1)
    return a / a.sum()
