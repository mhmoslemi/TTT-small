"""
Optimized step function search using adaptive smooth-max Adam with 
intensive peak chasing, LP refinement, and multi-scale exploration.
Focuses on aggressive peak reduction with careful temperature scheduling.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _score(p):
    n = len(p)
    c = _full_convolve(p)
    s = float(p.sum())
    if s < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return None
    return a / s

def _smooth_max_weights(c, beta):
    """Returns (smooth_max_value, weights) for the smooth max of c."""
    mval = float(c.max())
    if beta < 1e-12:
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = np.zeros_like(c)
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def _gradient_from_weights(w, p):
    """Compute gradient of weighted sum of convolution w.r.t. p."""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def _adam_optimize(p0, deadline, rng, lr=0.04, beta_init_frac=0.10,
                   beta_decay=0.95, patience=250):
    """Adam optimization in log-space with smooth max."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    
    z = np.log(p0)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    p = p0.copy()
    c = _full_convolve(p)
    beta = beta_init_frac * c.max()
    
    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * c.max())
    no_imp = 0
    
    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)
        g = _gradient_from_weights(w, p)
        
        # Project gradient to stay on the simplex
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            continue
        
        t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))

        z_shifted = z - z.max()
        e = np.exp(z_shifted)
        p = e / e.sum()
        
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

            best_z = z.copy()
            no_imp = 0

            z = best_z + noise * rng.normal(size=n)

            e = np.exp(z_shifted)

            c = _full_convolve(p)
            beta = beta_init_frac * c.max()
            m.fill(0.0)

def _peak_chase(p, deadline, eta_init=0.06, top_k=3):
    """Aggressive multiplicative peak chasing."""
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size
    c = _full_convolve(p)
    best_p = p.copy()
    best_score = float(2.0 * n * c.max())
    eta = eta_init
    no_imp = 0
    
    while time.monotonic() < deadline:
        M = float(c.max())
        score = float(2.0 * n * M)
        
        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Find top peaks
        sorted_idx = np.argsort(c)[::-1]
        top_idxs = sorted_idx[:top_k]
        peak_values = c[top_idxs]
        
        # Weight peaks by their value (softmax-like)
        peak_weight_scale = 0.1 * M
        pw = np.exp((peak_values - peak_values[0]) / peak_weight_scale)
        pw = pw / pw.sum()
        
        # Compute heat
        heat = np.zeros(n)
        for k in top_idxs:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            partner = k - np.arange(i_start, i_end + 1)
            heat[i_start:i_end + 1] += pw[list(top_idxs).index(k)] * p[partner]
        
        heat_max = float(heat.max())
        if heat_max < 1e-15:
            break
        heat = heat / heat_max

def _lp_solve(a, M):
    """LP: maximize sum(b) s.t. (a*b)[k] <= M, b >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

        if s_b < 1e-10:
            break

def _interp_to_size(p, n_new):
    """Interpolate a sequence to a new size."""
    n_old = len(p)
    if n_old == n_new:

    p_new = np.interp(x_new, x_old, p)
    p_new = np.maximum(p_new, 1e-10)
    return p_new / p_new.sum()

Optimized step function search combining L-BFGS with smooth-max objective,
adaptive peak chasing via multiplicative weights, and LP-based refinement.
Uses multi-phase strategy with careful temperature scheduling and restarts.
"""
import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(p):
    n = len(p)
    c = _full_convolve(p)
    s = float(p.sum())
    if s < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return None
    return a / s


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-12:
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = np.zeros_like(c)
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _gradient_from_weights(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

    e = np.exp(z)

            best_score = sc

    return best_p, best_score


def _adam_optimize(p0, deadline, rng, lr=0.03, beta_init_frac=0.06,
                   beta_decay=0.95, patience=200, base_lr=None):
    """Adam in log-space with smooth max."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    z = np.log(p0)
    z -= z.mean()
    if base_lr is None:
        base_lr = lr
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    p = p0.copy()
    c = _full_convolve(p)
    beta = max(1e-6, beta_init_frac * c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * c.max())
    no_imp = 0

    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)
        g = _gradient_from_weights(w, p)
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_imp += 10
            continue

        t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= z.mean()
        p = _z_to_p(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        if t % 25 == 0:
            beta = max(1e-6, beta * beta_decay)

        if no_imp > patience:
            noise = 0.04 + 0.08 * min(no_imp / patience, 2.0)
            z = best_z + noise * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= z.mean()
            p = _z_to_p(z)
            c = _full_convolve(p)
            beta = max(1e-5, beta_init_frac * c.max())
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _peak_chase(p, deadline, eta_init=0.05, top_k=3):
    """Multiplicative peak chasing."""
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size
    c = _full_convolve(p)
    best_p = p.copy()
    best_score = float(2.0 * n * c.max())
    eta = eta_init
    no_imp = 0

    while time.monotonic() < deadline:
        M = float(c.max())
        score = float(2.0 * n * M)

        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            no_imp = 0
        else:
            no_imp += 1

        sorted_idx = np.argsort(c)[::-1]
        top_idxs = sorted_idx[:top_k]
        peak_vals = c[top_idxs]

        pw_scale = max(0.05 * M, 1e-10)
        pw = np.exp((peak_vals - peak_vals[0]) / pw_scale)
        pw = pw / pw.sum()

        heat = np.zeros(n)
        for idx, k in enumerate(top_idxs):
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            partner = k - np.arange(i_start, i_end + 1)
            heat[i_start:i_end + 1] += pw[idx] * p[partner]

        heat_max = float(heat.max())
        if heat_max < 1e-15:
            break
        heat = heat / heat_max

        factor = 1.0 - eta * heat
        factor = np.maximum(factor, 0.05)
        p = p * factor
        p = np.maximum(p, 1e-15)
        p = p / p.sum()
        c = _full_convolve(p)

        if no_imp > 80:
            eta = max(0.008, eta * 0.92)
        elif no_imp == 0:
            eta = min(0.12, eta * 1.03)

    return best_p, best_score


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(p, deadline, t_init=0.25, max_solves=30):
    """LP-based refinement with decreasing target."""
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = p / p.sum()
    n = p.size
    c = _full_convolve(p)
    M = float(c.max())
    best_p = p.copy()
    best_score = float(2.0 * n * M)
    t = t_init
    count = 0

    while time.monotonic() < deadline and count < max_solves:
        count += 1
        b = _lp_solve(p, M)
        if b is None:
            b = _lp_solve(p, M * 1.01)
            if b is None:
                break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        p_new = (1.0 - t) * p + t * b_norm
        p_new = np.maximum(p_new, 1e-15)
        p_new = p_new / p_new.sum()
        sc = float(2.0 * n * _full_convolve(p_new).max())
        if sc < best_score - 1e-14:
            best_score = sc
            best_p = p_new.copy()
        p = p_new
        c = _full_convolve(p)
        M = float(c.max()) * 0.997
        t = max(0.08, t * 0.93)

    return best_p, best_score


def _make_cosine_profile(n, power=0.8):
    t = np.linspace(0.0, np.pi, n)
    a = np.abs(np.cos(t)) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_to_size(p, n_new):
    n_old = len(p)
    if n_old == n_new:
        return p.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x_new, x_old, p)
    p_new = np.maximum(p_new, 1e-10)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: L-BFGS + Adam + peak chasing + LP refinement.
    Multi-phase with adaptive time allocation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, label=""):
        nonlocal best_seq, best_score
        if p is not None and len(p) > 0:
            sc = _score(p)
            if np.isfinite(sc) and sc < best_score:
                best_seq = np.asarray(p, dtype=np.float64).copy()
                best_score = float(sc)
                print(f"  BEST: {best_score:.10f} n={p.size} [{label}] t={time.monotonic()-t_start:.0f}s")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 10:
            parent_raw = np.maximum(parent_raw, 0.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n_init = parent.size if parent is not None else 1024
    print(f"Parent n={n_init}, budget={budget_s}s")

    if parent is not None:
        par_sc = _score(parent)
        print(f"Parent score: {par_sc:.10f}")
        _update(parent, 'parent')

    # Phase 1: L-BFGS (fast initial convergence)
    p1_end = t_start + 300.0
    print(f"Phase 1: L-BFGS...")
    if parent is not None:
        p1, sc1 = _lbfgs_optimize(parent, p1_end, beta_frac=0.04, maxfun=6000)
        _update(p1, 'lbfgs1')
    else:
        p_init = _make_cosine_profile(1024, 0.8)
        p1, sc1 = _lbfgs_optimize(p_init, p1_end, beta_frac=0.05, maxfun=6000)
        _update(p1, 'lbfgs_init')

    # Phase 2: Adam refinement (medium convergence)
    p2_end = t_start + 700.0
    print(f"Phase 2: Adam, best={best_score:.10f}")
    if best_seq is not None:
        p2, sc2 = _adam_optimize(best_seq, p2_end, rng, lr=0.025,
                                  beta_init_frac=0.04, beta_decay=0.96, patience=180)
        _update(p2, 'adam1')

    # Phase 3: Peak chasing (direct attack on bottleneck)
    p3_end = t_start + 1000.0
    print(f"Phase 3: Peak chase, best={best_score:.10f}")
    if best_seq is not None:
        p3, sc3 = _peak_chase(best_seq, p3_end, eta_init=0.05, top_k=3)
        _update(p3, 'peak_chase')

    # Phase 4: Fine Adam (low LR, low temperature)
    p4_end = t_start + 1250.0
    print(f"Phase 4: Fine Adam, best={best_score:.10f}")
    if best_seq is not None:
        p4, sc4 = _adam_optimize(best_seq, p4_end, rng, lr=0.01,
                                  beta_init_frac=0.02, beta_decay=0.97, patience=120)
        _update(p4, 'adam_fine')

    # Phase 5: LP refinement
    p5_end = t_start + 1450.0
    print(f"Phase 5: LP, best={best_score:.10f}")
    if best_seq is not None and best_seq.size <= 2048:
        p5, sc5 = _lp_refine(best_seq, p5_end, t_init=0.25, max_solves=20)
        _update(p5, 'lp')

    # Phase 6: Try 2x size
    p6_end = t_start + 1650.0
    print(f"Phase 6: 2x size, best={best_score:.10f}")
    if best_seq is not None and best_seq.size <= 1024 and time.monotonic() < p6_end:
        n_big = min(best_seq.size * 2, 2048)
        p_big = _interp_to_size(best_seq, n_big)
        sc_big = _score(p_big)
        print(f"  Interp n={n_big}: {sc_big:.10f}")
        _update(p_big, f'interp_{n_big}')
        remaining = p6_end - time.monotonic() - 10
        if remaining > 60:
            p6, sc6 = _adam_optimize(p_big, time.monotonic() + remaining, rng,
                                      lr=0.02, beta_init_frac=0.04,
                                      beta_decay=0.96, patience=150)
            _update(p6, f'adam_{n_big}')

    # Phase 7: Final L-BFGS polish on best found
    print(f"Phase 7: Final L-BFGS, best={best_score:.10f}")
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = min(deadline - time.monotonic() - 5, 120.0)
        if remaining > 40:
            # Determine best size to polish
            polish_seq = best_seq
            p7, sc7 = _lbfgs_optimize(polish_seq, time.monotonic() + remaining,
                                       beta_frac=0.02, maxfun=3000)
            _update(p7, 'lbfgs_final')

    # Phase 8: Last Adam micro-polish
    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic() - 3
        if remaining > 20:
            p8, sc8 = _adam_optimize(best_seq, time.monotonic() + remaining, rng,
                                      lr=0.005, beta_init_frac=0.01,
                                      beta_decay=0.98, patience=60)
            _update(p8, 'adam_final')

    # Final validation
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 0.01:
            final = final / s
        else:
            final = _make_cosine_profile(1024, 0.8)
        final_sc = _score(final)
        print(f"FINAL: score={final_sc:.10f} n={final.size}")
        try:
            ev = evaluate_sequence([float(x) for x in final])
            print(f"EVALUATE: {ev:.10f}")
        except Exception:
            pass
        return [float(x) for x in final]

    fb = _make_cosine_profile(1024, 0.8)
    print("FALLBACK n=1024")
    return [float(x) for x in fb]
