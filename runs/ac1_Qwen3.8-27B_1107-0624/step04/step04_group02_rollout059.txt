Uses FFT-based convolution, log-sum-exp continuation with L-BFGS,
peak-targeting local search, geometric profile initialization,
and multi-length optimization with aggressive time management.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    """Assumes sum(a)=1, returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

    aa = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(wa * aa, N)
    h = h_full[n-1:2*n-1]  # h[0:n] = h_0, ..., h_{n-1}
    
    grad = 4.0 * n * a * h
    return obj, grad, w, c, cm


def _make_geometric_profile(n, r):
    a = r ** np.arange(n)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u**2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_exp_taper(n, alpha):
    """Exponential taper: a[i] = exp(-alpha * (i/(n-1) - 0.5)^2) (Gaussian) or one-sided exp."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-alpha * (t - 0.3))  # asymmetric exponential
    return a / a.sum()


def _make_bell_asymmetric(n, center, width):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - center) / width)**2)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _run_lbfgs_continuation(a_init, deadline, rng, max_wall_time=None, betas=None):
    """L-BFGS with log-sum-exp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if betas is None:
        beta_starts = [0.1, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.00075, 0.0004, 0.0002, 0.0001]
        betas = [b * max_c0 for b in beta_starts]

    if max_wall_time is not None:
        per_level = max(3.0, max_wall_time / len(betas))
    else:
        per_level = 30.0

    for beta in betas:
        if time.monotonic() > deadline - 2.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        def obj_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            return 2.0 * n * (cm_ + beta * float(np.log(s_)))

        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            # Gradient w.r.t. z: d obj / dz_j = d obj / da_i * da_i / dz_j
            # da_i / dz_j = a_i * (delta_{ij} - a_j)
            # d obj / da_i = 4*n * h_i (from _convolution_and_gradient)
            # So d obj / dz_j = sum_i 4*n*h_i * a_i*(delta_{ij} - a_j)
            # = 4*n*(h_j*a_j - a_j * sum_i h_i*a_i)
            # = 4*n*a_j*(h_j - sum_i h_i*a_i)
            
            N = _next_pow2(2 * n)
            wa = np.fft.rfft(w, N)
            aa = np.fft.rfft(a_[::-1], N)
            h_full = np.fft.irfft(wa * aa, N)
            h = h_full[n-1:2*n-1]
            
            W = float(np.dot(h, a_))
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g

        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 2000, 'ftol': 1e-20, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 10000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception as ex:
            pass

    return best_a, best_score


def _peak_shaving(a_init, deadline, rng, n_peaks=5):
    """Targeted peak reduction: iteratively push down the top peaks."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for it in range(5000):
        if time.monotonic() > deadline - 1.0:
            break

        c = _full_convolve(a, a)
        # Find top peaks
        top_idx = np.argsort(c)[-n_peaks:]
        
        # Compute subgradient of max at the top peaks
        grad = np.zeros(n)
        for k_val in top_idx:
            k_val = int(k_val)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                grad[lo:hi+1] += 2.0 * a[k_val - hi:k_val - lo + 1][::-1]
        grad /= len(top_idx)
        
        # Project to sum=0 (to maintain normalization)
        grad -= grad.mean()
        
        # Optional: add small random perturbation to escape local minima
        if rng.random() < 0.1:
            grad += 0.01 * rng.normal(size=n)
            grad -= grad.mean()
        
        # Take a step
        step = 0.01
        a_trial = a + step * grad / (np.linalg.norm(grad) + 1e-10)
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial = a_trial / a_trial.sum()
        
        sc = _score_norm_fn(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
        else:
            # Try smaller step
            step = 0.001
            a_trial = a + step * grad / (np.linalg.norm(grad) + 1e-10)
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial = a_trial / a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

    return best_a, best_score


def _local_search_block(a_init, deadline, rng):
    """Local search: perturb random blocks."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    temp = 0.1
    temp_decay = 0.999

    for it in range(20000):
        if time.monotonic() > deadline - 1.0:
            break

        block_size = int(rng.integers(2, max(3, min(50, n // 10))))
        start = int(rng.integers(0, n - block_size))

        a_trial = a.copy()
        # Multiplicative perturbation
        perturbation = rng.uniform(0.9, 1.1, size=block_size)
        a_trial[start:start+block_size] *= perturbation
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial /= a_trial.sum()

        sc = _score_norm_fn(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
        elif rng.random() < np.exp(-(sc - best_score) / max(temp, 1e-10)):
            a = a_trial.copy()
        temp *= temp_decay

    return best_a, best_score
