Let me think about a completely different angle: **What if we formulate this as a eigenvalue problem or use matrix methods?**

The convolution `c[k] = sum_i a[i]*a[k-i]` can be written as `a^T T_k a` where `T_k` is the " Toeplitz-like" matrix with 1s on the k-th anti-diagonal.

`min_a max_k a^T T_k a = max_{lambda in simplex} min_a sum_k lambda_k * a^T T_k a`

The inner minimization: `min_a sum_k lambda_k * a^T T_k a` subject to `sum(a) = 1`, `a >= 0`.

`sum_k lambda_k * a^T T_k a = a^T (sum_k lambda_k T_k) a = a^T Q(a, lambda) a`

where `Q = sum_k lambda_k T_k`. The matrix `Q` is a symmetric matrix (each `T_k` is symmetric).

So the inner problem is: `min a^T Q a` subject to `sum(a) = 1`, `a >= 0`.

If `Q` is PSD, this is a convex QP. The optimum (ignoring non-negativity) is at `a = Q^{-1} * 1 / (1^T Q^{-1} * 1)` (if Q is invertible). With non-negativity, it's a box-constrained QP.

This gives us a **bilevel optimization** or **Sion's minimax** approach:
- Outer: maximize over `lambda` in the simplex (dimension 2n-2)
- Inner: solve a QP (dimension n)

This is interesting but the outer problem is high-dimensional.

"""
Hybrid LP + FFT-gradient optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Primary method: LP-based alternating optimization (fix one convolution factor,
solve LP for the other). Secondary: FFT-accelerated smoothed gradient descent
with Adam. Includes multi-profile init, backtracking line search, and fine polish.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_unnorm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_solve(f, M, n):
    A = _build_A(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 30, 'presolve': True})
        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def _grad_smoothed(a, w, n):
    N = _next_pow2(3 * n)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    X_corr = np.fft.irfft(A * np.conj(W), N)
    X_conv = np.fft.irfft(A * W, N)
    idx = (-np.arange(n)) % N
    term1 = X_corr[idx]
    term2 = X_conv[0:n]
    grad_a = 2.0 * n * (term1 + term2)
    return grad_a


def _adam_smoothed(a_init, deadline, beta_schedule, max_iter=100000, lr=0.01):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_unnorm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    cur_lr = lr

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_phase = 0
        while it_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_phase += 1
            t += 1
            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shift = (c - cm) / beta
            e = np.exp(c_shift)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            grad_a = _grad_smoothed(a, w, n)
            if not np.all(np.isfinite(grad_a)):
                break
            mean_g = float(np.dot(a, grad_a))
            grad_z = a * (grad_a - mean_g)
            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            z = z - cur_lr * mh / (np.sqrt(vh) + eps)
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            if t % 50 == 0:
                sc = _score_unnorm(_softmax(z))
                if sc < best_score:
                    best_score = sc
                    best_a = _softmax(z).copy()
            if it_phase % 5000 == 0:
                cur_lr = max(0.001, cur_lr * 0.97)

    a_final = _softmax(z)
    sc_f = _score_unnorm(a_final)
    if sc_f < best_score:

    return best_a

        ]
        per_time = phase3_end - time.monotonic() - 10
        p_best = _adam_smoothed(best_seq, time.monotonic() + per_time,
                                beta_schedule, max_iter=200000, lr=0.012)
        update_best(p_best, _score(p_best), "gradient_polish")

        # Second gradient pass from perturbed
        if time.monotonic() < phase3_end - 30:
            p_pert = _perturb(best_seq, 0.03, rng)
            c_p = _full_convolve(p_pert, p_pert)
            max_cp = float(c_p.max())
            bs2 = [0.01 * max_cp, 0.005 * max_cp, 0.002 * max_cp,
                   0.001 * max_cp, 0.0005 * max_cp, 0.0002 * max_cp]
            per_time2 = phase3_end - time.monotonic() - 5
            p_best2 = _adam_smoothed(p_pert, time.monotonic() + per_time2,
                                     bs2, max_iter=100000, lr=0.008)
            update_best(p_best2, _score(p_best2), "gradient_perturb")

    print(f"Phase 3 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: More LP iterations from improved solution (1200s - 1500s) ===
    phase4_end = min(deadline, t_start + 1500.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        f = best_seq.copy()
        n = len(f)
        c_f = _full_convolve(f, f)
        M = float(c_f.max())
        sc_f = _score_unnorm(f)
        no_improve = 0
        lp_count = 0

        while time.monotonic() < phase4_end and no_improve < 30:
            g = _lp_solve(f, M, n)
            lp_count += 1
            if g is None:
                no_improve += 5
                continue
            g_sum = float(g.sum())
            if g_sum < 1e-15:
                no_improve += 5
                continue
            g_norm = g / g_sum

            improved = False
            for tval in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
                f_try = (1.0 - tval) * f + tval * g_norm
                f_try = np.maximum(f_try, 0.0)
                f_try = f_try / f_try.sum()
                sc_try = _score_unnorm(f_try)
                if sc_try < sc_f * (1.0 - 1e-10):
                    f = f_try
                    sc_f = sc_try
                    c_f = _full_convolve(f, f)
                    M = float(c_f.max())
                    improved = True
                    no_improve = 0
                    break
            if not improved:
                no_improve += 1

            update_best(f, _score(f), f"lp_phase4_{lp_count}")

    print(f"Phase 4 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Try larger n (1500s - 1700s) ===
    phase5_end = min(deadline, t_start + 1700.0)
    if best_seq is not None and len(best_seq) < 4096 and time.monotonic() < phase5_end - 30:
        target_n = 4096
        p_interp = _interp_profile(best_seq, target_n)
        update_best(p_interp, _score(p_interp), f"interp_n{target_n}")

        if time.monotonic() < phase5_end - 60:
            c_int = _full_convolve(p_interp, p_interp)
            max_ci = float(c_int.max())
            M_int = max_ci
            n_int = target_n
            f_int = p_interp.copy()
            sc_int = _score_unnorm(f_int)
            no_imp = 0
            for _ in range(50):
                if time.monotonic() > phase5_end - 30 or no_imp > 20:
                    break
                g = _lp_solve(f_int, M_int, n_int)
                if g is None:
                    no_imp += 5
                    continue
                gs = float(g.sum())
                if gs < 1e-15:
                    no_imp += 5
                    continue
                gn = g / gs
                imp = False
                for tv in [1.0, 0.5, 0.3, 0.1]:
                    ft = (1.0 - tv) * f_int + tv * gn
                    ft = np.maximum(ft, 0.0) / np.maximum(ft).sum()
                    ft = ft / ft.sum()
                    st = _score_unnorm(ft)
                    if st < sc_int * (1.0 - 1e-10):
                        f_int = ft
                        sc_int = st
                        c_int = _full_convolve(f_int, f_int)
                        M_int = float(c_int.max())
                        imp = True
                        no_imp = 0
                        break
                if not imp:
                    no_imp += 1
                update_best(f_int, _score(f_int), f"opt_n{target_n}")

    print(f"Phase 5 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final fine polish (1700s - deadline) ===
    if best_seq is not None and time.monotonic() < deadline - 5:
        c_final = _full_convolve(best_seq, best_seq)
        max_cf = float(c_final.max())
        beta_final = [0.0001 * max_cf, 0.00005 * max_cf, 0.00002 * max_cf, 0.00001 * max_cf]
        per_time = deadline - time.monotonic() - 3
        if per_time > 10:
            p_best = _adam_smoothed(best_seq, time.monotonic() + per_time,
                                    beta_final, max_iter=50000, lr=0.003)
            update_best(p_best, _score(p_best), "final_polish")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
