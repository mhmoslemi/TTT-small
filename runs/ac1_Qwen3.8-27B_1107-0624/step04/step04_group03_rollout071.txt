def _next_pow2(n):

def _full_convolve(a, b):

    fa = np.fft.rfft(a, N)

    return r

def _spectral_project(a, n_peaks=20):

    n = len(a)
    A = np.fft.rfft(a)
    mag = np.abs(A).copy()
    # Exclude DC (bin 0)
    if len(mag) > 1:
        sorted_mag = np.sort(mag[1:])[::-1]
        if n_peaks >= len(sorted_mag):
            T = sorted_mag[-1]
        else:
            T = sorted_mag[n_peaks - 1]
        T = max(T, 1e-15)
        # Soft-clip non-DC bins
        mag[1:] = np.minimum(mag[1:], max(mag[1:].max() * 0.95, T))
    # Reconstruct
    phase = A / np.maximum(np.abs(A), 1e-30)
    new_A = mag * phase
    new_a = np.fft.irfft(new_A, n)
    return np.maximum(new_a, 0.0)

def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_profile(n):
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 + np.cos(2.0 * np.pi * t)) / 2.0
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_profile(n, width=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.3) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

"""
Multi-phase optimizer for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses L-BFGS continuation, Nesterov-accelerated smooth gradient,
Polyak-step subgradient with iterate averaging, spectral projection,
and Adam polishing. Key improvements over prior: Polyak step with
averaging, spectral flattening, more aggressive beta schedule, and
better time management across 7 phases.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _gradient_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _objective_smooth(z, beta):

    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_obj(z, beta):
    return _objective_smooth(z, beta)


def _lbfgs_grad(z, beta):
    return _gradient_smooth(z, beta)

def _exact_subgradient_step(a, top_k=5):
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())

    if top_k >= len(c):
        active = np.ones(len(c), dtype=bool)
        k_active = len(c)
    else:
        top_indices = np.argsort(c)[::-1][:top_k]
        active = np.zeros(len(c), dtype=bool)
        active[top_indices] = True
        k_active = top_k

    w = np.zeros(len(c))
    w[active] = 1.0 / k_active
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 2.0 * a * (h - W)

    return grad, c, cm

def _spectral_project(a, n_peaks=20):
    n = len(a)
    A = np.fft.rfft(a)
    mag = np.abs(A).copy()
    if len(mag) > 1:
        sorted_mag = np.sort(mag[1:])[::-1]
        if n_peaks >= len(sorted_mag):
            T = sorted_mag[-1]
        else:
            T = sorted_mag[min(n_peaks, len(sorted_mag) - 1)]
        T = max(T, 1e-15)
        cap = max(mag[1:].max() * 0.92, T)
        mag[1:] = np.minimum(mag[1:], cap)
    phase = A / np.maximum(np.abs(A), 1e-30)
    new_A = mag * phase
    new_a = np.fft.irfft(new_A, n)
    return np.maximum(new_a, 0.0)

def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_profile(n):
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 + np.cos(2.0 * np.pi * t)) / 2.0
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_profile(n, width=0.3, center=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - center) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

        print(f"No parent, using linear profile n={n}")

    # Phase 0: Profile evaluation (25s)
    phase0_end = min(deadline, t_start + 25.0)
    sc_parent = _score_norm_fn(parent)
    if sc_parent < best_score:
        best_score = sc_parent
        best_seq = parent.copy()
    elapsed = time.monotonic() - t_start
    print(f"  [{elapsed:6.1f}s] parent: {best_score:.10f} (n={n})")

    for sn in [2048, 4096]:
        if time.monotonic() > phase0_end:
            break
        profiles = [
            _make_linear_profile(sn, 1.5),
            _make_linear_profile(sn, 2.5),
            _make_linear_profile(sn, 3.5),
            _make_quadratic_profile(sn, 1.5, 0.3),
            _make_quadratic_profile(sn, 2.0, -0.2),
            _make_cosine_profile(sn),
            _make_gaussian_profile(sn, 0.3, 0.3),
            _make_gaussian_profile(sn, 0.35, 0.4),
        ]
        for idx, p in enumerate(profiles):
            if len(p) != n and parent is not None:
                p = _interp_profile(p, n)
            sc = _score_norm_fn(p)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = p.copy()
                print(f"  [{time.monotonic()-t_start:6.1f}s] prof{idx}_n{sn}: {sc:.10f}")

    print(f"Phase 0 done: best={best_score:.10f}, n={len(best_seq)}")

    # Phase 1: L-BFGS continuation (250s)
    phase1_end = min(deadline, t_start + 275.0)
    if best_seq is not None:
        per_time = min(200.0, phase1_end - time.monotonic() - 5.0)
        if per_time > 15:
            p_best, sc_best = _run_lbfgs(best_seq, phase1_end, max_wall_time=per_time)
            if np.isfinite(sc_best) and sc_best < best_score:
                best_score = sc_best
                best_seq = p_best.copy()
                print(f"  [{time.monotonic()-t_start:6.1f}s] lbfgs_phase1: {sc_best:.10f}")

    print(f"Phase 1 done: best={best_score:.10f}")

    # Phase 2: Nesterov with aggressive continuation (300s)
    phase2_end = min(deadline, t_start + 575.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        beta_fracs = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                      5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7]
        p_best, sc_best = _nesterov_optimize(best_seq, phase2_end, beta_fracs,
                                             lr_init=0.012, mom_init=0.88)
        if np.isfinite(sc_best) and sc_best < best_score:
            best_score = sc_best
            best_seq = p_best.copy()
            print(f"  [{time.monotonic()-t_start:6.1f}s] nesterov_phase2: {sc_best:.10f}")

    print(f"Phase 2 done: best={best_score:.10f}")

    # Phase 3: Adam polishing (200s)
    phase3_end = min(deadline, t_start + 775.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        per_time = min(180.0, phase3_end - time.monotonic() - 10.0)
        if per_time > 15:
            p_best, sc_best = _adam_optimize(best_seq, min(phase3_end, time.monotonic() + per_time),
                                             lr=0.03, beta_frac=0.004)
            if np.isfinite(sc_best) and sc_best < best_score:
                best_score = sc_best
                best_seq = p_best.copy()
                print(f"  [{time.monotonic()-t_start:6.1f}s] adam_phase3: {sc_best:.10f}")

    print(f"Phase 3 done: best={best_score:.10f}")

    # Phase 4: Polyak subgradient with averaging (300s)
    phase4_end = min(deadline, t_start + 1075.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        per_time = min(250.0, phase4_end - time.monotonic() - 10.0)
        if per_time > 15:
            top_k_sched = [3, 5, 10, 20, 50, 100]
            p_best, sc_best = _polyak_subgradient(best_seq,
                                                  min(phase4_end, time.monotonic() + per_time),
                                                  top_k_schedule=top_k_sched, eta_init=1.0)
            if np.isfinite(sc_best) and sc_best < best_score:
                best_score = sc_best
                best_seq = p_best.copy()
                print(f"  [{time.monotonic()-t_start:6.1f}s] polyak_phase4: {sc_best:.10f}")

    print(f"Phase 4 done: best={best_score:.10f}")

    # Phase 5: Spectral-subgradient alternation (250s)
    phase5_end = min(deadline, t_start + 1325.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        per_time = min(200.0, phase5_end - time.monotonic() - 10.0)
        if per_time > 15:
            p_best, sc_best = _spectral_subgradient(best_seq,
                                                    min(phase5_end, time.monotonic() + per_time),
                                                    top_k=8, n_spectral_every=30)
            if np.isfinite(sc_best) and sc_best < best_score:
                best_score = sc_best
                best_seq = p_best.copy()
                print(f"  [{time.monotonic()-t_start:6.1f}s] spectral_phase5: {sc_best:.10f}")

    print(f"Phase 5 done: best={best_score:.10f}")

    # Phase 6: Fine Nesterov + Adam (200s)
    phase6_end = min(deadline, t_start + 1525.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        remaining = phase6_end - time.monotonic()
        if remaining > 60:
            per_time = min(100.0, remaining - 30.0)
            beta_fracs_tiny = [1e-6, 5e-7, 2e-7, 1e-7, 5e-8, 2e-8]
            p_best, sc_best = _nesterov_optimize(best_seq,
                                                 min(phase6_end - 30, time.monotonic() + per_time),
                                                 beta_fracs_tiny, lr_init=0.004, mom_init=0.95)
            if np.isfinite(sc_best) and sc_best < best_score:
                best_score = sc_best
                best_seq = p_best.copy()
                print(f"  [{time.monotonic()-t_start:6.1f}s] fine_nesterov: {sc_best:.10f}")

        remaining2 = phase6_end - time.monotonic()
        if remaining2 > 30:
            per_time2 = min(80.0, remaining2 - 5.0)
            p_best, sc_best = _adam_optimize(best_seq,
                                             min(phase6_end - 5, time.monotonic() + per_time2),
                                             lr=0.006, beta_frac=0.001)
            if np.isfinite(sc_best) and sc_best < best_score:
                best_score = sc_best
                best_seq = p_best.copy()
                print(f"  [{time.monotonic()-t_start:6.1f}s] fine_adam: {sc_best:.10f}")

    print(f"Phase 6 done: best={best_score:.10f}")

    # Phase 7: Restarts + final polish (remaining)
    phase7_end = deadline
    if best_seq is not None and time.monotonic() < phase7_end:
        n_best = len(best_seq)
        remaining = phase7_end - time.monotonic()

        if remaining > 60:
            n_restarts = 3
            per_restart = max(10.0, (remaining - 40.0) / (n_restarts + 1))

            for i in range(n_restarts):
                if time.monotonic() > phase7_end - 30:
                    break
                noise_level = 0.025 + 0.015 * i
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z - z.mean()
                z = z + noise_level * rng.normal(size=n_best)
                z = np.clip(z, -50, 50)
                z = z - z.mean()
                p_start = _softmax_fn(z)

                per_dl = min(phase7_end - 20 - i * 5, time.monotonic() + per_restart)
                beta_fracs_short = [1e-5, 5e-6, 2e-6, 1e-6, 5e-7]
                p_best, sc_best = _nesterov_optimize(p_start, per_dl,
                                                     beta_fracs_short, lr_init=0.006, mom_init=0.9)
                if np.isfinite(sc_best) and sc_best < best_score:
                    best_score = sc_best
                    best_seq = p_best.copy()
                    print(f"  [{time.monotonic()-t_start:6.1f}s] restart{i}: {sc_best:.10f}")

            if time.monotonic() < phase7_end - 10:
                rem = phase7_end - time.monotonic() - 5
                if rem > 10:
                    p_best, sc_best = _polyak_subgradient(best_seq,
                                                          min(phase7_end, time.monotonic() + rem),
                                                          top_k_schedule=[3, 5, 10],
                                                          eta_init=0.5)
                    if np.isfinite(sc_best) and sc_best < best_score:
                        best_score = sc_best
                        best_seq = p_best.copy()
                        print(f"  [{time.monotonic()-t_start:6.1f}s] final_polyak: {sc_best:.10f}")

    print(f"Phase 7 done: best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_linear_profile(2048, 1.5)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print("Evaluator returned inf/NaN")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    fallback = _make_linear_profile(2048, 1.5)
    print(f"FALLBACK: n=2048, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
