    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _solve_lp_direction(f, M, n_constraints=None):
    n = len(f)
    n_total = 2 * n - 1
    if n_constraints is None or n_constraints >= n_total:
        idx = np.arange(n_total)
    else:
        # Use all constraints (for correctness) but we could subset
        idx = np.arange(n_total)

    # Build sparse constraint matrix
    rows_list = []
    cols_list = []
    vals_list = []
    for k in idx:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows_list.append(k)
            cols_list.append(j)
            vals_list.append(f[k - j])

    if len(rows_list) == 0:
        return f.copy()

    H = lil_matrix((n_total, n), dtype=np.float64)
    for i in range(len(rows_list)):
        H[rows_list[i], cols_list[i]] = vals_list[i]
    H = csr_matrix(H)

    b_ub = np.full(n_total, M)
    c_obj = -np.ones(n)

    try:
        res = linprog(c_obj, A_ub=H, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        if res.status == 0 and res.x is not None:
            return res.x
    except Exception:
        pass
    return f.copy()


def _lp_refine(f, deadline, max_iters=200, step=0.15):
    n = len(f)
    f = np.asarray(f, dtype=np.float64).copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f = f / s
    best_f = f.copy()
    best_score = _score_norm(f)
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() > deadline - 3.0:
            break

        c = _full_convolve(f, f)
        M = float(c.max())

        t0 = time.monotonic()
        g = _solve_lp_direction(f, M)
        t_lp = time.monotonic() - t0

        if g is None or not np.all(np.isfinite(g)):
            break

        g_sum = float(g.sum())
        if g_sum < 1e-10:
            break

        g = g * (s / g_sum)

        if np.max(np.abs(g - f)) < 1e-12:
            stagnant += 1
            if stagnant > 5:
                break
            M *= 1.05
            g = _solve_lp_direction(f, M)
            g_sum = float(g.sum())
            if g_sum < 1e-10:
                break
            g = g * (s / g_sum)
            if np.max(np.abs(g - f)) < 1e-12:
                break
        else:
            stagnant = 0

        best_t = 0.05
        best_trial = np.inf
        for t_try in [0.5, 0.25, 0.15, 0.1, 0.05, 0.025]:
            f_trial = (1.0 - t_try) * f + t_try * g
            f_trial = np.maximum(f_trial, 0)
            st = f_trial.sum()
            if st < 1e-10:
                continue
            f_trial = f_trial / st
            sc = _score_norm(f_trial)
            if sc < best_trial:
                best_trial = sc
                best_t = t_try

        f = (1.0 - best_t) * f + best_t * g
        f = np.maximum(f, 0)
        f = f / f.sum()

        sc = _score_norm(f)
        if sc < best_score:
            best_score = sc
            best_f = f.copy()

        if it % 20 == 0:
            print(f"    LP iter {it}: score={best_score:.10f}, M={M:.8f}, "
                  f"lp_time={t_lp:.2f}s, t={best_t}")

    return best_f, best_score


def _lbfgs_continuation(a_init, deadline, rng, beta_scale=1.0, max_beta_levels=12):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.1, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015,
                   0.0008, 0.0004, 0.0002, 0.0001, 0.00005]
    beta_starts = beta_starts[:max_beta_levels]
    betas = [b * max_c0 * beta_scale for b in beta_starts]

    per_level = max(5.0, (deadline - time.monotonic()) / len(betas))

    for beta in betas:
        if time.monotonic() > deadline - 3.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        def obj_func(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            return 2.0 * n * (cm_ + beta * np.log(s_))

        def grad_func(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            W = float(np.dot(w, c_))
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g

        try:
            result = minimize(
                obj_func, z, jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': 2000, 'ftol': 1e-20, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 10000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _coordinate_descent(a_init, deadline, rng, n_trials=50000):
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(n_trials):
        if time.monotonic() > deadline - 2.0:
            break

        # Pick two coordinates to swap between
        j = rng.integers(0, n)
        j2 = rng.integers(0, n)
        if j == j2:
            continue

        # Try moving epsilon from j to j2
        # Optimal epsilon approx: balance the peak
        c = _full_convolve(a, a)
        peak_k = int(np.argmax(c))

        # Gradient at peak for coordinate j
        grad_j = 0.0
        if 0 <= peak_k - j <= n - 1:
            grad_j = 2.0 * a[peak_k - j]
        grad_j2 = 0.0
        if 0 <= peak_k - j2 <= n - 1:
            grad_j2 = 2.0 * a[peak_k - j2]

        # Move from higher gradient to lower gradient
        if grad_j <= grad_j2:
            j, j2 = j2, j
            grad_j, grad_j2 = grad_j2, grad_j

        # epsilon = fraction of a[j] to move
        eps_frac = rng.uniform(0.05, 0.4)
        eps = eps_frac * a[j]

        a_trial = a.copy()
        a_trial[j] -= eps
        a_trial[j2] += eps
        if a_trial[j] < 0:
            a_trial[j] = 0
            a_trial[j2] += a[j] * eps_frac
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        a_trial = a_trial / st

        sc = _score_norm(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()

        if it % 5000 == 0:
            print(f"    CD iter {it}: score={best_score:.10f}")

    return best_a, best_score


def _simulated_annealing(a_init, deadline, rng, n_trials=100000,
                         T_start=0.01, T_end=0.0001):
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    current_score = best_score

    n_iter = n_trials
    for it in range(n_iter):
        if time.monotonic() > deadline - 2.0:
            break

        progress = it / max(1, n_iter)
        T = T_start * (T_end / T_start) ** progress

        # Perturbation: perturb a random block
        block_size = max(2, min(50, n // 20))
        start = rng.integers(0, n - block_size)
        a_trial = a.copy()
        a_trial[start:start + block_size] *= rng.uniform(0.5, 1.5, size=block_size)
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-10:
            continue
        a_trial = a_trial / st

        sc = _score_norm(a_trial)
        delta = sc - current_score

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_trial.copy()
            current_score = sc
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()

        if it % 10000 == 0:
            print(f"    SA iter {it}: best={best_score:.10f}, T={T:.6f}")

    return best_a, best_score


def _adam_polish(a_init, deadline, rng, lr=0.01, k_top=3):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8

    max_iters = 50000
    for it in range(max_iters):
        if time.monotonic() > deadline - 2.0:
            break

        t += 1
        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        cm = float(c.max())

        # Top-K gradient
        k_actual = min(k_top, len(c))
        top_idx = np.argsort(c)[-k_actual:]
        grad = np.zeros(n)
        for k_val in top_idx:
            k_val = int(k_val)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * a_curr[k_val - hi:k_val - lo + 1][::-1]
        grad /= k_actual
        grad *= 4.0 * n

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if it % 50 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()

        lr = max(0.001, lr * 0.9999)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP-based SOTA refinement, logsumexp L-BFGS,
    coordinate descent, simulated annealing, and multi-scale optimization.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 4096
        parent = _make_linear_profile(n_parent, 1.5)
        sc_parent = _score_norm(parent)
        update_best(parent, sc_parent, "init_linear")

    # --- Phase 0: Diverse initialization (30s) ---
    phase0_end = min(deadline, t_start + 30.0)
    n_work = n_parent
    profiles = [
        _make_linear_profile(n_work, 0.5),
        _make_linear_profile(n_work, 1.0),
        _make_linear_profile(n_work, 1.5),
        _make_linear_profile(n_work, 2.0),
        _make_linear_profile(n_work, 3.0),
        _make_quadratic_profile(n_work, 0.5, 0.3),
        _make_quadratic_profile(n_work, 1.0, -0.3),
        _make_quadratic_profile(n_work, 1.5, 0.5),
        _make_gaussian_profile(n_work, 0.25),
        _make_gaussian_profile(n_work, 0.35),
        _make_u_profile(n_work, 0.3),
        _make_u_profile(n_work, 0.5),
        _make_u_profile(n_work, 0.7),
    ]

    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm(p)
        update_best(p, sc, f"prof{idx}")

    print(f"Phase 0 done: best={best_score:.10f}")

    # --- Phase 1: L-BFGS on best candidate (200s) ---
    phase1_end = min(deadline, t_start + 230.0)
    if best_seq is not None:
        dl = phase1_end - 5.0
        p_best, sc_best = _lbfgs_continuation(best_seq, dl, rng, beta_scale=1.0)
        update_best(p_best, sc_best, "lbfgs_phase1")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: LP-based refinement (500s) ---
    phase2_end = min(deadline, t_start + 730.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        # Use a moderate n for LP speed
        n_lp = min(2048, len(best_seq))
        p_lp = _interp_profile(best_seq, n_lp) if len(best_seq) != n_lp else best_seq.copy()
        sc_lp = _score_norm(p_lp)
        print(f"  LP start: n={n_lp}, score={sc_lp:.10f}")
        dl = phase2_end - 5.0
        p_best, sc_best = _lp_refine(p_lp, dl, max_iters=150, step=0.15)
        # Interpolate back to original size
        if len(p_best) != len(best_seq):
            p_best_large = _interp_profile(p_best, len(best_seq))
            sc_best_large = _score_norm(p_best_large)
            update_best(p_best_large, sc_best_large, f"lp_interp_n{len(best_seq)}")
            update_best(p_best, sc_best, f"lp_n{n_lp}")
        else:
            update_best(p_best, sc_best, f"lp_n{n_lp}")

    # Also try LP on the actual best_seq if it's not too large
    if best_seq is not None and len(best_seq) <= 2048 and time.monotonic() < phase2_end:
        dl = phase2_end - 5.0
        p_best, sc_best = _lp_refine(best_seq, dl, max_iters=50, step=0.1)
        update_best(p_best, sc_best, "lp_direct")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: L-BFGS fine polishing (200s) ---
    phase3_end = min(deadline, t_start + 930.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        dl = phase3_end - 5.0
        p_best, sc_best = _lbfgs_continuation(best_seq, dl, rng, beta_scale=0.3)
        update_best(p_best, sc_best, "lbfgs_fine")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Adam polish (150s) ---
    phase4_end = min(deadline, t_start + 1080.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        dl = phase4_end - 5.0
        p_best, sc_best = _adam_polish(best_seq, dl, rng, lr=0.008, k_top=3)
        update_best(p_best, sc_best, "adam_k3")

        if time.monotonic() < phase4_end - 30:
            dl2 = phase4_end - 5.0
            p_best2, sc_best2 = _adam_polish(best_seq, dl2, rng, lr=0.005, k_top=7)
            update_best(p_best2, sc_best2, "adam_k7")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Coordinate descent (150s) ---
    phase5_end = min(deadline, t_start + 1230.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        dl = phase5_end - 5.0
        p_best, sc_best = _coordinate_descent(best_seq, dl, rng, n_trials=30000)
        update_best(p_best, sc_best, "coord_descent")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Simulated annealing (150s) ---
    phase6_end = min(deadline, t_start + 1380.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        dl = phase6_end - 5.0
        p_best, sc_best = _simulated_annealing(best_seq, dl, rng,
                                               n_trials=50000,
                                               T_start=0.005, T_end=0.00005)
        update_best(p_best, sc_best, "sim_anneal")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Multi-scale - try larger n (150s) ---
    phase7_end = min(deadline, t_start + 1530.0)
    if best_seq is not None and time.monotonic() < phase7_end:
        n_curr = len(best_seq)
        for target_n in [n_curr * 2]:
            if target_n > 16384 or target_n <= n_curr:
                continue
            if time.monotonic() > phase7_end - 30:
                break
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{target_n}")

            if time.monotonic() < phase7_end - 40:
                dl = min(phase7_end - 10, time.monotonic() + 100.0)
                p_best, sc_best = _lbfgs_continuation(p_interp, dl, rng,
                                                      beta_scale=0.5,
                                                      max_beta_levels=6)
                update_best(p_best, sc_best, f"lbfgs_n{target_n}")

    print(f"Phase 7 done: best={best_score:.10f}")

    # --- Phase 8: Final intensive polishing (remaining time) ---
    if best_seq is not None:
        remaining = max(10.0, deadline - time.monotonic())
        if remaining > 30:
            dl = min(deadline - 5, time.monotonic() + remaining * 0.4)
            p_best, sc_best = _lbfgs_continuation(best_seq, dl, rng,
                                                  beta_scale=0.1,
                                                  max_beta_levels=8)
            update_best(p_best, sc_best, "final_lbfgs")

        if time.monotonic() < deadline - 20:
            dl = min(deadline - 5, time.monotonic() + remaining * 0.3)
            p_best, sc_best = _adam_polish(best_seq, dl, rng, lr=0.003, k_top=2)
            update_best(p_best, sc_best, "final_adam")

        if time.monotonic() < deadline - 10:
            dl = min(deadline - 3, time.monotonic() + remaining * 0.2)
            p_best, sc_best = _coordinate_descent(best_seq, dl, rng, n_trials=20000)
            update_best(p_best, sc_best, "final_cd")

    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_linear_profile(4096, 1.5)
            s = float(final_seq.sum())
        # Keep unnormalized (the evaluator handles normalization internally
        # but the score formula uses sum^2, so any scaling works)
        # Actually, let's keep sum = 1 for cleanliness
        final_seq = final_seq / s

        sc_final = _score_full(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_linear_profile(4096, 1.5)
    print(f"FALLBACK: n=4096, score={_score_full(fallback):.10f}")
    return [float(x) for x in fallback]
