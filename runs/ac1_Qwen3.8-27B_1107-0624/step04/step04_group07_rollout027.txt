"""
Optimization of non-negative step function coefficients to minimize
2*n*max(conv(a,a))/sum(a)^2 using smooth-max gradient descent,
LP polishing, multi-scale escalation, and spectral initialization.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _cross_correlate(w, a):
    """Compute cross-correlation of w (len 2n-1) and a (len n) at lags 0..n-1."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())

def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip to avoid overflow
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w

def _build_conv_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def _lp_primal(a, M, n):
    """Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    A = _build_conv_matrix(n, a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def _lp_fixed_point(a, deadline, max_iters=50, use_line_search=True):
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a)
        M = float(c.max())
        b = _lp_primal(a, M, n)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b = b / s_b
        
        if use_line_search:
            best_t = 0.5
            best_sc = _score_norm(a)
            for t_candidate in [0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
                a_t = (1.0 - t_candidate) * a + t_candidate * b
                a_t = np.maximum(a_t, 1e-15)
                a_t = a_t / a_t.sum()
                sc_t = _score_norm(a_t)
                if sc_t < best_sc:
                    best_sc = sc_t
                    best_t = t_candidate
            a = (1.0 - best_t) * a + best_t * b
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
        else:
            a = b
        
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score

def _adam_smooth_max(a, deadline, rng, lr=0.05, beta_init=None,
                     beta_decay=0.95, no_imp_limit=200):

    z = np.log(a)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    c = _full_convolve(a)
    maxc = float(c.max())
    beta = beta_init if beta_init is not None else 0.1 * maxc
    
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    
    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)
        g = _cross_correlate(w, a)
        
        dot = float(np.dot(a, g))
        gz = a * (g - dot)
        gz -= gz.mean()
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()
        
        a = np.exp(z - z.max())
        a = a / a.sum()
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        c = _full_convolve(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        
        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.003, lr * 0.93)
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        if t_adam % 50 == 0:
            beta = max(1e-4, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.1:
                    beta = max(1e-4, beta * 0.5)
        
        if no_imp > no_imp_limit:
            ns = 0.08 if best_score < 1.51 else 0.15
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a)
            beta = 0.1 * c.max()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0
    
    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score

def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
