def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a) = 1. Returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _convolve_peak_info(a, k_top=10):
    """Returns the convolution, the max value, and indices of top-K peaks."""
    c = _full_convolve(a, a)
    cm = float(c.max())
    k_actual = min(k_top, len(c))
    top_idx = np.argsort(c)[-k_actual:]
    return c, cm, top_idx

def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _topk_subgrad_descent(a_init, deadline, k_top=5, lr_init=0.01, momentum=0.9):
    """Top-K subgradient descent with momentum and adaptive step size."""
    n = len(a_init)
    a = a_init.copy()
    a = _project(a)
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    v = np.zeros(n)  # momentum
    t = 0
    lr = lr_init
    
    c, cm, top_idx = _convolve_peak_info(a, k_top)
    
    while time.monotonic() < deadline:
        t += 1
        
        # Compute subgradient
        grad = _topk_subgradient(a, top_idx, k_top)
        
        # Momentum update
        v = momentum * v + (1 - momentum) * grad
        a_new = a - lr * v
        a_new = _project(a_new)
        
        # Evaluate
        sc = _score_norm(a_new)
        
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            # Recompute top indices
            c, cm, top_idx = _convolve_peak_info(a, k_top)
            # Slightly increase lr on improvement
            lr = min(0.05, lr * 1.01)
        else:
            # Reduce lr
            lr = max(0.0001, lr * 0.95)
            # Don't update a (or partially update)
            if sc < best_score * 1.001:
                a = a_new.copy()
        
        # Periodically recompute top indices even if no improvement
        if t % 50 == 0:
            c, cm, top_idx = _convolve_peak_info(a, k_top)
        
        # Periodic restart with noise
        if t % 500 == 0:
            noise = 0.01 * np.random.default_rng(t).normal(size=n)
            a_noisy = _project(a + noise)
            sc_noisy = _score_norm(a_noisy)
            if sc_noisy < best_score:
                best_score = sc_noisy
                best_a = a_noisy.copy()
                a = a_noisy.copy()
    
    return best_a, best_score
