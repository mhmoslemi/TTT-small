OK so we want C to be NOT flat. We want c to be flat, which means C should be concentrated at DC.

So the "ideal" C is [1, 0, 0, ..., 0] (all energy at DC). The resulting c would be [1/(2n-1), 1/(2n-1), ...] (flat). But this requires P = [1, 0, 0, ..., 0], which means p = [1/(2n-1), 0, 0, ...] (a delta). But p must have length n and be non-negative, so p = [1, 0, 0, ..., 0] (a single spike). The autocorrelation of a single spike is also a single spike, not flat.

The issue is that we can't achieve both:
- C concentrated at DC (which would give flat c)
- p having length n with n > 1 (which requires P to have energy at non-DC frequencies)

The constraint is: P[k] = sum_{i=0}^{n-1} p[i] * exp(-2*pi*i*k/N) where N = 2n-1.

For a constant p[i] = 1/n:
P[k] = (1/n) * sum_{i=0}^{n-1} exp(-2*pi*i*k/N)

This is a "Dirichlet-like" sum. For k=0: P[0] = 1. For k != 0: |P[k]| = (1/n) * |sin(n*pi*k/N) / sin(pi*k/N)|.

The magnitude |P[k]| has peaks at k close to 0 and k close to N (which is the same as k close to 0 by symmetry), and zeros at k = N/n, 2N/n, ... (if n divides N).

For N = 2n-1, n doesn't divide N (since 2n-1 is not divisible by n for n > 1). So there are no exact zeros, but there are near-zeros.

The autocorrelation of the uniform sequence is a triangle, which has a peak of 1/n at the center. The score is 2.

To do better than 2, we need a non-uniform sequence. The question is: what shape?

Let me think about this more carefully using the Fourier approach.

We want to minimize max(c) = max(IDFT(C)) where C = |P|^2 and P[k] = sum_{i=0}^{n-1} p[i] exp(-2*pi*i*k/N).

The constraint is p >= 0, sum(p) = 1.

In the frequency domain, the constraint p >= 0 is non-trivial. But the constraint sum(p) = 1 means P[0] = 1.

Let's parameterize: P[0] = 1, P[k] = a_k + i*b_k for k = 1, ..., N-1 (with P[N-k] = conjugate(P[k])).

The constraint p >= 0 means: for all i in [0, n-1], (1/N) * sum_{k=0}^{N-1} P[k] * exp(2*pi*i*k/N) >= 0.

And p[i] = 0 for i in [n, N-1] means: for all i in [n, N-1], (1/N) * sum_{k=0}^{N-1} P[k] * exp(2*pi*i*k/N) = 0.

The second constraint is very restrictive! It means P must be the DFT of a signal that's zero for the second half. This is essentially a "sparse in time" constraint.

This makes the problem very hard to solve directly in the frequency domain. The GS-type iteration is a natural way to handle this: alternate between:
1. Frequency domain: modify C to reduce peaks
2. Time domain: project onto {p >= 0, p[i] = 0 for i >= n}

"""

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-10:
        return None
    return a / s


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 0.0))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-14):

    z -= float(np.mean(z))
    b1, b2, eps = 0.9, 0.999, 1e-8

    v = np.zeros(n)

    p = p0.copy()

            if w_sum <= 0:

            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0

        z = z - lr * mh / (np.sqrt(vh) + eps)

            m.fill(0.0)

def _peak_surgery(p, deadline, n_iter_max=3000, eta_init=0.03):
    """Direct peak reduction by adjusting contributing pairs."""
    a = np.asarray(p, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    eta = eta_init
    no_improve = 0
    rng = np.random.default_rng(999)

    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break
        M = float(c.max())
        peak_k = int(np.argmax(c))
        score = 2.0 * n * M
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        if i_start > i_end:
            break
        idx = np.arange(i_start, i_end + 1)
        partners = peak_k - idx
        contrib = a[idx] * a[partners]
        total = float(contrib.sum())
        if total < 1e-15:
            break
        top_frac = 0.3
        n_top = max(2, int(len(idx) * top_frac))
        top_local = np.argsort(contrib)[::-1][:n_top]
        hot_i = idx[top_local]
        hot_partners = partners[top_local]
        # Move mass from hot positions
        for i_pos in hot_i[:3]:
            for j_pos in hot_partners[:3]:
                if i_pos == j_pos:
                    continue
                # Move a fraction of mass from the larger to a random cold position
                if a[i_pos] < a[j_pos]:
                    src = i_pos
                else:
                    src = j_pos
                # Find a cold target (not contributing to top peaks)
                cold_mask = np.ones(n, dtype=bool)
                for k_offset in range(-2, 3):
                    kk = peak_k + k_offset
                    if 0 <= kk < 2 * n - 1:
                        s2 = max(0, kk - n + 1)
                        e2 = min(n - 1, kk)
                        if s2 <= e2:
                            cold_mask[s2:e2 + 1] = False
                            cold_mask[kk - np.arange(s2, e2 + 1)] = False
                cold_idx = np.where(cold_mask)[0]
                if len(cold_idx) == 0:
                    continue
                target = cold_idx[rng.integers(0, len(cold_idx))]
                amt = a[src] * eta
                a[src] -= amt
                a[target] += amt
        a = np.maximum(a, 1e-15)
        a /= a.sum()
        c = _full_convolve(a)
        if no_improve > 500:
            eta = max(0.003, eta * 0.85)
            no_improve = 0
        if no_improve > 2000:
            break

    return best_a, best_score


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(p, deadline, t_start=0.3, max_solves=30, m_shrink=0.997):
    """LP-based refinement with decreasing target."""
    n = len(p)
    a = np.asarray(p, dtype=np.float64).ravel().copy()
    a /= a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0
    t = t_start

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(a, M)
        if b is None:
            b = _lp_solve(a, M * 1.02)
            if b is None:
                break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new /= a_new.sum()
        sc_new = _score_norm(a_new)
        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()
        a = a_new
        c = _full_convolve(a)
        M_new = float(c.max())
        M = max(M_new * m_shrink, best_score / (2.0 * n) * 1.003)
        t = max(0.1, t * 0.93)

    return best_a, best_score


def _make_profile(n, kind='cos', param=1.0):
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cos':
        t = np.linspace(0, 1, n)
        a = 1.0 + param * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == 'raised_cos':
        t = np.linspace(0, 1, n)
        a = (1.0 + np.cos(2.0 * np.pi * t)) ** param
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + param * (t - 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == 'triangle':
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0) ** param
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == 'gaussian':
        t = np.linspace(0, 1, n)
        a = np.exp(-param * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == 'bimodal':
        t = np.linspace(0, 1, n)
        a = np.exp(-param * (t - 0.3) ** 2) + np.exp(-param * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    return np.ones(n) / n


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    best_seq = None
    best_score = np.inf

    def _update(p, label=""):
        nonlocal best_seq, best_score
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            p = np.maximum(p, 0.0)
            s = float(p.sum())
            if s < 1e-10:
                return
            p = p / s
            sc = _score_norm(p)
            if sc < best_score:
                best_seq = p.copy()
                best_score = float(sc)
                print(f"  NEW BEST: {sc:.10f} n={p.size} [{label}]")
            return sc
        return np.inf

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_p = _score_norm(parent)
        print(f"Parent: n={parent.size}, score={sc_p:.10f}")
        _update(parent, 'parent')
    else:
        print("No parent, using initialization")
        p_init = _make_profile(1024, 'raised_cos', 0.5)
        _update(p_init, 'init_cos')

    n_base = best_seq.size if best_seq is not None else 1024

    # === Phase 1: Spectral flattening (200s) ===
    phase1_end = t_start + 200.0
    print(f"\nPhase 1: Spectral flattening (target: {phase1_end - t_start:.0f}s)")
    if time.monotonic() < phase1_end:
        p1, sc1 = _spectral_flatten(best_seq, phase1_end, alpha_init=0.05)
        _update(p1, 'spec_flat')

    # === Phase 2: Adam main optimization (300s) ===
    phase2_end = t_start + 500.0
    print(f"\nPhase 2: Adam optimization, current best={best_score:.10f}")
    if time.monotonic() < phase2_end:
        p2, sc2 = _adam_logit(best_seq, phase2_end, rng, lr=0.04, beta_frac=0.05,
                              beta_decay=0.95, no_imp_limit=200)
        _update(p2, 'adam_main')

    # === Phase 3: Spectal targeted (150s) ===
    phase3_end = t_start + 650.0
    print(f"\nPhase 3: Targeted spectral, current best={best_score:.10f}")
    if time.monotonic() < phase3_end and best_seq.size <= 4096:
        p3, sc3 = _spectral_targeted(best_seq, phase3_end, alpha=0.03, n_top=3)
        _update(p3, 'spec_targeted')

    # === Phase 4: Peak surgery (200s) ===
    phase4_end = t_start + 850.0
    print(f"\nPhase 4: Peak surgery, current best={best_score:.10f}")
    if time.monotonic() < phase4_end and best_seq.size <= 2048:
        p4, sc4 = _peak_surgery(best_seq, phase4_end, n_iter_max=3000, eta_init=0.02)
        _update(p4, 'peak_surgery')

    # === Phase 5: LP refinement (150s) ===
    phase5_end = t_start + 1000.0
    print(f"\nPhase 5: LP refinement, current best={best_score:.10f}")
    if time.monotonic() < phase5_end and best_seq.size <= 2048:
        p5, sc5 = _lp_refine(best_seq, phase5_end, t_start=0.25, max_solves=20, m_shrink=0.998)
        _update(p5, 'lp_refine')
        # Quick Adam after LP
        if time.monotonic() < phase5_end - 10:
            p5a, sc5a = _adam_logit(best_seq, min(phase5_end, time.monotonic() + 40),
                                     rng, lr=0.02, beta_frac=0.04, beta_decay=0.96,
                                     no_imp_limit=100)
            _update(p5a, 'post_lp_adam')

    # === Phase 6: Second Adam pass with different params (200s) ===
    phase6_end = t_start + 1200.0
    print(f"\nPhase 6: Adam pass 2, current best={best_score:.10f}")
    if time.monotonic() < phase6_end:
        # Perturb and re-optimize
        perturbed = best_seq + 0.02 * rng.normal(size=best_seq.size)
        perturbed = np.maximum(perturbed, 1e-12)
        perturbed /= perturbed.sum()
        p6, sc6 = _adam_logit(perturbed, phase6_end, rng, lr=0.03, beta_frac=0.04,
                              beta_decay=0.95, no_imp_limit=150)
        _update(p6, 'adam_pass2')

    # === Phase 7: Multi-scale exploration (250s) ===
    phase7_end = t_start + 1450.0
    print(f"\nPhase 7: Multi-scale, current best={best_score:.10f}")
    if time.monotonic() < phase7_end:
        for n_try in [2048, 4096]:
            if time.monotonic() > phase7_end - 60:
                break
            if best_seq.size >= n_try:
                continue
            p_int = _interp_profile(best_seq, n_try)
            if p_int is None:
                continue
            sc_int = _score_norm(p_int)
            print(f"  Interpolated to n={n_try}: {sc_int:.10f}")
            _update(p_int, f'interp_n{n_try}')
            # Quick spectral + Adam at new scale
            per_spec = min(40.0, phase7_end - time.monotonic() - 60)
            if per_spec > 15:
                p_ns, sc_ns = _spectral_flatten(p_int, time.monotonic() + per_spec, alpha_init=0.03)
                _update(p_ns, f'spec_n{n_try}')
            per_adam = min(60.0, phase7_end - time.monotonic() - 10)
            if per_adam > 20:
                p_na, sc_na = _adam_logit(best_seq if best_seq.size == n_try else p_int,
                                           time.monotonic() + per_adam, rng,
                                           lr=0.03, beta_frac=0.05, beta_decay=0.95,
                                           no_imp_limit=100)
                _update(p_na, f'adam_n{n_try}')

    # === Phase 8: Final intensive polish (remaining time) ===
    remaining_end = deadline
    print(f"\nPhase 8: Final polish, current best={best_score:.10f}")
    if time.monotonic() < remaining_end and best_seq is not None:
        # Alternating fine-tuning
        cycles = 0
        while time.monotonic() < remaining_end - 10 and cycles < 10:
            cycles += 1
            # Fine Adam
            per = min(30.0, remaining_end - time.monotonic() - 5)
            if per < 10:
                break
            pf, scf = _adam_logit(best_seq, time.monotonic() + per, rng,
                                   lr=0.01, beta_frac=0.03, beta_decay=0.97,
                                   no_imp_limit=80)
            _update(pf, f'final_adam_{cycles}')
            # Fine spectral
            if time.monotonic() < remaining_end - 10 and best_seq.size <= 4096:
                per2 = min(20.0, remaining_end - time.monotonic() - 5)
                if per2 < 10:
                    break
                ps, scs = _spectral_flatten(best_seq, time.monotonic() + per2, alpha_init=0.01)
                _update(ps, f'final_spec_{cycles}')
            # Fine peak surgery
            if time.monotonic() < remaining_end - 10 and best_seq.size <= 2048:
                per3 = min(20.0, remaining_end - time.monotonic() - 5)
                if per3 < 10:
                    break
                pp, scp = _peak_surgery(best_seq, time.monotonic() + per3,
                                         n_iter_max=500, eta_init=0.01)
                _update(pp, f'final_surg_{cycles}')

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            # Keep as-is (scale invariant)
            pass
        else:
            final = _make_profile(1024, 'uniform')

        exact_sc = _score_norm(final)
        print(f"\nFINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL confirmed: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval check error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile(1024, 'raised_cos', 0.5)
    print(f"FALLBACK n=1024")
    return [float(x) for x in fb]
