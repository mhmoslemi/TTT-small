"""
Multi-phase Adam optimization with cosine-annealing LR, central-binomial init,
beta-continuation, multi-restart diversification, and multi-n exploration.
Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Key improvements: cosine LR schedule, central binomial coefficient profile,
adaptive beta, more diverse restarts, and larger-n exploration.
"""
import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _central_binomial_profile(n):
    """Generate normalized central binomial coefficient sequence (flat convolution)."""
    a = np.empty(n, dtype=np.float64)
    for k in range(n):
        # C(2k, k) / 4^k = product_{j=1}^{k} (2j-1)/(2j)
        if k == 0:
            a[k] = 1.0
        else:
            a[k] = a[k-1] * (2.0*k - 1.0) / (2.0*k)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "central_binomial":
        return _central_binomial_profile(n)
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "inv_sqrt":
        t = np.arange(n, dtype=np.float64) + 1.0
        a = 1.0 / np.sqrt(t)
        return a / a.sum()
    elif kind == "geometric":
        r = 0.99
        a = np.array([r**k for k in range(n)])
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_structured(a, rng, mode="fourier"):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    if mode == "shift":
        shift = int(rng.integers(1, max(2, n // 20)))
        z = np.roll(z, shift)
    elif mode == "fourier":
        freq = int(rng.integers(1, max(2, n // 4)))
        amp = 0.05 * rng.uniform(0.5, 2.0)
        t = np.arange(n)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif mode == "multi_fourier":
        n_freqs = int(rng.integers(2, 5))
        for _ in range(n_freqs):
            freq = int(rng.integers(1, max(2, n // 4)))
            amp = 0.03 * rng.uniform(0.5, 2.0)
            t = np.arange(n)
            phase = rng.uniform(0, 2 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif mode == "local":
        center = int(rng.integers(n // 4, 3 * n // 4))
        width = int(rng.integers(n // 20, n // 5))
        t = np.arange(n)
        z = z + 0.1 * rng.uniform(-1, 1) * np.exp(-0.5 * ((t - center) / width) ** 2)
    z = z - z.mean()
    return _softmax(z)


def _adam_optimize(a_init, deadline, beta_schedule, max_iter=500000, lr=0.02,
                   warmup_frac=0.05, momentum_reset_interval=50000):
    """Adam optimizer in log-space with cosine annealing and beta continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    total_iter = max_iter
    warmup_iters = max(100, int(total_iter * warmup_frac))

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        phase_iter_target = max_iter
        while it_in_phase < phase_iter_target:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            # Cosine annealing learning rate with warmup
            if t < warmup_iters:
                lr_t = lr * t / warmup_iters
            else:
                progress = (t - warmup_iters) / max(1, total_iter - warmup_iters)
                progress = min(1.0, progress)
                lr_t = lr * 0.5 * (1.0 + np.cos(np.pi * progress))
                lr_t = max(lr_t, lr * 0.01)

            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad = _compute_grad_z(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = lr_t * mh / (np.sqrt(vh) + eps)

            # Clip update magnitude
            max_update = 2.0
            update_norm = np.linalg.norm(update)
            if update_norm > max_update:
                update = update * (max_update / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            # Periodic momentum reset + small perturbation for exploration
            if momentum_reset_interval > 0 and t % momentum_reset_interval == 0:
                m_buf *= 0.5
                v_buf *= 0.5
                z = z + 0.01 * np.random.standard_normal(n)
                z = z - z.mean()

            if t % 200 == 0:
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
