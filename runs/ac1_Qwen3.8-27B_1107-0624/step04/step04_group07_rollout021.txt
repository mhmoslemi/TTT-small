    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]

def _cross_w(w, a):

    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)

    elif ptype == 'linear':

        a = 1.0 + p1 * (t - 0.5) * 2

        a = 1.0 + p1 * np.sin(2 * np.pi * p2 * t / n)

        a = 1.0 + p1 * t * t
