def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _score(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())

def _normalize(a):

def _lp_solve(a, M):
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2*n-1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

    M = c.max()
    threshold = M * 0.9

        hot_mask[int(np.argmax(c))] = True

    hot_indices = np.where(hot_mask)[0]
    for k in hot_indices:

        i_end = min(n - 1, k)
        if i_start <= i_end:
            partner = k - np.arange(i_start, i_end + 1)
            heat[i_start:i_end+1] += a[partner]
    
    heat_max = heat.max()
    if heat_max < 1e-15:
        return a

    factor = np.maximum(1.0 - eta * heat, 0.01)
    a_new = a * factor
    a_new = np.maximum(a_new, 1e-15)
    return a_new / a_new.sum()

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):

def _normalize(a):

def _binom_profile(n):

    a = np.zeros(n)
    for k in range(n):
        if k == 0:

        else:
            a[k] = a[k-1] * (2*k) * (2*k - 1) / (k * k * 4.0) if k > 0 else 1.0

def _u_shape_profile(n):
    """a[k] proportional to 1/sqrt((k+1)(n-k)), U-shaped."""
    k = np.arange(n)
    a = 1.0 / np.sqrt((k + 1.0) * (n - k))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _u_shape_power(n, p):
    """a[k] proportional to ((k+1)*(n-k))^(-p)."""
    k = np.arange(n)
    a = ((k + 1.0) * (n - k)) ** (-p)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _mirror_binom(n):
    """a[k] = min(binom(2k,k)/4^k, binom(2(n-1-k),n-1-k)/4^(n-1-k))."""
    left = np.zeros(n)
    left[0] = 1.0
    for k in range(1, n):
        left[k] = left[k-1] * (2.0*k - 1.0) / (2.0 * k)
    right = left[::-1]
    a = np.minimum(left, right)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _mixed_profile(n, alpha):
    """Blend between binom profile and uniform."""
    bp = _binom_profile(n)
    uniform = np.ones(n) / n
    a = (1 - alpha) * bp + alpha * uniform
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _adam_local_search(a, deadline, rng, lr=0.04, beta_frac=0.06, max_iter=5000, no_imp_limit=200):
    """Adam local search in log-space to polish a sequence."""
    n = len(a)

    a = a / a.sum()

    m = np.zeros(n)
    v = np.zeros(n)

def _lp_solve(a, M):
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2*n-1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a, M, t_mix=0.3):
    """One LP refinement step: find b with conv(a,b)<=M, mix with a."""
    b = _lp_solve(a, M)
    if b is None:
        return None
    s_b = float(b.sum())
    if s_b < 1e-10:
        return None
    b = b / s_b
    a_new = (1 - t_mix) * a + t_mix * b
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    return a_new

def _freq_flatten(a, alpha=0.4, n_iter=40):

def _chunk_swap(a, rng, n_swaps=3):
    """Swap random chunks."""
    n = len(a)
    a = a.copy()
    for _ in range(n_swaps):
        size = rng.integers(1, max(2, n // 8))
        i = rng.integers(0, n - size)
        j = rng.integers(0, n - size)
        if i == j:
            continue
        a[i:i+size], a[j:j+size] = a[j:j+size].copy(), a[i:i+size].copy()
    return a


def _section_reverse(a, rng):
    """Reverse a random section."""
    n = len(a)
    a = a.copy()
    start = rng.integers(0, max(1, n - 2))
    end = rng.integers(start + 2, n)
    a[start:end] = a[start:end][::-1]
    return a


def _interp_profile(p0, n_new):
    """Interpolate profile to new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return np.maximum(p0, 0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    return np.maximum(result, 0)

                    j = rng.integers(0, len(population))
                    if j != i:
                        cross_point = rng.integers(1, n - 1)
                        child = np.concatenate([population[i][:cross_point], population[j][cross_point:]])
                        child = np.maximum(child, 0)
                        cs = child.sum()
                        if cs > 1e-10:
                            child = child / cs
                            per = min(10.0, phase2_end - time.monotonic() - 5)
                            if per > 5:
                                child, _ = _adam_local_search(child, time.monotonic() + per, rng, lr=0.03)
                            offspring.append(child)

            if offspring:
                try_update(offchild := offspring[-1], f"gen{gen}_p{i}_op{op}")

        # Selection: rank all and keep top pop_size
        all_cands = population + offspring
        all_scores = [_score(c) for c in all_cands]
        idx_sorted = np.argsort(all_scores)
        population = [all_cands[idx_sorted[i]].copy() for i in range(min(pop_size, len(idx_sorted)))]
        # Ensure global best is in population
        if best_seq is not None:
            population[0] = best_seq.copy()

    # Phase 3: Multi-scale (200s)
    phase3_end = t_start + 1080.0
    print(f"Phase 3: Multi-scale, best={best_score:.10f}")

    if best_seq is not None:
        for n_new in [2048, 512, 4096]:
            if time.monotonic() > phase3_end - 30:
                break
            if n_new == n:
                continue
            p_interp = _interp_profile(best_seq, n_new)
            p_interp = p_interp / p_interp.sum()
            try_update(p_interp, f"interp_n{n_new}")

            if time.monotonic() < phase3_end - 40:
                per = min(60.0, phase3_end - time.monotonic() - 15)
                if per > 20:
                    p_pol, _ = _adam_local_search(p_interp, time.monotonic() + per, rng, lr=0.03, beta_frac=0.05)
                    try_update(p_pol, f"polish_n{n_new}")

            # Also try theoretical profiles at new n
            if time.monotonic() < phase3_end - 20:
                try_update(_binom_profile(n_new), f"binom_n{n_new}")
                try_update(_u_shape_profile(n_new), f"ushape_n{n_new}")

    # Phase 4: LP refinement-focused (150s)
    phase4_end = t_start + 1230.0
    print(f"Phase 4: LP refinement, best={best_score:.10f}")
    if best_seq is not None and best_seq.size <= 2048:
        p4, sc4 = _lp_refine_loop(best_seq, phase4_end, max_iters=40, t_start=0.3, m_shrink=0.997)
        try_update(p4, "lp_focus")

        # Adam after LP
        if time.monotonic() < phase4_end - 20:
            per = min(40.0, phase4_end - time.monotonic() - 10)
            if per > 15:
                p4a, _ = _adam_local_search(best_seq, time.monotonic() + per, rng, lr=0.02, beta_frac=0.04, no_imp_limit=80)
                try_update(p4a, "post_lp_adam")

    # Phase 5: Peak flood + freq (120s)
    phase5_end = t_start + 1350.0
    print(f"Phase 5: Flood + Freq, best={best_score:.10f}")
    if best_seq is not None:
        p5, sc5 = _peak_flood(best_seq, n_steps=200, eta=0.02)
        try_update(p5, "flood5")

        p5f, sc5f = _freq_flatten(best_seq, alpha=0.5, n_iter=50)
        try_update(p5f, "freq5")

    # Phase 6: Final intensive polish (remaining ~350s)
    phase6_end = deadline
    print(f"Phase 6: Final polish, best={best_score:.10f}")

    if best_seq is not None:
        remaining = phase6_end - time.monotonic()
        if remaining > 120:
            per1 = remaining * 0.35
            p6a, _ = _adam_local_search(best_seq, time.monotonic() + per1, rng, lr=0.015, beta_frac=0.03, no_imp_limit=100)
            try_update(p6a, "final_adam1")

        if time.monotonic() < phase6_end - 40 and best_seq.size <= 2048:
            per2 = min(50.0, phase6_end - time.monotonic() - 15)
            if per2 > 20:
                p6l, _ = _lp_refine_loop(best_seq, time.monotonic() + per2, max_iters=15, t_start=0.2, m_shrink=0.999)
                try_update(p6l, "final_lp")

        if time.monotonic() < phase6_end - 30:
            per3 = min(40.0, phase6_end - time.monotonic() - 10)
            if per3 > 15:
                p6f, _ = _peak_flood(best_seq, n_steps=100, eta=0.01)
                try_update(p6f, "final_flood")

        if time.monotonic() < phase6_end - 10:
            per4 = phase6_end - time.monotonic() - 5
            if per4 > 15:
                p6b, _ = _adam_local_search(best_seq, time.monotonic() + per4, rng, lr=0.01, beta_frac=0.02, no_imp_limit=50)
                try_update(p6b, "final_adam2")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _binom_profile(1024)

        exact_sc = _score(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    # Fallback
    fb = _binom_profile(1024)
    print(f"FALLBACK n=1024, score={_score(fb):.10f}")
    return [float(x) for x in fb]
