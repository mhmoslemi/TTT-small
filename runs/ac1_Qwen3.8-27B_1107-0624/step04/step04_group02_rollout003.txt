def _next_pow2(n):
    if n <= 0:

    return 1 << (n - 1).bit_length()


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)

def _score_full(a):

def _softmax_fn(z):

        # c[k_max] = sum_i a[i]*a[k_max-i]
        contributions = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                contributions[i] = a[i] * a[j]
        
        # Reduce the elements with highest contribution
        top_contrib = np.argsort(contributions)[-10:]
        a_trial = a.copy()
        for idx in top_contrib:
            a_trial[idx] *= 0.95
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial /= a_trial.sum()
        
        sc = _score_norm(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
    
    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _make_profile(n, kind="linear", param=1.5):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if kind == "linear":
        a = 1.0 + param * u
    elif kind == "quadratic":
        a = 1.0 + param * u * np.abs(u)
    elif kind == "cubic":
        a = 1.0 + param * u ** 3
    elif kind == "cosine":
        a = 0.5 * (1.0 + param * np.cos(np.pi * u))
    elif kind == "gaussian":
        a = np.exp(-u ** 2 / (2.0 * param ** 2))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: L-BFGS continuation on parent and diverse profiles
    Phase 2: Nesterov accelerated gradient with adaptive beta
    Phase 3: LP-based direction mixing
    Phase 4: Adam optimization
    Phase 5: Lag balancing and local search
    Phase 6: Multi-length exploration
    Phase 7: Final polishing
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:7.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "linear", 1.5)
        update_best(parent, _score_norm(parent), "default_parent")

    # === Phase 1: L-BFGS continuation on parent (400s) ===
    phase1_end = min(deadline, t_start + 400.0)
    print("\n=== Phase 1: L-BFGS continuation on parent ===")
    if time.monotonic() < phase1_end:
        dl = phase1_end - 5.0
        p_best, sc_best = _lbfgs_continuation(best_seq, dl, beta_scale=1.0)
        update_best(p_best, sc_best, "lbfgs_phase1")
        
        # Second pass with smaller beta
        if time.monotonic() < phase1_end - 30:
            dl2 = phase1_end - 5.0
            p_best2, sc_best2 = _lbfgs_continuation(p_best, dl2, beta_scale=0.3)
            update_best(p_best2, sc_best2, "lbfgs_phase1_fine")

    print(f"Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Nesterov accelerated gradient (300s) ===
    phase2_end = min(deadline, t_start + 700.0)
    print("\n=== Phase 2: Nesterov accelerated gradient ===")
    if time.monotonic() < phase2_end:
        dl = phase2_end - 5.0
        p_best, sc_best = _nesterov_logsumexp(best_seq, dl, beta_start=0.05, beta_end=0.001)
        update_best(p_best, sc_best, "nesterov")
        
        # Continue with smaller betas
        if time.monotonic() < phase2_end - 20:
            dl2 = phase2_end - 5.0
            p_best2, sc_best2 = _nesterov_logsumexp(best_seq, dl2, beta_start=0.01, beta_end=0.0001)
            update_best(p_best2, sc_best2, "nesterov_fine")

    print(f"Phase 2 done: best={best_score:.10f}")

    # === Phase 3: LP-based direction mixing (300s) ===
    phase3_end = min(deadline, t_start + 1000.0)
    print("\n=== Phase 3: LP-based direction mixing ===")
    if time.monotonic() < phase3_end and len(best_seq) <= 8192:
        n = len(best_seq)
        a = best_seq.copy()
        best_a_lp = a.copy()
        best_score_lp = _score_norm(a)
        
        n_iterations = 30
        t_mix = 0.1
        
        for it in range(n_iterations):
            if time.monotonic() > phase3_end - 5.0:
                break
            
            M = float(_compute_conv(a).max())
            g = _lp_direction(a, M, max_time=5.0)
            
            if g is not None:
                # Mix and evaluate
                a_trial = (1 - t_mix) * a + t_mix * g
                a_trial = np.maximum(a_trial, 0)
                if a_trial.sum() > 1e-15:
                    a_trial /= a_trial.sum()
                    sc = _score_norm(a_trial)
                    if sc < best_score_lp:
                        best_score_lp = sc
                        best_a_lp = a_trial.copy()
                        a = a_trial.copy()
                        # Compute update
                        elapsed = time.monotonic() - t_start
                        if sc < best_score:
                            update_best(a_trial, sc, f"lp_iter{it}")
            
            # Gradually increase mixing
            t_mix = min(0.3, 0.05 + 0.01 * it)
            
            # Occasionally do a gradient step after LP
            if it % 5 == 4 and time.monotonic() < phase3_end - 10:
                dl_mini = min(phase3_end - 5, time.monotonic() + 30.0)
                a_grad, sc_grad = _lbfgs_step(a, dl_mini, beta=0.005 * M)
                if sc_grad < best_score_lp:
                    best_score_lp = sc_grad
                    best_a_lp = a_grad.copy()
                    a = a_grad.copy()
        
        if best_score_lp < best_score:
            update_best(best_a_lp, best_score_lp, "lp_final")

    print(f"Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Adam optimization (200s) ===
    phase4_end = min(deadline, t_start + 1200.0)
    print("\n=== Phase 4: Adam optimization ===")
    if time.monotonic() < phase4_end:
        dl = phase4_end - 5.0
        p_best, sc_best = _adam_optimize(best_seq, dl, lr=0.01, beta_start=0.03, beta_end=0.001)
        update_best(p_best, sc_best, "adam")
        
        if time.monotonic() < phase4_end - 30:
            dl2 = phase4_end - 5.0
            p_best2, sc_best2 = _adam_optimize(best_seq, dl2, lr=0.005, beta_start=0.01, beta_end=0.0005)
            update_best(p_best2, sc_best2, "adam_fine")

    print(f"Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Lag balancing + local search (200s) ===
    phase5_end = min(deadline, t_start + 1400.0)
    print("\n=== Phase 5: Lag balancing + local search ===")
    if time.monotonic() < phase5_end:
        dl = min(phase5_end - 30, time.monotonic() + 100.0)
        p_best, sc_best = _lag_balancing(best_seq, dl, rng)
        update_best(p_best, sc_best, "lag_balance")
        
        if time.monotonic() < phase5_end - 10:
            dl2 = phase5_end - 5.0
            p_best2, sc_best2 = _local_search(best_seq, dl2, rng, n_trials=3000)
            update_best(p_best2, sc_best2, "local_search")

    print(f"Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Try different sequence lengths (200s) ===
    phase6_end = min(deadline, t_start + 1600.0)
    print("\n=== Phase 6: Multi-length exploration ===")
    if time.monotonic() < phase6_end and best_seq is not None:
        n_curr = len(best_seq)
        tried_lengths = set()
        
        # Try nearby lengths
        target_lengths = []
        if n_curr != 2048:
            target_lengths.append(2048)
        if n_curr != 8192:
            target_lengths.append(8192)
        if n_curr != 3072:
            target_lengths.append(3072)
        
        for tl in target_lengths:
            if time.monotonic() > phase6_end - 30:
                break
            
            p_interp = _interp_profile(best_seq, tl)
            sc_interp = _score_norm(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{tl}")
            
            # Quick optimization at this length
            if time.monotonic() < phase6_end - 20:
                dl = min(phase6_end - 10, time.monotonic() + 60.0)
                p_opt, sc_opt = _lbfgs_continuation(p_interp, dl, beta_scale=0.5)
                update_best(p_opt, sc_opt, f"lbfgs_n{tl}")

    print(f"Phase 6 done: best={best_score:.10f}")

    # === Phase 7: Final intensive polishing (remaining time) ===
    print("\n=== Phase 7: Final polishing ===")
    if best_seq is not None:
        remaining = max(10.0, deadline - time.monotonic())
        
        # L-BFGS with very small beta
        if remaining > 20:
            dl = min(deadline - 20, time.monotonic() + remaining * 0.4)
            p_best, sc_best = _lbfgs_continuation(best_seq, dl, beta_scale=0.2)
            update_best(p_best, sc_best, "final_lbfgs")
        
        # Adam with tiny lr
        if time.monotonic() < deadline - 20:
            dl = min(deadline - 10, time.monotonic() + min(60.0, (deadline - time.monotonic()) * 0.3))
            p_best, sc_best = _adam_optimize(best_seq, dl, lr=0.003, beta_start=0.005, beta_end=0.0005)
            update_best(p_best, sc_best, "final_adam")
        
        # Local search with very small perturbations
        if time.monotonic() < deadline - 5:
            dl = deadline - 5.0
            p_best, sc_best = _local_search(best_seq, dl, rng, n_trials=2000)
            update_best(p_best, sc_best, "final_local")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_profile(4096, "linear", 1.5)
            s = float(final_seq.sum())
        # Keep as is (don't normalize - the evaluator handles it via sum_a^2)
        # Actually, let's normalize to sum=1 for consistency
        final_seq = final_seq / s
        
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={float(final_seq.sum()):.6f}")
        
        # Validate with evaluator
        if len(final_seq) <= 8192:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))

                    print(f"Evaluator score: {sc_eval:.10f}")
                    if abs(sc_eval - sc_final) > 0.01:
                        print(f"WARNING: mismatch! internal={sc_final:.10f}, eval={sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")
        
        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profile(4096, "linear", 1.5)
    print(f"FALLBACK: n=4096, score={_score_norm(fallback):.10f}")
    return [float(x) for x in fallback]
