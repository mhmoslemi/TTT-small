"""
Optimization for minimizing max autocorrelation of non-negative step functions.
Multi-phase approach: (1) sharp smooth-max Adam, (2) LP with dual-guided weighting,
(3) stochastic peak-subset gradients, (4) multi-scale n=1024/2048, (5) direct peak
targeting via coordinate surgery, (6) final polish with very sharp smooth-max.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _safe_normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-15, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.ones(a.size) / a.size
    return a / s


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(np.clip(x, -60.0, 0.0))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _adam_logit_sharp(p0, deadline, rng, lr=0.05, beta_init=0.03, beta_decay=0.95,
                       no_imp_limit=250, sharp_phase=False):
    """Adam in log-space with adaptive sharp smooth-max."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-6, beta_init * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    it = 0

    while time.monotonic() < deadline:
        it += 1
        _, w = _smooth_max_weights(c, beta)
        g = _conv_cross_corr(w, p)
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 150 == 0:
                    lr = max(0.001, lr * 0.97)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if it % 20 == 0:
            beta = max(1e-7, beta * beta_decay)

        if no_imp > no_imp_limit:
            if best_score < 1.51:
                ns = 0.04
            elif best_score < 1.53:
                ns = 0.08
            else:
                ns = 0.15
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(beta_init * maxc, 1e-6)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)

            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if it % 30 == 0:

def _lp_solve_with_dual(a, M):
    """Solve LP and return solution + dual variables."""
    n = len(a)
    n_cons = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(n_cons):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(n_cons, n))
    c_obj = -np.ones(n)
    b_ub = M * np.ones(n_cons)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        dual = np.maximum(res.unequality.constraint_multiplicers, 0.0) if hasattr(res.unequality, 'constraint_multiplicers') else np.zeros(n_cons)
        return np.maximum(res.x, 0.0), dual
    return None, None


def _lp_refine_dual_guided(a, deadline, max_solves=60, t_start=0.35, m_shrink=0.997):
    """LP refinement using dual variables to create weighted gradients."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0
    t = t_start
    dual_cache = np.zeros(2 * n - 1)

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1

        b, dual = _lp_solve_with_dual(a, M)
        if b is None:
            b, dual = _lp_solve_with_dual(a, M * 1.005)
            if b is None:
                break

        s_b = float(b.sum())
        if s_b < 1e-10:
            break

        # Normalize dual for use as weights
        if dual is not None and dual.sum() > 1e-15:
            dual_cache = 0.5 * dual_cache + 0.5 * (dual / dual.sum())

        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)

        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        c = _full_convolve(a)
        M_new = float(c.max())
        M = max(M_new * m_shrink, (best_score / (2.0 * n)) * 1.003)
        t = max(0.1, t * 0.93)

    return best_a, best_score, dual_cache

def _peak_surgery(p0, deadline, rng, n_rounds=200):
    """Direct peak targeting: surgically reduce the argmax convolution value."""
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    n = p.size
    best_p = p.copy()
    best_score = _score_norm(p)
    it = 0

    while time.monotonic() < deadline and it < n_rounds:
        it += 1
        c = _full_convolve(p)
        M = float(c.max())
        score = float(2.0 * n * M)

        if score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()

        # Find top peak
        k_star = int(np.argmax(c))

        # Find contributing pairs
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        idx = np.arange(i_start, i_end + 1)
        contrib = p[idx] * p[k_star - idx]

        if contrib.max() < 1e-15:
            continue

        # Find the pair with max contribution
        local_max = int(np.argmax(contrib))
        i_big = i_start + local_max

        # Find a "safe" target position (one that doesn't contribute to top peaks)
        top_peaks = np.argsort(c)[::-1][:min(10, c.size)]
        bad_positions = set()
        for kp in top_peaks:
            ks = max(0, int(kp) - n + 1)
            ke = min(n - 1, int(kp))
            for ii in range(ks, ke + 1):
                bad_positions.add(ii)
                bad_positions.add(int(kp) - ii)

        safe_positions = [i for i in range(n) if i not in bad_positions and i != i_big]
        if not safe_positions:
            safe_positions = [i for i in range(n) if i != i_big]
        if not safe_positions:
            continue

        target = safe_positions[rng.integers(0, len(safe_positions))]

        # Move a small amount of mass
        frac = 0.02 + 0.03 * rng.random()
        amount = p[i_big] * frac
        p[i_big] -= amount
        p[target] += amount
        p = np.maximum(p, 1e-12)
        p = p / p.sum()

    return best_p, best_score


def _make_profile(n, kind, param):
    """Generate various profile initializations."""
    t = np.linspace(0.0, 1.0, n)
    if kind == 'linear':
        a = 1.0 + param * (t - 0.5)
    elif kind == 'quadratic':
        u = 2.0 * t - 1.0
        a = 1.0 + param * u ** 2
    elif kind == 'cubic':
        u = 2.0 * t - 1.0
        a = 1.0 + param * u + param * 0.5 * u ** 3
    elif kind == 'exp_decay':
        a = np.exp(-param * t)
    elif kind == 'bump_left':
        a = np.exp(-param * (t - 0.3) ** 2) + 0.3 * np.exp(-param * (t - 0.7) ** 2)
    elif kind == 'bump_right':
        a = 0.3 * np.exp(-param * (t - 0.3) ** 2) + np.exp(-param * (t - 0.7) ** 2)
    elif kind == 'two_bumps':
        a = np.exp(-param * (t - 0.25) ** 2) + np.exp(-param * (t - 0.75) ** 2)
    elif kind == 'cosine':
        a = 1.0 + param * np.cos(2.0 * np.pi * t)
    elif kind == 'sine_skew':
        a = 1.0 + param * np.sin(2.0 * np.pi * t) * t
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _safe_normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _safe_normalize(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Multi-phase optimizer for minimum-max-autocorrelation non-negative sequences."""
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, score, label=""):
        nonlocal best_seq, best_score
        if p is not None:
            sc = _score_norm(p)
            if np.isfinite(sc) and sc < best_score:
                best_seq = np.asarray(p, dtype=np.float64).copy()
                best_score = float(sc)
                print(f"  NEW BEST: {best_score:.10f} n={p.size} [{label}] t={time.monotonic()-t_start:.0f}s")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={parent.size}, score={_score_norm(parent):.10f}")
    else:
        parent = _make_profile(1024, 'linear', 2.0)
        print(f"No parent, using linear init n={parent.size}")

    # === Phase 1: Sharp Adam on parent (200s) ===
    p1_end = t_start + 200.0
    print(f"\n=== Phase 1: Sharp Adam (0-{p1_end-t_start:.0f}s) ===")
    p1, sc1 = _adam_logit_sharp(parent, p1_end, rng, lr=0.05, beta_init=0.03,
                                 beta_decay=0.95, no_imp_limit=200)
    _update(p1, sc1, 'phase1_adam')

    # === Phase 2: LP with dual guidance (300s) ===
    p2_end = t_start + 500.0
    print(f"\n=== Phase 2: LP + dual ({p1_end-t_start:.0f}s-{p2_end-t_start:.0f}s) ===")
    if best_seq is not None and best_seq.size <= 2048:
        p2, sc2, dual_w = _lp_refine_dual_guided(best_seq, p2_end, max_solves=50,
                                                   t_start=0.35, m_shrink=0.997)
        _update(p2, sc2, 'phase2_lp')

        # Dual-guided Adam
        if time.monotonic() < p2_end - 15:
            p2a, sc2a = _dual_guided_adam(best_seq, dual_w, time.monotonic() + 30,
                                           rng, lr=0.03, beta_init=0.02)
            _update(p2a, sc2a, 'phase2_dual_adam')

    # === Phase 3: Stochastic peak-subset Adam (200s) ===
    p3_end = t_start + 700.0
    print(f"\n=== Phase 3: Stochastic Adam ({p2_end-t_start:.0f}s-{p3_end-t_start:.0f}s) ===")
    if best_seq is not None:
        p3, sc3 = _stochastic_adam(best_seq, p3_end, rng, lr=0.04, n_peaks=48,
                                    beta_init=0.02)
        _update(p3, sc3, 'phase3_stoch')

        # Peak surgery
        if time.monotonic() < p3_end - 10:
            p3s, sc3s = _peak_surgery(best_seq, min(p3_end, time.monotonic() + 40),
                                       rng, n_rounds=100)
            _update(p3s, sc3s, 'phase3_surgery')

    # === Phase 4: Multi-scale n=2048 (300s) ===
    p4_end = t_start + 1000.0
    print(f"\n=== Phase 4: Multi-scale n=2048 ({p3_end-t_start:.0f}s-{p4_end-t_start:.0f}s) ===")
    if time.monotonic() < p4_end and best_seq is not None:
        # Interpolate to 2048
        p4_int = _interp_profile(best_seq, 2048)
        sc4_int = _score_norm(p4_int)
        print(f"  Interp to n=2048: {sc4_int:.10f}")
        _update(p4_int, sc4_int, 'interp_2048')

        # Adam at 2048
        per4 = min(120.0, p4_end - time.monotonic() - 10)
        if per4 > 30:
            p4a, sc4a = _adam_logit_sharp(p4_int, time.monotonic() + per4, rng,
                                            lr=0.04, beta_init=0.025,
                                            beta_decay=0.95, no_imp_limit=150)
            _update(p4a, sc4a, 'adam_2048')

        # Also try interpolating to 4096
        if time.monotonic() < p4_end - 60:
            p4b_int = _interp_profile(best_seq, 4096)
            sc4b = _score_norm(p4b_int)
            print(f"  Interp to n=4096: {sc4b:.10f}")
            _update(p4b_int, sc4b, 'interp_4096')

    # === Phase 5: Back to best n, LP + surgery (200s) ===
    p5_end = t_start + 1200.0
    print(f"\n=== Phase 5: Refine best ({p4_end-t_start:.0f}s-{p5_end-t_start:.0f}s) ===")
    if best_seq is not None and best_seq.size <= 2048:
        p5, sc5, _ = _lp_refine_dual_guided(best_seq, min(p5_end, time.monotonic() + 100),
                                              max_solves=30, t_start=0.25, m_shrink=0.998)
        _update(p5, sc5, 'phase5_lp')

        if time.monotonic() < p5_end - 20:
            p5s, sc5s = _peak_surgery(best_seq, time.monotonic() + 40, rng, n_rounds=80)
            _update(p5s, sc5s, 'phase5_surgery')

    # === Phase 6: Alternative initializations (200s) ===
    p6_end = t_start + 1400.0
    print(f"\n=== Phase 6: Alt inits ({p5_end-t_start:.0f}s-{p6_end-t_start:.0f}s) ===")
    if time.monotonic() < p6_end:
        n_try = 1024
        alts = [
            ('exp_3', 'exp_decay', 3.0),
            ('bumpL_8', 'bump_left', 8.0),
            ('bumpR_8', 'bump_right', 8.0),
            ('two_6', 'two_bumps', 6.0),
            ('sine_1.5', 'sine_skew', 1.5),
            ('cos_1.2', 'cosine', 1.2),
            ('quad_2', 'quadratic', 2.0),
        ]
        per_alt = max(10.0, (p6_end - time.monotonic()) / len(alts))
        for label, kind, param in alts:
            if time.monotonic() > p6_end - 5:
                break
            pa = _make_profile(n_try, kind, param)
            sc_a = _score_norm(pa)
            # Quick Adam
            per = min(per_alt, p6_end - time.monotonic() - 3)
            if per < 8:
                break
            p_alt, sc_alt = _adam_logit_sharp(pa, time.monotonic() + per, rng,
                                                lr=0.06, beta_init=0.04,
                                                beta_decay=0.94, no_imp_limit=100)
            _update(p_alt, sc_alt, f'alt_{label}')

    # === Phase 7: Final intensive polish (remaining time) ===
    p7_end = deadline
    print(f"\n=== Phase 7: Final polish ({p6_end-t_start:.0f}s-{p7_end-t_start:.0f}s) ===")
    if time.monotonic() < p7_end and best_seq is not None:
        remaining = p7_end - time.monotonic()
        print(f"  Time remaining: {remaining:.0f}s, current best: {best_score:.10f}")

        # Very sharp Adam
        per7a = remaining * 0.4
        if per7a > 20:
            p7a, sc7a = _adam_logit_sharp(best_seq, time.monotonic() + per7a, rng,
                                            lr=0.02, beta_init=0.01,
                                            beta_decay=0.97, no_imp_limit=100)
            _update(p7a, sc7a, 'final_sharp_adam')

        # LP pass
        if time.monotonic() < p7_end - 30 and best_seq.size <= 2048:
            per7b = min(60.0, p7_end - time.monotonic() - 10)
            if per7b > 20:
                p7b, sc7b, _ = _lp_refine_dual_guided(best_seq, time.monotonic() + per7b,
                                                        max_solves=15, t_start=0.2,
                                                        m_shrink=0.999)
                _update(p7b, sc7b, 'final_lp')

        # Peak surgery
        if time.monotonic() < p7_end - 15:
            per7c = min(40.0, p7_end - time.monotonic() - 5)
            if per7c > 10:
                p7c, sc7c = _peak_surgery(best_seq, time.monotonic() + per7c, rng,
                                            n_rounds=60)
                _update(p7c, sc7c, 'final_surgery')

        # Final Adam
        if time.monotonic() < p7_end - 5:
            per7d = p7_end - time.monotonic() - 5
            if per7d > 10:
                p7d, sc7d = _adam_logit_sharp(best_seq, time.monotonic() + per7d, rng,
                                                lr=0.01, beta_init=0.005,
                                                beta_decay=0.98, no_imp_limit=50)
                _update(p7d, sc7d, 'final_adam2')

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile(1024, 'linear', 2.0)

        exact_sc = _score_norm(final)
        print(f"\nFINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile(1024, 'linear', 2.0)
    print(f"FALLBACK n=1024 score={_score_norm(fb):.10f}")
    return [float(x) for x in fb]
