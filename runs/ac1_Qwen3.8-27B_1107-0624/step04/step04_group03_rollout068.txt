"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines adaptive L-BFGS, Nesterov acceleration, free-parameter gradient
descent with projection, subgradient line search, and Adam polishing.
Uses diverse initializations and multi-restart to escape local minima.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _conv_w_with_a(w, a):
    n = len(a)
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    return h


def _smooth_objective(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _smooth_gradient(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    h = _conv_w_with_a(w, a)
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _cv_of_h(w, a):
    h = _conv_w_with_a(w, a)
    mean_h = float(np.mean(h))
    if mean_h < 1e-15:
        return 1e10
    std_h = float(np.std(h))
    return std_h / mean_h


def _lbfgs_adaptive(a_init, deadline, max_wall_time=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    scale = max_c0

    if max_wall_time is None:
        max_wall_time = min(300.0, max(1.0, deadline - time.monotonic()))

    betas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002,
             0.0001, 5e-5, 2e-5, 1e-5, 5e-6]
    betas = [b * scale for b in betas]

    per_level = max_wall_time / len(betas)

    for i, beta in enumerate(betas):
        if time.monotonic() >= deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level)

        a_curr = _softmax_fn(z)
        c_curr = _full_convolve(a_curr, a_curr)
        w_curr, _ = _compute_weights(c_curr, beta)
        eff = np.exp(-np.sum(w_curr * np.log(w_curr + 1e-300)))
        if eff < 1.5 and i > 2:
            continue

        try:
            result = minimize(
                _smooth_objective, z, args=(beta,),
                method='L-BFGS-B', jac=_smooth_gradient,
                options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 3000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x.copy()
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                z = z_new
        except Exception:
            pass

    return best_a, best_score


def _nesterov_adaptive(a_init, deadline, beta_fracs, lr_init=0.02, mom_init=0.9):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    scale = max_c

    lr = lr_init
    mom = mom_init
    t = 0

    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-12, beta_frac * scale)

        stall_count = 0
        while time.monotonic() < deadline:
            t += 1
            grad_y = _smooth_gradient(y, beta)
            if not np.all(np.isfinite(grad_y)):
                grad_y = _smooth_gradient(z, beta)

            z_new = z - lr * grad_y
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -50, 50)

            y_new = z_new + mom * (z_new - z)
            y_new = y_new - y_new.mean()

            z = z_new
            y = y_new

            if t % 100 == 0:
                a_curr = _softmax_fn(z)
                sc = _score_norm_fn(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()
                    stall_count = 0
                else:
                    stall_count += 1
                    if stall_count > 8:
                        break

            if t % 200 == 0:
                mom = min(0.999, mom + 0.0003)

        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        max_c = float(c.max())
        scale = max_c
        lr = max(0.002, lr * 0.75)

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score

    """Optimize directly in a-space with projection onto sum=1, a>=0."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm_fn(a)

    lr = lr_init
    momentum = np.zeros(n)
    mom_coeff = 0.9
    c = _full_convolve(a, a)
    max_c = float(c.max())
    scale = max_c
    beta_frac = 0.01
    t = 0

    for it in range(200000):
        if time.monotonic() > deadline:
            break
        t += 1
        beta = max(1e-12, beta_frac * scale)

        a = a_init if False else a  # keep reference
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        h = _conv_w_with_a(w, a)
        grad = 2.0 * n * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        momentum = mom_coeff * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s_new
        a = a_new

        if t % 100 == 0:

            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                beta_frac *= 0.85
                c_new = _full_convolve(a, a)
                max_c = float(c_new.max())
                scale = max_c
            else:
                lr = max(0.0005, lr * 0.95)

    return best_a, best_score


def _subgradient_line_search(a_init, deadline):
    """Exact subgradient with backtracking line search in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    step = 0.01

    for it in range(5000):
        if time.monotonic() > deadline:
            break

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        k_star = int(np.argmax(c))

        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for m in range(lo, hi + 1):
            grad[m] = 2.0 * a[k_star - m]
        grad = 2.0 * n * grad

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad_dir = -grad / grad_norm

        found = False
        for trial in range(8):
            s_try = step / (2.0 ** trial)
            z_try = z + s_try * grad_dir
            z_try = z_try - z_try.mean()
            z_try = np.clip(z_try, -50, 50)
            a_try = _softmax_fn(z_try)
            sc_try = _score_norm_fn(a_try)
            if np.isfinite(sc_try) and sc_try < best_score:
                z = z_try
                if sc_try < best_score:
                    best_score = sc_try
                    best_a = a_try.copy()
                step = min(0.05, step * 1.3)
                found = True
                break
        if not found:
            step = step * 0.5

    return best_a, best_score


def _truncated_gauss_subgradient(a_init, deadline):
    """Subgradient using truncated active set with momentum."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    lr = 0.003
    momentum = np.zeros(n)
    mom_coeff = 0.95

    for it in range(10000):
        if time.monotonic() > deadline:
            break

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        threshold = cm * 0.995
        active = c >= threshold
        if not np.any(active):
            active[int(np.argmax(c))] = True

        w = np.zeros_like(c)
        w[active] = 1.0 / np.sum(active)
        W = float(np.dot(w, c))
        h = _conv_w_with_a(w, a)
        grad = 4.0 * n * a * (h - W)

        momentum = mom_coeff * momentum - lr * grad
        z = z + momentum
        z = z - z.mean()
        z = np.clip(z, -50, 50)

        if it % 50 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            else:
                lr = lr * 0.95

    return best_a, best_score


def _adam_optimize(a_init, deadline, lr=0.05, beta_frac=0.01):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    scale = max_c
    beta = max(1e-12, beta_frac * scale)

    for it in range(200000):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        h = _conv_w_with_a(w, a)
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a = _softmax_fn(z)

        if t % 200 == 0:
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
            beta = max(1e-12, beta * 0.85)
            c_new = _full_convolve(a, a)
            max_c = float(c_new.max())
            scale = max_c

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_sinusoidal_profile(n, freq, amp):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.sin(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_profile(n, width):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) ** 2) / (2.0 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _candidate_sort_key(item):
    return item[0]

    print(f"Phase 0 done: best={best_score:.10f} (n={n_best})")

    # --- Phase 1: Adaptive L-BFGS (350s) ---
    phase1_end = min(deadline, t_start + 375.0)
    per_time = min(300.0, phase1_end - time.monotonic())
    if per_time > 20:
        p_b, s_b = _lbfgs_adaptive(best_seq, phase1_end, max_wall_time=per_time)
        if s_b < best_score:
            best_score = s_b
            best_seq = p_b.copy()
            print(f"  [{time.monotonic()-t_start:5.1f}s] lbfgs: {s_b:.10f}")

    for idx in range(1, min(4, len(candidates))):
        if time.monotonic() > phase1_end - 20:
            break
        sc_c, p_c, pn_c, label_c = candidates[idx]
        if pn_c != n_best:
            continue
        per_time_c = min(50.0, phase1_end - time.monotonic() - 5)
        if per_time_c > 10:
            p_b, s_b = _lbfgs_adaptive(p_c, phase1_end, max_wall_time=per_time_c)
            if s_b < best_score:
                best_score = s_b
                best_seq = p_b.copy()
                print(f"  [{time.monotonic()-t_start:5.1f}s] lbfgs_{label_c}: {s_b:.10f}")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Nesterov aggressive (300s) ---
    phase2_end = min(deadline, t_start + 675.0)
    if time.monotonic() < phase2_end:
        beta_fracs = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002,
                      0.0001, 5e-5, 2e-5, 1e-5, 5e-6]
        per_time = min(280.0, phase2_end - time.monotonic())
        p_b, s_b = _nesterov_adaptive(best_seq, min(phase2_end - 5, time.monotonic() + per_time),
                                       beta_fracs, lr_init=0.015, mom_init=0.9)
        if s_b < best_score:
            best_score = s_b
            best_seq = p_b.copy()
            print(f"  [{time.monotonic()-t_start:5.1f}s] nesterov: {s_b:.10f}")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Free parameter optimization (250s) ---
    phase3_end = min(deadline, t_start + 925.0)
    if time.monotonic() < phase3_end:
        per_time = min(200.0, phase3_end - time.monotonic())
        p_b, s_b = _free_param_optimize(best_seq, min(phase3_end - 5, time.monotonic() + per_time),
                                         lr_init=0.003)
        if s_b < best_score:
            best_score = s_b
            best_seq = p_b.copy()
            print(f"  [{time.monotonic()-t_start:5.1f}s] free_param: {s_b:.10f}")

    # --- Phase 4: Subgradient line search (150s) ---
    phase4_end = min(deadline, t_start + 1075.0)
    if time.monotonic() < phase4_end:
        per_time = min(120.0, phase4_end - time.monotonic())
        p_b, s_b = _subgradient_line_search(best_seq, min(phase4_end - 5, time.monotonic() + per_time))
        if s_b < best_score:
            best_score = s_b
            best_seq = p_b.copy()
            print(f"  [{time.monotonic()-t_start:5.1f}s] subgrad_ls: {s_b:.10f}")

    # Truncated Gaussian subgradient
    if time.monotonic() < phase4_end - 10:
        per_time = min(60.0, phase4_end - time.monotonic() - 5)
        p_b, s_b = _truncated_gauss_subgradient(best_seq, min(phase4_end - 2, time.monotonic() + per_time))
        if s_b < best_score:
            best_score = s_b
            best_seq = p_b.copy()
            print(f"  [{time.monotonic()-t_start:5.1f}s] trunc_gauss: {s_b:.10f}")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Adam polishing (200s) ---
    phase5_end = min(deadline, t_start + 1275.0)
    if time.monotonic() < phase5_end:
        per_time = min(150.0, phase5_end - time.monotonic())
        p_b, s_b = _adam_optimize(best_seq, min(phase5_end - 5, time.monotonic() + per_time),
                                   lr=0.03, beta_frac=0.005)
        if s_b < best_score:
            best_score = s_b
            best_seq = p_b.copy()
            print(f"  [{time.monotonic()-t_start:5.1f}s] adam: {s_b:.10f}")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Multi-restart with noise (250s) ---
    phase6_end = min(deadline, t_start + 1525.0)
    n_best = len(best_seq)
    if time.monotonic() < phase6_end:
        n_restarts = 5
        per_restart = min(45.0, (phase6_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 10:
                break
            noise_level = 0.03 + 0.02 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_level * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            per_dl = min(phase6_end - 5, time.monotonic() + per_restart)
            beta_fracs_short = [0.003, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]
            p_b, s_b = _nesterov_adaptive(p_start, per_dl, beta_fracs_short,
                                           lr_init=0.01, mom_init=0.85)
            if s_b < best_score:
                best_score = s_b
                best_seq = p_b.copy()
                print(f"  [{time.monotonic()-t_start:5.1f}s] restart_{i}: {s_b:.10f}")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final fine polish (remaining) ---
    if best_seq is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()

        if remaining > 60:
            per_time = min(50.0, remaining - 40)
            beta_fracs_final = [5e-6, 2e-6, 1e-6, 5e-7, 2e-7]
            p_b, s_b = _nesterov_adaptive(best_seq, min(deadline - 40, time.monotonic() + per_time),
                                           beta_fracs_final, lr_init=0.003, mom_init=0.97)
            if s_b < best_score:
                best_score = s_b
                best_seq = p_b.copy()
                print(f"  [{time.monotonic()-t_start:5.1f}s] final_nesterov: {s_b:.10f}")

        if time.monotonic() < deadline - 20:
            per_time = min(30.0, deadline - time.monotonic() - 10)
            if per_time > 5:
                p_b, s_b = _adam_optimize(best_seq, min(deadline - 5, time.monotonic() + per_time),
                                           lr=0.005, beta_frac=0.0005)
                if s_b < best_score:
                    best_score = s_b
                    best_seq = p_b.copy()
                    print(f"  [{time.monotonic()-t_start:5.1f}s] final_adam: {s_b:.10f}")

    print(f"Phase 7 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_linear_profile(2048, 1.0)
            s = float(final_seq.sum())
            final_seq = final_seq / s

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned non-finite, using internal: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    fallback = _make_linear_profile(2048, 1.0)
    print(f"FALLBACK: n=2048, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
