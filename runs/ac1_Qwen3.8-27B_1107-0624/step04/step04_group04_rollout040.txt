for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses game-theoretic dual updates, adaptive active sets, and periodic

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _active_set_gradient(a, c, top_k=20, threshold_frac=0.995):
    """Compute the average gradient over the active set of peaks."""
    n = len(a)
    cm = float(c.max())
    # Active set: top-k peaks OR peaks above threshold
    idx_top = np.argsort(c)[-top_k:]
    idx_thresh = np.where(c >= cm * threshold_frac)[0]
    # Union of both
    active = np.unique(np.concatenate([idx_top, idx_thresh]))
    
    # Compute gradient: for each active k, grad_k[j] = 2*a[k-j] if 0<=k-j<n
    # Average gradient over active set
    avg_grad = np.zeros(n)
    for k in active:
        k = int(k)
        # grad_k[j] = 2*a[k-j] if 0 <= k-j < n, i.e., j in [max(0,k-n+1), min(k, n-1)]
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            avg_grad[lo:hi+1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    avg_grad /= len(active)
    return avg_grad, active


def _multiplicative_update(a, grad, eta):
    """Multiplicative (exponentiated) update preserving non-negativity."""
    # Center the gradient to preserve sum approximately
    grad_centered = grad - grad.mean()
    a_new = a * np.exp(-eta * grad_centered)
    s = a_new.sum()
    if s < 1e-300 or not np.isfinite(s):
        return a.copy()
    return a_new / s


def _logsumexp_smooth_obj(a, c, beta, top_k=0):
    """Compute logsumexp-smoothed max of convolution."""
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_norm = c_top - c_top.max()
        return c_top.max() + beta * float(np.log(np.sum(np.exp(c_norm / beta))))
    else:
        c_shifted = (c - cm) / beta
        return cm + beta * float(np.log(np.sum(np.exp(c_shifted))))

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

    # === Phase 2: Fine L-BFGS with decreasing beta ===
    print("=== Phase 2: Fine L-BFGS ===")
    phase2_end = min(deadline, t_start + 1000.0)
    
    if best_seq is not None and time.monotonic() < phase2_end:
        a = best_seq.copy() / best_seq.sum()
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        betas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        per_beta = max(5.0, (phase2_end - time.monotonic()) / len(betas))
        
        for beta in betas:
            if time.monotonic() > phase2_end - 10:
                break
            dl = min(phase2_end - 5, time.monotonic() + per_beta)
            a_new, sc_new = _lbfgs_smooth(a, dl, beta=beta * cm, top_k=20, max_iter_lbfgs=200)
            if sc_new < best_score:
                best_score = sc_new
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"    [beta={beta*cm:.6f}] score={sc_new:.10f}")
    
    print(f"  Phase 2 done: best={best_score:.10f}")
    
    # === Phase 3: KKT Balancing ===
    print("=== Phase 3: KKT Balancing ===")
    phase3_end = min(deadline, t_start + 1300.0)
    
    if best_seq is not None and time.monotonic() < phase3_end:
        a = best_seq.copy() / best_seq.sum()
        
        for it in range(5000):
            if time.monotonic() > phase3_end:
                break
            
            c = _full_convolve(a, a)
            cm = float(c.max())
            
            # Active set: peaks within 0.5% of max
            active = np.where(c >= cm * 0.9995)[0]
            if len(active) == 0:
                active = np.where(c >= cm * 0.995)[0]
            if len(active) == 0:
                break
            
            # Compute balanced direction
            h = np.zeros(len(a))
            for k_idx in active:
                k_val = int(k_idx)
                lo = max(0, k_val - len(a) + 1)
                hi = min(k_val, len(a) - 1)
                if lo <= hi:
                    h[lo:hi+1] += a[k_val - hi:k_val - lo + 1][::-1]
            h /= len(active)
            
            W = float(np.dot(h, a))
            
            # Find step size via line search
            best_step = 0
            best_cm = cm
            for step_factor in [0.1, 0.05, 0.02, 0.01, 0.005]:
                step = step_factor * (cm - float(c[active].min())) / (cm + 1e-15)
                step = max(step, 1e-6)
                z = np.log(np.maximum(a, 1e-15))
                z = z - z.mean()
                z = z - step * (h - W) / (np.abs(h - W).max() + 1e-15)
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                a_trial = _softmax_fn(z)
                c_trial = _full_convolve(a_trial, a_trial)
                cm_trial = float(c_trial.max())
                if cm_trial < best_cm:
                    best_cm = cm_trial
                    best_step = step
                    best_a_trial = a_trial.copy()
            
            if best_step > 0:
                a = best_a_trial
                sc = _score_norm_fn(a)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"    [KKT iter {it}] score={sc:.10f}")
            else:
                break
        
    print(f"  Phase 3 done: best={best_score:.10f}")
    
    # === Phase 4: Multi-restart with diverse perturbations ===
    print("=== Phase 4: Multi-restart ===")
    phase4_end = min(deadline, t_start + 1550.0)
    
    if best_seq is not None and time.monotonic() < phase4_end:
        n_restarts = 6
        per_restart = max(10.0, (phase4_end - time.monotonic()) / n_restarts)
        
        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 15:
                break
            
            # Different perturbation strategies
            strategies = [
                lambda: 0.1 * rng.normal(size=len(best_seq)),
                lambda: 0.2 * rng.normal(size=len(best_seq)),
                lambda: 0.05 * rng.normal(size=len(best_seq)) + 0.1 * np.cos(2*np.pi*rng.uniform(0,10)*np.arange(len(best_seq))/len(best_seq)),
                lambda: 0.15 * rng.normal(size=len(best_seq)),
                lambda: 0.3 * rng.normal(size=len(best_seq)),
                lambda: 0.08 * rng.normal(size=len(best_seq)),
            ]
            
            noise = strategies[i % len(strategies)]()
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            a_start = _softmax_fn(z)
            
            # Quick optimization: multiplicative gradient + L-BFGS
            eta_r = 0.05
            for inner_it in range(200):
                if time.monotonic() > phase4_end - 20:
                    break
                c_r = _full_convolve(a_start, a_start)
                cm_r = float(c_r.max())
                avg_grad_r, _ = _active_set_gradient(a_start, c_r, top_k=15, threshold_frac=0.995)
                a_start = _multiplicative_update(a_start, avg_grad_r, eta_r)
                c_r2 = _full_convolve(a_start, a_start)
                if float(c_r2.max()) >= cm_r:
                    eta_r *= 0.9
            
            # L-BFGS polish
            dl = min(phase4_end - 10, time.monotonic() + per_restart * 0.4)
            a_r, sc_r = _lbfgs_smooth(a_start, dl, beta=0.01, top_k=10, max_iter_lbfgs=100)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"    [restart {i}] score={sc_r:.10f}")
    
    print(f"  Phase 4 done: best={best_score:.10f}")
    
    # === Phase 5: Length variation ===
    print("=== Phase 5: Length variation ===")
    phase5_end = min(deadline, t_start + 1700.0)
    
    if best_seq is not None and time.monotonic() < phase5_end:
        n_curr = len(best_seq)
        for target_n in [2048, 8192]:
            if time.monotonic() > phase5_end - 60:
                break
            if target_n == n_curr:
                continue
            
            # Interpolate
            x0 = np.linspace(0, 1, n_curr)
            x1 = np.linspace(0, 1, target_n)
            p_interp = np.interp(x1, x0, best_seq)
            p_interp = np.maximum(p_interp, 1e-12)
            p_interp /= p_interp.sum()
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{target_n}")
            
            # Optimize at new length
            if time.monotonic() < phase5_end - 30:
                dl = min(phase5_end - 20, time.monotonic() + 60.0)
                a_new, sc_new = _lbfgs_smooth(p_interp, dl, beta=0.01, top_k=15, max_iter_lbfgs=150)
                update_best(a_new, sc_new, f"lbfgs_n{target_n}")
                
                if time.monotonic() < phase5_end - 20:
                    # Quick multiplicative gradient
                    a_mg = a_new.copy() / a_new.sum()
                    eta_mg = 0.05
                    for _ in range(100):
                        if time.monotonic() > phase5_end - 25:
                            break
                        c_mg = _full_convolve(a_mg, a_mg)
                        cm_mg = float(c_mg.max())
                        avg_g, _ = _active_set_gradient(a_mg, c_mg, top_k=15, threshold_frac=0.995)
                        a_mg_new = _multiplicative_update(a_mg, avg_g, eta_mg)
                        c_mg2 = _full_convolve(a_mg_new, a_mg_new)
                        if float(c_mg2.max()) < cm_mg:
                            a_mg = a_mg_new
                        else:
                            eta_mg *= 0.9
                    sc_mg = _score_norm_fn(a_mg)
                    update_best(a_mg, sc_mg, f"mg_n{target_n}")
    
    print(f"  Phase 5 done: best={best_score:.10f}")
    
    # === Phase 6: Final polish ===
    print("=== Phase 6: Final polish ===")
    
    if best_seq is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_final = best_seq.copy() / best_seq.sum()
            c_final = _full_convolve(a_final, a_final)
            cm_final = float(c_final.max())
            
            # Very fine L-BFGS
            dl = min(deadline - 5, time.monotonic() + min(60.0, remaining * 0.5))
            a_fp, sc_fp = _lbfgs_smooth(a_final, dl, beta=max(0.001 * cm_final, 1e-8),
                                          top_k=30, max_iter_lbfgs=300)
            if sc_fp < best_score:
                best_score = sc_fp
                best_seq = a_fp.copy()
                print(f"  Final L-BFGS: {sc_fp:.10f}")
        
        # KKT final
        if time.monotonic() < deadline - 10:
            a_k = best_seq.copy() / best_seq.sum()
            for _ in range(200):
                if time.monotonic() > deadline - 5:
                    break
                c_k = _full_convolve(a_k, a_k)
                cm_k = float(c_k.max())
                active_k = np.where(c_k >= cm_k * 0.999)[0]
                if len(active_k) == 0:
                    break
                h_k = np.zeros(len(a_k))
                for k_idx in active_k:
                    k_val = int(k_idx)
                    lo = max(0, k_val - len(a_k) + 1)
                    hi = min(k_val, len(a_k) - 1)
                    if lo <= hi:
                        h_k[lo:hi+1] += a_k[k_val - hi:k_val - lo + 1][::-1]
                h_k /= len(active_k)
                W_k = float(np.dot(h_k, a_k))
                step_k = 0.01 * (cm_k - float(c_k[active_k].min())) / (cm_k + 1e-15)
                z_k = np.log(np.maximum(a_k, 1e-15))
                z_k = z_k - z_k.mean()
                z_k = z_k - step_k * (h_k - W_k) / (np.abs(h_k - W_k).max() + 1e-15)
                z_k = np.clip(z_k, -80, 80)
                z_k = z_k - z_k.mean()
                a_trial = _softmax_fn(z_k)
                sc_trial = _score_norm_fn(a_trial)
                if sc_trial < best_score:
                    best_score = sc_trial
                    best_seq = a_trial.copy()
                    a_k = a_trial.copy()
    
    # === Return ===
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = np.ones(2048) / 2048
            final_score = _score_fn(final_seq)
        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")
        
        # Validate
        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 8192:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"  eval check: {e}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = np.ones(4096) / 4096
    print(f"  FALLBACK: uniform n=4096, score={_score_norm_fn(fallback):.10f}")
    return [float(x) for x in fallback]


def propose_candidate(seed=42, budget_s=1800):
