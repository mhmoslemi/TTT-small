"""
LP-based optimization with Adam polishing for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses linear programming to find descent directions, with Adam for fine-tuning.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a) = 1. Returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_lp_matrix(f, n):
    """Build sparse constraint matrix for the LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(f[i])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns g or None."""
    n = len(f)

            return res.x

        pass
    return None

def _interp_profile(p, n_new):

The constraint `conv(f, g)[k] <= M` is:
`sum_{i=0}^{n-1} f[i] * g[k-i] <= M` (with `g[j] = 0` if `j < 0` or `j >= n`)

This is equivalent to:
`sum_{j=max(0,k-n+1)}^{min(k,n-1)} f[k-j] * g[j] <= M`

So the (k, j) entry of the constraint matrix is `f[k-j]` if `max(0,k-n+1) <= j <= min(k,n-1)`.

This is correct. The matrix is banded (each row has at most n non-zeros, and the band width is n).

Let me also think about whether the LP is always feasible. `g = 0` is always feasible (all constraints are satisfied since 0 <= M). So the LP is always feasible. The question is whether the optimal solution gives `sum(g) > 0`.

Since `g = f` is feasible (because `conv(f,f)[k] <= M` by definition of M = max conv(f,f)), the optimal `sum(g*) >= sum(f) > 0`. Good.

Now, after the LP step, we update `f_new = (1-t)*f + t*g_norm`. The key question is: does this always reduce the max?

For small t, the first-order change in `conv(f_new, f_new)[k]` is:
`2t * (conv(f, g_norm)[k] - conv(f, f)[k])` (approximately, for the peak k)

Since `conv(f, g_norm)[k] <= M = conv(f, f)[k_peak]` for all k (actually, `conv(f, g_norm)[k] <= M * sum(f)/sum(g*) <= M`), the first-order change at the peak is non-positive.

But wait, it's non-positive for ALL k, meaning the first-order change is non-positive everywhere. So for small enough t, the max should decrease (or stay the same).

This means LP iterations should give monotonic improvement! The only issue is the second-order term `t^2 * conv(g_norm, g_norm)[k]`, which can cause the max to increase if t is too large.

1. Start with a moderate t (e.g., 0.5)
2. If the max increases, halve t
3. Repeat until the max decreases

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_lp_matrix(f, n):
    """Build sparse constraint matrix A where A[k,j] = f[k-j] if valid."""
    k = np.arange(2 * n - 1)
    j = np.arange(n)
    diff = k[:, np.newaxis] - j[np.newaxis, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_solve(f, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M elementwise, g >= 0."""
    n = len(f)
    A = _build_lp_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

                      options={'time_limit': time_limit, 'presolve': True})

            g = np.asarray(res.x, dtype=np.float64)
            g = np.maximum(g, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:

    return None

        a_best, sc_best = _adam_optimize(
            a_init, time.monotonic() + (phase2_end - time.monotonic() - 15), lr=0.01
        )
        update_best(a_best)

        # Second Adam run from LP result
        if time.monotonic() < phase2_end - 60:
            a_lp_norm = f / f.sum()
            a_best2, sc_best2 = _adam_optimize(
                a_lp_norm, time.monotonic() + (phase2_end - time.monotonic() - 10), lr=0.008
            )
            update_best(a_best2)

    print(f"Phase 2 done: best={best_score:.10f}, time={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Multi-restart (1100s - 1400s) ===
    phase3_end = min(deadline, t_start + 1400.0)
    print(f"\n=== Phase 3: Multi-restart ===")

    if best_seq is not None and time.monotonic() < phase3_end - 30:
        a_best_norm = best_seq / best_seq.sum()
        n_restarts = 4
        per_restart = (phase3_end - time.monotonic() - 20) / max(1, n_restarts)
        sigmas = [0.03, 0.05, 0.1, 0.15]

        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 20:
                break
            p_start = _perturb(a_best_norm, sigmas[i], rng)
            deadline_i = time.monotonic() + min(per_restart, phase3_end - time.monotonic() - 10)
            a_r, sc_r = _adam_optimize(p_start, deadline_i, lr=0.01)
            update_best(a_r)

    print(f"Phase 3 done: best={best_score:.10f}, time={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Larger n exploration (1400s - 1650s) ===
    phase4_end = min(deadline, t_start + 1650.0)
    print(f"\n=== Phase 4: Larger n exploration ===")

    if best_seq is not None and len(best_seq) < 8192 and time.monotonic() < phase4_end - 60:
        target_ns = [4096, 8192]
        for target_n in target_ns:
            if time.monotonic() > phase4_end - 80:
                break
            p_interp = _interp_profile(best_seq, target_n)
            update_best(p_interp)

            remaining = phase4_end - time.monotonic() - 40
            if remaining > 120:
                a_r, _ = _adam_optimize(p_interp, time.monotonic() + remaining, lr=0.008)
                update_best(a_r)

    print(f"Phase 4 done: best={best_score:.10f}, time={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Final polish (1650s - deadline) ===
    print(f"\n=== Phase 5: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_best_norm = best_seq / best_seq.sum()
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            a_final, _ = _adam_optimize(a_best_norm, time.monotonic() + remaining, lr=0.002)
            update_best(a_final)

    print(f"Phase 5 done: best={best_score:.10f}, time={time.monotonic()-t_start:.1f}s")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_uniform(256)
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]

    fallback = _make_uniform(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
