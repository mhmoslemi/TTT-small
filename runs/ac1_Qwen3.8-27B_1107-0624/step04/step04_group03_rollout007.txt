"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS continuation, Nesterov accelerated gradient, active-set subgradient,
and multi-restart strategy with adaptive hyperparameters.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _gradient_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _objective_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _subgradient_exact(a, top_k=5):
    """Compute exact subgradient of max(conv(a,a)) using top-k active lags."""
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    
    # Get top-k lags
    if top_k >= len(c):
        w = np.ones(len(c)) / len(c)
    else:
        top_indices = np.argsort(c)[-top_k:]
        w = np.zeros(len(c))
        top_vals = c[top_indices]
        top_max = top_vals.max()
        weights = np.exp((top_vals - top_max) / max(1e-10, top_max * 0.001))
        weights /= weights.sum()
        w[top_indices] = weights
    
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    # Gradient of (1/s^2) * 2n * sum(w*c) with respect to z (log domain)
    # d/dz_m [2n * sum(w*c) / s^2] where s = sum(a) = sum(exp(z))
    # = 2n * (s * dh/dz_m * ... ) -- let me just use the smooth grad formula
    grad = 4.0 * n * a * (h - W)
    return grad


def _active_set_subgradient_optimize(a_init, deadline, n_iter=20000, top_k=3):
    """Subgradient method with active set focusing on top lags."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    # Polyak step size: lr_k = (f_k - f*) / ||g_k||^2, but we don't know f*
    # Use diminishing step: lr_k = c / sqrt(k+1)
    lr_base = 0.1
    momentum = np.zeros(n)
    mom_coeff = 0.85
    
    t = 0
    no_improve = 0
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        
        a = _softmax_fn(z)
        grad = _subgradient_exact(a, top_k=top_k)
        if not np.all(np.isfinite(grad)):
            continue
        
        # Adaptive learning rate
        lr = lr_base / np.sqrt(t)
        
        # Momentum step
        momentum = mom_coeff * momentum - lr * grad
        z_new = z + momentum
        z_new = z_new - z_new.mean()
        z_new = np.clip(z_new, -30, 30)
        
        # Check improvement periodically
        if t % 50 == 0:
            a_curr = _softmax_fn(z_new)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:

                best_a = a_curr.copy()
                z = z_new
                no_improve = 0
            else:
                no_improve += 1

Multi-phase optimizer combining L-BFGS continuation, Nesterov-accelerated
gradient descent, exact subgradient polishing, and LP-based direction finding
to minimize 2*n*max(convolve(a,a))/sum(a)^2.
"""
import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm

    n = len(a)

    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _exact_subgradient(a, top_k=10):
    """Compute subgradient of max(conv(a,a)) w.r.t. log-domain variables."""
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    
    if top_k >= len(c):
        w = np.ones(len(c)) / len(c)
    else:
        top_indices = np.argsort(c)[-top_k:]
        top_vals = c[top_indices]
        # Soft weighting over top-k
        temperature = max(1e-10, (top_vals.max() - top_vals.min()) * 0.5)
        weights = np.exp((top_vals - top_vals.max()) / temperature)
        weights /= weights.sum()
        w = np.zeros(len(c))
        w[top_indices] = weights
    
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad, cm

def _lp_direction(f, M, max_time=30.0):
    """
    Solve LP to find direction b: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0.
    Returns b or None if LP fails.
    """
    n = len(f)
    num_lags = 2 * n - 1
    
    # Build constraint matrix A (sparse): A[k, m] = f[m] if max(0, k-n+1) <= m <= min(n-1, k)
    rows = []
    cols = []
    vals = []
    
    for k in range(num_lags):
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        for m in range(m_start, m_end + 1):
            rows.append(k)
            cols.append(m)
            vals.append(f[m])
    
    A = csr_matrix((vals, (rows, cols)), shape=(num_lags, n))
    
    # Objective: minimize -sum(b) = (-1)^T b
    c_obj = -np.ones(n)
    b_ub = np.full(num_lags, M)
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), 
                         method='highs', options={'time_limit': max_time})
        if result.success and result.x is not None:
            b = np.maximum(result.x, 0)
            s = b.sum()
            if s > 1e-15:
                return b / s
    except Exception:
        pass
    
    return None

    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
