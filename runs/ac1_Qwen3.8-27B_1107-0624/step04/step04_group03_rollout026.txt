```python
import numpy as np
from scipy.optimize import linprog
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_unit(a):
    """Score assuming sum(a) = 1"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _solve_lp(b, n):
    """Solve: min t s.t. (b*a)[k] <= t for all k, sum(a)=1, a>=0"""
    # Variables: a[0..n-1], t
    # c_vec = [0,...,0, 1] (minimize t)
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    
    # Inequality constraints: (b*a)[k] - t <= 0 for k = 0,...,2n-2
    # A_ub has shape (2n-1, n+1)
    # A_ub[k, i] = b[k-i] if 0 <= k-i < n, else 0
    # A_ub[k, n] = -1
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, n] = -1.0
    for k in range(2*n-1):
        # (b*a)[k] = sum_i b[k-i] * a[i]
        # For each i, the coefficient of a[i] in (b*a)[k] is b[k-i]
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        for i in range(i_start, i_end+1):
            A_ub[k, i] = b[k - i]
    
    b_ub = np.zeros(2*n - 1)
    
    # Equality constraint: sum(a) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: a[i] >= 0, t unbounded below (or >= 0)
    bounds = [(0, None)] * n + [(0, None)]
```

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines: (1) spectral flattening in frequency domain, (2) smooth gradient
descent with log-sum-exp continuation, (3) simulated annealing, (4) 
alternating LP subproblems, and (5) multi-length exploration.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_unit(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _gradient_smooth(a, beta):
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(a)
    return grad


def _spectral_flat_step(a, alpha):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    power = np.abs(A) ** 2
    mean_power = float(power.sum()) / len(power)
    
    new_power = alpha * power + (1.0 - alpha) * mean_power
    new_power = np.maximum(new_power, 1e-30)
    
    new_mag = np.sqrt(new_power)
    phase = np.angle(A)
    A_new = new_mag * np.exp(1j * phase)
    
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _adam_step(a, grad, state, lr):
    t = state['t'] + 1
    state['t'] = t
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = b1 * state['m'] + (1 - b1) * grad
    v = b2 * state['v'] + (1 - b2) * (grad * grad)
    mh = m / (1 - b1 ** t)
    vh = v / (1 - b2 ** t)
    state['m'] = m
    state['v'] = v
    step = lr * mh / (np.sqrt(vh) + eps)
    return step


def _project_simplex(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _run_adam_smooth(a_init, deadline, beta_init, lr_init=0.05):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_unit(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, beta_init * max_c)

    while time.monotonic() < deadline:
        t += 1
        a = _softmax_fn(z)
        grad_z = _gradient_smooth(a, beta)
        if not np.all(np.isfinite(grad_z)):
            continue

        step = _adam_step(np.zeros(n), grad_z, {'m': m, 'v': v, 't': t}, lr_init)
        z = z - step
        z = z - z.mean()
        z = np.clip(z, -50, 50)

        if t % 100 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_unit(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            beta = max(1e-12, beta * 0.95)
            c_new = _full_convolve(a_curr, a_curr)
            max_c = float(c_new.max())

    a_final = _softmax_fn(z)
    sc_final = _score_unit(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _simulated_annealing(a_init, deadline, T_init=0.1, T_min=0.001, cooling=0.999):
    n = len(a_init)
    a_curr = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_unit(a_init)
    T = T_init
    c_curr = _full_convolve(a_curr, a_curr)
    max_c_curr = float(c_curr.max())

    it = 0
    while time.monotonic() < deadline and T > T_min:
        it += 1
        # Generate perturbation
        if it % 10 == 0:
            # Spectral step
            alpha = 0.1 + 0.3 * T
            a_new = _spectral_flat_step(a_curr, alpha)
        else:
            # Gradient step with noise
            a = a_curr
            c = _full_convolve(a, a)
            w, cm = _compute_weights(c, 0.01 * max_c_curr)
            W = float(np.dot(w, c))
            a_rev = a[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            grad = 4.0 * n * a * (h - W)
            
            lr = 0.01 * T + 0.001
            a_new = a_curr - lr * grad
            a_new = a_new + 0.01 * T * np.random.randn(n)
            a_new = _project_simplex(a_new)

        sc_new = _score_unit(a_new)
        if not np.isfinite(sc_new):
            T *= cooling
            continue

        sc_curr = _score_unit(a_curr)
        delta = sc_new - sc_curr

        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            a_curr = a_new
            c_curr = _full_convolve(a_curr, a_curr)
            max_c_curr = float(c_curr.max())
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_curr.copy()

        T *= cooling

    return best_a, best_score


def _alternating_lp(a_init, deadline, n_iter_max=50):

    a_curr = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_unit(a_init)

    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break
        b = a_curr.copy()
        
        # Solve LP: min t s.t. (b*a)[k] <= t, sum(a)=1, a>=0
        # Use a reduced set of constraints for speed
        # Sample constraints if n is large

        num_constraints = min(2 * n - 1, 2000)
        if 2 * n - 1 <= num_constraints:
            ks = np.arange(2 * n - 1)
        else:
            ks = np.random.choice(2 * n - 1, num_constraints, replace=False)
        
        A_ub = np.zeros((len(ks), n + 1))
        A_ub[:, n] = -1.0
        for idx, k in enumerate(ks):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                A_ub[idx, i] = b[k - i]
        b_ub = np.zeros(len(ks))
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * (n + 1)
        
        try:
            result = linprog(np.zeros(n + 1) + np.array([0.0] * n + [1.0]),
                           A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                           bounds=bounds, method='highs',
                           options={'time_limit': 10.0})
            if result.success:
                a_new = _project_simplex(result.x[:n])
                sc = _score_unit(a_new)
                # Damped update
                a_new_mix = 0.5 * a_curr + 0.5 * a_new
                a_new_mix = _project_simplex(a_new_mix)
                sc_mix = _score_unit(a_new_mix)
                
                if np.isfinite(sc_mix) and sc_mix < best_score:
                    best_score = sc_mix
                    best_a = a_new_mix.copy()
                if np.isfinite(sc_mix) and sc_mix < _score_unit(a_curr):
                    a_curr = a_new_mix
        except Exception:
            pass

    return best_a, best_score
