"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS in log-space with adaptive log-sum-exp smoothing, coordinate descent
polishing, multi-scale search, and multi-restart diversification.

import numpy as np

from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_W(a, w, n):
    """W[m] = 2 * sum_k w[k] * a[k-m] for m=0,...,n-1."""
    b = a[::-1]
    conv_wb = _full_convolve(w, b)
    return 2.0 * conv_wb[n-1:2*n-1]

def _obj_grad_smooth(z, n, beta):
    """Smoothed objective and gradient."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    
    if beta < 1e-12:
        max_idx = int(np.argmax(c))
        w = np.zeros(len(c))
        w[max_idx] = 1.0
        value = 2.0 * n * cm
    else:
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if not np.isfinite(s) or s < 1e-300:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
            value = 2.0 * n * cm
        else:
            w = e / s
            value = 2.0 * n * (cm + beta * np.log(s))
    
    W_vec = _compute_W(a, w, n)
    W_scalar = float(np.dot(W_vec, a))
    grad = 2.0 * n * a * (W_vec - W_scalar)
    
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n)
    
    return value, grad

            # Check x=0

                best_f = f_lo

            # Since f is convex, the minimum is where f' changes from negative to positive.

                f_hi2 = float(np.max(t * (hi - 0.001*(hi-lo)) + rest)) if hi > lo else f_mid
                
                if f_mid < best_f:
                    best_f = f_mid
                    best_x = mid
                
                # Check subgradient at mid
                c_mid = t * mid + rest
                max_val = c_mid.max()
                active = c_mid >= max_val - 1e-15
                if active.any():
                    slope_min = t[active].min()
                    slope_max = t[active].max()
                    if slope_min <= 0 <= slope_max:
                        # Found the minimum
                        break
                    elif slope_min > 0:
                        # f is increasing at mid, go lower
                        hi = mid
                    else:
                        # f is decreasing at mid, go higher
                        lo = mid
                else:
                    hi = mid  # fallback
            
            if best_x >= 0 and best_f < f_at_ai - 1e-12:
                a[i] = best_x
                improved = True

        s = a.sum()
