```python
def lp_direction(f, M):
    """Find direction g that maximizes sum(g) subject to g >= 0, ||f*g||_inf <= M"""
    n = len(f)
    # f*g has length 2n-1
    # (f*g)[k] = sum_i f[i]*g[k-i] <= M for all k
    # g[i] >= 0 for all i
    # Maximize sum(g)
    
    # This is a linear program with:
    # - n variables (g[0], ..., g[n-1])
    # - (2n-1) + n constraints
    # Can solve with scipy's linprog
    
    from scipy.optimize import linprog
    c = -np.ones(n)  # minimize -sum(g)
    
    # Inequality constraints: A_ub @ g <= b_ub
    # For each k in 0..2n-2: sum_i f[i]*g[k-i] <= M
    # The coefficient of g[j] in constraint k is f[k-j] (if 0 <= k-j < n)
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = f[idx]
    
    b_ub = np.full(2*n-1, M)
    
    # Bounds: g[j] >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x
```

LP-guided optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
1. Linear programming to find optimal descent directions (avoids soft-min approximation bias)
2. Active-set peak equalization using dual variables
3. Hybrid LP + L-BFGS with more aggressive continuation
4. Frequency-domain initialization exploiting flat-spectrum structure
5. Multi-scale approach: optimize at coarser resolution then refine

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _lp_direction(a, M, max_time=5.0):
    """Solve LP: max sum(g) s.t. g >= 0, conv(a,g) <= M.
    Returns normalized direction g with sum=1, or None if failed."""
    n = len(a)
    conv_len = 2 * n - 1

    # Build constraint matrix: row k has a[k-j] at column j (if valid)
    # (a*g)[k] = sum_j a[k-j] * g[j] for valid k-j
    # coefficients of g[j] in row k: a[k-j] if 0 <= k-j < n

    # Use sparse construction via scipy
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []

    for k in range(conv_len):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            idx = k - j
            if 0 <= idx < n and a[idx] > 1e-15:
                rows.append(k)
                cols.append(j)
                vals.append(a[idx])

    A = csr_matrix((vals, (rows, cols)), shape=(conv_len, n))
    b_ub = np.full(conv_len, M + 1e-12)
    c_obj = -np.ones(n)

    t0 = time.monotonic()
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub,
                         bounds=[(0, None)] * n, method='highs',
                         options={'time_limit': max_time, 'presolve': True})
    except Exception:
        return None

    if time.monotonic() - t0 > max_time:
        return None

    if result.status not in (0, 2):  # 0=optimal, 2=infeasible (shouldn't happen)
        return None

    if result.x is None or not np.all(np.isfinite(result.x)):
        return None

    g = result.x
    s = g.sum()
    if s < 1e-10:
        return None
    return g / s

            continue

def _compute_weights(c, beta):

    return w, cm


def _soft_min_obj(z, beta, n):
    """Objective: smooth approximation of 2*n*max(conv(softmax(z), softmax(z)))."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    c_shifted = np.clip(c_shifted, -30, 30)
    log_sum_exp = float(np.sum(np.exp(c_shifted)))
    val = 2.0 * n * (cm + beta * np.log(log_sum_exp))
    return val

def _adaptive_beta(c):
    cm = float(c.max())
    if cm < 1e-15:
        return 1e-10
    sorted_c = np.sort(c)[::-1]
    k = min(20, len(sorted_c) - 1)
    gap = sorted_c[0] - sorted_c[k]
    if gap < 0.05 * cm:
        return max(1e-10, 0.1 * cm)
    elif gap > 0.5 * cm:
        return max(1e-10, 0.01 * cm)
    else:
        return max(1e-10, 0.5 * gap)


def _run_softmin_lbfgs(a_init, deadline, beta_list=None, maxiter_per_beta=2000):

                 0.0002, 0.00015, 0.0001, 0.00008]

    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    num_peaks = 5

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = best_a.copy()
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())
        
        # Get top peaks
        peak_indices = np.argsort(c)[::-1][:num_peaks]
        top_val = c[peak_indices[0]]

        # For each top peak, find contributing a[i] values and try to reduce them
        improved = False
        for pk in peak_indices:
            # Valid i range for c[pk] = sum_i a[i]*a[pk-i]
            i_min = max(0, pk - n + 1)
            i_max = min(n - 1, pk)
            valid_i = np.arange(i_min, i_max + 1)
            contribs = a_cur[valid_i] * a_cur[pk - valid_i]
            
            # Try reducing the top contributors
            top_k = np.argsort(contribs)[::-1][:5]
            for ti in top_k:
                idx = valid_i[ti]
                for delta in [0.0005, 0.001, 0.002, 0.005, 0.01]:
                    a_trial = a_cur.copy()
                    # Reduce a[idx] and redistribute to the smallest elements
                    reduction = a_trial[idx] * delta
                    a_trial[idx] -= reduction
                    a_trial = np.maximum(a_trial, 0)
                    # Redistribute to increase sum without changing it much
                    # Add to elements that contribute least to peaks
                    a_trial = a_trial + reduction / n  # uniform redistribution
                    s = a_trial.sum()
                    a_trial = a_trial / s
                    
                    sc = _score_norm(a_trial)
                    if sc < best_score - 1e-10:
                        best_score = sc
                        best_a = a_trial.copy()
                        a_cur = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try increasing small elements (asymmetric boost)
            small_idx = np.argsort(a_cur)[:3]
            for idx in small_idx:
                # Choose end elements for asymmetry
                pass
            # Try directional perturbations
            a_cur = best_a.copy()
            for _ in range(10):
                idx1 = rng.integers(0, n)
                idx2 = rng.integers(0, n)
                if idx1 == idx2:
                    continue
                delta = 0.005
                a_trial = a_cur.copy()
                a_trial[idx1] = max(0, a_trial[idx1] - delta * a_trial[idx1])
                a_trial[idx2] += delta * a_trial[idx1 + 1] if idx1 + 1 < n else delta * 0.1
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                sc = _score_norm(a_trial)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_a = a_trial.copy()
                    improved = True
                    break

        if not improved and rnd > 20:
            break

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage optimization combining LP-guided direction finding,
    soft-min L-BFGS, adaptive gradient descent, and peak equalization.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                sc_parent = _score_norm(parent)
                print(f"Parent: n={n_parent}, norm_score={sc_parent:.10f}, "
                      f"actual_score={_score_full(parent):.10f}")
                update_best(parent, sc_parent, "parent")
    except Exception:
        parent = None

    if parent is None:
        n_parent = 2048
        parent = _make_profiles(n_parent)[3]
        update_best(parent, _score_norm(parent), "default_profile")

    n_best = len(best_seq)

    # === Phase 1: LP-based optimization (500s) ===
    phase1_end = min(deadline, t_start + 500.0)
    print(f"Phase 1: LP-based optimization (n={n_best})")

    a_lp = best_seq.copy()
    a_lp_best, sc_lp = _lp_iterate(a_lp, phase1_end, max_iters=150, label="lp_main")
    update_best(a_lp_best, sc_lp, "lp_main")

    # Try LP from different perturbations of best
    for pert_scale in [0.02, 0.05]:
        if time.monotonic() > phase1_end - 30:
            break
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean() + pert_scale * rng.normal(size=n_best)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a_pert = _softmax_fn(z)
        per_dl = min(phase1_end - 10, time.monotonic() + 80.0)
        a_pert_best, sc_pert = _lp_iterate(a_pert, per_dl, max_iters=50, label=f"lp_pert{pert_scale}")
        update_best(a_pert_best, sc_pert, f"lp_pert{pert_scale}")

    print(f"  Phase 1 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # === Phase 2: Soft-min L-BFGS (600s) ===
    phase2_end = min(deadline, t_start + 1100.0)
    print(f"Phase 2: Soft-min L-BFGS")

    # Multi-restart
    n_restarts = 3
    for r in range(n_restarts):
        if time.monotonic() > phase2_end - 60:
            break

        if r == 0:
            a_start = best_seq.copy()
        else:
            # Perturb
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise = 0.03 * (r + 1) * rng.normal(size=n_best)
            z = z + noise
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            a_start = _softmax_fn(z)

        per_dl = min(phase2_end - 30, time.monotonic() + (phase2_end - time.monotonic()) / (n_restarts - r))
        a_best_r, sc_r = _run_softmin_lbfgs(a_start, per_dl, maxiter_per_beta=3000)
        update_best(a_best_r, sc_r, f"lbfgs_r{r}")

    print(f"  Phase 2 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # === Phase 3: Second round of LP (300s) ===
    phase3_end = min(deadline, t_start + 1400.0)

        a_lp2 = best_seq.copy()
        a_lp2_best, sc_lp2 = _lp_iterate(a_lp2, phase3_end, max_iters=100, label="lp_r2")
        update_best(a_lp2_best, sc_lp2, "lp_r2")

    print(f"  Phase 3 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    phase4_end = min(deadline, t_start + 1700.0)

            per_dl = min(phase5_end - 10, time.monotonic() + 30.0)
            try:

    return [float(x) for x in fallback]
