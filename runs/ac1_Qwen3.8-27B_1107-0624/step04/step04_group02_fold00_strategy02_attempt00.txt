    """Find the smallest power of 2 >= n."""
    if n <= 1:
        return 1
    k = 1
    while k < n:
        k *= 2
    return k

def _full_convolve(a, b):
    """Compute the full convolution using FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
