Optimized step function search using Frank-Wolfe, LP best-response, 
and smooth-max Adam. Targets score < 1.5027.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        c = np.convolve(a, a)
    else:
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _fw_subgrad(c, a, active_idx):
    """Compute subgradient g[i] = (1/|K|) * sum_{k in K} 2*a[|k-i|] if |k-i|<n else 0"""
    n = len(a)
    g = np.zeros(n)
    for k in active_idx:
        # g[i] += 2 * a[|k-i|] for |k-i| < n
        # For each i, |k-i| ranges from |k-(n-1)| to |k-0|
        # We can vectorize: for fixed k, the contribution to g is:
        #   g[i] += 2 * a[|k-i|] if abs(k-i) < n
        # This is like a "V-shaped" template centered at k
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k + (n - 1))
        idx = np.arange(lo, hi + 1)
        dist = np.abs(k - idx)
        g[lo:hi+1] += 2.0 * a[dist]
    g /= len(active_idx)
    return g


def _fw_step(a, t, i_star):
    """Frank-Wolfe update: a_new = (1-t)*a + t*e_{i_star}"""
    e = np.zeros_like(a)
    e[i_star] = 1.0
    return (1.0 - t) * a + t * e

Optimized step function search combining Frank-Wolfe, LP best-response with
geometric/convex mixing and adaptive M, smooth-max Adam in logit space, and
multi-scale escalation. Targets score < 1.5027 by aggressively flattening the
self-convolution peak through iterative LP best-response and gradient polishing.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

        fa = np.fft.rfft(a, N)

    return c

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-30:
        return None
    return a / s


def _build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_solve(a, M):
    n = len(a)
    A_ub = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

    """Find best t in (0,1) for (1-t)*a + t*b minimizing score."""
    best_t = 0.1

    for i in range(n_trials):

        trial = (1.0 - t) * a + t * b
        trial = np.maximum(trial, 0.0)
        s = float(trial.sum())
        if s < 1e-30:

        trial = trial / s
        sc = _score_norm(trial)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    return best_t, best_sc

    """Geometric mixing: a_new[i] proportional to a[i]^(1-t) * b[i]^t."""
    la = np.log(np.maximum(a, 1e-300))

    return e / s

def _frank_wolfe(a, deadline, max_iters=500000):

            active = np.array([int(np.argmax(c))])

        i_star = int(np.argmin(g))
        t_fw = 2.0 / (it + 2.0)
        e = np.zeros(n)
        e[i_star] = 1.0
        a_new = (1.0 - t_fw) * a + t_fw * e

        # Line search every 1000 iterations
        if it % 1000 == 0:
            sc_new = _score_norm(a_new)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
            # Try a few steps
            for tc in [0.5 * t_fw, 1.5 * t_fw, 2.0 * t_fw]:
                trial = (1.0 - tc) * a + tc * e
                sc_try = _score_norm(trial)
                if sc_try < best_score:
                    best_score = sc_try
                    best_a = trial.copy()
                    a = trial
                    c = _full_convolve(a)
                    break
            else:
                a = a_new
                c = _full_convolve(a)
        else:
            a = a_new
            if it % 100 == 0 or it < 100:
                c = _full_convolve(a)
            else:
                # Approximate update (skip full conv for speed)
                c = _full_convolve(a)

        # Periodic score check
        if it % 5000 == 0:
            sc_curr = _score_norm(a)
            if sc_curr < best_score:
                best_score = sc_curr
                best_a = a.copy()

    return best_a, best_score


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    mval = float(c.max())
    if beta < 1e-13:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-9
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 0.0))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-14):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _adam_logit(p0, deadline, rng, lr=0.03, beta_frac=0.06, beta_decay=0.95,
                no_imp_limit=400):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-4, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)
        g = _conv_cross_corr(w, p)
        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.003, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 50 == 0:
            beta = max(5e-5, beta * beta_decay)

        if no_imp > no_imp_limit:
            ns = 0.03
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-4, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_exponential(n, lam):
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}] t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Started. Parent n={pn}, budget={budget_s}s")

    # Phase 1: Initialization (30s)
    phase1_end = t_start + 30.0
    candidates = []

    if parent is not None and pn > 0:
        sc = _score_norm(parent)
        candidates.append((sc, parent.copy(), pn, 'parent'))

    n_base = 1024
    for n in [512, n_base, 2048]:
        profs = [
            ('lin1.0', _make_profile_linear(n, 1.0)),
            ('lin1.5', _make_profile_linear(n, 1.5)),
            ('lin2.0', _make_profile_linear(n, 2.0)),
            ('exp0.3', _make_profile_exponential(n, 0.3)),
            ('exp0.7', _make_profile_exponential(n, 0.7)),
            ('twobump', _make_profile_two_bump(n, 0.35, 0.65, 0.15)),
            ('cos1', _make_profile_cosine(n, 1, 0.5)),
        ]
        for label, p in profs:
            sc = _score_norm(p)
            candidates.append((sc, p, n, label))

    if parent is not None:
        for n in [512, 1024, 2048]:
            if pn != n:
                pi = _interp_profile(parent, n)
                if pi is not None:
                    sc = _score_norm(pi)
                    candidates.append((sc, pi, n, 'parent_int'))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f}")
    _update(candidates[0][1], candidates[0][0], 'init')

    # Phase 2: Frank-Wolfe on best candidates (300s)
    phase2_end = t_start + 330.0
    for sc, p, n, label in candidates[:4]:
        if time.monotonic() > phase2_end or n > 2048:
            break
        per = min(120.0, phase2_end - time.monotonic() - 5)
        if per < 10:
            break
        p_fw, sc_fw = _frank_wolfe(p, time.monotonic() + per)
        if p_fw is not None:
            _update(p_fw, sc_fw, f'FW_{label}_n{n}')

    print(f"Phase 2 done. Best: {best_score:.10f}")

    # Phase 3: Intensive LP refinement (400s)
    phase3_end = t_start + 730.0
    if best_seq is not None and best_seq.size <= 2048:
        n_lp = best_seq.size
        print(f"  LP refine n={n_lp}, score={best_score:.10f}")
        per_lp = min(400.0, phase3_end - time.monotonic() - 30)
        if per_lp > 30:
            p_lp, sc_lp = _lp_refine(best_seq, time.monotonic() + per_lp,
                                      max_solves=300, use_geo=True)
            _update(p_lp, sc_lp, 'LP_main')

        # Second LP pass with fresh start
        if time.monotonic() < phase3_end - 30 and best_seq.size <= 2048:
            per_lp2 = min(100.0, phase3_end - time.monotonic() - 10)
            if per_lp2 > 20:
                p_lp2, sc_lp2 = _lp_refine(best_seq, time.monotonic() + per_lp2,
                                            max_solves=100, use_geo=True)
                _update(p_lp2, sc_lp2, 'LP_pass2')

    # Phase 4: Adam polish (200s)
    phase4_end = t_start + 930.0
    if best_seq is not None and time.monotonic() < phase4_end:
        per_adam = min(200.0, phase4_end - time.monotonic() - 10)
        if per_adam > 20:
            p_ad, sc_ad = _adam_logit(best_seq, time.monotonic() + per_adam, rng,
                                       lr=0.03, beta_frac=0.05, beta_decay=0.96,
                                       no_imp_limit=300)
            _update(p_ad, sc_ad, 'Adam_main')

            # LP after Adam
            if time.monotonic() < phase4_end - 20 and best_seq.size <= 2048:
                per = min(60.0, phase4_end - time.monotonic() - 5)
                if per > 15:
                    p_la, sc_la = _lp_refine(best_seq, time.monotonic() + per,
                                              max_solves=50, use_geo=True)
                    _update(p_la, sc_la, 'LP_postAdam')

    print(f"Phase 4 done. Best: {best_score:.10f}")

    # Phase 5: Escalate to n=2048 (400s)
    phase5_end = t_start + 1330.0
    if best_seq is not None and best_seq.size <= 1024 and time.monotonic() < phase5_end:
        p_int = _interp_profile(best_seq, 2048)
        if p_int is not None:
            sc_int = _score_norm(p_int)
            _update(p_int, sc_int, 'interp_2048')

            # FW on escalated
            per_fw = min(120.0, phase5_end - time.monotonic() - 60)
            if per_fw > 20:
                p_fw2, sc_fw2 = _frank_wolfe(p_int, time.monotonic() + per_fw)
                if p_fw2 is not None:
                    _update(p_fw2, sc_fw2, 'FW_2048')

            # LP on escalated
            if time.monotonic() < phase5_end - 30:
                per = min(150.0, phase5_end - time.monotonic() - 10)
                if per > 20:
                    p_esc, sc_esc = _lp_refine(best_seq, time.monotonic() + per,
                                                max_solves=100, use_geo=True)
                    _update(p_esc, sc_esc, 'LP_2048')

            # Adam on escalated
            if time.monotonic() < phase5_end - 10:
                per = min(60.0, phase5_end - time.monotonic() - 5)
                if per > 15:
                    p_ad2, sc_ad2 = _adam_logit(best_seq, time.monotonic() + per, rng,
                                                 lr=0.02, beta_frac=0.04, beta_decay=0.97,
                                                 no_imp_limit=200)
                    _update(p_ad2, sc_ad2, 'Adam_2048')

    # Phase 6: Final alternating polish (remaining time)
    phase6_end = deadline - 5.0
    if best_seq is not None and time.monotonic() < phase6_end:
        n_final = best_seq.size
        if n_final > 2048:
            n_final = 2048
            best_seq = _interp_profile(best_seq, n_final)
            best_score = _score_norm(best_seq)

        # Alternate LP and Adam
        n_rounds = 6
        per_round = max(10.0, (phase6_end - time.monotonic()) / n_rounds)

        for i in range(n_rounds):
            if time.monotonic() > phase6_end - 10:
                break
            # LP round
            per = min(per_round * 0.6, phase6_end - time.monotonic() - 5)
            if per > 10:
                p_r, sc_r = _lp_refine(best_seq, time.monotonic() + per,
                                        max_solves=30, use_geo=True)
                _update(p_r, sc_r, f'finalLP_{i}')

            # Adam round
            per = min(per_round * 0.4, phase6_end - time.monotonic() - 5)
            if per > 8:
                p_r2, sc_r2 = _adam_logit(best_seq, time.monotonic() + per, rng,
                                           lr=0.01, beta_frac=0.03, beta_decay=0.98,
                                           no_imp_limit=100)
                _update(p_r2, sc_r2, f'finalAdam_{i}')

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(1024, 1.5)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size} time={time.monotonic()-t_start:.1f}s")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    fb = _make_profile_linear(1024, 1.5)
    print(f"FALLBACK n=1024")
    return [float(x) for x in fb]
