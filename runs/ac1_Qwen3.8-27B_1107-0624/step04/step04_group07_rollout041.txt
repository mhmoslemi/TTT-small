FFT-based spectral flattening and gradient optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses power spectrum shaping, smooth-max
gradient descent with backtracking, simulated annealing, and multi-scale
escalation. All core operations are O(n log n) via FFT.

import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c


def _score(a):
    n = a.size
    c = _full_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def _grad_smooth_max(a, beta):
    """Compute gradient of smooth-max of conv(a,a). Returns (smooth_max_value, grad)."""
    n = a.size
    c = _full_convolve(a)
    c_max = c.max()
    z = (c - c_max) / beta
    max_z = z.max()
    exp_z = np.exp(z - max_z)
    sum_exp = exp_z.sum()
    w = exp_z / sum_exp
    smooth_max = c_max + beta * (np.log(sum_exp) + max_z)

    # g[j] = 2 * sum_m w[m+j] * a[m]
    # = 2 * conv(a[::-1], w)[j + n - 1]
    N2 = _next_pow2(3 * n - 2)
    a_ref = a[::-1]
    fa_ref = np.fft.rfft(a_ref, N2)
    fw = np.fft.rfft(w, N2)
    conv_aw = np.fft.irfft(fa_ref * fw, N2)[:3 * n - 2]
    r = conv_aw[n - 1:2 * n - 1]
    g = 2.0 * r

    return smooth_max, g


def _score_gradient(a, beta):
    """Returns (score, gradient) for score = 2*n*smooth_max(conv(a,a))/(sum(a))^2."""
    n = a.size
    s = a.sum()
    if s < 1e-15:
        return np.inf, np.zeros(n)
    smooth_max, g = _grad_smooth_max(a, beta)
    score = 2.0 * n * smooth_max / (s * s)
    grad = 2.0 * n * (g * s - 2.0 * smooth_max) / (s ** 3)
    return score, grad

    elif ptype == 'U_shape':
        t = np.linspace(0, 1, n)
        a = (2 * t - 1) ** 2 + p1
    elif ptype == 'edge_heavy':
        t = np.arange(n) / max(1, n - 1)
        a = 1.0 / (0.05 + abs(t - 0.5) * 2.0) ** p1
    elif ptype == 'arch':
        t = np.linspace(0, 1, n)
        a = np.sqrt(np.maximum(1.0 - (2 * t - 1) ** 2, 0.01)) + p1
    elif ptype == 'W_shape':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * np.abs(np.sin(np.pi * t)) ** p2
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()


def _spectral_flatten(a, alpha=0.7, t=0.5):
    """Spectral flattening: shape the power spectrum to reduce autocorrelation peaks."""
    n = a.size
    N = _next_pow2(2 * n - 1)
    A = np.fft.fft(a, N)
    P = np.abs(A) ** 2

    # Shape: compress dynamic range
    P_min = P.min()
    P_max = P.max()
    if P_max - P_min < 1e-15:
        return a.copy()
    P_norm = (P - P_min) / (P_max - P_min) + 1e-10
    P_shaped = P_norm ** (1.0 / alpha)  # alpha < 1 makes peaks smaller relative to valleys

    # Preserve phase, use shaped magnitude
    phase = np.angle(A)
    mag_shaped = np.sqrt(P_shaped)
    # Renormalize to preserve total power
    mag_shaped = mag_shaped * np.sqrt(np.sum(np.abs(A) ** 2) / np.sum(mag_shaped ** 2 + 1e-30))
    A_new = mag_shaped * np.exp(1j * phase)
    b = np.fft.ifft(A_new).real[:n]
    b = np.maximum(b, 1e-12)
    b = b / b.sum()

    a_new = (1 - t) * a + t * b
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()


def _autocorr_flatten(a, alpha=0.8, t=0.5):
    """Flatten the autocorrelation directly in time domain."""
    n = a.size
    N = _next_pow2(2 * n - 1)
    A = np.fft.fft(a, N)
    P = np.abs(A) ** 2
    R = np.fft.ifft(P).real[:N]  # circular autocorrelation = linear when N >= 2n-1

    # Shape the autocorrelation to be flatter
    R_nonneg = np.maximum(R, 0)
    R_max = R_nonneg.max()
    if R_max < 1e-15:
        return a.copy()
    R_norm = R_nonneg / R_max + 1e-10
    R_shaped = R_norm ** alpha  # alpha < 1 flattens peaks

    # Go back to frequency domain
    P_new = np.fft.fft(R_shaped, N)
    P_new = np.maximum(np.real(P_new), 0)
    # Preserve original phase
    phase = np.angle(A)
    mag_new = np.sqrt(P_new)
    if mag_new.sum() > 1e-15:
        mag_new = mag_new * np.sqrt(np.sum(np.abs(A) ** 2) / np.sum(mag_new ** 2 + 1e-30))
    A_new = mag_new * np.exp(1j * phase)
    b = np.fft.ifft(A_new).real[:n]
    b = np.maximum(b, 1e-12)
    b = b / b.sum()

    a_new = (1 - t) * a + t * b
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()


def _grad_optimize(a, deadline, lr_init=0.3, beta_init=None, max_iters=20000,
                   beta_decay=0.998, restart_limit=300, rng=None):
    """Gradient descent with smooth max, backtracking, and restarts."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = a.size
    a = np.maximum(a, 1e-12).copy()
    a = a / a.sum()

    c = _full_convolve(a)
    if beta_init is None:
        beta = 0.05 * max(c.max() - c.min(), 1e-10)
    else:
        beta = beta_init

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    t_cur = 0.1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        _, grad = _score_gradient(a, beta)
        # Project to sum-preserving direction
        grad = grad - grad.mean()

        # Backtracking line search
        sc_cur = _score(a)
        improved = False
        for lr in [lr_init, lr_init * 0.5, lr_init * 0.25, lr_init * 0.1, lr_init * 0.05]:
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 1e-12)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < sc_cur - 1e-10:
                a = a_new
                sc_cur = sc_new
                improved = True
                break
        if not improved:
            no_imp += 1
        else:
            no_imp = 0

        sc_cur = _score(a)
        if sc_cur < best_score:
            best_score = sc_cur
            best_a = a.copy()

        # Decay beta
        if it % 50 == 0:
            beta = max(1e-4, beta * beta_decay)

        # Restart
        if no_imp > restart_limit:
            ns = 0.05 + 0.1 * min(no_imp / restart_limit, 3.0)
            z = np.log(np.maximum(a, 1e-12))
            z -= z.mean()
            z = z + ns * rng.normal(size=n)
            z = np.clip(z, -40, 40)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            beta = 0.05 * _full_convolve(a).max()
            no_imp = 0
            lr_init = min(lr_init * 1.5, 2.0)

        # Adaptive LR
        if it % 100 == 0:
            lr_init = max(0.01, lr_init * 0.98)

    return best_a, best_score


def _sa_optimize(a, deadline, n_iter=50000, rng=None):
    """Simulated annealing with move = perturb one or few elements."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = a.size
    a = np.maximum(a, 1e-12).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score

    T_init = 0.1
    T_final = 0.001
    cooling = (T_final / T_init) ** (1.0 / max(n_iter, 1))

    accept_count = 0
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        T = T_init * (cooling ** it)

        # Random move: perturb 1-3 random elements
        n_perturb = rng.integers(1, 4)
        idx = rng.integers(0, n, size=n_perturb)
        a_new = a.copy()
        for j in idx:
            factor = np.exp(rng.normal(0, 0.3 + T))
            a_new[j] = max(a_new[j] * factor, 1e-12)
        a_new = a_new / a_new.sum()

        sc_new = _score(a_new)
        delta = sc_new - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            a = a_new
            cur_score = sc_new
            accept_count += 1
            if sc_new < best_score:
                best_score = sc_new
                best_a = a.copy()

    return best_a, best_score


def _coord_descent(a, deadline, max_sweeps=100):
    """Coordinate descent: optimize each element independently."""
    n = a.size
    a = np.maximum(a, 1e-12).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            old = a[j]
            # Try multiplicative factors
            for factor in [0.5, 0.7, 1.0/0.7, 1.0/0.5]:
                a[j] = max(old * factor, 1e-12)
                sc = _score(a)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = old
                    break
            # Also try binary search around current for finer adjustment
            a[j] = old
            lo, hi = old * 0.8, old * 1.25
            for _ in range(5):
                mid = (lo + hi) / 2
                a[j] = mid
                sc = _score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    lo = mid
                else:
                    hi = mid
            a[j] = best_a[j]

        if not improved:
            break

    return best_a, best_score


def _evolutionary_search(base_a, deadline, pop_size=20, n_gen=100, rng=None):
    """Simple evolutionary strategy on the sequence."""
    if rng is None:
        rng = np.random.default_rng(123)
    n = base_a.size
    base_a = np.maximum(base_a, 1e-12).copy()
    base_a = base_a / base_a.sum()

    population = [base_a.copy()]
    scores = [_score(base_a)]
    best_a = base_a.copy()
    best_score = scores[0]

    sigma = 0.1
    for gen in range(n_gen):
        if time.monotonic() > deadline:
            break

        # Generate offspring
        for i in range(pop_size // 2):
            parent_idx = int(np.argmin(scores))  # always mutate best
            parent = population[parent_idx]
            child = parent + sigma * rng.normal(size=n)
            child = np.maximum(child, 1e-12)
            child = child / child.sum()
            population.append(child)
            scores.append(_score(child))

        # Selection: keep top pop_size/2
        pop_size_cur = len(population)
        idx = np.argsort(scores)[:pop_size_cur // 2]
        population = [population[i].copy() for i in idx]
        scores = [scores[i] for i in idx]

        if scores[0] < best_score:
            best_score = scores[0]
            best_a = population[0].copy()

        # Adapt sigma
        improved = sum(1 for i in range(len(population) // 2) if scores[i] < scores[0] + 0.001)
        if improved > pop_size // 4:
            sigma = min(0.5, sigma * 1.2)
        else:
            sigma = max(0.01, sigma * 0.8)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy FFT-based optimization combining spectral flattening,
    gradient descent, simulated annealing, evolutionary search, and
    coordinate descent to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent if available
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    print(f"Start: budget={budget_s}s, parent n={0 if parent is None else parent.size}")
    if parent is not None:
        ps = _score(parent)
        print(f"Parent score: {ps:.10f}")
        if ps < best_score:
            best_score = ps
            best_seq = parent.copy()

    # === Phase 1: Generate diverse candidates ===
    print("Phase 1: Generating candidates...")
    t_phase1 = deadline - 1755.0  # 25s for this phase
    candidates = []

    n_sizes = [128, 256, 512, 1024, 2048]
    profiles = [
        ('lin1.0', 'linear', 1.0, 0, 0),
        ('lin2.0', 'linear', 2.0, 0, 0),
        ('lin3.0', 'linear', 3.0, 0, 0),
        ('lin4.0', 'linear', 4.0, 0, 0),
        ('quad1', 'quadratic', 1.0, 0, 0),
        ('quad2', 'quadratic', 2.0, 0, 0),
        ('quad3', 'quadratic', 3.0, 0, 0),
        ('cos1', 'cosine', 0.8, 1, 0),
        ('cos2', 'cosine', 0.6, 2, 0),
        ('cos3', 'cosine', 0.5, 3, 0),
        ('cos4', 'cosine', 0.4, 4, 0),
        ('sin2', 'sine', 0.5, 2, 0),
        ('sin3', 'sine', 0.4, 3, 0),
        ('exp1', 'exp', 1.0, 0, 0),
        ('exp2', 'exp', 2.0, 0, 0),
        ('g03', 'gauss', 0.3, 0, 0),
        ('g05', 'gauss', 0.5, 0, 0),
        ('tg35', 'two_gauss', 0.3, 0.7, 0.15),
        ('U03', 'U_shape', 0.3, 0, 0),
        ('U05', 'U_shape', 0.5, 0, 0),
        ('edge1', 'edge_heavy', 1.0, 0, 0),
        ('edge15', 'edge_heavy', 1.5, 0, 0),
        ('edge2', 'edge_heavy', 2.0, 0, 0),
        ('arch1', 'arch', 0.3, 0, 0),
        ('W1', 'W_shape', 0.5, 2, 0),
    ]

    for n in n_sizes:
        for (label, ptype, p1, p2, p3) in profiles:
            p = _make_profile(n, ptype, p1, p2, p3)
            sc = _score(p)
            candidates.append((sc, p, n, label))

    if parent is not None:
        for n in [256, 512, 1024, 2048, 4096]:
            pi = _interp(parent, n)
            sc = _score(pi)
            candidates.append((sc, pi, n, 'parent'))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates. Best: {candidates[0][0]:.8f} n={candidates[0][2]} [{candidates[0][3]}]")

    if candidates[0][0] < best_score:
        best_score = candidates[0][0]
        best_seq = candidates[0][1].copy()

    # === Phase 2: Spectral flattening + gradient on top candidates ===
    print("Phase 2: Spectral flattening + gradient optimization...")
    t_phase2_end = t_start + 700.0

    top_candidates = candidates[:8]
    polished = []

    for (sc, p, n, label) in top_candidates:
        if time.monotonic() > t_phase2_end - 60:
            break
        if n > 4096:
            continue
        alloc = min(120.0, t_phase2_end - time.monotonic() - 30)
        if alloc < 30:
            break

        a_work = p.copy()
        cur_best = sc

        # Spectral flattening iterations
        spec_end = time.monotonic() + alloc * 0.4
        for it in range(2000):
            if time.monotonic() > spec_end:
                break
            # Alternate between spectrum and autocorrelation flattening
            if it % 3 == 0:
                a_new = _spectral_flatten(a_work, alpha=0.6 + 0.3 * (it / 2000), t=0.3)
            elif it % 3 == 1:
                a_new = _autocorr_flatten(a_work, alpha=0.7 + 0.2 * (it / 2000), t=0.3)
            else:
                a_new = _spectral_flatten(a_work, alpha=0.8, t=0.2)

            sc_new = _score(a_new)
            if sc_new < cur_best:
                a_work = a_new
                cur_best = sc_new
            else:
                # Revert and try with different parameters
                a_new = _spectral_flatten(a_work, alpha=0.5, t=0.1)
                sc_new2 = _score(a_new)
                if sc_new2 < cur_best:
                    a_work = a_new
                    cur_best = sc_new2

        if cur_best < best_score:
            best_score = cur_best
            best_seq = a_work.copy()
            print(f"  [spec_{label}_n{n}] {cur_best:.10f}")

        # Gradient refinement
        if time.monotonic() < t_phase2_end - 30:
            g_end = min(t_phase2_end - time.monotonic(), 60.0)
            if g_end > 10:
                a_g, sc_g = _grad_optimize(a_work, time.monotonic() + g_end,
                                           lr_init=0.5, beta_init=0.02,
                                           max_iters=10000, rng=rng)
                if sc_g < best_score:
                    best_score = sc_g
                    best_seq = a_g.copy()
                    print(f"  [grad_{label}_n{n}] {sc_g:.10f}")
                polished.append((sc_g, a_g, n, label + '_g'))
            else:
                polished.append((cur_best, a_work, n, label))
        else:
            polished.append((cur_best, a_work, n, label))

    polished.sort(key=lambda x: x[0])
    print(f"  After Phase 2: best={best_score:.10f}")

    # === Phase 3: Intensive gradient optimization on best ===
    print("Phase 3: Intensive gradient optimization...")
    t_phase3_end = t_start + 1200.0

    for i, (sc, p, n, label) in enumerate(polished[:3]):
        if time.monotonic() > t_phase3_end - 60:
            break
        alloc = min(200.0, t_phase3_end - time.monotonic() - 20)
        if alloc < 30:
            break

        # Multiple gradient runs with different betas
        for beta_try in [0.01, 0.03, 0.1]:
            if time.monotonic() > t_phase3_end - 20:
                break
            sub_alloc = min(60.0, t_phase3_end - time.monotonic() - 10)
            if sub_alloc < 15:
                break
            a_r, sc_r = _grad_optimize(p, time.monotonic() + sub_alloc,
                                       lr_init=0.3, beta_init=beta_try,
                                       max_iters=15000, rng=rng)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  [grad3_b{beta_try}_{label}] {sc_r:.10f}")

    print(f"  After Phase 3: best={best_score:.10f}")

    # === Phase 4: Simulated annealing and evolutionary search ===
    print("Phase 4: SA and evolutionary search...")
    t_phase4_end = t_start + 1500.0

    if best_seq is not None:
        # SA on best
        if time.monotonic() < t_phase4_end - 60:
            sa_end = min(150.0, t_phase4_end - time.monotonic())
            if sa_end > 30:
                a_sa, sc_sa = _sa_optimize(best_seq, time.monotonic() + sa_end,
                                           n_iter=30000, rng=rng)
                if sc_sa < best_score:
                    best_score = sc_sa
                    best_seq = a_sa.copy()
                    print(f"  [SA] {sc_sa:.10f}")

        # Evolutionary
        if time.monotonic() < t_phase4_end - 30:
            evo_end = min(120.0, t_phase4_end - time.monotonic())
            if evo_end > 20:
                a_evo, sc_evo = _evolutionary_search(best_seq, time.monotonic() + evo_end,
                                                     pop_size=16, n_gen=200, rng=rng)
                if sc_evo < best_score:
                    best_score = sc_evo
                    best_seq = a_evo.copy()
                    print(f"  [EVO] {sc_evo:.10f}")

        # Random restarts with gradient
        for ri in range(3):
            if time.monotonic() > t_phase4_end - 20:
                break
            ns = 0.03 + 0.05 * ri
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            z = z + ns * rng.normal(size=best_seq.size)
            z = np.clip(z, -40, 40)
            z -= z.mean()
            a_s = np.exp(z - z.max())
            a_s = a_s / a_s.sum()
            sub_end = min(60.0, t_phase4_end - time.monotonic() - 10)
            if sub_end < 15:
                break
            a_r, sc_r = _grad_optimize(a_s, time.monotonic() + sub_end,
                                       lr_init=0.3, beta_init=0.02,
                                       max_iters=5000, rng=rng)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  [restart{ri}] {sc_r:.10f}")

    print(f"  After Phase 4: best={best_score:.10f}")

    # === Phase 5: Multi-scale escalation ===
    print("Phase 5: Multi-scale escalation...")
    t_phase5_end = t_start + 1700.0

    if best_seq is not None and time.monotonic() < t_phase5_end:
        n_cur = best_seq.size
        # Try doubling
        if n_cur <= 1024:
            for target_n in [n_cur * 2, n_cur * 4]:
                if time.monotonic() > t_phase5_end - 60:
                    break
                if target_n > 8192:
                    continue
                p_int = _interp(best_seq, target_n)
                sc_int = _score(p_int)
                print(f"  Interp to n={target_n}: {sc_int:.8f}")
                if sc_int < best_score:
                    best_score = sc_int
                    best_seq = p_int.copy()

                # Quick optimization at new scale
                if time.monotonic() < t_phase5_end - 30:
                    sub_end = min(80.0, t_phase5_end - time.monotonic() - 10)
                    if sub_end > 20:
                        a_e, sc_e = _grad_optimize(p_int, time.monotonic() + sub_end,
                                                   lr_init=0.2, beta_init=0.02,
                                                   max_iters=5000, rng=rng)
                        if sc_e < best_score:
                            best_score = sc_e
                            best_seq = a_e.copy()
                            print(f"  [escalate_n{target_n}] {sc_e:.10f}")

        # Also try smaller n (sometimes smaller n has better score)
        if n_cur > 256 and time.monotonic() < t_phase5_end - 60:
            for small_n in [n_cur // 2, n_cur // 4]:
                if small_n < 64:
                    continue
                p_small = _interp(best_seq, small_n)
                sc_small = _score(p_small)
                if time.monotonic() < t_phase5_end - 30:
                    sub_end = min(50.0, t_phase5_end - time.monotonic() - 10)
                    if sub_end > 15:
                        a_s2, sc_s2 = _grad_optimize(p_small, time.monotonic() + sub_end,
                                                     lr_init=0.3, beta_init=0.02,
                                                     max_iters=3000, rng=rng)
                        if sc_s2 < best_score:
                            best_score = sc_s2
                            best_seq = a_s2.copy()
                            print(f"  [downscale_n{small_n}] {sc_s2:.10f}")

    print(f"  After Phase 5: best={best_score:.10f}")

    # === Phase 6: Final coordinate descent polish ===
    print("Phase 6: Final coordinate descent polish...")
    if best_seq is not None and time.monotonic() < deadline - 5:
        cd_end = deadline - 5
        # Only do coord descent if n is manageable
        if best_seq.size <= 512:
            a_cd, sc_cd = _coord_descent(best_seq, cd_end, max_sweeps=200)
            if sc_cd < best_score:
                best_score = sc_cd
                best_seq = a_cd.copy()
                print(f"  [coord] {sc_cd:.10f}")
        else:
            # For larger n, do partial coordinate descent on a subset
            a_work = best_seq.copy()
            sc_work = best_score
            n_sub = min(best_seq.size, 256)
            # Randomly select indices to optimize
            idx = rng.choice(best_seq.size, size=min(n_sub, best_seq.size), replace=False)
            for j in idx:
                if time.monotonic() > cd_end - 2:
                    break
                old = a_work[j]
                for factor in [0.5, 0.7, 1.4, 2.0, 0.8, 1.25]:
                    a_work[j] = max(old * factor, 1e-12)
                    a_work = a_work / a_work.sum()
                    sc = _score(a_work)
                    if sc < sc_work:
                        sc_work = sc
                    else:
                        a_work[j] = old
                        a_work = a_work / a_work.sum()
                        break
                a_work[j] = old
                a_work = a_work / a_work.sum()

            # Do a second pass
            for j in idx[:100]:
                if time.monotonic() > cd_end - 2:
                    break
                old = a_work[j]
                lo, hi = old * 0.85, old * 1.15
                for _ in range(8):
                    mid = (lo + hi) / 2
                    a_work[j] = mid
                    a_work = a_work / a_work.sum()
                    sc = _score(a_work)
                    if sc < sc_work:
                        sc_work = sc
                        lo = mid
                    else:
                        hi = mid
                a_work[j] = old
                a_work = a_work / a_work.sum()

            if sc_work < best_score:
                best_score = sc_work
                best_seq = a_work.copy()
                print(f"  [partial_coord] {sc_work:.10f}")

    # Final validation
    print(f"\nFINAL: score={best_score:.10f} n={0 if best_seq is None else best_seq.size}")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile(512, 'linear', 1.5, 0, 0)

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            print(f"EVALUATE: {eval_sc:.10f}")
            if np.isfinite(eval_sc) and eval_sc < best_score:
                best_score = eval_sc
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile(512, 'linear', 2.0, 0, 0)
    print("FALLBACK n=512")
    return [float(x) for x in fb]
