import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_actual(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_topk_weights(c, k, beta):
    kk = min(k, len(c))
    topk_idx = np.argpartition(c, -kk)[-kk:]
    topk_vals = c[topk_idx]
    topk_shifted = (topk_vals - float(topk_vals.max())) / beta
    e = np.exp(topk_shifted)
    w = e / e.sum()
    w_full = np.zeros(len(c))
    w_full[topk_idx] = w
    return w_full, topk_idx, topk_vals, w


def _grad_topk(z, k, beta):

    n = len(a)
    c = _full_convolve(a, a)

    W = float(np.dot(w, topk_vals))
    h = _full_convolve(w_full, a[::-1])[n - 1:n - 1 + n]

    cm = float(topk_vals.max())
    obj = cm + beta * float(np.log(np.sum(np.exp((topk_vals - cm) / beta))))
    return obj, grad, a, c, cm
