"""
Optimization for minimizing 2n*max(autocorrelation)/(sum)^2 of non-negative sequences.
Uses: (1) multi-peak equalizing subgradient with Adam, (2) LP-based directional jumps,
(3) structured restarts with targeted perturbations, (4) coordinate descent on bottleneck
pairs, (5) multi-scale exploration. Key insight: equalize top-K convolution peaks
simultaneously rather than chasing only the argmax.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _conv_cross_corr(w, p):
    """Compute cross-correlation: for each j, sum_k w[k]*p[k-j] (valid range)."""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    if base_lr is None:
        base_lr = lr

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    score = float(2.0 * n * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = score
    no_imp = 0

    topk = max(3, int(topk_frac * (2 * n - 1)))

    while time.monotonic() < deadline:
        # Compute weights for top-K peaks
        c_sorted = np.sort(c)[::-1]
        # Sharp softmax over top peaks relative to max
        topk_vals = c_sorted[:topk]
        rel = (topk_vals - maxc) / max(1e-10, maxc)
        w_peaks = np.exp(sharpness * rel)
        w_peaks = w_peaks / w_peaks.sum()

        # Get indices of top-K peaks
        topk_indices = np.argsort(c)[::-1][:topk]

        # Weighted gradient: sum of weighted cross-correlations
        g = np.zeros(n)
        for wi, ki in zip(w_peaks, topk_indices):
            g += wi * _conv_cross_corr(c[ki:ki+1].reshape(1), p)  # placeholder

        # Actually compute the gradient properly: for each peak position k,
        # the gradient of conv[k] w.r.t. a[j] is 2*a[k-j] (for valid k-j)
        # So the weighted gradient is sum_w w_k * 2*a[k-j]
        # Which equals: for each j, 2 * sum_k w_k * a[k-j] (valid)
        # This is 2 * (weighted sum of shifted a)
        # = 2 * cross-correlation of w_with_peaks and a
        
        # Build the weight vector over all 2n-1 positions
        w_full = np.zeros(2 * n - 1)
        w_full[topk_indices] = w_peaks
        g = 2.0 * _conv_cross_corr(w_full, p)

        # Convert to log-space gradient
        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 300 == 0:
                    lr = max(0.003, lr * 0.97)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        # Adaptive sharpness: increase when peaks are well-separated
        if t_adam % 50 == 0 and topk_vals.size >= 2:
            gap = topk_vals[0] - topk_vals[min(1, topk_vals.size - 1)]
            if gap > 0.1 * maxc:
                sharpness = min(20.0, sharpness * 1.05)
            else:
                sharpness = max(3.0, sharpness * 0.98)

        # Restart logic
        if no_imp > no_imp_limit:
            # Structured restart: perturb a random subset
            n_perturb = max(10, n // 20)
            pert_idx = rng.choice(n, size=n_perturb, replace=False)
            z = best_z.copy()
            z[pert_idx] += rng.normal(0, 0.3, size=n_perturb)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _lp_direction_jump(a, deadline, n_trials=20):
    """Use LP to find direction jumps. Try multiple M values."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        # Try M at current peak and slightly below
        M_try = M * (1.0 - 0.005 * trial)
        if M_try < best_score / (2.0 * n) * 0.99:
            break

        b = _lp_solve(a, M_try)
        if b is None:
            b = _lp_solve(a, M_try * 1.02)
            if b is None:
                continue

        s_b = float(b.sum())
        if s_b < 1e-10:
            continue

        b_norm = b / s_b
        # Try multiple blending ratios
        for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
            a_new = (1.0 - t) * a + t * b_norm
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            sc = _score_norm(a_new)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                c = _full_convolve(a)
                M = float(c.max())
                break

    return best_a, best_score


def _coord_descent_targeted(a, deadline, n_iter=500):
    """Targeted coordinate descent: move mass from peak-contributing positions."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(777)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a)
        M = float(c.max())
        peak_k = int(np.argmax(c))
        score = float(2.0 * n * M)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        # Find top contributors to the peak
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        contrib = a[idx] * a[peak_k - idx]
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            continue

        # The position with highest contribution
        top_i = idx[np.argmax(contrib)]
        # Find a "cold" position (low contribution to peak, low value overall)
        contrib_norm = contrib / total_contrib
        # Positions NOT involved in the peak
        involved = set(idx.tolist())
        cold_candidates = [j for j in range(n) if j not in involved]
        if not cold_candidates:
            cold_candidates = list(range(n))
        
        # Pick a cold position with low value
        cold_vals = np.array([a[j] for j in cold_candidates])
        target = cold_candidates[int(np.argmin(cold_vals))]

        # Move a fraction of mass
        frac = 0.02 + 0.03 * rng.random()
        amount = a[top_i] * frac
        if amount < 1e-15 or amount > a[top_i]:
            amount = a[top_i] * 0.01
        a[top_i] -= amount
        a[target] += amount
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

    return best_a, best_score


def _simulated_annealing(a, deadline, rng, n_iter=20000, T_start=0.01, T_end=0.001):
    """Simulated annealing with targeted moves."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    score = float(2.0 * n * M)

    n_total = 2 * n - 1
    log_T = np.log(max(T_start, 1e-10))
    log_T_end = np.log(max(T_end, 1e-10))

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        T = np.exp(log_T + (log_T_end - log_T) * it / max(1, n_iter))

        # Choose a move
        move_type = rng.integers(0, 3)
        a_try = a.copy()

        if move_type == 0:
            # Move mass between two random positions
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            amount = a[i] * (0.01 + 0.05 * rng.random())
            a_try[i] -= amount
            a_try[j] += amount
        elif move_type == 1:
            # Bump one position up, another down
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = 0.01 * a.mean()
            a_try[i] += delta
            a_try[j] -= delta
        else:
            # Small random perturbation to 3-5 positions
            k = rng.integers(3, 6)
            idx = rng.choice(n, size=k, replace=False)
            a_try[idx] += rng.normal(0, 0.005, size=k) * a.mean()

        a_try = np.maximum(a_try, 1e-15)
        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try = a_try / s

        c_try = _full_convolve(a_try)
        M_try = float(c_try.max())
        score_try = float(2.0 * n * M_try)

        if score_try < best_score - 1e-14:
            best_score = score_try
            best_a = a_try.copy()

        # Metropolis acceptance
        delta = score_try - score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_try
            score = score_try

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 1e-15)
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p0)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _make_profile_power(n, alpha):
    """a[i] = (i+1)^alpha normalized."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** alpha
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_exp(n, rate):
    """a[i] = exp(rate * i / n) normalized."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.exp(rate * t)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, label=""):
        nonlocal best_seq, best_score
        if p is not None and len(p) > 0:
            p = np.asarray(p, dtype=np.float64).ravel()
            p = np.maximum(p, 0.0)
            s = p.sum()
            if s < 1e-10:
                return
            p = p / s
            sc = _score_norm(p)
            if np.isfinite(sc) and sc < best_score:
                best_seq = p.copy()
                best_score = float(sc)
                print(f"  NEW BEST: {best_score:.10f} n={p.size} [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm(parent)
        print(f"Parent: n={parent.size}, score={sc_parent:.10f}")
        _update(parent, 'parent')
    else:
        parent = _make_profile_power(1024, 1.0)
        _update(parent, 'power_init')

    # --- Phase 1: Top-K Adam optimization (400s) ---
    phase1_end = t_start + 400.0
    print(f"Phase 1: TopK Adam (400s)...")
    p1, sc1 = _adam_topk(parent, phase1_end, rng, lr=0.04, topk_frac=0.08,
                         sharpness=10.0, no_imp_limit=300)
    _update(p1, 'topk_adam_1')

    # --- Phase 2: LP direction jumps (300s) ---
    phase2_end = t_start + 700.0
    print(f"Phase 2: LP jumps (300s), current={best_score:.10f}")
    if best_seq is not None and best_seq.size <= 2048:
        p2, sc2 = _lp_direction_jump(best_seq, phase2_end, n_trials=30)
        _update(p2, 'lp_jumps')

        # Follow up with Adam
        if time.monotonic() < phase2_end - 30:
            p2a, sc2a = _adam_topk(best_seq, min(phase2_end, time.monotonic() + 60),
                                   rng, lr=0.02, topk_frac=0.1,
                                   sharpness=12.0, no_imp_limit=200)
            _update(p2a, 'post_lp_adam')

    # --- Phase 3: Slower TopK Adam with more peaks (300s) ---
    phase3_end = t_start + 1000.0
    print(f"Phase 3: Slow TopK Adam more peaks (300s), current={best_score:.10f}")
    p3, sc3 = _adam_topk(best_seq, phase3_end, rng, lr=0.015, topk_frac=0.15,
                         sharpness=5.0, no_imp_limit=500)
    _update(p3, 'topk_adam_3')

    # --- Phase 4: Coordinate descent (150s) ---
    phase4_end = t_start + 1150.0
    print(f"Phase 4: Coord descent (150s), current={best_score:.10f}")
    if best_seq is not None:
        p4, sc4 = _coord_descent_targeted(best_seq, phase4_end, n_iter=800)
        _update(p4, 'coord_descent')

        # Quick Adam after CD
        if time.monotonic() < phase4_end - 10:
            p4a, sc4a = _adam_topk(best_seq, min(phase4_end, time.monotonic() + 30),
                                   rng, lr=0.01, topk_frac=0.1,
                                   sharpness=10.0, no_imp_limit=100)
            _update(p4a, 'post_cd_adam')

    # --- Phase 5: Simulated annealing (150s) ---
    phase5_end = t_start + 1300.0
    print(f"Phase 5: Simulated annealing (150s), current={best_score:.10f}")
    if best_seq is not None:
        p5, sc5 = _simulated_annealing(best_seq, phase5_end, rng,
                                       n_iter=15000, T_start=0.005, T_end=0.0005)
        _update(p5, 'sim_anneal')

        # Polish after SA
        if time.monotonic() < phase5_end - 10:
            p5a, sc5a = _adam_topk(best_seq, min(phase5_end, time.monotonic() + 30),
                                   rng, lr=0.01, topk_frac=0.1,
                                   sharpness=8.0, no_imp_limit=100)
            _update(p5a, 'post_sa_adam')

    # --- Phase 6: Multi-scale exploration (150s) ---
    phase6_end = t_start + 1450.0
    print(f"Phase 6: Multi-scale (150s), current={best_score:.10f}")
    if time.monotonic() < phase6_end and best_seq is not None:
        # Try interpolating to nearby sizes
        n0 = best_seq.size
        for n_new in [n0 - 128, n0 + 128, n0 - 64, n0 + 64]:
            if n_new < 256 or n_new > 4096:
                continue
            if time.monotonic() > phase6_end - 20:
                break
            p_int = _interp_profile(best_seq, n_new)
            _update(p_int, f'interp_n{n_new}')
            if time.monotonic() < phase6_end - 15:
                p_new, sc_new = _adam_topk(p_int, min(phase6_end, time.monotonic() + 30),
                                           rng, lr=0.03, topk_frac=0.1,
                                           sharpness=8.0, no_imp_limit=150)
                _update(p_new, f'adam_n{n_new}')

    # --- Phase 7: Try alternative initializations (150s) ---
    phase7_end = t_start + 1600.0
    print(f"Phase 7: Alt inits (150s), current={best_score:.10f}")
    if time.monotonic() < phase7_end:
        n_try = 1024
        alts = [
            ('pow_0.5', _make_profile_power(n_try, 0.5)),
            ('pow_2.0', _make_profile_power(n_try, 2.0)),

        ]
        # Add a "bump in middle" profile
        t = np.linspace(0, 1, n_try)
        bump = 1.0 + 1.5 * np.exp(-50.0 * (t - 0.5) ** 2)
        alts.append(('bump_mid', bump / bump.sum()))
        # Add "U-shape" profile
        u_shape = 1.0 + 2.0 * (2.0 * t - 1.0) ** 2
        alts.append(('u_shape', u_shape / u_shape.sum()))

        per_alt = max(10.0, (phase7_end - time.monotonic()) / len(alts))
        for label, pa in alts:
            if time.monotonic() > phase7_end - 10:
                break
            if pa is None:
                continue
            pa = pa / pa.sum()
            per = min(per_alt, phase7_end - time.monotonic() - 5)
            if per < 8:
                break
            p_alt, sc_alt = _adam_topk(pa, time.monotonic() + per, rng,
                                       lr=0.05, topk_frac=0.08,
                                       sharpness=8.0, no_imp_limit=100)
            _update(p_alt, f'alt_{label}')

    # --- Phase 8: Final intensive polish (remaining time) ---
    remaining = deadline - time.monotonic()
    print(f"Phase 8: Final polish ({remaining:.0f}s), current={best_score:.10f}")
    if remaining > 20 and best_seq is not None:
        # LP refinement
        if best_seq.size <= 2048 and remaining > 60:
            per_lp = min(60.0, remaining * 0.25)
            p8l, sc8l = _lp_direction_jump(best_seq, time.monotonic() + per_lp, n_trials=15)
            _update(p8l, 'final_lp')
            remaining = deadline - time.monotonic()

        # Very slow Adam
        if remaining > 30:
            per_adam = remaining * 0.5
            p8a, sc8a = _adam_topk(best_seq, time.monotonic() + per_adam, rng,
                                   lr=0.005, topk_frac=0.2,
                                   sharpness=4.0, no_imp_limit=300)
            _update(p8a, 'final_slow_adam')
            remaining = deadline - time.monotonic()

        # Final coordinate descent
        if remaining > 20:
            p8c, sc8c = _coord_descent_targeted(best_seq, time.monotonic() + remaining * 0.4,
                                                n_iter=300)
            _update(p8c, 'final_cd')

        # Last Adam
        remaining = deadline - time.monotonic()
        if remaining > 15:
            p8b, sc8b = _adam_topk(best_seq, time.monotonic() + remaining * 0.5, rng,
                                   lr=0.003, topk_frac=0.3,
                                   sharpness=3.0, no_imp_limit=200)
            _update(p8b, 'final_adam2')

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_power(1024, 1.0)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile_power(1024, 1.0)
    print(f"FALLBACK n=1024, score={_score_norm(fb):.10f}")
    return [float(x) for x in fb]
