    a = a / a.sum()  # normalize
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    # Backtracking parameters
    c1 = 1e-4  # Armijo parameter
    rho = 0.5  # step reduction
    
    for it in range(100000):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        cm = c.max()
        
        # Active set: top K indices
        K = max(5, min(50, len(c)))
        active_idx = np.argsort(c)[-K:]
        active_vals = c[active_idx]
        
        # Determine which are "truly active" (within some tolerance of max)
        threshold = cm - 0.01 * cm  # within 1% of max
        true_active = active_idx[active_vals >= threshold]
        
        if len(true_active) == 0:
            true_active = active_idx[:5]
        
        # Compute subgradient: average gradient over active set
        # grad_i = (1/|A|) * sum_{k in A} 2*a[k-i] (with boundary)
        grad = np.zeros(n)
        for k in true_active:
            k = int(k)
            # For each i, gradient contribution is 2*a[k-i] if 0 <= k-i < n
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        grad /= len(true_active)
        
        # Project gradient onto the tangent space of the simplex
        # (subtract the mean to ensure the step is in the direction that preserves sum)
        grad_tangent = grad - grad.mean()
        
        # Backtracking line search
        step = 0.1
        improved = False
        for ls in range(20):
            a_trial = a - step * grad_tangent
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                step *= rho
                continue
            a_trial = a_trial / s
            sc = _score_norm_fn(a_trial)
            if sc < best_score - c1 * step * np.abs(grad_tangent @ (c.max() - c.mean())):
                # Armijo condition (simplified)
                improved = True
                break
            if sc < _score_norm_fn(a) * (1 - 1e-10):
                # Just accept any improvement
                improved = True
                break
            step *= rho
        
        if not improved:
            # Try a smaller step or break
            step *= rho
            if step < 1e-15:
                break
            a_trial = a - step * grad_tangent
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            a = a_trial
        else:
            a = a_trial
            sc = _score_norm_fn(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        # Convergence check

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_active_gradient(a, active_idx):
    """Compute the average gradient over active set."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= len(active_idx)
    return grad


def _active_set_subgrad(a_init, deadline, max_iter=50000, k_min=5, k_max=100):
    """Active-set subgradient method with backtracking line search."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    c1 = 1e-4
    rho = 0.5
    step_init = 0.05

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        # Adaptive active set size
        K = min(k_max, max(k_min, it // 100 + k_min))
        K = min(K, len(c) // 2)
        active_idx = np.argsort(c)[-K:]
        active_vals = c[active_idx]

        # True active: within relative tolerance of max
        tol = 0.005 * cm + 1e-12
        true_mask = active_vals >= (cm - tol)
        true_active = active_idx[true_mask]
        if len(true_active) < 3:
            true_active = active_idx[:max(3, K // 4)]

        # Compute subgradient
        grad = _compute_active_gradient(a, true_active)
        grad_tangent = grad - grad.mean()

        # Skip if gradient is negligible
        if np.max(np.abs(grad_tangent)) < 1e-14:
            break

        # Backtracking line search
        step = step_init
        improved = False
        current_score = best_score
        for ls in range(25):
            a_trial = a - step * grad_tangent
            a_trial = np.maximum(a_trial, 0.0)
            s_tr = a_trial.sum()
            if s_tr < 1e-10:
                step *= rho
                continue
            a_trial = a_trial / s_tr
            sc = _score_norm_fn(a_trial)
            if np.isfinite(sc) and sc < current_score:
                improved = True
                break
            step *= rho
            if step < 1e-16:

        if improved:
            sc = _score_norm_fn(a_trial)
            if sc < best_score:
                best_score = sc

def _newton_kkt(a_init, deadline, max_iter=200, k_min=10, k_max=200):
    """Newton refinement on KKT conditions for the active set problem."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        # Active set
        K = min(k_max, max(k_min, 10 + it))
        K = min(K, len(c) // 2)
        active_idx = np.argsort(c)[-K:]
        active_vals = c[active_idx]

        # True active within tolerance
        tol = max(0.002 * cm, 1e-10)
        true_mask = active_vals >= (cm - tol)
        true_active = active_idx[true_mask].astype(int)
        if len(true_active) < 3:
            true_active = active_idx[:5].astype(int)

        A = len(true_active)

        # Compute current "pressure" vector: p[i] = (1/A) * sum_{k in A} 2*a[k-i]
        p = np.zeros(n)
        for k in true_active:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                p[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        p /= A

        # The KKT condition says p should be constant (= lambda) on the support of a
        # We want to find a direction d such that p + P*d is more constant
        # where (P*d)[i] = (1/A) * sum_k 2*d[k-i]
        # This is equivalent to: d should decrease where p > mean(p), increase where p < mean(p)
        # But we need d to be in the tangent space (sum(d) = 0)

        p_mean = p.mean()
        direction = -(p - p_mean)
        direction -= direction.mean()  # ensure zero sum

        # Normalize direction
        d_norm = np.linalg.norm(direction)
        if d_norm < 1e-14:
            break
        direction /= d_norm

        # Line search
        step = 0.1
        improved = False
        for ls in range(30):
            a_trial = a + step * direction
            a_trial = np.maximum(a_trial, 0.0)
            s_tr = a_trial.sum()
            if s_tr < 1e-10:
                step *= 0.5
                continue
            a_trial = a_trial / s_tr
            sc = _score_norm_fn(a_trial)
            if np.isfinite(sc) and sc < _score_norm_fn(a) * (1 - 1e-10):
                improved = True
                break
            step *= 0.5
            if step < 1e-15:
                break

        if improved:
            sc = _score_norm_fn(a_trial)
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
            a = a_trial
        else:
            if step < 1e-14:
                break
            a_trial = a + step * direction
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial / a_trial.sum()
            a = a_trial

    return best_a, best_score

def _adam_optimize(a_init, deadline, lr=0.03, top_k=10, n_max_iter=100000):
    """Adam optimizer for the smoothed objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-10, 0.005 * max_c)

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Top-k weighted gradient
        K = min(top_k, len(c))
        idx = np.argsort(c)[-K:]
        c_top = c[idx]
        c_norm = (c_top - c_top.max()) / beta
        e = np.exp(c_norm)
        s = e.sum()
        if s < 1e-300:
            w = np.zeros(len(c))
            w[idx[0]] = 1.0
        else:
            w = np.zeros(len(c))
            w[idx] = e / s

        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 200 == 0:
            beta = max(1e-14, beta * 0.9)
        if t % 5000 == 0 and best_score < 1.504:
            lr = max(0.001, lr * 0.95)

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _make_gaussian_profile(n, sigma_frac):
    """Gaussian profile centered at n/2."""
    t = np.linspace(0.0, 1.0, n)
    center = 0.5
    a = np.exp(-0.5 * ((t - center) / sigma_frac) ** 2)
    return a / a.sum()


def _make_bell_profile(n, width):
    """Bell-shaped profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 / (1.0 + ((t - 0.5) / width) ** 4)
    return a / a.sum()

        dl = min(phase1_end, time.monotonic() + 110.0)
        p_best, sc_best = _logsumexp_lbfgs(parent, dl, top_k=0)
        update_best(p_best, sc_best, "lbfgs_warm")

    # ---- Phase 2: Active-set subgradient (main workhorse) ----
    print("=== Phase 2: Active-set subgradient ===")
    phase2_end = min(deadline, t_start + 500.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        dl = min(phase2_end, time.monotonic() + 380.0)
        p_best, sc_best = _active_set_subgrad(best_seq, dl, max_iter=50000, k_min=10, k_max=150)
        update_best(p_best, sc_best, "subgrad")

    # ---- Phase 3: Newton KKT refinement ----
    print("=== Phase 3: Newton KKT ===")
    phase3_end = min(deadline, t_start + 800.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        dl = min(phase3_end, time.monotonic() + 300.0)
        p_best, sc_best = _newton_kkt(best_seq, dl, max_iter=500, k_min=20, k_max=300)
        update_best(p_best, sc_best, "newton_kkt")

    # ---- Phase 4: Mixed method (alternate subgrad + Newton) ----
    print("=== Phase 4: Mixed method ===")
    phase4_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        dl = min(phase4_end, time.monotonic() + 300.0)
        p_best, sc_best = _mixed_method(best_seq, dl, max_iter=30000)
        update_best(p_best, sc_best, "mixed")

    # ---- Phase 5: Adam polish ----
    print("=== Phase 5: Adam polish ===")
    phase5_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        remaining = phase5_end - time.monotonic()
        if remaining > 20:
            dl = min(phase5_end - 5, time.monotonic() + remaining * 0.7)
            p_best, sc_best = _adam_optimize(best_seq, dl, lr=0.02, top_k=20, n_max_iter=60000)
            update_best(p_best, sc_best, "adam1")

        if time.monotonic() < phase5_end - 10:
            dl2 = min(phase5_end - 5, time.monotonic() + (phase5_end - time.monotonic()) * 0.6)
            p_best2, sc_best2 = _adam_optimize(best_seq, dl2, lr=0.005, top_k=50, n_max_iter=40000)
            update_best(p_best2, sc_best2, "adam2")

    # ---- Phase 6: L-BFGS final pass with larger top_k ----
    print("=== Phase 6: L-BFGS final ===")
    phase6_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase6_end and best_seq is not None:
        remaining = phase6_end - time.monotonic()
        if remaining > 20:
            dl = min(phase6_end - 5, time.monotonic() + remaining * 0.8)
            p_best, sc_best = _logsumexp_lbfgs(best_seq, dl, top_k=20)
            update_best(p_best, sc_best, "lbfgs_final")

    # ---- Phase 7: Try different n values ----
    print("=== Phase 7: Multi-n exploration ===")
    phase7_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase7_end and best_seq is not None:
        n_curr = len(best_seq)
        target_ns = [2048, 4096, 8192]
        target_ns = [tn for tn in target_ns if tn != n_curr]
        # Also try some other sizes
        for extra_n in [3072, 6144]:
            if extra_n != n_curr and extra_n not in target_ns:
                target_ns.append(extra_n)

        for tn in target_ns[:3]:  # Limit to 3 different sizes
            if time.monotonic() > phase7_end - 40:
                break
            p_interp = _interp_profile(best_seq, tn)
            sc_interp = _score_norm_fn(p_interp)
            if np.isfinite(sc_interp) and sc_interp < best_score:
                update_best(p_interp, sc_interp, f"interp_n{tn}")

            remaining = phase7_end - time.monotonic()
            if remaining < 30:
                break

            # Quick optimization at this n
            per_opt = min(80.0, remaining * 0.6)
            dl = min(phase7_end - 10, time.monotonic() + per_opt)
            p_best, sc_best = _active_set_subgrad(p_interp, dl, max_iter=20000, k_min=5, k_max=80)
            update_best(p_best, sc_best, f"subgrad_n{tn}")

            if time.monotonic() < phase7_end - 20:
                dl2 = min(phase7_end - 10, time.monotonic() + min(40.0, remaining * 0.3))
                p_best2, sc_best2 = _adam_optimize(p_best, dl2, lr=0.01, top_k=10, n_max_iter=20000)
                update_best(p_best2, sc_best2, f"adam_n{tn}")

    # ---- Phase 8: Final active-set polish on best ----
    print("=== Phase 8: Final polish ===")
    if time.monotonic() < deadline and best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            dl = min(deadline - 3, time.monotonic() + min(60.0, remaining * 0.7))
            p_best, sc_best = _active_set_subgrad(best_seq, dl, max_iter=20000, k_min=20, k_max=300)
            update_best(p_best, sc_best, "final_subgrad")

        if time.monotonic() < deadline - 5 and best_seq is not None:
            remaining2 = deadline - time.monotonic()
            if remaining2 > 15:
                dl2 = min(deadline - 3, time.monotonic() + min(30.0, remaining2 * 0.8))
                p_best2, sc_best2 = _newton_kkt(best_seq, dl2, max_iter=200, k_min=50, k_max=500)
                update_best(p_best2, sc_best2, "final_newton")

    # ---- Final validation ----
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = _make_gaussian_profile(4096, 0.3)
            final_score = _score_fn(final_seq)

        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")

        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 10000:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  evaluate_sequence check: {e}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_gaussian_profile(4096, 0.3)
    print(f"  FALLBACK: n=4096, score={_score_norm_fn(fallback):.10f}")
    return [float(x) for x in fallback]
