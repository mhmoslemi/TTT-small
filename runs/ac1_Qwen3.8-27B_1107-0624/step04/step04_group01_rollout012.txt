    g_mean = float(np.dot(a, g))
    dz = a * (g - g_mean)

    call_count = [0]

    def objective(z):
        call_count[0] += 1
        if time.monotonic() > deadline:
            a = _softmax(z)
            return _score(a), np.zeros(n)
        a = _softmax(z)
        sc, grad = _score_grad_z(a, top_k=top_k)
        if not np.isfinite(sc):
            return 1e10, np.zeros(n)
        return sc, grad

    try:
        result = scipy_minimize(objective, z0, method='L-BFGS-B',
                                options={'maxiter': max_iter, 'gtol': 1e-10,
                                         'maxfun': 50000})
        a_final = _softmax(result.x)
        sc_final = _score(a_final)
        if np.isfinite(sc_final) and sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
    except Exception:
        pass

    return best_a, best_score


def _peak_shave(a, num_steps=1000, step_size=0.01):
    """Coordinate descent: shave the highest peak by redistributing mass."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_score = _score(a)

    for _ in range(num_steps):
        c = _full_convolve(a, a)
        peak_idx = int(np.argmax(c))
        cm = float(c[peak_idx])
        s = float(a.sum())
        score = 2.0 * n * cm / (s * s)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        # For peak at index peak_idx, contributions are a_j * a[peak_idx - j]
        # Find the pair with maximum product and redistribute
        contributions = []
        for j in range(n):
            k = peak_idx - j
            if 0 <= k < n:
                contributions.append((a[j] * a[k], j, k))

        if not contributions:
            break

        contributions.sort(reverse=True)
        prod_max, j_star, k_star = contributions[0]

        if j_star == k_star:
            # a_j^2 term, can't redistribute within same element
            continue

        # Transfer mass from the larger to the smaller
        if a[j_star] > a[k_star]:
            src, dst = j_star, k_star
        else:
            src, dst = k_star, j_star

        transfer = min(step_size * a[src], a[src] * 0.1)
        if transfer < 1e-12:
            break
        a[src] -= transfer
        a[dst] += transfer
        a = np.maximum(a, 0)
        a = a / a.sum()

    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "periodic3":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.35 * np.cos(6.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "periodic5":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.25 * np.cos(10.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    freq = rng.integers(1, max(2, n // 4))
    amp = 0.05 * rng.uniform(0.5, 2.0)
    phase = rng.uniform(0, 2 * np.pi)
    t = np.arange(n)
    z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_shift(a, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    shift = int(rng.integers(1, max(2, n // 16)))
    z = np.roll(z, shift)
    z = z - z.mean()
    return _softmax(z)


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses exact top-k subgradient, Adam, L-BFGS, peak shaving, and multi-restart.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        p = p / s  # Normalize (score is scale-invariant)
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        sc0 = _score(parent)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        update_best(parent, "parent")
    else:
        n_main = 2048
        parent = _make_profile(n_main, "uniform")
        sc0 = _score(parent)
        print(f"No parent, using uniform n={n_main}, score={sc0:.10f}")
        update_best(parent, "init_uniform")

    # --- Phase 0: Quick diverse exploration (0-60s) ---
    phase0_end = min(deadline, t_start + 60.0)
    for kind in ["sinusoidal", "bimodal", "tapered", "periodic3", "periodic5"]:
        if time.monotonic() > phase0_end:
            break
        p = _make_profile(n_main, kind)
        update_best(p, f"init_{kind}")

    # Small perturbations of parent
    for sigma in [0.02, 0.05, 0.1]:
        if time.monotonic() > phase0_end:
            break
        p = _perturb(parent, sigma, rng)
        update_best(p, f"perturb_{sigma}")

    print(f"Phase 0: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 1: Main Adam optimization (60s - 500s) ---
    phase1_end = min(deadline, t_start + 500.0)
    if best_seq is not None:
        # Coarse to fine top-k schedule
        top_k_sched = [10, 5, 3, 2, 1]
        p_best, sc_best = _adam_phase(
            best_seq, time.monotonic() + (phase1_end - time.monotonic() - 10),
            top_k_schedule=top_k_sched, max_iter=500000, lr=0.02
        )
        update_best(p_best, "adam_main")

    # Second Adam pass with perturbed start
    if best_seq is not None and time.monotonic() < phase1_end - 60:
        p_pert = _perturb(best_seq, 0.03, rng)
        top_k_sched2 = [5, 3, 2, 1]
        remaining = phase1_end - time.monotonic() - 10
        p_best, sc_best = _adam_phase(
            p_pert, time.monotonic() + remaining,
            top_k_schedule=top_k_sched2, max_iter=300000, lr=0.015
        )
        update_best(p_best, "adam_perturbed")

    print(f"Phase 1: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 2: L-BFGS polishing (500s - 700s) ---
    phase2_end = min(deadline, t_start + 700.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        remaining = phase2_end - time.monotonic() - 10
        p_best, sc_best = _lbfgs_phase(best_seq, time.monotonic() + remaining, top_k=3, max_iter=10000)
        update_best(p_best, "lbfgs_3")

    if best_seq is not None and time.monotonic() < phase2_end - 30:
        remaining = phase2_end - time.monotonic() - 5
        p_best, sc_best = _lbfgs_phase(best_seq, time.monotonic() + remaining, top_k=1, max_iter=5000)
        update_best(p_best, "lbfgs_1")

    print(f"Phase 2: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 3: Peak shaving (700s - 900s) ---
    phase3_end = min(deadline, t_start + 900.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        p_shaved, sc_shaved = _peak_shave(best_seq, num_steps=2000, step_size=0.005)
        update_best(p_shaved, "peak_shave")

        # Adam after peak shaving
        if time.monotonic() < phase3_end - 30:
            top_k_sched3 = [3, 2, 1]
            remaining = phase3_end - time.monotonic() - 5
            p_best, sc_best = _adam_phase(
                p_shaved, time.monotonic() + remaining,
                top_k_schedule=top_k_sched3, max_iter=200000, lr=0.008
            )
            update_best(p_best, "post_shave_adam")

    print(f"Phase 3: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 4: Multi-restart (900s - 1400s) ---
    phase4_end = min(deadline, t_start + 1400.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        n_restarts = 6
        per_restart = (phase4_end - time.monotonic() - 30) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 20:
                break
            print(f"  Restart {i+1}/{n_restarts} t={time.monotonic()-t_start:.1f}s")

            if i % 3 == 0:
                p_start = _perturb(best_seq, 0.03 + 0.01 * (i % 3), rng)
            elif i % 3 == 1:
                p_start = _perturb_fourier(best_seq, rng)
            else:
                p_start = _perturb_shift(best_seq, rng)

            top_k_s = [5, 3, 2, 1]
            p_best, sc_best = _adam_phase(
                p_start, time.monotonic() + per_restart * 0.6,
                top_k_schedule=top_k_s, max_iter=150000, lr=0.012
            )
            update_best(p_best, f"restart{i}_adam")

            # Quick L-BFGS polish
            if time.monotonic() < phase4_end - 15:
                p_best2, sc_best2 = _lbfgs_phase(
                    p_best, time.monotonic() + per_restart * 0.4,
                    top_k=1, max_iter=3000
                )
                update_best(p_best2, f"restart{i}_lbfgs")

    print(f"Phase 4: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 5: Final ultra-precise polish (1400s - deadline) ---
    if best_seq is not None and time.monotonic() < deadline - 10:
        # Very fine Adam with top-1
        remaining = deadline - time.monotonic() - 20
        p_best, sc_best = _adam_phase(
            best_seq, time.monotonic() + remaining * 0.5,
            top_k_schedule=[1, 1], max_iter=300000, lr=0.003
        )
        update_best(p_best, "ultra_fine_adam")

        # Peak shave again
        if time.monotonic() < deadline - 15:
            p_shaved, sc_shaved = _peak_shave(p_best, num_steps=1000, step_size=0.002)
            update_best(p_shaved, "final_shave")

        # Final L-BFGS
        if time.monotonic() < deadline - 5:
            p_best2, sc_best2 = _lbfgs_phase(
                best_seq, time.monotonic() + (deadline - time.monotonic() - 3),
                top_k=1, max_iter=3000
            )
            update_best(p_best2, "final_lbfgs")

    # --- Try larger n ---
    if best_seq is not None and len(best_seq) < 4096 and time.monotonic() < deadline - 60:
        target_n = 4096
        p_interp = _interp_profile(best_seq, target_n)
        sc_interp = _score(p_interp)
        update_best(p_interp, f"interp_n{target_n}")

        if time.monotonic() < deadline - 30:
            top_k_sched_l = [5, 3, 1]
            remaining = deadline - time.monotonic() - 10
            p_best, sc_best = _adam_phase(
                p_interp, time.monotonic() + remaining,
                top_k_schedule=top_k_sched_l, max_iter=100000, lr=0.01
            )
            update_best(p_best, f"opt_n{target_n}")

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
