def _next_pow2(n):

    while k < n:
        k <<= 1
    return max(2, k)

    n, m = len(a), len(b)

    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

For the problem: min max_k conv(a,a)[k] subject to sum(a) = 1, a >= 0.

If we think of a as a probability distribution on {0, ..., n-1}, then conv(a,a) is the distribution of the sum of two independent random variables with distribution a. We want to minimize the peak of this sum distribution.

The optimal distribution should make the sum as "uniform" as possible over {0, ..., 2n-2}. This is like the "discrete central limit theorem" - we want the sum to be as close to uniform as possible.

For a continuous analogue: if X and Y are i.i.d. with density f, we want f*f (convolution) to be as flat as possible. The optimal f is... well, if f*f = constant on [0, 2], then in Fourier space, |F|^2 = constant, so |F| = constant. This means F is a "flat spectrum" function, which for a compactly supported function is related to prolate spheroidal wave functions (PSWFs).

But for the discrete case with non-negativity constraint, it's different. The non-negativity constraint is key.

For a non-negative sequence a with sum 1, the convolution a*a is also non-negative and has sum 1 (since sum(a*a) = (sum a)^2 = 1). We want to minimize the peak of a*a. The minimum possible peak is 1/(2n-1) (if a*a is uniform over its support). But a*a has a special structure (it's a convolution of a with itself), so it can't be perfectly uniform.

For the uniform a (a_i = 1/n), a*a is triangular: (a*a)[k] = (k+1)/n^2 for k < n, (2n-1-k)/n^2 for k >= n. Peak = 1/n.

"""
Multi-strategy optimization for minimum peak convolution sequences.
Combines L-BFGS with smooth max (log-sum-exp), direct peak cycling,
multi-restart diversification, and adaptive n selection.
Gradient: dJ/dz[m] = 4*n*a[m]*(H[m] - W_c) where H = cross_corr(w, a_rev), W_c = dot(w, c).
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_smooth_max_and_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return cm, w
    w = e / s
    smooth_max = cm + beta * np.log(s)
    return smooth_max, w


def _obj_and_grad_z(z, beta):
    a = _softmax(z)
    n = len(a)
    c = _full_convolve(a, a)
    smooth_max, w = _compute_smooth_max_and_weights(c, beta)
    obj = 2.0 * n * smooth_max
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_c = float(np.dot(w, c))
    grad_z = 4.0 * n * a * (h - W_c)
    if not np.all(np.isfinite(grad_z)):
        grad_z = np.zeros_like(z)
    return obj, grad_z


def _lbgfs_optimize(a_init, deadline, beta_schedule, max_iter=5000, lr=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break

        def fun(gz):
            a_tmp = _softmax(gz)
            c_tmp = _full_convolve(a_tmp, a_tmp)
            sm, _ = _compute_smooth_max_and_weights(c_tmp, beta)
            return 2.0 * n * sm

        def fun_and_grad(gz):
            return _obj_and_grad_z(gz, beta)

        result = minimize(fun_and_grad, z, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'gtol': 1e-8,
                                   'maxls': 20, 'ftol': 1e-10})

        z = result.x
        a_final = _softmax(z)
        sc = _score_norm(a_final)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_final.copy()

        if time.monotonic() > deadline:
            break

    return best_a, best_score


def _direct_peak_cycle(a, deadline, max_iters=5000, top_k=5):
    n = len(a)
    best = a.copy()
    best_score = _score_norm(a)
    step = 0.1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Get top K peak positions
        order = np.argsort(c)[::-1][:top_k]

        # Average gradient of top peaks
        grad = np.zeros(n)
        for k_star in order:
            k_star = int(k_star)
            lo = max(0, k_star - n + 1)
            hi = min(n - 1, k_star)
            if hi >= lo:
                idx = k_star - np.arange(lo, hi + 1)
                grad[lo:hi + 1] += 2.0 * a[idx]

        grad -= np.mean(grad)

        # Line search
        improved = False
        for _ in range(30):
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                step *= 0.5
                continue
            a_new = a_new / s
            c_new = _full_convolve(a_new, a_new)
            c_new_max = float(c_new.max())
            if c_new_max < c_max * (1.0 - 1e-7):
                a = a_new
                step = min(0.5, step * 1.1)
                improved = True
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best = a.copy()
                break
            step *= 0.7

        if not improved:
            step *= 0.5
            if step < 1e-10:
                break

    return best, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "u_shape":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "v_shape":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edges":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * (2.0 * t - 1.0) ** 4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "three_lobe":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(3.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    n_freqs = rng.integers(1, max(2, n // 8))
    for _ in range(n_freqs):
        freq = rng.integers(1, max(2, n // 3))
        amp = rng.uniform(0.02, 0.15) * rng.standard_normal()
        t = np.arange(n)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi))
    z = z - z.mean()
    return _softmax(z)


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)

    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

    # --- Phase 5: Fine polish (1550s - 1750s) ---
    phase5_end = min(deadline, t_start + 1750.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        c_final = _full_convolve(best_seq, best_seq)
        max_cf = float(c_final.max())
        beta_fine = [0.001 * max_cf, 0.0005 * max_cf, 0.0002 * max_cf,
                     0.0001 * max_cf, 0.00005 * max_cf]
        per_time = phase5_end - time.monotonic() - 10
        if per_time > 10:
            p_best, sc_best = _lbgfs_optimize(
                best_seq, time.monotonic() + per_time, beta_fine, max_iter=3000
            )
            update_best(p_best, _score(p_best), "fine_lbgfs")

        per_time2 = phase5_end - time.monotonic() - 5
        if per_time2 > 10:
            p_best2, sc_best2 = _direct_peak_cycle(
                best_seq, time.monotonic() + per_time2,
                max_iters=5000, top_k=5
            )
            update_best(p_best2, _score(p_best2), "fine_peak")

    print(f"Phase 5: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 6: Ultra-fine polish (1750s - deadline) ---
    if best_seq is not None and time.monotonic() < deadline - 5:
        c_ultra = _full_convolve(best_seq, best_seq)
        max_cu = float(c_ultra.max())
        beta_ultra = [0.00005 * max_cu, 0.00002 * max_cu, 0.00001 * max_cu]
        per_time = deadline - time.monotonic() - 3
        if per_time > 5:
            p_best, sc_best = _lbgfs_optimize(
                best_seq, time.monotonic() + per_time, beta_ultra, max_iter=2000
            )
            update_best(p_best, _score(p_best), "ultra_fine")

    print(f"Phase 6: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
