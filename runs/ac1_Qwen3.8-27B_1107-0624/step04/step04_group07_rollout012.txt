"""
Improved LP fixed-point optimization with spectral initialization,
dual-guided support reduction, golden-section line search, and
multi-scale escalation for minimizing 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _build_conv_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _solve_lp_primal(a, M, n):
    A = _build_conv_matrix(n, a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _solve_lp_dual(a, n):
    A = _build_conv_matrix(n, a)
    c_obj = np.ones(2 * n - 1)
    A_ub = -A.T
    b_ub = -np.ones(n)
    bounds = [(0, None)] * (2 * n - 1)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return float(res.fun), np.maximum(res.x, 0.0)
    return None, None


def _golden_line_search(a, b, deadline, max_iters=20):
    golden = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - golden * (hi - lo)
    d = lo + golden * (hi - lo)
    a_c = (1.0 - c) * a + c * b
    a_c = np.maximum(a_c, 1e-15)
    a_c = a_c / a_c.sum()
    sc_c = _score_norm(a_c)
    a_d = (1.0 - d) * a + d * b
    a_d = np.maximum(a_d, 1e-15)
    a_d = a_d / a_d.sum()
    sc_d = _score_norm(a_d)
    for _ in range(max_iters):
        if time.monotonic() > deadline:
            break
        if sc_c < sc_d:
            hi = d
            d = c
            sc_d = sc_c
            c = hi - golden * (hi - lo)
            a_c = (1.0 - c) * a + c * b
            a_c = np.maximum(a_c, 1e-15)
            a_c = a_c / a_c.sum()
            sc_c = _score_norm(a_c)
        else:
            lo = c
            c = d
            sc_c = sc_d
            d = lo + golden * (hi - lo)
            a_d = (1.0 - d) * a + d * b
            a_d = np.maximum(a_d, 1e-15)
            a_d = a_d / a_d.sum()
            sc_d = _score_norm(a_d)
    t_best = (lo + hi) / 2.0
    a_best = (1.0 - t_best) * a + t_best * b
    a_best = np.maximum(a_best, 1e-15)
    a_best = a_best / a_best.sum()
    return a_best, _score_norm(a_best)


def _lp_fixed_point(a, deadline, max_iters=200, use_golden=True):
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    b_prev = None
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a)
        M = float(c.max())
        b = _solve_lp_primal(a, M, n)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b = b / s_b
        if use_golden:
            a_new, sc_new = _golden_line_search(a, b, deadline, max_iters=15)
        else:
            best_t = 0.5
            best_sc = _score_norm(b)
            for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
                a_t = (1.0 - t) * a + t * b
                a_t = np.maximum(a_t, 1e-15)
                a_t = a_t / a_t.sum()
                sc_t = _score_norm(a_t)
                if sc_t < best_sc:
                    best_sc = sc_t
                    best_t = t
            a_new = (1.0 - best_t) * a + best_t * b
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            sc_new = best_sc
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
        change = np.linalg.norm(a_new - a)
        a = a_new
        if change < 1e-10:
            break
    return best_a, best_score


def _gs_spectral_init(a, n_iters=100, blend=0.3):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    for it in range(n_iters):
        A = np.fft.rfft(a, N)
        amp = np.abs(A)
        mean_amp = np.sqrt(np.mean(np.abs(A) ** 2))
        if mean_amp < 1e-10:
            break
        new_amp = (1.0 - blend) * amp + blend * mean_amp
        new_A = new_amp * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(new_A, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score


def _adam_refine(a, deadline, rng, lr=0.04, beta_init=None, no_imp_limit=150):
    n = len(a)
    a = np.maximum(a, 1e-12).copy()
    a = a / a.sum()
    z = np.log(a)
    z -= z.mean()
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    c = _full_convolve(a)
    maxc = float(c.max())
    beta = beta_init if beta_init is not None else 0.05 * maxc
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        mval = float(c.max())
        x = (c - mval) / beta
        ex = np.exp(x)
        s_exp = float(ex.sum())
        if not np.isfinite(s_exp) or s_exp <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_exp
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h = np.fft.irfft(fw * fa_rev, N)
        g = h[n - 1:2 * n - 1]
        dot = float(np.dot(a, g))
        gz = a * (g - dot)
        gz -= gz.mean()
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()
        a = np.exp(z - z.max())
        a = a / a.sum()
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        c = _full_convolve(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.002, lr * 0.92)
        if t_adam % 30 == 0:
            beta = max(1e-5, beta * 0.93)
        if no_imp > no_imp_limit:
            ns = 0.06
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a)
            beta = 0.05 * c.max()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0
    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score


def _make_profile(n, ptype, p1=1.0, p2=0.0, p3=0.0):
    if ptype == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (t - 0.5) * 2.0
    elif ptype == 'quadratic':
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + p1 * u ** 2
    elif ptype == 'cubic':
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + p1 * u + p2 * u ** 3
    elif ptype == 'exp':
        t = np.arange(n) / max(1, n)
        a = np.exp(-p1 * t)
    elif ptype == 'cosine':
        t = np.arange(n)
        a = 1.0 + p1 * np.cos(2 * np.pi * p2 * t / n)
    elif ptype == 'sine':
        t = np.arange(n)
        a = 1.0 + p1 * np.sin(2 * np.pi * p2 * t / n)
    elif ptype == 'parabola':
        t = np.linspace(0, 1, n)
        a = (t - 0.5) ** 2 + p1
    elif ptype == 'gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / p1) ** 2)
    elif ptype == 'U_shape':
        t = np.linspace(0, 1, n)
        a = (2 * t - 1) ** 2 + p1
    elif ptype == 'edge_heavy':
        t = np.arange(n) / max(1, n - 1)
        a = 1.0 / (0.1 + abs(t - 0.5) * 2.0) ** p1
    elif ptype == 'power':
        t = np.linspace(0.01, 1.0, n)
        a = t ** p1
    elif ptype == 'anti_sym':
        t = np.linspace(0, 1, n)
        a = p1 * t + p2 * (1 - t) + 0.1
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new


def _dual_guided_support(a, n, threshold_frac=0.1):
    dual_obj, y_star = _solve_lp_dual(a, n)
    if y_star is None:
        return a, None
    A = _build_conv_matrix(n, a)
    Ay = np.asarray(A.T @ y_star).ravel()
    Ay_max = float(Ay.max())
    if Ay_max <= 0:
        return a, None
    threshold = Ay_max - threshold_frac * Ay_max
    support = Ay >= threshold
    if support.sum() < n // 4:
        support = np.ones(n, dtype=bool)
    if support.sum() == n:
        return a, None
    a_masked = a.copy()
    a_masked[~support] = 1e-15
    a_masked = a_masked / a_masked.sum()
    return a_masked, y_star

    # Phase 8: Final polish
    print("Phase 8: Final polish...")
    if time.monotonic() < deadline - 10 and best_seq is not None:
        per = deadline - time.monotonic() - 10
        if per > 15:
            if best_seq.size <= 1500:
                a_fp, sc_fp = _lp_fixed_point(best_seq, time.monotonic() + per,
                                              max_iters=30, use_golden=True)
            else:
                a_fp, sc_fp = _adam_refine(best_seq, time.monotonic() + per, rng,
                                           lr=0.02, no_imp_limit=80)
            if sc_fp < best_score:
                best_seq = a_fp.copy()
                best_score = sc_fp
                print(f"  BEST: {best_score:.10f} n={best_seq.size} [final]")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile(512, 'linear', 1.5, 0, 0)
        exact_sc = _score_norm(final)
        print(f"\nFINAL: score={exact_sc:.10f} n={final.size}")
        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")
        return [float(x) for x in final]

    fb = _make_profile(512, 'linear', 1.5, 0, 0)
    print("FALLBACK n=512")
    return [float(x) for x in fb]
