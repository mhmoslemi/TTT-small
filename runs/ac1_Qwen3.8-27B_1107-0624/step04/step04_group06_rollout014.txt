Primary method: Linearized self-convolution LP (solving min M s.t. linearized

The LP directly targets peak reduction by solving a linearized version of the

from scipy.sparse import csr_matrix

def _lp_linearized(a):
    """
    Solve LP: min M s.t. sum_i 2*a_{k-i}*a_i <= M + c_k for all k,
    sum a_i = 1, a_i >= 0, M >= 0.
    This is the linearization of (a*a)_k <= M at the current point a.
    Returns (a_new, M_new) or None if infeasible.
    """
    n = len(a)
    c = _full_convolve(a)
    n_k = 2 * n - 1

    # Build sparse constraint matrix A_ub of shape (n_k, n+1)
    # A_ub[k, i] = 2 * a^{(t)}_{k-i} if 0 <= k-i < n, else 0
    # A_ub[k, n] = -1  (coefficient of M)
    rows = []
    cols = []
    vals = []
    for k in range(n_k):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            partner = k - idx
            rv = 2.0 * a[partner]
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(rv.tolist())
        # M column
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)

    A_ub = csr_matrix((vals, (rows, cols)), shape=(n_k, n + 1))
    b_ub = c.copy()  # RHS: M + c_k => 2*a_{k-i}*a_i - M <= c_k

    # Equality: sum a_i = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Objective: minimize M
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    # Bounds
    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        a_new = np.maximum(res.x[:n], 0.0)
        M_new = res.x[n]
        s = float(a_new.sum())
        if s > 1e-10:
            a_new = a_new / s
        return a_new, M_new
    return None

    z -= float(np.mean(z))

        g = _grad_weighted_conv(w, p)

        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)

            m.fill(0.0)
            v.fill(0.0)

        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)

                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.002, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 30 == 0:
            beta = max(1e-5, beta * beta_decay)

        if no_imp > no_imp_limit:
            ns = 0.05 if best_score < 1.51 else 0.1
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z - float(np.max(z)))
            p = e + 1e-12
            p = p / p.sum()
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            no_imp = 0

    return best_p, best_score


def _lp_optimize(a, deadline, max_iters=2000, stagnation_limit=50):
    """
    Main LP-based optimizer using linearized self-convolution.
    At each step, solves the LP and tries to take a step that reduces the peak.
    Falls back to smooth gradient if LP stalls.
    """
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    n = a.size

    best_a = a.copy()
    best_peak = float(_full_convolve(a).max())
    best_score = float(2.0 * n * best_peak)
    c = _full_convolve(a)
    cur_peak = float(c.max())

    rng = np.random.default_rng(777)
    stagnation = 0
    lp_calls = 0

    while time.monotonic() < deadline:
        # Try LP
        result = _lp_linearized(a)
        if result is not None:
            lp_calls += 1
            a_lp, M_lp = result

            # Try different step sizes
            improved = False
            for alpha in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]:
                a_try = (1.0 - alpha) * a + alpha * a_lp
                a_try = np.maximum(a_try, 0.0)
                s = float(a_try.sum())
                if s < 1e-10:
                    continue
                a_try = a_try / s
                c_try = _full_convolve(a_try)
                p_try = float(c_try.max())
                if p_try < cur_peak - 1e-15:
                    a = a_try
                    c = c_try
                    cur_peak = p_try
                    improved = True
                    break

            if improved:
                if cur_peak < best_peak - 1e-12:
                    best_peak = cur_peak
                    best_score = float(2.0 * n * cur_peak)
                    best_a = a.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1

        # If LP is stalling, use smooth gradient
        if stagnation >= stagnation_limit:
            # Reset and do a burst of Adam steps
            sub_deadline = min(deadline, time.monotonic() + max(5.0, (deadline - time.monotonic()) * 0.05))
            a_new, sc_new = _adam_smooth(a, sub_deadline, rng,
                                          lr=0.03, beta_frac=0.06,
                                          beta_decay=0.94, no_imp_limit=100)
            a = a_new
            c = _full_convolve(a)
            cur_peak = float(c.max())
            if cur_peak < best_peak - 1e-12:
                best_peak = cur_peak
                best_score = float(2.0 * n * cur_peak)
                best_a = a.copy()
            stagnation = 0
            # Random perturbation to escape
            if time.monotonic() < deadline:
                a = a + 0.01 * rng.normal(size=n)
                a = np.maximum(a, 1e-6)
                a = a / a.sum()

        # Periodic print
        if lp_calls % 200 == 0 and lp_calls > 0:
            print(f"    LP iter {lp_calls}: score={best_score:.10f}, cur_peak={cur_peak:.10f}, stag={stagnation}")

    return best_a, best_score


def _peak_flood(a, deadline, eta=0.03, n_iter_max=3000):
    """Multiplicative peak-flooding update."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_peak = float(_full_convolve(a).max())
    best_score = float(2.0 * n * best_peak)
    c = _full_convolve(a)

    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break

        M = float(c.max())
        if M < best_peak - 1e-12:
            best_peak = M
            best_score = float(2.0 * n * M)
            best_a = a.copy()

        threshold = M * 0.95
        hot_mask = c >= threshold
        if not hot_mask.any():
            hot_mask[int(np.argmax(c))] = True

        heat = np.zeros(n)
        hot_indices = np.where(hot_mask)[0]
        for k in hot_indices:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                partner = k - np.arange(i_start, i_end + 1)
                heat[i_start:i_end + 1] += a[partner]

        heat_max = float(heat.max())
        if heat_max < 1e-15:
            break
        heat = heat / heat_max

        factor = 1.0 - eta * heat
        factor = np.maximum(factor, 0.01)
        a = a * factor
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        c = _full_convolve(a)

        if it % 200 == 0 and it > 0:
            if M >= best_peak:
                eta = max(0.005, eta * 0.9)
            else:
                eta = min(0.12, eta * 1.03)

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, label=""):
        nonlocal best_seq, best_score
        if p is not None and len(p) > 0:
            p = np.asarray(p, dtype=np.float64).ravel()
            p = np.clip(p, 0.0, 1000.0)
            s = float(p.sum())
            if s < 0.01:
                return
            p = p / s
            sc = _score_norm(p)
            if np.isfinite(sc) and sc < best_score:
                best_seq = p.copy()
                best_score = float(sc)
                print(f"  BEST: {best_score:.10f} n={p.size} [{label}]")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm(parent)
        print(f"Parent: n={parent.size}, score={sc_parent:.10f}")
        _update(parent, "parent")
    else:
        n_init = 1024
        t = np.linspace(0, 1, n_init)
        parent = 1.0 + 2.0 * (t - 0.5)
        parent = np.maximum(parent, 1e-12)
        parent = parent / parent.sum()
        _update(parent, "linear_init")

    pn = best_seq.size if best_seq is not None else 1024
    print(f"Starting optimization, n={pn}, best={best_score:.10f}, budget={budget_s}s")

    # === Phase 1: Quick Adam to polish (120s) ===
    phase1_end = t_start + 120.0
    print("Phase 1: Adam polish...")
    p1, sc1 = _adam_smooth(best_seq, phase1_end, rng, lr=0.04, beta_frac=0.07,
                           beta_decay=0.93, no_imp_limit=150)
    _update(p1, "adam1")

    # === Phase 2: LP optimization (main, 600s) ===
    phase2_end = t_start + 720.0
    print(f"Phase 2: LP optimization, score={best_score:.10f}...")
    p2, sc2 = _lp_optimize(best_seq, phase2_end, max_iters=3000, stagnation_limit=40)
    _update(p2, "lp_main")

    # === Phase 3: Adam after LP (150s) ===
    phase3_end = t_start + 870.0
    print(f"Phase 3: Post-LP Adam, score={best_score:.10f}...")
    if time.monotonic() < phase3_end:
        p3, sc3 = _adam_smooth(best_seq, phase3_end, rng, lr=0.025, beta_frac=0.05,
                               beta_decay=0.95, no_imp_limit=150)
        _update(p3, "adam_post_lp")

    # === Phase 4: Second LP round (300s) ===
    phase4_end = t_start + 1170.0
    print(f"Phase 4: LP round 2, score={best_score:.10f}...")
    if time.monotonic() < phase4_end:
        p4, sc4 = _lp_optimize(best_seq, phase4_end, max_iters=2000, stagnation_limit=30)
        _update(p4, "lp_round2")

    # === Phase 5: Try n=2048 (200s) ===
    phase5_end = t_start + 1370.0
    print(f"Phase 5: Multi-scale (n=2048), score={best_score:.10f}...")
    if time.monotonic() < phase5_end and best_seq.size <= 1024:
        n_new = 2048
        p_int = _interp_profile(best_seq, n_new)
        if p_int is not None:
            sc_int = _score_norm(p_int)
            print(f"  Interpolated n={n_new}: {sc_int:.10f}")
            _update(p_int, f"interp_{n_new}")

            if time.monotonic() < phase5_end - 10:
                per = min(100.0, phase5_end - time.monotonic() - 5)
                if per > 30:
                    p5, sc5 = _lp_optimize(p_int, time.monotonic() + per,
                                           max_iters=500, stagnation_limit=30)
                    _update(p5, f"lp_{n_new}")

    # === Phase 6: Try n=512 (100s) ===
    phase6_end = t_start + 1470.0
    print(f"Phase 6: Try n=512, score={best_score:.10f}...")
    if time.monotonic() < phase6_end and best_seq.size > 512:
        n_small = 512
        p_small = _interp_profile(best_seq, n_small)
        if p_small is not None:
            sc_small = _score_norm(p_small)
            print(f"  Interpolated n={n_small}: {sc_small:.10f}")
            _update(p_small, f"interp_{n_small}")
            if time.monotonic() < phase6_end - 5:
                per = min(50.0, phase6_end - time.monotonic() - 3)
                if per > 20:
                    p6, sc6 = _lp_optimize(p_small, time.monotonic() + per,
                                           max_iters=300, stagnation_limit=20)
                    _update(p6, f"lp_{n_small}")

    # === Phase 7: Final intensive polish (remaining) ===
    remaining = deadline - time.monotonic()
    print(f"Phase 7: Final polish, remaining={remaining:.0f}s, score={best_score:.10f}...")
    if remaining > 30:
        # Peak flood
        per1 = min(60.0, remaining * 0.3)
        if per1 > 20:
            p7f, sc7f = _peak_flood(best_seq, time.monotonic() + per1, eta=0.02)
            _update(p7f, "final_flood")

        # LP
        remaining2 = deadline - time.monotonic()
        if remaining2 > 40:
            per2 = min(120.0, remaining2 * 0.5)
            if per2 > 20:
                p7l, sc7l = _lp_optimize(best_seq, time.monotonic() + per2,
                                         max_iters=1000, stagnation_limit=25)
                _update(p7l, "final_lp")

        # Final Adam
        remaining3 = deadline - time.monotonic()
        if remaining3 > 15:
            p7a, sc7a = _adam_smooth(best_seq, deadline - 5, rng, lr=0.01,
                                     beta_frac=0.03, beta_decay=0.97, no_imp_limit=50)
            _update(p7a, "final_adam")

    # Final
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
                if eval_sc < exact_sc:
                    print("  (eval lower due to clipping/normalization)")
        except Exception as e:
            print(f"  Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    n_fb = 1024
    t = np.linspace(0, 1, n_fb)
    fb = 1.0 + 2.0 * (t - 0.5)
    fb = np.maximum(fb, 1e-12)
    fb = fb / fb.sum()
    print(f"FALLBACK n={n_fb}")
    return [float(x) for x in fb]
