"""
Optimized step function search combining adaptive LP refinement, smooth-max 
Adam in logit space, coordinate descent, and multi-restart local search.
Targets score < 1.5027 by aggressively flattening the self-convolution peak.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 512:
        c = np.convolve(a, a)
    else:
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        c = np.maximum(c, 0.0)
    return c


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    mval = float(c.max())
    if beta < 1e-13:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-9
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 0.0))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-14):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_solve(a, M):
    n = len(a)
    A_ub = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a, deadline, max_solves=40, t_init=0.3):
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    t = t_init
    stall_count = 0
    n_solves = 0

    while time.monotonic() < deadline and n_solves < max_solves:
        if time.monotonic() > deadline - 5:
            break
        n_solves += 1
        b = _lp_solve(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b

        # Line search for t
        best_sc = current_score
        best_t = t
        for tc in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]:
            a_try = (1.0 - tc) * a + tc * b_norm
            a_try = np.maximum(a_try, 1e-15)
            a_try = a_try / a_try.sum()
            sc_try = _score_norm(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_t = tc
        
        a_new = (1.0 - best_t) * a + best_t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)

        if sc_new < current_score - 1e-12:
            current_score = sc_new
            current_best = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 8:
                break

        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
        t = max(0.05, best_t * 0.8)

    return current_best, current_score


def _adam_logit(p0, deadline, rng, lr=0.05, beta_frac=0.08, beta_decay=0.93,
                no_imp_limit=400):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-4, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        if time.monotonic() > deadline - 1:
            break
        _, w = _smooth_max_and_weights(c, beta)
        g = _conv_cross_corr(w, p)

        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z_new = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z_new)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))
        p_new = _softmax_from_z(z_new)
        c_new = _full_convolve(p_new)
        maxc_new = float(c_new.max())
        score = float(2.0 * n * maxc_new)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p_new.copy()
                best_z = z_new.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.005, lr * 0.93)
            
            z = z_new
            p = p_new
            c = c_new
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 30 == 0:
            beta = max(5e-5, beta * beta_decay)

        if no_imp > no_imp_limit:
            ns = 0.05 if best_score < 1.51 else 0.1
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-4, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _local_search(p0, deadline, rng, n_perturb=500, scale_init=0.02):
    """Local random search: perturb random subsets of coordinates."""
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    n = p.size
    best_p = p.copy()
    best_score = _score_norm(p)
    scale = scale_init
    
    c = _full_convolve(p)
    
    for _ in range(n_perturb):
        if time.monotonic() > deadline - 1:
            break
        
        # Perturb a random subset of coordinates
        n_pert = max(1, int(n * 0.05))
        idx = rng.choice(n, size=n_pert, replace=False)
        delta = scale * rng.standard_normal(n_pert)
        
        p_try = p.copy()
        p_try[idx] = np.maximum(p_try[idx] + delta, 1e-12)
        p_try = p_try / p_try.sum()
        
        sc_try = _score_norm(p_try)
        
        if sc_try < best_score:
            best_score = sc_try
            best_p = p_try.copy()
            p = p_try.copy()
            scale = max(0.001, scale * 1.05)
        else:
            scale = max(0.001, scale * 0.95)
    
    return best_p, best_score


def _make_profile(n, kind, **params):
    t = np.arange(n, dtype=np.float64) / n
    if kind == 'linear':
        eta = params.get('eta', 1.0)
        a = 1.0 + eta * (t - 0.5)
    elif kind == 'quad':
        a1 = params.get('a1', 1.0)
        a2 = params.get('a2', -0.5)
        a = 1.0 + a1 * t + a2 * t * t
    elif kind == 'cubic':
        a1 = params.get('a1', 1.0)
        a3 = params.get('a3', -0.5)
        u = 2.0 * t - 1.0
        a = 1.0 + a1 * u + a3 * u**3
    elif kind == 'exp':
        lam = params.get('lam', 1.0)
        a = np.exp(-lam * t)
    elif kind == 'twobump':
        c1 = params.get('c1', 0.3)
        c2 = params.get('c2', 0.7)
        s = params.get('s', 0.15)
        a = np.exp(-0.5 * ((t - c1) / s)**2) + np.exp(-0.5 * ((t - c2) / s)**2)
    elif kind == 'cos':
        freq = params.get('freq', 1)
        amp = params.get('amp', 0.5)
        a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t)
    elif kind == 'power':
        alpha = params.get('alpha', 2.0)
        a = t**alpha + 1e-6
    elif kind == 'asym':
        center = params.get('center', 0.35)
        width = params.get('width', 0.2)
        skew = params.get('skew', 0.8)
        x = (t - center) / width
        a = np.exp(-0.5 * x * (x + skew))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0.copy() / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 1e-12)
    return p1 / p1.sum()
