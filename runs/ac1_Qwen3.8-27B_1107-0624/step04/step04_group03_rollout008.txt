"""

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _effective_active(w):
    s2 = float(np.sum(w * w))
    if s2 < 1e-300:
        return 1.0
    return 1.0 / s2


def _gradient_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _objective_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_obj(z, beta, n):
    return _objective_smooth(z, beta)


def _lbfgs_grad(z, beta, n):
    return _gradient_smooth(z, beta)


def _run_lbfgs_adaptive(a_init, deadline, max_wall_time=None):
    """L-BFGS with adaptive beta based on effective active set size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if max_wall_time is None:
        max_wall_time = min(300.0, max(1.0, deadline - time.monotonic()))

    # Start with large beta, adaptively decrease
    beta = 0.1 * max_c0
    min_beta = 1e-8 * max_c0

    target_active_low = 20
    target_active_high = 80

    total_time_spent = 0.0
    stuck_count = 0
    prev_score = best_score

    while time.monotonic() < deadline and total_time_spent < max_wall_time:
        t0 = time.monotonic()
        level_dl = min(deadline, t0 + max(2.0, max_wall_time - total_time_spent))

        try:
            result = minimize(
                _lbfgs_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_grad,
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-8,
                         'maxls': 50, 'maxfun': 1000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x.copy()
                a_new = _softmax_fn(z_new)
                sc_new = _score_norm_fn(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                z = z_new

                # Compute effective active set at new point
                c_new = _full_convolve(a_new, a_new)
                w_new, _ = _compute_weights(c_new, beta)
                eff_active = _effective_active(w_new)

                # Adapt beta
                if eff_active > target_active_high:
                    beta = max(min_beta, beta * 0.5)
                elif eff_active < target_active_low:
                    beta = min(0.5 * max_c0, beta * 1.5)
                else:
                    beta = max(min_beta, beta * 0.7)

                # Check for stagnation
                if abs(prev_score - best_score) < 1e-12:
                    stuck_count += 1
                else:
                    stuck_count = 0
                prev_score = best_score

                if stuck_count > 3:
                    beta = max(min_beta, beta * 0.3)
                    stuck_count = 0

        except Exception:
            pass

        total_time_spent += time.monotonic() - t0

        if beta <= min_beta * 1.1:
            break

    return best_a, best_score


def _nesterov_bb(a_init, deadline, beta_init_frac=0.01):
    """Nesterov with Barzilai-Borwein step size and adaptive beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, beta_init_frac * max_c)
    min_beta = 1e-9 * max_c

    mom = 0.9
    t = 0

    # BB state
    prev_z = z.copy()
    prev_grad = None
    bb_lr = 0.02

    target_active = 30

    while time.monotonic() < deadline:
        t += 1
        grad_y = _gradient_smooth(y, beta)
        if not np.all(np.isfinite(grad_y)):
            grad_y = _gradient_smooth(z, beta)

        # Compute BB step size
        if prev_grad is not None:
            s = z - prev_z
            y_bb = grad_y - prev_grad
            ss = float(np.dot(s, s))
            sy = float(np.dot(s, y_bb))
            yy = float(np.dot(y_bb, y_bb))
            if ss > 1e-15 and yy > 1e-15:
                if t % 2 == 0:
                    bb_candidate = ss / max(sy, 1e-15) if sy > 0 else 1.0 / (yy / ss)
                else:
                    bb_candidate = sy / yy if sy > 0 else ss / max(1e-15, yy)
                if np.isfinite(bb_candidate) and 1e-4 < bb_candidate < 0.5:
                    bb_lr = 0.5 * bb_lr + 0.5 * bb_candidate

        z_new = z - bb_lr * grad_y
        z_new = z_new - z_new.mean()
        z_new = np.clip(z_new, -60, 60)

        y_new = z_new + mom * (z_new - z)
        y_new = y_new - y_new.mean()

        prev_z = z.copy()
        prev_grad = grad_y.copy()
        z = z_new
        y = y_new

        # Slowly increase momentum
        mom = min(0.995, mom + 0.0001)

        # Periodic evaluation and beta adaptation
        if t % 30 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()

            c_check = _full_convolve(a_curr, a_curr)
            max_c = float(c_check.max())
            w_check, _ = _compute_weights(c_check, beta)
            eff = _effective_active(w_check)

            if eff > target_active * 2:
                beta = max(min_beta, beta * 0.5)
            elif eff < target_active * 0.3:
                beta = min(0.1 * max_c, beta * 2.0)
            else:
                beta = max(min_beta, beta * 0.8)

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score

    n = len(a_init)

    mom_coeff = 0.85
    lr = 0.003
    prev_z = z.copy()
    prev_grad = None
    bb_lr = lr

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Active set: lags within 0.1% of max
        threshold = cm * (1.0 - 1e-3)
        active = c >= threshold

        w = np.zeros_like(c)
        n_active = int(np.sum(active))

        a_rev = a[::-1]

            y_bb = grad - prev_grad
            ss = float(np.dot(s, s))
            sy = float(np.dot(s, y_bb))
            yy = float(np.dot(y_bb, y_bb))
            if ss > 1e-18 and yy > 1e-18 and sy > 0:
                bb_candidate = ss / sy
                if 1e-5 < bb_candidate < 0.1:
                    bb_lr = 0.3 * bb_lr + 0.7 * bb_candidate

        momentum = mom_coeff * momentum - bb_lr * grad
        z_new = z + momentum
        z_new = z_new - z_new.mean()
        z_new = np.clip(z_new, -60, 60)

        prev_z = z.copy()
        prev_grad = grad.copy()
        z = z_new

        if it % 200 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            else:
                mom_coeff = max(0.5, mom_coeff * 0.98)

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_profile(n, alpha, power):
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 + alpha * np.cos(2.0 * np.pi * t)) ** power
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n}, score={sc_parent:.10f}")
    else:
        n = 4096
        parent = _make_linear_profile(n, 1.0)
        sc_parent = _score_norm_fn(parent)
        print(f"No parent, using linear profile n={n}, score={sc_parent:.10f}")

    best_seq = parent.copy()
    best_score = sc_parent

    # --- Phase 0: Quick diverse initializations (30s) ---
    phase0_end = min(deadline, t_start + 30.0)
    candidates = [(best_score, parent.copy(), "parent")]

    test_profiles = [
        _make_linear_profile(n, 0.5),
        _make_linear_profile(n, 1.5),
        _make_quadratic_profile(n, 0.5, 0.3),
        _make_quadratic_profile(n, -0.5, 0.3),
        _make_cosine_profile(n, 0.3, 2.0),
        _make_cosine_profile(n, 0.5, 1.5),
        _make_cosine_profile(n, 0.2, 3.0),
    ]

    for idx, p in enumerate(test_profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p, f"prof{idx}"))
        if sc < best_score:
            best_score = sc
            best_seq = p.copy()
            print(f"  Phase0: profile {idx} improved to {sc:.10f}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}")

    # --- Phase 1: L-BFGS adaptive continuation (400s) ---
    phase1_end = min(deadline, t_start + 430.0)
    if best_seq is not None and time.monotonic() < phase1_end:
        per_time = phase1_end - time.monotonic() - 5
        if per_time > 20:
            p_best, sc_best = _run_lbfgs_adaptive(best_seq, phase1_end - 5, max_wall_time=per_time)
            update_best(p_best, sc_best, "lbfgs_adaptive")

    # Also try on second-best candidate
    if time.monotonic() < phase1_end - 60 and len(candidates) > 1:
        for sc, p, label in candidates[1:4]:
            if time.monotonic() > phase1_end - 20:
                break
            per_time = min(60.0, phase1_end - time.monotonic() - 10)
            if per_time > 15:
                p_best, sc_best = _run_lbfgs_adaptive(p, phase1_end - 10, max_wall_time=per_time)
                update_best(p_best, sc_best, f"lbfgs_{label}")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Nesterov with BB step size (350s) ---
    phase2_end = min(deadline, t_start + 780.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        per_time = phase2_end - time.monotonic() - 5
        if per_time > 30:
            p_best, sc_best = _nesterov_bb(best_seq, phase2_end - 5, beta_init_frac=0.005)
            update_best(p_best, sc_best, "nesterov_bb")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Exact subgradient polish (250s) ---
    phase3_end = min(deadline, t_start + 1030.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        per_time = phase3_end - time.monotonic() - 5
        if per_time > 30:
            p_best, sc_best = _exact_subgradient_polish(best_seq, phase3_end - 5)
            update_best(p_best, sc_best, "exact_subgrad")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Structured restarts (350s) ---
    phase4_end = min(deadline, t_start + 1380.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        n_best = len(best_seq)
        n_restarts = 6
        per_restart = (phase4_end - time.monotonic()) / max(1, n_restarts) - 5

        restart_types = [
            ("sin_low", lambda z: z + 0.03 * np.sin(2 * np.pi * 1 * np.arange(n_best) / n_best)),
            ("sin_2", lambda z: z + 0.02 * np.sin(2 * np.pi * 2 * np.arange(n_best) / n_best)),
            ("sin_3", lambda z: z + 0.02 * np.sin(2 * np.pi * 3 * np.arange(n_best) / n_best)),
            ("noise_small", lambda z: z + 0.02 * rng.normal(size=n_best)),
            ("noise_med", lambda z: z + 0.05 * rng.normal(size=n_best)),
            ("mirror", lambda z: np.maximum(z, z[::-1]) - z.mean()),
        ]

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 15:
                break
            if i >= len(restart_types):
                break
            name, perturb_fn = restart_types[i]
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = perturb_fn(z.copy())
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            per_dl = min(phase4_end - 10, time.monotonic() + max(10.0, per_restart))

            # Short L-BFGS
            if time.monotonic() < per_dl - 10:
                p_mid, sc_mid = _run_lbfgs_adaptive(p_start, min(per_dl - 10, time.monotonic() + min(30.0, per_restart * 0.5)),
                                                     max_wall_time=min(30.0, per_restart * 0.5))
                update_best(p_mid, sc_mid, f"restart_{name}_lbfgs")

            # Short Nesterov
            if time.monotonic() < per_dl - 5:
                p_end, sc_end = _nesterov_bb(p_mid, per_dl, beta_init_frac=0.002)
                update_best(p_end, sc_end, f"restart_{name}_nest")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multi-length exploration (200s) ---
    phase5_end = min(deadline, t_start + 1580.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        n_curr = len(best_seq)

        # Try larger n
        target_n = min(8192, n_curr * 2)
        if target_n > n_curr:
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            print(f"  Interp to n={target_n}: score={sc_interp:.10f} (vs {best_score:.10f})")
            if sc_interp < best_score:
                update_best(p_interp, sc_interp, f"interp_n{target_n}")
                if time.monotonic() < phase5_end - 60:
                    per_time = min(120.0, phase5_end - time.monotonic() - 20)
                    if per_time > 20:
                        p_best, sc_best = _run_lbfgs_adaptive(p_interp, phase5_end - 20, max_wall_time=per_time)
                        update_best(p_best, sc_best, f"lbfgs_n{target_n}")

        # Try smaller n (if current is large)
        if n_curr >= 4096:
            target_n2 = 2048
            p_interp2 = _interp_profile(best_seq, target_n2)
            sc_interp2 = _score_norm_fn(p_interp2)
            print(f"  Interp to n={target_n2}: score={sc_interp2:.10f}")
            if sc_interp2 < best_score:
                update_best(p_interp2, sc_interp2, f"interp_n{target_n2}")
                if time.monotonic() < phase5_end - 40:
                    per_time = min(60.0, phase5_end - time.monotonic() - 15)
                    if per_time > 15:
                        p_best, sc_best = _nesterov_bb(p_interp2, phase5_end - 15, beta_init_frac=0.005)
                        update_best(p_best, sc_best, f"nest_n{target_n2}")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Final fine polish (remaining time) ---
    phase6_end = deadline
    if best_seq is not None and time.monotonic() < phase6_end:
        remaining = phase6_end - time.monotonic()

        # Nesterov with tiny beta
        if remaining > 40:
            per_time = min(60.0, remaining - 20)
            p_best, sc_best = _nesterov_bb(best_seq, min(phase6_end - 25, time.monotonic() + per_time),
                                            beta_init_frac=0.0005)
            update_best(p_best, sc_best, "final_nesterov")

        # Exact subgradient
        if time.monotonic() < phase6_end - 15:
            per_time = min(40.0, phase6_end - time.monotonic() - 5)
            if per_time > 10:
                p_best, sc_best = _exact_subgradient_polish(best_seq, min(phase6_end - 5, time.monotonic() + per_time))
                update_best(p_best, sc_best, "final_subgrad")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_linear_profile(4096, 1.0) / _make_linear_profile(4096, 1.0).sum()

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned inf/NaN, using internal: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_linear_profile(4096, 1.0)
    fallback = fallback / fallback.sum()
    print(f"FALLBACK: n=4096, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
