```python
def _lp_solve(a, M):
    """Solve max sum(b) s.t. conv(a,b)[k] <= M for all k, b >= 0."""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    
    # Build constraints: for each lag k, sum_j a[j]*b[k-j] <= M
    # conv(a,b)[k] = sum_{j} a[j] * b[k-j] where 0<=j<n and 0<=k-j<n
    constraints = []
    for k in range(2 * n - 1):
        # j ranges from max(0, k-n+1) to min(n-1, k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # conv[k] = sum_{j=j_start}^{j_end} a[j] * b[k-j]
            # b index: k-j, ranges from k-j_end to k-j_start
            expr = 0
            for j in range(j_start, j_end + 1):
                expr = expr + a[j] * b[k - j]
            constraints.append(expr <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    prob.solve(solver=cp.CLARABEL)
    if b.value is not None:
        return np.maximum(b.value, 0.0)
    return None
```

geometric initial profiles, and multi-scale interpolation for minimizing
2*n*max(conv(a,a))/sum(a)^2.
Uses scipy HiGHS for LP solves, adaptive line search, and dual-guided
support identification for final refinement.

import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

    while k < n:
        k <<= 1
    return max(2, k)

    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def _precompute_structure(n):
    """Precompute row/col indices for the convolution constraint matrix."""
    rows = []
    cols = []
    for j in range(n):
        length = min(n, 2 * n - 1 - j)
        rows.extend(range(j, j + length))
        cols.extend([j] * length)
    return np.array(rows, dtype=np.int64), np.array(cols, dtype=np.int64)


def _build_A_dense(a, rows, cols):
    """Build dense constraint matrix where (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    vals = a[rows - cols]
    A = np.zeros((2 * n - 1, n))
    A[rows, cols] = vals
    return A


def _lp_solve_minmax(A, n):
    """Solve min t s.t. A@b <= t*1, sum(b)=1, b>=0. Returns (t, b*1)."""
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.hstack([np.ones((1, n)), np.zeros(1)])
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return float(res.x[-1]), np.maximum(res.x[:n], 0.0)
    return None, 1


def _lp_fixed_point(a, deadline, max_iters=80):
    """LP fixed-point iteration with line search."""
    n = len(a)
    a = _normalize(a)
    rows, cols = _precompute_structure(n)

    best_a = a.copy()
    best_score = _score(a)
    prev_score = best_score

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        A = _build_A_dense(a, rows, cols)
        t_val, b = _lp_solve_minmax(A, n)

        if t_val is None:
            break

        b = _normalize(b)

        # Line search over mixing parameter
        best_alpha = 1.0
        best_sc = best_score
        for alpha in [0.5, 0.7, 0.85, 1.0]:
            a_new = (1.0 - alpha) * a + alpha * b
            a_new = _normalize(a_new)
            sc = _score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * b
        a = _normalize(a)

        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Convergence check
        if abs(prev_score - sc) < 1e-10:
            if it > 5:
                break
        prev_score = sc

    return best_a, best_score


def _smooth_max_weights(c, beta):
    """Compute softmax weights for smooth max approximation."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w


def _cross_correlate(w, a):
    """Compute cross-correlation: (w * a_reversed) via FFT."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]


def _adam_smooth_max(a, deadline, rng, lr=0.04, beta_init=None,
                     beta_decay=0.96, no_imp_limit=300):
    """Adam optimizer with smooth-max approximation of the peak."""
    n = len(a)
    a = _normalize(a)
    z = np.log(np.maximum(a, 1e-12))
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = _full_convolve(a)
    maxc = float(c.max())
    beta = beta_init if beta_init is not None else 0.05 * maxc

    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)
        g = _cross_correlate(w, a)

        dot = float(np.dot(a, g))
        gz = a * (g - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()

        a = np.exp(z - z.max())
        a = _normalize(a)

        c = _full_convolve(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.002, lr * 0.9)
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        if t_adam % 30 == 0:
            beta = max(1e-5, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.05:
                    beta = max(1e-5, beta * 0.4)

        if no_imp > no_imp_limit:
            ns = 0.06 if best_score < 1.52 else 0.12
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = _normalize(a)
            c = _full_convolve(a)
            beta = 0.05 * c.max()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    a_best = np.exp(best_z - best_z.max())
    a_best = _normalize(a_best)
    return a_best, best_score


def _coord_descent(a, deadline, max_sweeps=30):
    """Coordinate descent: try scaling each element by factors."""
    n = len(a)
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score(a)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            for factor in [0.6, 0.8, 1.25, 1.6]:
                a[j] = max(old_val * factor, 1e-15)
                a[j+1:] = a[j+1:]  # no-op, just for clarity
                sc = _score(a)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = old_val
                    break
            a[j] = old_val
        if not improved:
            break

    return best_a, best_score


def _interp(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()


def _make_profile(n, ptype, p1=1.0, p2=0.0, p3=0.0):
    """Generate various profile shapes."""
    if ptype == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (t - 0.5)
    elif ptype == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (2 * t - 1) ** 2
    elif ptype == 'geometric':
        r = p1
        a = r ** np.arange(n)
    elif ptype == 'exp_decay':
        a = np.exp(-p1 * np.arange(n) / n)
    elif ptype == 'cosine':
        t = np.arange(n)
        a = 1.0 + p1 * np.cos(2 * np.pi * p2 * t / n)
    elif ptype == 'U_shape':
        t = np.linspace(0, 1, n)
        a = (2 * t - 1) ** 2 + p1
    elif ptype == 'gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / p1) ** 2)
    elif ptype == 'two_gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - p1) / p3) ** 2) + np.exp(-0.5 * ((t - p2) / p3) ** 2)
    elif ptype == 'edge_heavy':
        t = np.arange(n) / max(1, n - 1)
        a = 1.0 / (0.01 + abs(t - 0.5) * 2.0) ** p1
    elif ptype == 'plateau':
        t = np.linspace(0, 1, n)
        a = np.where(np.abs(t - 0.5) < p1, 1.0, 0.3)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization: LP fixed-point (primary), Adam smooth-max
    (secondary), coordinate descent (tertiary), and multi-scale escalation.
    Builds on parent sequence if available.
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # --- Get parent sequence ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s, deadline at t={deadline - t_start:.1f}s")

    # === Phase 1: Generate diverse initial candidates ===
    print("Phase 1: Generating initial candidates...")
    candidates = []

    n_sizes = [256, 512, 1024, 2048]
    profile_defs = [
        ('lin1', 'linear', 1.0, 0, 0),
        ('lin2', 'linear', 2.0, 0, 0),
        ('lin3', 'linear', 3.0, 0, 0),
        ('quad1', 'quadratic', 1.0, 0, 0),
        ('quad2', 'quadratic', 2.0, 0, 0),
        ('geo099', 'geometric', 0.99, 0, 0),
        ('geo0995', 'geometric', 0.995, 0, 0),
        ('geo0999', 'geometric', 0.999, 0, 0),
        ('exp05', 'exp_decay', 0.5, 0, 0),
        ('exp10', 'exp_decay', 1.0, 0, 0),
        ('exp20', 'exp_decay', 2.0, 0, 0),
        ('cos1', 'cosine', 0.8, 1, 0),
        ('cos2', 'cosine', 0.6, 2, 0),
        ('U03', 'U_shape', 0.3, 0, 0),
        ('U05', 'U_shape', 0.5, 0, 0),
        ('g03', 'gauss', 0.3, 0, 0),
        ('g04', 'gauss', 0.4, 0, 0),
        ('tg30', 'two_gauss', 0.3, 0.7, 0.15),
        ('tg25', 'two_gauss', 0.25, 0.75, 0.2),
        ('edge1', 'edge_heavy', 1.0, 0, 0),
        ('edge2', 'edge_heavy', 2.0, 0, 0),
        ('plat03', 'plateau', 0.3, 0, 0),
        ('plat05', 'plateau', 0.5, 0, 0),
    ]

    for n in n_sizes:
        for (label, ptype, p1, p2, p3) in profile_defs:
            p = _make_profile(n, ptype, p1, p2, p3)
            sc = _score(p)
            if np.isfinite(sc):
                candidates.append((sc, p, n, label))

    # Add parent-based candidates
    if parent is not None:
        for n in [512, 1024, 2048, 4096]:
            pi = _interp(parent, n)
            sc = _score(pi)
            if np.isfinite(sc):
                candidates.append((sc, pi, n, 'parent'))
        sc_p = _score(parent)
        if np.isfinite(sc_p):
            candidates.append((sc_p, parent.copy(), pn, 'parent_exact'))
            print(f"  Parent score: {sc_p:.10f} (n={pn})")

    # Add random candidates
    for n in [512, 1024]:
        for _ in range(5):
            z = rng.normal(size=n)
            a = np.exp(z - z.max())
            a = _normalize(a)
            sc = _score(a)
            if np.isfinite(sc):
                candidates.append((sc, a, n, 'rand'))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, best init: {candidates[0][0]:.8f} n={candidates[0][2]} [{candidates[0][3]}]")

    if candidates[0][0] < best_score:
        best_seq = candidates[0][1].copy()
        best_score = candidates[0][0]
        print(f"  BEST: {best_score:.10f} n={best_seq.size} [init]")

    # === Phase 2: LP fixed-point on top candidates ===
    print("Phase 2: LP fixed-point optimization...")
    phase2_end = t_start + min(900.0, budget_s * 0.5)
    polished = []
    count = 0

    # Sort by (score, n) to prefer good scores at smaller n for speed
    candidates_lp = sorted(candidates, key=lambda x: (x[0], x[2]))

    for (sc, p, n, label) in candidates_lp:
        if time.monotonic() > phase2_end or count >= 15:
            break
        if n > 4096:
            continue

        # Time allocation: more time for more promising candidates
        remaining = phase2_end - time.monotonic()
        if remaining < 30:
            break
        time_alloc = max(60.0, min(180.0, remaining / max(1, 15 - count)))
        per_end = min(phase2_end - 5, time.monotonic() + time_alloc)

        a_lp, sc_lp = _lp_fixed_point(p, per_end, max_iters=60)
        if sc_lp < best_score:
            best_seq = a_lp.copy()
            best_score = sc_lp
            print(f"  BEST: {best_score:.10f} n={a_lp.size} [LP_{label}_n{n}]")
        polished.append((sc_lp, a_lp, n, label + '_lp'))
        count += 1

    polished.sort(key=lambda x: x[0])
    print(f"  After Phase 2: best={best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    # === Phase 3: Adam refinement on best LP results ===
    print("Phase 3: Adam smooth-max refinement...")
    phase3_end = t_start + min(1400.0, budget_s * 0.75)

            break

            best_score = sc_ad

        # LP polish after Adam
        if n <= 2048 and time.monotonic() < phase3_end - 40:
            per2 = min(80.0, phase3_end - time.monotonic() - 10)
            if per2 > 20:
                a_lp2, sc_lp2 = _lp_fixed_point(a_ad, time.monotonic() + per2, max_iters=20)
                if sc_lp2 < best_score:
                    best_seq = a_lp2.copy()
                    best_score = sc_lp2
                    print(f"  BEST: {best_score:.10f} n={a_lp2.size} [LPpostAdam_{label}]")

            target_ns = [4096]

            print(f"  Interpolated to n={tn}: {sc_int:.8f}")

                best_score = sc_int
                print(f"  BEST: {best_score:.10f} n={tn} [interp]")

                        print(f"  BEST: {best_score:.10f} n={tn} [LP_esc]")

            # Adam at new scale
            if time.monotonic() < phase4_end - 40:
