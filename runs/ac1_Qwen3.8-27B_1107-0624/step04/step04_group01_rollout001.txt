"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp smoothing with geometric beta decay, Adam optimizer with
backtracking-inspired adaptive LR, diverse restarts, and multi-length search.
Key improvements over previous: better beta scheduling, active-set focusing,
longer convergence at each scale, and systematic n exploration.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adam_optimize_improved(a_init, deadline, beta_schedule, max_iter=200000, lr=0.01, patience=2000):
    """Adam with adaptive LR and patience-based convergence."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    prev_best = best_score
    stale_count = 0

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        phase_best = best_score
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad = _compute_grad_z(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            max_update = 0.3
            update_norm = np.linalg.norm(update)
            if update_norm > max_update:
                update = update * (max_update / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 50 == 0:
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_score:
                    if sc < best_score * (1 - 1e-7):
                        best_score = sc
                        best_a = a.copy()
                        stale_count = 0
                    else:
                        best_score = sc
                        best_a = a.copy()
                        stale_count = 0
                else:
                    stale_count += 50

                # Adaptive LR
                if it_in_phase % 2000 == 0:
                    if best_score < prev_best * 0.9999:
                        current_lr = min(0.04, current_lr * 1.05)
                    elif best_score > prev_best * 1.0001:
                        current_lr = max(0.002, current_lr * 0.85)
                    else:
                        current_lr = current_lr * 0.97
                    prev_best = best_score

                # Early stop if very stale
                if stale_count > patience:
                    break

            if time.monotonic() > deadline:
                break

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _activate_refine(a_init, deadline, n_active=5, lr=0.005, max_iter=50000):
    """
    Active-set refinement: only consider the top-n_active convolution values
    in the gradient computation. This focuses optimization on the critical peaks.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Use top-n_active for gradient (scaled log-sum-exp)
        idx = np.argpartition(c, -n_active)[-n_active:]
        c_top = c[idx].copy()
        c_max = c_top.max()
        c_shifted = (c_top - c_max) / (0.01 * c_max + 1e-10)
        e = np.exp(c_shifted)
        w_top = e / e.sum()

        # Build full w with only active set
        w = np.zeros_like(c)
        w[idx] = w_top

        grad = _compute_grad_z(a, c, w, n)
        if not np.all(np.isfinite(grad)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)

        update = current_lr * mh / (np.sqrt(vh) + eps)
        max_update = 0.1
        update_norm = np.linalg.norm(update)
        if update_norm > max_update:
            update = update * (max_update / update_norm)

        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if t % 100 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    """Add random Fourier modes to log-space."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 8))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.01, 0.05)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block(a, rng, block_size=16):
    """Perturb blocks of the sequence."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    n_blocks = n // block_size
    for _ in range(min(n_blocks, 10)):
        b = rng.integers(0, n_blocks)
        start = b * block_size
        end = start + block_size
        z[start:end] += rng.uniform(-0.1, 0.1)
    z = z - z.mean()
    return _softmax(z)


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _mirrored_extend(a):
    """Create a palindromic version of a by mirroring."""
    n = len(a)
    return np.concatenate([a, a[::-1]]) / (a.sum() * 2)
