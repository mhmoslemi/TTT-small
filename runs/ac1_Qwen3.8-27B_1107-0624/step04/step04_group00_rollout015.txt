def _solve_peak_lp(a, c, K=30, step_size=0.001):
    n = len(a)
    top_k = np.argsort(c)[::-1][:K]
    
    # Build A_ub for peak constraints
    A_peak = np.zeros((K, n + 1))
    b_peak = np.zeros(K)
    
    for idx, k in enumerate(top_k):
        A_peak[idx, 0] = -1.0
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                A_peak[idx, j + 1] = -2.0 * a[kj]
        b_peak[idx] = -c[k]
    
    # Non-negativity: -a[i] <= δ[i] <= step_size
    # δ[i] <= step_size: A_ub row has 1 at position i+1
    # -δ[i] <= a[i]: A_ub row has -1 at position i+1
    
    A_bounds = np.zeros((2 * n, n + 1))
    b_bounds = np.zeros(2 * n)
    for i in range(n):
        A_bounds[i, i + 1] = 1.0
        b_bounds[i] = step_size
        A_bounds[n + i, i + 1] = -1.0
        b_bounds[n + i] = a[i]
    
    A_ub = np.vstack([A_peak, A_bounds])
    b_ub = np.concatenate([b_peak, b_bounds])
    
    # Equality: sum(δ) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    # Objective: minimize t

        return delta
    return None

        # coeff of δ[j] = -2*a[k-j] if 0 <= k-j < n

            A_peak[idx, j_start + 1:j_end + 2] = coeffs
        b_peak[idx] = -c[k]
    
    A_eq = np.zeros((1, n + 1))

    b_eq = np.zeros(1)
    
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0

"""
LP-guided peak equalization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal peak-reduction directions,
combined with projected gradient descent, peak equalization LPs,
and adaptive step-size control. Novel approach: direct A-space
optimization with LP-based direction finding replaces softmax
parameterization for faster convergence on the max objective.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max()) / (s * s)


def _softmax_weights(c, beta):
    cm = float(c.max())
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w, cm
    return e / s, cm


def _compute_grad(a, w, n):
    """Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a."""
    n_a = len(a)
    n_w = len(w)
    N = _next_pow2(n_a + n_w - 1)
    a_flip = a[::-1]
    faf = np.fft.rfft(a_flip, N)
    fw = np.fft.rfft(w, N)
    conv_w_aflip = np.fft.irfft(fw * faf, N)
    grad = 2.0 * conv_w_aflip[n_a - 1: n_a - 1 + n_a]
    return grad


def _project_nonneg_sum(a, target_sum):
    """Project a onto {x >= 0, sum(x) = target_sum} approximately."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a = a * (target_sum / s)
    return a


def _solve_lp_reduce_max(a, c, K, step_size):
    """LP: minimize predicted max peak. Returns (delta, predicted_max)."""
    n = len(a)
    top_k = np.argsort(c)[::-1][:K]

    A_peak = np.zeros((K, n + 1))
    b_peak = np.zeros(K)

    for idx, k in enumerate(top_k):
        A_peak[idx, 0] = -1.0
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            A_peak[idx, js + 1] = -2.0 * a[k - js]
        b_peak[idx] = -c[k]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0

    bounds = [(0, None)] + [(-a[i], step_size) for i in range(n)]

    try:
        result = linprog(c_obj, A_ub=A_peak, b_ub=b_peak,
                         A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={"presolve": True, "time_limit": 10})
        if result.success:
            return result.x[1:], float(result.x[0])
    except Exception:
        pass
    return None, None


def _solve_lp_equalize(a, c, K, step_size):
    """LP: equalize top K peaks. Returns delta or None."""
    n = len(a)
    top_k = np.argsort(c)[::-1][:K]
    cm = float(c[top_k].max())

    # Variables: [t, m, delta_0, ..., delta_{n-1}]
    # minimize t - m (reduce max, increase margin)
    # Constraints:
    #   t >= c[k] + 2*conv(delta,a)[k]   => -t - 2*sum_j delta[j]*a[k-j] <= -c[k]
    #   t - m >= c[k] + 2*conv(delta,a)[k] => -t + m - 2*sum_j delta[j]*a[k-j] <= -c[k]

    n_vars = n + 2
    A1 = np.zeros((2 * K, n_vars))
    b1 = np.zeros(2 * K)

    for idx, k in enumerate(top_k):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # Row 2*idx: -t - 2*sum delta*a[k-j] <= -c[k]
        A1[2 * idx, 0] = -1.0
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            A1[2 * idx, js + 2] = -2.0 * a[k - js]
        b1[2 * idx] = -c[k]

        # Row 2*idx+1: -t + m - 2*sum delta*a[k-j] <= -c[k]
        A1[2 * idx + 1, 0] = -1.0
        A1[2 * idx + 1, 1] = 1.0
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            A1[2 * idx + 1, js + 2] = -2.0 * a[k - js]
        b1[2 * idx + 1] = -c[k]

    A_eq = np.zeros((1, n_vars))
    A_eq[0, 2:] = 1.0
    b_eq = np.zeros(1)

    c_obj = np.zeros(n_vars)
    c_obj[0] = 1.0
    c_obj[1] = -1.0

    bounds = [(0, None), (0, None)] + [(-a[i], step_size) for i in range(n)]

    try:
        result = linprog(c_obj, A_ub=A1, b_ub=b1,
                         A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={"presolve": True, "time_limit": 10})
        if result.success:
            return result.x[2:]
    except Exception:
        pass
    return None


def _gradient_step(a, beta, lr):
    """One projected gradient step with softmax-weighted objective."""
    c = _full_convolve(a, a)
    w, cm = _softmax_weights(c, beta)
    n = len(a)
    grad = _compute_grad(a, w, n)
    a_new = a - lr * grad
    a_new = _project_nonneg_sum(a_new, a.sum())
    return a_new, c

def _run_gradient_optimization(a, deadline, beta_list=None, rng=None):
    """Gradient-based optimization with multiple temperatures."""
    n = len(a)
    target_sum = a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    if rng is None:
        rng = np.random.default_rng(123)

    if beta_list is None:
        c0 = _full_convolve(a, a)
        cm0 = float(c0.max())
        beta_list = [0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001]
        beta_list = [b * cm0 for b in beta_list]

    for beta in beta_list:
        if time.monotonic() > deadline - 20:
            break

        phase_deadline = min(deadline - 10, time.monotonic() + (deadline - time.monotonic()) / (len(beta_list) + 1))
        lr = 0.001
        c = _full_convolve(a, a)

        for it in range(5000):
            if time.monotonic() > phase_deadline:
                break

            c = _full_convolve(a, a)
            w, cm = _softmax_weights(c, beta)
            grad = _compute_grad(a, w, n)

            # Normalize gradient
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            grad = grad / gnorm

            a_new = a - lr * gnorm * grad
            a_new = _project_nonneg_sum(a_new, target_sum)

            if np.random.random() < 0.1:
                a_new = a_new + lr * 0.01 * rng.normal(size=n)
                a_new = _project_nonneg_sum(a_new, target_sum)

            sc_new = _score_norm(a_new)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new.copy()
                lr = min(lr * 1.01, 0.01)
            else:
                lr *= 0.9
                a = best_a.copy()  # revert

            if lr < 1e-10:
                break

    return best_a, best_score


def _make_initial_profiles(n):

    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    for slope in [1.0, 1.5, 2.0, 3.0, 5.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-10)

    for rate in [1.0, 2.0, 3.0, 5.0, 8.0]:

        a = np.maximum(np.cos(np.pi * u / 2), 1e-10) ** p
        profiles.append(a / a.sum())
    return profiles
