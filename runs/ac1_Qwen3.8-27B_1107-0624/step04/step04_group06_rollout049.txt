def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _smooth_max_power(c, power):
    """Power-softmax: M_p = (sum c_k^p)^(1/p). Returns (value, weights)."""
    c = np.asarray(c, dtype=np.float64)
    c = np.maximum(c, 1e-30)
    if power < 1.01:
        mval = float(c.max())
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = np.zeros_like(c)
            w[mask] = 1.0 / cnt
        return mval, w
    cp = c ** power
    s = float(cp.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return c.max(), w
    val = s ** (1.0 / power)
    w = cp / s
    return val, w


def _smooth_max_topk(c, k, beta):
    """Smooth max focusing on top-k values with soft weighting."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if k >= len(c):
        return _smooth_max_power(c, 50)
    idx = np.argsort(c)[::-1][:k]
    topk = c[idx]
    # Soft-max over top-k
    x = (topk - topk.max()) / max(beta, 1e-10)
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return c.max(), w
    weights_topk = ex / s
    w = np.zeros_like(c)
    w[idx] = weights_topk
    val = mval + beta * float(np.log(s))
    return val, w


def _conv_grad(p, w):
    """
    Compute gradient of sum_k w_k * b_k where b = p*p (full convolution).
    grad_j = sum_k w_k * db_k/dp_j
    For b_k = sum_i p_i p_{k-i}: db_k/dp_j = p_{k-j} (if 0<=k-j<n) + p_{k-j} (if j>=k and 0<=k-j<n from other factor)
    Simplified: grad_j = 2 * sum_{m} w_{j+m} * p_m (approximately, ignoring k=2j correction)
    This is 2 * cross_correlation(p, w)_j
    """
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fp = np.fft.rfft(p, N)
    fw = np.fft.rfft(w, N)
    # cross-correlation: (p ⋆ w)[j] = sum_m p[m] w[m+j] (circular)
    corr = np.fft.irfft(np.conj(fp) * fw, N)
    # We want sum_m w[m+j] p[m] for m in [0,n-1], j+m in [0, 2n-2]
    # This is corr[j] for j < n (the circular version is close for well-separated signals)
    return 2.0 * corr[:n]

def _subgradient_polyak(p0, deadline, rng, eta_init=0.02, n_restarts=5):

    for restart in range(n_restarts):

def _lp_solve_tight(a, M):
    """Solve LP: max sum(b) s.t. a*b <= M, b >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_aggressive(a, deadline, max_solves=60):

            if b is not None and b.sum() > 1e-10:
                found = b
                break
            if shrink >= 1.0:
                found = b
                break

        if found is None or found.sum() < 1e-10:
            M *= 1.01
            continue

        s_b = float(found.sum())
        b_norm = found / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)

        sc_new = _score_norm(a_new)

        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        c = _full_convolve(a)

    return best_a, best_score

                continue
            a_try = a.copy()
            a_try[i_src] -= amount
            a_try[i_dst] += amount
            a_try = np.maximum(a_try, 1e-12)
            a_try = a_try / a_try.sum()
            sc_try = _score_norm(a_try)
            if sc_try < best_local_score:
                best_local_score = sc_try
                best_local = a_try.copy()

        a = best_local

    return best_a, best_score


def _peak_shave(a, deadline, n_iters=5000):
    """Direct peak shaving: iteratively reduce mass at peak-contributing positions."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a)
        M = float(c.max())
        score = float(2.0 * n * M)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        # Get the top 3 peaks
        top_idx = np.argsort(c)[::-1][:3]
        # For each peak, find the single most contributing position
        for k in top_idx:
            i_start = max(0, int(k) - n + 1)
            i_end = min(n - 1, int(k))
            if i_start > i_end:
                continue
            contrib = a[i_start:i_end + 1] * a[int(k) - np.arange(i_start, i_end + 1)]
            i_peak = i_start + int(np.argmax(contrib))
            # Reduce mass at i_peak, redistribute to random positions
            reduce_frac = 0.02
            amount = a[i_peak] * reduce_frac
            if amount < 1e-10:
                continue
            a[i_peak] -= amount
            # Redistribute to positions that don't contribute to current peaks
            n_redir = min(5, n)
            redir_idx = np.random.choice(n, size=n_redir, replace=False)
            a[redir_idx] += amount / n_redir

        a = np.maximum(a, 1e-12)
        a = a / a.sum()

    return best_a, best_score


def _make_cosine_init(n, alpha, beta=0.0):
    """Initialize with cosine shape: a_i ~ (1 + alpha*cos(2*pi*i/n) + beta*cos(4*pi*i/n))^2"""
    x = np.arange(n)
    a = 1.0 + alpha * np.cos(2 * np.pi * x / n) + beta * np.cos(4 * np.pi * x / n)
    a = a ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_edge_init(n, gamma):
    """Initialize with edge concentration: a_i ~ (1 + gamma*|2i/(n-1) - 1|)^2"""
    x = np.linspace(-1, 1, n)
    a = (1.0 + gamma * np.abs(x)) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_to_n(p, n_new):
    """Interpolate profile to new length."""
    p0 = np.asarray(p, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(_score_norm(p)):
            sc = _score_norm(p)
            if sc < best_score:
                best_seq = np.asarray(p, dtype=np.float64).copy()
                best_score = float(sc)
                print(f"  BEST: {best_score:.10f} n={p.size} [{label}] t={time.monotonic()-t_start:.0f}s")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    print(f"Parent: n={parent.size if parent is not None else 0}, budget={budget_s}s")

    if parent is not None:
        sc_p = _score_norm(parent)
        print(f"Parent score: {sc_p:.10f}")

    # === PHASE 1: Adam with power-softmax on parent (250s) ===
    phase1_end = t_start + 250.0
    print(f"\n=== Phase 1: Adam power-softmax (parent n={parent.size if parent is not None else '?'}) ===")
    if parent is not None:
        p1, sc1 = _adam_logit_power(parent, phase1_end, rng, lr=0.04,
                                    power_start=20.0, power_end=100.0,
                                    no_imp_limit=200)
        _update(p1, 'adam1')
    else:
        p1 = _make_cosine_init(1024, 1.5)
        _update(p1, 'cos_init')

    # === PHASE 2: Aggressive LP (300s) ===
    phase2_end = t_start + 550.0
    print(f"\n=== Phase 2: LP aggressive, current best={best_score:.10f} ===")
    if best_seq is not None and best_seq.size <= 2048:
        p2, sc2 = _lp_aggressive(best_seq, phase2_end, max_solves=80)
        _update(p2, 'lp_agg')

    # === PHASE 3: Adam after LP (200s) ===
    phase3_end = t_start + 750.0
    print(f"\n=== Phase 3: Post-LP Adam, current best={best_score:.10f} ===")
    if best_seq is not None:
        p3, sc3 = _adam_logit_power(best_seq, phase3_end, rng, lr=0.03,
                                    power_start=30.0, power_end=120.0,
                                    no_imp_limit=150)
        _update(p3, 'adam2')

    # === PHASE 4: Pair-swap local search (200s) ===
    phase4_end = t_start + 950.0
    print(f"\n=== Phase 4: Pair-swap search, current best={best_score:.10f} ===")
    if best_seq is not None and best_seq.size <= 2048:
        p4, sc4 = _pair_swap_search(best_seq, phase4_end, n_rounds=3000)
        _update(p4, 'pair_swap')

        # Quick Adam after pair swap
        if time.monotonic() < phase4_end - 30:
            p4a, sc4a = _adam_logit_power(best_seq, phase4_end, rng, lr=0.02,
                                          power_start=50.0, power_end=150.0,
                                          no_imp_limit=100)
            _update(p4a, 'post_swap_adam')

    # === PHASE 5: n=2048 exploration (300s) ===
    phase5_end = t_start + 1250.0
    print(f"\n=== Phase 5: n=2048 exploration, current best={best_score:.10f} ===")
    if time.monotonic() < phase5_end and best_seq is not None:
        n_new = 2048
        p_int = _interp_to_n(best_seq, n_new)
        sc_int = _score_norm(p_int)
        print(f"  Interpolated to n=2048: {sc_int:.10f}")
        _update(p_int, 'interp_2048')

        # Also try direct cosine init at n=2048
        p_cos2048 = _make_cosine_init(2048, 1.5)
        sc_cos = _score_norm(p_cos2048)
        print(f"  Cosine init n=2048: {sc_cos:.10f}")
        _update(p_cos2048, 'cos_2048')

        # Adam on the best n=2048 candidate
        base_2048 = p_int if sc_int < sc_cos else p_cos2048
        per = max(30.0, phase5_end - time.monotonic() - 10)
        if per > 30 and best_seq is not None or base_2048 is not None:
            p5, sc5 = _adam_logit_power(base_2048, time.monotonic() + per, rng,
                                        lr=0.04, power_start=20.0, power_end=80.0,
                                        no_imp_limit=200)
            _update(p5, 'adam_2048')

    # === PHASE 6: Peak shaving + final LP (200s) ===
    phase6_end = t_start + 1450.0
    print(f"\n=== Phase 6: Peak shave + LP, current best={best_score:.10f} ===")
    if best_seq is not None and best_seq.size <= 2048:
        # Peak shaving
        if time.monotonic() < phase6_end - 60:
            p6s, sc6s = _peak_shave(best_seq, min(phase6_end - 60, time.monotonic() + 80), n_iters=3000)
            _update(p6s, 'peak_shave')

        # Final LP
        if time.monotonic() < phase6_end - 20:
            p6l, sc6l = _lp_aggressive(best_seq, phase6_end, max_solves=30)
            _update(p6l, 'final_lp')

    # === PHASE 7: Final intensive polish (remaining) ===
    phase7_end = deadline
    print(f"\n=== Phase 7: Final polish, current best={best_score:.10f} ===")
    if best_seq is not None and time.monotonic() < phase7_end:
        remaining = phase7_end - time.monotonic()
        if remaining > 50:
            # Very fine Adam
            per = remaining * 0.6
            p7a, sc7a = _adam_logit_power(best_seq, time.monotonic() + per, rng,
                                          lr=0.015, power_start=80.0, power_end=200.0,
                                          no_imp_limit=80)
            _update(p7a, 'final_adam')

        if time.monotonic() < phase7_end - 20 and best_seq.size <= 2048:
            per2 = min(40.0, phase7_end - time.monotonic() - 10)
            if per2 > 15:
                p7s, sc7s = _pair_swap_search(best_seq, time.monotonic() + per2, n_rounds=500)
                _update(p7s, 'final_swap')

        # One more LP pass
        if time.monotonic() < phase7_end - 10 and best_seq.size <= 2048:
            p7l, sc7l = _lp_aggressive(best_seq, phase7_end, max_solves=15)
            _update(p7l, 'very_final_lp')

    # === Final validation ===
    print(f"\n=== Final: best_score={best_score:.10f} (target: 1.5027) ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_cosine_init(1024, 1.5)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_cosine_init(1024, 1.5)
    print(f"FALLBACK n=1024, score={_score_norm(fb):.10f}")
    return [float(x) for x in fb]
