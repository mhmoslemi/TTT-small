"""
Optimized step function search using gradient-based minimax on self-convolution.
Uses log-domain Adam with log-sum-exp smooth max, multiple strategies, and
restarts to flatten the self-convolution peak. Targets score < 1.5027.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 512:
        c = np.convolve(a, a)
    else:
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _convolve_gradient(c_idx, a, n):
    """Gradient of c[c_idx] w.r.t. a. Result is 2*a[c_idx - i] clipped to valid range."""
    k = c_idx
    grad = np.zeros(n)
    # db[k]/da[i] = 2*a[k-i] if 0 <= k-i < n
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad


def _softmax_from_z(z, floor=1e-15):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -80.0, 80.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _adam_minimax(p0, deadline, rng, lr=0.02, top_m=20, beta_init=None,
                  beta_decay=0.97, no_imp_limit=300, lr_decay=0.95,
                  lr_min=0.001, noise_scale=None):
    """Adam optimization in log-domain to minimize max of self-convolution."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    if beta_init is None:
        beta = maxc * 0.05
    else:
        beta = beta_init

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        # Compute smooth max and weights
        c_sorted_idx = np.argsort(c)[::-1]
        top_idx = c_sorted_idx[:top_m]
        c_top = c[top_idx]
        c_max = c_top[0]

        # Log-sum-exp of top values (shifted for numerical stability)
        c_shifted = (c_top - c_max) / beta
        c_shifted = np.minimum(c_shifted, 0.0)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if not np.isfinite(s) or s <= 0:
            e = np.zeros_like(c_top)
            e[0] = 1.0
            s = 1.0
        w_top = e / s

        # Weighted sum of gradients for top lags
        grad_obj = np.zeros(n)
        for i, k in enumerate(top_idx):
            if w_top[i] > 1e-6:
                grad_obj += w_top[i] * _convolve_gradient(k, p, n)

        # Account for normalization: f = max_b / S^2, S = sum(p) = 1 here
        # Since we work in log-domain with softmax, sum is always 1
        # Gradient in z-space: grad_z = p * (grad_obj - p @ grad_obj)
        dot = float(np.dot(p, grad_obj))
        gz = p * (grad_obj - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m_vec.fill(0.0)
            v_vec.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m_vec = b1 * m_vec + (1.0 - b1) * gz
        v_vec = b2 * v_vec + (1.0 - b2) * (gz * gz)
        mh = m_vec / (1.0 - b1 ** t_adam)
        vh = v_vec / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m_vec.fill(0.0)
            v_vec.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -80.0, 80.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(lr_min, lr * lr_decay)
        else:
            z = best_z.copy()
            m_vec.fill(0.0)
            v_vec.fill(0.0)
            t_adam = 0
            continue

        # Decay beta
        if t_adam % 20 == 0:
            beta = max(1e-5, beta * beta_decay)

        if no_imp > no_imp_limit:
            if noise_scale is None:
                if best_score < 1.51:
                    ns = 0.04
                elif best_score < 1.53:
                    ns = 0.08
                else:
                    ns = 0.15
            else:
                ns = noise_scale
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -80.0, 80.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = maxc * 0.05
            m_vec.fill(0.0)
            v_vec.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score

            fw2 = np.fft.rfft(w, N)
            fp2 = np.fft.rfft(p, N)
            # cross_corr[i] = sum_k w[k] * p[k-i] = (w * p^{-})[i+n-1]
            # where p^{-} is p reversed
            fp_rev = np.fft.rfft(p[::-1], N)
            g_fft = np.fft.irfft(fw2 * fp_rev, N)
            grad_obj = 2.0 * g_fft[n-1:n-1+n]

        dot = float(np.dot(p, grad_obj))
        gz = p * (grad_obj - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m_vec.fill(0.0)
            v_vec.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m_vec = b1 * m_vec + (1.0 - b1) * gz
        v_vec = b2 * v_vec + (1.0 - b2) * (gz * gz)
        mh = m_vec / (1.0 - b1 ** t_adam)
        vh = v_vec / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m_vec.fill(0.0)
            v_vec.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -80.0, 80.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 150 == 0:
                    lr = max(0.002, lr * 0.93)
        else:
            z = best_z.copy()
            m_vec.fill(0.0)
            v_vec.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 30 == 0:
            beta = max(1e-5, beta * 0.96)

        if no_imp > no_imp_limit:
            ns = 0.05 if best_score < 1.51 else 0.1
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -80.0, 80.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = maxc * beta_frac
            m_vec.fill(0.0)
            v_vec.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _coordinate_descent(p0, deadline, rng, step_size=0.01, max_iter=10000):
    """Coordinate descent: try adjusting pairs of elements to reduce max convolution."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p = np.maximum(p0, 1e-12)
    p = p / p.sum()
    n = p.size
    c = _full_convolve(p)
    best_score = float(2.0 * n * c.max())
    best_p = p.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Find the lag with maximum value
        k_max = int(np.argmax(c))
        
        # Try to reduce c[k_max] by moving mass from positions that contribute
        # c[k_max] = sum_j p[j] * p[k_max-j]
        # Moving mass from j to j' changes c[k_max]
        
        # Simple approach: try small perturbations
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        delta = 0.001 * p[i]
        p_new = p.copy()
        p_new[i] -= delta
        p_new[j] += delta
        if p_new[i] < 0 or p_new[j] < 0:
            continue
        p_new = p_new / p_new.sum()
        
        c_new = _full_convolve(p_new)
        sc_new = float(2.0 * n * c_new.max())
        
        if sc_new < best_score:
            best_score = sc_new
            best_p = p_new.copy()
            p = p_new
            c = c_new
    
    return best_p, best_score


def _make_initial(n, rng):
    """Create a diverse set of initial sequences."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Linear
    t = np.linspace(0, 1, n)
    for eta in [0.5, 1.0, 1.5, 2.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    # Quadratic
    for a2 in [0.5, 1.0, 1.5, 2.0]:
        a = 1.0 + a2 * (t - 0.5) ** 2
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    # Cosine
    for freq in [1, 2, 3, 4]:
        for amp in [0.3, 0.5, 0.7, 0.9]:
            a = 1.0 + amp * np.cos(2 * np.pi * freq * t)
            a = np.maximum(a, 1e-10)
            inits.append(a / a.sum())
    
    # Gaussian-like
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.3]:
            a = np.exp(-0.5 * ((t - center) / width) ** 2)
            a = np.maximum(a, 1e-10)
            inits.append(a / a.sum())
    
    # Random smooth
    for _ in range(10):
        a = np.abs(rng.normal(size=n)) + 0.1
        # Smooth it
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}] at {time.monotonic()-t_start:.1f}s")

    # Try to load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Started. Parent n={pn}, budget={budget_s}s")

    # Phase 1: Quick evaluation of initial candidates (first 30s)
    phase1_end = t_start + 30.0
    n_base = 1024
    
    candidates = []
    if parent is not None and 0 < pn <= 4096:
        sc = _score_norm(parent)
        candidates.append((sc, parent.copy(), pn, 'parent'))
        print(f"  Parent score: {sc:.10f}")
    
    for n in [512, 1024, 2048]:
        inits = _make_initial(n, rng)
        for i, p in enumerate(inits):
            sc = _score_norm(p)
            candidates.append((sc, p, n, f'init{n}_{i}'))
    
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f}")
    update(candidates[0][1], candidates[0][0], 'init')
    
    if time.monotonic() > phase1_end:
        print("  Phase 1 exceeded time, moving on")

    # Phase 2: Intensive Adam optimization on top candidates (30s - 600s)
    phase2_end = t_start + 600.0
    top_candidates = candidates[:6]
    
    for i, (sc, p, n, label) in enumerate(top_candidates):
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        per_time = min(remaining / (len(top_candidates) - i), 120.0)
        if per_time < 15:
            break
        
        print(f"  Adam v1 on {label} n={n} score={sc:.10f} for {per_time:.0f}s")
        p_b, sc_b = _adam_minimax(p, time.monotonic() + per_time, rng,
                                   lr=0.02, top_m=30, beta_init=sc/(2*n)*0.05,
                                   no_imp_limit=400)
        update(p_b, sc_b, f'adam1_{label}')
        
        if time.monotonic() < phase2_end - 30:
            per_time2 = min(phase2_end - time.monotonic(), 90.0)
            if per_time2 > 15:
                print(f"  Adam v2 on {label} n={n} for {per_time2:.0f}s")
                p_b2, sc_b2 = _adam_minimax_v2(p, time.monotonic() + per_time2, rng,
                                                lr=0.02, top_m=30, beta_frac=0.03,
                                                no_imp_limit=300)
                update(p_b2, sc_b2, f'adam2_{label}')

    print(f"Phase 2 done. Best: {best_score:.10f}")

    # Phase 3: Continue optimization on best sequence (600s - 1200s)
    phase3_end = t_start + 1200.0
    if best_seq is not None:
        n_best = best_seq.size
        print(f"  Phase 3: refining n={n_best}, score={best_score:.10f}")
        
        # Multiple Adam runs with different settings
        settings = [
            {'lr': 0.015, 'top_m': 50, 'no_imp_limit': 500, 'beta_frac': 0.02},
            {'lr': 0.01, 'top_m': 100, 'no_imp_limit': 400, 'beta_frac': 0.01},
            {'lr': 0.03, 'top_m': 20, 'no_imp_limit': 300, 'beta_frac': 0.05},
            {'lr': 0.008, 'top_m': 200, 'no_imp_limit': 600, 'beta_frac': 0.005},
        ]
        
        for i, s in enumerate(settings):
            if time.monotonic() > phase3_end:
                break
            per_time = min(phase3_end - time.monotonic(), 150.0)
            if per_time < 20:
                break
            
            # Start from current best with slight perturbation
            p_start = best_seq.copy()
            z = np.log(np.maximum(p_start, 1e-12))
            z -= z.mean()
            ns = 0.02 * (i + 1)
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -80, 80)
            z -= z.mean()
            p_start = _softmax_from_z(z)
            
            print(f"  Phase 3 run {i}: lr={s['lr']}, top_m={s['top_m']}")
            p_b, sc_b = _adam_minimax_v2(p_start, time.monotonic() + per_time, rng,
                                          lr=s['lr'], top_m=s['top_m'],
                                          beta_frac=s['beta_frac'],
                                          no_imp_limit=s['no_imp_limit'])
            update(p_b, sc_b, f'phase3_{i}')

    # Phase 4: Try different sizes (1200s - 1500s)
    phase4_end = t_start + 1500.0
    if time.monotonic() < phase4_end and best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048]
        elif n_cur <= 1024:
            target_ns = [2048]
        elif n_cur > 2048:
            target_ns = [1024]
        
        for tn in target_ns:
            if time.monotonic() > phase4_end - 60:
                break
            # Interpolate
            x0 = np.linspace(0, 1, n_cur)
            x1 = np.linspace(0, 1, tn)
            p_int = np.interp(x1, x0, best_seq)
            p_int = np.maximum(p_int, 1e-12)
            p_int = p_int / p_int.sum()
            
            sc_int = _score_norm(p_int)
            print(f"  Interp to n={tn}: {sc_int:.10f}")
            update(p_int, sc_int, f'interp_n{tn}')
            
            per = min(60.0, phase4_end - time.monotonic() - 10)
            if per > 20:
                p_e, sc_e = _adam_minimax_v2(p_int, time.monotonic() + per, rng,
                                              lr=0.02, top_m=30, beta_frac=0.03,
                                              no_imp_limit=200)
                update(p_e, sc_e, f'escalate_n{tn}')

    # Phase 5: Final intensive optimization (1500s - deadline)
    if time.monotonic() < deadline - 20 and best_seq is not None:
        n_best = best_seq.size
        remaining = deadline - time.monotonic() - 10
        print(f"  Phase 5: {remaining:.0f}s remaining, n={n_best}, score={best_score:.10f}")
        
        # Final long Adam run
        p_f, sc_f = _adam_minimax_v2(best_seq, time.monotonic() + remaining * 0.6, rng,
                                      lr=0.005, top_m=50, beta_frac=0.01,
                                      no_imp_limit=800)
        update(p_f, sc_f, 'final_adam')
        
        if time.monotonic() < deadline - 10:
            per2 = deadline - time.monotonic() - 5
            if per2 > 10:
                p_f2, sc_f2 = _adam_minimax(best_seq, time.monotonic() + per2, rng,
                                             lr=0.003, top_m=20,
                                             no_imp_limit=500)
                update(p_f2, sc_f2, 'final_adam2')

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = np.ones(1024) / 1024
        
        exact_sc = _score_norm(final)
        print(f"FINAL: local_score={exact_sc:.10f} n={final.size}")
        
        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")
        
        return [float(x) for x in final]

    # Fallback
    fb = np.ones(1024) / 1024
    print(f"FALLBACK: uniform n=1024")
    return [float(x) for x in fb]
