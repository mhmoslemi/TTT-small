def _adam_polish(a_init, deadline, lr=0.05, top_k=0, n_max_iter=50000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-10, 0.01 * max_c)

    if top_k == 0:
        adaptive_k = True
        cur_k = 3
    else:
        adaptive_k = False
        cur_k = top_k

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        use_k = cur_k if adaptive_k else top_k

        c_conv = _full_convolve(a, a)
        cm = float(c_conv.max())

        if use_k > 0 and use_k < len(c_conv):
            idx = np.argsort(c_conv)[-use_k:]
            w = np.zeros_like(c_conv)
            c_top = c_conv[idx]
            c_norm = (c_top - c_top.max()) / beta
            e = np.exp(c_norm)
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w[idx[0]] = 1.0
            else:
                w[idx] = e / s
        else:
            c_shifted = (c_conv - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                idx = int(np.argmax(c_conv))
                w = np.zeros_like(c_conv)
                w[idx] = 1.0
            else:
                w = e / s

        W = float(np.dot(w, c_conv))
        a_rev = a[::-1]
        w_rev = w[::-1]
        w_conv = _full_convolve(w_rev, a)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            grad = np.zeros_like(z)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 100 == 0:
            beta = max(1e-14, beta * 0.85)
            if adaptive_k and t % 500 == 0:
                cur_k = min(cur_k + 2, 50)
        if t % 2000 == 0:
            if best_score < 1.51:
                lr = max(0.002, lr * 0.95)

    return best_a, best_score


def _kkt_refine(a_init, deadline, max_iter=2000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        cm = float(c.max())
        threshold = cm * (1.0 - 1e-4)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            break

        h = np.zeros(n)
        for k_idx in active:
            k_val = int(k_idx)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                h[lo:hi + 1] += a[k_val - hi:k_val - lo + 1][::-1]
        h /= len(active)
        W = float(np.dot(h, a))

        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()
        spread = float(c[active].max() - c[active].min()) + 1e-15
        step = min(0.02 * spread / (cm + 1e-15), 0.03)
        denom = np.abs(h - W).max() + 1e-15
        z = z - step * (h - W) / denom
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a_new = _softmax_fn(z)

        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            step *= 0.5
            z = np.log(np.maximum(a, 1e-15))
            z = z - z.mean()
            z = z - step * (h - W) / denom
            z = np.clip(z, -80, 80)
            z = z - z.mean()
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new

        if it > 50 and abs(best_score - sc) < 1e-12:
            break

    return best_a, best_score


def _make_exp_profile(n, alpha):
    r = 1.0 - alpha
    a = r ** np.arange(n)
    return a / a.sum()


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _lp_optimize_refined(a_init, deadline, max_iters=100):
    """LP with adaptive active set and constraint generation."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, _score_norm_fn(a)
    a = a / s0

    best_a = a.copy()
    best_score = _score_norm_fn(a)
    t_mix = 0.4

    for it in range(max_iters):
        if time.monotonic() > deadline - 10:
            break

        c = _full_convolve(a, a)
        M = float(c.max())
        score = 2.0 * n * M
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if it % 20 == 0:
                print(f"    Refined LP iter {it}: score={score:.10f}")

        # Progressive tightening of active set
        if it < 10:
            thresh_frac = 0.95
        elif it < 30:
            thresh_frac = 0.98
        else:
            thresh_frac = 0.995

        active = c >= M * thresh_frac
        if np.sum(active) < 3:
            top_k = np.argsort(c)[-max(5, n // 10):]
            active = np.zeros(len(c), dtype=bool)
            active[top_k] = True

        x = _lp_direction(a, M, active)
        if x is None:
            break

        x = np.maximum(x, 0)
        sx = x.sum()
        if sx < 1e-15:
            break
        x = x / sx

        diff = np.max(np.abs(x - a))
        if diff < 1e-9:
            break

        a_new = (1.0 - t_mix) * a + t_mix * x
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn
        new_score = _score_norm_fn(a_new)

        if new_score <= score + 1e-12:
            a = a_new
            t_mix = min(0.8, t_mix * 1.02)
        else:
            t_mix = max(0.05, t_mix * 0.7)
            a_new = (1.0 - t_mix) * a + t_mix * x
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn > 1e-15:
                a_new = a_new / sn
                if _score_norm_fn(a_new) < score:
                    a = a_new

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    global height_sequence_1

    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Phase 0: Initialization ---
    print("=== Phase 0: Initialization ===")
    n_primary = 4096

    # Try parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm_fn(parent)
                update_best(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception:
        pass

    # Generate initial profiles
    profiles = []
    for alpha in [0.001, 0.002, 0.005, 0.01]:
        profiles.append(_make_exp_profile(n_primary, alpha))
    for slope in [-1.5, -1.0, -0.5, 0.5, 1.0, 1.5, 2.0, 3.0]:
        profiles.append(_make_linear_profile(n_primary, slope))

    # Add parent interpolated if different length
    if parent is not None and len(parent) != n_primary:
        p_interp = _interp_profile(parent, n_primary)
        profiles.append(p_interp)
        sc = _score_norm_fn(p_interp)
        update_best(p_interp, sc, "parent_interp")
    elif parent is not None:
        profiles.append(parent.copy())

    # Score all profiles, pick best few
    scored_profiles = []
    for idx, p in enumerate(profiles):
        sc = _score_norm_fn(p)
        scored_profiles.append((sc, p, f"prof{idx}"))
        update_best(p, sc, f"prof{idx}")

    scored_profiles.sort(key=lambda x: x[0])
    print(f"  Phase 0 done: best={best_score:.10f}, {len(scored_profiles)} profiles")

    # --- Phase 1: LP-based optimization (main phase) ---
    print("=== Phase 1: LP-based optimization ===")
    phase1_end = min(deadline, t_start + 500.0)
    count = 0

    for sc, p, label in scored_profiles[:3]:
        if time.monotonic() > phase1_end - 30:
            break
        if sc > best_score * 1.05:
            break
        remaining = phase1_end - time.monotonic()
        if remaining < 30:
            break
        print(f"  Running LP on {label} (n={len(p)}), budget={remaining:.0f}s")
        p_best, sc_best = _lp_optimize(p, phase1_end, max_iters=80)
        update_best(p_best, sc_best, f"lp_{label}")
        count += 1

    print(f"  Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Refined LP with tighter active sets ---
    print("=== Phase 2: Refined LP ===")
    phase2_end = min(deadline, t_start + 800.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        remaining = phase2_end - time.monotonic()
        if remaining > 30:
            print(f"  Refined LP on best, budget={remaining:.0f}s")
            p_best, sc_best = _lp_optimize_refined(best_seq, phase2_end, max_iters=80)
            update_best(p_best, sc_best, "lp_refined")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Adam polishing ---
    print("=== Phase 3: Adam polishing ===")
    phase3_end = min(deadline, t_start + 1200.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        remaining = phase3_end - time.monotonic()
        if remaining > 30:
            dl = min(phase3_end, time.monotonic() + remaining)
            print(f"  Adam round 1, budget={remaining:.0f}s")
            p_best, sc_best = _adam_polish(best_seq, dl, lr=0.04, top_k=0, n_max_iter=60000)
            update_best(p_best, sc_best, "adam1")

        if time.monotonic() < phase3_end - 10:
            remaining2 = phase3_end - time.monotonic()
            if remaining2 > 20:
                dl2 = min(phase3_end, time.monotonic() + remaining2)
                print(f"  Adam round 2 (top-k), budget={remaining2:.0f}s")
                p_best, sc_best = _adam_polish(best_seq, dl2, lr=0.02, top_k=8, n_max_iter=40000)
                update_best(p_best, sc_best, "adam2")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: KKT refinement ---
    print("=== Phase 4: KKT refinement ===")
    phase4_end = min(deadline, t_start + 1450.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        remaining = phase4_end - time.monotonic()
        if remaining > 20:
            dl = min(phase4_end, time.monotonic() + remaining)
            print(f"  KKT refine, budget={remaining:.0f}s")
            p_best, sc_best = _kkt_refine(best_seq, dl, max_iter=3000)
            update_best(p_best, sc_best, "kkt")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: LP again on refined sequence ---
    print("=== Phase 5: Final LP pass ===")
    phase5_end = min(deadline, t_start + 1650.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        remaining = phase5_end - time.monotonic()
        if remaining > 30:
            dl = min(phase5_end, time.monotonic() + remaining)
            print(f"  Final LP, budget={remaining:.0f}s")
            p_best, sc_best = _lp_optimize_refined(best_seq, dl, max_iters=50)
            update_best(p_best, sc_best, "final_lp")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Try different lengths ---
    print("=== Phase 6: Multi-length exploration ===")
    phase6_end = min(deadline, t_start + 1780.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        n_curr = len(best_seq)
        target_ns = []
        if n_curr != 2048:
            target_ns.append(2048)
        if n_curr != 8192:
            target_ns.append(8192)

        for tn in target_ns:
            if time.monotonic() > phase6_end - 20:
                break
            p_interp = _interp_profile(best_seq, tn)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{tn}")

            remaining = phase6_end - time.monotonic()
            if remaining > 40:
                # Quick LP at this length
                if tn <= 4096:
                    dl = min(phase6_end - 5, time.monotonic() + min(80, remaining * 0.7))
                    p_best, sc_best = _lp_optimize(p_interp, dl, max_iters=30)
                    update_best(p_best, sc_best, f"lp_n{tn}")
                else:
                    # For larger n, use Adam
                    dl = min(phase6_end - 5, time.monotonic() + min(60, remaining * 0.7))
                    p_best, sc_best = _adam_polish(p_interp, dl, lr=0.03, top_k=5, n_max_iter=20000)
                    update_best(p_best, sc_best, f"adam_n{tn}")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final polish ---
    print("=== Phase 7: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic() - 5
        if remaining > 15:
            dl = min(deadline - 5, time.monotonic() + min(30.0, remaining * 0.6))
            p_best, sc_best = _adam_polish(best_seq, dl, lr=0.01, top_k=15, n_max_iter=20000)
            update_best(p_best, sc_best, "final_adam")

        if time.monotonic() < deadline - 10:
            remaining2 = deadline - time.monotonic() - 10
            if remaining2 > 10:
                dl2 = min(deadline - 5, time.monotonic() + min(20.0, remaining2))
                p_best, sc_best = _kkt_refine(best_seq, dl2, max_iter=1000)
                update_best(p_best, sc_best, "final_kkt")

    # --- Final validation ---
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = _make_exp_profile(2048, 0.003)
            final_score = _score_fn(final_seq)
        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")

        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 16384:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  evaluate_sequence check: {e}")

        return [float(x) for x in final_seq]

    fallback = _make_exp_profile(2048, 0.003)
    print(f"  FALLBACK: n=2048, score={_score_norm_fn(fallback):.10f}")
    return [float(x) for x in fallback]
