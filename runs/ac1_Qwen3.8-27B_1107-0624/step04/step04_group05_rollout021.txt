"""
Optimized step function search using aggressive LP best-response refinement,
adaptive smooth-max gradient in logit space, and spectral domain nudging.
Targets score < 1.5027 by combining multiple optimization strategies with
careful time management and adaptive step sizing.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        c = np.convolve(a, a)
    else:
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        c = np.maximum(c, 0.0)
    return c


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    mval = float(c.max())
    if beta < 1e-13:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-9
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 0.0))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-14):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_solve(a, M):
    n = len(a)
    A_ub = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)

    M = float(c.max())
    t = t_init
    n_solves = 0
    stall_count = 0
    rng = np.random.default_rng(12345)

        s_b = float(b.sum())

        b_norm = b / s_b

        best_sc = current_score
        best_t = 0.0

            sc_try = _score_norm(a_try)

            m.fill(0.0)
            v.fill(0.0)

        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

            m.fill(0.0)
            v.fill(0.0)

            if score < best_score - 1e-14:

                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    current_lr = max(0.002, current_lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            continue

        # Adaptive beta

            if c.size >= 2:

                gap = top5[0] - top5[-1]

            current_lr *= lr_decay

        if no_imp > no_imp_limit:
            if noise_schedule is None:

                else:
                    ns = 0.15
            else:
                ns = noise_schedule
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-4, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_imp = 0

    return best_p, best_score


def _spectral_nudge(a, deadline, rng, n_iters=200):
    """Nudge the sequence in frequency domain to flatten the power spectrum."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    best_p = a.copy()
    best_score = _score_norm(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        a_pad[:n] = a
        A = np.fft.rfft(a_pad)

        # Compute power spectrum of the convolution (which is A^2)
        P = A * A

        # IDFT to get the convolution
        c_full = np.fft.irfft(P, N)[:2*n-1]
        c_full = np.maximum(c_full, 0)
        mean_c = c_full.mean()
        peak_c = c_full.max()

        # Use a soft threshold on the magnitude of A
        mag = np.abs(A)
        # Scale down components that contribute most to the peak
        # Simple approach: apply a power compression to the magnitude
        compression = 0.99  # very gentle
        mag_new = mag ** (1 - (1 - compression))  # this is just mag, let me use a different approach

        # Better: reduce the peak of |A|^2 by scaling down the largest components

        if max_mag2 > 0:

            threshold = 0.5 * max_mag2

        a_new_pad = np.fft.irfft(A_mod, N)
        a_new = np.abs(a_new_pad[:n])  # take absolute value to ensure non-negativity

        # Actually, a better approach: keep the phase but adjust magnitude
        # Let's try a different spectral nudge
        # Subtract a small multiple of the "excess" from the peak lags
        p = a.copy()
        p = p / p.sum()

        # Gradient step in time domain using the convolution error
        c = _full_convolve(p)
        excess = np.maximum(c - c.mean(), 0)
        # Gradient of max(c) w.r.t. p is the cross-correlation of the peak weights with p
        _, w = _smooth_max_and_weights(c, 0.01 * c.max())
        grad = _conv_cross_corr(w, p)
        # Negative gradient step
        step = 0.05
        p_new = p - step * grad
        p_new = np.maximum(p_new, 1e-12)
        p_new = p_new / p_new.sum()
        sc = _score_norm(p_new)

        if sc < best_score:
            best_score = sc
            best_p = p_new.copy()
            a = p_new
        else:
            # Try smaller step
            p_new2 = p - 0.01 * grad
            p_new2 = np.maximum(p_new2, 1e-12)
            p_new2 = p_new2 / p_new2.sum()
            sc2 = _score_norm(p_new2)
            if sc2 < best_score:
                best_score = sc2
                best_p = p_new2.copy()
                a = p_new2
            else:
                break

    return best_p, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p0)
    return _normalize_nonneg(p_new)


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-15, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _make_hann(n):
    t = np.arange(n)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * t / (n - 1)))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_flat_cos(n, k, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * k * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_polynomial(n, coeffs):
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for i, c in enumerate(coeffs):
        a += c * t**i
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}]")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Started. Parent n={pn}, budget={budget_s}s")

    # Phase 1: Initialization (10s)
    phase1_end = t_start + 10.0
    candidates = []

    if parent is not None:
        sc = _score_norm(parent)
        candidates.append((sc, parent.copy(), pn, 'parent'))
        _update(parent.copy(), sc, 'parent_init')

    # Generate diverse initializations
    n_base = 1024
    init_profiles = [
        ('hann', _make_hann(n_base)),
        ('flat_cos1', _make_flat_cos(n_base, 1, 0.5)),
        ('flat_cos1_0.3', _make_flat_cos(n_base, 1, 0.3)),
        ('flat_cos1_0.7', _make_flat_cos(n_base, 1, 0.7)),
        ('flat_cos2', _make_flat_cos(n_base, 2, 0.3)),
        ('flat_cos1+2', _make_flat_cos(n_base, 1, 0.3) * 0.5 + _make_flat_cos(n_base, 2, 0.2) * 0.5),
        ('poly2', _make_polynomial(n_base, [1.0, 0.0, -0.3])),
        ('poly3', _make_polynomial(n_base, [1.0, 0.5, -0.5, 0.1])),
        ('poly4', _make_polynomial(n_base, [1.0, 1.0, -1.5, 0.5, 0.1])),
    ]
    for label, p in init_profiles:
        sc = _score_norm(p)
        candidates.append((sc, p, n_base, label))

    # Interpolate parent to other lengths
    if parent is not None and pn <= 2048:
        for n_try in [512, 1024, 2048]:
            if n_try != pn:
                pi = _interp_profile(parent, n_try)
                if pi is not None:
                    sc = _score_norm(pi)
                    candidates.append((sc, pi, n_try, f'parent_n{n_try}'))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f}")
    if candidates[0][0] < best_score:
        _update(candidates[0][1], candidates[0][0], 'best_init')

    # Phase 2: Intensive LP refinement on best (10s - 500s)
    phase2_end = t_start + 500.0
    if best_seq is not None:
        n_cur = best_seq.size
        print(f"Phase 2: LP refine n={n_cur}, score={best_score:.10f}")
        p_lp, sc_lp = _lp_refine_intensive(
            best_seq, min(phase2_end, deadline),
            t_init=0.4, max_solves=500, perturbation_interval=50
        )
        _update(p_lp, sc_lp, 'LP_phase2')

    # Phase 3: Adam polish (500s - 900s)
    phase3_end = t_start + 900.0
    if best_seq is not None and time.monotonic() < phase3_end:
        per = min(phase3_end - time.monotonic() - 10, 350.0)
        if per > 20:
            print(f"Phase 3: Adam polish, time={per:.0f}s")
            p_ad, sc_ad = _adam_logit_improved(
                best_seq, time.monotonic() + per, rng,
                lr=0.04, beta_frac=0.07, beta_decay=0.95,
                no_imp_limit=300, lr_decay=0.997
            )
            _update(p_ad, sc_ad, 'adam_phase3')

    # Phase 4: Second LP round (900s - 1200s)
    phase4_end = t_start + 1200.0
    if best_seq is not None and time.monotonic() < phase4_end:
        print(f"Phase 4: Second LP, score={best_score:.10f}")
        per = min(phase4_end - time.monotonic() - 10, 280.0)
        if per > 20:
            p_lp2, sc_lp2 = _lp_refine_intensive(
                best_seq, time.monotonic() + per,
                t_init=0.2, max_solves=300, perturbation_interval=60
            )
            _update(p_lp2, sc_lp2, 'LP_phase4')

    # Phase 5: Multi-scale escalation (1200s - 1500s)
    phase5_end = t_start + 1500.0
    if best_seq is not None and time.monotonic() < phase5_end:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048]
        elif n_cur <= 1024:
            target_ns = [2048]
        elif n_cur <= 2048:
            target_ns = [4096]

        for tn in target_ns:
            if time.monotonic() > phase5_end - 60:
                break
            p_int = _interp_profile(best_seq, tn)
            if p_int is None:
                continue
            sc_int = _score_norm(p_int)
            print(f"  Escalate to n={tn}: {sc_int:.10f}")
            _update(p_int, sc_int, f'interp_n{tn}')

            per = min(100.0, phase5_end - time.monotonic() - 20)
            if per > 30:
                # Quick LP at new scale
                if tn <= 2048:
                    p_e, sc_e = _lp_refine_intensive(
                        p_int, time.monotonic() + min(60, per * 0.6),
                        t_init=0.3, max_solves=50, perturbation_interval=100
                    )
                    _update(p_e, sc_e, f'LP_n{tn}')

                if time.monotonic() < phase5_end - 10:
                    per2 = phase5_end - time.monotonic() - 5
                    if per2 > 15:
                        p_a, sc_a = _adam_logit_improved(
                            best_seq, time.monotonic() + min(40, per2), rng,
                            lr=0.03, beta_frac=0.05, beta_decay=0.95,
                            no_imp_limit=200
                        )
                        _update(p_a, sc_a, f'adam_n{tn}')

    # Phase 6: Multi-restart at best length (1500s - 1750s)
    phase6_end = deadline - 30.0
    if best_seq is not None and time.monotonic() < phase6_end:
        n_best = best_seq.size
        n_restarts = 4
        per_r = max(20.0, (phase6_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 15:
                break
            per_adam = min(per_r * 0.5, phase6_end - time.monotonic() - 20)
            if per_adam < 10:
                break

            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            ns = 0.02 + 0.03 * i
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -60, 60)
            z -= z.mean()
            p_s = _softmax_from_z(z)

            p_r, sc_r = _adam_logit_improved(
                p_s, time.monotonic() + per_adam, rng,
                lr=0.03, beta_frac=0.05, beta_decay=0.95,
                no_imp_limit=int(0.3 * min(n_best, 500))
            )
            _update(p_r, sc_r, f'restart{i}_n{n_best}')

            # Quick LP polish after restart
            if best_seq.size <= 2048 and time.monotonic() < phase6_end - 10:
                per_lp = min(25.0, phase6_end - time.monotonic() - 5)
                if per_lp > 8:
                    p_lp3, sc_lp3 = _lp_refine_intensive(
                        best_seq, time.monotonic() + per_lp,
                        t_init=0.15, max_solves=20, perturbation_interval=0
                    )
                    _update(p_lp3, sc_lp3, f'LP_restart{i}')

    # Phase 7: Final squeeze (last 30s)
    if time.monotonic() < deadline - 3 and best_seq is not None:
        per = deadline - time.monotonic() - 3
        if per > 10 and best_seq.size <= 2048:
            p_f, sc_f = _lp_refine_intensive(
                best_seq, time.monotonic() + per * 0.5,
                t_init=0.1, max_solves=20, perturbation_interval=0
            )
            _update(p_f, sc_f, 'final_LP')

        if time.monotonic() < deadline - 3:
            per2 = deadline - time.monotonic() - 3
            if per2 > 5:
                p_f2, sc_f2 = _adam_logit_improved(
                    best_seq, time.monotonic() + per2, rng,
                    lr=0.01, beta_frac=0.03, beta_decay=0.97,
                    no_imp_limit=50
                )
                _update(p_f2, sc_f2, 'final_adam')

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_hann(1024)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    # Fallback
    fb = _make_hann(1024)
    print(f"FALLBACK n=1024, score={_score_norm(fb):.10f}")
    return [float(x) for x in fb]
