    # For m = n-1+i: sum_k grad_c[k] * a[n-1-(n-1+i)+k] = sum_k grad_c[k] * a[k-i]
    # Yes! So h_i = (grad_c * a_flip)[n-1+i]
    # And this already accounts for the "both sides" contribution because the convolution 
    # naturally sums over all valid k.
    # So dM/da_i = h_i (not 2*h_i). Let me double-check with a simple example.
    # a = [1, 2], n=2. (a*a) = [1, 4, 4]. 
    # d(a*a)[1]/da_0 = 2*1*2 = 4? Let's see: (a*a)[1] = a_0*a_1 + a_1*a_0 = 4.
    # d/da_0 = a_1 + a_1 = 2*a_1 = 4. Yes.
    # Using convolution: a_flip = [2, 1]. grad_c = [1, 1, 1] (uniform for testing).
    # (grad_c * a_flip) = [2, 3, 3, 1]. Slice [n-1:2n-1] = [1:3] = [3, 3].
    # Hmm, that gives h = [3, 3], but the actual gradient should be:
    # d/da_0 sum_k grad_c[k]*(a*a)[k] = grad_c[0]*a_0 + grad_c[1]*2*a_1 + grad_c[2]*0 = 1*1 + 1*4 = 5
    # d/da_1 = grad_c[0]*0 + grad_c[1]*2*a_0 + grad_c[2]*a_1 = 0 + 2 + 2 = 4
    # Using my formula: h_0 = sum_k grad_c[k] * a[k-0] = grad_c[0]*a_0 + grad_c[1]*a_1 + grad_c[2]*a_2(invalid) = 1+2 = 3
    # h_1 = sum_k grad_c[k] * a[k-1] = grad_c[1]*a_0 + grad_c[2]*a_1 = 1+2 = 3
    # But actual gradient is [5, 4]. So my formula is wrong!
    # The issue: d(a*a)[k]/da_i is NOT just a[k-i]. It's a[k-i] + a[k-i] = 2*a[k-i] when i != k-i,
    # and a[k] when i = k-i.
    # More precisely: d(a*a)[k]/da_i = a[k-i] (if 0<=k-i<n) + a[k-i] (if 0<=k-i<n and k-i != i)
    # = 2*a[k-i] if i != k-i and 0<=k-i<n
    # = a[k] if i = k-i (i.e., 2i = k)
    # So the gradient is NOT simply a convolution. It's a "dilated" convolution.
    # Let me redo: d/da_i sum_k grad_c[k]*(a*a)[k] = sum_k grad_c[k] * (2*a[k-i] if i!=k-i, a[k] if i=k/2)
    # This is messy. Let me just use the correct formula:
    # d(a*a)[k]/da_i = a[k-i] + (a[k-i] if i != k-i else 0) ... no that's still wrong.
    # Let me be very explicit. (a*a)[k] = sum_j a_j * a[k-j] where the sum is over j such that 0<=j<n and 0<=k-j<n.
    # d/da_i: the term a_j * a[k-j] is differentiated w.r.t. a_i.
    # This gives a nonzero contribution when j = i (giving a[k-i]) or when k-j = i, i.e., j = k-i (giving a[k-i] = a[i]... wait)
    # When j = i: contribution is a[k-i] (derivative of a_j w.r.t. a_i times a[k-j] = a[k-i])
    # When j = k-i: contribution is a_j = a[k-i] (derivative of a[k-j] w.r.t. a_i, which is 1 when k-j = i, times a_j = a[k-i])
    # If i = k-i (i.e., 2i = k): both conditions are the same, so contribution is 2*a[i] but we only count it once? 
    # No: if 2i = k, then j = i = k-i, so it's the same term a_i * a_i. d/da_i of a_i^2 = 2*a_i.
    # So: d(a*a)[k]/da_i = a[k-i] (from j=i) + a[k-i] (from j=k-i) if i != k-i
    #                             = 2*a[i] if i = k-i (= k/2)
    # In all cases, it's: a[k-i] + a[k-i] = 2*a[k-i] when i != k-i, and 2*a[i] = 2*a[k/2] when i = k/2.
    # But a[k-i] when i = k-i is a[i] = a[k/2], so it's always 2*a[k-i]!
    # Wait: when i = k-i, a[k-i] = a[i], and the derivative is 2*a[i] = 2*a[k-i]. ✓
    # When i != k-i, the derivative is a[k-i] + a[k-i] = 2*a[k-i]. ✓
    # So d(a*a)[k]/da_i = 2*a[k-i] (when k-i is valid, i.e., 0 <= k-i < n), and 0 otherwise.
    # 
    # Great! So d/da_i sum_k grad_c[k]*(a*a)[k] = 2 * sum_k grad_c[k] * a[k-i] (where k-i is valid)
    # = 2 * (grad_c * a_flip)[n-1+i] (as I computed before)
    #
    # Let me re-verify: a = [1, 2], n=2, grad_c = [1, 1, 1]
    # h_0 = sum_k grad_c[k]*a[k-0] = 1*1 + 1*2 + 0 = 3
    # h_1 = sum_k grad_c[k]*a[k-1] = 0 + 1*1 + 1*2 = 3
    # dM/da = 2*h = [6, 6]
    # But I computed the actual gradient as [5, 4] above. Let me recheck.
    # (a*a)[0] = a_0^2 = 1, (a*a)[1] = 2*a_0*a_1 = 4, (a*a)[2] = a_1^2 = 4
    # sum grad_c[k]*(a*a)[k] = 1 + 4 + 4 = 9
    # d/da_0: (a*a)[0]: d/da_0 of a_0^2 = 2*a_0 = 2. (a*a)[1]: d/da_0 of 2*a_0*a_1 = 2*a_1 = 4. (a*a)[2]: 0.
    # Total: 1*2 + 1*4 + 1*0 = 6. ✓ (matches 2*h_0 = 6)
    # d/da_1: (a*a)[0]: 0. (a*a)[1]: d/da_1 of 2*a_0*a_1 = 2*a_0 = 2. (a*a)[2]: d/da_1 of a_1^2 = 2*a_1 = 4.
    # Total: 1*0 + 1*2 + 1*4 = 6. ✓ (matches 2*h_1 = 6)
    # 
    # So my earlier computation of [5, 4] was wrong! The correct gradient is [6, 6].
    # Great, so the formula is: dM/da_i = 2 * sum_k grad_c[k] * a[k-i] = 2 * h_i where h is the convolution.
    
    # Now for the logit parameterization: a = softmax(z), so da_i/dz_i = a_i(1-a_i), da_i/dz_j = -a_i*a_j (j!=i)
    # dM/dz = sum_i (dM/da_i) * da_i/dz ... this gets complicated.
    # Actually, in the logit space, the gradient is: dM/dz_i = dM/da_i * a_i - a_i * sum_j dM/da_j * a_j
    # = a_i * (dM/da_i - sum_j dM/da_j * a_j)
    # = a_i * (2*h_i - 2*sum_j h_j * a_j)
    # = 2*a_i * (h_i - W) where W = sum_j h_j * a_j
    
    grad_a = 2.0 * h
    W = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - W)
    if not np.all(np.isfinite(grad_z)):
        return np.zeros_like(z)
    return grad_z

def _power_smooth_obj(z, p, n):
    """Power-smoothed objective in logit space."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    M, _ = _power_smooth_max(c, p)
    return 2.0 * n * M

def _run_nesterov(a_init, deadline, p_start=100, p_end=5000, n_iter=50000, lr=0.01):
    """Nesterov-accelerated gradient descent with power smoothing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    # Nesterov state
    y = z.copy()
    t = 1.0
    
    # Momentum
    v = np.zeros(n)
    beta = 0.9
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        # Adaptive p
        progress = min(1.0, it / n_iter)
        p = p_start * (p_end / p_start) ** progress
        
        a_y = _softmax_fn(y)
        c = _full_convolve(a_y, a_y)
        M, grad_c = _power_smooth_max(c, p)
        a_rev = a_y[::-1]
        w_conv = _full_convolve(grad_c, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad_a = 2.0 * h
        W = float(np.dot(grad_a, a_y))
        grad_z = a_y * (grad_a - W)
        
        if not np.all(np.isfinite(grad_z)):
            continue
        
        # Momentum update
        v = beta * v + (1 - beta) * grad_z
        y = y - lr * v
        y = np.clip(y, -50, 50)
        y = y - y.mean()
        
        # Nesterov extrapolation
        t_new = 1 + (t - 1) * beta  # simplified
        z_new = y + ((t - 1) / t_new) * (y - z)
        z = z_new
        t = t_new
        
        # Evaluate
        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score

def _run_lbfgs_power(a_init, deadline, p_start=50, p_end=50000, n_levels=8):
    """L-BFGS with power smoothing continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if max_c0 < 1e-15:
        return best_a, best_score
    
    # Scale p based on the magnitude of c
    p_factors = np.linspace(np.log(p_start), np.log(p_end), n_levels)
    max_wall = max(1.0, deadline - time.monotonic())
    per_level = max_wall / n_levels
    
    for li in range(n_levels):
        if time.monotonic() > deadline:
            break
        p = float(np.exp(p_factors[li]))
        level_dl = min(deadline, time.monotonic() + per_level)
        
        try:
            result = minimize(
                _power_smooth_obj, z, args=(p, n),
                method='L-BFGS-B', jac=_power_smooth_grad_z,
                options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 2000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                if np.all(np.isfinite(a_new)):
                    sc = _score_norm_fn(a_new)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
                        z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score

def _topk_lbfgs(a_init, deadline, top_k=10):
    """L-BFGS with top-k active set (near-exact max gradient)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    def _topk_obj(z, k, n):
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        idx = np.argsort(c)[-k:]
        c_top = c[idx]
        M = c_top.max()
        log_s = float(np.log(np.sum(np.exp((c_top - M) / 0.01))))
        return 2.0 * n * (M + 0.01 * log_s)
    
    def _topk_grad(z, k, n):
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        idx = np.argsort(c)[-k:]
        c_top = c[idx]
        M = c_top.max()
        e = np.exp((c_top - M) / 0.01)
        w_top = e / e.sum()
        w = np.zeros(len(c))
        w[idx] = w_top
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad_a = 2.0 * h
        W = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - W)
        if not np.all(np.isfinite(grad_z)):
            return np.zeros_like(z)
        return grad_z
    
    try:
        result = minimize(
            _topk_obj, z, args=(top_k, n),
            method='L-BFGS-B', jac=_topk_grad,
            options={'maxiter': 1000, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxls': 50, 'maxfun': 5000}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            a_new = _softmax_fn(result.x)
            if np.all(np.isfinite(a_new)):
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
    except Exception:
        pass
    
    return best_a, best_score

def _active_set_refine(a_init, deadline, tol=0.001, max_iter=5000):
    """Refine using explicit active set identification and balanced gradients."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Active set: within tol * cm of max
        threshold = cm * (1.0 - tol)
        active = np.where(c >= threshold)[0]
        
        if len(active) < 2:
            # Add neighbors
            active = np.where(c >= cm * 0.99)[0]
        
        # Compute balanced gradient using active set
        # h_i = (1/|A|) * sum_{k in A} a[k-i] (for valid k-i)
        # Use convolution: create indicator vector of active set
        w = np.zeros(len(c))
        w[active] = 1.0 / len(active)
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        
        W = float(np.dot(h, a))
        
        # Direction: reduce a where h > W, increase where h < W
        diff = h - W
        max_diff = float(np.abs(diff).max())
        if max_diff < 1e-15:
            break  # At equilibrium
        
        # Step size
        step = 0.02 * max_diff / (cm + 1e-15)
        step = min(step, 0.1)
        
        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()
        z = z - step * diff / (max_diff + 1e-15)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a_new = _softmax_fn(z)
        
        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Halve step
            z = np.log(np.maximum(a, 1e-15))
            z = z - z.mean()
            z = z - 0.5 * step * diff / (max_diff + 1e-15)
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
            else:
                # Try smaller
                z = np.log(np.maximum(a, 1e-15))
                z = z - z.mean()
                z = z - 0.1 * step * diff / (max_diff + 1e-15)
                z = np.clip(z, -60, 60)
                z = z - z.mean()
                a_new = _softmax_fn(z)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new
                else:
                    break
        
        if it > 50 and best_score - (sc if np.isfinite(sc) else 0) < 1e-12:
            break
    
    return best_a, best_score

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

def _adam_with_active_set(a_init, deadline, lr=0.03, max_iter=80000):
    """Adam with adaptive active set (threshold-based, not top-k)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    # Adaptive threshold for active set
    tol = 0.01  # start: active if c >= max * (1 - tol)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Active set with adaptive tolerance
        threshold = cm * (1.0 - tol)
        active = np.where(c >= threshold)[0]
        k = len(active)
        
        # Compute gradient using active set weights
        # Use softmax over active set values
        c_act = c[active]
        exp_vals = np.exp((c_act - c_act.max()) / (0.005 * cm + 1e-15))
        w_act = exp_vals / exp_vals.sum()
        w = np.zeros(len(c))
        w[active] = w_act
        
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad_a = 2.0 * h
        W = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - W)
        
        if not np.all(np.isfinite(grad_z)):
            grad_z = np.zeros_like(z)
        
        # Adam update
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        
        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Adapt tolerance: tighten over time
        if t % 200 == 0:
            tol = max(0.0001, tol * 0.9)
        if t % 1000 == 0:
            lr = max(0.002, lr * 0.95)
    
    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining power-smoothed Nesterov, active-set Adam,
    top-k L-BFGS, and active-set refinement. Multi-scale with n in {2048, 4096, 8192}.
    """
    global height_sequence_1
    
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")
    
    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm_fn(parent)
                update_best(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent: {e}")
    
    # Multi-scale approach
    scales = [2048, 4096, 8192]
    
    # Start at each scale
    for scale_n in scales:
        if time.monotonic() > deadline - 60:
            break
        print(f"\n=== Scale n={scale_n} ===")
        
        # Get initial point for this scale
        if parent is not None:
            init_p = _interp_profile(parent, scale_n)
        else:
            # Fallback: exponential profile
            r = 0.998
            init_p = r ** np.arange(scale_n)
            init_p = init_p / init_p.sum()
        
        sc_init = _score_norm_fn(init_p)
        update_best(init_p, sc_init, f"init_n{scale_n}")
        
        remaining = deadline - time.monotonic()
        if remaining < 30:
            continue
        
        # Phase A: Power-smoothed L-BFGS (coarse to fine)
        time_budget_a = min(120.0, remaining * 0.25)
        dl_a = min(deadline - 60, time.monotonic() + time_budget_a)
        print(f"  A) Power L-BFGS: budget={time_budget_a:.1f}s")
        p_a, sc_a = _run_lbfgs_power(init_p, dl_a, p_start=20, p_end=20000, n_levels=6)
        update_best(p_a, sc_a, f"power_lbfgs_n{scale_n}")
        
        if time.monotonic() > deadline - 60:
            break
        
        # Phase B: Active-set Adam
        time_budget_b = min(200.0, (deadline - time.monotonic()) * 0.3)
        dl_b = min(deadline - 60, time.monotonic() + time_budget_b)
        print(f"  B) Active Adam: budget={time_budget_b:.1f}s")
        p_b, sc_b = _adam_with_active_set(p_a, dl_b, lr=0.03, max_iter=60000)
        update_best(p_b, sc_b, f"adam_n{scale_n}")
        
        if time.monotonic() > deadline - 40:
            break
        
        # Phase C: Top-k L-BFGS (near-exact)
        time_budget_c = min(60.0, (deadline - time.monotonic()) * 0.2)
        dl_c = min(deadline - 30, time.monotonic() + time_budget_c)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_grad_a_from_weights(w, a, n):
    """Compute gradient of weighted sum of convolution w.r.t. a.
    Returns 2*h where h_i = sum_k w[k]*a[k-i]."""
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    return 2.0 * h

    """Convert gradient w.r.t. a to gradient w.r.t. logit z."""
    W = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - W)
    if not np.all(np.isfinite(grad_z)):

    return grad_z

    """Compute power-smoothed objective value and gradient w.r.t. z."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    c_pos = np.maximum(c, 0.0)
    cp = c_pos ** p
    s = float(cp.sum())
    if s < 1e-300:
        return 2.0 * n * 0.0, np.zeros_like(z)
    M = s ** (1.0 / p)
    # g_k = c_k^(p-1) / (sum c^p)^((p-1)/p)
    denom = s ** ((p - 1.0) / p)
    g = (c_pos ** (p - 1)) / denom
    grad_a = _compute_grad_a_from_weights(g, a, n)
    grad_z = _grad_z_from_grad_a(grad_a, a)
    return 2.0 * n * M, grad_z


def _topk_smooth_obj_and_grad(z, k, n, beta_scale=0.005):
    """Compute top-k logsumexp smoothed objective and gradient."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    idx = np.argsort(c)[-k:]
    c_top = c[idx]
    M = float(c_top.max())

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _make_symmetric_profile(n, shape='bathtub'):
    """Create a symmetric profile suitable for initialization."""
    t = np.linspace(-1.0, 1.0, n)
    if shape == 'bathtub':

    elif shape == 'cosine':
        a = 0.5 + 0.5 * np.cos(np.pi * t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()

    Explores n={4096, 8192} with multi-restart and progressive polishing.
    """
    global height_sequence_1

    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm_fn(parent)
                update_best(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent: {e}")

    if parent is None:
        parent = _make_symmetric_profile(4096, 'bathtub')
        sc = _score_norm_fn(parent)
        update_best(parent, sc, "init_bathtub")

    # === Phase 1: Multi-scale optimization ===
    scales = [4096, 8192]

    for si, scale_n in enumerate(scales):
        if time.monotonic() > deadline - 120:
            break
        print(f"\n{'='*50}")
        print(f"Phase 1: Scale n={scale_n}")
        print(f"{'='*50}")

        # Initial point for this scale
        init_p = _interp_profile(parent, scale_n)
        sc_init = _score_norm_fn(init_p)
        update_best(init_p, sc_init, f"init_n{scale_n}")

        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        # Step 1: Momentum descent with power smoothing (coarse exploration)
        budget1 = min(120.0, remaining * 0.2)
        dl1 = min(deadline - 120, time.monotonic() + budget1)
        print(f"  1a) Momentum+power: {budget1:.0f}s")
        p1, sc1 = _run_momentum_descend(init_p, dl1, p_start=20, p_end=10000,
                                        lr=0.08, beta=0.9, max_iter=50000)
        update_best(p1, sc1, f"mom_n{scale_n}")

        if time.monotonic() > deadline - 100:
            break

        # Step 2: L-BFGS with power smoothing continuation
        budget2 = min(180.0, (deadline - time.monotonic()) * 0.2)
        dl2 = min(deadline - 100, time.monotonic() + budget2)
        p_list = [50, 200, 1000, 5000, 20000]
        print(f"  1b) L-BFGS power: {budget2:.0f}s")
        p2, sc2 = _run_lbfgs_power(p1, dl2, p_list=p_list, n_levels=len(p_list))
        update_best(p2, sc2, f"lbfgs_pw_n{scale_n}")

        if time.monotonic() > deadline - 80:
            break

        # Step 3: Adam with active set
        budget3 = min(200.0, (deadline - time.monotonic()) * 0.25)
        dl3 = min(deadline - 80, time.monotonic() + budget3)
        print(f"  1c) Adam active: {budget3:.0f}s")
        p3, sc3 = _adam_active(p2, dl3, lr=0.03, max_iter=60000, initial_tol=0.005)
        update_best(p3, sc3, f"adam_n{scale_n}")

        if time.monotonic() > deadline - 60:
            break

        # Step 4: Top-k L-BFGS (near-exact)
        budget4 = min(80.0, (deadline - time.monotonic()) * 0.15)
        dl4 = min(deadline - 40, time.monotonic() + budget4)
        print(f"  1d) Top-k L-BFGS: {budget4:.0f}s")
        p4, sc4 = _run_lbfgs_topk(p3, dl4, top_k=8, beta_scale=0.003)
        update_best(p4, sc4, f"topk_n{scale_n}")

        if time.monotonic() > deadline - 40:
            break

        # Step 5: Active set refinement
        budget5 = min(60.0, (deadline - time.monotonic()) * 0.15)
        dl5 = min(deadline - 20, time.monotonic() + budget5)
        print(f"  1e) Active refine: {budget5:.0f}s")
        p5, sc5 = _active_set_refine(p4, dl5, initial_tol=0.001, max_iter=5000)
        update_best(p5, sc5, f"refine_n{scale_n}")

        # Use best at this scale as parent for next
        scale_best = p5 if sc5 <= sc4 else p4
        parent = scale_best

    # === Phase 2: Multi-restart refinement ===
    if time.monotonic() < deadline - 80 and best_seq is not None:
        print(f"\n{'='*50}")
        print(f"Phase 2: Multi-restart")
        print(f"{'='*50}")
        n_best = len(best_seq)
        n_restarts = 4
        per_restart = min(100.0, (deadline - time.monotonic() - 30) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            noise_levels = [0.01, 0.03, 0.05, 0.08]
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_levels[i] * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            dl = min(deadline - 20, time.monotonic() + per_restart)
            # Adam then top-k L-BFGS
            per_a = per_restart * 0.55
            per_b = per_restart * 0.4
            p_r, sc_r = _adam_active(p_start, min(deadline - 30, time.monotonic() + per_a),
                                     lr=0.02, max_iter=20000, initial_tol=0.003)
            update_best(p_r, sc_r, f"r{i}_adam")

            if time.monotonic() < deadline - 15:
                p_r2, sc_r2 = _run_lbfgs_topk(p_r, min(deadline - 10, time.monotonic() + per_b),
                                              top_k=5, beta_scale=0.002)
                update_best(p_r2, sc_r2, f"r{i}_topk")

    # === Phase 3: Final aggressive polish ===
    if time.monotonic() < deadline - 10 and best_seq is not None:
        print(f"\n{'='*50}")
        print(f"Phase 3: Final polish")
        print(f"{'='*50}")
        remaining = deadline - time.monotonic()

        # Very tight active set refinement
        if remaining > 30:
            dl = min(deadline - 5, time.monotonic() + min(60.0, remaining * 0.5))
            p_f, sc_f = _active_set_refine(best_seq, dl, initial_tol=0.0001, max_iter=8000)
            update_best(p_f, sc_f, "final_refine")
            if sc_f < best_score:
                best_seq = p_f

        # Final top-k L-BFGS with very tight smoothing
        if time.monotonic() < deadline - 10:
            dl2 = min(deadline - 5, time.monotonic() + min(40.0, (deadline - time.monotonic()) * 0.5))
            p_f2, sc_f2 = _run_lbfgs_topk(best_seq, dl2, top_k=5, beta_scale=0.001)
            update_best(p_f2, sc_f2, "final_topk")

        # One more active set pass
        if time.monotonic() < deadline - 5:
            dl3 = min(deadline - 2, time.monotonic() + min(15.0, deadline - time.monotonic() - 3))
            p_f3, sc_f3 = _active_set_refine(best_seq, dl3, initial_tol=0.00005, max_iter=3000)
            update_best(p_f3, sc_f3, "final_refine2")

    # === Return ===
    print(f"\n{'='*50}")
    print(f"RESULT")
    print(f"{'='*50}")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = np.ones(2048) / 2048.0
            final_score = _score_fn(final_seq)

        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")
        print(f"  Sum={s:.6f}, max={final_seq.max():.8f}, min={final_seq.min():.8f}")

        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 16384:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  eval check: {e}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_symmetric_profile(4096, 'bathtub')
    print(f"  FALLBACK: n=4096, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
