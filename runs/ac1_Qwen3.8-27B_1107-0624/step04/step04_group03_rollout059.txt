2n*max(convolve(a,a))/sum(a)^2.
Key differences from previous approach: operates directly in a-space (not log-space),

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)


def _compute_h(a, active_mask):
    """Compute h[i] = (1/|S|) * sum_{k in S} a[k-i] using FFT."""
    n = len(a)
    n_active = active_mask.sum()
    if n_active == 0:

    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    return w_conv[n-1:2*n-1]


def _subgradient(a, active_frac, max_active):

    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.zeros(n), np.inf, c

    grad = h - W / s

    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_score = _score(a)

    lr = lr_init
    mom = mom_init
    t = 0

                n_act = 1

        a_new = a + v

            cm0 = c.max()
        
        # Check score periodically
        if t % 100 == 0:

            if sc < best_score:

"""
Direct subgradient optimizer with smooth max approximation for minimizing
2n*max(convolve(a,a))/sum(a)^2.
Key improvements: operates in a-space directly (not log-space) allowing zeros
and spiky sequences; adaptive temperature schedule; diverse restarts;
Nesterov momentum with adaptive step size; and active-set refinement.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_score_and_grad(a, beta):
    """Compute smooth score and gradient using log-sum-exp approximation."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf, np.zeros(n)
    c = _full_convolve(a, a)
    cm = c.max()
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    se = e.sum()
    if not np.isfinite(se) or se < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        W = cm
    else:
        w = e / se
        W = cm + beta * np.log(se)
    
    smooth_score = 2.0 * n * W / (s * s)
    
    # Gradient: 4n/s^2 * (h[i] - W/s) where h[i] = sum_k w[k]*a[k-i]
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = (4.0 * n / (s * s)) * (h - W / s)
    
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n)
    
    return smooth_score, grad


def _active_set_grad(a, active_frac, max_active):
    """Compute subgradient using hard active set."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.zeros(n)
    c = _full_convolve(a, a)
    cm = c.max()
    
    if active_frac >= 1.0:
        active = np.ones(len(c), dtype=bool)
    else:
        threshold = cm * (1.0 - active_frac)
        active = c >= threshold
    
    n_act = active.sum()
    if n_act > max_active:
        idx = np.argsort(c)[-max_active:]
        active = np.zeros(len(c), dtype=bool)
        active[idx] = True
        n_act = max_active
    if n_act == 0:
        active[np.argmax(c)] = True
        n_act = 1
    
    w = active.astype(np.float64) / n_act
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = np.dot(active, c) / n_act
    
    grad = (4.0 * n / (s * s)) * (h - W / s)
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n)
    return grad


def _nesterov_smooth(a_init, deadline, beta_start, beta_end, lr_init, mom_init,
                     beta_decay, lr_decay, check_interval=50):
    """Nesterov-accelerated gradient descent on smooth approximation."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()
    
    v = np.zeros(n)
    best_a = a.copy()
    best_score = _score(a)
    
    beta = beta_start
    lr = lr_init
    mom = mom_init
    t = 0
    active_frac = 0.1
    max_active = 500
    
    while time.monotonic() < deadline:
        t += 1
        
        # Compute gradient
        if beta > beta_end * 1.01:
            _, grad = _smooth_score_and_grad(a, beta)
        else:
            grad = _active_set_grad(a, active_frac, max_active)
        
        if not np.all(np.isfinite(grad)):
            lr *= 0.5
            continue
        
        # Nesterov step
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Check if step was beneficial (using smooth score for speed)
        if t % check_interval == 0:
            a = a_new
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            else:
                # Revert and reduce lr
                a = best_a.copy()
                v = np.zeros(n)
                lr *= 0.7
        else:
            a = a_new
        
        # Adapt parameters
        if t % 200 == 0:
            beta = max(beta_end, beta * beta_decay)
            lr = max(1e-6, lr * lr_decay)
            mom = min(0.999, mom + 0.002)
            active_frac = max(0.0, active_frac * 0.9)
            max_active = max(1, int(max_active * 0.95))
    
    # Final check
    sc = _score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score


def _adam_direct(a_init, deadline, beta, lr_init=0.01):
    """Adam optimizer in a-space with projection."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    
    best_a = a.copy()
    best_score = _score(a)
    
    while time.monotonic() < deadline:
        t += 1
        _, grad = _smooth_score_and_grad(a, beta)
        
        if not np.all(np.isfinite(grad)):
            continue
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        
        a_new = a - lr * mh / (np.sqrt(vh) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new
        
        if t % 100 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            beta = max(1e-8, beta * 0.85)
    
    sc = _score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score


def _make_linear_profile(n, slope=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_random_profile(n, rng, alpha=1.0):
    """Generate random non-negative profile with power-law tail."""
    a = rng.exponential(scale=1.0, size=n) ** alpha
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal_profile(n, center_frac=0.5, width=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - center_frac) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_profile(n, power=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.cos(np.pi * (t - 0.5)) ** power
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _nesterov_from_z(a_init, deadline, beta_start, beta_end, lr_init=0.02, mom_init=0.7):
    """Nesterov in log-space (softmax parameterization) as backup."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()
    
    best_a = a_init.copy()
    best_score = _score(a_init)
    
    beta = beta_start
    lr = lr_init
    mom = mom_init
    t = 0
    
    while time.monotonic() < deadline:
        t += 1
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        _, grad_a = _smooth_score_and_grad(a, beta)
        # Convert gradient from a-space to z-space: grad_z[i] = a[i] * grad_a[i]
        grad_z = a * grad_a
        grad_z = grad_z - grad_z.mean()
        
        if not np.all(np.isfinite(grad_z)):
            lr *= 0.5
            continue
        
        y = y - lr * grad_z
        z_new = y - mom * (y - z)
        z_new = z_new - z_new.mean()
        z_new = np.clip(z_new, -30, 30)
        z = z_new
        
        if t % 100 == 0:
            a_curr = np.exp(z - z.max())
            a_curr = a_curr / a_curr.sum()
            sc = _score(a_curr)
            if sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            
            beta = max(beta_end, beta * 0.95)
            lr = max(1e-5, lr * 0.98)
            mom = min(0.99, mom + 0.001)
    
    z_final = z - z.max()
    a_final = np.exp(z_final)
    a_final = a_final / a_final.sum()
    sc = _score(a_final)
    if sc < best_score:
        best_score = sc
        best_a = a_final.copy()
    
    return best_a, best_score

    fallback = _make_cosine_profile(4096, 2.0)
    sc_f = _score(fallback)
    print(f"FALLBACK: n=4096, score={sc_f:.10f}")
    return [float(x) for x in fallback]
