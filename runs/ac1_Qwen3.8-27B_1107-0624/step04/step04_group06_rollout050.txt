**Key insight**: The problem is equivalent to finding a probability distribution p (after normalizing) that minimizes `2n * max_k (p*p)[k]`.

The sum of all (p*p)[k] values is 1 (since sum(p)=1, so sum of convolution = 1).
There are 2n-1 values, so the average is 1/(2n-1).
The minimum possible maximum is 1/(2n-1) (if perfectly uniform), giving score = 2n/(2n-1) ≈ 1.

But we can't achieve perfectly uniform with non-negative p. The constraint is that p*p must be the convolution of a non-negative sequence with itself.

This is a well-studied problem in the context of "minimum peak-to-average power ratio" sequences.

**Approaches to try:**

1. **Iterative projection / water-filling on the convolution**:
   - Start with some initial p
   - Compute c = p*p
   - Find the peak positions
   - "Push down" the values of p that contribute to the peak

2. **Bisection on M with LP feasibility**:
   - For a given M, check if there exists p ≥ 0, sum(p)=1 with (p*p)[k] ≤ M for all k

    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

    return val, w

    # pressure[j] = sum_k w[k] * p[k-j] + sum_k w[k] * p[j-k]
    # 
    # First sum: let i = k-j, so k = i+j: sum_i w[i+j] * p[i] (for i in [0,n-1] and i+j in [0,2n-2])
    # Second sum: let i = j-k, so k = j-i: sum_i w[j-i] * p[i] (for i in [0,n-1] and j-i in [0,2n-2])
    #
    # First sum = <p, w shifted by j> = cross-correlation
    # Second sum = <p, w reversed and shifted by j>
    #
    # Let w_rev[k] = w[2n-2-k] (reverse of w)
    # Second sum = sum_i w[j-i] * p[i] = sum_i w_rev[2n-2-j+i] * p[i]
    # Hmm, this isn't clean either.
    #
    # Let me just use the convolution interpretation:
    # conv(p,p)[k] = sum_i p[i]*p[k-i]
    # d/dp[j] conv(p,p)[k] = p[k-j] + p[j-k] (with out-of-bounds = 0)
    #
    # So pressure[j] = sum_k w[k] * p[k-j] + sum_k w[k] * p[j-k]
    #
    # Define: f[j] = sum_k w[k] * p[k-j] (first term)
    #         g[j] = sum_k w[k] * p[j-k] (second term)
    #
    # f[j] = sum_{k=j}^{j+n-1} w[k] * p[k-j] (k-j in [0,n-1], k in [j, j+n-1]∩[0,2n-2])
    #      = sum_{i=0}^{n-1} w[i+j] * p[i] (with w out of range = 0)
    #
    # g[j] = sum_{k=max(0,j-n+1)}^{min(j,2n-2)} w[k] * p[j-k] (j-k in [0,n-1])
    #      = sum_{i=0}^{n-1} w[j-i] * p[i] (with w out of range = 0)
    #
    # Now, g[j] = sum_i p[i] * w[j-i] = (p * w)[j] (linear convolution, p has length n, w has length 2n-1)
    # The linear convolution (p*w) has length n + (2n-1) - 1 = 3n-2.
    # We need g[j] for j in [0, n-1].
    #
    # f[j] = sum_i p[i] * w[i+j] = sum_i p[i] * w_rev[(2n-1-1)-(i+j)] ... 
    # Hmm, let's define w_shifted differently.
    # f[j] = sum_i p[i] * w[i+j]
    # If I define w2[m] = w[m], then f[j] = sum_i p[i] * w2[i+j] = cross-correlation of p and w2 at lag j.
    # cross_corr_j = sum_i p[i] * w2[i+j]
    # This is the same as (p_rev * w2)[n-1+j] where p_rev = p[n-1::-1]
    # Because (p_rev * w2)[m] = sum_i p_rev[i] * w2[m-i] = sum_i p[n-1-i] * w2[m-i]
    # Let j' = n-1-i: = sum_{j'} p[j'] * w2[m-n+1+j']
    # So (p_rev * w2)[m] = sum_{j'} p[j'] * w2[m-n+1+j']
    # We want f[j] = sum_i p[i] * w2[i+j], so m-n+1+j' = i+j => m = j+n-1 (with j'=i)
    # So f[j] = (p_rev * w2)[j+n-1]
    
    # OK let me just compute both via a single FFT-based approach.
    # I'll compute the full linear convolution of p and w (length 3n-2), 
    # and also the convolution of p_rev and w.
    
    N = _next_pow2(3 * n + 1)  # Need N >= 3n-1 for p*w to not wrap
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p, N)
    fp_rev = np.fft.rfft(p[::-1], N)
    
    # g = p * w (linear convolution)
    pw = np.fft.irfft(fw * fp, N)
    # f[j] = (p_rev * w)[j+n-1]
    pwr = np.fft.irfft(fw * fp_rev, N)
    
    # g[j] for j in [0, n-1]
    term2 = pw[:n]  # (p*w)[j] for j=0..n-1
    # f[j] = (p_rev * w)[j+n-1] for j=0..n-1
    term1 = pwr[n-1:2*n-1]
    
    return term1 + term2


def _adam_logit_v2(p0, deadline, rng, lr=0.05, beta_frac=0.06, beta_min=1e-5,
                   beta_decay=0.94, restart_patience=200):
    """Adam optimizer in log-space with adaptive smooth-max."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc / max(n, 1))

    best_p = p0.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max(c, beta)
        g = _pressure_from_weights(w, p)

        # Normalize gradient to unit "effective" size
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        g = g / g_norm

        # Project gradient onto tangent space of simplex (in log-space)
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_p.copy()
            z = np.log(np.maximum(z, 1e-12))
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = np.log(np.maximum(best_p, 1e-12))
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-10:
                best_score = score
                best_p = p.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            z = np.log(np.maximum(best_p, 1e-12))
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        # Adapt beta
        if t_adam % 20 == 0:
            beta = max(beta_min, beta * beta_decay)

        # Restart if stuck
        if no_imp > restart_patience:
            if best_score < 1.51:
                ns = 0.03
            elif best_score < 1.53:
                ns = 0.08
            else:
                ns = 0.15
            z = np.log(np.maximum(best_p, 1e-12)) + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc / max(n, 1))
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _mwu_update(p0, deadline, eta_init=0.1, n_iter_max=10000):
    """Multiplicative weights update based on peak pressure."""
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    n = p.size
    
    best_p = p.copy()
    best_score = _score_norm(p)
    c = _full_convolve(p)
    eta = eta_init
    no_imp = 0
    
    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break
        
        M = float(c.max())
        score = float(2.0 * n * M)
        
        if score < best_score - 1e-10:
            best_score = score
            best_p = p.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Compute peak weights (hard max with small tolerance)
        tol = 0.01 * M
        peak_mask = c >= M - tol
        n_peaks = int(peak_mask.sum())
        w = peak_mask.astype(np.float64) / n_peaks
        
        # Compute pressure
        pressure = _pressure_from_weights(w, p)
        pressure = np.maximum(pressure, 0)
        
        # Normalize pressure
        p_max = float(pressure.max())
        if p_max < 1e-15:
            break
        pressure = pressure / p_max
        
        # Multiplicative update
        p = p * np.exp(-eta * pressure)
        p = np.maximum(p, 1e-12)
        p = p / p.sum()
        
        c = _full_convolve(p)
        
        # Adaptive eta
        if it % 50 == 0 and it > 0:
            if no_imp > 50:
                eta = max(0.01, eta * 0.9)
            elif no_imp == 0:
                eta = min(0.3, eta * 1.05)
        
        # Occasional restart
        if no_imp > 500:
            p = best_p * (1 + 0.05 * np.random.default_rng(it).normal(size=n))
            p = np.maximum(p, 1e-12)
            p = p / p.sum()
            c = _full_convolve(p)
            eta = eta_init
            no_imp = 0
    
    return best_p, best_score


def _lp_bisection(a, deadline, n_iters=30):
    """LP bisection: find the minimum M such that there exists
    b >= 0, sum(b) >= sum(a), with conv(a, b) <= M everywhere.
    Then blend a with b."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    s_a = float(a.sum())
    a = a / s_a  # normalize to sum=1
    
    c = _full_convolve(a)
    M_lo = float(c.max()) * 0.99  # current peak (should be feasible)
    M_hi = M_lo * 1.5
    
    best_a = a.copy()
    best_score = _score_norm(a)
    t_blend = 0.3
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        M_mid = (M_lo + M_hi) / 2
        
        # LP: maximize sum(b) subject to b >= 0, conv(a,b)[k] <= M_mid for all k
        # This is linear in b!
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                rows.extend([k] * len(idx))
                cols.extend(idx.tolist())
                vals.extend(a[k - idx].tolist())
        
        A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b_ub = M_mid * np.ones(2 * n - 1)
        bounds = [(0, None)] * n
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if res.success:
            b = np.maximum(res.x, 0)
            s_b = float(b.sum())
            if s_b > 1e-10:
                b_norm = b / s_b
                a_new = (1 - t_blend) * a + t_blend * b_norm
                a_new = np.maximum(a_new, 1e-15)
                a_new = a_new / a_new.sum()
                sc = _score_norm(a_new)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_a = a_new.copy()
                a = a_new
                c = _full_convolve(a)
                M_lo = float(c.max()) * 0.995  # new lower bound (should be feasible)
                # Try to decrease M_hi
                if M_mid > float(c.max()):
                    M_hi = M_mid
                else:
                    M_lo = M_mid
            else:
                M_lo = M_mid
        else:
            M_lo = M_mid
        
        t_blend = max(0.1, t_blend * 0.95)
    
    return best_a, best_score


def _make_init_constant(n):
    return np.ones(n) / n


def _make_init_cosh(n, alpha):
    """Cosh-shaped profile."""
    t = np.linspace(-1, 1, n)
    a = np.cosh(alpha * t)
    return a / a.sum()
