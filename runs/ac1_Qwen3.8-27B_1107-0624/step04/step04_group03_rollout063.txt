"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses softmax parameterization with log-sum-exp smoothing,
Nesterov-accelerated gradient descent with Polyak step size,
and multiple diverse restarts.
"""
import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_norm_fn(a):
    """Score = 2n * max(convolve(a,a)) for normalized a (sum=1)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_fn(a):
    """Full score = 2n * max(convolve(a,a)) / sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm

def _gradient_smooth(z, beta):
    """Gradient of smoothed objective w.r.t. z (log-space)."""
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

def _objective_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)

def _nesterov_optimize(a_init, deadline, beta_fracs, lr_init=0.02, mom_init=0.9, use_polyak=False):

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    
    lr = lr_init
    mom = mom_init
    t = 0
    
    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-12, beta_frac * max_c)
        
        while time.monotonic() < deadline:
            t += 1
            grad_y = _gradient_smooth(y, beta)
            if not np.all(np.isfinite(grad_y)):

            if use_polyak and t > 100:

                fy = _score_norm_fn(a_y)
                grad_norm = np.linalg.norm(grad_y)
                if grad_norm > 1e-10 and np.isfinite(fy):

            z_new = z_new - z_new.mean()

            y_new = y_new - y_new.mean()
            
            z = z_new
            y = y_new
            
            if t % 30 == 0:

                sc = _score_norm_fn(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()

        c = _full_convolve(a_curr, a_curr)

        lr = max(0.002, lr * 0.6)

    sc_final = _score_norm_fn(a_final)

def _adam_optimize(a_init, deadline, lr=0.05, beta_frac=0.01, n_max_iter=100000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, beta_frac * max_c)
    
    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        
        if not np.all(np.isfinite(grad)):
            continue
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = z - z.mean()
        a = _softmax_fn(z)
        
        if t % 100 == 0:
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
            beta = max(1e-12, beta * 0.85)
            c_new = _full_convolve(a, a)
            max_c = float(c_new.max())
    
    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _make_cosine_bell(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(np.pi * t))
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_gaussian(n, sigma=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_power_profile(n, p, q):
    t = np.linspace(0.0, 1.0, n)
    a = (t + 1e-8) ** p * (1.0 - t + 1e-8) ** q
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_sine_sq(n):
    t = np.linspace(0.0, 1.0, n)
    a = np.sin(np.pi * t) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

def _lbfgs_optimize(a_init, deadline, max_wall_time=None):

def _active_set_polish(a_init, deadline, n_iter=5000):
    """Subgradient method focusing on active (near-max) lags."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    lr = 0.003
    momentum = np.zeros(n)
    mom_coeff = 0.9
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Active set: lags within 0.5% of max
        threshold = cm * 0.9995
        active = c >= threshold
        if not np.any(active):
            active[int(np.argmax(c))] = True
        
        w = np.zeros_like(c)
        w[active] = 1.0 / np.sum(active)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        
        momentum = mom_coeff * momentum - lr * grad
        z = z + momentum
        z = z - z.mean()
        z = np.clip(z, -50, 50)
        
        if it % 50 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            else:
                lr = lr * 0.92
    
    return best_a, best_score

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key improvements over previous: better ramp-like initialization exploiting the
complementary structure (a[i]+a[n-1-i]≈const), Polyak step-size Nesterov,
more aggressive beta annealing, diverse restarts, and improved time management.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    """Score = 2n * max(convolve(a,a)) for normalized a (sum=1)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    """Full score = 2n * max(convolve(a,a)) / sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _gradient_smooth(z, beta):
    """Gradient of smoothed objective w.r.t. z (log-space)."""
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _objective_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _nesterov_optimize(a_init, deadline, beta_fracs, lr_init=0.02, mom_init=0.9,
                       use_polyak=False, eval_interval=50):
    """Nesterov-accelerated gradient descent with optional Polyak step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())

    lr = lr_init
    mom = mom_init
    t = 0
    score_change_count = 0

    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-12, beta_frac * max_c)
        prev_score = best_score
        iter_in_phase = 0

        while time.monotonic() < deadline:
            t += 1
            iter_in_phase += 1

            grad_y = _gradient_smooth(y, beta)
            if not np.all(np.isfinite(grad_y)):
                grad_y = _gradient_smooth(z, beta)

            if use_polyak and iter_in_phase > 50 and t % eval_interval == 0:
                a_y = _softmax_fn(y)
                fy = _score_norm_fn(a_y)
                grad_norm_sq = float(np.dot(grad_y, grad_y))
                if grad_norm_sq > 1e-10 and np.isfinite(fy):
                    gap = max(0.0, fy - best_score)
                    lr = max(0.001, min(0.05, gap / (grad_norm_sq + 1e-10) * 10.0))

            z_new = z - lr * grad_y
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -50, 50)

            y_new = z_new + mom * (z_new - z)
            y_new = y_new - y_new.mean()

            z = z_new
            y = y_new

            if t % eval_interval == 0:
                a_curr = _softmax_fn(z)
                sc = _score_norm_fn(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()
                    score_change_count += 1
                elif score_change_count > 0 and sc > best_score * 1.01:
                    lr = max(0.001, lr * 0.8)

            if iter_in_phase % 300 == 0:
                mom = min(0.999, mom + 0.0003)

        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        max_c = float(c.max())
        lr = max(0.001, lr * 0.5)

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _adam_optimize(a_init, deadline, lr=0.05, beta_frac=0.01, n_max_iter=100000):
    """Adam optimizer in log-space with progressive beta cooling."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, beta_frac * max_c)

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = z - z.mean()
        a = _softmax_fn(z)

        if t % 80 == 0:
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
            beta = max(1e-12, beta * 0.85)
            c_new = _full_convolve(a, a)
            max_c = float(c_new.max())

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _make_ramp_profile(n, power=1.0):
    """Ramp-like profile: a[i] ~ (i/n)^power. Exploits complementary structure."""
    t = np.linspace(0.0, 1.0, n)
    a = (t + 0.01) ** power
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_symmetric_ramp(n, power=1.0):
    """Symmetric version: high in middle, low at edges."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - (2.0 * t - 1.0) ** power
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_bell(n):

    a = 0.5 * (1.0 + np.cos(np.pi * t))
    a = np.maximum(a, 1e-12)

def _make_linear_profile(n, slope):

    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)

def _make_complementary_ramp(n, sharpness=2.0):
    """Profile designed to satisfy a[i] + a[n-1-i] ≈ const (complementary)."""
    t = np.linspace(0.0, 1.0, n)
    # a[i] = 0.5 + 0.5 * tanh(sharpness * (t - 0.5))
    # This gives a smooth S-curve that's complementary
    a = 0.5 + 0.5 * np.tanh(sharpness * (t - 0.5))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _lbfgs_optimize(a_init, deadline, max_wall_time=None):

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))

    betas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max_c0 for b in betas]

    per_level = max_wall_time / len(betas)

    for beta in betas:
        if time.monotonic() >= deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level)

        def obj(z_curr):
            return _objective_smooth(z_curr, beta)

        def grad(z_curr):
            return _gradient_smooth(z_curr, beta)

        try:
            result = minimize(
                obj, z,
                method='L-BFGS-B', jac=grad,
                options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxls': 30, 'maxfun': 1000}

            else:
                stall_count += 1
                if stall_count > 5:
                    lr = lr * 0.85
                    stall_count = 0

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: diverse ramp-based initialization -> L-BFGS ->
    Nesterov with Polyak step -> Adam -> active-set polish -> diverse restarts.
    Exploits the complementary structure a[i]+a[n-1-i]~const for better
    initializations and avoids symmetric local minima.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        print(f"Parent: n={n}, score={_score_norm_fn(parent):.10f}")
    else:
        n = 4096
        parent = _make_complementary_ramp(n, 2.0)
        print(f"No parent, using complementary ramp n={n}")

    # --- Phase 0: Evaluate diverse initial candidates (40s) ---
    phase0_end = min(deadline, t_start + 40.0)
    candidates = []

    sc_parent = _score_norm_fn(parent)
    candidates.append((sc_parent, parent.copy(), len(parent), "parent"))
    update_best(parent, sc_parent, "parent")

    for sn in [4096]:
        if time.monotonic() > phase0_end:
            break
        profiles = [
            _make_complementary_ramp(sn, 1.5),
            _make_complementary_ramp(sn, 2.0),
            _make_complementary_ramp(sn, 3.0),
            _make_ramp_profile(sn, 1.0),
            _make_ramp_profile(sn, 1.5),
            _make_power_profile(sn, 1.0, 0.5),
            _make_power_profile(sn, 1.5, 0.5),
            _make_linear_profile(sn, 1.0),
            _make_linear_profile(sn, 1.5),
            _make_cosine_bell(sn),
            _make_gaussian(sn, 0.35),
            _make_sine_sq(sn),
        ]
        for idx, p in enumerate(profiles):
            if time.monotonic() > phase0_end:
                break
            sc = _score_norm_fn(p)
            candidates.append((sc, p, sn, f"prof{idx}"))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}")
    print(f"  Top 5: {[(c[3], f'{c[0]:.8f}') for c in candidates[:5]]}")

    # --- Phase 1: L-BFGS on top 3 candidates (280s) ---
    phase1_end = min(deadline, t_start + 320.0)
    for idx, (sc, p, pn, label) in enumerate(candidates[:3]):
        if time.monotonic() >= phase1_end - 20:
            break
        per_time = min(90.0, phase1_end - time.monotonic() - 15)
        if per_time > 20:
            p_best, sc_best = _lbfgs_optimize(p, min(phase1_end - 10, time.monotonic() + per_time),
                                               max_wall_time=per_time)
            update_best(p_best, sc_best, f"lbfgs_{label}")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Nesterov with Polyak (400s) ---
    phase2_end = min(deadline, t_start + 720.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        beta_fracs = [0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003,
                      0.0002, 0.0001, 5e-5, 3e-5, 2e-5, 1e-5, 5e-6]
        p_best, sc_best = _nesterov_optimize(best_seq, phase2_end, beta_fracs,
                                             lr_init=0.015, mom_init=0.9,
                                             use_polyak=True, eval_interval=40)
        update_best(p_best, sc_best, "nesterov_phase2")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Adam polishing (250s) ---
    phase3_end = min(deadline, t_start + 970.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        per_time = min(230.0, phase3_end - time.monotonic() - 10)
        if per_time > 20:
            p_best, sc_best = _adam_optimize(best_seq, min(phase3_end - 5, time.monotonic() + per_time),
                                              lr=0.03, beta_frac=0.004)
            update_best(p_best, sc_best, "adam_phase3")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Multi-restart with diverse perturbations (320s) ---
    phase4_end = min(deadline, t_start + 1290.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        n_best = len(best_seq)
        n_restarts = 6
        per_restart = min(50.0, (phase4_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 10:
                break

            if i == 0:
                # Small Gaussian noise
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z - z.mean()
                z = z + 0.02 * rng.normal(size=n_best)
                z = np.clip(z, -50, 50)
                z = z - z.mean()
                p_start = _softmax_fn(z)
            elif i == 1:
                # Medium noise
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z - z.mean()
                z = z + 0.08 * rng.normal(size=n_best)
                z = np.clip(z, -50, 50)
                z = z - z.mean()
                p_start = _softmax_fn(z)
            elif i == 2:
                # Smooth low-frequency perturbation
                z_base = np.log(np.maximum(best_seq, 1e-15))
                z_base = z_base - z_base.mean()
                t_idx = np.arange(n_best) / n_best
                smooth_noise = np.zeros(n_best)
                for k in range(4):
                    amp = rng.uniform(0.02, 0.06)
                    freq = (k + 1) * rng.uniform(0.5, 2.0)
                    phase = rng.uniform(0, 2 * np.pi)
                    smooth_noise += amp * np.sin(2 * np.pi * freq * t_idx + phase)
                z = z_base + smooth_noise
                z = np.clip(z, -50, 50)
                z = z - z.mean()
                p_start = _softmax_fn(z)
            elif i == 3:
                # Blend with complementary ramp
                other = _make_complementary_ramp(n_best, rng.uniform(1.0, 3.0))
                alpha = rng.uniform(0.3, 0.6)
                p_start = alpha * best_seq + (1 - alpha) * other
                p_start = p_start / p_start.sum()
            elif i == 4:
                # Blend with linear profile
                other = _make_linear_profile(n_best, rng.uniform(0.5, 2.0))
                alpha = rng.uniform(0.5, 0.8)
                p_start = alpha * best_seq + (1 - alpha) * other
                p_start = p_start / p_start.sum()
            else:
                # Larger noise
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z - z.mean()
                z = z + 0.12 * rng.normal(size=n_best)
                z = np.clip(z, -50, 50)
                z = z - z.mean()
                p_start = _softmax_fn(z)

            per_dl = min(phase4_end - 5, time.monotonic() + per_restart)
            beta_fracs_short = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]
            p_best, sc_best = _nesterov_optimize(p_start, per_dl, beta_fracs_short,
                                                  lr_init=0.01, mom_init=0.85,
                                                  use_polyak=True, eval_interval=30)
            update_best(p_best, sc_best, f"restart{i}")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Fine polishing (250s) ---
    phase5_end = min(deadline, t_start + 1540.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        # Active set polish
        per_time = min(80.0, phase5_end - time.monotonic() - 120)
        if per_time > 15:
            p_best, sc_best = _active_set_polish(best_seq, min(phase5_end - 100, time.monotonic() + per_time))
            update_best(p_best, sc_best, "active_set")

        # Fine Nesterov with tiny beta
        if time.monotonic() < phase5_end - 40:
            per_time2 = min(70.0, phase5_end - time.monotonic() - 30)
            if per_time2 > 15:
                beta_fracs_tiny = [1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7]
                p_best, sc_best = _nesterov_optimize(best_seq, min(phase5_end - 20, time.monotonic() + per_time2),
                                                       beta_fracs_tiny, lr_init=0.003, mom_init=0.97,
                                                       use_polyak=True, eval_interval=30)
                update_best(p_best, sc_best, "fine_nesterov")

        # Fine Adam
        if time.monotonic() < phase5_end - 15:
            per_time3 = min(50.0, phase5_end - time.monotonic() - 10)
            if per_time3 > 10:
                p_best, sc_best = _adam_optimize(best_seq, min(phase5_end - 5, time.monotonic() + per_time3),
                                                   lr=0.005, beta_frac=0.0005)
                update_best(p_best, sc_best, "fine_adam")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Try different length (120s) ---
    phase6_end = min(deadline, t_start + 1660.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        n_curr = len(best_seq)
        for target_n in [4096, 3072, 3584, 4608]:
            if target_n == n_curr or time.monotonic() > phase6_end - 25:
                continue
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            if sc_interp < best_score:
                update_best(p_interp, sc_interp, f"interp_n{target_n}")

            if time.monotonic() < phase6_end - 35:
                per_time = min(30.0, phase6_end - time.monotonic() - 25)
                if per_time > 10:
                    p_best, sc_best = _nesterov_optimize(
                        p_interp, min(phase6_end - 15, time.monotonic() + per_time),
                        [0.003, 0.001, 0.0005, 0.0002, 0.0001],
                        lr_init=0.01, mom_init=0.85, use_polyak=True, eval_interval=30)
                    update_best(p_best, sc_best, f"nest_n{target_n}")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final ultra-fine polish (remaining) ---
    phase7_end = deadline
    if best_seq is not None and time.monotonic() < phase7_end:
        remaining = phase7_end - time.monotonic()
        if remaining > 25:
            per_time = min(40.0, remaining - 15)
            beta_fracs_final = [1e-6, 5e-7, 2e-7, 1e-7]
            p_best, sc_best = _nesterov_optimize(best_seq, min(phase7_end - 5, time.monotonic() + per_time),
                                                   beta_fracs_final, lr_init=0.002, mom_init=0.98,
                                                   use_polyak=True, eval_interval=20)
            update_best(p_best, sc_best, "final_nesterov")

    print(f"Phase 7 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_complementary_ramp(4096, 2.0)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned inf/NaN, using internal score: {sc_final:.10f}")
