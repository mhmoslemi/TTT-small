```python
"""
Optimized step function sequence minimizer using antisymmetric perturbation
and peak-targeting LP approach. Key innovation: explicitly exploit the fact
that antisymmetric perturbations reduce the center convolution peak at O(eps^2),
enabling escape from symmetric local minima. Combines LP-based direction finding
with L-BFGS refinement and multi-scale peak targeting.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _symmetrize(a):
    a = np.asarray(a, dtype=np.float64)
    return (a + a[::-1]) / 2.0


def _antisy_gradient(c, a, n, peak_indices):
    """Compute gradient of c[k] w.r.t. antisymmetric perturbation d.
    For antisymmetric d (d[i] = -d[n-1-i]), the first-order change in c[k]
    is 2*sum_i d[i]*a[k-i]. We return the coefficient vector for the free
    (first half) variables of d."""
    n_half = n // 2
    grad_half = np.zeros(n_half)
    for k in peak_indices:
        # c[k] = sum_i d[i]*a[k-i] (first order, factor of 2 absorbed)
        # d[i] for i in [0, n_half) is free; d[n-1-i] = -d[i]
        for i in range(n_half):
            # Contribution from d[i]: a[k-i] (if valid)
            if 0 <= k - i < n:
                grad_half[i] += a[k - i]
            # Contribution from d[n-1-i] = -d[i]: -a[k-(n-1-i)]
            j = n - 1 - i
            if 0 <= k - j < n:
                grad_half[i] -= a[k - j]
    return grad_half


def _solve_lp_direction(c, a, n, peak_indices, max_iter_lp=200):
    """Solve LP to find antisymmetric direction d that minimizes max increase
    at peak positions. Returns the direction (full length n) and the LP value."""
    n_half = n // 2
    n_peaks = len(peak_indices)
    if n_peaks == 0 or n_half == 0:
        return np.zeros(n), 0.0

    # Build coefficient matrix: A[j, i] = coefficient of d_free[i] in c[k_j] change
    A = np.zeros((n_peaks, n_half))
    for j, k in enumerate(peak_indices):
        for i in range(n_half):
            coeff = 0.0
            if 0 <= k - i < n:
                coeff += a[k - i]
            j2 = n - 1 - i
            if 0 <= k - j2 < n:
                coeff -= a[k - j2]
            A[j, i] = coeff

    # LP: minimize t
    # subject to: A @ d_free <= t  (for each peak)
    #             -1 <= d_free[i] <= 1
    # Variables: [d_free (n_half), t] -> total n_half + 1 variables

    # c_lp = [0, ..., 0, 1] (minimize t)
    c_lp = np.zeros(n_half + 1)
    c_lp[-1] = 1.0

    # A_ub: A @ d_free - t <= 0  =>  [A, -1] @ [d_free, t] <= 0
    A_ub = np.zeros((n_peaks, n_half + 1))
    A_ub[:, :n_half] = A
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_peaks)

    # Bounds: -1 <= d_free[i] <= 1, t unbounded below (but we minimize it)
    bounds = [(-1.0, 1.0)] * n_half
    bounds.append((None, None))

    try:
        res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            d_free = res.x[:n_half]
            t_val = res.x[-1]
            # Build full antisymmetric d
            d = np.zeros(n)
            d[:n_half] = d_free
            d[n - 1 : n - 1 - n_half : -1] = -d_free
            if n % 2 == 1:
                d[n_half] = 0.0  # center must be 0 for antisymmetry
            norm_d = np.linalg.norm(d)
            if norm_d < 1e-15:
                return np.zeros(n), 0.0
            d = d / norm_d
            return d, t_val
    except Exception:
        pass
    return np.zeros(n), 0.0


def _find_active_peaks(c, n, top_k=15):
    """Find indices of the top-k peaks in the convolution."""
    cm = c.max()
    # Get top k values
    sorted_idx = np.argsort(c)[::-1]
    active = []
    for idx in sorted_idx:
        if c[idx] > 0.7 * cm:  # at least 70% of max
            active.append(int(idx))
        if len(active) >= top_k:
            break
    if not active:
        active = [int(np.argmax(c))]
    return active


def _apply_antisymmetric_step(a, d, eps, deadline):
    """Apply antisymmetric perturbation with line search."""
    n = len(a)
    base_score = _score_norm_fn(a)

    best_a = a.copy()
    best_sc = base_score

    # Try a range of step sizes
    for trial_eps in [eps, eps * 0.5, eps * 0.25, eps * 2.0]:
        a_trial = a + trial_eps * d
        # Ensure non-negativity
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial = a_trial / s * np.sqrt(2.0 * n)  # normalize
        sc = _score_norm_fn(a_trial)
        if sc < best_sc:
            best_sc = sc
            best_a = a_trial.copy()

    return best_a, best_sc


def _antisymmetric_optimize(a_init, deadline, rng, max_iters=200):
    """Iteratively apply optimal antisymmetric perturbations to reduce peaks."""
    a = a_init.copy()
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    # Normalize to sum = sqrt(2n) for consistency
    target_sum = np.sqrt(2.0 * n)
    a = a * target_sum / a.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    eps = 0.01
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        peaks = _find_active_peaks(c, n, top_k=12)

        # Solve LP for optimal antisymmetric direction
        d, lp_val = _solve_lp_direction(c, a, n, peaks)
        if np.all(d == 0):
            # No improvement direction found
            stagnant += 1
            if stagnant > 5:
                # Try random antisymmetric direction
                d_rand = rng.normal(size=n)
                d_rand = d_rand - d_rand[::-1]  # make antisymmetric
                norm = np.linalg.norm(d_rand)
                if norm > 1e-15:
                    d = d_rand / norm
                else:
                    break
            else:
                continue

        # Apply step with adaptive epsilon
        a_new, sc_new = _apply_antisymmetric_step(a, d, eps, deadline)

        if sc_new < best_score - 1e-12:
            best_a = a_new.copy()
            best_score = sc_new
            a = a_new.copy()
            stagnant = 0
            # Slightly increase step size on success
            eps = min(eps * 1.05, 0.05)
        else:
            # No improvement - try smaller step or different direction
            eps = eps * 0.7
            stagnant += 1
            if stagnant > 10:

                z = np.log(np.maximum(a, 1e-15))
                z = z - z.mean()
                z = z + 0.02 * rng.normal(size=n)
                z = z - z.mean()
                a = _softmax_fn(z)
                a = a * target_sum / a.sum()
                eps = 0.005
                stagnant = 0

        # Print progress every 20 iterations
        if (it + 1) % 20 == 0:
            print(f"    Anti-sym iter {it+1}: score={best_score:.10f}, eps={eps:.4f}")

    return best_a, best_score


def _peak_logsumexp_obj(z, beta, n):
    """Objective: 2n * logsumexp((c - cm)/beta) + 2n*cm, approximating 2n*max(c)."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    shifted = (c - cm) / beta
    shifted = np.clip(shifted, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(shifted))))
    return 2.0 * n * lse


def _peak_logsumexp_grad(z, beta, n):
    """Gradient of the log-sum-exp peak objective."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    shifted = (c - cm) / beta
    shifted = np.clip(shifted, -50, 50)
    e = np.exp(shifted)
    w = e / e.sum()
    # Gradient of c[k] w.r.t. a[i] is a[k-i] (symmetric part)
    # Gradient of lse w.r.t. a[i] = sum_k w[k] * a[k-i]
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1 : 2 * n - 1]
    # d(lse)/d(a[i]) = h[i]
    # d(a[i])/d(z[i]) = a[i] * (1 - a[i]) (softmax derivative)
    # But since we normalize (softmax already normalizes), the gradient in z-space:
    # d(a)/d(z) = a*(I - a*a^T) for softmax
    # For simplicity, use: grad_z[i] = 2n * a[i] * (h[i] - dot(w,c))
    W = float(np.dot(w, c))
    grad = 2.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _lbfgs_optimize(a_init, deadline, beta_schedule=None):
    """L-BFGS optimization with log-sum-exp peak approximation and continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if beta_schedule is None:
        beta_schedule = [0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        beta_schedule = [b * max_c0 for b in beta_schedule]

    n_betas = len(beta_schedule)
    for idx, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_alloc = (deadline - time.monotonic()) / max(1, remaining)
        time_alloc = min(time_alloc, 180.0)
        level_dl = time.monotonic() + time_alloc

        try:
            result = minimize(
                _peak_logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_peak_logsumexp_grad,
                options={'maxiter': 2000, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': 8000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                if np.all(a_new >= 0) and a_new.sum() > 1e-10:

                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
                        print(f"    L-BFGS beta={beta:.6f}: score={best_score:.10f}")
        except Exception as ex:
            print(f"    L-BFGS error at beta={beta:.6f}: {ex}")

    return best_a, best_score


def _direct_peak_reduction(a, deadline, max_rounds=100):
    """Directly target the highest peak by reducing contributing elements."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    target_sum = np.sqrt(2.0 * n)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))
        cm = float(c[k_star])

        # Find indices contributing to this peak
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        valid = np.arange(lo, hi + 1)
        contribs = a[valid] * a[k_star - valid]
        total_contrib = contribs.sum()
        if total_contrib < 1e-15:
            break

        # Try reducing the top contributor
        top_pos = valid[np.argmax(contribs)]
        improved = False

        for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
            a_trial = a.copy()
            a_trial[top_pos] *= (1.0 - frac)
            # Redistribute to maintain sum
            deficit = target_sum - a_trial.sum()
            if deficit > 0:
                # Add to smallest elements
                small_idx = np.argsort(a_trial)[:10]
                a_trial[small_idx] += deficit / len(small_idx)
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial * target_sum / a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
                improved = True
                break

        if not improved:
            # Try adding to a small gap element (redistribute mass)
            small_idx = int(np.argmin(a))
            for frac in [0.01, 0.05, 0.1]:
                a_trial = a.copy()
                # Take from top contributor, give to small element
                a_trial[small_idx] += a_trial[top_pos] * frac
                a_trial[top_pos] *= (1.0 - frac)
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial * target_sum / a_trial.sum()
                sc = _score_norm_fn(a_trial)
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break

        if not improved:
            # Try antisymmetric nudge
            d = np.zeros(n)
            d[top_pos] = 1.0
            d[n - 1 - top_pos] = -1.0
            norm = np.linalg.norm(d)
            d /= norm
            for eps in [0.001, 0.005, 0.01, 0.02]:
                a_trial = a + eps * d
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial * target_sum / a_trial.sum()
                sc = _score_norm_fn(a_trial)
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break

        if not improved:
            break

    return best_a, best_score


def _adam_optimize(a_init, deadline, lr_init=0.1, max_iter=50000):
    """Adam optimizer in log-space with soft peak objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Adaptive beta
        sorted_c = np.sort(c)[::-1]
        k = min(10, len(sorted_c) - 1)
        gap = sorted_c[0] - sorted_c[k]
        beta = max(1e-8, 0.1 * gap) if gap > 1e-10 else 0.01 * cm

        # Compute gradient
        shifted = (c - cm) / beta
        shifted = np.clip(shifted, -50, 50)
        e = np.exp(shifted)
        w = e / e.sum()
        W = float(np.dot(w, c))

        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1 : 2 * n - 1]
        grad = 2.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps_adam)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Decayed learning rate
        if t % 2000 == 0:
            lr = max(0.01, lr * 0.9)

        # Periodic kick
        if t % 10000 == 0:
            z = z + 0.03 * np.random.default_rng(t).normal(size=n)
            z = z - z.mean()

    return best_a, best_score


def _make_initial_profiles(n):
    """Generate diverse initial profiles for exploration."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []

    # Symmetric bumps
    for width in [0.3, 0.5, 0.7, 1.0]:
        a = np.exp(-((u) ** 2) / (2 * width ** 2))
        profiles.append(a / a.sum())

    # Raised cosine
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())

    # Linear (asymmetric)
    for slope in [1.0, 2.0, -1.0, -2.0, 3.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Quadratic bump
    for center in [0.3, 0.5, 0.7]:
        a = np.exp(-5 * (t - center) ** 2)
        profiles.append(a / a.sum())

    return profiles


def _interp_to_length(a, n_new):
    """Interpolate sequence to new length."""
    a = np.asarray(a, dtype=np.float64)
    n_old = len(a)
    if n_old == n_new:
        return a / a.sum()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Load parent, create symmetric + asymmetric candidates
    2. Antisymmetric perturbation (key innovation)
    3. L-BFGS with continuation
    4. Direct peak reduction
    5. Adam refinement
    6. Final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            actual = _score_fn(p)
            print(f"  NEW BEST: norm={sc:.10f}, actual={actual:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        print(f"Parent loaded: n={n}, score={_score_fn(parent):.10f}")
        update_best(parent, _score_norm_fn(parent), "parent")
    else:
        n = 4096
        parent = _make_initial_profiles(n)[3]  # Gaussian bump
        print(f"Using default: n={n}")
        update_best(parent, _score_norm_fn(parent), "default")

    # Normalize parent to sum = sqrt(2n)
    target_sum = np.sqrt(2.0 * n)
    a = parent * target_sum / parent.sum()

    # === Phase 1: Symmetric optimization (60s) ===
    phase1_end = min(deadline, t_start + 60.0)
    print("Phase 1: Symmetric optimization")

    a_sym = _symmetrize(a)
    a_sym = a_sym * target_sum / a_sym.sum()
    sc_sym = _score_norm_fn(a_sym)
    if sc_sym < best_score:
        update_best(a_sym, sc_sym, "symmetric")

    # Quick L-BFGS on symmetric version
    if time.monotonic() < phase1_end:
        a_sym_opt, sc_sym_opt = _lbfgs_optimize(
            a_sym, phase1_end,
            beta_schedule=[0.02 * a_sym.max() * n, 0.005 * a_sym.max() * n, 0.001 * a_sym.max() * n]
        )
        update_best(a_sym_opt, sc_sym_opt, "sym_lbfgs")

    # Also try asymmetric initial optimization
    if time.monotonic() < phase1_end - 10:
        a_asym = a.copy()
        a_asym_opt, sc_asym_opt = _lbfgs_optimize(
            a_asym, phase1_end - 5,
            beta_schedule=[0.02 * a_asym.max() * n, 0.005 * a_asym.max() * n]
        )
        update_best(a_asym_opt, sc_asym_opt, "asym_lbfgs")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Antisymmetric perturbation (500s) ===
    phase2_end = min(deadline, t_start + 560.0)
    print("Phase 2: Antisymmetric perturbation")

    if time.monotonic() < phase2_end and best_seq is not None:
        a_start = best_seq * target_sum / best_seq.sum()
        a_anti, sc_anti = _antisymmetric_optimize(a_start, phase2_end, rng, max_iters=300)
        update_best(a_anti, sc_anti, "antisym")

        # Run again from the result with different parameters
        if time.monotonic() < phase2_end - 60:
            a_anti2, sc_anti2 = _antisymmetric_optimize(a_anti, phase2_end, rng, max_iters=200)
            update_best(a_anti2, sc_anti2, "antisym2")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Full L-BFGS with fine continuation (500s) ===
    phase3_end = min(deadline, t_start + 1060.0)
    print("Phase 3: L-BFGS fine continuation")

    if time.monotonic() < phase3_end and best_seq is not None:
        a_cur = best_seq * target_sum / best_seq.sum()
        c_cur = _full_convolve(a_cur, a_cur)
        max_c = float(c_cur.max())

        # Fine continuation schedule
        betas = [0.03, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        betas = [b * max_c for b in betas]

        a_lbfgs, sc_lbfgs = _lbfgs_optimize(a_cur, phase3_end, beta_schedule=betas)
        update_best(a_lbfgs, sc_lbfgs, "lbfgs_fine")

        # Restart from perturbed version
        if time.monotonic() < phase3_end - 60:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + 0.03 * rng.normal(size=n)
            z = z - z.mean()
            a_pert = _softmax_fn(z) * target_sum
            a_pert_opt, sc_pert = _lbfgs_optimize(
                a_pert, phase3_end,
                beta_schedule=betas[2:]
            )
            update_best(a_pert_opt, sc_pert, "lbfgs_restart")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Direct peak reduction (300s) ===
    phase4_end = min(deadline, t_start + 1360.0)
    print("Phase 4: Direct peak reduction")

    if time.monotonic() < phase4_end and best_seq is not None:
        a_cur = best_seq * target_sum / best_seq.sum()
        a_dr, sc_dr = _direct_peak_reduction(a_cur, phase4_end, max_rounds=200)
        update_best(a_dr, sc_dr, "peak_reduce")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Adam refinement (200s) ===
    phase5_end = min(deadline, t_start + 1560.0)
    print("Phase 5: Adam refinement")

    if time.monotonic() < phase5_end and best_seq is not None:
        a_cur = best_seq * target_sum / best_seq.sum()
        a_adam, sc_adam = _adam_optimize(a_cur, phase5_end, lr_init=0.05)
        update_best(a_adam, sc_adam, "adam")

        # Second Adam with different lr
        if time.monotonic() < phase5_end - 50:
            a_adam2, sc_adam2 = _adam_optimize(a_cur, phase5_end, lr_init=0.02)
            update_best(a_adam2, sc_adam2, "adam_slow")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Final combined polish (remaining) ===
    phase6_end = deadline
    print("Phase 6: Final polish")

    if time.monotonic() < phase6_end and best_seq is not None:
        a_cur = best_seq * target_sum / best_seq.sum()

        # Tiny beta L-BFGS
        c_cur = _full_convolve(a_cur, a_cur)
        max_c = float(c_cur.max())
        tiny_betas = [0.0005 * max_c, 0.0002 * max_c, 0.0001 * max_c]
        a_final, sc_final = _lbfgs_optimize(a_cur, phase6_end, beta_schedule=tiny_betas)
        update_best(a_final, sc_final, "final_tiny_lbfgs")

        # Final antisymmetric nudge
        if time.monotonic() < phase6_end - 30 and best_seq is not None:
            a_cur2 = best_seq * target_sum / best_seq.sum()
            a_nudge, sc_nudge = _antisymmetric_optimize(a_cur2, phase6_end, rng, max_iters=100)
            update_best(a_nudge, sc_nudge, "final_antisym")

    # === Try different n ===
    if time.monotonic() < deadline - 30 and best_seq is not None:
        print("Trying alternative sizes")
        for n_alt in [2048, 8192]:
            if time.monotonic() > deadline - 30:
                break
            if n_alt == len(best_seq):
                continue
            a_alt = _interp_to_length(best_seq, n_alt) * np.sqrt(2.0 * n_alt)
            sc_alt = _score_norm_fn(a_alt)
            if sc_alt < best_score:
                update_best(a_alt, sc_alt, f"interp_n{n_alt}")
            # Quick optimization
            if time.monotonic() < deadline - 30:
                a_opt, sc_opt = _lbfgs_optimize(a_alt, min(deadline - 10, time.monotonic() + 60),
                                                 beta_schedule=[0.01 * a_alt.max() * n_alt, 0.001 * a_alt.max() * n_alt])
                update_best(a_opt, sc_opt, f"opt_n{n_alt}")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            pass  # Keep as is; score is scale-invariant
        else:
            final_seq = _make_initial_profiles(256)[2]

        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        # Validate with evaluator if feasible
        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"  Evaluator confirms: {sc_eval:.10f}")
                else:
                    print(f"  Evaluator returned inf, using fallback")
                    fallback = _make_initial_profiles(512)[3]
                    return [float(x) for x in fallback]
            except Exception as ex:
                print(f"  evaluate_sequence error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    print("FALLBACK: using default profile")
    fallback = _make_initial_profiles(512)[3]
    print(f"  Fallback score: {_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
```
