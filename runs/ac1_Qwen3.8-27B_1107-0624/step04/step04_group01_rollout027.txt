```python
"""
LP-based direction finding combined with gradient polishing for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key innovation: Use linear programming to find optimal directions that
minimize the convolution peak, combined with log-sum-exp smoothed gradients.
"""
import numpy as np
from scipy.optimize import linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assume sum(a)=1, return 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix_vals(a):
    """Build the dense convolution matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = 2 * n - 1
    cols = n
    A = np.zeros((rows, cols), dtype=np.float64)
    for k in range(rows):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def _build_conv_matrix_sparse(a):
    """Build sparse convolution matrix."""
    from scipy.sparse import lil_matrix
    n = len(a)
    rows = 2 * n - 1
    A = lil_matrix((rows, n), dtype=np.float64)
    for k in range(rows):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def _solve_lp_direction(a, M):
    """
    Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns normalized g (sum=1) or None if infeasible.
    """
    n = len(a)
    A = _build_conv_matrix_sparse(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = res.x
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def _line_search(a, direction, f_current, max_iter=50):
    """Backtracking line search. Find t such that f((1-t)*a + t*direction) < f_current."""
    n = len(a)
    best_t = 0.0
    best_f = f_current
    t = 1.0
    for _ in range(max_iter):
        a_try = (1 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-10:
            t /= 2
            continue
        a_try = a_try / s
        f_try = _score_norm(a_try)
        if f_try < best_f:
            best_f = f_try
            best_t = t
        t /= 2
        if t < 1e-8:
            break
    return best_t, best_t > 0 and a_try / a_try.sum() if best_t > 0 else a


def _gradient_step(z, a, c, beta, lr):
    """
    One gradient step in log-space using log-sum-exp smoothing.
    Returns updated z.
    """
    n = len(a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    
    # Gradient of smoothed max w.r.t. a: 2 * (w * a reversed) circular-like
    # d/d a[m] of sum_k w[k] * (a*a)[k] = 2 * sum_k w[k] * a[k-m]
    # (by the symmetry of convolution)
    a_rev = a[::-1]
    # h[m] = sum_k w[k] * a[k-m] for valid k-m
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]  # This gives sum_k w[k] * a[k-m] for m=0,...,n-1
    
    # Gradient w.r.t. a: 2 * h (approximately, for the smooth approximation)
    grad_a = 2.0 * h
    
    # Gradient w.r.t. z: grad_a * a (chain rule through a = softmax(z) / sum... )
    # Actually, a = softmax(z) means a_i = exp(z_i) / sum(exp(z_j))
    # d a_i / d z_m = a_i * (delta_im - a_m)
    # d f / d z_m = sum_i (d f / d a_i) * (d a_i / d z_m)
    # = sum_i grad_a_i * a_i * (delta_im - a_m)

    z_new = z_new - z_new.mean()  # centering for numerical stability
    return z_new, grad_z


def _polyak_step_size(f_current, f_ref, a, direction):
    """Polyak step size: t = (f_current - f_ref) / (f(direction) - f_current) clipped."""
    f_dir = _score_norm(direction)
    denom = f_dir - f_current
    if denom > 1e-10:
        t = (f_current - f_ref) / denom
    else:
        t = 0.5
    return np.clip(t, 0.01, 1.0)
```
