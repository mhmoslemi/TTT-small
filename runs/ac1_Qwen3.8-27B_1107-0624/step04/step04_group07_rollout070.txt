```python
"""
Multi-scale optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.

Strategy:
- Start from the previously provided construction if available, plus diverse
  analytic/random profiles.
- Refine candidates with a softmax-parameterized Adam optimizer for a
  soft-max surrogate of the maximum convolution coefficient.
- Use an exact analytic one-coordinate polish: modifying a single coefficient
  changes the convolution in O(n) time, allowing many local improvements.
- Escalate through sequence lengths (512/1024/2048/3072) and finish with a
  long coordinate polish on the best sequence found.
"""
import numpy as np
import time

try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    _HAVE_SCIPY = True
except Exception:
    _HAVE_SCIPY = False


def _candidate_key(x):
    return x[0]


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _cross_w(w, a):
    """
    Compute g[j] = sum_k w[k] * a[k-j] for valid k-j.
    This is the convolution-gradient direction (up to a constant factor).
    """
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]


def _score_norm(a):
    """Score assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    c = _full_convolve(a)
    if c.size == 0:
        return np.inf
    mm = float(c.max())
    if not np.isfinite(mm):
        return np.inf
    return float(2.0 * a.size * mm)


def _score_seq(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.01:
        return np.inf
    c = _full_convolve(a)
    mm = float(c.max())
    if not np.isfinite(mm):
        return np.inf
    return float(2.0 * a.size * mm / (s * s))


def _soft_max_w(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([], dtype=np.float64)
    mval = float(c.max())
    x = np.clip((c - mval) / beta, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w


def _top_k_weights(c, k):
    c = np.asarray(c, dtype=np.float64)
    L = c.size
    if L == 0:
        return np.array([], dtype=np.float64)
    if k <= 0:
        return np.ones(L, dtype=np.float64) / L
    if k >= L:
        total = float(c.sum())
        if total > 0.0 and np.isfinite(total):
            w = c / total
            if np.all(np.isfinite(w)):
                return w
        return np.ones(L, dtype=np.float64) / L
    idx = np.argpartition(c, -k)[-k:]
    w = np.zeros(L, dtype=np.float64)
    w[idx] = 1.0 / float(k)
    return w


def _softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.array([], dtype=np.float64)
    zm = z - z.max()
    zm = np.clip(zm, -50.0, 50.0)
    p = np.exp(zm)
    p = np.maximum(p, 1e-15)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(z)
        s = float(p.sum())
    return p / s


def _interp(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n_old = a.size
    if n_old == 0:
        return np.ones(n_new, dtype=np.float64) / max(1, n_new)
    if n_old == 1:
        return np.full(n_new, float(a[0]), dtype=np.float64)
    if n_old == n_new:
        b = a.copy()
    else:
        x0 = np.linspace(0.0, 1.0, n_old)
        x1 = np.linspace(0.0, 1.0, n_new)
        b = np.interp(x1, x0, a)
    b = np.maximum(b, 1e-12)
    s = float(b.sum())
    if s <= 0.0 or not np.isfinite(s):
        b = np.ones(n_new, dtype=np.float64)
        s = float(b.sum())
    return b / s


def _make_profile(n, ptype, p1=0.0, p2=0.0, p3=0.0):
    n = max(1, int(n))
    if ptype == 'flat':
        a = np.ones(n, dtype=np.float64)
    elif ptype == 'linear':
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + float(p1) * (2.0 * t - 1.0)
    elif ptype == 'cosine':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + float(p1) * np.cos(2.0 * np.pi * float(p2) * t / n)
    elif ptype == 'sine':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + float(p1) * np.sin(2.0 * np.pi * float(p2) * t / n)
    elif ptype == 'parab':
        t = np.linspace(-1.0, 1.0, n)
        a = 1.0 + float(p1) * (t * t)
    elif ptype == 'U':
        t = np.linspace(-1.0, 1.0, n)
        a = float(p1) * (t * t) + (1.0 - float(p1))
    elif ptype == 'gauss':
        t = np.linspace(0.0, 1.0, n)
        sigma = max(1e-3, abs(float(p1)))
        a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
    elif ptype == 'edge':
        t = np.arange(n, dtype=np.float64) / max(1, n - 1)
        base = max(1e-3, float(p1) + abs(2.0 * t - 1.0))
        a = 1.0 / (base ** max(0.0, float(p2)))
    elif ptype == 'twopeak':
        t = np.linspace(0.0, 1.0, n)
        c1 = float(p1)
        c2 = float(p2)
        sig = max(1e-3, abs(float(p3)))
        a = (np.exp(-0.5 * ((t - c1) / sig) ** 2) +
             np.exp(-0.5 * ((t - c2) / sig) ** 2) + 0.01)
    elif ptype == 'ramp':
        t = np.linspace(0.0, 1.0, n)
        a = np.power(t, max(0.01, float(p1))) + 0.01
    else:
        a = np.ones(n, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        a = np.ones(n, dtype=np.float64)
        s = float(a.sum())
    return a / s


def _get_parent():
    try:
        raw = globals().get('height_sequence_1', None)
        if raw is None:
            return None
        a = np.asarray(raw, dtype=np.float64).ravel()
        if a.size == 0:
            return None
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if not np.isfinite(s) or s <= 0.01:
            return None
        return a / s
    except Exception:
        return None


def _make_candidates(parent, end_time, rng):
    candidates = []

    profile_defs = [
        ('lin05', 'linear', 0.5, 0.0, 0.0),
        ('lin10', 'linear', 1.0, 0.0, 0.0),
        ('lin15', 'linear', 1.5, 0.0, 0.0),
        ('lin20', 'linear', 2.0, 0.0, 0.0),
        ('cos1_1', 'cosine', 0.5, 1, 0.0),
        ('cos1_2', 'cosine', 0.6, 2, 0.0),
        ('cos1_3', 'cosine', 0.7, 3, 0.0),
        ('sin1_2', 'sine', 0.6, 2, 0.0),
        ('sin1_4', 'sine', 0.5, 4, 0.0),
        ('parab1', 'parab', 0.5, 0.0, 0.0),
        ('parab2', 'parab', 1.0, 0.0, 0.0),
        ('U03', 'U', 0.3, 0.0, 0.0),
        ('U05', 'U', 0.5, 0.0, 0.0),
        ('U07', 'U', 0.7, 0.0, 0.0),
        ('gauss03', 'gauss', 0.3, 0.0, 0.0),
        ('gauss04', 'gauss', 0.4, 0.0, 0.0),
        ('edge10', 'edge', 0.1, 1.0, 0.0),
        ('edge15', 'edge', 0.1, 1.5, 0.0),
        ('edge20', 'edge', 0.08, 2.0, 0.0),
        ('two30', 'twopeak', 0.3, 0.7, 0.1),
        ('two25', 'twopeak', 0.25, 0.75, 0.15),
        ('ramp05', 'ramp', 0.5, 0.0, 0.0),
        ('ramp20', 'ramp', 2.0, 0.0, 0.0),
    ]

    n_sizes = [256, 512, 1024, 2048]
    for n in n_sizes:
        for _label, ptype, p1, p2, p3 in profile_defs:
            if time.monotonic() > end_time:
                break
            a = _make_profile(n, ptype, p1, p2, p3)
            sc = _score_norm(a)
            if np.isfinite(sc):
                candidates.append((sc, a, n))
        if time.monotonic() > end_time:
            break

    for n in [1024, 2048]:
        for r in range(5):
            if time.monotonic() > end_time:
                break
            noise = 0.35 + 0.08 * r
            z = rng.normal(size=n) * noise
            z -= z.mean()
            z = np.clip(z, -6.0, 6.0)
            a = _softmax_from_z(z)
            sc = _score_norm(a)
            if np.isfinite(sc):
                candidates.append((sc, a, n))

    if parent is not None:
        pn = parent.size
        for n in [256, 512, 1024, 1536, 2048, 3072]:
            if time.monotonic() > end_time:
                break
            a = _interp(parent, n)
            sc = _score_norm(a)
            if np.isfinite(sc):
                candidates.append((sc, a, n))
        if 16 <= pn <= 4096:
            a = parent.copy()
            a = np.maximum(a, 1e-12)
            s = float(a.sum())
            if s > 0:
                a = a / s
                sc = _score_norm(a)
                if np.isfinite(sc):
                    candidates.append((sc, a, pn))

    if not candidates:
        fallback = _make_profile(512, 'linear', 1.0, 0.0, 0.0)
        candidates.append((_score_norm(fallback), fallback, 512))

    candidates.sort(key=_candidate_key)
    return candidates


def _best_in_range(cands, nmin, nmax):
    best = None
    bs = np.inf
    for sc, a, n in cands:
        if nmin <= n <= nmax and sc < bs:
            bs = float(sc)
            best = (bs, np.asarray(a, dtype=np.float64).copy(), int(n))
    return best


def _choose_track(cands, nmin, nmax, default_n):
    best = _best_in_range(cands, nmin, nmax)
    if best is not None:
        return best
    sc, a, _ = cands[0]
    aa = _interp(a, default_n)
    ss = _score_norm(aa)
    return (ss, aa, int(default_n))


def _build_conv_mat(n, a):
    rows = []
    cols = []
    vals = []
    a = np.asarray(a, dtype=np.float64).ravel()
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_solve(a, M, n):
    if not _HAVE_SCIPY:
        return None
    try:
        A = _build_conv_mat(n, a)
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = float(M) * np.ones(A.shape[0], dtype=np.float64)
        bounds = [(0.0, None)] * n
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            x = np.asarray(res.x, dtype=np.float64)
            x = np.maximum(x, 0.0)
            s = float(x.sum())
            if s > 0.0:
                return x
    except Exception:
        return None
    return None


def _lp_refine(a, end_time, max_iters=8):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0 or n > 256 or not _HAVE_SCIPY:
        return a.copy(), _score_norm(a)
    a = np.maximum(a, 1e-14)
    a = a / a.sum()
    best_a = a.copy()
    best_s = _score_norm(a)
    for _ in range(max_iters):
        if time.monotonic() >= end_time - 1.0:
            break
        c = _full_convolve(a)
        M = float(c.max())
        b = _lp_solve(a, M, n)
        if b is None:
            break
        sb = float(b.sum())
        if sb <= 0.0:
            break
        b = b / sb
        best_t = 0.0
        best_sc = best_s
        for tt in (0.25, 0.5, 0.75, 1.0):
            cand = (1.0 - tt) * a + tt * b
            cand = np.maximum(cand, 1e-14)
            cand = cand / cand.sum()
            sc = _score_norm(cand)
            if sc < best_sc:
                best_sc = sc
                best_t = tt
        if best_t == 0.0:
            break
        a = (1.0 - best_t) * a + best_t * b
        a = np.maximum(a, 1e-14)
        a = a / a.sum()
        sc = _score_norm(a)
        if sc < best_s:
            best_s = sc
            best_a = a.copy()
    return best_a, best_s


def _adam_refine(a, end_time, rng, base_lr=None, label=''):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = max(1, a.size)
    a = np.maximum(a, 1e-15)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        a = np.ones(n, dtype=np.float64)
        s = float(a.sum())
    a = a / s

    z = np.log(a)
    z -= z.mean()
    z_best = z.copy()
    a_best = a.copy()

    c = _full_convolve(a)
    score = float(2.0 * n * c.max())
    if not np.isfinite(score):
        score = np.inf
    best_score = score

    if base_lr is None:
        base_lr = 0.14 * (512.0 / float(n)) ** 0.25
    lr = float(base_lr)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    t = 0

    medc = float(np.median(c))
    scale = max(1e-10, float(c.max()) - medc)
    beta = max(1e-9, 0.8 * scale)
    no_imp = 0

    while time.monotonic() < end_time:
        if not np.isfinite(score) or not np.all(np.isfinite(a)):
            z = z_best.copy()
            a = a_best.copy()
            c = _full_convolve(a)
            score = float(2.0 * n * c.max())
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            lr = float(base_lr)
            no_imp = 0
            continue

        if np.isfinite(score) and score < best_score - 1e-13:
            best_score = float(score)
            z_best = z.copy()
            a_best = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > 130:
            ns = 0.09 if best_score < 1.51 else 0.16
            z = z_best + ns * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z -= z.mean()
            a = _softmax_from_z(z)
            c = _full_convolve(a)
            score = float(2.0 * n * c.max())
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            lr = float(base_lr)
            medc = float(np.median(c))
            scale = max(1e-10, float(c.max()) - medc)
            beta = max(1e-9, 0.8 * scale)
            no_imp = 0
            continue

        if t % 10 == 0:
            medc = float(np.median(c))
            scale = max(1e-10, float(c.max()) - medc)
            if no_imp > 35:
                beta *= 1.5
            elif no_imp > 12:
                beta *= 1.1
            else:
                beta *= 0.985
            beta = min(max(beta, 0.05 * scale), 1.5 * scale)

        use_top = (t % 4 == 3) or (no_imp > 8)
        if use_top:
            k = max(8, min(1024, len(c) // 64 + min(128, no_imp)))
            wt = _top_k_weights(c, k)
            ws, _ = _soft_max_w(c, beta)
            w = 0.55 * wt + 0.45 * ws
            sw = float(w.sum())
            if sw <= 0.0 or not np.isfinite(sw):
                w = np.ones_like(c) / max(1, c.size)
            else:
                w = w / sw
        else:
            _, w = _soft_max_w(c, beta)

        g = _cross_w(w, a)
        if not np.all(np.isfinite(g)):
            lr = max(float(base_lr) * 0.2, lr * 0.8)
            continue

        dot = float(np.dot(a, g))
        gz = a * (g - dot)
        gz = np.where(np.isfinite(gz), gz, 0.0)
        gz = np.clip(gz, -1.0, 1.0)

        t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        step = lr * mh / (np.sqrt(vh) + eps)
        step = np.clip(step, -0.3, 0.3)

        accepted = False
        for scale_step in (1.0, 0.5, 0.25, 0.1):
            zt = z - scale_step * step
            zt = np.clip(zt, -40.0, 40.0)
            zt = zt - zt.mean()
            at = _softmax_from_z(zt)
            ct = _full_convolve(at)
            st = float(2.0 * n * ct.max())
            if np.isfinite(st) and st <= score + 1e-12:
                z = zt
                a = at
                c = ct
                score = st
                accepted = True
                break

        if not accepted:
            lr = max(float(base_lr) * 0.2, lr * 0.8)
        else:
            if np.isfinite(score) and score < best_score - 1e-13:
                best_score = float(score)
                z_best = z.copy()
                a_best = a.copy()
                no_imp = 0
                lr = min(base_lr * 2.0, lr * 1.02)

    return a_best, best_score


def _coord_gradient(a, c):
    n = len(a)
    medc = float(np.median(c))
    scale = max(1e-10, float(c.max()) - medc)
    beta = max(1e-10, 0.8 * scale)
    ws, _ = _soft_max_w(c, beta)
    k = max(8, min(512, max(1, len(c)) // 64))
    wt = _top_k_weights(c, k)
    w = 0.5 * ws + 0.5 * wt
    sw = float(w.sum())
    if sw <= 0.0 or not np.isfinite(sw):
        w = np.ones_like(c) / max(1, c.size)
    else:
        w = w / sw
    g = _cross_w(w, a)
    dot = float(np.dot(a, g))
    gz = a * (g - dot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    return gz


def _try_coord_best(a, c, j, factors):
    n = a.size
    old = float(a[j])
    if old <= 0.0 or not np.isfinite(old):
        return None
    best_fac = 1.0
    best_s = np.inf
    best_c2 = None
    best_ssum = 1.0
    best_new = old
    for fac in factors:
        fac = float(fac)
        newv = old * fac
        if newv <= 0.0 or not np.isfinite(newv):
            continue
        delta = newv - old
        if abs(delta) < 1e-18:
            continue
        ssum = 1.0 + delta
        if ssum <= 0.0 or not np.isfinite(ssum):
            continue
        c2 = c.copy()
        c2[j:j + n] += (2.0 * delta) * a
        if 2 * j < c2.size:
            c2[2 * j] += delta * delta
        mx = float(c2.max())
        if not np.isfinite(mx) or mx < 0.0:
            mx = 0.0
        sc = 2.0 * n * mx / (ssum * ssum)
        if np.isfinite(sc) and sc < best_s:
            best_s = sc
            best_fac = fac
            best_c2 = c2
            best_ssum = ssum
            best_new = newv
    if best_fac == 1.0 or best_c2 is None:
        return None
    return (best_new, best_ssum, best_c2, best_s)


def _apply_coord_move(a, c, j, move):
    newv, ssum, c2, sc = move
    a[j] = newv
    a = a / a.sum()
    c = c2
    c = c / (ssum * ssum)
    np.maximum(c, 0.0, out=c)
    return a, c, sc


def _coord_polish(a, end_time, rng, label=''):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return a, np.inf
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    c = _full_convolve(a)
    best_score = float(2.0 * n * c.max())
    if not np.isfinite(best_score):
        best_score = np.inf
    best_a = a.copy()

    while time.monotonic() < end_time - 1.0:
        gz = _coord_gradient(a, c)
        if not np.all(np.isfinite(gz)):
            c = _full_convolve(a)
            gz = _coord_gradient(a, c)

        order = np.argsort(np.abs(gz))[::-1]
        improved = False
        accepts = 0
        for ii in order:
            if time.monotonic() > end_time - 0.5:
                break
            j = int(ii)
            old = float(a[j])
            if old <= 1e-14:
                continue
            if gz[j] > 1e-9:
                factors = (0.8, 0.65, 0.5)
            elif gz[j] < -1e-9:
                factors = (1.25, 1.5, 2.0)
            else:
                factors = (0.9, 1.1)
            move = _try_coord_best(a, c, j, factors)
            if move is None:
                continue
            a, c, sc = _apply_coord_move(a, c, j, move)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a.copy()
            improved = True
            accepts += 1

            if accepts >= 256:
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                c = _full_convolve(a)
                cur = float(2.0 * n * c.max())
                if np.isfinite(cur):
                    if cur < best_score:
                        best_score = cur
                        best_a = a.copy()
                accepts = 0

        if not improved:
            if time.monotonic() > end_time - 5.0:
                break
            nrand = min(n, 512)
            idxs = rng.choice(n, size=nrand, replace=False)
            any_imp = False
            for jj in idxs:
                if time.monotonic() > end_time - 0.5:
                    break
                j = int(jj)
                old = float(a[j])
                if old <= 1e-14:
                    continue
                move = _try_coord_best(a, c, j, (0.7, 1.4, 0.5, 1.7))
                if move is None:
                    continue
                a, c, sc = _apply_coord_move(a, c, j, move)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a.copy()
                any_imp = True
            if not any_imp:
                break

    return best_a, best_score


def _finalize_sequence(a, fallback):
    if a is None:
        return fallback
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0 or not np.all(np.isfinite(a)):
        return fallback
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return fallback
    a = a / s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Returns a non-negative coefficient list minimizing the target score.
    """
    t_start = time.monotonic()
    try:
        budget = max(60.0, float(budget_s))
    except Exception:
        budget = 1800.0
    deadline = t_start + budget - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    parent = _get_parent()
    fallback = parent.copy() if parent is not None else _make_profile(512, 'linear', 1.0, 0.0, 0.0)

    gen_end = t_start + 70.0
    if gen_end > deadline - 300.0:
        gen_end = max(t_start + 20.0, deadline - 300.0)
    candidates = _make_candidates(parent, gen_end, rng)

    if not candidates:
        return [float(x) for x in fallback]

    best_a = np.asarray(candidates[0][1], dtype=np.float64).copy()
    best_s = float(candidates[0][0])
    print(f"Initial best score={best_s:.10f} n={best_a.size}")

    small = _choose_track(candidates, 0, 512, 512)
    med = _choose_track(candidates, 513, 1536, 1024)
    large = _choose_track(candidates, 1537, 3072, 2048)

    small_result_a = None
    med_result_a = None

    # Small track: optional LP + Adam + coordinate polish.
    if time.monotonic() < deadline - 900.0:
        end_small = min(deadline - 900.0, time.monotonic() + 180.0)
        cur_a = np.asarray(small[1], dtype=np.float64).copy()
        cur_s = float(small[0])
        cur_n = int(small[2])

        if _HAVE_SCIPY and cur_n <= 192:
            lp_end = min(end_small - 80.0, time.monotonic() + 30.0)
            if lp_end > time.monotonic() + 5.0:
                a_lp, s_lp = _lp_refine(cur_a, lp_end, 6)
                if s_lp < cur_s:
                    cur_a = a_lp
                    cur_s = s_lp

        if end_small - time.monotonic() > 20.0:
            a_ad, s_ad = _adam_refine(cur_a, end_small, rng, base_lr=0.12, label='small')
            if s_ad < cur_s:
                cur_a = a_ad
                cur_s = s_ad

        if end_small - time.monotonic() > 10.0:
            a_co, s_co = _coord_polish(cur_a, end_small, rng, label='smallcoord')
            if s_co < cur_s:
                cur_a = a_co
                cur_s = s_co

        small_result_a = cur_a.copy()
        if cur_s < best_s:
            best_s = cur_s
            best_a = cur_a.copy()
            print(f"Small track score={best_s:.10f} n={best_a.size}")

    # Medium track: Adam + coordinate polish.
    if time.monotonic() < deadline - 700.0:
        end_med = min(deadline - 700.0, time.monotonic() + 520.0)
        med_n = int(med[2])
        cur_a = np.asarray(med[1], dtype=np.float64).copy()
        cur_s = float(med[0])

        if best_a is not None and best_s < cur_s + 0.02:
            inter = _interp(best_a, med_n)
            sc = _score_norm(inter)
            if np.isfinite(sc) and sc < cur_s:
                cur_a = inter
                cur_s = sc

        if end_med - time.monotonic() > 30.0:
            a_ad, s_ad = _adam_refine(cur_a, end_med, rng, base_lr=None, label='med')
            if s_ad < cur_s:
                cur_a = a_ad
                cur_s = s_ad

        if end_med - time.monotonic() > 20.0:
            a_co, s_co = _coord_polish(cur_a, end_med, rng, label='medcoord')
            if s_co < cur_s:
                cur_a = a_co
                cur_s = s_co

        med_result_a = cur_a.copy()
        if cur_s < best_s:
            best_s = cur_s
            best_a = cur_a.copy()
            print(f"Medium track score={best_s:.10f} n={best_a.size}")

    # Large track: start from best available interpolation, then refine.
    if time.monotonic() < deadline - 150.0:
        end_large = min(deadline - 150.0, time.monotonic() + 900.0)
        large_n = int(large[2])
        cur_a = np.asarray(large[1], dtype=np.float64).copy()
        cur_s = float(large[0])

        pool = [best_a, small_result_a, med_result_a]
        for src in pool:
            if src is None:
                continue
            src = np.asarray(src, dtype=np.float64).ravel()
            if src.size == 0:
                continue
            if src.size != large_n:
                inter = _interp(src, large_n)
                sc = _score_norm(inter)
                if np.isfinite(sc) and sc < cur_s:
                    cur_a = inter
                    cur_s = sc

        if end_large - time.monotonic() > 30.0:
            a_ad, s_ad = _adam_refine(cur_a, end_large, rng, base_lr=None, label='large')
            if s_ad < cur_s:
                cur_a = a_ad
                cur_s = s_ad

        if cur_s < best_s:
            best_s = cur_s
            best_a = cur_a.copy()
            print(f"Large track score={best_s:.10f} n={best_a.size}")

    # Final coordinate polish on the global best.
    if best_a is not None and time.monotonic() < deadline - 10.0:
        end_final = deadline - 5.0
        a_f, s_f = _coord_polish(best_a, end_final, rng, label='finalcoord')
        if s_f < best_s:
            best_s = s_f
            best_a = a_f
            print(f"Final coord score={best_s:.10f} n={best_a.size}")

    final_a = _finalize_sequence(best_a, fallback)
    final_score = _score_seq(final_a)
    print(f"FINAL internal score={final_score:.10f} n={final_a.size}")

    try:
        if 'evaluate_sequence' in globals():
            ev = float(evaluate_sequence([float(x) for x in final_a]))
            if np.isfinite(ev):
                print(f"FINAL evaluate_sequence={ev:.10f}")
    except Exception as e:
        print(f"evaluate_sequence check failed: {e}")

    return [float(x) for x in final_a]
```
