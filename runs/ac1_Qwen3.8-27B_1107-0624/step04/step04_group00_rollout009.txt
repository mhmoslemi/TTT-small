def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):

    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

                         'maxls': 50, 'maxfun': maxiter_per_beta * 3}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _run_active_gd(a_init, deadline, K_start=1, K_max=50, lr_init=0.1, max_iter=100000):
    """Active-set gradient descent with Adam and adaptive K."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Adaptive K: start small, grow as peaks become more balanced
        sorted_c = np.sort(c)[::-1]
        # K = number of peaks within 1% of the max
        K = max(K_start, int(np.sum(c > 0.999 * cm)))
        K = min(K, K_max)
        
        # Compute active-set gradient
        top_K = np.argsort(c)[::-1][:K]
        w = np.zeros_like(c)
        w[top_K] = 1.0 / K
        
        a_rev = a[::-1]
        conv_wb = _full_convolve(w, a_rev)
        h = conv_wb[n-1:2*n-1]
        
        da = 4.0 * n * h
        mean_da = float(np.dot(a, da))
        grad = a * (da - mean_da)
        
        if not np.all(np.isfinite(grad)):
            break
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        
        # Learning rate with decay
        lr = lr_init * (0.999 ** (t // 100))
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        
        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score
