def solve_lp_direction(f, M):

def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a, b):

    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

"""
Multi-phase optimizer: mirror descent + LP direction + local search.
Minimizes 2*n*max(self_convolve(a,a))/sum(a)^2 for non-negative a.
"""
import numpy as np
import time
import scipy.sparse as sp_dense

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * np.max(c))

def _score_full(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64).copy()
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _subgradient(a, w):
    """Compute subgradient g_j = sum_k w_k (a_{k-j} + a_{j-k})."""
    n = len(a)
    # term1_j = sum_k w_k * a_{k-j} = (w * a_flip)[n-1+j] for j=0..n-1
    # where a_flip = a[::-1]
    a_flip = a[::-1]
    # convolve(w, a_flip) has length len(w) + n - 1
    conv1 = _full_convolve(w, a_flip)
    term1 = conv1[n-1:2*n-1]  # indices n-1 to 2n-2 (n values)
    
    # term2_j = sum_k w_k * a_{j-k} = (w * a)[j] for j=0..n-1
    conv2 = _full_convolve(w, a)
    term2 = conv2[0:n]
    
    return term1 + term2

def _mirror_descent(a_init, deadline, rng, n_max=500000, tau0=0.05, tau1=0.001, eta0=0.5):
    """Mirror descent with annealing temperature."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    # Estimate max iterations

    c0 = _full_convolve(a, a)
    _ = np.max(c0)
    elapsed_1 = time.monotonic()
    time_per_iter = max(0.0005, (elapsed_1 - t_start) / 2)  # rough estimate
    n_iters = min(n_max, max(1000, int((deadline - t_start) / time_per_iter)))
    
    it = 0
    while time.monotonic() < deadline - 1.0 and it < n_iters:
        it += 1
        prog = it / n_iters
        tau = tau0 * (tau1 / tau0) ** prog
        eta = eta0 * (1.0 + 0.5 * (1.0 - prog))  # slightly decreasing
        
        c = _full_convolve(a, a)
        M = float(np.max(c))
        
        # Softmax weights (temperature-scaled)
        c_shifted = (c - M) / max(tau, 1e-10)
        e = np.exp(c_shifted)
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = e / s

        # Mirror descent update (exponential weights)
        z = np.log(np.maximum(a, 1e-30)) - eta * g
        z = z - np.max(z)
        a_new = np.exp(z)

        a_new = np.maximum(a_new, 1e-30)

        a = a_new

            sc = _score_norm(a)
            if sc < best_score:

                best_a = a.copy()

    if sc < best_score:

Now let me also think about the LP approach.

For the LP:
- Variables: $b \in \mathbb{R}^n$, $b \geq 0$.
- Objective: maximize $\mathbf{1}^T b$.
- Constraints: $(a * b)_k \leq M$ for selected lags $k$.

$(a * b)_k = \sum_i a_i b_{k-i}$ (with zero extension).

For a fixed lag $k$, this is $\sum_i a_i b_{k-i} = \sum_j a_{k-j} b_j$ (reindexing with $j = k-i$).

So the constraint for lag $k$ is: $\sum_j C[k,j] b_j \leq M$ where $C[k,j] = a_{k-j}$ if $0 \leq k-j \leq n-1$, else 0.

To build the sparse matrix $C$:
- $C$ is $(2n-1) \times n$.
- $C[k,j] = a[k-j]$ if $0 \leq k-j \leq n-1$.
- For a fixed $j$: nonzero for $k$ in $[j, j+n-1]$, with values $a[k-j] = a[0], a[1], \ldots, a[n-1]$.
- So column $j$ has nonzeros at rows $j, j+1, \ldots, j+n-1$, with values $a[0], a[1], \ldots, a[n-1]$.

This is a band matrix! Each column is a shifted copy of $a$.

Total nonzeros: $n \times n = n^2$ (each column has $n$ nonzeros). For $n=4096$, that's 16.8M nonzeros. That's a lot for an LP solver.

To reduce: only include "active" constraints. At each iteration, keep only lags where $(a*a)_k > M(1-\delta)$ for some $\delta$ (e.g., 0.05 or 0.1). This might reduce to 50-500 constraints.

With, say, 200 constraints, the LP is 200 x 4096, which is very manageable.

"""
Multi-phase optimizer: FFT-based mirror descent with smoothed subgradients,
LP-based direction finding (iterative linearization), and local search.
Exploits the non-symmetric structure of self-convolution to beat symmetric baselines.
Key insight: subgradient of max(a*a) at lag k is 2*a_{k-j}, enabling O(n log n) iterations.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * np.max(c))

def _score_full(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64).copy()
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _compute_subgradient(a, w):
    """Compute g_j = 2 * sum_k w_k * a_{k-j} (with zero extension).
    This equals 2 * cross_correlation(w, a)[j] for j=0..n-1."""
    n = len(a)
    # Cross-correlation: (w star a)[j] = sum_i w[i+j] * a[i]
    # = ifft(rfft(w, N) * conj(rfft(a, N)))[j]
    N = _next_pow2(3 * n - 2)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    cross = np.fft.irfft(fw * np.conj(fa), N)
    return 2.0 * cross[:n]

def _mirror_descent(a_init, deadline, tau0=0.05, tau1=0.001, eta=0.8, check_interval=200):
    """Mirror descent (exponential weights) with annealed temperature."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    # Precompute FFT size
    N_fft = _next_pow2(3 * n - 2)
    
    it = 0
    while time.monotonic() < deadline - 1.0:
        it += 1
        prog = min(1.0, it / 100000.0)
        tau = max(tau1, tau0 * (tau1 / tau0) ** prog)
        
        c = _full_convolve(a, a)
        M = float(np.max(c))
        
        # Softmax weights
        c_shifted = (c - M) / max(tau, 1e-12)
        e = np.exp(c_shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        
        # Subgradient via FFT
        fw = np.fft.rfft(w, N_fft)
        fa = np.fft.rfft(a, N_fft)
        cross = np.fft.irfft(fw * np.conj(fa), N_fft)
        g = 2.0 * cross[:n]
        
        # Exponential weights update
        z = np.log(np.maximum(a, 1e-30)) - eta * g
        z_max = np.max(z)
        if not np.isfinite(z_max):
            break
        z = z - z_max
        a_new = np.exp(z)
        s_new = a_new.sum()
        if s_new < 1e-300 or not np.isfinite(s_new):
            break
        a = a_new / s_new
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        
        if it % check_interval == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score

def _lp_direction(a, M, n_active=None):
    """Solve LP: max 1^T b s.t. (a*b)_k <= M for active k, b >= 0.
    Returns normalized direction or None on failure."""
    try:
        import cvxpy as cp
    except ImportError:
        return None
    
    n = len(a)
    c = _full_convolve(a, a)
    
    if n_active is None:
        # Use top constraints
        n_active = min(300, len(c))
    
    # Select active lags (where convolution is near the max)
    threshold = M * 0.9
    active = np.where(c >= threshold)[0]
    if len(active) < 20:
        active = np.argsort(c)[-n_active:]
    elif len(active) > n_active:
        # Keep the most active ones
        active_sorted = active[np.argsort(c[active])][-n_active:]
        active = active_sorted
    
    # Build constraint matrix (dense, since n_active * n is manageable for n_active <= 300)
    A_const = np.zeros((len(active), n))
    for idx, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # For j in [lo, hi]: coefficient is a[k-j]
            for j in range(lo, hi + 1):
                A_const[idx, j] = a[k - j]
    
    b = cp.Variable(n, nonneg=True)
    constraints = [A_const @ b <= M]
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    
    # Try solvers in order of preference
    solvers_to_try = ['CLARABEL', 'SCS', 'ECOS']
    for solver in solvers_to_try:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue
    
    if b.value is None or not np.all(np.isfinite(b.value)):
        return None
    
    b_val = np.maximum(b.value, 0.0)
    s = b_val.sum()
    if s < 1e-10:
        return None
    return b_val / s

def _adam_smoothed(a_init, deadline, lr=0.01, k_top=5, n_max=200000):
    """Adam with top-K smoothed gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15)).astype(np.float64)
    z = z - z.mean()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_a = _softmax_fn(z)
    best_score = _score_norm(best_a)
    
    N_fft = _next_pow2(3 * n - 2)
    
    for it in range(n_max):
        if time.monotonic() > deadline - 1.0:
            break
        t += 1
        
        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        M = float(np.max(c))
        
        # Top-K weights
        k_act = min(k_top, len(c))
        top_idx = np.argsort(c)[-k_act:]
        w = np.zeros(len(c))
        w[top_idx] = 1.0 / k_act
        
        # Subgradient
        fw = np.fft.rfft(w, N_fft)
        fa = np.fft.rfft(a_curr, N_fft)
        cross = np.fft.irfft(fw * np.conj(fa), N_fft)
        grad = 2.0 * cross[:n]
        
        # Adam update on log-scale
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        if t % 50 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
    
    a_final = _softmax_fn(z)
    sc = _score_norm(a_final)
    if sc < best_score:
        best_score = sc
        best_a = a_final.copy()
    
    return best_a, best_score

def _local_search(a_init, deadline, rng, n_trials=5000):
    """Block-based local search with multiplicative perturbations."""
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score_norm(a_init)
    a = a / s
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break
        
        # Choose block to perturb
        block_size = int(rng.integers(3, min(30, n // 4)))
        start = int(rng.integers(0, n - block_size))
        
        a_trial = a.copy()
        perturbation = rng.uniform(0.8, 1.2, size=block_size)
        a_trial[start:start + block_size] *= perturbation
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial = a_trial / a_trial.sum()
        
        sc = _score_norm(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
        elif rng.random() < 0.02 and sc < best_score * 1.005:
            a = a_trial.copy()
    
    return best_a, best_score

def _make_profile(n, p_type, **kwargs):
    """Generate various profile shapes."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    
    if p_type == 'linear':
        slope = kwargs.get('slope', 1.5)
        a = 1.0 + slope * (t - 0.5)
    elif p_type == 'quadratic':
        a1 = kwargs.get('a1', 1.0)
        a2 = kwargs.get('a2', 0.2)
        a = 1.0 + a1 * u + a2 * u ** 2
    elif p_type == 'cubic':
        a1 = kwargs.get('a1', 1.0)
        a3 = kwargs.get('a3', 0.2)
        a = 1.0 + a1 * u + a3 * u ** 3
    elif p_type == 'gaussian':
        width = kwargs.get('width', 0.3)
        a = np.exp(-u ** 2 / (2.0 * width ** 2))
    elif p_type == 'exponential':
        rate = kwargs.get('rate', 2.0)
        a = np.exp(rate * u)
    elif p_type == 'sigmoid':
        sharp = kwargs.get('sharp', 5.0)
        a = 1.0 / (1.0 + np.exp(-sharp * u))
    elif p_type == 'trapezoid':
        alpha = kwargs.get('alpha', 0.3)
        a = np.clip(1.0 - np.abs(u) / (1.0 - alpha), 0.1, 1.0)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

    p3_end = min(deadline, t_start + 900.0)
    if time.monotonic() < p3_end:
        # Use lower temperature for finer optimization
        a_m2, s_m2 = _mirror_descent(best_seq, p3_end, tau0=0.02, tau1=0.002, eta=0.4)
        update_best(a_m2, s_m2, "mirror_descent_2")
    
    # === Phase 4: Adam smoothed (200s) ===
    print(f"\n=== Phase 4: Adam Smoothed ===")
    p4_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < p4_end:
        a_adam, s_adam = _adam_smoothed(best_seq, p4_end, lr=0.015, k_top=3)
        update_best(a_adam, s_adam, "adam_k3")
        
        if time.monotonic() < p4_end - 30:
            a_adam2, s_adam2 = _adam_smoothed(best_seq, p4_end, lr=0.008, k_top=8)
            update_best(a_adam2, s_adam2, "adam_k8")
    
    # === Phase 5: Try larger n (300s) ===
    print(f"\n=== Phase 5: Larger n ===")
    p5_end = min(deadline, t_start + 1400.0)
    n_curr = len(best_seq)
    
    for target_n in [8192, 6144]:
        if time.monotonic() > p5_end - 60:
            break
        if target_n <= n_curr:
            continue
        
        print(f"  Trying n={target_n}...")
        p_interp = _interp_profile(best_seq, target_n)
        sc_interp = _score_norm(p_interp)
        update_best(p_interp, sc_interp, f"interp_n{target_n}")
        
        # Mirror descent at new n
        sub_deadline = min(p5_end - 30, time.monotonic() + 150.0)
        a_new, s_new = _mirror_descent(p_interp, sub_deadline, tau0=0.05, tau1=0.005, eta=0.5)
        update_best(a_new, s_new, f"mirror_n{target_n}")
        
        # Also try different profiles at this n
        if time.monotonic() < p5_end - 30:
            for ptype, kwargs in [('exponential', {'rate': 1.0}), ('exponential', {'rate': 2.0}),
                                   ('sigmoid', {'sharp': 4.0}), ('trapezoid', {'alpha': 0.4})]:
                p_try = _make_profile(target_n, ptype, **kwargs)
                sc_try = _score_norm(p_try)
                if sc_try < best_score * 1.01:
                    sub_dl = min(p5_end - 10, time.monotonic() + 30.0)
                    a_t, s_t = _mirror_descent(p_try, sub_dl, tau0=0.05, tau1=0.005, eta=0.5)
                    update_best(a_t, s_t, f"{ptype}_n{target_n}")
    
    # === Phase 6: Multi-restart (200s) ===
    print(f"\n=== Phase 6: Multi-Restart ===")
    p6_end = min(deadline, t_start + 1600.0)
    n_restarts = 4
    for i in range(n_restarts):
        if time.monotonic() > p6_end - 10:
            break
        
        noise = 0.05 + 0.03 * i
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        z = z + noise * rng.normal(size=len(best_seq))
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        p_start = _softmax_fn(z)
        
        sub_dl = min(p6_end - 5, time.monotonic() + 40.0)
        
        # Alternate between mirror descent and LP
        if i % 2 == 0:
            a_r, s_r = _mirror_descent(p_start, sub_dl, tau0=0.03, tau1=0.003, eta=0.5)
        else:
            a_r, s_r = _lp_refinement(p_start, sub_dl, n_max_iters=10)
        
        update_best(a_r, s_r, f"restart{i}")
    
    # === Phase 7: Final LP polish + local search (remaining) ===
    print(f"\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        remaining = max(10.0, deadline - time.monotonic())
        
        # LP polish
        lp_dl = min(deadline - 10, time.monotonic() + remaining * 0.4)
        if time.monotonic() < lp_dl:
            a_final, s_final = _lp_refinement(best_seq, lp_dl, n_max_iters=30)
            update_best(a_final, s_final, "final_lp")
        
        # Local search
        ls_dl = min(deadline - 5, time.monotonic() + remaining * 0.3)
        if time.monotonic() < ls_dl:
            a_ls, s_ls = _local_search(best_seq, ls_dl, rng, n_trials=3000)
            update_best(a_ls, s_ls, "local_search")
        
        # Final mirror descent (very fine)
        if time.monotonic() < deadline - 5:
            a_fine, s_fine = _mirror_descent(best_seq, deadline - 3, tau0=0.01, tau1=0.001, eta=0.3)
            update_best(a_fine, s_fine, "final_fine")
    
    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.maximum(best_seq, 0.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_profile(4096, 'exponential', rate=1.5)
            s = float(final_seq.sum())
        # Keep sum = 1 for clean normalization (score is scale-invariant)
        final_seq = final_seq / s
        
        sc_final = _score_full(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")
        
        # Validate with evaluator
        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    print("FALLBACK: using exponential profile")
    fallback = _make_profile(4096, 'exponential', rate=1.5)
    return [float(x) for x in fallback]
