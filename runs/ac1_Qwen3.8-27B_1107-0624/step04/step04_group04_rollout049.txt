    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

"""
Active-set guided optimization using p-norm smoothing, mirror descent,
and adaptive L-BFGS refinement. Minimizes 2*n*max(conv(a,a))/sum(a)^2
over non-negative sequences by combining fast multiplicative gradient
steps with periodic second-order refinement.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _cross_corr_weights(a, w):
    """Compute g[i] = sum_j a[j] * w[i+j] for i=0..n-1 using FFT."""
    n = len(a)
    N = _next_pow2(n + len(w) - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fw * np.conj(fa), N)
    return result[:n]


def _score_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (a.sum() ** 2))


def _pnorm_smoothed_obj(a, p, top_k=0):
    """Compute p-norm smoothed objective and gradient direction."""
    n = len(a)
    b = _full_convolve(a, a)
    b = np.maximum(b, 0)  # b should be non-negative for non-negative a
    
    if top_k > 0 and top_k < len(b):
        idx = np.argsort(b)[-top_k:]
        b_sel = b[idx]
        bp = b_sel ** p
        norm_p = bp.sum() ** (1.0 / p)
        if norm_p < 1e-300:
            return 0.0, np.zeros(n)
        w_full = np.zeros_like(b)
        w_full[idx] = bp / bp.sum()
    else:
        bp = b ** p
        norm_p = bp.sum() ** (1.0 / p)
        if norm_p < 1e-300:
            return 0.0, np.zeros(n)
        w_full = bp / bp.sum()
    
    # Gradient direction: g[i] = sum_j a[j] * w[i+j]
    g = _cross_corr_weights(a, w_full)
    obj = 2.0 * n * norm_p / (a.sum() ** 2)
    return obj, g


def _mirror_step(a, g, eta):
    """Multiplicative update: a_i <- a_i * exp(-eta * g_i), renormalized."""
    g_centered = g - g.mean()
    log_a = np.log(np.maximum(a, 1e-300))
    log_a -= log_a.mean()
    log_a = log_a - eta * g_centered
    log_a = np.clip(log_a, -80, 80)
    a_new = np.exp(log_a)
    a_new = a_new / a_new.sum()
    return a_new


def _backtracking_mirror(a, p, top_k, eta_init=0.5, max_iter=50):
    """Mirror descent with backtracking line search."""
    n = len(a)
    obj0, g = _pnorm_smoothed_obj(a, p, top_k)
    eta = eta_init
    best_a = a.copy()
    best_obj = obj0
    
    for it in range(max_iter):
        a_new = _mirror_step(a, g, eta)
        obj_new, _ = _pnorm_smoothed_obj(a_new, p, top_k)
        if np.isfinite(obj_new) and obj_new < obj0:
            best_a = a_new.copy()
            best_obj = obj_new
            return best_a, best_obj
        eta *= 0.5
        if eta < 1e-10:
            break
    
    return best_a, best_obj


def _run_mirror_descent(a_init, deadline, p_start=20, p_end=500, top_k_init=50, top_k_end=5):

    stalled = 0

            best_obj = obj_new
            a = a_new

def _kkt_direction(a, active_set):
    """Compute the KKT balancing direction for given active set."""
    n = len(a)
    # h[i] = mean_{k in A} a[k-i] (for valid indices)
    h = np.zeros(n)
    for k_idx in active_set:
        k_val = int(k_idx)
        lo = max(0, k_val - (n - 1))
        hi = min(n - 1, k_val)
        if lo <= hi:
            h[lo:hi + 1] += a[k_val - hi:k_val - lo + 1][::-1]
    h /= len(active_set)
    return h


def _kkt_iterate(a, deadline, tol=0.005, max_iter=5000):
    """KKT-based balancing: equalize active peaks."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        cm = float(c.max())
        threshold = cm * (1.0 - tol)
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0 or len(active) > 2 * n:
            tol = max(0.001, tol * 0.5)
            active = np.where(c >= cm * 0.99)[0]
        
        h = _kkt_direction(a, active)
        spread = float(np.max(h) - np.min(h))
        
        if spread < 1e-12:
            break
        
        # Multiplicative step: reduce a where h > mean, increase where h < mean
        eta = 0.1 * cm / (spread + 1e-15)
        eta = min(eta, 0.5)
        
        a_new = _mirror_step(a, h, eta)
        sc_new = _score_norm(a_new)
        
        if np.isfinite(sc_new) and sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            eta /= 10
            a_new = _mirror_step(a, h, eta)
            sc_new = _score_norm(a_new)
            if np.isfinite(sc_new) and sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
            else:
                break
    
    return best_a, best_score


def _make_profiles(n):
    """Generate diverse initial profiles."""
    profiles = []
    # Exponential
    for alpha in [0.001, 0.003, 0.008, 0.02, 0.05]:
        r = 1.0 - alpha
        a = r ** np.arange(n)
        profiles.append(a / a.sum())
    # Symmetric bump
    x = np.linspace(-1, 1, n)
    for width in [0.3, 0.5, 0.7, 1.0]:
        a = np.exp(-x ** 2 / (2 * width ** 2))
        profiles.append(a / a.sum())
    # Power
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, -0.5, -1.0]:
        a = (np.arange(n) + 1.0) ** p
        profiles.append(a / a.sum())
    # Sine modulated
    for freq in [1, 2, 3, 5]:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        profiles.append(a / a.sum())
    # Uniform
    profiles.append(np.ones(n) / n)
    return profiles


def _adam_refine(a_init, deadline, lr=0.05, p_start=50, top_k=20):
    """Adam optimizer in logit space with p-norm."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    p = p_start
    
    while time.monotonic() < deadline:
        t += 1
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        # Compute p-norm gradient
        b = _full_convolve(a, a)
        if top_k > 0 and top_k < len(b):
            idx = np.argsort(b)[-top_k:]
            b_sel = b[idx]
            bp = b_sel ** (p - 1)
            w = np.zeros_like(b)
            w[idx] = bp / (bp.sum() + 1e-300)
        else:
            bp = b ** (p - 1)
            w = bp / (bp.sum() + 1e-300)
        
        h = _cross_corr_weights(a, w)
        s = a.sum()
        grad = 2.0 * n * a * h - 2.0 * np.dot(w, b) * a / s
        grad = grad - grad.mean()
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        if t % 50 == 0:
            a_cur = np.exp(z - z.max())
            a_cur = a_cur / a_cur.sum()
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
            
            # Slowly increase p
            p = min(500.0, p * 1.01)
            # Reduce lr if close to convergence
            if best_score < 1.51:
                lr = max(0.005, lr * 0.995)
    
    # Final check
    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc = _score_norm(a_final)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a_final.copy()
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining p-norm mirror descent, Adam refinement,
    KKT balancing, and L-BFGS for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    """
    global height_sequence_1
    
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")
    
    # --- Load parent ---
    print("=== Loading parent ===")
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm(parent)
                update_best(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent: {e}")
    
    # --- Phase 0: Initial L-BFGS from parent ---
    print("=== Phase 0: Initial L-BFGS ===")
    phase0_end = min(deadline, t_start + 60.0)
    
    if parent is not None:
        n_par = len(parent)
        p_best, sc_best = _lbfgs_refine(parent, phase0_end, beta=1e-3, top_k=20)
        update_best(p_best, sc_best, "lbfgs0")
    
    # Also try some profiles if we have time
    if time.monotonic() < phase0_end - 10:
        n_try = 4096
        profiles = _make_profiles(n_try)
        for i, pf in enumerate(profiles[:4]):
            if time.monotonic() > phase0_end - 5:
                break
            sc = _score_norm(pf)
            update_best(pf, sc, f"prof{i}")
    
    print(f"  Phase 0 done: best={best_score:.10f}")
    
    if best_seq is None:
        best_seq = parent if parent is not None else _make_profiles(4096)[0]
        best_score = _score_norm(best_seq)
    
    # --- Phase 1: Mirror descent (main workhorse) ---
    print("=== Phase 1: Mirror descent ===")
    phase1_end = min(deadline, t_start + 500.0)
    
    a_work = best_seq.copy() / best_seq.sum()
    
    # Split time into chunks for different p regimes
    chunks = [
        (20, 80, 100, 30),    # p_start, p_end, top_k_start, top_k_end
        (80, 200, 50, 15),
        (200, 500, 30, 8),
        (500, 1000, 15, 5),
    ]
    
    per_chunk = (phase1_end - time.monotonic()) / len(chunks)
    for ci, (p0, p1, tk0, tk1) in enumerate(chunks):
        if time.monotonic() > phase1_end:
            break
        chunk_end = min(phase1_end, time.monotonic() + per_chunk)
        print(f"  Chunk {ci}: p=[{p0},{p1}], top_k=[{tk0},{tk1}], time={chunk_end - time.monotonic():.0f}s")
        
        a_work, sc_work = _run_mirror_descent(
            a_work, chunk_end,
            p_start=p0, p_end=p1,
            top_k_init=tk0, top_k_end=tk1
        )
        if sc_work < best_score:
            best_seq = a_work.copy()
            best_score = sc_work
            print(f"  Mirror done: {best_score:.10f}")
        a_work = best_seq.copy()
    
    print(f"  Phase 1 done: best={best_score:.10f}")
    
    # --- Phase 2: Adam refinement ---
    print("=== Phase 2: Adam refinement ===")
    phase2_end = min(deadline, t_start + 800.0)
    
    if time.monotonic() < phase2_end:
        remaining = phase2_end - time.monotonic()
        # Multiple Adam passes with different params
        adam_configs = [
            (0.05, 30, 20),
            (0.03, 50, 15),
            (0.02, 80, 10),
            (0.01, 200, 8),
        ]
        per_adam = remaining / len(adam_configs)
        for ai, (lr, p0, tk) in enumerate(adam_configs):
            if time.monotonic() > phase2_end - 5:
                break
            dl = min(phase2_end - 2, time.monotonic() + per_adam)
            p_best, sc_best = _adam_refine(best_seq, dl, lr=lr, p_start=p0, top_k=tk)
            update_best(p_best, sc_best, f"adam{ai}")
    
    print(f"  Phase 2 done: best={best_score:.10f}")
    
    # --- Phase 3: KKT balancing ---
    print("=== Phase 3: KKT balancing ===")
    phase3_end = min(deadline, t_start + 1000.0)
    
    if time.monotonic() < phase3_end:
        dl = min(phase3_end - 2, time.monotonic() + (phase3_end - time.monotonic()))
        # Multiple KKT passes with decreasing tolerance
        for tol in [0.01, 0.005, 0.002, 0.001]:
            if time.monotonic() > phase3_end - 5:
                break
            p_best, sc_best = _kkt_iterate(best_seq, min(phase3_end - 2, time.monotonic() + 60), tol=tol)
            update_best(p_best, sc_best, f"kkt_tol{tol}")
    
    print(f"  Phase 3 done: best={best_score:.10f}")
    
    # --- Phase 4: L-BFGS final refinement ---
    print("=== Phase 4: L-BFGS refinement ===")
    phase4_end = min(deadline, t_start + 1300.0)
    
    if time.monotonic() < phase4_end:
        remaining = phase4_end - time.monotonic()
        # Tight L-BFGS
        dl = min(phase4_end - 2, time.monotonic() + min(120.0, remaining * 0.5))
        p_best, sc_best = _lbfgs_refine(best_seq, dl, beta=1e-5, top_k=10)
        update_best(p_best, sc_best, "lbfgs_tight")
        
        if time.monotonic() < phase4_end - 10:
            dl2 = min(phase4_end - 2, time.monotonic() + min(60.0, (phase4_end - time.monotonic()) * 0.5))
            p_best2, sc_best2 = _lbfgs_refine(best_seq, dl2, beta=5e-6, top_k=5)
            update_best(p_best2, sc_best2, "lbfgs_verytight")
    
    print(f"  Phase 4 done: best={best_score:.10f}")
    
    # --- Phase 5: Multi-restart ---
    print("=== Phase 5: Multi-restart ===")
    phase5_end = min(deadline, t_start + 1600.0)
    
    if time.monotonic() < phase5_end and best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 3
        per_restart = (phase5_end - time.monotonic()) / max(1, n_restarts)
        
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            # Add noise in logit space
            noise_levels = [0.02, 0.05, 0.1]
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_levels[i] * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = np.exp(z)
            p_start = p_start / p_start.sum()
            
            # Quick mirror descent then L-BFGS
            per_md = per_restart * 0.5
            per_lb = per_restart * 0.45
            dl_md = min(phase5_end - 5, time.monotonic() + per_md)
            p_m, sc_m = _run_mirror_descent(p_start, dl_md, p_start=50, p_end=300,
                                            top_k_init=30, top_k_end=8)
            update_best(p_m, sc_m, f"r{i}_md")
            
            if time.monotonic() < phase5_end - 5:
                base = p_m if sc_m < best_score else best_seq
                dl_lb = min(phase5_end - 3, time.monotonic() + per_lb)
                p_lb, sc_lb = _lbfgs_refine(base, dl_lb, beta=1e-4, top_k=10)
                update_best(p_lb, sc_lb, f"r{i}_lbfgs")
    
    print(f"  Phase 5 done: best={best_score:.10f}")
    
    # --- Phase 6: Try different sizes ---
    print("=== Phase 6: Different sizes ===")
    phase6_end = min(deadline, t_start + 1750.0)
    
    if time.monotonic() < phase6_end and best_seq is not None:
        n_curr = len(best_seq)
        target_ns = []
        if n_curr != 2048:
            target_ns.append(2048)
        if n_curr != 8192:
            target_ns.append(8192)
        
        for tn in target_ns:
            if time.monotonic() > phase6_end - 30:
                break
            # Interpolate
            p_interp = np.interp(
                np.linspace(0, 1, tn),
                np.linspace(0, 1, n_curr),
                best_seq
            )
            p_interp = np.maximum(p_interp, 1e-10)
            p_interp = p_interp / p_interp.sum()
            sc_interp = _score_norm(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{tn}")
            
            # Quick optimize at this size
            if time.monotonic() < phase6_end - 20:
                remaining = phase6_end - time.monotonic()
                per_opt = min(80.0, remaining * 0.4)
                dl = min(phase6_end - 5, time.monotonic() + per_opt)
                p_best, sc_best = _run_mirror_descent(p_interp, dl,
                                                       p_start=30, p_end=200,
                                                       top_k_init=40, top_k_end=10)
                update_best(p_best, sc_best, f"md_n{tn}")
                
                if time.monotonic() < phase6_end - 15:
                    dl2 = min(phase6_end - 3, time.monotonic() + min(30.0, (phase6_end - time.monotonic()) * 0.4))
                    p_best2, sc_best2 = _lbfgs_refine(p_best, dl2, beta=1e-4, top_k=8)
                    update_best(p_best2, sc_best2, f"lbfgs_n{tn}")
    
    print(f"  Phase 6 done: best={best_score:.10f}")
    
    # --- Phase 7: Final polish ---
    print("=== Phase 7: Final polish ===")
    if time.monotonic() < deadline and best_seq is not None:
        remaining = deadline - time.monotonic()
        
        # Very tight KKT
        if remaining > 20:
            dl = min(deadline - 5, time.monotonic() + min(40.0, remaining * 0.4))
            p_best, sc_best = _kkt_iterate(best_seq, dl, tol=0.0005)
            update_best(p_best, sc_best, "final_kkt")
        
        # Final L-BFGS
        if time.monotonic() < deadline - 10:
            dl2 = min(deadline - 3, time.monotonic() + min(30.0, (deadline - time.monotonic()) * 0.4))
            p_best2, sc_best2 = _lbfgs_refine(best_seq, dl2, beta=1e-6, top_k=5)
            update_best(p_best2, sc_best2, "final_lbfgs")
    
    print(f"  Phase 7 done: best={best_score:.10f}")
    
    # --- Final output ---
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_norm(final_seq)
        else:
            final_seq = _make_profiles(4096)[0]
            final_score = _score_norm(final_seq)
        
        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")
        
        # Validate
        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 8192:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  eval check failed: {e}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_profiles(4096)[0]
    print(f"  FALLBACK: score={_score_norm(fallback):.10f}")
    return [float(x) for x in fallback]
