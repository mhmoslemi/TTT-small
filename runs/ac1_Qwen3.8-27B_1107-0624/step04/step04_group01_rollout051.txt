    # w has length 2n-1, so w_flipped has length 2n-1
    # conv(p, w_flipped) has length n + 2n-1 - 1 = 3n-2
    # cross_correlation(p, w)[j] = conv(p, w_flipped)[2n-2-j]
    
    w_flipped = w[::-1]
    pw = _full_convolve(p, w_flipped)
    # h[j] = pw[2n-2-j] for j=0..n-1
    h = pw[2*n-2- np.arange(n)][::-1]  # h[j] = pw[2n-2-j]
    # Wait, let me be more careful.
    # h[j] = sum_m p[m]*w[m+j]
    # Let's verify for j=0: h[0] = sum_m p[m]*w[m] = dot(p, w[:n])
    # pw[2n-2-0] = pw[2n-2] = sum_m p[m]*w_flipped[2n-2-m] = sum_m p[m]*w[2n-2-(2n-2-m)] = sum_m p[m]*w[m] ✓
    
    grad_p = 2.0 * h
    W = float(np.dot(p, grad_p))
    grad_z = p * (grad_p - W)
    
    obj = 2.0 * n * (cm + beta * np.log(s))  # = 2*n*logsumexp(c/beta)
    return obj, grad_z


def _lbfgsb_optimize(z_init, deadline, beta_schedule, max_iter_per_phase=5000):
    """L-BFGS-B optimization with beta continuation."""
    n = len(z_init)
    best_z = z_init.copy()
    best_obj = _smoothed_obj_and_grad(z_init, beta_schedule[0])[0]
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        def obj_grad(z):
            return _smoothed_obj_and_grad(z, beta)
        
        try:
            res = minimize(
                obj_grad, z_init, jac=True, method='L-BFGS-B',
                options={'maxiter': max_iter_per_phase, 'gtol': 1e-6,
                         'maxfun': max_iter_per_phase * 2}
            )
            z_init = res.x
            obj_val, _ = _smoothed_obj_and_grad(z_init, beta)
            if obj_val < best_obj:
                best_obj = obj_val
                best_z = z_init.copy()
        except Exception:
            pass
        
        # Check time
        if time.monotonic() > deadline - 10:
            break
    
    return best_z, best_obj


def _adam_smooth(z_init, deadline, beta_schedule, lr=0.01, max_iter=100000):
    """Adam optimizer in log-space with log-sum-exp smoothing."""
    n = len(z_init)
    z = z_init.copy()
    best_z = z.copy()
    best_obj = _smoothed_obj_and_grad(z, beta_schedule[0])[0]
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        for it in range(max_iter):
            if time.monotonic() > deadline:
                break
            t += 1
            obj, grad = _smoothed_obj_and_grad(z, beta)
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1-b1) * grad
            v_buf = b2 * v_buf + (1-b2) * (grad*grad)
            mh = m_buf / (1-b1**t)
            vh = v_buf / (1-b2**t)
            
            z = z - lr * mh / (np.sqrt(vh) + eps)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            
            if t % 200 == 0:
                sc = _smoothed_obj_and_grad(z, beta)[0]
                if sc < best_obj:
                    best_obj = sc
                    best_z = z.copy()
    
    return best_z, best_obj


def _make_initialization(n, kind, rng):
    """Create diverse initializations."""
    if kind == "uniform":
        return np.zeros(n)
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        return np.log(1.0 / (1.0 + 20.0*(t-0.5)**2)) - np.log(1.0/(1.0+20.0*0.25))
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-50*(t-0.3)**2) + np.exp(-50*(t-0.7)**2)
        return np.log(np.maximum(a, 1e-10))
    elif kind == "fourier":
        freq = rng.integers(2, max(3, n//4))
        t = np.arange(n)
        a = 1.0 + 0.5*np.cos(2*np.pi*freq*t/n)
        return np.log(np.maximum(a, 1e-10))
    elif kind == "chirp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3*np.cos(2*np.pi*(t**2 - t))
        return np.log(np.maximum(a, 1e-10))
    elif kind == "random":
        return 0.5 * rng.standard_normal(n)
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - 2.0*np.abs(t - 0.5)
        return np.log(np.maximum(a, 1e-10))
    else:
        return np.zeros(n)
