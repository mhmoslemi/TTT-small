    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient_z(a, c, w, n):
    """Gradient of 2n*L_beta wrt z (softmax param). = 4*n*a*(h - dot(w,c))."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _obj_grad(z, beta, n):
    """Return (objective, gradient) for L-BFGS."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        obj = 2.0 * n * c_max
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        obj = 2.0 * n * (c_max + beta * np.log(s))
        w = e / s
    grad = _compute_gradient_z(a, c, w, n)
    if not np.all(np.isfinite(grad)) or np.max(np.abs(grad)) > 1e10:
        grad = np.zeros_like(z)
    return obj, grad


def _lbfgs_phase(z_init, beta, n, deadline, max_iter=20000, maxfun=10000):
    """Run one L-BFGS phase with given beta."""
    def fun(z):
        return _obj_grad(z, beta, n)[0]

    def fun_grad(z):
        return _obj_grad(z, beta, n)

    try:
        res = sp_min(fun_grad, z_init, method='L-BFGS-B',
                     options={'maxiter': max_iter, 'maxfun': maxfun,
                              'gtol': 1e-10, 'ftol': 1e-12, 'iprint': 0})
        return res.x, res.fun
    except Exception:
        return z_init, _obj_grad(z_init, beta, n)[0]


def _adam_phase(z_init, beta_schedule, n, deadline, lr=0.01, max_iter=100000):
    """Adam optimizer in log-space with given beta schedule."""
    z = z_init.copy()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_z = z.copy()
    best_score = _score_norm(_softmax(z))

    for beta in beta_schedule:
        it_count = 0
        while it_count < max_iter:
            if time.monotonic() > deadline:
                return best_z, best_score
            it_count += 1
            t += 1

            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad = _compute_gradient_z(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            step = lr * mh / (np.sqrt(vh) + eps)
            max_step = 1.0
            sn = np.linalg.norm(step)
            if sn > max_step:
                step = step * max_step / sn
            z = z - step
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

                sc = _score_norm(_softmax(z))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_z = z.copy()
