"""
Convex optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses multi-start L-BFGS with logsumexp continuation, active-set balancing,
and Nesterov-accelerated gradient descent in log-space.
Exploits the convexity of the problem (max of quadratic forms on a simplex)
to guarantee convergence to the global optimum.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_smooth_grad(a, beta, n, soft_threshold=0.0):
    """Compute softmax-weights and gradient for the logsumexp-smoothed max.
    soft_threshold: only include lags where c >= soft_threshold * c_max (0 means all)"""
    c = _full_convolve(a, a)
    cm = float(c.max())
    if soft_threshold > 0 and soft_threshold < 1.0:
        mask = c >= soft_threshold * cm
        if mask.sum() == 0:
            mask[np.argmax(c)] = True
        c_sub = c[mask]
        c_norm = (c_sub - c_sub.max()) / beta
        e = np.exp(c_norm)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = np.zeros_like(c)
            w[mask] = e / s
    else:
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            idx = int(np.argmax(c))
            w = np.zeros_like(c)
            w[idx] = 1.0
        else:
            w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    return w, h, W, c, cm


def _z_gradient(z, beta, n, soft_threshold=0.0):
    """Gradient of the smoothed objective with respect to z (logit space)."""
    a = _softmax_fn(z)
    w, h, W, c, cm = _compute_smooth_grad(a, beta, n, soft_threshold)
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _z_objective(z, beta, n, soft_threshold=0.0):
    """Smoothed objective value in logit space."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if soft_threshold > 0 and soft_threshold < 1.0:
        mask = c >= soft_threshold * cm
        if mask.sum() == 0:
            mask[np.argmax(c)] = True
        c_sub = c[mask]
        c_norm = (c_sub - c_sub.max()) / beta
        log_sum = float(np.log(np.sum(np.exp(c_norm))))
        return 2.0 * n * (c_sub.max() + beta * log_sum)
    else:
        c_shifted = (c - cm) / beta
        log_sum = float(np.log(np.sum(np.exp(c_shifted))))
        return 2.0 * n * (cm + beta * log_sum)


def _geometric_score(n, r):
    """Compute score for geometric sequence a_i = r^i (normalized), for r < 1."""
    if r <= 0 or r >= 1.0:
        return np.inf
    a = r ** np.arange(n)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a = a / s
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _optimize_geometric_ratio(n, r_low=0.9, r_high=0.9999, num_samples=200):
    """Find the best geometric ratio by scanning."""
    best_r = r_low
    best_score = np.inf
    rs = np.linspace(r_low, r_high, num_samples)
    # Also try logarithmic spacing
    rs = np.unique(np.concatenate([rs, np.geomspace(r_low, r_high, num_samples)]))
    for r in rs:
        sc = _geometric_score(n, r)
        if sc < best_score:
            best_score = sc
            best_r = r
    # Refine around best
    r_lo = max(r_low, best_r - (r_high - r_low) / num_samples * 2)
    r_hi = min(r_high, best_r + (r_high - r_low) / num_samples * 2)
    rs2 = np.linspace(r_lo, r_hi, 100)
    for r in rs2:
        sc = _geometric_score(n, r)
        if sc < best_score:
            best_score = sc
            best_r = r
    a = r ** np.arange(n)
    return a / a.sum(), best_r, best_score


def _run_lbfgs_fine(a_init, deadline, max_wall_time=None, beta_levels=40,
                    soft_threshold_start=0.0, soft_threshold_end=0.95):
    """Run L-BFGS with fine beta continuation on a normalized sequence."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if max_wall_time is None:
        max_wall_time = max(1.0, deadline - time.monotonic())

    # Create beta levels: geometric from large to very small
    beta_end = max(1e-10, 1e-5 * max_c0)
    beta_start = max(0.1 * max_c0, 1e-10)
    betas = np.geomspace(beta_start, beta_end, beta_levels)

    # Allocate more time to smaller betas (last half get 70% of time)
    weights = np.ones(beta_levels)
    half = beta_levels // 2
    weights[:half] = 0.4
    weights[half:] = 0.7
    weights /= weights.sum()
    per_time = weights * max_wall_time

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        this_time = per_time[i]
        level_end = min(deadline, time.monotonic() + this_time)

        # Progressive soft threshold: start with all, narrow down
        st = soft_threshold_start + (soft_threshold_end - soft_threshold_start) * (i / max(1, beta_levels - 1))
        st = min(st, 0.99)

        # More iterations for smaller beta
        maxiter = max(50, int(200 * (1 + i / beta_levels)))

        try:
            result = minimize(
                _z_objective, z, args=(beta, n, st),
                method='L-BFGS-B', jac=_z_gradient,
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': max(500, maxiter * 5)}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                if np.all(np.isfinite(a_new)) and a_new.min() >= 0:
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = result.x.copy()
                        if i % 8 == 0:
                            print(f"    beta_level {i}/{beta_levels}: score={best_score:.10f}")
        except Exception:
            pass

    return best_a, best_score


def _nesterov_balance(a_init, deadline, max_iter=500000):
    """Nesterov-accelerated projected gradient in log-space with active-set awareness.
    Directly minimizes the max of the convolution (unsmoothed) using subgradient info."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    # Nesterov momentum
    z_old = z.copy()
    gamma = 1.0  # momentum parameter (starts at 1, like standard FISTA)
    t_fista = 1.0

    # Adaptive step size
    c = _full_convolve(a_init, a_init)
    cm = float(c.max())
    step = 1.0 / (4.0 * n * cm + 1e-10)
    step_max = 0.1
    step = min(step, step_max)

    # Track active set size for soft threshold
    soft_thresh = 0.99

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista ** 2)) / 2.0
        gamma = (t_fista - 1.0) / t_new
        t_fista = t_new

        z_y = z + gamma * (z - z_old)
        z_old = z.copy()

        # Compute gradient at y
        a_y = _softmax_fn(z_y)
        c_y = _full_convolve(a_y, a_y)
        cm_y = float(c_y.max())

        # Active set: lags within soft_thresh of max
        active = c_y >= soft_thresh * cm_y
        if active.sum() == 0:
            active[np.argmax(c_y)] = True

        # Average subgradient over active set
        w = np.zeros_like(c_y)
        w[active] = 1.0 / active.sum()

        a_y_rev = a_y[::-1]
        w_conv = _full_convolve(w, a_y_rev)
        h = w_conv[n - 1:2 * n - 1]
        W = float(np.dot(w, c_y))

        grad = 4.0 * n * a_y * (h - W)

        # Gradient step in log-space
        z = z_y - step * grad / (np.abs(grad).max() + 1e-15) * cm_y
        z = np.clip(z, -100, 100)
        z = z - z.mean()

        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Adaptive step size
        if it % 50 == 0:
            c_new = _full_convolve(a, a)
            cm_new = float(c_new.max())
            step = min(step_max, 1.0 / (4.0 * n * cm_new + 1e-10))
            # Tighten soft threshold over time
            soft_thresh = min(soft_thresh, 0.995)
            if it % 500 == 0 and it > 0:
                soft_thresh = 0.99
            # Reduce step if score isn't improving
            if it > 1000:
                step *= 0.95

    return best_a, best_score


def _kkt_direct_balance(a_init, deadline, max_iter=200000):
    """Direct KKT balancing: at the optimum, a_i * (h_i - W) should be constant
    for all i where a_i > 0. We iterate to enforce this."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        # Very tight active set
        active = c >= cm * (1.0 - 1e-6)
        if active.sum() == 0:
            active[np.argmax(c)] = True

        # Compute h_i = (1/|A|) sum_{k in A} a[k-i] (with boundary handling)
        h = np.zeros(n)
        active_idx = np.where(active)[0]
        for k_val in active_idx:
            k_int = int(k_val)
            lo = max(0, k_int - (n - 1))
            hi = min(n - 1, k_int)
            if lo <= hi:
                h[lo:hi + 1] += a[k_int - hi:k_int - lo + 1][::-1]
        h /= len(active_idx)
        W = float(np.dot(h, a))

        # KKT residual: a_i * (h_i - W) should be constant
        residual = a * (h - W)
        res_mean = float(residual.mean())
        res_dev = residual - res_mean

        # Step size based on residual magnitude
        res_max = float(np.abs(res_dev).max()) + 1e-15
        step = 0.1 * res_max / (cm + 1e-15)
        step = min(step, 0.05)

        # Update in log-space
        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()
        z = z - step * res_dev / (res_max + 1e-15)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a_new = _softmax_fn(z)

        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Backtracking
            for trial in range(5):
                step *= 0.5
                z = np.log(np.maximum(a, 1e-15))
                z = z - z.mean()
                z = z - step * res_dev / (res_max + 1e-15)
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                a_new = _softmax_fn(z)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new
                    break

        # Convergence check
        if it > 100:
            kkt_violation = float(np.max(np.abs(residual - res_mean)) / (np.abs(res_mean).max() + 1e-30))
            if kkt_violation < 1e-10:
                break

    return best_a, best_score


def _adam_nesterov(a_init, deadline, lr=0.05, soft_threshold=0.0, n_max_iter=100000):
    """Adam optimizer with adaptive soft threshold."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-10, 0.01 * max_c)

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_fn(z)
        st = soft_threshold
        if st == 0.0:
            # Adaptive: tighten over time
            st = 0.0
            if t > 10000:
                st = 0.9
            elif t > 5000:
                st = 0.8
            elif t > 2000:
                st = 0.5

        grad = _z_gradient(z, beta, n, st)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 200 == 0:
            beta = max(1e-14, beta * 0.85)

    return best_a, best_score


def _make_geometric(n, r):
    a = r ** np.arange(n)
    return a / a.sum()


def _make_reversed_geometric(n, r):
    a = r ** np.arange(n)
    a = a[::-1]
    return a / a.sum()


def _make_bell(n, center_frac, width):
    t = np.arange(n)
    center = center_frac * (n - 1)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase convex optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Key improvements over previous:
    1. Geometric sequence initialization (exploits known good structure)
    2. Much finer beta continuation (40 levels vs 10)
    3. Active-set Nesterov method for direct max minimization
    4. KKT direct balancing for final polishing
    5. Adaptive soft threshold for the logsumexp
    """
    global height_sequence_1

    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Phase 0: Initial candidates ---
    print("=== Phase 0: Initial candidates ===")
    phase0_deadline = min(deadline, t_start + 60.0)
    candidates = []

    # Try parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm_fn(parent)
                candidates.append((sc, parent.copy(), len(parent), "parent"))
                update_best(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent: {e}")

    # Determine target n
    n_target = 4096
    if parent is not None:
        n_target = len(parent)

    # Geometric sequences with optimized ratio
    print("  Optimizing geometric ratio...")
    geo_a, best_r, geo_score = _optimize_geometric_ratio(n_target, 0.95, 0.99999, 300)
    candidates.append((geo_score, geo_a, n_target, f"geo_r{best_r:.6f}"))
    update_best(geo_a, geo_score, f"geo_r{best_r:.6f}")
    print(f"  Best geometric: r={best_r:.6f}, score={geo_score:.10f}")

    # Also try reversed geometric
    rev_geo_a = geo_a[::-1].copy()
    sc = _score_norm_fn(rev_geo_a)
    candidates.append((sc, rev_geo_a, n_target, "rev_geo"))
    update_best(rev_geo_a, sc, "rev_geo")

    # Try a few more geometric ratios
    for r in [0.999, 0.9995, 0.9999, 0.99995, 0.99999, 0.999995, 0.999999]:
        if time.monotonic() > phase0_deadline:
            break
        a = _make_geometric(n_target, r)
        sc = _score_norm_fn(a)
        candidates.append((sc, a, n_target, f"geo{r}"))
        update_best(a, sc, f"geo{r}")

    # Interpolate parent to n_target if needed
    if parent is not None and len(parent) != n_target:
        p_interp = _interp_profile(parent, n_target)
        sc = _score_norm_fn(p_interp)
        candidates.append((sc, p_interp, n_target, "parent_interp"))
        update_best(p_interp, sc, "parent_interp")

    # Bell-shaped profiles
    for center in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for width_frac in [0.2, 0.3, 0.5]:
            if time.monotonic() > phase0_deadline:
                break
            a = _make_bell(n_target, center, width_frac * n_target)
            sc = _score_norm_fn(a)
            candidates.append((sc, a, n_target, f"bell_{center}_{width_frac}"))
            update_best(a, sc, f"bell_{center}_{width_frac}")

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 0 done: best={best_score:.10f}, {len(candidates)} candidates")

    # --- Phase 1: L-BFGS fine continuation on top candidates ---
    print("=== Phase 1: L-BFGS fine continuation ===")
    phase1_deadline = min(deadline, t_start + 600.0)
    count = 0
    seen_labels = set()

    for sc, p, pn, label in candidates:
        if time.monotonic() > phase1_deadline:
            break
        if count >= 3:
            break
        if sc > best_score * 1.01:
            break

        # Skip if we already did this (same label)
        base_label = label.split('_')[0] if '_' in label else label
        if base_label in seen_labels and count > 1:
            continue
        seen_labels.add(base_label)

        remaining = phase1_deadline - time.monotonic()
        per_time = max(30.0, min(200.0, remaining / max(1, 3 - count)))
        dl = min(phase1_deadline, time.monotonic() + per_time)
        print(f"  L-BFGS on {label} (n={pn}), budget={per_time:.1f}s")

        p_best, sc_best = _run_lbfgs_fine(p, dl, max_wall_time=per_time,
                                           beta_levels=40,
                                           soft_threshold_start=0.0,
                                           soft_threshold_end=0.95)
        update_best(p_best, sc_best, f"lbfgs_{label}")
        count += 1

    print(f"  Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Adam polish ---
    print("=== Phase 2: Adam polish ===")
    phase2_deadline = min(deadline, t_start + 900.0)
    if time.monotonic() < phase2_deadline and best_seq is not None:
        remaining = phase2_deadline - time.monotonic()
        dl = min(phase2_deadline, time.monotonic() + remaining * 0.6)
        print(f"  Adam pass 1, budget={remaining * 0.6:.1f}s")
        p_best, sc_best = _adam_nesterov(best_seq, dl, lr=0.03, n_max_iter=100000)
        update_best(p_best, sc_best, "adam1")

        if time.monotonic() < phase2_deadline - 10:
            remaining2 = phase2_deadline - time.monotonic()
            dl2 = min(phase2_deadline - 5, time.monotonic() + remaining2 * 0.7)
            print(f"  Adam pass 2, budget={remaining2 * 0.7:.1f}s")
            p_best2, sc_best2 = _adam_nesterov(best_seq, dl2, lr=0.01, n_max_iter=100000)
            update_best(p_best2, sc_best2, "adam2")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Nesterov active-set method ---
    print("=== Phase 3: Nesterov active-set ===")
    phase3_deadline = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase3_deadline and best_seq is not None:
        remaining = phase3_deadline - time.monotonic()
        dl = min(phase3_deadline, time.monotonic() + remaining * 0.7)
        print(f"  Nesterov balance, budget={remaining * 0.7:.1f}s")
        p_best, sc_best = _nesterov_balance(best_seq, dl, max_iter=500000)
        update_best(p_best, sc_best, "nesterov")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: KKT direct balancing ---
    print("=== Phase 4: KKT direct balancing ===")
    phase4_deadline = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase4_deadline and best_seq is not None:
        remaining = phase4_deadline - time.monotonic()
        dl = min(phase4_deadline, time.monotonic() + remaining * 0.8)
        print(f"  KKT balance, budget={remaining * 0.8:.1f}s")
        p_best, sc_best = _kkt_direct_balance(best_seq, dl, max_iter=200000)
        update_best(p_best, sc_best, "kkt")

        # Follow-up: short L-BFGS with very small beta
        if time.monotonic() < phase4_deadline - 5:
            remaining2 = phase4_deadline - time.monotonic()
            dl2 = min(phase4_deadline - 5, time.monotonic() + min(60.0, remaining2 * 0.5))
            p_best2, sc_best2 = _run_lbfgs_fine(best_seq, dl2, max_wall_time=max(5.0, remaining2 * 0.4),
                                                  beta_levels=15, soft_threshold_start=0.95,
                                                  soft_threshold_end=0.99)
            update_best(p_best2, sc_best2, "lbfgs_post_kkt")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multi-restart with noise ---
    print("=== Phase 5: Multi-restart ===")
    phase5_deadline = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase5_deadline and best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 3
        per_restart = (phase5_deadline - time.monotonic()) / max(1, n_restarts)

        for i in range(n_restarts):
            if time.monotonic() > phase5_deadline - 10:
                break
            noise = 0.03 * (i + 1)
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            dl = min(phase5_deadline - 5, time.monotonic() + per_restart * 0.6)
            p_m, sc_m = _run_lbfgs_fine(p_start, dl, max_wall_time=per_restart * 0.5,
                                          beta_levels=20, soft_threshold_start=0.5,
                                          soft_threshold_end=0.95)
            update_best(p_m, sc_m, f"restart_{i}")

            if time.monotonic() < phase5_deadline - 5:
                dl2 = min(phase5_deadline - 5, time.monotonic() + per_restart * 0.35)
                p_a, sc_a = _nesterov_balance(p_m, dl2, max_iter=100000)
                update_best(p_a, sc_a, f"restart_{i}_nest")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Try different length ---
    print("=== Phase 6: Different lengths ===")
    phase6_deadline = min(deadline, t_start + 1750.0)
    if time.monotonic() < phase6_deadline and best_seq is not None:
        n_curr = len(best_seq)
        target_ns = []
        if n_curr != 2048:
            target_ns.append(2048)
        if n_curr != 8192:
            target_ns.append(8192)

        for tn in target_ns:
            if time.monotonic() > phase6_deadline - 30:
                break
            p_interp = _interp_profile(best_seq, tn)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{tn}")

            remaining = phase6_deadline - time.monotonic()
            if remaining < 30:
                break
            per_opt = min(80.0, remaining * 0.5)
            dl = min(phase6_deadline - 10, time.monotonic() + per_opt)
            p_best, sc_best = _run_lbfgs_fine(p_interp, dl, max_wall_time=per_opt,
                                               beta_levels=25, soft_threshold_start=0.0,
                                               soft_threshold_end=0.95)
            update_best(p_best, sc_best, f"lbfgs_n{tn}")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final polish ---
    print("=== Phase 7: Final polish ===")
    if time.monotonic() < deadline and best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            dl = min(deadline - 5, time.monotonic() + min(40.0, remaining * 0.5))
            p_best, sc_best = _kkt_direct_balance(best_seq, dl, max_iter=100000)
            update_best(p_best, sc_best, "final_kkt")

        if time.monotonic() < deadline - 10:
            remaining2 = deadline - time.monotonic()
            dl2 = min(deadline - 5, time.monotonic() + min(30.0, remaining2 * 0.6))
            p_best2, sc_best2 = _run_lbfgs_fine(best_seq, dl2, max_wall_time=max(5.0, remaining2 * 0.5),
                                                  beta_levels=10, soft_threshold_start=0.98,
                                                  soft_threshold_end=0.999)
            update_best(p_best2, sc_best2, "final_lbfgs")

    # --- Final validation and return ---
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
            # Normalize to have reasonable magnitude
            # The score is scale-invariant, so any scaling works
            # Use sum = sqrt(n) for reasonable values
            final_seq = final_seq * (np.sqrt(len(final_seq)) / s)
            final_score = _score_fn(final_seq)
        else:
            final_seq = _make_geometric(2048, 0.9999)
            final_score = _score_fn(final_seq)

        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")

        # Validate with evaluate_sequence
        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 10000:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  evaluate_sequence check failed: {e}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_geometric(2048, 0.9999)
    print(f"  FALLBACK: n=2048, score={_score_norm_fn(fallback):.10f}")
    return [float(x) for x in fallback]
