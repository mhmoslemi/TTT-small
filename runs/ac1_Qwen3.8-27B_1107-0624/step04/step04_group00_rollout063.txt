def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _lse_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 2.0 * n * cm
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    lse = cm + beta * np.log(e.sum())
    return 2.0 * n * lse


def _lse_gradient(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _make_topk_indicator(top_indices, length):
    e = np.zeros(length)
    for k in top_indices:
        if 0 <= int(k) < length:
            e[int(k)] = 1.0
    return e


def _topk_objective(z, k_top, n):

    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    sorted_c = np.sort(c)[::-1]
    return 2.0 * n * float(np.sum(sorted_c[:k_top]))


def _topk_gradient(z, k_top, n):
    """Approximate gradient of top-k sum via hard weights."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    top_indices = np.argsort(c)[::-1][:k_top]
    w = _make_topk_indicator(top_indices, 2 * n - 1)
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c) / k_top)
    grad = 4.0 * n * a * (h - W) / k_top
    if not np.all(np.isfinite(grad)):

    return grad

def _structured_restart(a_base, rng, n):
    """Generate a structured perturbation of the base sequence."""
    z = np.log(np.maximum(a_base, 1e-15))
    z = z - z.mean()

    # Pick a random structured perturbation
    mode = rng.integers(0, 5)
    if mode == 0:
        # Sinusoidal perturbation
        freq = rng.uniform(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.3)
        t = np.linspace(0, 1, n)
        z += amp * np.sin(2 * np.pi * freq * t + phase)
    elif mode == 1:
        # Linear tilt
        amp = rng.uniform(-0.3, 0.3)
        t = np.linspace(0, 1, n)
        z += amp * (t - 0.5)
    elif mode == 2:
        # Quadratic bump
        amp = rng.uniform(-0.3, 0.3)
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.4)
        t = np.linspace(0, 1, n)
        z += amp * np.exp(-((t - center) / width) ** 2)
    elif mode == 3:
        # Step function perturbation
        threshold = rng.uniform(0.2, 0.8)
        amp = rng.uniform(-0.2, 0.2)
        t = np.linspace(0, 1, n)
        z += amp * (t > threshold).astype(float)
    else:
        # Gaussian noise (fallback)
        z += rng.uniform(0.05, 0.15) * rng.normal(size=n)

    z = np.clip(z, -60, 60)
    z = z - z.mean()
    return _softmax_fn(z)


def _adaptive_gd_v2(a_init, deadline, rng, lr_init=0.1, max_iter=30000):
    """Adaptive gradient descent with top-k blending."""

def _make_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []

    # Linear ramps (asymmetric)
    for slope in [1.0, 2.0, 3.0, 5.0]:
        a = 1.0 + slope * (t - 0.3)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Exponential
    for rate in [2.0, 3.0, 5.0, -2.0, -3.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())

    # Power-law (asymmetric)
    for p in [0.3, 0.5, 0.7, 1.5, 2.0]:
        a = np.maximum(t, 1e-10) ** p * np.maximum(1 - t, 1e-10) ** (2.0 - p)
        profiles.append(a / a.sum())

    # Two-level step
    for thresh in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for ratio in [0.1, 0.3, 0.5, 1.0, 2.0]:
            a = np.where(t < thresh, 1.0, ratio)
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())

    # Cosine with phase shift
    for phase in [0.0, 0.3, -0.3, 0.5, -0.5]:
        a = np.maximum(np.cos(np.pi * (u + phase)), 1e-12)
        profiles.append(a / a.sum())

    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

    # Phase 7: Final fine polish (remaining)
    phase7_end = deadline
    print("Phase 7: Final fine polish")

    if best_seq is not None and time.monotonic() < phase7_end - 20:
        n_final = len(best_seq)
        remaining = phase7_end - time.monotonic()

        # L-BFGS with very fine beta
        c_final = _full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())
        beta_fine = 5e-5 * max_c_final

        per_dl = min(phase7_end - 15, time.monotonic() + remaining * 0.5)
        try:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            result = minimize(
                _lse_objective, z, args=(beta_fine, n_final),
                method='L-BFGS-B', jac=_lse_gradient,
                options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': 10000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_seq = a_new.copy()
                    print(f"  Fine L-BFGS: {best_score:.10f}")
        except Exception:
            pass

        # Final peak surgery
        if time.monotonic() < phase7_end - 10:
            per_dl = min(phase7_end - 5, time.monotonic() + remaining * 0.3)
            p_best, sc_best = _peak_surgery_v2(best_seq, per_dl, rng, max_outer=100)
            update_best(p_best, sc_best, "final_surgery")

        # Final adaptive GD
        if time.monotonic() < phase7_end - 10:
            per_dl = min(phase7_end - 3, time.monotonic() + remaining * 0.2)
            p_best, sc_best = _adaptive_gd_v2(best_seq, per_dl, rng, lr_init=0.02)
            update_best(p_best, sc_best, "final_gd")

    print(f"  Phase 7 done: best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profiles(256)[2]

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 5000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    fallback = _make_profiles(256)[2]
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
