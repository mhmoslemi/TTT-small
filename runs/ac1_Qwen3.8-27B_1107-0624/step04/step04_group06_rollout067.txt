```
def _obj_jac(z, beta):
    a = softmax(z)
    c = full_convolve(a)
    val, w = smooth_max(c,beta)
    grad_a = grad_conv_weighted(a,w)
    dot = dot(a, grad_a)
    gz = a*(grad_a-dot)
    gz -= gz.mean() # optional, remove invariance
```

```python
"""
Optimization of non-negative step-function heights to minimize
2*n*max(conv(a,a))/(sum(a))^2.

Main ideas:
- Work on the normalized simplex (sum(a)=1, a>=0) via softmax/log parameters.
- Use FFT-exact gradients of a smooth-max approximation to the bottleneck convolution
  coefficient, driven by Scipy L-BFGS-B with a continuation schedule.
- Use a novel successive linear-programming (SLP) step: solve an LP for a new
  distribution q that minimizes the linearized maximum of conv(q,q), then line-search
  a convex combination of current a and q.
- Add multiplicative peak-flooding, stochastic coordinate moves, and Adam polish on
  logits for final fine-tuning.
"""

import time
import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix


# ----------------------------------------------------------------------
# Global state used only for safe time-boxing of Scipy callbacks.
# ----------------------------------------------------------------------
_TIMEBOX = {"deadline": 0.0}
_LP_CACHE = {}


class _Timeout(Exception):
    pass


def _lbfgs_callback(pk):
    dl = _TIMEBOX["deadline"]
    if dl and time.monotonic() > dl:
        return True
    return False


def _next_pow2(n):
    if n <= 1:
        return 1
    k = 1
    while k < n:
        k <<= 1
    return k


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.zeros(1, dtype=np.float64)
    a = np.maximum(a, 1e-15)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(a):
    """Assumes a is already normalized (sum ~= 1)."""
    n = a.size
    if n == 0:
        return np.inf
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _score_unit(a):
    a = _normalize(a)
    return _score_norm(a)


def _softmax_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -100.0, 100.0)
    z = z - float(z.max())
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    return p / s


def _logits_of(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 1e-30)
    z = np.log(a)
    z -= float(z.mean())
    return z


def _smooth_max(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    if c.size == 0:
        return 0.0, np.zeros(0, dtype=np.float64)
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        tol = 1e-12 * max(1.0, abs(mval))
        mask = c >= mval - tol
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w

    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    return val, ex / s


def _cross_w_a(w, a):
    """
    Cross-correlation corr[i] = sum_k w[k+i] * a[k] for valid linear indices.
    This is used for gradients of weighted convolution objectives.
    """
    n = len(a)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    N = _next_pow2(3 * n - 1)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    # IMPORTANT: this is W * conj(A), not conj(W) * A.
    cross = np.fft.irfft(W * np.conj(A), N)
    return cross[:n]


def _grad_conv_weighted(a, w):
    """
    Gradient of sum_k w[k] * conv(a,a)[k] with respect to a.
    grad[i] = 2 * sum_k w[k] * a[k-i].
    """
    return 2.0 * _cross_w_a(w, a)


# ----------------------------------------------------------------------
# Free-coordinate parameterization for L-BFGS-B (remove log-sum invariance).
# ----------------------------------------------------------------------
def _to_free(a, fixed=None):
    a = _normalize(a)
    n = a.size
    if fixed is None or fixed < 0 or fixed >= n:
        fixed = int(np.argmax(a))
    x = np.log(np.maximum(a, 1e-30))
    x -= x[fixed]
    y = np.empty(n - 1, dtype=np.float64)
    if fixed > 0:
        y[:fixed] = x[:fixed]
    if fixed < n - 1:
        y[fixed:] = x[fixed + 1:]
    return y, fixed


def _from_free(y, fixed):
    n = len(y) + 1
    x = np.empty(n, dtype=np.float64)
    if fixed > 0:
        x[:fixed] = y[:fixed]
    x[fixed] = 0.0
    if fixed < n - 1:
        x[fixed + 1:] = y[fixed:]
    return _softmax_z(x)


def _smooth_obj_free(y, args):
    beta, fixed, emergency_deadline = args
    if time.monotonic() > emergency_deadline:
        raise _Timeout()

    n = len(y) + 1
    x = np.empty(n, dtype=np.float64)
    if fixed > 0:
        x[:fixed] = y[:fixed]
    x[fixed] = 0.0
    if fixed < n - 1:
        x[fixed + 1:] = y[fixed:]

    a = _softmax_z(x)
    c = _full_convolve(a)
    val, w = _smooth_max(c, beta)

    ga = _grad_conv_weighted(a, w)
    dot = float(np.dot(a, ga))
    gz_full = a * (ga - dot)

    g = np.empty(n - 1, dtype=np.float64)
    if fixed > 0:
        g[:fixed] = gz_full[:fixed]
    if fixed < n - 1:
        g[fixed:] = gz_full[fixed + 1:]

    if not np.all(np.isfinite(g)):
        return 1e30, np.zeros(n - 1, dtype=np.float64)
    return float(val), g


def _lbfgs_step(a, beta, fixed, run_deadline):
    global _TIMEBOX
    y = _to_free(a, fixed)[0]
    emergency = run_deadline + 10.0
    _TIMEBOX["deadline"] = run_deadline

    args = (float(beta), int(fixed), float(emergency))
    try:
        res = minimize(
            _smooth_obj_free,
            y,
            method="L-BFGS-B",
            jac=True,
            args=args,
            callback=_lbfgs_callback,
            options={
                "maxiter": 100000,
                "maxfun": 2000000,
                "maxls": 20,
                "ftol": 1e-15,
                "gtol": 1e-14,
                "disp": False,
            },
        )
    except _Timeout:
        return None
    except Exception:
        return None

    if res is None or not hasattr(res, "x") or res.x is None:
        return None
    x = np.asarray(res.x, dtype=np.float64)
    if x.size != max(1, len(a) - 1) or not np.all(np.isfinite(x)):
        return None
    return _from_free(x, fixed)


def _lbfgs_continuation(a0, deadline, phase_budget):
    a = _normalize(a0)
    n = a.size
    best = a.copy()
    best_sc = _score_norm(a)
    cur = a.copy()

    cmax = float(_full_convolve(best).max())
    gammas = [
        1.0, 0.75, 0.55, 0.4, 0.3, 0.22, 0.16, 0.12, 0.09,
        0.065, 0.048, 0.034, 0.024, 0.017, 0.012, 0.0085,
        0.006, 0.0042, 0.003, 0.0021,
    ]

    phase_budget = max(1.0, float(phase_budget))
    run_time = max(3.0, min(45.0, phase_budget / float(len(gammas))))

    for gamma in gammas:
        if time.monotonic() >= deadline - 5.0:
            break
        beta = float(gamma) * max(1e-9, cmax)
        fixed = int(np.argmax(cur))
        run_deadline = min(deadline - 2.0, time.monotonic() + run_time)
        a_new = _lbfgs_step(cur, beta, fixed, run_deadline)

        if a_new is None or a_new.shape != a.shape or not np.all(np.isfinite(a_new)):
            continue

        a_new = _normalize(a_new)
        sc_new = _score_norm(a_new)
        if sc_new < best_sc - 1e-13:
            best = a_new.copy()
            best_sc = sc_new
            print(f"  L-BFGS gamma={gamma:.4f} beta={beta:.3e} score={best_sc:.12f}")
        if sc_new <= best_sc + 0.02 * max(1.0, best_sc):
            cur = a_new.copy()
        else:
            cur = best.copy()

        cmax = float(_full_convolve(best).max())
        cmax = max(1e-9, cmax)

    return best, best_sc

def _adam_full(a0, deadline, rng, lr=0.03, beta0_frac=0.07, beta_min_frac=0.0025,
               beta_decay=0.94, no_imp_limit=150):
    a = _normalize(a0)
    n = a.size
    best = a.copy()
    best_z = _logits_of(a)
    best_sc = _score_norm(a)

    z = best_z.copy()
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    c = _full_convolve(a)
    maxc = float(c.max())
    beta = max(1e-12, float(beta0_frac) * maxc)
    beta_min_abs = max(1e-7, float(beta_min_frac) * max(1e-9, maxc))
    base_lr = float(lr)
    no_imp = 0

    while time.monotonic() < deadline:
        _, w = _smooth_max(c, beta)
        ga = _grad_conv_weighted(a, w)
        if not np.all(np.isfinite(ga)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            continue

        dot = float(np.dot(a, ga))
        gz = a * (ga - dot)
        gz -= float(gz.mean())

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            continue

        t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        z = z - float(lr) * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(z.mean())
        a = _softmax_z(z)
        c = _full_convolve(a)
        maxc = float(c.max())
        sc = 2.0 * n * maxc

        if np.isfinite(sc):
            if sc < best_sc - 1e-13:
                best = a.copy()
                best_z = z.copy()
                best_sc = sc
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.002, lr * 0.96)

            beta = max(beta_min_abs, beta * float(beta_decay))
            if t % 30 == 0 and c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.05:
                    beta = max(beta_min_abs, beta * 0.5)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            continue

        if no_imp > max(60, int(no_imp_limit)):
            if best_sc < 1.503:
                ns = 0.02
            elif best_sc < 1.506:
                ns = 0.04
            elif best_sc < 1.51:
                ns = 0.08
            else:
                ns = 0.15

            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(z.mean())
            a = _softmax_z(z)
            c = _full_convolve(a)
            maxc = float(c.max())
            beta = max(beta_min_abs, float(beta0_frac) * max(1e-9, maxc))
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            lr = base_lr
            no_imp = 0

    return best, best_sc

def _peak_flood(a0, deadline, eta=0.04):
    a = _normalize(a0)
    n = a.size
    best = a.copy()
    best_sc = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    no_imp = 0
    cur_eta = float(eta)

    for it in range(20000):
        if time.monotonic() > deadline:
            break

        M = float(c.max())
        sc = 2.0 * n * M
        if sc < best_sc - 1e-13:
            best = a.copy()
            best_sc = sc
            no_imp = 0
        else:
            no_imp += 1

        threshold = 0.95 * M
        hot = c >= threshold
        if not np.any(hot):
            hot[int(np.argmax(c))] = True
        w = hot.astype(np.float64)

        heat = _cross_w_a(w, a)
        hmax = float(np.max(heat))
        if not np.isfinite(hmax) or hmax <= 1e-15:
            break

        factor = np.exp(-cur_eta * heat / hmax)
        a = a * factor
        a = np.maximum(a, 1e-15)
        a /= a.sum()

        c = _full_convolve(a)

        if it % 100 == 0:
            cur_eta = max(0.005, min(0.12, cur_eta * (0.9 if no_imp > 30 else 1.03)))
            if no_imp > 300:
                a = best.copy()
                c = _full_convolve(a)
                cur_eta = 0.02
                no_imp = 0

    return best, best_sc


# ----------------------------------------------------------------------
# SLP via LP: minimize t subject to 2*A(q) - c(a) <= t, sum(q)=1, q>=0.
# ----------------------------------------------------------------------
def _lp_cache_get(n):
    if n in _LP_CACHE:
        return _LP_CACHE[n]
    if n > 1200:
        return None
    m = 2 * n - 1
    i = np.arange(n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)
    rows = (i[:, None] + j[None, :]).ravel().astype(np.int32, copy=False)
    cols = np.repeat(i, n).astype(np.int32, copy=False)
    js = np.tile(j, n).astype(np.int32, copy=False)

    rows_m = np.arange(m, dtype=np.int32)
    cols_m = np.full(m, n, dtype=np.int32)
    rows_all = np.concatenate((rows, rows_m)).astype(np.int32, copy=False)
    cols_all = np.concatenate((cols, cols_m)).astype(np.int32, copy=False)

    data_all = np.empty(rows_all.size, dtype=np.float64)
    data_all[-m:] = -1.0

    eq_data = np.ones(n, dtype=np.float64)
    eq_rows = np.zeros(n, dtype=np.int32)
    eq_cols = np.arange(n, dtype=np.int32)
    A_eq = csr_matrix((eq_data, (eq_rows, eq_cols)), shape=(1, n + 1))

    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)
    bounds = [(0.0, None)] * (n + 1)

    cache = {
        "n": n,
        "m": m,
        "js": js,
        "rows_all": rows_all,
        "cols_all": cols_all,
        "data_all": data_all,
        "A_eq": A_eq,
        "c_obj": c_obj,
        "b_eq": b_eq,
        "bounds": bounds,
    }
    _LP_CACHE[n] = cache
    return cache


def _lp_linearized(a):
    a = _normalize(a)
    n = a.size
    cache = _lp_cache_get(n)
    if cache is None:
        return None

    m = cache["m"]
    data_all = cache["data_all"]
    np.take(a, cache["js"], out=data_all[:n * n])
    data_all[:n * n] *= 2.0
    data_all[n * n:] = -1.0

    A_ub = csr_matrix((data_all, (cache["rows_all"], cache["cols_all"])), shape=(m, n + 1))
    c = _full_convolve(a)

    res = linprog(
        cache["c_obj"],
        A_ub=A_ub,
        b_ub=c,
        A_eq=cache["A_eq"],
        b_eq=cache["b_eq"],
        bounds=cache["bounds"],
        method="highs",
    )

    if res is None or not hasattr(res, "x") or res.x is None:
        return None
    status = getattr(res, "status", None)
    if not (res.success or status == 1):
        return None

    q = np.asarray(res.x[:n], dtype=np.float64)
    if not np.all(np.isfinite(q)):
        return None
    q = np.maximum(q, 0.0)
    s = float(q.sum())
    if s <= 1e-12:
        return None
    return q / s


def _line_search_blend(a, d):
    a = _normalize(a)
    d = _normalize(d)
    base_sc = _score_norm(a)
    best_t = 0.0
    best_sc = base_sc
    best_a = a.copy()

    grid = (
        0.001, 0.002, 0.004, 0.008, 0.015, 0.025, 0.04,
        0.065, 0.1, 0.15, 0.22, 0.32, 0.45, 0.6, 0.75
    )
    for tt in grid:
        if not np.isfinite(tt) or tt <= 0.0 or tt >= 1.0:
            continue
        at = (1.0 - tt) * a + tt * d
        s = float(at.sum())
        if s <= 0.0 or not np.isfinite(s):
            continue
        at = at / s
        sc = _score_norm(at)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_t = tt
            best_a = at.copy()

    if best_t > 0.0:
        finals = (best_t * 0.5, best_t * 0.75, best_t * 1.25, best_t * 1.5,
                  best_t - 0.006, best_t + 0.006)
        for tt in finals:
            if not np.isfinite(tt) or tt <= 1e-6 or tt >= 1.0:
                continue
            at = (1.0 - tt) * a + tt * d
            s = float(at.sum())
            if s <= 0.0 or not np.isfinite(s):
                continue
            at = at / s
            sc = _score_norm(at)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_t = tt
                best_a = at.copy()

    return best_a, best_sc


def _slp_lp(a0, deadline, max_iters=100000):
    a = _normalize(a0)
    n = a.size
    best = a.copy()
    best_sc = _score_norm(a)
    u = np.ones(n, dtype=np.float64) / float(n)
    it = 0
    stalls = 0

    while time.monotonic() < deadline and it < max_iters:
        q = _lp_linearized(best)
        if q is None:
            break

        directions = []
        directions.append(q)
        r1 = 0.9 * q + 0.1 * u
        r1 = np.maximum(r1, 0.0)
        r1 /= r1.sum()
        directions.append(r1)
        r2 = 0.75 * q + 0.25 * u
        r2 = np.maximum(r2, 0.0)
        r2 /= r2.sum()
        directions.append(r2)

        improved = False
        for d in directions:
            if time.monotonic() >= deadline:
                break
            a_new, sc_new = _line_search_blend(best, d)
            if sc_new < best_sc - 1e-13:
                best = a_new
                best_sc = sc_new
                improved = True
                if it % 5 == 0:
                    print(f"  SLP-LP iter={it} score={best_sc:.12f}")
                breaks = [0]
                for _ in range(breaks[0]):
                    pass
                break
        it += 1
        if not improved:
            stalls += 1
            if stalls >= 3:
                break

    return best, best_sc

def _stochastic_moves(a0, deadline, rng, max_steps=500000):
    a = _normalize(a0)
    n = a.size
    best = a.copy()
    best_sc = _score_norm(a)

    fracs = (0.01, 0.02, 0.04, 0.08, 0.15, 0.25, 0.45)
    no_imp = 0
    heat_recompute_every = 300

    c = _full_convolve(best)
    M = float(c.max())
    w = c >= 0.95 * M
    if not np.any(w):
        w[int(np.argmax(c))] = True
    heat = _cross_w_a(w.astype(np.float64), a)
    hmax = float(np.max(heat))
    if hmax > 0:
        heat = heat / hmax
    else:
        heat = np.zeros(n, dtype=np.float64)

    src_w = np.maximum(a, 0.0) * np.maximum(heat, 0.0)
    src_s = float(src_w.sum())
    if src_s <= 0.0:
        src_w = np.ones(n, dtype=np.float64)
    src_cdf = np.cumsum(src_w)
    src_cdf = src_cdf / src_cdf[-1]

    max_a = float(np.max(a)) + 1e-12
    tgt_w = (1.0 - np.maximum(heat, 0.0)) / (a / max_a + 1e-3)
    tgt_w = np.maximum(tgt_w, 1e-6)
    tgt_s = float(tgt_w.sum())
    tgt_w = tgt_w / tgt_s
    tgt_cdf = np.cumsum(tgt_w)
    tgt_cdf = tgt_cdf / tgt_cdf[-1]

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break

        r1 = float(rng.random())
        r2 = float(rng.random())
        src = int(np.searchsorted(src_cdf, r1))
        tgt = int(np.searchsorted(tgt_cdf, r2))
        if src == tgt:
            continue
        if src < 0:
            src = 0
        if tgt < 0:
            tgt = 0
        if src >= n:
            src = n - 1
        if tgt >= n:
            tgt = n - 1

        a_src = float(a[src])
        if a_src <= 0.0:
            continue
        a_tgt = float(a[tgt])

        improved = False
        for frac in fracs:
            amt = a_src * frac
            if amt <= 0.0:
                continue
            a[src] = a_src - amt
            a[tgt] = a_tgt + amt

            sc = _score_norm(a)
            if sc < best_sc - 1e-13:
                best = a.copy()
                best_sc = sc
                no_imp = 0
                improved = True
                break
            else:
                a[src] = a_src
                a[tgt] = a_tgt

        if improved:
            if step % 20000 == 0:
                print(f"  stochastic step={step} score={best_sc:.12f}")
        else:
            no_imp += 1
            if no_imp > heat_recompute_every:
                c = _full_convolve(best)
                a = best.copy()
                M = float(c.max())
                w = c >= 0.95 * M
                if not np.any(w):
                    w[int(np.argmax(c))] = True
                heat = _cross_w_a(w.astype(np.float64), a)
                hmax = float(np.max(heat))
                if hmax > 0:
                    heat = heat / hmax
                else:
                    heat = np.zeros(n, dtype=np.float64)

                src_w = np.maximum(a, 0.0) * np.maximum(heat, 0.0)
                src_s = float(src_w.sum())
                if src_s <= 0.0:
                    src_w = np.ones(n, dtype=np.float64)
                src_cdf = np.cumsum(src_w)
                src_cdf = src_cdf / src_cdf[-1]

                max_a = float(np.max(a)) + 1e-12
                tgt_w = (1.0 - np.maximum(heat, 0.0)) / (a / max_a + 1e-3)
                tgt_w = np.maximum(tgt_w, 1e-6)
                tgt_w = tgt_w / tgt_w.sum()
                tgt_cdf = np.cumsum(tgt_w)
                tgt_cdf = tgt_cdf / tgt_cdf[-1]
                no_imp = 0

    return best, best_sc

# Simple multi-scale helpers and fallback profile.
# ----------------------------------------------------------------------
def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, max(2, n))
    a = 1.0 + float(eta) * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = _normalize(p0)
    n0 = p0.size
    if n0 == n_new:
        return p0.copy()
    if n0 == 1:
        return np.ones(n_new, dtype=np.float64) / float(n_new)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    out = np.interp(x1, x0, p0)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 0.0:
        return np.ones(n_new, dtype=np.float64) / float(n_new)
    return out / s

# Top-level search entry point.

def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load and normalize parent sequence, if available.
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is None:
        parent = _make_profile_linear(1024, 2.0)

    best = np.asarray(parent, dtype=np.float64).copy()
    best_sc = _score_unit(best)
    n_best = best.size
    print(f"Initial parent n={n_best} score={best_sc:.12f}")

    # Useful deadlines.
    total = max(60.0, deadline - t_start)
    end_lbfgs1 = t_start + 0.25 * total
    end_slp = end_lbfgs1 + 0.30 * total
    end_adam = end_slp + 0.17 * total
    end_peak = end_adam + 0.08 * total
    end_stoch = end_peak + 0.20 * total

    def _update(seq, sc, label):
        pass  # Not used; avoids stray closures. We do direct updates below.

    # Phase 1: L-BFGS-B continuation on smooth-max objective.
    if time.monotonic() < end_lbfgs1:
        a1, sc1 = _lbfgs_continuation(best, min(deadline, end_lbfgs1), end_lbfgs1 - time.monotonic())
        if a1 is not None and sc1 < best_sc - 1e-13:
            best = a1.copy()
            best_sc = sc1
            print(f"After L-BFGS 1: n={best.size} score={best_sc:.12f}")

    # Phase 2: Successive LP linearized step.
    if time.monotonic() < end_slp and best.size <= 1200:
        a2, sc2 = _slp_lp(best, min(deadline, end_slp))
        if a2 is not None and sc2 < best_sc - 1e-13:
            best = a2.copy()
            best_sc = sc2
            print(f"After SLP-LP: n={best.size} score={best_sc:.12f}")

    # Short L-BFGS polish after LP.
    if time.monotonic() < end_slp + 0.06 * total:
        mini_end = min(deadline, end_slp + 0.06 * total)
        a2b, sc2b = _lbfgs_continuation(best, mini_end, mini_end - time.monotonic())
        if a2b is not None and sc2b < best_sc - 1e-13:
            best = a2b.copy()
            best_sc = sc2b
            print(f"After post-LP L-BFGS: n={best.size} score={best_sc:.12f}")

    # Phase 3: Adam on logits.
    if time.monotonic() < end_adam:
        a3, sc3 = _adam_full(best, min(deadline, end_adam), rng, lr=0.03,
                             beta0_frac=0.07, beta_min_frac=0.0025,
                             beta_decay=0.94, no_imp_limit=120)
        if a3 is not None and sc3 < best_sc - 1e-13:
            best = a3.copy()
            best_sc = sc3
            print(f"After Adam: n={best.size} score={best_sc:.12f}")

    # Phase 4: Peak flooding.
    if time.monotonic() < end_peak:
        a4, sc4 = _peak_flood(best, min(deadline, end_peak), eta=0.035)
        if a4 is not None and sc4 < best_sc - 1e-13:
            best = a4.copy()
            best_sc = sc4
            print(f"After peak flood: n={best.size} score={best_sc:.12f}")

    # Phase 5: Stochastic local moves.
    if time.monotonic() < end_stoch:
        a5, sc5 = _stochastic_moves(best, min(deadline, end_stoch), rng, max_steps=400000)
        if a5 is not None and sc5 < best_sc - 1e-13:
            best = a5.copy()
            best_sc = sc5
            print(f"After stochastic: n={best.size} score={best_sc:.12f}")

    # Phase 6: Final polish and optional quick multi-scale trial.
    if time.monotonic() < deadline - 30.0:
        # Try one nearby scale if there is enough time.
        alt_ns = []
        if best.size <= 1536:
            alt_ns = [int(round(best.size * 1.5))]
        else:
            alt_ns = [1024]
        for nn in alt_ns:
            if time.monotonic() > deadline - 60.0:
                break
            if nn <= 0 or nn > 3000:
                continue
            p = _interp_profile(best, nn)
            sc_p = _score_norm(p)
            label = f"interp n={nn}"
            if sc_p < best_sc - 1e-12:
                best = p.copy()
                best_sc = sc_p
                print(f"{label}: score={best_sc:.12f}")

            if time.monotonic() < deadline - 40.0 and best.size == nn:
                mini_end = min(deadline - 20.0, time.monotonic() + 35.0)
                pa, sca = _adam_full(best, mini_end, rng, lr=0.02,
                                     beta0_frac=0.05, beta_min_frac=0.002,
                                     beta_decay=0.95, no_imp_limit=80)
                if pa is not None and sca < best_sc - 1e-12:
                    best = pa.copy()
                    best_sc = sca
                    print(f"Adam on n={nn}: score={best_sc:.12f}")

        # Final short Adam on best, at its current length.
        if time.monotonic() < deadline - 10.0:
            last_end = min(deadline - 2.0, time.monotonic() + 30.0)
            af, scf = _adam_full(best, last_end, rng, lr=0.015,
                                 beta0_frac=0.04, beta_min_frac=0.0018,
                                 beta_decay=0.96, no_imp_limit=60)
            if af is not None and scf < best_sc - 1e-12:
                best = af.copy()
                best_sc = scf
                print(f"Final Adam: n={best.size} score={best_sc:.12f}")

    # Final sanity normalization and validation.
    final = np.asarray(best, dtype=np.float64).ravel()
    final = np.clip(final, 0.0, 1000.0)
    s = float(final.sum())
    if not np.isfinite(s) or s < 0.01:
        final = _make_profile_linear(1024, 2.0)
        s = float(final.sum())
    final = final / s

    final_sc = _score_norm(final)
    print(f"FINAL internal score n={final.size} score={final_sc:.12f}")

    try:
        eval_sc = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(eval_sc):
            print(f"FINAL evaluate_sequence score={eval_sc:.12f}")
    except Exception:
        pass

    return [float(x) for x in final]
```
