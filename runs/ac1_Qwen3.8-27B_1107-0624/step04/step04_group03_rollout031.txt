"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Improvements over previous: adaptive beta continuation, better time management,
structured restarts, power-sum alternative to log-sum-exp, and basis-function
initialization for diverse starting points.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _smooth_values(c, beta):
    cm = float(c.max())
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w, cm
    w = e / s
    return w, cm


def _smooth_obj(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _smooth_values(c, beta)
    W = float(np.dot(w, c))
    return 2.0 * n * W


def _smooth_grad(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _smooth_values(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _lbfgs_phase(a_init, deadline, betas=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if betas is None:
        betas = [0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    betas = [b * max_c0 for b in betas]

    per_level = max(1.0, (deadline - time.monotonic()) / len(betas))

    for i, beta in enumerate(betas):
        if time.monotonic() >= deadline - 2:
            break
        level_dl = min(deadline - 1, time.monotonic() + per_level)
        try:
            result = minimize(
                _smooth_obj, z, args=(beta,),
                method='L-BFGS-B', jac=_smooth_grad,
                options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-9,
                         'maxls': 20, 'maxfun': 800}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x.copy()
                a_new = _softmax_fn(z_new)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new
                    c_new = _full_convolve(a_new, a_new)
                    max_c0 = float(c_new.max())
                per_level = max(1.0, (deadline - time.monotonic()) / max(1, len(betas) - i - 1))
        except Exception:
            pass

    a_final = _softmax_fn(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _adam_phase(a_init, deadline, lr=0.05, beta_frac=0.01):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, beta_frac * max_c)

    stagnant = 0
    last_best = best_score

    while time.monotonic() < deadline - 1:
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _smooth_values(c, beta)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            step_scale = lr * 0.1 / (grad_norm / n + 1e-10)
            step = step_scale * mh / (np.sqrt(vh) + eps)
        else:
            step = np.zeros(n)

        z = z + step
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        if t % 100 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm(a_curr)
            if np.isfinite(sc) and sc < best_score:
                if best_score - sc > 1e-8:
                    stagnant = 0
                else:
                    stagnant += 1
                best_score = sc
                best_a = a_curr.copy()
            else:
                stagnant += 1

            # Adaptive beta and lr
            if stagnant > 5:
                beta = max(1e-12, beta * 0.7)
                lr = max(0.001, lr * 0.8)
            elif stagnant == 0:
                beta = max(1e-12, beta * 0.9)

            c_new = _full_convolve(_softmax_fn(z), _softmax_fn(z))
            max_c = float(c_new.max())

    a_final = _softmax_fn(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _nesterov_phase(a_init, deadline, lr_init=0.02, mom_init=0.9, beta_fracs=None):

    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())

    if beta_fracs is None:
        beta_fracs = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002,
                      0.0001, 5e-5, 2e-5, 1e-5, 5e-6]

    lr = lr_init
    mom = mom_init
    t = 0
    stagnant = 0
    last_best = best_score

    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline - 1:
            break
        beta = max(1e-12, beta_frac * max_c)

        while time.monotonic() < deadline - 1:

            grad_y = _smooth_grad(y, beta)

                else:

def _subgrad_phase(a_init, deadline, lr_init=0.003, n_active_frac=0.01):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    lr = lr_init
    momentum = np.zeros(n)
    mom_coeff = 0.9
    stagnant = 0

    while time.monotonic() < deadline - 1:
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Active set: top fraction of convolution values
        n_active = max(1, int(n_active_frac * len(c)))
        top_indices = np.argsort(c)[-n_active:]
        w = np.zeros_like(c)
        w[top_indices] = 1.0 / n_active
        W = float(np.dot(w, c))

        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        momentum = mom_coeff * momentum - lr * grad
        z = z + momentum
        z = z - z.mean()
        z = np.clip(z, -60, 60)

        # Check progress every 50 steps
        if int(momentum.sum() * 1000) % 50 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm(a_curr)
            if np.isfinite(sc) and sc < best_score:
                stagnant = 0
                best_score = sc
                best_a = a_curr.copy()
            else:
                stagnant += 1
                if stagnant > 3:
                    lr = lr * 0.9
                    stagnant = 0

    a_final = _softmax_fn(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _make_profile(n, kind="linear", **kwargs):
    t = np.linspace(0.0, 1.0, n)
    if kind == "linear":
        slope = kwargs.get("slope", 1.5)
        a = 1.0 + slope * (t - 0.5)
    elif kind == "quadratic":
        a1 = kwargs.get("a1", 1.0)
        a2 = kwargs.get("a2", 0.2)
        u = 2.0 * t - 1.0
        a = 1.0 + a1 * u + a2 * u ** 2
    elif kind == "cubic":
        a1 = kwargs.get("a1", 0.8)
        a3 = kwargs.get("a3", -0.4)
        u = 2.0 * t - 1.0
        a = 1.0 + a1 * u + a3 * u ** 3
    elif kind == "edge_peak":
        # High at edges, low in middle (anti-triangular)
        amp = kwargs.get("amp", 0.5)
        a = 1.0 - amp * np.sin(np.pi * t)
    elif kind == "cosine":
        amp = kwargs.get("amp", 0.3)
        a = 1.0 + amp * np.cos(2 * np.pi * t)
    elif kind == "gaussian_bump":
        # Dip in the middle
        width = kwargs.get("width", 0.3)
        a = 1.0 - kwargs.get("depth", 0.5) * np.exp(-((t - 0.5) ** 2) / (2 * width ** 2))
    elif kind == "sine_sq":
        a = np.sin(np.pi * t) ** 2
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

    sc_parent = _score_norm(parent)
    update_best(parent, sc_parent, "parent")

    profiles_to_try = [
        ("linear_2", "linear", {"slope": 2.0}),
        ("linear_3", "linear", {"slope": 3.0}),
        ("edge_peak", "edge_peak", {"amp": 0.6}),
        ("gauss_dip", "gaussian_bump", {"depth": 0.5, "width": 0.3}),
        ("cos_03", "cosine", {"amp": 0.3}),
        ("quad", "quadratic", {"a1": 1.5, "a2": -0.3}),
    ]

    for label, kind, kw in profiles_to_try:
        if time.monotonic() > phase0_end:
            break
        p = _make_profile(n, kind, **kw)
        sc = _score_norm(p)
        if sc < best_score:
            update_best(p, sc, f"init_{label}")

    print(f"Phase 0 done: best={best_score:.10f}, n={len(best_seq)}")

    # --- Phase 1: L-BFGS continuation (350s) ---
    phase1_end = min(deadline, t_start + 380.0)
    if best_seq is not None and time.monotonic() < phase1_end:
        per_time = min(300.0, phase1_end - time.monotonic() - 10)
        if per_time > 20:
            p_best, sc_best = _lbfgs_phase(best_seq, min(phase1_end - 5, time.monotonic() + per_time))
            update_best(p_best, sc_best, "lbfgs_p1")

    # Try L-BFGS on a second profile if time allows
    if time.monotonic() < phase1_end - 40:
        p_alt = _make_profile(n, "edge_peak", {"amp": 0.4})
        per_time2 = min(80.0, phase1_end - time.monotonic() - 10)
        if per_time2 > 20:
            p_best2, sc_best2 = _lbfgs_phase(p_alt, min(phase1_end - 5, time.monotonic() + per_time2))
            update_best(p_best2, sc_best2, "lbfgs_alt")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Adam with adaptive beta (400s) ---
    phase2_end = min(deadline, t_start + 780.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        per_time = min(350.0, phase2_end - time.monotonic() - 10)
        if per_time > 20:
            p_best, sc_best = _adam_phase(best_seq, min(phase2_end - 5, time.monotonic() + per_time),
                                          lr=0.04, beta_frac=0.005)
            update_best(p_best, sc_best, "adam_p2")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Nesterov (300s) ---
    phase3_end = min(deadline, t_start + 1080.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        beta_fracs = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                      5e-5, 2e-5, 1e-5, 5e-6, 2e-6]
        per_time = min(250.0, phase3_end - time.monotonic() - 10)
        if per_time > 20:
            p_best, sc_best = _nesterov_phase(best_seq, min(phase3_end - 5, time.monotonic() + per_time),
                                              lr_init=0.015, mom_init=0.9, beta_fracs=beta_fracs)
            update_best(p_best, sc_best, "nesterov_p3")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Subgradient active set (200s) ---
    phase4_end = min(deadline, t_start + 1280.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        per_time = min(180.0, phase4_end - time.monotonic() - 10)
        if per_time > 20:
            p_best, sc_best = _subgrad_phase(best_seq, min(phase4_end - 5, time.monotonic() + per_time),
                                             lr_init=0.002, n_active_frac=0.005)
            update_best(p_best, sc_best, "subgrad_p4")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Restarted Adam (200s) ---
    phase5_end = min(deadline, t_start + 1480.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        n_restarts = 3
        per_restart = min(60.0, (phase5_end - time.monotonic() - 10) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            noise_level = 0.02 + 0.01 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_level * rng.normal(size=len(best_seq))
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            per_dl = min(phase5_end - 5, time.monotonic() + per_restart)
            p_best, sc_best = _adam_phase(p_start, per_dl, lr=0.03, beta_frac=0.003)
            update_best(p_best, sc_best, f"restart_adam_{i}")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Fine polishing (150s) ---
    phase6_end = min(deadline, t_start + 1630.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        # Fine L-BFGS with tiny beta
        per_time = min(80.0, phase6_end - time.monotonic() - 10)
        if per_time > 15:
            betas_fine = [1e-5, 5e-6, 2e-6, 1e-6]
            p_best, sc_best = _lbfgs_phase(best_seq, min(phase6_end - 30, time.monotonic() + per_time),
                                           betas=betas_fine)
            update_best(p_best, sc_best, "fine_lbfgs")

        # Fine subgradient
        if time.monotonic() < phase6_end - 15:
            per_time2 = min(60.0, phase6_end - time.monotonic() - 5)
            if per_time2 > 10:
                p_best, sc_best = _subgrad_phase(best_seq, min(phase6_end - 2, time.monotonic() + per_time2),
                                                 lr_init=0.001, n_active_frac=0.002)
                update_best(p_best, sc_best, "fine_subgrad")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Try length change if beneficial (80s) ---
    phase7_end = deadline
    if best_seq is not None and time.monotonic() < phase7_end:
        n_curr = len(best_seq)
        # Try larger
        if n_curr <= 4096:
            target_n = n_curr * 2
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm(p_interp)
            if sc_interp < best_score:
                update_best(p_interp, sc_interp, f"interp_n{target_n}")
                per_time = min(40.0, phase7_end - time.monotonic() - 5)
                if per_time > 10:
                    p_best, sc_best = _adam_phase(p_interp, min(phase7_end - 2, time.monotonic() + per_time),
                                                  lr=0.02, beta_frac=0.005)
                    update_best(p_best, sc_best, f"adam_n{target_n}")

        # Try smaller
        if n_curr >= 4096:
            target_n = n_curr // 2
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm(p_interp)
            if sc_interp < best_score:
                update_best(p_interp, sc_interp, f"interp_n{target_n}")

    print(f"Phase 7 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profile(4096, "linear", slope=2.0)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned inf/NaN, using internal: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    fallback = _make_profile(4096, "linear", slope=2.0)
    print(f"FALLBACK: n=4096, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
