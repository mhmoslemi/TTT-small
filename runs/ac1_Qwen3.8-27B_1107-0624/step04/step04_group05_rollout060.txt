"""
Optimized step function search using Successive Convex Approximation (SCA) Linear
Programming, smooth-max Adam in logit space, and LP best-response refinement.
The key innovation is the SCA-LP which linearizes the self-convolution peak at the
current iterate and solves for the next point, combined with adaptive beta smooth-max
gradient polishing and multi-restart exploration across sequence lengths.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        c = np.convolve(a, a)
    else:
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        c = np.maximum(c, 0.0)
    return c


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    mval = float(c.max())
    if beta < 1e-13:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-9
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 0.0))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _compute_grad_p(p, w):
    n = len(p)
    N = _next_pow2(3 * n - 2)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p, N)
    cc = np.fft.irfft(fw * np.conj(fp), N)
    return 2.0 * cc[:n]


def _softmax_from_z(z, floor=1e-14):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-15, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _build_cross_conv_matrix(a):
    n = len(a)
    m = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for m_idx in range(m):
        j_start = max(0, m_idx - n + 1)
        j_end = min(n - 1, m_idx)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([m_idx] * len(idx))
            cols.extend(idx.tolist())
            vals.extend((2.0 * a[m_idx - idx]).tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(m, n))
    return A


def _sca_lp_solve(a, c):
    n = len(a)
    m = 2 * n - 1
    A_cross = _build_cross_conv_matrix(a)
    A_ub = np.column_stack([A_cross.toarray(), -np.ones(m)])
    b_ub = c.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0, None)] * n + [(-np.inf, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        a_new = np.maximum(res.x[:n], 0.0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
        else:
            return None, None
        return a_new, float(res.x[n])
    return None, None


def _sca_lp_sparse(a, c):
    n = len(a)
    m = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for m_idx in range(m):
        j_start = max(0, m_idx - n + 1)
        j_end = min(n - 1, m_idx)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([m_idx] * len(idx))
            cols.extend(idx.tolist())
            vals.extend((2.0 * a[m_idx - idx]).tolist())
        rows.append(m_idx)
        cols.append(n)
        vals.append(-1.0)
    A_ub = csr_matrix((vals, (rows, cols)), shape=(m, n + 1))
    b_ub = c.copy()
    A_eq = csr_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))), shape=(1, n + 1))
    b_eq = np.array([1.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0, None)] * n + [(-np.inf, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        a_new = np.maximum(res.x[:n], 0.0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
        else:
            return None, None
        return a_new, float(res.x[n])
    return None, None


def _lp_best_response(a, M):
    n = len(a)
    A_cross = _build_cross_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_cross, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        b = np.maximum(res.x, 0.0)
        s = b.sum()
        if s > 1e-10:
            return b / s
    return None


def _adam_logit(p0, deadline, rng, lr=0.05, beta_frac=0.08, beta_decay=0.93,
                no_imp_limit=500, noise_scale=None):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    z = np.log(p0)
    z -= float(np.mean(z))
    b1, b2, eps = 0.9, 0.999, 1e-8
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_adam = 0
    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-4, beta_frac * maxc)
    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_and_weights(c, beta)
        g = _compute_grad_p(p, w)
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m_buf.fill(0.0)
            v_buf.fill(0.0)
            t_adam = 0
            continue
        t_adam += 1
        m_buf = b1 * m_buf + (1.0 - b1) * gz
        v_buf = b2 * v_buf + (1.0 - b2) * (gz * gz)
        mh = m_buf / (1.0 - b1 ** t_adam)
        vh = v_buf / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m_buf.fill(0.0)
            v_buf.fill(0.0)
            t_adam = 0
            continue
        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.005, lr * 0.93)
        else:
            z = best_z.copy()
            m_buf.fill(0.0)
            v_buf.fill(0.0)
            t_adam = 0
            continue
        if t_adam % 30 == 0:
            beta = max(5e-5, beta * beta_decay)
            if c.size >= 2:
                top5 = np.sort(c)[::-1][:min(5, c.size)]
                gap = top5[0] - top5[-1]
                if gap < beta * 0.05:
                    beta = max(5e-5, beta * 0.7)
        if no_imp > no_imp_limit:
            if noise_scale is None:
                if best_score < 1.51:
                    ns = 0.05
                elif best_score < 1.53:
                    ns = 0.1
                else:
                    ns = 0.2
            else:
                ns = noise_scale
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-4, beta_frac * maxc)
            m_buf.fill(0.0)
            v_buf.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _sca_refine(a, deadline, max_iters=50, alpha_start=0.5, alpha_decay=0.95):
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    alpha = alpha_start
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        a_sca, t_val = _sca_lp_sparse(a, c)
        if a_sca is None:
            break
        best_sc = current_score
        best_alpha = 0.0
        for al in [alpha, alpha * 0.5, alpha * 0.25, 0.1, 0.05]:
            a_try = (1.0 - al) * a + al * a_sca
            a_try = np.maximum(a_try, 1e-15)
            a_try = a_try / a_try.sum()
            sc_try = _score_norm(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_alpha = al
        if best_alpha <= 0:
            stall += 1
            if stall > 8:
                break
            alpha = max(0.02, alpha * 0.7)
            continue
        a = (1.0 - best_alpha) * a + best_alpha * a_sca
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        sc = _score_norm(a)
        if sc < current_score - 1e-12:
            current_score = sc
            current_best = a.copy()
            stall = 0
        c = _full_convolve(a)
        alpha = max(0.02, alpha * alpha_decay)

    return current_best, current_score


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_quadratic(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + a1 * t + a2 * t * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_exponential(n, lam):
    t = np.arange(n)
    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}]")

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Started. Parent n={pn}, budget={budget_s}s")

    # Phase 1: Diverse initialization
    phase1_end = t_start + 30.0
    candidates = []
    n_base = 1024

    for n in [512, n_base]:
        profs = [
            ('lin1.0', _make_profile_linear(n, 1.0)),
            ('lin1.5', _make_profile_linear(n, 1.5)),
            ('lin2.0', _make_profile_linear(n, 2.0)),
            ('lin2.5', _make_profile_linear(n, 2.5)),
            ('quad1', _make_profile_quadratic(n, 1.0, -0.5)),
            ('quad2', _make_profile_quadratic(n, 1.5, -1.0)),
            ('quad3', _make_profile_quadratic(n, 0.5, 0.5)),
            ('exp0.3', _make_profile_exponential(n, 0.3)),
            ('exp0.7', _make_profile_exponential(n, 0.7)),
            ('exp1.5', _make_profile_exponential(n, 1.5)),
            ('twobump1', _make_profile_two_bump(n, 0.3, 0.7, 0.12)),
            ('twobump2', _make_profile_two_bump(n, 0.25, 0.75, 0.18)),
            ('cos1', _make_profile_cosine(n, 1, 0.7)),
            ('cos2', _make_profile_cosine(n, 2, 0.5)),
        ]
        for label, p in profs:
            sc = _score_norm(p)
            candidates.append((sc, p, n, label))

    if parent is not None:
        sc = _score_norm(parent)
        candidates.append((sc, parent.copy(), pn, 'parent'))
        for n_try in [512, 1024, 2048]:
            if pn != n_try:
                pi = _interp_profile(parent, n_try)
                if pi is not None:
                    sc_i = _score_norm(pi)
                    candidates.append((sc_i, pi, n_try, 'parent_interp'))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f}")
    _update(candidates[0][1], candidates[0][0], 'init')

    # Phase 2: SCA-LP refinement on top candidates (30s - 300s)
    phase2_end = t_start + 300.0
    count = 0
    for sc, p, n, label in candidates:
        if time.monotonic() > phase2_end or count >= 6:
            break
        if n > 1024:
            continue
        per_end = min(phase2_end, time.monotonic() + 80.0)
        p_sca, sc_sca = _sca_refine(p, per_end, max_iters=30, alpha_start=0.5)
        _update(p_sca, sc_sca, f'SCA_{label}_n{n}')
        count += 1

    print(f"Phase 2 done. Best: {best_score:.10f}")

    # Phase 3: Adam polish on best (300s - 500s)
    phase3_end = t_start + 500.0
    if best_seq is not None:
        per = min(150.0, phase3_end - time.monotonic())
        if per > 20:
            p_ad, sc_ad = _adam_logit(best_seq, time.monotonic() + per, rng,
                                      lr=0.05, beta_frac=0.08, beta_decay=0.93,
                                      no_imp_limit=400)
            _update(p_ad, sc_ad, 'adam_phase3')

    print(f"Phase 3 done. Best: {best_score:.10f}")

    # Phase 4: SCA-LP on best after Adam (500s - 700s)
    phase4_end = t_start + 700.0
    if best_seq is not None and best_seq.size <= 1024:
        per = min(180.0, phase4_end - time.monotonic())
        if per > 30:
            p_sca2, sc_sca2 = _sca_refine(best_seq, time.monotonic() + per,
                                           max_iters=40, alpha_start=0.3)
            _update(p_sca2, sc_sca2, 'SCA_phase4')

    print(f"Phase 4 done. Best: {best_score:.10f}")

    # Phase 5: LP best-response refinement (700s - 900s)
    phase5_end = t_start + 900.0
    if best_seq is not None and best_seq.size <= 1024:
        a = best_seq.copy()
        c = _full_convolve(a)
        M = float(c.max()) * 0.99
        for it in range(30):
            if time.monotonic() > phase5_end:
                break
            b = _lp_best_response(a, M)
            if b is None:
                break
            a_new = 0.7 * a + 0.3 * b
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            sc = _score_norm(a_new)
            if sc < best_score - 1e-12:
                best_score = sc
                best_seq = a_new.copy()
                print(f"  LP_BR iter {it}: {sc:.10f}")
            a = a_new
            c = _full_convolve(a)
            M = float(c.max()) * 0.99

    print(f"Phase 5 done. Best: {best_score:.10f}")

    # Phase 6: Multi-restart (900s - 1500s)
    phase6_end = t_start + 1500.0
    if best_seq is not None and best_seq.size <= 1024:
        n_restarts = 6
        per_r = max(20.0, (phase6_end - time.monotonic()) / max(1, n_restarts + 1))
        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 30:
                break
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            ns = 0.03 + 0.03 * i
            z = z + ns * rng.normal(size=best_seq.size)
            z = np.clip(z, -60, 60)
            z -= z.mean()
            p_s = _softmax_from_z(z)

            per_adam = min(per_r * 0.4, phase6_end - time.monotonic() - 40)
            if per_adam < 10:
                break
            p_r, sc_r = _adam_logit(p_s, time.monotonic() + per_adam, rng,
                                     lr=0.04, beta_frac=0.06, beta_decay=0.94,
                                     no_imp_limit=300)
            _update(p_r, sc_r, f'restart{i}')

            if best_seq.size <= 1024 and time.monotonic() < phase6_end - 20:
                per_sca = min(30.0, phase6_end - time.monotonic() - 10)
                if per_sca > 15:
                    p_sca3, sc_sca3 = _sca_refine(best_seq, time.monotonic() + per_sca,
                                                   max_iters=15, alpha_start=0.25)
                    _update(p_sca3, sc_sca3, f'SCA_restart{i}')

    print(f"Phase 6 done. Best: {best_score:.10f}")

    # Phase 7: Multi-length escalation (1500s - 1700s)
    phase7_end = t_start + 1700.0
    if time.monotonic() < phase7_end and best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024]
        elif n_cur <= 1024:
            target_ns = [2048]
        for tn in target_ns:
            if time.monotonic() > phase7_end - 60:
                break
            p_int = _interp_profile(best_seq, tn)
            if p_int is None:
                continue
            sc_int = _score_norm(p_int)
            _update(p_int, sc_int, f'interp_n{tn}')
            if time.monotonic() < phase7_end - 30:
                per = min(30.0, phase7_end - time.monotonic() - 10)
                if per > 10:
                    p_e, sc_e = _adam_logit(p_int, time.monotonic() + per, rng,
                                             lr=0.04, beta_frac=0.06, beta_decay=0.94,
                                             no_imp_limit=200)
                    _update(p_e, sc_e, f'escalate_n{tn}')

    # Phase 8: Final polish (1700s - deadline)
    if time.monotonic() < deadline - 5 and best_seq is not None:
        if best_seq.size <= 1024:
            per = min(60.0, deadline - time.monotonic() - 5)
            if per > 20:
                p_f, sc_f = _sca_refine(best_seq, time.monotonic() + per * 0.5,
                                         max_iters=10, alpha_start=0.2)
                _update(p_f, sc_f, 'final_SCA')
        if time.monotonic() < deadline - 5:
            per2 = min(40.0, deadline - time.monotonic() - 5)
            if per2 > 10:
                p_f2, sc_f2 = _adam_logit(best_seq, time.monotonic() + per2, rng,
                                           lr=0.01, beta_frac=0.03, beta_decay=0.97,
                                           no_imp_limit=100)
                _update(p_f2, sc_f2, 'final_adam')

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(1024, 1.5)
        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")
        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass
        return [float(x) for x in final]

    fb = _make_profile_linear(1024, 1.5)
    print("FALLBACK n=1024")
    return [float(x) for x in fb]
