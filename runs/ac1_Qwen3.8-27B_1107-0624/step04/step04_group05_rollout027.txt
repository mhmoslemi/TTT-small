import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 512:
        c = np.convolve(a, a)
    else:
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        c = np.maximum(c, 0.0)
    return c


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    mval = float(c.max())
    if beta < 1e-13:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-9
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 0.0))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-15):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -70.0, 70.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-15, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_solve(a, M):
    n = len(a)
    A_ub = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_solve_tight(a, M, tightening=0.995, n_attempts=5):
    """Try tighter M values to get a more aggressive best response."""
    n = len(a)
    A_ub = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    for i in range(n_attempts):
        M_try = M * (tightening ** i)
        b_ub_arr = M_try * np.ones(2 * n - 1)
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
        if res.success:
            b = np.maximum(res.x, 0.0)
            if b.sum() > 1e-10:
                return b, M_try
    return None, M


def _line_search(a, b_norm, current_score, max_steps=12):
    """Backtracking line search for mixing a with b_norm."""
    best_sc = current_score
    best_t = 0.0
    a_cur = a.copy()
    for i in range(max_steps):
        t = 0.1 * (2 ** i) / (1 + 0.5 * i)
        t = min(1.0, t)
        a_try = (1.0 - t) * a_cur + t * b_norm
        a_try = np.maximum(a_try, 1e-15)
        a_try = a_try / a_try.sum()
        sc_try = _score_norm(a_try)
        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_t = t
            a_cur = a_try
        else:
            break
    return a_cur, best_sc, best_t


def _adam_logit(p0, deadline, rng, lr=0.05, beta_frac=0.06, beta_decay=0.94,
                no_imp_limit=600, noise_scale=None, max_steps=None):
    """Smooth-max Adam optimization in logit space with adaptive beta."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-4, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    step_count = 0

    while time.monotonic() < deadline:
        if max_steps is not None and step_count >= max_steps:
            break
        step_count += 1

        _, w = _smooth_max_and_weights(c, beta)
        g = _conv_cross_corr(w, p)

        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)

        lr_eff = lr * (0.95 ** (step_count // 500))
        lr_eff = max(lr_eff, 0.003)
        z = z - lr_eff * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -70.0, 70.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-15:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1

            if no_imp > 0 and no_imp % 300 == 0:
                lr = max(0.003, lr * 0.9)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if step_count % 20 == 0:
            beta = max(2e-5, beta * beta_decay)
            if c.size >= 3:
                top3 = np.sort(c)[::-1][:3]
                gap = top3[0] - top3[-1]
                if gap < beta * 0.1:
                    beta = max(2e-5, beta * 0.7)

        if no_imp > no_imp_limit:
            if noise_scale is None:
                if best_score < 1.51:
                    ns = 0.08
                elif best_score < 1.53:
                    ns = 0.15
                else:
                    ns = 0.25
            else:
                ns = noise_scale
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -70.0, 70.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-4, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _lp_refine(a, deadline, t_init=0.3, max_solves=80, use_tight=True, line_search=True):
    """LP best-response iteration with adaptive step size."""
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    stall_count = 0

    for _ in range(max_solves):
        if time.monotonic() >= deadline:
            break

        if use_tight:
            b, M_used = _lp_solve_tight(a, M, tightening=0.998, n_attempts=4)
        else:
            b = _lp_solve(a, M)
            M_used = M
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b

        if line_search:
            a_new, _, _ = _line_search(a, b_norm, current_score, max_steps=10)
        else:
            a_new = (1.0 - t_init) * a + t_init * b_norm
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()

        sc_new = _score_norm(a_new)

        if sc_new < current_score - 1e-13:
            current_score = sc_new
            current_best = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 4:
                break

        a = a_new
        c = _full_convolve(a)
        M = float(c.max())

    return current_best, current_score


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_quadratic(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + a1 * t + a2 * t * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_exponential(n, lam, direction=1):
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * direction * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp, phase=0.0):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_sine(n, freq, amp, phase=0.0):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.sin(2.0 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_u_shaped(n, depth):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + depth * ((2.0 * t - 1.0) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))

    # === Phase 2: Adam polish top candidates (40s-300s) ===
    phase2_end = t_start + 300.0
    polished = []
    count = 0
    for sc, p, n_c, label in candidates:
        if time.monotonic() > phase2_end or count >= 10:
            break
        per = min(phase2_end - time.monotonic() - 5, 25.0)
        if per < 5:
            break
        p_b, sc_b = _adam_logit(p, time.monotonic() + per, rng,
                                lr=0.06, beta_frac=0.06, beta_decay=0.93,
                                no_imp_limit=300)
        _update(p_b, sc_b, f'adam_{label}_n{n_c}')
        polished.append((sc_b, p_b, n_c, label))
        count += 1

    polished.sort(key=lambda x: x[0])
    print(f"Phase 2 done. Best: {best_score:.10f}")

    # === Phase 3: Intensive LP refinement (300s-900s) ===
    phase3_end = t_start + 900.0
    if best_seq is not None and best_seq.size <= 4096:
        n_lp = best_seq.size
        print(f"Phase 3: LP refine n={n_lp}, score={best_score:.10f}")

        # First LP pass with line search
        per_lp1 = min(phase3_end - time.monotonic() - 60, 250.0)
        if per_lp1 > 30:
            p_lp1, sc_lp1 = _lp_refine(best_seq, time.monotonic() + per_lp1,
                                        t_init=0.3, max_solves=60,
                                        use_tight=True, line_search=True)
            _update(p_lp1, sc_lp1, 'LP_pass1')

        # Adam after LP
        if time.monotonic() < phase3_end - 30:
            per_ad = min(80.0, phase3_end - time.monotonic() - 20)
            if per_ad > 10:
                p_ad, sc_ad = _adam_logit(best_seq, time.monotonic() + per_ad, rng,
                                           lr=0.03, beta_frac=0.04, beta_decay=0.95,
                                           no_imp_limit=400)
                _update(p_ad, sc_ad, 'adam_postLP')

        # Second LP pass
        if time.monotonic() < phase3_end - 10 and best_seq.size <= 4096:
            per_lp2 = min(phase3_end - time.monotonic() - 5, 120.0)
            if per_lp2 > 20:
                p_lp2, sc_lp2 = _lp_refine(best_seq, time.monotonic() + per_lp2,
                                            t_init=0.25, max_solves=40,
                                            use_tight=True, line_search=True)
                _update(p_lp2, sc_lp2, 'LP_pass2')

    print(f"Phase 3 done. Best: {best_score:.10f}")

    # === Phase 4: Multi-scale exploration (900s-1200s) ===
    phase4_end = t_start + 1200.0
    if best_seq is not None and time.monotonic() < phase4_end:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048]
        elif n_cur <= 1024:
            target_ns = [2048]
        elif n_cur <= 2048:
            target_ns = [4096]

        for tn in target_ns:
            if time.monotonic() > phase4_end - 60:
                break
            p_int = _interp_profile(best_seq, tn)
            if p_int is None:
                continue
            sc_int = _score_norm(p_int)
            print(f"  Scale to n={tn}: {sc_int:.10f}")
            _update(p_int, sc_int, f'scale_n{tn}')

            per = min(50.0, phase4_end - time.monotonic() - 20)
            if per > 10:
                p_e, sc_e = _adam_logit(p_int, time.monotonic() + per, rng,
                                         lr=0.05, beta_frac=0.05, beta_decay=0.94,
                                         no_imp_limit=300)
                _update(p_e, sc_e, f'scale_adam_n{tn}')

                if best_seq.size <= 4096 and time.monotonic() < phase4_end - 15:
                    per_lp = min(40.0, phase4_end - time.monotonic() - 5)
                    if per_lp > 15:
                        p_ls, sc_ls = _lp_refine(best_seq, time.monotonic() + per_lp,
                                                   t_init=0.2, max_solves=20,
                                                   use_tight=True, line_search=True)
                        _update(p_ls, sc_ls, f'scale_LP_n{best_seq.size}')

    print(f"Phase 4 done. Best: {best_score:.10f}")

    # === Phase 5: Aggressive multi-restart (1200s-1600s) ===
    phase5_end = t_start + 1600.0
    if best_seq is not None and time.monotonic() < phase5_end:
        n_best = best_seq.size
        n_restarts = 8
        total_time = phase5_end - time.monotonic()
        per_restart = total_time / (n_restarts + 1)

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 15:
                break

            # Different perturbation strategies
            if i % 3 == 0:
                # Gaussian noise on logit
                ns = 0.05 + 0.03 * i
                z = np.log(np.maximum(best_seq, 1e-12))
                z -= z.mean()
                z = z + ns * rng.normal(size=n_best)
            elif i % 3 == 1:
                # Sparse perturbation (perturb a subset of coordinates)
                z = np.log(np.maximum(best_seq, 1e-12))
                z -= z.mean()
                n_pert = max(10, n_best // 4)
                idx = rng.choice(n_best, size=n_pert, replace=False)
                z[idx] += 0.1 * rng.normal(size=n_pert)
            else:
                # Structured perturbation (smooth noise)
                ns = 0.04 + 0.02 * i
                noise = np.cumsum(rng.normal(size=n_best)) * ns
                noise -= noise.mean()
                z = np.log(np.maximum(best_seq, 1e-12))
                z -= z.mean()
                z = z + noise

            z = np.clip(z, -70, 70)
            z -= z.mean()
            p_s = _softmax_from_z(z)

            per_adam = min(per_restart * 0.5, phase5_end - time.monotonic() - 25)
            if per_adam < 8:
                break
            p_r, sc_r = _adam_logit(p_s, time.monotonic() + per_adam, rng,
                                     lr=0.05, beta_frac=0.06, beta_decay=0.94,
                                     no_imp_limit=int(0.2 * n_best))
            _update(p_r, sc_r, f'restart{i}_n{n_best}')

            if best_seq.size <= 4096 and time.monotonic() < phase5_end - 10:
                per_lp = min(30.0, phase5_end - time.monotonic() - 8)
                if per_lp > 15:
                    p_lpr, sc_lpr = _lp_refine(best_seq, time.monotonic() + per_lp,
                                                 t_init=0.2, max_solves=15,
                                                 use_tight=True, line_search=True)
                    _update(p_lpr, sc_lpr, f'LP_restart{i}')

    print(f"Phase 5 done. Best: {best_score:.10f}")

    # === Phase 6: Final intensive LP polish (1600s-1780s) ===
    phase6_end = deadline - 8.0
    if best_seq is not None and time.monotonic() < phase6_end:
        if best_seq.size <= 4096:
            per = min(60.0, phase6_end - time.monotonic() - 5)
            if per > 20:
                p_f, sc_f = _lp_refine(best_seq, time.monotonic() + per,
                                        t_init=0.15, max_solves=30,
                                        use_tight=True, line_search=True)
                _update(p_f, sc_f, 'final_LP')

        # Final tiny Adam step
        if time.monotonic() < phase6_end - 3:
            per2 = phase6_end - time.monotonic() - 3
            if per2 > 5:
                p_f2, sc_f2 = _adam_logit(best_seq, time.monotonic() + per2, rng,
                                           lr=0.01, beta_frac=0.03, beta_decay=0.97,
                                           no_imp_limit=50)
                _update(p_f2, sc_f2, 'final_adam')

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(1024, 1.5)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    fb = _make_profile_linear(1024, 1.5)
    print(f"FALLBACK n=1024")
    return [float(x) for x in fb]
