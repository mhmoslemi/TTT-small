If `j ≠ k-j` (i.e., `2j ≠ k`), these are two separate terms, so `∂b_k/∂a_j = a_{k-j} + a_{k-j} = 2a_{k-j}`.
If `j = k-j` (i.e., `2j = k`), it's a single term `a_j^2`, so `∂b_k/∂a_j = 2a_j = 2a_{k-j}`.

So in all cases: `∂b_k/∂a_j = 2 * a_{k-j}` (with `a_i = 0` for `i < 0` or `i ≥ n`).

Therefore: `∂f/∂a_j = 2n * sum_k w_k * 2 * a_{k-j} = 4n * sum_k w_k * a_{k-j}`

The sum `sum_k w_k * a_{k-j}` is the cross-correlation of `w` and `a` at lag `j`.

So the gradient is `4n * cross_correlate(w, a)`.

But we're optimizing on the simplex (sum = 1, a >= 0), so we need to project the gradient. The projected gradient in the interior of the simplex is:
`g_projected[j] = grad[j] - mean(grad)` (subtract the mean to maintain sum=0 for the update direction)

Then we use a mirror map (entropic) update:
`a_new = exp(log(a) - lr * g_projected) / sum(exp(log(a) - lr * g_projected))`

This is the "negative entropy" mirror prox or "exponential weights" update.

Gradient-driven optimization with LP polishing and multi-scale escalation
for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive smooth-max gradient with line search, LP fixed-point,
and diverse initialization strategies.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    if n <= 1:
        return 1
    k = 1
    while k < n:
        k <<= 1
    return k


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _cross_correlate(w, a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _smooth_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s


def _build_conv_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_primal(a, M, n):
    A = _build_conv_matrix(n, a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

        g -= g.mean()

        if not np.all(np.isfinite(g)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * (g * g)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)

        if do_line_search:
            best_z_candidate = z
            best_sc = float(2.0 * n * maxc)
            for lr_try in [lr * 0.5, lr, lr * 1.5, lr * 2.0]:
                z_try = z - lr_try * mh / (np.sqrt(vh) + eps)
                z_try = np.clip(z_try, -50, 50)
                z_try -= z_try.mean()
                a_try = np.exp(z_try - z_try.max())
                a_try = np.maximum(a_try, 1e-15)
                a_try = a_try / a_try.sum()
                sc_try = _score_norm(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_z_candidate = z_try.copy()
            z = best_z_candidate
        else:
            z = z - lr * mh / (np.sqrt(vh) + eps)
            z = np.clip(z, -50, 50)
            z -= z.mean()

        a = np.exp(z - z.max())
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        c = _full_convolve(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.005, lr * 0.9)

        if t_adam % 80 == 0:
            beta = max(1e-5, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.3:
                    beta = max(1e-5, beta * 0.5)

        if no_imp > patience:
            ns = 0.06 + 0.04 * (no_imp // patience)
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            c = _full_convolve(a)
            beta = 0.08 * c.max()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = lr_base
            no_imp = 0

    a_out = np.exp(best_z - best_z.max())
    a_out = np.maximum(a_out, 1e-15)
    a_out = a_out / a_out.sum()
    return a_out, best_score


def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new


def _make_profile(n, profile_type, param1=1.0, param2=0.0, param3=0.0):
    if profile_type == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + param1 * (t - 0.5)
    elif profile_type == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 + param1 * (2 * t - 1) ** 2
    elif profile_type == 'cubic':
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + param1 * u + param2 * u ** 3
    elif profile_type == 'exp':
        t = np.arange(n)
        a = np.exp(-param1 * t / n)
    elif profile_type == 'cosine':
        t = np.arange(n)
        a = 1.0 + param1 * np.cos(2 * np.pi * param2 * t / n)
    elif profile_type == 'sine':
        t = np.arange(n)
        a = 1.0 + param1 * np.sin(2 * np.pi * param2 * t / n)
    elif profile_type == 'parabola':
        t = np.linspace(0, 1, n)
        a = (t - 0.5) ** 2 + param1
    elif profile_type == 'two_gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - param1) / param3) ** 2) + np.exp(-0.5 * ((t - param2) / param3) ** 2)
    elif profile_type == 'gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / param1) ** 2)
    elif profile_type == 'U_shape':
        t = np.linspace(0, 1, n)
        a = (2 * t - 1) ** 2 + param1
    elif profile_type == 'edge_heavy':
        t = np.arange(n) / max(1, n - 1)
        a = 1.0 / (0.01 + abs(t - 0.5) * 2.0) ** param1
    elif profile_type == 'wavy':
        t = np.arange(n)
        a = 1.0 + param1 * np.cos(2 * np.pi * param2 * t / n) + param3 * np.cos(2 * np.pi * param2 * 2 * t / n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _dual_lp(a, n):
    A = _build_conv_matrix(n, a)
    c_obj = np.ones(2 * n - 1)
    A_ub = -A.T
    b_ub = -np.ones(n)
    bounds = [(0, None)] * (2 * n - 1)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return float(res.fun), np.maximum(res.x, 0.0)
    return None, None


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization: diverse init, gradient with line search,
    LP fixed-point, dual certificates, and multi-scale escalation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    # === Phase 1: Generate diverse initial candidates ===
    print("Phase 1: Generating initial candidates...")
    candidates = []

    n_sizes = [256, 512, 1024, 2048]
    profile_defs = [
        ('lin1.5', 'linear', 1.5, 0, 0),
        ('lin2.0', 'linear', 2.0, 0, 0),
        ('lin2.5', 'linear', 2.5, 0, 0),
        ('lin3.0', 'linear', 3.0, 0, 0),
        ('quad1', 'quadratic', 1.0, 0, 0),
        ('quad2', 'quadratic', 2.0, 0, 0),
        ('cubic1', 'cubic', 1.5, -1.0, 0),
        ('cubic2', 'cubic', 2.0, -1.5, 0),
        ('exp1.0', 'exp', 1.0, 0, 0),
        ('exp2.0', 'exp', 2.0, 0, 0),
        ('cos1', 'cosine', 0.8, 1, 0),
        ('cos2', 'cosine', 0.6, 2, 0),
        ('cos3', 'cosine', 0.5, 3, 0),
        ('sin2', 'sine', 0.7, 2, 0),
        ('para05', 'parabola', 0.5, 0, 0),
        ('para10', 'parabola', 1.0, 0, 0),
        ('g03', 'gauss', 0.3, 0, 0),
        ('tg30', 'two_gauss', 0.3, 0.7, 0.15),
        ('U03', 'U_shape', 0.3, 0, 0),
        ('U05', 'U_shape', 0.5, 0, 0),
        ('edge1', 'edge_heavy', 1.0, 0, 0),
        ('edge2', 'edge_heavy', 2.0, 0, 0),
        ('wavy1', 'wavy', 0.6, 3, 0.3),
        ('wavy2', 'wavy', 0.5, 5, 0.2),
        ('wavy3', 'wavy', 0.8, 2, 0.4),
    ]

    for n in n_sizes:
        for (label, ptype, p1, p2, p3) in profile_defs:
            p = _make_profile(n, ptype, p1, p2, p3)
            sc = _score_norm(p)
            candidates.append((sc, p, n, label))

    if parent is not None:
        for n in [512, 1024, 2048, 4096]:
            pi = _interp(parent, n)
            sc = _score_norm(pi)
            candidates.append((sc, pi, n, 'parent'))
        if 0 < pn <= 4096:
            sc = _score_norm(parent)
            candidates.append((sc, parent.copy(), pn, 'parent_exact'))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, best init: {candidates[0][0]:.8f} n={candidates[0][2]} [{candidates[0][3]}]")

    if candidates[0][0] < best_score:
        best_seq = np.asarray(candidates[0][1], dtype=np.float64).copy()
        best_score = float(candidates[0][0])
        print(f"  BEST: {best_score:.10f} n={best_seq.size} [init]")

    # === Phase 2: LP fixed-point on top small-n candidates ===
    print("Phase 2: LP fixed-point optimization...")
    phase2_end = t_start + 400.0
    polished = []

    # Sort by (score, n) - prefer small n for LP speed, good score
    candidates_lp = sorted(candidates, key=lambda x: (x[0], x[2]))

    count = 0
    for (sc, p, n, label) in candidates_lp:
        if time.monotonic() > phase2_end or count >= 8:
            break
        if n > 1024:
            continue
        time_alloc = max(30.0, min(60.0, 400.0 * (256.0 / float(n))))
        per_end = min(phase2_end, time.monotonic() + time_alloc)

        a_lp, sc_lp = _lp_fixed_point(p, per_end, max_iters=50, use_line_search=True)
        if sc_lp < best_score:
            best_seq = np.asarray(a_lp, dtype=np.float64).copy()
            best_score = float(sc_lp)
            print(f"  BEST: {best_score:.10f} n={a_lp.size} [LP_{label}_n{n}]")
        polished.append((sc_lp, a_lp, n, label + '_lp'))
        count += 1

    polished.sort(key=lambda x: x[0])
    print(f"  After Phase 2: best={best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    # === Phase 3: Gradient refinement with line search ===
    print("Phase 3: Gradient optimization with line search...")
    phase3_end = t_start + 900.0

    # Run gradient on top polished results and best candidates
    grad_inputs = []
    for (sc, p, n, label) in polished[:3]:
        grad_inputs.append((sc, p, n, 'grad_' + label))
    for (sc, p, n, label) in candidates[:3]:
        if (sc, p, n, label) not in grad_inputs:
            grad_inputs.append((sc, p, n, 'grad_init'))

    for (sc, p, n, label) in grad_inputs[:5]:
        if time.monotonic() > phase3_end - 30:
            break
        per = min(100.0, phase3_end - time.monotonic() - 10)
        if per < 20:
            break

        a_ad, sc_ad = _gradient_optimize(p, time.monotonic() + per, rng,
                                         lr_base=0.08, beta_decay=0.97,
                                         patience=300, do_line_search=True)
        if sc_ad < best_score:
            best_seq = np.asarray(a_ad, dtype=np.float64).copy()
            best_score = float(sc_ad)
            print(f"  BEST: {best_score:.10f} n={a_ad.size} [{label}]")

        # LP polish after gradient (only for small n)
        if n <= 1024 and time.monotonic() < phase3_end - 30:
            per2 = min(40.0, phase3_end - time.monotonic() - 10)
            if per2 > 15:
                a_lp2, sc_lp2 = _lp_fixed_point(a_ad, time.monotonic() + per2,
                                                max_iters=20, use_line_search=True)
                if sc_lp2 < best_score:
                    best_seq = np.asarray(a_lp2, dtype=np.float64).copy()
                    best_score = float(sc_lp2)
                    print(f"  BEST: {best_score:.10f} n={a_lp2.size} [LPpost_{label}]")

    # === Phase 4: Multi-scale escalation ===
    print("Phase 4: Multi-scale escalation...")
    phase4_end = t_start + 1350.0

    if best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048, 4096]
        elif n_cur <= 1024:
            target_ns = [2048, 4096]
        elif n_cur <= 2048:
            target_ns = [4096]
        else:
            target_ns = []

        for tn in target_ns:
            if time.monotonic() > phase4_end - 60:
                break
            p_int = _interp(best_seq, tn)
            sc_int = _score_norm(p_int)
            print(f"  Interpolated to n={tn}: {sc_int:.8f}")
            if sc_int < best_score:
                best_seq = p_int.copy()
                best_score = sc_int
                print(f"  BEST: {best_score:.10f} n={tn} [interp]")

            # Gradient on interpolated
            if time.monotonic() < phase4_end - 60:
                per = min(100.0, phase4_end - time.monotonic() - 30)
                if per > 25:
                    a_e, sc_e = _gradient_optimize(p_int, time.monotonic() + per, rng,
                                                   lr_base=0.06, beta_decay=0.97,
                                                   patience=400, do_line_search=True)
                    if sc_e < best_score:
                        best_seq = a_e.copy()
                        best_score = sc_e
                        print(f"  BEST: {best_score:.10f} n={tn} [Grad_esc]")

            # LP at new scale (only if feasible)
            if tn <= 1024 and time.monotonic() < phase4_end - 30:
                per2 = min(40.0, phase4_end - time.monotonic() - 15)
                if per2 > 15:
                    a_le, sc_le = _lp_fixed_point(p_int, time.monotonic() + per2,
                                                  max_iters=15, use_line_search=True)
                    if sc_le < best_score:
                        best_seq = a_le.copy()
                        best_score = sc_le
                        print(f"  BEST: {best_score:.10f} n={tn} [LP_esc]")

    # === Phase 5: Dual LP certificate and guided search ===
    print("Phase 5: Dual LP certificate...")
    phase5_end = t_start + 1550.0

    if best_seq is not None and time.monotonic() < phase5_end:
        n_test = min(best_seq.size, 1024)
        p_test = _interp(best_seq, n_test) if best_seq.size != n_test else best_seq.copy()

        dual_obj, y_star = _dual_lp(p_test, n_test)
        if dual_obj is not None and np.isfinite(dual_obj) and dual_obj > 0:
            lower_bound = 2.0 * n_test / dual_obj
            print(f"  Dual LP lower bound (n={n_test}): {lower_bound:.10f}")
            print(f"  Current best: {best_score:.10f}")

            A_mat = _build_conv_matrix(n_test, p_test)
            Ay = np.asarray(A_mat.T @ y_star).ravel()
            Ay_max = float(Ay.max())
            if Ay_max > 0:
                threshold = Ay_max - 0.1 * Ay_max
                support = Ay >= threshold

                if 0 < support.sum() < n_test:
                    p_masked = p_test.copy()
                    p_masked[~support] = 1e-10
                    p_masked = p_masked / p_masked.sum()
                    sc_m = _score_norm(p_masked)
                    print(f"  Masked (support={support.sum()}): {sc_m:.10f}")
                    if sc_m < best_score:
                        best_seq = p_masked.copy()
                        best_score = sc_m
                        print(f"  BEST: {best_score:.10f} n={n_test} [dual_masked]")

                    if time.monotonic() < phase5_end - 20 and n_test <= 1024:
                        per = min(40.0, phase5_end - time.monotonic() - 10)
                        if per > 15:
                            a_dm, sc_dm = _lp_fixed_point(p_masked, time.monotonic() + per,
                                                          max_iters=20, use_line_search=True)
                            if sc_dm < best_score:
                                best_seq = a_dm.copy()
                                best_score = sc_dm
                                print(f"  BEST: {best_score:.10f} n={n_test} [LP_dual])")

    # === Phase 6: Random restarts with diverse perturbation ===
    print("Phase 6: Random restarts...")
    phase6_end = deadline - 80.0

    if best_seq is not None and time.monotonic() < phase6_end:
        n_best = best_seq.size
        n_restarts = 6
        per_r = max(10.0, (phase6_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase6_end:
                break
            per = min(per_r, phase6_end - time.monotonic() - 5)
            if per < 10:
                break

            if i == 0:
                # Symmetric perturbation
                z = np.log(np.maximum(best_seq, 1e-12))
                z -= z.mean()
                noise = rng.normal(size=n_best)
                noise = 0.5 * (noise + noise[::-1])  # Make symmetric
                z = z + 0.08 * noise
                z = np.clip(z, -50, 50)
                z -= z.mean()
            elif i == 1:
                # Larger perturbation
                z = np.log(np.maximum(best_seq, 1e-12))
                z -= z.mean()
                z = z + 0.15 * rng.normal(size=n_best)
                z = np
