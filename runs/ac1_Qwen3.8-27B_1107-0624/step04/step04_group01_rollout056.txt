```python
import numpy as np
import time
from scipy.optimize import linprog, minimize

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def solve_lp(a, n):
    """Solve LP: min t s.t. conv(a,b)[k] <= t for all k, sum b = 1, b >= 0."""
    # Variables: b_0, ..., b_{n-1}, t (total n+1 variables)
    # Minimize t: c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Constraints: conv(a,b)[k] <= t for all k
    # For each k, the constraint is: sum_{i} a[i]*b[k-i] - t <= 0
    # which is: a[i_1]*b[j_1] + ... + a[i_m]*b[j_m] - t <= 0
    
    # Build A_ub (each row is one constraint)
    # Total constraints: 2n-1
    A_ub = np.zeros((2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    
    for k in range(2*n-1):
        # b[k-i] for i in max(0, k-n+1) .. min(k, n-1)
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i  # index into b
            if 0 <= j < n:
                A_ub[k, j] = a[i]
        A_ub[k, n] = -1.0  # -t term
    
    # Equality: sum b_i = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    
    # Bounds: b_i >= 0, t unbounded below (but will be positive)
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        b = res.x[:n]
        t = res.x[n]
        return b, t
    else:
        return None, None
```

"""
LP-based fixed point iteration with line search, gradient polishing,
and multi-restart for minimizing 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _peak(a):
    """Return max of conv(a,a) assuming sum(a)=1."""
    c = _full_convolve(a, a)
    return float(c.max())

def _build_conv_matrix(a):
    """Build convolution matrix C (2n-1 x n) such that C @ b = conv(a, b)."""
    n = len(a)
    # C[k, j] = a[k-j] if 0 <= k-j < n
    # For column j: C[j:j+n, j] = a
    rows = np.repeat(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    k_vals = rows + cols  # k = j + m where m is the offset within column j
    # Actually: for column j, row j+m gets value a[m] for m=0..n-1
    # So row = j + m, col = j, value = a[m]
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    cols = np.repeat(np.arange(n), n)
    data = np.tile(a, n)
    C = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return C

def _solve_lp_min_t(a):
    """Solve LP: min t s.t. conv(a,b) <= t, sum b = 1, b >= 0.
    Returns (b, t) or (None, None)."""
    n = len(a)
    C = _build_conv_matrix(a)
    # A_ub = [C, -1] (2n-1 x n+1)
    # b_ub = 0
    # A_eq = [1_n^T, 0] (1 x n+1)
    # b_eq = 1
    
    # Build dense A_ub for scipy (n=2048 → 4095 x 2049, ~67MB)
    C_dense = C.toarray()
    A_ub = np.hstack([C_dense, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.ones(1)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        b = res.x[:n]
        t = res.x[n]
        return b, t
    return None, None

def _lp_direction(a):
    """Get LP direction: solve LP, return b (normalized)."""
    b, t = _solve_lp_min_t(a)
    if b is None:
        return a.copy()
    s = b.sum()
    if s < 1e-10:
        return a.copy()
    return b / s

def _line_search(a, b, M):

def _subgradient_step(a, lr):

    c = _full_convolve(a, a)
    k_star = int(np.argmax(c))
    # Gradient of conv(a,a)[k_star] w.r.t. a: 2 * Q_kstar @ a
    # Q_kstar[i,j] = 1 if i+j = k_star

    # Actually, Q_k @ a has (Q_k @ a)[i] = sum_j Q_k[i,j] * a[j] = a[k-i] (if valid)
    # So the gradient of a^T Q_k a is 2 * Q_k @ a = 2 * [a[k-i] for valid i]
    # Wait, I need to be more careful.
    # a^T Q_k a = sum_{i,j: i+j=k} a[i] * a[j]
    # d/da[m] of this = sum_{j: m+j=k} a[j] + sum_{i: i+m=k} a[i] = 2 * a[k-m] (if valid)
    # (since the two sums are the same by symmetry)
    # Actually: d/da[m] sum_{i+j=k} a[i]a[j] = a[k-m] + a[k-m] = 2a[k-m] (if 0<=k-m<n)

    # So the formula 2*a[k-m] is correct in all cases.

        if 0 <= j < n:

    a_new = a - lr * grad
    # Project onto simplex: min ||x - a_new||^2 s.t. sum x = 1, x >= 0
    a_new = _project_simplex(a_new)
    return a_new

def _project_simplex(v):
    """Project v onto the probability simplex {x: sum x = 1, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
