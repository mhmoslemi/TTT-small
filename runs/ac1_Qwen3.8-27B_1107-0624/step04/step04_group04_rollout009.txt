import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    """Score assuming sum(a)=1: returns 2*n*max(conv)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    """Full score: 2*n*max(conv)/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_conv_and_weights(a, beta, top_k=0):
    """Compute convolution and softmax weights over top peaks."""
    c = _full_convolve(a, a)
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        w = np.zeros_like(c)
        c_top = c[idx]
        c_norm = (c_top - c_top.max()) / beta
        e = np.exp(c_norm)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            argmax_idx = int(np.argmax(c))
            w[argmax_idx] = 1.0
        else:
            w[idx] = e / s
    else:
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            argmax_idx = int(np.argmax(c))
            w = np.zeros_like(c)
            w[argmax_idx] = 1.0
        else:
            w = e / s
    W = float(np.dot(w, c))
    return w, W, c, cm


def _z_objective_and_grad(z, beta, n, top_k=0):
    """Objective and gradient in logit space for smoothed max."""
    a = _softmax_fn(z)
    w, W, c, cm = _compute_conv_and_weights(a, beta, top_k)
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_norm = c_top - c_top.max()
        log_sum = float(np.log(np.sum(np.exp(c_norm / beta))))
        obj = 2.0 * n * (c_top.max() + beta * log_sum)
    else:
        c_shifted = (c - cm) / beta
        log_sum = float(np.log(np.sum(np.exp(c_shifted))))
        obj = 2.0 * n * (cm + beta * log_sum)
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        grad = np.zeros_like(z)
    return obj, grad

def _z_gradient_only(z, beta, n, top_k=0):

def _active_set_refine(a_init, deadline, max_iter=5000):
    """Exact active-set subgradient method with backtracking line search."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    step_base = 0.1
    patience = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        # Active set: peaks within 0.5% of max
        threshold = cm * 0.9995
        active = np.where(c >= threshold)[0]
        k_active = len(active)

        if k_active == 0:
            break

        # Subgradient: average gradient over active set
        # grad_i of conv_k = a_{k-i} (with bounds)
        # grad_i of sum_{k in active} conv_k = sum_{k in active} a_{k-i}
        # This equals (convolve(indicator_active, a_rev))[n-1 : 2n-1]
        ind = np.zeros(2 * n - 1)
        ind[active] = 1.0 / k_active
        a_rev = a[::-1]
        grad_conv = _full_convolve(ind, a_rev)
        h = grad_conv[n - 1:2 * n - 1]

        # Project to simplex tangent: g_i = h_i - mean(h * a) / mean(a) ... 
        # Actually for softmax parametrization: grad_z_i = a_i * (h_i - sum(a * h))
        W = float(np.dot(a, h))
        grad_z = a * (h - W)

        # Line search
        step = step_base
        improved = False
        for ls in range(10):
            z_new = z - step * grad_z
            z_new = np.clip(z_new, -60, 60)
            z_new = z_new - z_new.mean()
            a_new = _softmax_fn(z_new)
            sc_new = _score_norm_fn(a_new)

            if np.isfinite(sc_new) and sc_new < best_score:
                z = z_new
                a = a_new
                best_score = sc_new
                best_a = a_new.copy()
                improved = True
                break
            step *= 0.5

        if not improved:
            patience += 1
            if patience > 50:
                # Try smaller active set
                threshold = cm * 0.999
                active = np.where(c >= threshold)[0]
                if len(active) == 0:
                    break
                patience = 0
            step_base *= 0.8
        else:
            patience = 0

        # Periodically try to grow the active set
        if it % 200 == 0:
            threshold = cm * (0.999 - 0.001 * min(it // 200, 5))
            active = np.where(c >= threshold)[0]

    return best_a, best_score


def _peak_leveling(a_init, deadline, max_iter=3000):
    """Directly level the top peaks by moving mass from peak-contributing positions."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        # Get top 5 peaks
        top_idx = np.argsort(c)[-5:]
        top_vals = c[top_idx]
        spread = float(top_vals.max() - top_vals.min()) + 1e-15

        # For each peak position k, the gradient of conv_k w.r.t. a is:
        # grad_i = a_{k-i} (for valid i)
        # We want to reduce the max peak. Direction: negative gradient of max peak.
        # But we want to reduce ALL top peaks, so use average gradient.
        ind = np.zeros(2 * n - 1)
        ind[top_idx] = 1.0 / len(top_idx)
        a_rev = a[::-1]
        h = _full_convolve(ind, a_rev)[n - 1:2 * n - 1]

        W = float(np.dot(a, h))
        grad_z = a * (h - W)

        # Adaptive step based on spread
        step = min(0.05, 0.01 * spread / (cm + 1e-15))

        z_new = z - step * grad_z / (np.abs(grad_z).max() + 1e-15)
        z_new = np.clip(z_new, -60, 60)
        z_new = z_new - z_new.mean()
        a_new = _softmax_fn(z_new)
        sc_new = _score_norm_fn(a_new)

        if np.isfinite(sc_new) and sc_new < best_score:
            z = z_new
            a = a_new
            best_score = sc_new
            best_a = a_new.copy()
        else:
            # Try smaller step
            z_new2 = z - step * 0.3 * grad_z / (np.abs(grad_z).max() + 1e-15)
            z_new2 = np.clip(z_new2, -60, 60)
            z_new2 = z_new2 - z_new2.mean()
            a_new2 = _softmax_fn(z_new2)
            sc_new2 = _score_norm_fn(a_new2)
            if np.isfinite(sc_new2) and sc_new2 < best_score:
                z = z_new2
                a = a_new2
                best_score = sc_new2
                best_a = a_new2.copy()

        if it > 100:
            # Check convergence
            c_check = _full_convolve(a, a)
            top_check = np.sort(c_check)[-5:]
            if top_check[0] / (top_check[-1] + 1e-15) > 0.9999:
                if it > 200:
                    break

    return best_a, best_score


def _adam_phased(a_init, deadline, lr=0.03, top_k_init=3, top_k_max=20,
                 beta_init_ratio=0.05, n_max_iter=60000):
    """Adam with adaptive top-k and beta, phased for convergence."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c_init = _full_convolve(a_init, a_init)
    max_c = float(c_init.max())
    beta = beta_init_ratio * max_c

    cur_k = top_k_init
    cur_lr = lr

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        obj, grad = _z_objective_and_grad(z, beta, n, cur_k)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - cur_lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -70, 70)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Adaptive adjustments
        if t % 50 == 0:
            beta = max(1e-10, beta * 0.9)
        if t % 300 == 0:
            cur_k = min(cur_k + 1, top_k_max)
        if t % 2000 == 0:
            cur_lr = max(0.002, cur_lr * 0.95)
            if best_score < 1.51:
                cur_lr = max(0.001, cur_lr * 0.9)

    return best_a, best_score


def _make_profiles(n):
    """Generate diverse initial profiles."""
    profiles = []
    # Raised cosine
    t = np.linspace(0, np.pi, n)
    a = 0.5 * (1 + np.cos(t))
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    # Hamming window
    w = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
    w = np.maximum(w, 1e-12)
    profiles.append(w / w.sum())
    # Hanning window
    w = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    w = np.maximum(w, 1e-12)
    profiles.append(w / w.sum())
    # Blackman window
    w = 0.42 - 0.5 * np.cos(2 * np.pi * np.arange(n) / (n - 1)) \
        + 0.08 * np.cos(4 * np.pi * np.arange(n) / (n - 1))
    w = np.maximum(w, 1e-12)
    profiles.append(w / w.sum())
    # Triangular
    mid = (n - 1) / 2.0
    a = 1.0 - np.abs(np.arange(n) - mid) / (mid + 1)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    # Exponential (various rates)
    for alpha in [0.001, 0.003, 0.005, 0.01, 0.02]:
        r = 1.0 - alpha
        a = r ** np.arange(n)
        profiles.append(a / a.sum())
    # Sinc-like (sinc squared)
    x = np.linspace(-1, 1, n)
    a = (np.sinc(x)) ** 2
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    # Flat top (nearly uniform with slight taper)
    a = np.ones(n)
    taper = max(1, n // 20)
    for i in range(taper):
        frac = (i + 1) / (taper + 1)
        a[i] = 0.5 + 0.5 * frac
        a[n - 1 - i] = 0.5 + 0.5 * frac
    profiles.append(a / a.sum())
    # Super-Gaussian (various exponents)
    x = np.linspace(-1, 1, n)
    for p in [1.5, 2.0, 3.0, 4.0, 6.0]:
        a = np.exp(-p * x ** 2)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Optimal-ish: try to make convolution flat
    # a_i proportional to sqrt of some window
    x = np.linspace(0, 1, n)
    a = np.sqrt(1 + 0.1 * np.sin(2 * np.pi * x))
    profiles.append(a / a.sum())
    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _refine_candidate(a, deadline, target_n=None):
    """Full refinement pipeline for a single candidate."""
    n = len(a)
    if target_n is not None and target_n != n:
        a = _interp_profile(a, target_n)
        n = target_n

    a = a / a.sum()
    sc = _score_norm_fn(a)
    best_a = a.copy()
    best_score = sc

    # Phase A: L-BFGS with beta continuation
    if time.monotonic() < deadline:
        t_a_end = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.35)
        a_new, sc_new = _lbfgs_phase(a, t_a_end, max_wall_time=t_a_end - time.monotonic(), top_k=0)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new

    # Phase B: Active set refinement
    if time.monotonic() < deadline:
        t_b_end = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.25)
        a_new, sc_new = _active_set_refine(best_a, t_b_end, max_iter=3000)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new

    # Phase C: Peak leveling
    if time.monotonic() < deadline:
        t_c_end = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.2)
        a_new, sc_new = _peak_leveling(best_a, t_c_end, max_iter=2000)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new

    # Phase D: Final Adam polish
    if time.monotonic() < deadline - 5:
        t_d_end = min(deadline - 2, time.monotonic() + (deadline - time.monotonic()) * 0.3)
        a_new, sc_new = _adam_phased(best_a, t_d_end, lr=0.02, top_k_init=3,
                                     top_k_max=15, beta_init_ratio=0.02)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new

    # Phase E: Final active set
    if time.monotonic() < deadline - 3:
        t_e_end = min(deadline - 1, time.monotonic() + min(30.0, (deadline - time.monotonic()) * 0.3))
        a_new, sc_new = _active_set_refine(best_a, t_e_end, max_iter=2000)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization search for minimizing the step function score.
    Uses L-BFGS with beta continuation, active-set refinement, peak leveling,
    and multi-n exploration.
    """
    # Access parent if available
    global height_sequence_1

    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # === Phase 0: Initial candidates ===
    print("=== Phase 0: Initial candidates ===")
    phase0_end = min(deadline, t_start + 30.0)
    candidates = []

    # Try parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm_fn(parent)
                candidates.append((sc, parent.copy(), len(parent), "parent"))
                _update_best(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent: {e}")

    # Main search length
    n_main = 4096

    # Generate profiles
    profiles = _make_profiles(n_main)
    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p, n_main, f"prof{idx}"))
        _update_best(p, sc, f"prof{idx}")

    # Interpolate parent to n_main
    if parent is not None and len(parent) != n_main:
        p_interp = _interp_profile(parent, n_main)
        sc = _score_norm_fn(p_interp)
        candidates.append((sc, p_interp, n_main, "parent_interp"))
        _update_best(p_interp, sc, "parent_interp")

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 0 done: best={best_score:.10f}, {len(candidates)} candidates")

    # === Phase 1: Refine top candidates ===
    print("=== Phase 1: Refine top candidates ===")
    phase1_end = min(deadline, t_start + 500.0)
    n_refine = 3
    for i in range(min(n_refine, len(candidates))):
        if time.monotonic() > phase1_end:
            break
        sc, p, pn, label = candidates[i]
        if sc > best_score * 1.01:
            break
        remaining = phase1_end - time.monotonic()
        per_time = remaining / (n_refine - i)
        dl = min(phase1_end, time.monotonic() + per_time)
        print(f"  Refining {label} (n={pn}, sc={sc:.8f}), budget={per_time:.0f}s")
        a_best, sc_best = _refine_candidate(p, dl, target_n=n_main)
        _update_best(a_best, sc_best, f"refine_{label}")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Intensive refinement on best ===
    print("=== Phase 2: Intensive refinement ===")
    phase2_end = min(deadline, t_start + 1000.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        remaining = phase2_end - time.monotonic()
        per_pass = remaining / 3
        for pass_idx in range(3):
            if time.monotonic() > phase2_end:
                break
            dl = min(phase2_end, time.monotonic() + per_pass)
            # Each pass: L-BFGS + active set + peak leveling
            n = len(best_seq)
            a = best_seq.copy()

            # L-BFGS with small top_k
            t1 = min(dl, time.monotonic() + per_pass * 0.4)
            a, sc = _lbfgs_phase(a, t1, max_wall_time=t1 - time.monotonic(), top_k=3)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()

            # Active set
            t2 = min(dl, time.monotonic() + per_pass * 0.3)
            a, sc = _active_set_refine(a, t2, max_iter=2000)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()

            # Peak leveling
            t3 = min(dl, time.monotonic() + per_pass * 0.2)
            a, sc = _peak_leveling(a, t3, max_iter=1500)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()

            # Adam
            t4 = min(dl - 1, time.monotonic() + per_pass * 0.09)
            a, sc = _adam_phased(a, t4, lr=0.01, top_k_init=5, top_k_max=10,
                                 beta_init_ratio=0.01)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()

            print(f"    Pass {pass_idx+1}: score={best_score:.10f}")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Multi-restart with structured perturbations ===
    print("=== Phase 3: Multi-restart ===")
    phase3_end = min(deadline, t_start + 1400.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        n_best = len(best_seq)
        n_restarts = 4
        per_restart = (phase3_end - time.monotonic()) / max(1, n_restarts)

        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 10:
                break
            # Different perturbation strategies
            if i == 0:
                # Gaussian noise
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z - z.mean()
                z = z + 0.05 * rng.normal(size=n_best)
            elif i == 1:
                # Shape perturbation: modify the "width"
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z - z.mean()
                freq = 2 * np.pi * rng.integers(1, 5) / n_best
                z = z + 0.03 * np.sin(freq * np.arange(n_best))
            elif i == 2:
                # Random block perturbation
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z - z.mean()
                block_size = n_best // 8
                start = rng.integers(0, n_best - block_size)
                z[start:start + block_size] += 0.05 * rng.normal(size=block_size)
            else:
                # Very small noise
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z - z.mean()
                z = z + 0.02 * rng.normal(size=n_best)

            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            dl = min(phase3_end - 5, time.monotonic() + per_restart)
            a_best, sc_best = _refine_candidate(p_start, dl, target_n=n_best)
            _update_best(a_best, sc_best, f"restart{i}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Try different n values ===
    print("=== Phase 4: Different lengths ===")
    phase4_end = min(deadline, t_start + 1650.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        n_curr = len(best_seq)
        target_ns = []
        if n_curr != 2048:
            target_ns.append(2048)
        if n_curr != 8192:
            target_ns.append(8192)

        for tn in target_ns:
            if time.monotonic() > phase4_end - 30:
                break
            p_interp = _interp_profile(best_seq, tn)
            sc_interp = _score_norm_fn(p_interp)
            _update_best(p_interp, sc_interp, f"interp_n{tn}")

            remaining = phase4_end - time.monotonic()
            if remaining < 30:
                break
            per_opt = min(80.0, remaining * 0.5)
            dl = min(phase4_end - 5, time.monotonic() + per_opt)
            a_best, sc_best = _refine_candidate(p_interp, dl, target_n=tn)
            _update_best(a_best, sc_best, f"refine_n{tn}")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Final ultra-fine polish ===
    print("=== Phase 5: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            # Very small step Adam
            dl = min(deadline - 5, time.monotonic() + min(40.0, remaining * 0.5))
            a, sc = _adam_phased(best_seq, dl, lr=0.005, top_k_init=8,
                                 top_k_max=15, beta_init_ratio=0.005)
            _update_best(a, sc, "final_adam")

        if time.monotonic() < deadline - 10:
            dl2 = min(deadline - 3, time.monotonic() + min(30.0, deadline - time.monotonic() - 5))
            a, sc = _active_set_refine(best_seq, dl2, max_iter=2000)
            _update_best(a, sc, "final_active")

        if time.monotonic() < deadline - 5:
            dl3 = min(deadline - 2, time.monotonic() + min(20.0, deadline - time.monotonic() - 3))
            a, sc = _peak_leveling(best_seq, dl3, max_iter=1000)
            _update_best(a, sc, "final_level")

    # === Final ===
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = _make_profiles(2048)[0]
            final_score = _score_fn(final_seq)

        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")

        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 8192:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  evaluate check: {e}")

        return [float(x) for x in final_seq]

    fallback = _make_profiles(2048)[0]
    print(f"  FALLBACK: n=2048, score={_score_norm_fn(fallback):.10f}")
    return [float(x) for x in fallback]
